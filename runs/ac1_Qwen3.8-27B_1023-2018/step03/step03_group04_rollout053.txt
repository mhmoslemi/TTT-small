from scipy.optimize import linprog


def _get_parent():

    except Exception:

def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_from_conv(a, c):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    M = float(c.max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M / (S * S)


def _score(a):
    c = _conv_fft(a)
    return _score_from_conv(a, c)


def _build_A(f, n):
    """Build (2n-1) x n matrix where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, M, n):
    if M < 1e-15:
        return None
    A = _build_A(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s

        S = 1.0
    else:
        a = a / S
        S = 1.0

    best_score = _score(a)
    conv = _conv_fft(a)
    cur_score = best_score

    golden = (math.sqrt(5.0) - 1.0) / 2.0
    max_cd_iters = 500

    for cd_iter in range(max_cd_iters):
        if time.monotonic() > stop_time - 3.0:
            break
        improved_this_pass = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break

                    S_new = S + x_try

                    M_new = conv_new.max()

                        cur_score = sc_new
                        conv = conv_new
                        S = S_new
                        improved_this_pass = True
                        if cur_score < best_score - 1e-15:
                            best_score = cur_score

                        break

                if a[i] == 0.0:

                    pass

            a[i] = best_x
            if best_x != v:
                improved = True
                if best_sc < best_rank - 1e-15:
                    best_rank = best_sc
                    best_a = a.copy()
        if not improved:
            break

    return best_a, best_rank


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    n_p = len(parent)

    # Fallback
    fallback = parent.copy()
    try:
        fb_score = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fb_score = math.inf
    if not math.isfinite(fb_score):
        fallback = np.ones(384, dtype=np.float64)
        fb_score = float(evaluate_sequence([float(x) for x in fallback]))

    best_seq = fallback.copy()
    best_score = fb_score

    def _track(f):
        nonlocal best_seq, best_score
        if f is None or len(f) == 0:
            return
        f = np.clip(np.asarray(f, dtype=np.float64).ravel(), 0, 1000)
        if f.sum() < 0.01:
            return
        if not np.all(np.isfinite(f)):
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in f]))
        except Exception:
            return
        if math.isfinite(exact) and exact < best_score:
            best_seq = f.copy()
            best_score = exact

    # Phase 1: LP optimization (55% of budget)
    phase1_end = t0 + (deadline - t0) * 0.55
    n = n_p
    rng_main = np.random.default_rng(seed)

    # Initializations
    inits = []
    p_norm = parent.copy()
    ps = p_norm.sum()
    if ps > 0.01:
        p_norm = p_norm / ps
    inits.append(("parent", p_norm.copy()))

    i_arr = np.arange(n, dtype=np.float64)
    # Cosine window
    cos_w = 0.5 + 0.5 * np.cos(2.0 * np.pi * i_arr / n)
    cos_w = np.clip(cos_w, 0.01, None)
    inits.append(("cosine", cos_w / cos_w.sum()))

    # Lopsided
    lop = np.exp(-i_arr / (n * 0.4))
    inits.append(("lopsided", lop / lop.sum()))

    # Bimodal
    bim = np.exp(-((i_arr - 0.25 * n) / (n * 0.1))**2) + np.exp(-((i_arr - 0.75 * n) / (n * 0.1))**2)
    if bim.sum() > 0:
        inits.append(("bimodal", bim / bim.sum()))

    # Random smooth
    rnd = np.abs(rng_main.normal(0, 1, n)) + 0.05
    # Smooth it
    from scipy.ndimage import uniform_filter1d
    rnd_smooth = uniform_filter1d(rnd, size=max(3, n // 50))
    rnd_smooth = np.clip(rnd_smooth, 0.0, None)
    if rnd_smooth.sum() > 0.01:
        inits.append(("random", rnd_smooth / rnd_smooth.sum()))

    # Parent + noise
    noise = np.clip(parent + rng_main.normal(0, 0.15 * max(1.0, np.mean(parent)), n), 0, None)
    if noise.sum() > 0.01:
        inits.append(("noisy", noise / noise.sum()))

    time_alloc = [0.40, 0.15, 0.12, 0.12, 0.11, 0.10]
    phase1_dur = phase1_end - t0
    for idx, (name, init) in enumerate(inits):
        if time.monotonic() > phase1_end - 3.0:
            break
        alloc = time_alloc[idx] if idx < len(time_alloc) else 0.1
        stop_i = min(phase1_end - 2.0, time.monotonic() + phase1_dur * alloc)
        if time.monotonic() >= stop_i:
            continue
        try:
            res, sc = _lp_optimize(init, stop_i, seed + 1000 * (idx + 1))
            _track(res)
        except Exception:
            pass

    # Phase 2: Coordinate descent on best (25% of budget)
    phase2_end = t0 + (deadline - t0) * 0.82
    if time.monotonic() < phase2_end - 5.0:
        try:
            # Use the efficient O(n) update version
            best_a, best_sc = _coord_descent(best_seq, phase2_end - 3.0, seed + 500)
            _track(best_a)
        except Exception:
            pass

    # If still time, run fast_cd as additional polish
    if time.monotonic() < phase2_end - 10.0:
        try:
            best_a2, best_sc2 = _fast_coord_descent(best_seq, phase2_end - 3.0, seed + 700)
            _track(best_a2)
        except Exception:
            pass

    # Phase 3: Try a couple other lengths (10%)
    phase3_end = t0 + (deadline - t0) * 0.92
    other_len = 256
    if time.monotonic() < phase3_end - 5.0 and n_p != other_len:
        try:
            init_other = np.interp(np.linspace(0, 1, other_len),
                                   np.linspace(0, 1, n_p), parent)
            stop_other = min(phase3_end - 2.0, time.monotonic() + (phase3_end - time.monotonic()) * 0.6)
            res_o, sc_o = _lp_optimize(init_other, stop_other, seed + 9000)
            _track(res_o)
        except Exception:
            pass

    if time.monotonic() < phase3_end - 3.0:
        try:
            best_a3, _ = _coord_descent(best_seq, phase3_end - 2.0, seed + 1100)
            _track(best_a3)
        except Exception:
            pass

    # Phase 4: Final polish (remaining time)
    if time.monotonic() < deadline - 5.0:
        try:
            best_a4, _ = _coord_descent(best_seq, deadline - 2.0, seed + 1300)
            _track(best_a4)
        except Exception:
            pass

    if time.monotonic() < deadline - 3.0:
        try:
            best_a5, _ = _fast_coord_descent(best_seq, deadline - 1.0, seed + 1500)
            _track(best_a5)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
