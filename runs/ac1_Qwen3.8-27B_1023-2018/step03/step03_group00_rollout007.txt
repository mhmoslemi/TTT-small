    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(np.maximum(p, 0.0), 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_direction(f, n, M, timeout=2.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _subgradient_step(a, n, step_size=0.1):
    """Take a subgradient step on the score function."""
    b = _conv_full(a)
    M = float(b.max())
    S = float(a.sum())
    if S < 0.01 or M < 1e-18:
        return a.copy()
    
    # Find near-peak lags
    threshold = M * 0.95
    peak_lags = np.where(b >= threshold)[0]
    
    if len(peak_lags) == 0:
        peak_lags = np.where(b >= M * 0.9)[0]
    
    # Compute subgradient
    sg = np.zeros(n)
    for k in peak_lags:
        for i in range(n):
            ji = int(k - i)
            if 0 <= ji < n:
                sg[i] += a[ji]
    sg /= len(peak_lags)
    
    # Score = 2n*M/S^2
    # d(score)/d(a_i) = (2n/S^2)*sg_i - 2*(2n*M/S^3)*1
    grad = (2.0 * n / (S * S)) * sg - 4.0 * n * M / (S * S * S)
    
    # Normalized step
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a.copy()
    
    a_new = a - step_size * grad / gnorm * S
    a_new = np.maximum(a_new, 0.0)
    return a_new

def _lp_optimize(f0, n, stop_time, seed=42):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    fsum = float(f.sum())
    if fsum < 1e-10:
        f = np.ones(n)
    
    best_f = f.copy()
    best_score = _score(f)
    
    t = 0.3
    t_min = 0.02
    t_max = 0.8
    stall = 0
    max_stall = 80
    M_factor = 1.0
    it = 0
    
    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        # LP direction
        g = _lp_direction(f, n, M * M_factor, timeout=1.5)
        if g is None:
            g = _lp_direction(f, n, M * 1.01, timeout=1.5)
        
        if g is not None and g.sum() > 1e-15:
            g = g * (fsum / g.sum())
            
            dist = np.linalg.norm(g - f) / max(1.0, np.linalg.norm(f))
            if dist > 1e-10:
                improved = False
                for _ in range(12):
                    f_new = (1 - t) * f + t * g
                    f_new = np.clip(f_new, 0, None)
                    sc = _score(f_new)
                    if sc < best_score - 1e-14:
                        f = f_new
                        best_f = f.copy()
                        best_score = sc
                        fsum = float(f.sum())
                        improved = True
                        t = min(t_max, t * 1.1)
                        stall = 0
                        break
                    else:
                        t = max(t_min, t * 0.5)
                
                if not improved:
                    stall += 1
                    M_factor = max(0.92, M_factor * 0.985)
            else:
                stall += 3
                M_factor = max(0.92, M_factor * 0.98)
        else:
            stall += 5
        
        # Periodic subgradient step
        if it % 10 == 9 and stall > 5:
            f_sg = _subgradient_step(f, n, step_size=0.05)
            sc_sg = _score(f_sg)
            if sc_sg < best_score - 1e-12:
                f = f_sg
                best_f = f.copy()
                best_score = sc_sg
                fsum = float(f.sum())
            
            M_factor = 1.0
            t = 0.3
        
        if stall >= max_stall:
            # Perturb and reset
            f = np.clip(f + rng.normal(0, 0.05 * max(1.0, f.mean()), n), 0, None)
            fsum_now = f.sum()
            if fsum_now > 1e-10:
                f = f / fsum_now * fsum
            t = 0.3
            M_factor = 1.0
            stall = 0
            max_stall += 40
            # Also try full perturbation
            if rng.random() < 0.3:
                f = np.abs(rng.normal(0, 1, n)) + 0.01
                f = f / f.sum() * fsum
        
        it += 1
    
    return best_f, best_score

def _refined_coordinate_descent(a, stop_time, seed=0):
    n = len(a)
    best = np.clip(a.copy(), 0, None).astype(np.float64)
    best_sc = _score(best)
    if not math.isfinite(best_sc):
        return best, best_sc
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 8
    passes = 0
    
    while time.monotonic() < stop_time - 3.0:
        improved = False
        passes += 1
        
        # Single-coordinate golden search
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = best[i]
            S0 = best.sum() - orig
            if S0 < 1e-12:
                continue
            
            # Find good range
            lo, hi = 0.0, max(3.0 * orig + 0.01, 1.0)
            # Try a few points
            candidates = set()
            for m in (0.0, 0.1, 0.2, 0.35, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0, 5.0):
                candidates.add(min(1000.0, orig * m))
            candidates.add(0.0)
            
            best_this = orig
            best_this_sc = best_sc
            for v in sorted(candidates):
                best[i] = v
                sc = _score(best)
                if sc < best_this_sc - 1e-15:
                    best_this = v
                    best_this_sc = sc
            
            # Refine around best
            if best_this != orig:
                center = best_this
                for delta in (0.1, 0.05, 0.02, 0.01, 0.005):
                    for sign in (-1, 1):
                        v = center + sign * delta * max(1.0, abs(center))
                        if v < 0 or v > 1000:
                            v = min(max(v, 0), 1000)
                        best[i] = v
                        sc = _score(best)
                        if sc < best_this_sc - 1e-15:
                            best_this = v
                            best_this_sc = sc
                            center = v
                best[i] = best_this
                if best_this_sc < best_sc - 1e-15:
                    best_sc = best_this_sc
                    improved = True
            else:
                best[i] = orig
        
        # Two-coordinate
        for _ in range(150):
            if time.monotonic() > stop_time - 3.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            best_ij_sc = best_sc
            for mi in (0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.6, 0.0):
                for mj in (0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.6, 0.0):
                    best[i] = oi * mi if oi > 1e-15 else 0.0
                    best[j] = oj * mj if oj > 1e-15 else 0.0
                    sc = _score(best)
                    if sc < best_ij_sc - 1e-15:
                        best_ij_sc = sc
                        improved = True
            best[i], best[j] = oi, oj
            if best_ij_sc < best_sc - 1e-15:
                best_sc = best_ij_sc
        
        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0
    
    return best, best_sc
