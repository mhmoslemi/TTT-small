def _fft_len(m):
    m = int(m)
    if m not in _fft_cache:
        _fft_cache[m] = int(next_fast_len(m))
    return _fft_cache[m]

def _conv(a):
    """Full linear convolution a*a"""
    n = len(a)
    if n == 0:
        return np.array([])
    L = _fft_len(2*n-1)
    fa = rfft(a, L)
    c = irfft(fa*fa, L)[:2*n-1]
    return c

    """Full linear convolution a*b"""
    n1, n2 = len(a), len(b)
    if n1 == 0 or n2 == 0:

    L = _fft_len(n1+n2-1)
    c = irfft(rfft(a, L)*rfft(b, L), L)[:n1+n2-1]

        cfs = _conv_cross(f, g)
        t_opt = _best_t(cf, cfs, cg)
        
        cands = sorted(set([max(0.0, min(1.0, t_opt)), max(0.0, min(1.0, t_opt*0.6)),
                           max(0.0, min(1.0, t_opt*1.3)), 0.25, 0.5, 0.75, 1.0]))
        best_new = f.copy()
        best_new_sc = _int_score(f)
        for tt in cands:
            new_f = _norm((1.0-tt)*f + tt*g)
            sc = _int_score(new_f)
            if sc < best_new_sc:
                best_new_sc = sc
                best_new = new_f
        
        if best_new_sc < _int_score(f) - 1e-12:
            f = best_new
            no_improve = 0
            _try(f)
        else:
            no_improve += 1
            if no_improve >= 8:
                # Add perturbation
                scale = 0.3/f.sum()*(1.0 + 0.1*min(no_improve, 15))
                f = _norm(f + rng.normal(0, scale, n))
                # Also try the LP direction directly
                f_try = _norm((1-t_opt)*best_seq + t_opt*g)
                if _int_score(f_try) < _int_score(f):
                    f = f_try
                no_improve = 0
            else:
                # Small step toward g
                f = _norm((1.0-0.1)*f + 0.1*g)
        
        # Periodic subgradient
        if count % 25 == 0 and time.monotonic() < lp1_stop - 8:
            sc_f = _int_score(f)
            c_f = _conv_self(f)
            alpha = 0.08 / (n * max(c_f.max(), 1e-10))
            f = _subgrad_refine(f, alpha, 3, lp1_stop)
            _try(f)
    
    _try(f)
    
    # === Phase 2: Extended LP with alternating directions (15%) ===
    lp2_stop = t0 + 0.72*(budget - 35)
    f_alt = best_seq.copy()
    n_alt = f_alt.size
    
    while time.monotonic() < lp2_stop:
        # Direction 1: fix f_alt, optimize other
        g1 = _build_and_solve_lp(f_alt, n_alt)
        if g1 is None:
            break
        cf1 = _conv_self(f_alt)
        cg1 = _conv_self(g1)
        cfs1 = _conv_cross(f_alt, g1)
        t1 = _best_t(cf1, cfs1, cg1)
        new_alt = _norm((1-t1)*f_alt + t1*g1)
        
        # Direction 2: fix new_alt, optimize
        g2 = _build_and_solve_lp(new_alt, n_alt)
        if g2 is not None:
            cf2 = _conv_self(new_alt)
            cg2 = _conv_self(g2)
            cfs2 = _conv_cross(new_alt, g2)
            t2 = _best_t(cf2, cfs2, cg2)
            new_alt2 = _norm((1-t2)*new_alt + t2*g2)
            if _int_score(new_alt2) < _int_score(new_alt):
                new_alt = new_alt2
        
        if _int_score(new_alt) < _int_score(f_alt):
            f_alt = new_alt
            _try(f_alt)
        else:
            break
    
    _try(f_alt)
    
    # === Phase 3: L-BFGS refinement (12%) ===
    lbfgs_stop = t0 + 0.84*(budget - 35)
    if time.monotonic() < lbfgs_stop - 20:
        start = best_seq.copy()
        a_out = _lbfgs_refine(start, lbfgs_stop, beta_list=(60.0, 200.0, 500.0))
        _try(a_out)
    
    # === Phase 4: Other lengths (7%) ===
    other_stop = t0 + 0.91*(budget - 35)
    cur_n = len(best_seq)
    other_lens = sorted(set([192, 320, 192, 384, 128, 448]) - {cur_n})[:3]
    
    for L in other_lens:
        if time.monotonic() > other_stop - 10:
            break
        if best_score < 1.48:
            break
        # Resample
        x0 = np.linspace(0, 1, len(best_seq))
        x1 = np.linspace(0, 1, L)
        a_L = np.interp(x1, x0, best_seq)
        a_L = _norm(a_L)
        
        sub_stop = min(other_stop - 8, time.monotonic() + max(15.0, (other_stop - time.monotonic())*0.35))
        for _ in range(40):
            if time.monotonic() > sub_stop:
                break
            g = _build_and_solve_lp(a_L, L)
            if g is None:
                break
            cf = _conv_self(a_L)
            cg = _conv_self(g)
            cfs = _conv_cross(a_L, g)
            tt = _best_t(cf, cfs, cg)
            a_new = _norm((1-tt)*a_L + tt*g)
            if _int_score(a_new) < _int_score(a_L):
                a_L = a_new
            else:
                a_L = _norm((1-0.15)*a_L + 0.15*g)
        _try(a_L)
    
    # === Phase 5: Final polish (8%) ===
    fin_stop = hard_dl
    a_fin, _ = _coord_descent(best_seq, fin_stop, n_passes=2)
    _try(a_fin)
    
    # Final LP pass
    if time.monotonic() < fin_stop - 4:
        nf = len(best_seq)
        for _ in range(15):
            if time.monotonic() > fin_stop - 3:
                break
            g = _build_and_solve_lp(best_seq, nf)
            if g is None:
                break
            cf = _conv_self(best_seq)
            cg = _conv_self(g)
            cfs = _conv_cross(best_seq, g)
            tt = _best_t(cf, cfs, cg)
            new = _norm((1-tt)*best_seq + tt*g)
            if _int_score(new) < _int_score(best_seq) - 1e-14:
                best_seq = new
                best_score = _int_score(best_seq)
            else:
                break
    
    # Validate and return
    seq = np.clip(np.asarray(best_seq, dtype=np.float64).ravel(), 0.0, 1000.0)
    lst = [float(x) for x in seq]
    sc = _exact(lst)
    if not math.isfinite(sc):
        lst = [float(x) for x in np.clip(fallback, 0, 1000)]
    return lst
