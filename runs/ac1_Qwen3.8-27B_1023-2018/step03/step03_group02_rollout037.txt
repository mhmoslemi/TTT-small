def _coordinate_descent(a, n_rounds=3, stop_time=None):
    """1D coordinate descent: for each coordinate, try to optimize it"""
    n = len(a)
    a_curr = a.copy()
    s = a_curr.sum()
    if s < 1e-10:
        return a_curr, _score(a_curr)
    a_curr = a_curr / s
    best_sc = _score(a_curr)
    rng = np.random.default_rng(12345)
    
    for rnd in range(n_rounds):
        if stop_time and time.monotonic() > stop_time:
            break
        order = rng.permutation(n)
        for i in order:
            if stop_time and time.monotonic() > stop_time:
                break
            orig = a_curr[i]
            # Try multipliers
            if orig > 1e-10:
                candidates = [orig * m for m in (0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.6, 2.5, 0.0)]
            else:
                candidates = [0.0, 0.001, 0.005, 0.01, 0.05]
            
            best_val = orig
            best_c = best_sc
            for cv in candidates:
                a_curr[i] = cv
                if a_curr.sum() < 1e-10:
                    a_curr[i] = orig
                    continue
                sc = _score(a_curr)
                if sc < best_c:
                    best_c = sc
                    best_val = cv
            a_curr[i] = best_val
    
    a_curr = a_curr / a_curr.sum()
    return a_curr, _score(a_curr)

def _spectral_smooth(a, sigma=2.0):
    """Apply Gaussian smoothing in frequency domain"""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    # Gaussian window in frequency
    freqs = np.arange(fsize // 2 + 1)
    w = np.exp(-0.5 * (freqs / sigma) ** 2)
    F_smooth = F * w
    a_smooth = irfft(F_smooth, fsize)[:n]
    return np.clip(a_smooth, 0, None)

        else:
            g_dir = _lp_direction(f, n, M, time_limit=8.0)

            if dist > 1e-9:
                # Golden section line search
                t_best, sc_best = _golden_section_line_search(f, g_dir, 1.0, 0.01, 0.95, max_iter=12)
                if sc_best < best_score - 1e-14:

                    f_new = np.clip(f_new, 0, None)
                    f_new = f_new / f_new.sum()
                    f = f_new
                    best_f = f.copy()
                    best_score = sc_best
                    b_current = _conv_full(f)
                    M = float(b_current.max())
                    stall = 0
                    iter_count += 1
                    continue
            else:
                stall += 2

        stall += 1
        if stall >= max_stall:
            # Structured perturbation
            strategy = stall % 5
            if strategy == 0:

                f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
            elif strategy == 1:

                f = _spectral_smooth(f, sigma=3.0)
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            elif strategy == 2:

                f2 = f.copy()
                blk = n // 3
                f2[:blk], f2[n-blk:] = f[n-blk:].copy(), f[:blk].copy()
                f = f2
            elif strategy == 3:

                f, sc = _subgradient_refine(f, n_steps=30, initial_lr=0.03)
            else:

                f = np.abs(rng.normal(0, 0.5, n)) + 0.001
            
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            b_current = _conv_full(f)
            M = float(b_current.max())
            stall = 0
        
        # Periodic subgradient refinement
        if iter_count > 0 and iter_count % 50 == 0:
            f, sc = _subgradient_refine(f, n_steps=20, initial_lr=0.015, stop_time=stop_time - 5.0)
            if sc < best_score:
                best_f = f.copy()
                best_score = sc
            b_current = _conv_full(f)
            M = float(b_current.max())
