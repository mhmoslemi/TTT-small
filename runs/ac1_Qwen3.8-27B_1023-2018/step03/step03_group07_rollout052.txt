For j in S: 2 * sum_{k in A*} lambda_k * a[(k-j) mod n... ]

KKT: For j in S: 2 * sum_{k in A*} lambda_k * a[k-j] (where a[k-j]=0 if k-j not in [0,n-1]) = lambda_0

And: sum_j a[j] = 1
And: for k in A*: sum_j a[j]*a[k-j] = M

This is a system of |A*| + 1 nonlinear equations in |S| + |A*| + 1 unknowns (a[j] for j in S, lambda_k for k in A*, lambda_0, and M).

import numpy as np

from scipy.optimize import linprog, minimize

def _norm(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _build_A(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def _lp_best_response(A, M, n, tlm=5.0):
    c_obj = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tlm})
    except Exception:
        return None
    if res.status == 0 and res.x is not None:
        g = np.asarray(res.x, dtype=np.float64)
        gs = g.sum()
        if gs > 1e-10 and np.all(np.isfinite(g)):
            return g / gs
    return None

def _do_line_search(a, direction, M_cur):
    best_alpha = 0.0
    best_val = M_cur
    # Coarse search
    for alpha in np.concatenate([np.linspace(0.05, 0.5, 10), np.linspace(0.5, 1.0, 10)]):
        cand = _norm((1 - alpha) * a + alpha * direction)
        val = float(np.convolve(cand, cand).max())
        if val < best_val:
            best_val = val
            best_alpha = alpha
    return best_alpha, best_val

def _optimize_lp(a_init, stop_time, seed=42, lp_tlm=5.0):
    n = len(a_init)
    a = _norm(a_init)
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0
    
    while time.monotonic() < stop_time - 2.0:
        c = np.convolve(a, a)
        M = float(c.max())
        
        if M < best_M * (1 - 1e-14):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 100:
            scale = min(0.05 + 0.01 * (stuck // 100), 0.2)
            a = _norm(a * np.exp(rng.normal(0, scale, n)))
            stuck = 0
            continue
        
        A = _build_A(a)
        g = _lp_best_response(A, M * (1 + 1e-8), n, lp_tlm)
        
        if g is not None:
            alpha, val = _do_line_search(a, g, M)
            if alpha > 0 and val < M * (1 - 1e-13):
                a = _norm((1 - alpha) * a + alpha * g)
                continue
        
        # Gradient fallback
        k_star = int(c.argmax())
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn > 1e-14:
            grad /= gn
            for step in [0.05, 0.02, 0.01]:

                val = float(np.convolve(cand, cand).max())
                if val < M:

def _polish_scipy(a_init, stop_time):
    """Use scipy minimize to polish."""
    n = len(a_init)
    a = _norm(a_init)
    
    # Parameterize using log: a[j] = exp(w[j]) / sum(exp(w))
    # But this can have numerical issues. Use direct with constraint instead.
    
    def objective(w):
        # w are free parameters, a = softmax(w)
        w_max = w.max()
        a_ = np.exp(w - w_max)
        a_ /= a_.sum()
        c = np.convolve(a_, a_)
        return float(c.max())
    
    def jac(w):
        w_max = w.max()
        a_ = np.exp(w - w_max)
        a_ /= a_.sum()
        c = np.convolve(a_, a_)
        k_star = int(c.argmax())
        # dc[k_star]/dw[j] = dc[k_star]/da[j] * da[j]/dw[j]
        # da[j]/dw[j] = a[j] - a[j]*sum(a) = a[j] - a[j] = 0 (since sum(a)=1)
        # da[j]/dw[l] = a[j]*(delta(j,l) - a[l])
        # dc[k]/da[j] = 2*a[k-j] (with bounds)
        grad_a = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad_a[j] = 2.0 * a_[idx]
        # Now dc/dw = (dc/da) @ (da/dw) = grad_a * a_ - a_ * sum(grad_a * a_)
        grad_w = grad_a * a_ - a_ * np.dot(grad_a, a_)
        return grad_w
    
    try:
        w0 = np.log(a + 1e-30)
        res = minimize(objective, w0, method='L-BFGS-B', jac=jac,
                       options={'maxiter': 5000, 'maxfun': 10000,
                                'factr': 1e10})
        if res.success or res.fun < objective(w0):
            w_max = res.x.max()
            a_new = np.exp(res.x - w_max)
            a_new /= a_new.sum()
            return _norm(a_new)
    except Exception:
        pass
    return a_init  # return original if polish failed
