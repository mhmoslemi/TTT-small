def _subgradient_step(a, n, step_size, top_k=20):
    """
    One subgradient step for minimizing max(conv(a,a)) subject to sum(a)=1, a>=0
    """
    b = _conv_full(a)
    M = float(b.max())

    active = np.where(b > 0.9 * M)[0]

    # Subgradient: grad[j] = 2 * sum_{k in active} a[k-j]
    grad = np.zeros(n)
    for k in active:

        jmax = min(n-1, k)
        if jmin <= jmax:

    # Normalize subgradient
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-12:
        return a.copy(), M
    
    # Step

    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy(), M
    a_new /= s
    return a_new, M

def _coordinate_polish(a, stop_time, seed):
    """Coordinate and pairwise search"""
    n = len(a)
    best = np.clip(a.astype(np.float64), 0, None).copy()
    s = best.sum()
    if s < 1e-10:
        return best, math.inf
    best = best / s
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 40
    
    while time.monotonic() < stop_time - 1.0:
        improved = False
        
        # Coordinate descent
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            candidates = []
            if orig > 1e-8:
                candidates = [0.0, orig*0.3, orig*0.5, orig*0.7, orig, orig*1.3, orig*1.6, orig*2.0]
            else:
                candidates = [0.0, 0.001, 0.005, 0.01, 0.05]
            
            best_val = orig
            best_val_sc = best_sc
            for cv in candidates:
                best[i] = cv
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score_fast(best)
                if sc < best_val_sc - 1e-14:
                    best_val = cv
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True
        
        # Pairwise moves
        n_pairs = min(200, n)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 1.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j: continue
            oi, oj = best[i], best[j]
            for frac in (0.0, 0.25, 0.5, 0.75, 1.0):
                best[i] = frac * oi
                best[j] = oj + oi - frac * oi
                if best.sum() < 0.01:
                    break
                sc = _score_fast(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj
        
        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0
    
    return best, best_sc

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n, idx=None):
    if idx is None:
        idx = np.arange(2 * n - 1)
    A = np.zeros((len(idx), n))
    for row, i in enumerate(idx):
        jmin = max(0, i - n + 1)
        jmax = min(n - 1, i)
        if jmin <= jmax:
            A[row, jmin:jmax + 1] = f[i - jmax:i - jmin + 1][::-1]
    return A


def _lp_solve(f, n, M, idx=None, time_limit=3.0):
    if idx is None:
        idx = np.arange(2 * n - 1)
    A = _build_A(f, n, idx)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(len(idx), M),
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs


def _lp_optimize(f0, n, stop_time, seed):

    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None)

    else:
        f = f[:n].copy()
    fs = float(f.sum())
    if fs < 1e-10:

    else:

    b = _conv_full(f)
    best_M = float(b.max())
    best_score = 2.0 * n * best_M  # sum is 1

    t = 0.3
    stall = 0
    max_stall = 60
    iters = 0

        M = float(b_cur.max())

                stall = 0

            if stall > max_stall:

                if fs > 1e-10:

            else:

def _subgradient_optimize(a0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    a = np.clip(np.asarray(a0, dtype=np.float64).ravel(), 0, None)
    if a.size < n:

    else:
        a = a[:n].copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_sc = _score_fast(a)

    alpha = 1.0
    t_count = 0
    no_imp = 0

    while time.monotonic() < stop_time - 0.5:
        b = _conv_full(a)
        M = float(b.max())

        # Active set
        thresh = M * 0.97
        active_mask = b >= thresh
        if active_mask.sum() > 30:
            active_mask = np.zeros_like(b)
            top_idx = np.argsort(-b)[:30]
            active_mask[top_idx] = True
        active = np.where(active_mask)[0]

        if len(active) == 0:
            break

        # Subgradient
        grad = np.zeros(n)
        for k in active:
            jmin = max(0, k - n + 1)
            jmax = min(n - 1, k)
            if jmin <= jmax:
                grad[jmin:jmax + 1] += 2.0 * a[k - jmax:k - jmin + 1][::-1]
        grad /= len(active)

        a_new = a - alpha * grad
        a_new = np.clip(a_new, 0, None)
        sn = a_new.sum()
        if sn > 1e-10:
            a_new /= sn
        else:
            a_new = a.copy()

        sc = _score_fast(a_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            a = a_new
            no_imp += 1

        t_count += 1
        if no_imp > 40:
            alpha *= 0.7
            no_imp = 0
            if alpha < 0.005:
                break

        # Periodic exploration
        if t_count % 200 == 100 and alpha > 0.1:
            a = np.clip(a + rng.normal(0, 0.03, n), 0, None)
            s2 = a.sum()
            if s2 > 1e-10:
                a /= s2

    return best_a, best_sc


def _coordinate_polish(a, stop_time, seed):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None)
    if best.size < n:
        best = np.concatenate([best, np.zeros(n - best.size)])
    else:
        best = best[:n].copy()
    s = best.sum()
    if s < 1e-10:
        return best, math.inf
    best = best / s
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 30

    while time.monotonic() < stop_time - 0.5:
        improved = False

        # Coordinate descent
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 0.5:
                break
            orig = best[i]
            if orig > 1e-8:
                cands = [0.0, orig * 0.4, orig * 0.6, orig * 0.8, orig,
                         orig * 1.2, orig * 1.5, orig * 2.0]
            else:
                cands = [0.0, 0.002, 0.01, 0.05]

            bv, bvs = orig, best_sc
            for cv in cands:
                best[i] = cv
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score_fast(best)
                if sc < bvs - 1e-14:
                    bv, bvs = cv, sc
            best[i] = bv
            if bvs < best_sc - 1e-14:
                best_sc = bvs
                improved = True

        # Pairwise
        n_pairs = min(150, n)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 0.5:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for frac in (0.0, 0.3, 0.5, 0.7, 1.0):
                best[i] = frac * oi
                best[j] = oj + oi - frac * oi
                if best.sum() < 0.01:
                    break
                sc = _score_fast(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc

(a*a)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * a[k-i]

For a specific j:
- a[j] appears in the product a[i]*a[k-i] when i=j (contribution a[k-j] if k-j in [0,n-1])
- a[j] appears when k-i=j, i.e., i=k-j (contribution a[j] if k-j in [0,n-1] and k-j ≠ j)
- If i=j=k-j (i.e., 2j=k), the term is a[j]^2, and d/d(a[j]) = 2*a[j]

Case 1: 0 <= k-j < n and j ≠ k-j:
  Two terms: a[j]*a[k-j] and a[k-j]*a[j]
  d/d(a[j]) = a[k-j] + a[k-j] = 2*a[k-j]

Case 2: 0 <= k-j < n and j = k-j (i.e., 2j=k, k even):
  One term: a[j]*a[j] = a[j]^2
  d/d(a[j]) = 2*a[j] = 2*a[k-j]

Case 3: k-j not in [0,n-1]:
  d/d(a[j]) = 0

In all cases: d/d(a[j]) of (a*a)[k] = 2*a[k-j] if 0 <= k-j < n, else 0.

So the subgradient for the max is:
subgrad[j] = 2/a(K) * sum_{k in K} a[k-j] (for j in [0,n-1])

where K is the set of active positions (where (a*a)[k] = max) and a(K) = |K|.

Or without averaging:
subgrad[j] = sum_{k in K} 2*a[k-j]

The subgradient descent step:
a_new = a - alpha * subgrad / ||subgrad||
a_new = clip(a_new, 0, None)
a_new = a_new / sum(a_new)

                    best_M = Mn
                    best_f = f.copy()
                    best_sc = 2.0 * n * Mn
                ok = True
                tgt = min(0.65, tgt * 1.12)
                stall = 0
                break
            tgt = max(0.03, tgt * 0.5)

        if not ok:
            stall += 1
        it += 1

    return best_f, best_sc


def _sg_refine(a0, n, stop_t, seed):
    rng = np.random.default_rng(seed)
    a = np.clip(np.asarray(a0, dtype=np.float64).ravel(), 0.0, None)
    if a.size < n:
        a = np.concatenate([a, np.zeros(n - a.size)])
    else:
        a = a[:n].copy()
    s = float(a.sum())
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_sc = _score_fast(a)
    alpha = 1.0
    no_imp = 0
    tc = 0

    while time.monotonic() < stop_t - 0.5:
        b = _conv_full(a)
        M = float(b.max())
        mask = b >= M * 0.97
        if mask.sum() > 30:
            mask = np.zeros(b.shape, dtype=bool)
            mask[np.argsort(-b)[:30]] = True
        act = np.where(mask)[0]
        if len(act) == 0:
            break

        grad = np.zeros(n)
        for k in act:
            j0 = max(0, k - n + 1)
            j1 = min(n - 1, k)
            if j0 <= j1:
                grad[j0 : j1 + 1] += 2.0 * a[k - j1 : k - j0 + 1][::-1]
        grad /= len(act)

        an = np.clip(a - alpha * grad, 0.0, None)
        sn = float(an.sum())
        an = an / sn if sn > 1e-10 else a.copy()

        sc = _score_fast(an)
        a = an
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        tc += 1
        if no_imp > 40:
            alpha *= 0.7
            no_imp = 0
            if alpha < 0.005:
                break
        if tc % 200 == 100 and alpha > 0.1:
            a = np.clip(a + rng.normal(0, 0.03, n), 0.0, None)
            s2 = float(a.sum())
            if s2 > 1e-10:
                a /= s2

    return best_a, best_sc


def _coord_polish(a, n, stop_t, seed):
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0.0, None)
    if best.size < n:
        best = np.concatenate([best, np.zeros(n - best.size)])
    else:
        best = best[:n].copy()
    s = float(best.sum())
    if s < 1e-10:
        return best, math.inf
    best = best / s
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_ni = 35

    while time.monotonic() < stop_t - 0.5:
        improved = False

        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_t - 0.5:
                break
            orig = best[i]
            if orig > 1e-8:
                cands = [0.0, orig * 0.4, orig * 0.6, orig * 0.8, orig,
                         orig * 1.25, orig * 1.6, orig * 2.0]
            else:
                cands = [0.0, 0.002, 0.01, 0.05]
            bv, bvs = orig, best_sc
            for cv in cands:
                best[i] = cv
                if float(best.sum()) < 0.01:
                    best[i] = orig
                    continue
                sc = _score_fast(best)
                if sc < bvs - 1e-14:
                    bv, bvs = cv, sc
            best[i] = bv
            if bvs < best_sc - 1e-14:
                best_sc = bvs
                improved = True

        n_pairs = min(150, n)
        for _ in range(n_pairs):
            if time.monotonic() > stop_t - 0.5:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for frac in (0.0, 0.3, 0.5, 0.7, 1.0):
                best[i] = frac * oi
                best[j] = oj + oi - frac * oi
                if float(best.sum()) < 0.01:
                    break
                sc = _score_fast(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
            if no_imp >= max_ni:
                break
        else:
            no_imp = 0

    return best, best_sc


def _try_update(best_seq, best_score, candidate, tag=""):
    try:
        rc = np.clip(candidate, 0.0, 1000.0)
        if not np.all(np.isfinite(rc)) or rc.sum() < 0.01:
            return best_seq, best_score
        exact = float(evaluate_sequence([float(x) for x in rc]))
        if math.isfinite(exact) and exact < best_score:
            return rc.copy(), exact
    except Exception:
        pass
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    fallback = parent.copy()
    fallback_sc = _score_fast(fallback) if n_p > 0 else math.inf
    if not math.isfinite(fallback_sc) or n_p == 0:
        fallback = np.ones(384.0) / 384.0
        fallback_sc = _score_fast(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc
    n = n_p

    i_arr = np.arange(n, dtype=np.float64)
    rng_main = np.random.default_rng(seed)

    inits = []
    inits.append(parent.copy())
    inits.append(0.5 + 0.5 * np.cos(math.pi * i_arr / (n - 1)))
    inits.append(np.abs(0.5 + 0.4 * np.cos(4.0 * math.pi * i_arr / n)) + 0.01)
    sm = np.convolve(rng_main.normal(0, 1, n),
                     np.exp(-0.5 * (np.arange(-6, 7) / 2.5) ** 2), "same")
    inits.append(np.abs(sm) + 0.01)
    inits.append(
        np.abs(parent + rng_main.normal(0, 0.12 * max(1.0, float(np.std(parent))), n))
        + 0.005
    )
    plate = np.ones(n)
    c = n // 2
    w = n // 4
    plate[c - w : c + w] = 1.8
    inits.append(plate)

    # --- Phase 1: LP projection on diverse inits (58% time) ---
    p1_end = t0 + budget * 0.58
    tp_init = max(12.0, (p1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > p1_end - 5.0:
            break
        if best_score < 1.5027:
            break
        stop = min(p1_end - 3.0, time.monotonic() + tp_init)
        if time.monotonic() >= stop:
            break
        try:
            res, sc = _lp_run(init, n, stop, seed + 10007 * (idx + 1))
            best_seq, best_score = _try_update(best_seq, best_score, res, "lp")
        except Exception:
            pass

    # --- Phase 2: Subgradient refine (14% time) ---
    p2_end = t0 + budget * 0.72
    if time.monotonic() < p2_end - 4.0:
        sg_stop = min(p2_end - 2.0, time.monotonic() + budget * 0.13)
        try:
            sg_res, sg_sc = _sg_refine(best_seq, n, sg_stop, seed + 55555)
            best_seq, best_score = _try_update(best_seq, best_score, sg_res, "sg")
        except Exception:
            pass

    # --- Phase 3: Coordinate polish (10% time) ---
    p3_end = t0 + budget * 0.82
    if time.monotonic() < p3_end - 3.0:
        cp_stop = min(p3_end - 2.0, time.monotonic() + budget * 0.09)
        try:
            cp_res, cp_sc = _coord_polish(best_seq, n, cp_stop, seed + 44444)
            best_seq, best_score = _try_update(best_seq, best_score, cp_res, "cp1")
        except Exception:
            pass

    # --- Phase 4: Second LP round from polished result (10% time) ---
    p4_end = deadline - 5.0
    if time.monotonic() < p4_end - 3.0:
        try:
            res, sc = _lp_run(best_seq, n, p4_end, seed + 33333)
            best_seq, best_score = _try_update(best_seq, best_score, res, "lp2")
        except Exception:
            pass

    # --- Final polish ---
    if time.monotonic() < deadline - 2.0:
        try:
            cp_res, cp_sc = _coord_polish(best_seq, n, deadline, seed + 22222)
            best_seq, best_score = _try_update(best_seq, best_score, cp_res, "cp2")
        except Exception:
            pass

    # Final exact validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            fb_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            fb_sc = math.inf
        if math.isfinite(fb_sc):
            return [float(x) for x in fallback]
        return [1.0]

    return [float(x) for x in final_seq]
