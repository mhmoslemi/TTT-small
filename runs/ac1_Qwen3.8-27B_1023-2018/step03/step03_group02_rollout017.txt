def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.where(u * np.arange(1, n + 1) > (cssv - 1), np.arange(1, n + 1)))
    theta = cssv[int(rho) - 1] / rho
    return np.maximum(v - theta, 0.0)


def _smooth_max_grad(a, alpha):
    """Gradient of (1/alpha)*log(sum(exp(alpha*(a*a)_k))) w.r.t. a."""
    n = len(a)
    b = _conv_full(a)
    b_max = b.max()
    e = np.exp(alpha * (b - b_max))
    w = e / e.sum()  # length 2n-1
    # grad_i = 2 * sum_{j=0}^{n-1} a_j * w_{i+j}
    # = 2 * (a_rev * w)_{n-1+i}
    fsize = next_fast_len(3 * n)
    a_rev = a[::-1]
    AR = rfft(a_rev, fsize)
    W = rfft(w, fsize)
    conv = irfft(AR * W, fsize)
    grad = 2.0 * conv[n - 1: n - 1 + n]
    return grad, b

def _pairwise_cd(a, stop_time, seed=0):

            f_new = _project_simplex(f_new)
            f_sum_new = f_new.sum()
            if f_sum_new < 0.01:
                t = max(0.05, t * 0.5)
                continue
            sc = _score(f_new)
            if sc < best_score - 1e-15:
                best_score = sc
                best_f = f_new.copy()
                f = f_new
                b_current = _conv_full(f)
                M = float(b_current.max())
                for j in range(n):
                    A[j:j + n, j] = f
                improved = True
                t = min(0.9, t * 1.12)
                stall = 0
                break
            else:
                t = max(0.05, t * 0.5)

        if not improved:
            stall += 1
            if stall > 30:
                f = _project_simplex(f + rng.normal(0, 0.08, n))
                b_current = _conv_full(f)
                M = float(b_current.max())
                for j in range(n):
                    A[j:j + n, j] = f
                t = 0.5
                stall = 0
        iter_count += 1

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n = len(parent)

    # Normalize to simplex
    u = parent.copy()
    u_sum = u.sum()
    if u_sum < 0.01:
        u = np.ones(n) / n
    else:
        u = u / u_sum

    best_u = u.copy()
    best_score = _score(u)

    # Validate with exact evaluator
    try:
        vs = float(evaluate_sequence([float(x) for x in best_u]))
        if math.isfinite(vs) and vs < best_score:
            best_score = vs
    except Exception:
        pass

    # Phase 1: LP refinement (30% of budget)
    lp_stop = t0 + budget * 0.30
    try:
        res, sc = _lp_refine(best_u, n, lp_stop, seed)
        if sc < best_score and np.all(np.isfinite(res)):
            best_u = res
            best_score = sc
    except Exception:
        pass

    # Phase 2: Gradient / multiplicative optimization (45% of budget)
    grad_stop = min(t0 + budget * 0.75, deadline - 120)
    configs = [
        ('grad', 5.0, 50.0),
        ('mw', 10.0, 100.0),
        ('grad', 50.0, 300.0),
        ('mw', 100.0, 500.0),
    ]
    time_per = max(15.0, (grad_stop - t0) / len(configs))

    for idx, (method, a1, a2) in enumerate(configs):
        if time.monotonic() > grad_stop - 5.0:
            break
        p_stop = min(grad_stop - 5.0, time.monotonic() + time_per)
        if time.monotonic() >= p_stop - 2.0:
            break
        try:
            if method == 'grad':
                res, sc = _gradient_optimize(best_u, p_stop, seed + 500 * idx + 1, a1, a2)
            else:
                res, sc = _mult_weight_optimize(best_u, p_stop, seed + 500 * idx + 1, a1, a2)
            if sc < best_score and np.all(np.isfinite(res)):
                best_u = res
                best_score = sc
        except Exception:
            pass

    # Phase 3: Coordinate descent (25% of budget)
    cd_stop = min(deadline - 30, t0 + budget * 0.93)
    cd_time = max(10.0, (cd_stop - t0) * 0.25)
    try:
        p1 = time.monotonic() + cd_time
        res, sc = _coord_descent(best_u, p1, seed + 9001)
        if sc < best_score and np.all(np.isfinite(res)):
            best_u = res
            best_score = sc
    except Exception:
        pass

    # Pairwise CD
    if time.monotonic() < cd_stop - 5.0:
        try:
            p2 = min(cd_stop - 5.0, time.monotonic() + cd_time * 0.7)
            res, sc = _pairwise_cd(best_u, p2, seed + 9002)
            if sc < best_score and np.all(np.isfinite(res)):
                best_u = res
                best_score = sc
        except Exception:
            pass

    # Phase 4: Final short gradient + CD
    if time.monotonic() < deadline - 15.0:
        final_stop = deadline - 5.0
        rem = final_stop - time.monotonic()
        try:
            gs = min(final_stop - rem * 0.4, time.monotonic() + rem * 0.5)
            res, sc = _gradient_optimize(best_u, gs, seed + 7777, 200.0, 800.0)
            if sc < best_score and np.all(np.isfinite(res)):
                best_u = res
                best_score = sc
        except Exception:
            pass
        try:
            res, sc = _coord_descent(best_u, final_stop - 2.0, seed + 8888)
            if sc < best_score and np.all(np.isfinite(res)):
                best_u = res
                best_score = sc
        except Exception:
            pass

    # Final validation
    final_u = np.clip(best_u, 0.0, 1000.0)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_u]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Fallback to parent
        try:
            fb = np.clip(parent, 0.0, 1000.0)
            fb_sc = float(evaluate_sequence([float(x) for x in fb]))
        except Exception:
            fb_sc = math.inf
        if math.isfinite(fb_sc):
            return [float(x) for x in fb]
        return [1.0]

    return [float(x) for x in final_u]
