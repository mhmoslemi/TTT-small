def _slices_for_n(n):
    """Precompute the valid segment for each convolution row when building A."""

        hi = min(n - 1, k)

        ln = hi - lo + 1

    _row_cache[n] = sl

def _exact_score(a):
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return float("inf")
    if arr.size == 0:
        return float("inf")
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return float("inf")
    n = arr.size
    try:
        c = np.convolve(arr, arr)
    except Exception:
        return float("inf")
    if c.size == 0:
        return float("inf")
    mb = float(np.max(c))
    if not np.isfinite(mb):
        return float("inf")
    return float(2.0 * n * mb / (s * s))


def _lp_slope_direction(a, ca, K, slices, S, time_left):
    """
    Solve the first-order active-set direction LP:
        minimize  mu
        subject to (a*h)_k - mu <= (a*a)_k for k in K
                   sum(h) = S, h >= 0
    """
    n = a.size
    K = np.asarray(K, dtype=int).ravel()
    K = K[(K >= 0) & (K < ca.size)]
    if K.size == 0:
        return None, None

    A = np.zeros((K.size, n), dtype=np.float64)
    for i, k in enumerate(K):
        k = int(k)
        lo, hi1, st, ln = slices[k]
        if ln > 0:
            A[i, lo:hi1] = a[st:st + ln][::-1]

    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0
    Aeq = np.ones((1, n), dtype=np.float64)
    b_eq = np.array([S], dtype=np.float64)
    bounds = [(0.0, None)] * n + [(-S * S - 1.0, S * S + 1.0)]

    try:
        tl = float(np.clip(0.25 * max(0.0, time_left), 0.4, 2.0))
        res = linprog(
            c,
            A_ub=A,
            b_ub=ca[K],
            A_eq=Aeq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": tl},
        )
    except Exception:
        return None, None

    if res is None or res.status != 0 or getattr(res, "x", None) is None:
        return None, None

    h = np.asarray(res.x[:n], dtype=np.float64)
    mu = float(res.x[-1])
    if not np.all(np.isfinite(h)) or not np.isfinite(mu):
        return None, None
    h = np.maximum(h, 0.0)
    hs = h.sum()
    if hs <= 1e-12:
        return None, None
    h = h * (S / hs)
    return h, mu


def _line_search(a, h, ca, M, S):
    """Line search from a to h, using precomputed current convolution ca."""
    n = a.size
    h = np.asarray(h, dtype=np.float64)
    if h.size != n or not np.all(np.isfinite(h)):
        return None, M
    h = np.maximum(h, 0.0)
    hs = h.sum()
    if hs <= 1e-12:
        return None, M
    h = h * (S / hs)

    try:
        ch = np.convolve(h, h)
        cah = np.convolve(a, h)
    except Exception:
        return None, M
    if not (np.all(np.isfinite(ch)) and np.all(np.isfinite(cah))):
        return None, M

    best_val = float(M)
    best_alpha = 0.0

    alphas = np.unique(np.concatenate((
        np.geomspace(1e-7, 1.0, 64),
        np.linspace(0.0, 1.0, 101),
    )))
    if alphas.size > 0:
        A = alphas[:, None]
        O = 1.0 - A
        vals = np.max(
            O * O * ca[None, :]
            + 2.0 * A * O * cah[None, :]
            + A * A * ch[None, :],
            axis=1,
        )
        vals = np.where(np.isfinite(vals), vals, np.inf)
        positive_mask = alphas > 1e-14
        if np.any(positive_mask):
            valid_idx = np.where(positive_mask)[0]
            top = valid_idx[np.argsort(vals[valid_idx])][:3]
            for ii in top:
                al = float(alphas[ii])
                vv = float(vals[ii])
                if al > 1e-14 and vv < best_val - 1e-12:
                    best_val = vv
                    best_alpha = al

            if best_alpha > 1.0 or best_alpha < 0.0:
                return None, best_val

            def f(alpha):
                alpha = float(alpha)
                oa = 1.0 - alpha
                return float(np.max(
                    oa * oa * ca
                    + 2.0 * alpha * oa * cah
                    + alpha * alpha * ch
                ))

            def golden_min(lo, hi, max_iter=35):
                if hi <= lo + 1e-12:
                    x = min(max(lo, 0.0), 1.0)
                    return x, f(x)

                        c = d
                        fc = fd
                        d = lo + invphi * (hi - lo)

                    best_alpha = ga

    if best_alpha <= 1e-14:
        return None, best_val

    cand = (1.0 - best_alpha) * a + best_alpha * h
    cand = np.maximum(cand, 0.0)

        return None, best_val

                if np.all(np.isfinite(p)):
                    return p

    for eta in (0.20, 0.10, 0.05, 0.02, 0.01):
        try:
            p = a * np.exp(-eta * grad)
        except Exception:
            p = a.copy()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 1e-12 * S / n)
        ps = p.sum()
        if ps > 1e-12:
            p = p * (S / ps)
            if np.all(np.isfinite(p)):
                return p

    if gnorm > 1e-12:
        d = -grad / gnorm
    else:
        d = rng.normal(0.0, 1.0, n)
    d = d - d.mean()
    dn = float(np.linalg.norm(d))
    if dn > 1e-12:
        p = a + 0.02 * S * d / dn
        p = np.maximum(p, 0.0)
        ps = p.sum()
        if ps > 1e-12:
            p = p * (S / ps)
            if np.all(np.isfinite(p)):
                return p
    return None


def _perturb_candidate(a, S, rng, sigma):
    n = a.size
    try:
        p = a * np.exp(rng.normal(0.0, sigma, n))
    except Exception:
        p = a.copy()
    p = np.where(np.isfinite(p), p, 0.0)
    floor = 1e-9 * S / n
    p = np.maximum(p, floor)

    if rng.random() < 0.30:
        eps = min(0.15, sigma)
        try:
            r = S * rng.dirichlet(np.ones(n, dtype=np.float64) * 0.5)
        except Exception:
            r = None
        if r is not None and np.all(np.isfinite(r)):
            p = (1.0 - eps) * p + eps * r

    ps = p.sum()
    if ps <= 1e-12 or not np.isfinite(ps):
        return a.copy()
    p = p * (S / ps)
    return np.where(np.isfinite(p), p, floor)


def _optimize_start(a0, stop_time, seed):
    arr0 = np.asarray(a0, dtype=np.float64).ravel()
    if arr0.size < 8:
        arr0 = np.ones(201, dtype=np.float64)
    n = arr0.size
    S = math.sqrt(2.0 * n)
    a = _normalize_to_S(arr0, S)

    best_a = a.copy()
    best_score = _exact_score(a)
    if not np.isfinite(best_score):
        best_a = np.ones(n, dtype=np.float64) * (S / n)
        best_score = _exact_score(best_a)

    rng = np.random.default_rng(seed)
    m = 2 * n - 1
    slices = _slices_for_n(n)

    sizes = [min(120, m), min(50, m), min(20, m), min(8, m), m]
    uniq = []
    for s in sizes:
        if s > 0 and s not in uniq:
            uniq.append(s)
    if not uniq:
        uniq = [max(1, m)]

    kidx = 0
    fails = 0
    it = 0
    perturb_count = 0

    while time.monotonic() < stop_time - 1.2:
        if not np.all(np.isfinite(a)) or a.size != n:
            a = best_a.copy().astype(np.float64)
            S = math.sqrt(2.0 * n)
            a = _normalize_to_S(a, S)
            continue

        try:
            ca = np.convolve(a, a)
        except Exception:
            a = best_a.copy()
            continue
        if ca.size != m or not np.all(np.isfinite(ca)):
            a = best_a.copy()
            continue

        M = float(np.max(ca))
        s = float(a.sum())
        if s <= 1e-12 or not np.isfinite(M):
            a = best_a.copy()
            continue

        sc = float(2.0 * n * M / (s * s))
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()

        if it % 40 == 0:
            sym = 0.5 * (a + a[::-1])
            sym = _normalize_to_S(sym, S)
            sc_sym = _exact_score(sym)
            if np.isfinite(sc_sym) and sc_sym < best_score - 1e-10:
                best_score = sc_sym
                best_a = sym.copy()
                a = sym.copy()
                it += 1
                continue

        if time.monotonic() > stop_time - 2.0:
            break

        ksize = uniq[kidx % len(uniq)]
        order = np.argsort(ca)[::-1]
        K = order[:ksize]

        improved = False
        h, mu = _lp_slope_direction(a, ca, K, slices, S, stop_time - time.monotonic())
        if h is not None and np.isfinite(mu) and mu < -1e-9:
            if float(np.linalg.norm(h - a)) > 1e-9 * S:
                variants = [h]
                if float(np.linalg.norm(h - h[::-1])) > 1e-7 * S:
                    h2 = 0.5 * (h + h[::-1])
                    h2 = _normalize_to_S(h2, S)
                    if np.all(np.isfinite(h2)) and not np.allclose(h2, h, rtol=1e-10, atol=1e-12):
                        variants.append(h2)

                for hv in variants:
                    if time.monotonic() > stop_time - 2.2:
                        break
                    cand, val = _line_search(a, hv, ca, M, S)
                    if cand is not None and val < M - 1e-10:
                        cs = float(cand.sum())
                        if cs <= 1e-12:
                            continue
                        new_score = float(2.0 * n * val / (cs * cs))
                        if not np.isfinite(new_score):
                            continue
                        a = cand
                        improved = True
                        fails = 0
                        kidx = 0
                        if new_score < best_score - 1e-12:
                            best_score = new_score
                            best_a = a.copy()
                        break

        if improved:
            it += 1
            continue

        if fails == 2:
            p = _gradient_candidate(a, ca, M, S, rng)
            if p is not None:
                cand, val = _line_search(a, p, ca, M, S)
                if cand is not None and val < M - 1e-10:
                    cs = float(cand.sum())
                    if cs > 1e-12:
                        new_score = float(2.0 * n * val / (cs * cs))
                        if np.isfinite(new_score):
                            a = cand
                            fails = 0
                            kidx = 0
                            if new_score < best_score - 1e-12:
                                best_score = new_score
                                best_a = a.copy()
                            it += 1
                            continue

        fails += 1
        if fails >= len(uniq):
            sigma = 0.020 + 0.012 * min(perturb_count, 10)
            if perturb_count > 6:
                sigma += 0.05
            a = _perturb_candidate(a, S, rng, sigma)
            fails = 0
            kidx = 0
            perturb_count += 1
        else:
            kidx += 1

        it += 1

    sym = 0.5 * (best_a + best_a[::-1])
    sym = _normalize_to_S(sym, math.sqrt(2.0 * best_a.size))
    sc_sym = _exact_score(sym)
    if np.isfinite(sc_sym) and sc_sym < _exact_score(best_a) - 1e-12:
        best_a = sym

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not np.isfinite(budget) or budget <= 1.0:
        budget = 1300.0

    # Keep a hard safety margin for final validation.
    deadline = t0 + max(5.0, budget - 25.0)

    parent = _get_parent()
    best_a = _normalize_to_S(parent, math.sqrt(2.0 * parent.size))
    best_score = _exact_score(best_a)

    if not np.isfinite(best_score) or best_a.size <= 0:
        best_a = np.ones(201, dtype=np.float64)
        best_a = _normalize_to_S(best_a, math.sqrt(2.0 * best_a.size))
        best_score = _exact_score(best_a)

    def _consider(a):
        nonlocal best_a, best_score
        if a is None:
            return
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return
        if arr.size < 2:
            return
        S = math.sqrt(2.0 * arr.size)
        arr = _normalize_to_S(arr, S)
        if not np.all(np.isfinite(arr)):
            return
        sc = _exact_score(arr)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_a = arr.copy()

    # Cheap geometric sanity moves before optimizing.
    rng0 = np.random.default_rng(int(seed) if np.isfinite(seed) else 42)
    _consider(0.5 * (best_a + best_a[::-1]))
    for sig in (0.03, 0.08):
        try:
            noisy = best_a * np.exp(rng0.normal(0.0, sig, best_a.size))
        except Exception:
            noisy = best_a.copy()
        _consider(_normalize_to_S(noisy, math.sqrt(2.0 * best_a.size)))

    avail = max(10.0, deadline - t0)
    if budget >= 400.0:
        primary_stop = t0 + 0.55 * avail
        sec_stop = t0 + 0.78 * avail
        len_stop = t0 + 0.93 * avail
        final_stop = deadline - 5.0
    elif budget >= 120.0:
        primary_stop = t0 + 0.70 * avail
        sec_stop = t0 + 0.90 * avail
        len_stop = final_stop = deadline - 5.0
    else:
        primary_stop = sec_stop = len_stop = final_stop = deadline - 5.0

    # Primary: strongest effort from the supplied construction.
    if time.monotonic() < primary_stop - 2.0:
        try:
            res = _optimize_start(best_a, primary_stop, int(seed) + 11)
            _consider(res)
        except Exception:
            pass

    # Secondary: same length, different basin initializations.
    if time.monotonic() < sec_stop - 10.0:
        base = best_a.copy()
        n = base.size
        S = math.sqrt(2.0 * n)
        x = np.linspace(0.0, 1.0, n)

        shape1 = 1.0 + 0.35 * np.sin(np.pi * x)
        shape2 = 1.0 + 0.25 * np.cos(2.0 * np.pi * x)
        shape3 = 1.0 + 0.20 * np.sin(4.0 * np.pi * x)
        shape4 = 1.0 + 0.12 * np.cos(6.0 * np.pi * x)
        flat = np.ones(n, dtype=np.float64)

        for sh in (shape1, shape2, shape3, shape4):
            sh[:] = np.maximum(sh, 1e-3)

        inits = []
        try:
            inits.append(base * np.exp(rng0.normal(0.0, 0.06, n)))
        except Exception:
            inits.append(base.copy())
        inits.append(0.70 * base + 0.30 * _normalize_to_S(shape1, S))
        inits.append(0.65 * base + 0.35 * _normalize_to_S(shape2, S))
        inits.append(0.55 * base + 0.45 * _normalize_to_S(shape3, S))
        if base.size > 20:
            inits.append(0.50 * base + 0.50 * _normalize_to_S(flat, S))

        inits = inits[:4]
        if inits:
            remaining = sec_stop - time.monotonic()
            per = max(6.0, remaining / max(1, len(inits)))
            for i, init in enumerate(inits):
                if time.monotonic() > sec_stop - 8.0:
                    break
                per_stop = min(time.monotonic() + per, sec_stop - 4.0)
                try:
                    res = _optimize_start(init, per_stop, int(seed) + 300 + 10 * i)
                    _consider(res)
                except Exception:
                    pass

    # Tertiary: try nearby resolutions, initialized by resampling the current best.
    if time.monotonic() < len_stop - 10.0:
        base_len = best_a.size
        cands = []
        for dn in (-12, -8, -4, -2, -1, 1, 2, 4, 8, 12, 16):
            nt = base_len + dn
            if 16 <= nt <= 500:
                cands.append(nt)
        cands = sorted(set(cands), key=lambda q: abs(q - base_len))[:4]

        remaining = len_stop - time.monotonic()
        per = max(5.0, remaining / max(1, len(cands)))
        resample_base = best_a.copy()
        for nt in cands:
            if time.monotonic() > len_stop - 10.0:
                break
            blen = resample_base.size
            if nt == blen:
                init = resample_base.copy()
            else:
                xo = np.linspace(0.0, 1.0, blen)
                xn = np.linspace(0.0, 1.0, nt)
                init = np.interp(xn, xo, resample_base)
            init = _normalize_to_S(init, math.sqrt(2.0 * nt))
            per_stop = min(time.monotonic() + per, len_stop - 6.0)
            try:
                res = _optimize_start(init, per_stop, int(seed) + 9000 + nt)
                _consider(res)
                if not np.allclose(best_a, resample_base, rtol=1e-10, atol=1e-12):
                    resample_base = best_a.copy()
            except Exception:
                pass

    # Final polish.
    if time.monotonic() < final_stop - 4.0:
        try:
            res = _optimize_start(best_a, final_stop, int(seed) + 20000)
            _consider(res)
        except Exception:
            pass

    # Final validation and safe return.
    final = np.asarray(best_a, dtype=np.float64).ravel()
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0 or final.sum() < 0.01:
        final = np.ones(20, dtype=np.float64) / 20.0

    fallback = np.ones(201, dtype=np.float64)
    fallback = _normalize_to_S(fallback, math.sqrt(2.0 * fallback.size))

    ev = globals().get("evaluate_sequence", None)
    try:
        if callable(ev):
            sc = float(ev([float(x) for x in final]))
            if not np.isfinite(sc):
                fb = [float(x) for x in fallback]
                scf = float(ev(fb))
                if np.isfinite(scf):
                    final = fallback
                raise ValueError
            # Keep the earlier exact best if it is somehow better.
            sc_best = float(_exact_score(final))
            if np.isfinite(sc_best) and sc_best > sc + 1e-12:
                pass
    except Exception:
        try:
            scf = float(_exact_score(fallback))
            scb = float(_exact_score(final))
            if not np.isfinite(scb) or np.isfinite(scf) and scf < scb:
                final = fallback
        except Exception:
            final = fallback

    return [float(x) for x in final]
