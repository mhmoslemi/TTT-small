def _project_simplex(v,target=1.0):
   v=np.asarray(v,float).ravel(); n=v.size
   if n==0: return np.ones(1)
   u=np.sort(v)[::-1]
   css=np.cumsum(u)
   idx=np.arange(1,n+1,float)
   cond=(css-target)/idx < u
   rho=np.nonzero(cond)[0]
   if rho.size==0: return np.full(n,target/n)
   theta=(css[rho[-1]]-target)/(rho[-1]+1)
   w=np.maximum(v-theta,0.0)
   s=w.sum()
   if s<=0: return np.full(n,target/n)
   return w/s*target
```

def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)

    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    with np.errstate(divide="ignore", invalid="ignore"):

def _lbfgs_block(p0, n, block_end, tau, beta):

    if n < 2:

        if time.monotonic() >= block_end:

        return val, gth

        if time.monotonic() >= block_end:
            raise _TimeUp

    bounds = [( -_TH_BOUNDS, _TH_BOUNDS) for _ in range(n - 1)]

    last = state["last"]

def _random_smooth(n, rng, base=None):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, endpoint=False, dtype=np.float64)
    kind = int(rng.integers(0, 6))
    if kind == 0:
        a = x ** float(rng.uniform(0.3, 2.5))
    elif kind == 1:
        a = (1.0 - x) ** float(rng.uniform(0.3, 2.5))
    elif kind == 2:
        c = float(rng.uniform(0.15, 0.85))
        w = float(rng.uniform(0.05, 0.35))
        a = np.exp(-((x - c) ** 2) / (2.0 * w * w)) + 0.08
    elif kind == 3:
        a = np.ones(n, dtype=np.float64)
        ncomp = int(rng.integers(2, 4))
        for k in range(1, ncomp + 1):
            amp = float(rng.uniform(0.05, 0.8 / k))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            a += amp * np.cos(2.0 * np.pi * k * x + ph)
    elif kind == 4:
        a = np.ones(n, dtype=np.float64)
        nb = int(rng.integers(1, 3))
        for _ in range(nb):
            c = float(rng.uniform(0.1, 0.9))
            w = float(rng.uniform(0.05, 0.30))
            amp = float(rng.uniform(0.2, 1.2))
            a += amp * np.exp(-((x - c) ** 2) / (2.0 * w * w))
    else:
        a = rng.uniform(0.25, 1.35, n).astype(np.float64)
    a = np.where(np.isfinite(a), a, 0.0)

    if base is not None:
        base = np.asarray(base, dtype=np.float64).ravel()
        if base.size == n and np.all(np.isfinite(base)):
            a = 0.55 * _normalize(base) + 0.45 * _normalize(a)

def _basis_modes(n):
    if n < 4:
        return np.empty((max(1, n - 1), 0), dtype=np.float64)
    x = np.arange(n - 1, dtype=np.float64) / float(n - 1)
    cols = []

    def add(v):
        v = np.asarray(v, dtype=np.float64).ravel()
        if v.size != n - 1:
            return
        v = np.where(np.isfinite(v), v, 0.0)
        v -= v[-1]
        sd = float(v.std())
        if sd > 1e-12:
            cols.append(v / sd)

    add(x)
    add(x * x - x)
    t = 2.0 * x - 1.0
    add(t)
    add(t * (1.0 - t * t))
    add(t * t * t - t)
    for k in (1, 2, 3, 4):
        add(np.sin(2.0 * np.pi * k * x))
        add(np.cos(2.0 * np.pi * k * x))
    for c, w in ((0.25, 0.16), (0.75, 0.16), (0.5, 0.10)):
        add(np.exp(-((x - c) ** 2) / (2.0 * w * w)))

    d = min(12, len(cols))
    if d == 0:
        return np.empty((n - 1, 0), dtype=np.float64)
    return np.column_stack(cols[:d])

    out = []
    b = _normalize(base)
    if b.size == n:
        out.append(b)
    out.append(np.ones(n, dtype=np.float64) / n)

    out.append(_normalize(np.maximum(1.0 + 0.55 * np.cos(4.0 * np.pi * x), 1e-3)))
    out.append(_normalize(np.maximum(1.0 + 0.85 * np.exp(-((x - 0.5) ** 2) / (2.0 * 0.14 * 0.14)), 1e-3)))
    return out

            continue
        M0 = float(_score_only(b) / (2.0 * n))
        tau = max(0.12 * M0, 1e-12)
        beta = 0.03
        invL = 1.0 / (2 * n - 1)

        def alpha_fun(alpha, theta0=theta0, U=U, n=n, tau=tau, beta=beta, invL=invL, state=state):
            alpha = np.asarray(alpha, dtype=np.float64).ravel()
            if alpha.size != d:
                alpha = np.zeros(d, dtype=np.float64)
            theta = theta0 + scale_factor * (U @ alpha)
            val, gth, _, _, _ = _eval_theta(theta, n, tau, beta, invL, state)
            grad = (scale_factor * (U.T @ gth)) if d else np.zeros(d, dtype=np.float64)
            return val, np.where(np.isfinite(grad), grad, 0.0)

        starts = [np.zeros(d, dtype=np.float64)]
        starts.append(rng.normal(0.0, 0.35, d))
        starts.append(rng.normal(0.0, 0.90, d))
        if d >= 3:
            lin = np.zeros(d, dtype=np.float64)
            lin[0] = 0.9
            lin[2:5] = 0.2
            starts.append(lin)

        for a0 in starts:
            if time.monotonic() >= deadline - 1.0:
                break
            a0 = np.where(np.isfinite(a0), a0, 0.0)
            per_end = min(deadline - 0.5, time.monotonic() + 1.1)

            def cb(_x):
                if time.monotonic() >= per_end:
                    raise _TimeUp

            try:
                minimize(
                    alpha_fun,
                    a0,
                    jac=True,
                    method="L-BFGS-B",
                    bounds=[(-4.0, 4.0) for _ in range(d)],
                    options={"maxiter": 260, "ftol": 1e-12, "gtol": 1e-7, "maxls": 20},
                    callback=cb,
                )
            except _TimeUp:
                pass
            except Exception:
                pass

    best = state["best"]
    if best is None or not np.all(np.isfinite(best)):
        best = base.copy()
    return best, float(state["best_sc"])


def _stochastic_polish(p, n, deadline, rng):
    p = _normalize(p)
    best = p.copy()
    best_sc = _score_only(best)
    sigma = 0.025
    no = 0
    while time.monotonic() < deadline - 2.0:
        r = float(rng.random())
        if r < 0.45:
            cand = _project_simplex(p + rng.normal(0.0, sigma, n))
        elif r < 0.75:
            cand = _normalize(0.78 * best + 0.22 * _random_smooth(n, rng, base=best))
        else:
            cand = _random_smooth(n, rng, base=best)
        cand = np.asarray(cand, dtype=np.float64).ravel()
        if cand.size != n or not np.all(np.isfinite(cand)):
            continue
        sc = _score_only(cand)
        if math.isfinite(sc) and sc < best_sc - 1e-15:
            best = cand.copy()
            best_sc = float(sc)
            no = 0
            sigma = max(0.002, sigma * 0.85)
        else:
            no += 1
            sigma = min(0.16, sigma * 1.02)
            if no > 250:
                no = 0
                p = _random_smooth(n, rng, base=best)
                sigma = 0.05
        if time.monotonic() < deadline - 2.0:
            p = best
    return best, best_sc


def _conv_matrix_rows(p, rows):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    rows = np.asarray(rows, dtype=np.int64).ravel()
    A = np.zeros((rows.size, n), dtype=np.float64)
    for i, k in enumerate(rows):
        k = int(k)
        if k < 0 or k >= 2 * n - 1:
            continue
        if k < n:
            length = k + 1
            A[i, :length] = p[:length][::-1]
        else:
            length = 2 * n - 1 - k
            s = k - n + 1
            A[i, s:s + length] = p[n - 1:n - length - 1:-1]
    return A


def _linprog_safe(c_obj, A_ub, b_ub, bounds, tl):
    tl = max(0.25, float(tl))
    try:
        return linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": tl, "presolve": True},
        )
    except Exception:
        pass
    try:
        return linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
    except Exception:
        pass
    try:
        return linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds)
    except Exception:
        return None


def _solve_lp_dir(p, M, rows, time_limit, rng):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 4 or M <= 0 or not math.isfinite(M) or len(rows) == 0:
        return None
    A = _conv_matrix_rows(p, rows)
    if A.size == 0:
        return None
    A = A / M
    c_obj = np.full(n, -1.0, dtype=np.float64)
    tie = rng.standard_normal(n)
    if n > 1:
        tie = (tie - tie.mean()) / (float(np.std(tie)) + 1e-12)
    c_obj += 1e-6 * tie
    b_ub = np.ones(A.shape[0], dtype=np.float64)
    bounds = [(0.0, 10.0) for _ in range(n)]
    res = _linprog_safe(c_obj, A, b_ub, bounds, time_limit)
    if res is not None and getattr(res, "success", False):
        x = np.asarray(res.x, dtype=np.float64).ravel()
        if x.size == n and np.all(np.isfinite(x)):
            x = np.clip(x, 0.0, None)
            s = float(x.sum())
            if s > 1e-12:
                return x / s
    return None


def _solve_lp_dual(p, M, rows, time_limit, rng):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 4 or M <= 0 or not math.isfinite(M) or len(rows) == 0:
        return None
    A = _conv_matrix_rows(p, rows)
    if A.size == 0:
        return None
    A = A / M
    k = A.shape[0]
    c_obj = np.ones(k, dtype=np.float64)
    tie = rng.standard_normal(k)
    if k > 1:
        tie = (tie - tie.mean()) / (float(np.std(tie)) + 1e-12)
    c_obj += 1e-6 * tie
    A_ub = -A.T
    b_ub = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, 1e6) for _ in range(k)]
    res = _linprog_safe(c_obj, A_ub, b_ub, bounds, time_limit)
    if res is not None and getattr(res, "success", False):
        lam = np.asarray(res.x, dtype=np.float64).ravel()
        if lam.size == k and np.all(np.isfinite(lam)):
            lam = np.clip(lam, 0.0, None)
            s = float(lam.sum())
            if s > 1e-12:
                full = np.zeros(2 * n - 1, dtype=np.float64)
                full[np.asarray(rows, dtype=np.int64)] = lam / s
                return full
    return None


def _lp_polish(p, n, deadline, rng):
    p = _normalize(p)
    best = p.copy()
    best_sc = _score_only(best)
    no = 0
    it = 0
    while time.monotonic() < deadline - 8.0:
        it += 1
        c = _conv_full(p)
        M = float(c.max())
        cur_sc = float(2.0 * n * M)
        if cur_sc < best_sc - 1e-15:
            best = p.copy()
            best_sc = cur_sc

        L = c.size
        K = int(min(L, max(48, n // 3)))
        if K <= L:
            if K < L:
                rows = np.argpartition(c, L - K)[L - K:]
            else:
                rows = np.randperm(L)
        else:
            rows = np.arange(L, dtype=np.int64)

        rem = deadline - time.monotonic() - 8.0
        if rem <= 1.5:
            break
        tl = min(12.0, max(0.8, rem * 0.25))

        g = _solve_lp_dir(p, M, rows, tl, rng)
        improved = False
        if g is not None:
            best_cand = None
            best_cand_sc = best_sc
            for alpha in (0.05, 0.10, 0.20, 0.35, 0.55, 0.80, 1.00):
                cand = (1.0 - alpha) * p + alpha * g
                cand = np.maximum(cand, 0.0)
                s = float(cand.sum())
                if s <= 1e-300:
                    continue
                cand = cand / s
                sc = _score_only(cand)
                if math.isfinite(sc) and sc < best_cand_sc - 1e-15:
                    best_cand_sc = sc
                    best_cand = cand
            if best_cand is not None:
                best = best_cand.copy()
                best_sc = float(best_cand_sc)
                p = best_cand.copy()
                improved = True

        if it % 2 == 0:
            rem = deadline - time.monotonic() - 8.0
            if rem > 4.0:
                lam = _solve_lp_dual(p, M, rows, min(8.0, max(0.6, rem * 0.20)), rng)
                if lam is not None:
                    grad = _grade_conv(p, lam)
                    best_cand = None
                    best_cand_sc = best_sc
                    for eta in (0.5, 2.0, 8.0, 30.0):
                        cand = _project_simplex(p - eta * grad)
                        cand = np.asarray(cand, dtype=np.float64).ravel()
                        if cand.size != n or not np.all(np.isfinite(cand)):
                            continue
                        sc = _score_only(cand)
                        if math.isfinite(sc) and sc < best_cand_sc - 1e-15:
                            best_cand_sc = sc
                            best_cand = cand
                    if best_cand is not None:
                        best = best_cand.copy()
                        best_sc = float(best_cand_sc)
                        p = best_cand.copy()
                        improved = True

        if improved:
            no = 0
        else:
            no += 1
            if no >= 4:
                break

    return best, best_sc


def _optimize_length(L, base_p, deadline, rng):
    p0 = _resample_mass(base_p, L)
    p0 = np.maximum(p0, 1e-9 / max(1, L))
    p0 = p0 / p0.sum()
    best = p0.copy()
    best_sc = _score_only(best)
    current = p0.copy()

    rem = deadline - time.monotonic()
    if rem > 12.0:
        basis_time = min(35.0, max(5.0, rem * 0.16))
        bp, bsc = _basis_search(base_p, L, time.monotonic() + basis_time, rng)
        if bsc < best_sc - 1e-15:
            best = bp.copy()
            best_sc = float(bsc)
        if rem - (time.monotonic() - (deadline - rem)) > 0:
            current = _normalize(0.55 * best + 0.45 * p0)

    block_defs = [
        (0.18, 0.12),
        (0.10, 0.04),
        (0.05, 0.01),
        (0.025, 0.0),
        (0.012, 0.0),
        (0.006, 0.0),
    ]
    for tf, beta in block_defs:
        rem = deadline - time.monotonic()
        if rem < 8.0:
            break
        allot = min(75.0, max(7.0, rem * 0.34))
        M = best_sc / (2.0 * L) if best_sc > 0 else 1.0 / L
        tau = max(M * tf, M * 1e-5, 1e-12)
        bb, last = _lbfgs_block(current, L, time.monotonic() + allot, tau, beta)
        scb = _score_only(bb)
        if math.isfinite(scb) and scb < best_sc - 1e-15:
            best = bb.copy()
            best_sc = float(scb)
        scl = _score_only(last)
        if math.isfinite(scl) and scl < best_sc - 1e-15:
            best = last.copy()
            best_sc = float(scl)
        if math.isfinite(scl) and scl < best_sc * 1.04:
            current = last.copy()
        else:
            current = best.copy()

    rem = deadline - time.monotonic()
    if rem > 25.0 and L >= 32:
        blend = _normalize(0.68 * best + 0.32 * _random_smooth(L, rng, base=best))
        M = best_sc / (2.0 * L) if best_sc > 0 else 1.0 / L
        tau = max(M * 0.012, M * 1e-5, 1e-12)
        allot = min(40.0, max(8.0, rem * 0.38))
        bb, last = _lbfgs_block(blend, L, time.monotonic() + allot, tau, 0.0)
        scb = _score_only(bb)
        if math.isfinite(scb) and scb < best_sc - 1e-15:
            best = bb.copy()
            best_sc = float(scb)
        scl = _score_only(last)
        if math.isfinite(scl) and scl < best_sc - 1e-15:
            best = last.copy()
            best_sc = float(scl)

    rem = deadline - time.monotonic()
    if rem > 20.0:
        sp, ssc = _stochastic_polish(best, L, time.monotonic() + min(30.0, rem - 4.0), rng)
        if ssc < best_sc - 1e-15:
            best = sp.copy()
            best_sc = float(ssc)

    best = _normalize(best)
    final_sc = _score_only(best)
    return best, float(final_sc)


def _direct_score(p):
    p = _normalize(p)
    n = p.size
    if n == 0:
        return float("inf")
    try:
        b = np.convolve(p, p)
        s = float(p.sum())
        if s <= 1e-300:
            return float("inf")
        return float(2.0 * n * float(np.max(b)) / (s * s))
    except Exception:
        return float("inf")


def _exact_score(p):
    p = _normalize(p)
    if p.size == 0:
        return float("inf")
    ev = globals().get("evaluate_sequence", None)
    if callable(ev):
        try:
            cand = [float(x) for x in np.asarray(p, dtype=np.float64).ravel()]
            if cand and all(math.isfinite(x) for x in cand):
                sc = float(ev(cand))
                if math.isfinite(sc):
                    return sc
        except Exception:
            pass
    return _direct_score(p)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(0.0, float(budget_s))
    except Exception:
        budget = 1300.0

    try:
        seed_i = int(seed)
    except Exception:
        seed_i = 42
    rng = np.random.default_rng(seed_i)

    if budget <= 2.0:
        p = _get_parent()
        return [float(x) for x in np.clip(_normalize(p), 0.0, 1000.0)] or [1.0]

    head = max(12.0, min(45.0, 0.03 * budget))
    return_limit = t0 + budget - max(3.0, head * 0.2)
    deadline = t0 + budget - head

    parent = _get_parent()
    best_p = _normalize(parent)
    best_n = int(best_p.size)
    if best_n < 2:
        best_p = np.ones(512, dtype=np.float64) / 512.0
        best_n = 512
    best_sc = _score_only(best_p)

    candidates = {}
    candidates[best_n] = (best_p.copy(), best_sc)
    current_base = best_p.copy()

    if budget > 350.0:
        reserve_final = max(40.0, min(190.0, 0.14 * budget))
        main_deadline = max(t0 + 20.0, return_limit - reserve_final)
        phase = max(1.0, main_deadline - t0)

        lengths = []
        if budget > 120.0:
            lengths.append(512)
        if budget > 650.0:
            lengths.append(1024)
        if budget > 1050.0:
            lengths.append(2048)
        if best_n not in lengths and 64 <= best_n <= 2048:
            lengths.append(best_n)
        lengths = sorted(set(lengths))

        weights = {512: 0.52, 1024: 0.34, 2048: 0.13, best_n: 0.10}
        total_w = sum(weights.get(L, 0.1) for L in lengths)
        if total_w <= 0.0:
            total_w = 1.0

        cum = 0.0
        start = t0
        for L in lengths:
            w = weights.get(L, 0.1)
            frac = w / total_w
            cum_frac = frac
            Ldeadline = min(main_deadline - 3.0, t0 + (sum(weights.get(x, 0.1) for x in lengths[:lengths.index(L) + 1]) / total_w) * phase)
            if time.monotonic() >= Ldeadline - 5.0:
                continue
            try:
                pL, scL = _optimize_length(L, current_base, Ldeadline, rng)
            except Exception:
                pL, scL = _resample_mass(current_base, L), _score_only(_resample_mass(current_base, L))
            pL = _normalize(pL)
            scL = _score_only(pL)
            old = candidates.get(L)
            if old is None or scL < old[1] - 1e-12:
                candidates[L] = (pL.copy(), float(scL))
            old = candidates.get(best_n)
            if scL < best_sc - 1e-12:
                best_p = pL.copy()
                best_n = int(L)
                best_sc = float(scL)
                current_base = pL.copy()

    # Final polish on the best internal candidate.
    rem = return_limit - time.monotonic()
    if rem > 25.0:
        pB = _normalize(best_p)
        nB = int(pB.size)
        if nB >= 2:
            if rem > 45.0 and nB <= 1536:
                M = best_sc / (2.0 * nB) if best_sc > 0 else 1.0 / nB
                tau = max(M * 0.010, M * 1e-5, 1e-12)
                allot = min(70.0, max(8.0, rem - 12.0))
                bb, last = _lbfgs_block(pB, nB, time.monotonic() + allot, tau, 0.0)
                fb = bb if _score_only(bb) <= _score_only(last) else last
                scb = _score_only(fb)
                if math.isfinite(scb) and scb < best_sc - 1e-15:
                    best_p = fb.copy()
                    best_sc = float(scb)
                pB = _normalize(best_p)

            rem = return_limit - time.monotonic()
            if rem > 40.0 and nB <= 600:
                lp_time = min(rem - 10.0, 260.0)
                try:
                    pl, scl = _lp_polish(pB, nB, time.monotonic() + lp_time, rng)
                    pl = _normalize(pl)
                    if scl < best_sc - 1e-15:
                        best_p = pl.copy()
                        best_sc = float(scl)
                except Exception:
                    pass
                candidates.setdefault(nB, (best_p.copy(), best_sc))

            rem = return_limit - time.monotonic()
            if rem > 22.0:
                try:
                    ps, scs = _stochastic_polish(best_p, int(best_p.size), time.monotonic() + min(35.0, rem - 8.0), rng)
                    if scs < best_sc - 1e-15:
                        best_p = ps.copy()
                        best_sc = float(scs)
                except Exception:
                    pass

        candidates[best_n] = (best_p.copy(), best_sc)

    # Exact validation / cross-check candidates, but always preserve fallback.
    final_best = best_p.copy()
    final_sc = best_sc

    if time.monotonic() < return_limit - 4.0:
        ordered = sorted(candidates.items(), key=lambda kv: kv[1][1])
        for L, (p, sc) in ordered:
            if time.monotonic() > return_limit - 4.0:
                break
            try:
                p = _normalize(p)
                if p.size != L:
                    continue
                esc = _exact_score(p)
                if math.isfinite(esc) and esc < final_sc - 1e-12:
                    final_best = p.copy()
                    final_sc = float(esc)
            except Exception:
                pass

    out = np.asarray(final_best, dtype=np.float64).ravel()
    if out.size == 0 or not np.all(np.isfinite(out)):
        out = np.ones(512, dtype=np.float64) / 512.0
    out = _normalize(np.clip(out, 0.0, 1000.0))
    if out.size == 0:
        return [1.0]
    return [float(x) for x in out]
