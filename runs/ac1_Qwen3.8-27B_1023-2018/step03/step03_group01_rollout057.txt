        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:

def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

    n = len(a)
    s = float(a.sum())
    if s < 0.01:

    b = _conv_fft(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

    rows = []
    cols = []
    vals = []

        hi = min(n, k + 1)
        for j in range(lo, hi):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_raw(f):
    """Score for f with sum=1: 2*n*max_conv."""
    n = len(f)
    b = _conv(f)
    return 2.0 * n * float(b.max())

def _score_full(a):
    """Score for arbitrary nonnegative a."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    f = a / s
    return 2.0 * n * float(_conv(f).max())

def _conv_matrix(f, n):

    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A[k, lo:hi] = f[k - np.arange(lo, hi)]
    return A

def _lp_direction(f, n):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0. Returns normalized g."""
    M = float(_conv(f).max())
    if M < 1e-15:
        return None
    A = _conv_matrix(f, n)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)]*n, method='highs',
                  options={'time_limit': 10, 'presolve': True})
    if res.status != 0:
        return None
    g = res.x
    gs = g.sum()
    if gs < 1e-14:
        return None
    return g / gs

def _line_search(f, g, n, best_sc, t_max=0.5, stop_time=None):
    """Find best t in [0, t_max]: score of (1-t)f + t*g."""
    t_lo, t_hi = 0.0, t_max
    best_t, best_f = 0.0, f.copy()
    for _ in range(20):
        if stop_time and time.monotonic() > stop_time:
            break
        t_mid = 0.5 * (t_lo + t_hi)
        cand = (1 - t_mid) * f + t_mid * g
        np.maximum(cand, 0, out=cand)
        cs = cand.sum()
        if cs < 1e-10:
            t_hi = t_mid
            continue
        cand = cand / cs
        sc = _score_raw(cand)
        if sc < best_sc:
            best_t = t_mid
            best_f = cand.copy()
            t_lo = t_mid
        else:
            t_hi = t_mid
        if t_hi - t_lo < 1e-6:
            break
    if best_t < 1e-8:
        return None
    return best_t, best_f

def _subgrad_phase(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = f0.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_sc = _score_raw(f)
    lr = 0.02
    momentum = np.zeros(n)
    beta = 0.8
    no_imp = 0
    
    for it in range(100000):
        if time.monotonic() > stop_time - 2:
            break
        
        b = _conv(f)
        M = float(b.max())
        # Find top-K active lags
        K = min(8, 2*n-1)
        idx = np.argpartition(b, -K)[-K:]
        w = b[idx] - b[idx].min() + 1e-12
        w /= w.sum()
        
        # Subgradient
        grad = np.zeros(n)
        for wi, k in zip(w, idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            j = np.arange(lo, hi)
            grad[j] += 2.0 * wi * f[k - j]
        
        # Projected subgradient step
        momentum = beta * momentum - lr * grad
        f_new = np.maximum(f + momentum, 0)
        # Project onto simplex (sum=1, >=0)
        f_new = _proj_simplex(f_new)
        f_new = f_new / max(f_new.sum(), 1e-10)
        
        sc = _score_raw(f_new)
        if sc < best_sc - 1e-12:
            f = f_new
            best_f = f_new.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.05, lr * 1.005)
        else:
            no_imp += 1
            lr = max(0.002, lr * 0.98)
            if no_imp > 500:
                f = f_new.copy()
                no_imp = 0
                lr = 0.015
                momentum = np.zeros(n)
    
    return best_f, best_sc

def _proj_simplex(v):
    """Project v onto {x: x>=0, sum(x)=1}."""
    n = len(v)
    if v.sum() < 1e-10:
        return np.ones(n) / n
    # Sort
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    return p

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_raw(f):
    """Score assuming sum(f)=1: returns 2*n*max(conv(f,f))."""
    n = len(f)
    b = _conv(f)
    return 2.0 * n * float(b.max())

def _score_full(a):
    """Score for arbitrary nonnegative array."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    f = a / s
    return 2.0 * n * float(_conv(f).max())

def _conv_matrix(f, n):
    """Dense matrix A: (A@g)_k = sum_j f[k-j]*g[j] for full conv."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A[k, lo:hi] = f[k - np.arange(lo, hi)]
    return A

def _lp_direction(f, n, time_limit=8.0):
    """Solve LP: max sum(g) s.t. f*g <= M, g>=0. Return normalized g or None."""
    b = _conv(f)
    M = float(b.max())
    if M < 1e-15:
        return None
    A = _conv_matrix(f, n)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs

def _line_search(f, g, n, best_sc, t_max=0.4, stop_time=None):
    """Bisection line search: minimize _score_raw((1-t)*f+t*g)."""
    t_lo, t_hi = 0.0, t_max
    best_t, best_f = 0.0, f.copy()
    for _ in range(22):
        if stop_time and time.monotonic() > stop_time:
            break
        t_mid = 0.5 * (t_lo + t_hi)
        cand = (1.0 - t_mid) * f + t_mid * g
        np.maximum(cand, 0.0, out=cand)
        cs = float(cand.sum())
        if cs < 1e-10:
            t_hi = t_mid
            continue
        cand /= cs
        sc = _score_raw(cand)
        if sc < best_sc:
            best_t = t_mid
            best_f = cand.copy()
            t_lo = t_mid
        else:
            t_hi = t_mid
        if t_hi - t_lo < 5e-7:
            break
    if best_t < 1e-8:
        return None
    return best_t, best_f

def _lp_phase(f0, n, stop_time, seed):

def _proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    s = float(v.sum())
    if s < 1e-12:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _subgrad_phase(f0, n, stop_time, seed):
    """Projected subgradient with momentum."""
    rng = np.random.default_rng(seed)
    f = np.asarray(f0, dtype=np.float64).ravel().copy()
    f = np.clip(f, 0, None)
    f = f / max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = _score_raw(f)
    lr = 0.015
    momentum = np.zeros(n)
    beta = 0.82
    no_imp = 0
    K_top = 6

    for it in range(200000):
        if time.monotonic() > stop_time - 2.0:
            break

        b = _conv(f)
        M = float(b.max())
        # Top-K active lags
        idx = np.argpartition(b, -K_top)[-K_top:]
        w = b[idx].copy()
        w -= w.min()
        w += 1e-12
        w /= w.sum()

        grad = np.zeros(n)
        for wi, k in zip(w, idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            j = np.arange(lo, hi)
            grad[j] += 2.0 * wi * f[k - j]

        momentum = beta * momentum - lr * grad
        f_new = np.maximum(f + momentum, 0.0)
        f_new = _proj_simplex(f_new)
        cs = float(f_new.sum())
        f_new = f_new / cs if cs > 1e-10 else best_f.copy()

        sc = _score_raw(f_new)
        if sc < best_sc - 1e-13:
            f = f_new
            best_f = f_new.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.04, lr * 1.003)
        else:
            no_imp += 1
            lr = max(0.002, lr * 0.985)
            if no_imp > 400:
                f = f_new.copy()
                no_imp = 0
                lr = 0.012
                momentum = np.zeros(n)
                beta = 0.75

    return best_f, best_sc

def _coord_phase(f0, n, stop_time, seed):

    rng = np.random.default_rng(seed)

    while time.monotonic() < stop_time - 3.0:

        for i in order:

                break
            orig = f[i]
            for mlt in mults:

                cs = f.sum()
                if cs < 1e-10:
                    f[i] = orig
                    break
                f_n = f / cs
                sc = _score_raw(f_n)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_f = np.array(f_n)
                    improved = True
                    f = f_n.copy()
                    orig = f[i]
                else:
                    f[i] = orig
            f = f / max(f.sum(), 1e-10)

        # 2-coordinate moves
        for _ in range(60):
            if time.monotonic() > stop_time - 3.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi in (0.4, 0.7, 1.4, 2.0, 0.0):
                for mj in (0.4, 0.7, 1.4, 0.0):
                    f[i] = oi * mi if oi > 1e-12 else (0.002 if mi > 0 else 0.0)
                    f[j] = oj * mj if oj > 1e-12 else (0.002 if mj > 0 else 0.0)
                    cs = f.sum()
                    if cs < 1e-10:

                    f_n = f / cs

            f[i], f[j] = oi, oj

            no_imp_rounds = 0
