    cur_max_b = float(conv.max())

    t = 0.15
    t_min, t_max = 0.01, 0.5
    stall = 0
    prev_stall_count = 0

    for _ in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break

        M = cur_max_b
        A = _build_A(f, n)
        g = _lp_solve(A, M, n, time_limit=4.0)

        if g is None:
            stall += 5
        else:
            dist = float(np.linalg.norm(g - f, ord=1))
            if dist < 1e-9:
                stall += 4
            else:
                found = False
                try_ts = sorted(
                    [t, t * 0.5, t * 0.25, t * 1.5,
                     0.04, 0.08, 0.15, 0.25, 0.4],
                    key=lambda x: abs(x - t),
                )
                for ti in try_ts[:8]:
                    f_new = (1.0 - ti) * f + ti * g
                    f_new = np.clip(f_new, 0, None)
                    fs_new = float(f_new.sum())
                    if fs_new < 1e-10:
                        continue
                    sc = _score(f_new)
                    if sc < best_score - 1e-14:
                        f = f_new
                        best_f = f.copy()
                        best_score = sc
                        conv = _conv_full(f)
                        cur_max_b = float(conv.max())
                        found = True
                        t = min(t_max, ti * 1.15)
                        stall = 0
                        break
                    else:
                        t = max(t_min, t * 0.45)
                if not found:
                    stall += 1

        if stall >= 25:
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            conv = _conv_full(f)
            cur_max_b = float(conv.max())
            t = 0.15
            stall = 0
        elif stall >= 12:
            f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            conv = _conv_full(f)
            cur_max_b = float(conv.max())
            t = 0.2
            stall = 0

    return best_f, best_score


def _coord_descent(f, stop_time, max_sweeps=80, seed=777):
    n = len(f)
    f = np.clip(f.astype(np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, _score(f)
    conv = _conv_full(f)
    best_score = 2.0 * n * float(conv.max()) / (S * S)
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 3.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            v = f[i]
            S_rest = S - v
            x_max = min(max(3.0 * v + 0.5, 1.0), 50.0)
            n_pts = 100

            start = i
            end = min(i + n, 2 * n - 1)
            infl = np.zeros(2 * n - 1)
            infl[start:end] = 2.0 * f[: end - start]

            x_grid = np.linspace(0.0, x_max, n_pts)
            w = x_grid - v

            if start > 0 or end < 2 * n - 1:
                conv_out = float(np.max(np.r_[conv[:start], conv[end:]]))
            else:
                conv_out = -1e30

            conv_in = conv[start:end] + w[:, None] * infl[start:end][None, :]
            idx_2i = 2 * i - start
            if 0 <= 2 * i <= 2 * n - 2 and 0 <= idx_2i < end - start:
                conv_in[:, idx_2i] += w * w

            max_in = conv_in.max(axis=1)
            max_b = np.maximum(max_in, conv_out)
            S_new = S_rest + x_grid
            valid = S_new > 0.01
            obj = np.full(n_pts, np.inf)
            obj[valid] = 2.0 * n * max_b[valid] / (S_new[valid] ** 2)

            bi = int(np.argmin(obj))
            best_obj = float(obj[bi])
            best_x = float(x_grid[bi])

            if best_x != v and best_obj < best_score:
                dw = x_max / n_pts
                x_fine = np.linspace(max(0.0, best_x - 3 * dw), min(x_max, best_x + 3 * dw), 50)
                w_f = x_fine - v
                conv_in_f = conv[start:end] + w_f[:, None] * infl[start:end][None, :]
                if 0 <= 2 * i <= 2 * n - 2 and 0 <= idx_2i < end - start:
                    conv_in_f[:, idx_2i] += w_f * w_f
                max_in_f = conv_in_f.max(axis=1)
                max_b_f = np.maximum(max_in_f, conv_out)
                S_f = S_rest + x_fine
                valid_f = S_f > 0.01
                obj_f = np.full(len(x_fine), np.inf)
                obj_f[valid_f] = 2.0 * n * max_b_f[valid_f] / (S_f[valid_f] ** 2)
                bf = int(np.argmin(obj_f))
                if obj_f[bf] < best_obj:
                    best_obj = float(obj_f[bf])
                    best_x = float(x_fine[bf])

            if best_x != v and best_obj < best_score - 1e-14:
                w_val = best_x - v
                f[i] = best_x
                S += w_val
                conv[start:end] += w_val * infl[start:end]
                if 0 <= 2 * i <= 2 * n - 2:
                    conv[2 * i] += w_val * w_val
                best_score = best_obj
                improved = True

        if not improved:
            break

    f = np.clip(f, 0, 1000)
    final_score = _score(f)
    return f, final_score


def _pair_polish(f, stop_time, max_rounds=25, seed=9999):
    n = len(f)
    f = f.astype(np.float64).copy()
    best_score = _score(f)
    rng = np.random.default_rng(seed)
    mults = (0.2, 0.5, 0.8, 1.3, 1.8, 2.5, 0.0)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 3.0:
            break
        improved = False
        n_tries = min(100, max(20, 6000 // n))
        for _ in range(n_tries):
            if time.monotonic() > stop_time - 3.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            best_pair = (oi, oj)
            best_sc = best_score
            for mi in mults:
                for mj in mults:
                    f[i] = oi * mi if oi > 1e-12 else (0.01 if mi > 0 else 0.0)
                    f[j] = oj * mj if oj > 1e-12 else (0.01 if mj > 0 else 0.0)
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        best_pair = (f[i], f[j])
            f[i], f[j] = best_pair
            if best_sc < best_score - 1e-14:
                best_score = best_sc
                improved = True
        if not improved:
            break
    return f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    fallback = parent.copy()
    try:
        fallback_sc = _score(fallback)
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(32) / 32.0
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update_best(seq, sc):
        nonlocal best_seq, best_score
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-13:
            best_seq = np.array(seq, dtype=np.float64).copy()
            best_score = exact
        elif not math.isfinite(best_score) and math.isfinite(exact):
            best_seq = np.array(seq, dtype=np.float64).copy()
            best_score = exact

    def _lp_and_polish(n, init_arr, lp_stop, polish_stop, sd):
        if time.monotonic() > lp_stop:
            return
        try:
            res, sc = _lp_optimize(init_arr, n, lp_stop, sd)
        except Exception:
            return
        if res is None or len(res) == 0 or not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0, 1000)
        if res.sum() < 0.01:
            return
        _update_best(res, sc)
        if time.monotonic() < polish_stop - 20:
            try:
                r2, s2 = _coord_descent(res, polish_stop, max_sweeps=50, seed=sd + 1)
            except Exception:
                r2, s2 = res, sc
            if r2 is not None and len(r2) > 0 and np.all(np.isfinite(r2)) and r2.sum() > 0.01:
                r2 = np.clip(r2, 0, 1000)
                _update_best(r2, s2)

    n = n_p
    ia = np.arange(n, dtype=np.float64)

    inits = []
    inits.append(parent.copy())
    inits.append(np.clip(0.3 + 0.7 * np.cos(np.pi * ia / n) ** 2, 0, None))
    inits.append(np.clip(0.5 + 0.5 * np.sin(6 * np.pi * ia / n), 0, None))
    inits.append(np.linspace(0.05, 2.5, n))
    b1 = np.exp(-((ia - 0.3 * n) ** 2) / (2 * (0.15 * n) ** 2))
    b2 = np.exp(-((ia - 0.7 * n) ** 2) / (2 * (0.15 * n) ** 2))
    inits.append(0.2 + b1 + b2)
    inits.append(np.clip(1.0 + 0.5 * np.cos(4 * np.pi * ia / n) + 0.3 * np.sin(2 * np.pi * ia / n), 0, None))

    for x in inits:
        s = x.sum()
        if s < 1e-6:
            x[:] = 1.0 / n
        else:
            x /= s

    # ---- Phase 1: LP + coord on parent length ----
    phase1_end = t0 + budget * 0.40
    tpi = max(10.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 5.0:
            break
        stop_lp = min(phase1_end - 3.0, time.monotonic() + tpi)
        stop_pol = min(hard_deadline, stop_lp + budget * 0.12)
        _lp_and_polish(n, init.copy(), stop_lp, stop_pol, seed + 200 * (idx + 1))

    # ---- Phase 2: Extended coordinate descent ----
    pol_end = t0 + budget * 0.58
    if time.monotonic() < pol_end - 15.0:
        try:
            best_seq, best_score = _coord_descent(best_seq, pol_end, max_sweeps=80, seed=424242)
        except Exception:
            pass
        _update_best(best_seq, best_score)

    # ---- Phase 3: Try other lengths ----
    phase3_end = t0 + budget * 0.82
    other_lengths = [96, 160, 256, 320, 448, 512, 640]
    for L in [n_p]:
        if L in other_lengths:
            other_lengths.remove(L)
    other_lengths.sort(key=lambda L: abs(L - n_p))

    rem = phase3_end - time.monotonic()
    if rem > 20:
        tpl = max(5.0, rem / max(1, len(other_lengths)))
        for L in other_lengths:
            if time.monotonic() > phase3_end - 10.0:
                break
            if best_score < 1.5028:
                break
            stop_lp = min(phase3_end - 5.0, time.monotonic() + tpl * 0.65)
            stop_pol = min(hard_deadline, stop_lp + tpl * 0.5)

            src = best_seq if best_score < 1.6 else parent
            if len(src) != L:
                init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(src)), src)
            else:
                init_l = src.astype(np.float64).copy()
            sl = init_l.sum()
            if sl > 1e-6:
                init_l /= sl
            else:
                init_l = np.ones(L) / L

            try:
                _lp_and_polish(L, init_l, stop_lp, stop_pol, seed + 50000 + L)
            except Exception:
                pass

    # ---- Phase 4: Final intensive polish ----
    final_end = hard_deadline - 12.0
    if time.monotonic() < final_end - 10.0:
        try:
            best_seq, best_score = _coord_descent(best_seq, final_end, max_sweeps=100, seed=31337)
        except Exception:
            pass
        _update_best(best_seq, best_score)

    if time.monotonic() < final_end - 8.0:
        try:
            best_seq, best_score = _pair_polish(best_seq, final_end, max_rounds=20, seed=55555)
        except Exception:
            pass
        _update_best(best_seq, best_score)

    if time.monotonic() < hard_deadline - 15.0:
        try:
            best_seq, best_score = _coord_descent(best_seq, hard_deadline - 10.0, max_sweeps=30, seed=88888)
        except Exception:
            pass
        _update_best(best_seq, best_score)

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    if final_sc < best_score:
        best_score = final_sc

    return [float(x) for x in best_seq]
