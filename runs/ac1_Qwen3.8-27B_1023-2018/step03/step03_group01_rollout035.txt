Now, the key insight: c[k] = sum_{i+j=k} f[i]f[j] is the probability that two independent random variables (with PMF f) sum to k. So c is the PMF of the sum of two i.i.d. random variables with PMF f.

We want to find a PMF f on {0, 1, ..., n-1} such that the PMF of X+X (where X ~ f) is as flat as possible.

This is a beautiful reformulation! The "flattest" distribution for X+X is the uniform distribution on {0, 1, ..., 2n-2}, which has PMF 1/(2n-1) at each point.

So we're looking for a PMF f on {0,...,n-1} such that the convolution f*f is as close to uniform as possible.

This is related to the concept of "dissociated" sequences in combinatorics, and to the "postage stamp problem."

For the uniform f (f[i] = 1/n), the convolution f*f[k] = (n-|k|)/n^2 for |k| < n (triangular). This is far from flat.

The "ideal" f is approximately f[k] ∝ 1/sqrt(k+1) (from the central binomial coefficient analysis), which gives a nearly flat convolution.

    n = len(v)

    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c


def _score(f):
    n = len(f)
    s = float(f.sum())
    if s < 0.01:
        return math.inf
    c = _conv_fft(f)
    M = float(np.max(c))
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M / (s * s)


def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1)
    return np.maximum(v - theta, 0.0)


def _init_central_binomial(n):
    f = np.empty(n)
    f[0] = 1.0
    for k in range(1, n):
        f[k] = f[k - 1] * (2 * k - 1) / (2 * k)
    s = f.sum()
    return f / s if s > 0 else np.ones(n) / n


def _init_inv_sqrt(n):
    f = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    return f / f.sum()


def _subgrad_optimize(f0, n, stop_time, seed):
    """Accelerated subgradient descent on the simplex."""
    rng = np.random.default_rng(seed)
    f = _project_simplex(f0.astype(np.float64).copy())
    best_f = f.copy()
    best_sc = _score(f)
    
    step = 0.3
    momentum = np.zeros(n)
    beta = 0.7
    no_imp = 0
    max_no_imp = 300
    
    it = 0
    while time.monotonic() < stop_time - 1.0:
        it += 1
        
        c = _conv_fft(f)
        M = float(c.max())
        S = float(f.sum())  # should be 1
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_f = f.copy()
            no_imp = 0
            step = min(0.5, step * 1.01)
        else:
            no_imp += 1
            step = max(0.01, step * 0.99)
            if no_imp >= max_no_imp:
                # Perturb and restart
                pert = rng.normal(0, 0.1, n)
                f = _project_simplex(f + pert)
                momentum *= 0.5
                no_imp = 0
                step = 0.2
        
        # Compute subgradient using active lags
        # Use lags within 95% of max for smoother gradient
        thresh = 0.95 * M
        active = np.where(c >= thresh)[0]
        
        # Weights proportional to (c[k] - thresh)
        w = np.zeros(2 * n - 1)
        w[active] = c[active] - thresh + 1e-12
        w_sum = w.sum()
        if w_sum < 1e-15:
            w[np.argmax(c)] = 1.0
            w_sum = 1.0
        else:
            w /= w_sum
        
        # Compute gradient via FFT (cross-correlation)
        fsize = next_fast_len(3 * n)
        W = rfft(w, fsize)
        F = rfft(f, fsize)
        corr = irfft(W * np.conj(F), fsize)
        # corr[i] = sum_j w[j] f[j-i] (circular, but we want linear)
        # Actually for rfft, the result is the real part of the circular correlation
        # We need the first n values of the linear correlation
        # Let me verify: corr[i] = sum_j w[j+i] * f[j] = sum_j w[j] * f[j-i]
        # The linear cross-correlation has length (2n-1) + n - 1 = 3n-2
        # With FFT size fsize >= 3n-2, we get the linear correlation in corr[:3n-2]
        # We want corr[i] for i = 0, ..., n-1
        
        grad_M = 2.0 * corr[:n].copy()
        
        # Full subgradient of F(f) = 2n * M / S^2
        # dF/df[i] = (2n/S^2) * dM/df[i] - (4n*M*1/S^3)
        # dM/df[i] = 2 * corr[i] (from the cross-correlation, each c[k] has gradient 2*f[k-i])
        grad = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M / (S * S * S)) * 1.0
        
        # Momentum update
        momentum = beta * momentum - step * grad
        f_new = _project_simplex(f + momentum)
        
        # Verify improvement (accept even if slightly worse, like SA)
        sc_new = _score(f_new)
        if sc_new < best_sc + 0.01:  # Accept if not much worse
            f = f_new
    
    return best_f, best_sc
