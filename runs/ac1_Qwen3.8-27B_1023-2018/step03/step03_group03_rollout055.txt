            # Let's just use the approach: for coordinate i, the change in b is:
            # For each valid k (0 <= k < 2n-1):
            #   j ranges over max(0, k-n+1) to min(k, n-1)
            #   b[k] = sum_j f[j]*f[k-j]
            #   If we change f[i] by d:
            #   delta_b[k] = d*f[k-i] (if 0<=k-i<n) + d*f[k-i] (if 0<=k-i<n and k-i != i) 
            #                                          = d*f[k-i] if i is in valid j range AND k-i is in valid j range
            #   Wait, I'll just do it the simple way:
            #   delta_b[k] = d * (number of times f[i] appears in b[k]) * f[partner]
            #   f[i] appears in b[k] as f[j] when j=i (requires 0<=k-i<n)
            #   f[i] appears in b[k] as f[k-j] when k-j=i, i.e., j=k-i (requires 0<=k-i<n and 0<=k-i<n)
            #   If i == k-i, i.e., k=2i: f[i] appears as both, so delta = 2*d*f[i] + d^2
            #   If i != k-i: delta = d*f[k-i] + d*f[k-i] = 2*d*f[k-i] (if both valid)
            #   But wait, if k-i < 0 or k-i >= n, the "partner" doesn't exist.
            #   Actually for the convolution, j must be in [0,n-1] AND k-j must be in [0,n-1].
            #   So f[k-i] exists only if 0 <= k-i <= n-1.
            #   And j=i exists only if 0 <= i <= n-1 (always true) AND 0 <= k-i <= n-1.
            #   And j=k-i exists only if 0 <= k-i <= n-1 AND 0 <= i <= n-1 (always true).
            
            # So: valid_partner = 0 <= k-i < n
            # If valid_partner AND k != 2*i: delta_b[k] = 2*d*f[k-i]
            # If valid_partner AND k == 2*i: delta_b[k] = 2*d*f[i] + d^2
            # If not valid_partner: delta_b[k] = 0
            
            # This is equivalent to: infl[k] = 2*f[k-i] for k in [i, i+n-1] (where k-i in [0,n-1])
            # And at k=2i: extra d^2 term
            start = i
            end = min(i + n, n_conv)
            infl[start:end] = 2.0 * f[:end - start]
            # At k=2i, the "2*f[k-i]" already includes 2*f[i], which is correct for the linear term
            # The quadratic term d^2 is handled separately
            
            # Find search range
            if v > 1e-8:
                x_max = max(3.0 * v, 0.5)
            else:
                x_max = 1.0
            x_max = min(x_max, 100.0)
            
            # Quick scan: find which lags are near the max
            b_current_max = float(conv.max())
            
            # Adaptive grid: more points near where the objective is "active"
            n_coarse = 80
            x_grid = np.linspace(0.0, x_max, n_coarse)
            
            best_x = v
            best_obj = float('inf')
            
            for x in x_grid:
                S_new = S_rest + x
                if S_new < 0.01:
                    continue
                w = x - v
                if abs(w) < 1e-14:
                    m_new = float(conv.max())
                else:
                    # Only need to check lags where b might change
                    # b changes only at lags in [start, end)
                    # The max could be at a lag outside [start, end) if that's already the global max
                    m_new = float(conv.max())
                    # Check the changed region
                    changed_max = float((conv[start:end] + w * infl[start:end]).max())
                    if 2 * i < n_conv:
                        changed_max = max(changed_max, conv[2 * i] + w * infl[2 * i] + w * w)
                    m_new = max(m_new, changed_max)
                
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x
            
            # Fine grid around best
            if best_x != v:
                dw = x_max / n_coarse
                n_fine = 40
                x_fine = np.linspace(max(0.0, best_x - 2 * dw), min(x_max, best_x + 2 * dw), n_fine)
                for x in x_fine:
                    S_new = S_rest + x
                    if S_new < 0.01:
                        continue
                    w = x - v
                    if abs(w) < 1e-14:
                        continue
                    m_new = float(conv.max())
                    changed_max = float((conv[start:end] + w * infl[start:end]).max())
                    if 2 * i < n_conv:
                        changed_max = max(changed_max, conv[2 * i] + w * infl[2 * i] + w * w)
                    m_new = max(m_new, changed_max)
                    obj = 2.0 * n * m_new / (S_new * S_new)
                    if obj < best_obj - 1e-16:
                        best_obj = obj
                        best_x = x
            
            # Accept if improved
            if best_x != v and best_obj < best_score - 1e-15:
                w = best_x - v
                f[i] = best_x
                S += w
                conv[start:end] += w * infl[start:end]
                if 0 <= 2 * i < n_conv:
                    # Need to also add w^2 at position 2i
                    # But we already added w*infl[2i] = w*2*f[0]... wait
                    # infl[2i-start] = 2*f[2i-start] = 2*f[i] (since 2i-start = 2i-i = i)
                    # So conv[2i] now has w*2*f[i] added, plus we need w^2
                    conv[2 * i] += w * w
                best_score = best_obj
                improved = True

        if not improved:
            # Try a few random pairs to break symmetry
            for _ in range(10):
                if time.monotonic() > stop_time - 4.0:
                    break
                i1, i2 = int(rng.integers(n)), int(rng.integers(n))
                if i1 == i2:
                    continue
                old1, old2 = f[i1], f[i2]
                # Try swapping or mixing
                f[i1], f[i2] = f[i2], f[i1]
                sc = _score_fast(f)
                if sc < best_score - 1e-14:
                    best_score = sc
                    conv = _conv_full(f)
                    improved = True
                else:
                    f[i1], f[i2] = old1, old2
            if not improved:
                break

    # Final verification
    f = np.clip(f, 0, 1000)
    if f.sum() > 0.01:
        final_score = _score_fast(f)
    else:
        final_score = math.inf
    return f, final_score


def _local_pair_polish(f, stop_time, max_rounds=20, seed=9999):
    """2-coordinate moves to escape 1D local optima."""
    n = len(f)
    f = f.copy()
    best_score = _score_fast(f)
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 4.0:
            break
        improved = False
        n_tries = min(60, max(10, int((stop_time - 4.0 - time.monotonic()) / max(0.01, (max_rounds - rnd) * 0.02))))
        for _ in range(n_tries):
            if time.monotonic() > stop_time - 4.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            best_pair_sc = best_score
            best_i, best_j = oi, oj
            multipliers_i = (0.2, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 0.0)
            multipliers_j = (0.2, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 0.0)
            for mi in multipliers_i:
                for mj in multipliers_j:
                    f[i] = max(0.0, oi * mi) if oi > 1e-12 else (0.01 if mi > 0 else 0.0)
                    f[j] = max(0.0, oj * mj) if oj > 1e-12 else (0.01 if mj > 0 else 0.0)
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score_fast(f)
                    if sc < best_pair_sc - 1e-14:
                        best_pair_sc = sc
                        best_i, best_j = f[i], f[j]
            f[i], f[j] = best_i, best_j
            if best_pair_sc < best_score - 1e-14:
                best_score = best_pair_sc
                improved = True
        if not improved:
            break
    return f, best_score


def _try_length(n, init, stop_lp, stop_polish, seed):
    """Run LP + coordinate polish for a given length."""
    if time.monotonic() > stop_lp:
        return None, math.inf
    
    try:
        res, sc = _lp_optimize(init, n, stop_lp, seed)
    except Exception:
        res, sc = init.copy(), _score_fast(init)
    
    if res is None or not np.all(np.isfinite(res)):
        return None, math.inf
    res = np.clip(res, 0, 1000)
    if res.sum() < 0.01:
        return None, math.inf
    
    # Coordinate polish
    if time.monotonic() < stop_polish - 10:
        try:
            res, sc = _coord_polish(res, stop_polish, max_sweeps=40, seed=seed + 1)
        except Exception:
            pass
    
    if time.monotonic() < stop_polish - 5:
        try:
            res, sc = _local_pair_polish(res, stop_polish, max_rounds=10, seed=seed + 2)
        except Exception:
            pass
    
    res = np.clip(res, 0, 1000)
    if res.sum() < 0.01:
        return None, math.inf
    sc = _score_fast(res)
    return res, sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    # Fallback
    fallback = parent.copy()
    try:
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(64) / 64.0
        fallback_sc = _score_fast(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update_best(seq, sc):
        nonlocal best_seq, best_score
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-13:
            best_seq = np.array(seq, dtype=np.float64).copy()
            best_score = exact
            return True
        return False

    rng_main = np.random.default_rng(seed)

    # === Phase 1: LP optimization on parent length ===
    phase1_end = t0 + budget * 0.25

    # Multiple initializations for the parent length
    inits = [parent.copy()]
    i_arr = np.arange(n_p, dtype=np.float64)
    # Cosine taper
    inits.append(0.2 + 0.8 * np.cos(np.pi * i_arr / n_p) ** 2)
    # Two-lobe
    inits.append(0.4 + 0.6 * np.cos(2 * np.pi * i_arr / n_p))
    inits.append(np.clip(inits[-1], 0, None))
    # Random smooth
    smooth = np.convolve(rng_main.uniform(0.3, 1.0, n_p), np.ones(20)/20, mode='same')
    inits.append(np.clip(smooth, 0.05, None))

    inits = [np.clip(x, 0, None) for x in inits]
    for x in inits:
        if x.sum() < 1e-6:
            x[:] = 1.0 / n_p

    time_per_init = max(20.0, (phase1_end - t0 - 5.0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 5.0:
            break
        if best_score < 1.5028:
            break
        stop_lp = min(phase1_end - 3.0, time.monotonic() + time_per_init * 0.65)
        stop_pol = min(hard_deadline - 30.0, stop_lp + time_per_init * 0.5)

        res, sc = _try_length(n_p, init.copy(), stop_lp, stop_pol, seed + 100 * (idx + 1))
        if res is not None and math.isfinite(sc):
            _update_best(res, sc)

    # === Phase 2: Extended coordinate polish on best ===
    phase2_end = t0 + budget * 0.50
    if time.monotonic() < phase2_end - 20.0:
        try:
            polished, sc = _coord_polish(best_seq, phase2_end, max_sweeps=60, seed=424242)
            if np.all(np.isfinite(polished)) and polished.sum() > 0.01:
                _update_best(polished, sc)
        except Exception:
            pass

    # === Phase 3: Try nearby lengths ===
    phase3_end = t0 + budget * 0.75
    other_lengths = []
    # Explore around parent length
    for delta in [-32, -16, 16, 32, 48, -48, 64, -64, 80, -80]:
        L = n_p + delta
        if 32 <= L <= 600:
            other_lengths.append(L)
    # Some special lengths
    for L in [128, 192, 256, 320, 512]:
        if L not in other_lengths and 32 <= L <= 600:
            other_lengths.append(L)
    other_lengths = sorted(set(other_lengths))
    other_lengths.sort(key=lambda L: abs(L - n_p))
    # Limit to reasonable number
    other_lengths = other_lengths[:12]

    time_per_len = max(15.0, (phase3_end - time.monotonic() - 5.0) / max(1, len(other_lengths)))

    for L in other_lengths:
        if time.monotonic() > phase3_end - 10.0:
            break
        if best_score < 1.50275:
            break

        # Interpolate best sequence to new length
        if len(best_seq) > 1:
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
        else:
            init_l = np.ones(L) / L
        init_l = np.clip(init_l, 0.01, None)

        stop_lp = min(phase3_end - 5.0, time.monotonic() + time_per_len * 0.5)
        stop_pol = min(hard_deadline - 20.0, stop_lp + time_per_len * 0.6)

        res, sc = _try_length(L, init_l, stop_lp, stop_pol, seed + 50000 + L)
        if res is not None and math.isfinite(sc):
            _update_best(res, sc)

    # === Phase 4: Intensive final polish ===
    final_end = hard_deadline - 10.0
    if time.monotonic() < final_end - 20.0:
        try:
            polished, sc = _coord_polish(best_seq, final_end, max_sweeps=80, seed=31337)
            if np.all(np.isfinite(polished)) and polished.sum() > 0.01:
                _update_best(polished, sc)
        except Exception:
            pass

    # Pair polish
    if time.monotonic() < final_end - 10.0:
        try:
            polished, sc = _local_pair_polish(best_seq, final_end, max_rounds=15, seed=55555)
            if np.all(np.isfinite(polished)) and polished.sum() > 0.01:
                _update_best(polished, sc)
        except Exception:
            pass

    # One more coordinate pass
    if time.monotonic() < hard_deadline - 12.0:
        try:
            polished, sc = _coord_polish(best_seq, hard_deadline - 8.0, max_sweeps=30, seed=88888)
            if np.all(np.isfinite(polished)) and polished.sum() > 0.01:
                _update_best(polished, sc)
        except Exception:
            pass

    # === Final validation ===
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        best_seq = fallback

    return [float(x) for x in best_seq]
