import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.array([1.0])
    if p.size == 0:
        p = np.array([1.0])
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.array([1.0])
    return p


def _conv(a):
    n = len(a)
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    return 2.0 * n * float(b.max()) / (s * s)


def _lp(f, M, tl=3.0):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                     bounds=[(0, None)] * n, method='highs',
                     options={'presolve': True, 'time_limit': tl})
    except Exception:
        return None, None
    if r.status not in (0, 3) or r.x is None:
        return None, None
    g = np.asarray(r.x, dtype=np.float64)
    gs = g.sum()
    if gs < 1e-12:
        return None, None
    du = np.asarray(r.ub, dtype=np.float64) if r.ub is not None else None
    return g / gs, du


def _press(f, du):
    n = len(f)
    p = np.zeros(n)
    for d in range(n):
        L = min(n, 2 * n - 1 - d)
        if L > 0:
            p[:L] += f[d] * du[d:d + L]
    return p

        g, du = _lp(f, M)
        iter_n += 1

        if g is None:
            f = np.clip(f + rng.normal(0, 0.1, n), 0, None)
            s = f.sum()
            f = f / s if s > 1e-10 else np.ones(n) / n
            M = float(_conv(f).max())
            stall += 3
            if stall > 150:
                break
            continue

        d = float(np.linalg.norm(g - f))
        if d < 1e-8:
            # Convergence: perturb

                p = _press(f, du)
                pm = max(p.mean(), 1e-10)
                f = np.clip(f + 0.05 * (p - pm) / pm, 0, None)
            else:
                f = np.clip(f + rng.normal(0, 0.1, n), 0, None)
            s = f.sum()
            f = f / s if s > 1e-10 else np.ones(n) / n

**Improved polish**:
1. **Coordinate descent with line search**: For each coordinate i, find the optimal value of a_i (holding others fixed) by 1D line search. The score as a function of a_i is:
   score(a_i) = 2n * max_k (a*a)(k) / (S)²
   where (a*a) depends on a_i in a specific way, and S = S₀ + (a_i - a_i₀).
   
   This is a 1D problem. I can use golden section search to find the optimal a_i.

2. **Pairwise optimization**: Simultaneously optimize two coordinates.

3. **Block optimization**: Optimize a block of coordinates.

For coordinate i:
(a*a)(k) = Σ_j a_j a_{k-j} = [terms not involving i] + a_i * (a_{k-i} + a_{i-k}) [if valid]
= [terms not involving i] + a_i * c_k(i)

where c_k(i) = a_{k-i} (if 0≤k-i<n) + a_{i-k} (if 0≤i-k<n).
If i = k-i (i.e., k=2i): c_k(i) = 2a_i.

So (a*a)(k) = B_k + a_i * c_k(i) where B_k = [terms not involving i].

The max over k of (a*a)(k) is a piecewise-linear function of a_i. The score is:
score = 2n * max_k (B_k + a_i * c_k(i)) / (S₀ + a_i - a_i_old)²

This is a ratio of a piecewise-linear function to a quadratic, which is unimodal (or near-unimodal). I can use golden section search.

But computing the max for each a_i evaluation requires O(n) work (compute all (a*a)(k)). With FFT, it's O(n log n). For n=384, that's fast.

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.array([1.0])
    if p.size == 0:
        p = np.array([1.0])
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.array([1.0])
    return p


def _conv(a):
    n = len(a)
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    return 2.0 * n * float(b.max()) / (s * s)


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, M, focused=None, tl=3.0):
    n = len(f)
    A = _build_A(f)
    if focused is not None:
        A = A[focused]
        b = np.full(len(focused), M)
    else:
        b = np.full(2 * n - 1, M)
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=b,
                     bounds=[(0, None)] * n, method='highs',
                     options={'presolve': True, 'time_limit': tl})
    except Exception:
        return None, None
    if r.status not in (0, 3) or r.x is None:
        return None, None
    g = np.asarray(r.x, dtype=np.float64)
    gs = g.sum()
    if gs < 1e-12:
        return None, None
    du = np.asarray(r.ub, dtype=np.float64) if r.ub is not None else None
    return g / gs, du


def _press(f, du):
    n = len(f)
    p = np.zeros(n)
    for d in range(n):
        L = min(n, 2 * n - 1 - d)
        if L > 0:
            p[:L] += f[d] * du[d:d + L]
    return p

            top_k = min(120, 2 * n - 1)
            fi = np.argsort(-b_c)[:top_k]
            g, _ = _lp_solve(f, M, focused=fi)
        else:
            g, du = _lp_solve(f, M)

        if g is None:
            f = np.clip(f + rng.normal(0, 0.1, n), 0, None)
            s = f.sum()
            f = f / s if s > 1e-10 else np.ones(n) / n

            stall += 3
            if stall > 150:

            # Perturb
            if du is not None:
                p = _press(f, du)
                pm = max(p.mean(), 1e-10)
                f = np.clip(f + 0.05 * (p - pm) / pm, 0, None)
            else:
                f = np.clip(f + rng.normal(0, 0.1, n), 0, None)
            s = f.sum()
            f = f / s if s > 1e-10 else np.ones(n) / n

                if k == 0:
                    f
