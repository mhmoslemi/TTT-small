                t_step = 0.12
                stall = 0

            if stall >= max_stall * 2:
                f = best.copy()
                b = _conv_full(f)
                M = float(b.max())
                t_step = 0.1
                stall = 0

    # Phase 2: Additional LP passes from best (15% of time)
    phase2_end = t0 + budget * 0.75
    if time.monotonic() < phase2_end - 10.0:
        f = best.copy()
        b = _conv_full(f)
        M = float(b.max())
        t_step = 0.1
        stall = 0
        max_stall_p2 = 40

        while time.monotonic() < phase2_end - 3.0:
            g = _lp_direction(f, n, M, time_limit=4.0)
            if g is not None:
                dist = float(np.linalg.norm(g - f))
                if dist > 1e-8:
                    f_new, sc_new = _line_search(f, g, best_sc, n)
                    if f_new is not None:
                        f = f_new
                        best = f_new.copy()
                        best_sc = sc_new
                        b = _conv_full(f)
                        M = float(b.max())
                        stall = 0
                        continue
            stall += 1
            if stall >= max_stall_p2:
                f = best.copy()
                f = f + rng.normal(0, 0.02, n)
                f = np.maximum(f, 0)
                f = f / f.sum()
                b = _conv_full(f)
                M = float(b.max())
                stall = 0

    # Phase 3: Polishing (25% of time)
    phase3_end = t0 + budget * 0.92
    polish_time = phase3_end - time.monotonic()
    if polish_time > 10.0:
        # Split polish into two passes
        half = polish_time / 2
        stop1 = time.monotonic() + half
        try:
            best_f, best_sc_p = _polish_sequence(best, stop1, seed + 100)
            if best_sc_p < best_sc:
                best = best_f
                best_sc = best_sc_p
        except Exception:
            pass

        stop2 = phase3_end - 2.0
        if time.monotonic() < stop2 - 10.0:
            try:
                best_f, best_sc_p = _polish_sequence(best, stop2, seed + 200)
                if best_sc_p < best_sc:
                    best = best_f
                    best_sc = best_sc_p
            except Exception:
            pass

    # Phase 4: Final subgradient + validation (remaining time)
    final_stop = hard_deadline - 5.0
    if time.monotonic() < final_stop - 3.0:
        # Quick subgradient sweep
        for eta in (0.001, 0.003, 0.005, 0.01, 0.002, 0.008):
            if time.monotonic() > final_stop - 3.0:
                break
            f_sub = _subgradient_step(best, n, eta)
            sc_sub = _score_fast(f_sub)
            if sc_sub < best_sc - 1e-15:
                best = f_sub
                best_sc = sc_sub

    # Final validation with exact evaluator
    final_seq = np.clip(best, 0, 1000)
    final_list = [float(x) for x in final_seq]
    final_sc = float(evaluate_sequence(final_list))

    if not math.isfinite(final_sc):
        # Fallback to parent
        parent_list = [float(x) for x in np.clip(parent, 0, 1000)]
        parent_sc = float(evaluate_sequence(parent_list))
        if math.isfinite(parent_sc):
            return parent_list
        return [1.0 / n] * n

    if final_sc > best_sc + 1e-6:
        # Something went wrong; re-verify parent
        parent_list = [float(x) for x in np.clip(parent, 0, 1000)]
        parent_sc = float(evaluate_sequence(parent_list))
        if math.isfinite(parent_sc) and parent_sc < final_sc:
            return parent_list

    return final_list
