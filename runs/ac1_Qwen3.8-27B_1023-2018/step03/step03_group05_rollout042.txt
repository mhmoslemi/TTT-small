import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    headroom = max(12.0, 0.015 * budget_s)
    deadline = t0 + budget_s - headroom

    rng = np.random.default_rng(seed)

    # --- Initialization from parent ---
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            a = np.asarray(h, dtype=np.float64).ravel()
        except Exception:
            a = np.ones(512, dtype=np.float64) / 512.0
    else:
        a = np.ones(512, dtype=np.float64) / 512.0

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(512, dtype=np.float64) / 512.0
    else:
        a = a / s
    n = a.size

    def _conv(x):
        xx = np.asarray(x, dtype=np.float64)
        nn = xx.size
        nf = next_fast_len(2 * nn - 1)
        f = rfft(xx, nf)
        return irfft(f * f, nf)[:2 * nn - 1]

    def _proj_simplex(v, target=1.0):
        v = np.asarray(v, dtype=np.float64)
        nn = v.size
        if nn == 0:
            return np.ones(1)
        u = np.sort(v)[::-1]
        css = np.cumsum(u)
        idx = np.arange(1, nn + 1, dtype=np.float64)
        with np.errstate(divide="ignore", invalid="ignore"):
            cond = (css - target) / idx < u
        nz = np.nonzero(cond)[0]
        if nz.size == 0:
            return np.full(nn, target / nn)
        rho = nz[-1]
        theta = (css[rho] - target) / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        sw = w.sum()
        if sw <= 1e-300:
            return np.full(nn, target / nn)
        return w / sw * target

    def _fft_grad(a_xx, w_lags):
        """Compute 2*(w_lags cross-corr a_xx) at indices 0..n-1."""
        nn = a_xx.size
        m = w_lags.size
        need = nn + m - 1
        nf = next_fast_len(need)
        fa = rfft(a_xx, nf)
        fw = rfft(w_lags[::-1], nf)
        cr = irfft(fa * fw, nf)[:need]
        idx = (m - 1) - np.arange(nn)
        idx = np.clip(idx, 0, need - 1)
        return 2.0 * cr[idx]

    def _soft_weights(c, M, temp_scale=0.08):
        T = max(M * temp_scale, 1e-14)
        z = (c - M) / T
        z = np.clip(z, -60.0, 0.0)
        p = np.exp(z)
        ps = p.sum()
        if ps > 0 and np.isfinite(ps):
            return p / ps
        w = np.zeros(c.size)
        w[np.argmax(c)] = 1.0
        return w

    def _lp_solve(a_cur, M_val, dl):
        """Solve LP: max sum(b) s.t. A*b <= M_val, b>=0. Returns (b, dual)."""
        rem = dl - time.monotonic()
        if rem < 2.0:
            return None, None
        nn = a_cur.size
        m = 2 * nn - 1
        # Build A: A[k,j] = a_cur[k-j] if 0<=k-j<nn
        # Use the banded structure
        A = np.zeros((m, nn), dtype=np.float64)
        for j in range(nn):
            A[j:j + nn, j] = a_cur
        A_scaled = A / M_val
        tl = max(0.5, min(8.0, 0.1 * rem))
        try:
            res = linprog(
                -np.ones(nn),
                A_ub=A_scaled,
                b_ub=np.ones(m),
                bounds=[(0, None)] * nn,
                method='highs',
                options={'time_limit': tl, 'presolve': True}
            )
        except Exception:
            try:
                res = linprog(-np.ones(nn), A_ub=A_scaled, b_ub=np.ones(m),
                              bounds=[(0, None)] * nn, method='highs')
            except Exception:
                return None, None
        if res is not None and res.success:
            b = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
            sb = b.sum()
            dual = None
            try:
                marg = getattr(res.ineqlin, 'marginals', None)
                if marg is not None:
                    dual = np.maximum(np.asarray(marg, dtype=np.float64), 0.0)
                    ds = dual.sum()
                    if ds > 1e-15:
                        dual = dual / ds
                    else:
                        dual = None
            except Exception:
                dual = None
            return b, dual
        return None, None

    # --- Main optimization ---
    best_a = a.copy()
    best_score = float('inf')
    try:
        sc0 = evaluate_sequence([float(x) for x in a])
        if np.isfinite(sc0):
            best_score = sc0
    except Exception:
        best_score = 2.0

    c = _conv(a)
    M = float(c.max())

    it = 0
    no_improve = 0
    lp_every = 2
    sigma = 0.04

    while time.monotonic() < deadline - 3.0:
        it += 1

        c = _conv(a)
        M = float(c.max())
        cur_score = 2.0 * n * M / 1.0  # sum(a)=1

        if np.isfinite(cur_score) and cur_score < best_score * (1 - 1e-14):
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Random restart when stuck
        if no_improve > 80:
            r = rng.random()
            if r < 0.4:
                a = _proj_simplex(a + rng.normal(0, sigma, n))
            elif r < 0.7:
                x = np.linspace(0, 1, n, endpoint=False)
                alpha = float(rng.uniform(0.3, 2.0))
                a = _proj_simplex(x ** alpha + 1e-6)
            elif r < 0.9:
                x = np.linspace(0, 1, n, endpoint=False)
                ctr = float(rng.uniform(0.2, 0.8))
                wdt = float(rng.uniform(0.05, 0.3))
                a = _proj_simplex(np.exp(-((x - ctr) ** 2) / (2 * wdt ** 2)) + 0.1)
            else:
                a = _proj_simplex(0.5 * best_a + 0.5 * _proj_simplex(rng.uniform(0.1, 1.5, n)))
            sigma = min(0.3, sigma * 1.5)
            no_improve = 0
            continue

        # Scale down noise when improving
        if no_improve == 0:
            sigma = max(0.005, sigma * 0.97)

        # LP step (every lp_every iterations)
        did_lp = False
        if n <= 600 and it % lp_every == 1 and (deadline - time.monotonic()) > 25.0:
            b_sol, dual = _lp_solve(a, M, deadline - 5.0)
            if b_sol is not None:
                sb = b_sol.sum()
                if sb > 1e-10:
                    b_norm = b_sol / sb
                    # Line search
                    improved = False
                    for t in (0.4, 0.6, 0.2, 0.8, 0.5, 0.3, 1.0):
                        cand = (1 - t) * a + t * b_norm
                        cc = _conv(cand)
                        if cc.max() < M * (1 - 1e-14):
                            a = cand
                            c = cc
                            M = float(cc.max())
                            improved = True
                            break
                    did_lp = True

                    # Use dual for gradient if available
                    if dual is not None and np.any(dual > 1e-12):
                        g = _fft_grad(a, dual)
                        g = np.maximum(g, 0.0)
                        gm = g.mean()
                        if gm > 1e-14:
                            for eta in (0.3, 0.6, 1.0, 0.15):
                                cand = _proj_simplex(a - eta * g / gm)
                                cc = _conv(cand)
                                if cc.max() < M * (1 - 1e-14):
                                    a = cand
                                    c = cc
                                    M = float(cc.max())
                                    break

        # Gradient step (always, or if LP didn't fire)
        if not did_lp or np.random.default_rng(it).random() < 0.5:
            w = _soft_weights(c, M, temp_scale=0.06)
            g = _fft_grad(a, w)
            g = np.maximum(g, 0.0)
            gm = g.mean()
            if gm > 1e-14:
                for eta in (0.4, 0.8, 0.2, 1.2, 0.6):
                    cand = _proj_simplex(a - eta * g / gm)
                    cc = _conv(cand)
                    if cc.max() < M * (1 - 1e-14):
                        a = cand
                        c = cc
                        M = float(cc.max())
                        break

        # Multi-start exploration: every ~200 iterations, try a different shape
        if it % 200 == 0 and (deadline - time.monotonic()) > 100.0:
            x = np.linspace(0, 1, n, endpoint=False)
            shape = _proj_simplex(x ** float(rng.uniform(0.2, 1.8)) + 1e-6)
            if _conv(shape).max() < M:
                a = shape
                c = _conv(a)
                M = float(c.max())

    # Final validation
    try:
        best_a = np.asarray(best_a, dtype=np.float64).ravel()
        best_a = np.where(np.isfinite(best_a), best_a, 0.0)
        best_a = np.maximum(best_a, 0.0)
        bs = best_a.sum()
        if bs < 1e-12:
            best_a = np.ones(32, dtype=np.float64) / 32.0
        else:
            best_a = best_a / bs

        final_sc = evaluate_sequence([float(x) for x in best_a])
        if not np.isfinite(final_sc):
            best_a = np.ones(32, dtype=np.float64) / 32.0
    except Exception:
        best_a = np.ones(32, dtype=np.float64) / 32.0

    return [float(x) for x in best_a]
