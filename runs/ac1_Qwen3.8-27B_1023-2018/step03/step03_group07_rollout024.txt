def _line_search(a, g, max_evals=30):
    """Find best alpha in [0,1] minimizing max((1-a)*a + a*g)^2 conv."""
    def obj(alpha):
        cand = (1.0 - alpha) * a + alpha * g
        cand = np.maximum(cand, 0.0)
        cs = cand.sum()
        if cs < 1e-15:
            return float('inf')
        return float(np.convolve(cand, cand).max())
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # ~0.618
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = obj(c1)
    f2 = obj(c2)
    
    best_val = min(f1, f2)
    best_alpha = c1 if f1 < f2 else c2
    
    for _ in range(max_evals):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = obj(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = obj(c2)
        
        best_val = min(best_val, f1, f2)
        best_alpha = c1 if f1 <= f2 else c2
        
        if hi - lo < 1e-6:
            break
    
    if best_alpha <= 1e-8:
        return None, float('inf')
    
    cand = (1.0 - best_alpha) * a + best_alpha * g
    cand = np.maximum(cand, 0.0)
    cand = cand / cand.sum()
    return cand, obj(best_alpha)

def _gradient_step(a, k_active, lr=0.01):

    n = a.size

    for k in k_active:

        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):

            if 0 <= idx < n:

    # is the average (or any convex combination). Let me use equal weights.

    grad -= grad.mean()

    new_a = a - lr * grad
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s < 1e-15:
        return None
    new_a /= s
    
    return new_a

import time
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()

        p = np.ones(201, dtype=np.float64) / 201.0

def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    c = np.convolve(a, a)
    return 2.0 * a.size * float(c.max()) / (s * s)


def _conv(a):
    return np.convolve(a, a)


def _build_A(a):
    n = a.size
    m = 2 * n - 1
    k = np.arange(m)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    mask = (idx >= 0) & (idx < n)
    A = np.where(mask, a[np.clip(idx, 0, n - 1)], 0.0)
    return A


def _lp_minimax(a, c):
    """LP: min t s.t. A@g <= c + t, sum(g)=1, g>=0."""
    n = a.size
    A = _build_A(a)
    m = A.shape[0]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.empty((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = c.astype(np.float64).copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(
            c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
            bounds=bounds, method="highs",
            options={"time_limit": 10.0,
                     "dual_feasibility_tolerance": 1e-12,
                     "primal_feasibility_tolerance": 1e-12},
        )
    except Exception:
        return None, 0.0
    if res.status != 0 or res.x is None:
        return None, 0.0
    g = np.asarray(res.x[:n], dtype=np.float64)
    t_val = float(res.x[n])
    gsum = g.sum()
    if gsum <= 1e-15:
        return None, t_val
    return g / gsum, t_val


def _pgs_step(a, k_active, eta):
    """Projected gradient step onto simplex {x>=0, sum=1}."""
    n = a.size
    grad = np.zeros(n)
    ka = np.asarray(k_active)
    for k in ka:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += 2.0 * a[k - j]
    grad /= len(ka)
    v = a - eta * grad
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.maximum(v - mid, 0.0).sum() > 1.0:
            lo = mid
        else:
            hi = mid
    a_new = np.maximum(v - (lo + hi) * 0.5, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return None
    return a_new / s


def _line_search(a, g, cur_M, n_eval_max=35):
    """Golden-section then bisection line search for best mix fraction."""
    calls = [0]

    def obj(t):
        calls[0] += 1
        x = (1.0 - t) * a + t * g
        x = np.maximum(x, 0.0)
        xsum = x.sum()
        if xsum < 1e-15:
            return 1e30
        x = x / xsum
        return float(np.convolve(x, x).max())

    # coarse scan
    grid = np.linspace(0.05, 1.0, 12)
    best_t, best_v = 0.0, cur_M
    for t in grid:
        v = obj(t)
        if v < best_v:
            best_v, best_t = v, t
    if best_t == 0.0:
        return None, cur_M

    # golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = max(0.0, best_t - 0.12)
    hi = min(1.0, best_t + 0.12)
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = obj(c1), obj(c2)
    for _ in range(18):
        if calls[0] >= n_eval_max:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = obj(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = obj(c2)
        if f1 < best_v:
            best_v, best_t = f1, c1
        if f2 < best_v:
            best_v, best_t = f2, c2
        if hi - lo < 1e-7:
            break

    x = (1.0 - best_t) * a + best_t * g
    x = np.maximum(x, 0.0)
    x /= x.sum()
    return x, best_v


def _optimize(a_init, stop_time, seed=0):
    n = a_init.size
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck, perturb, eta = 0, 0, 0.008
    while time.monotonic() < stop_time - 0.3:

        M = float(b.max())
        if M < best_M * (1.0 - 1e-14):
            best_M, best_a, stuck = M, a.copy(), 0
        else:
            stuck += 1
        if stuck > 100:
            sc = min(0.04 + 0.01 * perturb, 0.5)
            a = np.maximum(a * np.exp(rng.normal(0, sc, n)), 1e-10)
            a /= a.sum()
            stuck, perturb = 0, perturb + 1
            if perturb > 50:
                break
            continue
        thr = M * 0.995
        k_act = np.where(b >= thr)[0]
        moved = False
        g, _ = _lp_minimax(a, b)
        if g is not None and np.abs(g - a).sum() > 1e-5:
            cand, val = _line_search(a, g, M)
            if cand is not None and val < M * (1 - 1e-12):
                a, moved = cand, True
        if not moved:
            for et in (eta, eta * 0.5, eta * 0.2):
                na = _pgs_step(a, k_act, et)

                v = float(np.convolve(na, na).max())

                    break
        if not moved:

            a /= a.sum()
            eta = max(eta * 0.7, 1e-5)
    return best_a
