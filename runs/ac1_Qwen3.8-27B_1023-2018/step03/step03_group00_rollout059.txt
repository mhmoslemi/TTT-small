            fd = fc
            c = a_hi - (a_hi - a_lo) / gr
            fc = _sc(c)
        else:
            a_lo, c = c, d
            fc = fd
            d = a_lo + (a_hi - a_lo) / gr
            fd = _sc(d)

    t_best = (a_lo + a_hi) / 2.0
    return t_best


def _optimize_lp(a_init, n, stop_time, rng, tag=""):
    a = np.clip(a_init.copy(), 0, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = _score(a)
    stall = 0
    max_stall = 100
    t_interp = 0.35
    it = 0

    while time.monotonic() < stop_time - 5.0 and it < 500:
        it += 1
        c_arr = _conv_full(a)
        M = float(c_arr.max())
        if M < 1e-20:
            break

        g = _lp_direction(a, n, M)
        if g is None:
            stall += 4
            if stall > max_stall:
                break
            a = np.clip(a + rng.normal(0, 0.015, n), 0, None)
            s = a.sum()
            if s > 1e-12:
                a = a / s
            continue

        g = g / g.sum()
        dist = float(np.linalg.norm(g - a))
        if dist < 1e-9:
            stall += 1
            if stall > max_stall:
                break
            continue

        # Try several t values including golden section
        t_gs = _golden_line_search(a, g, n)
        t_candidates = sorted(set([
            max(0.01, t_interp),
            max(0.01, t_interp * 0.5),
            max(0.01, t_interp * 1.5),
            t_gs,
            max(0.01, t_gs * 0.6),
        ]))

        best_t = 0.0
        best_t_sc = best_sc
        for t in t_candidates:
            f_new = (1.0 - t) * a + t * g
            f_new = np.clip(f_new, 0, None)
            s_new = f_new.sum()
            if s_new < 1e-12:
                continue
            f_new = f_new / s_new
            sc = _score(f_new)
            if sc < best_t_sc - 1e-15:
                best_t = t
                best_t_sc = sc

        if best_t_sc < best_sc - 1e-14:
            a[(slice(None))] = ((1.0 - best_t) * a + best_t * g)
            a = np.clip(a, 0, None)
            a = a / a.sum()
            best_a = a.copy()
            best_sc = best_t_sc
            t_interp = min(0.75, t_interp * 1.08)
            stall = 0
        else:
            stall += 1
            t_interp = max(0.05, t_interp * 0.7)
            if stall >= max_stall:
                a = np.clip(a + rng.normal(0, 0.04, n), 0, None)
                s = a.sum()
                if s > 1e-12:
                    a = a / s
                t_interp = 0.3
                stall = 0
                max_stall += 30

    return best_a, best_sc


def _subgradient_polish(a, stop_time, rng, n_steps=200):
    """Subgradient-based refinement on the min-max objective."""
    a = np.clip(a.copy(), 0, None).astype(np.float64)
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a, math.inf
    a = a / S
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.02

    for step in range(n_steps):
        if time.monotonic() > stop_time - 5.0:
            break
        c = _conv_full(a)
        # Find all near-max lags
        threshold = c.max() * 0.999
        active = np.where(c >= threshold)[0]
        # Subgradient: average over active lags
        g = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    g[i] += 2.0 * a[j]
        g /= max(1, len(active))
        # Multiplicative update (mirror descent on simplex)
        a_new = a * np.exp(-eta * g)
        s = a_new.sum()
        if s < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a_new = np.clip(a_new, 0, None)
        sc = _score(a_new)
        if sc < best_sc - 1e-15:
            best_a = a_new.copy()
            best_sc = sc
            a = a_new
            eta = min(0.1, eta * 1.05)
        else:
            eta = max(0.003, eta * 0.8)

    return best_a, best_sc


def _polish_cd(a, stop_time, rng):
    """Coordinate descent with adaptive multipliers."""
    a = np.clip(a.copy(), 0, None).astype(np.float64)
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a, math.inf
    a = a / S
    best_sc = _score(a)
    no_improve = 0
    max_no_imp = 12
    mults = [0.0, 0.05, 0.15, 0.3, 0.5, 0.7, 0.85, 1.15, 1.35, 1.6, 2.0, 3.0]

    while time.monotonic() < stop_time - 5.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break
            orig = a[i]
            best_m = 1.0
            for m in mults:
                a[i] = orig * m
                if a.sum() < 0.01:
                    a[i] = orig
                    continue
                sc = _score(a)
                if sc < best_sc - 1e-15:
                    best_m = m
                    best_sc = sc
                    improved = True
            a[i] = orig * best_m

        # Two-coordinate moves
        for _ in range(60):
            if time.monotonic() > stop_time - 5.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = a[i], a[j]
            for mi in (0.3, 0.6, 1.4, 2.0, 0.0):
                for mj in (0.3, 0.6, 1.4, 2.0, 0.0):
                    a[i] = oi * mi
                    a[j] = oj * mj
                    if a.sum() < 0.01:
                        a[i], a[j] = oi, oj
                        continue
                    sc = _score(a)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
            a[i], a[j] = oi, oj

        if not improved:
            no_improve += 1
            if no_improve >= max_no_imp:
                break
        else:
            no_improve = 0

    return a, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 15.0

    parent = _get_parent()
    n_p = len(parent)
    rng = np.random.default_rng(seed)

    # Normalize parent to sum=1
    p = np.clip(parent, 0, None).astype(np.float64).copy()
    ps = p.sum()
    if ps < 1e-10:
        p = np.ones(n_p) / n_p
    else:
        p = p / ps

    best_seq = p.copy()
    best_score = _score(p)

    # Phase 1: LP optimization on parent length with multiple inits
    phase1_end = t0 + budget_s * 0.45
    n = n_p
    i_arr = np.arange(n, dtype=np.float64)

    inits = [p.copy()]
    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n)) + 1e-6)
    # Asymmetric bump
    inits.append(np.exp(-0.5 * ((i_arr - n * 0.6) / (n * 0.18)) ** 2) + 1e-4)
    # Two-lobe
    inits.append(
        np.exp(-0.5 * ((i_arr - n * 0.3) / (n * 0.12)) ** 2)
        + np.exp(-0.5 * ((i_arr - n * 0.7) / (n * 0.12)) ** 2) + 1e-4
    )
    # Parent + noise
    inits.append(np.abs(p + rng.normal(0, 0.1 * p.mean(), n)) + 1e-4)

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 8.0:
            break
        init = np.clip(init, 0, None).copy()
        s = init.sum()
        if s < 1e-10:
            continue
        init = init / s
        stop = min(phase1_end - 4.0, time.monotonic() + max(20.0, (phase1_end - t0) / len(inits)))
        try:
            res, sc = _optimize_lp(init, n, stop, np.random.default_rng(seed + 1000 * idx))
        except Exception:
            continue
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq, best_score = res.copy(), sc

    # Phase 2: Subgradient polish on best
    if time.monotonic() < deadline - 60.0:
        sg_end = min(deadline - 50.0, time.monotonic() + budget_s * 0.08)
        try:
            best_seq, best_score = _subgradient_polish(best_seq, sg_end, rng)
        except Exception:
            pass

    # Phase 3: LP on a few other lengths
    phase3_end = t0 + budget_s * 0.75
    other_lengths = []
    for dl in [-32, -16, 16, 32, -64, 64]:
        L = n_p + dl
        if 64 <= L <= 1200 and L not in other_lengths:
            other_lengths.append(L)

    for L in other_lengths:
        if time.monotonic() > phase3_end - 10.0:
            break
        if len(best_seq) == L:
            init_l = best_seq.copy()
        else:
            init_l = np.interp(
                np.linspace(0, 1, L),
                np.linspace(0, 1, len(best_seq)),
                best_seq,
            )
        init_l = np.clip(init_l, 0, None)
        s = init_l.sum()
        if s < 1e-10:
            continue
        init_l = init_l / s
        stop = min(phase3_end - 4.0, time.monotonic() + max(10.0, (phase3_end - phase1_end) / max(1, len(other_lengths))))
        try:
            res, sc = _optimize_lp(init_l, L, stop, np.random.default_rng(seed + 50000 + L))
        except Exception:
            continue
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq, best_score = res.copy(), sc

    # Phase 4: Additional LP pass on current best (warm start)
    if time.monotonic() < deadline - 40.0:
        lp2_end = min(deadline - 30.0, time.monotonic() + budget_s * 0.10)
        try:
            best_seq, best_score = _optimize_lp(best_seq.copy(), len(best_seq), lp2_end, np.random.default_rng(seed + 777777))
        except Exception:
            pass

    # Phase 5: Coordinate descent polish
    cd_end = deadline - 10.0
    if time.monotonic() < cd_end - 15.0:
        try:
            best_seq, best_score = _polish_cd(best_seq, cd_end - 5.0, np.random.default_rng(seed + 31337))
        except Exception:
            pass

    # Phase 6: Second subgradient polish
    if time.monotonic() < deadline - 12.0:
        try:
            sg_end2 = min(deadline - 8.0, time.monotonic() + 25.0)
            best_seq, best_score = _subgradient_polish(best_seq, sg_end2, np.random.default_rng(seed + 55555))
        except Exception:
            pass

    # Phase 7: Final tiny CD
    if time.monotonic() < deadline - 6.0:
        try:
            best_seq, best_score = _polish_cd(best_seq, deadline - 4.0, np.random.default_rng(seed + 11111))
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Fallback to parent
        try:
            final_sc2 = float(evaluate_sequence([float(x) for x in p]))
        except Exception:
            final_sc2 = math.inf
        if math.isfinite(final_sc2):
            return [float(x) for x in p]
        return [1.0]

    return [float(x) for x in best_seq]
