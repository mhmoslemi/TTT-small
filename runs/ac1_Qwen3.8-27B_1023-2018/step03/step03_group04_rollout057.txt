import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len

def _conv_full(a):
    """Compute full linear convolution via FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_eval(a):
    """Compute the evaluator score."""
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    conv = _conv_full(a)
    M = float(conv.max())
    return 2.0 * n * M / (S * S)

def _subgrad_step(a, conv, active_frac=0.1, lr=0.01):
    """One step of subgradient descent on max(conv) with sum(a)=1 constraint."""
    n = len(a)
    M = float(conv.max())
    if M < 1e-15:
        return a.copy()
    
    # Active set: indices where conv is within active_frac of max
    thresh = M * (1.0 - active_frac)
    active = np.where(conv >= thresh)[0]
    
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    
    # Subgradient: for each active k, d conv[k]/da[i] = 2*a[k-i] if valid
    sg = np.zeros(n)
    for k in active:
        for i in range(n):
            idx = int(k) - int(i)
            if 0 <= idx < n:
                sg[i] += 2.0 * a[idx]
    sg /= len(active)
    
    # Line search / step
    a_new = a - lr * sg
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    a_new /= s
    return a_new
