**Alternative: Semidefinite programming (SDP)**

**Another alternative: Frank-Wolfe (conditional gradient) method**

The Frank-Wolfe method is designed for minimizing a convex function over a convex set. Our problem is non-convex, but we can use a "non-convex" FW variant.

The constraint set is: C = {a : a >= 0, sum(a) = 1} (a simplex).
The objective is: f(a) = 2n * max_k (a*a)[k] (after normalizing sum = 1).

The FW method:
1. Compute subgradient g = ∇f(a) (or subgradient of the max)
2. Solve: v = argmin_{b in C} g^T b
3. Update: a_new = (1-t) * a + t * v

The linear minimization oracle (step 2) over the simplex is:
v[i] = 1 if i = argmin_j g[j], else 0

This is very cheap! But the FW method is slow to converge (O(1/t) rate).

However, with a "shrinking" strategy (reducing the step size), it can be competitive. And it avoids the expensive LP solve.

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv(a):
    """Linear autocorrelation via FFT."""
    n = len(a)
    if n <= 1:
        return np.array([a[0]**2]) if n == 1 else np.array([0.0])
    fsize = next_fast_len(2*n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_arr(a):
    """Compute the AC1 upper bound score."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_conv_matrix(f):
    """Build matrix A where (A @ g) = linear_convolution(f, g)."""
    n = len(f)
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A


def _solve_lp(f, n, M, time_limit=8.0):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A = _build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.maximum(res.x.astype(np.float64), 0.0)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs


def _lp_optimize(f_init, n, stop_time, seed, verbose=False):
    """Run LP-based optimization iterations."""
    rng = np.random.default_rng(seed)
    f = np.clip(f_init.astype(np.float64), 0.0, None)
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs
    
    best_f = f.copy()
    best_sc = _score_arr(f)
    
    if verbose:
        print(f"  Start: score={best_sc:.10f}")
    
    t_scale = 1.0
    stall = 0
    max_stall = 50
    it = 0
    no_improve = 0
    
    while time.monotonic() < stop_time - 5.0:
        it += 1
        
        b = _conv(f)
        M = float(b.max())
        if M < 1e-15:
            break
        
        # Solve LP
        g = _solve_lp(f, n, M)
        if g is None:
            stall += 5
            if stall >= max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.05, n), 0.0, None)
            f = f / max(f.sum(), 1e-10)
            t_scale = 0.5
            continue
        
        # Check fixed point
        dist = float(np.linalg.norm(g - f))
        if dist < 1e-8:
            stall += 3
            if stall >= max_stall:
                break
            # Perturb using active-lag information
            active = np.where(b >= M * 0.999)[0]
            n_active = len(active)
            if n_active > 0 and n_active < 2*n-1:
                # Reduce f at positions contributing to active lags
                perturbation = np.zeros(n)
                for k in active[:10]:
                    lo = max(0, int(k) - n + 1)
                    hi = min(n, int(k) + 1)
                    perturbation[lo:hi] -= 0.02 * f[lo:hi].max()
                f = np.clip(f + perturbation + rng.normal(0, 0.03, n), 0.0, None)
            else:
                f = np.clip(f + rng.normal(0, 0.08, n), 0.0, None)
            f = f / max(f.sum(), 1e-10)
            t_scale = 0.5
            continue
        
        # Line search: try geometric step sizes
        direction = g - f
        best_step = None
        best_step_sc = best_sc
        
        step_sizes = []
        # Generate adaptive step sizes based on t_scale
        base = min(1.0, t_scale)
        for i in range(8):
            step_sizes.append(base * (0.5 ** i))
        # Sort descending
        step_sizes = sorted(set([s for s in step_sizes if s > 0.001]), reverse=True)
        
        for t in step_sizes:
            if time.monotonic() > stop_time - 5.0:
                break
            fn = f + t * direction
            fn = np.maximum(fn, 0.0)
            fn_sum = float(fn.sum())
            if fn_sum < 0.01:
                continue
            sc = _score_arr(fn)
            if sc < best_step_sc - 1e-15:
                best_step_sc = sc
                best_step = fn.copy()
        
        if best_step is not None:
            f = best_step
            sc = _score_arr(f)
            if sc < best_sc - 1e-14:
                best_f = f.copy()
                best_sc = sc
                no_improve = 0
                t_scale = min(1.0, t_scale * 1.15)
            else:
                no_improve += 1
            stall = 0
        else:
            t_scale = max(0.01, t_scale * 0.5)
            stall += 2
            no_improve += 1
        
        if it % 50 == 0 and verbose:
            print(f"  Iter {it}: score={best_sc:.10f}, t_scale={t_scale:.3f}")
        
        # If stuck, do a more aggressive restart
        if no_improve > 100:
            f = np.clip(f + rng.normal(0, 0.15, n), 0.0, None)
            f = f / max(f.sum(), 1e-10)
            t_scale = 0.5
            no_improve = 0
            stall = 0
        
        # Early termination if we're doing well
        if best_sc < 1.5028 and it > 50:
            break
    
    return best_f, best_sc


def _coord_descent(f_init, n, stop_time, seed):
    """Coordinate descent polish."""
    rng = np.random.default_rng(seed)
    f = np.clip(f_init.astype(np.float64), 0.0, None)
    fs = float(f.sum())
    f = f / fs if fs > 1e-10 else np.ones(n)/n
    best_sc = _score_arr(f)
    
    multipliers = (0.5, 0.7, 0.85, 1.0, 1.1, 1.3, 1.5, 2.0, 0.001)
    
    improved_total = 0
    while time.monotonic() < stop_time - 5.0:
        round_improved = 0
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break
            orig = f[i]
            best_val = orig
            for m in multipliers:
                f[i] = orig * m if m > 0.001 else 0.0
                sc = _score_arr(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_val = f[i]
                    round_improved += 1
            f[i] = best_val
        
        if round_improved == 0:
            break
        improved_total += round_improved
    
    return f, best_sc
```

```python
def _subgradient_polish(f_init, n, stop_time, seed):

    rng = np.random.default_rng(seed)

            break

        # Find the lag(s) achieving max
        max_idx = np.where(b >= M - 1e-10)[0]
        
        # Compute subgradient (average over active lags)
        grad = np.zeros(n)
        for k in max_idx:
            k = int(k)
            # b[k] = sum_{i: 0<=i<n, 0<=k-i<n} f[i]*f[k-i]
            # db[k]/df[j] = f[k-j] (if 0<=k-j<n) + f[j-k] (if 0<=j-k<n, j!=k-j)

            hi = min(n, k + 1)
            for i in range(lo, hi):
                j = k - i
                if j != i:
                    grad[i] += f[j]
                    grad[j] += f[i]
                else:
                    grad[i] += 2.0 * f[i]

        s = float(f.sum())
        obj_grad = (2.0 * n / (s*s)) * grad - (4.0 * n * M / (s*s*s)) * np.ones(n)
        
        # Update with momentum
        momentum = beta * momentum - lr * obj_grad
        f_new = np.maximum(f + momentum, 0.0)
        f_new = f_new / f_new.sum()
        
        sc = _score_arr(f_new)

                momentum *= 0.5
                beta = 0.8
                no_imp = 0
```

```python
def _solve_lp_reduced(f, n, M, n_keep=100, time_limit=3.0):
    """Solve LP with only the top n_keep constraints."""
    b = _conv(f)
    # Keep the top n_keep lags
    if len(b) > n_keep:
        top_idx = np.argpartition(b, -n_keep)[-n_keep:]
        A_rows = top_idx
    else:
        A_rows = np.arange(len(b))
    
    # Build reduced matrix
    A_full = _build_conv_matrix(f)
    A = A_full[A_rows, :]
    
    c = -np.ones(n)
    b_ub = M * np.ones(len(A_rows))
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    
    if res.status != 0:
        return None
    g = np.maximum(res.x, 0.0)
    gs = float(g.sum())
    if gs < 1e-10:
        return None
    g = g / gs
    
    # Verify: check if g satisfies ALL constraints
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv(a):
    """Linear autocorrelation via FFT."""
    n = len(a)
    if n <= 1:
        return np.array([a[0] * a[0]])
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _conv2(a, b):
    """Linear convolution of two sequences via FFT."""
    la, lb = len(a), len(b)
    n = la + lb - 1
    fsize = next_fast_len(n)
    A = rfft(a, fsize)
    B = rfft(b, fsize)
    c = irfft(A * B, fsize)[:n]
    np.maximum(c, 0.0, out=c)
    return c


def _score_arr(a):
    """Compute the AC1 upper bound score from a numpy array."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_conv_matrix(f):
    """Build matrix A where (A @ g) = linear_convolution(f, g)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp(f, n, M, time_limit=8.0):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A = _build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.maximum(res.x.astype(np.float64), 0.0)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs


def _lp_optimize(f_init, n, stop_time, seed):
    """LP-based iterative optimization to minimize the AC1 upper bound."""
    rng = np.random.default_rng(seed)
    f = np.clip(f_init.astype(np.float64), 0.0, None)
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs

    best_f = f.copy()
    best_sc = _score_arr(f)

    t_scale = 0.8
    stall = 0
    max_stall = 60
    it = 0
    no_improve = 0

    while time.monotonic() < stop_time - 4.0:
        it += 1

        b = _conv(f)
        M = float(b.max())
        if M < 1e-15:
            break

        # Solve the LP
        g = _solve_lp(f, n, M, time_limit=6.0)
        if g is None:
            stall += 5
            if stall >= max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.05, n), 0.0, None)
            f = f / max(f.sum(), 1e-10)
            t_scale = 0.5
            continue

        # Check for fixed point
        dist = float(np.linalg.norm(g - f))
        if dist < 1e-8:
            stall += 3
            if stall >= max_stall:
                break
            # Perturb based on active lags
            active = np.where(b >= M * 0.9999)[0]
            if 0 < len(active) < 2 * n - 1:
                pert = np.zeros(n)
                for k in active[:15]:
                    kk = int(k)
                    lo = max(0, kk - n + 1)
                    hi = min(n, kk + 1)
                    pert[lo:hi] -= 0.03 * f[lo:hi].max()
                f = np.clip(f + pert + rng.normal(0, 0.04, n), 0.0, None)
            else:
                f = np.clip(f + rng.normal(0, 0.1, n), 0.0, None)
            f = f / max(f.sum(), 1e-10)
            t_scale = 0.5
            continue

        # Line search with geometric step sizes
        direction = g - f
        best_step_sc = best_sc
        best_fn = None

        # Adaptive step sizes
        base = min(1.0, t_scale)
        steps = sorted(set([base * (0.55 ** i) for i in range(10) if base * (0.55 ** i) > 0.002]),
                       reverse=True)

        for t in steps:
            if time.monotonic() > stop_time - 4.0:
                break
            fn = f + t * direction
            np.maximum(fn, 0.0, out=fn)
            fn_sum = float(fn.sum())
            if fn_sum < 0.01:
                continue
            sc = _score_arr(fn)
            if sc < best_step_sc - 1e-15:
                best_step_sc = sc
                best_fn = fn.copy()

        if best_fn is not None:
            f = best_fn
            sc = _score_arr(f)
            if sc < best_sc - 1e-14:
                best_f = f.copy()
                best_sc = sc
                no_improve = 0
                t_scale = min(1.0, t_scale * 1.12)
            else:
                no_improve += 1
            stall = 0
        else:
            t_scale = max(0.02, t_scale * 0.5)
            stall += 2
            no_improve += 1

        # Aggressive restart if stuck
        if no_improve > 80:
            f = np.clip(f + rng.normal(0, 0.12, n), 0.0, None)
            f = f / max(f.sum(), 1e-10)
            t_scale = 0.5
            no_improve = 0
            stall = 0

        # Early stop if excellent
        if best_sc < 1.50275 and it > 100:
            break

    return best_f, best_sc


def _subgrad_polish(f_init, n, stop_time, seed):
    """Subgradient method with momentum for fast polynomial decrease."""
    rng = np.random.default_rng(seed)
    f = np.clip(f_init.astype(np.float64), 0.0, None)
    f = f / f.sum()
    best_f = f.copy()
    best_sc = _score_arr(f)

    lr = 0.005
    momentum = np.zeros(n)
    beta = 0.85
    no_imp = 0
    best_lr = lr

    for it in range(300000):
        if time.monotonic() > stop_time - 3.0:
            break

        b = _conv(f)
        M = float(b.max())
        s = float(f.sum())

        # Find active lags (those achieving the max)
        tol = M * 1e-7 + 1e-14
        active = np.where(b >= M - tol)[0]

        # Compute subgradient of max(b) w.r.t. f
        grad_M = np.zeros(n)
        for k in active:
            kk = int(k)
            lo = max(0, kk - n + 1)
            hi = min(n, kk + 1)
            # grad[k][j] = 2*f[kk-j] for j in [lo, hi)
            f_idx = kk - np.arange(hi - 1, lo - 1, -1)
            grad_M[lo:hi] += 2.0 * f[f_idx]
        grad_M /= len(active)

        # Full objective gradient: d(2n*M/s^2)/df = (2n/s^2)*grad_M - (4n*M/s^3)*ones
        obj_grad = (2.0 * n / (s * s)) * grad_M - (4.0 * n * M / (s * s * s)) * np.ones(n)

        # Update with momentum and projection
        momentum = beta * momentum - lr * obj_grad
        f_new = np.maximum(f + momentum, 0.0)
        f_new_sum = float(f_new.sum())
        if f_new_sum < 0.01:
            f_new = f.copy()
        else:
            f_new = f_new / f_new_sum

        sc = _score_arr(f_new)
        if sc < best_sc - 1e-15:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.03, lr * 1.003)
        else:
            f = f_new
            no_imp += 1
            lr = max(0.001, lr * 0.985)
            if no_imp > 300:
                lr = best_lr
                momentum *= 0.3
                beta = 0.75
                no_imp = 0

    return best_f, best_sc


def _coord_polish(f_init, n, stop_time, seed):

    rng = np.random.default_rng(seed)

    f = f / f.sum()

    for _round in range(200):
        if time.monotonic() > stop_time - 3.0:
            break
        round_improved = 0
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:

                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_val = f[i]
                    round_improved += 1
            f[i] = best_val

        if round_improved == 0:
            break

    return f, best_sc


def _get_initializations(parent, n, seed):

    rng = np.random.default_rng(seed)

    # Parent with noise

    inits['pnoise'] = np.abs(parent + rng.normal(0, noise_scale, n)) + 0.01

    # Sum-shaped (trapezoidal)
    half = n // 2
    inits['trapezoid'] = np.minimum(i_arr + 1, n - i_arr).astype(np.float64)

    # Multi-frequency
    sup = np.full(n, 0.15)
    for fr in (1, 2, 3, 5):
        ph = rng.uniform(0, 2 * math.pi)
        sup += 0.2 * np.cos(2 * math.pi * fr * i_arr / n + ph)
    inits['multifreq'] = np.abs(sup) + 0.01
```
