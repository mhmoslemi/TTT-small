def _project_simplex(v):
    """Project v onto the probability simplex {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

Actually, I just realized something important. Let me re-examine the LP formulation.

The LP is:
$$\max \sum_j g_j \text{ s.t. } g_j \geq 0 \text{ and } (f * g)_k \leq M \text{ for all } k$$

The motivation is: if $f$ is the current sequence and we move toward $g$, the new self-convolution should have lower peaks. But this is just a heuristic - it doesn't guarantee improvement.

A potentially better LP would be:
$$\min M' \text{ s.t. } ((1-t)f + tg * ((1-t)f + tg))_k \leq M' \text{ for all } k, g_j \geq 0$$

But this is nonlinear in $(t, g, M')$.

Another option: use a second-order expansion. The self-convolution of $(1-t)f + tg$ is:
$((1-t)f + tg) * ((1-t)f + tg) = (1-t)^2(f*f) + 2t(1-t)(f*g) + t^2(g*g)$

For small $t$, this is approximately:
$(f*f) + 2t(f*g) + O(t^2)$

So the condition $((1-t)f + tg) * ((1-t)f + tg) \leq M'$ becomes:
$(f*f)_k + 2t(f*g)_k \leq M'$ for all $k$ (to first order)

If we set $M' = M(1 - \epsilon)$ for some small $\epsilon > 0$, the constraint becomes:
$(f*f)_k + 2t(f*g)_k \leq M(1 - \epsilon)$
$2t(f*g)_k \leq M(1 - \epsilon) - (f*f)_k$

For lags where $(f*f)_k = M$ (active lags), this becomes:
$2t(f*g)_k \leq -\epsilon M$

But this requires $(f*g)_k < 0$ for active lags, which is impossible since $f, g \geq 0$. So the first-order approach doesn't work for reducing the peak.

For lags where $(f*f)_k < M$, we can potentially increase the cross-convolution without violating the constraint. But the active lags are the bottleneck.

This means the LP approach as stated is indeed a heuristic: it finds a $g$ that is "in the right direction" (minimizes cross-convolution with $f$ at active lags while maximizing total mass), but the actual improvement in the self-convolution comes from the quadratic terms.

OK, so the LP is a good direction-finding heuristic but not a direct optimization. The gradient method is more principled for the actual non-convex problem.

Let me design the gradient method more carefully.

The problem (with $\sum a = 1$):
$$\min_{a \in \Delta_n} f(a) = 2n \max_k (a*a)_k$$

where $\Delta_n = \{a : a \geq 0, \sum a = 1\}$ is the probability simplex.

This is a smooth problem almost everywhere (the max is piecewise smooth). At points where multiple lags achieve the max, the subdifferential is the convex hull of the individual gradients.

Algorithm:
1. Compute $B = a * a$, $M = B_{\max}$
2. Find active set $K = \{k : B_k \geq M - \epsilon\}$ (with small $\epsilon$ for numerical robustness)
3. Compute subgradient: $g = \frac{2n}{|K|} \sum_{k \in K} \nabla B_k$
   where $(\nabla B_k)_i = 2a_{k-i}$ (valid indices)
4. Take a projected gradient step: $a \leftarrow \text{Proj}_{\Delta_n}(a - \eta g)$
5. Line search: decrease $\eta$ until $f$ improves
6. Repeat

For the step size, I can use Armijo backtracking: find the largest $\eta$ (from a max) such that:
$f(a - \eta g) \leq f(a) - c \eta \|g\|^2$ for $c = 0.01$.

For the projection onto the simplex, I'll use the standard algorithm.

Let me also add a "smoothing" step: after computing the subgradient, I can smooth it slightly to avoid the non-smoothness of the max. This is like using a Moreau envelope.

Actually, a simpler approach: instead of using the exact active set, I can use a soft version:
$g_{soft} = \sum_k \sigma(B_k) \cdot \nabla B_k$ where $\sigma$ is a soft-max (e.g., log-sum-exp or truncated exponential).

This gives a smooth approximation to the gradient of the max.

    """Euclidean projection onto probability simplex."""

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)

    return p

def _conv(a):
    n = len(a)
    fl = next_fast_len(2 * n - 1)
    F = rfft(a, fl)
    c = irfft(F * F, fl)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    return 2.0 * n * float(b.max()) / (s * s)

def _proj(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    idx = np.where(u * np.arange(1, n + 1) > cv - 1.0)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    r = idx[-1]
    return np.maximum(v - (cv[r] - 1.0) / (r + 1.0), 0.0)

def _subgrad_M(a, K):
    n = len(a)
    g = np.zeros(n)
    for k in K:
        i0, i1 = max(0, k - n + 1), min(k, n - 1)
        if i0 <= i1:
            g[i0:i1 + 1] += a[k - i1:k - i0 + 1][::-1]
    g *= 2.0 / len(K)
    return g

def _build_A(f, idx):
    n = len(f)
    A = np.zeros((len(idx), n))
    for ii, k in enumerate(idx):
        i0, i1 = max(0, k - n + 1), min(k, n - 1)
        if i0 <= i1:
            A[ii, i0:i1 + 1] = f[k - i1:k - i0 + 1][::-1]
    return A

def _lp_dir(f, M, top_k=150, tl=4.0):
    n = len(f)
    b = _conv(f)
    idx = np.argsort(-b)[:top_k]
    A = _build_A(f, idx)
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(len(idx), M),

        return None
    if r.status not in (0, 3) or r.x is None:
        return None
    g = np.asarray(r.x, dtype=np.float64)
    s = float(g.sum())
    return g / s if s > 1e-12 else None

                if abs(d) < 1e-15:
                    continue
                if k0 <= k1:
                    B[k0:k1 + 1] += 2.0 * d * a_sl
                    if k0 <= 2 * i <= k1:
                        B[2 * i] += d * d
                a[i] = nv
                Mn = float(np.maximum(B, 0.0).max())
                scn = 2.0 * n * Mn
                if scn < bvs - 1e-15:
                    bv, bvs = nv, scn
                B[:] = ov_B
                a[i] = ov
            if abs(bv - ov) > 1e-15:
                d = bv - ov
                a[i] = bv
                if k0 <= k1:
                    B[k0:k1 + 1] += 2.0 * d * a_sl
                    if k0 <= 2 * i <= k1:
                        B[2 * i] += d * d
                B = np.maximum(B, 0.0)
                bestsc = bvs
                imp = True
        if outer % 200 == 199:
            B = _conv(a)
        if imp:
            noimp = 0
            best = a.copy()
            if bestsc < 2.0 * n * float(B.max()):
                bestsc = 2.0 * n * float(B.max())
        else:
            noimp += 1
            if noimp >= 6:
                a = _proj(a + rng.normal(0, 0.03, n))
                B = _conv(a)
                noimp = 0
    return best, bestsc

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_p = len(parent)

    try:
        fb_sc = _score(parent)
    except Exception:
        fb_sc = math.inf
    if not math.isfinite(fb_sc) or fb_sc >= 2.0:
        parent = np.ones(384) / 384
        fb_sc = _score(parent)

    best_seq = parent.copy()
    best_score = fb_sc

    rng = np.random.default_rng(seed)
    n = n_p

    # Build diverse initializations
    inits = [parent.copy()]
    i_arr = np.arange(n, dtype=np.float64)
    taper = 0.5 + 0.5 * np.cos(np.pi * i_arr / max(n - 1, 1))
    inits.append(taper)
    sup = np.ones(n) * 0.15
    for fr in (1, 2, 3, 5, 8):
        ph = rng.uniform(0, 2 * math.pi)
        sup += 0.25 * np.cos(2 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.005)
    base = rng.normal(0, 1, n)
    kern = np.exp(-0.5 * (np.arange(-8, 9) / 3.0) ** 2)
    kern /= kern.sum()
    sm = np.convolve(base, kern, 'same')
    inits.append(np.abs(sm) + 0.005)
    inits.append(np.abs(parent + rng.normal(0, 0.08 * max(1.0, np.std(parent)), n)) + 0.001)

    # Phase 1: LP optimization (58% of time)
    p1_end = t0 + budget * 0.58
    time_per_init = max(15.0, (p1_end - t0) / len(inits))
    for idx, init in enumerate(inits):
        if time.monotonic() > p1_end - 5.0 or best_score < 1.503:
            break
        stop_i = min(p1_end - 3.0, time.monotonic() + time_per_init)
        if time.monotonic() >= stop_i:
            break
        try:
            r, sc = _lp_opt(init, stop_i, seed + 7919 * (idx + 1))
            if np.all(np.isfinite(r)) and r.sum() > 0.01:
                sc = _score(r)
                if sc < best_score:
                    best_seq = r.copy()
                    best_score = sc
        except Exception:
            pass

    # Phase 2: Gradient refinement (17% of time)
    p2_end = t0 + budget * 0.75
    if time.monotonic() < p2_end - 5.0 and best_score > 1.5027:
        try:
            r, sc = _grad_opt(best_seq, p2_end - 3.0, seed + 99991)
            if np.all(np.isfinite(r)) and r.sum() > 0.01:
                sc = _score(r)
                if sc < best_score:
                    best_seq = r.copy()
                    best_score = sc
        except Exception:
            pass

    # Phase 3: Coordinate descent (13% of time)
    p3_end = deadline - 5.0
    if time.monotonic() < p3_end - 5.0:
        try:
            r, _ = _coord_opt(best_seq, p3_end - 2.0, seed + 55501)
            if np.all(np.isfinite(r)) and r.sum() > 0.01:
                sc = _score(r)
                if sc < best_score:
                    best_seq = r.copy()
                    best_score = sc
        except Exception:
            pass

    # Phase 4: Second gradient pass (4% of time)
    p4_end = deadline - 3.0
    if time.monotonic() < p4_end - 8.0 and best_score > 1.5025:
        try:
            r, sc = _grad_opt(best_seq, p4_end - 2.0, seed + 77777)
            if np.all(np.isfinite(r)) and r.sum() > 0.01:
                sc = _score(r)
                if sc < best_score:
                    best_seq = r.copy()
                    best_score = sc
        except Exception:
            pass

    # Phase 5: Final coordinate polish
    final_stop = t0 + budget - 6.0
    if time.monotonic() < final_stop - 3.0:
        try:
            r, _ = _coord_opt(best_seq, final_stop - 1.0, seed + 31337)
            if np.all(np.isfinite(r)) and r.sum() > 0.01:
                sc = _score(r)
                if sc < best_score:
                    best_seq = r.copy()
                    best_score = sc
        except Exception:
            pass

    # Final validation
    final = np.clip(best_seq, 0.0, 1000.0)
    try:
        fs = float(evaluate_sequence([float(x) for x in final]))
    except Exception:
        fs = math.inf
    if not math.isfinite(fs):
        try:
            fs = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            return [1.0]
        return [float(x) for x in parent]
    return [float(x) for x in final]
