        for sig in [0.1, 0.15, 0.2, 0.3]:
            seq = np.exp(-0.5 * ((xn - mu) / sig) ** 2)
            inits.append(seq / seq.sum())

    # Cosine window variants
    for f in [0.5, 1.0, 1.5]:
        seq = 0.5 + 0.5 * np.cos(2 * np.pi * f * xn)
        seq = np.maximum(seq, 0.01)
        inits.append(seq / seq.sum())

    # Blend parent with structured inits
    pn = best_a.copy()
    for init in inits[:8]:
        blended = 0.5 * pn + 0.5 * init
        inits.append(blended / blended.sum())

    per_init = max(3.0, (phase2_end - time.monotonic()) / max(1, len(inits)) * 0.7)
    for idx, init in enumerate(inits):
        if time.monotonic() > phase2_end - 3:
            break
        alloc = min(time.monotonic() + per_init, phase2_end - 2)
        try:
            result = _optimize_lp(init, alloc, rng)
            _update(result)
        except Exception:
            pass

    # Phase 3: Try nearby lengths (10% of time)
    phase3_end = t0 + (deadline - t0) * 0.75
    for dn in [-4, -2, -1, 0, 1, 2, 4, 6, 8]:
        if time.monotonic() > phase3_end - 5:
            break
        nt = n_parent + dn
        if nt < 16 or nt > 400:
            continue
        x_old = np.linspace(0, 1, n_parent)
        x_new = np.linspace(0, 1, nt)
        init_new = np.interp(x_new, x_old, best_a)
        init_new = np.clip(init_new, 0, 1000.0)
        if init_new.sum() < 1e-12:
            continue
        init_new = init_new / init_new.sum()
        try:
            result = _optimize_lp(init_new, time.monotonic() + 20, rng)
            _update(result)
        except Exception:
            pass

    # Phase 4: Final extended polish (remaining time)
    if time.monotonic() < deadline - 5:
        result = _optimize_multi_M(best_a, deadline, rng)
        _update(result)

    # Final validation with exact evaluator
    try:
        final_list = [float(v) for v in best_a]
        fs = float(evaluate_sequence(final_list))
        if np.isfinite(fs):
            return final_list
    except Exception:
        pass

    fallback = [1.0] * 32
    return fallback
