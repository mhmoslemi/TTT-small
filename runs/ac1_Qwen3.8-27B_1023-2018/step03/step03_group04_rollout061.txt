import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0 or p.sum() < 0.01:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c


def _score_vec(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    M = float(_conv_full(a).max())
    return 2.0 * n * M / (S * S)


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(f, M):
    n = len(f)
    if M <= 1e-15:
        return None
    A = _build_A(f)
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={"presolve": True, "time_limit": 3})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s


def _lp_optimize(f0, stop_time, seed, max_iters=200000):
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = np.clip(f0.astype(np.float64).copy(), 0, None)
    S = f.sum()
    f = f / S if S > 1e-10 else np.ones(n) / n

    best_f = f.copy()
    best_M = float(_conv_full(f).max())
    stall = 0
    max_stall = 300

    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break
        conv = _conv_full(f)
        M = float(conv.max())
        if M < best_M - 1e-14:
            best_f = f.copy()
            best_M = M
            stall = 0
        else:
            stall += 1

        g = _lp_dir(f, M)
        if g is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
            f /= f.sum() if f.sum() > 1e-10 else 1
            continue

        if np.max(np.abs(g - f)) < 1e-10:
            stall += 1
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.01, n), 0, None)
            f /= f.sum() if f.sum() > 1e-10 else 1
            continue

        # Golden section line search
        a_lo, a_hi = 0.0, 1.0
        phi = (1 + 5**0.5) / 2
        f1, f2 = None, None
        M1, M2 = math.inf, math.inf
        for _ in range(25):
            if time.monotonic() > stop_time:
                break
            m1 = a_hi - (a_hi - a_lo) / phi
            m2 = a_lo + (a_hi - a_lo) / phi
            f1 = np.clip((1 - m1) * f + m1 * g, 0, None)
            f1 /= f1.sum() if f1.sum() > 1e-10 else 1
            M1 = float(_conv_full(f1).max())
            f2 = np.clip((1 - m2) * f + m2 * g, 0, None)
            f2 /= f2.sum() if f2.sum() > 1e-10 else 1
            M2 = float(_conv_full(f2).max())
            if M1 < M2:
                a_hi = m2
            else:
                a_lo = m1
            if a_hi - a_lo < 1e-5:
                break

        t_star = (a_lo + a_hi) / 2
        f_new = np.clip((1 - t_star) * f + t_star * g, 0, None)
        f_new /= f_new.sum() if f_new.sum() > 1e-10 else 1
        M_new = float(_conv_full(f_new).max())
        f = f_new

    best_M = float(_conv_full(best_f).max())
    return best_f, 2.0 * n * best_M
