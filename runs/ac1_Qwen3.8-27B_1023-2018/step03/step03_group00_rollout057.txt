Given a = [a0, a1, ..., a_{n-1}], conv = a * a (full, length 2n-1).
conv[k] = sum_{p=max(0,k-n+1)}^{min(k,n-1)} a[p] * a[k-p]

- conv[k] changes only if i is in [max(0,k-n+1), min(k,n-1)], i.e., k-i is in [0, n-1] and k-i is in [max(0,k-n+1), min(k,n-1)]
- The condition k-i in [0,n-1] means k in [i, i+n-1]
- Also need k-i <= k (always true) and k-i >= k-n+1, i.e., i <= n-1 (always true)
- And i <= k (true since k >= i) and i >= k-n+1 (i >= k-n+1, since k <= i+n-1, k-n+1 <= i, true)

So affected k range: [i, i+n-1] (clamped to [0, 2n-2]).

For a specific k in this range, j = k - i (so 0 <= j <= n-1):
- The term a[i]*a[j] appears in conv[k] (as p=i)
- The term a[j]*a[i] appears in conv[k] (as p=j, if j≠i)
- If j = i (k = 2i): the term a[i]*a[i] = a[i]^2 appears once

- If j = i: ov^2

- If j ≠ i: 2 * nv * a[j]
- If j = i: nv^2

- If j ≠ i: 2 * (nv - ov) * a[j] = 2 * d * a[j]

        f = np.concatenate([f, np.zeros(n - f.size)])
    if f.sum() < 1e-10:
        f = np.ones(n) * 0.5
    best_f = f.copy()
    best_sc = _score(f)
    t = 0.35
    stall = 0
    max_stall = 50
    while time.monotonic() < stop - 3.0:
        b = _conv(f)
        M = float(b.max())
        if M < 1e-15:
            break
        g = _lp_dir(f, n, M, timeout=min(8.0, max(2.0, stop - time.monotonic() - 4.0)))
        if g is None:
            stall += 4
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.08 * max(f.mean(), 1.0), n), 0, None)
                if f.sum() < 0.01:
                    f = best_f.copy()
                stall = 0
                max_stall += 30
            continue
        g_sum = g.sum()
        if g_sum < 1e-12:
            stall += 2
            continue
        g = g * (f.sum() / g_sum)
        improved = False
        for tt in (t, t * 0.65, t * 0.4, t * 0.2, 0.08, 0.03, 0.01):
            if tt < 0.005:
                break
            fn = np.clip((1.0 - tt) * f + tt * g, 0, None)
            if fn.sum() < 0.01:
                continue
            sc = _score(fn)
            if sc < best_sc - 1e-13:
                f = fn
                best_f = fn.copy()
                best_sc = sc
                t = min(0.75, t * 1.12)
                stall = 0
                improved = True
                break
            else:
                t = max(0.01, t * 0.6)
        if not improved:
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.12 * max(f.mean(), 1.0), n), 0, None)
                if f.sum() < 0.01:
                    f = best_f.copy()
                t = 0.3
                stall = 0
                max_stall = 70
    return best_f, best_sc


def _coord_descent(a, stop, seed=0, passes=None):
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = _conv(a)
    S = float(a.sum())
    best_sc = 2.0 * n * float(conv.max()) / (S * S)
    best_a = a.copy()
    no_imp = 0
    max_no_imp = 6
    rng = np.random.default_rng(seed)
    p = 0
    while time.monotonic() < stop - 3.0:
        if passes is not None and p >= passes:
            break
        any_imp = False
        S = float(a.sum())
        for i in range(n):
            if time.monotonic() > stop - 3.0:
                break
            ov = float(a[i])
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            conv_aff = conv[k_lo:k_hi + 1].copy()
            peak_unaff = 0.0
            if k_lo > 0:
                peak_unaff = max(peak_unaff, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                peak_unaff = max(peak_unaff, float(conv[k_hi + 1:].max()))
            if ov > 1e-14:
                trials = [ov * m for m in (0.05, 0.2, 0.4, 0.7, 1.0, 1.3, 1.7, 2.5, 4.0, 0.0)]
            else:
                trials = [0.0, 0.001, 0.01, 0.1, 0.5, 1.0, 2.0, 5.0]
            best_tv = ov
            best_trial_sc = best_sc
            for tv in trials:
                tv = float(max(0.0, tv))
                d = tv - ov
                if abs(d) < 1e-14:
                    continue
                ns = S + d
                if ns < 0.01:
                    continue
                peak_aff = 0.0
                for idx in range(len(conv_aff)):
                    k = k_lo + idx
                    j = k - i
                    if j == i:
                        nk = conv_aff[idx] + d * (ov + tv)
                    else:
                        nk = conv_aff[idx] + 2.0 * d * a[j]
                    if nk > peak_aff:
                        peak_aff = nk
                new_peak = max(peak_aff, peak_unaff)
                sc = 2.0 * n * new_peak / (ns * ns)
                if sc < best_trial_sc - 1e-14:
                    best_trial_sc = sc
                    best_tv = tv
            if abs(best_tv - ov) > 1e-14:
                d = best_tv - ov
                a[i] = best_tv
                S = S + d
                for idx in range(len(conv_aff)):
                    k = k_lo + idx
                    j = k - i
                    if j == i:
                        conv[k] = conv_aff[idx] + d * (ov + best_tv)
                    else:
                        conv[k] = conv_aff[idx] + 2.0 * d * a[j]
                if best_trial_sc < best_sc - 1e-14:
                    best_sc = best_trial_sc
                    best_a = a.copy()
                    any_imp = True
        p += 1
        if not any_imp:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0
            max_no_imp = min(12, max_no_imp + 1)
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    n_p = len(parent)
    best_seq = parent.copy()
    best_score = _score(best_seq)

    def _try_update(arr, sc):
        nonlocal best_seq, best_score
        if arr is None or len(arr) == 0:
            return
        arr = np.clip(np.asarray(arr, dtype=np.float64), 0, 1000)
        if arr.sum() < 0.01:
            return
        if not np.all(np.isfinite(arr)):
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in arr]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-14:
            best_seq = arr.copy()
            best_score = exact

    # Phase 1: LP optimization on parent length
    phase1_end = t0 + budget * 0.45
    inits = [parent.copy()]

    # Add a few diverse inits
    i_arr = np.arange(n_p, dtype=np.float64)
    rng_main = np.random.default_rng(seed)

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n_p)))

    # Multi-frequency
    sup = np.ones(n_p) * 0.3
    for fr in (1, 2, 3, 5):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.15 * np.cos(2.0 * math.pi * fr * i_arr / n_p + ph)
    inits.append(np.abs(sup) + 0.02)

    # Parent perturbed
    sig = 0.1 * max(1.0, float(parent.mean()))
    inits.append(np.abs(parent + rng_main.normal(0, sig, n_p)) + 0.01)

    # Gaussian
    center = (n_p - 1) / 2.0
    inits.append(np.exp(-0.5 * ((i_arr - center) / (n_p / 5.0)) ** 2))

    time_per_init = max(15.0, (phase1_end - t0) / len(inits))
    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 8.0:
            break
        stop = min(phase1_end - 4.0, time.monotonic() + time_per_init)
        try:
            res_f, res_sc = _lp_loop(init, n_p, stop, seed + 1000 * (idx + 1))
            if res_sc < best_score - 1e-14:
                _try_update(res_f, res_sc)
        except Exception:
            pass

    # Phase 2: Coordinate descent polish on best
    phase2_end = t0 + budget * 0.65
    if time.monotonic() < phase2_end - 10.0:
        try:
            res_f, res_sc = _coord_descent(best_seq, phase2_end - 3.0, seed=seed + 7777)
            if res_sc < best_score - 1e-14:
                _try_update(res_f, res_sc)
        except Exception:
            pass

    # Phase 3: Another LP pass on improved best (warm start)
    phase3_end = t0 + budget * 0.80
    if time.monotonic() < phase3_end - 15.0:
        try:
            res_f, res_sc = _lp_loop(best_seq, phase3_end - 3.0, seed + 55555)
            if res_sc < best_score - 1e-14:
                _try_update(res_f, res_sc)
        except Exception:
            pass
        # Follow with coord descent
        if time.monotonic() < phase3_end - 5.0:
            try:
                res_f, res_sc = _coord_descent(best_seq, phase3_end - 3.0, seed=seed + 88888)
                if res_sc < best_score - 1e-14:
                    _try_update(res_f, res_sc)
            except Exception:
                pass

    # Phase 4: Try a couple other lengths
    phase4_end = t0 + budget * 0.92
    other_lens = [384, 448, 320]
    for L in other_lens:
        if time.monotonic() > phase4_end - 15.0:
            break
        stop = min(phase4_end - 4.0, time.monotonic() + (phase4_end - phase3_end) / max(1, len(other_lens)))
        if stop - time.monotonic() < 10.0:
            break
        init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
        try:
            res_f, res_sc = _lp_loop(init_l, L, stop, seed + L * 1000)
            if time.monotonic() < stop - 3.0:
                res_f2, res_sc2 = _coord_descent(res_f, stop, seed + L * 2000)
                if res_sc2 < res_sc:
                    res_f, res_sc = res_f2, res_sc2
            if res_sc < best_score - 1e-14:
                _try_update(res_f, res_sc)
        except Exception:
            pass

    # Phase 5: Final coordinate polish
    final_end = deadline - 5.0
    if time.monotonic() < final_end - 5.0:
        try:
            res_f, res_sc = _coord_descent(best_seq, final_end, seed=seed + 999999)
            if res_sc < best_score - 1e-14:
                _try_update(res_f, res_sc)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            return [1.0]
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    # Re-verify with exact evaluator
    try:
        val = float(evaluate_sequence([float(x) for x in best_seq]))
        if not math.isfinite(val):
            return [1.0]
    except Exception:
        return [1.0]

    return [float(x) for x in best_seq]
