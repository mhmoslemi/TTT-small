    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n_p, dtype=np.float64)

    inits = [
        parent.copy(),
        np.exp(-i_arr / (n_p * 0.25)),
        0.5 + 0.5 * np.cos(np.pi * i_arr / (n_p - 1)),
        np.exp(-((i_arr - n_p * 0.2) / (n_p * 0.12)) ** 2) + np.exp(-((i_arr - n_p * 0.8) / (n_p * 0.12)) ** 2),
        np.abs(rng_main.normal(0, 1, n_p)) + 0.05,
        np.clip(parent + rng_main.normal(0, 0.15 * max(1.0, np.mean(parent)), n_p), 0, None),
    ]

    time_alloc = [0.40, 0.12, 0.12, 0.10, 0.10, 0.16]
    phase1_avail = phase1_end - t0
    for idx, init in enumerate(inits):
        if time.monotonic() >= phase1_end - 5:
            break
        alloc = time_alloc[idx] if idx < len(time_alloc) else 0.1
        stop = min(phase1_end - 3.0, time.monotonic() + phase1_avail * alloc)
        try:
            res, sc = _lp_optimize(init, stop, seed + 100 * (idx + 1))
            _track(res)
        except Exception:
            pass

    # Subgradient refinement on best
    if time.monotonic() < phase1_end - 10:
        try:
            res, sc = _subgrad_step(best_seq, min(phase1_end - 3, time.monotonic() + 60), seed + 555)
            _track(res)
        except Exception:
            pass

    # Phase 2: More LP iterations on the best (20% budget)
    phase2_end = min(deadline, t0 + budget * 0.72)
    if time.monotonic() < phase2_end - 10:
        try:
            stop2 = min(phase2_end - 3.0, time.monotonic() + budget * 0.18)
            res, sc = _lp_optimize(best_seq, stop2, seed + 9999)
            _track(res)
        except Exception:
            pass

    # Phase 3: Explore other lengths (15% budget)
    phase3_end = min(deadline, t0 + budget * 0.87)
    other_lengths = [192, 256, 320, 448, 512, 640]
    other_lengths = [L for L in other_lengths if L != n_p]
    if n_p not in other_lengths and time.monotonic() < phase3_end - 10:
        other_lengths.insert(0, n_p)
    phase3_avail = phase3_end - max(phase2_end, time.monotonic())
    if phase3_avail > 10:
        time_per_len = max(5.0, phase3_avail / max(1, len(other_lengths)))
        for L in other_lengths:
            if time.monotonic() > phase3_end - 3:
                break
            stop = min(phase3_end - 2.0, time.monotonic() + time_per_len)
            if L == n_p:
                init_l = best_seq.copy()
            else:
                init_l = np.interp(np.linspace(0, 1, L),
                                   np.linspace(0, 1, len(parent)), parent)
            try:
                res, sc = _lp_optimize(init_l, stop, seed + 50000 + L)
                _track(res)
            except Exception:
                pass

    # Phase 4: Polish and final refinement (remaining time)
    polish_end = deadline - 8.0
    if time.monotonic() < polish_end - 10:
        try:
            polished, _ = _polish(best_seq, min(polish_end - 5, time.monotonic() + 120))
            _track(polished)
        except Exception:
            pass

    if time.monotonic() < polish_end - 5:
        try:
            res, sc = _lp_optimize(best_seq, min(polish_end - 3, time.monotonic() + 80), seed + 42424)
            _track(res)
        except Exception:
            pass

    if time.monotonic() < polish_end - 3:
        try:
            polished, _ = _polish(best_seq, polish_end - 2.0)
            _track(polished)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
