    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """Build the constraint matrix for the LP.
    A[k,j] = f[j] if 0 <= k-j < n, else 0.
    So A has shape (2n-1, n).
    """
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f
    return A

def _lp_minimize_M(f, n, S, timeout=5.0):
    """LP: minimize M subject to A@t <= M*e, t >= 0, sum(t) = S.
    Variables: [t_0, ..., t_{n-1}, M] (n+1 variables)
    Constraints:
      - A@t - M <= 0  (2n-1 constraints)
      - t >= 0  (n constraints, handled by bounds)
      - sum(t) = S (1 equality)
    """
    # Variables: x = [t_0,...,t_{n-1}, M]
    # Minimize: x[n] (just M)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality: A@t - M <= 0
    # A_in @ x <= b_in
    A_in = np.zeros((2 * n - 1, n + 1))
    for j in range(n):
        A_in[j : j + n, j] = f
    # Subtract M: column n is -1
    A_in[:, n] = -1.0
    b_in = np.zeros(2 * n - 1)
    
    # Equality: sum(t) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds: t_j >= 0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_in, b_ub=b_in, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method="highs",
                      options={"time_limit": timeout, "presolve": True})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    g = np.asarray(res.x[:n], dtype=np.float64)
    M_val = float(res.x[n])
    
    g = np.clip(g, 0.0, 1000.0)
    if g.sum() < 1e-15:
        return None
    return g, M_val

def _lp_direction_maxsum(f, n, M, timeout=5.0):
    """Original LP: maximize sum(g) subject to A@g <= M, g >= 0."""
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"time_limit": timeout, "presolve": True})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1000.0)
    if g.sum() < 1e-15:
        return None
    return g

def _subgradient_step(a, n, alpha):
    """One subgradient step for minimizing max(b) while keeping sum approximately fixed."""
    b = _conv_full(a)
    M = float(b.max())
    S = float(a.sum())
    
    # Active set: lags within 1% of M
    threshold = M * 0.99
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    
    # Subgradient of max(b) w.r.t. a_j
    grad = np.zeros(n)
    for k in active:
        # (a*a)_k = sum_j a_j * a_{k-j}, d/d a_j = a_{k-j}
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            grad[j] += a[k - j]
    grad /= len(active)
    
    # Subgradient of f = 2nM/S^2
    full_grad = 2.0 * n / (S * S) * grad - 4.0 * n * M / (S ** 3)
    
    # Step
    a_new = a - alpha * full_grad
    a_new = np.clip(a_new, 0, None)
    
    return a_new, M, S

def _optimize_lp(a0, n, stop_time, seed, method="minM"):
    rng = np.random.default_rng(seed)
    a = np.clip(np.asarray(a0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while a.size < n:
        a = np.concatenate([a, np.zeros(n - a.size)])
    a = a[:n]
    
    S = float(a.sum())
    if S < 1e-10:
        a = np.ones(n)
        S = n
    
    best_a = a.copy()
    best_score = _score(a)
    
    t = 0.3
    stall = 0
    max_stall = 100
    no_improve = 0
    
    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(a)
        M = float(b.max())
        S_cur = float(a.sum())
        
        if method == "minM":
            res = _lp_minimize_M(a, n, S_cur, timeout=4.0)
            if res is None:
                # Fallback to max-sum LP
                g = _lp_direction_maxsum(a, n, M, timeout=4.0)
            else:
                g, M_lp = res
        else:
            g = _lp_direction_maxsum(a, n, M, timeout=4.0)
        
        if g is None:
            stall += 5
            if stall > max_stall:
                break
            # Add noise
            a = np.clip(a + rng.normal(0, 0.02 * max(1.0, a.mean()), n), 0, None)
            a = a / a.sum() * S_cur
            continue
        
        g = g * (S_cur / g.sum())
        
        # Line search
        best_t = t
        best_t_sc = best_score
        found = False
        for trial in range(12):
            a_new = np.clip((1 - t) * a + t * g, 0, None)
            sc = _score(a_new)
            if sc < best_t_sc - 1e-14:
                best_t = t
                best_t_sc = sc
            if sc < best_score - 1e-14:
                a = a_new
                best_a = a.copy()
                best_score = sc
                t = min(0.8, t * 1.2)
                stall = 0
                no_improve = 0
                found = True
                break
            else:
                t = max(0.02, t * 0.6)
        
        if not found:
            stall += 1
            no_improve += 1
            if stall >= max_stall:
                # Restart with perturbation
                a = np.clip(best_a + rng.normal(0, 0.05, n), 0, None)
                a = a / a.sum() * S_cur
                t = 0.3
                stall = 0
                max_stall += 30
        
    return best_a, best_score

def _subgradient_optimize(a0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    a = np.clip(np.asarray(a0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    a = a / a.sum()  # normalize to sum=1
    
    best_a = a.copy()
    best_score = _score(a)
    
    alpha = 0.01
    min_alpha = 1e-6
    max_alpha = 0.1
    
    for it in range(5000):
        if time.monotonic() > stop_time - 3.0:
            break
        
        b = _conv_full(a)
        M = float(b.max())
        S = 1.0  # normalized
        
        # Active set
        threshold = M * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient of max(b) w.r.t. a
        grad_max = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            idx = range(j_lo, j_hi + 1)
            k_vals = np.array([k - j for j in idx])
            grad_max[j_lo:j_hi+1] += a[k_vals]
        grad_max /= len(active)
        
        # For f = max(b) with constraint sum(a)=1, use Lagrangian
        # grad of f subject to sum=1: grad_max - mean(grad_max)
        # (project gradient onto the hyperplane sum=0)
        grad_proj = grad_max - np.mean(grad_max)
        
        a_new = a - alpha * grad_proj
        a_new = np.clip(a_new, 0, None)
        if a_new.sum() < 0.01:
            a_new = a
        else:
            a_new = a_new / a_new.sum()  # renormalize
        
        sc = _score(a_new)
        if sc < best_score - 1e-14:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            alpha = min(max_alpha, alpha * 1.05)
        else:
            a = a_new if sc < _score(a) else a
            alpha = max(min_alpha, alpha * 0.9)
            if sc < best_score:
                # small improvement
                pass
    
    return best_a, best_score

def _coordinate_polish(a, stop_time, seed=4242):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    
    no_imp = 0
    max_no_imp = 15
    
    while time.monotonic() < stop_time - 3.0 and no_imp < max_no_imp:
        improved = False
        no_imp += 1
        
        # Single coordinate
        for i in range(n):
            if time.monotonic() > stop_time - 3.0:
                break
            orig = best[i]
            if orig < 1e-14:
                continue
            best_val = orig
            best_val_sc = best_sc
            for mlt in (0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.6, 0.0):
                best[i] = orig * mlt
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-15:
                    best_val = best[i]
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-15:
                best_sc = best_val_sc
                improved = True
        
        # Random pairs
        for _ in range(200):
            if time.monotonic() > stop_time - 3.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.5, 0.8, 1.2, 1.5):
                for mj in (0.5, 0.8, 1.2, 1.5):
                    best[i] = oi * mi
                    best[j] = oj * mj
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj
        
        if improved:
            no_imp = 0
    
    return best, best_sc
