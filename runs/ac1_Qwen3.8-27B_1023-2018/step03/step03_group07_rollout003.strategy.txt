def _evaluate(seq):
    a = np.clip(np.array(seq), 0, 1000)
    s = np.sum(a)
    if s < 0.01: return float('inf')
    conv = np.convolve(a, a)
    return 2*len(seq)*np.max(conv)/s**2

def evolve_sequence(base, tol=1e-3):

        half_width = max(1, half_width+1)
