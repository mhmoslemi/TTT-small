            if best_val_M < best_M - 1e-15:

                best_sc = 2.0 * n * best_M

                    best[j] = oj * mj if oj > 1e-12 else (0.01 if mj > 0 else 0.0)

                        improved = True

            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

def _conv(a, fsize):
    F = rfft(a, fsize)
    return irfft(F * F, fsize)[:2 * len(a) - 1]


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _lp_solve(f, M, timeout=2.0):
    n = len(f)
    A = _build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True, "time_limit": timeout})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s

    if time.monotonic() < deadline - 10.0:
        try:
            polished, _ = _polish(best_seq, deadline - 5.0, seed + 31337)
            _track(polished)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf
    if not math.isfinite(final_sc):
        return [1.0]
    return [float(x) for x in best_seq]
