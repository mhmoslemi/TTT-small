def _get_parent():
    obj = globals().get("height_sequence_1", None)
    if obj is None:

    try:
        arr = np.asarray(obj, dtype=np.float64).ravel()
    except Exception:

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n <= 80:
        c = np.convolve(a, a, mode="full")
    else:
        L = next_fast_len(2 * n - 1)
        fa = rfft(a, L)
        c = irfft(fa * fa, L)[: 2 * n - 1]
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return c
    c[~np.isfinite(c)] = 0.0
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return INF
    s = float(a.sum())
    if s < 1e-300:
        return INF
    c = _conv(a)
    if c.size == 0:
        return INF
    m = float(c.max())
    if not math.isfinite(m):
        return INF
    return float(2.0 * n * m / (s * s))


def _conv_matrix(a):

    n = a.size
    L = 2 * n - 1
    if toeplitz is not None:

                A[~np.isfinite(A)] = 0.0

def _rhs_vector(c_f, kind):
    if kind == 0:
        return None
    M = float(c_f.max()) if c_f.size else 0.0
    if not math.isfinite(M) or M <= 0.0:
        return None
    p = c_f / M
    p = np.clip(p, 0.0, 1.0)
    if kind == 1:

    elif kind == 2:

    b = np.maximum(b, 0.05)
    return b


def _lp_direction(f, c_f, M, kind, time_limit):
    n = f.size
    if n == 0 or M <= 0.0 or not math.isfinite(M):
        return None

    h = f / M
    A = _conv_matrix(h)
    b = _rhs_vector(c_f, kind)
    if b is None:
        b = np.ones(2 * n - 1, dtype=np.float64)
    else:
        b = np.asarray(b, dtype=np.float64)
        if b.size != 2 * n - 1:
            b = np.ones(2 * n - 1, dtype=np.float64)
        b[~np.isfinite(b)] = 1.0
        np.maximum(b, 0.05, out=b)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = None
    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": float(max(0.1, time_limit))},
        )
    except Exception:
        res = None

    if res is None:
        try:
            res = linprog(
                c_obj,
                A_ub=A,
                b_ub=b,
                bounds=bounds,
                method="highs",
                options={},
            )
        except Exception:
            return None
        if res is None:
            try:
                res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds)
            except Exception:
                return None

    x = getattr(res, "x", None)
    if x is None:
        return None
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size != n or not np.all(np.isfinite(x)):
        return None
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-16:
        return None
    mx = float(x.max())
    if not math.isfinite(mx) or mx > 1e12:
        x = np.clip(x, 0.0, 1e6)
        s = float(x.sum())
        if s <= 1e-16:
            return None
    return x / s


def _max_q_array(ts, A, B, C):
    ts = np.asarray(ts, dtype=np.float64)
    u = 1.0 - ts
    cand = (u * u)[:, None] * A[None, :]
    cand = cand + (2.0 * ts * u)[:, None] * B[None, :]
    cand = cand + (ts * ts)[:, None] * C[None, :]
    np.maximum(cand, 0.0, out=cand)
    return cand.max(axis=1)


def _cross_conv(f, g):
    n = f.size
    L = next_fast_len(2 * n - 1)
    fg = irfft(rfft(f, L) * rfft(g, L), L)[: 2 * n - 1]
    fg = np.asarray(fg, dtype=np.float64)
    if fg.size == 0:
        return fg
    fg[~np.isfinite(fg)] = 0.0
    np.maximum(fg, 0.0, out=fg)
    return fg


def _line_search_quad(f, g, c_f, stop_time=None):
    n = f.size
    sf = float(f.sum())
    if sf <= 1e-300 or not math.isfinite(sf):
        return None, INF
    if abs(sf - 1.0) > 1e-10:
        f = f / sf
        c_f = _conv(f)

    sg = float(g.sum())
    if sg <= 1e-300 or not math.isfinite(sg):
        return None, INF
    g = g / sg

    if float(np.linalg.norm(g - f)) < 1e-8:
        return None, INF

    A = c_f
    C = _conv(g)
    B = _cross_conv(f, g)

    m0 = float(A.max())
    current_score = 2.0 * n * m0
    if not math.isfinite(current_score):
        return None, INF

    ts = np.linspace(0.0, 1.0, 81, dtype=np.float64)
    extra = np.array([1e-4, 5e-4, 1e-3, 2e-3, 5e-3, 1e-2, 2e-2], dtype=np.float64)
    ts = np.unique(np.concatenate([ts, extra]))

    mts = _max_q_array(ts, A, B, C)
    idx = int(np.argmin(mts))
    if not math.isfinite(mts[idx]) or mts[idx] >= m0 - 1e-13:
        return None, INF

    if idx == 0:
        lo = 0.0
        hi = float(ts[min(1, len(ts) - 1)])
    elif idx == len(ts) - 1:
        lo = float(ts[max(0, len(ts) - 2)])
        hi = 1.0
    else:
        lo = float(ts[idx - 1])
        hi = float(ts[idx + 1])

    tr = np.linspace(lo, hi, 25, dtype=np.float64)
    mtr = _max_q_array(tr, A, B, C)
    j = int(np.argmin(mtr))
    t = float(tr[j])
    best_m = float(mtr[j])

    if not math.isfinite(best_m) or best_m >= m0 - 1e-13:
        return None, INF

    fn = (1.0 - t) * f + t * g
    np.maximum(fn, 0.0, out=fn)
    ns = float(fn.sum())
    if ns <= 1e-300:
        return None, INF
    fn = fn / ns

    sc = _score(fn)
    if not math.isfinite(sc) or sc >= current_score - 1e-14:
        return None, INF
    return fn, sc


def _pair_polish(f0, n, stop_time, seed=0, grid=21):
    f = _normalize(f0)
    n = f.size
    if n < 2:
        return f, _score(f)
    if time.monotonic() >= stop_time:
        return f, _score(f)

    base = _conv(f)
    sf = float(f.sum())
    best_score = 2.0 * n * float(base.max()) / (sf * sf)
    best_f = f.copy()
    K = 2 * n - 1

    rng = np.random.default_rng(seed)

    offsets = list(range(1, min(14, n)))
    for off in (n // 10, n // 4, n // 3, n // 2):
        if 0 < off < n and off not in offsets:
            offsets.append(off)
    if not offsets:
        offsets = [1]

    rand_pairs = max(600, min(1200, 2 * n))
    no_sweeps = 0
    sweep = 0

    def _attempt(i, j):
        nonlocal f, base, best_score, best_f
        x = float(f[i])
        y = float(f[j])
        if x <= 1e-18 or y <= 1e-18:
            return False

        L = np.zeros(K, dtype=np.float64)
        L[i : i + n] += 2.0 * f
        L[j : j + n] -= 2.0 * f

        epsb = 1e-14
        lo = -x * (1.0 - epsb) if x > 0.0 else 0.0
        hi = y * (1.0 - epsb) if y > 0.0 else 0.0
        span = hi - lo
        if not math.isfinite(span) or span <= 1e-16 * max(1.0, x, y):
            return False

        dvals = np.linspace(lo, hi, grid, dtype=np.float64)
        fracs = np.array([0.0, 0.002, 0.01, 0.02, 0.05, 0.10], dtype=np.float64)
        extra = lo + span * fracs
        extra = np.concatenate([extra, hi - span * fracs])
        extra = extra[(extra >= lo) & (extra <= hi)]
        if extra.size:
            dvals = np.unique(np.concatenate([dvals, extra]))
        if dvals.size < 2:
            return False

        cand = base + dvals[:, None] * L[None, :]
        d2 = dvals * dvals
        k1 = 2 * i
        k2 = 2 * j
        k3 = i + j
        if k1 == k2:
            cand[:, k1] += d2
        else:
            cand[:, k1] += d2
            cand[:, k2] += d2
        cand[:, k3] -= 2.0 * d2
        np.maximum(cand, 0.0, out=cand)
        m = cand.max(axis=1)
        idx = int(np.argmin(m))
        bestv = float(m[idx])

        current_m = best_score / (2.0 * n)
        tol = max(1e-15, 1e-12 * best_score)
        tolM = tol / (2.0 * n)
        if not math.isfinite(bestv) or bestv >= current_m - tolM:
            return False

        d = float(dvals[idx])
        if dvals.size > 3:
            if idx == 0:
                lo2 = float(dvals[0])
                hi2 = float(dvals[min(2, dvals.size - 1)])
            elif idx == dvals.size - 1:
                lo2 = float(dvals[max(0, dvals.size - 3)])
                hi2 = float(dvals[-1])
            else:
                lo2 = float(dvals[idx - 1])
                hi2 = float(dvals[idx + 1])
            rv = np.linspace(lo2, hi2, 15, dtype=np.float64)
            rc = base + rv[:, None] * L[None, :]
            rd2 = rv * rv
            if k1 == k2:
                rc[:, k1] += rd2
            else:
                rc[:, k1] += rd2
                rc[:, k2] += rd2
            rc[:, k3] -= 2.0 * rd2
            np.maximum(rc, 0.0, out=rc)
            rm = rc.max(axis=1)
            j2 = int(np.argmin(rm))
            if math.isfinite(rm[j2]) and rm[j2] < bestv:
                bestv = float(rm[j2])
                d = float(rv[j2])

        if bestv >= current_m - tolM:
            return False

        old_f = f.copy()
        old_base = base
        f[i] = x + d
        f[j] = y - d
        if f[i] < 0.0:
            f[i] = 0.0
        if f[j] < 0.0:
            f[j] = 0.0
        ns = float(f.sum())
        if ns <= 1e-15 or not math.isfinite(ns):
            f = old_f
            base = old_base
            return False

        f = f / ns
        new_base = _conv(f)
        new_score = 2.0 * n * float(new_base.max())
        if math.isfinite(new_score) and new_score < best_score - tol:
            base = new_base
            best_score = new_score
            best_f = f.copy()
            return True

        f = old_f
        base = old_base
        return False

    while time.monotonic() < stop_time - 1.5:
        if time.monotonic() >= stop_time - 1.5:
            break
        sweep += 1
        improved = False

        off = offsets[(sweep - 1) % len(offsets)]
        for i in range(n - off):
            if time.monotonic() > stop_time - 1.5:
                break
            if _attempt(i, i + off):
                improved = True

        if time.monotonic() >= stop_time - 1.5:
            break

        for _ in range(rand_pairs):
            if time.monotonic() > stop_time - 1.5:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            if _attempt(i, j):
                improved = True

        if improved:
            no_sweeps = 0
        else:
            no_sweeps += 1
            if no_sweeps >= 12:
                break

    f = _normalize(best_f)
    return f, _score(f)


def _lp_loop(f0, n, stop_time, seed):
    f = _normalize(f0)
    n = f.size
    best_f = f.copy()
    best_score = _score(f)

    if n < 2 or time.monotonic() >= stop_time - 3.0:
        return best_f, best_score

    rng = np.random.default_rng(seed)
    kind_cycle = (0, 0, 1, 0, 2, 0, 5, 0, 3, 0, 4)
    stall = 0
    it = 0

    while time.monotonic() < stop_time - 4.0:
        it += 1
        now = time.monotonic()

        if stall >= 6 and now + 5.0 < stop_time:
            pair_stop = min(stop_time - 5.0, now + 3.0)
            try:
                pf, ps = _pair_polish(f, n, pair_stop, seed + 1000 + 17 * it)
                if math.isfinite(ps) and ps < best_score - 1e-15:
                    f = _normalize(pf)
                    best_f = f.copy()
                    best_score = ps
                    stall = 0
                else:
                    stall += 1
            except Exception:
                stall += 1

        now = time.monotonic()
        if now >= stop_time - 4.0:
            break

        c_f = _conv(f)
        M = float(c_f.max()) if c_f.size else 0.0
        if not math.isfinite(M) or M <= 0.0:
            break

        kind = 0 if stall < 3 else kind_cycle[it % len(kind_cycle)]
        tlim = max(0.5, min(6.0, stop_time - now - 7.0))
        if tlim < 0.7:
            break

        g = _lp_direction(f, c_f, M, kind, tlim)
        if g is None and kind != 0:
            g = _lp_direction(f, c_f, M, 0, max(0.5, tlim - 0.5))

        if g is None:
            stall += 2
        else:
            if float(np.linalg.norm(g - f)) < 1e-7:
                stall += 2
            else:
                fn, sc = _line_search_quad(f, g, c_f)
                if fn is not None and math.isfinite(sc) and sc < best_score - 1e-15:
                    f = _normalize(fn)
                    best_f = f.copy()
                    best_score = sc
                    stall = 0
                else:
                    stall += 1

        if stall >= 10:
            sigma = 0.02 + 0.015 * min(4, (stall - 10) // 3)
            try:
                f = f * np.exp(rng.normal(0.0, sigma, n))
                f = _normalize(f)
                stall = 0
            except Exception:
                f = best_f.copy()
                stall = 0

        if it % 25 == 0 and stall == 0 and time.monotonic() + 4.0 < stop_time:
            pair_stop = min(stop_time - 4.0, time.monotonic() + 2.0)
            try:
                pf, ps = _pair_polish(f, n, pair_stop, seed + 2000 + 7 * it)
                if math.isfinite(ps) and ps < best_score - 1e-15:
                    f = _normalize(pf)
                    best_f = f.copy()
                    best_score = ps
                    stall = 0
            except Exception:
                pass

    return best_f, best_score


def _trim_search(a, max_trim=40, stop_time=None):
    a = _normalize(a)
    n = a.size
    if n <= 2:
        return a, _score(a)

    best = a.copy()
    best_sc = _score(best)
    mt = min(max_trim, n - 2)

    for L in range(mt + 1):
        if stop_time is not None and time.monotonic() >= stop_time:
            break
        maxR = min(mt, n - L - 2)
        for R in range(maxR + 1):
            if L == 0 and R == 0:
                continue
            cand = a[L : n - R]
            if cand.size < 2:
                continue
            if float(cand.sum()) < 0.01:
                continue
            sc = _score(cand)
            if math.isfinite(sc) and sc < best_sc - 1e-14:
                best = cand.copy()
                best_sc = sc

    best = _normalize(best)
    return best, _score(best)


def _exact_score(lst):
    try:
        val = evaluate_sequence(lst)
    except Exception:
        val = _score(np.asarray(lst, dtype=np.float64))
    try:
        return float(val)
    except Exception:
        return INF


def _validate_and_return(cands):
    best_list = None
    best_val = INF
    default_len = 384
    for a in cands:
        arr = np.asarray(a, dtype=np.float64).ravel()
        if arr.size == 0:
            continue
        default_len = max(default_len, arr.size)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if float(arr.sum()) < 0.01:
            continue
        lst = [float(x) for x in arr]
        sc = _exact_score(lst)
        if math.isfinite(sc) and sc < best_val:
            best_val = sc
            best_list = lst
    if best_list is None:
        best_list = [1.0] * default_len
    return best_list


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(30.0, float(budget_s))
    except Exception:
        budget = 1300.0
    deadline = t0 + budget - 25.0

    seed = int(seed) if isinstance(seed, (int, np.integer)) else 42

    parent = _get_parent()
    if parent is None:
        parent = np.ones(384, dtype=np.float64)
    parent = _normalize(parent)

    n_main = int(parent.size)
    best_arr = parent.copy()
    best_score = _score(best_arr)

    if budget < 80:
        return _validate_and_return([best_arr, parent])

    stop_main = min(deadline, t0 + budget * 0.55)
    if time.monotonic() < stop_main - 5.0:
        try:
            arr, sc = _lp_loop(best_arr, n_main, stop_main, seed)
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_arr = arr
                best_score = sc
        except Exception:
            pass

    stop_pair1 = min(deadline, t0 + budget * 0.70)
    if time.monotonic() < stop_pair1 - 5.0:
        try:
            arr, sc = _pair_polish(best_arr, len(best_arr), stop_pair1, seed + 101)
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_arr = arr
                best_score = sc
        except Exception:
            pass

    stop_lp2 = min(deadline, t0 + budget * 0.84)
    if time.monotonic() < stop_lp2 - 20.0:
        try:
            arr, sc = _lp_loop(best_arr, len(best_arr), stop_lp2, seed + 777)
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_arr = arr
                best_score = sc
        except Exception:
            pass

    stop_trim = min(deadline, t0 + budget * 0.89)
    if time.monotonic() < stop_trim:
        try:
            arr, sc = _trim_search(best_arr, max_trim=40, stop_time=stop_trim)
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_arr = arr
                best_score = sc
        except Exception:
            pass

    if len(best_arr) != n_main:
        stop_lp3 = min(deadline - 25.0, t0 + budget * 0.94)
        if time.monotonic() < stop_lp3 - 15.0:
            try:
                arr, sc = _lp_loop(best_arr, len(best_arr), stop_lp3, seed + 2024)
                if math.isfinite(sc) and sc < best_score - 1e-15:
                    best_arr = arr
                    best_score = sc
            except Exception:
                pass

    stop_final = deadline
    if time.monotonic() < stop_final - 5.0:
        try:
            arr, sc = _pair_polish(best_arr, len(best_arr), stop_final, seed + 9999)
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_arr = arr
                best_score = sc
        except Exception:
            pass

    try:
        arr, sc = _trim_search(best_arr, max_trim=20, stop_time=time.monotonic() + 8.0)
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_arr = arr
            best_score = sc
    except Exception:
        pass

    return _validate_and_return([best_arr, parent, np.ones(max(1, n_main), dtype=np.float64)])
