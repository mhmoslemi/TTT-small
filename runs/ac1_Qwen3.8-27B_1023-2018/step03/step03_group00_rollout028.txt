            return
        if not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0, 1000)
        if res.sum() < 0.01:
            return
        if sc < best_score - 1e-14:
            try:
                exact = float(evaluate_sequence([float(x) for x in res]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-14:
                best_seq = res.copy()
                best_score = exact

    # Phase 1: LP on parent length with diverse inits
    n = n_parent
    phase1_stop = t0 + budget * 0.38
    i_arr = np.arange(n, dtype=np.float64)

    inits = [parent.copy()]

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n)))

    # Multi-frequency
    sup = np.ones(n) * 0.3
    for fr in (1, 3, 7, 12):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.15 * np.cos(2.0 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.02)

    # Random abs-normal
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.05)

    # Gaussian
    center = (n - 1) / 2.0
    inits.append(np.exp(-0.5 * ((i_arr - center) / (n / 4.0)) ** 2))

    # Two-lobe
    inits.append(np.exp(-0.5 * ((i_arr - n * 0.25) / (n * 0.1)) ** 2) +
                 np.exp(-0.5 * ((i_arr - n * 0.75) / (n * 0.1)) ** 2))

    # Parent + noise
    sig = 0.12 * max(1.0, float(np.mean(parent)))
    inits.append(np.abs(parent + rng_main.normal(0, sig, n)) + 0.01)

    # Bounded random
    inits.append(rng_main.uniform(0.1, 1.5, n))

    time_per_init = max(12.0, (phase1_stop - t0) / len(inits))
    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_stop - 6:
            break
        stop = min(phase1_stop - 4, time.monotonic() + time_per_init)
        try_optimize(n, init, stop, seed + 7777 * (idx + 1))

    # Phase 2: Other lengths
    phase2_stop = t0 + budget * 0.58
    other_lengths = [96, 128, 160, 192, 224, 256, 288, 320, 352, 384, 448, 480, 512, 544]
    other_lengths = [L for L in other_lengths if L != n]
    time_per_len = max(7.0, (phase2_stop - max(time.monotonic(), t0)) / max(1, len(other_lengths)))

    for L in other_lengths:
        if time.monotonic() > phase2_stop - 5:
            break
        stop = min(phase2_stop - 3, time.monotonic() + time_per_len)
        init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n_parent), parent)
        try_optimize(L, init_l, stop, seed + 50000 + L)

    # Phase 3: Re-optimize best found (warm start)
    phase3_stop = t0 + budget * 0.78
    if time.monotonic() < phase3_stop - 15:
        try_optimize(len(best_seq), best_seq.copy(), phase3_stop - 4, seed + 999999)

    # Phase 4: Polish
    polish_stop = deadline - 8
    if time.monotonic() < polish_stop - 8:
        try:
            best_seq, best_score = polish(best_seq, polish_stop, np.random.default_rng(seed * 7 + 3))
        except Exception:
            pass

    # Phase 5: Second polish with different seed if time remains
    if time.monotonic() < deadline - 12:
        try:
            bs2, sc2 = polish(best_seq, deadline - 6, np.random.default_rng(seed * 13 + 17))
            if sc2 < best_score - 1e-14:
                best_seq, best_score = bs2, sc2
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Fallback to parent
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    return [float(x) for x in best_seq]
