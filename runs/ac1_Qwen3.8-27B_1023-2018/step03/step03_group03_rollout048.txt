```python
def _coord_polish_fast(f, stop_time, max_sweeps=100, seed=777):
    n = len(f)
    f = f.copy().astype(np.float64)
    S = float(f.sum())
    conv = _conv_full(f)
    best_score = 2.0 * n * conv.max() / (S * S)
    rng = np.random.default_rng(seed)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time:
                break
            v = f[i]
            S_rest = S - v
            
            # Influence vector
            infl = np.zeros(2 * n - 1)
            start = i
            end = min(i + n, 2 * n - 1)
            infl[start:end] = 2.0 * f[:end - start]
            
            # 1D search: x in [0, x_max]
            x_max = max(5.0 * v, S / n * 5.0) if v > 1e-10 else S / n * 2.0
            x_max = min(x_max, 10.0 * S / n)
            
            m = 200
            x_grid = np.linspace(0.0, x_max, m)
            d = x_grid - v
            new_S = S_rest + x_grid
            valid = new_S > 0.01
            if not valid.any():
                continue
            
            # Vectorized: (m, 2n-1)
            new_conv = conv[None, :].copy() + d[:, None] * infl[None, :]
            # Quadratic correction at 2i
            if 0 <= 2*i <= 2*n-2:
                new_conv[:, 2*i] += d**2
            
            max_b = new_conv.max(axis=1)
            obj = 2.0 * n * max_b / new_S**2
            obj[~valid] = np.inf
            
            idx = np.argmin(obj)
            if obj[idx] < best_score - 1e-14:
                # Accept
                f[i] = x_grid[idx]
                delta = x_grid[idx] - v
                S += delta
                conv[start:end] += delta * infl[start:end]
                if 0 <= 2*i <= 2*n-2:
                    conv[2*i] += delta * delta
                best_score = obj[idx]
                improved = True
        
        if not improved:
            break
    
    return f, best_score
```

```python
def _coord_descent(f, stop_t, max_sweeps=200, seed=0):
    n = len(f)
    f = f.copy()
    S = float(f.sum())
    if S < 0.01:
        return f, math.inf
    conv = _conv(f)
    best_sc = 2*n*conv.max()/(S*S)
    rng = np.random.default_rng(seed)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > stop_t:
            break
        improved = False
        for i in rng.permutation(n):
            v = f[i]
            Sr = S - v
            st = i
            en = min(i+n, 2*n-1)
            seg = en - st
            if seg <= 0:
                continue
            # influence: 2*f[k-i] for k in [i, i+n-1]
            infl_seg = 2.0 * f[:seg]
            # max outside segment
            if st > 0:
                m_out = conv[:st].max()
            else:
                m_out = -1
            if en < 2*n-1:
                m_out = max(m_out, conv[en:].max())
            
            # 1D search
            x_max = max(5*v, S/n*3) if v > 1e-12 else S/n*5
            x_max = min(x_max, 20*S/n)
            m_pts = 200
            xg = np.linspace(0, x_max, m_pts)
            d = xg - v
            nS = Sr + xg
            # new conv in segment
            nc = conv[st:en][None,:] + d[:,None]*infl_seg[None,:]
            if 2*i >= st and 2*i < en:
                nc[:, 2*i-st] += d*d
            # max of new conv
            nc_max = np.maximum(nc.max(axis=1), m_out)
            obj = 2*n*nc_max/np.maximum(nS*nS, 1e-30)
            obj[nS < 0.01] = math.inf
            
            idx = int(np.argmin(obj))
            if obj[idx] < best_sc - 1e-14:
                f[i] = xg[idx]
                dd = xg[idx] - v
                S += dd
                conv[st:en] += dd*infl_seg
                if 2*i >= st and 2*i < en:
                    conv[2*i] += dd*dd
                best_sc = obj[idx]
                improved = True
        
        if not improved:
            break
    
    return f, best_sc
```

```python
            seg_len = en - st
            if seg_len <= 0:
                continue
            
            infl_seg = 2.0 * f[:seg_len]
            
            # max outside
            m_out = 0.0
            if st > 0:
                m_out = max(m_out, float(conv[:st].max()))
            if en < 2*n-1:
                m_out = max(m_out, float(conv[en:2*n-1].max()))
            
            # Search
            x_max = max(4*v, 3*S/n) if v > 1e-12 else 6*S/n
            x_max = min(x_max, 50*S/n, 1000)
            m_pts = 256
            xg = np.linspace(0, x_max, m_pts)
            d = xg - v
            nS = Sr + xg
            nc = conv[st:en][None,:] + d[:,None]*infl_seg[None,:]
            if 2*i >= st and 2*i < en:
                nc[:, 2*i-st] += d*d
            nc_max = np.maximum(nc.max(axis=1), m_out)
            obj = 2.0*n*np.maximum(nc_max, 0) / np.maximum(nS*nS, 1e-30)
            obj[nS < 0.01] = math.inf
            
            idx = int(np.argmin(obj))
            if obj[idx] < best_sc - 1e-14 and xg[idx] >= 0:
                f[i] = xg[idx]
                dd = xg[idx] - v
                S += dd
                conv[st:en] += dd*infl_seg
                if 2*i >= st and 2*i < en:
                    conv[2*i] += dd*dd
                best_sc = float(obj[idx])
                improved = True
        
        if not improved:
            break
    
    # Final recompute
    sc = _score(f)
    return f, sc

def _pair_moves(f, stop_t, max_rounds=50, seed=0):
    n = len(f)
    f = f.copy()
    S = float(f.sum())
    best_sc = _score(f)
    rng = np.random.default_rng(seed)
    mults = np.array([0.0, 0.1, 0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5, 4.0])
    
    for rnd in range(max_rounds):
        if time.monotonic() > stop_t:
            break
        improved = False
        n_try = min(60, max(10, 3600 // n))
        for _ in range(n_try):
            if time.monotonic() > stop_t:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            vi, vj = f[i], f[j]
            S_rest = S - vi - vj
            best_val = best_sc
            best_i, best_j = vi, vj
            for mi in mults:
                ni = vi*mi if vi > 1e-12 else (0.02*S/n if mi > 0 else 0.0)
                for mj in mults:
                    nj = vj*mj if vj > 1e-12 else (0.02*S/n if mj > 0 else 0.0)
                    if S_rest + ni + nj < 0.01:
                        continue
                    f[i], f[j] = ni, nj
                    sc = _score(f)
                    if sc < best_val:
                        best_val = sc
                        best_i, best_j = ni, nj
            f[i], f[j] = best_i, best_j
            S = float(f.sum())
            if best_val < best_sc - 1e-14:
                best_sc = best_val
                improved = True
        if not improved:
            break
    return f, best_sc
```
