def _get_parent():

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / max(a.size, 1)
    return a / s


def _conv_full(a):
    n = a.size
    if n == 0:
        return np.array([0.0])
    nfft = next_fast_len(2 * n - 1)
    fa = rfft(a, nfft)
    c = irfft(fa * fa, nfft)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _grad_softmax(a, w):
    """grad[i] = 2 * sum_j w[i+j] * a[j] for j=0..n-1"""
    n = a.size
    # Build the (n, n) matrix W where W[i,j] = w[i+j]
    # cross[i] = dot(W[i,:], a)
    # Use FFT for efficiency when n is large
    if n > 256:
        L = next_fast_len(3 * n - 2)
        w_ext = np.zeros(L)
        w_ext[:2 * n - 1] = w
        a_ext = np.zeros(L)
        a_ext[:n] = a
        # cross[i] = sum_{j=0}^{n-1} w[i+j]*a[j]
        # This is the cross-correlation of (w, a) at lags 0..n-1
        # = real part of ifft(fft(w) * conj(fft(a))) shifted
        W_m = rfft(w_ext, L)
        A_m = rfft(a_ext, L)
        corr = irfft(W_m * np.conj(A_m), L)
        # corr[k] = sum_j w_ext[j+k] * a_ext[j] ... need to check indexing
        # Actually: cross-correlation (w P v)[m] = sum_n w[n+m] v[n]
        # With FFT: ifft(FFT(w) * conj(FFT(v)))[m] gives sum_n w[n+m] conj(v[n])
        # For real: sum_n w[n+m] v[n]
        # We want cross[i] = sum_{j=0}^{n-1} w[i+j]*a[j] for i=0..n-1
        # So cross[i] = (w P a)[i] where P is the "pre-correlation"
        # corr[m] = sum_n w[n+m]*a[n] for m = 0, 1, ...
        cross = corr[:n]
    else:
        cross = np.array([np.dot(w[i:i + n], a) for i in range(n)])
    cross = np.where(np.isfinite(cross), cross, 0.0)
    return 2.0 * cross


def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.array([target])
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    with np.errstate(divide="ignore", invalid="ignore"):
        cond = (css - target) / idx < u
    nz = np.nonzero(cond)[0]
    if nz.size == 0:
        return np.full(n, target / n, dtype=np.float64)
    rho = nz[-1]
    theta = (css[rho] - target) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s <= 1e-300:
        return np.full(n, target / n, dtype=np.float64)
    return w / s * target


def _exact_score(a):

        if a.size == 0 or not np.all(np.isfinite(a)):
            return float("inf")
        seq = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(seq))
    except Exception:
        return float("inf")


def _build_conv_matrix(n):
    """Build the (2n-1, n) convolution matrix."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(k, n - 1)
        for j in range(jlo, jhi + 1):
            A[k, j] = 1.0
    return A


def _solve_lp(a, M, deadline, rng):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0."""
    n = a.size
    if n < 2 or n > 600 or M <= 0 or not np.isfinite(M):
        return None
    if time.monotonic() > deadline - 2.0:
        return None
    # Build constraint matrix: (a*b)[k] = sum_i a[|k-i| offset]... 
    # Actually (a*b)[k] = sum_j a[j]*b[k-j], so for fixed a, the matrix row k has:
    # A[k,j] = a[k-j] if 0<=k-j<n, else 0
    m = 2 * n - 1
    A_ub = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        jlo = max(0, k - n + 1)
        jhi = min(k, n - 1)
        for j in range(jlo, jhi + 1):
            A_ub[k, j] = a[k - j]
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(m, M, dtype=np.float64)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'time_limit': 5.0, 'presolve': True})
        if res is not None and getattr(res, 'success', False):
            x = np.asarray(res.x, dtype=np.float64)
            x = np.maximum(x, 0.0)
            s = x.sum()
            if s > 1e-12:
                return x / s
    except Exception:
        pass
    return None


def _soft_max_grad(a, c, M, T):
    """Compute softmax-weighted gradient of smooth max surrogate."""
    n = a.size
    z = (c - M) / T
    z = np.clip(z, -60.0, 0.0)
    w = np.exp(z)
    ws = float(w.sum())
    if ws < 1e-300 or not np.isfinite(ws):
        w = np.zeros(c.size)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= ws
    grad = _grad_softmax(a, w)
    return grad, w

                time.monotonic() + main_time,
                rng,
                use_lp=True,
                lp_interval=50
            )
            if np.isfinite(rs) and rs < best_score:
                best_a = _normalize(ra)
                best_score = float(rs)
        except Exception:
            pass

    # Phase 3: Second optimization run with different parameters
    rem = deadline - time.monotonic() - 10.0
    if rem > 30:
        try:
            # Slightly perturb best and re-optimize
            pert = best_a + rng.normal(0, 0.005, n)
            init_a2 = _normalize(pert)
            ra2, rs2 = _optimize_core(
                init_a2,
                time.monotonic() + min(rem * 0.5, 200.0),
                rng,
                use_lp=True,
                lp_interval=30
            )
            if np.isfinite(rs2) and rs2 < best_score:
                best_a = _normalize(ra2)
                best_score = float(rs2)
        except Exception:
            pass

    # Phase 4: Final polish - very fine-tuned optimization
    rem = deadline - time.monotonic() - 8.0
    if rem > 20:
        try:
            ra3, rs3 = _optimize_core(
                _normalize(best_a),
                time.monotonic() + min(rem * 0.8, 300.0),
                rng,
                use_lp=False,
                lp_interval=9999
            )
            if np.isfinite(rs3) and rs3 < best_score:
                best_a = _normalize(ra3)
                best_score = float(rs3)
        except Exception:
            pass

    # Final validation
    try:
        best_a = np.asarray(best_a, dtype=np.float64).ravel()
        best_a = np.maximum(best_a, 0.0)
        s = float(best_a.sum())
        if s < 0.01:
            best_a = np.ones(n, dtype=np.float64) / n
        final_sc = _exact_score(best_a)
        if np.isfinite(final_sc):
            best_score = min(best_score, final_sc)
        else:
            best_a = np.ones(n, dtype=np.float64) / n
            best_score = _exact_score(best_a)
    except Exception:
        best_a = np.ones(n, dtype=np.float64) / n
        best_score = _exact_score(best_a)

    if not np.isfinite(best_score):
        return [1.0]

    out = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in out]
