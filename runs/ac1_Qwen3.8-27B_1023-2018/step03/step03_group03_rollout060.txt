score(f) = 2n * max_k(c_k) / S^2 where c = f*f, S = sum(f)

c_after[2i] = c[2i] + 2vw + w^2

This is a 1D optimization problem in w. The function is:
g(w) = 2n * max_k(c[k] + delta_k(w)) / (S+w)^2

- delta_k(w) = 2w*f[k-i] for k ≠ 2i
- delta_2i(w) = 2vw + w^2

The max of linear functions (in w) is convex, and dividing by (S+w)^2 (which is decreasing in w for w > -S) makes g(w) non-monotone.

The optimal w can be found by:
1. Identifying which k achieves the max at the optimum (active constraint)
2. For each candidate active set, solving the 1D subproblem

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except:
        p = np.ones(198)
    if p.size == 0: p = np.ones(198)
    p = np.clip(p, 0, 1000)
    p = np.where(np.isfinite(p), p, 0.0)
    return p

def _conv_sq(a):
    """Autocorrelation via FFT"""
    n = len(a)
    sz = next_fast_len(2*n-1)
    F = rfft(a, sz)
    c = irfft(F*F, sz)[:2*n-1]
    return c

def _score(a, conv=None):
    n = len(a)
    S = float(a.sum())
    if S < 0.01: return 1e18
    if conv is None: conv = _conv_sq(a)
    return 2.0*n*float(conv.max())/(S*S)

def _build_lp_matrix(f, n):
    """Build the constraint matrix A for the LP: (f*g)[k] <= M"""
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _solve_lp(f, n, M, time_limit=8.0):
    """Solve: max sum(g) s.t. (f*g) <= M, g >= 0"""
    A = _build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0,None)]*n,
                      method='highs', options={'presolve':True, 'time_limit':time_limit})
    except:
        return None
    if res.status != 0 or res.x is None: return None
    g = res.x.copy()
    gs = g.sum()
    if gs < 1e-12: return None
    return g/gs

def _lp_step(f, n, stop_t, max_iters=200, seed=42):
    """Main LP-based optimization loop"""
    rng = np.random.default_rng(seed)
    f = np.clip(f.copy(), 0, None)
    S = f.sum()
    if S < 1e-10: f = np.ones(n)/n; S = 1.0
    f = f/S  # normalize to sum=1
    
    best_f = f.copy()
    best_sc = _score(f)
    t_val = 0.3
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > stop_t: break
        
        conv = _conv_sq(f)
        M = float(conv.max())
        
        # Solve LP
        g = _solve_lp(f, n, M, time_limit=min(10.0, max(2.0, stop_t - time.monotonic() - 1.0)))
        
        if g is None:
            # Perturb and retry
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            S = f.sum()
            if S > 1e-10: f /= S
            else: f = best_f.copy()
            stall += 3
            if stall > 50: break
            continue
        
        dist = np.linalg.norm(g - f)
        if dist < 1e-8:
            stall += 1
            if stall > 80: break
            # Try perturbation
            f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
            f /= f.sum()
            stall = 0
            continue
        
        # Line search for step size
        improved = False
        for attempt in range(15):
            t_new = np.clip(t_val, 0.01, 0.8)
            f_try = (1 - t_new)*f + t_new*g
            f_try = np.clip(f_try, 0, None)
            St = f_try.sum()
            if St < 0.01:
                t_val *= 0.5
                continue
            sc = _score(f_try)
            if sc < best_sc - 1e-14:
                f = f_try
                best_f = f.copy()
                best_sc = sc
                improved = True
                t_val = min(0.7, t_val*1.1)
                stall = 0
                break
            else:
                t_val *= 0.5
        
        if not improved:
            # Small step anyway to make progress
            t_new = max(0.01, t_val*0.3)
            f_try = (1-t_new)*f + t_new*g
            f_try = np.clip(f_try, 0, None)
            f_try /= f_try.sum()
            sc = _score(f_try)
            if sc < best_sc:
                f = f_try
                best_f = f.copy()
                best_sc = sc
                stall = 0
            else:
                stall += 1
                if stall > 100: break
                # Perturb
                f = np.clip(f + rng.normal(0, 0.1, n), 0, None)
                f /= f.sum()
                stall = 0
    
    return best_f, best_sc

def _coord_descent(f, stop_t, max_sweeps=100, seed=0):
    """Coordinate descent with golden-section search per coordinate"""
    n = len(f)
    f = np.clip(f.copy(), 0, None)
    S = float(f.sum())
    if S < 0.01: return f, 1e18
    conv = _conv_sq(f)
    best_sc = 2.0*n*float(conv.max())/(S*S)
    rng = np.random.default_rng(seed)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > stop_t: break
        any_improve = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_t: break
            
            v = f[i]
            S_rest = S - v
            
            # Precompute influence: for k where f[k-i] is valid
            # c_after[k] = c[k] + 2*(x-v)*f[k-i] for k != 2i
            # c_after[2i] += 2*v*(x-v) + (x-v)^2
            
            # Determine valid range for k: need 0 <= k-i < n, so k in [i, i+n-1], intersect [0, 2n-2]
            k_lo = max(0, i)
            k_hi = min(2*n-2, i+n-1)
            
            # influence[k] = 2*f[k-i] for k in [k_lo, k_hi]
            # (for k != 2i)
            influence = 2.0 * f[k_lo-i : k_hi-i+1]  # length k_hi-k_lo+1
            # Special index for 2i within this range:
            two_i = 2*i
            if k_lo <= two_i <= k_hi:
                offset_2i = two_i - k_lo
            else:
                offset_2i = -1
            
            # Golden-section search for x (new value of f[i])
            # Objective: g(x) = 2n * max(c_after) / (S_rest + x)^2
            # Range: x in [0, x_max]
            x_max = max(3.0*v, 0.5) if v > 1e-8 else 1.0
            x_max = min(x_max, 100.0)
            
            def obj(x):
                S_new = S_rest + x
                if S_new < 0.01: return 1e18
                w = x - v
                if abs(w) < 1e-14:
                    m = float(conv.max())
                else:
                    # Compute new max
                    # c_after[k] = conv[k] + w * influence[k-k_lo] for k in [k_lo, k_hi]
                    # plus special term at 2i
                    new_c = conv.copy()
                    new_c[k_lo:k_hi+1] += w * influence
                    if offset_2i >= 0:
                        new_c[two_i] += w*w
                    m = float(new_c.max())
