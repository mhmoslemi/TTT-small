    if len(active) == 0:

    a_new = np.clip(a - sg, 0, None)
    a_new /= max(a_new.sum(), 1e-10)
    
    return a_new, M

from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(np.max(b))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_conv_mat(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp(f, M, time_limit=8.0):
    n = len(f)
    A = _build_conv_mat(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(np.sum(g))
    if gs < 1e-12:
        return None
    return g / gs
