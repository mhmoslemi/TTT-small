        best_trial = None
        best_trial_M = M
        for factor in [0.3, 0.7, 1.0, 1.5, 2.5]:
            e = eta * factor
            a_trial = _eg_step(a, g_eff, e)
            c_trial = _conv_full(a_trial)
            M_trial = float(c_trial.max())
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_trial = a_trial

        if best_trial is not None:
            a = best_trial
            c = _conv_full(a)
            M = float(c.max())
            if M < best_M * (1 - 1e-10):
                best_M = M
                best_a = a.copy()
                patience = 0
                eta = min(eta * 1.02, 0.2)
            else:
                patience += 1
                eta = max(eta * 0.97, 0.005)
        else:
            patience += 1
            eta = max(eta * 0.95, 0.003)

        # Decay temperature
        temp_frac = max(0.005, temp_frac * 0.995)

        # Restart if stuck
        if patience > 60:
            r = float(rng.random())
            if r < 0.4:
                a = a + rng.normal(0, 0.02, n)
                a = _project_simplex(a, 1.0)
            elif r < 0.7:
                a = 0.7 * best_a + 0.3 * _random_init(n, rng, int(rng.integers(0, 5)))
                a = _project_simplex(a, 1.0)
            else:
                a = _random_init(n, rng, int(rng.integers(0, 5)))
            c = _conv_full(a)
            M = float(c.max())
            v = np.zeros(n)
            patience = 0
            eta = 0.05
            temp_frac = 0.03

    return best_a, best_M


def _polish(a, deadline, rng):
    """Fine polishing with small steps."""
    a = _normalize(a)
    n = a.size
    c = _conv_full(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M

    for it in range(2000):
        if time.monotonic() >= deadline:
            break
        w = _soft_weights(c, 0.005)
        g = _compute_grad_fft(a, w)
        best_trial = None
        best_trial_M = M
        for e in [0.005, 0.01, 0.02, 0.04]:
            a_trial = _eg_step(a, g, e)
            c_trial = _conv_full(a_trial)
            M_trial = float(c_trial.max())
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_trial = a_trial
        if best_trial is not None:
            a = best_trial
            c = _conv_full(a)
            M = float(c.max())
            if M < best_M:
                best_M = M
                best_a = a.copy()
        else:
            break
    return best_a, best_M


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(1.0, float(budget_s))
    except Exception:
        budget = 1300.0

    if budget <= 2.0:
        try:
            p = _get_parent()
            sc = _exact_score(p)
            if np.isfinite(sc):
                return [float(x) for x in p]
        except Exception:
            pass
        return [1.0]

    headroom = max(15.0, 0.06 * budget)
    deadline = t0 + budget - headroom
    if deadline <= t0 + 5.0:
        deadline = t0 + max(5.0, budget * 0.9)

    rng = np.random.default_rng(int(seed) if np.isfinite(float(seed)) else 42)

    # Get parent
    parent = _get_parent()
    best_a = _normalize(parent)
    best_score = _exact_score(best_a)

    if not np.isfinite(best_score):
        best_a = np.ones(512, dtype=np.float64) / 512.0
        best_score = _exact_score(best_a)

    n = int(best_a.size)

    # Phase 1: Main optimization on parent length
    phase1_deadline = deadline - max(60.0, 0.15 * budget)
    if phase1_deadline > t0 + 30.0:
        a1, M1 = _optimize(best_a, phase1_deadline, rng, use_lp=True, lp_interval=4)
        sc1 = _exact_score(a1)
        if np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()

    # Phase 2: Try a different initialization
    rem = deadline - time.monotonic() - max(30.0, 0.1 * budget)
    if rem > 60.0:
        phase2_deadline = time.monotonic() + min(rem, 0.25 * budget)
        init2 = 0.6 * best_a + 0.4 * _random_init(n, rng, 3)
        init2 = _project_simplex(init2, 1.0)
        a2, M2 = _optimize(init2, phase2_deadline, rng, use_lp=True, lp_interval=5)
        sc2 = _exact_score(a2)
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()

    # Phase 3: Polish the best solution
    pol_deadline = deadline - 10.0
    if pol_deadline > time.monotonic() + 20.0:
        a3, M3 = _polish(best_a, pol_deadline, rng)
        sc3 = _exact_score(a3)
        if np.isfinite(sc3) and sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()

    # Final validation
    try:
        best_a = np.asarray(best_a, dtype=np.float64).ravel()
        if best_a.size == 0 or not np.all(np.isfinite(best_a)):
            raise ValueError
        best_a = np.clip(best_a, 0.0, 1000.0)
        best_a = _normalize(best_a)
        final_score = _exact_score(best_a)
        if np.isfinite(final_score):
            best_score = min(best_score, final_score)
        else:
            raise ValueError
    except Exception:
        # Fallback to parent
        try:
            p = _get_parent()
            p = _normalize(p)
            ps = _exact_score(p)
            if np.isfinite(ps):
                return [float(x) for x in p]
        except Exception:
            pass
        return [1.0]

    out = np.clip(best_a, 0.0, 1000.0)
    out = _normalize(out)
    if out.size == 0:
        return [1.0]
    return [float(x) for x in out]
