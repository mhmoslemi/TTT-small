import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(200, dtype=np.float64)
    if p.size == 0:
        p = np.ones(200, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(200, dtype=np.float64)
    return p

def _conv_full(a):
    n = len(a)
    if n == 1:
        return np.array([a[0]*a[0]])
    fsize = next_fast_len(2*n - 1)
    F = rfft(a, fsize)
    c = irFFT(F * F, fsize)[:2*n-1]
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_direction(f, b, n, timeout=8.0):
    """Solve: max sum(g) s.t. Ag <= b, g >= 0."""
    A = _build_A(f, n)
    c_obj = np.full(n, -1.0)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0.0, None)]*n,
                      method='highs',
                      options={'presolve': True, 'time_limit': timeout,
                               'mip_rel_gap': 0.0})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1000.0)
    if g.sum() < 1e-15:
        return None
    return g

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = v.copy()
    n = len(v)
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    # Water-filling
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def _subgrad_step(f, b, n, eta):
    """Take a subgradient step for minimizing max(b_k)."""
    max_val = float(b.max())
    eps = 1e-8 * max(1.0, abs(max_val))
    K_star = np.where(b >= max_val - eps)[0]
    if len(K_star) == 0:
        K_star = np.array([int(np.argmax(b))])
    
    # Average subgradient
    # For k in K*, grad_k[j] = 2*f[k-j] for valid k-j
    grad = np.zeros(n)
    count = 0
    for k in K_star[:20]:  # top 20
        for j in range(max(0, k-n+1), min(k+1, n)):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += 2.0 * f[idx]
        count += 1
    if count > 0:
        grad /= count
    
    # Step: f - eta * grad, project to simplex
    f_new = f - eta * grad
    f_new = _project_simplex(f_new)
    return f_new

                        best[i], best[j] = oi, oj

                    sc = _score(best)
                    if sc < best_sc - 1e-15:

                k = int(rng.integers(n))
                pert = rng.normal(0, 0.01)

                    best[k] = old
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                else:
                    best[k] = old
        
        if not improved:
            no_imp += 1
            if no_imp >= 8:
                break
        else:
            no_imp = 0
    
    best = best / best.sum() if best.sum() > 1e-10 else best
    return best, _score(best)
