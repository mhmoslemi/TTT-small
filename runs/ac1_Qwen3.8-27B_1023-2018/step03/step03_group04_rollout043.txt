    """Compute subgradient of max(conv) at f."""
    n = len(f)
    M = float(conv.max())
    if M < 1e-15:
        return None

    active = np.where(conv >= thresh)[0]

        hi = min(n - 1, k)

    theta = (cssv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)
