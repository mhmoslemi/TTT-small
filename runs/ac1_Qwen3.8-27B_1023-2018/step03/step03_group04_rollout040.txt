def _project_simplex(v):

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find largest m such that u[m-1] > (cssv[m-1] - 1) / m

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0 or p.sum() < 0.01:
        p = np.ones(320, dtype=np.float64)
    return p


def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _conv_cross(a, b):
    fsize = next_fast_len(len(a) + len(b) - 1)
    A = rfft(a, fsize)
    B = rfft(b, fsize)
    c = irfft(A * B, fsize)[:len(a) + len(b) - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    M = float(_conv_fft(a).max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M / (S * S)


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    d = u * np.arange(1, n + 1) + 1.0 - cssv
    rho = np.where(d > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    m = rho[-1] + 1
    theta = (cssv[m - 1] - 1.0) / m
    return np.maximum(v - theta, 0.0)


def _lp_direction(f, M):
    n = len(f)
    A = _build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True,
                               "primal_feasibility_tolerance": 1e-9,
                               "dual_feasibility_tolerance": 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s

    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    fs = f.sum()

    else:

    best_score = _score(f)

    max_stall = 200
    iteration = 0
    last_improve = 0

    while time.monotonic() < stop_time - 1.0 and iteration < 200000:

        M = float(conv.max())

        if g is None:

            fs = f.sum()
            f = f / fs if fs > 1e-10 else best_f.copy()
            iteration += 1
            continue

        if dist < 1e-7:
            stall += 3

            fs = f.sum()

            return 2.0 * n * float(val.max())

        # Golden section on [0, t_max]
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, t_max
        d = lo + (1.0 - gr) * (hi - lo)

                e = d
                fe = fd

                d = e
                fd = fe

                fe = _score_line(e)

def _gradient_polish(f, stop_time, seed=42):
    n = len(f)
    f = np.clip(np.asarray(f, dtype=np.float64).ravel(), 0, None).copy()
    fs = f.sum()
    if fs < 1e-10:
        return f, math.inf
    f = f / fs
    best_f = f.copy()
    best_sc = _score(f)
    rng = np.random.default_rng(seed)

    t_smooth = 10.0
    lr = 0.05
    no_imp = 0
    max_no_imp = 2000

    for it in range(500000):
        if time.monotonic() > stop_time - 1.0:
            break

        b = _conv_fft(f)
        tb = t_smooth * b
        tb -= tb.max()
        w = np.exp(tb)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= ws

        L = next_fast_len(3 * n)
        F = rfft(f, L)
        W = rfft(w, L)
        cross = irfft(np.conj(F) * W, L)[:n]
        grad = 2.0 * cross

        v = f - lr * grad
        f_new = _project_simplex(v)
        sc = _score(f_new)

        if sc < best_sc - 1e-12:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.2, lr * 1.01)
            if it % 500 == 0:
                t_smooth = min(200.0, t_smooth * 1.02)
        else:
            no_imp += 1
            lr = max(0.005, lr * 0.9)
            if no_imp > max_no_imp:
                break
            if no_imp % 100 == 0:
                f = _project_simplex(f + rng.normal(0, 0.005, n))

    return best_f, best_sc


def _coord_polish(f, stop_time, seed=123):
    n = len(f)
    best = np.clip(np.asarray(f, dtype=np.float64).ravel(), 0, None).copy()
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 5

    multipliers = [0.0, 0.2, 0.3, 0.5, 0.7, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0, 5.0]

    while time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in multipliers:
                new_val = orig * mlt if orig > 1e-12 else (0.01 if mlt > 0 else 0.0)
                best[i] = new_val
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-13:
                    best_val = new_val
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-13:
                best_sc = best_val_sc
                improved = True

        # 2-coordinate moves
        n_2coord = min(200, n)
        for _ in range(n_2coord):
            if time.monotonic() > stop_time - 1.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.0, 0.5, 1.5, 2.0):
                for mj in (0.0, 0.5, 1.5, 2.0):
                    ni = oi * mi if oi > 1e-12 else (0.01 if mi > 0 else 0.0)
                    nj = oj * mj if oj > 1e-12 else (0.01 if mj > 0 else 0.0)
                    best[i], best[j] = ni, nj
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= max_no_imp:

            _track(res)
        except Exception:
            pass

    # Phase 5: Final LP refinement if time allows
    if time.monotonic() < deadline - 10.0:
        try:
            stop = min(deadline - 8.0, time.monotonic() + 60.0)
            _run_lp(best_seq, stop, seed + 9999)
        except Exception:
            pass

    # Final coordinate polish
    if time.monotonic() < deadline - 5.0:
        try:
            res, sc = _coord_polish(best_seq, deadline - 4.0, seed + 333)
            _track(res)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
