    bm = float(b.max())
    shifted = b - bm
    logsum = bm + (1.0 / beta) * math.log(float(np.sum(np.exp(beta * shifted)) / (2 * N - 1)))
    score = 2.0 * N * logsum / (s * s)
    return score


def _smoothed_grad(a, beta=80.0):
    N = len(a)
    s = float(a.sum())
    b = _conv_full(a)
    bm = float(b.max())
    w = np.exp(beta * (b - bm))
    w /= w.sum()
    logsum = bm + (1.0 / beta) * math.log(float(w.sum()) / (2 * N - 1))
    
    # grad of logsum w.r.t. a_i = (1/beta) * sum_k w_k * db_k/d a_i
    grad_logsum = np.zeros(N)
    for k in range(2 * N - 1):
        if w[k] < 1e-15:
            continue
        lo = max(0, k - N + 1)
        hi = min(N - 1, k)
        for i in range(lo, hi + 1):
            idx = k - i
            if 0 <= idx < N:
                grad_logsum[i] += w[k] * a[idx]
    grad_logsum *= (1.0 / beta)
    
    score = 2.0 * N * logsum / (s * s)
    d = 2.0 * N * (grad_logsum * s - 2.0 * logsum * a) / (s ** 3)
    return score, d


def _golden_line_search(f, g, score0, stop_time, max_iter=40):
    """Golden section search for t in [0,1] minimizing score((1-t)*f + t*g)"""
    gr = (math.sqrt(5) - 1) / 2  # ~0.618
    a_lo, a_hi = 0.0, 1.0
    t1 = a_hi - gr * (a_hi - a_lo)
    t2 = a_lo + gr * (a_hi - a_lo)
    f1 = _score((1 - t1) * f + t1 * g)
    f2 = _score((1 - t2) * f + t2 * g)
    
    for _ in range(max_iter):
        if time.monotonic() > stop_time:
            break
        if f1 < f2:
            a_hi = t2
            t2 = t1
            f2 = f1
            t1 = a_hi - gr * (a_hi - a_lo)
            f1 = _score((1 - t1) * f + t1 * g)
        else:
            a_lo = t1
            t1 = t2
            f1 = f2
            t2 = a_lo + gr * (a_hi - a_lo)
            f2 = _score((1 - t2) * f + t2 * g)
    
    t_opt = (a_lo + a_hi) / 2
    best_t = t_opt
    best_sc = min(f1, f2)
    # Check endpoints and midpoint
    for t_check in [0.0, 0.25, 0.5, 0.75, 1.0, t_opt]:
        sc = _score((1 - t_check) * f + t_check * g)
        if sc < best_sc:
            best_sc = sc
            best_t = t_check
    return best_t, best_sc


def _lp_optimize(f0, n, stop_time, seed=42):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n)
        fs = float(f.sum())
    f = f / fs  # normalize to sum=1
    
    best_f = f.copy()
    best_score = _score(f)
    
    stall = 0
    max_stall = 300
    t_curr = 0.5
    
    while time.monotonic() < stop_time - 5.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-20:
            break
        
        # LP direction
        g = _lp_direction(f, n, M, timeout=6.0)
        if g is None:
            stall += 5
            if stall > max_stall:
                break
            # Perturb and retry
            noise = rng.normal(0, 0.02 * f.mean(), n)
            f = np.clip(f + noise, 0, None)
            fs2 = f.sum()
            if fs2 > 1e-10:
                f = f / fs2
            else:
                f = best_f.copy()
            continue
        
        g_sum = float(g.sum())
        g = g / g_sum  # normalize to sum=1
        
        # Check if we're at a fixed point
        dist = float(np.linalg.norm(g - f))
        if dist < 1e-10:
            stall += 3
            if stall > max_stall:
                break
            # Perturb to escape
            f = np.clip(f + rng.normal(0, 0.03 * f.mean(), n), 0, None)
            fs3 = f.sum()
            if fs3 > 1e-10:
                f = f / fs3
            stall = 0
            max_stall += 50
            continue
        
        # Line search
        t_opt, sc_try = _golden_line_search(f, g, _score(f), stop_time - 3.0, max_iter=25)
        
        if sc_try < best_score - 1e-14:
            f_new = (1 - t_opt) * f + t_opt * g
            f_new = np.clip(f_new, 0, None)
            fs_new = f_new.sum()
            if fs_new < 1e-10:
                stall += 1
                continue
            f_new = f_new / fs_new
            actual_sc = _score(f_new)
            if actual_sc < best_score - 1e-15:
                best_f = f_new.copy()
                best_score = actual_sc
                f = f_new.copy()
                stall = 0
                if actual_sc < sc_try:
                    t_curr = min(1.0, t_opt * 1.2)
            else:
                stall += 1
        else:
            stall += 1
            if stall >= 50:
                # Aggressive perturbation
                f = np.clip(f + rng.normal(0, 0.05 * f.mean(), n), 0, None)
                fs4 = f.sum()
                if fs4 > 1e-10:
                    f = f / fs4
                stall = 0
                max_stall += 100
        
        if stall > max_stall:
            break
    
    return best_f, best_score


def _nonlinear_polish(a, stop_time, seed=12345):
    """Use scipy minimize with smoothed objective for polishing"""
    n = len(a)
    x0 = np.clip(np.asarray(a, dtype=np.float64).copy(), 0, None)
    s0 = x0.sum()
    if s0 < 1e-10:
        return a, _score(a)
    x0 = x0 / s0
    
    best_x = x0.copy()
    best_sc = _score(x0)
    
    for beta in [100.0, 200.0, 500.0]:
        if time.monotonic() > stop_time - 5.0:
            break
        remaining = (stop_time - time.monotonic()) * 0.4
        try:
            res = minimize(
                _smoothed_obj, x0, args=(beta,),
                jac=_smoothed_grad,
                method='L-BFGS-B',
                bounds=[(0.0, None)] * n,
                options={'maxiter': max(50, int(remaining * 2)),
                         'maxfun': max(200, int(remaining * 5)),
                         'ftol': 1e-15, 'gtol': 1e-10}
            )
            if res.x is not None:
                x_new = np.clip(res.x, 0, None)
                s_new = x_new.sum()
                if s_new > 1e-10:
                    x_new = x_new / s_new
                    sc = _score(x_new)
                    if sc < best_sc - 1e-14:
                        best_x = x_new.copy()
                        best_sc = sc
                        x0 = x_new.copy()
        except Exception:
            pass
    
    return best_x, best_sc


def _coord_descent(a, stop_time, seed=999):
    """Aggressive coordinate and pair descent"""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).copy(), 0, None)
    s = best.sum()
    if s < 1e-10:
        return best, _score(best)
    best = best / s
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    
    while time.monotonic() < stop_time - 4.0 and no_imp < 30:
        improved = False
        
        # Single coordinate: try multiple multipliers
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 4.0:
                break
            orig = best[i]
            if orig < 1e-14:
                # Try setting to small positive
                for val in [0.001, 0.005, 0.01, 0.02 * best.mean()]:
                    best[i] = val
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
                        break
                else:
                    best[i] = 0
                continue
            best_val = orig
            best_val_sc = best_sc
            for mlt in [0.0, 0.05, 0.1, 0.2, 0.4, 0.6, 0.8, 1.05, 1.2, 1.5, 2.0, 3.0]:
                best[i] = orig * mlt
                sc = _score(best)
                if sc < best_val_sc - 1e-15:
                    best_val = best[i]
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-15:
                best_sc = best_val_sc
                improved = True
        
        # Pair coordinate descent
        for _ in range(200):
            if time.monotonic() > stop_time - 4.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            pair_best_si, pair_best_sj = oi, oj
            pair_best_sc = best_sc
            for mi in [0.0, 0.3, 0.5, 0.7, 1.0, 1.4, 2.0]:
                for mj in [0.0, 0.3, 0.5, 0.7, 1.0, 1.4, 2.0]:
                    if mi == 1.0 and mj == 1.0:
                        continue
                    vi = oi * mi if oi > 1e-15 else 0.0
                    vj = oj * mj if oj > 1e-15 else 0.0
                    best[i], best[j] = vi, vj
                    sc = _score(best)
                    if sc < pair_best_sc - 1e-15:
                        pair_best_si, pair_best_sj = vi, vj
                        pair_best_sc = sc
            best[i], best[j] = pair_best_si, pair_best_sj
            if pair_best_sc < best_sc - 1e-15:
                best_sc = pair_best_sc
                improved = True
        
        # Small random perturbations
        if not improved:
            for _ in range(50):
                if time.monotonic() > stop_time - 4.0:
                    break
                mask = rng.random(n) < 0.05
                k_indices = np.where(mask)[0]
                if len(k_indices) == 0:
                    continue
                pert = rng.normal(0, 0.03 * best.mean(), len(k_indices))
                old_vals = best[k_indices].copy()
                best[k_indices] = np.clip(best[k_indices] + pert, 0, None)
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                else:
                    best[k_indices] = old_vals
        
        if not improved:
            no_imp += 1
        else:
            no_imp = 0
    
    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 10.0  # 10s headroom
    
    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(416, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(416, dtype=np.float64)
    parent = np.clip(parent, 0, 1000)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    n_p = len(parent)
    n = n_p
    
    # Normalize parent
    p_sum = parent.sum()
    if p_sum < 1e-10:
        parent = np.ones(n)
        p_sum = float(parent.sum())
    f_init = parent / p_sum
    
    best_seq = f_init.copy()
    best_score = _score(f_init)
    
    # Verify with exact evaluator
    try:
        ex = float(evaluate_sequence([float(x) for x in best_seq]))
        if math.isfinite(ex) and ex < best_score:
            best_score = ex
    except Exception:
        pass
    
    rng_main = np.random.default_rng(seed)
    
    # Phase 1: LP optimization on parent (45% budget)
    phase1_end = t0 + budget * 0.45
    inits = [f_init.copy()]
    
    # Raised cosine
    i_arr = np.arange(n, dtype=np.float64)
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n)))
    
    # Multi-frequency
    sup = 0.3 * np.ones(n)
    for fr in [1, 2, 3, 5, 8, 13]:
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.15 * np.cos(2.0 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.02)
    
    # Parent + noise
    sig = 0.1 * f_init.mean()
    inits.append(np.abs(parent + rng_main.normal(0, sig, n)) + 0.01)
    
    # Random
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.05)
    
    # Gaussian bumps
    c1, c2 = n * 0.3, n * 0.7
    w = n / 5.0
    inits.append(np.exp(-0.5 * ((i_arr - c1) / w) ** 2) + np.exp(-0.5 * ((i_arr - c2) / w) ** 2))
    
    # Tapered
    taper = 1.0 - 0.3 * (2.0 * i_arr / (n - 1) - 1.0) ** 2
    inits.append(f_init * (1.0 + 0.5 * (taper - 1.0)))
    
    time_per_init = max(15.0, (phase1_end - t0) / len(inits))
    
    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 8.0:
            break
        s_init = init.sum()
        if s_init < 1e-10:
            continue
        init_n = init / s_init
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        try:
            res, sc = _lp_optimize(init_n, n, stop, seed + 7777 * (idx + 1))
            if res is not None and len(res) == n:
                res = np.clip(res, 0, None)
                rs = res.sum()
                if rs > 1e-10:
                    res = res / rs
                    exact = _score(res)
                    if exact < best_score - 1e-14:
                        best_seq = res.copy()
                        best_score = exact
        except Exception:
            pass
    
    # Phase 2: Second LP pass on best (15% budget)
    phase2_end = t0 + budget * 0.60
    if time.monotonic() < phase2_end - 20.0:
        stop2 = phase2_end - 5.0
        try:
            res2, sc2 = _lp_optimize(best_seq.copy(), n, stop2, seed + 99999)
            if res2 is not None and len(res2) == n:
                res2 = np.clip(res2, 0, None)
                rs2 = res2.sum()
                if rs2 > 1e-10:
                    res2 = res2 / rs2
                    exact2 = _score(res2)
                    if exact2 < best_score - 1e-14:
                        best_seq = res2.copy()
                        best_score = exact2
        except Exception:
            pass
    
    # Phase 3: Nonlinear polishing (15% budget)
    phase3_end = t0 + budget * 0.75
    if time.monotonic() < phase3_end - 10.0:
        try:
            res3, sc3 = _nonlinear_polish(best_seq, phase3_end - 3.0, seed=55555)
            if res3 is not None and len(res3) == n:
                res3 = np.clip(res3, 0, None)
                rs3 = res3.sum()
                if rs3 > 1e-10:
                    res3 = res3 / rs3
                    exact3 = _score(res3)
                    if exact3 < best_score - 1e-14:
                        best_seq = res3.copy()
                        best_score = exact3
        except Exception:
            pass
    
    # Phase 4: Coordinate descent polishing (20% budget)
    phase4_end = deadline - 8.0
    if time.monotonic() < phase4_end - 10.0:
        try:
            res4, sc4 = _coord_descent(best_seq, phase4_end, seed=77777)
            if res4 is not None and len(res4) == n:
                res4 = np.clip(res4, 0, None)
                rs4 = res4.sum()
                if rs4 > 1e-10:
                    res4 = res4 / rs4
                    exact4 = _score(res4)
                    if exact4 < best_score - 1e-14:
                        best_seq = res4.copy()
                        best_score = exact4
        except Exception:
            pass
    
    # Final: one more quick coord descent if time remains
    if time.monotonic() < deadline - 6.0:
        try:
            res5, sc5 = _coord_descent(best_seq, deadline - 4.0, seed=31337)
            if res5 is not None and len(res5) == n:
                res5 = np.clip(res5, 0, None)
                rs5 = res5.sum()
                if rs5 > 1e-10:
                    res5 = res5 / rs5
                    exact5 = _score(res5)
                    if exact5 < best_score - 1e-14:
                        best_seq = res5.copy()
                        best_score = exact5
        except Exception:
            pass
    
    # Final validation with exact evaluator
    final_list = [float(x) for x in best_seq]
    try:
        final_sc = float(evaluate_sequence(final_list))
    except Exception:
        final_sc = math.inf
    
    if not math.isfinite(final_sc):
        # Fallback to parent
        fallback_list = [float(x) for x in parent]
        try:
            fs = float(evaluate_sequence(fallback_list))
            if math.isfinite(fs):
                return fallback_list
        except Exception:
            pass
        return [1.0]
    
    best_score = final_sc
    return final_list
