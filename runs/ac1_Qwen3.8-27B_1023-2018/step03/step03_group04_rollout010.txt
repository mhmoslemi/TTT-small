```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0 or p.sum() < 0.01:
        p = np.ones(320, dtype=np.float64)
    return p

def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _exact_score(a):
    n = len(a)
    c = np.convolve(a, a)
    M = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    return 2.0 * n * M / (S * S)

def _precompute_structure(n):
    """Pre-compute the index structure for the convolution matrix."""
    rows = []
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(k, n - 1)
        rows.append((jlo, jhi))
    return rows

def _build_A(f, structure):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k, (jlo, jhi) in enumerate(structure):
        if jlo <= jhi:
            A[k, jlo:jhi+1] = f[k - np.arange(jlo, jhi+1)]
    return A

def _lp_direction(f, M, structure):
    n = len(f)
    if M < 1e-15:
        return None
    A = _build_A(f, structure)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0, None)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s

def _lp_optimize(f0, stop_time, seed, structure=None):
    """Main LP-based optimization."""
    rng = np.random.default_rng(seed)
    n = len(f0)
    if structure is None:
        structure = _precompute_structure(n)
    
    f = np.clip(f0.astype(np.float64).copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs
    
    best_f = f.copy()
    best_M = float(_conv_fft(f).max())
    best_score = 2.0 * n * best_M  # sum is 1
    
    t = 0.4
    stall = 0
    max_stall = 200
    iteration = 0
    
    while time.monotonic() < stop_time and iteration < 100000:
        conv = _conv_fft(f)
        M = float(conv.max())
        score = 2.0 * n * M
        
        if score < best_score - 1e-10:
            best_f = f.copy()
            best_score = score
            best_M = M
            stall = 0
        
        g = _lp_direction(f, M, structure)
        if g is None:
            stall += 5
            if stall > max_stall:
                f = np.clip(best_f + rng.normal(0, 0.05, n), 0, None)
                f /= max(f.sum(), 1e-10)
                t = 0.4
                stall = 0
            continue
        
        dist = float(np.linalg.norm(g - f, ord=np.inf))
        if dist < 1e-10:
            stall += 3
            if stall > max_stall:
                f = np.clip(best_f + rng.normal(0, 0.05, n), 0, None)
                f /= max(f.sum(), 1e-10)
                t = 0.4
                stall = 0
            continue
        
        # Line search
        best_t = t
        best_sc = score
        candidate_f = None
        
        for tc in [t, t*0.5, t*1.5, t*0.25, min(1.0, t*2.0)]:
            tc = max(0.01, min(1.0, tc))
            f_new = (1.0 - tc) * f + tc * g
            f_new = np.clip(f_new, 0, None)
            fs_new = f_new.sum()
            if fs_new < 1e-10:
                continue
            f_new = f_new / fs_new
            sc = 2.0 * n * float(_conv_fft(f_new).max())
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_t = tc
                candidate_f = f_new
        
        if candidate_f is not None:
            f = candidate_f
            if best_sc < best_score - 1e-12:
                best_f = f.copy()
                best_score = best_sc
                t = min(1.0, t * 1.05)
            else:
                t = max(0.05, t * 0.8)
        else:
            stall += 1
            t = max(0.05, t * 0.6)
            if stall >= max_stall:
                f = np.clip(best_f + rng.normal(0, 0.08, n), 0, None)
                f /= max(f.sum(), 1e-10)
                t = 0.4
                stall = 0
        
        iteration += 1
    
    return best_f, 2.0 * n * float(_conv_fft(best_f).max())

def _subgradient_refine(f0, stop_time, seed):
    """Subgradient method for refinement."""
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = np.clip(f0.astype(np.float64).copy(), 0, None)
    f /= max(f.sum(), 1e-10)
    
    best_f = f.copy()
    best_M = float(_conv_fft(f).max())
    
    lr = 0.02
    for it in range(5000):
        if time.monotonic() > stop_time:
            break
        conv = _conv_fft(f)
        M = float(conv.max())
        if 2*n*M < 2*n*best_M - 1e-10:
            best_f = f.copy()
            best_M = M
        
        # Active set
        thresh = 0.9 * M
        active = np.where(conv >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        
        # Subgradient
        sg = np.zeros(n)
        for k in active:
            jlo = max(0, k - n + 1)
            jhi = min(k, n - 1)
            js = np.arange(jlo, jhi+1)
            sg[js] += 2.0 * f[k - js]
        sg /= len(active)
        
        # Step
        f_new = f - lr * sg
        f_new = np.clip(f_new, 0, None)
        fs_new = f_new.sum()
        if fs_new < 1e-10:
            break
        f_new /= fs_new
        
        new_M = float(_conv_fft(f_new).max())
        if new_M < M - 1e-12:
            f = f_new
            lr = min(0.1, lr * 1.02)
        else:
            lr = max(0.001, lr * 0.7)
    
    return best_f

def _polish(f0, stop_time):
    """Coordinate descent polish."""
    n = len(f0)
    f = np.clip(f0.astype(np.float64).copy(), 0, None)
    f /= max(f.sum(), 1e-10)
    best_sc = 2.0 * n * float(_conv_fft(f).max())
    rng = np.random.default_rng(42)
    
    multipliers = [0.0, 0.1, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0]
    no_imp = 0
    max_no_imp = 15
    
    while time.monotonic() < stop_time and no_imp < max_no_imp:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            best_val = orig
            best_val_sc = 2.0 * n * float(_conv_fft(f).max())
            for m in multipliers:
                f[i] = orig * m if orig > 1e-12 else (0.01 if m > 0 else 0.0)
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = 2.0 * n * float(_conv_fft(f).max())
                if sc < best_val_sc - 1e-13:
                    best_val = f[i]
                    best_val_sc = sc
            f[i] = best_val
            if best_val_sc < best_sc - 1e-13:
                best_sc = best_val_sc
                improved = True
        
        if not improved:
            no_imp += 1
        else:
