def _conv(a):
    n = len(a)

        return np.array([a[0]*a[0]])
    fsize = next_fast_len(2*n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    return c

def _score_normalized(a):
    """Score assuming sum(a) = 1. Returns 2*n*max(conv)."""
    n = len(a)

def _lp_direction(f, M, stall_rounds=0):
    """Find LP direction: max sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    # Build constraint matrix A: (2n-1) x n
    # A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    # This is a Toeplitz-like matrix
    A = np.zeros((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"time_limit": 10, "presolve": True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s

def _conv(a):

    return c


def _score_norm(a):
    """Score assuming sum(a)=1: returns 2*n*max(conv)."""
    n = len(a)

def _build_A(f):
    """Build the (2n-1) x n convolution constraint matrix for f."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _lp_direction(f, M, time_limit=8.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    if M < 1e-18:
        return None
    A = _build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"time_limit": time_limit, "presolve": True,
                               "dual_feasibility_tolerance": 1e-9,
                               "primal_feasibility_tolerance": 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-12:
        return None
    return g / s

    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
