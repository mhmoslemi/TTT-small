import math

from scipy.optimize import linprog

# --- Helper functions ---

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p.copy()

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    c[c < 0] = 0
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_conv_matrix(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_dir(f, M, tl=3.0):
    n = len(f)
    A = _build_conv_matrix(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': tl})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs

def _smooth_max_gradient(a, conv, tau=60.0):
    n = len(a)

        if time.monotonic() > lp_end:

        res, sc = _lp_optimize(init_arr, n, lp_end, sd, max_iters=200)
        if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
            res = np.clip(res, 0, 1000)
            _update(res, sc)
        if time.monotonic() < pol_end - 15.0:
            try:
                res2, sc2 = _coord_polish(res if res is not None else init_arr,
                                          pol_end, max_sweeps=25, seed=sd + 1)
                _update(res2, sc2)
            except Exception:
                pass

    # === Phase 1: LP + polish on parent length ===
    n = n_p
    i_arr = np.arange(n, dtype=np.float64)

    inits = [parent.copy()]
    # Cosine taper
    inits.append(np.clip(0.3 + 0.7 * np.cos(np.pi * i_arr / n) ** 2, 0, None))
    # Two-hump
    inits.append(np.clip(0.4 + 0.6 * np.cos(3 * np.pi * i_arr / n), 0, None))
    # Exponential taper
    inits.append(np.exp(-2.0 * i_arr / n) + 0.1)

    phase1_end = t0 + budget * 0.35
    time_per_init = (phase1_end - t0) / len(inits)

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 5.0:
            break
        if best_score < 1.503:
            break
        init = np.clip(init, 0, None)
        if init.sum() < 1e-6:
            init = np.ones(n)
        stop_lp = min(phase1_end - 3.0, time.monotonic() + time_per_init)
        stop_pol = min(hard_deadline - 30, stop_lp + budget * 0.12)
        try:
            _run_lp_polish(n, init, stop_lp, stop_pol, seed + 100 * (idx + 1))
        except Exception:
            pass

    # === Phase 2: Smooth gradient descent on best ===
    grad_end = t0 + budget * 0.50
    if time.monotonic() < grad_end - 20.0:
        try:
            res_g, sc_g = _smooth_grad_descent(best_seq, grad_end, max_iters=8000, seed=12345)
            _update(res_g, sc_g)
        except Exception:
            pass

    # === Phase 3: Extended coordinate polish ===
    pol_end = t0 + budget * 0.68
    if time.monotonic() < pol_end - 20.0:
        try:
            res_c, sc_c = _coord_polish(best_seq, pol_end, max_sweeps=40, seed=424242)
            _update(res_c, sc_c)
        except Exception:
            pass

    # === Phase 4: Try other lengths ===
    phase4_end = t0 + budget * 0.85
    other_lengths = [128, 192, 256, 384, 448, 512, 640]
    if n_p not in other_lengths:
        other_lengths.append(n_p)
    other_lengths = sorted(set(other_lengths), key=lambda L: abs(L - n_p))

    time_per_len = max(8.0, (phase4_end - time.monotonic()) / max(1, len(other_lengths)))

    for L in other_lengths:
        if time.monotonic() > phase4_end - 10.0:
            break
        if best_score < 1.5028:
            break
        if L == n_p:
            continue  # Already done in phase 1

        stop_lp = min(phase4_end - 5.0, time.monotonic() + time_per_len * 0.6)
        stop_pol = min(hard_deadline - 20, stop_lp + time_per_len * 0.7)

        # Interpolate from best or parent
        src = best_seq if best_score < 1.6 else parent
        init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(src)), src)
        init_l = np.clip(init_l, 0, None)

        try:
            _run_lp_polish(L, init_l, stop_lp, stop_pol, seed + 50000 + L)
        except Exception:
            pass

    # === Phase 5: Final intensive polish ===
    final_end = hard_deadline - 12.0
    if time.monotonic() < final_end - 15.0:
        try:
            res_f, sc_f = _coord_polish(best_seq, final_end, max_sweeps=50, seed=31337)
            _update(res_f, sc_f)
        except Exception:
            pass

    # One more smooth gradient pass if time allows
    if time.monotonic() < final_end - 10.0:
        try:
            res_g2, sc_g2 = _smooth_grad_descent(best_seq, final_end, max_iters=3000, seed=55555)
            _update(res_g2, sc_g2)
        except Exception:
            pass

    # Final coordinate polish
    if time.monotonic() < hard_deadline - 10.0:
        try:
            res_c2, sc_c2 = _coord_polish(best_seq, hard_deadline - 5.0, max_sweeps=15, seed=88888)
            _update(res_c2, sc_c2)
        except Exception:
            pass

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0, 1.0, 1.0, 1.0]
        return [float(x) for x in fallback]

    if final_sc < best_score:
        best_score = final_sc

    return [float(x) for x in best_seq]
