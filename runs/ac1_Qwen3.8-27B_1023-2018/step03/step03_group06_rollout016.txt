# FFT utilities
_fft_cache = {}

def _fft_len(m):
    m = int(m)
    if m <= 0:
        return 1
    if m not in _fft_cache:
        _fft_cache[m] = int(next_fast_len(m))
    return _fft_cache[m]

def _conv(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n1, n2 = len(a), len(b)
    if n1 == 0 or n2 == 0:
        return np.array([], dtype=np.float64)
    m = n1 + n2 - 1
    L = _fft_len(m)
    c = irfft(rfft(a, L) * rfft(b, L), L)[:m]
    return np.maximum(c, 0.0)

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / max(1, len(a))
    return a / s

def _score(a):
    """Internal score (normalized to sum=1)."""
    a = _normalize(a)
    n = len(a)
    b = _conv(a)
    return 2.0 * n * b.max()

    """Use the evaluator for exact scoring."""
    try:
        fn = globals().get("evaluate_sequence")
        if fn is None:
            return math.inf
        return float(fn(lst))

def _lp_direction(a, time_limit=5.0):
    """Find g minimizing max(conv(a,g)) s.t. g>=0, sum(g)=1."""
    n = len(a)
    C = _conv_matrix(a)  # (2n-1) x n
    
    # LP: min M' s.t. C*g - M' <= 0, sum(g) = 1, g >= 0
    # Variables: g[0..n-1], M'
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M'
    
    A_ub = np.hstack([C, -np.ones((2*n-1, 1))])  # (2n-1) x (n+1)
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(0.0, None)]  # g >= 0, M' >= 0
    
    t_start = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except:
        return None
    
    if res is None or res.status != 0 or res.x is None:
        return None
    
    g = res.x[:n]
    g = np.clip(g, 0.0, None)
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s
