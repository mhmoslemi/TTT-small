    L = 2 * n - 1

    g = 2.0 * cr[idx]

def _conv(a):
    n = a.size
    nf = next_fast_len(2 * n - 1)
    f = rfft(a, nf)
    c = irfft(f * f, nf)[:2 * n - 1].copy()
    np.maximum(c, 0.0, out=c)
    return c


def _bld_A(f, n):
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    fr = f[::-1]
    for k in range(n):
        A[k, :k+1] = fr[:k+1]
    for k in range(n, m):
        A[k, k-n+1:] = fr[:m-k]
    return A


def _grad(a, w):

    if sw < 1e-15:
        return np.zeros(n)

    cr = irfft(fa * fw, nf)[:3*n-2]

        b = np.zeros(m)
