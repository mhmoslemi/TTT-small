import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

_COARSE = np.array(
    [0.0, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0, 5.0],
    dtype=np.float64,
)
_FINE = np.array([1e-5, 1e-4, 1e-3, 5e-3, 2e-2, 1e-1], dtype=np.float64)
_LOGR = 10.0 ** np.linspace(-2.0, 1.0, 10)


def _clean_array(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    return a


def _normalize(a):
    a = _clean_array(a)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    s = float(a.sum())
    if s <= 1e-12:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    a = a / s
    return a


def _get_parent():
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size > 0 and float(p.sum()) > 1e-12:
                return p
    except Exception:
        pass
    return np.ones(384, dtype=np.float64)


def _conv(a, b=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if b is None:
        b = a
        m = n
    else:
        b = np.asarray(b, dtype=np.float64)
        m = b.size
    L = next_fast_len(n + m - 1)
    fa = rfft(a, L)
    if b is a:
        c = irfft(fa * fa, L)[: n + m - 1]
    else:
        fb = rfft(b, L)
        c = irfft(fa * fb, L)[: n + m - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_c(c, S, n):
    if S <= 1e-15 or n <= 0:
        return math.inf
    m = float(np.max(c)) if c.size else 0.0
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * n * m / (S * S))


def _score(a):
    a = _clean_array(a)
    if a.size == 0:
        return math.inf
    S = float(a.sum())
    if S <= 1e-15:
        return math.inf
    c = _conv(a)
    return _score_c(c, S, a.size)


def _exact_score(a):
    a = _clean_array(a)
    if a.size == 0:
        return math.inf
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    ev = globals().get("evaluate_sequence", None)
    if ev is None:
        return _score(a)
    try:
        vals = [float(x) for x in a]
        sc = float(ev(vals))
        if math.isfinite(sc) and sc >= 0.0:
            return sc
        return math.inf
    except Exception:
        return _score(a)


def _resample(arr, L):
    arr = _clean_array(arr)
    L = int(L)
    if L <= 0:
        L = 1
    if arr.size == L:
        out = arr.copy()
        s = float(out.sum())
        if s <= 1e-12:
            out = np.ones(L, dtype=np.float64)
        else:
            out = out / s
        return _clean_array(out)
    if arr.size == 0:
        return np.ones(L, dtype=np.float64) / float(L)
    if arr.size == 1:
        return np.full(L, float(arr[0]), dtype=np.float64)
    xp = np.linspace(0.0, 1.0, arr.size)
    x = np.linspace(0.0, 1.0, L)
    out = np.interp(x, xp, arr)
    out = np.clip(out, 0.0, None)
    s = float(out.sum())
    if s <= 1e-12:
        out = np.ones(L, dtype=np.float64) / float(L)
    else:
        out = out / s
    return _clean_array(out)


def _perturb(a, sigma, rng):
    a = _normalize(a)
    n = a.size
    klen = int(max(3, min(31, max(3, n // 16))))
    if klen % 2 == 0:
        klen += 1
    r = np.arange(-klen // 2, klen // 2 + 1, dtype=np.float64)
    sig = max(1.0, float(klen) / 3.0)
    kern = np.exp(-0.5 * (r / sig) ** 2)
    kern /= kern.sum()
    noise = np.convolve(rng.normal(0.0, 1.0, n), kern, mode="same")
    a2 = a * (1.0 + sigma * noise)
    a2 += 1e-7 * rng.random(n)
    a2 = np.clip(a2, 0.0, None)
    s = float(a2.sum())
    if s <= 1e-12:
        return a.copy()
    return a2 / s


def _lp_move(p, c, n, stop_time, K=80, tl=2.0):
    now = time.monotonic()
    if now >= stop_time - 1.0:
        return None
    p = _normalize(p)
    c = _conv(p)
    L = float(c.max())
    if not math.isfinite(L) or L <= 0.0:
        return None
    K = int(min(max(10, K), max(10, 2 * n - 1)))
    if K >= 2 * n - 1:
        idx = np.arange(2 * n - 1)
    else:
        idx = np.argpartition(c, -K)[-K:]
    A = np.zeros((K, n), dtype=np.float64)
    for row, k in enumerate(idx):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if hi >= lo:
            js = np.arange(lo, hi + 1, dtype=int)
            A[row, js] = p[k - js]
    tl = min(2.0, max(0.5, float(stop_time) - time.monotonic()))
    try:
        res = linprog(
            -np.ones(n, dtype=np.float64),
            A_ub=A,
            b_ub=np.full(K, float(L), dtype=np.float64),
            bounds=(0.0, 10.0),
            method="highs",
            options={"presolve": True, "time_limit": tl},
        )
    except Exception:
        return None
    if res is None or getattr(res, "status", None) not in (0, 3):
        return None
    g = getattr(res, "x", None)
    if g is None:
        return None
    g = _clean_array(g)
    if g.size != n:
        return None
    gs = float(g.sum())
    if gs <= 1e-12:
        return None
    g = g / gs
    cp = c
    cg = _conv(g)
    cm = _conv(p, g)
    best_t = 0.0
    best_sc = _score_c(cp, 1.0, n)
    for t in (0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0):
        c_t = ((1.0 - t) ** 2) * cp + (2.0 * t * (1.0 - t)) * cm + (t * t) * cg
        sc = _score_c(c_t, 1.0, n)
        if math.isfinite(sc) and sc < best_sc - 1e-13:
            best_sc = sc
            best_t = t
    if best_t <= 0.0:
        return None
    return (1.0 - best_t) * p + best_t * g


def _build_candidates(x, S, n, best_score, max_cand):
    base = S / float(n) if n > 0 else 1.0
    yb = 0.0
    if math.isfinite(best_score) and best_score > 0.0 and n > 0:
        lb = best_score / (2.0 * float(n))
        if lb > 0.0:
            yb = math.sqrt(lb) * S
    scale = max(float(x), float(base), float(yb), 1e-12)
    max_y = min(
        1000.0,
        max(20.0 * base, 2.0 * float(x), 3.0 * float(yb), 10.0 * scale, 1e-8),
    )
    parts = [np.array([0.0], dtype=np.float64)]
    if x > 1e-300:
        parts.append(_COARSE * float(x))
    parts.append(_COARSE * float(base))
    if yb > 1e-300:
        parts.append(_COARSE * float(yb))
    if x > 1e-300:
        parts.append(float(x) * (1.0 - _FINE))
        parts.append(float(x) * (1.0 + _FINE))
    parts.append(float(base) * (1.0 - _FINE))
    parts.append(float(base) * (1.0 + _FINE))
    parts.append(_LOGR * scale)
    cand = np.concatenate(parts)
    cand = cand[(cand >= 0.0) & (cand <= max_y) & np.isfinite(cand)]
    if cand.size == 0:
        cand = np.array([0.0, max(0.0, min(float(x), max_y))], dtype=np.float64)
    cand = np.unique(cand)
    if cand.size > int(max_cand):
        idx = np.unique(
            np.concatenate(
                (
                    np.array([0, cand.size - 1], dtype=int),
                    np.linspace(0, cand.size - 1, max(1, int(max_cand) - 2)).astype(int),
                )
            )
        )
        cand = cand[idx]
    return cand


def _coordinate_refine(a0, stop_time, seed=0, max_lp=4):
    a = _normalize(a0)
    n = int(a.size)
    if n <= 0:
        return np.ones(1, dtype=np.float64), _score(np.ones(1))
    if n == 1:
        a = np.array([1.0], dtype=np.float64)
        return a, 2.0

    c = _conv(a)
    cur_score = _score_c(c, 1.0, n)
    if not math.isfinite(cur_score):
        cur_score = 2.0
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng(seed)
    lp_count = 0
    no_imp = 0
    last_best = best_score
    pass_num = 0

    while time.monotonic() < stop_time - 2.0:
        pass_num += 1
        S = float(a.sum())
        if S > 1e-12:
            a = a / S
            S = 1.0
        else:
            a = best_a.copy()
            S = 1.0
        c = _conv(a)
        cur_score = _score_c(c, S, n)
        if not math.isfinite(cur_score):
            cur_score = best_score
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()

        if n <= 512:
            max_cand = 70
        elif n <= 1024:
            max_cand = 55
        else:
            max_cand = 45

        for i in range(n):
            if time.monotonic() > stop_time - 1.5:
                break
            x = float(a[i])
            if i > 0:
                left = float(c[:i].max())
            else:
                left = 0.0
            if i + n < 2 * n - 1:
                right = float(c[i + n:].max())
            else:
                right = 0.0
            unaff = left if left > right else right

            cand = _build_candidates(x, S, n, best_score, max_cand)
            if cand.size == 0:
                continue
            delta = cand - x
            Sv = S + delta
            good = Sv > 1e-12
            if not np.any(good):
                continue

            aff = c[i : i + n][None, :] + 2.0 * delta[:, None] * a[None, :]
            aff[:, i] += delta * delta
            maxv = aff.max(axis=1)
            maxv = np.maximum(maxv, unaff)
            sc = np.full(aff.shape[0], math.inf, dtype=np.float64)
            denom = Sv[good] * Sv[good]
            sc[good] = (2.0 * float(n) * maxv[good]) / denom

            bi = int(np.argmin(sc))
            ybest = float(cand[bi])
            scbest = float(sc[bi])
            if not math.isfinite(scbest):
                continue
            if scbest < cur_score - 1e-13:
                d = ybest - x
                if d != 0.0:
                    c[i : i + n] += 2.0 * d * a
                    c[2 * i] += d * d
                    a[i] = ybest
                    S += d
                    cur_score = scbest

        S = float(a.sum())
        if S > 1e-12:
            a = a / S
            S = 1.0
        else:
            a = best_a.copy()
            S = 1.0
        c = _conv(a)
        cur_score = _score_c(c, S, n)
        if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()

        if best_score < last_best - 1e-12:
            last_best = best_score
            no_imp = 0
        else:
            no_imp += 1

        if no_imp >= 2:
            if max_lp > 0 and lp_count < max_lp and time.monotonic() < stop_time - 12.0:
                K = 100 if n <= 512 else 80 if n <= 1024 else 60
                try:
                    newp = _lp_move(
                        a,
                        c,
                        n,
                        min(stop_time - 5.0, time.monotonic() + 3.0),
                        K=K,
                    )
                except Exception:
                    newp = None
                if newp is not None:
                    newp = _normalize(newp)
                    sc = _score(newp)
                    if math.isfinite(sc):
                        a = newp
                        c = _conv(a)
                        cur_score = sc
                        if cur_score < best_score - 1e-13:
                            best_score = cur_score
                            best_a = a.copy()
                        lp_count += 1
                        no_imp = 0
                        last_best = min(last_best, best_score)
                        continue

            sig = 0.02 + 0.02 * min(max(0, no_imp - 1), 10)
            if best_score > 1.7:
                sig = min(0.25, 2.0 * sig)
            sig = min(0.35, sig)

            cand_a = best_a.copy()
            cand_sc = best_score
            for s in (sig, sig * 1.7, sig * 2.8):
                pa = _perturb(best_a, s, rng)
                sc = _score(pa)
                if math.isfinite(sc) and sc < cand_sc:
                    cand_a = pa
                    cand_sc = sc
            a = _normalize(cand_a)
            c = _conv(a)
            cur_score = _score_c(c, 1.0, n)
            if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
                best_score = cur_score
                best_a = a.copy()
            no_imp = 0
            last_best = min(last_best, best_score)

        if time.monotonic() > stop_time - 5.0:
            break

    return _normalize(best_a), float(best_score)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 25.0
    if deadline <= t0 + 20.0:
        return [1.0]

    parent = _normalize(_get_parent())
    best_a = parent.copy()
    best_sc = _exact_score(best_a)
    if not math.isfinite(best_sc):
        best_a = np.ones(384, dtype=np.float64) / 384.0
        best_sc = _exact_score(best_a)
    if not math.isfinite(best_sc):
        return [1.0]

    # Fixed resolution ladder.  The supplied construction is length 384;
    # larger lengths are generated by interpolating the current best whenever
    # possible, which preserves progress while allowing finer discretization.
    lengths = [384, 768, 1024, 1536, 2048]
    if parent.size not in lengths:
        lengths.append(int(parent.size))
    lengths = sorted(set(int(x) for x in lengths if int(x) >= 64 and int(x) <= 2200))
    if len(lengths) > 5:
        keep = {int(parent.size)}
        rest = sorted(set(lengths) - keep, reverse=True)[:4]
        lengths = sorted(keep | set(rest))

    phases = lengths + ["final"]
    raw = [max(0.3, L / float(max(1, parent.size))) for L in lengths]
    final_frac = 0.12
    work_frac = 1.0 - final_frac
    sraw = float(sum(raw))
    fracs = [work_frac * (r / sraw if sraw > 0 else 0.0) for r in raw] + [final_frac]

    total_work = max(10.0, deadline - t0)
    cum = 0.0
    phase_idx = 0
    final_ran = False

    for ph in phases:
        cum += fracs[phase_idx]
        phase_idx += 1
        stop_abs = t0 + total_work * min(1.0, cum)
        if stop_abs > deadline:
            stop_abs = deadline
        if time.monotonic() >= stop_abs - 1.0:
            break
        stop = stop_abs - 1.0

        if ph == "final":
            init = best_a.copy()
            max_lp = 3
            final_ran = True
        else:
            L = int(ph)
            if best_a.size > 0 and math.isfinite(best_sc):
                source = best_a
            else:
                source = parent
            init = source.copy() if int(source.size) == L else _resample(source, L)
            max_lp = 3 if L <= 768 else 4

        try:
            new_a, new_sc = _coordinate_refine(init, stop, seed=seed + 991 * (phase_idx * 7 + 3), max_lp=max_lp)
        except Exception:
            new_a, new_sc = init, best_sc

        if new_a is not None and len(new_a) > 0 and math.isfinite(new_sc):
            es = _exact_score(new_a)
            if math.isfinite(es) and es < best_sc - 1e-15:
                best_a = _clean_array(new_a)
                best_sc = es

        if time.monotonic() > stop_abs - 5.0:
            break

    # If the final phase was skipped, use all remaining time for final polish.
    if (not final_ran) and time.monotonic() < deadline - 5.0:
        try:
            new_a, new_sc = _coordinate_refine(best_a, deadline - 3.0, seed=seed + 123457, max_lp=4)
            if new_a is not None and len(new_a) > 0 and math.isfinite(new_sc):
                es = _exact_score(new_a)
                if math.isfinite(es) and es < best_sc - 1e-15:
                    best_a = _clean_array(new_a)
                    best_sc = es
        except Exception:
            pass

    final_a = _clean_array(best_a)
    if final_a.size == 0:
        final_a = np.ones(1, dtype=np.float64)
    final_sc = _exact_score(final_a)
    if not math.isfinite(final_sc):
        fallback = _clean_array(parent)
        if fallback.size == 0:
            fallback = np.ones(1, dtype=np.float64)
        final_sc = _exact_score(fallback)
        final_a = fallback
    if not math.isfinite(final_sc):
        return [1.0]

    return [float(x) for x in final_a]
