    fsize = next_fast_len(2 * n - 1)

def _subgrad(f0, n, stop, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.astype(float), 0, None)
    s = f.sum()
    f = f / s if s > 1e-10 else np.ones(n) / n
    bf, bsc = f.copy(), _score(f)
    lr, mom, beta, ni = 0.015, np.zeros(n), 0.8, 0
    while time.monotonic() < stop - 2:
        b = _conv(f)
        M, s = float(b.max()), float(f.sum())
        K = min(10, 2 * n - 1)
        tk = np.argpartition(b, -K)[-K:]
        w = b[tk].copy()
        w = (w - w.min() + 1e-10)
        w /= w.sum()
        gm = np.zeros(n)
        for wk, k in zip(w, tk):
            k = int(k)
            lo, hi = max(0, k - n + 1), min(n, k + 1)
            for i in range(lo, hi):
                j = k - i
                if 0 <= j < n:
                    gm[i] += wk * 2.0 * f[j]
        gf = (2.0 * n / s ** 2) * gm - (4.0 * n * M / s ** 3)
        mom = beta * mom - lr * gf
        fnew = np.maximum(f + mom, 0)
        s2 = fnew.sum()
        if s2 < 1e-10:
            fnew = f.copy()
        else:
            fnew /= s2
        sc = _score(fnew)
        if sc < bsc - 1e-14:
            f, bf, bsc, ni = fnew, fnew.copy(), sc, 0
            lr = min(0.06, lr * 1.008)
        else:
            ni += 1
            lr = max(0.002, lr * 0.985)
            if ni > 250:
                f = fnew.copy()
                ni, lr, beta = 0, 0.012, 0.7
                mom[:] = 0
    return bf, bsc

def _coord(f0, n, stop, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.astype(float), 0, None)
    s = f.sum()
    f = f / s if s > 1e-10 else np.ones(n) / n
    bsc = _score(f)
    ni = 0
    ms = (0.25, 0.5, 0.75, 1.25, 1.5, 2.0, 0.1)
    while time.monotonic() < stop - 2:
        imp = False
        for i in rng.permutation(n):
            if time.monotonic() > stop - 2:
                return f, bsc
            o = f[i]
            for m in ms:
                f[i] = o * m if o > 1e-12 else (0.05 if m > 0 else 0)
                if _score(f) < bsc - 1e-13:
                    bsc = _score(f)
                    imp = True
            f[i] = o
            if o > 1e-6:
                for d in (0.08, -0.08):
                    f[i] = max(0, o * (1 + d))
                    if _score(f) < bsc - 1e-13:
                        bsc = _score(f)
                        imp = True
                    f[i] = o
        if not imp:
            ni += 1
            if ni >= 10:
                break
        else:
            ni = 0
    return f, bsc

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lpdir(f, n, M, tl=8.0):
    A = _cmat(f, n)
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                    bounds=[(0, None)] * n, method='highs',
                    options={'presolve': True, 'time_limit': tl})
    except Exception:
        return None
    if r.status != 0 or r.x is None:
        return None
    g = np.clip(np.asarray(r.x, dtype=np.float64), 0, None)
    gs = float(g.sum())
    return g / gs if gs > 1e-14 else None


def _ls(f, g, bsc, tp, st):
    cands = sorted(set([max(0.02, min(0.95, tp * x))
                        for x in (0.4, 0.6, 0.8, 1.0, 1.3, 1.7, 2.2)]))
    bt, bf, loc_best = 0.0, None, bsc
    for t in cands:
        if time.monotonic() > st:
            break
        fn = (1.0 - t) * f + t * g
        np.maximum(fn, 0.0, out=fn)
        if fn.sum() < 0.01:
            continue
        sc = _score(fn)
        if sc < loc_best:
            loc_best = sc
            bt, bf = t, fn.copy()
    if bf is None:
        return None, None
    return bt, bf


def _lp_opt(f0, n, stop, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.astype(np.float64), 0, None)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    bf, bsc = f.copy(), _score(f)
    t, stall = 0.30, 0
    it = 0
    while time.monotonic() < stop - 2.0:
        it += 1
        b = _conv(f)
        M = float(b.max())
        if M < 1e-15:
            break
        g = _lpdir(f, n, M, tl=8.0)
        if g is None:
            stall += 3
            if stall > 60:
                break
            f = np.clip(f + 0.12 * rng.standard_normal(n), 0, None)
            s2 = f.sum()
            f /= s2 if s2 > 1e-10 else 1.0
            t = 0.25
            continue
        if np.linalg.norm(g - f) < 1e-7:
            stall += 3
            if stall > 60:
                break
            f = np.clip(f + 0.18 * rng.standard_normal(n), 0, None)
            s2 = f.sum()
            f /= s2 if s2 > 1e-10 else 1.0
            t, stall = 0.30, 0
            continue
        nt, nf = _ls(f, g, bsc, t, stop)
        if nf is not None:
            f = nf
            bf, bsc = f.copy(), _score(f)
            t = min(0.80, nt * 1.15)
            stall = 0
        else:
            t = max(0.04, t * 0.55)
            stall += 1
            if stall >= 45:
                f = np.clip(f + 0.25 * rng.standard_normal(n), 0, None)
                s2 = f.sum()
                f /= s2 if s2 > 1e-10 else 1.0
                t, stall = 0.30, 0
        if it % 80 == 0 and bsc < 1.503:
            break
    return bf, bsc


def _subgrad(f0, n, stop, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.astype(np.float64), 0, None)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    bf, bsc = f.copy(), _score(f)
    lr, mom, beta = 0.015, np.zeros(n), 0.82
    ni = 0
    while time.monotonic() < stop - 2.0:
        b = _conv(f)
        M, s = float(b.max()), float(f.sum())
        K = min(12, 2 * n - 1)
        tk = np.argpartition(b, -K)[-K:]
        w = b[tk].copy()
        w = w - w.min() + 1e-10
        w /= w.sum()
        gm = np.zeros(n)
        for wk, k_val in zip(w, tk):
            k = int(k_val)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            idx = np.arange(lo, hi)
            js = k - idx
            valid = (js >= 0) & (js < n)
            gm[idx[valid]] += wk * 2.0 * f[js[valid]]
        gf = (2.0 * n / (s * s)) * gm - (4.0 * n * M / (s * s * s))
        mom = beta * mom - lr * gf
        fnew = np.maximum(f + mom, 0.0)
        s2 = fnew.sum()
        if s2 < 1e-10:
            fnew = f.copy()
        else:
            fnew /= s2
        sc = _score(fnew)
        if sc < bsc - 1e-14:
            f, bf, bsc, ni = fnew, fnew.copy(), sc, 0
            lr = min(0.06, lr * 1.006)
        else:
            ni += 1
            lr = max(0.002, lr * 0.985)
            if ni > 300:
                f = fnew.copy()
                ni, lr, beta = 0, 0.012, 0.70
                mom[:] = 0.0
    return bf, bsc


def _coord(f0, n, stop, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.astype(np.float64), 0, None)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    bsc = _score(f)
    ni = 0
    ms = (0.25, 0.5, 0.75, 1.25, 1.5, 2.0, 0.1)
    while time.monotonic() < stop - 2.0:
        imp = False
        for i in rng.permutation(n):
            if time.monotonic() > stop - 2.0:
                return f, bsc
            o = f[i]
            for m in ms:
                f[i] = o * m if o > 1e-12 else (0.05 if m > 0 else 0.0)
                if _score(f) < bsc - 1e-13:
                    bsc = _score(f)
                    imp = True
            f[i] = o
            if o > 1e-6:
                for d in (0.08, -0.08, 0.15, -0.15):
                    f[i] = max(0.0, o * (1.0 + d))
                    if _score(f) < bsc - 1e-13:
                        bsc = _score(f)
                        imp = True
                    f[i] = o
        if not imp:
            ni += 1
            if ni >= 12:
                break
        else:
            ni = 0
    return f, bsc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    dl = t0 + budget - 20.0

    parent = _get_parent()
    n = parent.size

    best_seq = parent.copy()
    best_sc = _score(best_seq)
    if not math.isfinite(best_sc):
        best_seq = np.ones(384)
        n = 384
        best_sc = _score(best_seq)

    def upd(c):
        nonlocal best_seq, best_sc
        if c is None:
            return
        c = np.clip(np.asarray(c, dtype=np.float64).ravel(), 0, 1000)
        if c.size == 0 or c.sum() < 0.01:
            return
        try:
            sc = _score(c)
        except Exception:
            return
        if sc < best_sc - 1e-15:
            best_seq, best_sc = c.copy(), sc

    rng = np.random.default_rng(seed)
    ia = np.arange(n, dtype=np.float64)

    inits = [
        ("P", parent.copy()),
        ("C", 0.5 + 0.5 * np.cos(2.0 * math.pi * ia / n)),
        ("B", np.abs(0.42 + 0.5 * np.cos(2.0 * math.pi * ia / n)
                       + 0.08 * np.cos(4.0 * math.pi * ia / n)) + 0.01),
        ("R", np.abs(rng.normal(0, 1, n)) + 0.05),
        ("S", np.abs(np.sinc((ia - n / 2.0) / (n * 0.4))) + 0.05),
        ("N", np.abs(parent + 0.3 * max(1.0, float(parent.mean()))
                      * rng.standard_normal(n)) + 0.01),
    ]
    mf = np.full(n, 0.15)
    for fr in (1, 2, 3, 5, 7):
        mf += 0.25 * np.cos(2.0 * math.pi * fr * ia / n
                            + rng.uniform(0, 2 * math.pi))
    inits.append(("M", np.abs(mf) + 0.01))

    hp = 1.0 - 0.7 * np.cos(2.0 * math.pi * ia / n)
    inits.append(("H", np.abs(hp) + 0.01))

    # ---- Phase 1: LP optimisation (58% budget) ----
    lp_end = t0 + budget * 0.58
    # Give parent more time
    parent_time = (lp_end - t0) * 0.35
    other_time = (lp_end - t0 - parent_time) / max(len(inits) - 1, 1)

    for idx, (nm, ini) in enumerate(inits):
        if time.monotonic() > lp_end - 5.0:
            break
        if best_sc < 1.503:
            break
        if idx == 0:
            s_end = min(lp_end - 3.0, time.monotonic() + parent_time)
        else:
            s_end = min(lp_end - 3.0, time.monotonic() + other_time)
        if s_end - time.monotonic() < 15.0:
            s_end = min(lp_end - 3.0, time.monotonic() + 20.0)
        try:
            r, _ = _lp_opt(ini, n, s_end, seed + 1000 * (idx + 1))
            upd(r)
        except Exception:
            pass

    # Continue LP on best if time remains
    if time.monotonic() < lp_end - 8.0 and best_sc >= 1.503:
        try:
            r, _ = _lp_opt(best_seq, n, lp_end, seed + 77777)
            upd(r)
        except Exception:
            pass

    # ---- Phase 2: Subgradient polish (17%) ----
    sg_end = t0 + budget * 0.75
    if time.monotonic() < sg_end - 10.0:
        try:
            r, _ = _subgrad(best_seq, n, sg_end, seed + 99001)
            upd(r)
        except Exception:
            pass

    # ---- Phase 3: Multi-length exploration (10%) ----
    ml_end = t0 + budget * 0.85
    if time.monotonic() < ml_end - 10.0 and best_sc > 1.504:
        for L in (256, 320, 448, 512):
            if L == n or time.monotonic() > ml_end - 6.0:
                continue
            s_end = min(ml_end - 4.0, time.monotonic() + 30.0)
            il = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n), best_seq)
            try:
                r, _ = _lp_opt(il, L, s_end, seed + 50000 + L)
                upd(r)
            except Exception:
                pass

    # ---- Phase 4: Coordinate polish (7%) ----
    cp_end = dl - 8.0
    if time.monotonic() < cp_end - 10.0:
        try:
            r, _ = _coord(best_seq, len(best_seq), cp_end, seed + 77001)
            upd(r)
        except Exception:
            pass

    # ---- Final subgradient burst ----
    if time.monotonic() < dl - 8.0:
        try:
            r, _ = _subgrad(best_seq, len(best_seq), dl - 6.0, seed + 33001)
            upd(r)
        except Exception:
            pass

    # ---- Final validation with exact evaluator ----
    fl = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        sc = evaluate_sequence(fl)
        if math.isfinite(sc) and sc > 0:
            return fl
    except Exception:
        pass

    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass

    return [1.0] * len(parent) if parent.size > 0 else [1.0]
