import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p

def _conv_full(a):
    n = len(a)
    if n <= 600:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """Build the constraint matrix for the LP: A[g] = f * g (full convolution)."""
    # A has shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # For variable g[j], it contributes f[k-j] to A[k, j] for k = j, ..., j+n-1
        A[j:j+n, j] = f
    return A

def _lp_direction(f, n, M, time_limit=5.0):
    """Solve max sum(g) s.t. f*g <= M, g >= 0. Returns normalized g or None."""
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None
    return g / g_sum

def _coord_polish(f, stop_time, max_sweeps=40, seed=777):
    """Coordinate descent with exact 1D optimization for each coordinate."""
    n = len(f)
    f = np.clip(f.astype(np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, _score(f)
    conv = _conv_full(f)
    best_score = 2.0 * n * conv.max() / (S * S)
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break

            v = f[i]
            S_rest = S - v

            # Influence: for k in [i, i+n-1], c_k changes by 2*delta*f[k-i]
            # For k=2i: additional delta^2
            # The affected range is k from i to i+n-1 (clamped to [0, 2n-2])
            start = i
            end = min(i + n, 2 * n - 1)
            # f[k-i] for k in [start, end) is f[0] to f[end-start-1]
            infl = f[:end - start]  # length = end - start

            # Precompute: max of unaffected entries
            # Unaffected: k < start or k >= end
            if start > 0:
                max_unaffected_left = float(conv[:start].max())
            else:
                max_unaffected_left = 0.0
            if end < 2 * n - 1:
                max_unaffected_right = float(conv[end:].max())
            else:
                max_unaffected_right = 0.0
            max_unaffected = max(max_unaffected_left, max_unaffected_right)

            # Search range for new value
            x_max = max(3.0 * v + 0.5, 1.0) if v > 1e-8 else 2.0
            x_max = min(x_max, 100.0)

            # Coarse grid
            n_pts = 100
            x_grid = np.linspace(0.0, x_max, n_pts)

            best_x = v
            best_obj = best_score

            for x in x_grid:
                S_new = S_rest + x
                if S_new < 0.01:
                    continue
                w = x - v
                # New max in affected range:
                # For k in [start, end): new_c[k] = conv[k] + 2*w*f[k-i]
                # For k = 2i: additional w^2
                new_affected = conv[start:end] + 2.0 * w * infl
                if i < end:  # 2i is in [start, end) if start <= 2i < end
                    # 2i >= start means 2i >= i, always true for i >= 0
                    # 2i < end means 2i < min(i+n, 2n-1), always true for i < n
                    idx_2i = 2 * i - start
                    new_affected[idx_2i] += w * w
                m_affected = float(new_affected.max())
                m_new = max(m_affected, max_unaffected)
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_x = x

            # Fine grid around best_x
            if best_x != v:
                dw = x_max / n_pts
                x_fine = np.linspace(max(0, best_x - 3 * dw), min(x_max, best_x + 3 * dw), 50)
                for x in x_fine:
                    S_new = S_rest + x
                    if S_new < 0.01:
                        continue
                    w = x - v
                    if abs(w) < 1e-14:
                        continue
                    new_affected = conv[start:end] + 2.0 * w * infl
                    idx_2i = 2 * i - start
                    if 0 <= idx_2i < len(new_affected):
                        new_affected[idx_2i] += w * w
                    m_affected = float(new_affected.max())
                    m_new = max(m_affected, max_unaffected)
                    obj = 2.0 * n * m_new / (S_new * S_new)
                    if obj < best_obj - 1e-15:

                f[i] = best_x

                conv[start:end] += 2.0 * w * infl
                idx_2i = 2 * i - start

def _block_polish(f, stop_time, block_size=4, max_rounds=20, seed=12345):
    """Optimize blocks of coordinates simultaneously."""
    n = len(f)
    f = np.clip(f.astype(np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, _score(f)
    best_score = _score(f)
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        # Pick a random block
        start_idx = int(rng.integers(0, max(1, n - block_size)))
        block = slice(start_idx, start_idx + block_size)
        orig_block = f[block].copy()
        S_rest = S - float(orig_block.sum())

        # Try random perturbations of the block
        for trial in range(200):
            if time.monotonic() > stop_time - 5.0:
                break
            pert = rng.uniform(-0.3, 0.3, block_size)
            new_block = np.clip(orig_block * (1.0 + pert), 0, None)
            if new_block.sum() < 1e-10 and S_rest < 0.01:
                continue
            f[block] = new_block
            sc = _score(f)
            if sc < best_score - 1e-14:
                best_score = sc
                improved = True
                orig_block = new_block.copy()
                S = float(f.sum())

        if not improved:
            if rnd > 5:
                break

    return f, best_score

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    if n <= 800:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: 2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _lp_direction(f, n, M, time_limit=4.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None
    return g / g_sum


def _lp_optimize(f0, n, stop_time, seed, max_iters=300):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f = f[:n]
    if f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum

    best_f = f.copy()
    best_score = _score(f)
    conv = _conv_full(f)
    current_max_b = float(conv.max())

    t = 0.15
    t_min = 0.02
    t_max = 0.5
    stall = 0
    max_stall = 60
    iteration = 0

    while time.monotonic() < stop_time - 3.0 and iteration < max_iters:
        g_dir = _lp_direction(f, n, current_max_b, time_limit=3.0)
        if g_dir is None:
            stall += 4
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
                conv = _conv_full(f)
                current_max_b = float(conv.max())
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-7:
            stall += 4
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
                conv = _conv_full(f)
                current_max_b = float(conv.max())
            continue

        improved = False
        # Try multiple t values adaptively
        for trial in range(14):
            f_new = (1.0 - t) * f + t * g_dir
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 1e-10:
                t = max(t_min, t * 0.5)
                continue
            f_new = f_new / fs_new
            sc = _score(f_new)
            if sc < best_score - 1e-14:
                f = f_new
                best_f = f.copy()
                best_score = sc
                conv = _conv_full(f)
                current_max_b = float(conv.max())
                improved = True
                t = min(t_max, t * 1.12)
                stall = 0
                break
            else:
                t = max(t_min, t * 0.5)

        if not improved:
            stall += 1
            if stall >= max_stall:
                # Larger perturbation to escape
                f = np.clip(f + rng.normal(0, 0.12, n), 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs
                else:
                    f = best_f.copy()
                conv = _conv_full(f)
                current_max_b = float(conv.max())
                t = 0.15
                stall = 0
        iteration += 1

    return best_f, best_score


def _coord_polish_vec(f, stop_time, max_sweeps=60, seed=777):
    """Vectorized coordinate descent with exact 1D optimization."""
    n = len(f)
    f = np.clip(f.astype(np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, _score(f)
    conv = _conv_full(f).copy()
    best_score = 2.0 * n * conv.max() / (S * S)
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break

            v = f[i]
            S_rest = S - v

            start = i
            end = min(i + n, 2 * n - 1)
            infl = f[: end - start]
            idx_2i = 2 * i - start

            # Precompute max of unaffected entries
            max_unaff = 0.0
            if start > 0:
                max_unaff = max(max_unaff, float(conv[:start].max()))
            if end < 2 * n - 1:
                max_unaff = max(max_unaff, float(conv[end:].max()))

            # Search range
            x_max = max(3.0 * v + 0.3, 0.5) if v > 1e-8 else 1.5
            x_max = min(x_max, 50.0)

            # Vectorized grid search
            n_pts = 120
            x_grid = np.linspace(0.0, x_max, n_pts)
            w_grid = x_grid - v

            # Build matrix: (end-start) x n_pts
            new_aff = conv[start:end, :None] + 2.0 * w_grid[None, :] * infl[:, None]
            if 0 <= idx_2i < len(infl):
                new_aff[idx_2i, :] += w_grid ** 2

            m_aff = new_aff.max(axis=0)
            m_new = np.maximum(m_aff, max_unaff)
            S_new_grid = S_rest + x_grid
            valid = S_new_grid > 0.01
            if not valid.any():
                continue
            obj = np.where(valid, 2.0 * n * m_new[valid] / (S_new_grid[valid] ** 2), np.inf)
            best_idx = int(np.argmin(obj))
            if obj[best_idx] < best_score - 1e-14:
                # Fine grid around best
                x_b = x_grid[best_idx]
                dw = x_max / n_pts
                x_fine = np.linspace(max(0, x_b - 4 * dw), min(x_max, x_b + 4 * dw), 80)
                w_fine = x_fine - v
                new_aff_f = conv[start:end, :None] + 2.0 * w_fine[None, :] * infl[:, None]
                if 0 <= idx_2i < len(infl):
                    new_aff_f[idx_2i, :] += w_fine ** 2
                m_aff_f = new_aff_f.max(axis=0)
                m_new_f = np.maximum(m_aff_f, max_unaff)
                S_new_f = S_rest + x_fine
                valid_f = S_new_f > 0.01
                if valid_f.any():
                    obj_f = np.where(valid_f, 2.0 * n * m_new_f[valid_f] / (S_new_f[valid_f] ** 2), np.inf)
                    fi = int(np.argmin(obj_f))
                    if obj_f[fi] < best_score - 1e-14:
                        x_b = x_fine[fi]
                        best_score = float(obj_f[fi])

                # Accept
                w = x_b - v
                f[i] = x_b
                S += w
                conv[start:end] += 2.0 * w * infl
                if 0 <= idx_2i < len(conv):
                    conv[2 * i] += w * w
                improved = True

        if not improved:
            if sweep > 2:
                break

    f = np.clip(f, 0, 1000)
    final_score = _score(f)
    return f, final_score


def _smooth_obj_and_grad(f, alpha):
    """Compute soft-max smoothed objective and gradient."""
    n = len(f)
    S = float(f.sum())
    if S < 0.01:
        return math.inf, None
    conv = _conv_full(f)
    max_c = float(conv.max())
    shifted = conv - max_c
    exp_shifted = np.exp(alpha * shifted)
    log_sum_exp = max_c + np.log(exp_shifted.sum()) / alpha
    weights = exp_shifted / exp_shifted.sum()

    # Gradient of soft_max wrt f:
    # grad_i = 2 * sum_j f[j] * weights[i+j] for valid indices
    # = 2 * (convolve(f[::-1], weights, 'full'))[n-1 : 2n-1]
    gc = np.convolve(f[::-1], weights, mode="full")
    grad_sm = 2.0 * gc[n - 1 : 2 * n - 1]

    obj = 2.0 * n * log_sum_exp / (S * S)
    grad_obj = 2.0 * n * (grad_sm - 2.0 * log_sum_exp / S) / (S * S)

    return float(obj), grad_obj


def _smooth_optimize(f0, stop_time, alpha_start=30.0, alpha_end=300.0, seed=123):
    """Optimize using soft-max smoothed objective with L-BFGS-B."""
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    n = len(f)
    best_f = f.copy()
    best_hard = _score(f)

    # An Anneal alpha and run L-BFGS-B
    n_steps = 8
    alphas = np.geomspace(alpha_start, alpha_end, n_steps)
    time_per_step = max(3.0, (stop_time - time.monotonic() - 5.0) / n_steps)

    for alpha in alphas:
        if time.monotonic() > stop_time - 5.0:
            break
        step_deadline = time.monotonic() + time_per_step

        def fun(x):
            if time.monotonic() > step_deadline:
                return math.inf, np.zeros_like(x)
            obj, grad = _smooth_obj_and_grad(x, alpha)
            if not math.isfinite(obj):
                return 1e10, np.zeros_like(x)
            if grad is None:
                return 1e10, np.zeros_like(x)
            return obj, grad

        try:
            res = minimize(
                fun, f, method="L-BFGS-B",
                bounds=[(0, 1000)] * n,
                options={"maxiter": 80, "maxfun": 200,
                         "ftol": 1e-16, "gtol": 1e-12},
            )
        except Exception:
            continue

        if res.x is not None and np.all(np.isfinite(res.x)):
            res_x = np.clip(res.x, 0, 1000)
            if res_x.sum() > 0.01:
                hard_sc = _score(res_x)
                if hard_sc < best_hard - 1e-14:
                    best_f = res_x.copy()
                    best_hard = hard_sc
                f = res_x.copy()

    return best_f, best_hard


def _block_polish(f, stop_time, block_size=3, max_rounds=25, seed=12345):
    """Optimize small blocks of coordinates via random search."""
    n = len(f)
    f = np.clip(f.astype(np.float64).copy(), 0, None)
    best_score = _score(f)
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        n_blocks = max(1, n // block_size)
        for b in range(n_blocks):
            if time.monotonic() > stop_time - 5.0:
                break
            start_idx = b * block_size
            end_idx = min(start_idx + block_size, n)
            block = slice(start_idx, end_idx)
            orig_block = f[block].copy()

            for trial in range(80):
                if time.monotonic() > stop_time - 5.0:
                    break
                pert = rng.uniform(-0.4, 0.4, end_idx - start_idx)
                new_block = np.clip(orig_block * (1.0 + pert), 0, None)
                f[block] = new_block
                if f.sum() < 0.01:
                    f[block] = orig_block
                    continue
                sc = _score(f)
                if sc < best_score - 1e-14:
                    best_score = sc
                    orig_block = new_block.copy()
                    improved = True

        if not improved:
            if rnd > 3:
                break

    return f, best_score
