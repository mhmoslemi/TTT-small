```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv(a):
    """Linear convolution via FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    """Exact score matching the evaluator."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)


def _norm1(a):
    """Normalize to sum=1, maintaining nonnegativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _build_conv_matrix(f, n):
    """A[k, j] = f[k-j] if 0<=k-j<n else 0. So (A@x)_k = (f*x)_k."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(f, n, M):
    """LP: max 1'g s.t. (f*g)_k <= M, g >= 0. Returns normalized g."""
    A = _build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, 1000.0)] * n,
                      method='highs',
                      options={'time_limit': 15, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
    gs = g.sum()
    if gs < 1e-14:
        return None
    return g / gs


def _subgrad_dir(f, n, peak_frac=0.9):
    """Compute subgradient direction for max(f*f) using peak lags.
    d(f*f)_k / d(f_i) = 2*f_{k-i} if 0<=k-i<n else 0."""
    b = _conv(f)
    M = b.max()
    if M < 1e-18:
        return np.zeros(n), M, b
    thresh = M * peak_frac
    peak_idx = np.where(b >= thresh)[0]
    w = b[peak_idx] - thresh + 1e-16  # weight peaks by how close to max
    w /= w.sum()
    
    sg = np.zeros(n)
    for wi, k in zip(w, peak_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            partner = k - i
            if 0 <= partner < n:
                sg[i] += 2.0 * wi * f[partner]
    return sg, M, b


def _proj_subgrad(f0, n, t_dead, seed):
    """Projected subgradient with momentum and adaptive step."""
    rng = np.random.default_rng(seed)
    f = _norm1(f0.copy())
    best_f = f.copy()
    best_sc = _score(f)
    mom = np.zeros(n)
    beta = 0.85
    lr = 0.03
    no_imp = 0
    
    while time.monotonic() < t_dead:
        sg, M, b = _subgrad_dir(f, n, peak_frac=0.92)
        
        # Score = 2*n*M (since sum=1)
        sc = 2.0 * n * M
        
        # Update
        f_new = f - lr * sg / (sg.max() + 1e-15)
        f_new = np.maximum(f_new, 0)
        f_new = _norm1(f_new)
        
        mom = beta * mom + (f_new - f)
        f_new = _norm1(np.maximum(f_new + 0.3 * mom, 0))
        
        sc_new = _score(f_new)
        if sc_new < best_sc - 1e-14:
            best_f[:] = f_new
            best_sc = sc_new
            f = f_new
            no_imp = 0
            lr = min(0.08, lr * 1.01)
        else:
            no_imp += 1
            lr = max(0.003, lr * 0.985)
            f = _norm1(np.maximum(f_new, 0))  # still move for exploration
            if no_imp > 300:
                f = _norm1(np.maximum(f0 + rng.normal(0, 0.2, n), 0))
                best_f[:] = f
                best_sc = _score(f)
                no_imp = 0
                lr = 0.03
                mom[:] = 0
    
    return best_f, best_sc


def _coord_decscent(f0, n, t_dead, seed):
    """Coordinate and pair descent."""
    rng = np.random.default_rng(seed)
    f = _norm1(f0.copy())
    best_sc = _score(f)
    best_f = f.copy()
    mults = (0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0)
    stall_rounds = 0
    
    while time.monotonic() < t_dead:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > t_dead:
                return best_f, best_sc
            orig = f[i]
            for m in mults:
                f[i] = orig * m if orig > 1e-12 else (0.01 if m != 1.0 else 0.0)
                sc = _score(f)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    improved = True
                    orig = f[i]
                else:
                    f[i] = orig
            # Restore and try additive
            f[i] = orig
            if orig > 1e-8:
                for frac in (0.9, 1.1):
                    f[i] = orig * frac
                    f = _norm1(f)
                    sc = _score(f)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        improved = True
                    else:
                        f[i] = orig / (f.sum())  # undo normalization
                        f = _norm1(f)
        
        # Pair moves
        for _ in range(25):
            if time.monotonic() > t_dead:
                return best_f, best_sc
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi in (0.5, 1.5):
                for mj in (0.5, 1.5):
                    f[i] = oi * mi if oi > 1e-12 else 0.0
                    f[j] = oj * mj if oj > 1e-12 else 0.0
                    sc = _score(f)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        improved = True
                        break
                else:
                    continue
                break
            f[i], f[j] = oi, oj
            f = _norm1(f)
        
        best_f[:] = f
        
        if not improved:
            stall_rounds += 1
            if stall_rounds >= 8:
                break
        else:
            stall_rounds = 0
    
    return best_f, best_sc


def _lp_loop(f0, n, t_dead, seed):
    """LP-guided optimization."""
    rng = np.random.default_rng(seed)
    f = _norm1(f0.copy())
    best_f = f.copy()
    best_sc = _score(f)
    t_mv = 0.3
    stall = 0
    
    for it in range(500):
        if time.monotonic() > t_dead:
            break
        
        b = _conv(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        g = _lp_dir(f, n, M)
        if g is None:
            stall += 4
            if stall > 30:
                break
            f = _norm1(f + rng.normal(0, 0.04, n))
            continue
        
        d = np.linalg.norm(g - f, ord=np.inf)
        if d < 1e-7:
            stall += 4
            if stall > 30:
                break
            f = _norm1(f + rng.normal(0, 0.06, n))
            continue
        
        # Line search
        best_t = 0.0
        best_ls_sc = best_sc
        tl, th = 0.0, 1.0
        for _ in range(10):
            tm = 0.5 * (tl + th)
            if tm < 1e-8:
                break
            cand = _norm1((1 - tm) * f + tm * g)
            sc = _score(cand)
            if sc < best_ls_sc - 1e-14:
                best_ls_sc = sc
                best_t = tm
                tl = tm
            else:
                th = tm
            if th - tl < 5e-5:
                break
        
        if best_t > 0.001:
            f = _norm1((1 - best_t) * f + best_t * g)
            sc_new = _score(f)
            if sc_new < best_sc - 1e-14:
                best_f[:] = f
                best_sc = sc_new
                t_mv = min(0.6, max(0.1, best_t * 1.1))
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 25:
            f = _norm1(f + rng.normal(0, 0.1, n))
            t_mv = 0.3
            stall = 0
    
    return best_f, best_sc
```

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _n1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(f, n, M):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, 1000.0)] * n,
                      method='highs',
                      options={'time_limit': 12, 'presolve': True,
                               'dual_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None or res.fun is None or not np.isfinite(res.fun):
        return None
    g = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
    gs = g.sum()
    if gs < 1e-14:
        return None
    return g / gs


def _subgrad(f, n, frac=0.90):
    b = _conv(f)
    M = float(b.max())
    if M < 1e-18:
        return np.zeros(n), M, b
    thresh = M * frac
    pm = b >= thresh
    pw = np.where(pm, b - thresh + 1e-16, 0.0)
    pw_sum = pw.sum()
    if pw_sum < 1e-30:
        pw = np.zeros(b.shape); pw[np.argmax(b)] = 1.0
    else:
        pw /= pw_sum
    sg = np.zeros(n)
    peaks = np.where(pm)[0]
    for k in peaks:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            sg[lo:hi + 1] += 2.0 * pw[k] * f[k - hi:k - lo + 1][::-1]
    return sg, M, b


def _lp_loop(f0, n, t_dead, seed):
    rng = np.random.default_rng(seed)
    f = _n1(f0.copy())
    best_f = f.copy()
    best_sc = _score(f)
    stall = 0
    for it in range(500):
        if time.monotonic() > t_dead:
            break
        b = _conv(f)
        M = float(b.max())
        if M < 1e-18:
            break
        g = _lp_dir(f, n, M)
        if g is None:
            stall += 4
            if stall > 30:
                break
            f = _n1(f + rng.normal(0, 0.04, n))
            continue
        if math.isfinite(np.linalg.norm(g - f, ord=np.inf)) and np.linalg.norm(g - f, ord=np.inf) < 1e-7:
            stall += 4
            if stall > 30:
                break
            f = _n1(f + rng.normal(0, 0.06, n))
            continue
        best_t, best_ls = 0.0, best_sc
        tl, th = 0.0, 1.0
        for _ in range(10):
            tm = 0.5 * (tl + th)
            if tm < 1e-8:
                break
            cand = _n1((1 - tm) * f + tm * g)
            sc = _score(cand)
            if sc < best_ls - 1e-14:
                best_ls = sc
                best_t = tm
                tl = tm
            else:
                th = tm
            if th - tl < 5e-5:
                break
        if best_t > 0.001:
            f = _n1((1 - best_t) * f + best_t * g)
            sc_new = _score(f)
            if sc_new < best_sc - 1e-14:
                best_f[:] = f
                best_sc = sc_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 25:
            f = _n1(f + rng.normal(0, 0.1, n))
            stall = 0
    return best_f, best_sc


def _subgrad_run(f0, n, t_dead, seed):
    rng = np.random.default_rng(seed)
    f = _n1(f0.copy())
    best_f = f.copy()
    best_sc = _score(f)
    mom = np.zeros(n)
    beta = 0.85
    lr = 0.03
    no_imp = 0
    while time.monotonic() < t_dead:
        sg, M, b = _subgrad(f, n, 0.92)
        sg_norm = sg.max()
        if sg_norm < 1e-18:
            break
        sg_n = sg / sg_norm
        f_raw = f - lr * sg_n
        f_new = _n1(f_raw)
        mom = beta * mom + (f_new - f)
        f_new = _n1(f_raw + 0.3 * mom)
        sc_new = _score(f_new)
        if sc_new < best_sc - 1e-14:
            best_f[:] = f_new
            best_sc = sc_new
            f = f_new
            no_imp = 0
            lr = min(0.08, lr * 1.01)
        else:
            no_imp += 1
            lr = max(0.003, lr * 0.985)
            f = _n1(f_raw)
            if no_imp > 300:
                f = _n1(f0 + rng.normal(0, 0.2, n))
                best_f[:] = f
                best_sc = _score(f)
                no_imp = 0
                lr = 0.03
                mom[:] = 0.0
                beta = 0.7
    return best_f, best_sc


def _spade(f0, n, t_dead, seed):
    rng = np.random.default_rng(seed)
    f = _n1(f0.copy())
    best_f = f.copy()
    best_sc = _score(f)
    d = rng.normal(0, 1, n)
    no_imp = 0
    sc_prev_plus = _score(_n1(f + 0.01 * d))
    for it in range(200000):
        if time.monotonic() > t_dead:
            break
        c = 0.01
        fp = _n1(f + c * d)
        fm = _n1(f - c * d)
        scp = _score(fp)
        scm = _score(fm)
        if math.isfinite(scp) and math.isfinite(scm):
            gi = (scp - scm) / (2.0 * c)
            if abs(gi) > 1e-15:
                f_new = _n1(f - 0.5 * gi * d)
                sc = _score(f_new)
                if sc < best_sc - 1e-14:
                    best_f[:] = f_new
                    best_sc = sc
                    f = f_new
                    no_imp = 0
                else:
                    no_imp += 1
                    f = _n1(f - 0.1 * gi * d)
                    if no_imp > 500:
                        f = _n1(f0 + rng.normal(0, 0.15, n))
                        d = rng.normal(0, 1, n)
                        best_f[:] = f
                        best_sc = _score(f)
                        no_imp = 0
                        sc_prev_plus = _score(_n1(f + 0.01 * d))
            else:
                no_imp += 1
                if no_imp > 500:
                    f = _n1(f0 + rng.normal(0, 0.15, n))
                    d = rng.normal(0, 1, n)
                    best_f[:] = f
                    best_sc = _score(f)
                    no_imp = 0
                    sc_prev_plus = _score(_n1(f + 0.01 * d))
        if it % 500 == 499:
            d = rng.normal(0, 1, n)
    return best_f, best_sc


def _coord_desc(f0, n, t_dead, seed):
    rng = np.random.default_rng(seed)
    f = _n1(f0.copy())
    best_sc = _score(f)
    best_f = f.copy()
    mults = (0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0)
    stall_r = 0
    while time.monotonic() < t_dead:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > t_dead:
                return best_f, best_sc
            orig = f[i]
            for m in mults:
                f[i] = orig * m if orig > 1e-12 else (0.01 if m != 1.0 else 0.0)
                sc = _score(f)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    improved = True
                    orig = f[i]
                else:
                    f[i] = orig
            f[i] = orig
            if orig > 1e-8:
                for fr in (0.9, 0.95, 1.05, 1.1):
                    f[i] = orig * fr
                    f = _n1(f)
                    sc = _score(f)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        improved = True
                    else:
                        f = _n1(f)
                        f[i] = orig
        for _ in range(20):
            if time.monotonic() > t_dead:
                return best_f, best_sc
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi in (0.5, 1.5):
                for mj in (0.5, 1.5):
                    f[i] = oi * mi if oi > 1e-12 else 0.0
                    f[j] = oj * mj if oj > 1e-12 else 0.0
                    sc = _score(f)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        improved = True
                        break
                else:
                    continue
                break
            f[i], f[j] = oi, oj
            f = _n1(f)
        best_f[:] = f
        if not improved:
            stall_r += 1
            if stall_r >= 6:
                break
        else:
            stall_r = 0
    return best_f, best_sc
