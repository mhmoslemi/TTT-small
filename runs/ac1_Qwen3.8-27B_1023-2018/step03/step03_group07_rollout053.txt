                try:
                    res = minimize(obj, current, jac=jac, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 300, 'ftol': 1e-15})
                    if res.x is not None:
                        cand = np.maximum(res.x, 0.0)
                        cs = cand.sum()
                        if cs > 0.01:
                            cand = cand / cs
                            sc = _score(cand)
                            if np.isfinite(sc) and sc < local_best_sc * (1 - 1e-14):
                                local_best_sc = sc
                                local_best = cand.copy()
                            if not res.success:
                                break
                            current = cand.copy()
                        else:
                            break
                    else:
                        break
                except Exception:
                    break

        return local_best

    slsqp_result = _slsqp_optimize(lp_result, stop_time=phase2_end)
    _try_update(slsqp_result)

    # Also try SLSQP from a few alternative starts
    alt_starts = []
    x = np.arange(n, dtype=np.float64)
    # Smooth variants of parent
    for blur in [0.3, 0.5, 0.7]:
        alt = np.maximum(parent + blur * (parent - np.roll(parent, 1)), 0)
        alt = alt / alt.sum()
        alt_starts.append(alt)
    # Player-like: raised cosine
    alt_starts.append(0.5 * (1 + np.cos(2 * np.pi * x / n)) / np.mean(0.5 * (1 + np.cos(2 * np.pi * x / n))))
    # Power law
    for pw in [0.5, 1.5]:
        alt = (1 - x / (n - 1 + 1e-9)) ** pw
        alt = np.maximum(alt, 1e-6) / np.sum(np.maximum((1 - x / (n - 1 + 1e-9)) ** pw, 1e-6))
        alt_starts.append(alt)

    for i, alt in enumerate(alt_starts[:4]):
        if time.monotonic() > phase2_end - 10:
            break
        alloc = min(time.monotonic() + (phase2_end - time.monotonic()) / (len(alt_starts[:4]) - i), phase2_end - 5)
        try:
            r = _slsqp_optimize(alt, stop_time=alloc, betas=(100, 300))
            _try_update(r)
        except Exception:
            pass

    # ====================== Phase 3: Coordinate descent polish ======================
    phase3_end = t0 + (deadline - t0) * 0.88

    def _coord_descent(a_init, stop_time):
        a = np.array(a_init, dtype=np.float64).copy()
        a = np.clip(a, 0, 1000)
        a = np.maximum(a, 1e-12)
        a = a / a.sum()
        nn = a.size
        local_best = a.copy()
        local_best_sc = _score(a)

        order = rng.permutation(nn)
        sweeps = 0
        while time.monotonic() < stop_time:
            improved = False
            for j in order:
                if time.monotonic() > stop_time:
                    break
                # Save old
                old_j = a[j]
                others = a.sum() - a[j]
                if others < 1e-12:
                    continue
                # Max t such that a[j] <= t, sum stays positive
                t_low = 0.0
                t_high = 4.0 * old_j + 0.5 / nn  # generous upper bound
                if t_high < 1e-15:
                    t_high = 1.0 / (nn * 10)

                def _sub_score(t):
                    a2 = a.copy()
                    a2[j] = t
                    a2 = np.maximum(a2, 0)
                    s2 = a2.sum()
                    if s2 < 1e-10:
                        return float('inf')
                    c2 = np.convolve(a2, a2)
                    return float(np.max(c2)) / (s2 * s2)

                # Golden section search
                gr = (np.sqrt(5) - 1) / 2
                lo = t_low
                hi = t_high
                f_lo = _sub_score(lo)
                f_hi = _sub_score(hi)
                # Check boundaries
                best_t = j  # placeholder
                best_f = min(f_lo, f_hi)
                if f_lo <= f_hi:
                    best_t = lo
                else:
                    best_t = hi

                for _ in range(40):
                    if hi - lo < 1e-12:
                        break
                    m1 = lo + gr * (hi - lo)
                    m2 = hi - gr * (hi - lo)
                    f1 = _sub_score(m1)
                    f2 = _sub_score(m2)
                    if f1 <= f2:
                        hi = m2
                        if f2 < best_f:
                            best_f = f2
                            best_t = m2
                        if f1 < best_f:
                            best_f = f1
                            best_t = m1
                    else:
                        lo = m1
                        if f1 < best_f:
                            best_f = f1
                            best_t = m1

                if best_t >= 0 and best_t < 1000:
                    new_a = a.copy()
                    new_a[j] = best_t
                    new_a = np.maximum(new_a, 1e-14)
                    new_a = new_a / new_a.sum()
                    sc = _score(new_a)
                    if np.isfinite(sc) and sc < local_best_sc * (1 - 1e-14):
                        a = new_a
                        local_best_sc = sc
                        local_best = a.copy()
                        improved = True

            sweeps += 1
            if not improved and sweeps > 3:
                # Nudge
                a = a * np.exp(rng.normal(0, 0.01, nn))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
            if sweeps > 5000:
                break

        return local_best

    cd_result = _coord_descent(best, stop_time=phase3_end)
    _try_update(cd_result)

    # ====================== Phase 4: Try nearby lengths ======================
    phase4_end = t0 + (deadline - t0) * 0.94
    if time.monotonic() < phase4_end - 20:
        for dn in [-3, -2, -1, 1, 2, 3]:
            if time.monotonic() > phase4_end - 10:
                break
            nt = n + dn
            if nt < 10 or nt > 500:
                continue
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, best)
            init_new = np.clip(init_new, 0, 1000)
            sn = init_new.sum()
            if sn < 0.01:
                continue
            init_new = init_new / sn
            try:
                r = _slsqp_optimize(init_new, stop_time=time.monotonic() + 25, betas=(100, 300))
                _try_update(r)
            except Exception:
                pass

    # ====================== Phase 5: Final SLSQP polish on absolute best ======================
    if time.monotonic() < deadline - 15:
        final_try = _slsqp_optimize(best, stop_time=deadline - 5, betas=(200, 500, 1000, 3000))
        _try_update(final_try)

    # ====================== Final validation ======================
    out = [float(max(0.0, min(1000.0, x))) for x in best]
    try:
        fs = float(evaluate_sequence(out))
        if not np.isfinite(fs):
            out = [1.0 / 32] * 32
    except Exception:
        out = [1.0 / 32] * 32

    return out
