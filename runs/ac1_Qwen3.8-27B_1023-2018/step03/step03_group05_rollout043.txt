def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = len(v)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = (css - target) / idx < u
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target/n)
    theta = (css[rho[-1]] - target) / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-300:
        return np.full(n, target/n)
    return w / s * target

def _get_parent():
    h = globals().get("height_sequence_1", None)

        except Exception:

            s = float(p.sum())
            if s > 1e-300:
                return p / s
    return np.ones(512, dtype=np.float64) / 512.0


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones(a.size) / a.size if a.size > 0 else np.ones(1)
    return a / s


def _conv_full(a):
    n = a.size
    nfft = next_fast_len(2 * n - 1)
    f = rfft(a, nfft)
    c = irfft(f * f, nfft)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _weighted_grad(a, w, n):
    nfft = next_fast_len(3 * n - 2)
    fa = rfft(a, nfft)
    fw = rfft(w[::-1], nfft)
    cr = irfft(fa * fw, nfft)[:3 * n - 2]
    idx = 2 * n - 2 - np.arange(n)
    g = 2.0 * cr[idx]
    return np.maximum(g, 0.0)


def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.ones(1)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (css - target) / idx < u
    rho = np.nonzero(cond)[0]
    if rho.size == 0:
        return np.full(n, target / n)
    theta = (css[rho[-1]] - target) / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s < 1e-300:
        return np.full(n, target / n)
    return w / s * target


def _resample(a, m):
    n = a.size
    m = int(m)
    if m <= 0:
        return np.ones(1)
    if n == m:
        return a.copy()
    if m > n:
        idx = (np.arange(m, dtype=np.int64) * n) // m
        b = a[idx]
    else:
        b = np.zeros(m, dtype=np.float64)
        idx = (np.arange(n, dtype=np.int64) * m) // n
        np.add.at(b, idx, a)
    return _normalize(b)

def _soft_weights(c, T):
    M = c.max()
    if M <= 0:
        w = np.zeros(c.size)
        w[0] = 1.0
        return w
    z = (c - M) / max(T, 1e-15)
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    s = float(w.sum())
    if s < 1e-300:
        w = np.zeros(c.size)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    return w

    n = a.size
    a = a.copy()

    M = float(c.max())
    best_a = a.copy()

    T = M * 0.01  # temperature
    T_floor = M * 1e-6
    step = 0.1
    patience = 0
    
    while time.monotonic() < deadline:
        # Compute weights
        w = _soft_weights(c, T)
        
        # Compute gradient
        g = _weighted_grad(a, w, n)
        
        # Gradient magnitude check
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-12:
            break
        
        # Line search

        for s in [step * 0.1, step * 0.3, step, step * 3, step * 10]:

                continue
            a_trial = _project_simplex(a - s * g)
            c_trial = _conv_full(a_trial)
            M_trial = float(c_trial.max())
            if M_trial < M - 1e-14:
                a = a_trial
                c = c_trial
                M = M_trial
                improved = True
                break
            elif M_trial < M:
                a = a_trial
                c = c_trial
                M = M_trial

                best_a = a.copy()
            patience = 0
            # Adapt step

            # Cool down

            step *= 0.5
            if step < 1e-5:

                    M = float(c.max())
                    step = 0.1
                    T = M * 0.01
                    patience = 0

def _analytical_init(n, rng):
    """Generate analytical initial shapes."""
    x = np.linspace(0, 1, n, endpoint=False)
    kind = int(rng.integers(0, 8))
    
    if kind == 0:
        # Parabolic
        alpha = float(rng.uniform(0.3, 1.2))
        a = 1.0 - alpha * x**2
    elif kind == 1:
        # Cosine taper
        a = 0.5 + 0.5 * np.cos(np.pi * x)
    elif kind == 2:
        # Gaussian
        c = float(rng.uniform(0.3, 0.7))
        w = float(rng.uniform(0.15, 0.4))
        a = np.exp(-((x - c)**2) / (2*w**2))
    elif kind == 3:
        # Power
        alpha = float(rng.uniform(0.3, 2.0))
        a = x**alpha
    elif kind == 4:
        # Bi-gaussian
        a = np.exp(-x**2 / 0.08) + np.exp(-(1-x)**2 / 0.08)
    elif kind == 5:
        # Raised cosine squared
        a = (0.5 + 0.5 * np.cos(np.pi * x))**2
    elif kind == 6:
        # Sine
        k = int(rng.integers(1, 4))
        a = np.sin(np.pi * x * k)**2 + 0.1
    else:
        # Uniform with noise
        a = np.ones(n) + 0.3 * rng.standard_normal(n)

    return _normalize(a)

def _conv_full(a):
    n = a.size
    nfft = next_fast_len(2 * n - 1)
    f = rfft(a, nfft)
    c = irfft(f * f, nfft)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _weighted_grad(a, w, n):
