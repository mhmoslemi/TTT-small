(f*g)[k] <= (M + (f*f)[k]) / 2

            f = best_f_new
            M_cur = best_sc_local
            if M_cur < float(_conv_fft(best_f).max()) - 1e-14:
                best_f = f.copy()
                best_M = M_cur
                stall = 0
                no_improve = 0
            else:
                no_improve += 1
            t = float(np.clip(best_t * 1.05, 0.05, 1.0))
        else:
            stall += 1
            no_improve += 1
            t = max(0.05, t * 0.8)
            if stall >= max_stall:
                strength = 0.03 + 0.02 * min(stall // max_stall, 3)
                f = np.clip(best_f + rng.normal(0, strength, n), 0, None)
                f /= f.sum()
                t = 1.0
                stall = 0
        
        if no_improve >= max_no_improve:
            # Larger restart
            f = np.clip(best_f + rng.normal(0, 0.1, n), 0, None)
            f /= f.sum()
            t = 1.0
            stall = 0
            no_improve = 0
            # Check if we can break
            if _conv_fft(f).max() >= best_M * 1.01:
                # Perturbation made things worse, go back to best
                f = best_f.copy()
                # One more round with smaller perturbation
                f = np.clip(f + rng.normal(0, 0.01, n), 0, None)
                f /= f.sum()
    
    return best_f, 2.0 * n * best_M

        c = irfft(F * F, fsize)[:2*n-1]

        for i in order:

            best_val = orig
            best_val_M = best_sc
            for m in mults:
                f[i] = orig * m if orig > 1e-14 else (0.01 if m > 0 else 0.0)
                if f.sum() < 1e-10:
                    f[i] = orig
                    continue
                M = quick_M(f)
                # We need to account for the normalization
                # Actually, the score is 2n*M/S^2, and if we change one element, S changes
                # For the normalized version, we just look at M after normalization
                # But for speed, let's just look at M without re-normalizing
                # (normalization is a constant factor)
                if M < best_val_M - 1e-14:

                    best_val = f[i]

    f = f / f.sum()
