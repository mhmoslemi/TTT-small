```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _score(a):
    """Compute the score using FFT for speed."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    b = irfft(F*F, fsize)[:2*n-1]
    np.maximum(b, 0, out=b)
    m = float(b.max())
    return 2.0*n*m/(s*s)

def _conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    b = irfft(F*F, fsize)[:2*n-1]
    np.maximum(b, 0, out=b)
    return b

def _build_A(f, n):
    """Build the constraint matrix for the LP."""
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_direction(f, n, M):
    """Solve the LP to find the best direction g."""
    A = _build_A(f, n)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status != 0:
        return None
    g = res.x
    gs = g.sum()
    if gs < 1e-14:
        return None
    return g / gs  # normalize to sum 1

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()  # ensure exact sum=1
```
