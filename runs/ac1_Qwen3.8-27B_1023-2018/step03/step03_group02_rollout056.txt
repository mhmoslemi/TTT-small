def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):

def _build_A_rows(f, n, idx):
    """Build rows of the convolution matrix for specified indices."""
    rows = []
    for k in idx:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            row[j] = f[k - j]
        rows.append(row)
    return np.array(rows)


def _lp_focused(f, n, M, top_k=100, time_limit=5.0):

    c = -np.ones(n)
    try:

            c, A_ub=A, b_ub=np.full(len(sorted_idx), M),

        )

        return None

def _get_parent():

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A_subset(f, n, idx):
    """Build constraint matrix rows for given convolution indices."""
    rows = np.zeros((len(idx), n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            rows[i, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    return rows


def _lp_direction(f, n, M, top_k, time_limit=5.0):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for top-k positions, g>=0."""
    b = _conv_full(f)
    sorted_idx = np.argsort(-b)[:top_k]
    A = _build_A_subset(f, n, sorted_idx)
    c = -np.ones(n)
    try:
        res = linprog(
            c, A_ub=A, b_ub=np.full(len(sorted_idx), M),
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum

def _subgradient_step(f, n, top_positions=10):
    """Compute a subgradient descent direction for the score."""
    b = _conv_full(f)
    S = float(f.sum())
    S2 = S * S
    top_idx = np.argsort(-b)[:top_positions]
    # Average subgradient from top positions
    grad = np.zeros(n)
    for k in top_idx:
        conv_k = b[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # d(conv_k)/d(f_j) = 2*f_{k-j} for j in [lo, hi]
        f_window = f[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * f_window
        # -2*S contribution (from d(1/S^2)/dS = -2/S)
        grad += 0  # handled separately
    # Full subgradient of score = 2n*max_conv/S^2:
    # d/d(f_j) = 2n * (d max_conv/d f_j) / S^2 - 2n * max_conv * 2 / S^3
    # Approximate using top positions
    max_b = float(b.max())
    direction = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        f_window = f[k - hi:k - lo + 1][::-1]
        direction[lo:hi + 1] += 2.0 * f_window
    direction /= len(top_idx)
    # Subtract mean to handle the 1/S^2 term (for fixed sum normalization)
    direction = direction - direction.mean()
    # Negate for descent
    return -direction, max_b
