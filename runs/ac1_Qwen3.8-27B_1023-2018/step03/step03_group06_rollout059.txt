This is a QCQP, which is NP-hard in general for indefinite quadratics. But for our specific structure, the quadratics `c_k(a) = sum_{i+j=k} a_i a_j` are actually **copositive** on the nonnegative orthant (since `a_i a_j >= 0` for `a >= 0`). So the problem might be solvable.

`c_k(a) = sum_{i+j=k} a_i a_j`. The Hessian has entries `∂²c_k/∂a_i∂a_j = 2` if `i+j=k` and `i≠j`, and `1` if `2i=k`. Wait, the Hessian of `c_k` is a matrix `H_k` where `H_k[i,j] = 2` if `i+j=k` and `i≠j`, `H_k[i,i] = 1` if `2i=k`, else 0.

Is `H_k` positive semidefinite? Let's see: `x^T H_k x = 2 * sum_{i+j=k, i<j} x_i x_j + sum_{2i=k} x_i^2`.

This is `sum_{i+j=k} x_i x_j + (sum_{i+j=k, i<j} x_i x_j + sum_{2i=k} x_i^2)`.

Hmm, it's not obviously PSD. For example, if `k=2`, `c_2 = a_0 a_2 + a_1^2 + a_2 a_0 = 2a_0 a_2 + a_1^2`. The Hessian is `[[0, 0, 2], [0, 2, 0], [2, 0, 0]]`. Eigenvalues: 2, 2, -2. Not PSD!

So the problem is NOT convex. The QCQP is non-convex.

However, on the nonnegative orthant, `c_k(a) >= 0` and the behavior is different. But the non-convexity remains.

So we can't use a convex solver directly. The LP approach (linearizing one side) is a valid outer approximation.

Let me think about the step size more carefully. The convolution of the blended sequence is:

- At the crossing point, the max is determined by the two crossing quadratics
- The minimum of the max occurs either at a crossing point or at a single-quadratic minimum

import math

def _fft_len(m):
    m = int(max(m, 1))
    if m not in _fft_cache:
        _fft_cache[m] = int(next_fast_len(m))
    return _fft_cache[m]

def _conv_f(a):
    """Full auto-convolution."""
    n = len(a)
    if n <= 1:
        return a * a
    m = 2 * n - 1
    L = _fft_len(m)
    fa = rfft(a, L)
    return np.maximum(irfft(fa * fa, L)[:m], 0.0)

    """Cross convolution."""
    n1, n2 = len(a), len(b)
    m = n1 + n2 - 1

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize

class _TO(Exception): pass

_fft_cache = {}
def _fL(m):
    m = int(max(m, 1))
    if m not in _fft_cache:
        _fft_cache[m] = int(next_fast_len(m))
    return _fft_cache[m]

def _acf(a):
    """Auto-convolution (linear)."""
    n = len(a)
    if n <= 1:
        return a * a
    L = _fL(2 * n - 1)
    fa = rfft(a, L)
    return np.maximum(irfft(fa * fa, L)[:2*n-1], 0.0)

def _cconv(a, b):
    """Cross-convolution (linear)."""
    L = _fL(len(a) + len(b) - 1)
    return np.maximum(irfft(rfft(a, L) * rfft(b, L), L)[:len(a)+len(b)-1], 0.0)

def _score(a):
    """AC1 score."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return 1e18
    return 2.0 * float(len(a)) * float(_acf(a).max()) / (s * s)

def _eval_exact(lst):
    """Exact evaluation using the provided evaluator."""
    try:
        r = float(evaluate_sequence([float(x) for x in lst]))
        return r
    except Exception:
        return 1e18

def _norm(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0), 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / max(1, len(a))
    return a / s

def _lp_step(f, n, time_limit=3.0):
    """LP: find g in simplex minimizing max_k conv(f, g)[k]."""
    A = np.zeros((2 * n - 1, n + 1), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    A[:, n] = -1.0
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize free_var
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.zeros(2 * n - 1),
                      A_eq=A_eq, b_eq=np.array([1.0]),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    
    if res is None or getattr(res, 'status', 1) != 0:
        return None
    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.clip(g, 0.0, None)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs

def _best_t(cf, cfs, cg):
    """Best step size t in [0,1] for blending (1-t)*f + t*g."""
    a0 = cf
    b0 = 2.0 * (cfs - cf)
    c0 = cf - 2.0 * cfs + cg
    
    # Coarse grid
    ts = np.linspace(0.0, 1.0, 101)
    vals = np.max(a0[None, :] + b0[None, :] * ts[:, None] + c0[None, :] * ts[:, None] ** 2, axis=1)
    i = int(np.argmin(vals))
    
    # Fine grid around best
    lo = ts[max(0, i - 2)]
    hi = ts[min(len(ts) - 1, i + 2)]
    ts2 = np.linspace(lo, hi, 201)
    vals2 = np.max(a0[None, :] + b0[None, :] * ts2[:, None] + c0[None, :] * ts2[:, None] ** 2, axis=1)
    return float(ts2[int(np.argmin(vals2))])

def _lp_loop(f_init, stop_time, seed=0, max_noimp=30):
    """LP-based optimization loop."""
    rng = np.random.default_rng(seed)
    f = _norm(f_init).copy()
    n = len(f)
    best_f = f.copy()
    best_sc = _score(f)
    noimp = 0
    
    while time.monotonic() < stop_time:
        # Compute LP direction
        fp = np.maximum(f, 1e-14)
        fp = fp / float(fp.sum())
        
        g = _lp_step(fp, n)
        if g is None:
            break
        
        # Compute convolutions
        cf = _acf(f)
        cfs = _cconv(f, g)
        cg = _acf(g)
        
        # Find best step
        t_opt = _best_t(cf, cfs, cg)
        
        # Try candidate step sizes
        cands = list({
            max(0.0, min(1.0, t_opt)),
            max(0.0, min(1.0, 0.5 * t_opt)),
            max(0.0, min(1.0, 1.5 * t_opt)),
            0.01, 0.05, 0.1, 0.2, 0.5
        })
        cands.sort()
        
        best_local = f.copy()
        best_local_sc = best_sc
        
        for t in cands:
            nf = (1.0 - t) * f + t * g
            nf = np.clip(nf, 0.0, None)
            s = float(nf.sum())
            if s < 1e-15:
                continue
            nf = nf / s
            sc = _score(nf)
            if sc < best_local_sc:
                best_local_sc = sc
                best_local = nf.copy()
        
        if best_local_sc < best_sc - 1e-13:
            f = best_local
            best_f = f.copy()
            best_sc = best_local_sc
            noimp = 0
        else:
            noimp += 1
            # Add noise
            sigma = 0.3 * float(np.mean(f)) * (1.0 + 0.15 * min(float(noimp), 12.0))
            f = f + rng.standard_normal(n) * sigma
            f = np.clip(f, 0.0, None)
            f = f / max(float(f.sum()), 1e-15)
            sc = _score(f)
            if sc < best_sc:
                best_f = f.copy()
                best_sc = sc
                noimp = 0
            if noimp >= max_noimp:
                break
    
    return best_f, best_sc

def _subgrad_step(f, eta):
    """One subgradient step on the simplex."""
    n = len(f)
    c = _acf(f)
    kmax = int(np.argmax(c))
    # Subgradient of max_k c_k at f: 2 * P_{kmax} f
    grad = np.zeros(n)
    for j in range(n):
        k_j = kmax - j
        if 0 <= k_j < n:
            grad[j] = 2.0 * f[k_j]
    # Projected gradient step onto simplex
    f_new = f - eta * grad
    f_new = _norm(f_new)
    return f_new

def _lbfgs_refine(f_init, stop_time, eta=100.0, maxiter=200):
    """L-BFGS refinement using smoothed max."""
    f0 = _norm(f_init)
    n = len(f0)
    best_f = f0.copy()
    best_sc = _score(f0)
    
    if n < 2:
        return best_f, best_sc
    
    # Use softmax parameterization
    y0 = np.zeros(n)
    
    c_init = _acf(f0)
    M_init = float(c_init.max())
    if M_init < 1e-15:
        return best_f, best_sc
    beta = eta / M_init
    
    def obj_grad(y):
        if time.monotonic() > stop_time:
            raise _TO
        y = np.asarray(y, dtype=np.float64)
        if not np.all(np.isfinite(y)):
            return 1e100, np.zeros(n)
        
        # Softmax
        ym = y - float(y.max())
        e = np.exp(ym)
        Z = float(e.sum())
        if Z < 1e-300:
            return 1e100, np.zeros(n)
        a = e / Z
        
        c = _acf(a)
        bc = beta * c
        bc_max = float(bc.max())
        bc_s = bc - bc_max
        w = np.exp(bc_s)
        S = float(w.sum())
        if S < 1e-300:
            w = np.ones(len(c)) / len(c)
            S = float(len(c))
        w = w / S
        
        fun = (1.0 / beta) * (bc_max + math.log(S))
        
        # Gradient computation
        # dF/dy_j = sum_k w_k * dc_k/dy_j
        # dc_k/dy_j = dc_k/da_j - a_j * sum_i dc_k/da_i * a_i
        #           = dc_k/da_j - 2*a_j*c_k  (by Euler: sum_i dc_k/da_i * a_i = 2*c_k)
        # dc_k/da_j = 2*a_{k-j} if 0 <= k-j < n, else 0
        
        grad = np.zeros(n)
        for j in range(n):
            k_start = j
            k_end = j + n  # exclusive
            if k_start >= len(c):
                continue
            k_end = min(k_end, len(c))
            # dc_k/da_j = 2*a[k-j] for k in [j, j+n-1]
            dk = 2.0 * a[0:k_end - k_start]
            # full vector
            full_dk = np.zeros(len(c))
            full_dk[k_start:k_end] = dk
            grad[j] = float(np.dot(w, full_dk - 2.0 * a[j] * c))
        
        # Remove mean (gauge)
        grad -= float(grad.mean())
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
        
        return float(fun), grad
    
    try:
        res = minimize(obj_grad, y0, jac=True, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-13, 'gtol': 1e-9, 'maxfun': maxiter * 3})
        if hasattr(res, 'x') and res.x is not None:
