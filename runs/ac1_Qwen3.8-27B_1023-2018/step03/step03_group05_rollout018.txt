def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.full(a.size, target / max(1, a.size))
    return a * (target / s)

def _conv(a):
    n = a.size
    if n == 0:
        return np.array([0.0])
    nfft = next_fast_len(2 * n - 1)
    f = rfft(a, nfft)
    c = irfft(f * f, nfft)[:2 * n - 1]
    return np.where(np.isfinite(c), c, 0.0)

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float('inf')
    c = _conv(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return float(2 * a.size * c.max() / s**2)

def _proj_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    with np.errstate(divide='ignore', invalid='ignore'):
        mask = (css - target) / idx < u
    rho = np.nonzero(mask)[0]
    if rho.size == 0:
        return np.full(n, target / n)
    theta = (css[rho[-1]] - target) / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)

def _build_conv_matrix(a):
    """Build the matrix A such that (A @ g)[k] = (a * g)[k] for all k."""
    n = a.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            A[k, i_lo:i_hi+1] = a[k - np.arange(i_lo, i_hi+1)]
    return A

def _lp_direction(a, M, deadline, rng):
    """LP: max sum(g) s.t. A@x * g <= M, g >= 0, where A is the conv matrix."""
    n = a.size
    rem = deadline - time.monotonic()
    if rem < 1.0 or n > 1024 or M <= 0:
        return None
    
    A = _build_conv_matrix(a)
    A_s = A / M
    m = 2 * n - 1
    tie = rng.standard_normal(n)
    tie = (tie - tie.mean()) / (np.std(tie) + 1e-12)
    c_obj = -1.0 + 1e-4 * tie
    b_ub = np.ones(m)
    
    try:
        res = linprog(c_obj, A_ub=A_s, b_ub=b_ub, bounds=(0, None), method='highs',
                      options={'time_limit': max(0.1, min(5.0, 0.1 * rem))})
        if res.success:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None

k=2: A_2 = [[0,0,1],[0,0,0],[1,0,0]], which has eigenvalues {1,0,-1}. Not PSD!

So the objective is not convex in general. This means the QP might have multiple local minima. Projected gradient might get stuck.

Given this, the LP approach is safer because it always gives a valid direction.

Actually, I just realized something important. The LP:
max sum(g) s.t. (y*g)[k] <= M for all k, g >= 0

The optimal solution g* is a vertex of the feasible polytope. The feasible polytope is defined by (2n-1) inequalities and n nonnegativity constraints. A vertex has at most (2n-1) active constraints (since we have n variables, a vertex is determined by n linearly independent active constraints).

The direction g* tells us where to "push" the sequence. The idea is: by moving y towards g*, we reduce the peaks of y*y while maintaining feasibility.

The key question: how to choose the step size? The parent uses a fixed set of alphas. A better approach: binary search for the largest alpha such that y + alpha*(g*-y) still has max(y*y) <= M (or within some tolerance).

Actually, that's not quite right. We want to minimize max((y + alpha*(g-y)) * (y + alpha*(g-y))). This is a function of alpha that we can evaluate. Since it's the max of quadratic functions of alpha, it's a convex piecewise-quadratic function. We can minimize it with golden section search or by checking the vertices.

- Evaluate f(alpha) = max_conv(y + alpha*(g-y)) for alpha in [0, 1].
- Find the minimum using golden section search (about 40 evaluations).

        if aa2 is not None and np.isfinite(sc2) and sc2 < best_sc:
            best_sc, best_a = sc2, aa2
    return best_a, best_sc

# ── main entry point ────────────────────────────────────────────────────

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(0.0, float(budget_s))
    except Exception:
        budget = 1300.0

    if budget < 3.0:
        p = _get_parent()
        return [float(x) for x in _norm(p)]

    head = max(8.0, min(30.0, 0.02 * budget))
    dline = t0 + budget - head

    rng = np.random.default_rng(
        int(seed) if np.isfinite(float(seed)) else 42)

    parent = _get_parent()
    best_a = _norm(parent)
    best_sc = _exact(best_a)
    if not np.isfinite(best_sc):
        best_a = np.ones(64) / 64.0
        best_sc = _exact(best_a)

    # Phase 1 – basis search for new shape
    if budget > 250 and dline - time.monotonic() > 140:
        bt = min(55.0, max(18.0, 0.04 * budget))
        bd = time.monotonic() + bt
        try:
            ba, bs = _basis(256, bd, rng)
            if ba is not None and np.isfinite(bs) and bs < best_sc:
                best_a, best_sc = _norm(ba), bs
        except Exception:
            pass

    # Phase 2 – multi-length LP + subgradient refinement
    Ls = []
    for L in (256, 384, 512, 640):
        if L not in Ls:
            Ls.append(L)
    pL = best_a.size
    if pL not in Ls and 16 <= pL <= 1024:
        Ls.append(pL)

    wts = {}
    for L in Ls:
        w = {256: 1.0, 384: 1.2, 512: 1.1, 640: 0.7}.get(L, 0.5)
        if L == pL:
            w += 0.4
        if L > 768:
            w *= 0.4
        if L < 64:
            w *= 0.5
        wts[L] = w

    tot = sum(wts.values()) or 1.0
    phase = max(0.0, (dline - time.monotonic()) * 0.82)

    for L in Ls:
        rem = dline - time.monotonic() - 20.0
        if rem < 15.0:
            break
        tL = (wts[L] / tot) * phase
        tL = min(tL, rem - 5.0)
        if tL < 12.0:
            continue

        Lint = int(L)
        if Lint < 2:
            continue

        # main run from current best
        tA = min(tL * 0.72, dline - time.monotonic() - 25.0)
        if tA > 10.0:
            try:
                init = _resample(best_a, Lint)
                ra, sa = _refine(init, time.monotonic() + tA, rng)
                if ra is not None and np.isfinite(sa) and sa < best_sc:
                    best_a, best_sc = _norm(ra), sa
            except Exception:
                pass

        # exploratory run
        rem2 = dline - time.monotonic() - 25.0
        tB = min(tL * 0.25, rem2)
        if tB > 12.0:
            try:
                ini = _rand_smooth(Lint, rng, base=_resample(best_a, Lint))
                rb, sb = _refine(ini, time.monotonic() + tB, rng)
                if rb is not None and np.isfinite(sb) and sb < best_sc:
                    best_a, best_sc = _norm(rb), sb
            except Exception:
                pass

    # Phase 3 – final polish
    rem = dline - time.monotonic()
    if rem > 30.0:
        try:
            rp, sp = _refine(best_a, time.monotonic() + min(rem - 12.0, 250.0), rng)
            if rp is not None and np.isfinite(sp) and sp < best_sc:
                best_a, best_sc = _norm(rp), sp
        except Exception:
            pass

    # Final validation
    try:
        out = _norm(best_a)
        fs = _exact(out)
        if not np.isfinite(fs):
            fb = np.ones(64) / 64.0
            return [float(x) for x in _norm(fb)]
        if fs < best_sc:
            best_sc = fs
        return [float(x) for x in out]
    except Exception:
        return [float(x) for x in _norm(np.ones(64))]
