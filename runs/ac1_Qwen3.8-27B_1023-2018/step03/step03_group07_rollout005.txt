        if rng.random() < 0.5:

def _to_valid_array(x):
    try:
        arr = np.asarray(x, dtype=np.float64).ravel()
    except Exception:
        return np.ones(1, dtype=np.float64)
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    return arr


def _normalize(a):
    arr = _to_valid_array(a)
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def _get_parent():
    val = globals().get("height_sequence_1", None)
    if val is None:

def _conv(u, v=None):
    u = np.asarray(u, dtype=np.float64).ravel()
    if v is None:
        v = u
    v = np.asarray(v, dtype=np.float64).ravel()
    nu = u.size
    nv = v.size
    if nu == 0 or nv == 0:
        return np.zeros(0, dtype=np.float64)
    if nu * nv <= 200000:
        c = np.convolve(u, v)
    else:
        L = next_fast_len(nu + nv - 1)
        c = irfft(rfft(u, L) * rfft(v, L), L)[:nu + nv - 1]
    c = np.asarray(c, dtype=np.float64)
    if c.size:
        c = np.where(c > 0.0, c, 0.0)
    return c


def _M(a):
    c = _conv(a)
    if c.size == 0:
        return float("inf")

def _corr_valid(w, a):
    w = np.asarray(w, dtype=np.float64).ravel()
    a = np.asarray(a, dtype=np.float64).ravel()
    nw = w.size
    na = a.size
    if na == 0:
        return np.zeros(0, dtype=np.float64)
    if nw < na:
        w = np.concatenate((w, np.zeros(na - nw, dtype=np.float64)))
        nw = na
    valid_len = nw - na + 1
    if valid_len <= 0:
        return np.zeros(0, dtype=np.float64)

    if na <= 1500 or nw * na <= 500000:
        out = np.correlate(w, a, mode="valid")
        out = out[:na].astype(np.float64, copy=False)
        if out.size < na:
            out = np.concatenate((out, np.zeros(na - out.size, dtype=np.float64)))
        return out

    out = irfft(rfft(w, L) * rfft(a[::-1], L), L)
    start = na - 1
    res = out[start:start + na].astype(np.float64, copy=True)
    if res.size < na:

    return res


def _get_builder(n):
    n = int(n)
    if n in _BUILDER_CACHE:
        return _BUILDER_CACHE[n]
    m = 2 * n - 1
    rows = np.empty(n * n, dtype=np.int64)
    cols = np.empty(n * n, dtype=np.int64)
    idx = np.empty(n * n, dtype=np.int64)
    p = 0
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        cnt = hi - lo + 1

        cols[p:p + cnt] = js
        idx[p:p + cnt] = k - js
        p += cnt

        cols = cols[:p]
        idx = idx[:p]

def _lp_minmax(a, builder, time_limit):
    m, rows, cols, idx = builder
    n = a.size
    A = np.zeros((m, n + 1), dtype=np.float64)
    if rows.size:
        A[rows, cols] = a[idx]
    A[:, n] = -1.0

    obj = np.zeros(n + 1, dtype=np.float64)
    obj[n] = 1.0
    Aeq = np.ones((1, n + 1), dtype=np.float64)
    Aeq[0, n] = 0.0
    b_eq = np.array([1.0], dtype=np.float64)
    bounds = [(0.0, None)] * n + [(0.0, None)]
    b_ub = np.zeros(m, dtype=np.float64)

    res = _run_linprog(obj, A, b_ub, Aeq, b_eq, bounds, time_limit)
    if res is None or getattr(res, "x", None) is None:
        return None, float("inf")

    x = np.asarray(res.x, dtype=np.float64)
    if x.size != n + 1 or not np.all(np.isfinite(x)):
        return None, float("inf")

    g = np.maximum(x[:n], 0.0)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None, float("inf")
    g /= s
    t = float(x[n])
    if not np.isfinite(t):
        t = float("inf")
    return g, max(0.0, t)


def _lp_maxsum(a, builder, M, time_limit):
    m, rows, cols, idx = builder
    n = a.size
    A = np.zeros((m, n), dtype=np.float64)
    if rows.size:
        A[rows, cols] = a[idx]
    obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(m, max(_M(a), 0.0) * (1.0 + 1e-7) + 1e-12, dtype=np.float64)
    b_ub = np.full(m, M * (1.0 + 1e-7) + 1e-12, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = _run_linprog(obj, A, b_ub, None, None, bounds, time_limit)
    if res is None or getattr(res, "x", None) is None:
        return None
    x = np.asarray(res.x, dtype=np.float64)

        return None
    g = np.maximum(x, 0.0)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    return g / s


def _line_direct(a, g, M_a):

    if g.size != a.size:
        return None
    if not (np.all(np.isfinite(a)) and np.all(np.isfinite(g))):
        return None
    if np.max(np.abs(g - a)) < 1e-13:
        return None

    bestM = float(M_a)
    bestC = None
    for al in (0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3, 0.5, 0.75, 1.0):

        Mn = _M(cand)

            + (alphas * alphas)[:, None] * c_g[None, :]
        )
        return np.max(terms, axis=1)

    span = max(1, 8)
    lo = float(alphas[max(0, i - span)])
    hi = float(alphas[min(len(alphas) - 1, i + span)])
    alphas2 = np.linspace(lo, hi, 401)
    vals2 = maxvalues(alphas2)
    j = int(np.argmin(vals2))
    if float(vals2[j]) < val_best:
        alpha_best = float(alphas2[j])
        val_best = float(vals2[j])

    candidates = [alpha_best, lo, hi, 0.0, 1.0]
    if vals.size:
        top = np.argsort(vals)[:5]
        for tt in top:
            candidates.append(float(alphas[tt]))
    for d in (-0.03, -0.02, -0.01, -0.005, 0.0, 0.005, 0.01, 0.02, 0.03):
        candidates.append(alpha_best + d)

    unique = []
    seen = set()
    for al in candidates:
        al = float(al)
        if not (0.0 <= al <= 1.0):
            continue
        key = round(al, 12)
        if key not in seen:
            seen.add(key)
            unique.append(al)

    bestM = float(M_a)
    bestC = None
    for al in unique:
        cand = (1.0 - al) * a + al * g
        cand = np.maximum(cand, 0.0)
        s = float(cand.sum())
        if s <= 1e-12:
            continue
        cand /= s
        Mn = _M(cand)
        if Mn < bestM - _MOVE_TOL:
            bestM = Mn
            bestC = cand
    return bestC


def _gradient_step(a, c_a, M, tau):
    if M <= 0 or not np.isfinite(M):
        return None

    if tau is None or tau <= 0 or not np.isfinite(tau):
        active = c_a >= M - max(1e-12, 1e-5 * M)
        if not active.any():
            active = np.zeros_like(c_a, dtype=bool)
            active[int(np.argmax(c_a))] = True
        w = active.astype(np.float64)
        s = float(w.sum())
        if s <= 0.0:
            return None
        w /= s
    else:
        tau = max(float(tau), 1e-15)
        z = (c_a - M) / tau
        e = np.exp(np.clip(z, -700.0, 700.0))
        s = float(e.sum())
        if s <= 0.0 or not np.isfinite(s):
            return None
        w = e / s

    grad = 2.0 * _corr_valid(w, a)
    if grad.size != a.size or not np.all(np.isfinite(grad)):
        return None
    grad = grad - float(np.mean(grad))
    amax = float(np.max(np.abs(grad)))
    if not np.isfinite(amax) or amax <= 1e-14:
        return None

    step = -grad / amax
    bestM = M
    bestC = None
    best_alpha = 0.0

    alphas = (
        2e-4, 5e-4, 1e-3, 2e-3, 5e-3,
        1e-2, 2e-2, 5e-2, 1e-1, 2e-1, 4e-1
    )
    for al in alphas:
        cand = a - al * step
        cand = np.maximum(cand, 0.0)
        s = float(cand.sum())
        if s <= 1e-12:
            continue
        cand /= s
        Mn = _M(cand)
        if Mn < bestM - _MOVE_TOL:
            bestM = Mn
            bestC = cand
            best_alpha = al

    if bestC is None:
        return None

    lo = max(0.0, best_alpha - 0.02)
    hi = min(0.5, best_alpha + 0.02)
    for al in np.linspace(lo, hi, 31):
        cand = a - al * step
        cand = np.maximum(cand, 0.0)
        s = float(cand.sum())
        if s <= 1e-12:
            continue
        cand /= s
        Mn = _M(cand)
        if Mn < bestM - _MOVE_TOL:
            bestM = Mn
            bestC = cand

    return bestC


def _structured_init(n, typ, rng):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    t = int(typ) % 12

    if t == 0:
        y = np.ones(n, dtype=np.float64)
    elif t == 1:
        beta = float(rng.uniform(0.5, 6.0))
        y = np.exp(-beta * x)
    elif t == 2:
        beta = float(rng.uniform(0.5, 6.0))
        y = np.exp(-beta * (1.0 - x))
    elif t == 3:
        c = float(rng.uniform(0.2, 0.9))
        phi = float(rng.uniform(0.0, 1.0))
        wave = 1 if float(rng.random()) < 0.5 else 2
        if float(rng.random()) < 0.5:
            y = 1.0 + c * np.sin(2.0 * np.pi * wave * (x - 0.25))
        else:
            y = 1.0 + c * np.cos(2.0 * np.pi * wave * (x - phi))
    elif t == 4:
        a1 = float(rng.uniform(0.05, 0.70))
        a2 = float(rng.uniform(0.05, 0.70))
        y = np.power(x + 1e-3, -a1) * np.power(1.0 - x + 1e-3, -a2)
    elif t == 5:
        c1, c2 = np.sort(rng.uniform(0.1, 0.9, size=2))
        if c1 == c2:
            c2 = min(0.95, c1 + 0.2)
        w1 = float(0.03 + rng.uniform(0.0, 0.15))
        w2 = float(max(0.03, w1 * rng.uniform(0.4, 2.0)))
        amp2 = float(rng.uniform(0.3, 1.2))
        y = np.exp(-((x - c1) / w1) ** 2) + amp2 * np.exp(-((x - c2) / w2) ** 2)
    elif t == 6:
        y = np.abs(rng.normal(size=n)) + 0.1
        k = max(3, int(np.sqrt(n)))
        if k > n:
            k = n
        kernel = np.ones(k, dtype=np.float64) / float(k)
        y = np.convolve(y, kernel, mode="same")
        y = np.maximum(y, 1e-6)
    elif t == 7:
        dh = np.abs(rng.normal(size=n)) * 0.1 + 1e-6
        y = np.cumsum(dh)
        y = np.maximum(y, 1e-8)
    elif t == 8:
        y = np.maximum((1.0 - x) ** 3.0, 1e-8)
    elif t == 9:
        y = np.maximum(x ** 3.0, 1e-8)
    elif t == 10:
        y = np.ones(n, dtype=np.float64)
        y[int(0.05 * n):int(0.55 * n)] += float(rng.uniform(0.3, 1.0))
        y[int(0.55 * n):int(0.95 * n)] += float(rng.uniform(-0.5, 0.2))
        y = np.maximum(y, 1e-6)
    else:
        y = np.maximum(np.exp(-((x - 0.35) ** 2) / 0.02)
                       + 0.7 * np.exp(-((x - 0.75) ** 2) / 0.02), 1e-6)

    y = np.where(np.isfinite(y), y, 0.0)
    y = np.maximum(y, 0.0)
    s = float(y.sum())
    if not np.isfinite(s) or s <= 1e-12:
        y = np.ones(n, dtype=np.float64)
        s = float(y.sum())
    return y / s


def _focus_refine(a0, stop_time, seed=0, use_lp=True):
    a = _normalize(a0)
    n = a.size
    if n == 0:
        a = np.ones(1, dtype=np.float64)
        n = 1

    best = a.copy()
    bestM = _M(a)
    if not np.isfinite(bestM):
        best = np.ones(n, dtype=np.float64) / float(n)
        bestM = _M(best)

    rng = np.random.default_rng(seed)
    do_lp = bool(use_lp) and n <= 420
    builder = _get_builder(n) if do_lp else None
    lp_time = 0.15 if n <= 300 else 0.05

    it = 0
    no_best = 0
    restart_level = 0
    max_iter = 1000000

    while it < max_iter:
        now = time.monotonic()
        if now >= stop_time - 0.15:
            break
        it += 1

        c_a = _conv(a)
        M = float(np.max(c_a)) if c_a.size else float("inf")
        if not np.isfinite(M):
            a = best.copy()
            continue

        if M < bestM - _MOVE_TOL:
            bestM = M
            best = a.copy()
            no_best = 0
        else:
            no_best += 1

        if no_best > 120:
            restart_level += 1
            scale = 0.03 + 0.02 * min(restart_level, 5)
            a = best.copy() * np.exp(rng.normal(0.0, scale, n))
            a = np.maximum(a, 1e-10)
            a /= float(a.sum())
            no_best = 0
            continue

        improved = False

        tau_frac = 0.01 * (0.95 ** min(no_best, 100))
        tau = max(1e-9, M * tau_frac)
        cand = _gradient_step(a, c_a, M, tau)
        if cand is not None:
            a = cand
            improved = True

        if not improved and builder is not None:
            rem = stop_time - time.monotonic()
            tl = min(lp_time, max(0.01, rem / 5.0))
            if rem > tl + 0.1:
                g, _ = _lp_minmax(a, builder, tl)
                if g is not None:
                    cand = _line_combo(a, g, c_a, M)
                    if cand is not None:
                        a = cand
                        improved = True

        if not improved:
            cand = _gradient_step(a, c_a, M, 0.0)
            if cand is not None:
                a = cand
                improved = True

        if not improved and builder is not None and (it % 25 == 0 or (no_best > 10 and no_best % 15 == 0)):
            rem = stop_time - time.monotonic()
            tl = min(lp_time, max(0.01, rem / 5.0))
            if rem > tl + 0.1:
                g = _lp_maxsum(a, builder, M, tl)
                if g is not None:
                    cand = _line_combo(a, g, c_a, M)
                    if cand is not None:
                        a = cand
                        improved = True

        if not improved:
            scale = 0.01 + 0.004 * min(20, no_best // 5)
            a = a * np.exp(rng.normal(0.0, scale, n))
            a = np.maximum(a, 1e-10)
            a /= float(a.sum())

    return best


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 1300.0
    if budget < 20:
        budget = max(budget, 20.0)

    rng = np.random.default_rng(seed)
    hard_deadline = t0 + max(10.0, budget - 20.0)

    parent = _get_parent()
    if parent is not None:
        base = _normalize(parent)
    else:
        base = np.ones(201, dtype=np.float64) / 201.0

    best_seq = base.copy()
    best_score = _score_any(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(16, dtype=np.float64) / 16.0
        best_score = 2.0

    def update_best(candidate):
        nonlocal best_seq, best_score
        try:
            arr = _to_valid_array(candidate)
            if arr.size == 0:
                return
            sc = _score_any(arr)
            if np.isfinite(sc) and sc < best_score - _SCORE_TOL:
                best_score = sc
                best_seq = _normalize(arr).copy()
        except Exception:
            pass

    phase1_stop = t0 + min(budget * 0.50, 650.0)
    phase1_stop = min(phase1_stop, hard_deadline - 30.0)
    if phase1_stop < t0 + 5.0:
        phase1_stop = t0 + max(1.0, min(5.0, hard_deadline - t0))

    res = _focus_refine(best_seq.copy(), phase1_stop, seed=123456, use_lp=True)
    update_best(res)
    if time.monotonic() > hard_deadline - 5.0:
        return [float(x) for x in np.clip(best_seq, 0.0, 1000.0)]

    phase2_stop = t0 + min(budget * 0.85, 1080.0)
    phase2_stop = min(phase2_stop, hard_deadline - 10.0)
    if phase2_stop <= phase1_stop:
        phase2_stop = 0.5 * (phase1_stop + hard_deadline - 10.0)

    if time.monotonic() < phase2_stop - 15.0:
        base_n = int(best_seq.size)
        tasks = []

        for t in (1, 3, 4, 5, 7):
            tasks.append(("mix_best", int(t)))

        for L in (base_n - 20, base_n + 20, 256, 128, 321):
            if 32 <= L <= 360 and L != base_n:
                tasks.append(("resample_length", int(L)))

        for t in (0, 3, 4, 5, 10):
            tasks.append(("structured", int(t)))

        task_idx = 0
        while task_idx < len(tasks):
            now = time.monotonic()
            rem = phase2_stop - now - 10.0
            if rem < 12.0:
                break

            left = max(1, len(tasks) - task_idx)
            per = max(8.0, min(120.0, rem / float(left)))
            if left > 8:
                per = min(per, 35.0)
            elif left > 3:
                per = min(per, 60.0)

            kind, val = tasks[task_idx]
            ncur = int(best_seq.size)
            if ncur < 8:
                ncur = 128
                x_old = np.linspace(0.0, 1.0, max(1, best_seq.size), dtype=np.float64)
                x_new = np.linspace(0.0, 1.0, ncur, dtype=np.float64)
                best_seq = _normalize(np.interp(x_new, x_old, best_seq))
                best_score = _score_any(best_seq)

            try:
                if kind == "mix_best":
                    rnd = _structured_init(ncur, val, rng)
                    init = 0.72 * best_seq + 0.28 * rnd
                elif kind == "resample_length":
                    L = int(val)
                    x_old = np.linspace(0.0, 1.0, max(1, best_seq.size), dtype=np.float64)
                    x_new = np.linspace(0.0, 1.0, L, dtype=np.float64)
                    init = np.interp(x_new, x_old, best_seq)
                else:
                    init = _structured_init(ncur, val, rng)
                init = _normalize(init)

                stop_task = min(now + per, phase2_stop - 5.0)
                if stop_task > now + 6.0:
                    r = _focus_refine(init, stop_task, seed=987654 + task_idx * 31, use_lp=(init.size <= 420))
                    update_best(r)
            except Exception:
                pass

            task_idx += 1
            if time.monotonic() > phase2_stop - 8.0:
                break

    final_stop = hard_deadline - 2.0
    if time.monotonic() < final_stop - 3.0:
        try:
            r = _focus_refine(best_seq.copy(), final_stop, seed=424242, use_lp=(best_seq.size <= 420))
            update_best(r)
        except Exception:
            pass

    final_arr = np.clip(np.asarray(best_seq, dtype=np.float64).ravel(), 0.0, 1000.0)
    final_arr = np.where(np.isfinite(final_arr), final_arr, 0.0)
    s = float(final_arr.sum())
    if final_arr.size == 0 or not np.isfinite(s) or s < 0.01:
        final_arr = np.ones(16, dtype=np.float64) / 16.0
    else:
        final_arr = final_arr / s
        final_arr = np.clip(final_arr, 0.0, 1000.0)

    final_list = [float(x) for x in final_arr]

    try:
        exact = float(evaluate_sequence(final_list))
    except Exception:
        exact = _score_any(final_arr)

    if not np.isfinite(exact):
        final_list = [1.0] * 16
        try:
            exact = float(evaluate_sequence(final_list))
        except Exception:
            exact = 2.0
        if not np.isfinite(exact):
            final_list = [1.0]

    return final_list
