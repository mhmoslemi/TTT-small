```python
def _line_search(f, g, c_f=None, c_s=None, c_g=None):
    """
    Find t in [0,1] minimizing max(((1-t)f+tg)*((1-t)f+tg))
    Returns (best_t, best_a, best_max)
    """
    if c_f is None: c_f = _cfs(f)
    if c_s is None: c_s = _cf2(f, g)
    if c_g is None: c_g = _cfs(g)
    
    b = 2*(c_s - c_f)  # linear coefficient
    d = c_g - 2*c_s + c_f  # quadratic coefficient
    
    def Q(t):
        return np.max(c_f + b*t + d*t*t)
    
    # Coarse grid
    ts = np.linspace(0, 1, 50)
    vals = np.array([Q(t) for t in ts])
    i = argmin(vals)
    
    # Refine
    lo = max(0, ts[max(0,i-2)])
    hi = min(1, ts[min(len(ts)-1, i+2)])
    
    # Golden section
    gr = (1 - 5**0.5) / 2
    a_, b_ = lo, hi
    c_ = b_ - gr*(b_ - a_)
    d_ = a_ + gr*(b_ - a_)
    fc = Q(c_)
    fd = Q(d_)
    for _ in range(40):
        if fc < fd:
            b_ = d_; d_ = c_; fd = fc
            c_ = b_ - gr*(b_ - a_)
            fc = Q(c_)
        else:
            a_ = c_; c_ = d_; fc = fd
            d_ = a_ + gr*(b_ - a_)
            fd = Q(d_)
    
    t_star = (a_ + b_) / 2
    a_star = (1 - t_star) * f + t_star * g
    a_star = _norm(a_star)
    return t_star, a_star, Q(t_star)
```

```python
    """Compute Φ(softmax(y)) and its gradient w.r.t. y."""
    y = np.asarray(y)
    # Clip for numerical stability
    y = np.clip(y, -30, 30)
    a = np.exp(y - y.max())
    a = a / a.sum()

    # Softmax weights
    bc = beta * c - beta * c.max()
    w = np.exp(bc)
```

```python
def _lp_dir2(f, c_f, stop=None):
    """
    Solve LP2: min t s.t. 2*(f*g)[k] - c_f[k] <= t, sum(g)=1, g>=0
    """
    n = f.size
    f = np.maximum(f, 1e-14)
    f = f / f.sum()
    
    m = 2*n - 1
    A = np.zeros((m, n+1))
    for j in range(n):
        A[j:j+n, j] = 2.0 * f
    A[:, -1] = -1.0
    b_ub = c_f.copy()  # length m
    
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=np.array([1.0]),
                      bounds=[(0,None)]*n + [(None,None)],
                      method='highs', options={'presolve': True, 'time_limit': 3})
    except:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    g = np.clip(res.x[:n], 0, None)
    gs = g.sum()
    if gs < 1e-14:
        return np.ones(n) / n
    return g / gs
```
