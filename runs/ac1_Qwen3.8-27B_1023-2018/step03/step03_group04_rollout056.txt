    F = rfft(a, fsize)

    s = float(g.sum())
    if s < 1e-10:
        return None
    return g/s

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0
    
    parent = _get_parent()
    n = len(parent)
    
    # Initialize from parent
    f = parent.copy().astype(np.float64)
    f = np.clip(f, 0, 1000)
    fs = f.sum()
    if fs < 0.01:
        f = np.ones(384)
        fs = 384.0
    f = f / fs  # normalize to sum=1
    
    best_f = f.copy()
    best_score = _score(f)
    
    # Track with evaluator
    def track(f_candidate):
        nonlocal best_f, best_score
        fc = np.clip(f_candidate, 0, 1000)
        if fc.sum() < 0.01:
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in fc]))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score:
            best_f = fc.copy()
            best_score = sc
    
    track(f)
    
    # Main LP optimization loop
    rng = np.random.default_rng(seed)
    t_step = 0.3
    t_min = 0.01
    t_max = 0.8
    stall = 0
    max_stall = 200
    iteration = 0
    
    while time.monotonic() < deadline - 30.0:
        conv = _conv(f)
        M = float(conv.max())
        
        g = _lp_direction(f, M)
        if g is None:
            stall += 5
            if stall > max_stall:
                # Reset to best with perturbation
                f = np.clip(best_f + rng.normal(0, 0.02, n), 0, None)
                fs = f.sum()
                f = f/fs if fs > 1e-10 else best_f.copy()
                t_step = 0.3
                stall = 0
            continue
        
        dist = float(np.linalg.norm(g - f))
        if dist < 1e-9:
            stall += 3
            if stall > max_stall:
                f = np.clip(best_f + rng.normal(0, 0.02, n), 0, None)
                fs = f.sum()
                f = f/fs if fs > 1e-10 else best_f.copy()
                t_step = 0.3
                stall = 0
            continue
        
        # Line search
        best_t = t_step
        best_sc = _score(f)
        candidate_f = None
        
        t_list = [t_step, t_step*1.3, t_step*0.7, t_step*0.5, t_step*1.8]
        t_list = sorted(set(max(t_min, min(t_max, t)) for t in t_list))
        
        for tc in t_list:
            f_new = (1-tc)*f + tc*g
            f_new = np.clip(f_new, 0, None)
            fs_new = f_new.sum()
            if fs_new < 1e-10:
                continue
            f_new = f_new / fs_new
            sc = _score(f_new)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_t = tc
                candidate_f = f_new.copy()
        
        if candidate_f is not None:
            f = candidate_f
            track(f)
            if best_score < _score(best_f) - 1e-14:  # Actually track already updates
                stall = 0
            if best_f is f:
                t_step = min(t_max, t_step * 1.03)
            else:
                # Check if we improved
                if float(np.linalg.norm(f - best_f)) > 1e-10:
                    t_step = min(t_max, t_step * 1.01)
                else:
                    t_step = max(t_min, t_step * 0.8)
        else:
            stall += 1
            t_step = max(t_min, t_step * 0.7)
            if stall >= max_stall:
                # Perturbation
                strength = 0.01 + 0.03*(stall//max_stall)
                f = np.clip(best_f + rng.normal(0, strength, n), 0, None)
                fs = f.sum()
                f = f/fs if fs > 1e-10 else best_f.copy()
                t_step = 0.2
                stall = 0
        
        iteration += 1
    
    # Polish phase
    polish_deadline = deadline - 5.0
    _polish(best_f, polish_deadline)
    track(best_f)

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_f]))
    except Exception:
        final_sc = math.inf
