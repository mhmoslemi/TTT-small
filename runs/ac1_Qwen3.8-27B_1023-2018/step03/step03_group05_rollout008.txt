        tie = rng.standard_normal(n) * 1e-6
        if n > 1:
            tie = (tie - tie.mean()) / (np.std(tie) + 1e-12)
        c_obj = -1.0 + 1e-4 * tie
        b_ub = np.ones(m)
        tl = max(0.5, min(8.0, 0.3 * rem))
        res = linprog(c_obj, A_ub=A_scaled, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method="highs",
                      options={"time_limit": tl, "presolve": True})
        if res is not None and getattr(res, "success", False):
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if x.size == n and np.all(np.isfinite(x)):
                x = np.maximum(x, 0.0)
                s = float(x.sum())
                if s > 1e-12:
                    return x / s
    except Exception:
        pass
    return None

def _local_optimize(a0, deadline, rng, use_lp=True):
    a = _normalize(a0)
    n = a.size
    if n < 2:
        return a, float("inf")

    c = _conv_full(a)
    M = float(c.max())
    best = a.copy()
    best_M = M
    best_score = _fft_score(a)

    T = M * 0.08
    sigma = 0.02
    patience = 0
    lp_count = 0

    while time.monotonic() < deadline - 2.0:
        patience += 1

        # Compute gradient direction
        w = _softmax_weights(c, T)
        grad = _grad_fft(a, w)
        gnorm = float(np.linalg.norm(grad))

        if gnorm > 1e-14 and patience <= 30:
            # Projected gradient step with line search
            improved = False
            for eta in (0.3, 0.7, 1.5, 3.0, 5.0):
                if time.monotonic() >= deadline - 2.0:
                    break
                cand = _project_simplex(a - eta * grad)
                cc = _conv_full(cand)
                mm = float(cc.max())
                sc = _fft_score(cand)
                if sc < best_score - 1e-12:
                    best = cand.copy()
                    best_M = mm
                    best_score = sc
                    improved = True
                    break
                if mm < M * (1.0 - 1e-10):
                    a = cand
                    c = cc
                    M = mm
                    improved = True
                    break
            if improved:
                a = _normalize(a)
                c = _conv_full(a)
                M = float(c.max())
                if M < best_M:
                    best = a.copy()
                    best_M = M
                    best_score = _fft_score(a)
                    patience = 0
                    T = max(M * 0.03, 1e-12)
                    T = min(T, M * 0.15)
                    sigma = max(0.003, sigma * 0.97)
                continue

        # LP-based direction
        if use_lp and n <= 512 and (patience % 3 == 1) and time.monotonic() < deadline - 5.0:
            lp_count += 1
            dirb = _solve_lp(a, M, deadline - 3.0, rng)
            if dirb is not None:
                improved = False
                for alpha in (0.05, 0.15, 0.3, 0.5, 0.8, 1.0):
                    if time.monotonic() >= deadline - 2.0:
                        break
                    cand = _project_simplex(a + alpha * (dirb - a))
                    cc = _conv_full(cand)
                    mm = float(cc.max())
                    sc = _fft_score(cand)
                    if sc < best_score - 1e-12:
                        best = cand.copy()
                        best_M = mm
                        best_score = sc
                        a = cand
                        c = cc
                        M = mm
                        improved = True
                        break
                if improved:
                    patience = 0
                    T = max(M * 0.03, 1e-12)
                    continue

        # Geometric update
        if gnorm > 1e-14 and patience <= 40:
            gm = float(grad.mean())
            if gm > 1e-16:
                for eta in (0.5, 1.0, 2.0):
                    if time.monotonic() >= deadline - 2.0:
                        break
                    v = a * np.exp(-eta * grad / gm)
                    cand = _normalize(v)
                    cc = _conv_full(cand)
                    mm = float(cc.max())
                    sc = _fft_score(cand)
                    if sc < best_score - 1e-12:
                        best = cand.copy()
                        best_M = mm
                        best_score = sc
                        a = cand
                        c = cc
                        M = mm
                        patience = 0
                        break

        # Stuck: random perturbation
        if patience > 25:
            r = rng.random()
            if r < 0.4:
                a = _project_simplex(a + rng.normal(0, sigma, n))
            elif r < 0.7:
                # Smooth random perturbation
                k = int(rng.integers(1, max(2, n // 32)))
                freq = 2 * np.pi * k / n
                phase = rng.uniform(0, 2 * np.pi)
                pert = sigma * np.cos(freq * np.arange(n) + phase)
                a = _project_simplex(a + pert)
            else:
                # Blend with a smooth shape
                x = np.linspace(0, 1, n)
                alpha = rng.uniform(0.3, 2.0)
                shape = x ** alpha + 1e-6
                a = _project_simplex(0.7 * a + 0.3 * _normalize(shape))
            c = _conv_full(a)
            M = float(c.max())
            sc = _fft_score(a)
            if sc < best_score:
                best = a.copy()
                best_M = M
                best_score = sc
            patience = 0
            sigma = min(0.3, sigma * 1.4)
            T = max(M * 0.04, 1e-12)

        # Anneal temperature
        T = max(M * 0.02, T * 0.995)

    return best, best_score

def _resample(a, m):
    a = _normalize(a)
    n = a.size
    m = int(m)
    if m <= 0:
        return np.ones(1)
    if n == m:
        return a.copy()
    if m > n:
        idx = (np.arange(m) * n) // m
        counts = np.bincount(idx, minlength=n).astype(np.float64)
        counts[counts <= 0] = 1.0
        b = a[idx] / counts[idx]
    else:
        b = np.zeros(m)
        idx = (np.arange(n) * m) // n
        np.add.at(b, idx, a)
        b = np.maximum(b, 0.0)
    return _normalize(b)

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(0.0, float(budget_s))
    except Exception:
        budget = 1300.0

    if budget <= 2.0:
        return [1.0]

    headroom = max(5.0, min(20.0, 0.015 * budget))
    deadline = t0 + budget - headroom

    rng = np.random.default_rng(int(seed) if np.isfinite(float(seed)) else 42)

    parent = _get_parent()
    if parent is None:
        parent = np.ones(192)

    best_a = _normalize(parent)
    best_score = _fft_score(best_a)
    best_exact = _exact_score(best_a)

    if not np.isfinite(best_exact):
        best_a = np.ones(64)
        best_a = best_a / best_a.sum()
        best_exact = _exact_score(best_a)
        best_score = _fft_score(best_a)

    # Determine lengths to try
    lengths = []
    if parent.size >= 512:
        lengths.append(512)
    for L in (384, 256, 320, 448):
        if L not in lengths:
            lengths.append(L)
    # Add parent length if different
    if parent.size not in lengths and 32 <= parent.size <= 600:
        lengths.append(parent.size)
    lengths = [L for L in lengths if 32 <= L <= 600]
    if not lengths:
        lengths = [192]

    # Time allocation
    remaining = deadline - time.monotonic()
    if remaining < 30:
        # Very tight: just return best
        pass
    else:
        # Allocate time proportionally, favoring 512 slightly
        weights = {}
        for L in lengths:
            if L == 512:
                weights[L] = 1.4
            elif L == 384:
                weights[L] = 1.0
            elif L == 256:
                weights[L] = 0.8
            else:
                weights[L] = 0.6
        total_w = sum(weights.values())
        phase_time = remaining * 0.88

        for L in lengths:
            if time.monotonic() >= deadline - 25.0:
                break
            timeL = (weights[L] / total_w) * phase_time
            timeL = min(timeL, deadline - time.monotonic() - 22.0)
            if timeL < 15.0:
                continue

            use_lp = (L <= 512 and timeL > 40.0)

            # Main run: seed from best
            sub_time = min(timeL * 0.8, deadline - time.monotonic() - 20.0)
            if sub_time > 12.0:
                try:
                    init = _resample(best_a, L)
                    loc, loc_sc = _local_optimize(
                        init, time.monotonic() + sub_time, rng, use_lp=use_lp
                    )
                    if np.isfinite(loc_sc) and loc_sc < best_score + 1e-9:
                        sc_exact = _exact_score(loc)
                        if np.isfinite(sc_exact) and sc_exact < best_exact:
                            best_a = _normalize(loc)
                            best_score = loc_sc
                            best_exact = sc_exact
                        elif np.isfinite(loc_sc) and loc_sc < best_score:
                            best_a = _normalize(loc)
                            best_score = loc_sc
                except Exception:
                    pass

            # Secondary: exploratory start
            rem2 = deadline - time.monotonic() - 20.0
            sub_time2 = min((timeL - sub_time) * 0.9, rem2)
            if sub_time2 > 12.0:
                try:
                    x = np.linspace(0, 1, L, endpoint=False)
                    kind = int(rng.integers(0, 4))
                    if kind == 0:
                        alpha = rng.uniform(0.5, 2.0)
                        init2 = x ** alpha
                    elif kind == 1:
                        alpha = rng.uniform(0.5, 2.0)
                        init2 = (1.0 - x) ** alpha
                    elif kind == 2:
                        c = rng.uniform(0.2, 0.8)
                        wdt = rng.uniform(0.05, 0.3)
                        init2 = np.exp(-((x - c) ** 2) / (2 * wdt * wdt)) + 0.05
                    else:
                        init2 = _resample(best_a, L)
                        init2 = init2 + rng.normal(0, 0.1, L)
                    init2 = np.maximum(init2, 1e-6)
                    init2 = _normalize(init2)
                    loc2, loc_sc2 = _local_optimize(
                        init2, time.monotonic() + sub_time2, rng, use_lp=use_lp
                    )
                    if np.isfinite(loc_sc2) and loc_sc2 < best_score + 1e-9:
                        sc2 = _exact_score(loc2)
                        if np.isfinite(sc2) and sc2 < best_exact:
                            best_a = _normalize(loc2)
                            best_score = loc_sc2
                            best_exact = sc2
                except Exception:
                    pass

    # Final polish on best candidate
    if time.monotonic() < deadline - 15.0:
        pol_time = min(120.0, deadline - time.monotonic() - 10.0)
        if pol_time > 20.0:
            try:
                use_lp_f = int(best_a.size) <= 512 and pol_time > 50.0
                loc_f, sc_f = _local_optimize(
                    best_a, time.monotonic() + pol_time, rng, use_lp=use_lp_f
                )
                if np.isfinite(sc_f) and sc_f < best_score + 1e-9:
                    sc_exact_f = _exact_score(loc_f)
                    if np.isfinite(sc_exact_f) and sc_exact_f < best_exact:
                        best_a = _normalize(loc_f)
                        best_score = sc_f
                        best_exact = sc_exact_f
            except Exception:
                pass

    # Final validation
    try:
        best_a = np.asarray(best_a, dtype=np.float64).ravel()
        if best_a.size == 0 or not np.all(np.isfinite(best_a)):
            raise ValueError
        best_a = np.maximum(best_a, 0.0)
        best_a = _normalize(best_a)
        final_sc = _exact_score(best_a)
        if not np.isfinite(final_sc):
            raise ValueError
        if final_sc < best_exact:
            best_exact = final_sc
    except Exception:
        fb = np.ones(64) / 64.0
        fs = _exact_score(fb)
        if np.isfinite(fs):
            best_a = fb
            best_exact = fs
        else:
            return [1.0]

    out = np.clip(best_a, 0.0, 1000.0)
    s = out.sum()
    if s < 0.01:
        return [1.0 / 64.0] * 64
    out = out / s
    return [float(x) for x in out]
