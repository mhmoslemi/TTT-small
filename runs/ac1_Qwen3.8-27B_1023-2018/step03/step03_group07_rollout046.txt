import numpy as np
import time
from scipy.optimize import linprog

def _exact_score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * a.size * float(c.max()) / (s * s)

def _build_A(a):
    n = a.size
    m = 2 * n - 1
    # A[k, j] = a[k-j] if 0 <= k-j < n
    k_idx = np.arange(m)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    A = np.where(mask, a[np.clip(diff, 0, n-1)], 0.0)
    return A

def _lp_solve(A, M, n, time_limit=2.0):
    c_obj = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gsum = g.sum()
    if not np.isfinite(g).all() or gsum <= 1e-15:
        return None
    return g / gsum

def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * c.max() / (s * s)

    return float(np.convolve(a, a).max())

def _build_A(a):
    n = a.size
    m = 2 * n - 1
    k_idx = np.arange(m)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx

    c_obj = -np.ones(n)
    b_ub = np.full(A.shape[0], M + 1e-12)

                  bounds=[(0.0, None)] * n,
                  method='highs',

        return None
