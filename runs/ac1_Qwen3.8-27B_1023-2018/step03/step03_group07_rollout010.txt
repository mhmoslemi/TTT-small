Actually, I want to add one more technique: **coordinate descent on pairs of variables**. This can be very effective for problems with non-convex objectives.

For a fixed pair (i, j), we can optimize a_i and a_j while keeping other variables fixed. The objective becomes:
max_k (a*a)[k] where a appears in (a*a)[k] as:
sum_j' a[j'] * a[k-j']

For fixed a (except a_i, a_j), this is quadratic in (a_i, a_j):
(a*a)[k] = c_k + a_i * d_k + a_j * e_k + a_i * a_j * f_k + ...

where c_k, d_k, e_k, f_k depend on k and the fixed variables.

This is a 2D problem that can be solved by parameter search. But implementing this efficiently is complex.

import numpy as np

from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            return p
    except Exception:
        pass
    return np.ones(201, dtype=np.float64) / 201.0

def _exact_score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * a.size * float(c.max()) / (s * s)

def _conv(a):
    return np.convolve(a, a)

def _build_A(a):
    n = a.size
    m = 2 * n - 1
    ks = np.arange(m)[:, None]
    js = np.arange(n)[None, :]
    idx = ks - js
    ok = (idx >= 0) & (idx < n)
    return np.where(ok, a[np.clip(idx, 0, n - 1)], 0.0)

def _lp_solve(A, M, n, tl=2.0):
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(A.shape[0], M),
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': tl})
    except Exception:
        return None, None
    if res.status not in (0, 2) or res.x is None:
        return None, None
    g = np.asarray(res.x, dtype=np.float64)
    gs = g.sum()
    if not np.isfinite(g).all() or gs < 1e-15:
        return None, None
    g_norm = g / gs
    dw = None
    try:
        marg = np.asarray(res.ineqlin.marginals, dtype=np.float64)
        dw = np.maximum(marg, 0.0)
        if dw.sum() < 1e-12:
            dw = None
    except Exception:
        pass
    return g_norm, dw

def _subgrad(a, k):
    n = a.size
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return g
    g[lo:hi + 1] = a[k - hi : k - lo + 1][::-1]
    return g

    best = a.copy()
    best_M = float(_conv(a).max())
    rng = np.random.default_rng(seed)
    stale = 0

    for it in range(mxit):
        if time.monotonic() > t_end - 0.3:
            break
        b = _conv(a)
        M = float(b.max())
        ks = int(b.argmax())
        if M < best_M * (1.0 - 1e-13):
            best_M = M
            best = a.copy()
            stale = 0
        else:
            stale += 1

        if stale > 80:
            m = stale // 80
            if m == 1:
                a += rng.normal(0, 0.015, n)
            elif m == 2:
                f = rng.integers(1, 5)
                a += 0.008 * np.cos(2 * np.pi * f * np.arange(n) / n)
            elif m == 3:

            a = np.maximum(a, 1e-10)

            stale = 0
