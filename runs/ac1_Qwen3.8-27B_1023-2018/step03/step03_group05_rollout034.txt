            if r < 0.4:
                noise = rng.normal(0, 0.05, n)
                x = _project_simplex(x + noise)
            elif r < 0.7:
                # Smooth restart
                alpha = float(rng.uniform(0.1, 0.5))
                x = _project_simplex((1 - alpha) * best_x + alpha * rng.dirichlet(np.ones(n)))
            else:
                x = best_x + rng.normal(0, 0.02, n)
                x = _project_simplex(x)
            patience = 0
            T = T_base
            lr = 0.15
            continue

        # Compute weights
        w = _softmax_weights(c, T)

        # Compute gradient
        g = _grad_smooth_max(x, w)
        g = np.maximum(g, 0.0)  # gradient is nonnegative for nonnegative x
        g_mean = g.mean()
        if g_mean < 1e-15:
            patience += 10
            continue

        # Normalized gradient direction
        d = g / g_mean

        # Try multiplicative update
        improved = False
        for factor in [lr, lr * 0.5, lr * 0.25, lr * 2.0]:
            if factor <= 0 or factor > 5.0:
                continue
            x_new = x * np.exp(-np.clip(factor * d, -8.0, 8.0))
            x_new = np.maximum(x_new, 1e-15)
            x_new = x_new / x_new.sum()
            c_new = _conv_full(x_new)
            M_new = float(c_new.max())
            if M_new < M - 1e-14:
                x = x_new
                M = M_new
                improved = True
                break

        if not improved:
            # Try gradient descent with projection
            for factor in [lr * 0.1, lr * 0.05, lr * 0.02]:
                x_new = x - factor * d * M
                x_new = _project_simplex(x_new)
                c_new = _conv_full(x_new)
                M_new = float(c_new.max())
                if M_new < M - 1e-14:
                    x = x_new
                    M = M_new
                    improved = True
                    break

        if not improved:
            lr *= 0.9
            T *= 0.95
            T = max(T, T_min)
        else:
            lr = min(lr * 1.03, 3.0)
            T *= 0.998
            T = max(T, T_min)

    return best_x, best_M


def _resample(a, m):
    n = len(a)
    if m == n:
        return a.copy()
    if m > n:
        idx = (np.arange(m, dtype=np.int64) * n) // m
        counts = np.bincount(idx, minlength=n).astype(np.float64)
        counts[counts <= 0] = 1.0
        b = a[idx] / counts[idx]
    else:
        b = np.zeros(m, dtype=np.float64)
        idx = (np.arange(n, dtype=np.int64) * m) // n
        np.add.at(b, idx, a)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s > 1e-300:
        b /= s
    else:
        b = np.ones(m, dtype=np.float64) / m
    return b


def _random_smooth(n, rng):
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    kind = int(rng.integers(0, 5))
    if kind == 0:
        alpha = float(rng.uniform(0.3, 3.0))
        a = x ** alpha + 1e-6
    elif kind == 1:
        alpha = float(rng.uniform(0.3, 3.0))
        a = (1.0 - x) ** alpha + 1e-6
    elif kind == 2:
        c = float(rng.uniform(0.15, 0.85))
        wdt = float(rng.uniform(0.08, 0.4))
        a = np.exp(-((x - c) ** 2) / (2 * wdt * wdt)) + 0.1
    elif kind == 3:
        a = np.ones(n)
        for _ in range(int(rng.integers(1, 4))):
            c = float(rng.uniform(0.1, 0.9))
            wdt = float(rng.uniform(0.05, 0.3))
            amp = float(rng.uniform(0.2, 1.0))
            a += amp * np.exp(-((x - c) ** 2) / (2 * wdt * wdt))
    else:
        a = rng.uniform(0.3, 1.5, n)
    a = np.maximum(a, 1e-6)
    a /= a.sum()
    return a


def _gradient_polish(x0, deadline, rng):
    """Final polishing with very small steps and exact max subgradient."""
    n = len(x0)
    x = np.array(x0, dtype=np.float64)
    x = _project_simplex(x)

    best_x = x.copy()
    best_M = float(_conv_full(x).max())
    lr = 0.02

    while time.monotonic() < deadline:
        c = _conv_full(x)
        M = float(c.max())

        if M < best_M - 1e-15:
            best_M = M
            best_x = x.copy()

        # Use exact subgradient (peak only) or top-K
        K = max(4, min(32, n // 16))
        if K >= len(c):
            idx = np.arange(len(c))
        else:
            idx = np.argpartition(c, len(c) - K)[len(c) - K:]
        w = np.zeros(len(c), dtype=np.float64)
        w[idx] = 1.0 / K

        g = _grad_smooth_max(x, w)
        g = np.maximum(g, 0.0)
        g_mean = g.mean()
        if g_mean < 1e-15:
            break

        d = g / g_mean
        improved = False
        for factor in [lr, lr * 0.5, lr * 0.2]:
            x_new = x * np.exp(-factor * d)
            x_new = _project_simplex(x_new)
            M_new = float(_conv_full(x_new).max())
            if M_new < M - 1e-15:
                x = x_new
                improved = True
                break

        if improved:
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.85
            if lr < 1e-5:
                # Small random kick
                x = _project_simplex(x + rng.normal(0, 0.005, n))

    return best_x, best_M


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(1.0, float(budget_s))
    except Exception:
        budget = 1300.0

    headroom = max(15.0, min(40.0, 0.03 * budget))
    deadline = t0 + budget - headroom
    if deadline <= t0 + 5.0:
        deadline = t0 + max(3.0, budget * 0.85)

    rng = np.random.default_rng(int(seed) if np.isfinite(float(seed)) else 42)

    # Get parent sequence
    parent = None
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0 and np.all(np.isfinite(p)):
                p = np.clip(p, 0.0, 1000.0)
                if p.sum() > 1e-300:
                    parent = p / p.sum()
        except Exception:
            pass

    if parent is None:
        parent = np.ones(512, dtype=np.float64) / 512.0

    # Evaluate parent
    best_score = _exact_score(parent)
    best_a = parent.copy()

    if not np.isfinite(best_score):
        best_a = np.ones(128, dtype=np.float64) / 128.0
        best_score = _exact_score(best_a)

    # Strategy: try multiple n values and initializations
    # Primary: n=512 (same as parent, should improve)
    # Secondary: n=1024, n=256, n=768

    n_primary = 512
    n_list = [n_primary]
    if parent is not None and len(parent) != n_primary:
        n_list.append(len(parent))
    for nn in [1024, 768, 256]:
        if nn not in n_list:
            n_list.append(nn)

    # Allocate time: 60% to primary, rest split
    total_time = deadline - time.monotonic()
    time_alloc = {}
    time_alloc[n_primary] = total_time * 0.55
    remaining = total_time * 0.45
    for nn in n_list[1:]:
        if nn == 1024:
            time_alloc[nn] = remaining * 0.4
        elif nn == 768:
            time_alloc[nn] = remaining * 0.2
        elif nn == 256:
            time_alloc[nn] = remaining * 0.2
        else:
            time_alloc[nn] = remaining * 0.2

    for n in n_list:
        if time.monotonic() >= deadline - 10.0:
            break

        t_alloc = time_alloc.get(n, 30.0)
        t_alloc = min(t_alloc, deadline - time.monotonic() - 10.0)
        if t_alloc < 15.0:
            continue

        # Generate initializations for this n
        inits = []

        # 1. Resample parent
        if parent is not None:
            inits.append(_resample(parent, n))

        # 2. Random smooth
        inits.append(_random_smooth(n, rng))
        inits.append(_random_smooth(n, rng))

        # 3. Blended
        if parent is not None:
            blended = 0.7 * _resample(parent, n) + 0.3 * _random_smooth(n, rng)
            inits.append(_project_simplex(blended))

        # Run optimization on each initialization
        per_init_time = t_alloc / max(1, len(inits))
        per_init_time = max(10.0, per_init_time)

        for idx_init, init in enumerate(inits):
            if time.monotonic() >= deadline - 10.0:
                break

            sub_deadline = time.monotonic() + min(per_init_time, deadline - time.monotonic() - 10.0)
            if sub_deadline <= time.monotonic() + 5.0:
                break

            try:
                # Phase 1: Main optimization
                x_opt, M_opt = _optimize_sequence(
                    init, sub_deadline, rng,
                    T_base=max(_conv_full(init).max() * 0.08, 1e-10)
                )

                # Phase 2: Polish with exact subgradient
                if time.monotonic() < deadline - 15.0:
                    polish_deadline = min(time.monotonic() + (sub_deadline - time.monotonic()),
                                          deadline - 12.0)
                    if polish_deadline - time.monotonic() > 8.0:
                        x_opt, M_opt = _gradient_polish(x_opt, polish_deadline, rng)

                # Evaluate
                sc = _exact_score(x_opt)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = _project_simplex(x_opt)

            except Exception:
                pass

    # Final polish on the best candidate
    if time.monotonic() < deadline - 10.0:
        try:
            final_deadline = deadline - 5.0
            best_a, _ = _gradient_polish(best_a, final_deadline, rng)
            sc = _exact_score(best_a)
            if np.isfinite(sc):
                best_score = sc
        except Exception:
            pass

    # Final validation
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s > 1e-300:
        best_a = best_a / s
    else:
        best_a = np.ones(128, dtype=np.float64) / 128.0

    final_sc = _exact_score(best_a)
    if not np.isfinite(final_sc):
        fallback = np.ones(128, dtype=np.float64) / 128.0
        return [float(t) for t in fallback]

    return [float(t) for t in best_a]
