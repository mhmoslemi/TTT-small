def propose_candidate(seed=42, budget_s=1300, **kwargs):

    total = float(budget_s)
    deadline = t0 + total - 12.0

        parent = np.ones(384)

        parent = np.ones(384)

    n = int(parent.size)
    fsize = next_fast_len(2 * n - 1)
    rng = np.random.default_rng(seed)

    def sc(a):
        s = float(a.sum())
        if s < 0.01:
            return math.inf
        F = rfft(a, fsize)
        b = irfft(F * F, fsize)[:2 * n - 1]
        m = float(b.max())
        if not math.isfinite(m):
            return math.inf
        return 2.0 * n * m / (s * s)

    # Initialize from parent, normalized
    f = np.clip(parent, 0, None).copy()
    f = f / f.sum()
    best = f.copy()
    best_sc = sc(f)

    def build_A(fv):
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = fv[j]
        return A

    # Main LP loop
