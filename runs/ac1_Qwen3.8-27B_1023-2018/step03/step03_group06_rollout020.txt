```python
def _lp_step(f, stop):
    """One LP iteration: find g maximizing sum(g) s.t. g>=0, conv(f,g)<=M"""
    n = len(f)
    M = float(_conv(f).max())
    if M < 1e-30: return f.copy(), 0.0
    
    # Build constraint matrix: A @ g <= M, where A[k,i] = f[i] if 0<=k-i<n
    # A is (2n-1) x n
    # For memory efficiency with n=256, this is 511x256 ≈ 130K entries, fine.
    
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]  # f[i] contributes to positions i, i+1, ..., i+n-1
    
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    
    obj = np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    
    res = linprog(obj, A_ub=A, b_ub=b_ub, bounds=bounds, 
                  method='highs', options={'presolve': True})
    
    if res.status != 0 or res.x is None:
        return f.copy(), 0.0
    
    g = np.clip(res.x, 0, None)
    gs = float(g.sum())
    if gs < 1e-15:
        return f.copy(), 0.0
    g = g / gs  # normalize
    
    return g, M
```

Let me now think about the gradient step more carefully.

For the multi-peak gradient:
1. Compute c = conv(a, a)
2. Find top-K positions (K=5, say)
3. For each position p, the gradient is g_p[j] = 2*a[p-j] if 0 <= p-j < n
4. Average the gradients: g_avg = (1/K) * sum_k g_{p_k}
5. Subtract the mean to zero out the sum: g_avg -= g_avg.mean()
6. Update: a_new = a - lr * g_avg
7. Project: a_new = clip(a_new, 0, None); a_new = a_new / sum(a_new)

The learning rate lr should be adaptive. A good choice: lr = alpha * a.mean() / ||g_avg||, where alpha is a step size (like 0.1).

Actually, since we're on the simplex, a better approach is:
- Compute the gradient g (direction of increasing max_b)
- We want to decrease max_b, so move in direction -g
- But we need to stay on the simplex
- Use a line search: a(t) = normalize(clip(a - t*g, 0, None))
- Find t that minimizes _sc(a(t))

This is essentially a projected gradient method with line search.

For the line search, I can try t in [0.001, 0.01, 0.05, 0.1, 0.2, 0.5] and pick the best.

Let me estimate: each evaluation of _sc involves one FFT convolution (O(n log n)) and a max. For n=256, this is ~10us. So trying 6 t-values costs ~60us. Very fast.

The gradient computation is O(n) per peak, so K=5 peaks cost O(5n) = O(1280). Also fast.

So each gradient step is ~100us. In 1300s, I could do millions of gradient steps. But I'll use the LP for the heavy lifting and gradient steps for polishing.

```python
    a2 = gg - 2*fg + ff

def _grad_step(a, K=5, stop=None):
    """Multi-peak gradient descent step on the simplex."""
    n = len(a)
    c = _cov(a)
    # Top K positions
    idx = np.argpartition(c, -K)[-K:]
    # Average gradient
    g = np.zeros(n)
    for p in idx:
        p = int(p)
        lo = max(0, p - n + 1)
        hi = min(n - 1, p)
        for j in range(lo, hi+1):
            g[j] += 2.0 * a[p - j]
    g /= K
    # Center (remove mean to preserve sum approximately)
    g -= g.mean()
    # Scale gradient
    gn = np.linalg.norm(g)
    if gn < 1e-15: return a.copy()
    g = g / gn
    
    # Line search
    best_a = a.copy()
    best_s = _sc(a)
    for lr in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
        new = a - lr * g * a.mean() * 10  # scale by mean to make lr meaningful
        new = np.clip(new, 0, None)
        ns = new.sum()
        if ns < 1e-15: continue
        new = new / ns
        s = _sc(new)
        if s < best_s:

            best_a = new.copy()
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

_fcl = {}

def _fl(m):
    m = int(m)
    if m <= 0:
        return 1
    v = _fcl.get(m)
    if v is None:
        v = int(next_fast_len(m))
        _fcl[m] = v
    return v

def _cov(a, b=None):
    if b is None:
        b = a
    n1, n2 = len(a), len(b)
    if n1 == 0 or n2 == 0:
        return np.zeros(1)
    m = n1 + n2 - 1
    L = _fl(m)
    c = irfft(rfft(a, L) * rfft(b, L), L)[:m]
    return np.maximum(c, 0.0)

def _sc(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s < 1e-12:
        return math.inf
    n = a.size
    c = _cov(a)
    mb = float(c.max())
    if not math.isfinite(mb) or mb < 0:
        return math.inf
    return 2.0 * n * mb / (s * s)

def _nm(a):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / max(1, a.size)
    return a / s

def _gp():
    try:
        h = globals().get('height_sequence_1')
        if h is None:
            return None
        p = np.asarray(h, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size == 0 or p.sum() < 1e-15:
            return None
        return _nm(p)
    except Exception:
        return None

def _ex(a):
    try:
        fn = globals().get('evaluate_sequence')
        if fn is None:
            return math.inf
        return float(fn([float(x) for x in np.asarray(a, dtype=np.float64).ravel()]))
    except Exception:
        return math.inf

def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        fi = f[i]
        if fi > 0.0:
            A[i:i + n, i] = fi
    return A

def _lp_dir(f):
    """LP: maximize sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    ff = _cov(f)
    M = float(ff.max())
    if M < 1e-30:
        return None
    A = _build_A(f)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    try:
        res = linprog(
            -np.ones(n, dtype=np.float64),  # minimize -sum(g) = maximize sum(g)
            A_ub=A, b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method='highs',
            options={'presolve': True}
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs

def _best_t(ff, fg, gg):
    """Find t in [0,1] minimizing max_k of (1-t)^2*ff_k + 2t(1-t)*fg_k + t^2*gg_k."""
    a2 = gg - 2.0 * fg + ff
    b1 = 2.0 * (fg - ff)
    c0 = ff.copy()
    # Coarse grid
    ts = np.linspace(0.0, 1.0, 512)
    Q = c0[:, None] + b1[:, None] * ts[None, :] + a2[:, None] * (ts[None, :] ** 2)
    vals = Q.max(axis=0)
    j = int(vals.argmin())
    lo = ts[max(0, j - 3)]
    hi = ts[min(len(ts) - 1, j + 3)]
    if hi - lo < 1e-6:
        return float(ts[j])
    # Fine grid
    ts2 = np.linspace(lo, hi, 1024)
    Q2 = c0[:, None] + b1[:, None] * ts2[None, :] + a2[:, None] * (ts2[None, :] ** 2)
    vals2 = Q2.max(axis=0)
    j2 = int(vals2.argmin())
    return float(ts2[j2])

def _sgrad(a, s=150.0):
    """Smooth gradient of max_k(conv(a,a)_k) using softmax weighting."""
    n = len(a)
    c = _cov(a)
    cm = c.max()
    w = np.exp(s * (c - cm))
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones_like(c) / c.size
    else:
        w = w / ws
    # g[j] = 2 * sum_k w[k] * a[k-j] (valid)
    # = 2 * conv(w, a_rev)[n-1 + j] for j = 0..n-1
    a_rev = a[::-1]
    corr = _cov(w, a_rev)  # length (2n-1) + n - 1 = 3n-2
    g = corr[n - 1 : 2 * n - 1] * 2.0
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    g -= g.mean()
    return g

def _grad_step(a, s=150.0):
    """One gradient descent step on the simplex."""
    g = _sgrad(a, s=s)
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return a.copy()
    g = g / gn
    best_a = a.copy()
    best_s = _sc(a)
    am = a.mean()
    for lr in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15]:
        new = a - lr * g * am * 5.0
        new = np.clip(new, 0.0, None)
        ns = new.sum()
        if ns < 1e-15:
            continue
        new = new / ns
        sc = _sc(new)
        if sc < best_s:
            best_s = sc
            best_a = new.copy()
    return best_a

def _cpolish(a, stop, n_passes=2):
    """Coordinate descent polish."""
    a = _nm(a)
    n = a.size
    best = a.copy()
    best_sc = _sc(best)
    mults = (0.0, 0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0, 5.0)
    for _ in range(n_passes):
        if time.monotonic() > stop:
            break
        imp = False
        dflt = 1.0 / max(1.0, 4.0 * n)
        for j in range(n):
            if time.monotonic() > stop:
                break
            orig = best[j]
            base = max(orig, dflt)
            bj = orig
            bs = best_sc
            for m in mults:
                v = base * m
                if v > 1000.0:
                    continue
                best[j] = v
                sc = _sc(best)
                if sc < bs:
                    bs = sc
                    bj = v
            best[j] = bj
            if bs < best_sc - 1e-15:
                best_sc = bs
                imp = True
        if not imp:
            break
    return best, best_sc
```

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

_fcl = {}

def _fl(m):
    m = int(m)
    if m <= 0:
        return 1
    v = _fcl.get(m)
    if v is None:
        v = int(next_fast_len(m))
        _fcl[m] = v
    return v

def _cov(a, b=None):
    if b is None:
        b = a
    n1, n2 = len(a), len(b)
    if n1 == 0 or n2 == 0:
        return np.zeros(1)
    m = n1 + n2 - 1
    L = _fl(m)
    c = irfft(rfft(a, L) * rfft(b, L), L)[:m]
    return np.maximum(c, 0.0)

def _sc(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s < 1e-12:
        return math.inf
    n = a.size
    c = _cov(a)
    mb = float(c.max())
    if not math.isfinite(mb):
        return math.inf
    return 2.0 * n * mb / (s * s)

def _nm(a):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / max(1, a.size)
    return a / s

def _gp():
    try:
        h = globals().get('height_sequence_1')
        if h is None:
            return None
        p = np.asarray(h, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size == 0 or p.sum() < 1e-15:
            return None
        return _nm(p)
    except Exception:
        return None

def _ex(a):
    try:
