def _update_best(arr, forced=False):

    if arr is None: return

    mean=s/a.size
    # threshold trims
    for eps in (1e-12,1e-10,1e-8):
       c=an.copy()
       c[c <= eps] = 0.0
       if c.sum() < 0.01: continue
       # strip zeros

       if not mask.any(): continue

import numpy as np
import time
import math

from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.linalg import toeplitz


_FFT_SIZE_CACHE = {}


def _norm1(a, force_positive_eps=0.0):
    """Normalize to sum 1, clipping to finite nonnegative values."""
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return np.ones(1, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, force_positive_eps)
    s = float(a.sum())
    if s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def _get_parent():
    h = None
    try:
        h = globals()["height_sequence_1"]
    except Exception:
        h = None
    if h is None:
        return None
    try:
        a = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if a.size == 0 or float(a.sum()) <= 1e-12:
        return None
    return a


def _resample(a, target_len):
    a = np.asarray(a, dtype=np.float64).ravel()
    target_len = int(max(1, target_len))
    if a.size == 0:
        return np.ones(target_len, dtype=np.float64) / target_len
    if a.size == target_len:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, target_len)
    return np.interp(x_new, x_old, a)


def _conv_fft(a):
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([float(a[0] * a[0])], dtype=np.float64)
    full_len = 2 * n - 1
    fsize = _FFT_SIZE_CACHE.get(n)
    if fsize is None:
        fsize = next_fast_len(full_len)
        _FFT_SIZE_CACHE[n] = fsize
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:full_len]
    c = np.where(np.isfinite(c), c, 0.0)
    np.maximum(c, 0.0, out=c)
    return c


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 1e-12:
        return math.inf
    b = _conv_fft(a)
    if b.size == 0:
        return math.inf
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _build_K(a):
    """Dense operator for bilinear convolution: (K a)b = a*b."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    L = 2 * n - 1
    K = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        K[j:j + n, j] = a
    return K


def _fit_linprog(c, A_ub, b_ub, bounds, max_time=None):
    attempts = []
    if max_time is not None and max_time > 0.4:
        attempts.append(("highs", {"time_limit": float(max_time), "presolve": True}))
    attempts.append(("highs", None))
    attempts.append(("interior-point", None))

    for method, opts in attempts:
        for opt_choice in ([opts] if opts is not None else [None]) :
            try:
                if opt_choice is None:
                    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
                else:
                    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method, options=opt_choice)
                if res is not None and getattr(res, "status", 1) == 0:
                    return res
            except Exception:
                # If option-related failure, try the same method without options once.
                if opt_choice is not None and opts == opt_choice:
                    try:
                        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
                        if res is not None and getattr(res, "status", 1) == 0:
                            return res
                    except Exception:
                        res = None
    return None


def _lp_direction(K, M, relax=1.0, max_time=10.0):
    try:
        n = int(K.shape[1])
        L = int(K.shape[0])
        if M <= 1e-15 or relax <= 1e-12 or n == 0:
            return None, 0.0
        A = K / (M * relax)
        b = np.full(L, 1.0 + 1e-9, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = _fit_linprog(c, A, b, bounds, max_time=max_time)
        if res is None or getattr(res, "x", None) is None:
            return None, 0.0

        g = np.asarray(res.x, dtype=np.float64)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        S = float(g.sum())
        if S <= relax + 1e-8:
            return None, 0.0
        h = g / S
        return h, S
    except Exception:
        return None, 0.0


def _segment_best(f, h, prev_t=None):
    f = np.asarray(f, dtype=np.float64).ravel()
    h = np.asarray(h, dtype=np.float64).ravel()
    if f.size != h.size or f.size == 0:
        return None, math.inf, None

    cands = []
    if prev_t is not None and math.isfinite(prev_t):
        for mult in (0.35, 0.55, 0.75, 1.0, 1.2, 1.45):
            cands.append(prev_t * mult)
    cands += [
        0.05, 0.08, 0.12, 0.16, 0.20, 0.25, 0.30,
        0.38, 0.46, 0.55, 0.65, 0.75, 0.85, 0.92, 1.0
    ]

    seen = set()
    best_x = None
    best_sc = math.inf
    best_t = None

    def _eval(t):
        nonlocal best_x, best_sc, best_t
        if not math.isfinite(t):
            return
        if t < -1e-12 or t > 2.0:
            return
        key = round(float(t), 8)
        if key in seen:
            return
        seen.add(key)

        x = (1.0 - t) * f + t * h
        if x.min() < -1e-9:
            return
        x = np.maximum(x, 0.0)
        s = float(x.sum())
        if s <= 1e-12:
            return
        x = x / s
        sc = _score_fft(x)
        if math.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_x = x.copy()
            best_t = float(t)

    for t in cands:
        _eval(t)

    if best_t is not None:
        lo = max(0.0, best_t - 0.04)
        hi = min(2.0, best_t + 0.04)
        for tt in np.linspace(lo, hi, 9):
            _eval(float(tt))

    return best_x, best_sc, best_t


def _gn_step(a):
    """Gauss--Newton-style step toward a flat convolution profile."""
    a = _norm1(a)
    n = int(a.size)
    if n < 2 or n > 768:
        return None

    try:
        b = _conv_fft(a)
        M = float(b.max())
        if M <= 1e-15:
            return None
        L = 2 * n - 1
        r = float(L) / (2.0 * n - 1.0) - b
        r = np.where(np.isfinite(r), r, 0.0)
        r = r - float(r.mean())

        autoc = np.convolve(a, a[::-1], mode="full")
        first = autoc[n - 1::-1]
        KTK = toeplitz(first, first).astype(np.float64)
        KTr = np.convolve(r, a[::-1], mode="full")[n - 1:n - 1 + n]

        tr = float(np.trace(KTK))
        lam = 1e-8 * max(tr / float(n), 1e-12) + 1e-12

        rhs = 0.5 * KTr
        Aq = KTK + lam * np.eye(n, dtype=np.float64)
        try:
            delta = np.linalg.solve(Aq, rhs)
        except Exception:
            delta = np.linalg.lstsq(Aq, rhs, rcond=None)[0]

        if delta is None or not np.all(np.isfinite(delta)):
            return None

        na = float(np.linalg.norm(a))
        dnorm = float(np.linalg.norm(delta))
        if na > 1e-15 and dnorm > 2.0 * na:
            delta = delta * (2.0 * na / dnorm)

        best_x = None
        best_sc = math.inf
        for alpha in (0.5, 0.25, 0.125, 0.75, 1.0, 0.0625, 0.03125):
            x = a + alpha * delta
            x = np.maximum(x, 0.0)
            s = float(x.sum())
            if s <= 1e-12:
                continue
            x = x / s
            sc = _score_fft(x)
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_x = x.copy()

        if best_x is None:
            return None
        if best_sc < _score_fft(a) - 1e-12:
            return best_x
        return None
    except Exception:
        return None


def _perturb(a, rng, strength):
    a = _norm1(a)
    n = a.size
    if n == 0:
        return a
    scale = float(a.sum()) / float(n)
    if scale <= 1e-15:
        scale = 1.0
    x = a + scale * float(strength) * rng.standard_normal(n)
    x = np.clip(x, 0.0, None)
    s = float(x.sum())
    if s <= 1e-12:
        return a.copy()
    return x / s


def _local_polish(a, stop, rng):
    a = _norm1(a)
    best = a.copy()
    best_sc = _score_fft(best)
    n = best.size
    if n <= 0 or stop <= time.monotonic():
        return best, best_sc

    rng = np.random.default_rng(rng)
    no_imp = 0
    max_no_imp = 8

    while time.monotonic() < stop and no_imp < max_no_imp:
        improved = False

        # One-coordinate multiplicative moves.
        order = rng.permutation(n)
        for ii in order:
            if time.monotonic() >= stop:
                break
            i = int(ii)
            orig = float(best[i])
            if orig > 1e-14:
                moves = (0.0, 0.5, 0.8, 0.95, 1.05, 1.2, 1.5)
                for m in moves:
                    x = best.copy()
                    x[i] = orig * m
                    s = float(x.sum())
                    if s <= 1e-12:
                        continue
                    x = x / s
                    sc = _score_fft(x)
                    if sc < best_sc - 1e-13:
                        best = x
                        best_sc = sc
                        improved = True
                        break
            else:
                mean = float(best.sum()) / float(n)
                for v in (mean * 1e-3, mean * 1e-2):
                    x = best.copy()
                    x[i] = v
                    s = float(x.sum())
                    if s <= 1e-12:
                        continue
                    x = x / s
                    sc = _score_fft(x)
                    if sc < best_sc - 1e-13:
                        best = x
                        best_sc = sc
                        improved = True
                        break

        # Two-coordinate mass-transfer moves, preserving total sum.
        if time.monotonic() < stop:
            n_pairs = 90 if n <= 400 else 45
            for _ in range(n_pairs):
                if time.monotonic() >= stop:
                    break
                ij = rng.integers(0, n, size=2)
                i = int(ij[0])
                j = int(ij[1])
                if i == j:
                    continue
                ai = float(best[i])
                aj = float(best[j])
                if ai <= 1e-15 and aj <= 1e-15:
                    continue
                for f in (0.1, 0.25, 0.5, 0.8):
                    x = best.copy()
                    x[i] = ai * (1.0 - f)
                    x[j] = aj + f * ai
                    if x.min() >= -1e-12:
                        s = float(x.sum())
                        if s > 1e-12:
                            x = x / s
                            sc = _score_fft(x)
                            if sc < best_sc - 1e-13:
                                best = x
                                best_sc = sc
                                improved = True
                                break

                    x = best.copy()
                    x[i] = ai + f * aj
                    x[j] = aj * (1.0 - f)
                    if x.min() >= -1e-12:
                        s = float(x.sum())
                        if s > 1e-12:
                            x = x / s
                            sc = _score_fft(x)
                            if sc < best_sc - 1e-13:
                                best = x
                                best_sc = sc
                                improved = True
                                break

        if improved:
            no_imp = 0
        else:
            no_imp += 1

    return best, best_sc


def _choose_initialization(length, base_array, extra_candidates, seed):

    i = np.arange(length, dtype=np.float64)

    cands.append((0.5 + 0.5 * np.cos(2.0 * np.pi * u)) ** 2)

    for p in (0.13, 0.21, 0.33):

    scored = []

    if not scored:
        a = np.ones(length, dtype=np.float64) / float(length)
        return a, []

    scored.sort(key=lambda z: z[1])
    best = scored[0][0].copy()
    alts = [x[0].copy() for x in scored[1:6]]

def _optimize_length(a0, stop, seed, alt_inits=None, use_gn=True):
    a0 = _norm1(a0)
    n = int(a0.size)
    if n == 0:
        a0 = np.ones(1, dtype=np.float64)
        n = 1

    pool = []
    for arr in [a0] + list(alt_inits or []):
        aa = _norm1(arr)
        if aa.size == n:
            sc = _score_fft(aa)
            if math.isfinite(sc):
                pool.append((aa, sc))
    if not pool:
        pool.append((a0.copy(), _score_fft(a0)))

    pool.sort(key=lambda z: z[1])
    a = pool[0][0].copy()
    best = a.copy()
    best_sc = pool[0][1]
    cur_sc = best_sc
    prev_t = None
    stall = 0
    it = 0
    last_gn_time = 0.0
    rng = np.random.default_rng(int(seed))
    max_iters = 1000000

    while time.monotonic() < max(stop - 2.0, time.monotonic()) and it < max_iters:
        it += 1
        now = time.monotonic()
        if now >= stop - 2.0:
            break

        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best = a.copy()

        # Gauss--Newton flattening step, used less frequently.
        if use_gn and n <= 768 and (stall >= 8 or it % 28 == 0) and now - last_gn_time > 7.0 and stop - now > 12.0:
            last_gn_time = now
            xg = _gn_step(a)
            if xg is not None:
                scg = _score_fft(xg)
                if scg < cur_sc - 1e-12:
                    a = xg
                    cur_sc = scg
                    if cur_sc < best_sc - 1e-12:
                        best = a.copy()
                        best_sc = cur_sc
                    stall = 0
                    prev_t = None
                    continue

        K = None
        M = math.inf
        try:
            K = _build_K(a)
            bcur = K @ a
            if bcur.size:
                M = float(bcur.max())
        except Exception:
            K = None
            M = math.inf

        lp_accepted = False

        if K is not None and math.isfinite(M) and stop - now > 3.0:
            h, Sg = _lp_direction(K, M, relax=1.0, max_time=min(15.0, max(2.0, stop - now - 2.0)))
            if h is not None:
                cand, sc, tm = _segment_best(a, h, prev_t)
                if cand is not None and sc < cur_sc - 1e-12:
                    a = cand
                    cur_sc = sc
                    prev_t = tm
                    if cur_sc < best_sc - 1e-12:
                        best = a.copy()
                        best_sc = cur_sc
                    stall = 0
                    lp_accepted = True
                    continue

            # Occasionally solve mildly relaxed/tightened LPs for a different direction.
            if not lp_accepted and 4 < stall < 45 and it % 4 == 0 and stop - now > 5.0:
                for relax in (1.005, 0.995, 1.01):
                    h, Sg = _lp_direction(K, M, relax=relax, max_time=min(8.0, max(1.0, stop - now - 1.0)))
                    if h is not None:
                        cand, sc, tm = _segment_best(a, h, prev_t)
                        if cand is not None and sc < cur_sc - 1e-12:
                            a = cand
                            cur_sc = sc
                            prev_t = tm
                            if cur_sc < best_sc - 1e-12:
                                best = a.copy()
                                best_sc = cur_sc
                            stall = 0
                            lp_accepted = True
                            break
        else:
            stall += 4

        if not lp_accepted:
            stall += 1

        # If stagnating, run structured local polish; if that fails, perturb/restart.
        if stall >= 18 and stop - now > 15.0:
            lps = min(stop - 2.0, now + 20.0)
            try:
                x2, sc2 = _local_polish(a, lps, rng)
            except Exception:
                x2, sc2 = None, math.inf
            if x2 is not None and sc2 < cur_sc - 1e-12:
                a = x2
                cur_sc = sc2
                prev_t = None
                if cur_sc < best_sc - 1e-12:
                    best = a.copy()
                    best_sc = cur_sc
                stall = 0
            else:
                strength = min(0.30, 0.03 + 0.01 * min(5, stall // 18))
                reset = False
                if stall > 45 and len(pool) > 1:
                    idx = int(rng.integers(0, len(pool)))
                    if idx != 0 and not np.allclose(pool[idx][0], a, atol=1e-7, rtol=0.0):
                        a = pool[idx][0].copy()
                        cur_sc = pool[idx][1]
                        stall = 12
                        reset = True
                if not reset:
                    base_for_perturb = best if stall % 3 == 0 else a
                    a = _perturb(base_for_perturb, rng, strength)
                    sc = _score_fft(a)
                    if not math.isfinite(sc) or sc > best_sc + 0.05:
                        a = _perturb(best, rng, 0.02)
                        sc = _score_fft(a)
                        if not math.isfinite(sc) or sc > best_sc + 0.02:
                            a = best.copy()
                            sc = best_sc
                    cur_sc = sc
                    stall = 5

    # Final bounded polish with whatever time remains.
    if stop - time.monotonic() > 6.0:
        try:
            x2, sc2 = _local_polish(a, min(stop, time.monotonic() + max(2.0, (stop - time.monotonic()) * 0.25)), rng)
            if sc2 < best_sc - 1e-12:
                best = x2
                best_sc = sc2
        except Exception:
            pass

    if stop - time.monotonic() > 3.0 and not np.allclose(best, a, atol=1e-7, rtol=0.0):
        try:
            x2, sc2 = _local_polish(best, min(stop, time.monotonic() + 2.0), rng)
            if sc2 < best_sc - 1e-12:
                best = x2
                best_sc = sc2
        except Exception:
            pass

    return _norm1(best), _score_fft(best)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))

    headroom = max(5.0, min(30.0, budget * 0.03))
    hard_deadline = t0 + budget - headroom

    final_reserve = max(5.0, min(120.0, budget * 0.08))
    final_reserve = min(final_reserve, max(2.0, hard_deadline - t0 - 5.0))
    work_end = hard_deadline - final_reserve

    parent_raw = _get_parent()
    if parent_raw is None:
        parent_raw = np.ones(384, dtype=np.float64)

    base = parent_raw
    base_len = int(base.size)
    if base_len > 1024:
        base = _resample(base, 1024)
        base_len = 1024
    if base_len <= 0:
        base = np.ones(384, dtype=np.float64)
        base_len = 384

    best_seq = _norm1(base).copy()
    best_score = math.inf

    def _update_best(arr, forced=True):
        nonlocal best_seq, best_score
        if arr is None:
            return
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        if a.size == 0:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        ss = float(a.sum())
        if ss < 0.01:
            return
        an = a / ss
        candidates = [an]

        mean = ss / float(a.size)
        for eps in (1e-12, 1e-10, 1e-8):
            c = an.copy()
            c[c <= eps] = 0.0
            if c.sum() < 0.01:
                continue
            mask = c > 0
            if not mask.any():
                continue
            i0 = int(np.argmax(mask))
            i1 = int(a.size - np.argmax(mask[::-1]))
            cc = c[i0:i1]
            if cc.size > 0 and cc.sum() > 1e-12:
                cc = np.where(np.isfinite(cc), cc, 0.0)
                cc = cc / float(cc.sum())
                candidates.append(cc)

        m = int(an.size)
        for frac in (0.05, 0.10):
            k = int(m * frac)
            if k <= 0 or k >= m:
                continue
            if float(an[:k].sum()) <= 1e-4 * mean * k + 1e-12:
                cc = an[k:]
                if cc.size > 0 and cc.sum() > 1e-12:
                    cc = cc / float(cc.sum())
                    candidates.append(cc)
            if float(an[-k:].sum()) <= 1e-4 * mean * k + 1e-12:
                cc = an[:-k]
                if cc.size > 0 and cc.sum() > 1e-12:
                    cc = cc / float(cc.sum())
                    candidates.append(cc)

        for cand in candidates:
            try:
                cand = np.asarray(cand, dtype=np.float64).ravel()
                if cand.size == 0:
                    continue
                cand = np.where(np.isfinite(cand), cand, 0.0)
                cand = np.clip(cand, 0.0, 1000.0)
                cs = float(cand.sum())
                if cs < 0.01:
                    if cs > 1e-15:
                        cand = cand / cs
                        cs = float(cand.sum())
                if cs < 0.01:
                    continue
                sc = float(evaluate_sequence([float(x) for x in cand]))
            except Exception:
                continue
            if math.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_seq = cand.copy()

    _update_best(best_seq, forced=True)
    if not math.isfinite(best_score):
        best_seq = np.ones(32, dtype=np.float64)
        try:
            best_score = float(evaluate_sequence([float(x) for x in best_seq]))
        except Exception:
            best_score = math.inf

    lengths = [32, 48, 64, 96, 128, 160, 192, 256, 320, 384, 512, 640, 768, 1024]
    lengths.append(base_len)
    lengths = sorted(set(L for L in lengths if 16 <= L <= 1024))

    weights = {}
    for L in lengths:
        w = 1.0
        if L == base_len:
            w = 10.0
        elif L == 384:
            w = max(w, 4.0)
        if L <= 64:
            w *= 1.5
        elif L <= 128:
            w *= 1.2
        elif L >= 1024:
            w *= 0.6
        weights[L] = w

    work_budget = max(0.0, work_end - time.monotonic())
    total_w = float(sum(weights.values()))
    if total_w <= 0:
        total_w = 1.0
    alloc = {L: max(4.0, work_budget * weights[L] / total_w) for L in lengths}
    s_alloc = float(sum(alloc.values()))
    if s_alloc > work_budget and work_budget > 0.0:
        scale = work_budget / s_alloc
        for L in alloc:
            alloc[L] = max(3.0, alloc[L] * scale)

    extra_candidates = []

    if work_budget > 8.0:
        for L in sorted(lengths, key=lambda z: (-weights[z], z)):
            now = time.monotonic()
            if now >= work_end - 2.0:
                break
            stop = min(work_end - 2.0, now + alloc.get(L, 0.0))
            if stop - now < 4.0:
                continue

            try:
                init, alts = _choose_initialization(L, base, extra_candidates, int(seed) + 100000 + int(L))
            except Exception:
                init, alts = _norm1(_resample(base, L)), []

            try:
                new_arr, new_sc = _optimize_length(
                    init,
                    stop,
                    int(seed) + 200 + int(L),
                    alt_inits=alts,
                    use_gn=(L <= 768),
                )
            except Exception:
                new_arr, new_sc = init, _score_fft(init)

            try:
                _update_best(new_arr)
            except Exception:
                pass

            if new_arr is not None:
                extra_candidates.append((new_arr, new_sc))
                if len(extra_candidates) > 6:
                    extra_candidates.sort(key=lambda z: z[1])
                    extra_candidates = extra_candidates[:6]

    # Final focused refinement on the currently best exact candidate.
    if time.monotonic() < hard_deadline - 12.0 and best_seq is not None:
        final_stop = hard_deadline - 8.0
        best_arr = _norm1(best_seq)
        alts = []
        for arr, sc in extra_candidates[:4]:
            try:
                alts.append(_resample(arr, best_arr.size))
            except Exception:
                pass
        try:
            fin_arr, fin_sc = _optimize_length(
                best_arr,
                final_stop,
                int(seed) + 987654,
                alt_inits=alts,
                use_gn=(best_arr.size <= 768),
            )
        except Exception:
            fin_arr, fin_sc = best_arr, _score_fft(best_arr)
        try:
            _update_best(fin_arr)
        except Exception:
            pass

    # Final validation, with fallbacks.
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
        except Exception:
            final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            fallback = np.asarray(best_seq, dtype=np.float64).ravel()
        except Exception:
            fallback = None
        if fallback is None or fallback.size == 0:
            fallback = _norm1(base).copy()
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if math.isfinite(final_sc):
            best_seq = fallback
            final_sc = final_sc
        else:
            fallback = np.ones(32, dtype=np.float64)
            try:
                final_sc = float(evaluate_sequence([float(x) for x in fallback]))
            except Exception:
                final_sc = math.inf
            if math.isfinite(final_sc):
                best_seq = fallback
                final_sc = final_sc
            else:
                return [1.0]

    if best_seq is None or len(best_seq) == 0:
        return [1.0]

    return [float(x) for x in best_seq]
