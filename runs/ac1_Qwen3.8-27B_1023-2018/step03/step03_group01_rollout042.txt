def _subgrad_direction(f, n, K=5):
    """Compute top-K subgradient direction for the non-smooth max."""
    b = _conv_full(f)
    s = float(f.sum())
    s2 = s * s
    
    # Get top K lags
    topk = min(K, 2 * n - 1)
    idx = np.argpartition(b, -topk)[-topk:]
    weights = b[idx].copy()
    wmin = weights.min()
    weights = weights - wmin + 1e-12
    weights /= weights.sum()
    
    # Subgradient of max b_k w.r.t. f_j:
    # db_k/df_j = 2 * f[k-j] (for valid k-j)
    # Objective: 2n * max_b / s^2
    # d(obj)/df_j = 2n/s^2 * db_k/df_j - 4n*max_b/s^3 (from derivative of 1/s^2)
    # Simplify: direction = db_k/df_j - max_b/s * ones
    
    grad_m = np.zeros(n)
    for w, k in zip(weights, idx):
        k = int(k)
        # f[k-j] for valid j: j ranges from max(0, k-n+1) to min(k, n-1)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        # f[k-j] for j in [lo, hi): index into f is k-j
        f_indices = k - np.arange(lo, hi)
        valid = (f_indices >= 0) & (f_indices < n)
        if valid.any():
            grad_m[lo:hi][valid] += w * 2.0 * f[f_indices[valid]]
    
    M = float(b.max())
    # Gradient of the full objective (ignoring the max, using weighted approximation)
    # F = 2n * (weighted max) / s^2
    # dF/df_j = 2n * [2*f_{k*-j}/s^2 - (weighted_max)*2/s^3]
    # = (4n/s^3) * [f_{k*-j} * s - weighted_max]
    wm = float(weights @ b[idx])  # weighted max approximation
    direction = grad_m - (wm / s) * 2.0 * np.ones(n)
    # We want to move in the direction that DECREASES the objective
    # So the update direction is -direction (for gradient descent)
    return -direction, M

            if g_dir is not None:
                dist = float(np.linalg.norm(g_dir - f))
                if dist > 1e-6:
                    res_t, f_new, sc_ls = _line_search(f, g_dir, n, best_sc, stop_time)
                    if f_new is not None and sc_ls < best_sc - 1e-14:
                        f = f_new
                        best_f = f.copy()
                        best_sc = sc_ls
                        stall = 0
                        continue
                if dist < 1e-6:
                    stall += 3
            else:
                stall += 2
        else:
            # Subgradient step
            direction, M = _subgrad_direction(f, n, K=7)
            # Normalize direction
            dn = float(np.linalg.norm(direction))
            if dn < 1e-14:
                stall += 2
                continue
            direction /= dn
            
            # Adaptive step size
            lr = 2.0 / (it + 1.0) * 0.5
            momentum = 0.7 * momentum + 0.3 * direction
            f_new = f - lr * momentum
            np.maximum(f_new, 0.0, out=f_new)
            sn = float(f_new.sum())
            if sn < 1e-10:
                f_new = f.copy()
                stall += 1
                continue
            f_new = f_new / sn
            
            sc = _score(f_new)
            if sc < best_sc - 1e-14:
                f = f_new
                best_f = f.copy()
                best_sc = sc
                stall = 0
            else:
                f = f_new  # still move (stochastic-like)
                stall += 1
        
        # Stalling handling
        if stall >= max_stall:
            # Perturb
            noise_scale = 0.15 * np.std(f)
            f = np.clip(f + rng.normal(0, noise_scale, n), 0, None)
            s2 = float(f.sum())
            if s2 < 1e-10:
                f = best_f.copy()
            else:
                f = f / s2
            stall = 0
            momentum = np.zeros(n)
    
    return best_f, best_sc


def _coord_polish(f0, n, stop_time, seed):
    """Coordinate descent with Brent's method for 1D optimization."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    s = float(f.sum())
    f = f / s
    best_sc = _score(f)
    no_imp = 0
    
    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                return f, best_sc
            
            orig = f[i]
            
            # Try multiple scales
            for mlt in (0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 0.0):
                f[i] = orig * mlt if orig > 1e-12 else (0.01 if mlt > 0 else 0.0)
                sc = _score(f)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
                    orig = f[i]
                else:
                    f[i] = orig
            
            # Fine-tune with Brent if value is nonzero
            if f[i] > 1e-8:
                def obj_1d(val):
                    f[i] = max(0.0, val)
                    return _score(f)
                try:
                    res = minimize_scalar(obj_1d, bounds=(0.0, 3.0 * f[i]),
                                         method='bounded', options={'xatol': 1e-4})
                    f[i] = max(0.0, res.x)
                    sc = _score(f)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
                except Exception:
                    f[i] = orig
        
        # 2-coordinate moves
        for _ in range(min(60, n)):
            if time.monotonic() > stop_time - 1.0:
                return f, best_sc
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi in (0.5, 1.5, 2.0):
                for mj in (0.5, 1.5, 0.3):
                    f[i] = max(0.01, oi * mi) if oi > 1e-12 else 0.01
                    f[j] = max(0.01, oj * mj) if oj > 1e-12 else 0.01
                    sc = _score(f)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
            f[i], f[j] = oi, oj
        
        if not improved:
            no_imp += 1
            if no_imp >= 8:
                break
        else:
            no_imp = 0
    
    return f, best_sc


def _direct_optimize(f0, n, stop_time, seed):
    """Direct non-smooth optimization using projected gradient with smoothing."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    s = float(f.sum())
    f = f / s
    best_f = f.copy()
    best_sc = _score(f)
    
    lr = 0.02
    momentum = np.zeros(n)
    beta = 0.8
    tau = 0.1  # smoothing parameter for log-sum-exp
    no_imp = 0
    
    for it in range(50000):
        if time.monotonic() > stop_time - 1.0:
            break
        
        b = _conv_full(f)
        s = float(f.sum())
        s2 = s * s
        M = float(b.max())
        
        # Smoothed approximation: use log-sum-exp
        # softmax of b / (tau * s^2)
        scaled = b / (tau * s2)
        scaled -= scaled.max()
        exps = np.exp(scaled)
        softmax = exps / exps.sum()
        
        # Gradient of smoothed objective
        # F_smooth = 2n * tau * s^2 * log(sum exp(b_k/(tau*s^2))) / s^2
        # = 2n * tau * log(sum exp(b_k/(tau*s^2)))
        # dF/df_j = 2n * tau * (1/sum_exp) * sum_k exp(b_k/(tau*s^2)) * db_k/df_j / (tau*s^2)
        # = 2n / (s^2 * sum_exp) * sum_k exp(b_k/(tau*s^2)) * 2*f_{k-j}
        # Using softmax weights: = (4n/s^2) * sum_k softmax_k * f_{k-j}
        
        grad = np.zeros(n)
        # Only use lags where softmax is significant
        significant = np.where(softmax > 1e-6 / (2 * n - 1))[0]
        for k in significant:
            k = int(k)
            w = softmax[k] * 2.0
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            f_indices = k - np.arange(lo, hi)
            valid = (f_indices >= 0) & (f_indices < n)
            if valid.any():
                grad[lo:hi][valid] += w * f[f_indices[valid]]
        
        grad = (4.0 * n / s2) * grad - (4.0 * n * M / (s2 * s)) * np.ones(n)
        
        momentum = beta * momentum - lr * grad
        f_new = f + momentum
        np.maximum(f_new, 0.0, out=f_new)
        sn = float(f_new.sum())
        if sn < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / sn
        
        sc = _score(f_new)
        if sc < best_sc - 1e-14:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.05, lr * 1.01)
        else:
            no_imp += 1
            lr = max(0.003, lr * 0.98)
            if no_imp > 200:
                f = f_new.copy()
                no_imp = 0
                lr = 0.02
                beta = 0.7
                momentum = np.zeros(n)
                tau = max(0.05, tau * 0.9)
    
    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 15.0
    
    parent = _get_parent()
    n_p = len(parent)
    
    # Initial best
    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not math.isfinite(best_score):
        best_seq = np.ones(384) / 384.0
        n_p = 384
        n_p = len(best_seq)
        best_score = _score(best_seq)
    
    def _try_update(cand):
        nonlocal best_seq, best_score
        if cand is None:
            return
        cand = np.clip(np.asarray(cand, dtype=np.float64).ravel(), 0, 1000)
        if cand.size == 0 or cand.sum() < 0.01:
            return
        try:
            sc = _score(cand)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_seq = cand.copy()
            best_score = sc
    
    # Build initializations
    rng_main = np.random.default_rng(seed)
    n = n_p
    i_arr = np.arange(n, dtype=np.float64)
    
    inits = []
    inits.append(parent.copy())
    
    # Cosine bump
    inits.append(0.5 + 0.5 * np.cos(2 * math.pi * i_arr / n))
    
    # Multi-frequency
    sup = np.full(n, 0.2)
    for fr in (1, 2, 3, 5, 7, 11):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.25 * np.cos(2 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.02)
    
    # Random
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.03)
    
    # Anti-palindromic bias
    ap = np.ones(n)
    for i in range(n):
        partner = n - 1 - i
        if i < partner:
            ap[i] = 1.5
            ap[partner] = 0.3
    inits.append(ap + rng_main.normal(0, 0.1, n))
    inits[-1] = np.clip(inits[-1], 0.01, None)
    
    # Parent + noise
    noise_scale = 0.4 * max(1.0, float(np.mean(parent)))
    pn = np.abs(parent + rng_main.normal(0, noise_scale, n)) + 0.01
    inits.append(pn)
    
    # Sinc-like
    sinc = np.sinc((i_arr - n / 2) / (n * 0.25))
    inits.append(np.abs(sinc) + 0.05)
    
    # Two-bump
    bump1 = np.exp(-((i_arr - n * 0.25) / (n * 0.15)) ** 2)
    bump2 = np.exp(-((i_arr - n * 0.75) / (n * 0.15)) ** 2)
    inits.append(bump1 + 0.5 * bump2 + 0.05)
    
    # Phase 1: LP-based Frank-Wolfe on multiple initializations (55% time)
    fw_end = t0 + budget * 0.55
    time_per_init = max(20.0, (fw_end - t0 - 10) / len(inits))
    
    for idx, init in enumerate(inits):
        if time.monotonic() > fw_end - 5.0:
            break
        if best_score < 1.503:
            break
        stop = min(fw_end - 4.0, time.monotonic() + time_per_init)
        try:
            res, sc = _fw_phase(init, n, stop, seed + 1000 * (idx + 1), lp_every=2)
            _try_update(res)
        except Exception:
            pass
    
    # Phase 2: Direct smoothing optimization (20% time)
    sm_end = min(deadline, t0 + budget * 0.75)
    if time.monotonic() < sm_end - 15.0:
        try:
            res, sc = _direct_optimize(best_seq, len(best_seq), sm_end, seed + 55001)
            _try_update(res)
        except Exception:
            pass
    
    # Phase 3: More LP iterations on the best (10% time)
    lp2_end = min(deadline, t0 + budget * 0.85)
    if time.monotonic() < lp2_end - 10.0:
        try:
            res, sc = _fw_phase(best_seq, len(best_seq), lp2_end, seed + 99001, lp_every=1)
            _try_update(res)
        except Exception:
            pass
    
    # Phase 4: Coordinate descent polish (10% time)
    cd_end = deadline - 5.0
    if time.monotonic() < cd_end - 10.0:
        try:
            res, sc = _coord_polish(best_seq, len(best_seq), cd_end, seed + 77001)
            _try_update(res)
        except Exception:
            pass
    
    # Final subgradient burst
    if time.monotonic() < deadline - 8.0:
        try:
            res, sc = _direct_optimize(best_seq, len(best_seq), deadline - 6.0, seed + 33001)
            _try_update(res)
        except Exception:
            pass
    
    # Final validation
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        sc = evaluate_sequence(final_list)
        if math.isfinite(sc) and sc > 0:
            return final_list
    except Exception:
        pass
    
    # Fallback to parent
    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass
    
    return [1.0] * 384
