import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    return p

def _conv_full(a):
    n = len(a)
    if n <= 400:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: 2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_sparse_A(f, n):
    """Build the constraint matrix for the LP as a sparse matrix."""
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            idx = k - j
            if f[idx] != 0:
                rows.append(k)
                cols.append(j)
                vals.append(f[idx])
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def _lp_direction(f, n, M, time_limit=3.0):
    """Find direction g that maximizes sum(g) s.t. f*g <= M, g >= 0."""
    A = _build_sparse_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None
    return g / g_sum

def _gradient_step(f, conv, n, eta):
    """One
