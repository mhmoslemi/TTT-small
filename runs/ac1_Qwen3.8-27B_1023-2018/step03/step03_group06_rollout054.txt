def _conv_same(a):
    n = len(a)
    m = 2*n - 1
    L = next_fast_len(m)
    fa = rfft(a, L)
    c = irfft(fa*fa, L)[:m]
    return np.maximum(c, 0)

def _cross_conv(a, b):
    n1, n2 = len(a), len(b)
    m = n1 + n2 - 1
    L = next_fast_len(m)
    c = irfft(rfft(a, L) * rfft(b, L), L)[:m]

But wait, I should also account for the actual convergence rate. The LP-based method is essentially a subgradient method (or a projected subgradient method). The convergence rate is $O(1/\sqrt{K})$ for the objective value, which means we need many iterations. But since we're doing a full line search at each step (not a fixed step), the convergence can be much faster, especially near the optimum.

Let me also add a "restart" mechanism: if we stall for too long, restart from the best solution with a different random perturbation.

And at the end, I'll do coordinate descent:
- For each coordinate $i$, try adjusting $a_i$ to minimize the max convolution
- This is a 1-D problem (fix all other coordinates, optimize $a_i$)
- The max convolution as a function of $a_i$ is the max of a set of quadratics
- Find the minimum via golden section

Actually, coordinate descent is:
- Fix all $a_j$ for $j \neq i$
- $(a*a)[k] = \text{terms not involving } a_i + a_i \cdot (\text{coefficient involving } a_i)$
- For a specific $k$: $(a*a)[k] = c_k' + a_i \cdot \hat{a}_k + \delta_{k, 2i < 2n-1} \cdot a_i^2$ (roughly)
- The max over $k$ is the max of quadratics in $a_i$
- Minimize subject to $a_i \geq 0$ and $\sum a = 1$ (so other coordinates scale)

This is a bit involved. Let me simplify: for coordinate $i$, try a few values (current, current*0.5, current*1.5, current + delta, etc.) and pick the best. This is what the previous program's `_coordinate_polish` does.

The simpler one is: find $d$ that minimizes the max cross-convolution at active lags. This is like finding a direction that "evenly reduces" the contribution to the active constraints. It's a valid subgradient direction.

The "improved" one incorporates the current constraint values. import numpy as np
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
import time
import math

__fft_cache = {}

def _fft_len(m):
    m = int(m)
    if m not in _fft_cache:
        _fft_cache[m] = int(next_fast_len(m))
    return _fft_cache[m]

def _conv_same(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    L = _fft_len(m)
    fa = rfft(a, L)
    return np.maximum(irfft(fa * fa, L)[:m], 0.0)

def _lin_conv(a, b):
    n1, n2 = len(a), len(b)
    if n1 == 0 or n2 == 0:
        return np.array([], dtype=np.float64)
    m = n1 + n2 - 1
    L = _fft_len(m)
    return irfft(rfft(a, L) * rfft(b, L), L)[:m]

def _score_int(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    n = len(a)
    c = _conv_same(a)
    return float(2.0 * n * c.max() / (s * s))

def _exact_score(a):
    try:
        lst = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
        return float(evaluate_sequence(lst))
    except Exception:
        return float('inf')

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _lp_direction(f, n_active=30, use_mixed=False):
    n = len(f)
    c = _conv_same(f)
    M = float(c.max())
    sorted_idx = np.argsort(c)[::-1][:n_active]

    A_ub = np.zeros((n_active, n + 1), dtype=np.float64)
    b_ub = np.zeros(n_active, dtype=np.float64)

    for i, k in enumerate(sorted_idx):
        j_lo = max(0, k - n + 1)
        j_hi = min(n, k + 1)
        for j in range(j_lo, j_hi):
            fi = k - j
            if 0 <= fi < n:
                A_ub[i, j] = 2.0 * f[fi] if use_mixed else f[fi]
        A_ub[i, n] = -1.0
        if use_mixed:
            b_ub[i] = -c[k]

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0

    bounds = [(-f[i], 10.0) for i in range(n)] + [(None, None)]
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.array([0.0]),
                  bounds=bounds, method='highs', options={'presolve': True})
    if res is None or res.status != 0 or res.x is None:
        return None, M
    return res.x[:n].astype(np.float64), M

def _line_search(f, d, t_max_hint=1.0):
    neg = d < -1e-14
    if np.any(neg):
        t_max = t_max_hint
        t_max = min(t_max, float((f[neg] / (-d[neg])).min()) * 0.999)
    else:
        t_max = t_max_hint
    t_max = max(0.0, min(t_max, 5.0))
    if t_max < 1e-10:
        return 0.0

    c_f = _conv_same(f)
    c_fd = _lin_conv(f, d)
    c_dd = _lin_conv(d, d)

    def phi(t):
        return float(np.max(c_f + 2.0 * t * c_fd + t * t * c_dd))

    best_t = 0.0
    best_v = phi(0.0)
    ts = np.linspace(0.0, t_max, 65)
    for t in ts[1:]:
        v = phi(float(t))
        if v < best_v:
            best_v = v
            best_t = float(t)

    if 0.0 < best_t < t_max:
        idx = int(np.argmin(ts))
        step = t_max / 64.0
        lo = max(0.0, best_t - 2.0 * step)
        hi = min(t_max, best_t + 2.0 * step)
        for _ in range(35):
            m1 = lo + 0.381966 * (hi - lo)
            m2 = lo + 0.618034 * (hi - lo)
            v1 = phi(m1)
            v2 = phi(m2)
            if v1 < v2:
                hi = m2
            else:
                lo = m1
            if hi - lo < 1e-10:
                break
        best_t = (lo + hi) * 0.5
        best_v = phi(best_t)

    return best_t

def _coord_polish(f_arr, stop_time, seed=123):
    f_work = _normalize(f_arr.copy())
    n = len(f_work)
    best_sc = _score_int(f_work)
    rng = np.random.default_rng(seed)
    order = rng.permutation(n)
    mults = (0.0, 0.1, 0.25, 0.4, 0.6, 0.8, 1.0, 1.25, 1.5, 2.0, 3.0)

    for _ in range(3):
        if time.monotonic() > stop_time - 1.0:
            break
        improved = False
        for idx in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = float(f_work[idx])
            base = orig if orig > 1e-15 else 1.0 / n
            best_val = orig
            best_v = best_sc
            for m in mults:
                val = base * m
                if val >= 0.0 and val <= 1000.0:
                    f_work[idx] = val
                    sc = _score_int(f_work)
                    if sc < best_v:
                        best_v = sc
                        best_val = val
            f_work[idx] = best_val
            if best_v < best_sc:
                best_sc = best_v
                improved = True
        if not improved:
            break
    return f_work, best_sc

def _try_lm_polish(f_arr, stop_time, seed=99):
    """Simple Levenberg-Marquardt style: perturb coordinates and accept if better."""
    f_work = _normalize(f_arr.copy())
    n = len(f_work)
    best_sc = _score_int(f_work)
    rng = np.random.default_rng(seed)
    scale = 0.05 / math.sqrt(n)

    for _ in range(200):
        if time.monotonic() > stop_time - 1.0:
            break
        pert = rng.normal(0, scale, n)
        cand = f_work + pert
        cand = np.clip(cand, 0, 1000)
        cand = _normalize(cand)
        sc = _score_int(cand)
        if sc < best_sc:
            best_sc = sc
            f_work = cand
        else:
            scale *= 0.95
    return f_work, best_sc
