from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(416, dtype=np.float64)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    return p


def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=4.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout,
                               'primal_feasibility_tolerance': 1e-9,
                               'dual_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res.status != 0 or getattr(res, 'x', None) is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g


def _lp_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n)
    f = f / f_sum * math.sqrt(2.0 * n)

    best_f = f.copy()
    best_score = _score(f)

    t = 0.4
    t_min = 0.03
    t_max = 0.85
    stall = 0
    max_stall = 120
    convergence_threshold = 5e-10
    iteration = 0

    while time.monotonic() < stop_time - 6.0:
        if time.monotonic() - t0 > budget:
            break

        b = _conv(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, n, M, timeout=3.0)
        if g is None:
            stall += 8
            if stall > max_stall:
                break
            noise = rng.normal(0, 0.08 * max(1.0, f.mean()), n)
            f = np.clip(f + noise, 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs * math.sqrt(2.0 * n)
            else:
                f = best_f.copy()
            continue

        g_sum = float(g.sum())
        if g_sum < 1e-15:
            stall += 3
            continue

        f_snap = f.sum()
        g = g * (f_snap / g_sum)

        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < convergence_threshold:
            stall += 2
            if stall > max_stall:
                break
            continue

        improved = False
        best_t = t
        best_t_sc = _score(f)

        # Line search: try several t values
        ts = [t, t * 0.7, t * 0.4, t * 1.4, t * 0.25, t * 2.0]
        ts = [max(t_min, min(t_max, x)) for x in ts]
        ts = sorted(set(ts), key=lambda x: abs(x - t), reverse=True)

        for tt in ts:
            if time.monotonic() > stop_time - 6.0:
                break
            f_new = (1.0 - tt) * f + tt * g
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 0.01:
                continue
            sc = _score(f_new)
            if sc < best_t_sc - 1e-13:
                best_t = tt
                best_t_sc = sc
            if sc < best_score - 1e-13:
                f = f_new.copy()
                best_f = f.copy()
                best_score = sc
                improved = True
                if sc < best_t_sc:
                    t = min(t_max, tt * 1.15)
                else:
                    t = max(t_min, tt * 0.65)
                stall = 0
                break
        else:
            if best_t != t and best_t_sc < _score(f) - 1e-13:
                f_new = (1.0 - best_t) * f + best_t * g
                f_new = np.clip(f_new, 0, None)
                if f_new.sum() > 0.01:
                    sc = _score(f_new)
                    if sc < best_score:
                        f = f_new
                        best_f = f.copy()
                        best_score = sc
                        improved = True

        if not improved:
            stall += 1
            t = max(t_min, t * 0.7)
            if stall >= max_stall:
                # Perturbation to escape
                scale = 1.0 + 0.15
                noise_std = 0.1 * max(1.0, f.mean())
                f = np.clip(f * scale + rng.normal(0, noise_std, n), 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * math.sqrt(2.0 * n)
                t = 0.4
                stall = 0
                max_stall += 30

        iteration += 1

    return best_f, best_score


def _polish(a, stop_time, seed=4242):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 0.01:
        return best, _score(best)
    best = best / fs * math.sqrt(2.0 * n)
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 8

    while time.monotonic() < stop_time - 6.0:
        improved = False

        # Single-coordinate sweep
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 6.0:
                return best, best_sc
            orig = best[i]
            if orig < 1e-14:
                # Try setting to small positive
                for val in (0.01, 0.05, 0.1):
                    best[i] = val
                    if best.sum() < 0.01:
                        best[i] = orig
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        orig = val
                        best_sc = sc
                        improved = True
                    else:
                        best[i] = 0.0
                best[i] = orig
                continue

            multipliers = (0.05, 0.15, 0.3, 0.5, 0.7, 1.0, 1.3, 1.6, 2.2, 3.5, 0.0)
            best_val = orig
            best_val_sc = best_sc
            for mlt in multipliers:
                best[i] = orig * mlt
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-14:
                    best_val = best[i]
                    best_val_sc = sc
            if best_val != orig or best_val_sc < best_sc - 1e-14:
                best[i] = best_val
                best_sc = best_val_sc
                improved = True
            else:
                best[i] = orig

        # Two-coordinate
        for _ in range(60):
            if time.monotonic() > stop_time - 6.0:
                return best, best_sc
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.3, 0.5, 0.7, 1.4, 2.0, 0.0):
                for mj in (0.3, 0.5, 0.7, 1.4, 2.0, 0.0):
                    if mi == 1.0 and mj == 1.0:
                        continue
                    best[i] = oi * mi if oi > 1e-14 else 0.0
                    best[j] = oj * mj if oj > 1e-14 else 0.0
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
                        break
            best[i], best[j] = oi, oj

        # Random perturbations
        if not improved:
            for _ in range(20):
                if time.monotonic() > stop_time - 6.0:
                    break
                k = int(rng.integers(n))
                pert = rng.normal(0, 0.03 * max(1.0, best.mean()))
                old = best[k]
                best[k] = max(0.0, old + pert)
                if best.sum() < 0.01:
                    best[k] = old
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                else:
                    best[k] = old

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    global t0, budget
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 18.0

    parent = _get_parent()
    n_p = len(parent)

    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(256)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try(n, init_arr, stop, sd):
        nonlocal best_seq, best_score
        if time.monotonic() > stop:
            return
        try:
            res, sc = _lp_optimize(init_arr, n, stop, sd)
        except Exception:
            return
        if res is None or len(res) == 0:
            return
        if not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0, 1000)
        if res.sum() < 0.01:
            return
        if sc < best_score - 1e-14:
            try:
                exact = float(evaluate_sequence([float(x) for x in res]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-13:
                best_seq = res.copy()
                best_score = exact

    # Phase 1: LP optimization on parent length with multiple inits
    n = n_p
    phase1_end = t0 + budget * 0.45
    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    inits = []
    inits.append(parent.copy())

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n)))

    # Multi-frequency
    sup = np.ones(n) * 0.3
    for fr in (1, 2, 3, 5, 7, 11):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.15 * np.cos(2.0 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.02)

    # Random abs-normal
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.05)

    # Parent + noise
    sig = 0.2 * max(1.0, float(np.mean(parent)))
    inits.append(np.abs(parent + rng_main.normal(0, sig, n)) + 0.01)

    # Gaussian bump
    center = (n - 1) / 2.0
    width = n / 3.5
    inits.append(np.exp(-0.5 * ((i_arr - center) / width) ** 2) + 0.01)

    # Two-lobe
    two = np.exp(-0.5 * ((i_arr - n * 0.28) / (n * 0.13)) ** 2) + \
          np.exp(-0.5 * ((i_arr - n * 0.72) / (n * 0.13)) ** 2) + 0.01
    inits.append(two)

    # Periodic comb
    period = max(2, n // 8)
    comb = np.ones(n)
    for k in range(0, n, period):
        comb[k] = 2.5
    inits.append(comb + 0.05)

    # Parent interpolated with different cosines
    cos_init = parent.copy()
    for k in range(n):
        cos_init[k] *= (1.0 + 0.3 * np.cos(2.0 * math.pi * k / n))
    inits.append(np.abs(cos_init) + 0.01)

    num_inits = len(inits)
    time_per_init = max(15.0, (phase1_end - t0) / num_inits)

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 8.0:
            break
        if best_score < 1.42:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        _try(n, init, stop, seed + 7777 * (idx + 1))

    # Run parent init again with more time if it's still the best
    if best_seq is fallback and time.monotonic() < phase1_end - 20.0:
        stop_extra = min(phase1_end - 5.0, time.monotonic() + (phase1_end - time.monotonic() - 10))
        _try(n, parent.copy(), stop_extra, seed + 999001)

    # Phase 2: Explore other lengths
    phase2_end = t0 + budget * 0.65
    other_lengths = [192, 224, 256, 288, 320, 352, 384, 416, 448, 480, 512, 544, 576]
    other_lengths = [L for L in other_lengths if L >= 128]
    if n_p in other_lengths:
        other_lengths.remove(n_p)
    time_per_len = max(8.0, (phase2_end - max(phase1_end, t0)) / max(1, len(other_lengths)))

    for L in other_lengths:
        if time.monotonic() > phase2_end - 8.0:
            break
        if best_score < 1.42:
            break
        stop = min(phase2_end - 5.0, time.monotonic() + time_per_len)
        if abs(L - len(best_seq)) <= 8:
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
        else:
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(parent)), parent)
        _try(L, init_l, stop, seed + 50000 + L)

    # Phase 3: Second LP pass on best (warm start)
    phase3_end = t0 + budget * 0.82
    if time.monotonic() < phase3_end - 25.0:
        stop_p3 = min(phase3_end - 5.0, time.monotonic() + (phase3_end - time.monotonic() - 10))
        _try(len(best_seq), best_seq.copy(), stop_p3, seed + 999999)

    # Also try a fresh init at the best length
    if time.monotonic() < phase3_end - 30.0:
        n_best = len(best_seq)
        i_b = np.arange(n_best, dtype=np.float64)
        extra_init = np.abs(np.sin(2.0 * math.pi * i_b * 3 / n_best) +
                           0.5 * np.cos(2.0 * math.pi * i_b * 7 / n_best) + 1.0) + 0.02
        stop_p3b = min(phase3_end - 5.0, time.monotonic() + 60.0)
        _try(n_best, extra_init, stop_p3b, seed + 123456)

    # Phase 4: Polish
    polish_end = deadline - 8.0
    if time.monotonic() < polish_end - 10.0:
        try:
            best_seq, best_score = _polish(best_seq, polish_end, seed=4242)
        except Exception:
            pass

    # Second polish with different seed
    if time.monotonic() < deadline - 12.0:
        try:
            best_seq2, sc2 = _polish(best_seq, deadline - 8.0, seed=87654)
            if sc2 < best_score - 1e-14:
                best_seq = best_seq2
                best_score = sc2
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    if final_sc < best_score:
        best_score = final_sc

    # Ensure we have headroom
    if time.monotonic() > t0 + budget - 5.0:
        pass  # Return immediately

    return [float(x) for x in best_seq]
