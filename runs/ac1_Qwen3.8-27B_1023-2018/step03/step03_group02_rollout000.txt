def _smooth_obj_jac(z, kappa, K, n):
    z = np.asarray(z, dtype=float).ravel()
    if z.size != n: z = z[:n]; pad...
    z = np.where(np.isfinite(z), z, 0.0)
    zm = np.max(z)
    a = np.exp(z-zm)
    s = a.sum()
    if s <= 1e-300:
       a = np.ones(n)/n
    else: a /= s
    c = _conv(a)
    if K.size == 0:
       return 0.0, np.zeros(n)
    vals = c[K]
    if vals.size == 0: return 0, zeros
    vmax = float(vals.max())
    zeta = kappa * (vals - vmax)
    zeta = np.clip(zeta, -50.0, 0.0)
    e = np.exp(zeta)
    den = float(e.sum())
    if den <= 0: w = np.zeros_like(e); active_idx=[np.argmax(vals)]...
    else: w = e/den
    obj = (vmax + math.log(den))/kappa if den>1e-300 else vmax
    grad_a = np.zeros(n)
    center = n-1
    # active threshold
    thresh = max(1e-8, 1e-6 / K.size)
    active = np.nonzero(w > thresh)[0]
    if active.size == 0:
       active = np.array([int(np.argmax(vals))])
    for idx in active:
       wi = float(w[idx]); l=int(K[idx]); d=l-center
       if d == 0: grad_a += 2.0*wi*a
       elif d > 0:
           m = n-d
           if m>0:
              grad_a[:m] += wi*a[d:]
              grad_a[d:] += wi*a[:-d]
       else:
           dd=-d; m=n-dd
           if m>0:
              grad_a[dd:] += wi*a[:-dd]
              grad_a[:m] += wi*a[dd:]
    gdot = float(np.dot(grad_a,a))
    gz = a * (grad_a - gdot*a)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    return float(obj), gz

def _parent_array():

    return None


def _clean(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    return np.clip(a, 0.0, 1000.0)


def _normalize(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    n = a.size
    if s <= 1e-12:
        return np.full(n, float(target) / max(1, n), dtype=np.float64)
    a = a * (float(target) / s)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 <= 1e-12:
        return np.full(n, float(target) / max(1, n), dtype=np.float64)
    a = a * (float(target) / s2)
    return a


def _conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 1024 or rfft is None:
        c = np.convolve(a, a, mode="full").astype(np.float64, copy=False)
        return c
    L = next_fast_len(2 * n - 1)
    F = rfft(a, L)
    c = irfft(F * F, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    np.maximum(c, 0.0, out=c)
    return c


def _fast_score(a, kind):
    a = _clean(a)
    n = a.size
    if n == 0:
        return _INF
    s = float(a.sum())
    if s < 0.01 or not math.isfinite(s):

    c = _conv(a)

def _K_all(n):
    return np.arange(2 * n - 1, dtype=int) if n > 0 else np.array([], dtype=int)


def _K_off(n):
    if n <= 1:
        return np.array([], dtype=int)
    return np.r_[np.arange(0, n - 1, dtype=int), np.arange(n, 2 * n - 1, dtype=int)]


def _softmax_array(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([1.0], dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    a = np.exp(z)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones_like(a)
        s = float(a.sum())
    return a / s


def _smooth_obj_jac(z, kappa, K, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        if z.size < n:
            z = np.concatenate([z, np.zeros(n - z.size)])
        else:
            z = z[:n]
    z = np.where(np.isfinite(z), z, 0.0)
    zm = float(np.max(z))
    a = np.exp(z - zm)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.full(n, 1.0 / max(1, n))
    else:
        a = a / s

    c = _conv(a)
    if K.size == 0:
        return 0.0, np.zeros(n, dtype=np.float64)

    vals = c[K]
    if vals.size == 0:
        return 0.0, np.zeros(n, dtype=np.float64)

    vmax = float(vals.max())
    zeta = kappa * (vals - vmax)
    zeta = np.clip(zeta, -50.0, 0.0)
    e = np.exp(zeta)
    den = float(e.sum())
    if den <= 1e-300:
        w = np.zeros_like(vals)
        w[int(np.argmax(vals))] = 1.0
        obj = vmax
    else:
        w = e / den
        obj = (vmax + math.log(den)) / kappa

    grad_a = np.zeros(n, dtype=np.float64)
    ctr = n - 1
    thresh = max(1e-8, 1e-6 / max(1, K.size))
    active = np.nonzero(w > thresh)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(vals))], dtype=int)

    for idx in active:
        wi = float(w[idx])
        if wi <= 1e-20:
            continue
        l = int(K[idx])
        d = l - ctr
        if d == 0:
            grad_a += 2.0 * wi * a
        elif d > 0:
            m = n - d
            if m > 0:
                grad_a[:m] += wi * a[d:]
                grad_a[d:] += wi * a[:-d]
        else:
            dd = -d
            m = n - dd
            if m > 0:
                grad_a[dd:] += wi * a[:-dd]
                grad_a[:m] += wi * a[dd:]

    gdot = float(np.dot(grad_a, a))
    gz = a * (grad_a - gdot * a)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    return float(obj), gz


def _lbfgs_refine(f0, kind, stop, seed):
    n = f0.size
    if n <= 1 or minimize is None:
        return _normalize(f0), _exact_score(f0)

    best = _normalize(f0)
    bsc = _exact_score(best)
    K = _K_off(n) if kind == "off" else _K_all(n)
    if K.size == 0:
        return best, bsc

    for kappa in (250.0, 700.0, 1800.0):
        if time.monotonic() > stop - 5.0:
            break
        f = best.copy() if math.isfinite(bsc) else _normalize(f0)
        z = np.log(f + 1e-12)
        z = z - float(np.mean(z))

        def fun(zv, karr=K, nn=n, kk=kappa):
            return _smooth_obj_jac(zv, kk, karr, nn)

        try:
            res = minimize(
                fun,
                z,
                jac=True,
                method="L-BFGS-B",
                options={"maxiter": 60, "maxls": 12, "ftol": 1e-13, "gtol": 1e-9, "disp": False},
            )
        except Exception:
            res = None

        if res is not None and getattr(res, "x", None) is not None:
            a = _softmax_array(res.x)
            a = _normalize(a)
            sc = _exact_score(a)
            if math.isfinite(sc) and sc < bsc - 1e-14:
                best = a.copy()
                bsc = sc

    return best, bsc


def _line_search(cur, direction, cur_sc):
    base = _normalize(direction)
    for t in (1.0, 0.85, 0.65, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.01):
        if t >= 1.0:
            cand = base
        else:
            cand = (1.0 - t) * _normalize(cur) + t * base
        cand = _normalize(cand)
        sc = _exact_score(cand)
        if math.isfinite(sc) and (not math.isfinite(cur_sc) or sc < cur_sc * (1.0 - 1e-12)):
            return cand, sc, min(1.0, max(1e-3, t))
    return None, None, None


def _pg_candidate(f, c, kind, eta, rw=32, kappa=250.0):
    n = f.size
    K = _K_off(n) if kind == "off" else _K_all(n)
    if K.size == 0:
        return f.copy()
    vals = c[K]
    if vals.size == 0:
        return f.copy()

    m = min(int(rw), vals.size)
    idx = np.argpartition(-vals, m - 1)[:m] if m < vals.size else np.arange(vals.size)
    tv = vals[idx]
    vmax = float(tv.max())
    z = kappa * (tv - vmax)
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    den = float(e.sum())
    if den <= 1e-300:
        return f.copy()
    w = e / den

    grad = np.zeros(n, dtype=np.float64)
    ctr = n - 1
    for wi, ll in zip(w, K[idx]):
        wi = float(wi)
        if wi <= 1e-10:
            continue
        d = int(ll) - ctr
        if d == 0:
            grad += 2.0 * wi * f
        elif d > 0:
            mm = n - d
            if mm > 0:
                grad[:mm] += wi * f[d:]
                grad[d:] += wi * f[:-d]
        else:
            dd = -d
            mm = n - dd
            if mm > 0:
                grad[dd:] += wi * f[:-dd]
                grad[:mm] += wi * f[dd:]

    new = f - eta * grad
    return _normalize(new)


def _lp_direction(f, c, kind, r, time_limit, rng):
    n = f.size
    if n <= 1 or c.size < 2 * n - 1 or linprog is None:
        return None

    K_all = _K_off(n) if kind == "off" else _K_all(n)
    if K_all.size == 0:
        return None

    vals = c[K_all]
    if vals.size == 0:
        return None

    valmax = float(vals.max())
    if not math.isfinite(valmax) or valmax <= 1e-16:
        return None

    r = int(min(max(1, r), vals.size))
    part = np.argpartition(-vals, r - 1)[:r] if r < vals.size else np.arange(vals.size)
    order = part[np.argsort(-vals[part])]
    K = K_all[order]
    M = c[K].astype(np.float64, copy=True)
    M = M + 1e-12 * max(1.0, valmax)
    M = np.maximum(M, 1e-15)

    A = None
    if sp is not None:
        rows = []
        cols = []
        data = []
        for row, l in enumerate(K):
            j0 = max(0, l - (n - 1))
            j1 = min(n - 1, l)
            if j1 >= j0:
                m = j1 - j0 + 1
                rows.extend([row] * m)
                cols.extend(range(j0, j1 + 1))
                data.extend(f[l - j1 : l - j0 + 1][::-1])
        if data:
            try:
                A = sp.coo_matrix(
                    (np.asarray(data, dtype=np.float64),
                     (np.asarray(rows, dtype=int), np.asarray(cols, dtype=int))),
                    shape=(len(K), n),
                ).tocsr()
            except Exception:
                A = None
    if A is None:
        A = np.zeros((len(K), n), dtype=np.float64)
        for row, l in enumerate(K):
            j0 = max(0, l - (n - 1))
            j1 = min(n - 1, l)
            if j1 >= j0:
                A[row, j0:j1 + 1] = f[l - j1 : l - j0 + 1][::-1]

    try:
        res = linprog(
            -np.ones(n, dtype=np.float64),
            A_ub=A,
            b_ub=np.asarray(M, dtype=np.float64),
            bounds=[(0.0, 1.0)] * n,
            method="highs",
            options={
                "presolve": True,
                "time_limit": max(0.1, float(time_limit)),
            },
        )
    except Exception:
        return None

    if res is None or getattr(res, "x", None) is None:
        return None
    if getattr(res, "status", None) not in (0, 3):
        return None

    g = np.asarray(res.x, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.clip(g, 0.0, None)
    gs = float(g.sum())
    if gs <= 1e-12:
        return None
    return g / gs


def _lp_refine(f0, kind, stop, seed):
    n = f0.size
    if n <= 1:
        return _normalize(f0), _exact_score(f0)

    cur = _normalize(f0)
    best = cur.copy()
    cur_sc = _exact_score(cur)
    best_sc = cur_sc
    if not math.isfinite(cur_sc):
        cur = np.full(n, 1.0 / n)
        cur_sc = _exact_score(cur)
        best = cur.copy()
        best_sc = cur_sc

    c = _conv(cur)
    rng = np.random.default_rng(seed)
    no_imp = 0
    it = 0

    while time.monotonic() < stop - 2.0:
        it += 1
        if cur_sc < best_sc - 1e-14:
            best_sc = cur_sc
            best = cur.copy()

        improved = False
        g = None
        if linprog is not None and time.monotonic() < stop - 6.0:
            r = 150 if n <= 512 else 80
            tl = min(3.5, max(0.35, stop - time.monotonic() - 6.0))
            g = _lp_direction(cur, c, kind, r, tl, rng)

        if g is not None and np.all(np.isfinite(g)):
            if float(np.linalg.norm(g - cur)) > 1e-7:
                new, sc, _ = _line_search(cur, g, cur_sc)
                if new is not None and math.isfinite(sc) and sc < cur_sc * (1.0 - 1e-12):
                    cur = new
                    cur_sc = sc
                    c = _conv(cur)
                    improved = True
                    no_imp = 0

        if not improved:
            opts = (
                (0.05, 16, 150.0),
                (0.2, 16, 250.0),
                (1.0, 32, 350.0),
                (3.0, 32, 500.0),
                (10.0, 64, 700.0),
            )
            for eta, rw, kp in opts:
                if time.monotonic() > stop - 2.0:
                    break
                cand = _pg_candidate(cur, c, kind, eta, rw, kp)
                sc = _exact_score(cand)
                if math.isfinite(sc) and sc < cur_sc * (1.0 - 1e-12):
                    cur = cand
                    cur_sc = sc
                    c = _conv(cur)
                    improved = True
                    no_imp = 0
                    break

        if not improved:
            no_imp += 1
            if no_imp >= 18:
                scale = 0.04 + 0.05 * min(1.0, no_imp / 40.0)
                cur = _normalize(cur + rng.normal(0.0, scale, n))
                if math.isfinite(best_sc) and rng.random() < 0.35:
                    other = best.copy()
                    if other.size != n:
                        other = _normalize(other)
                    cur = _normalize(0.65 * other + 0.35 * cur)
                cur_sc = _exact_score(cur)
                if math.isfinite(cur_sc) and cur_sc < best_sc - 1e-14:
                    best_sc = cur_sc
                    best = cur.copy()
                c = _conv(cur)
                no_imp = 0
            elif no_imp >= 12 and it % 25 == 0 and math.isfinite(best_sc):
                cur = best.copy()
                cur_sc = best_sc
                c = _conv(cur)
                no_imp = 6

    if cur_sc < best_sc - 1e-14:
        best = cur.copy()
        best_sc = cur_sc
    return best, best_sc


def _coordinate_polish(f0, kind, stop, seed):
    n = f0.size
    f = _normalize(f0)
    best = f.copy()
    bsc = _exact_score(best)
    cur = f.copy()
    cur_sc = bsc
    cur_fast = _fast_score(f, kind)
    rng = np.random.default_rng(seed)
    passes = 0

    while time.monotonic() < stop - 4.0 and passes < 30:
        passes += 1
        improved_any = False
        mean = float(f.sum()) / max(1, n)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop - 4.0:
                break
            v0 = float(f[i])
            base = max(abs(v0), 0.05 * mean, 1e-4)
            cands = {0.0, v0}
            for m in (0.05, 0.1, 0.2, 0.4, 0.65, 0.85, 1.2, 1.5, 1.9, 2.6, 4.0):
                cands.add(v0 * m if v0 > 1e-12 else base * m)
            for m in (0.05, 0.15, 0.35, 0.7, 1.5, 3.0, 6.0):
                cands.add(mean * m)

            bestv = v0
            bestfast = cur_fast
            bestdetail = cur_sc

            for v in cands:
                if not math.isfinite(v) or v < 0.0 or v > 1000.0:
                    continue
                f[i] = float(v)
                fs = _fast_score(f, kind)
                if math.isfinite(fs) and (not math.isfinite(bestfast) or fs < bestfast * (1.0 - 1e-12)):
                    sc = _exact_score(f)
                    if math.isfinite(sc) and (not math.isfinite(bestdetail) or sc < bestdetail * (1.0 - 1e-12)):
                        bestdetail = sc
                        bestfast = fs
                        bestv = float(v)

            f[i] = bestv
            if math.isfinite(bestdetail) and bestdetail < cur_sc * (1.0 - 1e-12):
                cur_sc = bestdetail
                cur_fast = bestfast
                if cur_sc < bsc * (1.0 - 1e-12):
                    bsc = cur_sc
                    best = f.copy()
                improved_any = True

        f = _normalize(f)
        sc = _exact_score(f)
        if math.isfinite(sc):
            cur_sc = sc
            if sc < bsc * (1.0 - 1e-12):
                bsc = sc
                best = f.copy()
                improved_any = True
        cur_fast = _fast_score(f, kind)

        if not improved_any and passes >= 4:
            break

    return best, bsc


def _best_scale(a):
    a = _clean(a)
    if a.size == 0 or float(a.sum()) <= 1e-12:
        n = a.size if a.size > 0 else 384
        a = np.ones(n, dtype=np.float64)
        return a, _exact_score(a)

    s = float(a.sum())
    mx = float(a.max())
    cands = [a.copy()]

    def add(arr):
        arr = _clean(arr)
        if arr.size == a.size and float(arr.sum()) >= 0.01 and float(arr.max()) <= 1000.0:
            cands.append(arr)

    for ts in (1.0, math.sqrt(2.0 * float(a.size)), float(a.size), 10.0):
        if ts > 0.0:
            add(a * (ts / s))
    for mt in (1.0, 10.0, 100.0):
        if mx > 0.0:
            add(a * (mt / mx))

    best = a.copy()
    bsc = _INF
    for arr in cands:
        sc = _exact_score(arr)
        if math.isfinite(sc) and sc < bsc - 1e-15:
            best = arr.copy()
            bsc = sc
    if not math.isfinite(bsc):
        best = _normalize(a)
        bsc = _exact_score(best)
    return best, bsc


def _detect_kind(n, parent):
    probes = [np.ones(n, dtype=np.float64)]
    if parent is not None and parent.size == n:
        probes.append(parent)
    else:
        i = np.arange(n, dtype=np.float64)
        probes.append(0.25 + 0.75 * np.cos(pi = math.pi, x=i / max(1, n - 1)))

    full_err = 0.0
    off_err = 0.0
    cnt = 0
    for a in probes:
        aa = _normalize(a)
        exact = _exact_score(aa)
        if not math.isfinite(exact):
            continue
        c = _conv(aa)
        sn = aa.size
        s = float(aa.sum())
        if s <= 0.0 or c.size == 0:
            continue
        full = 2.0 * float(sn) * float(c.max()) / (s * s)
        ctr = sn - 1
        if sn == 1:
            off = 0.0
        else:
            ml = float(c[:ctr].max()) if ctr > 0 else 0.0
            mr = float(c[ctr + 1:].max()) if ctr + 1 < c.size else 0.0
            off = 2.0 * float(sn) * max(ml, mr) / (s * s)
        full_err += abs(exact - full) / max(1.0, abs(exact))
        off_err += abs(exact - off) / max(1.0, abs(exact))
        cnt += 1

    if cnt == 0:
        return "full"
    return "off" if off_err < full_err else "full"


def _off_search(n, best0, bsc0, parent, stop, seed):
    best = _normalize(best0 if best0 is not None else np.ones(n, dtype=np.float64))
    bsc = bsc0 if math.isfinite(bsc0) else _exact_score(best)
    if not math.isfinite(bsc):
        best = np.full(n, 1.0 / n)
        bsc = _exact_score(best)

    rng = np.random.default_rng(seed)
    inits = []

    if parent is not None:
        p = parent if parent.size == n else np.interp(
            np.linspace(0.0, 1.0, n),
            np.linspace(0.0, 1.0, parent.size),
            parent
        )
        p = _normalize(p)
        if float(p.sum()) > 1e-12:
            inits.append(p)
            inits.append((p + p[::-1]) * 0.5)
            for pw in (0.55, 0.75, 1.35, 1.8):
                inits.append(np.power(p + 1e-12, pw))
            noise = rng.normal(0.0, 1.0, n)
            kern = np.exp(-0.5 * (np.arange(-6, 7) / 2.6) ** 2)
            kern = kern / max(1e-12, kern.sum())
            smooth = np.convolve(noise, kern, mode="same")
            inits.append(0.7 * p + 0.3 * np.abs(smooth))
            inits.append(np.abs(p + 0.25 * smooth))
            for q in (0.55, 0.75, 0.88):
                thr = np.quantile(p, q)
                inits.append((p >= thr).astype(float))

    i = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n, dtype=np.float64))
    inits.append(0.5 + 0.5 * np.cos(2.0 * math.pi * i / float(max(1, n))))
    if n > 1:
        inits.append(0.2 + 0.8 * (0.5 + 0.5 * np.cos(math.pi * i / (n - 1))))
    for p in (0.35, 0.5, 0.65):
        inits.append((rng.random(n) < p).astype(float))

    for a in inits[:36]:
        aa = _normalize(a)
        sc = _exact_score(aa)
        if math.isfinite(sc) and sc < bsc - 1e-14:
            best = aa.copy()
            bsc = sc

    now = time.monotonic()
    rem = max(0.0, stop - now)
    if rem > 450.0 and minimize is not None:
        smooth_stop = now + min(280.0, 0.25 * rem)
        best, bsc = _lbfgs_refine(best, "off", smooth_stop, seed + 1)

    now = time.monotonic()
    rem = max(0.0, stop - now)
    coord_reserve = 0.0
    if rem > 260.0:
        coord_reserve = min(320.0, 0.30 * rem)
    elif rem > 110.0:
        coord_reserve = min(140.0, 0.30 * rem)
    stop_lp = stop - coord_reserve - 5.0

    if time.monotonic() < stop_lp - 15.0:
        best, bsc = _lp_refine(best, "off", stop_lp, seed + 2)

    if time.monotonic() < stop - 5.0:
        best, bsc = _coordinate_polish(best, "off", stop, seed + 3)

    return best, bsc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget
    stop = t0 + max(5.0, budget - 15.0)
    if stop > hard - 1.0:
        stop = hard - 1.0
    if stop <= t0:
        stop = t0 + max(1.0, 0.5 * budget)

    parent = _parent_array()
    if parent is not None:
        if parent.size <= 4096:
            n = int(parent.size)
            p = _clean(parent)
        else:
            n = 384
            idx = np.linspace(0.0, parent.size - 1, n).astype(int)
            p = _clean(parent[idx])
        if p.size != n:
            p = _clean(np.interp(np.linspace(0.0, 1.0, n), np.linspace(0.0, 1.0, p.size), p))
        if float(p.sum()) <= 1e-12:
            p = None
    else:
        n = 384
        p = None

    uniform = _normalize(np.ones(n, dtype=np.float64))
    best = uniform.copy()
    best_sc = _exact_score(best)

    if p is not None:
        pn = _normalize(p)
        sc = _exact_score(pn)
        if math.isfinite(sc) and sc < best_sc - 1e-14:
            best = pn.copy()
            best_sc = sc

    kind = "full"
    if budget > 25.0:
        kind = _detect_kind(n, p)

    run_off = bool(math.isfinite(best_sc) and best_sc < 1.998)
    if kind == "off" and (stop - time.monotonic()) > 60.0:
        run_off = True

    if run_off:
        try:
            best, best_sc = _off_search(n, best, best_sc, p, stop, int(seed))
        except Exception:
            pass

    try:
        best, _ = _best_scale(best)
    except Exception:
        best = _normalize(np.ones(n, dtype=np.float64))

    final = [float(x) for x in best]
    final_sc = _exact_score(np.asarray(final))
    if not math.isfinite(final_sc):
        fallback = [1.0] * n
        final = fallback
        final_sc = _exact_score(np.asarray(final))
    if not math.isfinite(final_sc):
        return [1.0]
    return final
