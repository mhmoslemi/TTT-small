            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-12:
            best_seq = np.array(seq, dtype=np.float64).copy()
            best_score = exact

    # ===== Phase 1: LP optimization on parent length =====
    phase1_end = t0 + budget * 0.35
    n = n_p

    # Multiple diverse initializations
    i_arr = np.arange(n, dtype=np.float64)
    inits = []
    inits.append(parent / parent.sum())  # normalized parent
    inits.append(np.ones(n) / n)
    inits.append(np.clip(0.3 + 0.7 * np.cos(np.pi * i_arr / (n - 1)) ** 2, 0, None) / (0.3 + 0.7 * np.cos(np.pi * i_arr / (n - 1)) ** 2).sum())
    inits.append(np.clip(0.5 + 0.5 * np.cos(2 * np.pi * 2 * i_arr / n), 0, None) / (0.5 + 0.5 * np.cos(2 * np.pi * 2 * i_arr / n)).sum())
    rng0 = np.random.default_rng(seed)
    noisy = np.clip(parent + rng0.normal(0, 0.1, n), 0, None)
    if noisy.sum() > 0.01:
        inits.append(noisy / noisy.sum())

    inits = [np.clip(x, 0, None) / max(x.sum(), 1e-10) for x in inits]

    time_per_init = max(15.0, (phase1_end - t0) / len(inits))
    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        if best_score < 1.5028:
            break
        stop_lp = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        try:
            f_try, sc = _lp_optimize(init, n, stop_lp, seed + 100 * (idx + 1))
        except Exception:
            continue
        if np.all(np.isfinite(f_try)) and f_try.sum() > 0.01:
            f_try = np.clip(f_try, 0, 1000)
            _update(f_try, sc)

    # Additional LP rounds on best
    if time.monotonic() < phase1_end - 20.0:
        try:
            f_try, sc = _multi_start_lp(best_seq, n, phase1_end, seed + 5000)
        except Exception:
            pass
        if np.all(np.isfinite(f_try)) and f_try.sum() > 0.01:
            _update(f_try, sc)

    # ===== Phase 2: Coordinate descent polish =====
    phase2_end = t0 + budget * 0.60
    if time.monotonic() < phase2_end - 10.0:
        try:
            f_polished, sc = _coord_descent(best_seq, phase2_end, max_sweeps=150, seed=4242)
        except Exception:
            f_polished, sc = best_seq, best_score
        if np.all(np.isfinite(f_polished)) and f_polished.sum() > 0.01:
            _update(f_polished, sc)

    # ===== Phase 3: Try other lengths =====
    phase3_end = t0 + budget * 0.80
    other_lengths = []
    for L in [128, 160, 198, 256, 320, 384, 448, 512, 600, 700]:
        if L != n_p:
            other_lengths.append(L)
    other_lengths.sort(key=lambda L: abs(L - n_p))

    time_per_len = max(12.0, (phase3_end - time.monotonic()) / max(1, len(other_lengths)))

    for L in other_lengths[:6]:
        if time.monotonic() > phase3_end - 15.0:
            break
        if best_score < 1.5028:
            break

        # Interpolate best to new length
        if best_seq.size > 1:
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, best_seq.size), best_seq)
        else:
            init_l = np.ones(L) / L
        init_l = np.clip(init_l, 0, None)
        if init_l.sum() < 0.01:
            init_l = np.ones(L) / L
        else:
            init_l = init_l / init_l.sum()

        stop_lp = min(phase3_end - 10.0, time.monotonic() + time_per_len * 0.6)
        try:
            f_try, sc = _lp_optimize(init_l, L, stop_lp, seed + 50000 + L)
        except Exception:
            continue
        if np.all(np.isfinite(f_try)) and f_try.sum() > 0.01:
            f_try = np.clip(f_try, 0, 1000)
            _update(f_try, sc)
            # Quick polish
            if time.monotonic() < phase3_end - 20.0 and best_score < 1.55:
                try:
                    f_p, sc_p = _coord_descent(f_try, min(phase3_end - 15.0, time.monotonic() + 60), max_sweeps=40, seed=L)
                except Exception:
                    pass
                if np.all(np.isfinite(f_p)) and f_p.sum() > 0.01:
                    _update(f_p, sc_p)

    # ===== Phase 4: Intensive coordinate descent on best =====
    final_end = hard_deadline - 20.0
    if time.monotonic() < final_end - 30.0:
        try:
            f_pol, sc = _coord_descent(best_seq, final_end, max_sweeps=300, seed=99999)
        except Exception:
            pass
        if np.all(np.isfinite(f_pol)) and f_pol.sum() > 0.01:
            _update(f_pol, sc)

    # Pair moves for final squeeze
    if time.monotonic() < hard_deadline - 15.0:
        n_b = len(best_seq)
        best_seq = best_seq.copy()
        rng_p = np.random.default_rng(seed + 7777)
        cur_sc = best_score
        for _ in range(200):
            if time.monotonic() > hard_deadline - 12.0:
                break
            i, j = int(rng_p.integers(n_b)), int(rng_p.integers(n_b))
            if i == j:
                continue
            oi, oj = best_seq[i], best_seq[j]
            best_pair_sc = cur_sc
            best_pair = (oi, oj)
            for mi in (0.0, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
                for mj in (0.0, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
                    ni = oi * mi if oi > 1e-12 else (0.01 if mi > 0 else 0.0)
                    nj = oj * mj if oj > 1e-12 else (0.01 if mj > 0 else 0.0)
                    best_seq[i], best_seq[j] = ni, nj
                    if best_seq.sum() < 0.01:
                        best_seq[i], best_seq[j] = oi, oj
                        continue
                    sc = _score(best_seq)
                    if sc < best_pair_sc - 1e-14:
                        best_pair_sc = sc
                        best_pair = (ni, nj)
            best_seq[i], best_seq[j] = best_pair
            if best_pair_sc < cur_sc - 1e-14:
                cur_sc = best_pair_sc
            else:
                best_seq[i], best_seq[j] = oi, oj
        _update(best_seq, _score(best_seq))

    # ===== Final validation =====
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
