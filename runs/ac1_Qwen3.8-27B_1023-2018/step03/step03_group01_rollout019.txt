Define F(a) = max_k (a*a)[k] subject to sum(a) = 1, a >= 0.

The subgradient of F at a (for the active k) is:
∂F/∂a_i = 2 * sum_{j: k-j valid} a[k-j] for each active k.

Wait, (a*a)[k] = sum_j a[j] * a[k-j]. The gradient w.r.t. a_i is:
d/d a_i [(a*a)[k]] = a[k-i] + a[i-k] (if both indices valid)

Actually: (a*a)[k] = sum_j a[j] * a[k-j].
d/d a_i = a[k-i] + (if k-i != i) a[i-k]
= a[k-i] + a[i-k] (when i ≠ k-i, i.e., 2i ≠ k)
= 2*a[i] (when 2i = k, i.e., i = k/2)

So the subgradient of max_k (a*a)[k] is the convex hull of the gradients of the active lags.

For the modified objective F(a) = max_k (a*a)[k] / (sum a)^2:
Since we fix sum(a) = 1: F(a) = max_k (a*a)[k], and the constraint sum(a) = 1, a >= 0.

Projected subgradient:
a_{new} = project_{sum=1, a>=0}(a - lr * grad)

where project is: subtract mean, clip at 0, renormalize. Actually for the constraint {a >= 0, sum(a) = 1}, the projection is the Euclidean projection onto the probability simplex.

This is the "water-filling" algorithm:
1. a - lr * grad
2. Find threshold theta such that sum(max(0, a_i - lr*grad_i - theta)) = 1
3. Projected point = max(0, a_i - lr*grad_i - theta)

This is O(n log n) per step.

The advantage: no LP needed, just FFT + subgradient + projection. Can do many iterations quickly.

The disadvantage: subgradient method is slow to converge (O(1/√T) rate).

import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()

def _conv(a):
    """FFT-based linear convolution (autocorrelation)."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fast(a):
    """Fast score using FFT."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):

    return 2.0 * n * m / (s * s)

    """Build convolution matrix: (A g)[k] = (f*g)[k]"""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_round(f, n, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Return normalized g or None."""
    A = _make_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                     bounds=[(0, None)] * n, method='highs',
                     options={'presolve': True, 'time_limit': 5})

        return None
    g = np.asarray(res.x, dtype=np.float64)

def _subgrad_phase(f_init, n, stop_time, seed):
    """Subgradient with momentum on max-conv."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f_init, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs if fs > 1e-10 else np.ones(n) / n
    
    best_f = f.copy()
    best_sc = _score_fast(f)
    lr = 0.005
    momentum = np.zeros(n)
    beta = 0.8
    no_imp = 0
    K = 7  # top-K lags
    
    while time.monotonic() < stop_time - 2.0:
        b = _conv(f)
        M = float(b.max())
        
        # Top-K lags
        if K >= 2 * n - 1:
            idx = np.arange(2 * n - 1)
        else:
            idx = np.argpartition(b, -K)[-K:]
        
        # Weight by how close to max
        weights = b[idx] - 0.9 * M + 1e-12
        weights = np.maximum(weights, 1e-12)
        weights /= weights.sum()
        
        # Subgradient of max_k (f*f)[k]
        grad = np.zeros(n)
        for w, k in zip(weights, idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * w * f[j]
        
        # Projected gradient step on simplex {a >= 0, sum(a) = 1}
        v = f - lr * grad / M  # normalize gradient
        # Project onto simplex: subtract theta, clip at 0
        v_sorted = np.sort(v)[::-1]
        css = np.cumsum(v_sorted)
        rho = np.nonzero(v_sorted * np.arange(1, n+1) > (css - 1.0))[0][-1]
        theta = (css[rho] - 1.0) / (rho + 1.0)
        f_new = np.maximum(v - theta, 0.0)
        f_new = f_new / f_new.sum()
        
        # Momentum
        momentum = beta * momentum + (f_new - f)
        f_mom = np.maximum(f + 0.5 * momentum, 0.0)
        f_mom = f_mom / f_mom.sum()
        
        # Accept if improvement or random
        sc = _score_fast(f_mom)
        if sc < best_sc - 1e-14:
            f = f_mom
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.02, lr * 1.005)
        else:
            no_imp += 1
            lr = max(0.001, lr * 0.98)
            if no_imp > 200:
                f = f_new.copy()  # take the non-momentum step
                no_imp = 0
                lr = 0.005
                beta = 0.7
                momentum[:] = 0
    
    return best_f, best_sc

def _coord_phase(f_init, n, stop_time, seed):

    f = np.clip(np.asarray(f_init, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())

    best_sc = _score_fast(f)
    no_full_pass_no_imp = 0
    
    multipliers = (0.3, 0.5, 0.7, 0.9, 1.0, 1.1, 1.3, 1.5, 2.0, 3.0)
    
    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                return f, best_sc
            orig = f[i]
            best_i = orig
            best_sc_i = best_sc
            for m in multipliers:
                val = orig * m
                if val < 0:
                    val = 0.0
                f[i] = val
                sc = _score_fast(f)
                if sc < best_sc_i - 1e-13:
                    best_sc_i = sc
                    best_i = val
            f[i] = best_i
            if best_i != orig:
                improved = True
            if best_sc_i < best_sc:
                best_sc = best_sc_i
        
        # 2-coordinate moves
        for _ in range(30):
            if time.monotonic() > stop_time - 2.0:
                return f, best_sc
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi, mj in [(0.5, 1.5), (1.5, 0.5), (0.7, 1.3), (1.3, 0.7), (0.3, 2.0)]:
                f[i] = max(0.0, oi * mi) if oi > 1e-12 else 0.01
                f[j] = max(0.0, oj * mj) if oj > 1e-12 else 0.01
                sc = _score_fast(f)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
            f[i], f[j] = oi, oj

F(a) = max_k (a*a)[k] where (a*a)[k] = sum_i a[i] * a[k-i] (valid indices)

dF/da_i for active lag k:
d(a*a)[k] / da_i = a[k-i] + (if i ≠ k-i) a[i-k]
                   = a[k-i] + a[i-k] (when k-i ≠ i and both valid)
                   = 2*a[i] (when k-i = i, i.e., i = k/2)

So the subgradient at a for the max function (with active set S) is:
grad_i = sum_{k in S, active} w_k * (a[k-i] + a[i-k])  [with appropriate validity checks]

where w_k are the weights (Lagrange multipliers for the active constraints, summing to 1).

grad[i] += w * (a[k-i] + a[i-k]) where both indices valid.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _make_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, n, M):
    A = _make_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 8})
    except Exception:
        return None
    if res is None or res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs


def _lp_phase(f_init, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f_init, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs if fs > 1e-10 else np.ones(n) / n

    best_f = f.copy()
    best_sc = _score(f)
    t_cur = 0.35
    stall = 0
    max_stall = 70
    it = 0

    while time.monotonic() < stop_time - 2.0:
        it += 1
        b = _conv(f)
        M = float(b.max())
        if M < 1e-15:
            break

        # Try LP with current M
        g = _lp_solve(f, n, M)
        # Also try tighter LP periodically
        if g is None and it % 3 == 0:
            g = _lp_solve(f, n, M * 0.97)
        if g is None:
            stall += 4
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            fs2 = f.sum()
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            continue

        dist = float(np.max(np.abs(g - f)))
        if dist < 1e-7:
            stall += 3
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
                fs2 = f.sum()
                f = f / fs2 if fs2 > 1e-10 else best_f.copy()
                t_cur = 0.35
                stall = 0
            continue

        # Line search (binary)
        t_lo, t_hi = 0.0, min(0.6, t_cur * 1.3)
        t_best = None
        f_try = None

        for _ in range(14):
            if time.monotonic() > stop_time - 2.0:
                break

            if fn.sum() < 1e-10:
                t_hi = t_mid
                continue
            sc = _score(fn)
            if sc < best_sc - 1e-14:

                t_best = t_mid
                f_try = fn.copy()
            else:

            stall = 0

            stall += 1

            f = f / fs2 if fs2 > 1e-10 else best_f.copy()

def _subgrad_phase(f_init, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f_init, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs if fs > 1e-10 else np.ones(n) / n

    best_f = f.copy()
    best_sc = _score(f)
    lr = 0.004
    beta = 0.8
    momentum = np.zeros(n)
    no_imp = 0
    K = 7

    i_range = np.arange(n)

    while time.monotonic() < stop_time - 2.0:
        b = _conv(f)
        M = float(b.max())
        if M < 1e-15:
            break

        if K >= 2 * n - 1:
            idx = np.arange(2 * n - 1)
        else:
            idx = np.argpartition(b, -K)[-K:]

        # Weights: emphasize lags near max
        w_raw = b[idx] - 0.85 * M + 1e-12
        w_raw = np.maximum(w_raw, 1e-12)
        weights = w_raw / w_raw.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, idx):
            k = int(k)
            # Term 1: a[k-i] valid when 0 <= k-i < n
            v1 = (k - i_range >= 0) & (k - i_range < n)
            if v1.any():
                grad[v1] += w * f[k - i_range[v1]]
            # Term 2: a[i-k] valid when 0 <= i-k < n and 2i != k
            v2 = (i_range - k >= 0) & (i_range - k < n) & (2 * i_range != k)
            if v2.any():
                grad[v2] += w * f[i_range[v2] - k]

        # Normalize gradient
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        grad /= gn

        # Step
        v = f - lr * grad
        # Project onto simplex {a >= 0, sum = 1}
        v_sort = np.sort(v)[::-1]
        css = np.cumsum(v_sort)
        rho_vals = np.nonzero(v_sort * np.arange(1, n + 1) > (css - 1.0))[0]
        if len(rho_vals) == 0:
            f_new = np.ones(n) / n
        else:
            rho = rho_vals[-1]
            theta = (css[rho] - 1.0) / (rho + 1.0)
            f_new = np.maximum(v - theta, 0.0)
            f_new = f_new / f_new.sum()

        # Momentum
        momentum = beta * momentum + (f_new - f)
        f_mom = np.maximum(f + 0.4 * momentum, 0.0)
        f_mom = f_mom / f_mom.sum()

        sc = _score(f_mom)
        if sc < best_sc - 1e-14:
            f = f_mom
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.015, lr * 1.003)
        else:
            no_imp += 1
            lr = max(0.0005, lr * 0.985)
            if no_imp > 150:
                f = f_new.copy()
                no_imp = 0
                lr = 0.004
                beta = 0.7
                momentum[:] = 0.0

    return best_f, best_sc


def _coord_phase(f_init, n, stop_time, seed):

    no_pass = 0

            best_local = best_sc
