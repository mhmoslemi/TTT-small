import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:

def _conv(a):
    """Full convolution via FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fast(a):
    """Compute 2*n*max_conv / sum^2 assuming sum=1."""
    n = len(a)
    c = _conv(a)
    M = float(c.max())
    return 2.0 * n * M

def _build_A(f):
    """Build the (2n-1) x n matrix for the LP."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_direction(f, M):
    """Solve the LP to find the best direction g."""
    n = len(f)
    if M < 1e-15:
        return None
    A = _build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True,
                               "dual_feasibility_tolerance": 1e-9,
                               "primal_feasibility_tolerance": 1e-9,
                               "time_limit": 30})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s  # normalize to sum=1

def _get_parent():

        h = globals()["height_sequence_1"]

    except Exception:

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max_conv"""
    n = len(a)
    M = float(_conv(a).max())
    return 2.0 * n * M


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, M, time_limit=20):
    n = len(f)
    if M < 1e-15:
        return None
    A = _build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True,
                               "dual_feasibility_tolerance": 1e-9,
                               "primal_feasibility_tolerance": 1e-9,
                               "time_limit": time_limit})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s

def _subgradient_refine(a, stop_time, seed=777):
    """Subgradient descent on the max-convolution objective."""
    n = len(a)
    f = np.clip(a.copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        return f, math.inf
    f = f / fs
    best_f = f.copy()
    best_sc = _score_norm(f)
    rng = np.random.default_rng(seed)

    lr = 0.03
    for it in range(500):
        if time.monotonic() > stop_time - 3.0:
            break
        conv = _conv(f)
        M = float(conv.max())
        thresh = 0.90 * M
        active = np.where(conv >= thresh)[0]

        subgrad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    subgrad[i] += 2.0 * f[j]
        subgrad /= max(1, len(active))

        # Normalize subgradient for stable step
        sg_norm = float(np.linalg.norm(subgrad))
        if sg_norm < 1e-12:
            break
        subgrad /= sg_norm

        f_new = f - lr * subgrad
        f_new = np.clip(f_new, 0, None)
        fs_new = f_new.sum()
        if fs_new < 1e-10:
            break
        f_new = f_new / fs_new
        sc_new = _score_norm(f_new)
        if sc_new < best_sc - 1e-12:
            best_f = f_new.copy()
            best_sc = sc_new
            f = f_new
            lr = min(0.08, lr * 1.02)
        else:
            lr = max(0.003, lr * 0.7)

    return best_f, best_sc
