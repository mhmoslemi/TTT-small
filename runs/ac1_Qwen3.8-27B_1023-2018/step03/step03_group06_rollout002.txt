def _central(n, tilt=0.0, kind='central', p=1.0):
 n=int(n)
 if n<=0: return np.ones(1)
 if kind=='central':
   x=np.empty(n); x[0]=1
   for k in range(1,n): x[k]=x[k-1]*(2*k-1)/(2*k)
 elif kind=='power':
   v=np.arange(1,n+1,dtype=float); x=v**(-float(p))
 elif kind=='exp':
   x=np.exp(-p*np.linspace(0,1,n))
 elif kind=='uniform': x=np.ones(n)

 if tilt and n>1:
   x *= np.exp(tilt*np.linspace(0,1,n-1))
 return _normalize(x)

import numpy as np

class _Timeout(Exception):
    pass


_fft_cache = {}


def _fft_len(m: int) -> int:
    m = int(max(1, m))
    v = _fft_cache.get(m)
    if v is None:
        v = int(next_fast_len(m))
        _fft_cache[m] = v
    return v


def _conv_same(a: np.ndarray) -> np.ndarray:
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = _fft_len(2 * n - 1)
    F = rfft(a, L)
    c = irfft(F * F, L)[: 2 * n - 1]
    return np.maximum(c, 0.0)


def _conv_ab(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    na = int(a.size)
    nb = int(b.size)
    if na == 0 or nb == 0:
        return np.array([], dtype=np.float64)
    L = _fft_len(na + nb - 1)
    c = irfft(rfft(a, L) * rfft(b, L), L)[: na + nb - 1]
    return np.maximum(c, 0.0)


def _valid_conv(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """h_i = sum_{r=0}^{n-1} w[i+r] a[r]."""
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = _fft_len(3 * n - 2)
    F = rfft(a, L)
    G = rfft(w[::-1], L)
    h = irfft(F * G, L)[: 3 * n - 2]
    return h[n - 1 : 2 * n - 1][::-1]


def _normalize(a, fallback_len: int = 32) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        n = int(fallback_len)
        if n <= 0:
            n = 32
        return np.ones(n, dtype=np.float64) / float(n)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s < 1e-12:
        return a / float(a.size) if a.size else np.ones(int(fallback_len), dtype=np.float64) / max(1, int(fallback_len))
    return a / s


def _ratio(a) -> float:
    """max(a*a) / (sum(a))^2.  Score is 2*n*ratio."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.where(np.isfinite(a), a, 0.0)

        return math.inf
    return m / (s * s)

def _central(n: int, tilt: float = 0.0, kind: str = "central", p: float = 1.0) -> np.ndarray:
    n = int(n)
    if n <= 0:
        return np.ones(1)
    if kind == "central":
        x = np.empty(n, dtype=np.float64)
        x[0] = 1.0
        for k in range(1, n):
            x[k] = x[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    elif kind == "power":
        v = np.arange(1, n + 1, dtype=np.float64)
        x = np.power(v, -float(p))
    elif kind == "exp":
        x = np.exp(-float(p) * np.linspace(0.0, 1.0, n))

    else:

def _interp(a, L: int) -> np.ndarray:

                grad_a = np.zeros(n, dtype=np.float64)
            dot = float(np.dot(aa, grad_a))
            gy = aa[idx] * (grad_a[idx] - dot)
            gy = np.where(np.isfinite(gy), gy, 0.0)
            return float(fun), gy

        try:
            res = minimize(
                fg,
                y0,
                jac=True,
                method="L-BFGS-B",
                options={
                    "maxiter": 220,
                    "maxfun": int(maxfun_per),
                    "ftol": 1e-12,
                    "gtol": 1e-9,
                    "maxls": 50,
                },
            )
        except _Timeout:
            break
        except Exception:
            break

        if not hasattr(res, "x") or res.x is None:
            break
        a_new = _recon_gauge(np.asarray(res.x, dtype=np.float64), gauge)
        if a_new is None:
            break
        r_new = _ratio(a_new)
        if math.isfinite(r_new) and r_new < best_r - 1e-15:
            best = a_new.copy()
            best_r = r_new
            a = best
        elif math.isfinite(r_new) and r_new <= best_r + 1e-10:
            a = a_new.copy()

    return best, 2.0 * n * best_r


def _lp_solve(f: np.ndarray):
    n = int(f.size)
    if n < 2:
        return None
    A = np.zeros((2 * n - 1, n + 1), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f
    A[:, n] = -1.0

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    b_ub = np.zeros(2 * n - 1, dtype=np.float64)
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
    except Exception:
        return None
    if res is None or getattr(res, "status", 1) != 0 or getattr(res, "x", None) is None:
        return None
    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.clip(g, 0.0, 1.0)
    s = float(g.sum())
    if s < 1e-12:
        return None
    return g / s


def _best_mix_t(cf: np.ndarray, cfs: np.ndarray, cg: np.ndarray) -> float:
    base = cf
    b = 2.0 * (cfs - cf)
    co = cf - 2.0 * cfs + cg
    ts = np.linspace(0.0, 1.0, 1024)
    vals = np.max(base[None, :] + b[None, :] * ts[:, None] + co[None, :] * (ts[:, None] ** 2), axis=1)
    return float(ts[int(np.argmin(vals))])


def _lp_update(a0: np.ndarray, stop: float, max_iters: int = 10, seed: int = 0):
    f = _normalize(a0)
    n = int(f.size)
    best = f.copy()
    best_r = _ratio(best)
    rng = np.random.default_rng(seed)
    noimp = 0
    max_iters = max(1, int(max_iters))

    for it in range(max_iters):
        if time.monotonic() > stop - 25:
            break
        g = _lp_solve(best)
        if g is None:
            break
        g = _normalize(g)

        cf = _conv_same(best)
        cfs = _conv_ab(best, g)
        cg = _conv_same(g)
        t0 = _best_mix_t(cf, cfs, cg)

        cands = {
            0.0, 1.0, float(t0),
            0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.75,
            float(max(0.0, t0 - 0.05)), float(min(1.0, t0 + 0.05)),
            float(max(0.0, 0.5 * t0)), float(min(1.0, 1.5 * t0)),
        }
        base = cf
        b = 2.0 * (cfs - cf)
        co = cf - 2.0 * cfs + cg
        new_best = best
        new_r = best_r
        for tt in sorted(cands):
            tt = float(np.clip(tt, 0.0, 1.0))
            r = float(np.max(base + b * tt + co * (tt * tt)))
            if math.isfinite(r) and r < new_r:
                new_r = r
                new_best = (1.0 - tt) * best + tt * g

        if math.isfinite(new_r) and new_r < best_r - 1e-15:
            cand = _normalize(new_best)
            r2 = _ratio(cand)
            if math.isfinite(r2) and r2 < best_r - 1e-15:
                best = cand
                best_r = r2
                noimp = 0
                continue

        noimp += 1
        if noimp >= 3:
            break
        # Gentle perturb to escape a non-improving linearized direction.
        dp = 1.0 / float(n)
        pert = best + rng.normal(0.0, max(1e-6, 0.05 * dp), size=n)
        pert = np.clip(pert, 0.0, None)
        s = float(pert.sum())
        if s > 1e-12:
            pert = pert / s
            rp = _ratio(pert)
            if math.isfinite(rp) and rp < best_r - 1e-15:
                best = pert
                best_r = rp
                noimp = 0

    return best, 2.0 * n * best_r


def _coord_score(x: float, c: np.ndarray, lin: np.ndarray, qidx: int, old: float) -> float:
    x = float(x)
    if not math.isfinite(x) or x < 0.0:
        return 1e100
    d = x - old
    S = 1.0 + d
    if S <= 1e-12:
        return 1e100
    arr = c + lin * d
    if 0 <= qidx < arr.size:
        arr[qidx] += d * d
    m = float(arr.max())
    if not math.isfinite(m):
        return 1e100
    return m / (S * S)


def _coord_polish(a0: np.ndarray, stop: float, seed: int = 1, refine: bool = True):
    a = _normalize(a0)
    n = int(a.size)
    if n < 2 or time.monotonic() > stop - 5:
        return a, 2.0 * n * _ratio(a)
    rng = np.random.default_rng(seed)

    def _pass(use_refine: bool):
        nonlocal a
        if time.monotonic() > stop - 4:
            return False
        c = _conv_same(a)
        score = float(c.max())
        if not math.isfinite(score):
            return False

        try:
            active = c >= max(score - 1e-10 * max(score, 1e-300), 0.0)
            w = active.astype(np.float64)
            ws = float(w.sum())
            if ws > 0:
                w = w / ws
            else:
                w = np.full(c.size, 1.0 / float(c.size), dtype=np.float64)
            grad = 2.0 * _valid_conv(w, a)
            if not np.all(np.isfinite(grad)):
                grad = np.zeros(n, dtype=np.float64)
            order = np.argsort(-np.abs(grad))
        except Exception:
            order = np.arange(n, dtype=int)

        improved = False
        for i in order:
            if time.monotonic() > stop - 3:
                break
            i = int(i)
            old = float(a[i])
            mean = 1.0 / float(n)

            cands = [0.0, old]
            for m in (0.0, 0.1, 0.2, 0.4, 0.6, 0.75, 0.85, 0.9, 0.95, 1.05, 1.1, 1.25, 1.5, 2.0, 3.0, 5.0):
                v = old * float(m)
                if math.isfinite(v):
                    cands.append(v)
            for m in (0.0, 0.05, 0.1, 0.2, 0.35, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0, 5.0):
                v = mean * float(m)
                if math.isfinite(v):
                    cands.append(v)
            for v in (1e-8 * mean, 1e-6 * mean, 1e-4 * mean, 1e-2 * mean):
                if math.isfinite(v):
                    cands.append(v)

            tmp = [float(v) for v in cands if math.isfinite(v) and 0.0 <= v <= 1000.0]
            if not tmp:
                tmp = [0.0]
            tmp.sort()
            xs_list = []
            prev = None
            for v in tmp:
                if prev is None or v > prev + 1e-13 * max(1.0, abs(prev)):
                    xs_list.append(v)
                    prev = v
            xs = np.asarray(xs_list, dtype=np.float64)
            if xs.size == 0:
                continue

            lin = np.zeros(2 * n - 1, dtype=np.float64)
            lin[i : i + n] = 2.0 * a
            qidx = 2 * i
            if qidx >= lin.size:
                qidx = -1

            scores = np.empty(xs.size, dtype=np.float64)
            for st in range(0, xs.size, 16):
                chunk = xs[st : st + 16]
                D = chunk - old
                Sx = 1.0 + D
                newc = c[None, :] + lin[None, :] * D[:, None]
                if qidx >= 0:
                    newc[:, qidx] += D * D
                mrow = np.max(newc, axis=1)
                with np.errstate(divide="ignore", invalid="ignore"):
                    sc = np.where(Sx > 1e-12, mrow / (Sx * Sx), np.inf)
                scores[st : st + 16] = sc

            pos = int(np.argmin(scores))
            r = float(scores[pos])
            x = float(xs[pos])
            if not math.isfinite(r):
                continue

            if use_refine and r < score - 1e-15:
                if xs.size > 1 and 0 < pos < xs.size - 1:
                    lo = float(xs[pos - 1])
                    hi = float(xs[pos + 1])
                elif xs.size > 1 and pos == 0:
                    lo = 0.0
                    hi = float(xs[1])
                elif xs.size > 1:
                    lo = float(xs[-2])
                    hi = min(1000.0, max(float(xs[-1]) * 1.25, float(xs[-1]) + mean))
                else:
                    span = max(1e-6, 0.2 * mean)
                    lo = max(0.0, x - span)
                    hi = min(1000.0, x + span if x > 0 else 2.0 * span)
                if hi > lo and (hi - lo) > 1e-14:
                    try:
                        res = minimize_scalar(
                            lambda z: _coord_score(float(z), c, lin, qidx, old),
                            bounds=(float(lo), float(hi)),
                            method="bounded",
                            options={"maxiter": 30, "xatol": 1e-10},
                        )
                        xr = float(res.x)
                        rr = float(res.fun)
                        if math.isfinite(rr) and rr < r - 1e-18:
                            # Re-evaluate to avoid any trust-region artifact.
                            rr = _coord_score(xr, c, lin, qidx, old)
                            if math.isfinite(rr) and rr < r:
                                r = rr
                                x = xr
                    except Exception:
                        pass

            if r < score - 1e-14:
                delta = x - old
                Snew = 1.0 + delta
                if Snew > 1e-12 and math.isfinite(Snew):
                    c_try = c + lin * delta
                    if qidx >= 0:
                        c_try[qidx] += delta * delta
                    a_try = a.copy()
                    a_try[i] = x
                    f = 1.0 / Snew
                    a_try *= f
                    c_try *= f * f
                    if np.all(np.isfinite(a_try)) and np.all(np.isfinite(c_try)):
                        c_try = np.maximum(c_try, 0.0)
                        new_score = float(c_try.max())
                        if math.isfinite(new_score) and new_score < score - 1e-15:
                            a = a_try
                            c = c_try
                            score = new_score
                            improved = True
        return improved

    last_improved = False
    for _ in range(3):
        if time.monotonic() > stop - 10:
            break
        last_improved = _pass(False)
        if not last_improved:
            break

    if refine and time.monotonic() < stop - 15 and (last_improved or _ratio(a) < 1e6):
        _pass(True)

    return a, 2.0 * n * _ratio(a)


def _optimise_one(a0: np.ndarray, stop: float, allow_lp: bool, seed: int):
    a = _normalize(a0)
    n = int(a.size)
    now = time.monotonic()
    if time.monotonic() > stop - 10:
        return a
    total = max(1.0, stop - now)

    if n <= 256:
        mf = 450
        lp_it = 10
    elif n <= 384:
        mf = 300
        lp_it = 6
    elif n <= 512:
        mf = 180
        lp_it = 4
    else:
        mf = 100
        lp_it = 0

    # Low-to-medium log-sum-exp smoothing.
    s1 = min(stop - max(30.0, 0.35 * total), now + max(15.0, 0.35 * total))
    a1 = a
    if s1 - now > 12.0:
        a1, _ = _lbfgs_run(a, (16.0, 64.0, 256.0), s1, maxfun_per=mf)

    a2 = a1
    if allow_lp and np.all(np.isfinite(a1)) and time.monotonic() < stop - max(60.0, 0.30 * total):
        lp_stop = min(stop - max(45.0, 0.25 * total), time.monotonic() + max(25.0, 0.22 * total))
        if lp_stop - time.monotonic() > 20.0 and lp_it > 0:
            a2, _ = _lp_update(a1, lp_stop, max_iters=lp_it, seed=int(seed))

    a3 = a2
    if time.monotonic() < stop - max(40.0, 0.20 * total):
        hb_stop = min(stop - max(25.0, 0.15 * total), time.monotonic() + max(15.0, 0.15 * total))
        if hb_stop - time.monotonic() > 12.0:
            a3, _ = _lbfgs_run(a2, (256.0, 1024.0, 2048.0), hb_stop, maxfun_per=max(60, mf // 2))

    final_stop = stop - 4.0
    if time.monotonic() < final_stop and np.all(np.isfinite(a3)):
        a4, _ = _coord_polish(a3, final_stop, seed=int(seed) + 77, refine=True)
        return a4
    return a3


def _make_initializations(L: int, parent, best_arr, rng) -> list:
    inits = []
    if parent is not None:
        inits.append(_interp(parent, L))
    if best_arr is not None and best_arr.size:
        inits.append(_interp(best_arr, L))

    inits.append(_central(L, 0.0))
    for t in (1.0, 2.0, 3.0, -1.0):
        inits.append(_central(L, t))
    for p in (0.35, 0.45, 0.5, 0.65, 0.8):
        inits.append(_central(L, kind="power", p=p))
    inits.append(_central(L, kind="uniform"))
    inits.append(_central(L, kind="exp", p=2.0))
    inits.append(_central(L, kind="exp", p=5.0))
    inits.append(_normalize(rng.exponential(1.0, size=L)))
    inits.append(_normalize(np.abs(rng.normal(0.0, 1.0, size=L)) + 0.05))

    out = []
    seen = []
    for a in inits:
        try:
            b = _normalize(a)
            key = tuple(np.round(b, 10).tolist())
            if key in seen:
                continue
            seen.append(key)
            out.append(b)
        except Exception:
            continue
    return out


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not math.isfinite(budget) or budget < 10.0:
        budget = 1300.0
    hard = t0 + budget
    if budget > 180.0:
        deadline = hard - 32.0
    elif budget > 60.0:
        deadline = hard - 12.0
    else:
        deadline = hard - 3.0

    parent = _get_parent()
    if parent is None:
        parent = _central(256)
    else:
        parent = _normalize(parent)

    best_arr = _normalize(parent)
    fallback = best_arr.copy()
    best_score = math.inf

    def _try_candidate(arr):
        nonlocal best_arr, best_score, fallback
        if arr is None:
            return
        a = np.asarray(arr, dtype=np.float64).ravel()
        if a.size == 0:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if not math.isfinite(s) or s < 1e-12:
            return
        a = a / s
        sc_int = _score_internal(a)
        if not math.isfinite(sc_int):
            return
        # Call the exact evaluator when internal score is promising, or as a rescue
        # if exact evaluator was unavailable earlier.
        if best_score == math.inf or sc_int < best_score - 1e-10:
            lst = [float(x) for x in a]
            sc = _exact_score(lst)
            if not math.isfinite(sc):
                sc = sc_int
            if math.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_arr = a.copy()
                fallback = a.copy()

    _try_candidate(best_arr)
    if not math.isfinite(best_score):
        _try_candidate(_central(int(best_arr.size)))

    def _finalize():
        seq = np.clip(np.asarray(best_arr, dtype=np.float64).ravel(), 0.0, 1000.0)
        if seq.size == 0 or float(seq.sum()) < 1e-8:
            seq = np.clip(np.asarray(fallback, dtype=np.float64).ravel(), 0.0, 1000.0)
        if seq.size == 0 or float(seq.sum()) < 1e-8:
            seq = np.clip(parent if parent is not None else _central(64), 0.0, 1000.0)
        if seq.size == 0 or float(seq.sum()) < 1e-8:
            seq = np.ones(64, dtype=np.float64) / 64.0
        lst = [float(x) for x in seq]
        sc = _exact_score(lst)
        if not math.isfinite(sc):
            fb = np.clip(np.asarray(fallback, dtype=np.float64).ravel(), 0.0, 1000.0)
            if fb.size == 0 or float(fb.sum()) < 1e-8:
                fb = np.ones(64, dtype=np.float64) / 64.0
            fb_lst = [float(x) for x in fb]
            sc2 = _exact_score(fb_lst)
            if math.isfinite(sc2):
                return fb_lst
        return lst

    rng = np.random.default_rng(seed)

    def _process_length(L: int, stop: float, allow_lp: bool, seed_val: int):
        L = int(L)
        now = time.monotronic() if hasattr(time, "monotonic") else time.time()
        now = time.monotonic()
        if L < 24 or stop - now < 35:
            return

        inits = _make_initializations(L, parent, best_arr, rng)
        scored = []
        for a in inits:
            if time.monotonic() > stop - 10:
                break
            try:
                r = _ratio(a)
            except Exception:
                r = math.inf
            if math.isfinite(r):
                scored.append((r, a))
        if not scored:
            return
        scored.sort(key=lambda z: z[0])

        want_second = (stop - now > 340 and len(scored) >= 2 and (2.0 * L * scored[1][0] < best_score + 0.45 or best_score == math.inf))
        reserve_second = 90.0 if want_second else 0.0
        first_stop = stop - reserve_second

        a_opt = _optimise_one(scored[0][1], first_stop, allow_lp=allow_lp, seed=seed_val)
        _try_candidate(a_opt)

        if want_second and time.monotonic() < stop - 25:
            a_opt2 = _optimise_one(scored[1][1], stop - 5.0, allow_lp=False, seed=seed_val + 123)
            _try_candidate(a_opt2)

    # Small-budget path: quick polish only.
    if budget < 80.0:
        if time.monotonic() < deadline - 2.0:
            try:
                a, _ = _coord_polish(best_arr, min(deadline - 2.0, t0 + max(5.0, budget - 12.0)), seed=9, refine=True)
                _try_candidate(a)
            except Exception:
                pass
        return _finalize()

    # Quick direct coordinate polish of the supplied parent first.
    quick_stop = min(deadline - 70.0, t0 + 55.0)
    if quick_stop - time.monotonic() > 12.0:
        try:
            a, _ = _coord_polish(best_arr, quick_stop, seed=11, refine=True)
            _try_candidate(a)
        except Exception:
            pass

    P = int(best_arr.size)
    preferred = [P]
    cand_l = set()
    for L in (int(P * 4 // 3), int(P * 3 // 2), int(P * 2), int(P * 5 // 4), int(P * 3 // 4), 384, 512, 768):
        if 96 <= int(L) <= 1024:
            cand_l.add(int(L))
    others = sorted([L for L in cand_l if L != P], key=lambda x: abs(x - P))
    preferred.extend(others[:4])

    # De-duplicate while preserving preferred order.
    order = []
    for L in preferred:
        if L not in order:
            order.append(L)

    base_alloc = max(60.0, deadline - time.monotonic())
    shares = [0.42, 0.26, 0.18, 0.12, 0.07]

    for idx, L in enumerate(order):
        now = time.monotonic()
        rem = deadline - now
        if rem < 70.0:
            break
        share = shares[idx] if idx < len(shares) else 0.05
        alloc = base_alloc * share
        if L == P:
            alloc = max(alloc, min(rem - 15.0, 650.0))
        else:
            alloc = max(55.0, min(alloc, 360.0))
        alloc = min(alloc, rem - 12.0)
        if alloc < 30.0:
            break
        stop = time.monotonic() + alloc
        allow_lp = bool(L <= (512 if budget > 500.0 else 384))
        try:
            _process_length(L, stop, allow_lp, seed=int(rng.integers(1, 1_000_000_000)))
        except Exception:
            pass

        if math.isfinite(best_score) and best_score < 1.495:
            break

    # Final safeguard: if we are near the absolute deadline, stop all work.
    if time.monotonic() >= hard - 14.0:
        return _finalize()

    # Very last polish on the globally best sequence.
    if time.monotonic() < deadline - 12.0:
        n_best = int(best_arr.size)
        if n_best <= 1100:
            try:
                a, _ = _coord_polish(best_arr, deadline - 8.0, seed=987654, refine=True)
                _try_candidate(a)
            except Exception:
                pass

    return _finalize()
