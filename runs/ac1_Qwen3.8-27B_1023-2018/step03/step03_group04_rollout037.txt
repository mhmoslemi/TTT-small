This gives a gradient direction, and we can do projected gradient descent on the objective max(f*f) subject to sum(f) = 1, f ≥ 0.

    return A

def _lp_direction(f, M):

    n = len(f)

    b_ub = np.full(2 * n - 1, M)

                      bounds=[(0, None)] * n,

        return None
    g = np.maximum(res.x, 0.0)
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s

    n = len(f0)

        if dist < 1e-9:

            if stall > max_stall:

            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            fs = f.sum()
            f = f / fs if fs > 1e-10 else best_f.copy()
            continue
        
        # Line search
        best_t = None
        best_sc = best_score
        best_f_new = None

            f_new = np.clip(f_new, 0, None)

            sc = _score_normalized(f_new)

                best_t = tc
                best_f_new = f_new.copy()
        
        if best_f_new is not None:

            if best_sc < best_score - 1e-14:

                best_score = best_sc
                stall = 0

                f = f / fs if fs > 1e-10 else best_f.copy()
                t = 0.2
                stall = 0
        
        iteration += 1
    
    return best_f, best_score
