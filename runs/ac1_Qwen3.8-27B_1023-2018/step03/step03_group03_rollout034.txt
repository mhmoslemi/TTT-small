import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c


def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    C = _conv(a)
    m = float(C.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (S * S)


def _lp_direction(f, M, time_limit=8.0):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": time_limit})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-15:
        return None
    return g / s


def _coord_descent_vec(f, stop_time, max_sweeps=500, seed=42, n_grid=400):

        return f, _score(f)
    C = _conv(f)
    score = 2.0 * n * C.max() / (S * S)
    rng = np.random.default_rng(seed)
    two_n_minus_1 = 2 * n - 1

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() > stop_time:
                break

            S_rest = S - v

            m_start = j
            m_end = min(j + n, two_n_minus_1)
            n_aff = m_end - m_start
            infl = f[:n_aff]

            if m_start > 0:
                c_max_rest = max(c_max_rest, float(C[:m_start].max()))
            if m_end < two_n_minus_1:
                c_max_rest = max(c_max_rest, float(C[m_end:].max()))

def _pair_polish(f, stop_time, max_rounds=30, seed=9999):
    n = len(f)
    f = f.copy()

    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time:

        improved = False
        n_trials = min(120, max(20, int((stop_time - time.monotonic()) * 2)))
        for _ in range(n_trials):

            j = int(rng.integers(n))
            if i == j:

            oi, oj = f[i], f[j]

        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    if final_sc < best_score:
        best_score = final_sc

    return [float(x) for x in best_seq]
