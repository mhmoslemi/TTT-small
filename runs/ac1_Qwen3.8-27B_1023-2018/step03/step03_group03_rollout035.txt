import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    if n <= 512:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n, active_mask=None):
    """Build constraint matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    if active_mask is not None:
        rows = np.where(active_mask)[0]
    else:
        rows = np.arange(2 * n - 1)
    A = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A[idx, j_lo:j_hi + 1] = f[j_lo - k:(j_hi + 1) - k]
    return A, rows


def _lp_direction(f, n, time_limit=4.0, active_frac=1.0):
    """Solve LP to find direction g maximizing sum(g) s.t. f*g <= M, g >= 0."""
    b = _conv_full(f)
    M = float(b.max())
    if M < 1e-15:
        return None, M
    if active_frac < 1.0:
        threshold = M * active_frac
        active_mask = b >= threshold
        if active_mask.sum() < 5:
            active_mask = b >= 0.5 * M
    else:
        active_mask = None
    A, rows = _build_A(f, n, active_mask)
    c = -np.ones(n)
    b_ub = M * np.ones(len(rows))
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": time_limit,
                               "dual_feasibility_tolerance": 1e-9})
    except Exception:
        return None, M
    if res.status not in (0, 3) or res.x is None:
        return None, M
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0, None)
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None, M
    return g / g_sum, M


def _coord_descent(f, stop_time, max_sweeps=80, seed=777, verbose=False):
    """Coordinate descent with efficient convolution update."""
    n = len(f)
    f = np.clip(f.astype(np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, _score(f)
    conv = _conv_full(f)
    best_score = 2.0 * n * conv.max() / (S * S)
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 3.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            v = f[i]
            S_rest = S - v
            # Influence: positions [i, i+n-1] get 2*f[k-i] added for change delta
            # At position 2*i, extra delta^2
            start = i
            end = min(i + n, 2 * n - 1)
            infl = 2.0 * f[:end - start]

            # Quick check: if v is 0 and influence is all positive, can't decrease max
            # Grid search for best delta
            x_max = max(4.0 * v, 0.5 / n) if v > 1e-10 else 2.0 / n
            x_max = min(x_max, 5.0 / n)

            n_pts = 80
            x_grid = np.linspace(0.0, x_max, n_pts)

            best_x = v
            best_obj = best_score

            # Vectorized evaluation
            # new_conv[k] = conv[k] + (x-v)*infl[k-start] for k in [start, end)
            # plus (x-v)^2 at k=2*i
            # We need max over all k
            # Precompute: max outside affected range
            if start > 0 and end < 2 * n - 1:
                ext_max = max(float(conv[:start].max()) if start > 0 else 0.0,
                              float(conv[end:].max()) if end < 2 * n - 1 else 0.0)
            elif start > 0:
                ext_max = float(conv[:start].max()) if start > 0 else 0.0
            else:
                ext_max = 0.0

            for x in x_grid:
                S_new = S_rest + x
                if S_new < 0.01:
                    continue
                w = x - v
                if abs(w) < 1e-14:
                    m_new = float(conv.max())
                else:
                    aff = conv[start:end] + w * infl
                    if 0 <= 2 * i <= 2 * n - 2:
                        aff[2 * i - start] += w * w
                    m_new = max(ext_max, float(aff.max()))
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_x = x

            if best_x != v and best_obj < best_score - 1e-14:
                w = best_x - v
                f[i] = best_x
                S += w
                conv[start:end] += w * infl
                if 0 <= 2 * i <= 2 * n - 2:
                    conv[2 * i] += w * w
                best_score = best_obj
                improved = True

        if not improved:
            # Try a few random perturbations to escape
            for _ in range(10):
                idx = rng.integers(0, n)
                f[idx] = max(0.0, f[idx] + rng.normal(0, 0.02 / n))
            f_norm = f.sum()
            if f_norm > 0.01:
                f = f / f_norm
                conv = _conv_full(f)
                best_score = 2.0 * n * conv.max() / (f_norm * f_norm)
            else:
                break
        if sweep > 0 and sweep % 20 == 0 and not improved:
            break

    f = np.clip(f, 0, 1000)
    final_score = _score(f)
    return f, final_score


def _pair_polish(f, stop_time, max_rounds=30, seed=9999):
    """2-coordinate moves to escape 1D local optima."""
    n = len(f)
    f = f.copy()
    best_score = _score(f)
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 3.0:
            break
        improved = False
        n_trials = 60
        for _ in range(n_trials):
            if time.monotonic() > stop_time - 3.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            mults = (0.2, 0.5, 1.0, 2.0, 4.0, 0.0)
            best_vals = (oi, oj)
            best_sc = best_score
            for mi in mults:
                vi = oi * mi if oi > 1e-12 else (0.01 / n if mi > 0 else 0.0)
                for mj in mults:
                    vj = oj * mj if oj > 1e-12 else (0.01 / n if mj > 0 else 0.0)
                    f[i], f[j] = vi, vj
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        best_vals = (vi, vj)
            f[i], f[j] = best_vals
            if best_sc < best_score - 1e-14:
                best_score = best_sc
                improved = True
        if not improved:
            break
    return f, best_score


def _lp_optimize(f0, n, stop_time, seed, max_iters=300):
    """LP-based optimization with adaptive step size."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f = f[:n]
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum

    best_f = f.copy()
    best_score = _score(f)
    iter_count = 0
    stall = 0
    max_stall = 80
    t = 0.15

    while time.monotonic() < stop_time - 5.0 and iter_count < max_iters:
        # Check for convergence
        g_dir, M = _lp_direction(f, n, time_limit=3.0, active_frac=1.0)
        iter_count += 1

        if g_dir is None:
            stall += 3
            if stall > max_stall:
                break
            # Perturb and continue
            f = np.clip(f + rng.normal(0, 0.03 / n, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            continue

        # Check if direction is meaningfully different
        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-7:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.05 / n, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            continue

        # Line search (geometric decrease)
        improved = False
        t_cur = t
        for _ in range(10):
            f_new = (1.0 - t_cur) * f + t_cur * g_dir
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 1e-10:
                t_cur *= 0.5
                continue
            sc = _score(f_new)
            if sc < best_score - 1e-13:
                f = f_new
                best_f = f.copy()
                best_score = sc
                t = min(0.5, t * 1.08)
                stall = 0
                improved = True
                break
            else:
                t_cur *= 0.5
                if t_cur < 0.005:
                    break

        if not improved:
            stall += 1
            t = max(0.05, t * 0.8)
            if stall >= max_stall:
                # Reset with perturbation
                f = np.clip(f + rng.normal(0, 0.1 / n, n), 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs
                else:
                    f = best_f.copy()
                stall = 0
                t = 0.15
                # Try with reduced active set
                for _ in range(3):
                    if time.monotonic() > stop_time - 5.0:
                        break
                    g_dir2, M2 = _lp_direction(f, n, time_limit=3.0, active_frac=0.8)
                    if g_dir2 is not None:
                        f_new = 0.7 * f + 0.3 * g_dir2
                        f_new = np.clip(f_new, 0, None)
                        fs_new = f_new.sum()
                        if fs_new > 1e-10:
                            f_new = f_new / fs_new
                            sc2 = _score(f_new)
                            if sc2 < best_score:
                                f = f_new
                                best_f = f.copy()
                                best_score = sc2

    return best_f, best_score
