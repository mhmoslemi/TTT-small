            # = conv[k_start:k_end][None,:] + 2*(x_grid-v)[:,None]*influence[None,:]
            # But this is n_pts * n which might be large for n=384, n_pts=60: 23K, fine.
            
            affected = conv[k_start:k_end][np.newaxis, :] + 2.0 * (x_grid - v)[:, np.newaxis] * influence[np.newaxis, :]
            
            # Fix the 2i point
            if k_2i >= k_start and k_2i < k_end:
                idx = k_2i - k_start
                # The formula 2*(x-v)*f[i] gives the wrong value at k=2i
                # Correct: (x^2 - v^2) = (x-v)(x+v) = (x-v)^2 + 2*v*(x-v)
                # What we computed: 2*(x-v)*f[i] = 2*(x-v)*v (since f[i] = v)
                # Difference: (x-v)^2 + 2*v*(x-v) - 2*v*(x-v) = (x-v)^2
                affected[:, idx] += (x_grid - v) ** 2
            
            aff_max = affected.max(axis=1)  # shape (n_pts,)
            total_max = np.maximum(aff_max, unaff_max)
            
            S_new = S_rest + x_grid
            valid = S_new > 0.01
            if not valid.any():
                continue

            scores[valid] = 2.0 * n * total_max[valid] / (S_new[valid] ** 2)
            
            best_idx = scores.argmin()
            best_x = x_grid[best_idx]
            best_sc = scores[best_idx]
            
            if best_sc < best_score - 1e-14 and best_x != v:
                # Refine around best_x
                step = (x_hi - x_lo) / n_pts
                x_fine = np.linspace(max(0, best_x - 2*step), best_x + 2*step, 30)
                x_fine = np.clip(x_fine, 0, x_hi)
                
                affected_f = conv[k_start:k_end][np.newaxis, :] + 2.0 * (x_fine - v)[:, np.newaxis] * influence[np.newaxis, :]
                if k_2i >= k_start and k_2i < k_end:
                    idx = k_2i - k_start
                    affected_f[:, idx] += (x_fine - v) ** 2
                
                aff_max_f = affected_f.max(axis=1)
                total_max_f = np.maximum(aff_max_f, unaff_max)
                S_new_f = S_rest + x_fine
                valid_f = S_new_f > 0.01
                scores_f = np.full(len(x_fine), math.inf)
                scores_f[valid_f] = 2.0 * n * total_max_f[valid_f] / (S_new_f[valid_f] ** 2)
                
                best_f_idx = scores_f.argmin()
                best_x = x_fine[best_f_idx]
                best_sc = scores_f[best_f_idx]
                
                if best_sc < best_score - 1e-14:
                    # Apply update
                    w = best_x - v
                    f[i] = best_x
                    S += w
                    conv[k_start:k_end] += 2.0 * w * influence
                    if k_2i >= k_start and k_2i < k_end:
                        conv[k_2i] += w * w  # (x-v)^2 = w^2
                    best_score = best_sc
                    improved = True
        
        if not improved:
            break
    
    f = np.clip(f, 0, 1000)
    final_score = _score(f)
    return f, final_score

Wait, there's an issue with the update to conv. When I update conv for coordinate i, I add 2*w*influence to the affected range. But influence = 2*f[0:n'] already has the factor of 2. Let me recheck.

conv[k] += 2*(x-v)*f[k-i] for k in affected range, k≠2i
conv[2i] += (x^2-v^2) = (x-v)^2 + 2*v*(x-v)

So the update is:
conv[k_start:k_end] += 2*w * f[0:len]  (where len = k_end - k_start)
conv[2i] += w*w + 2*w*v - 2*w*f[i]  = w*w (since f[i] = v)

Wait: 2*(x-v)*f[k-i] at k=2i gives 2*w*v. And (x^2-v^2) = 2*w*v + w^2. So the correction at 2i is w^2.

So:
conv[k_start:k_end] += 2*w * f[0:len]
conv[2i] += w*w

This is what I had. Good.

But wait, I need to be careful: f[0:len] where len = k_end - k_start. And k_start = i, so k_end - k_start = min(n, 2n-1-i). And f[k-i] for k in [k_start, k_end) is f[0] to f[k_end-1-k_start] = f[0] to f[min(n, 2n-1-i)-1].

So influence = 2.0 * f[:min(n, 2*n-1-i)].

Let me simplify: the affected range is k in [i, min(i+n, 2n-1)]. The length is min(n, 2n-1-i). And f[k-i] for k in this range is f[0] to f[min(n, 2n-1-i)-1].

So:
k_start = i
k_end = min(i + n, 2 * n - 1)
len_aff = k_end - k_start
influence = 2.0 * f[:len_aff]

And 2i is in [k_start, k_end) iff i <= 2i < i+n, i.e., 0 <= i < n, which is always true. Also 2i <= 2n-2, so k_2i = min(2*i, 2*n-2).

This looks correct.

The `.max(axis=1)` is fast. This should be much faster than the previous approach of copying the full array.

The constraint matrix A is (2n-1) x n, where A[k,j] = f[k-j] if 0 <= k-j < n, else 0. This is a banded matrix (each column has n nonzeros, each row has up to n nonzeros).

For n=384, the LP has 767 constraints and 384 variables. HiGHS should handle this in about 0.1-1 seconds.

import numpy as np

import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    return p

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(f):
    n = len(f)
    s = float(f.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(f)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """Build the convolution constraint matrix A: A[k,j] = f[k-j] if valid."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_direction(f, conv, n, time_limit=5.0):
    """Find LP direction: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    M = float(conv.max())
    if M < 1e-15:
        return None
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs

def _lp_optimize(f0, n, stop_time, seed, max_iters=200):
    """Iterative LP-based optimization."""
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs  # normalize to sum=1
    
    best_f = f.copy()
    conv = _conv_full(f)
    best_score = 2.0 * n * float(conv.max())  # since sum=1
    
    t = 0.15
    stall = 0
    iteration = 0
    
    while time.monotonic() < stop_time - 3.0 and iteration < max_iters:
        g_dir = _lp_direction(f, conv, n, time_limit=4.0)
        
        if g_dir is None:
            stall += 10
            if stall > 50:
                break
            # Perturb
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            f = f / f.sum()
            conv = _conv_full(f)
            iteration += 1
            continue
        
        dist = float(np.linalg.norm(g_dir - f, ord=2))
        if dist < 1e-6:
            stall += 5
            if stall > 50:
                break
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            f = f / f.sum()
            conv = _conv_full(f)
            iteration += 1
            continue
        
        # Line search
        improved = False
        for trial in range(10):
            f_new = (1.0 - t) * f + t * g_dir
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 1e-10:
                t *= 0.5
                continue
            f_new = f_new / fs_new  # renormalize
            conv_new = _conv_full(f_new)
            sc = 2.0 * n * float(conv_new.max())
            if sc < best_score - 1e-12:
                f = f_new
                conv = conv_new
                best_f = f.copy()
                best_score = sc
                t = min(0.5, t * 1.1)
                stall = 0
                improved = True
                break
            else:
                t = max(0.01, t * 0.5)
        
        if not improved:
            stall += 1
            if stall >= 40:
                f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
                f = f / f.sum()
                conv = _conv_full(f)
                t = 0.15
                stall = 0
        iteration += 1
    
    return best_f, _score(best_f)

def _fast_coord(f, stop_time, max_sweeps=200, seed=0):
    """Fast vectorized coordinate descent."""
    n = len(f)
    f = np.clip(f.copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, math.inf
    conv = _conv_full(f)
    best_score = 2.0 * n * float(conv.max()) / (S * S)
    rng = np.random.default_rng(seed)
    
    n_pts_coarse = 50
    n_pts_fine = 30
    
    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 2.0:
            break
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            
            v = f[i]
            S_rest = S - v
            
            k_start = i
            k_end = min(i + n, 2 * n - 1)
            len_aff = k_end - k_start
            k_2i = min(2 * i, 2 * n - 2)
            
            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                um = float(conv[:k_start].max())
                if um > unaff_max:
                    unaff_max = um
            if k_end < 2 * n - 1:
                um = float(conv[k_end:].max())
                if um > unaff_max:
                    unaff_max = um
            
            # Influence vector: 2*f[0:len_aff]
            infl = 2.0 * f[:len_aff]
            base_aff = conv[k_start:k_end].copy()
            
            # Search range
            if v > 1e-8:
                x_hi = min(3.0 * v + 0.5, 200.0)
            else:
                x_hi = 1.0
            x_lo = 0.0
            
            x_grid = np.linspace(x_lo, x_hi, n_pts_coarse)
            dv = x_grid - v
            
            # Vectorized affected computation
            # affected[p, q] = base_aff[q] + 2*dv[p]*f[q] = base_aff + dv[:,None]*infl[None,:]
            aff = base_aff[np.newaxis, :] + dv[:, np.newaxis] * infl[np.newaxis, :]
            
            # Fix 2i point
            if k_2i >= k_start and k_2i < k_end:
                idx2 = k_2i - k_start
                aff[:, idx2] += dv ** 2
            
            aff_max = aff.max(axis=1)
            total_max = np.maximum(aff_max, unaff_max)
            
            S_new = S_rest + x_grid
            valid = S_new > 0.01
            if not valid.any():
                continue
            
            scores = np.full(n_pts_coarse, math.inf)
            scores[valid] = 2.0 * n * total_max[valid] / (S_new[valid] ** 2)
            
            bi = int(scores.argmin())
            bx = x_grid[bi]
            bsc = scores[bi]
            
            if bsc < best_score - 1e-13 and abs(bx - v) > 1e-10:
                # Fine search
                step = x_hi / n_pts_coarse
                x_fine = np.linspace(max(0, bx - 2 * step), bx + 2 * step, n_pts_fine)
                df = x_fine - v
                
                aff_f = base_aff[np.newaxis, :] + df[:, np.newaxis] * infl[np.newaxis, :]
                if k_2i >= k_start and k_2i < k_end:
                    aff_f[:, k_2i - k_start] += df ** 2
                
                aff_max_f = aff_f.max(axis=1)
                total_max_f = np.maximum(aff_max_f, unaff_max)
                S_new_f = S_rest + x_fine
                valid_f = S_new_f > 0.01
                scores_f = np.full(n_pts_fine, math.inf)
                scores_f[valid_f] = 2.0 * n * total_max_f[valid_f] / (S_new_f[valid_f] ** 2)
                
                bf = int(scores_f.argmin())
                bx = x_fine[bf]
                bsc = scores_f[bf]
                
                if bsc < best_score - 1e-13:
                    w = bx - v
                    f[i] = bx
                    S += w
                    conv[k_start:k_end] += w * infl
                    if k_2i >= k_start and k_2i < k_end:
                        conv[k_2i] += w * w
                    best_score = bsc
                    improved = True
        
        if not improved:

    f = np.clip(f, 0, 1000)
    return f, _score(f)

def _pair_moves(f, stop_time, n_trials=200, seed=0):
    """Randomized 2-coordinate moves."""
    n = len(f)
    f = f.copy()
    best_score = _score(f)
    rng = np.random.default_rng(seed)
    
    mults = [0.2, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 0.0]
    
    for _ in range(n_trials):
        if time.monotonic() > stop_time - 3.0:
            break
        i, j = int(rng.integers(n)), int(rng.integers(n))
        if i == j:
            continue
        vi, vj = f[i], f[j]
        best_pair_sc = best_score
        best_vals = (vi, vj)
        
        for mi in mults:
            fi = vi * mi if vi > 1e-10 else (0.01 if mi > 0 else 0)
            for mj in mults:
                fj = vj * mj if vj > 1e-10 else (0.01 if mj > 0 else 0)
                if fi + f.sum() - vi - vj < 0.01:
                    continue
                f[i], f[j] = fi, fj
                sc = _score(f)
                if sc < best_pair_sc:
                    best_pair_sc = sc
                    best_vals = (fi, fj)
        
        f[i], f[j] = best_vals
    
    f = np.clip(f, 0, 1000)
    sc = _score(f)
    return f, sc
