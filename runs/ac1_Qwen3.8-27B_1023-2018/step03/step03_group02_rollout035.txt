def _conv_full(a):
    n = len(a)

def _build_conv_matrix(f, n):

    A = np.zeros((2 * n - 1, n))

    return A

    n = len(a)

        return best, _score(best)

    rng = np.random.default_rng(seed)
    
    while time.monotonic() < stop_time - 1.0:
        improved = False
        # 1D coordinate descent
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            mults = [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.6, 2.0, 0.0,

                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-15:
                    best_val = nv
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-15:
                best_sc = best_val_sc
                improved = True
        
        if not improved:
            break
    
    return best, best_sc

def _subgradient_pass(f0, n, stop_time, seed, n_iters=2000):
    """Fast subgradient method to quickly approach the optimum."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f_sum = f.sum()

        f_sum = n

    best_f = f.copy()
    best_score = _score(f)

    lr0 = 0.1
    for it in range(n_iters):
        if time.monotonic() > stop_time - 1.0:
            break
        b = _conv_full(f)
        M = b.max()
        K_star = np.where(b >= M - 1e-12 * M)[0]
        K_star = K_star[K_star < 2 * n - 1]

        # Subgradient: average of 2*R_k(f) over active k

            if math.isfinite(lp_sc2) and lp_sc2 < best_score:

                best_score = lp_sc2
        except Exception:
            pass

    # Phase 4: Other initializations (15% of time)
    other_stop = t0 + budget * 0.80
    inits = []
    # Cosine taper
    taper = 0.5 + 0.5 * np.cos(np.pi * i_arr / max(1, n - 1))
    inits.append(taper)
    # Bimodal
    bimodal = (np.exp(-0.5 * ((i_arr - n * 0.3) / (n * 0.12)) ** 2) +
               np.exp(-0.5 * ((i_arr - n * 0.7) / (n * 0.12)) ** 2))
    inits.append(bimodal + 0.01)
    # Multi-frequency
    sup = np.ones(n) * 0.2
    for fr in (1, 2, 3, 5, 7):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.25 * np.cos(2 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.005)
    # Parent + noise
    inits.append(np.abs(parent + rng_main.normal(0, 0.08 * max(1, np.std(parent)), n)) + 0.001)
    # Smooth random
    base = rng_main.normal(0, 1, n)
    kern = np.exp(-0.5 * (np.arange(-8, 9) / 3.0) ** 2)
    kern /= kern.sum()
    inits.append(np.abs(np.convolve(base, kern, 'same')) + 0.005)

    time_per_init = max(8.0, (other_stop - max(t0, time.monotonic())) / max(1, len(inits)))
    for idx, init in enumerate(inits):
        if time.monotonic() > other_stop - 5.0:
            break
        if best_score < 1.5028:
            break
        stop = min(other_stop - 3.0, time.monotonic() + time_per_init)
        if time.monotonic() >= stop:
            break
        try:
            res, sc = _lp_optimize(init, n, stop, seed + 7919 * (idx + 1))
            if np.all(np.isfinite(res)) and res.sum() > 0.01:
                exact = float(evaluate_sequence([float(x) for x in np.clip(res, 0, 1000)]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = np.clip(res, 0, 1000).copy()
                    best_score = exact
        except Exception:
            pass

    # Phase 5: Polish (remaining time before deadline)
    polish_stop = deadline - 5.0
    if time.monotonic() < polish_stop - 5.0:
        # Coordinate descent polish
        p1_stop = min(polish_stop, time.monotonic() + (polish_stop - time.monotonic()) * 0.4)
        try:
            best_seq, best_score = _polish(best_seq, p1_stop, seed=5001)
        except Exception:
            pass

        # Pair polish
        p2_stop = min(polish_stop, time.monotonic() + (polish_stop - time.monotonic()) * 0.3)
        if time.monotonic() < p2_stop:
            try:
                best_seq, best_score = _pair_polish(best_seq, p2_stop, seed=6001)
            except Exception:
                pass

        # Final coordinate pass
        if time.monotonic() < polish_stop - 2.0:
            try:
                best_seq, best_score = _polish(best_seq, polish_stop, seed=7001)
            except Exception:
                pass

    # Final validation
    final_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    final_seq = final_seq[np.isfinite(final_seq)]
    if len(final_seq) == 0:
        final_seq = np.ones(n)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
            if math.isfinite(final_sc):
                return [float(x) for x in fallback]
        except Exception:
            pass
        return [1.0]

    return [float(x) for x in final_seq]
