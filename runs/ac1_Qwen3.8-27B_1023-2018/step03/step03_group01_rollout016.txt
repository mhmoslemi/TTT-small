def _subgrad_step(f, n, lr):
    b = _conv_fft(f)
    M = float(b.max())
    # Active set: indices where b[k] is within epsilon of M
    eps = 1e-6 * M
    active = np.where(b >= M - eps)[0]
    
    # Subgradient: average of gradients over active set
    s = float(f.sum())
    grad = np.zeros(n)
    for k in active:
        # grad_k[i] = 2 * f[k-i] if 0 <= k-i <= n-1
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            grad[i] += 2.0 * f[k - i]

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
