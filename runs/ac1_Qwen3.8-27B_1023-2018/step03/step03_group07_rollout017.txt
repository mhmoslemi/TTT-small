        if idx < len(inits):

            inits.append(blend / blend.sum())

    p2_end = t0 + min(budget * 0.70, 700.0)
    if time.monotonic() < p2_end - 10:
        per_init = max(4.0, (p2_end - time.monotonic()) / max(1, len(inits)))
        for idx, init in enumerate(inits):
            if time.monotonic() > p2_end - 5:
                break
            alloc = min(time.monotonic() + per_init, p2_end - 2.0)
            if alloc - time.monotonic() < 2.0:
                continue
            try:
                result = _optimize_lp(init, stop_time=alloc, max_iters=50000, seed_lp=100 + idx)
                _update_best(result)
            except Exception:
                pass

    # Phase 3: Try slightly different lengths (10% time)
    p3_end = t0 + min(budget * 0.80, 800.0)
    if time.monotonic() < p3_end - 10:
        alt_lengths = [n - 10, n - 5, n + 5, n + 10]
        for nt in alt_lengths:
            if time.monotonic() > p3_end - 5:
                break
            if nt < 20 or nt > 500:
                continue
            # Resample parent to new length
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, parent)
            init_new = np.clip(init_new, 0, 1000.0)
            if init_new.sum() < 0.01:
                continue
            init_new = init_new / init_new.sum()
            alloc = min(time.monotonic() + 30.0, p3_end - 2.0)
            try:
                result = _optimize_lp(init_new, stop_time=alloc, max_iters=30000, seed_lp=500 + nt)
                _update_best(result)
            except Exception:
                pass

    # Phase 4: Final polishing of best (remaining time)
    if time.monotonic() < deadline - 3:
        a_polish = np.clip(best_seq, 0.0, 1000.0).astype(np.float64)
        a_polish = a_polish / a_polish.sum()
        result = _optimize_lp(a_polish, stop_time=deadline - 1.0,
                              max_iters=200000, seed_lp=999)
        _update_best(result)

    # Final validation with exact evaluator
    final_arr = np.clip(best_seq, 0.0, 1000.0)
    if final_arr.size == 0 or not np.isfinite(final_arr).all():
        return [1.0 / 32.0] * 32
    try:
        final_score = float(evaluate_sequence([float(v) for v in final_arr]))
    except Exception:
        final_score = float('inf')
    if not np.isfinite(final_score):
        return [1.0 / 32.0] * 32

    return [float(v) for v in final_arr]
