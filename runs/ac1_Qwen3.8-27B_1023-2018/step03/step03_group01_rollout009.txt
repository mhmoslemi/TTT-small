def _pg_step(f, n, lr, lam=0.5):
    """One projected gradient step on simplex for min max_k b[k]."""
    b = _conv(f)
    M = b.max()
    k = int(np.argmax(b))
    
    # Gradient of b[k] w.r.t. f
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    for i in range(lo, hi):
        grad[i] = 2.0 * f[k - i]
    
    # Projected gradient step
    f_new = f - lr * grad
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy(), M
    f_new /= s
    
    return f_new, M

def _conv(a):
    n = len(a)

    np.maximum(c, 0.0, out=c)
    return c

            continue

        # Normalize direction to same sum as f
        f_sum = f.sum()
        direction = g / gs * f_sum  # preserve sum direction

        # Line search: try t in {0.1, 0.2, 0.3, 0.5, 0.7, 1.0}
        best_t = 0
        best_f_t = f.copy()
        best_sc_t = best_sc
        for t in (0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
            f_t = (1.0 - t) * f + t * direction
            f_t = np.maximum(f_t, 0.0)
            s_t = f_t.sum()
            if s_t < 1e-10:
                continue
            f_t /= s_t
            sc_t = _score(f_t)
            if sc_t < best_sc_t:
                best_sc_t = sc_t
                best_t = t
                best_f_t = f_t.copy()

        if best_t > 0 and best_sc_t < best_sc - 1e-14:
            f = best_f_t
            best_f = f.copy()
            best_sc = best_sc_t
            stall = 0
        else:
            stall += 1
            if stall > 20:
                f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
                f = _norm1(f) or best_f.copy()
                stall = 0

    return best_f, best_sc


def _coord_polish(f0, n, stop_time, seed):
    """Coordinate and pair coordinate optimization."""
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0, None)
    f = _norm1(f)
    if f is None:
        return np.ones(n) / n, math.inf
    best_f = f.copy()
    best_sc = _score(f)
    no_full_pass = 0

    multipliers = (0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.0)

    while time.monotonic() < stop_time - 3.0 and no_full_pass < 8:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = f[i]
            for mlt in multipliers:
                val = orig * mlt if orig > 1e-15 else (0.01 if mlt > 0 else 0.0)
                f[i] = val
                sc = _score(f)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_f = f.copy()
                    improved = True
                    orig = val
            f[i] = orig
            # Fine adjustments
            if orig > 1e-10:
                for dt_frac in (0.02, -0.02, 0.05, -0.05):
                    f[i] = max(0.0, orig * (1.0 + dt_frac))
                    sc = _score(f)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        best_f = f.copy()
                        improved = True
                f[i] = orig

        if not improved:
            no_full_pass += 1
        else:
            no_full_pass = 0

    return best_f, best_sc


def _make_init(name, n, rng, parent=None):
    i_arr = np.arange(n, dtype=np.float64)
    if name == "parent":
        if parent is not None and len(parent) == n:
            f = parent.copy()
        elif parent is not None:
            f = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        else:
            f = np.ones(n)
        f = np.maximum(f, 0.01)
    elif name == "cosine":
        f = 0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n))
        f = f + 0.01
    elif name == "step":
        m = n // 2
        f = np.ones(n)
        f[:m] = 1.0
        f[m:] = 0.4
    elif name == "step2":
        m1 = n // 3
        m2 = 2 * n // 3
        f = np.ones(n)
        f[:m1] = 1.0
        f[m1:m2] = 0.7
        f[m2:] = 0.3
    elif name == "gauss":
        sigma = n * 0.25
        f = np.exp(-0.5 * ((i_arr - n / 2) / sigma) ** 2)
        f = f + 0.05
    elif name == "decay":
        f = 1.0 / (0.1 + i_arr / n)
    elif name == "multifreq":
        f = np.full(n, 0.3)
        for fr in (1, 2, 3, 5, 7, 11):
            ph = rng.uniform(0, 2 * math.pi)
            f += 0.25 * np.cos(2 * math.pi * fr * i_arr / n + ph)
        f = np.abs(f) + 0.02
    elif name == "rand":
        f = rng.uniform(0.1, 1.0, n)
    elif name == "pnoise":
        if parent is not None:
            f = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            noise = rng.normal(0, 0.3, n)
            f = np.abs(f + noise) + 0.02
        else:
            f = np.abs(rng.normal(0.5, 0.3, n)) + 0.02
    else:
        f = np.ones(n)
    f = np.clip(f, 0.0, 1000.0)
    f = np.where(np.isfinite(f), f, 0.0)
    return f


def _build_lp_A(f, n):
    """Build truncated A for LP: conv(f, g) <= M."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_p = len(parent)

    # Fallback
    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not math.isfinite(best_score):
        best_seq = np.ones(384) / 384.0
        n_p = 384
        best_score = _score(best_seq)

    def _try_update(cand):
        nonlocal best_seq, best_score
        if cand is None:
            return
        cand = np.clip(np.asarray(cand, dtype=np.float64).ravel(), 0, 1000)
        if cand.size == 0 or cand.sum() < 0.01:
            return
        try:
            sc = _score(cand)
        except Exception:
            return
        if sc < best_score - 1e-15:
            best_seq = cand.copy()
            best_score = sc

    rng_main = np.random.default_rng(seed)

    # === Phase 1: Multi-start subgradient (50% of budget) ===
    phase1_end = t0 + budget * 0.50
    inits = [
        ("parent", n_p),
        ("cosine", n_p),
        ("step", n_p),
        ("step2", n_p),
        ("gauss", n_p),
        ("decay", n_p),
        ("multifreq", n_p),
        ("pnoise", n_p),
    ]

    time_per_init = max(20.0, (phase1_end - t0 - 5.0) / len(inits))

    for idx, (name, n_init) in enumerate(inits):
        if time.monotonic() > phase1_end - 3.0:
            break
        stop = min(phase1_end - 2.0, time.monotonic() + time_per_init)
        try:
            init_f = _make_init(name, n_init, rng_main, parent)
            res, sc = _pg_optimize(init_f, n_init, stop, seed + 10000 + idx)
            _try_update(res)
        except Exception:
            pass

    # === Phase 2: LP refinement on best (30% of budget) ===
    phase2_end = t0 + budget * 0.80
    if time.monotonic() < phase2_end - 15.0 and best_score > 1.49:
        n_best = len(best_seq)
        stop_lp = min(phase2_end - 5.0, time.monotonic() + (phase2_end - time.monotonic()) * 0.7)
        try:
            res, sc = _lp_refine(best_seq, n_best, stop_lp, seed + 50001, max_iters=100)
            _try_update(res)
        except Exception:
            pass

        # Subgradient after LP
        if time.monotonic() < phase2_end - 5.0:
            try:
                res, sc = _pg_optimize(best_seq, len(best_seq),
                                       min(phase2_end - 3.0, time.monotonic() + 60.0),
                                       seed + 60001)
                _try_update(res)
            except Exception:
                pass

    # === Phase 3: Try other lengths (10% of budget) ===
    phase3_end = t0 + budget * 0.90
    if time.monotonic() < phase3_end - 10.0:
        other_lengths = [256, 320, 448, 512]
        for L in other_lengths:
            if time.monotonic() > phase3_end - 5.0:
                break
            stop_l = min(phase3_end - 3.0, time.monotonic() + 30.0)
            try:
                init_l = _make_init("parent", L, rng_main, parent)
                res, sc = _pg_optimize(init_l, L, stop_l, seed + 70000 + L)
                _try_update(res)
            except Exception:
                pass

    # === Phase 4: Coordinate polish (10% of budget) ===
    phase4_end = deadline - 5.0
    if time.monotonic() < phase4_end - 10.0:
        try:
            res, sc = _coord_polish(best_seq, len(best_seq), phase4_end, seed + 90001)
            _try_update(res)
        except Exception:
            pass

    # Final subgradient burst
    if time.monotonic() < deadline - 10.0:
        try:
            res, sc = _pg_optimize(best_seq, len(best_seq), deadline - 8.0, seed + 99001)
            _try_update(res)
        except Exception:
            pass

    # Final coordinate micro-polish
    if time.monotonic() < deadline - 5.0:
        try:
            res, sc = _coord_polish(best_seq, len(best_seq), deadline - 3.0, seed + 88001)
            _try_update(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        sc = evaluate_sequence(final_list)
        if math.isfinite(sc) and sc > 0:
            return final_list
    except Exception:
        pass

    # Fallback to parent
    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass

    return [1.0] * 320
