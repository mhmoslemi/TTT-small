import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p

def _conv_full(a):
    n = len(a)
    if n <= 256:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _conv_cross(a, b):
    n = len(a)
    if n <= 256:
        return np.convolve(a, b, mode="full")
    fsize = next_fast_len(n + len(b) - 1)
    F = rfft(a, fsize)
    G = rfft(b, fsize)
    c = irfft(F * G, fsize)[:n + len(b) - 1]
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    # A[i,j] = f[j] if 0 <= i-j < n, else 0
    # i ranges from 0 to 2n-2, j ranges from 0 to n-1
    # Condition: 0 <= i-j < n => i-n+1 <= j <= i
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_direction(f, n, M, timeout=5.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=bounds, method="highs",
            options={"presolve": True, "time_limit": timeout, 
                    "dual_feasibility_tolerance": 1e-10,
                    "primal_feasibility_tolerance": 1e-10},
        )
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1000.0)
    if g.sum() < 1e-15:
        return None
    return g

    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:

    best_score = _score(f)

    max_stall = 150

def _gradient_polish(a, stop_time, seed=4242):
    """Gradient-based polishing using subgradient of max convolution."""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 15
    
    # Subgradient-based method
    lr = 0.01
    while time.monotonic() < stop_time - 4.0:
        improved = False
        
        # Compute subgradient of M = max conv
        conv = _conv_full(best)
        M = float(conv.max())
        S = float(best.sum())
        
        # Active set: indices where conv ≈ M
        tol = 1e-10 * M + 1e-15
        active = np.where(conv >= M - tol)[0]
        
        # Subgradient: uniform over active set
        subgrad = np.zeros(n)
        for k in active:
            # d conv[k] / d a[i] = a[k-i] if 0 <= k-i < n
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            idx = j_min + np.arange(j_max - j_min + 1)
            subgrad[idx] += best[k - idx]
        subgrad /= len(active)
        
        # Gradient of score = 2n/M_factor * (subgrad/S - 2*M/S^2)
        # Actually: d(2n*M/S^2)/d a_i = 2n*(subgrad_i/S^2 - 2*M/S^3)
        grad = 2.0 * n * (subgrad / (S * S) - 2.0 * M / (S * S * S))
        
        # Normalized gradient step
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm
        
        # Try multiple step sizes
        best_t = 0
        best_sc_t = best_sc
        for t in [0.001, 0.003, 0.01, 0.03, 0.05, 0.1, 0.2, 0.3]:
            cand = np.clip(best - t * grad * max(1.0, best.mean()), 0, None)
            if cand.sum() < 0.01:
                continue
            sc = _score(cand)
            if sc < best_sc_t - 1e-15:
                best_sc_t = sc
                best_t = t
                improved = True
        
        if best_t > 0:
            best = np.clip(best - best_t * grad * max(1.0, best.mean()), 0, None)
            best_sc = best_sc_t
            lr = min(1.0, lr * 1.1)
        else:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
            lr = max(0.001, lr * 0.7)
        
        # Single coordinate polish
        order = rng.permutation(n)
        for i in order[:min(n, 200)]:
            if time.monotonic() > stop_time - 4.0:
                break
            orig = best[i]
            if orig < 1e-12:
                continue
            best_val = orig
            best_val_sc = best_sc
            for mlt in (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 0.0):
                best[i] = orig * mlt
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-15:
                    best_val = best[i]
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-15:
                best_sc = best_val_sc
                improved = True
        
        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0
    
    return best, best_sc

    no_imp = 0
    max_no_imp = 20

        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 4.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            if orig > 1e-12:

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)

    return p


def _conv_full(a):
    n = len(a)
    if n <= 256:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=4.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    remaining = timeout
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=bounds, method="highs",
            options={
                "presolve": True,
                "time_limit": max(0.5, remaining),
                "dual_feasibility_tolerance": 1e-10,
                "primal_feasibility_tolerance": 1e-10,
            },
        )
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1000.0)
    if g.sum() < 1e-15:
        return None
    return g

def _subgradient_step(a, stop_time, lr_init=0.02):
    """One round of subgradient descent on the max-conv objective."""
    n = len(a)
    best = a.copy()
    S = float(best.sum())
    conv = _conv_full(best)
    M = float(conv.max())
    tol = 1e-8 * M + 1e-15
    active = np.where(conv >= M - tol)[0]

    subgrad = np.zeros(n)
    for k in active:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        idx = np.arange(j_min, j_max + 1)
        subgrad[idx] += best[k - idx]
    subgrad /= len(active)

    grad = 2.0 * n * (subgrad / (S * S) - 2.0 * M / (S * S * S))
    g_norm = np.linalg.norm(grad)
    if g_norm < 1e-15:
        return best
    grad /= g_norm
    scale = max(1.0, best.mean())
    return np.clip(best - lr_init * grad * scale, 0, None)
