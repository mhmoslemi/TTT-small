    c = -np.ones(n)  # maximize sum = minimize -sum
    b_ub = np.full(len(active_indices), M)

        res = linprog(c, A_ub=A, b_ub=b_ub,

                      method='highs',

        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = g.sum()
    if g_sum < 1e-12:
        return None
    return g / g_sum

This is the coefficient of `x^k` in `(sum_j a[j] x^j)^2`. So `conv(a,a)` is the squared power of the polynomial `P(x) = sum_j a[j] x^j`.

This is equivalent to: find a polynomial `P(x) = sum_j a[j] x^j` with nonnegative coefficients, `P(1) = 1`, to minimize the maximum coefficient of `P(x)^2`.

By the uncertainty principle / Fejér-Riesz type results, the minimum is related to the "flatness" of the spectrum of P.

If `P(x) = (1 - x^n)/(n(1-x))` (geometric), then `P(x)^2` has coefficients that form a triangle, and the max is `1/n`. Score = 2.

To get below 2, we need a polynomial whose square has flatter coefficients. This is related to "low autocorrelation sequences" but for the convolution (not autocorrelation).

For the convolution (not autocorrelation), the structure is different. `P(x)^2` has the property that its coefficients are symmetric: `conv[k] = conv[2n-2-k]`. So we only need to consider `k = 0, ..., n-1`.

But the constraint is `a[j] >= 0`, which limits the polynomial to have nonnegative coefficients.

This is a well-studied problem in the theory of "flat polynomials" or "low-discrepancy sequences."

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    return p

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n, k_indices):
    """Build constraint matrix rows for given k indices."""
    n_rows = len(k_indices)
    all_rows = []
    all_cols = []
    all_vals = []
    for i, k in enumerate(k_indices):
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min > j_max:
            continue
        js = np.arange(j_min, j_max + 1)
        f_idx = k - js
        mask = (f_idx >= 0) & (f_idx < n)
        js = js[mask]
        f_idx = f_idx[mask]
        fv = f[f_idx]
        nonzero = fv > 1e-15
        js = js[nonzero]
        fv = fv[nonzero]
        if len(js) > 0:
            all_rows.extend([i] * len(js))
            all_cols.extend(js.tolist())
            all_vals.extend(fv.tolist())
    
    A = csr_matrix((all_vals, (all_rows, all_cols)), shape=(n_rows, n), dtype=np.float64)
    return A

def _solve_lp(f, n, M, k_indices, time_limit=5.0):
    A = _build_A(f, n, k_indices)
    c = -np.ones(n)
    b_ub = np.full(len(k_indices), M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0, None)
    g_sum = g.sum()
    if g_sum < 1e-12:
        return None
    return g / g_sum
