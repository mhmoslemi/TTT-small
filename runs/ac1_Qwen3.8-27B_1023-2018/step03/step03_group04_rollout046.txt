def _conv_fft(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

    """Fast local score (without evaluator overhead)."""
    n = len(a)
    S = float(a.sum())

    M = float(conv.max())

        return math.inf

def _build_conv_matrix(f):
    """Build the (2n-1) x n matrix A where A @ g = f * g (full conv)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_direction(f, M):

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_local(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    conv = _conv_fft(a)
    M = float(conv.max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M / (S * S)


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, M):
    n = len(f)
    if M < 1e-15:
        return None
    A = _build_A(f)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True,
                               "dual_feasibility_tolerance": 1e-9,
                               "primal_feasibility_tolerance": 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s


def _subgradient_step(f, conv, lr=0.05):
    n = len(f)
    M = float(conv.max())
    thresh = 0.92 * M
    active = np.where(conv >= thresh)[0]
    subgrad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            subgrad[i] += 2.0 * f[k - i]
    subgrad /= max(1, len(active))
    f_new = f - lr * subgrad
    f_new = np.clip(f_new, 0, None)
    s = f_new.sum()
    if s < 1e-10:
        return None
    return f_new / s


def _main_optimize(f0, stop_time, seed, max_iters=50000):
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs

    best_f = f.copy()
    best_score = _score_local(f)

    t = 0.25
    t_min = 0.02
    t_max = 0.7
    stall = 0
    max_stall = 180
    iteration = 0
    last_improve = 0

    while time.monotonic() < stop_time - 1.5 and iteration < max_iters:
        conv = _conv_fft(f)
        M = float(conv.max())
        score = 2.0 * n * M  # f normalized to sum 1

        g = _lp_direction(f, M)

        if g is None or np.linalg.norm(g - f) < 1e-8:
            stall += 5
            if stall > max_stall:
                strength = 0.04 + 0.04 * min(1.0, stall / (4 * max_stall))
                if rng.random() < 0.3:
                    f = np.clip(best_f + rng.normal(0, strength, n), 0, None)
                else:
                    f = np.clip(f + rng.normal(0, strength, n), 0, None)
                fs = f.sum()
                f = f / fs if fs > 1e-10 else best_f.copy()
                t = 0.25
                stall = 0
            elif stall > 80:
                sg = _subgradient_step(f, conv, lr=0.03 + 0.02 * rng.random())
                if sg is not None:
                    sc_s = _score_local(sg)
                    if sc_s < best_score * 1.005:
                        f = sg
                else:
                    f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                    fs = f.sum()
                    f = f / fs if fs > 1e-10 else best_f.copy()
            else:
                f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
                fs = f.sum()
                f = f / fs if fs > 1e-10 else best_f.copy()
            iteration += 1
            continue

        # Line search
        candidates = [t, t*1.4, t*0.6, t*0.3, t*2.0, t*0.15]
        candidates = [max(t_min, min(t_max, c)) for c in candidates]
        # Deduplicate
        seen = set()
        uni = []
        for c in candidates:
            k = round(c, 5)
            if k not in seen:
                seen.add(k)
                uni.append(c)

        best_t = t
        best_sc = score
        cand_f = None
        for tc in uni:
            f_new = (1.0 - tc) * f + tc * g
            f_new = np.clip(f_new, 0, None)
            s2 = f_new.sum()
            if s2 < 1e-10:
                continue
            f_new = f_new / s2
            sc = _score_local(f_new)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_t = tc
                cand_f = f_new.copy()

        if cand_f is not None:
            f = cand_f
            if best_sc < best_score - 1e-12:
                best_f = f.copy()
                best_score = best_sc
                stall = 0
                last_improve = iteration
                t = min(t_max, t * 1.04)
            else:
                t = max(t_min, t * 0.7)
        else:
            stall += 1
            t = max(t_min, t * 0.55)
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
                fs = f.sum()
                f = f / fs if fs > 1e-10 else best_f.copy()
                t = 0.25
                stall = 0

        iteration += 1

    return best_f, best_score


def _polish(a, stop_time, seed=4242):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score_local(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 12

    m1 = [0.2, 0.4, 0.6, 0.8, 1.1, 1.3, 1.6, 2.2, 0.0, 0.05, 3.0]

    while time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            bv = orig
            bv_sc = best_sc
            for m in m1:
                nv = orig * m if orig > 1e-12 else (0.005 if m > 0 else 0.0)
                best[i] = nv
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score_local(best)
                if sc < bv_sc - 1e-14:
                    bv = nv
                    bv_sc = sc
            best[i] = bv
            if bv_sc < best_sc - 1e-14:
                best_sc = bv_sc
                improved = True

        # 2-coordinate moves
        n2 = min(120, n // 3)
        for _ in range(n2):
            if time.monotonic() > stop_time - 1.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.4, 0.7, 1.3, 1.7, 0.0, 2.5):
                for mj in (0.4, 0.7, 1.3, 1.7, 0.0, 2.5):
                    best[i] = oi * mi if oi > 1e-12 else (0.005 if mi > 0 else 0.0)
                    best[j] = oj * mj if oj > 1e-12 else (0.005 if mj > 0 else 0.0)
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score_local(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        # Group moves: zero/set small blocks
        if not improved and no_imp > 4:
            blk = max(2, n // 32)
            s = int(rng.integers(0, n - blk))
            saved = best[s:s+blk].copy()
            best[s:s+blk] = 0.0
            if best.sum() > 0.01:
                sc = _score_local(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                else:
                    best[s:s+blk] = saved
            else:
                best[s:s+blk] = saved

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc


def _parametric_init(n, family, rng):
    i = np.arange(n, dtype=np.float64)
    u = i / max(1, n - 1)
    if family == 0:
        return np.exp(-u / (0.2 + 0.3 * rng.random()))
    elif family == 1:
        alpha = 0.3 + 0.7 * rng.random()
        return (1.0 - u) ** alpha
    elif family == 2:
        return 0.3 + 0.7 * np.cos(np.pi * u)
    elif family == 3:
        a1, a2 = rng.uniform(0.1, 0.4), rng.uniform(0.6, 0.9)
        w1, w2 = rng.uniform(0.5, 2.0), rng.uniform(0.5, 2.0)
        return w1 * np.exp(-((u - a1) / 0.15) ** 2) + w2 * np.exp(-((u - a2) / 0.15) ** 2)
    elif family == 4:
        return (1.0 + np.sin(2 * np.pi * rng.uniform(1, 4) * u)) * 0.5 + 0.1
    elif family == 5:
        gamma = 0.5 + 1.5 * rng.random()
        return (1.0 / (1.0 + i / (n * 0.3))) ** gamma
    else:
        return np.abs(rng.normal(0, 1, n)) + 0.05
