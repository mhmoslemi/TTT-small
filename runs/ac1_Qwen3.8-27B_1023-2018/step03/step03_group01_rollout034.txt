def _conv(a):
    """Full convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    # Numerical errors can give tiny negative values
    np.maximum(c, 0.0, out=c)
    return c

    if s < 0.01:

        b = _conv(a)
    m = float(b.max())

        # Use a threshold: b[k] >= M - 0.05*M
        threshold = M * 0.95
        active_mask = b >= threshold
        active_k = np.where(active_mask)[0].tolist()
        
        # Also always include the argmax
        argmax_k = int(np.argmax(b))
        if argmax_k not in active_k:
            active_k.append(argmax_k)
        
        # Solve LP
        g = _lp_direction(f, n, M, active_k, time_limit=15.0)
        if g is None:
            stall += 5
            if stall > max_stall:
                # Jump
                f = np.clip(f + rng.normal(0, 0.1, n), 0, None)
                s = f.sum()
                f = f / s if s > 1e-10 else best_f.copy()
                b = _conv(f)
                sc = _score_fast(f, b)
                if sc < best_sc:
                    best_f = f.copy()
                    best_sc = sc
                stall = 0
            continue
        
        dist = float(np.linalg.norm(g - f))
        if dist < 1e-6:
            stall += 3
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.15, n), 0, None)
                s = f.sum()
                f = f / s if s > 1e-10 else best_f.copy()
                b = _conv(f)
                stall = 0
            continue
        
        # Line search
        res_t, f_new = _line_search(f, g, n, best_sc, max_iter=15, stop_time=stop_time)
        if f_new is not None:
            sc_new = _score_fast(f_new)
            if sc_new < best_sc - 1e-13:
                f = f_new
                best_f = f.copy()
                best_sc = sc_new
                b = _conv(f)
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            # Try smaller step
            f_try = f + 0.1 * (g - f)
            np.maximum(f_try, 0, out=f_try)
            sc_try = _score_fast(f_try)
            if sc_try < best_sc - 1e-13:
                f = f_try
                best_f = f.copy()
                best_sc = sc_try
                b = _conv(f)
                stall = 0
        
        if stall >= max_stall:
            # Perturbation
            sigma = min(0.3, 0.05 * (1 + stall / max_stall))
            f = np.clip(f + rng.normal(0, sigma, n), 0, None)
            s = f.sum()
            f = f / s if s > 1e-10 else best_f.copy()
            b = _conv(f)
            if _score_fast(f, b) < best_sc:
                best_f = f.copy()
                best_sc = _score_fast(f, b)
            stall = 0
    
    return best_f, best_sc
