One idea: **alternating LP**. Instead of conv(f, g) <= M, use conv(g, f) <= M (which is the same by bilinearity, so no change). Or use a different constraint: conv(g, g) <= M. But the latter is quadratic in g, not LP.

Another idea: **two LPs**. Solve:
1. max sum(g) s.t. conv(f, g) <= M, g >= 0
2. max sum(h) s.t. conv(h, f) <= M, h >= 0

By symmetry, these give the same solution. So that doesn't help.

What about **dual LP**? The dual is:
min M * sum(y) s.t. A^T y >= 1, y >= 0

The solution y* tells us which constraints are binding. If we know the binding constraints, we can solve a smaller LP.

But in practice, the primal LP is fast enough.

    def H(t):
        v = (1.0 - t) * (1.0 - t) * cf + 2.0 * t * (1.0 - t) * cfx + t * t * cx
        return float(v.max())
    ts = np.linspace(0.0, 1.0, 160)

    best_t = 0.0

def _polish(a, stop, seed):
    n = len(a)
    rng = np.random.default_rng(seed)
    a = np.clip(a.copy(), 0, None)
    a /= max(a.sum(), 1e-10)
    bs = _score(a)
    no_imp = 0
    
    while time.monotonic() < stop - 1.0:
        improved = False
        
        # Coordinate descent
        for i in rng.permutation(n):
            if time.monotonic() > stop - 1.0:
                break
            orig = a[i]
            bi, bis = orig, bs
            for m in (0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 0.0):
                nv = orig * m if orig > 1e-8 else m
                a[i] = nv
                if a.sum() < 0.01:
                    a[i] = orig
                    continue
                sc = _score(a)
                if sc < bis:
                    bi, bis = nv, sc
            a[i] = bi
            if bis < bs - 1e-14:
                bs = bis
                improved = True
        
        # Pair moves
        for _ in range(40):
            if time.monotonic() > stop - 1.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = a[i], a[j]
            for r in (0.2, 0.5, 1.5, 2.0):
                a[i] = r * oi
                a[j] = oj + oi - r * oi
                if a[j] < -1e-10 or a.sum() < 0.01:
                    continue
                sc = _score(a)
                if sc < bs - 1e-14:
                    bs = sc
                    improved = True
            a[i], a[j] = oi, oj
        
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 5:
                break
    
    return a, bs

def _line_search(cf, cfx, cx, n):
    def H(t):
        v = (1.0 - t) * (1.0 - t) * cf + 2.0 * t * (1.0 - t) * cfx + t * t * cx
        return float(v.max())
    
    # Coarse grid
    ts = np.linspace(0.0, 1.0, 150)
    vals = np.array([H(t) for t in ts])
    i = int(vals.argmin())
    best_v, best_t = float(vals[i]), float(ts[i])
    
    # Fine grid around best
    lo = max(0.0, float(ts[max(0, i - 3)]))
    hi = min(1.0, float(ts[min(len(ts) - 1, i + 3)]))
    ts2 = np.linspace(lo, hi, 80)
    vals2 = np.array([H(t) for t in ts2])
    j = int(vals2.argmin())
    if vals2[j] < best_v:
        best_v, best_t = float(vals2[j]), float(ts2[j])
    
    return best_t, 2.0 * n * best_v
