def _generate_initials(n, parent, seed):

    if parent is not None and parent.size == n:

                stall = 0

        tau = max(1e-12, 0.5 * spread)
        w = np.exp((top_val - mv) / tau)
        w = np.where(np.isfinite(w), w, 0.0)
        ws = float(w.sum())
        if ws <= 0.0:
            w = np.ones_like(top_val)
        else:

def _lp_improve(a, mask, internal_eval, stop, seed=0, time_limit=8.0):

    if gs <= 1e-14:

    if float(np.linalg.norm(g - a)) < 1e-7:
        return a

    best_a = a.copy()
    best_sc = internal_eval(a)
    rng = np.random.default_rng(seed)

    for t in (0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 1.0):

    # Uniform is globally optimal for the supplied nonnegative evaluator:
    # max(a*a) >= sum(a)^2/n, hence score >= 2. If the actual evaluator shows
    # no score below 2, return it immediately. Otherwise run a search for the
    # actual objective.
    initial_n = base_n
    for a in _generate_initials(initial_n, parent, seed):
        add_candidate(a)

    # Ensure at least a uniform candidate was real.
    unif = np.full(initial_n, 1.0 / max(1, initial_n), dtype=np.float64)
    add_candidate(unif)

    cand_list.sort(key=lambda z: z[0])
    top_cands = [a for sc, a in cand_list if math.isfinite(sc)][:4]
    if best_arr is None and top_cands:
        best_arr = top_cands[0]
        best_actual = _actual_eval(best_arr)

    # If the supplied evaluator behaves as proven, stop at the analytic optimum.
    if math.isfinite(best_actual) and best_actual >= 2.0 - 1e-8:
        sc_un = _actual_eval(unif)
        if math.isfinite(sc_un) and (best_arr is None or sc_un <= best_actual + 1e-12):
            best_arr = unif
            best_actual = sc_un
        out = _to_list(best_arr if best_arr is not None else unif)
        try:
            sc_out = evaluate_sequence(out)
            if not math.isfinite(sc_out) or sc_out <= 0:
                out = _to_list(unif)
        except Exception:
            out = _to_list(unif)
        return out

    # If the actual evaluator admits values below 2, perform a bounded search.
    def _dedupe_masks(masks):
        uniq = []
        for m in masks:
            mm = np.asarray(m, dtype=bool)
            ok = True
            for u in uniq:
                if mm.size == u.size and np.array_equal(mm, u):
                    ok = False
                    break
            if ok and mm.size > 0:
                uniq.append(mm)
        return uniq

    def _refresh_top(k=5):
        cand_list.sort(key=lambda z: z[0])
        return [a for sc, a in cand_list if math.isfinite(sc)][:k]

    # Phase 1: subgradient steps on several masks/candidates.
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 25.0:
        stop1 = min(deadline - 45.0, time.monotonic() + max(15.0, 0.22 * remaining))
        tops = _refresh_top(3)
        if tops and best_arr is not None:
            mask_best = _infer_mask(best_arr, best_actual)
            mask_off = _off_mask(base_n)
            masks1 = _dedupe_masks([mask_best, mask_off])
            if not masks1:
                masks1 = [np.ones(2 * base_n - 1, dtype=bool)]
            for ci, cand in enumerate(tops):
                for mi, mask in enumerate(masks1):
                    if time.monotonic() > stop1:
                        break
                    per_stop = min(stop1, time.monotonic() + 12.0)
                    try:
                        res = _subgrad_optimize(cand, mask, per_stop, seed + 111 * (ci + mi + 1))
                    except Exception:
                        res = None
                    if res is not None:
                        add_candidate(res)

    # Phase 2: coordinate descent with internal masked scores.
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 25.0:
        stop2 = min(deadline - 25.0, time.monotonic() + max(30.0, 0.45 * remaining))
        tops = _refresh_top(3)
        if tops and best_arr is not None:
            mask_best = _infer_mask(best_arr, best_actual)
            mask_off = _off_mask(base_n)
            masks2 = _dedupe_masks([mask_best, mask_off])
            if not masks2:
                masks2 = [np.ones(2 * base_n - 1, dtype=bool)]
            for ci, cand in enumerate(tops):
                for mi, mask in enumerate(masks2):
                    if time.monotonic() > stop2:
                        break
                    evalm = lambda x, m=mask: _score_mask_arr(x, m)
                    per_stop = min(stop2, time.monotonic() + 20.0)
                    try:
                        res = _coordinate_optimize(cand, evalm, per_stop, seed + 222 * (ci + mi + 1))
                    except Exception:
                        res = None
                    if res is not None:
                        add_candidate(res)

    # Phase 3: a few LP-guided directions, if affordable.
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 55.0 and base_n <= 768:
        stop3 = min(deadline - 20.0, time.monotonic() + max(20.0, 0.18 * remaining))
        tops = _refresh_top(2)
        if tops and best_arr is not None:
            mask_best = _infer_mask(best_arr, best_actual)
            mask_off = _off_mask(base_n)
            masks3 = _dedupe_masks([mask_best, mask_off])
            if not masks3:
                masks3 = [np.ones(2 * base_n - 1, dtype=bool)]
            for ci, cand in enumerate(tops):
                for mi, mask in enumerate(masks3):
                    if time.monotonic() > stop3:
                        break
                    evalm = lambda x, m=mask: _score_mask_arr(x, m)
                    per_stop = min(stop3, time.monotonic() + 8.0)
                    try:
                        res = _lp_improve(cand, mask, evalm, per_stop, seed + 333 * (ci + mi + 1))
                    except Exception:
                        res = None
                    if res is not None:
                        add_candidate(res)

    # Phase 4: explore a few nearby lengths by interpolation.
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 50.0 and best_arr is not None:
        stop4 = deadline - 10.0
        Ls = set()
        for L in (128, 192, 256, 320, 384, 448, 512, 640, 768, 896, 1024):
            if L >= 64 and L <= 1024:
                Ls.add(L)
        for base in (base_n // 2, base_n, int(1.25 * base_n), int(1.5 * base_n), int(2.0 * base_n)):
            if 64 <= base <= 1024:
                Ls.add(base)
        Ls.discard(base_n)
        for L in sorted(Ls):
            if time.monotonic() > stop4 - 5.0:
                break
            try:
                a = np.interp(
                    np.linspace(0.0, 1.0, L),
                    np.linspace(0.0, 1.0, base_n),
                    best_arr[:base_n] if best_arr.size >= base_n else best_arr,
                )
            except Exception:
                continue
            a = _normalize(a)
            if a is None:
                continue
            sc = add_candidate(a)
            if not math.isfinite(sc):
                continue
            maskL = _infer_mask(a, sc)
            evalL = lambda x, m=maskL: _score_mask_arr(x, m)
            per_stop = min(stop4 - 5.0, time.monotonic() + 12.0)
            try:
                res = _coordinate_optimize(a, evalL, per_stop, seed + 4444 + L)
            except Exception:
                res = None
            if res is not None:
                add_candidate(res)

    # Final cleanup: short coordinate polish on the current best, then validate.
    tops = _refresh_top(1)
    if tops and time.monotonic() < deadline - 8.0:
        cand = tops[0]
        maskF = _infer_mask(cand, _actual_eval(cand))
        evalF = lambda x, m=maskF: _score_mask_arr(x, m)
        per_stop = min(deadline - 8.0, time.monotonic() + 12.0)
        try:
            res = _coordinate_optimize(cand, evalF, per_stop, seed + 999999)
        except Exception:
            res = None
        if res is not None:
            add_candidate(res)

    if best_arr is None:
        best_arr = np.full(base_n, 1.0 / max(1, base_n), dtype=np.float64)

    # Ensure a valid exact-score fallback is preserved.
    sc_un = _actual_eval(np.full(base_n, 1.0 / max(1, base_n), dtype=np.float64))
    if math.isfinite(sc_un) and sc_un < best_actual + 1e-12:
        best_arr = np.full(base_n, 1.0 / max(1, base_n), dtype=np.float64)
        best_actual = sc_un

    out = _to_list(best_arr)
    try:
        sc_out = evaluate_sequence(out)
        if not math.isfinite(sc_out) or sc_out <= 0:
            out = _to_list(np.full(base_n, 1.0 / max(1, base_n), dtype=np.float64))
    except Exception:
        out = _to_list(np.full(base_n, 1.0 / max(1, base_n), dtype=np.float64))

    return out
