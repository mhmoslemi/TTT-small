    g[mask] -= a[mask] * w[idx[mask]]
    return np.maximum(g, 0.0)


def _aw(a, w, n):
    n2 = 2 * n - 1
    L = n + n2 - 1
    nf = next_fast_len(L + 1)
    ca = rfft(a, nf)
    cw = rfft(w, nf)
    return irfft(ca * np.conj(cw), nf)[:L]


def _apexM(c, n):
    m = 2 * n - 1
    M = c.max()
    if not np.isfinite(M) or M <= 0:
        z = np.ones(n)
    else:
        T = max(M * 0.03, 1e-12)
        z = np.exp(np.clip((c - M) / T, -50, 0))
        s = z.sum()
        z = z / s if s > 0 else np.ones(n)
    K = max(4, min(24, n // 16))
    idx = np.argpartition(c, m - K)[m - K:] if K < m else np.arange(m)
    wt = np.zeros(m)
    wt[idx] = 1.0 / K
    return 0.6 * wt + 0.4 * z


def _try(cand, bmp, bm, bc):
    if cand is None:
        return bmp, bm, bc
    cand = np.asarray(cand, dtype=np.float64).ravel()
    if cand.size == 0 or not np.all(np.isfinite(cand)):
        return bmp, bm, bc
    c = _conv(cand)
    mm = c.max()
    if mm < bm - 1e-15 * bm:
        return cand, mm, c
    return bmp, bm, bc


def _resample(a, m):
    a = _norm1(a)
    n = a.size
    m = int(m)
    if m <= 0:
        return np.ones(1)
    if n == m:
        return a.copy()
    if m > n:
        idx = (np.arange(m) * n) // m
        cnt = np.bincount(idx, minlength=n).astype(float)
        cnt[cnt <= 0] = 1
        b = a[idx] / cnt[idx]
    else:
        b = np.zeros(m)
        idx = (np.arange(n) * m) // n
        np.add.at(b, idx, a)
        np.maximum(b, 0, out=b)
    return _norm1(b)


def _refine(a, dl, rng):
    a = _norm1(a)
    n = a.size
    if n < 2:
        return a
    c = _conv(a)
    M = c.max()
    best = a.copy()
    bM = M
    itr = 0
    while time.monotonic() < dl - 0.5:
        itr += 1
        if M < bM - 1e-13 * bM:
            best, bM, _ = a, M, c
        rem = dl - time.monotonic()
        if rem < 3:
            break
        dLP = None
        dDual = None
        lp_freq = 1 if itr < 60 else (2 if itr < 200 else 3)
        if itr % lp_freq == 0 and n <= 768:
            dLP = _lp_dir(a, M, dl, rng)
            if itr % max(lp_freq, 4) == 0 and n <= 512 and rem > 6:
                try:
                    A = _A_mat(a)
                except Exception:
                    A = None
                if A is not None:
                    dDual = _lp_dual(A, M, n, dl, rng)
        dW = None
        if dDual is not None:
            dW = dDual
        elif dLP is not None:
            c2 = _conv(dLP)
            dW = _apexM(c2, n)
        if dLP is not None:
            dvec = dLP - a
            if np.linalg.norm(dvec) > 1e-14:
                cand, M, c = _try(a + dvec, None, M, c)
        if dW is not None:
            g = _wgrad(a, dW)
            gn = np.linalg.norm(g)
            if gn > 1e-14:
                step = max(n, 2) * bM / (float(g.sum()) + 1e-30)
                t0 = min(1.0, step * 0.8)
                t = t0
                improved = False
                for _ in range(18):
                    v = a - t * g
                    np.maximum(v, 0, out=v)
                    s = v.sum()
                    if s > 1e-12:
                        v /= s
                    else:
                        t *= 0.5
                        if t < 1e-8:
                            break
                        continue
                    nc = _conv(v)
                    nm = nc.max()
                    if nm < M - 1e-15:
                        a, c, M, improved = v, nc, nm, True
                        break
                    t *= 0.5
                    if t < 1e-8:
                        break
                if not improved and t0 > 1e-8:
                    v = a - t * g
                    np.maximum(v, 0, out=v)
                    s = v.sum()
                    if s > 1e-12:
                        v /= s
                        nc = _conv(v)
                        if nc.max() < M:
                            a, c, M = v, nc, nc.max()
        else:
            w = _apexM(c, n)
            g = _wgrad(a, w)
            gn = np.linalg.norm(g)
            if gn > 1e-14:
                step = max(n, 2) * bM / (float(g.sum()) + 1e-30)
                t = min(1.0, step * 0.5)
                for _ in range(10):
                    v = a - t * g
                    np.maximum(v, 0, out=v)
                    s = v.sum()
                    if s > 1e-12:
                        v /= s
                    else:
                        t *= 0.5
                        continue
                    nc = _conv(v)
                    if nc.max() < M:
                        a, c, M = v, nc, nc.max()
                        break
                    t *= 0.5
                    if t < 1e-8:
                        break
        if M < bM - 1e-13 * bM:
            best, bM = a.copy(), M
        if itr > 80 and itr % 40 == 0 and rem > 25:
            r = rng.random()
            if r < 0.5:
                a = _norm1(a + rng.normal(0, 0.005, n))
            else:
                a = _norm1(0.9 * best + 0.1 * _rnd_shape(n, rng))
            c = _conv(a)
            M = c.max()
    if best is not None:
        _best_c = _conv(best)
        if _best_c.max() < bM:
            bM = _best_c.max()
    return _norm1(best)


def _rnd_shape(n, rng, base=None):
    x = np.linspace(0, 1, n, endpoint=False)
    k = int(rng.integers(0, 5))
    if k == 0:
        a = x ** rng.uniform(0.3, 2.5)
    elif k == 1:
        a = (1 - x) ** rng.uniform(0.3, 2.5)
    elif k == 2:
        c, w2 = rng.uniform(0.2, 0.8), rng.uniform(0.05, 0.35)
        a = np.exp(-((x - c) ** 2) / (2 * w2 ** 2)) + 0.1
    elif k == 3:
        a = np.ones(n)
        for kk in range(1, max(1, n // 50) + 1):
            a += rng.uniform(0.05, 0.8 / kk) * np.cos(2 * np.pi * kk * x + rng.uniform(0, 2 * np.pi))
    else:
        a = np.ones(n)
        for _ in range(int(rng.integers(2, 4))):
            a += rng.uniform(0.1, 0.8) * np.exp(-((x - rng.uniform(0.1, 0.9)) ** 2) / (2 * rng.uniform(0.04, 0.3) ** 2))
    a = np.maximum(a, 1e-6)
    if base is not None and len(base) == n:
        a = 0.5 * _norm1(base) + 0.5 * a
    return _norm1(a)


def _polish(a, dl, rng):
    a = _norm1(a)
    n = a.size
    c = _conv(a)
    M = c.max()
    best, bM = a.copy(), M
    for _ in range(500):
        if time.monotonic() >= dl - 0.5:
            break
        w = _apexM(c, n)
        g = _wgrad(a, w)
        if np.linalg.norm(g) < 1e-16:
            break
        step = max(n, 2) * bM / (g.sum() + 1e-30)
        t = min(0.3, step * 0.5)
        improved = False
        for _ in range(6):
            v = a - t * g
            np.maximum(v, 0, out=v)
            s = v.sum()
            if s > 1e-12:
                v /= s
            else:
                t *= 0.5
                continue
            nc = _conv(v)
            if nc.max() < M - 1e-16:
                a, c, M = v, nc, nc.max()
                improved = True
                break
            t *= 0.5
            if t < 1e-12:
                break
        if M < bM - 1e-15 * bM:
            best, bM = a.copy(), M
        if not improved:
            a = _norm1(a + rng.normal(0, 1e-5, n))
            c = _conv(a)
            M = c.max()
    _best_c = _conv(best)
    return _norm1(best)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(1.0, float(budget_s))
    except Exception:
        budget = 1300.0
    if budget <= 2.0:
        p = _get_parent()
        sc = _exact_score(p)
        if np.isfinite(sc):
            return [float(x) for x in p]
        return [1.0]
    dl = t0 + budget - max(8.0, 0.02 * budget)
    if dl <= t0 + 3:
        dl = t0 + max(3.0, budget * 0.9)
    rng = np.random.default_rng(int(seed) if np.isfinite(float(seed)) else 42)
    best_a = _norm1(_get_parent())
    best_sc = _exact_score(best_a)
    if not np.isfinite(best_sc):
        best_a = np.ones(256) / 256
        best_sc = _exact_score(best_a)
    def _maybe_upd(c, tag):
        nonlocal best_a, best_sc
        if c is None:
            return
        c = _norm1(c)
        s2 = _exact_score(c)
        if np.isfinite(s2) and s2 < best_sc:
            best_a, best_sc = c, s2
    n0 = best_a.size
    for L in sorted(set([512, 384, 256, n0]), key=lambda x: (-{512: 3, 384: 2, 256: 1}.get(x, 0), x)):
        if time.monotonic() >= dl - 25:
            break
        L = int(L)
        if L < 8:
            continue
        rem = dl - time.monotonic() - 20
        wl = {512: 1.0, 384: 0.55, 256: 0.35}.get(L, 0.3)
        tl = max(15.0, wl * rem * 0.55)
        tl = min(tl, dl - time.monotonic() - 18)
        if tl < 12:
            continue
        init = _resample(best_a, L)
        try:
            res = _refine(init, time.monotonic() + tl, rng)
            _maybe_upd(res, f"main_{L}")
        except Exception:
            pass
        if time.monotonic() < dl - 35:
            try:
                r2 = _refine(_rnd_shape(L, rng, base=best_a), time.monotonic() + min(25, dl - time.monotonic() - 25), rng)
                _maybe_upd(r2, f"r2_{L}")
            except Exception:
                pass
    polT = dl - time.monotonic() - 5
    if polT > 15:
        polT = min(polT, 300)
        try:
            p = _polish(_resample(best_a, int(best_a.size)), time.monotonic() + polT, rng)
            _maybe_upd(p, "polish")
        except Exception:
            pass
        pol2 = dl - time.monotonic() - 5
        if pol2 > 10:
            try:
                p2 = _polish(_resample(best_a, int(best_a.size)), time.monotonic() + min(pol2 * 0.4, 120), rng)
                _maybe_upd(p2, "polish2")
            except Exception:
                pass
    for L in range(max(2, best_a.size - 32), min(best_a.size + 33, max(best_a.size + 1, 768))):
        if time.monotonic() >= dl - 3:
            break
        if L < 8:
            continue
        c = _resample(best_a, L)
        sc = _exact_score(c)
        if np.isfinite(sc) and sc < best_sc:
            best_a, best_sc = c, sc
    try:
        best_a = _norm1(best_a)
        fs = _exact_score(best_a)
        if np.isfinite(fs) and fs < best_sc:
            best_sc = fs
    except Exception:
        pass
    try:
        out = np.asarray(best_a, dtype=np.float64).ravel()
        if out.size < 1 or not np.all(np.isfinite(out)):
            out = np.ones(64) / 64
        out = np.clip(out, 0.0, 1000.0)
        out = _norm1(out)
        if out.size < 1:
            return [1.0]
        return [float(x) for x in out]
    except Exception:
        return [1.0]
