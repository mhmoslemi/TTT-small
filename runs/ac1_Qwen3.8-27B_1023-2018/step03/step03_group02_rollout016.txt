    grad = np.zeros(n)
    for k in active:

        lo = max(0, k - n + 1)

    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    else:
        a_new = a.copy()
    
    return a_new, m

def _conv(a):
    n = len(a)
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _sc(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _mkA(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp(f, n, M, tl=3.0, tk=None):

    c = -np.ones(n)
    if tk and tk < 2 * n - 1:

        ix = np.argsort(-fg)[:tk]
        As = A[ix]
    else:
        As = A

                    options={'presolve': True, 'time_limit': tl})

        return None

        return None
    g = r.x.copy()

    return g / s
