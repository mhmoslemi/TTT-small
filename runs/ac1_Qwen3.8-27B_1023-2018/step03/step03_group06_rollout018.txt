def _fft_len(m):

def _conv_self(a):
    """Full auto-convolution of a (length n) → length 2n-1."""
    n = a.size
    L = _fft_len(2*n-1)
    fa = rfft(a, L)
    c = irfft(fa*fa, L)[:2*n-1]
    return np.maximum(c, 0.0)  # Numerical safety

def _conv_cross(a, b):
    """Full convolution a * b."""
    n1, n2 = a.size, b.size
    L = _fft_len(n1+n2-1)
    c = irfft(rfft(a, L) * rfft(b, L), L)[:n1+n2-1]

    n = f.size
    M = float(_conv_self(f).max())

    A = np.zeros((2*n-1, n+1))
    for j in range(n):
        A[j:j+n, j] = f
    A[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, 1.0)]*n + [(0.0, None)]
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0  # minimize x[n]
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0.001, stop - time.monotonic())})

        return None, M

    gs = g.sum()
    if gs < 1e-12:
        return None, M
    g = g / gs
    return g, M

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize

_fft_cache = {}

def _fft_len(m):
    m = int(max(1, m))
    if m not in _fft_cache:
        _fft_cache[m] = int(next_fast_len(m))
    return _fft_cache[m]

def _conv_self(a):
    n = a.size
    L = _fft_len(2 * n - 1)
    fa = rfft(a, L)
    c = irfft(fa * fa, L)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _conv_cross(a, b):
    n1, n2 = a.size, b.size
    L = _fft_len(n1 + n2 - 1)
    c = irfft(rfft(a, L) * rfft(b, L), L)[:n1 + n2 - 1]
    return np.maximum(c, 0.0)

def _valid_conv(d, a):
    n = a.size
    ar = a[::-1].copy()
    c = _conv_cross(d, ar)
    return c[n - 1:2 * n - 1]

def _score(a):
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0)
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a = a / s
    return 2.0 * a.size * _conv_self(a).max()

def _exact(lst):
    try:
        return evaluate_sequence(lst)
    except Exception:
        return float('inf')

def _normalize(a):
    a = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones_like(a) / max(1, a.size)
    return a / s

def _best_t(cf, m, q):
    ts = np.linspace(0.0, 1.0, 65)
    vals = np.max(cf[None] + m[None] * ts[:, None] + q[None] * ts[:, None] ** 2, axis=1)
    i = int(np.argmin(vals))
    lo, hi = ts[max(0, i - 2)], ts[min(len(ts) - 1, i + 2)]
    if hi <= lo:
        hi = min(1.0, lo + 0.05)
    ts2 = np.linspace(lo, hi, 257)
    vals2 = np.max(cf[None] + m[None] * ts2[:, None] + q[None] * ts2[:, None] ** 2, axis=1)
    return float(ts2[int(np.argmin(vals2))])

def _lbfgs_smooth(a0, eta, stop, maxiter=200, maxfun=800):
    n = a0.size
    b = int(np.argmax(a0))
    base = max(float(a0[b]), 1e-300)
    idx = np.array([i for i in range(n) if i != b], dtype=int)
    y0 = np.log(np.maximum(a0[idx], 1e-300) / base)
    y0 = np.clip(y0, -50.0, 50.0)

    def _a_from_y(y):
        y = np.clip(y, -50.0, 50.0)
        ex = np.exp(y)
        Z = 1.0 + ex.sum()
        a = np.empty(n)
        a[b] = 1.0 / Z
        a[idx] = ex / Z
        return a

    def fg(y):
        if time.monotonic() > stop:
            raise RuntimeError("timeout")
        a = _a_from_y(y)
        c = _conv_self(a)
        ec = eta * c
        ec -= ec.max()
        w = np.exp(ec)
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones_like(c) / c.size
            ws = 1.0
        w /= ws
        fun = c.max() + np.log(ws) / eta
        G = 2.0 * _valid_conv(w, a)
        dot = float(a.dot(G))
        gy = a[idx] * (G[idx] - dot)
        gy = np.where(np.isfinite(gy), gy, 0.0)
        return float(fun), gy

    best_a = a0.copy()
    best_f = _score(best_a)
    try:
        res = minimize(fg, y0, jac=True, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'maxfun': maxfun,
                                'ftol': 1e-14, 'gtol': 1e-12, 'maxls': 50})
        if hasattr(res, 'x') and res.x is not None:

            sc = _score(a_new)
            if math.isfinite(sc) and sc < best_f:
                best_a, best_f = a_new, sc
    except Exception:
        pass
    return best_a, best_f

def _lp_solve(f, n):
    A = np.zeros((2 * n - 1, n + 1))
    for j in range(n):
        A[j:j + n, j] = f
    A[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res is None or res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x[:n], 0.0, 1.0)
    gs = g.sum()
    if gs < 1e-12:
        return None
    return g / gs

def _lp_loop(f0, stop, max_iter=50000, seed=0):
    f = _normalize(f0)
    n = f.size
    best = f.copy()
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    it = 0
    while it < max_iter and time.monotonic() < stop:
        it += 1
        g = _lp_solve(f, n)
        if g is None:
            break
        cf = _conv_self(f)
        cfs = _conv_cross(f, g)
        cg = _conv_self(g)
        m = 2.0 * (cfs - cf)
        q = cf - 2.0 * cfs + cg
        t_star = _best_t(cf, m, q)
        cands = sorted(set([0.0, 1.0, t_star, 0.02, 0.05, 0.1, 0.2, 0.5,
                            max(0.0, t_star * 0.5), min(1.0, t_star * 1.5)]))
        best_t, best_sc_t = t_star, best_sc
        for tt in cands:
            tt = float(np.clip(tt, 0.0, 1.0))
            h = (1.0 - tt) * f + tt * g
            h = np.maximum(h, 0.0)
            hs = h.sum()
            if hs < 1e-12:
                continue
            h /= hs
            sc = _score(h)
            if sc < best_sc_t:
                best_sc_t = sc
                best_t = tt
        if best_sc_t < best_sc - 1e-14:
            f = (1.0 - best_t) * f + best_t * g
            f = _normalize(f)
            best, best_sc = f.copy(), best_sc_t
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 3:
                f = f + rng.normal(0.0, 0.3 / math.sqrt(n), n)
                f = _normalize(f)
                no_imp = 0
            if no_imp >= 60:
                break
    return best, best_sc

def _coord_polish(a0, stop):
    a = _normalize(a0)
    n = a.size
    best = a.copy()
    best_sc = _score(best)
    rng = np.random.default_rng(777)
    mults = (0.0, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0, 1.3, 1.7, 2.2, 3.0, 4.5)
    for _ in range(4):
        if time.monotonic() > stop:
            break
        improved = False
        for idx in rng.permutation(n):
            if time.monotonic() > stop:
                break
            orig = float(best[idx])
            base = orig if orig > 1e-15 else 1.0 / (4.0 * n)
            bv, bs = orig, best_sc
            for mm in mults:
                v = base * float(mm)
                v = max(0.0, min(1000.0, v))
                best[idx] = v
                s = _score(best)
                if s < bs:
                    bs, bv = s, v
            best[idx] = bv
            if bs < best_sc:
                best_sc, improved = bs, True
        if not improved:
            break
    return best, best_sc
