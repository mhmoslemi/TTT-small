import math
from scipy.fft import rfft, irfft, next_fast_len

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """Build constraint matrix for LP: A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        # A[k, j] = f[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        A[j:j+n, j] = f
    return A

def _lp_solve(f, n, M, time_limit=10.0):
    """Solve max sum(g) s.t. f*g <= M, g >= 0."""
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'mip_rel_gap': 0.1})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs  # normalize to sum=1

def _line_search(f, g_dir, n, best_sc, t_max=0.5, stop_time=None):
    """Find best t in [0, t_max] minimizing score of (1-t)*f + t*g_dir."""
    t_lo, t_hi = 0.0, t_max
    best_t, best_f = 0.0, f.copy()
    best_local = best_sc
    
    for _ in range(20):
        if stop_time and time.monotonic() > stop_time:
            break
        t_mid = 0.5 * (t_lo + t_hi)
        fn = (1.0 - t_mid) * f + t_mid * g_dir
        np.maximum(fn, 0.0, out=fn)
        fs = float(fn.sum())
        if fs < 1e-10:
            t_hi = t_mid
            continue
        sc = _score(fn)
        if sc < best_local - 1e-15:
            best_local = sc
            best_t = t_mid
            best_f = fn.copy()
            t_lo = t_mid
        else:
            t_hi = t_mid
        if t_hi - t_lo < 1e-5:
            break
    
    if best_t < 1e-6:
        return None, None
    return best_t, best_f

def _lp_loop(f0, n, stop_time, seed, max_iters=500):
    """Iterative LP-guided optimization."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs
    
    best_f = f.copy()
    best_sc = _score(f)
    t_cur = 0.3
    stall = 0
    max_stall = 40
    
    for it in range(max_iters):
        if time.monotonic() > stop_time - 1.0:
            break
        
        b = _conv(f)
        M = float(b.max())
        if M < 1e-15:
            break
        
        g_dir = _lp_solve(f, n, M, time_limit=max(2.0, (stop_time - time.monotonic()) * 0.3))
        if g_dir is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            f = f / f.sum() if f.sum() > 1e-10 else best_f.copy()
            continue
        
        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-7:
            stall += 3
            if stall > max_stall:
                break
            continue
        
        res_t, f_new = _line_search(f, g_dir, n, best_sc, t_max=min(0.5, t_cur * 1.2), stop_time=stop_time)
        
        if f_new is not None:
            sc_new = _score(f_new)
            if sc_new < best_sc - 1e-14:
                f = f_new
                best_f = f.copy()
                best_sc = sc_new
                t_cur = min(0.6, res_t * 1.15)
                stall = 0
            else:
                t_cur = max(0.02, t_cur * 0.6)
                stall += 1
        else:
            t_cur = max(0.02, t_cur * 0.5)
            stall += 1
        
        if stall >= max_stall:
            # Perturb and reset
            f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
            fs2 = f.sum()
            f = f / fs2 if fs2 > 1e-10 else best_f.copy()
            t_cur = 0.3
            stall = 0
    
    return best_f, best_sc

def _subgrad_loop(f0, n, stop_time, seed, max_iters=50000):
    """Subgradient optimization with momentum."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs
    
    best_f = f.copy()
    best_sc = _score(f)
    
    lr = 0.005
    momentum_arr = np.zeros(n)
    beta = 0.9
    no_imp = 0
    patience = 500
    
    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break
        
        b = _conv(f)
        M = float(b.max())
        s = float(f.sum())
        
        # Top-k lags
        top_k = min(10, 2 * n - 1)
        idx = np.argpartition(b, -top_k)[-top_k:]
        weights = b[idx].copy()
        weights = weights - weights.min() + 1e-12
        weights /= weights.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for w, k in zip(weights, idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                i = k - j
                if j == i:
                    grad[j] += w * f[j]
                else:
                    grad[j] += w * 2.0 * f[i]
        
        # Score gradient
        scale1 = 2.0 * n / (s * s)
        scale2 = 4.0 * n * M / (s * s * s)
        grad_sc = scale1 * grad - scale2
        
        # Momentum step
        momentum_arr = beta * momentum_arr - lr * grad_sc
        f_new = f + momentum_arr
        f_new = np.maximum(f_new, 0.0)
        fs_new = float(f_new.sum())
        if fs_new < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / fs_new
        
        sc = _score(f_new)
        if sc < best_sc - 1e-15:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.02, lr * 1.005)
        else:
            no_imp += 1
            if no_imp > patience:
                # Adaptive: accept anyway but reduce learning rate
                f = f_new
                lr = max(0.001, lr * 0.95)
                if no_imp > patience * 3:
                    # Restart momentum
                    momentum_arr = np.zeros(n)
                    beta = 0.7
                    lr = 0.005
                    no_imp = 0
    
    return best_f, best_sc

def _multigrad_loop(f0, n, stop_time, seed, max_iters=100000):
    """Multiplicative gradient descent (mirror descent)."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f = np.maximum(f, 1e-10)  # Ensure strictly positive
    fs = float(f.sum())
    f = f / fs
    
    best_f = f.copy()
    best_sc = _score(f)
    
    lr = 0.1
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break
        
        b = _conv(f)
        M = float(b.max())
        s = float(f.sum())
        
        top_k = min(8, 2 * n - 1)
        idx = np.argpartition(b, -top_k)[-top_k:]
        weights = b[idx].copy()
        weights = weights - weights.min() + 1e-12
        weights /= weights.sum()
        
        grad_F = np.zeros(n)
        for w, k in zip(weights, idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                i = k - j
                if j == i:
                    grad_F[j] += w * f[j]
                else:
                    grad_F[j] += w * 2.0 * f[i]
        
        # We want to minimize max_k b[k] / s^2
        # Multiplicative update: f_new[j] = f[j] * exp(-lr * g[j] / f[j])
        # where g[j] is the gradient of the objective
        # For the multiplicative update to work well, we use:
        # f_new[j] = f[j] * (grad_F[j] / f[j])^{-lr}  (for gradient descent)
        # Or more stably: f_new[j] = f[j] * exp(-lr * (grad_F[j] - mean) / f[j])
        
        # Simpler: use the ratio approach
        # f_new[j] proportional to f[j] * (target / current)^lr
        # where target is the "average" contribution
        
        # Actually, let's use a simpler multiplicative update:
        # f_new[j] = f[j] * (1 - lr * grad_sc[j]) where grad_sc is normalized
        # This is just gradient descent with a multiplicative constraint
        
        scale1 = 2.0 * n / (s * s)
        scale2 = 4.0 * n * M / (s * s * s)
        grad_sc = scale1 * grad_F - scale2
        
        # Normalize gradient for stability
        grad_sc = grad_sc - grad_sc.mean()  # Center it
        grad_max = np.abs(grad_sc).max()
        if grad_max > 1e-15:
            grad_sc = grad_sc / grad_max
        
        f_new = f * np.exp(-lr * grad_sc)
        f_new = np.maximum(f_new, 1e-15)
        fs_new = float(f_new.sum())
        f_new = f_new / fs_new
        
        sc = _score(f_new)
        if sc < best_sc - 1e-15:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.5, lr * 1.01)
        else:
            no_imp += 1
            lr = max(0.01, lr * 0.98)
            if no_imp > 2000:
                break
    
    return best_f, best_sc

def _coord_polish(f0, n, stop_time, seed):
    """Coordinate-wise optimization focusing on active lags."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    f = f / fs
    best_sc = _score(f)
    
    no_full_pass = 0
    
    while time.monotonic() < stop_time - 2.0 and no_full_pass < 5:
        improved = False
        b = _conv(f)
        M = float(b.max())
        
        # Find coordinates involved in top lags
        top_k = min(10, 2 * n - 1)
        idx = np.argpartition(b, -top_k)[-top_k:]
        
        active_coords = set()
        for k in idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                active_coords.add(j)
        
        active_coords = list(active_coords)
        rng.shuffle(active_coords)
        
        for i in active_coords:
            if time.monotonic() > stop_time - 2.0:
                return f, best_sc
            orig = f[i]
            if orig < 1e-12:
                continue
            for factor in (0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0):
                f[i] = orig * factor
                sc = _score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                else:
                    f[i] = orig
                    break  # Try next factor from a different direction
        
        # Also try some random coordinates
        for _ in range(50):
            if time.monotonic() > stop_time - 2.0:
                return f, best_sc
            i = int(rng.integers(n))
            orig = f[i]
            if orig < 1e-12:
                continue
            for factor in (0.8, 0.9, 1.1, 1.2):
                f[i] = orig * factor
                sc = _score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                    break
            else:
                f[i] = orig
        
        if not improved:
            no_full_pass += 1
        else:
            no_full_pass = 0
    
    return f, best_sc
