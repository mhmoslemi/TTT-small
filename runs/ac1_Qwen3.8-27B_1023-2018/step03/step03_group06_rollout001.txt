def _conv2(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()

        return np.zeros(0, dtype=np.float64)
    m = n1 + n2 - 1
    L = _fft_len(m)
    c = irfft(rfft(a, L) * rfft(b, L), L)[:m]

def _lbfgs_refine(a0, stop: float, etas=(20.0, 100.0, 500.0, 2000.0), maxfun_per_stage: int = None):
    a = _norm(a0)
    n = a.size
    if n < 2:
        return a, 2.0 * float(n) * _maxc(a)

    if maxfun_per_stage is None:
        if n <= 384:
            maxfun_per_stage = 550
        elif n <= 1024:
            maxfun_per_stage = 320
        else:
            maxfun_per_stage = 160

    best = a.copy()
    best_M = float(_maxc(a))

    for eta in etas:
        now = time.monotonic()
        if now > stop - 8.0:
            break

        cur_M = float(_maxc(a))
        base_M = max(cur_M, 1e-12)
        beta = float(eta) / base_M

        ref = int(np.argmax(a))
        idx = np.array([k for k in range(n) if k != ref], dtype=int)
        if idx.size == 0:
            break

        y0 = np.log(np.maximum(a[idx], 1e-300) / float(a[ref]))
        y0 = np.where(np.isfinite(y0), y0, 0.0)
        y0 = np.clip(y0, -60.0, 60.0)

        def fg(y, _beta=beta, _idx=idx, _n=n, _ref=ref):
            y = np.asarray(y, dtype=np.float64).ravel()
            if y.size and not np.all(np.isfinite(y)):
                return 1e100, np.zeros_like(y)

            an = _softmax_from_y(y, _idx, _n, _ref)
            c = _conv_same(an)
            if c.size == 0:
                return 1e100, np.zeros_like(y)
            maxc = float(np.max(c))
            if not math.isfinite(maxc):
                return 1e100, np.zeros_like(y)

            x = _beta * (c - maxc)
            x = np.clip(x, -700.0, 0.0)
            w = np.exp(x)
            Z = float(w.sum())
            if not math.isfinite(Z) or Z <= 1e-300:
                p = np.ones_like(c, dtype=np.float64) / float(c.size)
                fun = maxc
            else:
                p = w / Z
                fun = maxc + math.log(Z) / _beta

            v = _valid_conv(p, an)
            if v.size != _n:
                v = np.zeros(_n, dtype=np.float64)
            g = 2.0 * v
            dot = float(np.dot(an, g))
            grad = an[_idx] * (g[_idx] - dot)
            grad = np.where(np.isfinite(grad), grad, 0.0)
            if not math.isfinite(fun):
                return 1e100, np.zeros_like(y)
            return float(fun), grad

        mf = int(max(40, min(maxfun_per_stage, 2500)))
        try:
            res = minimize(
                fg,
                y0,
                jac=True,
                method="L-BFGS-B",
                bounds=[(-60.0, 60.0)] * idx.size,
                options={
                    "maxiter": int(min(900, mf + 300)),
                    "maxfun": mf,
                    "ftol": 1e-13,
                    "gtol": 1e-10,
                    "maxls": 50,
                },
            )
        except Exception:
            break

        if res is None or getattr(res, "x", None) is None:
            break

        a_new = _softmax_from_y(np.asarray(res.x, dtype=np.float64), idx, n, ref)
        M_new = _maxc(a_new)

        if math.isfinite(M_new) and M_new < best_M - 1e-14:
            best_M = float(M_new)
            best = a_new.copy()
            a = a_new
        elif math.isfinite(M_new) and M_new <= cur_M + 1e-12:
            a = a_new
        else:
            a = best.copy()

    return best, 2.0 * float(n) * float(best_M)

            arr, _ = _lp_loop(best_arr, hard_deadline - 8.0, seed=seed + 7)
            consider(arr)
        if hard_deadline - time.monotonic() > 15.0:
            arr, _ = _lbfgs_refine(best_arr, hard_deadline - 5.0, etas=(30.0, 150.0), maxfun_per_stage=220)
            consider(arr)
        return finalize()

    # Main refinement at the parent/base length.
    base_alloc = min(650.0, max(160.0, 0.50 * budget))
    base_stop = min(deadline, t0 + base_alloc)

    if base_alloc > 90.0:
        lp_stop = min(base_stop - 0.32 * base_alloc, t0 + 0.50 * base_alloc)
        if n_init <= 1024 and lp_stop - time.monotonic() > 25.0:
            arr, _ = _lp_loop(best_arr, lp_stop, seed=seed + 11)
            consider(arr)

        lbfgs_stop = min(base_stop - 0.08 * base_alloc, t0 + 0.80 * base_alloc)
        if lbfgs_stop - time.monotonic() > 15.0:
            if base_alloc >= 450.0:
                etas = (5.0, 20.0, 80.0, 300.0, 1200.0)
            else:
                etas = (20.0, 100.0, 400.0)
            arr, _ = _lbfgs_refine(best_arr, lbfgs_stop, etas=etas)
            consider(arr)

        if base_stop - time.monotonic() > 8.0:
            max_pass = 3 if best_arr.size <= 512 else 1
            arr, _ = _coordinate_polish(best_arr, base_stop - 3.0, max_passes=max_pass, seed=seed + 12)
            consider(arr)

    # Explore nearby / larger resolutions if there remains budget and we are not essentially at target.
    now = time.monotonic()
    other_stop = deadline - 70.0
    if best_score > 1.5026 and now < other_stop - 130.0:
        cur = int(best_arr.size)
        if cur <= 320:
            candidates = [512, 384, 1024, 768]
        elif cur <= 640:
            candidates = [1024, 768, 512, 384]
        elif cur <= 1200:
            candidates = [768, 512, 384]
        else:
            candidates = [1024, 768, 512]

        candidates = [L for L in candidates if L != cur and 64 <= L <= 1280]
        rem = max(0.0, other_stop - now)
        fr = [0.52, 0.30, 0.13, 0.05]

        for k, L in enumerate(candidates[:4]):
            now = time.monotonic()
            if now > other_stop - 90.0:
                break
            if k >= len(fr):
                break
            frac = fr[k]
            alloc = rem * frac
            alloc = min(alloc, 320.0 if L <= 600 else 230.0)
            if alloc < 70.0:
                alloc = max(35.0, rem * 0.10)
            stopL = min(other_stop - 25.0, time.monotonic() + max(35.0, alloc))
            if time.monotonic() >= stopL - 12.0:
                break

            init = _interp_seq(best_arr, L)
            consider(init)
            aL = init

            if L <= 1024 and stopL - time.monotonic() > max(30.0, 0.30 * alloc):
                aL, _ = _lp_loop(aL, stopL - 0.35 * alloc, seed=seed + 100 + k)
                consider(aL)

            if stopL - time.monotonic() > 20.0:
                aL, _ = _lbfgs_refine(
                    aL,
                    stopL - 12.0,
                    etas=(20.0, 100.0, 500.0),
                    maxfun_per_stage=300 if L <= 512 else 210,
                )
                consider(aL)

            if stopL - time.monotonic() > 8.0:
                max_pass = 2 if L <= 512 else 1
                aL, _ = _coordinate_polish(aL, stopL - 3.0, max_passes=max_pass, seed=seed + 200 + k)
                consider(aL)

    # Final polish on the best sequence found so far.
    fin_stop = deadline - 20.0
    now = time.monotonic()
    if now < fin_stop - 25.0 and best_arr.size <= 1280:
        stopF = min(fin_stop - 8.0, time.monotonic() + min(90.0, max(20.0, 0.10 * budget)))
        if stopF - time.monotonic() > 18.0:
            arr, _ = _lbfgs_refine(
                best_arr,
                stopF,
                etas=(80.0, 400.0, 1800.0),
                maxfun_per_stage=350 if best_arr.size <= 512 else 220,
            )
            consider(arr)

    if fin_stop - time.monotonic() > 12.0:
        max_pass = 2 if best_arr.size <= 512 else 1
        arr, _ = _coordinate_polish(best_arr, fin_stop - 4.0, max_passes=max_pass, seed=seed + 987654321)
        consider(arr)

    return finalize()
