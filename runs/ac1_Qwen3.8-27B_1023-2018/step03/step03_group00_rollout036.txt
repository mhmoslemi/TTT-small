def _subgradient_step(f, n, step_size):
    """Take a subgradient step on the normalized problem"""
    S = f.sum()
    if S < 1e-10:
        return f
    f_norm = f / S  # Normalize to sum = 1
    
    b = _conv_full(f_norm)
    M = b.max()
    K_star = np.where(b >= M * (1 - 1e-10))[0]
    
    # Subgradient of max(conv(f,f)) at f
    # d h_k / d f_i = 2 * f_{k-i} (with boundary convention)
    grad = np.zeros(n)
    for k in K_star:
        for i in range(n):
            idx = k - i
            if 0 <= idx < n:
                grad[i] += 2.0 * f_norm[idx]
    grad /= len(K_star)
    
    # Subtract mean to maintain sum constraint
    grad = grad - grad.mean()
    
    # Subgradient descent
    f_new = f_norm - step_size * grad
    f_new = np.clip(f_new, 0, None)
    s_new = f_new.sum()
    if s_new > 1e-10:
        f_new = f_new / s_new  # Renormalize
    else:

    return f_new

def _lp_direction(f, n, M, timeout=2.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0"""
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:

                      method='highs',

        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)

        b = _conv_full(f)
        M = float(b.max())

        f_sum = f.sum()

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:

    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    """Build Toeplitz matrix A such that A @ g = conv(f, g)."""
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    mask = (idx >= 0) & (idx < n)
    A[mask] = f[idx[mask]]
    return A


def _lp_direction(f, n, M, timeout=2.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _subgradient_escape(f, n, rng):
    """Take a subgradient step to escape from LP stagnation."""
    f_sum = f.sum()
    if f_sum < 1e-10:
        return None
    f_norm = f / f_sum
    b = _conv_full(f_norm)
    M = b.max()
    if M < 1e-18:
        return None
    K_star = np.where(b >= M * (1 - 1e-8))[0]
    if len(K_star) == 0:
        return None
    grad = np.zeros(n)
    for k in K_star:
        ki = int(k)
        for i in range(n):
            idx = ki - i
            if 0 <= idx < n:
                grad[i] += 2.0 * f_norm[idx]
    grad /= float(len(K_star))
    grad -= grad.mean()
    step = rng.uniform(0.005, 0.03)
    f_new = f - step * f_sum * grad
    f_new = np.clip(f_new, 0, None)
    s_new = f_new.sum()
    if s_new < 1e-10:
        return None
    f_new = f_new * (f_sum / s_new)
    return f_new

            best_val_sc = best_sc

                    best_val_sc = sc

            if best_val_sc < best_sc - 1e-14:

        # Pairs

            i = int(rng.integers(n))
