    n: length
    b: current autocorrelation (length 2n-1)
    S: sum of a
    """

                    b[k] += delta * a[k-i]
                improved = True
        
        if not improved:
            break
    
    return a, b, S

    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

    b = _conv_full(a)

    if not math.isfinite(m) or m < 0:

    return 2.0 * n * m / (s * s)
