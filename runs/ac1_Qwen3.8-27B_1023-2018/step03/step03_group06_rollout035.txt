        _fft_cache[m] = v
    return v

def _conv(a):
    """Linear autocorrelation of a via FFT. Returns array of length 2n-1."""
    n = a.size
    if n == 0: return np.array([], dtype=np.float64)
    L = _fft_len(2*n-1)
    fa = rfft(a, L)
    c = irfft(fa*fa, L)[:2*n-1]
    return np.maximum(c, 0)

def _conv_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n1, n2 = a.size, b.size
    if n1 == 0 or n2 == 0: return np.array([], dtype=np.float64)
    L = _fft_len(n1+n2-1)
    c = irfft(rfft(a, L)*rfft(b, L), L)[:n1+n2-1]
    return np.maximum(c, 0)

def _norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0), 0, 1000)
    s = a.sum()
    if s < 1e-12: return np.ones(a.size)/a.size
    return a/s

    """Score assuming sum=1: 2n*max(conv)"""
    a = _norm(a)
    n = a.size
    return 2*n*_conv(a).max()

    y = np.log(np.clip(a, 1e-30, None))
    y[0] = 0.0  # fix reference

        if stop is not None and time.monotonic() > stop:
            raise _Timeout
        y = np.zeros(n)
        y[1:] = yf
        y = np.clip(y, -50, 50)

    try:

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize

class _Timeout(Exception):
    pass

_fc = {}
def _fl(m):
    m = int(max(1, m))
    if m not in _fc:
        _fc[m] = int(next_fast_len(m))
    return _fc[m]

def _conv(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = _fl(2 * n - 1)
    fa = rfft(a, L)
    return np.maximum(irfft(fa * fa, L)[:2 * n - 1], 0.0)

def _convab(a, b):
    n1, n2 = a.size, b.size
    if n1 == 0 or n2 == 0:
        return np.array([], dtype=np.float64)
    L = _fl(n1 + n2 - 1)
    return np.maximum(irfft(rfft(a, L) * rfft(b, L), L)[:n1 + n2 - 1], 0.0)

def _norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0), 0.0, 1000.0)
    s = a.sum()
    if not math.isfinite(s) or s < 1e-15:
        n = a.size if a.size > 0 else 32
        return np.ones(n, dtype=np.float64) / n
    return a / s

def _sc(a):
    a = _norm(a)
    n = a.size
    return 2.0 * n * float(_conv(a).max())

def _eval(a):
    try:
        f = globals().get('evaluate_sequence')
        if f is None:
            return math.inf
        return float(f([float(x) for x in np.asarray(a, dtype=np.float64).ravel()]))
    except Exception:
        return math.inf

def _parent_seq():
    try:
        h = globals()['height_sequence_1']
        if h is None:
            return None
        p = np.asarray(h, dtype=np.float64).ravel()
        if p.size == 0:
            return None
        p = np.clip(np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0), 0.0, 1000.0)
        if p.sum() < 1e-12:
            return None
        return p / p.sum()
    except Exception:
        return None

def _lp_round(f, stop, min_remain=3.0, max_iters=5000):
    n = f.size
    best = f.copy()
    best_sc = 2.0 * n * float(_conv(f).max())
    iters = 0

    A_ub = np.zeros((2 * n - 1, n + 1), dtype=np.float64)
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0
    b_ub = np.zeros(2 * n - 1, dtype=np.float64)

    no_imp = 0
    rng_l = np.random.default_rng(12345)

    while time.monotonic() < stop - min_remain and iters < max_iters:
        iters += 1
        # Build LP matrix using current f with small epsilon
        eps = 1e-10
        fp = np.maximum(f, eps)
        fp = fp / fp.sum()
        for j in range(n):
            A_ub[j:j + n, j] = fp
        A_ub[:, n] = -1.0

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'time_limit': 5.0, 'mip_rel_gap': 0.0})
        except Exception:
            break

        if res is None or getattr(res, 'status', 1) != 0 or res.x is None:
            break

        g = np.clip(res.x[:n], 0.0, 1.0)
        gs = g.sum()
        if gs < 1e-12:
            break
        g = g / gs

        # Line search: golden section on actual score
        phi = (1.0 + math.sqrt(5.0)) / 2.0
        a_lo, a_hi = 0.0, 1.0
        a1 = a_hi - (a_hi - a_lo) / phi
        a2 = a_lo + (a_hi - a_lo) / phi

        def score_t(t):
            x = (1.0 - t) * f + t * g
            # x has sum=1, non-negative (convex combo of non-negative, sum-1 vectors)
            return 2.0 * n * float(_conv(np.maximum(x, 0.0)).max())

        f1 = score_t(a1)
        f2 = score_t(a2)
        for _ in range(25):
            if time.monotonic() > stop - min_remain:
                break
            if f1 < f2:
                a_hi = a2
                a2 = a1
                f2 = f1
                a1 = a_hi - (a_hi - a_lo) / phi
                f1 = score_t(a1)
            else:
                a_lo = a1
                a1 = a2
                f1 = f2
                a2 = a_lo + (a_hi - a_lo) / phi
                f2 = score_t(a2)
        t_best = (a_lo + a_hi) / 2.0
        new_f = (1.0 - t_best) * f + t_best * g
        new_f = np.maximum(new_f, 0.0)
        sc_new = 2.0 * n * float(_conv(new_f).max())

        if sc_new < best_sc - 1e-13:
            best = new_f.copy()
            best_sc = sc_new
            f = best
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 6:
                # Perturb to escape
                scale = 0.02 * max(1.0 / n, f.max()) * (1.0 + 0.5 * no_imp)
                f = f + rng_l.normal(0, scale, n)
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs < 1e-15:
                    f = best.copy()
                else:
                    f = f / fs
                sc_f = 2.0 * n * float(_conv(f).max())
                if sc_f < best_sc:
                    best = f.copy()
                    best_sc = sc_f
                no_imp = 0

    return best

def _lbfgs_smooth(a_init, eta, n, max_iter=300, stop=None):
    a = _norm(a_init)
    # Parameterize: y[0]=0, optimize y[1:]
    a_safe = np.maximum(a, 1e-30)
    y0 = np.log(a_safe / a_safe[0])
    y0[0] = 0.0

    def fg(yf):
        if stop is not None and time.monotonic() > stop:
            raise _Timeout
        yf = np.asarray(yf, dtype=np.float64)
        y = np.zeros(n)
        y[1:] = yf
        y = np.clip(y, -40.0, 40.0)
        a = np.exp(y - y.max())
        a /= a.sum()
        c = _conv(a)
        c_max = c.max()
        ec = np.exp(eta * (c - c_max))
        Z = ec.sum()
        if Z < 1e-300:
            return 1e100, np.zeros(n - 1)
        w = ec / Z
        H = c_max + math.log(Z) / eta
        c_w = float(w @ c)
        # v_j = sum_m w[j+m] * a[m], j=0..n-1
        v = np.array([float(np.dot(w[j:j + n], a)) for j in range(n)])
        grad_full = 2.0 * a * (v - c_w)
        grad = grad_full[1:]
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(H), grad

    try:
        res = minimize(fg, np.zeros(n - 1), jac=True, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'maxfun': max_iter * 4,
                                'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 50})
    except _Timeout:
        return a
    except Exception:
        return a

    if hasattr(res, 'x') and res.x is not None:
        y = np.zeros(n)
        y[1:] = res.x
        a_new = np.exp(y - y.max())
        a_new /= a_new.sum()
        if np.all(np.isfinite(a_new)) and a_new.sum() > 1e-15:
            return a_new
    return a

def _coord_descent(a0, stop, seed=77, max_passes=5):
    a = _norm(a0)
    n = a.size
    best = a.copy()
    best_sc = 2.0 * n * float(_conv(best).max())
    rng = np.random.default_rng(seed)

    mults = (0.0, 0.1, 0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0, 5.0)
    default_scale = 1.0 / (2.0 * n)

    for _ in range(max_passes):
        if time.monotonic() > stop:
            break
        improved = False
        order = rng.permutation(n)
        for idx in order:
            if time.monotonic() > stop:
                break
            orig = float(best[idx])
            base = orig if orig > 1e-15 else default_scale
            best_val = orig
            best_val_sc = best_sc
            vals = {orig}
            for m in mults:
                v = base * float(m)
                if 0.0 <= v <= 1000.0:
                    vals.add(v)
            for v in sorted(vals):
                best[idx] = v
                sc = 2.0 * n * float(_conv(best).max())
                if sc < best_val_sc:
                    best_val_sc = sc
                    best_val = v
            best[idx] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True
        if not improved:
            break
    return best

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1300.0
    hard_deadline = t0 + budget
    deadline = hard_deadline - 20.0

    parent = _parent_seq()
    if parent is None:
        parent = np.ones(256, dtype=np.float64) / 256.0
    n = parent.size

    best = parent.copy()
    best_sc = _eval(best)
    if not math.isfinite(best_sc):
        best_sc = _sc(best)
    rng = np.random.default_rng(seed)

    def try_update(a):
        nonlocal best, best_sc
        a = _norm(a)
        if a.size != best.size:
            return
        sc_int = 2.0 * a.size * float(_conv(a).max())
        if not math.isfinite(sc_int) or sc_int > best_sc:
            return
        sc = _eval(a)
        if math.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best = a.copy()

    # Phase 1: LP iterations from parent
    p1_stop = min(deadline - 300.0, t0 + 0.15 * budget)
    if p1_stop > t0 + 20.0:
        arr = _lp_round(best, p1_stop)
        try_update(arr)

    # Phase 2: L-BFGS with moderate eta
    p2_stop = min(deadline - 250.0, t0 + 0.35 * budget)
    if p2_stop > time.monotonic() + 15.0:
        for eta in (20.0, 60.0, 150.0):
            if time.monotonic() > p2_stop - 10.0:
                break
            sub_stop = min(p2_stop - 5.0, time.monotonic() + 60.0)
            if sub_stop - time.monotonic() < 10.0:
                break
            arr = _lbfgs_smooth(best, eta, n, max_iter=250, stop=sub_stop)
            try_update(arr)
            if best_sc < 1.503:
                break

    # Phase 3: More LP from improved point
    p3_stop = min(deadline - 200.0, t0 + 0.55 * budget)
    if p3_stop > time.monotonic() + 15.0:
        arr = _lp_round(best, p3_stop, max_iters=8000)
        try_update(arr)

    # Phase 4: L-BFGS with high eta
    p4_stop = min(deadline - 120.0, t0 + 0.75 * budget)
    if p4_stop > time.monotonic() + 15.0:
        for eta in (300.0, 600.0, 1200.0):
            if time.monotonic() > p4_stop - 10.0:
                break
            sub_stop = min(p4_stop - 5.0, time.monotonic() + 80.0)
            if sub_stop - time.monotonic() < 10.0:
                break
            arr = _lbfgs_smooth(best, eta, n, max_iter=300, stop=sub_stop)
            try_update(arr)

    # Phase 5: LP again
    p5_stop = min(deadline - 60.0, t0 + 0.88 * budget)
    if p5_stop > time.monotonic() + 10.0:
        arr = _lp_round(best, p5_stop, max_iters=6000)
        try_update(arr)

    # Phase 6: Final L-BFGS + coordinate descent
    p6_stop = deadline - 10.0
    if p6_stop > time.monotonic() + 8.0:
        arr = _lbfgs_smooth(best, 800.0, n, max_iter=200, stop=p6_stop - 5.0)
        try_update(arr)
    if p6_stop > time.monotonic() + 5.0:
        arr = _coord_descent(best, p6_stop, seed=999)
        try_update(arr)

    # Final verification
    final = np.clip(np.asarray(best, dtype=np.float64).ravel(), 0.0, 1000.0)
    if final.size == 0 or final.sum() < 0.01:
        final = np.ones(32, dtype=np.float64) / 32.0
    final_sc = _eval(final)
    if not math.isfinite(final_sc):
        fallback = np.ones(n, dtype=np.float64) / n
        final = fallback
    return [float(x) for x in final]
