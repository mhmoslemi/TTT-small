```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize_scalar

def _fft_conv(a):
    n = len(a)
    if n <= 200:
        return np.convolve(a, a)
    m = next_fast_len(2*n-1)
    F = rfft(a, m)
    c = irfft(F*F, m)[:2*n-1]
    np.maximum(c, 0, out=c)
    return c

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    return 2.0*n*b.max()/(s*s)
```

```python
            best_idx = np.argmin(scores)
            if scores[best_idx] < best_score - 1e-14:
                # Refine with minimize_scalar
                # Interval around best
                lo = max(0, xs[max(0, best_idx-2)])
                hi = min(x_max, xs[min(G-1, best_idx+2)])
                
                def obj(x):

                    sn = Sr + x
                    if sn < 0.01:
                        return math.inf
                    nc = conv + 2*w*a
                    if 2*i < 2*n-1:
                        nc = nc.copy()
                        nc[2*i] += w*w
                    return 2.0*n*nc.max()/(sn*sn)
                
                try:
                    res = minimize_scalar(obj, bounds=(lo, hi), method='bounded', options={'xatol': 1e-10})
                    if res.fun < scores[best_idx]:
                        best_idx = -1  # flag for refined value
                        xs[best_idx % G] = res.x  # not quite right
                except:
                    pass

                x_new = xs[best_idx]
                w = x_new - v
                f[i] = x_new
                S += w
                conv[i:i+n] += 2*w*a[i:i+n]
                if 2*i < 2*n-1:
                    conv[2*i] += w*w
                best_score = scores[best_idx]
                improved = True
```

Let me also think about the LP move more carefully.

The LP: maximize sum(g) subject to:
- (f*g)(k) <= M for k = 0, ..., 2n-2
- g >= 0

where M = max(f*f) = conv.max()

The constraint (f*g)(k) = sum_j f(|k-j|) * g(j)... no wait.

(f*g)(k) = sum_{j=0}^{n-1} f(k-j) * g(j) where f(m) = 0 if m not in [0,n-1].

So the constraint matrix is A[k,j] = f(k-j) if 0 <= k-j <= n-1, else 0.

This means row k of A is: f(k), f(k-1), ..., f(0), 0, ..., 0, ..., 0, f(0), f(1), ..., f(k-n) (shifted).

Actually, A[k,j] = f(k-j) for j in [max(0, k-n+1), min(k, n-1)].

So A is a (2n-1) x n matrix where each row is a shifted version of f. This is a Toeplitz-like (Hankel+Toeplitz) structure.

For n=384: A is 767 x 384, with about n nonzeros per row, so about 767*384/2 ≈ 147,000 nonzeros. Very sparse!

I can use a sparse LP solver. HiGHS handles sparse problems well.

```python
from scipy.sparse import csr_matrix

def _build_A(f):
    n = len(f)
    # A[k,j] = f(k-j) if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(k, n-1)
        for j in range(j_start, j_end+1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
```

For n=384, this has about sum_k min(j_end - j_start + 1) = sum_k (min(k,n-1) - max(0,k-n+1) + 1) nonzeros.

For k=0 to n-1: k+1 nonzeros. For k=n to 2n-2: 2n-1-k nonzeros.
Total: n(n+1)/2 + (n-1)n/2 = n^2. So about 384^2 = 147,456 nonzeros. Fine.

Now, the LP:
- c = -ones(n) (maximize sum = minimize -sum)
- A_ub = A (sparse)
- b_ub = M * ones(2n-1)
- bounds: [(0, None)] * n

This should be solvable in a fraction of a second by HiGHS.

After getting g, normalize: g = g / g.sum()
Then do a Frank-Wolfe step: f_new = (1-t)*f + t*g for some t.

The optimal t can be found by 1D search on [0,1] minimizing score((1-t)f + tg).

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix


def _fft_conv(a):
    n = len(a)
    if n <= 256:
        return np.convolve(a, a)
    m = next_fast_len(2 * n - 1)
    F = rfft(a, m)
    c = irfft(F * F, m)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(b.max())
    if m < 0:
        m = 0.0
    return 2.0 * n * m / (s * s)


def _exact_score(a):
    try:
        return float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        return math.inf


def _build_sparse_A(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j0 = max(0, k - n + 1)
        j1 = min(k, n - 1)
        for j in range(j0, j1 + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_direction(f, M, time_limit=10.0):
    n = len(f)
    A = _build_sparse_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit, "dual_feasibility_tolerance": 1e-7},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum


def _cd_sweep(f, stop_time, max_sweeps=500, grid_size=60):
    n = len(f)
    f = f.copy()
    S = float(f.sum())
    if S < 0.01:
        return f, math.inf
    conv = _fft_conv(f).copy()
    best_score = 2.0 * n * conv.max() / (S * S)
    two_n_minus_1 = 2 * n - 1

    for sweep in range(max_sweps):
        if time.monotonic() > stop_time:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > stop_time:
                break
            v = f[i]
            Sr = S - v

            # Influence: a[k] = f(k-i) for k-i in [0, n-1]
            a = np.zeros(two_n_minus_1)
            a[i:i + n] = f

            # Determine search range for x (new value of f[i])
            x_max = max(4.0 * v + 0.5, 1.0) if v > 1e-8 else 2.0
            x_max = min(x_max, 300.0)

            # Grid evaluation
            xs = np.linspace(0.0, x_max, grid_size)
            ws = xs - v
            S_new = Sr + xs

            # New conv [g, k] = conv[k] + 2*w[g]*a[k] + w[g]^2 * delta(k, 2i)
            delta_2i = np.zeros(two_n_minus_1)
            if 2 * i < two_n_minus_1:
                delta_2i[2 * i] = 1.0

            new_convs = conv[None, :] + 2.0 * ws[:, None] * a[None, :] + (ws * ws)[:, None] * delta_2i[None, :]
            maxes = new_convs.max(axis=1)

            valid = S_new > 0.01
            scores = np.full(grid_size, 1e30)
            if valid.any():
                scores[valid] = 2.0 * n * maxes[valid] / (S_new[valid] * S_new[valid])

            best_idx = int(np.argmin(scores))
            if scores[best_idx] < best_score - 1e-13:
                # Refine locally
                lo = max(0.0, xs[max(0, best_idx - 2)])
                hi = min(x_max, xs[min(grid_size - 1, best_idx + 2)])
                if hi - lo > 1e-12:
                    def obj_1d(x):
                        w = x - v
                        sn = Sr + x
                        if sn < 0.01:
                            return 1e30
                        nc = conv + 2.0 * w * a
                        if 2 * i < two_n_minus_1:
                            nc = nc.copy()
                            nc[2 * i] += w * w
                        return 2.0 * n * nc.max() / (sn * sn)
                    try:
                        res_m = minimize_scalar(obj_1d, bounds=(lo, hi), method='bounded',
                                                options={'xatol': 1e-12})
                        if res_m.fun < scores[best_idx]:
                            xs[best_idx] = res_m.x
                            scores[best_idx] = res_m.fun
                    except Exception:
                        pass

                x_new = xs[best_idx]
                w = x_new - v
                f[i] = x_new
                S += w
                conv[i:i + n] += 2.0 * w * a[i:i + n]
                if 2 * i < two_n_minus_1:
                    conv[2 * i] += w * w
                best_score = scores[best_idx]
                improved = True

        if not improved:
            break

    f = np.clip(f, 0.0, 1000.0)
    final_score = _score(f)
    return f, final_score


def _lp_move(f, stop_time, max_lp_iters=30):
    n = len(f)
    f = f.copy()
    f = np.clip(f, 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, _score(f)
    f = f / S  # normalize to sum=1
    conv = _fft_conv(f).copy()
    M = float(conv.max())
    best_score = 2.0 * n * M

    for it in range(max_lp_iters):
        if time.monotonic() > stop_time:
            break
        M = float(conv.max())
        if M < 1e-15:
            break

        g = _lp_direction(f, M, time_limit=5.0)
        if g is None:
            break

        dist = float(np.linalg.norm(g - f))
        if dist < 1e-8:
            break

        # Line search for optimal step size
        best_t = 0.0
        best_sc = best_score
        for t in np.linspace(0.0, 1.0, 20):
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.clip(f_trial, 0, None)
            st = float(f_trial.sum())
            if st < 0.01:
                continue
            f_trial = f_trial / st
            ct = _fft_conv(f_trial)
            sc = 2.0 * n * float(ct.max())
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_t = t

        if best_t < 0.01:
            break

        f = (1.0 - best_t) * f + best_t * g
        f = np.clip(f, 0, None)
        S = float(f.sum())
        if S < 0.01:
            break
        f = f / S
        conv = _fft_conv(f).copy()
        M = float(conv.max())
        best_score = 2.0 * n * M

    f = np.clip(f, 0.0, 1000.0)
    final_score = _score(f)
    return f, final_score


def _random_perturb(f, rng, scale=0.1):
    n = len(f)
    f_new = f.copy()
    f_new += rng.normal(0, scale, n)
    f_new = np.clip(f_new, 0, None)
    s = float(f_new.sum())
    if s < 0.01:
        f_new = f.copy()
    else:
        f_new = f_new / s
    return f_new

            return

    phase2_end = t0 + budget * 0.60
```
