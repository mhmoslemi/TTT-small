            return math.inf
        c = conv_fn(a)
        m = float(c.max())
        return float(2.0 * len(a) * m) / (s*s)

        fb_sc = eval_local(parent, conv_p)
    except:

    def update(seq):
        nonlocal best_seq, best_sc
        try:

        except:
            sc = math.inf

            best_seq = np.array(seq, dtype=np.float64).copy()
            best_sc = sc
    
    # --- LP step ---
    def lp_dir(f, n, time_lim):
        conv_fn, _ = make_conv(n)
        c = conv_fn(f)
        M = float(c.max())
        if M < 1e-15:
            return None
        # Build convolution matrix A: A[k,j] = f[j] if 0<=k-j<=n-1
        A = np.zeros((2*n-1, n))
        for j in range(n):
            A[j:j+n, j] = f
        cvec = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        try:
            res = linprog(cvec, A_ub=A, b_ub=b_ub,
                          bounds=[(0,None)]*n,
                          method='highs',
                          options={'time_limit': time_lim, 'presolve': True})
        except:
            return None
        if res.status != 0 or res.x is None:
            return None
        g = np.maximum(res.x, 0)
        gs = g.sum()
        if gs < 1e-15:
            return None
        return g / gs
    
    # --- LP Optimization ---
    def lp_opt(f0, n, stop, seed_val, max_it=1000):
        conv_fn, _ = make_conv(n)
        f = np.clip(f0.copy(), 0, None)
        fs = f.sum()
        if fs < 1e-10:
            f[:] = 1.0/n; fs = 1.0
        f /= fs
        best_f = f.copy()
        best_sc = eval_local(f, conv_fn)
        rng = np.random.default_rng(seed_val)
        t_step = 0.25
        stall = 0
        max_stall = 60
        
        for it in range(max_it):
            remain = stop - time.monotonic()
            if remain < 3: break
            
            g = lp_dir(f, n, min(6.0, remain*0.4))
            if g is None:
                stall += 3
                if stall > max_stall: break
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                fs = f.sum()
                f = f/fs if fs > 0 else best_f.copy()
                continue
            
            d = float(np.linalg.norm(g - f))
            if d < 1e-8:
                stall += 2
                if stall > max_stall: break
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
                fs = f.sum()
                f = f/fs if fs > 0 else best_f.copy()
                continue
            
            improved = False
            for _ in range(8):
                fn = (1-t_step)*f + t_step*g
                fn = np.clip(fn, 0, None)
                fns = fn.sum()
                if fns < 1e-10:
                    t_step = max(0.02, t_step*0.5)
                    continue
                sc = eval_local(fn, conv_fn)
                if sc < best_sc - 1e-14:
                    f = fn
                    best_f = f.copy()
                    best_sc = sc
                    improved = True
                    t_step = min(0.7, t_step*1.04)
                    stall = 0
                    break
                else:
                    t_step = max(0.02, t_step*0.5)
            
            if not improved:
                stall += 1
                if stall >= max_stall:
                    f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
                    fs = f.sum()
                    f = f/fs if fs > 0 else best_f.copy()
                    t_step = 0.2
                    stall = max(0, stall - 20)
        
        return best_f, best_sc
    
    # --- Coordinate Descent ---
    def cd_opt(f0, n, stop, seed_val, max_sweeps=200):
        conv_fn, _ = make_conv(n)
        f = np.clip(f0.copy(), 0, None)
        S = float(f.sum())
        if S < 0.01:
            f[:] = 1.0/n; S = 1.0
        c = conv_fn(f)
        best_sc = 2.0*n*float(c.max())/(S*S)
        rng = np.random.default_rng(seed_val)
        
        for sw in range(max_sweeps):
            if time.monotonic() > stop: break
            any_improved = False
            for i in rng.permutation(n):
                if time.monotonic() > stop: break
                
                v = f[i]
                Sr = S - v
                # Modified conv range: k in [i, min(i+n, 2n-1)]
                ks = i
                ke = min(i+n, 2*n-1)
                ld = ke - ks  # length of affected region
                
                # infl[k] = 2*f[k-i] for k in [ks, ke)
                # f_idx = k-i, ranges from 0 to ld-1
                f_idx = np.arange(ld)
                infl = 2.0 * f[f_idx]
                
                # rest_max
                if ks > 0:
                    rm1 = float(c[:ks].max())
                else:
                    rm1 = 0.0
                if ke < 2*n-1:
                    rm2 = float(c[ke:].max())
                else:
                    rm2 = 0.0
                rest_max = max(rm1, rm2)
                
                # Precompute base values in affected region
                base = c[ks:ke].copy()
                
                # w range
                w_lo = -v + 1e-6
                w_hi = max(2.0*v, 0.5) if v > 0.01 else 1.0
                w_hi = min(w_hi, 20.0)
                
                if w_lo >= w_hi:
                    continue
                
                # Check if 2i is in affected range
                two_i = 2*i
                in_range = (two_i >= ks) and (two_i < ke)
                
                def obj_w(w):
                    Sn = Sr + w
                    if Sn < 0.005:
                        return math.inf
                    # Max in affected region
                    vals = base + w * infl
                    if in_range:
                        vals[two_i - ks] += w*w
                    am = float(np.maximum(vals.max(), rest_max))
                    return 2.0*n*am/(Sn*Sn)
                
                # Grid search
                npg = 60
                ws = np.linspace(w_lo, w_hi, npg)
                best_w = 0.0
                best_oi = obj_w(0.0)
                
                for w in ws:
                    Sn = Sr + w
                    if Sn < 0.005:
                        continue
                    vals = base + w * infl
                    if in_range:
                        vals = vals.copy()
                        vals[two_i - ks] += w*w
                    am = max(float(vals.max()), rest_max)
                    oi = 2.0*n*am/(Sn*Sn)
                    if oi < best_oi - 1e-15:
                        best_oi = oi
                        best_w = w
                
                # Fine search
                if best_w != 0.0:
                    span = (w_hi - w_lo) / npg
                    fl = max(w_lo, best_w - 2.5*span)
                    fh = min(w_hi, best_w + 2.5*span)
                    wf = np.linspace(fl, fh, 30)
                    for w in wf:
                        Sn = Sr + w
                        if Sn < 0.005:
                            continue
                        vals = base + w * infl
                        if in_range:
                            vals = vals.copy()
                            vals[two_i - ks] += w*w
                        am = max(float(vals.max()), rest_max)
                        oi = 2.0*n*am/(Sn*Sn)
                        if oi < best_oi - 1e-15:
                            best_oi = oi
                            best_w = w
                
                if abs(best_w) > 1e-10 and best_oi < best_sc - 1e-14:
                    f[i] += best_w
                    S += best_w
                    c[ks:ke] += best_w * infl
                    if in_range:
                        c[two_i] += best_w*best_w
                    best_sc = best_oi
                    any_improved = True
            
            if not any_improved:
                break
        
        f = np.clip(f, 0, 1000)
        # Verify
        sc_v = eval_local(f, conv_fn)
        return f, sc_v
