    for it in range(num_iters):
        if time.monotonic() > stop_time:
            break
        c = _conv_full(a)
        k_max = int(np.argmax(c))
        M = c[k_max]
        
        # Find the coefficient that contributes most to c[k_max]
        # c[k_max] = sum_j a[j] * a[k_max - j]
        # Contribution of a[i] to c[k_max] is a[k_max - i] + a[i - k_max] (if valid)
        # Plus cross terms if k_max - i == i (i.e., i = k_max/2)
        
        # Actually, ∂c[k]/∂a[i] = a[k-i] (if valid) + a[i-k] (if valid)
        # For k = k_max:
        grad = np.zeros(n)
        for i in range(n):
            if 0 <= k_max - i < n:
                grad[i] += a[k_max - i]
            if 0 <= i - k_max < n:
                grad[i] += a[i - k_max]
        
        # Decrease a[i] where grad[i] is largest
        i_max = int(np.argmax(grad))
        delta = 0.1 * a[i_max]
        a[i_max] = max(0, a[i_max] - delta)
        
        sc = _score(a)
        if sc < best_sc:
            best = a.copy()
            best_sc = sc
        else:
            a[i_max] += delta  # revert
    
    return best, best_sc
