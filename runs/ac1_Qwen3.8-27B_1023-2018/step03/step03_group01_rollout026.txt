def _projected_grad(f0, n, stop_time, seed, max_its=50000):
    """Projected subgradient with momentum for final polish."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f = f / f.sum()
    best_f = f.copy()
    best_sc = _score(f)
    lr = 0.008
    momentum = np.zeros(n)
    beta = 0.8
    no_imp = 0
    fsize = next_fast_len(2 * n - 1)

    for it in range(max_its):
        if time.monotonic() > stop_time:
            break
        if it % 500 == 0 and time.monotonic() > stop_time - 0.5:
            break

        b = _conv_fft(f)
        M = float(b.max())
        s = float(f.sum())

        # Find top peaks for gradient
        topk = min(8, 2 * n - 1)
        idx = np.argpartition(b, -topk)[-topk:]
        weights = b[idx].copy()
        weights = weights - weights.min() + 1e-10
        weights /= weights.sum()

        grad_M = np.zeros(n)
        for w, k in zip(weights, idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                j = k - i
                if 0 <= j < n:
                    grad_M[i] += w * 2.0 * f[j]

        # Gradient of score = 2n*M/s^2
        grad_score = (2.0 * n / (s * s)) * grad_M - (4.0 * n * M / (s * s * s)) * np.ones(n)

        momentum = beta * momentum - lr * grad_score
        f_new = f + momentum
        f_new = np.maximum(f_new, 0.0)
        fsn = float(f_new.sum())
        if fsn < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / fsn

        sc = _score(f_new)
        if sc < best_sc - 1e-14:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.03, lr * 1.005)
        else:
            no_imp += 1
            lr = max(0.001, lr * 0.98)
            f = f_new  # still move to maintain exploration
            if no_imp > 200:
                momentum = np.zeros(n)
                lr = 0.005
                no_imp = 0
                beta = 0.7

    return best_f, best_sc


def _coord_descent(f0, n, stop_time, seed):

    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)

            if time.monotonic() > stop_time - 1.0:
                return f, best_sc
            orig = f[i]

                    f[i] = orig * frac
                    sc = _score(f)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
                        orig = f[i]
                    else:
                        f[i] = orig
            f[i] = orig

        if not improved:
            no_full_pass += 1
            if no_full_pass >= max_no_improve:
                break
        else:
            no_full_pass = 0

    return f, best_sc


def _sparse_restart(f0, n, rng, num_sparse=30):
    """Create a sparse perturbation of the current best."""
    f = f0.copy()
    mask = rng.random(n) < (num_sparse / n)
    f[mask] *= rng.uniform(0.1, 3.0, mask.sum())
    f[~mask] *= rng.uniform(0.8, 1.2, (~mask).sum())
    f = np.maximum(f, 0.0)
    fs = f.sum()
    if fs > 1e-10:
        f = f / fs
    return f


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    try:
        best_score = _score(best_seq)
    except Exception:
        best_score = math.inf
    if not math.isfinite(best_score):
        best_seq = np.ones(n_p) / n_p
        best_score = _score(best_seq)

    def _try_update(cand):
        nonlocal best_seq, best_score
        if cand is None:
            return
        cand = np.clip(np.asarray(cand, dtype=np.float64).ravel(), 0, 1000)
        if cand.size == 0 or cand.sum() < 0.01:
            return
        try:
            sc = _score(cand)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq = cand.copy()
            best_score = sc

    # === Phase 1: LP optimization on parent length ===
    n = n_p
    rng_main = np.random.default_rng(seed)

    # Generate diverse initializations
    inits = [(("parent", parent.copy()))]
    if n_p > 20:
        i_arr = np.arange(n_p, dtype=np.float64)
        # Cosine
        inits.append(("cos", 0.5 + 0.5 * np.cos(2 * math.pi * i_arr / n_p + 0.5)))
        # Multi-harmonic
        mh = np.full(n_p, 0.15)
        for fr in (1, 2, 3, 5, 8):
            ph = rng_main.uniform(0, 2 * math.pi)
            mh += 0.25 * np.cos(2 * math.pi * fr * i_arr / n_p + ph)
        inits.append(("multi", np.abs(mh) + 0.02))
        # Sinc
        sinc = np.sinc((i_arr - n_p / 2) / (n_p * 0.25))
        inits.append(("sinc", np.abs(sinc) + 0.05))
        # Exponential decay
        expd = np.exp(-2.0 * np.abs(i_arr - n_p / 2) / n_p)
        inits.append(("exp", 0.1 + 0.9 * expd))
        # Parent noise
        ns = 0.4 * max(1.0, float(np.mean(parent)))
        inits.append(("pnoise", np.abs(parent + rng_main.normal(0, ns, n_p)) + 0.01))
        # Random
        inits.append(("rand", np.abs(rng_main.normal(0, 1, n_p)) + 0.03))

    # Allocate 50% of budget for LP phase
    lp_end = t0 + budget * 0.50
    time_per_init = max(12.0, (lp_end - t0 - 5) / len(inits))

    for idx, (name, init) in enumerate(inits):
        if time.monotonic() > lp_end - 3.0:
            break
        if best_score < 1.5032:
            break
        stop = min(lp_end - 2.0, time.monotonic() + time_per_init)
        try:
            res, sc = _lp_main(init, n, stop, seed + 1000 * (idx + 1))
            _try_update(res)
        except Exception:
            pass

    # === Phase 2: Spectral surgery on best ===
    spec_end = min(deadline, t0 + budget * 0.62)
    if time.monotonic() < spec_end - 5.0:
        try:
            res, sc = _spectral_surgery(best_seq, len(best_seq), spec_end, seed + 55001, iters=200)
            _try_update(res)
        except Exception:
            pass

    # === Phase 3: More LP with spectral-optimized init ===
    lp2_end = min(deadline, t0 + budget * 0.72)
    if time.monotonic() < lp2_end - 5.0 and best_score > 1.5032:
        try:
            res, sc = _lp_main(best_seq, n, lp2_end, seed + 22001, max_its=1500)
            _try_update(res)
        except Exception:
            pass

    # === Phase 4: Projected gradient polish ===
    pg_end = min(deadline, t0 + budget * 0.82)
    if time.monotonic() < pg_end - 5.0:
        try:
            res, sc = _projected_grad(best_seq, len(best_seq), pg_end, seed + 77001)
            _try_update(res)
        except Exception:
            pass

    # === Phase 5: Try different lengths ===
    other_end = min(deadline, t0 + budget * 0.90)
    if time.monotonic() < other_end - 5.0 and best_score > 1.504:
        other_lengths = [n_p + d for d in (-8, -4, 4, 8, 16, -16, 32) if n_p + d >= 64]
        other_lengths = sorted(set(other_lengths))
        per_len = max(10.0, (other_end - time.monotonic() - 3.0) / len(other_lengths))
        for L in other_lengths:
            if time.monotonic() > other_end - 3.0:
                break
            if L <= 0 or L == n_p:
                continue
            stop = min(other_end - 2.0, time.monotonic() + per_len)
            # Interpolate
            x_old = np.linspace(0, 1, n_p)
            x_new = np.linspace(0, 1, L)
            init_l = np.interp(x_new, x_old, best_seq)
            try:
                res, sc = _lp_main(init_l, L, stop, seed + 50000 + L)
                _try_update(res)
            except Exception:
                pass

    # === Phase 6: Coordinate descent polish ===
    coord_end = deadline - 8.0
    if time.monotonic() < coord_end - 5.0:
        try:
            res, sc = _coord_descent(best_seq, len(best_seq), coord_end, seed + 88001)
            _try_update(res)
        except Exception:
            pass

    # === Phase 7: Final gradient burst ===
    if time.monotonic() < deadline - 5.0:
        try:
            res, sc = _projected_grad(best_seq, len(best_seq), deadline - 4.0, seed + 33333)
            _try_update(res)
        except Exception:
            pass

    # Final LP one more time if time permits
    if time.monotonic() < deadline - 5.0 and best_score > 1.5035:
        try:
            res, sc = _lp_main(best_seq, len(best_seq), deadline - 3.0, seed + 99999, max_its=500)
            _try_update(res)
        except Exception:
            pass

    # === Final validation ===
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        sc = evaluate_sequence(final_list)
        if math.isfinite(sc) and sc > 0 and sc < 1e6:
            return final_list
    except Exception:
        pass

    # Fallback: parent
    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass

    return [1.0] * 384
