            g_max /= gn
            improved = False
            for alpha in (0.03, 0.015, 0.007, 0.003, 0.001):
                c = a - alpha * g_max
                c = np.maximum(c, 0.0)
                cs = c.sum()
                if cs < 1e-12:
                    continue
                c = c / cs
                v = float(np.convolve(c, c).max())
                if v < M - 1e-13:
                    a = c
                    improved = True
                    break
        if not improved and stuck > 10:
            # Tiny random kick
            a = a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

    return best_a


def _optimize_polish(a0, stop_time, seed=0):
    """High-precision polishing: very small steps, tight tolerances."""
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0

    while time.monotonic() < stop_time - 0.2:
        b = np.convolve(a, a)
        M = float(b.max())
        k_star = int(b.argmax())

        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 60:
            # Very small perturbation
            a = a * np.exp(rng.normal(0, 0.005, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck = 0
            if stuck > 200:
                break
            continue

        A = _build_A(a)
        g = _lp_response(A, M * (1 + 1e-14), n, tl=0.5)

        if g is not None and np.abs(g - a).max() > 1e-10:
            cand, val = _line_search(a, g, M)
            if cand is not None and val < M - 1e-15:
                a = cand
                continue

        # Gradient micro-step
        g_max = _grad_max(a, k_star)
        g_max -= g_max.mean()
        gn = np.linalg.norm(g_max)
        if gn > 1e-14:
            g_max /= gn
            for alpha in (0.005, 0.002, 0.001, 0.0005):
                c = a - alpha * g_max
                c = np.maximum(c, 0.0)
                cs = c.sum()
                if cs < 1e-12:
                    continue
                c = c / cs
                v = float(np.convolve(c, c).max())
                if v < M - 1e-14:
                    a = c
                    break

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + budget_s - 22.0

    parent = _get_parent()
    n = parent.size

    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(32, dtype=np.float64) / 32.0
        best_score = 2.0

    def _update(cand):
        nonlocal best_seq, best_score
        try:
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                return True
        except Exception:
            pass
        return False

    # === Phase 1: Core optimization from parent (50% of budget) ===
    a_init = np.clip(parent, 0, 1000).copy()
    a_init = a_init / a_init.sum()
    p1_end = t0 + budget_s * 0.45
    try:
        res = _optimize_core(a_init, p1_end, seed=1)
        _update(res)
    except Exception:
        pass

    # === Phase 2: Alternative initializations (25% of budget) ===
    p2_end = t0 + budget_s * 0.70
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    inits.append(np.maximum(rc, 0.01) / np.maximum(rc, 0.01).sum())

    # Gaussian windows
    for w in (0.12, 0.18, 0.25, 0.35):
        ctr = (n - 1) / 2.0
        g = np.exp(-0.5 * ((x - ctr) / (w * (n - 1) / 2)) ** 2)
        inits.append(g / g.sum())

    # Sine window
    sw = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sw / sw.sum())

    # Blend with parent
    pn = a_init.copy()
    for f in (0.3, 0.5, 0.7):
        bl = f * pn + (1 - f) * inits[0]
        inits.append(bl / bl.sum())

    remaining = p2_end - time.monotonic()
    per_init = max(4.0, remaining / max(1, len(inits)))
    for idx, init in enumerate(inits):
        if time.monotonic() > p2_end - 3:
            break
        alloc = min(time.monotonic() + per_init, p2_end - 1)
        try:
            res = _optimize_core(init, alloc, seed=200 + idx)
            _update(res)
        except Exception:
            pass

    # === Phase 3: Polish best result (remaining budget) ===
    a_pol = np.clip(best_seq, 0, 1000).copy()
    a_pol = a_pol / a_pol.sum()
    try:
        res = _optimize_polish(a_pol, deadline, seed=999)
        _update(res)
    except Exception:
        pass

    # === Final validation ===
    try:
        fs = float(evaluate_sequence([float(v) for v in best_seq]))
    except Exception:
        fs = float('inf')

    if not np.isfinite(fs) or fs > 100.0:
        # Fallback: uniform
        return [1.0 / 32.0] * 32

    return [float(v) for v in best_seq]
