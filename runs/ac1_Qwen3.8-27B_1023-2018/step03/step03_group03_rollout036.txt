    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]

    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    if n <= 1024:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: 2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    """Build the constraint matrix for the LP: A has shape (2n-1, n)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _lp_direction(f, n, time_limit=10.0):
    """Find LP direction: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    b = _conv_full(f)
    M = float(b.max())
    if M < 1e-15:
        return None, M, b
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None, M, b
    if res.status != 0 or res.x is None:
        return None, M, b
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None, M, b
    return g / g_sum, M, b


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def _lp_optimize(f0, n, stop_time, seed, max_iters=300):
    """LP-based optimization with adaptive step sizes."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum

    best_f = f.copy()
    best_score = _score(f)

    t = 0.3
    t_min = 0.005
    t_max = 0.9
    stall = 0
    max_stall = 40
    iteration = 0
    prev_score = best_score

    while time.monotonic() < stop_time - 5.0 and iteration < max_iters:
        g_dir, M, b = _lp_direction(f, n, time_limit=8.0)
        if g_dir is None:
            stall += 4
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f /= fs
            else:
                f = best_f.copy()
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-7:
            stall += 3
            if stall > max_stall:
                break
            # Try perturbation
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f /= fs
            else:
                f = best_f.copy()
            continue

        # Binary search for optimal step
        f_curr_score = _score(f)
        best_t = 0.0
        best_sc = f_curr_score
        lo, hi = 0.0, 1.0
        for _ in range(18):
            mid = (lo + hi) / 2
            f_new = np.clip(f + mid * (g_dir - f), 0, None)
            fs_new = f_new.sum()
            if fs_new < 1e-10:
                hi = mid
                continue
            f_new = f_new / fs_new
            sc = _score(f_new)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_t = mid
                lo = mid
            else:
                hi = mid

        if best_t > 1e-4 and best_sc < best_score - 1e-13:
            f_new = np.clip(f + best_t * (g_dir - f), 0, None)
            f_new /= f_new.sum()
            f = f_new
            best_f = f.copy()
            best_score = best_sc
            stall = 0
        elif best_t > 1e-4 and best_sc < f_curr_score - 1e-13:
            # Accept even if not better than global best (progressive)
            f_new = np.clip(f + best_t * (g_dir - f), 0, None)
            f_new /= f_new.sum()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                # Larger perturbation
                f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f /= fs
                else:
                    f = best_f.copy()
                stall = 0
                if time.monotonic() > stop_time - 20:
                    break

        iteration += 1

    return best_f, best_score


def _grad_optimize(f0, n, stop_time, seed, max_iters=20000):
    """Gradient-based optimization using smoothed max."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum
    best_f = f.copy()
    best_score = _score(f)

    alpha = 0.05
    alpha_min = 1e-10
    tau = 80.0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 5.0:
            break

        b = _conv_full(f)
        S = float(f.sum())
        M = float(b.max())
        if M < 1e-18:
            break
        score = 2.0 * n * M / (S * S)

        if score < best_score - 1e-14:
            best_score = score
            best_f = f.copy()

        # Compute softmax weights for smoothed max
        b_shifted = b - M
        tau_b = tau * b_shifted
        tau_b -= tau_b.max()
        p = np.exp(tau_b)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(2 * n - 1)
            p[np.argmax(b)] = 1.0
            p_sum = 1.0
        p /= p_sum

        # Gradient of smoothed max: grad[i] = 2 * sum_j p[i+j] * f[j]
        # = 2 * (p * f_rev)[i + n - 1]
        f_rev = f[::-1]
        conv_size = 2 * n - 1
        P = rfft(p, conv_size)
        Fr = rfft(f_rev, conv_size)
        conv_val = irfft(P * Fr, conv_size)[:conv_size]
        grad = 2.0 * conv_val[n - 1 : 2 * n - 1]

        # Score gradient
        score_grad = (2.0 * n / (S * S)) * (grad - 2.0 * M / S)

        # Gradient step with projection
        g_raw = f - alpha * score_grad
        f_new = _project_simplex(g_raw)

        sc_new = _score(f_new)
        if sc_new < score - 1e-15:
            f = f_new
            alpha = max(alpha_min, alpha * 0.98)
        else:
            alpha = max(alpha_min, alpha * 0.5)
            if alpha < alpha_min:
                # Try tiny perturbation
                f = np.clip(f + rng.normal(0, 0.005, n), 0, None)
                f /= f.sum()
                alpha = 0.01

        # Periodic restart
        if it % 2000 == 1999:
            f = np.clip(f + rng.normal(0, 0.01, n), 0, None)
            f /= f.sum()
            alpha = 0.02

    return best_f, best_score


def _coord_polish(f0, stop_time, max_sweeps=80, seed=777):
    """Coordinate descent with exact 1D optimization."""
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, _score(f)
    f = f / S
    S = 1.0
    conv = _conv_full(f)
    best_score = 2.0 * n * conv.max()
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break

            v = f[i]
            S_rest = S - v

            # Influence vector
            infl = np.zeros(2 * n - 1)
            start = i
            end = min(i + n, 2 * n - 1)
            infl[start:end] = 2.0 * f[:end - start]

            # Search range
            x_max = max(5.0 * v, 0.5) if v > 1e-8 else 1.0
            x_max = min(x_max, 50.0)

            # Coarse grid
            n_pts = 100
            x_grid = np.linspace(0.0, x_max, n_pts)

            best_x = v
            best_obj = best_score

            for x in x_grid:
                S_new = S_rest + x
                if S_new < 0.01:
                    continue
                w = x - v
                if abs(w) < 1e-14:
                    m_new = conv.max()
                else:
                    tmp = conv.copy()
                    tmp[start:end] += w * infl[start:end]
                    if 0 <= 2 * i <= 2 * n - 2:
                        tmp[2 * i] += w * w
                    m_new = float(tmp.max())
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-16:
                    best_obj = obj
                    best_x = x

            # Fine grid
            if best_x != v:
                dw = x_max / n_pts
                x_lo = max(0.0, best_x - 3 * dw)
                x_hi = min(x_max, best_x + 3 * dw)
                x_fine = np.linspace(x_lo, x_hi, 50)
                for x in x_fine:
                    S_new = S_rest + x
                    if S_new < 0.01:
                        continue
                    w = x - v
                    if abs(w) < 1e-14:
                        continue
                    tmp = conv.copy()
                    tmp[start:end] += w * infl[start:end]
                    if 0 <= 2 * i <= 2 * n - 2:
                        tmp[2 * i] += w * w
                    m_new = float(tmp.max())
                    obj = 2.0 * n * m_new / (S_new * S_new)
                    if obj < best_obj - 1e-16:
                        best_obj = obj
                        best_x = x

            # Accept
            if best_x != v and best_obj < best_score - 1e-15:
                w = best_x - v
                f[i] = best_x
                S += w
                conv[start:end] += w * infl[start:end]
                if 0 <= 2 * i <= 2 * n - 2:
                    conv[2 * i] += w * w
                best_score = best_obj
                improved = True

        if not improved and sweep > 3:
            break

    f = np.clip(f, 0, 1000)
    final_score = _score(f)
    return f, final_score


def _pair_polish(f0, stop_time, max_rounds=50, seed=9999):
    """2-coordinate moves to escape local optima."""
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / f.sum()
    best_score = _score(f)
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        n_trials = 60
        for _ in range(n_trials):
            if time.monotonic() > stop_time - 5.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-12 and oj < 1e-12:
                continue

            multipliers = (0.2, 0.5, 1.0, 2.0, 4.0, 0.0)
            best_pair = (oi, oj)
            best_pair_sc = best_score

            for mi in multipliers:
                ni = oi * mi if oi > 1e-12 else (0.01 if mi > 0 else 0.0)
                for mj in multipliers:
                    nj = oj * mj if oj > 1e-12 else (0.01 if mj > 0 else 0.0)
                    f[i], f[j] = ni, nj
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_pair_sc - 1e-14:
                        best_pair_sc = sc
                        best_pair = (ni, nj)

            f[i], f[j] = best_pair
            if best_pair_sc < best_score - 1e-14:
                best_score = best_pair_sc
                best_f = f.copy()
                improved = True

        if not improved:
            break

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    # Fallback
    fallback = parent.copy()
    try:
        fallback_sc = _score(fallback)
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(32) / 32.0
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update_best(seq, sc=None):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64)
        seq = np.clip(seq, 0, 1000)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        if sc is None:
            sc = _score(seq)
        if not math.isfinite(sc):
            return
        # Verify with exact evaluator
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-14:
            best_seq = seq.copy()
            best_score = exact

    def _try_pipeline(n, init_arr, stop_time, phase_label=""):
        nonlocal best_seq, best_score
        if time.monotonic() > stop_time - 10:
            return
        init_arr = np.clip(np.asarray(init_arr, dtype=np.float64).ravel(), 0, None)
        while init_arr.size < n:
            init_arr = np.concatenate([init_arr, np.zeros(n - init_arr.size)])
        init_arr = init_arr[:n]
        if init_arr.sum() < 1e-10:
            init_arr = np.ones(n) / n

        # LP phase
        t_lp_end = min(stop_time - 30, t0 + (stop_time - t0) * 0.4)
        if time.monotonic() < t_lp_end - 5:
            try:
                res, sc = _lp_optimize(init_arr, n, t_lp_end, seed)
            except Exception:
                res, sc = init_arr, _score(init_arr)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res = res / res.sum()
                _update_best(res, sc)

        # Gradient phase
        t_grad_end = min(stop_time - 20, t0 + (stop_time - t0) * 0.6)
        if time.monotonic() < t_grad_end - 5 and best_score < 1.51:
            try:
                res, sc = _grad_optimize(best_seq, len(best_seq), t_grad_end, seed + 1)
            except Exception:
                res, sc = best_seq, best_score
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update_best(res, sc)

        # Coord descent
        t_cd_end = min(stop_time - 5, t0 + (stop_time - t0) * 0.85)
        if time.monotonic() < t_cd_end - 5:
            try:
                res, sc = _coord_polish(best_seq, t_cd_end, max_sweeps=60, seed=seed + 2)
            except Exception:
                res, sc = best_seq, best_score
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update_best(res, sc)

    # === Phase 1: Primary optimization on parent length ===
    phase1_end = t0 + budget * 0.45
    _try_pipeline(n_p, parent, phase1_end)

    # === Phase 2: Try alternative initializations on same length ===
    if time.monotonic() < phase1_end + budget * 0.10 and best_score > 1.503:
        i_arr = np.arange(n_p, dtype=np.float64)
        alt_inits = [
            np.clip(0.2 + 0.8 * np.cos(np.pi * i_arr / n_p) ** 2, 0, None),
            np.clip(0.5 + 0.5 * np.cos(2 * np.pi * i_arr / n_p), 0, None),
            np.clip(1.0 - 0.5 * (i_arr / n_p - 0.5) ** 2, 0, None),
        ]
        alt_inits = [x / x.sum() for x in alt_inits if x.sum() > 0.01]
        alt_stop = min(t0 + budget * 0.55, hard_deadline - 60)
        for idx, init in enumerate(alt_inits):
            if time.monotonic() > alt_stop:
                break
            if best_score < 1.503:
                break
            t_alloc = min(alt_stop - 10, time.monotonic() + budget * 0.06)
            try:
                res, sc = _lp_optimize(init, n_p, t_alloc, seed + 200 + idx)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res = res / res.sum()
                    _update_best(res, sc)
            except Exception:
                pass

    # === Phase 3: Intensive coordinate descent on best ===
    t_cd_end = min(t0 + budget * 0.75, hard_deadline - 40)
    if time.monotonic() < t_cd_end - 10:
        try:
            best_seq, best_score = _coord_polish(best_seq, t_cd_end, max_sweeps=80, seed=424242)
        except Exception:
            pass
        _update_best(best_seq, best_score)

    # === Phase 4: Try nearby lengths ===
    t_len_end = min(t0 + budget * 0.88, hard_deadline - 50)
    if time.monotonic() < t_len_end - 30 and best_score > 1.5028:
        other_lengths = []
        for L in [n_p - 16, n_p - 8, n_p + 8, n_p + 16, n_p * 2, n_p // 2]:
            if L >= 16 and L <= 1024 and L not in other_lengths:
                other_lengths.append(L)
        other_lengths.sort(key=lambda L: abs(L - n_p))

        time_per_len = max(15.0, (t_len_end - time.monotonic()) / max(1, len(other_lengths)))
        for L in other_lengths[:4]:
            if time.monotonic() > t_len_end - 10:
                break
            if best_score < 1.5028:
                break
            # Interpolate best to new length
            init_l = np.interp(
                np.linspace(0, 1, L),
                np.linspace(0, 1, len(best_seq)),
                best_seq
            )
            init_l = np.clip(init_l, 0, None)
            if init_l.sum() < 0.01:
                init_l = np.ones(L) / L
            init_l = init_l / init_l.sum()

            stop_lp = min(t_len_end - 5, time.monotonic() + time_per_len * 0.6)
            stop_pol = min(t_len_end - 2, stop_lp + time_per_len * 0.4)
            try:
                res, sc = _lp_optimize(init_l, L, stop_lp, seed + 50000 + L)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res = res / res.sum()
                    _update_best(res, sc)
                if time.monotonic() < stop_pol - 5:
                    res2, sc2 = _coord_polish(best_seq, stop_pol, max_sweeps=20, seed=seed + L)
                    _update_best(res2, sc2)
            except Exception:
                pass

    # === Phase 5: Final intense polish ===
    final_end = hard_deadline - 10.0
    if time.monotonic() < final_end - 10:
        try:
            best_seq, best_score = _coord_polish(best_seq, final_end - 10, max_sweeps=50, seed=31337)
        except Exception:
            pass
        _update_best(best_seq, best_score)

    if time.monotonic() < final_end - 5:
        try:
            best_seq, best_score = _pair_polish(best_seq, final_end - 3, max_rounds=30, seed=55555)
        except Exception:
            pass
        _update_best(best_seq, best_score)

    # Final coordinate polish
    if time.monotonic() < hard_deadline - 8:
        try:
            best_seq, best_score = _coord_polish(best_seq, hard_deadline - 6, max_sweeps=20, seed=88888)
        except Exception:
            pass
        _update_best(best_seq, best_score)

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
