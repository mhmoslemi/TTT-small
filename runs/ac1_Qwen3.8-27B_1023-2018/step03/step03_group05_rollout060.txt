    f = rfft(a, nfft)

def _build_A(a):
    n = a.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(n):
        A[k, :k + 1] = a[:k + 1][::-1]
    for k in range(n, m):
        L = m - k
        s = k - n + 1
        A[k, s:s + L] = a[n - 1:n - L - 1:-1]
    return A


def _softmax_weights(c, M, temperature=None):
    if temperature is None:
        temperature = max(M * 0.08, 1e-8)
    z = (c - M) / temperature

    w = np.exp(z)

    if s > 0 and np.isfinite(s):
        return w / s
    w = np.zeros_like(c)
    w[int(np.argmax(c))] = 1.0
    return w


def _gradient_softmax(a, w):
