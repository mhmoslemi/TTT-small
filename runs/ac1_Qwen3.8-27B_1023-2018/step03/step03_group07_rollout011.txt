```python
import numpy as np
import time
from scipy.optimize import linprog

def _convolution_matrix(a, n):
    """Build matrix B where B @ v gives (a*v) full convolution.
    B has shape (2n-1, n).
    B[k, j] = a[k-j] if 0 <= k-j < n else 0.
    """
    B = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                B[k, j] = a[idx]
    return B
```

```python
    Returns (v, M_opt)."""
    n = a.size
    m = 2*n - 1
    
    # Build B: B[k,j] = a[k-j] if 0<=k-j<n
    k_idx = np.arange(m)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    B = np.where(mask, a[np.clip(diff, 0, n-1).astype(int)], 0.0)
    
    # LP: min M s.t. B@v - M <= 0 (m constraints), sum(v) = 1, v >= 0, M >= 0
    # Variables: [v_0,...,v_{n-1}, M]
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0  # minimize M
    
    A_ub = np.hstack([B, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    v = res.x[:n]
    M = res.x[-1]
    v = np.maximum(v, 0)
    v = v / v.sum()
    return v, M
```

```python
import numpy as np
import time
from scipy.optimize import linprog


def _score(a):
    """Compute exact score matching evaluator."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * a.size * c.max() / (s * s))


def _conv_matrix(a):
    """Build B (shape 2n-1 x n) where B @ v = full convolution of a and v."""
    n = a.size
    m = 2 * n - 1
    k = np.arange(m)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    mask = (idx >= 0) & (idx < n)
    B = np.where(mask, a[np.clip(idx, 0, n - 1).astype(np.int64)], 0.0)
    return B


def _lp_br(a):
    """Solve LP: min M s.t. (a*v)[k] <= M for all k, sum(v)=1, v>=0, M>=0."""
    n = a.size
    m = 2 * n - 1
    B = _conv_matrix(a)

    c_obj = np.concatenate([np.zeros(n), [1.0]])
    A_ub = np.hstack([B, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 2.0, 'primal_feasibility_tolerance': 1e-11,
                               'dual_feasibility_tolerance': 1e-11})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    v = np.maximum(res.x[:n], 0.0)
    vs = v.sum()
    if vs < 1e-15:
        return None
    return v / vs


def _line_search(a, v, M_ref, n_trials=20):
    """Find best t in [0,1] minimizing max(((1-t)*a+t*v) * ((1-t)*a+t*v))."""
    best_t = 0.0
    best_val = M_ref
    # Coarse
    for t in np.linspace(0.0, 1.0, n_trials):
        cand = (1.0 - t) * a + t * v
        cand = np.maximum(cand, 1e-15)
        cand = cand / cand.sum()
        val = float(np.convolve(cand, cand).max())
        if val < best_val:
            best_val = val
            best_t = t
    # Fine around best
    if 0 < best_t < 1:
        lo = max(0, best_t - 1.0 / n_trials)
        hi = min(1, best_t + 1.0 / n_trials)
        for _ in range(30):
            if hi - lo < 1e-6:
                break
            mid = (lo + hi) / 2.0
            cand = (1.0 - mid) * a + mid * v
            cand = np.maximum(cand, 1e-15)
            cand = cand / cand.sum()
            val = float(np.convolve(cand, cand).max())
            if val < best_val:
                best_val = val
                best_t = mid
                hi = mid
            else:
                lo = mid
    if best_t < 1e-8:
        return None, best_val
    cand = (1.0 - best_t) * a + best_t * v
    cand = np.maximum(cand, 1e-15)
    cand = cand / cand.sum()
    return cand, best_val


def _gradient_dir(a, k_max):
    """Gradient of (a*a)[k_max] w.r.t. a, projected to sum-zero."""
    n = a.size
    k = k_max
    grad = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    for j in range(j_lo, j_hi + 1):
        idx = k - j
        if 0 <= idx < n:
            grad[j] += 2.0 * a[idx]
    grad -= grad.mean()
    return grad


def _optimize_lp(a0, stop_time, seed=0, max_iters=80000):
    """LP best-response with line search and perturbations."""
    n = a0.size
    a = np.clip(a0.copy(), 1e-10, 1000.0)
    a = a / a.sum()

    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())

    rng = np.random.default_rng(seed)
    stuck = 0
    perturb_count = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break

        v = _lp_br(a)
        if v is None:
            a = a * np.exp(rng.normal(0, 0.04, n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            continue

        cand, val = _line_search(a, v, best_M)
        if cand is not None and val < best_M * (1.0 - 1e-13):
            best_M = val
            best_a = cand.copy()
            a = cand
            stuck = 0
        else:
            # Try direct v
            M_v = float(np.convolve(v, v).max())
            if M_v < best_M * (1.0 - 1e-12):
                best_M = M_v
                best_a = v.copy()
                a = v.copy()
                stuck = 0
            else:
                stuck += 1

            if stuck > 40:
                scale = min(0.03 + 0.03 * (stuck // 40), 0.5)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                stuck = 0
                perturb_count += 1
                if perturb_count > 40:
                    break
                continue

            # Gradient step as sub-move
            b = np.convolve(a, a)
            k_star = int(b.argmax())
            grad = _gradient_dir(a, k_star)
            gn = np.linalg.norm(grad)
            if gn > 1e-12:

                    cand = np.maximum(cand, 1e-10)
                    cand = cand / cand.sum()
                    val = float(np.convolve(cand, cand).max())
                    if val < best_M * (1.0 - 1e-13):
                        best_M = val
                        best_a = cand.copy()
                        a = cand
                        stuck = 0
                        break

def _optimize_gradient_polish(a0, stop_time, seed=0, max_iters=50000):
    """Polish via subgradient descent on the active constraints."""
    n = a0.size
    a = np.clip(a0.copy(), 1e-10, 1000.0)
    a = a / a.sum()
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break

        b = np.convolve(a, a)
        M = float(b.max())
        # Active set: positions where b is within 1% of max
        active = np.where(b >= M * 0.999)[0]
        if len(active) == 0:
            active = [int(b.argmax())]

        # Averagesubgradient
        grad = np.zeros(n)
        for k in active:
            grad += _gradient_dir(a, k)
        grad /= len(active)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        grad = grad / gn
        # Line search
        improved = False
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.0005]:
            cand = a - alpha * grad
            cand = np.maximum(cand, 1e-10)
            cand = cand / cand.sum()
            val = float(np.convolve(cand, cand).max())
            if val < best_M * (1.0 - 1e-13):

                best_a = cand.copy()
                a = cand
                improved = True
                break

            stuck = 0
        else:
            stuck += 1

                a = np.maximum(a, 1e-10)
                a = a / a.sum()
```
