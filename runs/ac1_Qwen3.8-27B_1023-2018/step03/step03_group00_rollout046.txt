    c = irfft(F * F, fsize)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        m_lo = max(0, k - n + 1)
        m_hi = min(k, n - 1)
        if m_lo <= m_hi:
            A[k, m_lo:m_hi+1] = f[k - np.arange(m_lo, m_hi + 1)]
    return A


def _slp_optimize(f0, n, stop_time, seed, lp_timeout=2.0):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    if f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    if f.sum() < 1e-10:
        f = np.ones(n)
    
    best_f = f.copy()
    best_score = _score(f)
    
    t = 0.5
    stall = 0
    max_stall = 50
    iteration = 0
    
    while time.monotonic() < stop_time - 5.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        A = _build_A(f, n)
        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method="highs",
                         options={"presolve": True, "time_limit": lp_timeout,
                                  "dual_feasibility_tolerance": 1e-9})
        except Exception:
            stall += 10
            if stall > max_stall:
                break
            continue
        
        if res.status != 0 or res.x is None:
            stall += 5
            if stall > max_stall:
                break
            # Add noise and continue
            f = np.clip(f + 0.01 * rng.standard_normal(n) * max(1, f.mean()), 0, None)
            if f.sum() < 1e-10:
                f = best_f.copy()
            continue
        
        g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            stall += 3
            continue
        
        # Rescale g to match f's sum
        f_sum = float(f.sum())
        g = g * (f_sum / g_sum)
        
        # Check if g is close to f (convergence)
        dist = np.linalg.norm(g - f) / max(1.0, np.linalg.norm(f))
        if dist < 1e-10:
            stall += 1
            if stall > max_stall:
                break
            continue
        
        # Line search
        cur_sc = _score(f)
        best_t_trial = t
        best_sc_trial = cur_sc
        improved = False
        
        for _ in range(16):
            f_trial = (1 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0)
            if f_trial.sum() < 0.01:
                t *= 0.5
                continue
            sc = _score(f_trial)
            if sc < best_sc_trial - 1e-14:
                best_t_trial = t
                best_sc_trial = sc
            if sc < best_score - 1e-14:
                f = f_trial.copy()
                best_f = f.copy()
                best_score = sc
                improved = True
                t = min(0.8, t * 1.2)
                stall = 0
                break
            else:
                t = max(0.01, t * 0.5)
        
        if not improved:
            stall += 1
            if stall >= max_stall:
                # Kick with noise
                scale = 1.0 + 0.05
                f = np.clip(f * scale + 0.03 * max(1, f.mean()) * rng.standard_normal(n), 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * best_f.sum()
                t = 0.5
                stall = 0
                max_stall += 20
        
        iteration += 1
    
    return best_f, best_score


def _polish(a, stop_time, seed=4242):
    n = len(a)
    best = np.maximum(np.asarray(a, dtype=np.float64).copy(), 0)
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 15
    
    while time.monotonic() < stop_time - 5.0:
        improved = False
        
        # Coordinate descent - single coordinate
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break
            orig = best[i]
            if orig < 1e-14:
                continue
            best_val = orig
            best_val_sc = best_sc
            for mlt in (0.1, 0.25, 0.4, 0.6, 0.8, 1.1, 1.3, 1.6, 2.2, 3.5):
                best[i] = orig * mlt
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-15:
                    best_val = best[i]
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-15:
                best_sc = best_val_sc
                improved = True
        
        # Two-coordinate
        for _ in range(80):
            if time.monotonic() > stop_time - 5.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.3, 0.5, 0.7, 1.3, 1.8):
                for mj in (0.3, 0.5, 0.7, 1.3, 1.8):
                    best[i] = oi * mi
                    best[j] = oj * mj
                    if best.sum() < 0.01:
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj
        
        # Random perturbations
        if not improved:
            for _ in range(50):
                if time.monotonic() > stop_time - 5.0:
                    break
                k = int(rng.integers(n))
                old = best[k]
                best[k] = max(0.0, old + 0.02 * max(1, best.mean()) * rng.standard_normal())
                if best.sum() < 0.01:
                    best[k] = old
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                else:
                    best[k] = old
        
        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0
    
    return best, best_sc
