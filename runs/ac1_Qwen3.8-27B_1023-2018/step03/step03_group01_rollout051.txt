    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_p = len(parent)

    # Initialize best
    best_seq = parent.copy()
    fs_init = float(best_seq.sum())
    if fs_init < 0.01:
        best_seq = np.ones(n_p) / n_p
        fs_init = 1.0
    best_seq = np.clip(best_seq / fs_init * fs_init, 0, 1000)
    best_score = _score(best_seq)
    if not math.isfinite(best_score):
        best_seq = np.ones(384) / 384.0
        n_p = 384
        best_score = _score(best_seq)

    def _try_update(cand):
        nonlocal best_seq, best_score
        if cand is None or cand.size == 0:
            return
        cand = np.clip(np.asarray(cand, dtype=np.float64).ravel(), 0, 1000)
        if cand.sum() < 0.01:
            return
        try:
            sc = _score(cand)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq = cand.copy()
            best_score = sc

    # Phase 1: LP optimization on parent length
    n = n_p
    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    # Build diverse initializations
    inits = []
    inits.append(("parent", parent.copy()))

    # Cosine taper (reduces edge peaks)
    cos_taper = 0.15 + 0.85 * (0.5 + 0.5 * np.cos(2 * math.pi * i_arr / n))
    inits.append(("cos_taper", cos_taper))

    # Raised cosine (narrower)
    rc = 0.1 + 0.9 * (0.5 + 0.5 * np.cos(2 * math.pi * i_arr / (n * 1.5)))
    rc = np.clip(rc, 0, None)
    inits.append(("raised_cos", rc))

    # Multi-frequency superposition
    sup = np.full(n, 0.15)
    for fr in (1, 2, 3, 5):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.25 * np.cos(2 * math.pi * fr * i_arr / n + ph)
    sup = np.abs(sup) + 0.02
    inits.append(("multifreq", sup))

    # Random smooth (low-pass noise)
    rnd = rng_main.normal(0, 1, n)
    from scipy.ndimage import gaussian_filter1d
    rnd_smooth = gaussian_filter1d(rnd, sigma=n * 0.05)
    rnd_smooth = np.abs(rnd_smooth) + 0.05
    inits.append(("smooth_rand", rnd_smooth))

    # Parent + directed noise (reduce edges)
    p_mod = parent.copy()
    p_mod[:5] *= 0.3
    p_mod[-5:] *= 0.3
    p_mod = p_mod + rng_main.normal(0, 0.03 * max(1.0, np.mean(parent)), n)
    p_mod = np.abs(p_mod) + 0.01
    inits.append(("p_trimmed", p_mod))

    lp_end = t0 + budget * 0.55
    time_per_init = max(10.0, (lp_end - t0 - 5) / len(inits))

    for idx, (name, init) in enumerate(inits):
        if time.monotonic() > lp_end - 5.0:
            break
        if best_score < 1.5027:
            break
        stop = min(lp_end - 4.0, time.monotonic() + time_per_init)
        try:
            res, sc = _lp_loop(init, n, stop, seed + 1000 * (idx + 1))
            _try_update(res)
        except Exception:
            pass

    # Phase 2: Subgradient polish
    sg_end = min(deadline, t0 + budget * 0.72)
    if time.monotonic() < sg_end - 5.0:
        try:
            res, sc = _subgrad_polish(best_seq, len(best_seq), sg_end, seed + 99001)
            _try_update(res)
        except Exception:
            pass
        # Second subgrad pass with different params
        if time.monotonic() < sg_end - 5.0:
            try:
                res, sc = _subgrad_polish(best_seq, len(best_seq), sg_end, seed + 77002)
                _try_update(res)
            except Exception:
                pass

    # Phase 3: Try other lengths
    other_end = min(deadline, t0 + budget * 0.86)
    if time.monotonic() < other_end - 10.0 and best_score > 1.503:
        other_lengths = [n_p - 32, n_p, n_p + 32, 320, 448, 512]
        other_lengths = sorted(set(L for L in other_lengths if 64 <= L <= 1024))
        for L in other_lengths:
            if time.monotonic() > other_end - 5.0:
                break
            stop = min(other_end - 4.0, time.monotonic() + 20.0)
            if L == n_p:
                init_l = parent.copy()
            else:
                init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n_p), parent)
            try:
                res, sc = _lp_loop(init_l, L, stop, seed + 50000 + L)
                _try_update(res)
            except Exception:
                pass

    # Phase 4: Coordinate polish
    polish_end = deadline - 8.0
    if time.monotonic() < polish_end - 20.0:
        try:
            res, sc = _coord_polish(best_seq, len(best_seq), polish_end, seed + 55001)
            _try_update(res)
        except Exception:
            pass

    # Phase 5: Final subgrad burst
    if time.monotonic() < deadline - 12.0:
        try:
            res, sc = _subgrad_polish(best_seq, len(best_seq), deadline - 10.0, seed + 33001)
            _try_update(res)
        except Exception:
            pass

    # Phase 6: Quick LP restart from best
    if time.monotonic() < deadline - 15.0 and best_score > 1.503:
        try:
            res, sc = _lp_loop(best_seq, len(best_seq), deadline - 8.0, seed + 44001, aggressive=True)
            _try_update(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        sc = evaluate_sequence(final_list)
        if math.isfinite(sc) and sc > 0:
            return final_list
    except Exception:
        pass

    # Fallback
    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass

    return [1.0] * 384
