                stall += 1
                t = max(0.02, t * 0.6)

            if stall >= max_stall:
                # Escape: larger perturbation
                strength = 0.05 + 0.05 * min(stall // max_stall, 3)
                f = np.clip(f + rng.normal(0, strength, nn), 0, None)
                Sf = f.sum()
                if Sf > 1e-10:
                    f = f / Sf
                else:
                    f = best_f_local.copy()
                t = 0.3
                stall = 0

        return best_f_local, best_sc_local

    # Coordinate descent polish
    def polish(f0, stop_time, seed_val=4242):
        nn = len(f0)
        best = np.clip(f0.copy(), 0, None)
        S = best.sum()
        if S < 1e-10:
            return best, math.inf
        best = best / S
        best_sc = score_fast(best)
        rng = np.random.default_rng(seed_val)
        no_imp = 0
        max_no_imp = 6

        multipliers = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.6, 2.0, 3.0, 0.0, 5.0, 0.1]

        while time.monotonic() < stop_time - 2.0:
            improved = False

            # 1-coordinate moves (full sweep)
            order = rng.permutation(nn)
            for i in order:
                if time.monotonic() > stop_time - 2.0:
                    break
                orig = best[i]
                best_val = orig
                best_val_sc = best_sc
                for m in multipliers:
                    new_val = orig * m if orig > 1e-14 else (0.01 if m > 0 else 0.0)
                    best[i] = new_val
                    if best.sum() < 0.01:
                        best[i] = orig
                        continue
                    sc = score_fast(best)
                    if sc < best_val_sc - 1e-14:
                        best_val = new_val
                        best_val_sc = sc
                best[i] = best_val
                if best_val_sc < best_sc - 1e-14:
                    best_sc = best_val_sc
                    improved = True

            # 2-coordinate moves
            n_2coord = 120
            for _ in range(n_2coord):
                if time.monotonic() > stop_time - 2.0:
                    break
                i, j = int(rng.integers(nn)), int(rng.integers(nn))
                if i == j:
                    continue
                oi, oj = best[i], best[j]
                for mi in (0.5, 1.5, 0.0, 2.0, 0.2):
                    for mj in (0.5, 1.5, 0.0, 2.0, 0.2):
                        best[i] = oi * mi if oi > 1e-14 else (0.01 if mi > 0 else 0.0)
                        best[j] = oj * mj if oj > 1e-14 else (0.01 if mj > 0 else 0.0)
                        if best.sum() < 0.01:
                            best[i], best[j] = oi, oj
                            continue
                        sc = score_fast(best)
                        if sc < best_sc - 1e-14:
                            best_sc = sc
                            improved = True
                best[i], best[j] = oi, oj

            if not improved:
                no_imp += 1
                if no_imp >= max_no_imp:
                    break
            else:
                no_imp = 0

        return best, best_sc

    # Subgradient refinement
    def subgrad_refine(f0, stop_time, seed_val=99):
        nn = len(f0)
        f = np.clip(f0.copy(), 0, None)
        S = f.sum()
        if S < 1e-10:
            return f, math.inf
        f = f / S
        best_f = f.copy()
        best_sc = score_fast(f)
        rng = np.random.default_rng(seed_val)

        lr = 0.03
        for it in range(300):
            if time.monotonic() > stop_time - 2.0:
                break
            conv = conv_fft(f)
            M = float(conv.max())
            # Active set: lags within 10% of max
            thresh = 0.90 * M
            active = np.where(conv >= thresh)[0]
            if len(active) == 0:
                break
            # Subgradient of max conv component
            subgrad = np.zeros(nn)
            for k in active:
                # d conv[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
                # But for self-conv: conv[k] = sum_i f[i]*f[k-i]
                # d/d f[i] = f[k-i] (if valid) + f[i-k] (if valid, i != k-i)
                for i in range(nn):
                    j = k - i
                    if 0 <= j < nn:
                        subgrad[i] += f[j]
            subgrad /= len(active)
            # Normalize gradient
            gnorm = np.linalg.norm(subgrad)
            if gnorm < 1e-12:
                break
            subgrad /= gnorm

            f_new = f - lr * subgrad
            f_new = np.clip(f_new, 0, None)
            Sn = f_new.sum()
            if Sn < 1e-10:
                break
            f_new = f_new / Sn
            sc = score_fast(f_new)
            if sc < best_sc - 1e-13:
                best_f = f_new.copy()
                best_sc = sc
                f = f_new
                lr = min(0.08, lr * 1.03)
            else:
                lr = max(0.005, lr * 0.7)

        return best_f, best_sc

    # === Main optimization phases ===

    # Phase 1: Main LP optimization (55% of budget)
    phase1_end = t0 + budget * 0.55
    try:
        res, sc = lp_optimize(parent, phase1_end, seed, max_iters=800)
        track(res)
    except Exception:
        pass

    # Phase 2: Subgradient refinement (10%)
    phase2_end = t0 + budget * 0.65
    if time.monotonic() < phase2_end - 5.0:
        try:
            ref_f, ref_sc = subgrad_refine(best_seq, phase2_end, seed + 777)
            track(ref_f)
        except Exception:
            pass

    # Phase 3: Polish (15%)
    phase3_end = t0 + budget * 0.80
    if time.monotonic() < phase3_end - 5.0:
        try:
            polished, _ = polish(best_seq, phase3_end, seed + 100)
            track(polished)
        except Exception:
            pass

    # Phase 4: Perturbation + re-optimize (10%)
    phase4_end = deadline - 20.0
    if time.monotonic() < phase4_end - 10.0:
        rng = np.random.default_rng(seed + 999)
        n_trials = 4
        time_per_trial = max(5.0, (phase4_end - time.monotonic()) / (n_trials + 1))
        for trial in range(n_trials):
            if time.monotonic() > phase4_end - 3.0:
                break
            if trial % 2 == 0:
                perturbed = np.clip(best_seq + rng.normal(0, 0.03, n), 0, None)
            else:
                perturbed = np.clip(best_seq * (1.0 + 0.1 * rng.normal(0, 1, n)), 0, None)
            Sp = perturbed.sum()
            perturbed = perturbed / Sp if Sp > 1e-10 else best_seq.copy()
            stop_trial = min(phase4_end - 2.0, time.monotonic() + time_per_trial)
            try:
                res, sc = lp_optimize(perturbed, stop_trial, seed + 3000 + trial * 13, max_iters=200)
                track(res)
            except Exception:
                pass

    # Phase 5: Final polish (remaining time)
    if time.monotonic() < deadline - 5.0:
        try:
            polished, _ = polish(best_seq, deadline - 3.0, seed + 500)
            track(polished)
        except Exception:
            pass

    # Final subgradient if time allows
    if time.monotonic() < deadline - 4.0:
        try:
            ref_f, _ = subgrad_refine(best_seq, deadline - 2.0, seed + 1234)
            track(ref_f)
        except Exception:
            pass

    # Final validation
    final_sc = eval_exact(best_seq)
    if not math.isfinite(final_sc):
        # Fallback
        fallback = parent.copy()
        fb_sc = eval_exact(fallback)
        if math.isfinite(fb_sc):
            return [float(x) for x in fallback]
        return [1.0] * 384

    return [float(x) for x in best_seq]
