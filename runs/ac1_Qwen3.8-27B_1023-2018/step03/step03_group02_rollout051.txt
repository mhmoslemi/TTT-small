def _lp_direction(f, n, M, active_idx=None, time_limit=15.0):
    """
    Solve: max sum(g) s.t. (f*g)_k <= M, g >= 0
    Returns normalized g or None.
    """
    # Build constraint matrix rows
    if active_idx is not None:
        rows = active_idx
    else:
        rows = np.arange(2 * n - 1)
    
    # Build A_subset: shape (len(rows), n)
    A_sub = np.zeros((len(rows), n))
    for j in range(n):
        # Column j: A[k,j] = f[k-j] for valid k-j
        # For row index r in rows: A[rows[r], j] = f[rows[r]-j] if 0 <= rows[r]-j < n
        idx = rows - j  # shape (len(rows),)
        valid = (idx >= 0) & (idx < n)
        A_sub[valid, j] = f[idx[valid]]
    
    c = -np.ones(n)
    b_ub = np.full(len(rows), M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    g_sum = g.sum()
    if g_sum < 1e-12:
        return None
    return g / g_sum

                break

            # b_k = b_k_base + 2 * a_{k-j} * a_j  (for k != 2j)
            # b_{2j} = b_{2j}_base + a_j^2  (for k = 2j, if 0 <= 2j < 2n-1)

            # Restoring: b_k = b_k_base + 2*a_{k-j}*a_j for k != 2j
            # b_{2j} = b_{2j}_base + a_j^2
            
            # The max we want to minimize: max_k [b_k_base + 2*a_{k-j}*a_j] (with correction for k=2j)
            # + constraint: sum a_i = 1 → a_j = 1 - sum_{i≠j} a_i = 1 - (a.sum() - a_orig) [before change]

            # So changing a_j means we must adjust another coordinate.
            # This makes pure coordinate descent on a fixed-sum simplex tricky.
            
            # Alternative: work in the simplex by adjusting a_j and compensating with a uniform redistribution.
            # Or: just try perturbation directions.
            
            a[j] = a_orig  # restore
            # Try a few values for a_j, redistributing the difference over other coordinates
            base_sum = 1.0 - a_orig
            for mult in (0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.0):
                a_new_j = a_orig * mult
                diff = a_new_j - a_orig
                if abs(diff) < 1e-15:

                # Redistribute diff over other coordinates proportionally
                a_try = a.copy()
                a_try[j] = a_new_j
                others = np.arange(n) != j
                if base_sum > 1e-15:
                    a_try[others] -= diff * (a[others] / base_sum)
                a_try = np.clip(a_try, 0, None)

                    best_sc = sc

            no_improve += 1
            if no_improve >= max_no:
                break
        else:
            no_improve = 0
    
    return best_a, best_sc

        t_lo, t_hi = 0.0, min(1.0, max(0.1, t * 2.5))
        best_t = 0.0
        best_sc_at_t = best_score

        for _ in range(10):
            if time.monotonic() > stop_time - 4.0:
                break
            t_mid = (t_lo + t_hi) / 2
            f_try = np.clip((1 - t_mid) * f + t_mid * g, 0, None)
            s_try = f_try.sum()
            if s_try < 1e-10:
                t_hi = t_mid
                continue
            f_try = f_try / s_try
            sc_try = _score_fast(f_try)
            if sc_try < best_sc_at_t - 1e-15:
                best_sc_at_t = sc_try
                best_t = t_mid
                t_lo = t_mid
            else:
                t_hi = t_mid

        if best_t > 1e-6:
            f = np.clip((1 - best_t) * f + best_t * g, 0, None)
            f = _normalize(f)
            sc = _score_fast(f)
            if sc < best_score - 1e-15:
                best_f = f.copy()
                best_score = sc
            b_cur = _conv_full(f)
            M = float(b_cur.max())
            stall = 0
            t = max(0.1, min(1.0, best_t * 1.1))
        else:
            stall += 1
            if stall >= max_stall:
                strategy = stall % 3
                if strategy == 0:
                    f = np.clip(f + rng.normal(0, 0.1, n), 0, None)
                elif strategy == 1:
                    blk = max(1, n // 4)
                    f2 = f.copy()
                    f2[:blk], f2[n - blk:] = f[n - blk:].copy(), f[:blk].copy()
                    f = f2
                else:
                    kern = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
                    kern /= kern.sum()
                    noise = np.convolve(rng.normal(0, 0.4, n), kern, 'same')
                    f = np.clip(f * (1 + noise), 0, None)
                f = _normalize(f)
                b_cur = _conv_full(f)
                M = float(b_cur.max())
                t = 0.5
                stall = 0
            else:
                t = max(0.05, t * 0.6)

        iters += 1

    return best_f, best_score


def _slsqp_polish(a0, stop_time, max_iter=300):
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, None).copy()
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score_fast(a)

    def obj_fn(v):
        v = np.clip(v, 0, None)
        s = v.sum()
        if s < 1e-15:
            return 1.0
        v = v / s
        b = _conv_full(v)
        return b.max()

    def con_sum(v):
        return v.sum() - 1.0

    try:
        res = minimize(obj_fn, a, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': con_sum}],
                       bounds=[(0, None)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None:
            x = np.clip(res.x, 0, None)
            x = _normalize(x)
            sc = _score_fast(x)
            if sc < best_sc:
                best_a = x
                best_sc = sc
    except Exception:
        pass

    return best_a, best_sc


def _coord_polish(a0, stop_time, seed=0):
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, None).copy()
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score_fast(a)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no = 6

    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        base_sum_noj = 1.0

        for j in order:
            if time.monotonic() > stop_time - 2.0:
                break
            a_orig = a[j]
            if a_orig < 1e-15:
                continue
            others_sum = 1.0 - a_orig
            if others_sum < 1e-12:
                continue
            multipliers = (0.3, 0.6, 0.8, 1.0, 1.2, 1.5, 2.0, 0.0)
            for mlt in multipliers:
                a_new_j = a_orig * mlt
                diff = a_new_j - a_orig
                if abs(diff) < 1e-15:
                    continue
                a_try = a.copy()
                a_try[j] = a_new_j
                a_try[order != j] -= diff * (a[order != j] / others_sum)
                a_try = np.clip(a_try, 0, None)
                s_try = a_try.sum()
                if s_try < 0.01:
                    continue
                a_try = a_try / s_try
                sc = _score_fast(a_try)
                if sc < best_sc - 1e-14:
                    a = a_try
                    best_a = a.copy()
                    best_sc = sc
                    improved = True

        if not improved:
            no_imp += 1
            if no_imp >= max_no:
                break
        else:
            no_imp = 0

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    n_parent = len(parent)

    best_seq = np.clip(parent, 0, 1000).copy()
    best_score = float(evaluate_sequence([float(x) for x in best_seq]))
    if not math.isfinite(best_score):
        best_seq = np.ones(n_parent)
        best_score = _score_fast(best_seq)

    n = n_parent
    i_arr = np.arange(n, dtype=np.float64)
    rng_main = np.random.default_rng(seed)

    # Build diverse initializations
    inits = []
    inits.append(parent.copy())

    # Cosine taper
    taper = 0.5 + 0.5 * np.cos(np.pi * i_arr / (n - 1))
    inits.append(taper)

    # Multi-frequency
    sup = np.ones(n) * 0.2
    for fr in (1, 2, 3, 5, 7):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.15 * np.cos(2 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.002)

    # Smooth random
    base = rng_main.normal(0, 1, n)
    kern = np.exp(-0.5 * (np.arange(-10, 11) / 3.5) ** 2)
    kern /= kern.sum()
    smoothed = np.convolve(base, kern, 'same')
    inits.append(np.abs(smoothed) + 0.003)

    # Parent + perturbation
    nscale = 0.06 * max(1.0, np.std(parent))
    inits.append(np.abs(parent + rng_main.normal(0, nscale, n)) + 0.001)

    # Zipper
    z = np.ones(n)
    z[::2] *= 1.6
    z[1::2] *= 0.5
    inits.append(z)

    # Phase allocation
    phase1_frac = 0.55
    phase2_frac = 0.60
    phase3_frac = 0.05
    phase1_end = t0 + budget * phase1_frac
    phase2_end = t0 + budget * phase2_frac
    phase3_end = t0 + budget * (phase2_frac + phase3_frac)

    # Phase 1: LP optimization on multiple inits
    time_per_init = max(10.0, (phase1_end - t0) / max(1, len(inits)))
    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        if best_score < 1.5028:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        if time.monotonic() >= stop:
            break
        try:
            res, sc = _lp_optimize(init, n, stop, seed + 10000 * (idx + 1))
            if np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_c = np.clip(res, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in res_c]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = res_c.copy()
                    best_score = exact
        except Exception:
            pass

    # Continue LP on best found if time remains before phase2
    if time.monotonic() < phase2_end - 15.0:
        try:
            res, sc = _lp_optimize(best_seq, n, phase2_end - 5.0, seed + 99999)
            if np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_c = np.clip(res, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in res_c]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = res_c.copy()
                    best_score = exact
        except Exception:
            pass

    # Phase 2: SLSQP polish
    if time.monotonic() < phase3_end - 10.0:
        slsqp_stop = min(phase3_end - 5.0, time.monotonic() + (phase3_end - phase2_end) * 0.5)
        try:
            best_seq, best_score = _slsqp_polish(best_seq, slsqp_stop)
            exact = float(evaluate_sequence([float(x) for x in np.clip(best_seq, 0, 1000)]))
            if math.isfinite(exact) and exact < best_score:
                best_score = exact
        except Exception:
            pass

        # More SLSQP with different starting point
        if time.monotonic() < phase3_end - 8.0:
            slsqp_stop2 = min(phase3_end - 4.0, time.monotonic() + 60.0)
            start2 = best_seq * (1 + rng_main.normal(0, 0.02, n))
            start2 = np.clip(start2, 0, None)
            try:
                res2, sc2 = _slsqp_polish(start2, slsqp_stop2, max_iter=200)
                if sc2 < best_score:
                    best_seq = res2
                    best_score = sc2
            except Exception:
                pass

    # Phase 3: Coordinate descent polish
    if time.monotonic() < deadline - 8.0:
        coord_stop = min(deadline - 5.0, time.monotonic() + (deadline - phase3_end) * 0.8)
        try:
            best_seq, best_score = _coord_polish(best_seq, coord_stop, seed=777)
        except Exception:
            pass

    # Final short LP if time allows
    if time.monotonic() < deadline - 5.0:
        try:
            res, sc = _lp_optimize(best_seq, n, min(deadline - 3.0, time.monotonic() + 30.0),
                                   seed + 54321, lp_time=5.0)
            if np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_c = np.clip(res, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in res_c]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = res_c.copy()
                    best_score = exact
        except Exception:
            pass

    # Final coordinate polish
    if time.monotonic() < deadline - 3.0:
        try:
            best_seq, best_score = _coord_polish(best_seq, min(deadline - 2.0, time.monotonic() + 20.0), seed=1234)
        except Exception:
            pass

    # Final validation
    final_seq = np.clip(best_seq, 0, 1000)
    final_seq = np.where(np.isfinite(final_seq), final_seq, 0.0)
    if final_seq.sum() < 0.01:
        final_seq = np.ones(1)

    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        fb = np.clip(parent, 0, 1000)
        try:
            fb_sc = float(evaluate_sequence([float(x) for x in fb]))
        except Exception:
            fb_sc = math.inf
        if math.isfinite(fb_sc):
            return [float(x) for x in fb]
        return [1.0]

    return [float(x) for x in final_seq]
