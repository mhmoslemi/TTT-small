    M = float(conv.max())
    # Find active lags (within 2% of max)
    thresh = 0.98 * M
    active = np.where(conv >= thresh)[0]
    if len(active) == 0:
        return f.copy()
    # Subgradient: for each active lag k, d(conv)_k / d(a_i) = 2*a[k-i] (valid range)
    subgrad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            subgrad[i] += 2.0 * f[k - i]
    subgrad /= float(len(active))
    f_new = f - learning_rate * subgrad
    f_new = np.clip(f_new, 0, None)
    S = f_new.sum()
    if S < 1e-10:
        return f.copy()
    return f_new / S


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    n_p = len(parent)

    # Fallback
    try:
        fb_score = float(evaluate_sequence([float(x) for x in parent]))
    except Exception:
        fb_score = math.inf
    if not math.isfinite(fb_score):
        parent = np.ones(384)
        n_p = 384
        fb_score = float(evaluate_sequence([float(x) for x in parent]))

    best_seq = parent.copy()
    best_score = fb_score
    rng = np.random.default_rng(seed)

    def _track(f):
        nonlocal best_seq, best_score
        if f is None or len(f) == 0:
            return
        f = np.clip(np.asarray(f, dtype=np.float64), 0, 1000.0)
        if not np.all(np.isfinite(f)):
            return
        if f.sum() < 0.01:
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in f]))
        except Exception:
            sc = math.inf
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_seq = f.copy()

    # Phase 1: LP optimization on parent length (60% of budget)
    phase1_end = t0 + budget * 0.60
    n = n_p

    # Multiple initializations
    inits = []
    inits.append(parent.copy())

    # Cosine window
    i_arr = np.arange(n, dtype=np.float64)
    cos_w = 0.5 + 0.5 * np.cos(np.pi * i_arr / (n - 1))
    inits.append(cos_w)

    # Slight perturbation of parent
    noisy = np.clip(parent + rng.normal(0, 0.1 * max(1.0, float(np.mean(parent))), n), 0, None)
    inits.append(noisy)

    # Gaussian centered
    gauss = np.exp(-((i_arr - n / 2) / (n * 0.25)) ** 2)
    inits.append(gauss)

    # Allocate time
    phase1_dur = max(5.0, phase1_end - t0)
    time_alloc = [0.45, 0.20, 0.20, 0.15]

    for idx, init in enumerate(inits):
        if time.monotonic() >= phase1_end - 3.0:
            break
        alloc_t = time_alloc[idx] if idx < len(time_alloc) else 0.1
        stop = min(phase1_end - 4.0, time.monotonic() + phase1_dur * alloc_t)
        if time.monotonic() >= stop:
            continue
        try:
            res_f, res_sc = _lp_loop(init, stop, seed + 1000 * (idx + 1))
            _track(res_f)
        except Exception:
            pass

    # Gradient refinement on best
    if time.monotonic() < phase1_end - 5.0:
        try:
            f_cur = best_seq.copy()
            S = f_cur.sum()
            f_cur = f_cur / S if S > 1e-10 else f_cur
            lr = 0.01
            for _ in range(100):
                if time.monotonic() > phase1_end - 3.0:
                    break
                f_new = _gradient_step(f_cur, lr, n)
                sc = _score(f_new)
                if sc < _score(f_cur) - 1e-14:
                    f_cur = f_new
                    _track(f_cur)
                    lr = min(0.05, lr * 1.02)
                else:
                    lr = max(0.001, lr * 0.7)
        except Exception:
            pass

    # Phase 2: Polish (25% of budget)
    phase2_end = t0 + budget * 0.85
    if time.monotonic() < phase2_end - 10.0:
        try:
            pol_f, pol_sc = _polish(best_seq, phase2_end - 5.0, seed + 777)
            _track(pol_f)
        except Exception:
            pass

    # Phase 3: Second LP round on the polished sequence (10%)
    if time.monotonic() < deadline - 20.0:
        stop3 = min(deadline - 15.0, time.monotonic() + budget * 0.08)
        try:
            res_f, res_sc = _lp_loop(best_seq, stop3, seed + 9999)
            _track(res_f)
        except Exception:
            pass

    # Phase 4: Final polish
    if time.monotonic() < deadline - 8.0:
        try:
            pol_f, pol_sc = _polish(best_seq, deadline - 5.0, seed + 5555)
            _track(pol_f)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    return [float(x) for x in best_seq]
