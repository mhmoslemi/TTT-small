                unaff_max = max(unaff_max, float(b[k_end:kn].max()))
            
            aff_b = b[k_start:k_end]
            aff_len = k_end - k_start
            infl = f[:aff_len]  # f[k-i] for k in [i, i+aff_len-1] = f[0], f[1], ..., f[aff_len-1]
            
            v_ref = max(v_old, 2.0/n)
            v_max = min(30.0 * v_ref, 30.0)
            
            # Vectorized grid search
            n_grid = 200
            v_grid = np.linspace(0.0, v_max, n_grid)
            deltas = v_grid - v_old
            S_new_grid = S_rest + v_grid
            
            # For each grid point, compute max(aff_b + 2*delta*infl) (ignoring center correction for now)
            # Shape: (n_grid, aff_len)
            # This might use too much memory: 200 * 384 * 8 = 614KB - fine
            
            # Actually, let's do it more carefully to handle the center correction
            best_v = v_old
            best_obj = best_score
            valid = S_new_grid > 0.01
            if not np.any(valid):
                continue
            
            # Vectorized: compute new_aff for all grid points
            # new_aff[g, k] = aff_b[k] + 2 * deltas[g] * infl[k]
            # Center correction: if k_center in range, new_aff[g, center_idx] += deltas[g]^2
            
            # Memory: n_grid × aff_len = 200 × 384 = 76800 floats = 614KB. Fine.
            new_aff = aff_b[np.newaxis, :] + 2.0 * deltas[:, np.newaxis] * infl[np.newaxis, :]
            
            if k_start <= k_center < k_end:
                ci = k_center - k_start
                new_aff[:, ci] += deltas ** 2
            
            m_aff_grid = new_aff.max(axis=1)
            m_new_grid = np.maximum(unaff_max, m_aff_grid)
            
            with np.errstate(divide='ignore', invalid='ignore'):
                obj_grid = 2.0 * n * m_new_grid / (S_new_grid ** 2)
            obj_grid[~valid] = math.inf
            obj_grid[~np.isfinite(obj_grid)] = math.inf
            
            gi = int(np.argmin(obj_grid))
            if obj_grid[gi] < best_obj - 1e-15:
                best_obj = float(obj_grid[gi])
                best_v = float(v_grid[gi])
            
            # Fine search around best
            if best_v != v_old:
                step = v_max / n_grid
                v_fine = np.linspace(max(0.0, best_v - 5*step), min(v_max, best_v + 5*step), 100)
                d_fine = v_fine - v_old
                S_fine = S_rest + v_fine
                valid_f = S_fine > 0.01
                
                new_aff_f = aff_b[np.newaxis, :] + 2.0 * d_fine[:, np.newaxis] * infl[np.newaxis, :]
                if k_start <= k_center < k_end:
                    ci = k_center - k_start
                    new_aff_f[:, ci] += d_fine ** 2
                m_aff_f = new_aff_f.max(axis=1)
                m_new_f = np.maximum(unaff_max, m_aff_f)
                with np.errstate(divide='ignore', invalid='ignore'):
                    obj_f = 2.0 * n * m_new_f / (S_fine ** 2)
                obj_f[~valid_f] = math.inf
                obj_f[~np.isfinite(obj_f)] = math.inf
                
                gi_f = int(np.argmin(obj_f))
                if obj_f[gi_f] < best_obj - 1e-16:
                    best_obj = float(obj_f[gi_f])
                    best_v = float(v_fine[gi_f])
            
            if best_v != v_old and best_obj < best_score - 1e-14:
                delta = best_v - v_old
                b[k_start:k_end] = aff_b + 2.0 * delta * infl
                if k_start <= k_center < k_end:
                    b[k_center] += delta * delta
                f[i] = best_v
                S += delta
                best_score = best_obj
                improved = True
        
        if not improved:
            break
    
    f = np.clip(f, 0, 1000)
    if f.sum() < 0.01:
        return f, math.inf
    b = _conv_full(f)
    final_score = 2.0 * n * float(b.max()) / (float(f.sum()) ** 2)
    return f, final_score


def _lp_direction(f, n, time_limit=5.0):
    """LP-based direction finding."""
    b = _conv_full(f)
    M = float(b.max())
    if M < 1e-15:
        return None
    # Build A: A[k, j] = f[j] for j in [max(0,k-n+1), min(n-1,k)]
    # Actually: (f*g)[k] = sum_j f[j]*g[k-j] for valid j, k-j
    # So A[k, j] = f[j] if 0 <= k-j <= n-1, i.e., k-n+1 <= j <= k
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)]*n, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-15:
        return None
    return g / s


def _lp_step(f, n, stop_time, seed, max_iters=50):
    """LP-based optimization with line search."""
    rng = np.random.default_rng(seed)
    f = f.copy()
    best_score = _score(f)
    best_f = f.copy()
    t = 0.2
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break
        g = _lp_direction(f, n, time_limit=4.0)
        if g is None:
            stall += 3
            if stall > 50:
                break
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            if f.sum() < 0.01:
                f = best_f.copy()
            continue
        if np.linalg.norm(g - f) < 1e-8:
            stall += 3
            if stall > 50:
                break
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            if f.sum() < 0.01:
                f = best_f.copy()
            continue
        
        improved = False
        for _ in range(10):
            f_new = (1-t)*f + t*g
            if f_new.sum() < 0.01:
                t *= 0.5
                continue
            sc = _score(f_new)
            if sc < best_score - 1e-13:
                f = f_new
                best_f = f.copy()
                best_score = sc
                t = min(0.8, t * 1.1)
                stall = 0
                improved = True
                break
            else:
                t *= 0.5
        if not improved:
            stall += 1
            if stall > 30:
                f = best_f + rng.normal(0, 0.08, n)
                f = np.clip(f, 0, None)
                if f.sum() < 0.01:
                    f = best_f.copy()
                stall = 0
    
    return best_f, best_score
