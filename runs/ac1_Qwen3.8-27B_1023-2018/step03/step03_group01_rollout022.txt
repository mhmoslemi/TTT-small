        if time.monotonic() > lp_end - 4.0:
            break
        if best_score < 1.503:
            break
        stop = min(lp_end - 3.0, time.monotonic() + time_per_init)
        try:
            res, sc = _lp_loop(init, n, stop, seed + 731 * (idx + 1))
            _try_update(res)
        except Exception:
            pass

    # Phase 2: Subgradient polish (12%)
    sg_end = min(deadline, t0 + budget * 0.82)
    if time.monotonic() < sg_end - 8.0:
        try:
            res, sc = _sg_polish(best_seq, len(best_seq), sg_end, seed + 444001)
            _try_update(res)
        except Exception:
            pass

    # Phase 3: Try other lengths (10%)
    other_end = min(deadline, t0 + budget * 0.92)
    if time.monotonic() < other_end - 8.0 and best_score > 1.504:
        other_lengths = [256, 320, 448, 512]
        for L in other_lengths:
            if time.monotonic() > other_end - 5.0:
                break
            if L == n_p:
                continue
            stop = min(other_end - 4.0, time.monotonic() + 20.0)
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n_p), best_seq)
            try:
                res, sc = _lp_loop(init_l, L, stop, seed + 880000 + L)
                _try_update(res)
            except Exception:
                pass

    # Phase 4: Coordinate polish (6%)
    polish_end = deadline - 4.0
    if time.monotonic() < polish_end - 5.0:
        try:
            res, sc = _coord_polish(best_seq, len(best_seq), polish_end, seed + 555001)
            _try_update(res)
        except Exception:
            pass

    # Phase 5: Final subgradient burst
    if time.monotonic() < deadline - 3.0:
        try:
            res, sc = _sg_polish(best_seq, len(best_seq), deadline - 2.0, seed + 666001)
            _try_update(res)
        except Exception:
            pass

    # ---------- Final validation and return ----------
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        sc = evaluate_sequence(final_list)
        if math.isfinite(sc) and sc > 0 and len(final_list) > 0:
            return final_list
    except Exception:
        pass

    # Fallback: parent
    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass

    return [1.0] * n_p
