def _fflen(m: int) -> int:
    m = int(m)
    if m <= 0:
        return 1
    v = _flen_cache.get(m)
    if v is None:
        v = int(next_fast_len(m))
        _flen_cache[m] = v
    return v


def _conv_same(a: np.ndarray) -> np.ndarray:
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        v = float(a[0]) * float(a[0])
        return np.array([max(0.0, v)], dtype=np.float64)
    m = 2 * n - 1
    L = _fflen(m)
    fa = rfft(a, L)
    c = irfft(fa * fa, L)[:m]
    return np.maximum(c, 0.0)


def _conv2(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    n1 = int(a.size)
    n2 = int(b.size)
    if n1 == 0 or n2 == 0:
        return np.empty(0, dtype=np.float64)
    m = n1 + n2 - 1
    L = _fflen(m)
    c = irfft(rfft(a, L) * rfft(b, L), L)[:m]
    return np.maximum(c, 0.0)


def _valid_conv(lam: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    c = _conv2(lam, a[::-1])
    return c[n - 1:2 * n - 1]

    vals2 = np.max(base + lin * ts2[:, None] + quad * (ts2[:, None] ** 2), axis=1)
    if not np.all(np.isfinite(vals2)):
        vals2 = np.where(np.isfinite(vals2), vals2, 1e100)
    j = int(np.argmin(vals2))
    return float(ts2[j]), float(vals2[j])


def _lp_one(f: np.ndarray):
    f = _normalize(f)
    n = int(f.size)
    if n < 2 or n > 700:
        return None, math.inf, True

    cf = _conv_same(f)
    M = float(np.max(cf)) if cf.size else 1.0
    if not math.isfinite(M) or M <= 1e-300:
        return None, math.inf, True

    cur_sc = _score_np(f)
    fmax = float(np.max(f))
    floor = max(1e-12, 1e-12 * fmax)
    fp = np.maximum(f, floor)

    T = M * (1.0 + 1e-12) + 1e-18
    b_ub = T * np.ones(2 * n - 1, dtype=np.float64)

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = fp

    c = -np.ones(n, dtype=np.float64)
    cap = 1.0e6
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0.0, cap)] * n,
            method="highs",
            options={"presolve": True},
        )
    except Exception:
        return None, math.inf, True

    if res is None:
        return None, math.inf, True
    status = int(getattr(res, "status", 1))
    x = getattr(res, "x", None)
    if status != 0 or x is None:
        return None, math.inf, True

    g = np.asarray(x, dtype=np.float64).ravel()
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.clip(g, 0.0, cap)
    gs = float(g.sum())
    if not math.isfinite(gs) or gs <= 1e-12:
        return None, cur_sc, False
    g = g / gs

    if np.allclose(g, f, rtol=1e-10, atol=1e-12):
        return None, cur_sc, False

    cfs = _conv2(f, g)
    cg = _conv_same(g)

    t0, _ = _best_blend_t(cf, cfs, cg)
    cands = set()
    for tt in (0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.75, 1.0):
        cands.add(float(tt))
    for tt in (t0, 0.5 * t0, 1.5 * t0):
        if math.isfinite(tt):
            cands.add(float(np.clip(tt, 0.0, 1.0)))

    best_t = 0.0
    best_sc = cur_sc
    best_new = None

    for tt in sorted(cands):
        if tt <= 1e-14 or tt >= 0.999999999:
            new = g
        else:
            new = (1.0 - tt) * f + tt * g
            snew = float(new.sum())
            if snew <= 1e-12:
                continue
            new = new / snew
        sc = _score_np(new)
        if math.isfinite(sc) and sc < best_sc - 1e-13:
            best_sc = sc
            best_new = new
            best_t = tt

    if best_new is None or not math.isfinite(best_sc):
        return None, cur_sc, False
    return best_new, best_sc, False


def _lp_loop(f0: np.ndarray, stop: float, seed: int = 0, max_noimp: int = 60):
    f = _normalize(f0)
    n = int(f.size)
    best = f.copy()
    best_sc = _score_np(f)
    f_sc = best_sc
    rng = np.random.default_rng(seed)
    noimp = 0
    fail_count = 0
    min_rem = 90.0 if n > 350 else 45.0

    while time.monotonic() < stop - min_rem:
        new, sc, failed = _lp_one(f)

        if failed:
            fail_count += 1
            noimp += 1
            if fail_count >= 8:
                break

        elif new is not None and math.isfinite(sc):
            fail_count = 0
            if sc < best_sc - 1e-12:
                best = np.asarray(new, dtype=np.float64).copy()
                best_sc = sc
                f = best.copy()
                f_sc = sc
                noimp = 0
            elif sc < f_sc - 1e-13:
                f = np.asarray(new, dtype=np.float64).copy()
                f_sc = sc
                noimp = 0
            else:
                noimp += 1
        else:
            noimp += 1

        if noimp > 0:
            sd = (0.035 / math.sqrt(max(1, n))) * (1.0 + 0.10 * min(float(noimp), 15.0))
            fpert = f + rng.normal(0.0, sd, n)
            fpert = np.where(np.isfinite(fpert), fpert, 0.0)
            fpert = np.clip(fpert, 0.0, None)
            sp = float(fpert.sum())
            if sp <= 1e-12:
                fpert = f.copy()
            else:
                fpert = fpert / sp
            f = _normalize(fpert)
            f_sc = _score_np(f)
            if math.isfinite(f_sc) and f_sc < best_sc - 1e-12:
                best = f.copy()
                best_sc = f_sc
                noimp = 0

            if noimp >= max_noimp:
                break

    return best, best_sc, f


def _coordinate_polish(a0: np.ndarray, stop: float, seed: int = 1, max_passes: int = 3):
    a = _normalize(a0)
    n = int(a.size)
    best = a.copy()
    best_sc = _score_np(best)
    if n < 2:
        return best, best_sc

    if n > 700:
        max_passes = min(max_passes, 1)
    elif n > 400:
        max_passes = min(max_passes, 2)

    rng = np.random.default_rng(seed)
    mults = (0.0, 0.15, 0.35, 0.55, 0.75, 1.0, 1.15, 1.35, 1.7, 2.2, 3.2, 5.0)

    for _ in range(max_passes):
        if time.monotonic() > stop - 0.25:
            break
        improved = False
        order = rng.permutation(n)
        for idx in order:
            if time.monotonic() > stop - 0.25:
                break

            orig = float(best[idx])
            mean = 1.0 / float(n)
            base = orig if orig > 1e-15 else 0.25 * mean

            vals = {orig}
            for mm in mults:
                v = base * float(mm)
                if 0.0 <= v <= 1000.0:
                    vals.add(float(v))

            if orig > 0.0:
                for dd in (0.2, 0.5, 1.0, 2.0):
                    v = orig * (1.0 + dd)
                    if 0.0 <= v <= 1000.0:
                        vals.add(float(v))
                    v = orig * (1.0 - dd)
                    if v > 0.0:
                        vals.add(float(v))
            if orig <= 0.0:
                for mm in (1.0, 2.0, 4.0, 8.0):
                    v = 0.25 * mean * float(mm)
                    if 0.0 <= v <= 1000.0:
                        vals.add(float(v))

            local_best_val = orig
            local_best_sc = best_sc
            local_cand = None

            for val in vals:
                val = float(np.clip(val, 0.0, 1000.0))
                cand = best.copy()
                cand[idx] = val
                s = float(cand.sum())
                if s <= 1e-12:
                    continue
                cand /= s
                sc = _score_np(cand)
                if math.isfinite(sc) and sc < local_best_sc - 1e-14:
                    local_best_sc = sc
                    local_best_val = val
                    local_cand = cand

            if local_cand is not None and local_best_sc < best_sc - 1e-13:
                best = local_cand
                best_sc = local_best_sc
                improved = True

        if not improved:
            break

    return best, best_sc


def _safe_return(best, fallback=None) -> list:
    try:
        a = _normalize(best)
        if a.size == 0:
            raise ValueError
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if not math.isfinite(s) or s < 0.01:
            raise ValueError
        lst = [float(x) for x in a]
        if math.isfinite(_exact_score(lst)):
            return lst
    except Exception:
        pass

    if fallback is not None:
        try:
            f = _normalize(fallback)
            f = np.where(np.isfinite(f), f, 0.0)
            f = np.clip(f, 0.0, 1000.0)
            if f.size > 0 and float(f.sum()) >= 0.01:
                lst = [float(x) for x in f]
                if math.isfinite(_exact_score(lst)):
                    return lst
        except Exception:
            pass

    return [1.0 / 256.0] * 256


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(5.0, float(budget_s))
    except Exception:
        budget = 1300.0
    hard = t0 + budget
    reserve = 30.0 if budget > 120.0 else max(3.0, 0.08 * budget)
    deadline = hard - reserve

    parent = _get_parent()
    if parent is None:
        base_len = 256
        parent = _normalize(np.ones(base_len, dtype=np.float64))
    else:
        base_len = int(parent.size)
        if base_len > 700:
            parent = _interp_parent(parent, 512)
            base_len = 512
        elif base_len < 16:
            parent = _interp_parent(parent, 256)
            base_len = 256

    best = parent.copy()
    best_score = _exact_score(best)
    if not math.isfinite(best_score):
        best_score = _score_np(best)
    if not math.isfinite(best_score):
        best = _normalize(np.ones(base_len, dtype=np.float64))
        best_score = _score_np(best)

    fallback = best.copy()
    cur = best.copy()
    rng = np.random.default_rng(seed)

    def _consider(a):
        nonlocal best, best_score
        if a is None:
            return
        a = _normalize(a)
        if a.size == 0:
            return
        si = _score_np(a)
        if not math.isfinite(si):
            return
        if si >= best_score - 1e-10:
            return
        se = _exact_score(a)
        if math.isfinite(se) and se < best_score - 1e-12:
            best = a.copy()
            best_score = se

    if budget < 80.0 or deadline - time.monotonic() <= 10.0:
        stop = min(deadline - 2.0, time.monotonic() + max(5.0, 0.7 * budget))
        if stop - time.monotonic() > 3.0:
            a, _ = _coordinate_polish(best, stop, seed=123, max_passes=1)
            _consider(a)
        return _safe_return(best, fallback)

    # Quick initialization selection at the base length.
    inits = _make_inits(base_len, parent, rng)
    cur_sc = _score_np(cur)
    for a in inits:
        si = _score_np(a)
        if math.isfinite(si) and si < cur_sc - 1e-12:
            cur = a
            cur_sc = si
    _consider(cur)

    # Modest soft-max refinement of the strongest start.
    M = cur_sc / (2.0 * float(base_len)) if math.isfinite(cur_sc) and cur_sc > 0 else 0.004
    etasA = _make_etas(M, (0.25, 0.75, 2.0))
    stopA = min(deadline - max(300.0, 0.40 * budget), time.monotonic() + min(120.0, 0.18 * budget))
    if stopA - time.monotonic() > 20.0:
        maxfunA = 800 if base_len <= 320 else 380
        a, sc, it = _soft_minimize(cur, etasA, stopA, maxfun=maxfunA)
        _consider(a)
        if a is not None:
            cur = _normalize(a)
            sc_cur = sc
        else:
            sc_cur = _score_np(cur)

    # Main LP/directional phase at the base length.
    lp_stop = min(deadline - 80.0, t0 + 0.66 * budget)
    if base_len <= 700 and lp_stop - time.monotonic() > 40.0:
        a, sc, it = _lp_loop(
            cur,
            lp_stop,
            seed=int(rng.integers(1, 1_000_000)),
            max_noimp=80,
        )
        _consider(a)
        if a is not None:
            cur = _normalize(a)

    # Sharper soft-max polish from the current best local basin.
    cur_sc = _score_np(cur)
    ncur = int(cur.size)
    M = cur_sc / (2.0 * float(ncur)) if math.isfinite(cur_sc) and cur_sc > 0 else 0.003
    soft_stop = min(deadline - 160.0, time.monotonic() + min(280.0, 0.22 * budget))
    if soft_stop - time.monotonic() > 35.0:
        etasC = _make_etas(M, (1.0, 2.0, 4.0, 8.0))
        maxfunC = 700 if ncur <= 320 else 330
        a, sc, it = _soft_minimize(cur, etasC, soft_stop, maxfun=maxfunC)
        _consider(a)
        if a is not None:
            cur = _normalize(a)

    # Coordinate polish on the recorded best candidate.
    coord_stop = min(deadline - 120.0, time.monotonic() + min(70.0, 0.06 * budget))
    if coord_stop - time.monotonic() > 18.0:
        a, _ = _coordinate_polish(best, coord_stop, seed=987654321, max_passes=3)
        _consider(a)

    # Optional exploratory restart at the current base length.
    if deadline - time.monotonic() > 280.0 and base_len <= 600:
        inits2 = _make_inits(base_len, best, rng)
        if len(inits2) > 1:
            k = int(rng.integers(1, len(inits2)))
            cand = inits2[k]
            if np.linalg.norm(cand - best) > 1e-7:
                stop_r = min(deadline - 150.0, time.monotonic() + 70.0)
                if stop_r - time.monotonic() > 25.0:
                    sc_r = _score_np(cand)
                    M_r = sc_r / (2.0 * float(base_len)) if math.isfinite(sc_r) and sc_r > 0 else 0.003
                    etas_r = _make_etas(M_r, (0.5, 1.5, 3.0))
                    a, sc, it = _soft_minimize(cand, etas_r, stop_r, maxfun=380)
                    _consider(a)
                    if a is not None and deadline - time.monotonic() > 120.0:
                        lp_r = min(deadline - 100.0, time.monotonic() + 45.0)
                        if lp_r - time.monotonic() > 25.0:
                            a2, sc2, it2 = _lp_loop(a, lp_r, seed=int(rng.integers(1, 1_000_000)), max_noimp=25)
                            _consider(a2)

    # Try a coarser length if enough time remains.  This often helps when the
    # supplied 256-grid construction is already close to the discrete limit.
    if deadline - time.monotonic() > 230.0:
        n2 = 512
        if best.size == n2:
            start5 = best.copy()
        else:
            start5 = _interp_parent(best, n2)

        _consider(start5)
        sc5 = _score_np(start5)

        stop5a = min(deadline - 120.0, time.monotonic() + 80.0)
        if stop5a - time.monotonic() > 20.0:
            M5 = sc5 / (2.0 * float(n2)) if math.isfinite(sc5) and sc5 > 0 else 0.0015
            etas5 = _make_etas(M5, (0.5, 1.0, 2.0))
            a5, sc5, it5 = _soft_minimize(start5, etas5, stop5a, maxfun=420)
            _consider(a5)
            if a5 is not None:
                start5 = _normalize(a5)

        if n2 <= 700 and deadline - time.monotonic() > 180.0:
            start_lp5 = best if best.size == n2 else start5
            stop5lp = min(deadline - 90.0, time.monotonic() + 80.0)
            if stop5lp - time.monotonic() > 40.0:
                a5, sc5, it5 = _lp_loop(start_lp5, stop5lp, seed=int(rng.integers(1, 1_000_000)), max_noimp=30)
                _consider(a5)
                if a5 is not None:
                    start5 = _normalize(a5)

        if deadline - time.monotonic() > 100.0:
            startS = best if best.size == n2 else start5
            scS = _score_np(startS)
            M5 = scS / (2.0 * float(n2)) if math.isfinite(scS) and scS > 0 else 0.0015
            etas5b = _make_etas(M5, (1.0, 2.0, 4.0))
            stop5b = min(deadline - 50.0, time.monotonic() + 50.0)
            if stop5b - time.monotonic() > 18.0:
                a5, sc5, it5 = _soft_minimize(startS, etas5b, stop5b, maxfun=280)
                _consider(a5)

    # Final safe polish.
    final_stop = deadline - 5.0
    if final_stop - time.monotonic() > 15.0:
        maxp = 2 if best.size > 500 else 3
        a, _ = _coordinate_polish(best, final_stop, seed=31337, max_passes=maxp)
        _consider(a)

    if deadline - time.monotonic() > 35.0 and best.size <= 600:
        scb = _score_np(best)
        Mb = scb / (2.0 * float(best.size)) if math.isfinite(scb) and scb > 0 else 0.002
        etasF = _make_etas(Mb, (2.0, 4.0))
        stopF = min(deadline - 4.0, time.monotonic() + max(10.0, 0.5 * (deadline - time.monotonic())))
        if stopF - time.monotonic() > 10.0:
            a, sc, it = _soft_minimize(best, etasF, stopF, maxfun=220)
            _consider(a)

    return _safe_return(best, fallback)
