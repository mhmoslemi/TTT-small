    F = rfft(a, fs); c = irfft(F*F, fs)[:2*n-1]
    np.maximum(c, 0, out=c); return c

def _lp_dir(f, n, M, tlim=3.0):
    """LP: maximize sum(g) s.t. f*g <= M, g >= 0."""
    A = np.zeros((2 * n - 1, n))

        A[j:j + n, j] = f
    c = -np.ones(n)

                    bounds=[(0.0, None)] * n,
                    method='highs',
                    options={'presolve': True, 'time_limit': tlim})
    except Exception:
        return None

        return None
    g = np.clip(np.asarray(r.x), 0.0, 1000.0)
