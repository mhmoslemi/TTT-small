    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None, M
    return g / g_sum, M

def _lp_optimize(f0, n, stop_time, seed, max_iters=200):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n].copy()
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum

    best_f = f.copy()
    best_score = _score(f)
    
    t = 0.2
    t_min = 0.02
    t_max = 0.7
    stall = 0
    max_stall = 80
    iteration = 0

    while time.monotonic() < stop_time - 3.0 and iteration < max_iters:
        g_dir, M = _lp_direction(f, n, time_limit=4.0)
        if g_dir is None:
            stall += 5
            if stall > max_stall:
                break
            # Random perturbation
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            iteration += 1
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-6:
            stall += 3
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            iteration += 1
            continue

        improved = False
        for _ in range(15):
            f_new = (1.0 - t) * f + t * g_dir
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 1e-10:
                t = max(t_min, t * 0.5)
                continue
            sc = _score(f_new)
            if sc < best_score - 1e-14:
                f = f_new / fs_new  # normalize
                # Actually, we shouldn't normalize here since score is scale-invariant
                f = f_new
                best_f = f.copy()
                best_score = sc
                improved = True
                t = min(t_max, t * 1.08)
                stall = 0
                break
            else:
                t = max(t_min, t * 0.5)

        if not improved:
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs
                else:
                    f = best_f.copy()
                t = 0.15
                stall = 0
        iteration += 1

    return best_f, best_score

def _coord_descent(f, stop_time, max_sweeps=100, seed=777):
    n = len(f)
    f = np.clip(f.astype(np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, math.inf
    
    conv = _conv_full(f)
    best_score = 2.0 * n * conv.max() / (S * S)
    rng = np.random.default_rng(seed)
    
    # Precompute the "influence" vectors for each coordinate
    # infl_i[k] = 2*f[k-i] for k in [i, i+n-1] (clipped to [0, 2n-2])
    # Plus the diagonal correction at k=2i
    
    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break
            
            v = f[i]
            S_rest = S - v
            
            # Influence vector
            start = i
            end = min(i + n, 2 * n - 1)
            L = end - start
            infl = 2.0 * f[:L]  # f[k-i] for k in [i, i+L-1]
            
            # Determine search range for new value x
            if v > 1e-10:
                x_lo = max(0.0, v * 0.01)
                x_hi = min(200.0, max(10.0 * v, S * 0.1))
            else:
                x_lo = 0.0
                x_hi = S * 0.05
            
            # Coarse grid
            n_coarse = 100
            x_grid = np.linspace(x_lo, x_hi, n_coarse)
            
            best_x = v
            best_obj = best_score
            
            # Vectorized evaluation
            # For each x, delta = x - v
            # new_conv[k] = conv[k] + delta * infl[k-start] for k in [start, end)
            # new_conv[2i] += delta^2 (if 2i in [start, end))
            # new_S = S_rest + x
            # obj = 2n * max(new_conv) / new_S^2
            
            deltas = x_grid - v
            # new_conv = conv + outer(deltas, infl_at_positions)
            # This is memory-heavy for large n. Use a loop for the max.
            
            for idx_x in range(n_coarse):
                x = x_grid[idx_x]
                S_new = S_rest + x
                if S_new < 0.01:
                    continue
                w = x - v
                if abs(w) < 1e-14:
                    m_new = conv.max()
                else:
                    # Update only the affected part
                    # max of (conv[:start], conv[start:end] + w*infl, conv[end:] + w^2 correction at 2i)
                    m_left = conv[:start].max() if start > 0 else 0.0
                    affected = conv[start:end].copy()
                    affected += w * infl
                    if start <= 2 * i < end:
                        affected[2 * i - start] += w * w
                    m_affected = affected.max()
                    m_right = conv[end:].max() if end < 2 * n - 1 else 0.0
                    m_new = max(m_left, m_affected, m_right)
                
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_x = x
            
            # Fine grid around best_x
            if best_x != v:
                dw = (x_hi - x_lo) / n_coarse
                n_fine = 50
                x_fine = np.linspace(max(0, best_x - 3 * dw), min(x_hi, best_x + 3 * dw), n_fine)
                for x in x_fine:
                    S_new = S_rest + x
                    if S_new < 0.01:
                        continue
                    w = x - v
                    if abs(w) < 1e-14:
                        continue
                    affected = conv[start:end].copy()
                    affected += w * infl
                    if start <= 2 * i < end:
                        affected[2 * i - start] += w * w
                    m_left = conv[:start].max() if start > 0 else 0.0
                    m_affected = affected.max()
                    m_right = conv[end:].max() if end < 2 * n - 1 else 0.0
                    m_new = max(m_left, m_affected, m_right)
                    obj = 2.0 * n * m_new / (S_new * S_new)
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_x = x
            
            # Accept
            if best_x != v and best_obj < best_score - 1e-14:
                w = best_x - v
                f[i] = best_x
                S += w
                conv[start:end] += w * infl
                if start <= 2 * i < end:
                    conv[2 * i] += w * w
                best_score = best_obj
                improved = True
        
        if not improved:
            break

    f = np
