        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js <= je:
            A[i, js:je+1] = f[k-je:k-js+1][::-1]
    return A

def _solve_lp(f, conv, stop_time, top_k=200):
    n = len(f)
    M = float(conv.max())
    if M < 1e-15:
        return None
    
    # Adaptive: use at most top_k, but at least 50
    top_k = max(50, min(top_k, 2*n-1))
    top_idx = np.argsort(conv)[-top_k:][::-1]
    
    A = _build_A(f, top_idx)
    c_obj = -np.ones(n)
    b_ub = np.full(top_k, M)
    
    rem = stop_time - time.monotonic()
    if rem < 3:
        return None
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                     bounds=[(0, None)] * n,
                     method='highs',
                     options={"presolve": True, "time_limit": max(1, int(rem)-5)})
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s

def _opt_loop(f_init, stop_time, seed):
    rng = np.random.default_rng(seed)
    n = len(f_init)
    f = np.clip(f_init.astype(np.float64), 0, None).copy()
    fs = f.sum()
    f = f/fs if fs > 1e-10 else np.ones(n)/n
    
    best_f = f.copy()
    best_sc, _ = _score_fast(f)
    
    momentum = np.zeros(n)
    mu = 0.3  # momentum coefficient
    stall = 0
    max_stall = 30
    t = 0.25
    
    while time.monotonic() < stop_time - 3:
        conv = _conv(f)
        M = float(conv.max())
        
        g = _solve_lp(f, conv, stop_time, top_k=200)
        if g is None:
            stall += 3
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
            fs = f.sum()
            f = f/fs if fs > 1e-10 else best_f.copy()
            momentum = np.zeros(n)
            continue
        
        d = g - f  # direction
        dn = float(np.linalg.norm(d))
        if dn < 1e-6:
            stall += 2
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            fs = f.sum()
            f = f/fs if fs > 1e-10 else best_f.copy()
            momentum = np.zeros(n)
            continue
        
        # Momentum
        momentum = mu * momentum + (1-mu) * d / dn
        # Use d directly for the step (momentum is for escape)
        
        # Line search
        best_t = 0.0
        best_lc = best_sc
        for tc in [t, t*0.5, t*2.0, 0.1, 0.5, 0.8]:
            tc = max(0.02, min(0.98, tc))
            if time.monotonic() > stop_time - 3:
                break
            f_new = (1-tc)*f + tc*g
            f_new = np.clip(f_new, 0, None)
            fs = f_new.sum()
            if fs < 0.01:
                continue
            sc, _ = _score_fast(f_new)
            if sc < best_lc - 1e-14:
                best_lc = sc
                best_t = tc
        
        if best_t > 0 and best_lc < best_sc - 1e-12:
            f_new = np.clip((1-best_t)*f + best_t*g, 0, None)
            f = f_new / f_new.sum()
            best_f = f.copy()
            best_sc = _score_fast(f)[0]
            stall = 0
            t = min(0.6, t * 1.03)
        else:
            stall += 1
            t = max(0.05, t * 0.75)
            if stall >= max_stall:
                # Escape: use momentum direction
                esc = np.clip(f + 0.1 * momentum + rng.normal(0, 0.05, n), 0, None)
                fs = esc.sum()
                f = esc/fs if fs > 1e-10 else best_f.copy()
                momentum = np.zeros(n)
                t = 0.25
                stall = 0
                if rng.random() < 0.2:
                    f = np.clip(best_f + rng.normal(0, 0.015, n), 0, None)
                    fs = f.sum()
                    f = f/fs if fs > 1e-10 else best_f.copy()
    
    return best_f, best_sc

def _polish(a, stop_time, seed=42):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64), 0, None).copy()
    s = best.sum()
    if s < 1e-10:
        return best, math.inf
    best = best / s
    best_sc, _ = _score_fast(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 10
    
    while time.monotonic() < stop_time - 2:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2:
                break
            orig = best[i]
            bv = orig
            bvs = best_sc
            for m in [0.3, 0.5, 0.7, 0.8, 1.2, 1.5, 2.0, 0.0]:
                nv = orig*m if orig > 1e-15 else (0.01 if m > 0 else 0.0)
                best[i] = nv
                s = best.sum()
                if s < 0.01:
                    best[i] = orig
                    continue
                sc, _ = _score_fast(best)
                if sc < bvs - 1e-14:
                    bv = nv
                    bvs = sc
            best[i] = bv
            if bvs < best_sc - 1e-14:
                best_sc = bvs
                improved = True
        
        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0
    
    return best, best_sc

def propose_candidate(seed=42, budget_s=1300, **kwargs):

    phase1_end = t0 + budget * 0.55

    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.1)

            pass

        try:

        except Exception:

    try:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        fs = math.inf
    if not math.isfinite(fs):
        return [1.0] * n_p
    
    return [float(x) for x in best_seq]
