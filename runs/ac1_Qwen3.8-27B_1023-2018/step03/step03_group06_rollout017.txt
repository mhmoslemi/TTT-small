This is the "valid" cross-correlation of p and reversed... Actually:
`sum_j p[i+j] * a[j]` is the convolution of p (flipped) with a, evaluated at position... 

Let me just use FFT:
- Create array `p_ext` of length L, with p in positions [0, conv_len-1]
- Create array `a_ext` of length L, with a in positions [0, n-1]
- The cross-correlation: `corr[i] = sum_j p[i+j] * a[j]` = `ifft(fft(p_ext) * conj(fft(a_ext)))`... 

Actually, the cross-correlation of p and a is:
`(p ⋆ a)[i] = sum_j p[j] * a[j+i]` (or similar convention)

Let me just use the direct computation for n=256. 256 iterations of dot product of length up to 256 = 65K multiplications. That's negligible.

Actually wait, I can vectorize it:
```python
# p has length 2n-1, a has length n
# grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j] for i=0,...,n-1
# This is the "toeplitz" inner product
# Build a Toeplitz matrix from p: T[i,j] = p[i+j] for i+j < 2n-1, else 0
# Then grad = 2 * T @ a
```

Building a 256x511 Toeplitz matrix and multiplying by a 256-vector is O(n * conv_len) = O(131K). Fast enough.

Or even simpler with numpy:
```python
# Use strided view
from numpy.lib.stride_tricks import as_strided
# Or just:
# For i in range(n): grad[i] = 2 * dot(p[i:i+n], a)
# Vectorized: create matrix where row i is p[i:i+n]
idx = np.arange(n)[:, None] + np.arange(n)[None, :]  # n x n matrix
grad = 2.0 * (p[idx] * a[None, :]).sum(axis=1)
```

This creates an n×n = 65K element index. Fine.

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize, linprog


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1300.0
    deadline = t0 + budget - 15.0

    # --- Get parent ---
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = None
    if parent is None or parent.size == 0:
        parent = np.ones(256, dtype=np.float64) / 256.0
    parent = np.where(np.isfinite(parent), parent, 0.0)
    parent = np.clip(parent, 0.0, 1000.0)
    ps = float(parent.sum())
    if ps < 1e-10:
        parent = np.ones(256, dtype=np.float64) / 256.0
        ps = 256.0
    parent = parent / ps

    best_seq = parent.copy()
    best_score = _safe_exact_score(parent)
    if not math.isfinite(best_score):
        best_score = _internal_score(parent, 256)
    if not math.isfinite(best_score):
        best_score = 2.0

    rng = np.random.default_rng(seed)

    # --- FFT helpers ---
    _fft_cache = {}

    def _get_fft_len(m):
        m = int(m)
        if m not in _fft_cache:
            _fft_cache[m] = int(next_fast_len(m))
        return _fft_cache[m]

    def _conv(a, n):
        cl = 2 * n - 1
        L = _get_fft_len(cl)
        fa = rfft(a, L)
        c = irfft(fa * fa, L)[:cl]
        return np.maximum(c, 0.0)

    def _score(a, n):
        s = float(a.sum())
        if s < 1e-12:
            return float('inf')
        c = _conv(a, n)
        return 2.0 * float(n) * float(c.max()) / (s * s)

    def _safe_exact_score(a):
        try:
            fn = globals().get("evaluate_sequence", None)
            if fn is None:
                return float('inf')
            lst = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
            rv = float(fn(lst))
            return rv if math.isfinite(rv) else float('inf')
        except Exception:
            return float('inf')

    def _internal_score(a, n):
        return _score(a, n)

    def _try_update(arr, n):
        nonlocal best_seq, best_score
        if arr is None:
            return
        a = np.asarray(arr, dtype=np.float64).ravel()
        if a.size != n:
            return
        a = np.clip(a, 0.0, None)
        s = float(a.sum())
        if s < 1e-12:
            return
        a = a / s
        sc_int = _score(a, n)
        if not math.isfinite(sc_int):
            return
        if sc_int >= best_score - 1e-12:
            return
        sc = _safe_exact_score(a)
        if not math.isfinite(sc):
            sc = sc_int
        if math.isfinite(sc) and sc < best_score - 1e-13:
            best_score = sc
            best_seq = a.copy()

    # --- Smooth max (LSE) objective + gradient ---
    def _lse_obj_grad(a, T, n):
        c = _conv(a, n)
        Tc = T * c
        Tc_shift = Tc - Tc.max()
        exp_Tc = np.exp(Tc_shift)
        Z = float(exp_Tc.sum())
        if Z < 1e-300:
            return Tc.max() / T, np.zeros(n)
        p = exp_Tc / Z
        lse_val = (Tc.max() + math.log(Z)) / T
        # grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
        # Build index matrix
        n_half = n
        idx = np.arange(n)[:, None] + np.arange(n)[None, :]  # n x n
        valid = idx < (2 * n - 1)
        P_slice = np.where(valid, p[idx], 0.0)
        grad = 2.0 * (P_slice * a[None, :]).sum(axis=1)
        return lse_val, grad

    # --- L-BFGS refinement with LSE ---
    def _refine_lse(a0, n, T_list, total_time, stop_fn, maxiter=300):
        a = np.asarray(a0, dtype=np.float64).ravel().copy()
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 1e-12:
            return a, float('inf')
        a = a / s
        best_local = a.copy()
        best_local_sc = _score(a, n)

        for T in T_list:
            if stop_fn():
                break
            if time.monotonic() > total_time:
                break

            def fg(y):
                if time.monotonic() > total_time:
                    raise _Timeout
                a_loc = np.clip(np.asarray(y, dtype=np.float64), 0.0, None)
                sl = float(a_loc.sum())
                if sl < 1e-12:
                    return 1e100, np.zeros(n)
                a_loc = a_loc / sl
                val, grad = _lse_obj_grad(a_loc, T, n)
                # Gradient w.r.t. unnormalized y (project to simplex direction)
                # d(a_i)/d(y_i) = (delta_i - a_i) / sum(a)
                # Since we normalize, the effective gradient in y-space:
                # d(val)/dy_i = (grad_i - sum(a*grad)) / sum(a) ... 
                # Actually with a = y/sum(y), dval/dy_i = (grad_i - mean-like) 
                # Let's just use the gradient of val w.r.t. a, clamped:
                g = np.zeros(n)
                dot = float(a_loc @ grad)
                g = grad - a_loc * dot  # project to tangent of sum=1
                g = np.maximum(g, -100.0)  # don't push below 0 too hard
                if not np.all(np.isfinite(g)):
                    g = np.zeros(n)
                return float(val), g

            try:
                res = minimize(fg, a.copy(), jac=True, method='L-BFGS-B',
                               options={'maxiter': maxiter, 'ftol': 1e-14,
                                        'gtol': 1e-10, 'maxls': 40,
                                        'maxfun': int(maxiter * 2)})
            except _Timeout:
                break
            except Exception:
                break

            if hasattr(res, 'x') and res.x is not None:
                a_new = np.clip(res.x, 0.0, None)
                sn = float(a_new.sum())
                if sn > 1e-12:
                    a_new = a_new / sn
                    sc = _score(a_new, n)
                    if math.isfinite(sc) and sc < best_local_sc:
                        best_local_sc = sc
                        best_local = a_new.copy()
                        a = best_local.copy()

        return best_local, best_local_sc

    class _Timeout(Exception):
        pass

    # --- LP-based direction: find g that minimizes max(f*g) subject to norm(g)=1, g>=0 ---
    def _lp_direction(f, n, stop_t):
        """Find a non-negative g (sum=1) that reduces the active convolution values."""
        c = _conv(f, n)
        M = float(c.max())
        if not math.isfinite(M) or M < 1e-15:
            return None

        # Active constraints: lags where c[k] >= 0.98 * M
        active = np.where(c >= 0.98 * M)[0]
        if len(active) == 0:
            return None

        # LP: maximize sum(g) subject to (f*g)[k] <= M for k in active, g >= 0, sum(g)=1
        # (f*g)[k] = sum_{j: 0<=j<n, 0<=k-j<n} f[j]*g[k-j]
        # For g of length n: (f*g) has length 2n-1
        # Constraint: for each active k, sum_{j} f[j] * A[k,j] * g[j] <= M
        # where A[k,j] = 1 if 0 <= k-j < n else 0

        n_vars = n + 1  # g_0..g_{n-1} and slack t
        obj = np.zeros(n_vars)
        obj[n] = 1.0  # maximize t (or minimize -t)

        # Build constraint matrix
        A_ub = np.zeros((len(active), n_vars))
        b_ub = np.zeros(len(active))
        for idx, k in enumerate(active):
            for j in range(n):
                if 0 <= k - j < n:
                    A_ub[idx, j] = f[j]
            A_ub[idx, n] = -1.0  # minus t
            b_ub[idx] = 0.0  # f*g - t <= 0, i.e., f*g <= t

        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 30})
        except Exception:
            return None

        if res is None or getattr(res, 'status', -1) != 0:
            return None
        g = np.asarray(res.x[:n], dtype=np.float64)
        g = np.clip(g, 0.0, None)
        gs = float(g.sum())
        if gs < 1e-12:
            return None
        g = g / gs
        return g

    def _blend_step(f, g, n, stop_t):
        """Find best blend t in f->g."""
        cf = _conv(f, n)
        cg = _conv(g, n)
        # cfs[k] = (f*g)[k] = sum_j f[j]*g[k-j]
        cl = 2 * n - 1
        L = _get_fft_len(cl)
        cfs = irfft(rfft(f, L) * rfft(g, L), L)[:cl]
        cfs = np.maximum(cfs, 0.0)

        base = cf
        bcoef = 2.0 * (cfs - cf)
        gcoef = cf - 2.0 * cfs + cg

        ts = np.linspace(0.0, 1.0, 41)
        vals = np.max(base[None, :] + bcoef[None, :] * ts[:, None] + gcoef[None, :] * ts[:, None] ** 2, axis=1)
        i_best = int(np.argmin(vals))
        lo = ts[max(0, i_best - 2)]
        hi = ts[min(len(ts) - 1, i_best + 2)]
        if hi <= lo:
            hi = min(1.0, lo + 0.05)
            lo = max(0.0, hi - 0.05)
        ts2 = np.linspace(lo, hi, 81)
        vals2 = np.max(base[None, :] + bcoef[None, :] * ts2[:, None] + gcoef[None, :] * ts2[:, None] ** 2, axis=1)
        j_best = int(np.argmin(vals2))
        return float(ts2[j_best])

    # --- Main optimization loop ---
    def _optimize_n(a0, n, time_limit, label=""):
        a = np.asarray(a0, dtype=np.float64).ravel().copy()
        if a.size != n:
            a = _interp_to(a, n)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
            s = 1.0
        a = a / s
        local_best = a.copy()
        local_best_sc = _score(a, n)

        elapsed = lambda: time.monotonic() - t0
        remaining = lambda: time_limit - elapsed()

        # Phase A: LSE gradient refinement
        if remaining() > 30:
            for T in [30.0, 100.0, 300.0, 1000.0, 3000.0]:
                if remaining() < 20:
                    break
                sub_stop = time.monotonic() + min(remaining() - 8, 80.0)
                arr, sc = _refine_lse(a, n, [T], sub_stop,
                                      stop_fn=lambda: time.monotonic() >= sub_stop,
                                      maxiter=200)
                _try_update(arr, n)
                a = arr.copy()
                if sc < local_best_sc:
                    local_best_sc = sc
                    local_best = arr.copy()

        # Phase B: LP-based iterations
        if remaining() > 40:
            noimp = 0
            while remaining() > 25 and noimp < 20:
                g = _lp_direction(a, n, time.monotonic() + min(20, remaining() * 0.1))
                if g is None:
                    # Perturb and retry
                    a = a + rng.normal(0, 0.01 / n, n)
                    a = np.clip(a, 0, None)
                    a = a / a.sum()
                    noimp += 1
                    continue
                t_best = _blend_step(a, g, n, time.monotonic() + 5)
                new_a = (1 - t_best) * a + t_best * g
                new_a = np.clip(new_a, 0, None)
                sn = new_a.sum()
                if sn > 1e-12:
                    new_a = new_a / sn
                    sc_new = _score(new_a, n)
                    if math.isfinite(sc_new) and sc_new < local_best_sc - 1e-14:
                        local_best_sc = sc_new
                        local_best = new_a.copy()
                        a = new_a.copy()
                        noimp = 0
                    else:
                        noimp += 1
                        # Small noise
                        a = a + rng.normal(0, 0.005 / n, n)
                        a = np.clip(a, 0, None)
                        a = a / a.sum()
        return local_best, local_best_sc

    def _interp_to(arr, target_n):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size == target_n:
            out = arr.copy()
        elif arr.size == 0:
            out = np.ones(target_n) / target_n
        else:
            x0 = np.linspace(0, 1, arr.size)
            x1 = np.linspace(0, 1, target_n)
            out = np.interp(x1, x0, arr)
        out = np.clip(out, 0, None)
        s = out.sum()
        if s < 1e-12:
            out = np.ones(target_n) / target_n
        else:
            out = out / s
        return out

    # --- Coordinate polish ---
    def _coord_polish(a0, n, time_limit):
        a = np.asarray(a0, dtype=np.float64).ravel().copy()
        a = np.clip(a, 0, None)
        a = a / a.sum()
        local_best = a.copy()
        local_best_sc = _score(a, n)
        default_val = 1.0 / n
        mults = [0.0, 0.1, 0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5, 4.0]

        for pass_num in range(4):
            if time.monotonic() > time_limit:
                break
            improved = False
            for i in range(n):
                if time.monotonic() > time_limit - 2:
                    break
                orig = a[i]
                base = max(orig, default_val * 0.1)
                best_v = orig
                best_sc = local_best_sc
                for m in mults:
                    v = base * m
                    a[i] = v
                    s = a.sum()
                    if s < 1e-12:
                        a[i] = orig
                        continue
                    a_n = a / s
                    sc = _score(a_n, n)
                    if math.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_v = v
                a[i] = best_v
                s = a.sum()
                a = a / s
                if best_sc < local_best_sc:
                    local_best_sc = best_sc
                    local_best = a.copy()
                    improved = True
            if not improved:
                break
        return local_best, local_best_sc

    # --- Execute optimization ---
    n_parent = int(parent.size)

    # Quick initial score check
    _try_update(parent, n_parent)

    # Main optimization at parent length
    if deadline - time.monotonic() > 30:
        stop1 = min(deadline - 60, time.monotonic() + 0.4 * budget)
        best_seq, _ = _optimize_n(parent, n_parent, stop1, "parent_len")
        _try_update(best_seq, n_parent)

    # LSE refinement with high T
    if deadline - time.monotonic() > 30:
        stop2 = min(deadline - 40, time.monotonic() + min(180, 0.2 * budget))
        arr, _ = _refine_lse(best_seq, n_parent, [500.0, 2000.0, 8000.0], stop2,
                             stop_fn=lambda: False, maxiter=400)
        _try_update(arr, n_parent)

    # More LP iterations
    if deadline - time.monotonic() > 50:
        stop3 = min(deadline - 35, time.monotonic() + min(200, 0.15 * budget))
        a_cur = best_seq.copy()
        noimp = 0
        while time.monotonic() < stop3 and noimp < 30:
            g = _lp_direction(a_cur, n_parent, time.monotonic() + 15)
            if g is None:
                noimp += 1
                a_cur = a_cur + rng.normal(0, 0.008 / n_parent, n_parent)
                a_cur = np.clip(a_cur, 0, None)
                a_cur = a_cur / a_cur.sum()
                continue
            t_b = _blend_step(a_cur, g, n_parent, time.monotonic() + 5)
            new_a = (1 - t_b) * a_cur + t_b * g
            new_a = np.clip(new_a, 0, None)
            sn = new_a.sum()
            if sn > 1e-12:
                new_a = new_a / sn
                sc_new = _score(new_a, n_parent)
                if math.isfinite(sc_new) and sc_new < _score(a_cur, n_parent) - 1e-14:
                    a_cur = new_a
                    noimp = 0
                    _try_update(a_cur, n_parent)
                else:
                    noimp += 1
                    a_cur = a_cur + rng.normal(0, 0.003 / n_parent, n_parent)
                    a_cur = np.clip(a_cur, 0, None)
                    a_cur = a_cur / a_cur.sum()

    # Try other lengths
    other_lengths = [320, 384, 256, 192, 448]
    n_cur = int(best_seq.size)
    other_lengths = [L for L in other_lengths if L != n_cur]
    other_lengths.sort()

    if deadline - time.monotonic() > 80:
        per_time = (deadline - time.monotonic() - 50) / max(1, len(other_lengths))
        per_time = max(25.0, min(per_time, 120.0))
        for L in other_lengths[:3]:
            if time.monotonic() > deadline - 50:
                break
            a_init = _interp_to(best_seq, L)
            stopL = time.monotonic() + per_time
            arr_L, _ = _optimize_n(a_init, L, stopL, f"len_{L}")
            _try_update(arr_L, L)

    # Final LSE polish at best length
    n_best = int(best_seq.size)
    if deadline - time.monotonic() > 40:
        stopF = min(deadline - 20, time.monotonic() + min(100, 0.08 * budget))
        arr_F, _ = _refine_lse(best_seq, n_best, [2000.0, 10000.0], stopF,
                               stop_fn=lambda: False, maxiter=300)
        _try_update(arr_F, n_best)

    # Coordinate polish
    if deadline - time.monotonic() > 20:
        stopC = min(deadline - 10, time.monotonic() + min(60, 0.05 * budget))
        arr_C, _ = _coord_polish(best_seq, n_best, stopC)
        _try_update(arr_C, n_best)

    # Final validation
    seq = np.clip(best_seq, 0.0, 1000.0).ravel()
    if seq.size == 0 or seq.sum() < 0.01:
        seq = np.ones(32) / 32.0
    result = [float(x) for x in seq]
    final_sc = _safe_exact_score(result)
    if not math.isfinite(final_sc):
        result = [float(x) for x in np.clip(parent, 0.0, 1000.0)]
        final_sc = _safe_exact_score(result)
    if not math.isfinite(final_sc):
        result = [1.0 / 32.0] * 32

    return result
```
