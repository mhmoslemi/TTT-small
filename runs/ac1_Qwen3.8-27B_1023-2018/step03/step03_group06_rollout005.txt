import numpy as np

_fft_len_cache = {}


def _fft_len(m):
    m = int(m)
    if m <= 0:
        return 1
    v = _fft_len_cache.get(m)
    if v is None:
        v = int(next_fast_len(m))
        _fft_len_cache[m] = v
    return v


def _conv_same(a):
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    L = _fft_len(m)
    fa = rfft(a, L)
    c = irfft(fa * fa, L)[:m]
    return c


def _conv2(a, b):
    n1, n2 = int(a.size), int(b.size)
    if n1 == 0 or n2 == 0:
        return np.array([], dtype=np.float64)
    m = n1 + n2 - 1
    L = _fft_len(m)
    c = irfft(rfft(a, L) * rfft(b, L), L)[:m]
    return c


def _valid_conv(d, a):
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    nd = int(d.size)
    if nd < n:
        d = np.concatenate([d, np.zeros(n - nd)])
        nd = n
    ar = a[::-1].copy()
    c = _conv2(d, ar)
    return c[n - 1:2 * n - 1]

def _grad_peak(a, k):
    n = int(a.size)
    grad = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if hi >= lo:
        j_idx = k - np.arange(lo, hi + 1)
        j_idx = j_idx[(j_idx >= 0) & (j_idx < n)]
        i_idx = np.arange(lo, hi + 1)[(j_idx >= 0) & (j_idx < n)]
        grad[i_idx] = 2.0 * a[j_idx]
    return grad


def _subgrad_max(a, top_k=3):

    mval = float(c.max())
    n = int(a.size)
    if top_k <= 1:

        return mval, _grad_peak(a, idx)

    g = np.zeros(n, dtype=np.float64)
    for idx in order:
        g += _grad_peak(a, int(idx))
    return mval, g / float(top_k)


def _softmax_from_y(y, baseline_idx, n):
    y = np.asarray(y, dtype=np.float64)
    y = np.nan_to_num(y, nan=-20.0, posinf=20.0, neginf=-20.0)
    y = np.clip(y, -20.0, 20.0)
    ex = np.exp(y)
    Z = 1.0 + float(ex.sum())
    if not math.isfinite(Z) or Z <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    a = np.empty(n, dtype=np.float64)
    b = int(baseline_idx)
    a[:b] = ex[:b] / Z
    a[b] = 1.0 / Z
    if b + 1 < n:
        a[b + 1:] = ex[b:] / Z
    s = float(a.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s

    best = f.copy()
    best_sc = _score(f)
    if n < 2:

    b = int(np.argmax(f))
    idx = np.array([i for i in range(n) if i != b], dtype=int)
    base = max(float(f[b]), 1e-300)
    y0 = np.log(np.maximum(f[idx], 1e-300) / base)
    y0 = np.clip(np.nan_to_num(y0), -20.0, 20.0)

    def fg(y):
        if time.monotonic() > stop:
            raise _Timeout
        yv = np.asarray(y, dtype=np.float64)

        a = _softmax_from_y(yv, b, n)
        c = _conv_same(a)
        maxc = float(c.max())
        if not math.isfinite(maxc) or maxc <= 1e-300:
            return 1e100, np.zeros_like(yv)

        if soft_T > 0.05:
            t_reg = max(soft_T, 1e-6) / maxc
            shifted = c / maxc - 1.0
            max_shifted = 0.0
            log_sum = shifted.max()
            w = np.exp(shifted - log_sum)
            s_w = float(w.sum())
            if s_w <= 0 or not math.isfinite(s_w):
                return 1e100, np.zeros_like(yv)
            w = w / s_w
            fun = math.log(s_w) + log_sum + math.log(maxc)
            grad_a = np.zeros(n, dtype=np.float64)
            for kk in range(len(c)):
                if w[kk] > 1e-10:

def _coord_polish(a0, stop, seed=1, max_passes=3):
    a = _normalize(a0)
    n = int(a.size)
    best = a.copy()
    best_sc = _score(best)
    if n < 2:
        return best, best_sc
    rng = np.random.default_rng(seed)

    for _ in range(max_passes):
        if time.monotonic() > stop - 0.5:
            break
        improved = False
        order = rng.permutation(n)
        default_scale = 1.0 / (4.0 * n)
        mults = (0.0, 0.1, 0.25, 0.5, 0.75, 1.0, 1.3, 1.7, 2.5, 4.0)

        for idx_int in order:
            if time.monotonic() > stop - 0.5:
                break
            orig = float(best[idx_int])
            base = orig if orig > 1e-15 else default_scale
            best_val = orig
            best_val_sc = best_sc

            for m in mults:
                val = base * float(m)
                if val > 1000.0:
                    continue
                best[idx_int] = val
                sc = _score(best)
                if math.isfinite(sc) and sc < best_val_sc:
                    best_val_sc = sc
                    best_val = val

            best[idx_int] = best_val
            if best_val_sc < best_sc:
                best_sc = best_val_sc
                improved = True

        if not improved:
            break

    return best, best_sc


def _interp_to_len(a, L):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(L, dtype=np.float64) / float(L)
    if a.size == L:
        return _normalize(a)
    if a.size == 1:
        return _normalize(np.full(L, a[0]))
    x0 = np.linspace(0.0, 1.0, a.size)
    x1 = np.linspace(0.0, 1.0, L)
    v = np.interp(x1, x0, a)
    return _normalize(v)


def _make_inits(L, parent, rng):
    inits = []
    if parent is not None:
        inits.append(_interp_to_len(parent, L))
    inits.append(np.ones(L, dtype=np.float64) / float(L))

    x = np.arange(L, dtype=np.float64)
    tc = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (L - 1)))
    inits.append(_normalize(tc + 0.1))

    tc2 = 0.5 * (1.0 + np.cos(np.pi * 2.0 * (x - (L - 1) / 2.0) / (L - 1)))
    inits.append(_normalize(np.abs(tc2) + 0.2))

    for sig in (0.3, 0.5, 0.7):
        mu = L / 2.0
        v = np.exp(-0.5 * ((x - mu) / (sig * L / 2.0)) ** 2)
        inits.append(_normalize(v))

    inits.append(_normalize(rng.exponential(1.0, size=L)))
    inits.append(_normalize(np.abs(rng.normal(0, 1, size=L)) + 0.1))

    xnorm = np.linspace(0, 1, L)
    for p in (0.3, 0.5, 1.5, 2.0):
        inits.append(_normalize(xnorm ** p))
        inits.append(_normalize((1 - xnorm) ** p))
        inits.append(_normalize(np.minimum(xnorm, 1 - xnorm) ** p + 0.05))

    inits.append(_normalize(np.abs(np.sin(np.pi * xnorm * 2.0)) ** 1.5 + 0.1))
    inits.append(_normalize(np.abs(np.sin(np.pi * xnorm * 3.0)) ** 2.0 + 0.1))

    keep = []
    seen = set()
    for a in inits:
        a = _normalize(a)
        if a.size != L:
            continue
        key = tuple(np.round(a * 1000).astype(int).tolist())
        if key in seen:
            continue
        seen.add(key)
        sc = _score(a)
        if math.isfinite(sc):
            keep.append((sc, a))
    keep.sort(key=lambda z: z[0])
    return [a for _, a in keep[:8]]


def _peak_deflate(a, n_peaks=3, strength=0.05, stop=None):
    a = _normalize(a)
    n = int(a.size)
    c = _conv_same(a)
    order = np.argsort(c)[::-1][:n_peaks]
    for k in order:
        k = int(k)
        gk = _grad_peak(a, k)
        a = a - strength * gk / float(n)
        a = np.clip(a, 0.0, None)
        s = float(a.sum())
        if s < 1e-12:
            a = np.ones(n, dtype=np.float64) / float(n)
        else:
            a = a / s
    return a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(60.0, float(budget_s))
    except Exception:
        budget = 1300.0
    hard_deadline = t0 + budget
    deadline = hard_deadline - 35.0

    parent = _get_parent()
    if parent is None:
        n_base = 256
        parent = np.ones(n_base, dtype=np.float64) / float(n_base)
    else:
        parent = _normalize(parent)
        n_base = int(parent.size)

    best_seq = parent.copy()
    best_sc = _score(best_seq)
    fallback_seq = parent.copy()
    fallback_sc = best_sc

    rng = np.random.default_rng(seed)

    def _try(arr):
        nonlocal best_seq, best_sc, fallback_seq, fallback_sc
        if arr is None:
            return
        a = _normalize(arr)
        if a.size == 0:
            return
        sc = _score(a)
        if not math.isfinite(sc):
            return
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_seq = a.copy()
        if sc < fallback_sc - 1e-12:
            fallback_sc = sc
            fallback_seq = a.copy()

    # === Phase 1: Quick L-BFGS on best init (n_base) ===
    p1_stop = min(deadline - 300.0, t0 + 0.12 * budget)
    if p1_stop - time.monotonic() > 10.0:
        arr, sc = _lbfgs_optimize(best_seq, n_base, p1_stop, eta=50.0, maxfun=500, top_k=3)
        _try(arr)
        if time.monotonic() < p1_stop - 5.0:
            arr2, _ = _lbfgs_optimize(arr, n_base, p1_stop, eta=150.0, maxfun=350, top_k=3)
            _try(arr2)

    # === Phase 2: LP iterations (main workhorse) ===
    p2_stop = min(deadline - 250.0, t0 + 0.50 * budget)
    if p2_stop - time.monotonic() > 30.0 and n_base <= 512:
        arr, _ = _lp_update(best_seq, p2_stop, seed=12345,
                            max_iters=2000)
        _try(arr)

    # === Phase 3: More L-BFGS with high eta ===
    p3_stop = min(deadline - 200.0, t0 + 0.65 * budget)
    if p3_stop - time.monotonic() > 15.0:
        arr, _ = _lbfgs_optimize(best_seq, n_base, p3_stop, eta=300.0, maxfun=700, top_k=5)
        _try(arr)
        if time.monotonic() < p3_stop - 10.0:
            arr2, _ = _lbfgs_optimize(arr, n_base, p3_stop, eta=600.0, maxfun=500, top_k=3)
            _try(arr2)

    # === Phase 4: LP again on refined sequence ===
    p4_stop = min(deadline - 150.0, t0 + 0.80 * budget)
    if p4_stop - time.monotonic() > 25.0 and n_base <= 512:
        arr, _ = _lp_update(best_seq, p4_stop, seed=67890, max_iters=1500)
        _try(arr)

    # === Phase 5: Try other lengths ===
    p5_stop = min(deadline - 80.0, t0 + 0.90 * budget)
    if p5_stop - time.monotonic() > 40.0:
        other_lengths = [192, 320, 128, 384]
        other_lengths = [L for L in other_lengths if L != n_base]
        other_lengths = sorted(other_lengths)[:3]
        per_alloc = max(20.0, (p5_stop - time.monotonic()) / max(1, len(other_lengths)))
        for L in other_lengths:
            if time.monotonic() > p5_stop - 8.0:
                break
            stopL = min(p5_stop - 5.0, time.monotonic() + per_alloc)
            arr = _interp_to_len(best_seq, L)
            # Quick L-BFGS
            sub_stop = min(stopL - 5.0, time.monotonic() + max(15.0, 0.5 * (stopL - time.monotonic())))
            if sub_stop - time.monotonic() > 10.0 and L <= 384:
                arr, _ = _lbfgs_optimize(arr, L, sub_stop, eta=40.0, maxfun=400, top_k=3)
                _try(arr)
            # Quick LP
            if L <= 384 and time.monotonic() < stopL - 10.0:
                lp_s = min(stopL - 3.0, time.monotonic() + max(10.0, 0.3 * (stopL - time.monotonic())))
                if lp_s - time.monotonic() > 8.0:
                    arr2, _ = _lp_update(arr, L, lp_s, seed=int(rng.integers(1, 99999)), max_iters=300)
                    _try(arr2)

    # === Phase 6: Final polish on best ===
    fin_stop = deadline - 10.0
    if fin_stop - time.monotonic() > 30.0:
        cur_n = int(best_seq.size)
        arr, _ = _lbfgs_optimize(best_seq, cur_n, min(fin_stop - 15.0, time.monotonic() + 60.0),
                                 eta=500.0, maxfun=500, top_k=5, soft_T=0.02)
        _try(arr)

    if fin_stop - time.monotonic() > 15.0:
        cur_n = int(best_seq.size)
        arr, _ = _coord_polish(best_seq, min(fin_stop - 5.0, time.monotonic() + 30.0),
                               seed=98765, max_passes=2)
        _try(arr)

    # Final: peak deflation + re-score
    if deadline - time.monotonic() > 5.0:
        arr3 = _peak_deflate(best_seq, n_peaks=5, strength=0.02)
        _try(arr3)
        arr4 = _peak_deflate(best_seq, n_peaks=3, strength=0.03)
        _try(arr4)

    # === Final validation and return ===
    final_arr = np.clip(np.asarray(best_seq, dtype=np.float64).ravel(), 0.0, 1000.0)
    if final_arr.size == 0 or float(final_arr.sum()) < 0.01:
        final_arr = np.ones(32, dtype=np.float64) / 32.0

    sc_final = _exact_score(final_arr)
    if not math.isfinite(sc_final):
        fb = np.clip(np.asarray(fallback_seq, dtype=np.float64).ravel(), 0.0, 1000.0)
        if fb.size == 0 or float(fb.sum()) < 0.01:
            fb = np.ones(32, dtype=np.float64) / 32.0
        sc_fb = _exact_score(fb)
        if math.isfinite(sc_fb):
            return [float(x) for x in fb]
        # Last resort: uniform
        u = np.ones(32, dtype=np.float64) / 32.0
        return [float(x) for x in u]

    # Try fallback too and pick best
    if fallback_seq is not None:
        f_arr = np.clip(np.asarray(fallback_seq, dtype=np.float64).ravel(), 0.0, 1000.0)
        sc_fb = _exact_score(f_arr)
        if math.isfinite(sc_fb) and sc_fb < sc_final:
            return [float(x) for x in f_arr]

    return [float(x) for x in final_arr]
