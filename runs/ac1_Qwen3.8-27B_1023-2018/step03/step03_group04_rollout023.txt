def _conv_fft(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

        return math.inf
    M = float(_conv_fft(a).max())

    return 2.0 * n * M / (S * S)

def _lp_direction(f, M, active_frac=0.15):

    n = len(f)

        return None

def _project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
