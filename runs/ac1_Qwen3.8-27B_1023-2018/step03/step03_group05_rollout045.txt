def _weighted_gradient(x, w):
    """Compute gradient of weighted sum of convolution: d/dx sum_k w[k]*(x*x)[k]."""
    n = x.size
    m = 2 * n - 1
    need = 2 * n - 1
    ws = np.zeros(need, dtype=np.float64)
    ws[:min(w.size, need)] = w[:min(w.size, need)]
    ws = np.where(np.isfinite(ws), ws, 0.0)
    s = float(ws.sum())
    if s <= 0:
        return np.zeros(n, dtype=np.float64)
    ws /= s
    nfft = next_fast_len(3 * n - 2)
    fx = rfft(x, nfft)
    fw = rfft(ws[::-1], nfft)
    cr = irfft(fx * fw, nfft)[:3 * n - 2]
    cr = np.where(np.isfinite(cr), cr, 0.0)
    idx = 2 * n - 2 - np.arange(n)
    g = 2.0 * cr[idx]
    return g

def _resample_to(a, m):
    """Resample sequence to length m preserving mass."""
    a = _normalize(a)
    n = a.size
    m = int(m)
    if m <= 0:
        return np.ones(1)
    if n == m:
        return a.copy()
    if m > n:
        idx = (np.arange(m) * n) // m
        counts = np.bincount(idx, minlength=n).astype(np.float64)
        counts[counts <= 0] = 1.0
        b = a[idx] / counts[idx]
    else:
        b = np.zeros(m)
        idx = (np.arange(n) * m) // n
        np.add.at(b, idx, a)
    return _normalize(b)
