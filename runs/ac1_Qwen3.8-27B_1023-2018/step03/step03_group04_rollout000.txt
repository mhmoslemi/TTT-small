def _conv_from_F(F, fsize, L):

def _build_cross(p):
    n = p.size

    A = np.empty((L, n), dtype=np.float64)
    for j in range(n):

        Fl = (1.0 - a) * Fp + a * Fh
        c = irfft(Fl * Fl, fsize)[:L]
        m = float(c.max())
        if m < best_m - 1e-13:
            best_m = m
            best_a = float(a)

    if best_a is None:
        return None, None

    centers = [
        best_a,
        max(1e-5, best_a - 0.1),
        min(1.0, best_a + 0.1),
        max(1e-5, 0.75 * best_a),
        min(1.0, 1.25 * best_a),
    ]
    for center in centers:
        for d in (0.005, 0.01, 0.02, 0.05):
            for a in (center - d, center + d):
                if not (1e-5 < a < 1.0):
                    continue
                Fl = (1.0 - a) * Fp + a * Fh
                c = irfft(Fl * Fl, fsize)[:L]
                m = float(c.max())
                if m < best_m - 1e-13:
                    best_m = m
                    best_a = float(a)

    return best_a, best_m


def _lambda_direction(p, q, M, goal, A, lam, tlimit=8.0):
    n = p.size
    L = q.size
    if M <= 0.0:
        return None, None
    goal = float(min(max(goal, 1e-15), M))
    lam = float(np.clip(lam, 1e-3, 0.9))

    b = goal - (1.0 - 2.0 * lam) * q
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    A_ub = np.empty((L, n + 1), dtype=np.float64)
    A_ub[:, :n] = A
    A_ub[:, :n] *= (2.0 * lam)
    A_ub[:, -1] = -1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    bounds = [(0.0, None)] * n + [(-goal, None)]

    res = _lp_solve(c, A_ub, b, A_eq, b_eq, bounds, tlimit)
    if res is None or res.status != 0 or res.x is None:
        return None, None

    h = np.asarray(res.x[:n], dtype=np.float64).copy()
    np.maximum(h, 0.0, out=h)
    s = float(h.sum())
    if s <= 1e-12:
        return None, None
    h /= s
    t = float(res.x[-1])
    return h, t


def _sum_direction(p, A, M, alpha=1.0, tlimit=6.0):
    n = p.size
    L = 2 * n - 1
    cap = alpha * M
    if cap <= 0.0:
        return None

    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(L, cap, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = _lp_solve(c, A, b_ub, None, None, bounds, tlimit)
    if res is None or res.status != 0 or res.x is None:
        return None

    h = np.asarray(res.x, dtype=np.float64).copy()
    np.maximum(h, 0.0, out=h)
    s = float(h.sum())
    if s <= 1.0 + 1e-7:
        return None
    h /= s
    return h


def _optimize(init, stop, seed, track, target=_TARGET):
    p0 = _to_prob(init)
    n = p0.size
    if n <= 1:
        try:
            track(p0)
        except Exception:
            pass
        return p0, 2.0

    L = 2 * n - 1
    fsize = next_fast_len(L)

    p = p0.copy()
    Fp = _fft(p, fsize)
    q = _conv_from_F(Fp, fsize, L)
    M = float(q.max())
    score = 2.0 * n * M
    A = _build_cross(p)

    best_p = p.copy()
    best_score = score
    try:
        track(best_p)
    except Exception:
        pass

    rng = np.random.default_rng(int(seed) if seed is not None else 0)
    lam = 0.30
    stall = 0
    target_M = float(target) / (2.0 * float(n))

    def _set_p(p_x):
        nonlocal p, Fp, q, M, score, A
        p = _to_prob(p_x)
        Fp = _fft(p, fsize)
        q = _conv_from_F(Fp, fsize, L)
        M = float(q.max())
        score = 2.0 * n * M
        A = _build_cross(p)

    def _try_move(h, suggest):
        nonlocal best_score, best_p, lam, stall
        Fh = _fft(h, fsize)
        step, bm = _line_search(Fp, Fh, M, float(suggest), fsize, L)
        if step is None or bm >= M - 1e-12:
            return False
        p_new = (1.0 - step) * p + step * h
        _set_p(p_new)
        if score < best_score:
            best_score = score
            best_p = p.copy()
            try:
                track(best_p)
            except Exception:
                pass
        lam = float(np.clip(0.7 * lam + 0.3 * step, 0.05, 0.7))
        stall = 0
        return True

    while time.monotonic() < stop - 2.0:
        if M <= 1e-300:
            break

        if stall > 30:
            scale = 0.5 / float(n)
            base_p = best_p.copy()
            _set_p(base_p + scale * rng.normal(0.0, 1.0, n))
            if score < best_score:
                best_score = score
                best_p = p.copy()
                try:
                    track(best_p)
                except Exception:
                    pass
            stall = 0
            lam = 0.30
            continue

        goal = min(M, target_M)
        if goal <= 1e-300:
            goal = M
        base = max(0.0, M - goal)
        dir_tol = max(1e-12, 1e-8 * max(1.0, M))

        accepted = False
        lams = [
            lam,
            max(0.05, 0.6 * lam),
            min(0.7, 1.5 * lam),
        ]
        ntry = 1 if stall < 3 else (2 if stall < 8 else 3)

        if time.monotonic() < stop - 6.0:
            for lam_try in lams[:ntry]:
                if time.monotonic() > stop - 6.0:
                    break
                rem = stop - time.monotonic()
                if rem < 6.0:
                    break
                tlimit = max(1.0, min(12.0, rem - 6.0))
                h, t = _lambda_direction(p, q, M, goal, A, lam_try, tlimit)
                if h is None:
                    continue
                if not (t < base - dir_tol):
                    continue
                if _try_move(h, lam_try):
                    accepted = True
                    break

        if (not accepted) and (time.monotonic() < stop - 6.0) and stall >= 2:
            caps = [1.0]
            if stall >= 6:
                caps = [1.0, 0.85, 0.65]
            for cap_alpha in caps:
                if time.monotonic() > stop - 6.0:
                    break
                rem = stop - time.monotonic()
                if rem < 6.0:
                    break
                tlimit = max(1.0, min(8.0, rem - 6.0))
                h = _sum_direction(p, A, M, cap_alpha, tlimit)
                if h is None:
                    continue
                if float(np.max(np.abs(h - p))) < 1e-7:
                    continue
                if _try_move(h, 0.30):
                    accepted = True
                    break

        if (not accepted) and (time.monotonic() < stop - 4.0):
            try:
                if M > 1e-300:
                    z = (q / max(M, 1e-300)) * 20.0
                    w = np.exp(z - z.max())
                    ws = float(w.sum())
                    if ws > 1e-300:
                        w /= ws
                        D = 2.0 * (A.T.dot(w))
                        D = D - float(D.mean())
                        nd = float(np.linalg.norm(D, 1))
                        if nd > 1e-14:
                            for eta in (0.5, 2.0, 8.0, 32.0, 128.0):
                                if time.monotonic() > stop - 4.0:
                                    break
                                h = p - eta * D
                                h = _to_prob(h)
                                if float(np.max(np.abs(h - p))) < 1e-6:
                                    continue
                                if _try_move(h, 0.20):
                                    accepted = True
                                    break
            except Exception:
                pass

        if not accepted:
            scale = (0.5 / float(n)) * (1.0 + 0.05 * min(stall, 40))
            base_p = best_p if (stall > 6 and rng.random() < 0.7) else p
            h = base_p + scale * rng.normal(0.0, 1.0, n)
            h = _to_prob(h)
            if _try_move(h, 0.20):
                accepted = True
            else:
                if stall >= 6 and rng.random() < 0.20:
                    _set_p(h)
                    if score < best_score:
                        best_score = score
                        best_p = p.copy()
                        try:
                            track(best_p)
                        except Exception:
                            pass
                    stall = 0
                    lam = 0.30
                else:
                    stall += 1
                    if stall % 10 == 0:
                        lam = float(np.clip(lam * (0.8 if lam > 0.25 else 1.2), 0.05, 0.7))

        if time.monotonic() > stop - 2.5:
            break

    try:
        track(best_p)
    except Exception:
        pass
    return best_p, best_score


def _resample(src, L):
    src = np.asarray(src, dtype=np.float64).ravel()
    src = np.where(np.isfinite(src), src, 0.0)
    src = np.clip(src, 0.0, 1000.0)
    if src.size == 0:
        return np.ones(L, dtype=np.float64) / float(L)
    if src.size == 1:
        return np.ones(L, dtype=np.float64) / float(L)
    if src.size == L:
        return _to_prob(src)
    x_old = np.linspace(0.0, 1.0, src.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, L, dtype=np.float64)
    y = np.interp(x_new, x_old, src)
    return _to_prob(y)


def _polish_sequence(init, stop, seed=12345):
    p = _to_prob(init)
    n = p.size
    if n <= 1:
        return p, 2.0

    L = 2 * n - 1
    fsize = next_fast_len(L)
    Fp = _fft(p, fsize)
    q = _conv_from_F(Fp, fsize, L)
    M = float(q.max())
    best_p = p.copy()
    best_M = M

    rng = np.random.default_rng(seed)
    improved = True

    while time.monotonic() < stop - 2.0 and improved:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop - 1.5:
                break

            a = float(p[i])
            e = np.zeros(L, dtype=np.float64)
            if i < L:
                e[i:i + n] = p

            cands = {0.0, a}
            if a > 1e-15:
                for m in (0.125, 0.25, 0.5, 0.75, 1.5, 2.0, 3.0, 4.0):
                    cands.add(a * m)

            base = 1.0 / float(n)
            for m in (0.0, 0.25, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0):
                v = base * m
                if 0.0 <= v <= 1.0:
                    cands.add(float(v))
            for d in (base, 2.0 * base, 5.0 * base, 10.0 * base):
                v1 = a + d
                v2 = a - d
                if 0.0 <= v1 <= 1.0:
                    cands.add(float(v1))
                if 0.0 <= v2 <= 1.0:
                    cands.add(float(v2))

            best_x = a
            best_m = M
            two_i = 2 * i
            for x in cands:
                x = float(x)
                if x < 0.0 or x > 1.0:
                    continue
                dx = x - a
                if abs(dx) < 1e-16:
                    continue
                Snew = 1.0 + dx
                if Snew <= 1e-12:
                    continue
                tmp = q + 2.0 * dx * e
                if two_i < L:
                    tmp = tmp.copy()
                    tmp[two_i] += dx * dx
                m = float(tmp.max()) / (Snew * Snew)
                if m < best_m - 1e-13:
                    best_m = m
                    best_x = x

            if best_x != a:
                p_new = p.copy()
                p_new[i] = best_x
                p = _to_prob(p_new)
                Fp = _fft(p, fsize)
                q = _conv_from_F(Fp, fsize, L)
                M = float(q.max())
                if M < best_M - 1e-13:
                    best_M = M
                    best_p = p.copy()
                    improved = True

    return best_p, best_M


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))

    headroom = 15.0 if budget >= 300.0 else max(3.0, 0.1 * budget)
    hard_deadline = t0 + max(5.0, budget - headroom)

    parent = _get_parent()

    best_seq = parent.copy()
    best_score = math.inf

    def track(p):
        nonlocal best_seq, best_score
        try:
            arr = np.asarray(p, dtype=np.float64).ravel()
        except Exception:
            return
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size == 0:
            return
        if float(arr.sum()) < 0.01:
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in arr]))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = arr.copy()

    try:
        track(parent)
    except Exception:
        best_score = math.inf
        best_seq = np.ones(384, dtype=np.float64)

    def remaining():
        return hard_deadline - time.monotonic()

    # Main optimization at parent length.
    if remaining() > 5.0:
        phase1_stop = min(hard_deadline - 60.0, t0 + 0.62 * budget)
        if phase1_stop < time.monotonic() + 1.0:
            phase1_stop = min(hard_deadline - 10.0, time.monotonic() + 1.0)
        try:
            p1, s1 = _optimize(parent, phase1_stop, seed, track)
            track(p1)
        except Exception:
            pass

    # If not already below target, briefly explore nearby lengths.
    if math.isfinite(best_score) and best_score > _TARGET + 1e-6 and remaining() > 120.0:
        expl_stop = hard_deadline - 70.0
        lengths = []
        for L in (256, 320, 384, 448, 512, 192, 128):
            if L != parent.size and L not in lengths:
                lengths.append(L)
        idx = 0
        while idx < len(lengths) and time.monotonic() < expl_stop - 20.0:
            L = lengths[idx]
            alloc = max(12.0, min(75.0, (expl_stop - time.monotonic()) / max(1, len(lengths) - idx)))
            stopL = min(expl_stop, time.monotonic() + alloc)
            if stopL <= time.monotonic() + 3.0:
                break
            init = _resample(parent, int(L))
            try:
                pL, sL = _optimize(init, stopL, int(seed) + 1000 + int(L), track)
                track(pL)
            except Exception:
                pass
            idx += 1

    # Final dedicated optimization on the best exact candidate.
    if remaining() > 25.0:
        final_stop = min(hard_deadline - 20.0, time.monotonic() + max(20.0, 0.5 * (hard_deadline - 20.0 - time.monotonic())))
        if final_stop > time.monotonic() + 5.0:
            try:
                pB, sB = _optimize(best_seq, final_stop, int(seed) + 99999, track)
                track(pB)
            except Exception:
                pass

    # Coordinate polishing, with exact tracking of the polished candidate.
    if remaining() > 12.0:
        polish_stop = hard_deadline - 5.0
        try:
            poly, ps = _polish_sequence(best_seq, polish_stop, seed=int(seed) + 777)
            track(poly)
        except Exception:
            pass

    try:
        out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
        out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
        out_arr = np.clip(out_arr, 0.0, 1000.0)
        final_sc = float(evaluate_sequence([float(x) for x in out_arr]))
        if not math.isfinite(final_sc):
            raise ValueError
        out_arr = out_arr
    except Exception:
        out_arr = np.ones(384, dtype=np.float64)

    return [float(x) for x in out_arr]
