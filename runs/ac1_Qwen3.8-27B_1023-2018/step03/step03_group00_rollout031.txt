    f_init = f_init / f_init.sum()

    def obj_and_grad(z):
        call_count[0] += 1
        if time.monotonic() > stop_time:
            return 1e10, np.zeros_like(z)
        
        z_shifted = z - z.max()  # numerical stability
        w = np.exp(z_shifted)
        f = w / w.sum()
        
        b = np.convolve(f, f, mode='full')
        
        # Log-sum-exp
        b_max = b.max()
        exp_vals = np.exp(tau * (b - b_max))
        log_sum = b_max + np.log(exp_vals.sum()) / tau
        
        # Softmax probabilities
        p = exp_vals / exp_vals.sum()

        g = np.zeros(n)

    # Use L-BFGS-B
    try:

                         options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12,

    f0 = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 1e-15, None).copy()
    f0 = f0 / f0.sum()
    
    z0 = np.log(f0)  # will use these as log-softmax inputs
    # Actually, for softmax: f = softmax(z) = exp(z) / sum(exp(z))
    # If f0 is already the target, z0 = log(f0) works since softmax(log(f0)) = f0
    
    if tau_steps is None:
        tau_steps = [20, 50, 100, 200, 500, 1000]
    
    best_f = f0.copy()
    best_m = _conv_full(f0).max()
    
    N_fft = next_fast_len(4 * n)
    
    def compute_g_fft(p, f):
        """g_j = 2 * sum_k p_k * f_{k-j} for j in 0..n-1"""
        p_hat = rfft(p, N_fft)
        f_pad = np.zeros(len(p))
        f_pad[:n] = f
        f_pad_hat = rfft(f_pad, N_fft)
        corr = irfft(p_hat * np.conj(f_pad_hat), N_fft)

def _active_set_polish(f0, n, stop_time, seed=42):

    # determine the "gradient" of M. We do small steps in the negative
    # subgradient direction, projected onto the simplex.
    
    step_size = 0.01
    no_imp = 0
    max_no_imp = 50

        if time.monotonic() > stop_time - 2.0:
            break
        if no_imp > max_no_imp:
            break

        M = b.max()
        
        # Find active constraints (within tolerance)
        tol = 1e-8 * M + 1e-15
        active = np.where(b >= M - tol)[0]
        
        # Subgradient of M w.r.t. f
        # dM/df_i = (1/|S|) * sum_{k in S} 2*f_{k-i} (for valid k-i)
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    grad[i] += 2.0 * f[ki]
        grad /= len(active)
        
        # Projected gradient step (onto simplex: sum(f)=1, f>=0)
        # Direction: -grad, projected to sum=0
        d = -grad + np.mean(grad)  # ensure sum(d) = 0
        # Also ensure f + step*d >= 0
        # Max step before any component hits 0
        neg_mask = d < -1e-15
        if neg_mask.any():
            max_step = np.min(f[neg_mask] / (-d[neg_mask])) * 0.99
        else:
            max_step = step_size
        
        actual_step = min(step_size, max_step)
        f_new = f + actual_step * d
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()  # re-normalize
        
        sc = _score(f_new)
        if sc < best_sc - 1e-15:
            f = f_new
            best_f = f_new.copy()
            best_sc = sc
            no_imp = 0
            step_size = min(0.1, step_size * 1.1)
        else:
            no_imp += 1
            step_size *= 0.7
        
        # Every 100 iters, add small noise
        if iteration % 100 == 99:
            f = f + rng.normal(0, 1e-4, n)
            f = np.maximum(f, 0)
            f = f / f.sum()
    
    return best_f, best_sc

def _conv_full(a):
    return np.convolve(a, a, mode="full")


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=3.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs",
                      options={"presolve": True, "time_limit": timeout,
                               "dual_feasibility_tolerance": 1e-10})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g0 = np.asarray(res.x, dtype=np.float64)
    g0 = np.clip(g0, 0.0, 1000.0)
    if g0.sum() < 1e-15:
        return None
    # Ensure strictly positive for LP to be meaningful
    if np.all(g0 == 0):
        return None
    return g0

        b = _conv_full(f)
        M = float(b.max())

    f0 = f0 / f0.sum()
    
    z = np.log(f0)  # initial log-variables (not quite softmax input, but close)
    
    best_f = f0.copy()
    best_m = float(_conv_full(f0).max())
    
    N_fft = next_fast_len(4 * n)
    tau_steps = [15, 40, 100, 300, 800, 2000]

        if time.monotonic() > stop_time - 3.0:
            break
        
        max_iter = max(30, min(200, int(60000 / n)))
        
        def obj_and_grad(zv):
            zs = zv - zv.max()
            w = np.exp(zs)
            sw = w.sum()
            fv = w / sw

            obj = 2.0 * n * log_sum
            
            p = e / e.sum()
            
            # Compute g_j = 2 * sum_k p_k * f_{k-j}
            p_hat = rfft(p, N_fft)
            f_pad = np.zeros(len(b))

            corr = irfft(p_hat * np.conj(f_pad_hat), N_fft)
            g = 2.0 * corr[:n]
            
            G = float(np.dot(g, fv))
            
            grad = (2.0 * n / tau) * fv * (g - G)

                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-14, 'maxcor': 25, 'iprint': 0})

        except Exception:
            pass

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _conv_full(a):
    return np.convolve(a, a, mode="full")


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=4.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs",
                      options={"presolve": True, "time_limit": timeout,
                               "dual_feasibility_tolerance": 1e-10})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g0 = np.asarray(res.x, dtype=np.float64)
    g0 = np.clip(g0, 0.0, 1000.0)
    if g0.sum() < 1e-15:
        return None
    return g0


def _lp_optimize(f0, n, stop_time, seed=42):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n

    best_f = f.copy()
    best_score = _score(f)
    if not math.isfinite(best_score):
        return f, best_score

    t = 0.3
    stall = 0
    max_stall = 400

    while time.monotonic() < stop_time - 2.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g0 = _lp_direction(f, n, M)
        if g0 is None:
            stall += 5
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.015 * max(1, f.mean()), n), 0, None)
                f = f / f.sum() * best_f.sum()
                t = 0.3
                stall = 0
                max_stall = min(max_stall + 100, 3000)
                continue

        s_f = float(f.sum())
        s_g = float(g0.sum())
        g = g0 * (s_f / s_g)

        dist = float(np.linalg.norm(g - f) / max(1.0, np.linalg.norm(f)))
        if dist < 5e-11:
            stall += 2
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 5e-5 * max(1, f.mean()), n), 0, None)
            f = f / f.sum() * best_f.sum()
            continue

        improved = False
        ct = t
        for _ in range(25):
            f_new = (1.0 - ct) * f + ct * g
            f_new = np.maximum(f_new, 0)
            fs2 = f_new.sum()
            if fs2 < 0.01:
                ct *= 0.5
                continue
            sc = _score(f_new)
            if sc < best_score - 1e-15:
                f = f_new.copy()
                best_f = f.copy()
                best_score = sc
                t = min(0.9, ct * 1.15)
                improved = True
                stall = 0
                break
            else:
                ct *= 0.55
            if ct < 1e-7:
                break

        if not improved:
            stall += 1
            t = max(0.02, t * 0.85)
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.025 * max(1, f.mean()), n), 0, None)
                f = f / f.sum() * best_f.sum()
                t = 0.3
                stall = 0
                max_stall = min(max_stall + 100, 5000)

    return best_f, best_score


def _smoothed_gradient_optimize(f0, n, stop_time):
    f0 = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-15, None)
    f0 = f0 / f0.sum()

    z = np.log(f0)
    best_f = f0.copy()
    best_m = float(_conv_full(f0).max())
    N_fft = next_fast_len(4 * n)
    tau_steps = [15, 40, 100, 300, 800, 2000, 5000]

    for tau in tau_steps:
        if time.monotonic() > stop_time - 3.0:
            break
        max_iter = max(50, min(300, int(80000 / n)))

        def obj_and_grad(zv, _tau=tau):
            zs = zv - zv.max()
            w = np.exp(zs)
            sw = w.sum()
            fv = w / sw
            b = _conv_full(fv)
            b_max = b.max()
            tb = np.clip(_tau * (b - b_max), -50, 50)
            e = np.exp(tb)
            log_sum = b_max + np.log(e.sum()) / _tau
            obj = 2.0 * n * log_sum
            p = e / e.sum()
            p_hat = rfft(p, N_fft)
            f_pad = np.zeros(len(b))
            f_pad[:n] = fv
            f_pad_hat = rfft(f_pad, N_fft)
            corr = irfft(p_hat * np.conj(f_pad_hat), N_fft)
            g_vec = 2.0 * corr[:n]
            G = float(np.dot(g_vec, fv))
            grad = (2.0 * n / _tau) * fv * (g_vec - G)
            return obj, grad

        try:
            res = minimize(obj_and_grad, z.copy(), jac=True, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'ftol': 1e-16,
                                    'gtol': 1e-14, 'maxcor': 30, 'iprint': 0})
            if res.x is not None and np.all(np.isfinite(res.x)):
                z = res.x
                zs = z - z.max()
                w = np.exp(zs)
                fn = w.sum()
                if fn > 1e-10:
                    f_new = w / fn
                    f_new = np.clip(f_new, 0, None)
                    fnew = f_new.sum()
                    if fnew > 1e-10:
                        f_new /= fnew
                        m_new = float(_conv_full(f_new).max())
                        if m_new < best_m - 1e-16:
                            best_f = f_new.copy()
                            best_m = m_new
        except Exception:
            pass

    return best_f, 2.0 * n * best_m


def _active_set_polish(f0, n, stop_time, seed=0):
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    s = f.sum()
    if s < 1e-10:
        return f, _score(f)
    f = f / s
    best_f = f.copy()
    best_sc = _score(f)
    rng = np.random.default_rng(seed)
    step = 0.005
    no_imp = 0
    max_no = 80

    for _ in range(50000):
        if time.monotonic() > stop_time - 2.0 or no_imp > max_no:
            break
        b = _conv_full(f)
        M = b.max()
        tol = max(1e-9 * M, 1e-16)
        active = np.where(b >= M - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * f[k - hi:k - lo]
        grad /= len(active)
        d = -grad + np.mean(grad)
        neg = d < -1e-15
        ms = 0.1
        if neg.any():
            ms = min(ms, float(np.min(f[neg] / (-d[neg]))) * 0.99)
        st = max(1e-10, min(step, ms))
        f_new = f + st * d
        f_new = np.maximum(f_new, 0)
        fn = f_new.sum()
        if fn < 1e-10:
            no_imp += 1
            step *= 0.7
            continue
        f_new /= fn
        sc = _score(f_new)
        if sc < best_sc - 1e-16:
            f = f_new.copy()
            best_f = f.copy()
            best_sc = sc
            step = min(0.05, step * 1.1)
            no_imp = 0
        else:
            no_imp += 1
            step *= 0.75
        if _ % 200 == 199 and no_imp > 20:
            f = f + rng.normal(0, 1e-4, n)
            f = np.maximum(f, 0)
            f /= f.sum()
            step = 0.005

    return best_f, best_sc


def _coord_polish(f0, n, stop_time, seed=0):
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    s = f.sum()
    if s < 1e-10:
        return f, _score(f)
    f = f / s
    best_sc = _score(f)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no = 60

    while time.monotonic() < stop_time - 3.0 and no_imp < max_no:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = f[i]
            if orig < 1e-14:
                continue
            for mult in (0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.6, 2.2, 0.0):
                f[i] = orig * mult
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_sc - 1e-16:
                    best_sc = sc
                    improved = True
            if not improved:
                f[i] = orig
        for _ in range(150):
            if time.monotonic() > stop_time - 3.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi in (0.4, 0.7, 1.3, 1.7, 0.0):
                for mj in (0.4, 0.7, 1.3, 1.7, 0.0):
                    f[i] = oi * mi
                    f[j] = oj * mj
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_sc - 1e-16:
                        best_sc = sc
                        improved = True
            f[i], f[j] = oi, oj
        for _ in range(40):
            if time.monotonic() > stop_time - 3.0:
                break
            k = int(rng.integers(n))
            old = f[k]
            f[k] = max(0.0, old + rng.normal(0, 0.008 * max(1.0, f.mean())))
            if f.sum() < 0.01:
                f[k] = old
                continue
            sc = _score(f)
            if sc < best_sc - 1e-16:
                best_sc = sc
                improved = True
            else:
                f[k] = old
        if not improved:
            no_imp += 1
            f = f + rng.normal(0, 0.003 * max(1.0, f.mean()), n)
            f = np.maximum(f, 0)
            f /= f.sum()
        else:
            no_imp = 0

    return f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    deadline = t0 + budget - 18.0

    parent = _get_parent()
    n = len(parent)
    best_f = parent.copy()
    best_score = _score(best_f)
    if not math.isfinite(best_score):
        best_f = np.ones(100) / 100.0
        n = len(best_f)
        best_score = _score(best_f)

    def _update(f_new, sc):
        nonlocal best_f, best_score
        if f_new is None or len(f_new) == 0:
            return
        f_new = np.clip(np.asarray(f_new, dtype=np.float64).ravel(), 0, 1000)
        if f_new.sum() < 0.01 or not np.all(np.isfinite(f_new)):
            return
        if sc < best_score - 1e-15:
            best_f = f_new.copy()
            best_score = sc

    # Phase 1: LP
    p1_end = t0 + budget * 0.45
    if time.monotonic() < p1_end - 10:
        try:
            f1, s1 = _lp_optimize(best_f, n, p1_end, seed + 1)
            _update(f1, s1)
        except Exception:
            pass

    # Second LP pass (warm start)
    p1b_end = min(p1_end + budget * 0.08, t0 + budget * 0.60)
    if time.monotonic() < p1b_end - 10:
        try:
            f1b, s1b = _lp_optimize(best_f, n, p1b_end, seed + 999)
            _update(f1b, s1b)
        except Exception:
            pass

    # Phase 2: Smoothed gradient
    p2_end = t0 + budget * 0.78
    if time.monotonic() < p2_end - 15:
        try:
            f2, s2 = _smoothed_gradient_optimize(best_f, n, p2_end)
            _update(f2, s2)
        except Exception:
            pass

    # Phase 3: Active set polish
    p3_end = t0 + budget * 0.88
    if time.monotonic() < p3_end - 10:
        try:
            f3, s3 = _active_set_polish(best_f, n, p3_end, seed + 2)
            _update(f3, s3)
        except Exception:
            pass

    # Phase 4: Coord polish
    p4_end = deadline - 8
    if time.monotonic() < p4_end - 10:
        try:
            f4, s4 = _coord_polish(best_f, n, p4_end, seed + 3)
            _update(f4, s4)
        except Exception:
            pass

    # Second coord polish with different seed
    if time.monotonic() < deadline - 5:
        try:
            f4b, s4b = _coord_polish(best_f, n, deadline - 3, seed + 777)
            _update(f4b, s4b)
        except Exception:
            pass

    # Final validation
    try:
        final_list = [float(x) for x in best_f]
        final_sc = float(evaluate_sequence(final_list))
    except Exception:
        final_list = [float(x) for x in parent]
        final_sc = float(evaluate_sequence(final_list))

    if not math.isfinite(final_sc):
        return [1.0]
    return final_list
