    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0 or p.sum() < 0.01:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_norm(a):
    n = len(a)
    M = float(_conv_fft(a).max())
    return 2.0 * n * M


def _score_fast(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    M = float(_conv_fft(a).max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M / (S * S)


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_optimize(f0, stop_time, seed, max_iter=500000):
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = np.clip(f0.copy(), 0, None)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s

    best_f = f.copy()
    best_score = _score_norm(f)

    t = 0.25
    no_improve = 0
    max_no_improve = 300
    iteration = 0
    b_ub_buf = np.empty(2 * n - 1)

    while time.monotonic() < stop_time - 0.3 and iteration < max_iter:
        conv = _conv_fft(f)
        M = float(conv.max())
        if M < 1e-15:
            break

        A = _build_A(f)
        b_ub_buf[:] = M
        c_vec = -np.ones(n)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_vec, A_ub=A, b_ub=b_ub_buf,
                          bounds=bounds, method='highs',
                          options={"presolve": True, "time_limit": 5.0})
        except Exception:
            no_improve += 10
            if no_improve > max_no_improve:
                break
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            s = f.sum()
            f = f / s if s > 1e-10 else best_f.copy()
            continue

        if res.status != 0 or res.x is None:
            no_improve += 5
            if no_improve > max_no_improve:
                break
            f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
            s = f.sum()
            f = f / s if s > 1e-10 else best_f.copy()
            continue

        g = np.asarray(res.x, dtype=np.float64)
        gs = float(g.sum())
        if gs < 1e-10:
            no_improve += 5
            if no_improve > max_no_improve:
                break
            f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
            s = f.sum()
            f = f / s if s > 1e-10 else best_f.copy()
            continue
        g /= gs

        dist = float(np.sqrt(np.sum((g - f) ** 2)))
        if dist < 1e-10:
            no_improve += 8
            if no_improve > max_no_improve:
                break
            f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
            s = f.sum()
            f = f / s if s > 1e-10 else best_f.copy()
            continue

        # Geometric line search
        best_t = t
        best_sc = best_score
        best_fn = None
        t_cands = [t * 0.4, t * 0.7, t, t * 1.4, t * 2.0, t * 3.0]
        t_cands = [max(0.01, min(1.0, tc)) for tc in t_cands]
        t_cands = sorted(set(t_cands), key=lambda x: -abs(x - t))

        for tc in t_cands:

            fn = np.clip(fn, 0, None)
            s = fn.sum()
            if s < 1e-10:
                continue
            fn = fn / s

                best_sc = sc
                best_t = tc
                best_fn = fn

                f = f / s if s > 1e-10 else best_f.copy()

    return best_f, best_score


def _polish(a, stop_time, seed=42):
    n = len(a)
    best = np.clip(a.copy(), 0, None)
    s = best.sum()
    if s < 1e-10:
        return best, math.inf
    best /= s
    best_sc = _score_norm(best)
    rng = np.random.default_rng(seed)

    no_imp = 0
    max_no_imp = 8
    multipliers_1d = [0.0, 0.05, 0.1, 0.25, 0.5, 0.75, 1.25, 1.5, 2.0, 3.0, 5.0]

    while time.monotonic() < stop_time - 0.5:
        improved = False

        # Single-coordinate moves
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 0.5:
                break
            orig = best[i]
            bv = orig
            bsc = best_sc
            for m in multipliers_1d:
                best[i] = orig * m if orig > 1e-14 else (0.0001 if m > 0 else 0.0)
                sn = best.sum()
                if sn < 1e-10:
                    best[i] = orig
                    continue
                sc = _score_fast(best)
                if sc < bsc - 1e-14:
                    bsc = sc
                    bv = best[i]
            best[i] = bv
            if bsc < best_sc - 1e-14:
                s = best.sum()
                if s > 1e-10:
                    best /= s
                    best_sc = _score_norm(best)
                    improved = True

        # Pair moves
        n_pairs = min(300, 3 * n)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 0.5:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.0, 0.3, 0.7, 1.4, 2.0):
                for mj in (0.0, 0.3, 0.7, 1.4, 2.0):
                    best[i] = oi * mi if oi > 1e-14 else (0.0001 if mi > 0 else 0.0)
                    best[j] = oj * mj if oj > 1e-14 else (0.0001 if mj > 0 else 0.0)
                    s2 = best.sum()
                    if s2 < 1e-10:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score_fast(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc


def _hann_init(n):
    i = np.arange(n)
    return np.sin(np.pi * (i + 0.5) / n) ** 2

    sig = n * 0.25
    return np.exp(-((i - center) / sig) ** (2 * p)) + 1e-6
