L(b, M, λ, μ) = M + sum_j μ_j (1 - sum_j b_j) - sum_k λ_k ((b*b)_k - M)

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar
from scipy.fft import rfft, irfft


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    safety = 25.0
    deadline = t0 + budget - safety

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:

        parent = np.ones(n) / n
    parent = parent / parent.sum()

    def exact_score(a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        s = float(a.sum())
        if s < 0.01:
            return float('inf')
        c = np.convolve(a, a)
        return float(2.0 * len(a) * c.max() / (s * s))

        try:
            sc = exact_score(cand)
        except Exception:
            return False
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0).copy()
            return True
        return False

    rng = np.random.default_rng(seed)

    def build_active_A(a, active_lags):
        """Build constraint matrix for given active lags."""
        n = a.size
        m = len(active_lags)
        A = np.zeros((m, n), dtype=np.float64)
        for idx, k in enumerate(active_lags):
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_max >= j_min:
                js = np.arange(j_min, j_max + 1)
                A[idx, js] = a[k - js]
        return A

    def solve_lp_max_sum(A, M, n, time_limit=1.0):
        """Solve max sum(g) s.t. A@g <= M, g >= 0."""
        try:
            c_obj = -np.ones(n)
            b_ub = np.full(A.shape[0], M)
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                         bounds=[(0.0, None)] * n, method='highs',
                         options={'time_limit': time_limit,
                                  'dual_feasibility_tolerance': 1e-9,
                                  'primal_feasibility_tolerance': 1e-9})
            if res.status == 0 and res.x is not None:
                g = np.asarray(res.x, dtype=np.float64)
                if g.sum() > 1e-15:
                    return g
        except Exception:
            pass
        return None

    def solve_lp_min_cross(A_full, n, time_limit=1.0):
        """Solve min M' s.t. (a*g)_k <= M' for all k, sum(g)=1, g>=0."""
        try:
            # Variable: [g_0,...,g_{n-1}, M']
            m = A_full.shape[0]
            c = np.zeros(n + 1)
            c[n] = 1.0  # minimize M'
            # A_ub: [A_full, -1_m] * [g; M'] <= 0
            A_ub = np.zeros((m, n + 1))
            A_ub[:, :n] = A_full
            A_ub[:, n] = -1.0
            b_ub = np.zeros(m)
            # A_eq: sum(g) = 1
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            bounds = [(0.0, None)] * n + [(None, None)]
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': time_limit,
                                  'dual_feasibility_tolerance': 1e-9,
                                  'primal_feasibility_tolerance': 1e-9})
            if res.status == 0 and res.x is not None:
                g = np.asarray(res.x[:n], dtype=np.float64)
                if g.sum() > 1e-15:
                    return g
        except Exception:
            pass
        return None

    def line_search_conv(a, direction, M_current, time_left=2.0):
        """Find best alpha in [0,1] minimizing max_conv((1-alpha)*a + alpha*direction)."""
        def obj(alpha):
            cand = (1.0 - alpha) * a + alpha * direction
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-12:
                return 1e10
            cand = cand / cs
            return float(np.convolve(cand, cand).max())

        try:

                                 options={'xatol': 1e-7, 'maxiter': 40})

                cand = (1.0 - alpha) * a + alpha * direction
                cand = np.maximum(cand, 0.0)
                cand = cand / cand.sum()
                return cand, float(res.fun)
        except Exception:
            pass
        return None, M_current

    def gradient_step(a, b, M, k_star, n, rng_local):
        """Try gradient descent on the active lag."""
        grad = np.zeros(n)
        j_start = max(0, k_star - n + 1)

            return None
        grad = grad / gnorm
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            cand = a - alpha * grad
            cand = np.maximum(cand, 0.0)

            if cs < 1e-12:
                continue
            cand = cand / cs
            val = float(np.convolve(cand, cand).max())
            if val < M * (1.0 - 1e-11):
                return cand
        return None

    def optimize_from(a_init, stop_time, max_iters=200000, seed_val=0):
        """Main LP-based iterative optimization."""
        rng_local = np.random.default_rng(seed_val)
        n = a_init.size

        a = a / a.sum()

        local_best = a.copy()
        local_best_M = float(np.convolve(a, a).max())
        stuck = 0
        perturb_count = 0

        for it in range(max_iters):

                break

            b = np.convolve(a, a)
            M = float(b.max())
            k_max = int(b.argmax())

            if M < local_best_M * (1.0 - 1e-13):
                local_best_M = M
                local_best = a.copy()
                stuck = 0
            else:
                stuck += 1

            if stuck > 70:

                a = a * np.exp(rng_local.normal(0, scale, n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                stuck = 0
                perturb_count += 1
                if perturb_count > 40:
                    break
                continue

            # Determine active set
            frac = 0.88
            thresh = frac * M
            active = np.where(b >= thresh)[0]

            # Ensure we have enough constraints
            if len(active) < 8:
                half_w = max(5, n // 15)

                    kk = k_max + dk
                    if 0 <= kk < len(b):

                            active = np.append(active, kk)
                active = np.unique(active)

            # Cap the active set size
            max_active = 60
            if len(active) > max_active:
                order = np.argsort(b)[::-1][:max_active]
                active = order

            # Build constraint matrix
            A_active = build_active_A(a, active)

            # Strategy 1: max sum(g) LP
            found_move = False
            M_lp = M * (1.0 + 1e-9)
            g = solve_lp_max_sum(A_active, M_lp, n, time_limit=0.8)

            if g is None:
                M_lp = M * (1.0 + 1e-5)
                g = solve_lp_max_sum(A_active, M_lp, n, time_limit=0.5)

            if g is not None and np.any(g > 1e-10):
                g = g / g.sum()
                # Try several directions from LP solution
                directions = [g, 0.7 * g + 0.3 * a, 0.5 * g + 0.5 * a]

                        break
                    dr = dr / np.maximum(dr.sum(), 1e-12)
                    cand, val = line_search_conv(a, dr, M)
                    if cand is not None and val < M * (1.0 - 1e-11):
                        a = cand
                        found_move = True
                        break

            if found_move:
                continue

            # Strategy 2: min cross-correlation LP (use full A for this)
            if not found_move and time.monotonic() < stop_time - 0.5:
                A_full = build_active_A(a, np.arange(len(b)))

                        if time.monotonic() > stop_time:
                            break
                        dr = dr / np.maximum(dr.sum(), 1e-12)
                        cand, val = line_search_conv(a, dr, M)
                        if cand is not None and val < M * (1.0 - 1e-11):
                            a = cand
                            found_move = True
                            break

            if found_move:
                continue

            # Strategy 3: Gradient step
            cand = gradient_step(a, b, M, k_max, n, rng_local)

                a = cand
                continue

            # Strategy 4: Multi-lag gradient (sum of top 3 lags)
            top3 = np.argsort(b)[::-1][:3]
            multi_grad = np.zeros(n)
            for kk in top3:
                j_start = max(0, kk - n + 1)
                j_end = min(n - 1, kk)
