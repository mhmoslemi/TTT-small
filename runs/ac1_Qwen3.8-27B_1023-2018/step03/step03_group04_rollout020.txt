import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)  # numerical cleanup
    return c

def _score_fast(a):
    """Fast score computation (expects a is nonnegative)."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return math.inf
    M = _conv(a).max()
    return 2.0 * n * M / (S * S)

def _build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix."""
    n = len(f)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_solve(f, M, n):
    """Solve the LP to find g maximizing sum(g) s.t. f*g <= M, g >= 0."""
    if M < 1e-15:
        return None
    A = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)]*n,
                      method='highs',
                      options={"presolve": True,
                               "dual_feasibility_tolerance": 1e-10,
                               "primal_feasibility_tolerance": 1e-10,
                               "time_limit": 5})
    except Exception:
        return None
    if res.status == 0 and res.x is not None and res.x.sum() > 1e-10:
        g = np.asarray(res.x, dtype=np.float64)
        return g / g.sum()
    return None

            continue
        d = g - f
        dn = float(np.linalg.norm(d))
        if dn < 1e-11:
            if stall > 80:
                break
            pert = 0.015 * max(1.0, float(np.median(np.maximum(f, 1e-6))))
            f = np.clip(f + rng.normal(0, pert, n), 0, None)
            S2 = f.sum()
            f = f / S2 if S2 > 1e-10 else best_f.copy()
            stall += 1
            continue
        momentum = (1 - mom_alpha) * momentum + mom_alpha * (d / dn)
        best_new_sc = best_sc
        best_new_f = None
        dirs = [g]
        mom_dir = f + 0.4 * momentum
        mom_dir = np.clip(mom_dir, 0, None)
        S_mom = mom_dir.sum()
        if S_mom > 1e-10:
            mom_dir /= S_mom
            if float(np.linalg.norm(mom_dir - f)) > 1e-9 and float(np.dot(d, momentum)) > 0:
                dirs.append(mom_dir)
        for direction in dirs:
            for tc in (t, t * 0.5, t * 1.5, t * 0.3, 0.05, 0.4, 0.8):
                tc = max(0.005, min(1.0, tc))
                fn = (1.0 - tc) * f + tc * direction
                fn = np.clip(fn, 0, None)
                S2 = fn.sum()
                if S2 < 1e-10:
                    continue
                fn /= S2
                sc = _score(fn)
                if sc < best_new_sc:
                    best_new_sc = sc
                    best_new_f = fn.copy()
        if best_new_f is not None:
            f = best_new_f
            if best_new_sc < best_sc - 1e-14:
                best_f = f.copy()
                best_sc = best_new_sc
                stall = 0
                t = min(1.0, t * 1.03)
            else:
                t = max(0.02, t * 0.85)
                stall += 1
        else:
            stall += 1
            t = max(0.02, t * 0.7)
        if stall >= max_stall:
            rf = 0.03 + 0.05 * rng.random()
            f = (1.0 - rf) * best_f + rf * rng.random(n)
            f = np.clip(f, 0, None)
            S2 = f.sum()
            f = f / S2 if S2 > 1e-10 else best_f.copy()
            t = 0.25
            stall = 0
            momentum *= 0.05
    return best_f, best_sc


def _polish(a, stop_time, seed=4242):
    n = a.size
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    S = best.sum()
    if S < 1e-10:
        return best, math.inf
    best /= S
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    mults = [0.0, 0.1, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0, 5.0]
    while time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            bv, bvs = orig, best_sc
            for m in mults:
                v = orig * m if orig > 1e-14 else (0.01 if m > 0.001 else 0.0)
                best[i] = v
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < bvs:
                    bv, bvs = v, sc
            best[i] = bv
            if bvs < best_sc:
                best_sc = bvs
                improved = True
        n_pairs = max(40, n // 3)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 1.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.0, 0.5, 1.0, 1.5, 2.0):
                for mj in (0.0, 0.5, 1.0, 1.5, 2.0):
                    vi = oi * mi if oi > 1e-14 else (0.01 if mi > 0.001 else 0.0)
                    vj = oj * mj if oj > 1e-14 else (0.01 if mj > 0.001 else 0.0)
                    best[i], best[j] = vi, vj
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj
        if not improved:
            no_imp += 1
            if no_imp >= 8:
                break
        else:
            no_imp = 0
    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_p = parent.size

    best_seq = parent.copy()
    try:
        best_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        best_sc = math.inf
    if not math.isfinite(best_sc):
        best_seq = np.ones(256)
        best_sc = _score(best_seq)

    def track(f):
        nonlocal best_seq, best_sc
        f = np.clip(np.asarray(f, dtype=np.float64).ravel(), 0, 1000)
        if f.size == 0 or f.sum() < 0.01 or not np.all(np.isfinite(f)):
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in f]))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_sc:
            best_seq = f.copy()
            best_sc = sc

    # Phase 1: LP optimization at parent length (52%)
    p1_end = t0 + budget * 0.52
    n = n_p
    i_ar = np.arange(n, dtype=np.float64)
    rng0 = np.random.default_rng(seed)

    inits = [
        parent.copy(),
        np.exp(-i_ar / (n * 0.25)),
        0.3 + 0.7 * np.cos(np.pi * i_ar / max(1, n - 1)) ** 2,
        np.exp(-((i_ar - n * 0.3) / (n * 0.15)) ** 2) + np.exp(-((i_ar - n * 0.7) / (n * 0.15)) ** 2) + 0.05,
        np.abs(rng0.standard_normal(n)) + 0.1,
        np.clip(parent + rng0.normal(0, 0.3 * max(1.0, float(np.mean(parent))), n), 0.01, None),
        np.clip(1.0 + 0.3 * np.sin(2 * np.pi * 3.0 * i_ar / n), 0.01, None),
        np.clip(1.0 + 0.2 * np.cos(2 * np.pi * 5.0 * i_ar / n + 0.5), 0.01, None),
    ]
    fracs = [0.30, 0.12, 0.10, 0.10, 0.08, 0.10, 0.09, 0.08]
    p1_dur = max(5.0, p1_end - time.monotonic())
    for idx, init in enumerate(inits):
        if time.monotonic() >= p1_end - 2.0:
            break
        fr = fracs[idx] if idx < len(fracs) else 0.05
        stop_i = min(p1_end - 2.0, time.monotonic() + p1_dur * fr)
        if stop_i <= time.monotonic():
            continue
        try:
            res, sc = _lp_optimize(init, stop_i, seed + 100 * (idx + 1))
            track(res)
        except Exception:
            pass

    # Phase 2: Other lengths (23%)
    p2_end = t0 + budget * 0.75
    if time.monotonic() < p2_end - 5.0:
        for L in [256, 320, 512]:
            if time.monotonic() >= p2_end - 3.0:
                break
            if L == n_p:
                init_l = np.interp(np.linspace(0, 1, L),
                                   np.linspace(0, 1, max(1, len(best_seq))), best_seq)
            else:
                init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n_p), parent)
            remaining = p2_end - 2.0 - time.monotonic()
            n_left = 0
            for Lx in [256, 320, 512]:
                if Lx >= L:
                    n_left += 1
            alloc = max(5.0, remaining / max(1, n_left))
            stop_i = min(p2_end - 2.0, time.monotonic() + alloc)
            try:
                res, sc = _lp_optimize(init_l, stop_i, seed + 50000 + L)
                track(res)
            except Exception:
                pass

    # Phase 3: Polish (remaining time)
    pol_stop = deadline - 6.0
    if time.monotonic() < pol_stop - 10.0:
        try:
            pf, ps = _polish(best_seq, pol_stop)
            track(pf)
        except Exception:
            pass

    if time.monotonic() < deadline - 6.0:
        try:
            pf2, ps2 = _polish(best_seq, deadline - 4.0, seed=seed + 9999)
            track(pf2)
        except Exception:
            pass

    # Final validation
    try:
        fin_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        fin_sc = math.inf
    if not math.isfinite(fin_sc):
        return [1.0] * 16

    return [float(x) for x in best_seq]
