def _fft_conv_same(a):
    """Compute linear convolution a * a (same length as 2n-1)."""
    n = len(a)
    if n == 0:
        return np.array([])
    m = 2*n - 1
    L = next_fast_len(m)
    fa = rfft(a, L)
    c = irfft(fa * fa, L)[:m]
    return np.maximum(c, 0)

def _fft_conv(a, b):
    """Compute linear convolution a * b."""
    n1, n2 = len(a), len(b)
    if n1 == 0 or n2 == 0:
        return np.array([])
    m = n1 + n2 - 1
    L = next_fast_len(m)
    c = irfft(rfft(a, L) * rfft(b, L), L)[:m]
    return np.maximum(c, 0)

def _score(a):

    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return math.inf
    c = _fft_conv_same(a)
    return 2 * len(a) * c.max() / (s * s)

def _lp_update(f, deadline):
    """One LP iteration: find g maximizing sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    f = _normalize(f)
    c = _fft_conv_same(f)
    M = c.max()
    
    # Build constraint matrix A (2n-1) x n
    # A[k, i] = f[i] if 0 <= k-i < n else 0
    A = np.zeros((2*n-1, n))
    for i in range(n):
        if f[i] > 0:
            lo = i  # k - i >= 0 => k >= i
            hi = min(i + n - 1, 2*n - 2)  # k - i <= n-1 => k <= i + n - 1
            if hi >= 0:
                A[lo:hi+1, i] = f[i]
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Solve LP
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.status != 0:
        return f, M
    
    g = np.clip(res.x, 0, None)
    gs = g.sum()
    if gs < 1e-15:
        return f, M
    g = g / gs

= sum_{i!=b} ga[i] * a[i](delta_{ik} - a[k]) + ga[b] * (-a[b]*a[k])

So: gy[k] = a[k] * (ga[k] - dot(a, ga)) for k != b.

# L-BFGS refinement
def _lbfgs_refine(f, etas, deadline, maxfun=600):
    n = len(f)
    f = _norm(f)
    best = f.copy()
    best_sc = _score_arr(f)
    
    for eta in etas:
        if time.monotonic() > deadline:
            break
        c = _conv_same(f)
        M = c.max()
        if M < 1e-15:
            break
        
        b = int(np.argmax(f))
        idx = np.array([i for i in range(n) if i != b])
        base = max(f[b], 1e-300)
        y0 = np.clip(np.log(np.maximum(f[idx], 1e-300) / base), -30, 30)
        
        def fg(y):
            y = np.clip(np.nan_to_num(y, nan=-30, posinf=30, neginf=-30), -30, 30)
            a = np.zeros(n, dtype=np.float64)
            a[b] = 1.0
            a[idx] = np.exp(y)
            Z = a.sum()
            if Z < 1e-300:
                return 1e100, np.zeros(len(y))
            a /= Z
            
            c = _conv_same(a)
            maxc = float(c.max())
            x = eta * (c - maxc)
            x = np.clip(x, -30, 0)
            w = np.exp(x)
            sw = w.sum()
            if sw < 1e-300:
                return 1e100, np.zeros(len(y))
            fun = eta * maxc + math.log(sw)
            w = w / sw

            L2 = next_fast_len(2 * n)
            ga = 2.0 * irfft(np.conj(rfft(a, L2)) * rfft(w, L2), L2)[:n]
            ga = np.where(np.isfinite(ga), ga, 0.0)
            dot = float(np.dot(a, ga))
            gy = a[idx] * (ga[idx] - dot)
            gy = np.where(np.isfinite(gy), gy, 0.0)
            return float(fun), gy
        
        try:
            res = minimize(fg, y0, jac=True, method='L-BFGS-B',
                          options={'maxiter': 200, 'maxfun': maxfun,
                                   'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 30})
            if hasattr(res, 'x') and res.x is not None:

# Coordinate descent polish
def _coord_polish(a, deadline):
    n = len(a)
    a = _norm(a)
    best = a.copy()
    best_sc = _score_arr(a)
    rng = np.random.default_rng(0)
    mults = [0.0, 0.1, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0, 5.0]
    
    for _pass in range(5):
        if time.monotonic() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            orig = float(a[i])
            avg = float(a.mean())
            cands = set()
            cands.add(orig)
            for m in mults:
                v = orig * m
                if 0 <= v <= 1000:
                    cands.add(v)
            for m in (0.5, 1.0, 2.0, 4.0):
                v = avg * m
                if 0 <= v <= 1000:
                    cands.add(v)
            
            bv, bs = orig, best_sc
            for v in cands:
                a[i] = v
                s = a.sum()
                if s < 1e-15:
                    a[i] = orig
                    continue
                sc = 2.0 * n * float(_conv_same(a).max()) / (s * s)
                if sc < bs:
                    bs, bv = sc, v
            a[i] = bv
            if bs < best_sc - 1e-15:
                best = a.copy()
                best_sc = bs
                improved = True
        if not improved:
            break
    
    return best, best_sc
