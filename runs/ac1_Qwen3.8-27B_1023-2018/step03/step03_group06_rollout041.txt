        t_opt = _best_t(a, g, cf, cfs, cg)

        cands = sorted(set([0.0, t_opt, max(0.0, 0.5 * t_opt), min(1.0, 1.5 * t_opt), 0.1, 0.2, 0.4, 0.6, 0.8]))
        best_t = 0.0
        best_sc = best_s
        for t in cands:
            a_new = (1.0 - t) * a + t * g
            a_new = _norm(a_new)
            sc = _score_norm(a_new)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        a_new = (1.0 - best_t) * a + best_t * g
        a_new = _norm(a_new)
        if best_sc < best_s - 1e-14:
            best = a_new.copy()
            best_s = best_sc
            a = best
        else:
            # No improvement: add small noise and continue
            noise = 0.02 * (1.0 / n)
            a_noisy = a + rng.normal(0, noise, n)
            a_noisy = np.clip(a_noisy, 0, None)
            a_noisy = _norm(a_noisy)
            if _score_norm(a_noisy) < best_s:
                best = a_noisy.copy()
                best_s = _score_norm(a_noisy)
            a = best
            # If stuck for too long, try larger noise
            if it % 20 == 0:
                a_noisy = a + rng.normal(0, 0.05 / n, n)
                a_noisy = np.clip(a_noisy, 0, None)
                a_noisy = _norm(a_noisy)
                if _score_norm(a_noisy) < best_s:
                    best = a_noisy.copy()
                    best_s = _score_norm(a_noisy)
                a = best
    return best, best_s

def _smooth_optimize(a0, stop, beta_init=50.0, maxfun=600):
    """L-BFGS with log-sum-exp smooth approximation of max."""
    n = a0.size
    if n < 2:
        return a0, _score_norm(a0)
    best = a0.copy()
    best_s = _score_norm(best)

    c = _conv(best)
    M = float(c.max())
    if M < 1e-15:
        return best, best_s

    # Work in log-space: a_i = exp(y_i) / sum(exp(y))
    b_idx = int(np.argmax(best))
    idx = np.array([i for i in range(n) if i != b_idx], dtype=int)
    base = max(best[b_idx], 1e-30)
    y0 = np.log(np.maximum(best[idx], 1e-30) / base)
    y0 = np.clip(y0, -50.0, 50.0)

    def _softmax_y(y, b, n, idx):
        y = np.clip(np.nan_to_num(y), -50.0, 50.0)
        ex = np.exp(y)
        Z = 1.0 + ex.sum()
        a = np.empty(n, dtype=np.float64)
        a[b] = 1.0 / Z
        a[idx] = ex / Z
        return a

    def fg(y, beta):
        if time.monotonic() > stop:
            raise ValueError("timeout")
        y = np.nan_to_num(np.asarray(y, dtype=np.float64), nan=-30.0, posinf=30.0, neginf=-30.0)
        y = np.clip(y, -50.0, 50.0)
        a = _softmax_y(y, b_idx, n, idx)
        c = _conv(a)
        # Log-sum-exp approximation of max
        c_pos = c - c.max()
        Z_lse = float(np.exp(beta * c_pos).sum())
        M_lse = c.max() + math.log(Z_lse) / beta
        if not math.isfinite(M_lse):
            return 1e100, np.zeros_like(y)
        # Gradient of log-sum-exp w.r.t. c
        w = np.exp(beta * (c - c.max())) / Z_lse
        # dc/da_i = 2*a_{k-i} summed over k with weight w_k
        # grad_a_i = 2 * sum_k w_k * a_{k-i}
        ar = a[::-1].copy()
        wc = _conv(w, ar)
        grad_a = 2.0 * wc[n - 1:2 * n - 1]
        # Convert to y-space: a_i = exp(y_i)/Z for i in idx
        # da_i/dy_j = a_i*(delta_ij - a_j)
        dot = float(np.dot(a, grad_a))
        grad_y = a[idx] * (grad_a[idx] - dot)
        # Subtract gradient of log(Z) = sum a_i * 1 = 1, but that's already handled
        # Actually we're minimizing M_lse, not M_lse + anything else
        # Since sum(a)=1 is enforced by softmax, we just minimize M_lse
        fun = M_lse
        grad_y = np.where(np.isfinite(grad_y), grad_y, 0.0)
        return fun, grad_y

    for beta in [beta_init, beta_init * 3, beta_init * 10]:
        if time.monotonic() > stop - 5:
            break
        try:
            res = minimize(fg, y0, args=(beta,), jac=True, method='L-BFGS-B',
                           options={'maxiter': 300, 'maxfun': maxfun, 'ftol': 1e-14,
                                    'gtol': 1e-10, 'maxls': 50})
        except Exception:
            break
        if res.x is not None:
            a_new = _softmax_y(res.x, b_idx, n, idx)
            sc = _score_norm(a_new)
            if sc < best_s:
                best = a_new.copy()
                best_s = sc
            # Update y0 for next beta
            base2 = max(best[b_idx], 1e-30)
            y0 = np.log(np.maximum(best[idx], 1e-30) / base2)
            y0 = np.clip(y0, -50.0, 50.0)

    return best, best_s

def _mirror_descent(a0, stop, eta=0.01):
    """Multiplicative weights / mirror descent."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_s = _score_norm(a)

    for _ in range(5000):
        if time.monotonic() > stop:
            break
        c = _conv(a)
        M = float(c.max())
        if M < 1e-15:
            break
        # Active set
        tol = 1e-4 * M
        active = (c >= M - tol).astype(float)
        if active.sum() < 1:
            active = (c >= M).astype(float)
            if active.sum() < 1:
                active[int(np.argmax(c))] = 1.0
        # Subgradient: g_i = (2/|A|) * sum_k active_k * a_{k-i}
        ar = a[::-1].copy()
        v = _conv(active, ar)[n - 1:2 * n - 1]
        g = 2.0 * v / active.sum()
        # Mirror descent: a_i *= exp(-eta * g_i)
        a_new = a * np.exp(-eta * g)
        a_new = _norm(a_new)
        sc = _score_norm(a_new)
        if sc < best_s:
            best = a_new.copy()
            best_s = sc
        a = a_new
    return best, best_s

def _coord_polish(a0, stop, n_passes=2):
    """Coordinate descent polish."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_s = _score_norm(a)
    rng = np.random.default_rng(777)

    for _ in range(n_passes):
        if time.monotonic() > stop - 1:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop - 1:
                break
            orig = a[i]
            best_val = orig
            best_sc = best_s
            for val in [0.0, orig * 0.1, orig * 0.3, orig * 0.5, orig * 0.7,
                        orig, orig * 1.3, orig * 1.7, orig * 2.5, 4.0 / n]:
                a[i] = val
                sc = _score_norm(a)
                if sc < best_sc:
                    best_sc = sc
                    best_val = val
            a[i] = best_val
            if best_sc < best_s - 1e-15:
                best = a.copy()
                best_s = best_sc
                improved = True
        if not improved:
            break
    return best, best_s

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1300.0
    hard_end = t0 + budget
    deadline = hard_end - 20.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.where(np.isfinite(parent), parent, 0.0)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() < 1e-10:
            parent = np.ones(256, dtype=np.float64) / 256.0
    except Exception:
        parent = np.ones(256, dtype=np.float64) / 256.0
    parent = _norm(parent)

    best = parent.copy()
    best_s = _score_norm(parent)
    n0 = parent.size

    def _try(a):
        nonlocal best, best_s
        a = _norm(a)
        if a.size == 0:
            return
        sc = _score_norm(a)
        if math.isfinite(sc) and sc < best_s - 1e-14:
            best_s = sc
            best = a.copy()

    def _finalize():
        seq = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
        if seq.size == 0 or seq.sum() < 0.01:
            seq = np.ones(32, dtype=np.float64) / 32.0
        # Verify with exact evaluator
        lst = [float(x) for x in seq]
        try:
            sc = _exact(lst)
            if not math.isfinite(sc):
                lst = [float(x) for x in parent]
        except Exception:
            pass
        return lst

    rng = np.random.default_rng(seed)

    # === Phase 1: LP iterations at base length (main driver) ===
    lp_stop1 = min(deadline - 300.0, t0 + 0.40 * budget)
    if time.monotonic() < lp_stop1 - 10:
        a_lp, _ = _lp_iterate(best, lp_stop1, seed=int(rng.integers(1, 10**7)))
        _try(a_lp)

    # === Phase 2: Smooth optimization (L-BFGS) ===
    sm_stop = min(deadline - 250.0, t0 + 0.55 * budget)
    if time.monotonic() < sm_stop - 10:
        a_sm, _ = _smooth_optimize(best, sm_stop, beta_init=80.0, maxfun=800)
        _try(a_sm)

    # === Phase 3: More LP iterations from improved point ===
    lp_stop2 = min(deadline - 200.0, t0 + 0.68 * budget)
    if time.monotonic() < lp_stop2 - 10:
        a_lp2, _ = _lp_iterate(best, lp_stop2, seed=int(rng.integers(1, 10**7)))
        _try(a_lp2)

    # === Phase 4: Mirror descent for additional refinement ===
    md_stop = min(deadline - 150.0, t0 + 0.78 * budget)
    if time.monotonic() < md_stop - 10:
        for eta in [0.005, 0.002, 0.001]:
            if time.monotonic() > md_stop - 5:
                break
            a_md, _ = _mirror_descent(best, min(md_stop - 2, time.monotonic() + 40.0), eta=eta)
            _try(a_md)

    # === Phase 5: Another smooth pass with higher beta ===
    sm2_stop = min(deadline - 100.0, t0 + 0.85 * budget)
    if time.monotonic() < sm2_stop - 10:
        a_sm2, _ = _smooth_optimize(best, sm2_stop, beta_init=200.0, maxfun=500)
        _try(a_sm2)

    # === Phase 6: Try other lengths ===
    other_stop = min(deadline - 60.0, t0 + 0.92 * budget)
    cur_n = best.size
    other_lens = [L for L in [128, 384, 512, 192] if L != cur_n][:3]
    for L in other_lens:
        if time.monotonic() > other_stop - 15:
            break
        if best_s < 1.01:
            break
        # Interpolate
        x0 = np.linspace(0, 1, cur_n)
        x1 = np.linspace(0, 1, L)
        a_L = np.interp(x1, x0, best)
        a_L = _norm(a_L)
        sc_L = _score_norm(a_L)
        # Quick LP refinement
        alloc = max(15.0, (other_stop - time.monotonic()) / max(1, len(other_lens)))
        a_L2, _ = _lp_iterate(a_L, time.monotonic() + alloc, seed=int(rng.integers(1, 10**7)))
        _try(a_L2)

    # === Phase 7: Final LP + smooth ===
    fin_stop = min(deadline - 30.0, t0 + 0.96 * budget)
    if time.monotonic() < fin_stop - 10:
        a_fin, _ = _lp_iterate(best, min(fin_stop - 15, time.monotonic() + 60.0), seed=int(rng.integers(1, 10**7)))
        _try(a_fin)
        if time.monotonic() < fin_stop - 8:
            a_sm3, _ = _smooth_optimize(best, fin_stop - 3, beta_init=400.0, maxfun=400)
            _try(a_sm3)

    # === Phase 8: Coordinate polish ===
    cpol_stop = deadline - 5.0
    if time.monotonic() < cpol_stop - 3:
        a_cp, _ = _coord_polish(best, cpol_stop, n_passes=2)
        _try(a_cp)

    return _finalize()
