```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    return p

def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_fft(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_solve(f, n, M):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": 20.0},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs
```

```python
import math

from scipy.optimize import linprog

def _get_parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(320, dtype=np.float64)
    if p.size == 0:

    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(320, dtype=np.float64)
    return p

def _conv_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_fft(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_solve(f, n, M):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": 15.0})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs

def _line_search(f, g, n, best_sc):
    """Golden section line search for (1-t)*f + t*g"""
    a_lo, a_hi = 0.0, 1.0
    gr = (math.sqrt(5) - 1) / 2  # golden ratio
    
    def F(t):
        fn = (1.0 - t) * f + t * g
        np.maximum(fn, 0.0, out=fn)
        s = float(fn.sum())
        if s < 1e-10:
            return math.inf
        return _score(fn)
    
    f_lo = F(a_lo)
    f_hi = F(a_hi)
    
    # First check if t=1 is already good
    if f_hi < f_lo:
        return 1.0, F
    
    t1 = a_hi - gr * (a_hi - a_lo)
    t2 = a_lo + gr * (a_hi - a_lo)
    ft1 = F(t1)
    ft2 = F(t2)
    
    best_t = 0.0
    best_f = f_lo
    
    for _ in range(20):
        if ft1 < ft2:
            a_hi = t2
            t2 = t1
            ft2 = ft1
            t1 = a_hi - gr * (a_hi - a_lo)
            ft1 = F(t1)
        else:
            a_lo = t1
            t1 = t2
            ft1 = ft2
            t2 = a_lo + gr * (a_hi - a_lo)
            ft2 = F(t2)
        
        if ft1 < best_f:
            best_f = ft1
            best_t = t1
        if ft2 < best_f:
            best_f = ft2
            best_t = t2
        
        if a_hi - a_lo < 1e-5:
            break
    
    return best_t, F
```
