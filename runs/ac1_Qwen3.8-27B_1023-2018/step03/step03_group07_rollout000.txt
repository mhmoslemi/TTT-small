import numpy as np

_CTX = {}


class _Ctx:
    __slots__ = ("n", "m", "center", "mask", "clip")

    def __init__(self, n: int):
        self.n = int(n)
        self.m = 2 * self.n - 1
        self.center = self.n - 1
        k = np.arange(self.m, dtype=np.int64)[:, None]
        j = np.arange(self.n, dtype=np.int64)[None, :]
        idx = k - j
        self.mask = (idx >= 0) & (idx < self.n)
        self.clip = idx.clip(0, self.n - 1)


def _get_ctx(n: int) -> _Ctx:
    n = int(n)
    ctx = _CTX.get(n)
    if ctx is None:
        ctx = _Ctx(n)
        _CTX[n] = ctx
    return ctx


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    if not np.all(np.isfinite(a)):
        a = np.ones(a.size, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def _eval_array(a) -> float:
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
        if arr.size == 0:
            return float("inf")
        arr = np.clip(arr, 0.0, 1000.0)
        if not np.all(np.isfinite(arr)):
            return float("inf")
        if float(arr.sum()) < 0.01:
            return float("inf")

def _resample(a, n: int):
    n = int(n)
    if n <= 0:
        return np.array([1.0])
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a)
    return _normalize(out)


def _raw_info(a, score, c, ctx, default_exclude_center=False):
    n = ctx.n
    s = float(a.sum())
    maxc = float(np.max(c)) if c.size else 0.0
    if np.isfinite(score):
        m = score * s * s / (2.0 * float(n))
    else:
        m = 0.0

    if not np.isfinite(m) or m <= 0.0:
        m = maxc
    elif maxc > 0.0 and m > maxc + max(1e-12, 1e-8 * maxc):
        m = maxc

    if m <= 0.0 and maxc > 0.0:
        # If the objective reports zero, keep that raw value if it is plausibly
        # an off-peak objective; otherwise fall back to maxc.
        center_val = float(c[ctx.center]) if c.size else 0.0
        if default_exclude_center and center_val > 0.0:
            m = 0.0
        else:
            m = maxc

    tol = max(1e-12, 1e-8 * max(1.0, abs(float(m))))

    if not np.isfinite(center_val):
        exclude_center = bool(default_exclude_center)
    else:
        exclude_center = bool(center_val > m + tol)

    return float(m), float(tol), exclude_center


def _uniform_exclude_center(ctx) -> bool:

    u = np.ones(n, dtype=np.float64) / float(n)
    su = _eval_array(u)

    mu = su / (2.0 * float(n))
    if not np.isfinite(mu):
        return False
    center = float(cu[ctx.center]) if cu.size else 0.0
    tol = max(1e-10, 1e-7 * max(1.0, abs(mu)))
    return bool(center > mu + tol)


def _build_A(a, rows, ctx):

    if rows is None or len(rows) == 0:
        return np.zeros((0, n), dtype=np.float64)
    idx = ctx.clip[rows]
    vals = np.take(a, idx, axis=0).astype(np.float64)
    return np.where(ctx.mask[rows], vals, 0.0)


def _select_rows(c, m, tol, exclude_center, ctx, mode="active", frac=0.15):

    center = ctx.center

    if mode == "all":

        if exclude_center:

        return rows

    if mode == "top":

    if rows.size > max(60, total // 2):
        rows = _select_rows(c, m, tol, exclude_center, ctx, "top", 0.30)
    return rows


def _lp_minmax(a, A, deadline, time_limit=0.35):
    now = time.monotonic()
    if now >= deadline - max(0.25, time_limit):
        return None
    R, n = A.shape
    if R == 0 or n <= 0:
        return None

    obj = np.zeros(n + 1, dtype=np.float64)
    obj[n] = 1.0

    A_ub = np.empty((R, n + 1), dtype=np.float64)
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(R, dtype=np.float64)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    bounds = [(0.0, None)] * n + [(None, None)]

    specs = (
        ("highs", True),
        ("highs", False),
        (None, False),
    )

    for method, use_limit in specs:
        opts = {
            "output_flag": 0,
            "primal_feasibility_tolerance": 1e-9,
            "dual_feasibility_tolerance": 1e-9,
        }
        if use_limit and time_limit is not None:
            opts["time_limit"] = float(time_limit)
            opts["presolve"] = True

        try:
            if method is None:
                res = linprog(obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds, options=opts)
            else:
                res = linprog(obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds, method=method, options=opts)
        except Exception:
            continue

        if res is not None and getattr(res, "status", 1) == 0 and getattr(res, "x", None) is not None:
            g = np.asarray(res.x[:n], dtype=np.float64)
            g = np.clip(g, 0.0, None)
            if not np.all(np.isfinite(g)):
                continue
            s = float(g.sum())
            if s > 1e-12:
                return g / s
        break

    return None


def _subgradient_dirs(a, c, m, tol, exclude_center, ctx):
    n = ctx.n
    center = ctx.center

    active_tol = max(1e-10, 0.03 * abs(float(m)))
    rows = np.where(c >= m - active_tol)[0]
    if exclude_center:
        rows = rows[rows != center]

    if rows.size == 0:
        p75 = float(np.percentile(c, 75)) if c.size else 0.0
        rows = np.where(c >= p75)[0]
        if exclude_center:
            rows = rows[rows != center]

    if rows.size > 35:
        local_order = np.argsort(c[rows], kind="stable")
        rows = rows[local_order[-35:]]

    grad = np.zeros(n, dtype=np.float64)
    for k in rows:
        d = int(k - center)
        dd = abs(d)
        if dd == 0:
            grad += 2.0 * a
        elif dd < n:
            grad[:n - dd] += a[dd:]
            grad[dd:] += a[:-dd]

    gproj = grad - float(grad.mean())
    gnorm = float(np.linalg.norm(gproj))
    if not np.isfinite(gnorm) or gnorm <= 1e-15:
        return []
    gproj /= gnorm

    dirs = []
    for beta in (0.01, 0.03, 0.10, 0.30, 0.80, 1.50):
        g = a - beta * gproj
        g = np.maximum(g, 0.0)
        s = float(g.sum())
        if s <= 1e-12:
            continue
        g = g / s
        if not np.allclose(g, a, atol=1e-7, rtol=1e-7):
            dirs.append(g)
    return dirs


def _line_search(a, g, base_score, deadline, max_evals=24):
    best_s = float(base_score)
    best_t = 0.0
    best_cand = None
    evals = 0

    def try_t(t):
        nonlocal evals, best_s, best_t, best_cand
        if t <= 0.0 or t > 1.0:
            return
        if evals >= max_evals:
            return
        if time.monotonic() >= deadline:
            return
        cand = (1.0 - t) * a + t * g
        cand = np.maximum(cand, 0.0)
        s = float(cand.sum())
        if s <= 1e-300:
            return
        if abs(s - 1.0) > 1e-7:
            cand = cand / s
        sc = _eval_array(cand)
        evals += 1
        if np.isfinite(sc) and sc < best_s - 1e-13:
            best_s = float(sc)
            best_t = float(t)
            best_cand = cand

    coarse = (0.001, 0.003, 0.008, 0.018, 0.04, 0.08, 0.15, 0.28, 0.45, 0.70, 1.0)
    for t in coarse:
        try_t(t)
        if time.monotonic() >= deadline or evals >= max_evals:
            break

    if best_t == 0.0:
        return None, best_s

    center = best_t
    radius = 0.08
    for _ in range(4):
        if time.monotonic() >= deadline or evals >= max_evals or radius < 1e-5:
            break
        for fac in (-1.0, -0.5, 0.0, 0.5, 1.0):
            t = center + fac * radius
            try_t(t)
        center = best_t
        radius *= 0.45

    return best_cand, best_s


def _generate_directions(a, c, m, tol, exclude_center, ctx, best_a, stop_time, use_lp=True):
    dirs = []
    n = a.size

    if best_a is not None and best_a.size == n:
        if not np.allclose(a, best_a, atol=1e-8, rtol=1e-8):
            dirs.append(np.asarray(best_a, dtype=np.float64).copy())

    if not exclude_center:
        u = np.ones(n, dtype=np.float64) / float(n)
        if not np.allclose(a, u, atol=1e-8, rtol=1e-8):
            dirs.append(u)

    if use_lp and time.monotonic() < stop_time - 0.5:
        rem = max(0.0, stop_time - time.monotonic())
        max_lp = 3 if rem > 180.0 else (2 if rem > 60.0 else 1)

        modes = [("active", None), ("top", 0.08), ("top", 0.25)]
        if exclude_center:
            modes.append(("top", 0.45))
        else:
            modes.append(("all", None))

        seen_lp = 0
        for mode, frac in modes:
            if seen_lp >= max_lp:
                break
            if time.monotonic() >= stop_time - 0.7:
                break
            rows = _select_rows(c, m, tol, exclude_center, ctx, mode, 0.10 if frac is None else frac)
            if rows is None or rows.size == 0:
                continue
            A = _build_A(a, rows, ctx)
            tl = 0.45 if rows.size > 120 else 0.30
            g = _lp_minmax(a, A, stop_time, time_limit=tl)
            seen_lp += 1
            if g is not None and not np.allclose(g, a, atol=1e-7, rtol=1e-7):
                dirs.append(g)

    sg = _subgradient_dirs(a, c, m, tol, exclude_center, ctx)
    dirs.extend(sg[:2])

    uniq = []
    for g in dirs:
        if g is None:
            continue
        g = np.asarray(g, dtype=np.float64).ravel()
        if g.size != n or not np.all(np.isfinite(g)):
            continue
        g = np.clip(g, 0.0, 1000.0)
        s = float(g.sum())
        if s <= 1e-300:
            continue
        g = g / s
        duplicate = False
        for h in uniq:
            if np.allclose(g, h, atol=1e-6, rtol=1e-6):
                duplicate = True
                break
        if not duplicate:
            uniq.append(g)
    return uniq


def _perturb_current(a, best, rng, stop_time):
    n = a.size
    cands = []

    def add(cand):
        if time.monotonic() >= stop_time:
            return
        c = _normalize(cand)
        if c.size != n:
            return
        sc = _eval_array(c)
        if np.isfinite(sc):
            cands.append((float(sc), c))

    add(best * np.exp(rng.normal(0.0, 0.02, n)))
    add(best * np.exp(rng.normal(0.0, 0.05, n)))

    for sigma in (0.03, 0.08, 0.15, 0.28):
        add(a * np.exp(rng.normal(0.0, sigma, n)))

    for lam in (0.2, 0.5, 0.8):
        add((1.0 - lam) * a + lam * best)

    for _ in range(4):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        e = np.zeros(n, dtype=np.float64)
        e[i] = 0.6
        e[j] = 0.4
        e /= e.sum()
        for lam in (0.3, 0.6, 0.9):
            add((1.0 - lam) * a + lam * e)

    for w in (3, 5, 11):
        k = np.ones(w, dtype=np.float64) / float(w)
        add(np.convolve(a, k, mode="same"))

    x = np.arange(n, dtype=np.float64)
    for amp in (0.1, 0.3, 0.5):
        f = float(rng.uniform(0.5, 5.0))
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        add(a * (1.0 + amp * np.cos(2.0 * math.pi * f * x / float(n) + ph)))

    if not cands:
        sc_a = _eval_array(a)
        sc_b = _eval_array(best)
        return a.copy(), float(sc_a), best.copy(), float(sc_b)

    cands.sort(key=lambda x: x[0])
    best_sc, best_cand = cands[0]

    idx = 0
    if len(cands) >= 3:
        r = float(rng.random())
        if r < 0.60:
            idx = 0
        elif r < 0.85:
            idx = 1
        else:
            idx = 2
    elif len(cands) == 2:
        idx = 0 if float(rng.random()) < 0.70 else 1

    new_sc, new_cand = cands[idx]

    if new_sc > best_sc + 1e-9:
        blend = _normalize(0.55 * best_cand + 0.45 * new_cand)
        scb = _eval_array(blend)
        if np.isfinite(scb) and scb < new_sc:
            new_sc = float(scb)
            new_cand = blend
        if np.isfinite(scb) and scb < best_sc:
            best_sc = float(scb)
            best_cand = blend

    return new_cand, float(new_sc), best_cand, float(best_sc)


def _optimize(a_init, stop_time, seed=0):
    a = _normalize(a_init)
    ctx = _get_ctx(a.size)
    n = ctx.n

    score = _eval_array(a)
    if not np.isfinite(score):
        u = np.ones(n, dtype=np.float64) / float(n)
        a = u
        score = _eval_array(u)
        if not np.isfinite(score):
            score = float("inf")

    best_a = a.copy()
    best_score = float(score)
    if best_score <= 1e-10:
        return best_a

    rng = np.random.default_rng(seed)
    default_exclude = _uniform_exclude_center(ctx)
    stuck = 0

    while time.monotonic() < stop_time:
        if best_score <= 1e-10:
            break

        c = np.convolve(a, a)
        m, tol, exclude_center = _raw_info(a, score, c, ctx, default_exclude)

        stop_inner = stop_time - 0.2
        dirs = _generate_directions(a, c, m, tol, exclude_center, ctx, best_a, stop_inner, use_lp=True)
        improved = False

        for g in dirs:
            if time.monotonic() >= stop_inner:
                break
            if np.allclose(g, a, atol=1e-8, rtol=1e-8):
                continue

            rem = max(0.0, stop_time - time.monotonic())
            max_evals = 26 if rem > 120.0 else (18 if rem > 45.0 else 10)

            cand, sc = _line_search(a, g, score, stop_time - 0.1, max_evals=max_evals)
            if cand is None or not np.isfinite(sc):
                continue

            if sc < best_score - 1e-13:
                best_a = cand.copy()
                best_score = float(sc)

            if sc < score - 1e-13:
                a = cand.copy()
                score = float(sc)
                improved = True
                stuck = 0

                if best_score <= 1e-10:
                    break

                if best_score >= 2.0 - 1e-7:
                    cb = np.convolve(best_a, best_a)
                    mb, _, excl_b = _raw_info(best_a, best_score, cb, ctx, default_exclude)
                    if not excl_b:
                        break
                break

        if not improved:
            stuck += 1
            if stuck >= 3 and time.monotonic() < stop_time - 5.0:
                new_a, new_score, new_best, new_best_score = _perturb_current(a, best_a, rng, stop_time - 0.5)
                if np.isfinite(new_best_score) and new_best_score < best_score:
                    best_a = new_best.copy()
                    best_score = float(new_best_score)
                if np.isfinite(new_score):
                    a = new_a.copy()
                    score = float(new_score)
                else:
                    a = best_a.copy()
                    score = float(best_score)
                stuck = 0
            elif time.monotonic() >= stop_time:
                break

    final_sc = _eval_array(best_a)
    if np.isfinite(final_sc) and final_sc < best_score:
        best_a = _normalize(best_a)
        best_score = float(final_sc)

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 30.0
    deadline = t0 + max(10.0, budget - 35.0)

    parent = _get_parent()
    n = int(parent.size)
    ctx = _get_ctx(n)
    default_exclude = _uniform_exclude_center(ctx)
    rng = np.random.default_rng(seed)

    best_a = None
    best_score = float("inf")

    def consider(arr):
        nonlocal best_a, best_score
        if arr is None:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size == 0:
            return
        arr = np.clip(arr, 0.0, 1000.0)
        if not np.all(np.isfinite(arr)):
            arr = np.ones(n, dtype=np.float64)
        s = float(arr.sum())
        if s <= 1e-300:
            arr = np.ones(n, dtype=np.float64) / float(n)
        else:
            arr = arr / s
        sc = _eval_array(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_a = arr.copy()

    uniform = np.ones(n, dtype=np.float64) / float(n)
    inits = [parent, uniform]
    if n > 1:
        inits.append(parent[::-1])

    for w in (3, 7, 15):
        try:
            inits.append(np.convolve(parent, np.ones(w, dtype=np.float64) / float(w), mode="same"))
        except Exception:
            pass

    if n > 1:
        x = np.arange(n, dtype=np.float64)
        for amp, f, ph in [(0.20, 1.0, 0.0), (0.40, 2.0, math.pi / 3.0), (0.60, 3.0, math.pi)]:
            mod = parent * (1.0 + amp * np.cos(2.0 * math.pi * f * x / float(n) + ph))
            inits.append(mod)

    inits.append(parent * np.exp(rng.normal(0.0, 0.10, n)))

    for arr in inits:
        consider(arr)

    if best_a is None:
        best_a = uniform
        best_score = _eval_array(uniform)

    # If the exact evaluator behaves like the literal nonnegative formula with
    # the zero lag included, the uniform vector is globally optimal (score = 2).
    if np.isfinite(best_score) and best_score >= 2.0 - 1e-7:
        cb = np.convolve(best_a, best_a)
        _, _, exclude_b = _raw_info(best_a, best_score, cb, ctx, default_exclude)
        if not exclude_b:
            out = np.clip(np.asarray(best_a, dtype=np.float64), 0.0, 1000.0).tolist()
            if out and all(np.isfinite(v) for v in out):
                try:
                    final_sc = evaluate_sequence(out)
                except Exception:
                    final_sc = float("inf")
                if np.isfinite(final_sc):
                    return out
            return [1.0] * max(1, n)

    primary_stop = min(deadline - 8.0, max(t0 + 10.0, deadline - 90.0))
    if time.monotonic() < primary_stop:
        try:
            res = _optimize(best_a, primary_stop, seed=int(seed) + 1)
            consider(res)
        except Exception:
            pass

    # Polish the current global best if meaningful time remains.
    if time.monotonic() < deadline - 25.0:
        polish_stop = min(deadline - 20.0, time.monotonic() + max(8.0, (deadline - time.monotonic()) * 0.60))
        if polish_stop > time.monotonic() + 5.0:
            try:
                res = _optimize(best_a, polish_stop, seed=int(seed) + 777)
                consider(res)
            except Exception:
                pass

    # Light-dimensional length refinement around the parent length.
    if time.monotonic() < deadline - 18.0:
        lengths = []
        for d in (-8, -4, -2, -1, 1, 2, 4, 8):
            nt = n + d
            if 16 <= nt <= 450 and nt != n:
                lengths.append(nt)
        if lengths:
            lengths = sorted(set(lengths))

            for nt in lengths:
                if time.monotonic() >= deadline - 12.0:
                    break
                alloc = max(0.0, min(12.0, (deadline - 12.0 - time.monotonic()) * 0.55))
                if alloc < 2.0:
                    break
                init = _resample(best_a, int(nt))
                consider(init)
                try:
                    res = _optimize(init, time.monotonic() + alloc, seed=10000 + int(nt))
                    consider(res)
                except Exception:
                    pass

    if best_a is None:
        best_a = uniform

    final_arr = np.clip(np.asarray(best_a, dtype=np.float64).ravel(), 0.0, 1000.0)
    if final_arr.size == 0 or not np.all(np.isfinite(final_arr)) or float(final_arr.sum()) < 0.01:
        final_arr = np.ones(max(1, n), dtype=np.float64)
        final_arr /= float(final_arr.size)

    out = [float(v) for v in final_arr]
    try:
        final_sc = evaluate_sequence(out)
    except Exception:
        final_sc = float("inf")

    if not np.isfinite(final_sc) or len(out) == 0:
        out = [1.0] * max(1, n)

    return out
