        if diff < 1e-10 * max(1.0, np.linalg.norm(f)):
            stall += 2
            if stall > max_stall:
                break
            continue

        b_g = np.convolve(g, g, mode="full")
        b_fg = np.convolve(f, g, mode="full")

        # Efficient line search using precomputed convolutions
        best_local_t = 0.0
        best_local_sc = best_sc
        for t in t_vals:
            one_minus_t = 1.0 - t
            combined = one_minus_t * one_minus_t * b_f + 2.0 * t * one_minus_t * b_fg + t * t * b_g
            max_b = float(combined.max())
            sc = 2.0 * n * max_b / (f_sum * f_sum)
            if sc < best_local_sc:
                best_local_sc = sc
                best_local_t = t

        if best_local_t > 0:
            t = best_local_t
            f = (1.0 - t) * f + t * g
            f = np.maximum(f, 0)
            f_sum = float(f.sum())
            if f_sum < 0.01:
                f = best_f.copy()
                f_sum = float(best_f.sum())
                continue

            sc = _score_fast(f)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = f.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall >= max_stall:
            f = np.maximum(best_f + rng.normal(0, 0.08 * max(0.01, best_f.mean()), n), 0)
            f_sum = float(f.sum())
            stall = 0
            max_stall += 15

        iteration += 1

    return best_f, best_sc

def _subgradient_polish(a, stop_time, seed):
    """Subgradient method targeting the active peak."""
    n = len(a)
    best = np.maximum(a.astype(np.float64).copy(), 0)
    best_s = float(best.sum())
    if best_s < 0.01:
        return best, math.inf
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    lr = 0.1 / n
    iterations = 0
    patience = 200

    while time.monotonic() < stop_time:
        b = np.convolve(best, best, mode="full")
        k_star = int(np.argmax(b))
        M = float(b[k_star])

        # Subgradient of (a*a)[k_star] w.r.t. a
        grad = np.zeros(n)
        for j in range(n):
            k1 = k_star - j
            k2 = j  # a[j] * a[k_star - j], derivative w.r.t. a[j] is a[k_star-j]
            if 0 <= k1 < n:
                grad[j] += best[k1]
            k3 = j  # from the other side: a[k_star-j]*a[j], derivative w.r.t. a[i] where i = k_star-j
            # Actually d/d a[i] of sum_j a[j]*a[k-j] includes a[k-i] + a[i-k] for valid indices
            k4 = k_star - i if 0 <= k_star - i < n else -1

        # Redo properly: (a*a)[k] = sum_j a[j]*a[k-j] for valid j
        # d/d a[i] = a[k-i] (from j=i term) + a[i-k+... ] 
        # More carefully: if i is in the sum as j=i, contribution is a[i]*a[k-i], deriv w.r.t a[i] is a[k-i]
        # if k-i is in the sum as j=k-i, contribution is a[k-i]*a[i], deriv w.r.t a[i] is a[k-i]
        # So d/d a[i] = a[k-i] if k-i is valid, plus a[k-i] if i is valid (always), 
        # but we need to be careful about double counting when i = k-i

        grad = np.zeros(n)
        for i in range(n):
            val = 0.0
            # i appears as j in the sum: a[i]*a[k-i], valid if 0 <= k-i < n
            if 0 <= k_star - i < n:
                val += best[k_star - i]
            # i appears as (k-j) in the sum: a[j]*a[i] where j = k-i, valid if 0 <= k-i < n
            # This is the same term! So for i != k-i, we get a[k-i] + a[k-i] = 2*a[k-i]
            # For i == k-i (i.e., 2i = k), we get a[i] (single term, derivative is a[i])
            if 2 * i != k_star and 0 <= k_star - i < n:
                val += best[k_star - i]
            grad[i] = val

        # Project gradient onto the constraint sum(a) = best_s
        mean_grad = grad.mean()
        grad_dir = grad - mean_grad  # zero-mean direction (preserves sum)

        # Take step
        step = lr * grad_dir
        new_a = best - step
        new_a = np.maximum(new_a, 0)
        # Renormalize to preserve sum
        s_new = float(new_a.sum())
        if s_new < 0.01:
            continue
        new_a = new_a * (best_s / s_new)

        sc = _score_fast(new_a)
        if sc < best_sc - 1e-14:
            best = new_a.copy()
            best_sc = sc
            iterations = 0
            lr = min(lr * 1.1, 1.0 / n)
        else:
            iterations += 1
            lr *= 0.9
            if iterations > patience:
                break

    return best, best_sc

def _polish(a, stop_time, seed):
    """Multi-method polishing."""
    n = len(a)
    best = np.maximum(a.astype(np.float64).copy(), 0)
    best_s = float(best.sum())
    if best_s < 0.01:
        return best, math.inf
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 30

    while time.monotonic() < stop_time:
        improved = False

        # Single-coordinate
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                return best, best_sc
            orig = best[i]
            if orig < 1e-12:
                continue
            best_i_val = orig
            best_i_sc = best_sc
            for mlt in [0.0, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.6, 2.0, 3.0]:
                best[i] = orig * mlt
                s = float(best.sum())
                if s < 0.01:
                    best[i] = orig
                    continue
                b = np.convolve(best, best, mode="full")
                m = float(b.max())
                sc = 2.0 * n * m / (s * s)
                if sc < best_i_sc - 1e-15:
                    best_i_sc = sc
                    best_i_val = orig * mlt
            best[i] = best_i_val
            if best_i_sc < best_sc - 1e-15:
                best_sc = best_i_sc
                improved = True

        # Pairs
        n_pairs = min(300, max(50, int((stop_time - time.monotonic()) * 5)))
        for _ in range(n_pairs):
            if time.monotonic() > stop_time:
                return best, best_sc
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            if oi < 1e-12 and oj < 1e-12:
                continue
            for mi in [0.0, 0.3, 0.5, 0.7, 1.0, 1.4, 2.0]:
                for mj in [0.0, 0.3, 0.5, 0.7, 1.0, 1.4, 2.0]:
                    if mi == 1.0 and mj == 1.0:
                        continue
                    best[i] = oi * mi if oi > 1e-15 else 0.0
                    best[j] = oj * mj if oj > 1e-15 else 0.0
                    s = float(best.sum())
                    if s < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    b = np.convolve(best, best, mode="full")
                    m = float(b.max())
                    sc = 2.0 * n * m / (s * s)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        # Random perturbations
        if not improved:
            for _ in range(40):
                if time.monotonic() > stop_time:
                    return best, best_sc
                k = int(rng.integers(n))
                old = best[k]
                best[k] = max(0.0, old + rng.normal(0, 0.04 * max(0.01, best.mean())))
                s = float(best.sum())
                if s < 0.01:
                    best[k] = old
                    continue
                b = np.convolve(best, best, mode="full")
                m = float(b.max())
                sc = 2.0 * n * m / (s * s)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                else:
                    best[k] = old

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = _score_fast(best_seq)
    if not math.isfinite(best_score):
        best_seq = np.ones(200)
        best_score = _score_fast(best_seq)

    if not math.isfinite(best_score):
        best_seq = np.ones(100)
        best_score = 2.0

    n = n_p

    # Phase 1: Main LP optimization on parent
    phase1_end = min(t0 + budget * 0.45, deadline - 60.0)
    if time.monotonic() < phase1_end:
        try:
            res, sc = _lp_optimize(parent, phase1_end, seed)
            if sc < best_score:
                best_score = sc
                best_seq = res.copy()
        except Exception:
            pass

    # Phase 2: LP from best with different seed (refinement)
    phase2_end = min(t0 + budget * 0.60, deadline - 50.0)
    if time.monotonic() < phase2_end:
        try:
            res, sc = _lp_optimize(best_seq, phase2_end, seed + 7777)
            if sc < best_score:
                best_score = sc
                best_seq = res.copy()
        except Exception:
            pass

    # Phase 3: Try alternative initializations
    if time.monotonic() < deadline - 50.0:
        rng = np.random.default_rng(seed)
        i_arr = np.arange(n, dtype=np.float64)
        inits = []

        # Gaussian bump
        center = (n - 1) / 2.0
        inits.append(np.exp(-0.5 * ((i_arr - center) / (n * 0.18)) ** 2))

        # Raised cosine
        inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n)))

        # Best with noise
        inits.append(np.maximum(best_seq + rng.normal(0, 0.1 * max(0.01, best_seq.mean()), n), 0))

        # Multi-bump
        bump = np.zeros(n)
        for frac in [0.2, 0.5, 0.8]:
            bump += np.exp(-0.5 * ((i_arr - n * frac) / (n * 0.1)) ** 2)
        inits.append(bump)

        remaining = deadline - 50.0 - time.monotonic()
        if remaining > 10:
            time_per = max(5.0, remaining / len(inits))
            for idx, init in enumerate(inits):
                if time.monotonic() > deadline - 50.0:
                    break
                stop = min(deadline - 50.0, time.monotonic() + time_per)
                try:
                    res, sc = _lp_optimize(init, stop, seed + 50000 * (idx + 1))
                    if sc < best_score:
                        best_score = sc
                        best_seq = res.copy()
                except Exception:
                    pass

    # Phase 4: Subgradient polish
    sg_end = min(t0 + budget * 0.85, deadline - 30.0)
    if time.monotonic() < sg_end - 5.0:
        try:
            res, sc = _subgradient_polish(best_seq, sg_end, seed + 12345)
            if sc < best_score:
                best_score = sc
                best_seq = res.copy()
        except Exception:
            pass

    # Phase 5: Coordinate polish
    polish_end = deadline - 10.0
    if time.monotonic() < polish_end - 5.0:
        try:
            res, sc = _polish(best_seq, polish_end, seed)
            if sc < best_score:
                best_score = sc
                best_seq = res.copy()
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Fallback
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    return [float(x) for x in best_seq]
