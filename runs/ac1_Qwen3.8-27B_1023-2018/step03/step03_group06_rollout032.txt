import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize, linprog

class _Timeout(Exception):
    pass

_fft_cache = {}

def _fft_len(m):
    m = int(m)
    if m <= 0:
        return 1
    v = _fft_cache.get(m)
    if v is None:
        v = int(next_fast_len(m))
        _fft_cache[m] = v
    return v

def _conv_same(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    L = _fft_len(m)
    fa = rfft(a, L)
    c = irfft(fa * fa, L)[:m]
    return np.maximum(c, 0.0)

def _valid_corr(w, a):
    """r[j] = sum_{i=0}^{n-1} w[j+i] * a[i] for j=0..n-1"""
    n = a.size
    ar = a[::-1]
    m = w.size + n - 1
    L = _fft_len(m)
    c = irfft(rfft(w, L) * rfft(ar, L), L)
    return c[n-1:2*n-1]

def _score_internal(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-12:
        return math.inf
    a = a / s
    n = a.size
    c = _conv_same(a)
    return 2.0 * n * c.max()

def _exact_score(a):
    try:
        fn = globals().get('evaluate_sequence')
        if fn is None:
            return math.inf
        lst = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
        return float(fn(lst))
    except Exception:

def _softmax(y):
    y = y - y.max()
    ey = np.exp(y)
    s = ey.sum()
    if s < 1e-300:
        return np.full_like(y, 1.0/y.size)
    return ey / s

def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u - (cssv - 1.0) / np.arange(1, n + 1)
    rho = np.max(np.where(cond > 0)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0)

    try:

        p = np.clip(p, 0, 1000)
        s = p.sum()
        if s < 1e-12 or p.size == 0:
