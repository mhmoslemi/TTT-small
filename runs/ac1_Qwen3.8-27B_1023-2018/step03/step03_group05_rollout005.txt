```
def _project_simplex(v, target=1.0):
    v=np.asarray(v,float).ravel()
    n=v.size
    if n==0: return np.array([target])
    u=np.sort(v)[::-1]
    css=np.cumsum(u)
    idx=np.arange(1,n+1,float)
    cond=(css-target)/idx < u
    rho=np.nonzero(cond)[0]
    if rho.size==0: return np.full(n,target/n)
    theta=(css[rho[-1]]-target)/(rho[-1]+1)
    w=np.maximum(v-theta,0)
    s=w.sum()
    if s<=0: return np.full(n,target/n)
    return w/s*target
```

```
def _normalize(a, target=1.0):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.array([target], dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.full(arr.size, target / float(arr.size), dtype=np.float64)
    out = (target / s) * arr
    return np.where(np.isfinite(out), out, 0.0)
```

```
def _random_smooth(n, rng, base=None):
    n=int(n)
    if n<=0: return np.ones(1)
```

```python
def _get_parent():
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0 and np.all(np.isfinite(p)):

def _normalize(a, target=1.0):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.array([float(target)], dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.full(arr.size, float(target) / float(arr.size), dtype=np.float64)
    out = (float(target) / s) * arr
    return np.where(np.isfinite(out), out, 0.0)


def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.array([float(target)], dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    with np.errstate(divide="ignore", invalid="ignore"):
        cond = (css - target) / idx < u
    rho = np.nonzero(cond)[0]
    if rho.size == 0:
        return np.full(n, float(target) / float(n), dtype=np.float64)
    theta = (css[rho[-1]] - target) / float(rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.full(n, float(target) / float(n), dtype=np.float64)
    return (w / s) * float(target)


def _resample_cdf(a, m):
    a = _normalize(a)
    m = int(m)
    if m <= 0:
        return np.ones(1, dtype=np.float64)
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.full(m, 1.0 / m, dtype=np.float64)
    old_edge = np.linspace(0.0, 1.0, n + 1, dtype=np.float64)
    old_cdf = np.concatenate(([0.0], np.cumsum(a)))
    new_edge = np.linspace(0.0, 1.0, m + 1, dtype=np.float64)
    new_cdf = np.interp(new_edge, old_edge, old_cdf)
    b = np.diff(new_cdf)
    return _normalize(b)


def _fft_conv(a):
    arr = _normalize(a)
    n = arr.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    if n == 1:
        return np.array([float(arr[0] * arr[0])], dtype=np.float64)
    nfft = next_fast_len(2 * n - 1)
    fa = rfft(arr, nfft)
    c = irfft(fa * fa, nfft)[: 2 * n - 1]
    c = np.asarray(c, dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    np.maximum(c, 0.0, out=c)
    return c


def _corr_lag(a, w):
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    need_w = 2 * n - 1
    if w.size < need_w:
        w = np.pad(w, (0, need_w - w.size), mode="constant")
    elif w.size > need_w:
        w = w[:need_w]
    w = np.where(np.isfinite(w), w, 0.0)

    need = 3 * n - 2
    nfft = next_fast_len(need)
    fa = rfft(a, nfft)
    fw = rfft(w[::-1], nfft)
    cr = irfft(fa * fw, nfft)[:need]
    cr = np.asarray(cr, dtype=np.float64)
    idx = (2 * n - 2) - np.arange(n, dtype=np.int64)
    q = cr[idx]
    q = np.where(np.isfinite(q), q, 0.0)
    np.maximum(q, 0.0, out=q)
    return q


def _softmax_c(c, T=None, K=None):
    c = np.asarray(c, dtype=np.float64).ravel()
    m = c.size
    if m == 0:
        return np.zeros(0, dtype=np.float64)
    M = float(c.max())
    if not math.isfinite(M) or M <= 0.0:
        w = np.zeros(m, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    if T is None or T <= 0.0:
        if K is None:
            K = max(4, min(48, m // 25))
        K = int(max(1, min(K, m)))
        if K >= m:
            idx = np.arange(m)
        else:
            idx = np.argpartition(c, -K)[-K:]
        w = np.zeros(m, dtype=np.float64)
        w[idx] = 1.0 / float(K)
        return w
    z = (c - M) / T
    z = np.where(z > -50.0, z, -50.0)
    z = np.minimum(z, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if math.isfinite(s) and s > 0.0:
        return e / s
    idx = np.argsort(c)[-max(4, min(24, m)):]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0 / float(idx.size)
    return w


def _grad_from_weights(a, w):
    return 2.0 * _corr_lag(a, w)


def _score_fft(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    c = _fft_conv(a)
    M = float(c.max())
    if not math.isfinite(M):
        return float("inf")
    return float(2.0 * n * M / (s * s))


def _exact_score_np(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return float("inf")
    arr = np.where(np.isfinite(arr), arr, 0.0)
    np.maximum(arr, 0.0, out=arr)
    np.minimum(arr, 1000.0, out=arr)
    s = float(arr.sum())
    if s < 0.01:
        return float("inf")
    b = np.convolve(arr, arr)
    M = float(b.max())
    if not math.isfinite(M):
        return float("inf")
    return float(2.0 * arr.size * M / (s * s))


def _random_smooth(n, rng, base=None):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)

    if base is not None:
        b = np.asarray(base, dtype=np.float64).ravel()
        if b.size > 0 and np.all(np.isfinite(b)):
            if b.size == n:
                base = _normalize(b)
            else:
                base = _resample_cdf(_normalize(b), n)
        else:
            base = np.ones(n, dtype=np.float64) / float(n)
    else:
        base = np.ones(n, dtype=np.float64) / float(n)

    kind = int(rng.integers(0, 8))
    if kind == 0:
        alpha = float(rng.uniform(0.3, 3.0))
        beta = float(rng.uniform(0.3, 3.0))
        mix = float(rng.uniform(0.3, 0.7))
        a = mix * (x ** alpha) + (1.0 - mix) * ((1.0 - x) ** beta)
    elif kind == 1:
        c = float(rng.uniform(0.2, 0.8))
        w = float(rng.uniform(0.05, 0.30))
        amp = float(rng.uniform(0.5, 2.0))
        a = 0.25 + amp * np.exp(-((x - c) ** 2) / (2.0 * w * w))
    elif kind == 2:
        c = float(rng.uniform(0.4, 0.8))
        low = float(rng.uniform(0.1, 0.5))
        high = float(rng.uniform(1.0, 3.0))
        w = float(rng.uniform(0.02, 0.15))
        a = low + (high - low) * np.clip((x - c) / w + 0.5, 0.0, 1.0)
    elif kind == 3:
        a = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 5))
        for k in range(1, K + 1):
            amp1 = float(rng.uniform(0.0, 1.0 / k))
            ph1 = float(rng.uniform(0.0, 2.0 * np.pi))
            amp2 = float(rng.uniform(0.0, 1.0 / k))
            ph2 = float(rng.uniform(0.0, 2.0 * np.pi))
            a += amp1 * np.cos(2.0 * np.pi * k * x + ph1)
            a += amp2 * np.sin(2.0 * np.pi * k * x + ph2)
    elif kind == 4:
        a = base + 0.1 * rng.normal(size=n)
    elif kind == 5:
        alpha = float(rng.uniform(0.3, 3.0))
        beta = float(rng.uniform(0.3, 3.0))
        a = (x ** alpha) + 0.1 * ((1.0 - x) ** beta)
    elif kind == 6:
        beta = float(rng.uniform(0.3, 3.0))
        alpha = float(rng.uniform(0.3, 3.0))
        a = ((1.0 - x) ** beta) + 0.1 * (x ** alpha)
    else:
        a = base + rng.normal(0.0, 0.10, n)

    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-6)
    if rng.random() < 0.45:
        a = 0.55 * a + 0.45 * base
    return _normalize(a)


def _build_A(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        end = min(m, i + n)
        A[i:end, i] = a[: end - i]
    return A


def _lp_linearized(p, M, stage_deadline, lp_s=5.0):
    p = _normalize(p)
    n = p.size
    if n < 10 or n > 750:
        return None
    now = _now()
    rem = stage_deadline - now
    if rem < 4.0:
        return None

    A = _build_A(p)
    c = _fft_conv(p)
    m = c.size

    A_ub = np.empty((m, n + 1), dtype=np.float64)
    A_ub[:, :n] = 2.0 * A
    A_ub[:, n] = -1.0
    b_ub = c.copy()
    del A

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[-1] = 1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    bounds = [(0.0, None)] * n + [(-np.inf, np.inf)]
    tl = max(1.0, min(float(lp_s), 0.55 * rem))

    res = None
    try:
        res = linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"time_limit": tl, "presolve": True},
        )
    except Exception:
        try:
            res = linprog(
                c_obj,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
            )
        except Exception:
            res = None

    if res is not None and getattr(res, "x", None) is not None:
        x = np.asarray(res.x, dtype=np.float64).ravel()[:n]
        x = np.where(np.isfinite(x), x, 0.0)
        x = np.maximum(x, 0.0)
        s = float(x.sum())
        if s > 1e-9:
            return x / s
    return None


def _step_candidates(p, g):
    out = []
    if p.size == 0:
        return out
    g = np.asarray(g, dtype=np.float64).ravel()
    if g.size != p.size:
        return out
    gc = g - float(np.mean(g))
    mx = float(np.max(np.abs(gc))) if gc.size else 0.0
    if mx > 1e-14:
        gn = gc / mx
        for eta in (0.03, 0.08, 0.20, 0.50, 1.00):
            out.append(_project_simplex(p - eta * gn))
    sd = float(np.std(gc)) if gc.size else 0.0
    if sd > 1e-14:
        gd = gc / sd
        for eta in (0.08, 0.25, 0.80, 2.00):
            e = np.exp(np.clip(-eta * gd, -25.0, 25.0))
            out.append(_normalize(p * e))
    return out[:12]


def _choose_cands(p, cands, current_M, max_eval=32):
    best = p
    best_M = current_M
    cnt = 0
    for cand in cands:
        if cnt >= max_eval:
            break
        cand = _normalize(cand)
        if cand.size != p.size:
            continue
        if not np.all(np.isfinite(cand)):
            continue
        c = _fft_conv(cand)
        M = float(c.max())
        if math.isfinite(M) and M < best_M * (1.0 - 1e-12):
            best = cand
            best_M = M
        cnt += 1
    return best, best_M


def _local_polish(p, deadline, rng, use_lp=False):
    p = _normalize(p)
    n = p.size
    c = _fft_conv(p)
    M = float(c.max())
    best = p.copy()
    best_M = M
    if not math.isfinite(best_M):
        best = np.ones(max(1, n), dtype=np.float64) / float(max(1, n))
        best_M = float(_fft_conv(best).max())

    it = 0
    patience = 0
    last_lp = -1e9

    while _now() < deadline - 3.0:
        it += 1
        if M < best_M * (1.0 - 1e-12):
            best = p.copy()
            best_M = M
            patience = 0
        else:
            patience += 1

        if patience > 35:
            p = best.copy()
            c = _fft_conv(p)
            M = float(c.max())
            patience = 0
            continue

        cands = []
        for Tf in (0.03, 0.08):
            w = _softmax_c(c, Tf * M)
            g = _grad_from_weights(p, w)
            cands.extend(_step_candidates(p, g))
        w = _softmax_c(c, None)
        g = _grad_from_weights(p, w)
        cands.extend(_step_candidates(p, g))

        if patience > 12:
            cands.append(_project_simplex(p + rng.normal(0.0, 0.03, n)))
            cands.append(_project_simplex(0.85 * p + 0.15 * _random_smooth(n, rng, base=p)))

        if use_lp and n <= 750 and patience > 4 and (_now() - last_lp) > 25.0:
            last_lp = _now()
            rem = deadline - _now()
            if rem > 10.0:
                dirb = _lp_linearized(p, M, _now() + min(6.0, rem - 3.0), 5.0)
                if dirb is not None:
                    for tt in (0.10, 0.30, 0.60, 1.00):
                        cands.append((1.0 - tt) * p + tt * dirb)

        if cands:
            bp, bM = _choose_cands(p, cands, M)
            if bM < M * (1.0 - 1e-12):
                p = bp
                M = bM
                c = _fft_conv(p)

        if not np.all(np.isfinite(p)):
            p = best.copy()
            c = _fft_conv(p)
            M = float(c.max())

    return best


def _lbfgs_stage(p, stage_deadline, T=None, maxfun=400, bounds_lim=25.0):
    p = _normalize(p)
    n = p.size
    c0 = _fft_conv(p)
    M0 = float(c0.max())
    if not math.isfinite(M0) or M0 <= 0.0:
        return p
    if T is None or T <= 0.0:
        T = 0.05 * M0
    best = p.copy()
    best_M = M0

    with np.errstate(divide="ignore", invalid="ignore"):
        z0 = np.log(np.maximum(p, 1e-300))
    z0 = np.where(np.isfinite(z0), z0, 0.0)
    z0 = z0 - float(np.mean(z0))
    z0 = np.clip(z0, -bounds_lim, bounds_lim)

    def _time_cb(k):
        if _now() >= stage_deadline:
            raise _Timeout

    def _fun(z):
        nonlocal best, best_M
        if _now() >= stage_deadline:
            raise _Timeout
        z = np.asarray(z, dtype=np.float64).ravel()
        if z.size != n or not np.all(np.isfinite(z)):
            return (1e50, np.zeros(n, dtype=np.float64))
        z = z - float(np.mean(z))
        z = np.clip(z, -bounds_lim, bounds_lim)
        zm = z - float(np.max(z))
        e = np.exp(np.clip(zm, -50.0, 0.0))
        s = float(e.sum())
        if not math.isfinite(s) or s <= 0.0:
            return (1e50, np.zeros(n, dtype=np.float64))
        pn = e / s
        c = _fft_conv(pn)
        M = float(c.max())
        if not math.isfinite(M) or M <= 0.0:
            return (1e50, np.zeros(n, dtype=np.float64))
        if M < best_M * (1.0 - 1e-13):
            best = pn.copy()
            best_M = M

        z64 = (c - M) / T
        z64 = np.where(z64 > -50.0, z64, -50.0)
        z64 = np.minimum(z64, 0.0)
        ee = np.exp(z64)
        ss = float(ee.sum())
        if math.isfinite(ss) and ss > 0.0:
            val = M + T * math.log(ss)
            w = ee / ss
        else:
            val = M
            w = _softmax_c(c, None)

        q = _corr_lag(pn, w)
        gp = 2.0 * q
        gp = np.where(np.isfinite(gp), gp, 0.0)
        dot = float(np.dot(gp, pn))
        gz = pn * (gp - pn * dot)
        gz = np.where(np.isfinite(gz), gz, 0.0)
        ng = float(np.max(np.abs(gz))) if gz.size else 0.0
        if ng > 1e8:
            gz *= 1e8 / ng
        return (float(2.0 * n * val), gz)

    res = None
    try:
        bounds = [(-bounds_lim, bounds_lim)] * n
        res = minimize(
            _fun,
            z0,
            jac=True,
            method="L-BFGS-B",
            bounds=bounds,
            options={
                "maxiter": max(60, int(maxfun)),
                "maxfun": max(60, int(maxfun)),
                "ftol": 1e-15,
                "gtol": 1e-8,
                "maxls": 20,
            },
            callback=_time_cb,
        )
    except _Timeout:
        pass
    except Exception:
        pass

    zx = None
    if res is not None:
        zx = getattr(res, "x", None)
    if zx is not None:
        zx = np.asarray(zx, dtype=np.float64).ravel()
        if zx.size == n and np.all(np.isfinite(zx)):
            zx = np.clip(zx - float(np.mean(zx)), -bounds_lim, bounds_lim)
            zm = zx - float(np.max(zx))
            e = np.exp(np.clip(zm, -50.0, 0.0))
            s = float(e.sum())
            if math.isfinite(s) and s > 0.0:
                pn = e / s
                M = float(_fft_conv(pn).max())
                if math.isfinite(M) and M < best_M * (1.0 - 1e-12):
                    best = pn.copy()
                    best_M = M
    return best


def _lbfgs_refine(p, deadline, max_total=40.0):
    p = _normalize(p)
    M = float(_fft_conv(p).max())
    if not math.isfinite(M):
        return p
    factors = (0.08, 0.03, 0.012)
    fracs = (0.45, 0.35, 0.20)
    for fac, fr in zip(factors, fracs):
        if _now() >= deadline - 8.0:
            break
        stage_seconds = max_total * fr
        if stage_seconds < 4.0:
            break
        stage_deadline = min(deadline - 4.0, _now() + stage_seconds)
        T = fac * M
        sec = max(1.0, stage_deadline - _now())
        maxfun = int(max(80, min(2200, sec * 90)))
        p2 = _lbfgs_stage(p, stage_deadline, T=T, maxfun=maxfun)
        M2 = float(_fft_conv(p2).max())
        if math.isfinite(M2) and M2 < M * (1.0 - 1e-12):
            p = p2
            M = M2
    return p


def _lowdim_fourier_global(n, deadline, rng, base=None, K=5):
    n = int(n)
    if n < 64 or _now() + 12.0 >= deadline:
        return None
    if base is None:
        base = np.ones(n, dtype=np.float64) / float(n)
    else:
        base = _resample_cdf(_normalize(base), n)

    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    D = 1 + 2 * K
    B = np.empty((D, n), dtype=np.float64)
    B[0, :] = 1.0
    jj = 1
    for k in range(1, K + 1):
        w = 2.0 * np.pi * k
        B[jj, :] = np.cos(w * x)
        B[jj + 1, :] = np.sin(w * x)
        jj += 2

    M0 = float(_fft_conv(base).max())
    if not math.isfinite(M0) or M0 <= 0.0:
        return base
    T = max(1e-14, 0.10 * M0)

    best = base.copy()
    best_M = M0
    best_soft = 1e100
    best_theta = None

    try:
        lb = np.log(np.maximum(base, 1e-12))
        theta0, _, _, _ = np.linalg.lstsq(B.T, lb, rcond=None)
        theta0 = np.where(np.isfinite(theta0), theta0, 0.0)
        theta0 = np.clip(theta0, -3.0, 3.0)
    except Exception:
        theta0 = np.zeros(D, dtype=np.float64)
    best_theta = theta0.copy()

    def _p_from_theta(theta):
        theta = np.asarray(theta, dtype=np.float64).ravel()
        if theta.size != D or not np.all(np.isfinite(theta)):
            return None
        l = B.T @ theta
        l = np.where(np.isfinite(l), l, 0.0)
        l = l - float(np.max(l))
        e = np.exp(np.clip(l, -50.0, 50.0))
        s = float(e.sum())
        if not math.isfinite(s) or s <= 0.0:
            return None
        return e / s

    def _objective(theta):
        nonlocal best, best_M, best_soft, best_theta
        if _now() >= deadline:
            raise _Timeout
        pn = _p_from_theta(theta)
        if pn is None:
            return 1e50
        c = _fft_conv(pn)
        M = float(c.max())
        if not math.isfinite(M) or M <= 0.0:
            return 1e50
        if M < best_M * (1.0 - 1e-12):
            best = pn.copy()
            best_M = M
            best_theta = np.asarray(theta, dtype=np.float64).copy()
        z = (c - M) / T
        z = np.where(z > -50.0, z, -50.0)
        z = np.minimum(z, 0.0)
        ee = np.exp(z)
        ss = float(ee.sum())
        val = M + T * math.log(ss) if ss > 0.0 else M
        val = float(2.0 * n * val)
        if math.isfinite(val) and val < best_soft - 1e-12:
            best_soft = val
        return val

    bounds = [(-3.0, 3.0)] * D
    if differential_evolution is not None and _now() + 10.0 < deadline:
        def _de_cb(xk, convergence):
            if _now() >= deadline - 4.0:
                raise _Timeout

        rem = deadline - _now()
        maxiter = int(max(24, min(90, rem * 2.5)))
        try:
            de_res = differential_evolution(
                _objective,
                bounds=bounds,
                maxiter=maxiter,
                popsize=15,
                seed=int(rng.integers(1, 2**31)),
                polish=False,
                callback=_de_cb,
            )
        except Exception:
            de_res = None
        if de_res is not None and getattr(de_res, "x", None) is not None:
            try:
                _objective(de_res.x)
            except Exception:
                pass
    else:
        for _ in range(40):
            try:
                _objective(theta0 + rng.normal(0.0, 0.8, D))
            except Exception:
                break

    # A few direct random evaluations can help if DE was truncated.
    for _ in range(24):
        if _now() >= deadline - 3.0:
            break
        try:
            _objective(theta0 + rng.normal(0.0, 0.9, D))
        except Exception:
            break

    return best
```
