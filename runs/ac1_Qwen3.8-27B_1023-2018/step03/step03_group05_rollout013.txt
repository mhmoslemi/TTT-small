def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0 : sum(x) = target}."""
    n = v.size
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.full(n, target / n)
    return v * (target / s)

def _soft_max_grad(c, a, T, K=16):
    """Compute weighted subgradient of soft-max of c w.r.t. a."""
    n = a.size
    m = c.size
    M = c.max()
    # Weight top-K entries
    if m <= K:
        idx = np.arange(m)
    else:
        idx = np.argpartition(c, m - K)[m - K:]
    sub = c[idx]
    w = np.exp((sub - M) / max(T, 1e-15))
    ws = w.sum()
    if ws < 1e-30:
        # Only the argmax
        k0 = idx[np.argmax(sub)]
        g = np.zeros(n)
        for mm in range(n):
            if 0 <= k0 - mm < n:
                g[mm] = 2.0 * a[k0 - mm]
        return g
    
    w = w / ws
    # Weighted gradient
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        k0 = idx[ii]
        for mm in range(n):
            if 0 <= k0 - mm < n:
                g[mm] += 2.0 * w[ii] * a[k0 - mm]
    return g

        if patience > 30:

            if r < 0.4:

            elif r < 0.7:
                # Perturb the shape
                x = np.linspace(0, 1, n)
                shape = best_a + 0.1 * np.sin(2 * np.pi * rng.integers(1, 5) * x + rng.uniform(0, 2*np.pi))
                a = _project_simplex(np.maximum(shape, 0))
            else:

                x = np.linspace(0, 1, n)
                alpha = rng.uniform(0.3, 2.0)
                a = _project_simplex(x**alpha + rng.normal(0, 0.05, n))
            c = _conv_fft(a)
            M = c.max()
            patience = 0

        # LP direction (periodically)
        improved = False
        if use_lp and it % lp_every == 0 and n <= 512:
            b_dir = _lp_direction(a, M, deadline - 10.0, rng)
            if b_dir is not None:

                    M_new = c_new.max()
                    if M_new < best_M * (1.0 - 1e-12):

def _get_parent():

    if h is not None:
        try:
            p = np.asarray(h, dtype=np.float64).ravel()

    """Full convolution via FFT."""
    n = a.size
    nfft = next_fast_len(2 * n - 1)
    f = rfft(a, nfft)
    c = irfft(f * f, nfft)[:2 * n - 1]
    return np.maximum(c, 0.0)

    s = float(a.sum())

    c = _conv(a)
    return float(2.0 * a.size * c.max() / (s * s))

    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = a.size
    if n < 4 or M <= 0:
        return None
    rem = deadline - time.monotonic()
    if rem < 2.0:
        return None
    
    m = 2 * n - 1
    # Build A: A[k,j] = a[k-j] if 0<=k-j<n
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a

    c_obj += 1e-7 * rng.standard_normal(n)

            c_obj, A_ub=A, b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method='highs',
            options={'time_limit': min(rem * 0.5, 10.0), 'presolve': True}
        )
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception:
        pass
    return None

    m = c.size
    M = c.max()

    if m <= K:
        idx = np.arange(m)
    else:
        idx = np.argpartition(c, m - K)[m - K:]

    if T is None:
        T = max(M * 0.02, 1e-8)
    w = np.exp((c[idx] - M) / T)
    ws = w.sum()
    if ws < 1e-30:
        k0 = idx[np.argmax(c[idx])]
        g = np.zeros(n)
        for mm in range(n):
            jj = k0 - mm
            if 0 <= jj < n:
