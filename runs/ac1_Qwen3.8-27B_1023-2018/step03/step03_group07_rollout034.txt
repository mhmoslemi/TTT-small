import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(200, dtype=np.float64) / 200.0
    p = np.clip(p, 0.0, 1000.0)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(200, dtype=np.float64) / 200.0
    return p


def _fft_conv(a, nfft):
    """Compute linear convolution a*a via FFT, return full array."""
    fa = rfft(a, nfft)
    return irfft(fa * fa, nfft)[:2 * len(a) - 1]


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)


def _lp_optimize(a0, stop_time, max_iters=500000, seed=0):
    """
    LP best-response iteration to minimize max(a*a) subject to sum(a)=1, a>=0.
    Returns the best sequence found (normalized to sum 1).
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    s = float(a.sum())
    if s <= 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())

    rng = np.random.default_rng(seed)
    no_improve = 0
    nfft = next_fast_len(2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() > stop_time - 0.5:
            break

        # Compute current state
        c = _fft_conv(a, nfft)
        M = float(c.max())

        if M < best_M * (1.0 - 1e-13):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Perturb if stuck
        if no_improve > 40:
            scale = min(0.02 + 0.005 * (no_improve // 40), 0.25)
            a = a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            no_improve = 0
            continue

        # Build constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0
        # Shape: (2n-1) x n
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = a

        # LP: max sum(g) s.t. A@g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * (1.0 + 1e-9))
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'time_limit': 5.0,
                                   'dual_feasibility_tolerance': 1e-12,
                                   'primal_feasibility_tolerance': 1e-12})
        except Exception:
            continue

        if res.status != 0 or res.x is None:
            continue

        g = np.asarray(res.x, dtype=np.float64)
        gsum = float(g.sum())
        if gsum <= 1e-14 or not np.isfinite(g).all():
            continue

        # If LP says sum <= 1 + eps, we're at (or near) a fixed point
        if gsum < 1.0 + 1e-7:
            no_improve += 5
            continue

        g = g / gsum

        # Check if direction is meaningful
        if float(np.max(np.abs(g - a))) < 1e-10:
            no_improve += 2
            continue

        # Line search: minimize max(((1-t)*a + t*g) conv ((1-t)*a + t*g))
        best_t = 0.0
        best_val = M

        # Coarse scan
        for t in [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]:
            cand = (1.0 - t) * a + t * g
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-12:
                continue
            cand = cand / cs
            val = float(_fft_conv(cand, nfft).max())
            if val < best_val:
                best_val = val
                best_t = t

        # Bisection refinement
        if best_t > 0:
            lo = max(0.0, best_t - 0.12)
            hi = min(1.0, best_t + 0.12)
            for _ in range(25):
                if hi - lo < 1e-7:
                    break
                mid = (lo + hi) * 0.5
                cand = (1.0 - mid) * a + mid * g
                cand = np.maximum(cand, 0.0)
                cs = cand.sum()
                if cs < 1e-12:
                    hi = mid
                    continue
                cand = cand / cs
                val = float(_fft_conv(cand, nfft).max())
                if val < best_val:
                    best_val = val
                    best_t = mid
                    hi = mid
                else:
                    lo = mid

            if best_t > 0:
                a = (1.0 - best_t) * a + best_t * g
                a = np.maximum(a, 0.0)
                a = a / a.sum()

    return best_a


def _subgrad_polish(a0, stop_time, max_iters=50000, seed=0):
    """
    Subgradient descent on max(a*a) with simplex projection.
    Used as a polishing step after LP convergence.
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())

    rng = np.random.default_rng(seed)
    nfft = next_fast_len(2 * n - 1)
    step = 0.01

    for it in range(max_iters):
        if time.monotonic() > stop_time - 0.5:
            break

        c = _fft_conv(a, nfft)
        M = float(c.max())

        if M < best_M * (1.0 - 1e-13):
            best_M = M
            best_a = a.copy()

        # Active constraints (within tolerance of M)
        tol = 1e-10
        active = np.where(np.abs(c - M) <= tol)[0]
        if len(active) == 0:
            active = np.array([int(c.argmax())])

        # Uniform subgradient
        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                idx = k - j
                grad[j] += a[idx]
        grad /= float(len(active))

        # Projected step onto simplex {x: sum(x)=1, x>=0}
        # Use projection: project(a - step*grad) onto simplex
        v = a - step * grad
        # Water-filling projection
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        w = np.arange(1, n + 1)
        rho = np.where(u * w > (cssv - 1.0))[0]
        if len(rho) == 0:
            a_new = np.ones(n) / n
        else:
            rho = rho[-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(v - theta, 0.0)

        M_new = float(_fft_conv(a_new, nfft).max())
        if M_new < M * (1.0 - 1e-14):
            a = a_new
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.5
            if step < 1e-10:
                # Try perturbation
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                step = 0.005

    return best_a
