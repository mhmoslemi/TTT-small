import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    return p

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

    for j in range(n):
        A[j:j + n, j] = f
    return A

    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                    bounds=[(0,None)]*n, method='highs',
                    options={'presolve':True, 'time_limit':tl})
    except:
        return None

        g = np.asarray(r.x, dtype=np.float64)
        s = g.sum()
        if s > 1e-12:
            return g/s
    return None

                b = _conv_full(f)
                M = float(b.max())

def _conv(a):

def _buildA(f, n):

                r, s = _pair_polish(best_seq, p3b_end, seed=7777)
                r = np.clip(r, 0, 1000)
                if np.all(np.isfinite(r)) and r.sum() > 0.01:
                    ets = float(evaluate_sequence([float(x) for x in r]))
                    if math.isfinite(ets) and ets < best_score:
                        best_seq = r.copy()
                        best_score = ets
            except Exception:
                pass

        # Final short coordinate polish
        p3c_end = p3_end - 1.0
        if time.monotonic() < p3c_end - 1.0:
            try:
                r, s = _polish(best_seq, p3c_end, seed=12345)
                r = np.clip(r, 0, 1000)
                if np.all(np.isfinite(r)) and r.sum() > 0.01:
                    ets = float(evaluate_sequence([float(x) for x in r]))
                    if math.isfinite(ets) and ets < best_score:
                        best_seq = r.copy()
                        best_score = ets
            except Exception:
                pass

    # Final validation
    final = np.clip(best_seq, 0, 1000)
    try:
        fsc = float(evaluate_sequence([float(x) for x in final]))
        if not math.isfinite(fsc) or fsc > best_score:
            final = np.clip(parent, 0, 1000)
    except Exception:
        final = np.clip(parent, 0, 1000)

    return [float(x) for x in final]
