        m = 2 * n - 1
        A_s = A / M
        tie = rng.standard_normal(n)
        tie = (tie - tie.mean()) / (np.std(tie) + 1e-12)
        c_obj = -1.0 + 1e-6 * tie
        b_ub = np.ones(m, dtype=np.float64)
        rem = max(0.5, deadline - time.monotonic() - 1.0)
        tl = max(0.5, min(15.0, 0.3 * rem))
        res = linprog(c_obj, A_ub=A_s, b_ub=b_ub, bounds=(0.0, None),
                      method='highs', options={'time_limit': tl, 'presolve': True})
        if res is not None and getattr(res, 'success', False):
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if x.size == n and np.all(np.isfinite(x)):
                x = np.maximum(x, 0.0)
                s = float(x.sum())
                if s > 1e-12:
                    return x / s
    except Exception:
        pass
    return None


def _resample(a, m):
    a = _normalize(a)
    n = a.size
    m = int(m)
    if m <= 0 or n == 0:
        return np.ones(1, dtype=np.float64)
    if n == m:
        return a.copy()
    if m > n:
        idx = (np.arange(m, dtype=np.int64) * n) // m
        counts = np.bincount(idx, minlength=n).astype(np.float64)
        counts[counts <= 0] = 1.0
        b = a[idx] / counts[idx]
    else:
        b = np.zeros(m, dtype=np.float64)
        idx = (np.arange(n, dtype=np.int64) * m) // n
        np.add.at(b, idx, a)
        b = np.maximum(b, 0.0)
    return _normalize(b)


def _optimize_smooth(a_init, deadline, rng, T_start=0.1, T_end=0.005, use_lp=True):
    a = _normalize(a_init.copy())
    n = a.size
    T = T_start
    best_a = a.copy()
    best_M = float(_conv(a).max())
    best_score = _exact(a)
    c = _conv(a)

    n_steps = 300
    ratio = (T_end / T_start) ** (1.0 / n_steps)

    step = 0
    no_improve = 0
    max_no_improve = 40

    while time.monotonic() < deadline - 3.0:
        step += 1
        rem = deadline - time.monotonic()
        if rem < 2.0:
            break

        grad, w, M_now = _smooth_max_grad(a, T)
        gn = float(np.linalg.norm(grad))

        if gn < 1e-12:
            T = max(T_end, T * ratio)
            continue

        # LP direction periodically
        lp_used = False
        if use_lp and step % 5 == 0 and rem > 20.0 and n <= 640:
            dirb = _lp_direction(a, best_M, deadline - 5.0, rng)
            if dirb is not None and np.any(np.abs(dirb - a) > 1e-6):
                for alpha in (0.1, 0.2, 0.35, 0.5, 0.7, 1.0):
                    if time.monotonic() >= deadline - 5.0:
                        break
                    cand = _normalize(a + alpha * (dirb - a))
                    cc = _conv(cand)
                    MM = float(cc.max())
                    if MM < best_M * (1 - 1e-10):
                        a = cand
                        best_M = MM
                        best_a = a.copy()
                        sc = _exact(a)
                        if sc < best_score:
                            best_score = sc
                        c = cc
                        no_improve = 0
                        lp_used = True
                        break
                    elif no_improve > max_no_improve and alpha == 0.35:
                        a = cand
                        c = cc
                        no_improve = 0
                        lp_used = True
                        break

        if lp_used:
            T = max(T_end, T * ratio)
            continue

        # Gradient descent step
        if no_improve > max_no_improve:
            r = float(rng.random())
            if r < 0.5:
                a = _project_simplex(a + rng.normal(0.0, 0.02, n))
            else:
                x = np.linspace(0, 1, n, endpoint=False)
                alpha = float(rng.uniform(0.5, 2.0))
                perturb = a.copy()
                perturb *= (1.0 + 0.3 * np.exp(-((x - 0.5)**2) / (2*0.1**2)))
                a = _project_simplex(0.7 * a + 0.3 * perturb)
            c = _conv(a)
            best_M = float(c.max())
            no_improve = 0
            T = max(T_end, T * ratio)
            continue

        # Multiplicative gradient step with backtracking
        eta = 2.0 / (gn / n + 1e-10)
        improved = False
        for _bt in range(6):
            log_step = np.clip(-eta * grad / (T + 1e-12), -5.0, 5.0)
            cand = a * np.exp(log_step)
            cs = float(cand.sum())
            if cs > 1e-300:
                cand = cand / cs
            else:
                cand = _project_simplex(a - eta * grad)
            cc = _conv(cand)
            MM = float(cc.max())
            if MM < M_now * (1 - 1e-8) or step > 3:
                a = cand
                c = cc
                M_now = MM
                improved = True
                break
            eta *= 0.5

        if improved:
            if M_now < best_M * (1 - 1e-10):
                best_a = a.copy()
                best_M = M_now
                sc = _exact(a)
                if sc < best_score:
                    best_score = sc
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        T = max(T_end, T * ratio)

    # Final check
    sc = _exact(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _optimize_lp_iter(a_init, deadline, rng):
    a = _normalize(a_init.copy())
    n = a.size
    best_a = a.copy()
    best_M = float(_conv(a).max())
    best_score = _exact(a)
    c = _conv(a)
    it = 0

    while time.monotonic() < deadline - 5.0:
        it += 1
        M_now = float(c.max())

        dirb = _lp_direction(a, M_now, deadline - 3.0, rng)
        if dirb is None:
            if it > 3:
                break
            continue

        improved = False
        for alpha in (0.3, 0.5, 0.7, 0.9, 1.0):
            if time.monotonic() >= deadline - 4.0:
                break
            cand = _normalize(a + alpha * (dirb - a))
            cc = _conv(cand)
            MM = float(cc.max())
            if MM < M_now * (1 - 1e-10):
                a = cand
                c = cc
                M_now = MM
                if M_now < best_M * (1 - 1e-10):
                    best_a = a.copy()
                    best_M = M_now
                    sc = _exact(a)
                    if sc < best_score:
                        best_score = sc
                improved = True
                break

        if not improved and it > 2:
            a = _project_simplex(a + rng.normal(0.0, 0.015, n))
            c = _conv(a)

    return best_a, best_score


def _poly_init(n, alpha, rng):
    """Generate a polynomial-based initialization."""
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    kind = int(rng.integers(0, 8))
    if kind == 0:
        a = x ** alpha
    elif kind == 1:
        a = (1.0 - x) ** alpha
    elif kind == 2:
        a = np.sin(np.pi * x) ** alpha
    elif kind == 3:
        a = 1.0 + 0.8 * np.cos(2.0 * np.pi * x)
    elif kind == 4:
        a = np.exp(-((x - 0.5) ** 2) / (2 * 0.15**2))
    elif kind == 5:
        a = np.ones(n) + 0.5 * np.cos(2.0 * np.pi * x) + 0.2 * np.cos(4.0 * np.pi * x)
    elif kind == 6:
        a = np.maximum(0.0, 1.0 - 4.0 * (x - 0.5) ** 2)
    else:
        a = np.ones(n) * (1.0 + 0.3 * np.random.RandomState(123).sin(2*np.pi*x))
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-6)
    return _normalize(a)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(0.0, float(budget_s))
    except Exception:
        budget = 1300.0

    if budget <= 2.0:
        p = _get_parent()
        sc = _exact(p)
        if np.isfinite(sc):
            return [float(x) for x in p]
        return [1.0]

    headroom = max(8.0, min(30.0, 0.025 * budget))
    deadline = t0 + budget - headroom

    rng = np.random.default_rng(int(seed) if np.isfinite(float(seed)) else 42)

    parent = _get_parent()
    best_a = _normalize(parent.copy())
    best_score = _exact(best_a)

    if not np.isfinite(best_score):
        best_a = np.ones(256, dtype=np.float64) / 256.0
        best_score = _exact(best_a)

    # Define length candidates
    pn = int(best_a.size) if best_a.size > 0 else 512
    lengths = sorted(set([pn, 512, 384, 640]))
    lengths = [L for L in lengths if 64 <= L <= 1024]
    if not lengths:
        lengths = [512]

    # Allocate time across lengths
    time_per = max(15.0, (deadline - 30.0) / len(lengths))

    for L in lengths:
        if time.monotonic() >= deadline - 25.0:
            break
        rem_phase = min(time_per, deadline - time.monotonic() - 20.0)
        if rem_phase < 15.0:
            continue

        Lint = int(L)
        phase_end = time.monotonic() + rem_phase

        # Multiple initializations
        inits = []
        # From parent
        inits.append(_resample(best_a, Lint))
        # Parametric shapes
        for alpha in (0.8, 1.2, 1.5):
            inits.append(_poly_init(Lint, alpha, rng))
        # Random smooth
        x = np.linspace(0, 1, Lint, endpoint=False)
        base = _resample(best_a, Lint)
        inits.append(_project_simplex(base * (1.0 + 0.3 * np.sin(np.pi * x))))

        # Phase A: Smooth max gradient optimization
        subA = min(rem_phase * 0.55, phase_end - time.monotonic() - 10.0)
        if subA > 12.0:
            for init_a in inits[:4]:
                if time.monotonic() >= phase_end - 10.0:
                    break
                sub_deadline = min(time.monotonic() + subA / max(1, len(inits[:4])), phase_end - 8.0)
                sub_deadline = max(sub_deadline, time.monotonic() + 8.0)
                try:
                    loc_a, loc_sc = _optimize_smooth(
                        init_a, sub_deadline, rng,
                        T_start=0.12, T_end=0.004,
                        use_lp=(Lint <= 640 and subA > 30)
                    )
                    if loc_sc < best_score:
                        best_a = _normalize(loc_a)
                        best_score = float(loc_sc)
                except Exception:
                    pass

        # Phase B: LP-focused refinement on best
        remB = phase_end - time.monotonic() - 5.0
        if remB > 15.0:
            initB = _resample(best_a, Lint)
            try:
                loc_a, loc_sc = _optimize_lp_iter(
                    initB, time.monotonic() + remB, rng
                )
                if loc_sc < best_score:
                    best_a = _normalize(loc_a)
                    best_score = float(loc_sc)
            except Exception:
                pass

    # Final polish on the overall best
    final_rem = deadline - time.monotonic() - 5.0
    if final_rem > 20.0:
        try:
            Lbest = int(best_a.size)
            loc_a, loc_sc = _optimize_smooth(
                best_a, time.monotonic() + min(final_rem * 0.8, 200.0), rng,
                T_start=0.06, T_end=0.002,
                use_lp=(Lbest <= 640 and final_rem > 60)
            )
            if loc_sc < best_score:
                best_a = _normalize(loc_a)
                best_score = float(loc_sc)
        except Exception:
            pass

    # One more LP pass if time
    rem_final = deadline - time.monotonic() - 3.0
    if rem_final > 15.0:
        try:
            loc_a, loc_sc = _optimize_lp_iter(
                best_a, time.monotonic() + min(rem_final, 100.0), rng
            )
            if loc_sc < best_score:
                best_a = _normalize(loc_a)
                best_score = float(loc_sc)
        except Exception:
            pass

    # Final validation
    try:
        best_a = np.asarray(best_a, dtype=np.float64).ravel()
        best_a = np.clip(best_a, 0.0, 1000.0)
        best_a = _normalize(best_a)
        fs = _exact(best_a)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
    except Exception:
        pass

    # Sanity check
    try:
        out = np.asarray(best_a, dtype=np.float64).ravel()
        if out.size == 0 or not np.all(np.isfinite(out)):
            raise ValueError("invalid")
        out = np.clip(out, 0.0, 1000.0)
        out = _normalize(out)
        fs = _exact(out)
        if np.isfinite(fs):
            return [float(x) for x in out]
    except Exception:
        pass

    # Fallback
    fb = np.ones(256, dtype=np.float64) / 256.0
    fs = _exact(fb)
    if np.isfinite(fs):
        return [float(x) for x in fb]
    return [1.0]
