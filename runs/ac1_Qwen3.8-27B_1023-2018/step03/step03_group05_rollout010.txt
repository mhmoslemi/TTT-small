                    if sv > 1e-300 and np.isfinite(sv):
                        cand = v / sv
                        cand_c = _conv_full(cand)
                        cand_M = float(cand_c.max())
                        if cand_M < nM:
                            na, nc, nM = cand, cand_c, cand_M
                            break

        if na is None:
            patience += 1
            lr *= 0.92
            if patience > max_patience:
                # Perturbation
                r = rng.random()
                if r < 0.5:
                    a = _project_simplex(a + rng.normal(0, 0.003, n))
                else:
                    # Small LP jump
                    if n <= 512 and lp_count < 8:
                        lp_dir = _lp_direction(a, M, deadline - 5.0, rng)
                        if lp_dir is not None:
                            for alpha in [0.1, 0.25, 0.5]:
                                cand = _project_simplex(a + alpha * (lp_dir - a))
                                cand_c = _conv_full(cand)
                                cand_M = float(cand_c.max())
                                if cand_M < M * 1.01:
                                    a = cand
                                    c = cand_c
                                    M = cand_M
                                    break
                            lp_count += 1
                c = _conv_full(a)
                M = float(c.max())
                lr = 0.02
                patience = 0
                T_frac = max(T_frac * 1.5, 0.005)
            continue

        a = na
        c = nc
        M = nM
        lr = min(lr * 1.05, 0.15)
        patience = 0

        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

    return best_a, best_M


def _parametric_optimize(a_init, deadline, rng):
    n = a_init.size
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    D = 14
    FM = np.zeros((D, n), dtype=np.float64)
    FM[0] = 1.0
    FM[1] = x
    FM[2] = x * x
    for k in range(3, D):
        FM[k] = np.cos(2.0 * np.pi * (k - 2) * x)

    def _recon(th):
        log_a = FM @ th
        log_a -= np.max(log_a)
        a_raw = np.exp(np.clip(log_a, -25.0, 0.0))
        a_raw = np.maximum(a_raw, 1e-12)
        return _normalize(a_raw)

    def _obj(th):
        a_p = _recon(th)
        c_p = _conv_full(a_p)
        M_p = float(c_p.max())
        if M_p <= 0 or not np.isfinite(M_p):
            return 1e8
        T_p = M_p * 0.008
        w_p = _soft_weights(c_p, T_p)
        logsum = math.log(max(float(np.sum(np.exp((c_p - M_p) / T_p))), 1e-300))
        return 2.0 * n * (M_p + T_p * logsum)

    a_norm = _normalize(a_init)
    log_a = np.log(np.maximum(a_norm, 1e-300))
    try:
        th0, _, _, _ = np.linalg.lstsq(FM.T, log_a, rcond=None)
    except Exception:
        th0 = np.zeros(D, dtype=np.float64)
    if not np.all(np.isfinite(th0)):
        th0 = np.zeros(D, dtype=np.float64)

    best_a = _recon(th0)
    best_sc = _exact_score(best_a)

    maxit = max(200, int((deadline - time.monotonic()) * 1.5))
    maxit = min(maxit, 1500)
    try:
        res = minimize(_obj, th0, method="Nelder-Mead",
                       options={"maxiter": maxit, "xatol": 1e-7, "fatol": 1e-9})
        th1 = res.x if res.x is not None else th0
        if np.all(np.isfinite(th1)):
            a1 = _recon(th1)
            sc1 = _exact_score(a1)
            if np.isfinite(sc1) and sc1 < best_sc:
                best_a, best_sc = a1, sc1
    except Exception:
        pass

    return best_a, best_sc


def _dual_weight_refine(a, c, M, deadline, rng):
    """Use LP dual (shadow prices) to get better weights for gradient."""
    n = a.size
    m = 2 * n - 1
    if n > 512 or time.monotonic() >= deadline - 5.0:
        return None
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(n):
        A[k, :k + 1] = a[:k + 1][::-1]
    for k in range(n, m):
        L = m - k
        s = k - n + 1
        A[k, s:s + L] = a[n - 1:n - L - 1:-1]

    # Dual: minimize sum(lam) s.t. A.T @ lam >= 1, lam >= 0
    A_dual = -A.T
    b_dual = -np.ones(n, dtype=np.float64)
    tie = rng.standard_normal(m)
    c_dual = 1.0 + 1e-5 * (tie - tie.mean())
    rem = deadline - time.monotonic()
    tl = max(0.3, min(8.0, 0.5 * rem))
    try:
        res = linprog(c_dual, A_ub=A_dual, b_ub=b_dual, bounds=(0.0, None),
                      method="highs", options={"time_limit": tl})
        if res is not None and getattr(res, "success", False):
            lam = np.asarray(res.x, dtype=np.float64).ravel()
            lam = np.maximum(lam, 0.0)
            s = lam.sum()
            if s > 1e-12:
                return lam / s
    except Exception:
        pass
    return None


def _resample(a, m):
    a = _normalize(a)
    n = a.size
    m = int(m)
    if m <= 0:
        return np.ones(1, dtype=np.float64)
    if n == m:
        return a.copy()
    if m > n:
        idx = (np.arange(m, dtype=np.int64) * n) // m
        counts = np.bincount(idx, minlength=n).astype(np.float64)
        counts[counts <= 0] = 1.0
        b = a[idx] / counts[idx]
    else:
        b = np.zeros(m, dtype=np.float64)
        idx = (np.arange(n, dtype=np.int64) * m) // n
        np.add.at(b, idx, a)
    return _normalize(b)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(0.0, float(budget_s))
    except Exception:
        budget = 1300.0

    if budget <= 2.0:
        return [1.0]

    head = max(8.0, min(30.0, 0.025 * budget))
    deadline = t0 + budget - head
    if deadline <= t0 + 3.0:
        deadline = t0 + max(3.0, budget * 0.85)

    rng = np.random.default_rng(int(seed) if np.isfinite(float(seed)) else 42)

    parent = _get_parent()
    best_a = _normalize(parent)
    best_score = _exact_score(best_a)

    if not np.isfinite(best_score):
        best_a = np.ones(512, dtype=np.float64) / 512.0
        best_score = _exact_score(best_a)
        if not np.isfinite(best_score):
            return [1.0]

    n_parent = best_a.size

    # Phase 1: Parametric search for global structure (n=512)
    if deadline - time.monotonic() > 80.0:
        p1_time = min(60.0, (deadline - time.monotonic()) * 0.25)
        p1_deadline = time.monotonic() + p1_time
        try:
            a_p, sc_p = _parametric_optimize(best_a, p1_deadline, rng)
            if a_p is not None and np.isfinite(sc_p) and sc_p < best_score:
                best_a = _normalize(a_p)
                best_score = float(sc_p)
        except Exception:
            pass

    # Phase 2: Main gradient descent on n=512
    if deadline - time.monotonic() > 60.0:
        n_main = 512
        a_main = _resample(best_a, n_main)
        gdt = min(deadline - time.monotonic() - 40.0, (deadline - t0) * 0.45)
        gdt = max(gdt, 40.0)
        gd_deadline = time.monotonic() + gdt
        try:
            a_gd, M_gd = _gradient_descent(a_main, gd_deadline, rng)
            sc_gd = _exact_score(a_gd)
            if np.isfinite(sc_gd) and sc_gd < best_score:
                best_a = _normalize(a_gd)
                best_score = float(sc_gd)
        except Exception:
            pass

    # Phase 3: Try a second gradient descent with different init
    if deadline - time.monotonic() > 50.0:
        n2 = 512
        a_init2 = _resample(best_a, n2)
        # Perturb
        a_init2 = _project_simplex(a_init2 + rng.normal(0, 0.01, n2))
        gdt2 = min(deadline - time.monotonic() - 30.0, 80.0)
        if gdt2 > 20.0:
            gd_deadline2 = time.monotonic() + gdt2
            try:
                a_gd2, M_gd2 = _gradient_descent(a_init2, gd_deadline2, rng)
                sc_gd2 = _exact_score(a_gd2)
                if np.isfinite(sc_gd2) and sc_gd2 < best_score:
                    best_a = _normalize(a_gd2)
                    best_score = float(sc_gd2)
            except Exception:
                pass

    # Phase 4: Try different lengths
    for L in [448, 384, 256]:
        if deadline - time.monotonic() < 35.0:
            break
        a_L = _resample(best_a, L)
        t_L = min(deadline - time.monotonic() - 25.0, 50.0)
        if t_L < 20.0:
            continue
        try:
            a_Lg, MLg = _gradient_descent(a_L, time.monotonic() + t_L, rng)
            sc_L = _exact_score(a_Lg)
            if np.isfinite(sc_L) and sc_L < best_score:
                best_a = _normalize(a_Lg)
                best_score = float(sc_L)
        except Exception:
            pass

    # Phase 5: Final polish on best candidate
    if deadline - time.monotonic() > 30.0:
        nF = best_a.size
        tF = min(deadline - time.monotonic() - 10.0, 120.0)
        if tF > 20.0:
            try:
                aF, MF = _gradient_descent(best_a.copy(), time.monotonic() + tF, rng)
                scF = _exact_score(aF)
                if np.isfinite(scF) and scF < best_score:
                    best_a = _normalize(aF)
                    best_score = float(scF)
            except Exception:
                pass

    # Final validation
    try:
        best_a = np.asarray(best_a, dtype=np.float64).ravel()
        best_a = np.maximum(best_a, 0.0)
        best_a = _normalize(best_a)
        final_sc = _exact_score(best_a)
        if np.isfinite(final_sc) and final_sc < best_score:
            best_score = final_sc
    except Exception:
        final_sc = float("inf")

    if not np.isfinite(final_sc):
        fb = np.ones(512, dtype=np.float64) / 512.0
        fs = _exact_score(fb)
        if np.isfinite(fs):
            return [float(x) for x in fb]
        return [1.0]

    out = np.clip(best_a, 0.0, 1000.0)
    out = _normalize(out)
    if out.size == 0:
        return [1.0]
    return [float(x) for x in out]
