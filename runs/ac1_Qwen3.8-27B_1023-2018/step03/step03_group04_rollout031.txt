    n = len(a)
    M = float(_conv(a).max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M

    A = np.zeros((2 * n - 1, n))
    for j in range(n):

def _lp_direction(f, M):
    n = len(f)
    if M <= 1e-15:
        return None
    A = _build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True,
                               "dual_feasibility_tolerance": 1e-9,
                               "primal_feasibility_tolerance": 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s

def _optimize_lp(f0, stop_time, seed, max_iters=50000):

    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs

    best_f = f.copy()
    best_sc = _score_norm(f)

    t = 0.3
    t_min, t_max = 0.01, 0.8
    stall = 0
    max_stall = 120
    iteration = 0

    while time.monotonic() < stop_time and iteration < max_iters:
        conv = _conv(f)
        M = float(conv.max())
        sc = 2.0 * n * M

        g = _lp_direction(f, M)

        if g is None:
            strength = 0.05 + 0.05 * min(stall / max_stall, 1.0)
            f = np.clip(f + rng.normal(0, strength, n), 0, None)
            fs = f.sum()
            f = f / fs if fs > 1e-10 else np.ones(n) / n
            stall += 5
            if stall > max_stall:
                f = np.clip(best_f + rng.normal(0, 0.02, n), 0, None)
                fs = f.sum()
                f = f / fs if fs > 1e-10 else best_f.copy()
                stall = 0
            iteration += 1
            continue

        dist = float(np.linalg.norm(g - f))
        if dist < 1e-7:
            stall += 3
            if stall > max_stall:
                f = np.clip(best_f + rng.normal(0, 0.03, n), 0, None)
                fs = f.sum()
                f = f / fs if fs > 1e-10 else best_f.copy()
                stall = 0
            iteration += 1
            continue

        # Line search
        best_t = -1
        best_sc_ls = sc
        best_f_ls = None

        # Try a range of step sizes
        base_ts = [t * 0.25, t * 0.5, t * 0.7, t, t * 1.3, t * 1.6, t * 2.0, t * 3.0]
        ts_list = [max(t_min, min(t_max, tc)) for tc in base_ts]
        ts_list = sorted(set(round(x, 8) for x in ts_list))

        for tc in ts_list:
            f_new = (1.0 - tc) * f + tc * g
            f_new = np.clip(f_new, 0, None)
            fs_new = f_new.sum()
            if fs_new < 1e-10:
                continue
            f_new = f_new / fs_new
            sc_new = _score_norm(f_new)
            if sc_new < best_sc_ls - 1e-14:
                best_sc_ls = sc_new
                best_t = tc
                best_f_ls = f_new.copy()

        if best_f_ls is not None:
            f = best_f_ls
            if best_sc_ls < best_sc:
                best_f = f.copy()
                best_sc = best_sc_ls
                stall = 0
                t = min(t_max, t * 1.03)
            else:
                t = max(t_min, t * 0.6)
                stall += 1
        else:
            stall += 1
            t = max(t_min, t * 0.5)

        if stall >= max_stall:
            strength = 0.1 + 0.05 * min(stall // max_stall, 3)
            f = np.clip(f + rng.normal(0, strength, n), 0, None)
            fs = f.sum()
            f = f / fs if fs > 1e-10 else best_f.copy()
            t = 0.3
            stall = 0
            if rng.random() < 0.4:
                f = np.clip(best_f + rng.normal(0, 0.02, n), 0, None)
                fs = f.sum()
                f = f / fs if fs > 1e-10 else best_f.copy()

        iteration += 1

    return best_f, best_sc

def _polish(a, stop_time, seed=4242):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score_norm(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 10

    multipliers = [0.05, 0.1, 0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 4.0, 0.0]

    while time.monotonic() < stop_time:
        improved = False

        # 1-coordinate moves (all coordinates)
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in multipliers:
                if mlt == 0.0:
                    new_val = 0.0
                elif orig > 1e-14:
                    new_val = orig * mlt
                else:
                    new_val = 0.001 * mlt
                best[i] = new_val
                s = best.sum()
                if s < 0.01:
                    best[i] = orig
                    continue
                tmp = best / s
                sc = _score_norm(tmp)
                if sc < best_val_sc - 1e-13:
                    best_val = new_val
                    best_val_sc = sc
            if best_val != orig:
                best[i] = best_val
                if best_val_sc < best_sc:
                    s = best.sum()
                    best = best / s
                    best_sc = best_val_sc
                    improved = True
            else:
                best[i] = orig

        # 2-coordinate moves (random pairs)
        n_pairs = min(200, n * 2)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.5, 1.5, 0.0, 2.0, 0.2):
                for mj in (0.5, 1.5, 0.0, 2.0, 0.2):
                    best[i] = oi * mi if oi > 1e-14 and mi > 0 else (0.001 if mi > 0 else 0.0)
                    best[j] = oj * mj if oj > 1e-14 and mj > 0 else (0.001 if mj > 0 else 0.0)
                    s = best.sum()
                    if s < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    tmp = best / s
                    sc = _score_norm(tmp)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_norm(a):
    """Score with sum(a)=1: returns 2*n*max(a*a)."""
    n = len(a)
    M = float(_conv(a).max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, M):
    n = len(f)
    if M <= 1e-15:
        return None
    A = _build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={"presolve": True,
                               "dual_feasibility_tolerance": 1e-9,
                               "primal_feasibility_tolerance": 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s

    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 12

    multipliers = [0.05, 0.1, 0.2, 0.5, 1.5, 2.0, 4.0, 0.0]

    while time.monotonic() < stop_time:
        improved = False

        # 1-coordinate moves
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:

            best_val = orig
            best_val_sc = best_sc
            for mlt in multipliers:
                if mlt == 0.0:
                    new_val = 0.0
                elif orig > 1e-14:
                    new_val = orig * mlt
                else:
                    new_val = 0.0001
                best[i] = new_val
                s = best.sum()
                if s < 0.01:
                    best[i] = orig
                    continue
                tmp = best / s
                sc = _score_norm(tmp)
                if sc < best_val_sc - 1e-13:
                    best_val = new_val
                    best_val_sc = sc
            if best_val != orig:
                best[i] = best_val
                if best_val_sc < best_sc:
                    s = best.sum()
                    best = best / s
                    best_sc = best_val_sc
                    improved = True
            else:
                best[i] = orig

        # 2-coordinate moves
        n_pairs = 150
        for _ in range(n_pairs):
            if time.monotonic() > stop_time:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.5, 1.5, 0.0, 2.0):
                for mj in (0.5, 1.5, 0.0, 2.0):
                    best[i] = oi * mi if oi > 1e-14 and mi > 0 else (0.0001 if mi > 0 else 0.0)
                    best[j] = oj * mj if oj > 1e-14 and mj > 0 else (0.0001 if mj > 0 else 0.0)
                    s = best.sum()
                    if s < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    tmp = best / s
                    sc = _score_norm(tmp)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc
