    lo = max(0.0, best_alpha - 0.15)
    hi = min(1.0, best_alpha + 0.15)
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    f1 = float(np.convolve(np.maximum((1-a1)*a + a1*g, 0)/np.maximum((1-a1)*a + a1*g,0).sum(),
                           np.maximum((1-a1)*a + a1*g, 0)/np.maximum((1-a1)*a + a1*g,0).sum()).max())
    f2 = float(np.convolve(np.maximum((1-a2)*a + a2*g, 0)/np.maximum((1-a2)*a + a2*g,0).sum(),
                           np.maximum((1-a2)*a + a2*g, 0)/np.maximum((1-a2)*a + a2*g,0).sum()).max())
    for _ in range(n_refine):
        if hi - lo < 1e-6:
            break
        if f1 < f2:
            hi = a2; a2 = a1; f2 = f1
            a1 = hi - gr * (hi - lo)
            c1 = np.maximum((1-a1)*a + a1*g, 0.0)
            f1 = float(np.convolve(c1/c1.sum(), c1/c1.sum()).max()) if c1.sum()>1e-12 else float('inf')
        else:
            lo = a1; a1 = a2; f1 = f2
            a2 = lo + gr * (hi - lo)
            c2 = np.maximum((1-a2)*a + a2*g, 0.0)
            f2 = float(np.convolve(c2/c2.sum(), c2/c2.sum()).max()) if c2.sum()>1e-12 else float('inf')
        mid = (lo + hi) / 2
        cm = np.maximum((1-mid)*a + mid*g, 0.0)
        if cm.sum() > 1e-12:
            vm = float(np.convolve(cm/cm.sum(), cm/cm.sum()).max())
            if vm < best_val:
                best_val = vm
                best_alpha = mid
    cand = np.maximum((1 - best_alpha) * a + best_alpha * g, 0.0)
    if cand.sum() < 1e-12:
        return None, M
    cand /= cand.sum()
    return cand, best_val


def _optimize_lp(a0, stop_time, seed=0):
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())

    rng = np.random.default_rng(seed)
    stuck = 0
    perturb_count = 0

    while time.monotonic() < stop_time - 0.5:
        b = np.convolve(a, a)
        M = float(b.max())
        k_star = int(b.argmax())

        if M < best_M * (1.0 - 1e-13):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 120 and perturb_count < 40:
            scale = min(0.03 + 0.015 * perturb_count, 0.25)
            a = a * np.exp(rng.normal(0.0, scale, n))
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            stuck = 0
            perturb_count += 1
            continue
        if stuck > 300:
            break

        A = _build_A(a)
        found = False

        for mu in [1.0, 1.0 + 1e-7, 1.0 + 1e-5, 1.0 + 1e-3, 1.0 + 5e-3]:
            M_lp = M * mu
            g = _solvelp(A, M_lp, n)
            if g is None:
                continue
            if np.linalg.norm(g - a) < 1e-5:
                continue
            cand, val = _line_search(a, g, M)
            if cand is not None and val < M * (1.0 - 1e-12):
                a = cand
                found = True
                break

        if found:
            continue

        # Gradient step on active lag
        grad = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        for j in range(j_lo, j_hi + 1):
            grad[j] += 2.0 * a[k_star - j]
        # Also try the symmetric lag
        k_sym = 2 * (n - 1) - k_star
        if k_sym != k_star:
            j_lo2 = max(0, k_sym - n + 1)
            j_hi2 = min(k_sym, n - 1)
            for j in range(j_lo2, j_hi2 + 1):
                grad[j] += 2.0 * a[k_sym - j]
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn > 1e-10:
            grad /= gn
            for alpha in [0.08, 0.04, 0.02, 0.01, 0.005]:
                cand = np.maximum(a + alpha * grad, 0.0)
                cs = cand.sum()
                if cs < 1e-12:
                    continue
                cand /= cs
                v = float(np.convolve(cand, cand).max())
                if v < M * (1.0 - 1e-10):
                    a = cand
                    found = True
                    break

        if not found:
            a = a * np.exp(rng.normal(0.0, 0.02, n))
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            perturb_count += 1

    return best_a


def _qp_polish(a0, stop_time, tol=1e-10):
    """Polish by minimizing max of top-3 active lags via active-set QP."""
    from scipy.optimize import minimize

    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    a /= a.sum()
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())

    for iteration in range(50):
        if time.monotonic() > stop_time - 1.0:
            break

        b = np.convolve(a, a)
        M = float(b.max())
        if M < best_M:
            best_M = M
            best_a = a.copy()

        # Identify active lags (within 1% of max)
        active = np.where(b >= M * 0.999)[0]
        if len(active) == 0:
            active = np.array([int(b.argmax())])
        active = active[:5]  # Limit to top 5

        # Solve: min t s.t. (a*a)_{k_i} <= t, a >= 0, sum(a) = 1
        p = len(active)

        def objective(vars_):
            aa = vars_[:-1]
            t = vars_[-1]
            # Penalty for violations
            penalty = 0.0
            for k in active:
                j_lo = max(0, k - n + 1)
                j_hi = min(k, n - 1)
                val = sum(aa[j] * aa[k - j] for j in range(j_lo, j_hi + 1))
                violation = max(0.0, val - t)
                penalty += 1e6 * violation * violation
            return t + penalty

        def gradient(vars_):
            aa = vars_[:-1]
            t = vars_[-1]
            g = np.zeros(n + 1)
            for k in active:
                j_lo = max(0, k - n + 1)
                j_hi = min(k, n - 1)
                for j in range(j_lo, j_hi + 1):
                    g[j] += 2.0 * aa[k - j]
                    if k - j != j:
                        g[k - j] += 2.0 * aa[j]
                    g[-1] += -1.0
                    violation = max(0.0, sum(aa[j2] * aa[k - j2] for j2 in range(j_lo, j_hi + 1)) - t)
                    if violation > 0:
                        g[j] += 4e6 * violation * aa[k - j]
                        if k - j != j:
                            g[k - j] += 4e6 * violation * aa[j]
                        g[-1] -= 4e6 * violation
            return g

        x0 = np.append(a, M * 1.01)
        bounds = [(0.0, None)] * n + [(0.0, None)]
        constraints = [{'type': 'eq', 'fun': lambda v: v[:-1].sum() - 1.0,
                        'jac': lambda v: np.append(-np.ones(n), 0.0)}]

        res = minimize(objective, x0, jac=gradient, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 200, 'ftol': 1e-12})

        if res.success or res.fun < M:
            new_a = np.clip(res.x[:-1], 0, None)
            ns = new_a.sum()
            if ns > 1e-10:
                new_a /= ns
                new_M = float(np.convolve(new_a, new_a).max())
                if new_M < M * (1 - 1e-10):
                    a = new_a
                else:
                    break
            else:
                break
        else:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 25.0

    parent = _get_parent()
    n_parent = parent.size

    best_seq = parent.copy()
    best_sc = _score(best_seq)
    if not np.isfinite(best_sc):
        best_seq = np.ones(64, dtype=np.float64) / 64.0
        best_sc = 2.0

    def update(cand):
        nonlocal best_seq, best_sc
        try:
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_sc:
                best_seq = np.asarray(cand, dtype=np.float64).copy()
                best_sc = sc
                return True
        except Exception:
            pass
        return False

    # Phase 1: Primary LP from parent
    a0 = np.clip(parent, 0, 1000.0).astype(np.float64)
    s0 = a0.sum()
    if s0 > 1e-15:
        a0 /= s0
    else:
        a0 = np.ones(n_parent) / n_parent

    t1 = min(t0 + budget * 0.35, deadline - 300.0)
    result = _optimize_lp(a0, stop_time=t1, seed=1)
    update(result)

    # Phase 2: Continue from current best
    a1 = np.clip(best_seq, 0, 1000.0).astype(np.float64)
    s1 = a1.sum()
    if s1 > 1e-15:
        a1 /= s1
    t2 = min(t0 + budget * 0.55, deadline - 200.0)
    result = _optimize_lp(a1, stop_time=t2, seed=3)
    update(result)

    # Phase 3: QP polish
    if time.monotonic() < deadline - 100:
        t3 = min(time.monotonic() + 60, deadline - 80)
        result = _qp_polish(best_seq, stop_time=t3)
        update(result)

    # Phase 4: Alternative initializations
    x = np.arange(n_parent, dtype=np.float64)
    inits = []

    # Cosine harmonics
    for c in [1, 2, 3, 4, 5, 6]:
        s = 1.0 + 0.5 * np.cos(c * np.pi * x / (n_parent - 1))
        s = np.maximum(s, 1e-10)
        inits.append(s / s.sum())

    # Exponential decay
    for al in [0.5, 1.0, 2.0, 3.0, 5.0]:
        s = np.exp(-al * x / n_parent)
        inits.append(s / s.sum())
        s = np.exp(-al * (n_parent - 1 - x) / n_parent)
        inits.append(s / s.sum())

    # Centered Gaussian
    ctr = (n_parent - 1) / 2.0
    for w in [0.15, 0.25, 0.35, 0.5, 0.7]:
        s = np.exp(-0.5 * ((x - ctr) / (w * n_parent * 0.5)) ** 2)
        inits.append(s / s.sum())

    # V-shape (minimum at center)
    for w in [0.3, 0.5, 0.7, 0.9]:
        s = 1.0 + w * np.abs(x - ctr) / (n_parent * 0.5)
        inits.append(s / s.sum())

    # Inverted V (maximum at center)
    for w in [0.3, 0.5, 0.7, 0.9]:
        s = 1.0 - w * np.abs(x - ctr) / (n_parent * 0.5)
        s = np.maximum(s, 1e-10)
        inits.append(s / s.sum())

    # Power law
    for p in [0.5, 1.0, 2.0, 3.0]:
        s = (1.0 - x / (n_parent - 1 + 1e-10)) ** p + 1e-10
        inits.append(s / s.sum())

    # Blend with parent
    p_norm = np.clip(parent, 0, 1000.0).astype(np.float64)
    p_norm /= p_norm.sum()
    for i in range(0, min(8, len(inits)), 2):
        blend = 0.4 * p_norm + 0.6 * inits[i]
        inits.append(blend / blend.sum())

    t4 = min(t0 + budget * 0.78, deadline - 80.0)
    rem = t4 - time.monotonic()
    if rem > 15:
        per_init = max(5.0, rem / len(inits))
        for idx, init in enumerate(inits):
            if time.monotonic() > t4 - 5:
                break
            t_alloc = min(time.monotonic() + per_init, t4 - 2)
            try:
                result = _optimize_lp(init, stop_time=t_alloc, seed=200 + idx)
                update(result)
            except Exception:
                pass

    # Phase 5: Try different lengths
    t5 = min(t0 + budget * 0.90, deadline - 50.0)
    for dn in [-8, -4, -2, 2, 4, 8, 12, -12, 16, -16, 24, -24]:
        if time.monotonic() > t5 - 20:
            break
        nt = n_parent + dn
        if nt < 16 or nt > 512:
            continue
        try:
            xo = np.linspace(0, 1, n_parent)
            xn = np.linspace(0, 1, nt)
            init = np.interp(xn, xo, parent)
            init = np.clip(init, 0, 1000.0)
            init = init / init.sum()
            t_alloc = min(time.monotonic() + 25, t5 - 3)
            result = _optimize_lp(init, stop_time=t_alloc, seed=300 + abs(dn))
            update(result)
        except Exception:
            pass

    # Phase 6: Final intensive polish
    if time.monotonic() < deadline - 10:
        af = np.clip(best_seq, 0, 1000.0).astype(np.float64)
        sf = af.sum()
        if sf > 1e-15:
            af /= sf
        result = _optimize_lp(af, stop_time=deadline, seed=777)
        update(result)

        # Final QP polish
        if time.monotonic() < deadline - 3:
            result = _qp_polish(best_seq, stop_time=deadline)
            update(result)

    # Final validation
    try:
        final_score = float(evaluate_sequence([float(v) for v in best_seq]))
    except Exception:
        final_score = float('inf')

    if not np.isfinite(final_score):
        return [1.0] * 32

    return [float(v) for v in best_seq]
