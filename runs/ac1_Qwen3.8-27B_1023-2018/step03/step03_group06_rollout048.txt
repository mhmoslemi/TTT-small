Dropping rank: min tr(C_max * X) subject to tr(C_k * X) <= t, sum X[i,j] = 1, X >= 0.
This is an SDP with a 256x256 matrix - too large for our purposes.

So the LP approach (which is NOT a relaxation of the true problem, but a heuristic) seems like the best structured approach.

Let me also think about a simple but effective approach: **iterative water-filling**.

Algorithm:
1. Start with a normalized sequence.
2. Compute c = conv(a, a), M = max(c).
3. For the positions where c is closest to M (the "hot" positions), reduce the a values that contribute most to them.
4. Redistribute the removed mass to "cold" positions.
5. Repeat.

This is essentially a subgradient method on max(c).

For the gradient: grad[j] = 2 * a[k*-j] where k* is the argmax of c.
So we decrease a[j] where grad[j] is large (i.e., where a[k*-j] is large).

A simple multiplicative update: a[j] *= exp(-eta * grad[j]) = exp(-2*eta*a[k*-j])
Then renormalize.

This is like a "heat equation" on the autocorrelation - it diffuses mass away from the peaks.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize
from scipy.signal import correlate as _corr


class _Timeout(Exception):
    pass


_fft_cache = {}


def _fft_len(m):
    m = int(m)
    if m <= 0:
        return 1
    v = _fft_cache.get(m)
    if v is None:
        v = int(next_fast_len(m))
        _fft_cache[m] = v
    return v


def _conv_full(a, b=None):
    """Full linear convolution using FFT."""
    if b is None:
        b = a
    n1, n2 = len(a), len(b)
    m = n1 + n2 - 1
    L = _fft_len(m)
    fa = rfft(a, L)
    fb = rfft(b, L)
    c = irfft(fa * fb, L)[:m]
    return np.maximum(c, 0.0)


def _conv_auto(a):
    return _conv_full(a, a)


def _valid_corr(p, a):
    """Compute v[j] = sum_{m=0}^{n-1} p[j+m]*a[m] for j=0..n-1 where p has len 2n-1, a has len n."""
    n = len(a)
    return _corr(p, a, mode='valid')


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s < 1e-12:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s < 0.01:
        return math.inf
    n = len(a)
    c = _conv_auto(a)
    mb = float(c.max())
    return 2.0 * n * mb / (s * s)


def _exact_score(lst):
    try:
        return float(evaluate_sequence(lst))
    except Exception:
        return math.inf


def _softmax_params(y, n):
    """Convert log-parameters to probability vector (sum=1, all positive)."""
    y = np.asarray(y, dtype=np.float64)
    y = y - y.max()
    ex = np.exp(y)
    Z = ex.sum()
    if Z <= 0 or not math.isfinite(Z):
        return np.ones(n) / n
    return ex / Z


def _log_sum_exp_objective(y, beta, a0, n, deadline):
    """Objective and gradient for log-sum-exp approximation of max autocorrelation."""
    if time.monotonic() > deadline:
        raise _Timeout
    a = _softmax_params(y, n)
    c = _conv_auto(a)
    bc = beta * c
    bc_max = bc.max() - 30.0  # numerical stability
    ex = np.exp(np.clip(bc - bc_max, -700, 700))
    Z = ex.sum()
    if Z <= 0 or not math.isfinite(Z):
        return 1e10, np.zeros(n)
    p = ex / Z  # softmax over c
    fun = beta * (c @ p)  # = beta * weighted average, approximates max
    
    # Gradient wrt a: grad_a[j] = 2 * sum_m p[j+m] * a[m]
    grad_a = 2.0 * _valid_corr(p, a)
    
    # Gradient wrt y: grad_y[j] = a[j] * (grad_a[j] - sum_k a[k]*grad_a[k])
    dot = float(np.dot(a, grad_a))
    grad_y = a * (grad_a - dot)
    grad_y = np.where(np.isfinite(grad_y), grad_y, 0.0)
    return float(fun), grad_y


def _lbfgs_refine(a0, betas, deadline, maxfun_per=800):
    """L-BFGS refinement with log-sum-exp objective, using softmax parameterization."""
    a = _normalize(a0)
    n = len(a)
    if n < 2:
        return a, _score(a)
    
    # Exclude the max coordinate from reparameterization (fix it)
    b = int(np.argmax(a))
    idx = np.array([i for i in range(n) if i != b], dtype=int)
    base = max(float(a[b]), 1e-300)
    y0 = np.log(np.maximum(a[idx], 1e-300) / base)
    y0 = np.clip(y0, -25.0, 25.0)
    
    best = a.copy()
    best_sc = _score(a)
    
    def fg(y, beta=1.0):
        if time.monotonic() > deadline:
            raise _Timeout
        y = np.asarray(y, dtype=np.float64)
        if not np.all(np.isfinite(y)):
            return 1e10, np.zeros(len(y))
        # Reconstruct a
        a = np.zeros(n)
        a[b] = base
        ex = np.exp(np.clip(y, -25, 25))
        Z = 1.0 + ex.sum()
        a[idx] = ex / Z * base
        s = a.sum()
        a = a / s
        
        c = _conv_auto(a)
        bc = beta * c
        mval = bc.max()
        ex_c = np.exp(np.clip(bc - mval, -700, 700))
        Zc = ex_c.sum()
        if Zc <= 0:
            return 1e10, np.zeros(len(y))
        p = ex_c / Zc
        
        fun = beta * float(np.dot(c, p))
        
        grad_a = 2.0 * _valid_corr(p, a)
        dot = float(np.dot(a, grad_a))
        grad_full = a * (grad_a - dot)
        grad_y = grad_full[idx]
        grad_y = np.where(np.isfinite(grad_y), grad_y, 0.0)
        return float(fun), grad_y
    
    for beta in betas:
        if time.monotonic() > deadline - 2:
            break
        try:
            res = minimize(
                fg, y0, args=(beta,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': 200, 'maxfun': int(maxfun_per),
                         'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 30}
            )
        except _Timeout:
            break
        except Exception:
            break
        
        if res is not None and res.x is not None:
            y_new = np.asarray(res.x, dtype=np.float64)
            a_new = np.zeros(n)
            a_new[b] = base
            ex = np.exp(np.clip(y_new, -25, 25))
            Z = 1.0 + ex.sum()
            a_new[idx] = ex / Z * base
            a_new = _normalize(a_new)
            sc = _score(a_new)
            if math.isfinite(sc) and sc < best_sc:
                best = a_new.copy()
                best_sc = sc
                y0 = np.log(np.maximum(a_new[idx], 1e-300) / base)
    
    return best, best_sc


def _lp_update(f0, n, deadline, max_iters=5000, seed=0):
    """LP-based iterative update."""
    f = _normalize(f0)
    n = int(n)
    if n < 2:
        return f, _score(f)
    
    best = f.copy()
    best_sc = _score(f)
    rng = np.random.default_rng(seed)
    no_imp = 0
    
    # Precompute constraint matrix structure (but fill values each iteration)
    # A[k,j] = f[j] if 0 <= k-j < n
    # Actually: (f * b)[k] = sum_j f[k-j] * b[j] where 0<=k-j<n
    # So A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    
    obj = np.zeros(n)
    obj[:] = 1.0
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n
    
    min_rem = 20.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline - min_rem:
            break
        
        c_f = _conv_auto(f)
        M = float(c_f.max())
        if M <= 0:
            break
        
        # Build A_ub: (2n-1) x n, A_ub[k,j] = f[k-j] if 0<=k-j<n
        A_ub = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            # A_ub[k, j] = f[k-j] for k such that 0<=k-j<n => j <= k <= j+n-1
            k_lo = j
            k_hi = min(j + n - 1, 2 * n - 2)
            A_ub[k_lo:k_hi + 1, j] = f[n - 1 - (k_hi - j): n - 1 - (k_lo - j) + 1]
        
        # Actually let me redo this more carefully.
        # (f * b)[k] = sum_{j: 0<=j<n, 0<=k-j<n} f[k-j] * b[j]
        # So A_ub[k,j] = f[k-j] if 0 <= k-j < n, else 0
        
        A_ub = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            for k in range(max(0, k for k in [j]), min(2*n-1, j+n)):
                q = k - j
                if 0 <= q < n:
                    A_ub[k, j] = f[q]
        
        # That's O(n^2) which is fine for n=256
        # Faster construction:
        A_ub = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            k_start = j
            k_end = j + n
            k_end = min(k_end, 2 * n - 1)
            # A_ub[k, j] = f[k - j] for k = j, ..., k_end-1
            A_ub[k_start:k_end, j] = f[:k_end - k_start]
        
        b_ub = np.full(2 * n - 1, M)
        
        try:
            res = linprog(obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
        except Exception:
            break
        
        if res is None or res.status != 0 or res.x is None:
            break
        
        g = np.asarray(res.x, dtype=np.float64)
        g = np.clip(g, 0, None)
        g = _normalize(g)
        
        # Find best t in (0,1)
        cf = _conv_auto(f)
        cfs = _conv_full(f, g)
        cg = _conv_auto(g)
        
        # Score at t: 2n * max((1-t)^2*cf + 2t(1-t)*cfs + t^2*cg) / (sum = 1, since both sum to 1)
        # Wait, after normalization sum is 1, so the mix also has sum 1.
        # max_b(t) = max_k [(1-t)^2 cf[k] + 2t(1-t) cfs[k] + t^2 cg[k]]
        
        base_c = cf.copy()
        cross_c = 2.0 * cfs
        g_c = cg - 2.0 * cfs + cg  # coefficient of t^2: cg, coefficient of t: 2*cfs - 2*cf
        
        # Actually: mix = (1-t)f + t*g, normalized. Since sum(f)=sum(g)=1, sum(mix)=1.
        # conv(mix, mix) = (1-t)^2 conv(f,f) + 2t(1-t) conv(f,g) + t^2 conv(g,g)
        quad_coef = cg - 2.0 * cf
        lin_coef = 2.0 * (cfs - cf)
        
        ts = np.linspace(0, 1, 41)
        vals = np.max(base_c[:, None] + lin_coef[:, None] * ts[None, :] + quad_coef[:, None] * ts[None, :]**2, axis=0)
        t0 = float(ts[np.argmin(vals)])
        
        # Refine
        lo, hi = max(0, t0 - 0.05), min(1, t0 + 0.05)
        ts2 = np.linspace(lo, hi, 201)
        vals2 = np.max(base_c[:, None] + lin_coef[:, None] * ts2[None, :] + quad_coef[:, None] * ts2[None, :]**2, axis=0)
        t0 = float(ts2[np.argmin(vals2)])
        
        t0 = max(0.0, min(1.0, t0))
        
        # Try a few candidate t values
        candidates = {0.0, 1.0, t0, 0.5 * t0, min(1.0, 1.5 * t0), 0.1, 0.3, 0.7}
        candidates = {max(0.0, min(1.0, c)) for c in candidates}
        
        acc = f.copy()
        acc_sc = best_sc
        for tt in sorted(candidates):
            new = (1 - tt) * f + tt * g
            new = np.maximum(new, 0)
            s_new = new.sum()
            if s_new < 1e-12:
                continue
            new = new / s_new
            sc = _score(new)
            if math.isfinite(sc) and sc < acc_sc:
                acc_sc = sc
                acc = new.copy()
        
        if acc_sc < best_sc - 1e-13:
            best = acc.copy()
            best_sc = acc_sc
            f = acc
            no_imp = 0
        else:
            no_imp += 1
            # Add noise to escape
            if no_imp > 5:
                scale = 0.02 * max(float(f.mean()), 1e-6) * (1 + 0.1 * no_imp)
                f = f + rng.normal(0, scale, n)
                f = _normalize(f)
                sc_f = _score(f)
                if sc_f < best_sc:
                    best = f.copy()
                    best_sc = sc_f
                    no_imp = 0
            if no_imp > 20:
                break
    
    return best, best_sc


def _coord_descent(a0, deadline, n_passes=2):
    """Coordinate descent: try adjusting each coordinate."""
    a = _normalize(a0)
    n = len(a)
    best = a.copy()
    best_sc = _score(a)
    
    multis = [0.0, 0.1, 0.3, 0.5, 0.7, 1.0, 1.3, 1.6, 2.0, 3.0]
    
    for p in range(n_passes):
        if time.monotonic() > deadline - 1:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline - 1:
                break
            orig = best[j]
            base = orig if orig > 1e-10 else max(1.0 / (4 * n), 1e-6)
            best_j = orig
            best_j_sc = best_sc
            
            for m in multis:
                val = base * m
                if val < 0:
                    continue
                best[j] = val
                s = best.sum()
                if s < 1e-12:
                    best[j] = best_j
                    continue
                best[j] = val
                sc = _score(best)
                if sc < best_j_sc:
                    best_j_sc = sc
                    best_j = val
            
            best[j] = best_j
            if best_j_sc < best_sc:
                best_sc = best_j_sc
                improved = True
        
        if not improved:
            break
    
    return best, best_sc


def _interpolate(src, L):
    """Interpolate a sequence to length L."""
    if len(src) == L:
        return _normalize(src)
    x0 = np.linspace(0, 1, len(src))
    x1 = np.linspace(0, 1, L)
    v = np.interp(x1, x0, src)
    return _normalize(np.maximum(v, 0))


def _get_parent():
    try:
        h = globals()['height_sequence_1']

    if h is None:

            lst = [float(x) for x in seq]
        return lst
    
    n_base = len(best_seq)
    
    # Phase 1: Quick L-BFGS at base length (moderate beta)
    p1_deadline = min(deadline - 300, t_start + 0.15 * budget)
    if time.monotonic() < p1_deadline - 5:
        arr, _ = _lbfgs_refine(best_seq, [30, 80, 200], p1_deadline, maxfun_per=500)
        try_candidate(arr)
    
    # Phase 2: LP polishing at base length
    p2_deadline = min(deadline - 200, t_start + 0.45 * budget)
    if time.monotonic() < p2_deadline - 5:
        arr, _ = _lp_update(best_seq, n_base, p2_deadline, max_iters=8000, seed=int(rng.integers(1, 10**6)))
        try_candidate(arr)
    
    # Phase 3: Aggressive L-BFGS at base length (high beta)
    p3_deadline = min(deadline - 150, t_start + 0.65 * budget)
    if time.monotonic() < p3_deadline - 5:
        arr, _ = _lbfgs_refine(best_seq, [300, 600, 1000], p3_deadline, maxfun_per=700)
        try_candidate(arr)
    
    # Phase 4: More LP at base length
    p4_deadline = min(deadline - 100, t_start + 0.80 * budget)
    if time.monotonic() < p4_deadline - 5 and best_score > 1.49:
        arr, _ = _lp_update(best_seq, n_base, p4_deadline, max_iters=6000, seed=int(rng.integers(1, 10**6)))
        try_candidate(arr)
    
    # Phase 5: Try other lengths
    other_stop = min(deadline - 50, t_start + 0.92 * budget)
    other_lengths = [128, 192, 320, 384, 512]
    current_n = len(best_seq)
    other_lengths = [L for L in other_lengths if L != current_n]
    
    for L in other_lengths[:3]:
        if time.monotonic() > other_stop - 10:
            break
        if best_score < 1.49:
            break
        alloc = max(20.0, (other_stop - time.monotonic()))
        stop_L = min(other_stop - 5, time.monotonic() + alloc * 0.6)
        
        arr = _interpolate(best_seq, L)
        # Quick L-BFGS
        sub_stop = min(stop_L - 3, time.monotonic() + max(10, alloc * 0.3))
        if time.monotonic() < sub_stop - 3:
            arr, _ = _lbfgs_refine(arr, [50, 150], sub_stop, maxfun_per=350)
            try_candidate(arr)
        
        # Quick LP
        if L <= 384 and time.monotonic() < stop_L - 3:
            arr, _ = _lp_update(arr, L, stop_L - 2, max_iters=2000, seed=7)
            try_candidate(arr)
    
    # Phase 6: Final refinement at best length
    n_best = len(best_seq)
    fin_deadline = deadline - 10
    if time.monotonic() < fin_deadline - 10:
        arr, _ = _lbfgs_refine(best_seq, [500, 1200], fin_deadline - 5, maxfun_per=500)
        try_candidate(arr)
    
    # Phase 7: Coordinate descent polish
    if time.monotonic() < deadline - 3:
        n_best = len(best_seq)
        if n_best <= 512:
            arr, _ = _coord_descent(best_seq, deadline - 2, n_passes=2)
            try_candidate(arr)
    
    return finalize()
