import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize
from scipy.linalg import toeplitz

TARGET = 1.50274349
EARLY_MARGIN = 5e-4


def _default_init(n=384):
    i = np.arange(n, dtype=np.float64)
    a = (
        1.0
        + 0.45 * np.cos(2.0 * np.pi * i / n)
        + 0.25 * np.cos(4.0 * np.pi * i / n + 0.37)
        + 0.12 * np.cos(9.0 * np.pi * i / n + 1.13)
    )
    a = np.maximum(a, 1e-4)
    a = a / a.sum()
    return a


def _get_parent():
    try:
        h = globals().get("height_sequence_1")
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size > 0 and p.sum() > 1e-12:
                return p
    except Exception:
        pass
    return _default_init(384)


def _normalize(a):
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return None
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        return None
    return a / s


def _self_conv(a):
    n = int(a.size)
    if n < 32:
        c = np.convolve(a, a)
    else:
        f = next_fast_len(2 * n - 1)
        fa = rfft(a, f)
        c = irfft(fa * fa, f)
        c = c[: 2 * n - 1]
    if c.size:
        c = np.maximum(c, 0.0)
    return c


def _fast_score(a):
    n = int(a.size)
    s = float(a.sum())
    if s <= 1e-15:
        return math.inf
    b = _self_conv(a)
    if b.size == 0:
        return math.inf
    m = float(b.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _exact_score(arr):
    try:
        fn = globals().get("evaluate_sequence")
        if fn is None:
            return _fast_score(np.asarray(arr, dtype=np.float64))
        return float(fn([float(x) for x in arr]))
    except Exception:
        return math.inf


def _cross_weight(w, a):
    n = int(a.size)
    L = 2 * n - 1
    if n <= 512:
        c = np.convolve(w[::-1], a, mode="full")
        return c[L - n : L][::-1]
    f = next_fast_len(L + n - 1)
    fw = rfft(w[::-1], f)
    fa = rfft(a, f)
    c = irfft(fw * fa, f)[: L + n - 1]
    return c[L - n : L][::-1]


def _smooth_score_grad(a, beta):
    n = int(a.size)
    b = _self_conv(a)
    L = b.size
    mx = float(b.max())
    if not math.isfinite(mx) or mx <= 0.0:
        return 0.0, np.zeros_like(a)

    x = beta * (b - mx)
    ex = np.exp(np.clip(x, -745.0, 0.0))
    s = float(ex.sum())
    if s <= 1e-300:
        ex = np.zeros(L, dtype=np.float64)
        idx = np.where(b >= mx - 1e-15)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(b))])
        ex[idx] = 1.0
        s = float(ex.size)
    w = ex / s
    val = (2.0 * n / beta) * (beta * mx + math.log(s))
    cross = _cross_weight(w, a)
    grad = 4.0 * n * cross
    if not math.isfinite(val):
        val = 1e100
    if not np.all(np.isfinite(grad)):
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(val), grad


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    zm = float(z.max())
    ex = np.exp(z - zm)
    s = float(ex.sum())
    if not math.isfinite(s) or s <= 1e-300:
        ex = np.ones_like(z)
        s = float(ex.size)
    a = ex / s
    return a


def _z_from_a(a):
    a = _normalize(a)
    if a is None:
        a = _default_init(198)
    z = np.log(np.maximum(a, 1e-12))
    z = z - float(z.mean())
    z = np.clip(z, -50.0, 50.0)
    return z


def _lbfgs_smooth(a0, stop_time, stages=None, stop_flag=None, on_improve=None):
    if stages is None:
        stages = (
            (10.0, 80),
            (40.0, 70),
            (160.0, 60),
            (700.0, 45),
            (3000.0, 35),
            (12000.0, 25),
            (50000.0, 18),
        )

    a = _normalize(a0)
    if a is None:
        return None

    best_a = a.copy()
    best_sc = _fast_score(a)
    if not math.isfinite(best_sc):
        return best_a

    bounds = [(-50.0, 50.0)] * int(a.size)

    for beta, maxit in stages:
        if time.monotonic() > stop_time - 4.0:
            break
        if stop_flag is not None and stop_flag.get("v"):
            break

        z = _z_from_a(best_a)
        flag_stop = {"v": False}

        def fun(z, _beta=beta):
            if stop_flag is not None and stop_flag.get("v"):
                return (1e100, np.zeros_like(z))
            aa = _softmax(z)
            val, ga = _smooth_score_grad(aa, _beta)
            gz = aa * ga - float(np.dot(aa, ga)) * aa
            # Center z softly to remove the softmax flat direction.
            val += 1e-12 * float(np.dot(z, z))
            gz = gz + 2e-12 * z
            if not math.isfinite(val):
                return (1e100, np.zeros_like(z))
            if not np.all(np.isfinite(gz)):
                gz = np.nan_to_num(gz, nan=0.0, posinf=0.0, neginf=0.0)
            return (float(val), gz)

        def cb(i, xk, f):
            if (stop_flag is not None and stop_flag.get("v")) or (
                time.monotonic() > stop_time - 2.0
            ):
                flag_stop["v"] = True
            return flag_stop["v"]

        try:
            res = minimize(
                fun,
                z,
                method="L-BFGS-B",
                jac=True,
                bounds=bounds,
                callback=cb,
                options={
                    "maxiter": int(max(5, maxit)),
                    "ftol": 1e-15,
                    "gtol": 1e-12,
                    "maxfun": int(max(50, 3 * maxit)),
                },
            )
        except Exception:
            res = None

        if res is not None and getattr(res, "x", None) is not None:
            try:
                new_a = _softmax(np.asarray(res.x, dtype=np.float64))
            except Exception:
                new_a = None
            if new_a is not None:
                sc = _fast_score(new_a)
                if math.isfinite(sc) and sc < best_sc - 1e-15:
                    best_sc = float(sc)
                    best_a = new_a.copy()
                    if on_improve is not None:
                        on_improve(best_a)
        if flag_stop["v"]:
            break

    return best_a


def _lp_tangent(a, K=None, cap=None, time_limit=1.2):
    n = int(a.size)
    L = 2 * n - 1
    b = _self_conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return None, None, None

    if K is None or K >= L:
        idx = np.arange(L)
    else:
        K = int(max(1, min(K, L)))
        if K < L:
            idx = np.argpartition(-b, K)[:K]
            idx = idx[np.argsort(-b[idx])]
        else:
            idx = np.arange(L)

    first_col = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
    first_row = np.zeros(n, dtype=np.float64)
    first_row[0] = a[0]
    A = toeplitz(first_col, first_row)
    A_sub = 2.0 * A[idx]
    kk = int(A_sub.shape[0])

    A_ub = np.empty((kk, n + 1), dtype=np.float64)
    A_ub[:, :n] = A_sub
    A_ub[:, n] = -1.0
    b_ub = b[idx].astype(np.float64, copy=True)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0

    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    lower = np.zeros(n, dtype=np.float64)
    upper = np.ones(n, dtype=np.float64)
    if cap is not None and float(cap) > 0.0:
        capf = float(cap)
        upper = np.minimum(1.0, a + capf)
        if float(upper.sum()) < 1.0 - 1e-12:
            upper = np.ones(n, dtype=np.float64)

    bounds = [(lower[i], upper[i]) for i in range(n)] + [(0.0, 10.0)]
    opts = {"presolve": True, "time_limit": max(0.1, float(time_limit))}

    res = None
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=[1.0],
            bounds=bounds,
            method="highs",
            options=opts,
        )
    except Exception:
        try:
            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=[1.0],
                bounds=bounds,
                method="highs",
            )
        except Exception:
            return None, None, None

    if res is None or res.status not in (0, 3) or getattr(res, "x", None) is None:
        return None, None, None

    y = np.asarray(res.x[:n], dtype=np.float64)
    sy = float(y.sum())
    if sy <= 1e-12:
        return None, None, None
    y = y / sy
    pred = float(res.x[-1])
    if not math.isfinite(pred):
        pred = m
    return y, pred, m


def _line_search(a, y, current_score, stop_time=None):
    if not math.isfinite(current_score):
        return None, None
    d = y - a
    best_th = 0.0
    best_sc = float(current_score)

    thetas = (
        1.0,
        0.85,
        0.65,
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.02,
        0.01,
        0.005,
        0.002,
        0.001,
    )

    def score_at(th):
        x = a + th * d
        x = np.clip(x, 0.0, None)
        sx = float(x.sum())
        if sx <= 1e-15:
            return math.inf
        return _fast_score(x / sx)

    for th in thetas:
        if stop_time is not None and time.monotonic() > stop_time:
            break
        if th <= 0.0:
            continue
        sc = score_at(th)
        if math.isfinite(sc) and sc < best_sc - 1e-14:
            best_sc = sc
            best_th = th

    if best_th > 0.0:
        if best_th < 0.05:
            cand = (0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.04, 0.07, 0.1)
        elif best_th > 0.9:
            cand = (0.6, 0.8, 0.9, 0.95, 0.98, 1.0)
        else:
            lo = max(0.0, best_th * 0.55)
            hi = min(1.0, best_th * 1.45)
            cand = tuple(float(t) for t in np.linspace(lo, hi, 12))
        seen = set()
        for th in cand:
            th = float(th)
            if th <= 0.0 or th > 1.0 or abs(th) in seen:
                continue
            seen.add(abs(th))
            sc = score_at(th)
            if math.isfinite(sc) and sc < best_sc - 1e-14:
                best_sc = sc
                best_th = th

    if best_th > 0.0 and best_sc < float(current_score) - 1e-14:
        x = a + best_th * d
        x = np.clip(x, 0.0, None)
        sx = float(x.sum())
        if sx > 1e-15:
            return x / sx, float(best_sc)
    return None, None


def _smooth_noise(a, rng, scale):
    n = int(a.size)
    if n < 3:
        return rng.normal(0.0, scale, n)
    base = rng.normal(0.0, scale, n)
    k = np.exp(-0.5 * (np.arange(-5, 6) / 1.7) ** 2)
    k = k / k.sum()
    return np.convolve(base, k, mode="same")


def _lp_drive(a0, stop_time, seed=123, K=None, cap=None, time_limit=1.2,
              stop_flag=None, on_improve=None, max_iter=5000):
    a = _normalize(a0)
    if a is None:
        return None

    cur_sc = _fast_score(a)
    if not math.isfinite(cur_sc):
        return a

    best_a = a.copy()
    best_sc = cur_sc
    no_improve = 0
    it = 0

    while it < int(max_iter):
        if time.monotonic() > stop_time - 1.5:
            break
        if stop_flag is not None and stop_flag.get("v"):
            break
        it += 1

        # If the current perturbed point is far worse, snap back to the local best.
        if best_a is not None and cur_sc > best_sc + 1e-4:
            a = best_a.copy()
            cur_sc = best_sc

        rem = stop_time - time.monotonic()
        if rem < 2.0:
            break

        y, pred, m = _lp_tangent(
            a,
            K=K,
            cap=cap,
            time_limit=min(float(time_limit), max(0.25, rem - 1.5)),
        )

        if y is None:
            no_improve += 3
            if no_improve > 25:
                break
            continue

        dist = float(np.linalg.norm(y - a))
        if dist < 1e-10:
            no_improve += 2
            if no_improve > 20:
                break
            continue

        if pred is not None and m is not None and pred >= m - 1e-12:
            no_improve += 1
            if no_improve > 18:
                break
            continue

        x, sc = _line_search(a, y, cur_sc, stop_time=stop_time - 0.5)
        if x is not None and math.isfinite(sc) and sc < cur_sc - 1e-14:
            a = x.copy()
            cur_sc = float(sc)
            if cur_sc < best_sc - 1e-15:
                best_sc = cur_sc
                best_a = a.copy()
                if on_improve is not None:
                    on_improve(best_a)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 8 and no_improve <= 20:
                rng = np.random.default_rng((seed + 17 * it) % 1000000007)
                scale = 0.008 + 0.018 * min(1.0, no_improve / 8.0)
                noise = _smooth_noise(a, rng, scale)
                a2 = _normalize(np.maximum(a * (1.0 + noise), 0.0))
                if a2 is not None:
                    sc2 = _fast_score(a2)
                    if math.isfinite(sc2):
                        a = a2
                        cur_sc = sc2
                no_improve = 0
            elif no_improve > 20:
                break

    return best_a


def _coordinate_polish(a0, stop_time, seed=31, max_passes=500,
                       stop_flag=None, on_improve=None):
    a = _normalize(a0)
    if a is None:
        return None

    n = int(a.size)
    L = 2 * n - 1
    b = _self_conv(a)
    cur_sc = _fast_score(a)
    if not math.isfinite(cur_sc):
        return a

    best_a = a.copy()
    best_sc = cur_sc
    rng = np.random.default_rng(seed)
    no_improve = 0

    for _ in range(int(max_passes)):
        if time.monotonic() > stop_time - 1.0:
            break
        if stop_flag is not None and stop_flag.get("v"):
            break
        if cur_sc > best_sc + 1e-4:
            a = best_a.copy()
            b = _self_conv(a)
            cur_sc = best_sc

        m_val = cur_sc / (2.0 * n)
        if not math.isfinite(m_val) or m_val <= 0.0:
            break

        wexp = np.exp(np.clip(25.0 * (b - m_val), -745.0, 0.0))
        sw = float(wexp.sum())
        if sw <= 1e-300:
            wexp = np.zeros(L)
            wexp[int(np.argmax(b))] = 1.0
            sw = 1.0
        w = wexp / sw
        g = 2.0 * _cross_weight(w, a)

        pos_idx = np.where(a > 1e-10)[0]
        if pos_idx.size == 0:
            break

        K_pairs = int(min(18, max(4, n // 18)))
        donor_rank = np.argsort(-g[pos_idx])[:K_pairs]
        donors = pos_idx[donor_rank]
        receivers = np.argsort(g)[: max(K_pairs * 2, 8)]

        fracs = (0.01, 0.03, 0.06, 0.12, 0.25, 0.5)
        best_pair = None
        best_delta = 0.0
        best_new_sc = cur_sc

        # Structured pair transfers.
        for i in donors:
            i = int(i)
            ai = float(a[i])
            if ai <= 1e-12:
                continue
            for j in receivers:
                j = int(j)
                if i == j:
                    continue
                for fr in fracs:
                    delta = fr * ai
                    if delta <= 1e-14:
                        continue
                    arr = b.copy()
                    lin = 2.0 * delta
                    arr[i : i + n] -= lin * a
                    arr[j : j + n] += lin * a
                    q = delta * delta
                    pi = 2 * i
                    if pi < L:
                        arr[pi] += q
                    pj = 2 * j
                    if pj < L:
                        arr[pj] += q
                    sidx = i + j
                    if sidx < L:
                        arr[sidx] -= 2.0 * q
                    mn = float(arr.max())
                    sc = 2.0 * n * mn
                    if math.isfinite(sc) and sc < best_new_sc - 1e-14:
                        best_new_sc = sc
                        best_pair = (i, j)
                        best_delta = delta

        # Small random exploration.
        for _ in range(40):
            if time.monotonic() > stop_time - 1.0:
                break
            ii = int(pos_idx[int(rng.integers(pos_idx.size))])
            jj = int(rng.integers(n))
            if ii == jj:
                continue
            ai = float(a[ii])
            if ai <= 1e-12:
                continue
            for fr in (0.005, 0.02, 0.05):
                delta = fr * ai
                if delta <= 1e-14:
                    continue
                arr = b.copy()
                lin = 2.0 * delta
                arr[ii : ii + n] -= lin * a
                arr[jj : jj + n] += lin * a
                q = delta * delta
                pi = 2 * ii
                if pi < L:
                    arr[pi] += q
                pj = 2 * jj
                if pj < L:
                    arr[pj] += q
                sidx = ii + jj
                if sidx < L:
                    arr[sidx] -= 2.0 * q
                mn = float(arr.max())
                sc = 2.0 * n * mn
                if math.isfinite(sc) and sc < best_new_sc - 1e-14:
                    best_new_sc = sc
                    best_pair = (ii, jj)
                    best_delta = delta
                    break

        if best_pair is not None:
            i, j = best_pair
            a[i] -= best_delta
            a[j] += best_delta
            a = np.maximum(a, 0.0)
            ssum = float(a.sum())
            if ssum <= 1e-12:
                break
            a = a / ssum
            b = _self_conv(a)
            cur_sc = _fast_score(a)
            if math.isfinite(cur_sc) and cur_sc < best_sc - 1e-15:
                best_sc = cur_sc
                best_a = a.copy()
                no_improve = 0
                if on_improve is not None:
                    on_improve(best_a)
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 25:
            break

    return best_a


def _coarse_upsample(a, block=8, n=None):
    a = _normalize(a)
    if a is None:
        return a
    if n is None:
        n = int(a.size)
    block = int(max(1, block))
    if block <= 1:
        return a

    m = n // block
    if m < 2:
        return a

    if n % block == 0:
        coarse = a.reshape(m, block).sum(axis=1)
    else:
        edges = np.linspace(0, n, m + 1).astype(int)
        coarse = np.array(
            [a[edges[i] : edges[i + 1]].sum() if edges[i + 1] > edges[i] else 0.0
            for i in range(m)]
        )
    sc = float(coarse.sum())
    if sc <= 1e-12:
        coarse = np.full(m, 1.0 / m)
    else:
        coarse = coarse / sc

    if coarse.size == 1:
        return np.full(n, 1.0 / n)

    fine = np.interp(
        np.linspace(0.0, 1.0, n),
        np.linspace(0.0, 1.0, coarse.size),
        coarse,
    )
    return _normalize(fine)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    deadline = t0 + max(10.0, budget) - 20.0
    stop_flag = {"v": False}

    best_seq = None
    best_exact = math.inf

    def consider(cand):
        nonlocal best_seq, best_exact
        if stop_flag.get("v"):
            return
        an = _normalize(cand)
        if an is None:
            return
        fs = _fast_score(an)
        if not math.isfinite(fs):
            return
        # Exact check only when the surrogate is close to the incumbent.
        if best_seq is not None and fs > best_exact + 1e-9:
            return
        ex = _exact_score(an)
        if math.isfinite(ex) and (best_seq is None or ex < best_exact - 1e-15):
            best_seq = an.copy()
            best_exact = float(ex)
            if best_exact < TARGET - EARLY_MARGIN:
                stop_flag["v"] = True

    parent = _get_parent()
    consider(parent)

    if best_seq is None:
        fb = _normalize(_default_init(384))
        if fb is not None:
            consider(fb)
        else:
            best_seq = np.ones(1, dtype=np.float64)
            best_exact = _exact_score(best_seq)
            if not math.isfinite(best_exact):
                return [1.0]

    if not stop_flag.get("v"):
        parent_norm = _normalize(best_seq)
        main_a = parent_norm.copy()

        # Main length-384 (or parent length) LP phases, from wide to local.
        lp_stages = (
            # (cumulative fraction, cap, K)
            (0.10, 0.20, 512),
            (0.22, 0.08, 512),
            (0.36, 0.03, 512),
            (0.50, 0.01, None),
            (0.64, None, None),
        )
        for si, (frac, cap, K) in enumerate(lp_stages):
            if stop_flag.get("v"):
                break
            end = min(t0 + frac * budget, deadline - 120.0)
            if end <= time.monotonic() + 5.0:
                continue
            start_a = best_seq.copy() if best_seq is not None else main_a
            nnow = int(start_a.size)
            Lnow = 2 * nnow - 1
            Kuse = Lnow if K is None else int(max(50, min(K, Lnow)))
            try:
                res = _lp_drive(
                    start_a,
                    end,
                    seed=1000 + 37 * (si + 1),
                    K=Kuse,
                    cap=cap,
                    time_limit=1.1,
                    stop_flag=stop_flag,
                    on_improve=consider,
                )
            except Exception:
                res = None
            if res is not None:
                consider(res)

        # Smoothed quasi-Newton polish on the parent length.
        if not stop_flag.get("v"):
            end = min(t0 + 0.76 * budget, deadline - 80.0)
            if end > time.monotonic() + 15.0:
                try:
                    res = _lbfgs_smooth(
                        best_seq,
                        end,
                        stop_flag=stop_flag,
                        on_improve=consider,
                    )
                    if res is not None:
                        consider(res)
                except Exception:
                    pass

        # Exact pair-transfer polish.
        if not stop_flag.get("v"):
            end = min(t0 + 0.86 * budget, deadline - 45.0)
            if end > time.monotonic() + 10.0:
                try:
                    res = _coordinate_polish(
                        best_seq,
                        end,
                        seed=977,
                        max_passes=1000,
                        stop_flag=stop_flag,
                        on_improve=consider,
                    )
                    if res is not None:
                        consider(res)
                except Exception:
                    pass

        # Restart phases: symmetrization, coarse-to-fine, smooth perturbations.
        if not stop_flag.get("v"):
            end = min(t0 + 0.95 * budget, deadline - 25.0)
            a_best = best_seq

            if stop_flag.get("v") is False and end > time.monotonic() + 5.0:
                consider(0.5 * (a_best + a_best[::-1]))

            inits = []
            rng = np.random.default_rng(int(seed) + 20240517)

            for block in (8, 16, 32):
                if stop_flag.get("v"):
                    break
                try:
                    inits.append(_coarse_upsample(a_best, block, int(a_best.size)))
                except Exception:
                    pass

            for scale in (0.08, 0.18, 0.35):
                if stop_flag.get("v"):
                    break
                noise = _smooth_noise(a_best, rng, scale)
                inits.append(_normalize(np.maximum(a_best * (1.0 + noise), 0.0)))

            inits.append(_normalize(a_best + 0.25 * np.full(int(a_best.size), 1.0 / int(a_best.size))))
            inits.append(_normalize(0.75 * a_best + 0.25 * _default_init(int(a_best.size))))

            for ri, init in enumerate(inits):
                if stop_flag.get("v"):
                    break
                if time.monotonic() > end - 12.0:
                    break
                per = max(12.0, (end - time.monotonic()) / max(1, len(inits) - ri))
                e1 = min(end - 5.0, time.monotonic() + per)
                if e1 <= time.monotonic() + 3.0:
                    continue

                ii = _normalize(init)
                if ii is None:
                    continue
                consider(ii)

                try:
                    res = _lp_drive(
                        ii,
                        e1,
                        seed=91000 + 101 * (ri + 1),
                        K=min(512, 2 * int(ii.size) - 1),
                        cap=0.06,
                        time_limit=0.9,
                        stop_flag=stop_flag,
                        on_improve=consider,
                    )
                    if res is not None:
                        consider(res)
                except Exception:
                    res = ii

                if not stop_flag.get("v") and time.monotonic() < end - 6.0:
                    e2 = min(end - 2.0, time.monotonic() + 18.0)
                    try:
                        res2 = _coordinate_polish(
                            res,
                            e2,
                            seed=777000 + ri,
                            max_passes=120,
                            stop_flag=stop_flag,
                            on_improve=consider,
                        )
                        if res2 is not None:
                            consider(res2)
                    except Exception:
                        pass

            # One alternate longer length if there is still meaningful time.
            if not stop_flag.get("v") and time.monotonic() < end - 45.0:
                try:
                    Lalt = 512
                    init_alt = np.interp(
                        np.linspace(0.0, 1.0, Lalt),
                        np.linspace(0.0, 1.0, int(a_best.size)),
                        a_best,
                    )
                    init_alt = _normalize(init_alt)
                    if init_alt is not None:
                        consider(init_alt)
                        ealt = min(end - 5.0, time.monotonic() + min(50.0, 0.45 * (end - time.monotonic())))
                        if ealt > time.monotonic() + 8.0:
                            res_alt = _lp_drive(
                                init_alt,
                                ealt,
                                seed=555123,
                                K=min(450, 2 * Lalt - 1),
                                cap=0.06,
                                time_limit=1.2,
                                stop_flag=stop_flag,
                                on_improve=consider,
                            )
                            if res_alt is not None:
                                consider(res_alt)
                            if not stop_flag.get("v") and time.monotonic() < ealt - 6.0:
                                res_c = _coordinate_polish(
                                    res_alt if res_alt is not None else init_alt,
                                    ealt - 2.0,
                                    seed=888321,
                                    max_passes=250,
                                    stop_flag=stop_flag,
                                    on_improve=consider,
                                )
                                if res_c is not None:
                                    consider(res_c)
                except Exception:
                    pass

    # Final short polish and validation.
    final_a = _normalize(best_seq)
    if final_a is None:
        final_a = np.ones(1, dtype=np.float64)

    exact = _exact_score(final_a)
    if not math.isfinite(exact):
        return [1.0]

    if not stop_flag.get("v") and time.monotonic() < deadline - 25.0 and exact > TARGET - EARLY_MARGIN:
        try:
            res = _coordinate_polish(
                final_a,
                deadline - 8.0,
                seed=314159,
                max_passes=2000,
                stop_flag=stop_flag,
                on_improve=consider,
            )
            if res is not None:
                consider(res)
        except Exception:
            pass

    final_a = _normalize(best_seq)
    if final_a is None:
        final_a = np.ones(1, dtype=np.float64)

    final_exact = _exact_score(final_a)
    if not math.isfinite(final_exact):
        return [float(x) for x in np.ones(1, dtype=np.float64)]

    return [float(x) for x in final_a]
