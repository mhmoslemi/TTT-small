def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    """Build the convolution matrix A where (A @ g)[k] = (f*g)[k]"""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, time_limit=3.0):
    """Find direction g maximizing sum(g) s.t. (f*g)_k <= M, g >= 0"""
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None

                f_new = f_new / s_new
                sc = _score(f_new)

                    best_sc = sc
                    improved = True

    fejer_like = (np.sin(np.pi * x) / (np.pi * x + 1e-12)) ** 2
    fejer_like[np.abs(np.pi * x) < 1e-10] = 1.0
    inits.append(fejer_like)

    # Smooth random
    base = rng_main.normal(0, 1, n)
    kern = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
    kern /= kern.sum()
    smoothed = np.convolve(base, kern, 'same')
    inits.append(np.abs(smoothed) + 0.003)

    # Anti-peak (reduce center emphasis)
    anti = best_seq.copy()
    center = n // 2
    w = n // 6
    anti[center - w:center + w] *= 0.75
    inits.append(anti + 0.001)

    time_per_init = max(12.0, (phase2_end - t0) / max(1, len(inits)))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase2_end - 6.0:
            break
        if best_score < 1.503:
            break
        stop = min(phase2_end - 3.0, time.monotonic() + time_per_init)
        if time.monotonic() >= stop:
            break
        try:
            init_norm = np.clip(init, 0, None)
            s_init = init_norm.sum()
            if s_init > 1e-10:
                init_norm = init_norm / s_init
            else:
                init_norm = np.ones(n) / n
            res, sc = _lp_optimize(init_norm, n, stop, seed + 9973 * (idx + 1), lp_time=2.0)
            if np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_clipped = np.clip(res, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in res_clipped]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = res_clipped.copy()
                    best_score = exact
        except Exception:
            pass

    # === Phase 3: Polish + final LP ===
    if time.monotonic() < phase3_end - 5.0:
        # Gradient polish passes
        n_polish = 4
        polish_time_per = max(8.0, (phase3_end - time.monotonic() - 30.0) / max(1, n_polish))
        for pidx in range(n_polish):
            p_stop = time.monotonic() + polish_time_per
            if p_stop > phase3_end - 25.0:
                p_stop = phase3_end - 25.0
            if time.monotonic() >= p_stop - 1.0:
                break
            try:
                result, sc = _gradient_polish(best_seq, p_stop, seed=4000 + pidx * 137)
                if np.all(np.isfinite(result)) and result.sum() > 0.01:
                    result_clipped = np.clip(result, 0, 1000)
                    exact = float(evaluate_sequence([float(x) for x in result_clipped]))
                    if math.isfinite(exact) and exact < best_score:
                        best_seq = result_clipped.copy()
                        best_score = exact
            except Exception:
                pass

        # Final LP round
        if time.monotonic() < phase3_end - 15.0:
            final_lp_stop = phase3_end - 5.0
            try:
                f_norm = np.clip(best_seq, 0, None)
                s = f_norm.sum()
                if s > 1e-10:
                    f_norm = f_norm / s
                res, sc = _lp_optimize(f_norm, n, final_lp_stop, seed + 88888, lp_time=3.0)
                if np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res_clipped = np.clip(res, 0, 1000)
                    exact = float(evaluate_sequence([float(x) for x in res_clipped]))
                    if math.isfinite(exact) and exact < best_score:
                        best_seq = res_clipped.copy()
                        best_score = exact
            except Exception:
                pass

        # Final short polish
        final_polish_stop = t0 + budget - 5.0
        if time.monotonic() < final_polish_stop - 3.0:
            try:
                result, sc = _gradient_polish(best_seq, final_polish_stop, seed=66666)
                if np.all(np.isfinite(result)) and result.sum() > 0.01:
                    result_clipped = np.clip(result, 0, 1000)
                    exact = float(evaluate_sequence([float(x) for x in result_clipped]))
                    if math.isfinite(exact) and exact < best_score:
                        best_seq = result_clipped.copy()
                        best_score = exact
            except Exception:
                pass

    # Final validation
    final_seq = np.clip(best_seq, 0, 1000)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        return [float(x) for x in fallback]

    return [float(x) for x in final_seq]
