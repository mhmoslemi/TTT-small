```python
def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}"""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    sorted_v = np.sort(v)[::-1]
    css = np.cumsum(sorted_v)
    rho = np.arange(1, n+1)
    cond = css - target > rho * sorted_v
    if not np.any(cond):
        return np.full(n, target/n)
    rho = np.where(cond)[0][-1] + 1
    theta = (css[rho-1] - target) / rho
    return np.maximum(v - theta, 0.0)

def _conv_max(a):

    n = a.size

    c = irfft(fa * fa, N)[:2*n-1]

    fa = rfft(a, N)
    c = irfft(fa * fa, N)[:2*n-1]
    c[c < 0] = 0  # numerical errors

    M = c.max()
```

```python
    top_k = np.argpartition(c, -K)[-K:]
    w = c[top_k] - c.max() + c.max()  # just c[top_k]
    # Soft weight: exponential
    T = max(M * 0.01, 1e-15)

        hi = min(n-1, k)
        if lo <= hi:
            grad[lo:hi+1] += wi * a[k - np.arange(lo, hi+1)]
    return grad
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize


def _normalize(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.full(a.size, target / max(a.size, 1), dtype=np.float64)
    return a / s * target


def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (css - target) / rho < u
    idx = np.nonzero(cond)[0]
    if idx.size == 0:
        return np.full(n, target / n, dtype=np.float64)
    k = idx[-1]
    theta = (css[k] - target) / (k + 1)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s <= 1e-300:
        return np.full(n, target / n, dtype=np.float64)
    return (w / s) * target


def _conv(a):
    n = a.size
    if n == 0:
        return np.array([0.0])
    N = next_fast_len(2 * n - 1)
    fa = rfft(a, N)
    c = irfft(fa * fa, N)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    np.maximum(c, 0.0, out=c)
    return c


def _subgrad(a, c, K=16):
    """Compute subgradient of max(conv(a,a)) using soft active set."""
    n = a.size
    M = float(c.max())
    if not np.isfinite(M) or M <= 0:
        return np.zeros(n)
    # Soft weights on top-K
    if K >= c.size:
        top_idx = np.arange(c.size)
    else:
        top_idx = np.argpartition(c, -K)[-K:]
    T = max(M * 0.03, 1e-15)
    z = (c[top_idx] - M) / T
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    sw = float(w.sum())
    if sw <= 0:
        w = np.zeros_like(w)
        w[0] = 1.0
    else:
        w /= sw
    grad = np.zeros(n, dtype=np.float64)
    for i, k in enumerate(top_idx):
        wi = 2.0 * float(w[i])
        if wi < 1e-15:
            continue
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            a_idx = k - np.arange(lo, hi + 1)
            grad[lo:hi + 1] += wi * a[a_idx]
    return grad


def _lp_direction(a, M, deadline, rng):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M, b>=0. Returns normalized b."""
    n = a.size
    if n < 2 or n > 600 or M <= 0:
        return None
    if time.monotonic() >= deadline:
        return None
    m = 2 * n - 1
    # Build constraint matrix A (m x n): A[k,j] = a[j] if 0<=k-j<n else 0
    # (a*b)_k = sum_j a[j]*b[k-j] -> A[k, k-j] += a[j]
    # Actually (a*b)_k = sum_{j} a[j] b[k-j], so A[k,j] = a[j] if 0<=k-j<n
    # A[k,j] = a[j] if j <= k < j+n, i.e., k-j in [0,n-1]
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        # a[j]*b[k-j] contributes to row k = j + (k-j) = k where k-j in [0,n-1]
        # So for fixed j, A[k, j] = a[j] for k in [j, j+n-1]
        lo = j
        hi = min(j + n, m)
        A[lo:hi, j] = a[j]
    # Scale
    A = A / M
    # Objective: maximize sum(b) -> minimize -sum(b)
    c_obj = -np.ones(n, dtype=np.float64)
    c_obj += 1e-6 * np.asarray(rng.standard_normal(n), dtype=np.float64)
    b_ub = np.ones(m, dtype=np.float64)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                      options={"time_limit": max(0.5, min(8.0, deadline - time.monotonic() - 1.0)),
                               "presolve": True})
        if res is not None and getattr(res, 'success', False):
            b = np.asarray(res.x, dtype=np.float64)
            b = np.maximum(b, 0.0)
            s = float(b.sum())
            if s > 1e-12:
                return b / s
    except Exception:
        pass
    return None

        best_cand_M = M
        best_cand_a = None

        step_candidates = []
        if gnorm > 1e-14:

                    if cm < best_cand_M * (1 - 1e-10):

            c = best_cand_c
            M = best_cand_M
            if M < best_M * (1 - 1e-12):

def _smooth_init(n, kind, rng):
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    if kind == 0:
        a = (1.0 + np.cos(2.0 * np.pi * x)) ** 2
    elif kind == 1:
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * x)
    elif kind == 2:
        c = float(rng.uniform(0.3, 0.7))
        s = float(rng.uniform(0.15, 0.4))
        a = np.exp(-((x - c) ** 2) / (2 * s * s)) + 0.05
    elif kind == 3:
        a = np.sin(np.pi * (x + 0.5 / n)) ** 2
    elif kind == 4:
        alpha = float(rng.uniform(0.5, 2.0))
        a = (x + 0.01) ** alpha * (1.0 - x + 0.01) ** alpha
    else:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 1e-8)
    return _normalize(a)


def _basis_optimize(n, D, deadline, rng, init_params=None):
    """Optimize in Chebyshev coefficient space."""
    x = np.linspace(0, 1, n, endpoint=False)
    t = 2 * x - 1
    B = np.zeros((D, n))
    B[0] = 1.0
    if D > 1:
        B[1] = t
    for k in range(2, D):
        B[k] = 2 * t * B[k-1] - B[k-2]

    def reconstruct(th):
        th = np.asarray(th, dtype=np.float64).ravel()
        loga = B @ th
        loga = np.where(np.isfinite(loga), loga, 0.0)
        loga -= float(loga.max())
        a = np.exp(np.clip(loga, -40, 0))
        a = np.maximum(a, 1e-15)
        return _normalize(a)

    def objective(th):
        a = reconstruct(th)
        c = _conv(a)
        return float(c.max())

    if init_params is None:
        init_params = rng.standard_normal(D) * 0.3

    best_a = reconstruct(init_params)
    best_obj = objective(init_params)

    rem = deadline - time.monotonic()
    maxiter = int(min(300, max(30, rem * 0.8)))
    try:
        res = minimize(objective, init_params, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-8, 'fatol': 1e-10})
        if res is not None and hasattr(res, 'x') and res.x is not None:
            a_new = reconstruct(res.x)
            obj_new = objective(a_new)
            if obj_new < best_obj:
                best_a, best_obj = a_new, obj_new
    except Exception:
        pass
    return best_a, best_obj


def _exact_score(a):
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            return float('inf')
        a = np.clip(a, 0, 1000)
        if a.sum() < 0.01:
            return float('inf')
        return float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        return float('inf')


def _resample(a, m):
    n = a.size
    a = _normalize(a)
    if n == m:
        return a.copy()
    if m > n:
        idx = (np.arange(m) * n) // m
        counts = np.bincount(idx, minlength=n).astype(np.float64)
        counts[counts <= 0] = 1.0
        b = a[idx] / counts[idx]
    else:
        idx = (np.arange(n) * m) // n
        b = np.zeros(m)
        np.add.at(b, idx, a)
    return _normalize(b)

        best_score = _exact_score(best_a)
    if not np.isfinite(best_score):
        return [1.0]

    n_parent = best_a.size

    # Phase 1: Local optimization from parent (main phase)
    if deadline - time.monotonic() > 30.0:

                if time.monotonic() >= deadline - 15.0:

                # Blend with best

                sub_dl = time.monotonic() + min(sub_time / 3.0, deadline - time.monotonic() - 10.0)

                    if np.isfinite(sc) and sc < best_score:

                        best_score = float(sc)
                except Exception:
                    pass

    # Phase 3: Chebyshev basis optimization on best shape
    if deadline - time.monotonic() > 25.0:
        n_cur = best_a.size
        D = min(14, max(6, n_cur // 40 + 4))
        b_dl = time.monotonic() + min(40.0, deadline - time.monotonic() - 10.0)
        try:

            log_best = np.log(np.maximum(best_a, 1e-15))
            th_init, _, _, _ = np.linalg.lstsq(B.T if False else np.zeros((1,1)), log_best, rcond=None) if False else (None, None, None, None)
            # Build basis and fit
            x = np.linspace(0, 1, n_cur, endpoint=False)
            t_param = 2 * x - 1
            Bmat = np.zeros((D, n_cur))
            Bmat[0] = 1.0
            if D > 1:
                Bmat[1] = t_param
            for kk in range(2, D):
                Bmat[kk] = 2 * t_param * Bmat[kk-1] - Bmat[kk-2]
            th0, _, _, _ = np.linalg.lstsq(Bmat.T, log_best, rcond=None)
            if np.all(np.isfinite(th0)):

                    best_score = float(sc)
        except Exception:
            pass

    # Phase 4: Final polish with very small steps
```
