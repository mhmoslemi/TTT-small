```
import numpy as np, math, time
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize, linprog

class _Ctx:
    def __init__(self,n,tau=None, mode='l'):
       self.n=n; self.L=2*n-1; self.mode=mode; self.tau=tau
       self.N=next_fast_len(3*n-2)
    def a_from_x(self,x): ...
    def _raw(self,a,w):
       r=w[::-1]
       F=rfft(a,self.N)
       FR=rfft(r,self.N)
       conv=irfft(F*FR,self.N)
       return conv[self.n-1:self.L][::-1]
    def _fun(self,x):
       a=self.a_from_x(x)
       F=rfft(a,self.N)
       y=irfft(F*F,self.N)[:self.L]
       np.maximum(y,0,out=y)
       if self.mode=='quad':
           val=float(np.dot(y,y))
           raw=self._raw(a,y)
           dot=float(np.dot(a,raw))
           grad=4.0*a[:-1]*(raw[:-1]-dot)
       else:
           tau=self.tau
           m=float(y.max())
           z=tau*(y-m)
           # If tau too huge, exp(z) underflow okay. If tau small, okay.
           e=np.exp(z); s=e.sum()
           if not finite or s<=0:
              idx=int(np.argmax(y)); p=np.zeros(self.L, dtype=float); p[idx]=1.0; val=m
           else:
              p=e/s; val=m + math.log(s)/tau
           raw=self._raw(a,p)
           dot=float(np.dot(a,raw))
           grad=2.0*a[:-1]*(raw[:-1]-dot)
       # Clean grad
```

```python
            dot = float(np.dot(a, raw))
            grad = 4.0 * a[:-1] * (raw[:-1] - dot)
        elif self.mode == "top":
            m = float(y.max())
            k = max(1, min(self.topk, self.L))
            idx = np.argsort(y)[-k:]
            p = np.zeros(self.L, dtype=float)
            p[idx] = 1.0 / k
            val = m
            raw = self._raw_with_F(F, p)
            dot = float(np.dot(a, raw))
            grad = 2.0 * a[:-1] * (raw[:-1] - dot)
        else:
            tau = self.tau
            m = float(y.max())
            z = tau * (y - m)
            e = np.exp(z)
            sE = float(e.sum())
            if not math.isfinite(sE) or sE <= 1e-300:
                p = np.zeros(self.L, dtype=float)
                p[int(np.argmax(y))] = 1.0
                val = m
            else:
                p = e / sE
                val = m + math.log(sE) / tau
            raw = self._raw_with_F(F, p)
            dot = float(np.dot(a, raw))
            grad = 2.0 * a[:-1] * (raw[:-1] - dot)

        if not np.all(np.isfinite(grad)):
            grad = np.zeros_like(grad)
        if not math.isfinite(val):
            return math.inf, np.zeros_like(grad)
        return float(val), grad


def _run_minimize(ctx, x, stop):
    x = np.asarray(x, dtype=float).ravel()
    if x.size == 0:
        return x
    while time.monotonic() < stop - 1.0:
        try:
            res = minimize(
                ctx.fun,
                x,
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": 20,
                    "ftol": 1e-12,
                    "gtol": 1e-8,
                    "maxls": 10,
                    "iprint": 0,
                },
            )
        except Exception:
            break

        if res.x is not None and np.all(np.isfinite(res.x)):
            x = np.asarray(res.x, dtype=float).ravel()
        else:
            break

        if not math.isfinite(float(res.fun)):
            break
        if bool(getattr(res, "success", False)):
            break
    return x


def _optimize_init(a0, stop, seed=0, final=False):
    a0 = _normalize(a0)
    if a0 is None:
        return None, math.inf

    n = a0.size
    best_a = a0.copy()
    best_sc = _eval_exact(best_a)
    x = _x_from_a(a0, n)
    best_x = x.copy()

    if final:
        phases = [
            ("log", 1e4, 1.0),
            ("log", 5e4, 1.5),
            ("top", None, 2.0),
        ]
    else:
        phases = [
            ("quad", None, 1.5),
            ("log", 1e3, 0.75),
            ("log", 5e3, 0.9),
            ("log", 2e4, 1.0),
            ("log", 5e4, 1.0),
            ("log", 1e5, 0.9),
            ("top", None, 1.6),
        ]

    rng = np.random.default_rng(seed)

    for idx, (mode, tau, w) in enumerate(phases):
        now = time.monotonic()
        rem = stop - now
        if rem < 3.0:
            break
        rem_w = sum(wi for _, _, wi in phases[idx:])
        if rem_w <= 0:
            break
        phase_time = rem * (w / rem_w)
        if idx == len(phases) - 1:
            phase_stop = stop
        else:
            phase_stop = now + max(2.0, min(phase_time, rem - 1.0))

        if mode == "quad":
            ctx = _OptCtx(n, mode="quad")
        elif mode == "log":
            ctx = _OptCtx(n, mode="log", tau=tau, topk=max(16, min(128, n // 6)))
        else:
            ctx = _OptCtx(n, mode="top", topk=max(16, min(128, n // 6)))

        x = _run_minimize(ctx, x, phase_stop)

        a = _a_from_x(x, n)
        scf = _score_fft(a)
        if scf < best_sc + 1e-9:
            sc = _eval_exact(a)
            if math.isfinite(sc) and sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a.copy()
                best_x = x.copy()

    for xx in (x, best_x):
        if xx is not None and len(xx) == n - 1:
            a = _a_from_x(xx, n)
            sc = _eval_exact(a)
            if math.isfinite(sc) and sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a.copy()

    # Occasionally retry from the best stored state if the local path drifted.
    if time.monotonic() < stop - 5.0 and best_x is not None:
        x = best_x + rng.normal(0.0, 0.02, max(0, n - 1))
        x = np.where(np.isfinite(x), x, 0.0)
        phase_stop = min(stop, time.monotonic() + max(4.0, (stop - time.monotonic()) * 0.25))
        ctx = _OptCtx(n, mode="top", topk=max(16, min(128, n // 6)))
        x = _run_minimize(ctx, x, phase_stop)
        a = _a_from_x(x, n)
        sc = _eval_exact(a)
        if math.isfinite(sc) and sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _build_A_active(f, idx):
    n = f.size
    A = np.zeros((len(idx), n), dtype=float)
    for r, k in enumerate(idx):
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        if i1 >= i0:
            A[r, i0:i1 + 1] = f[k - i1:k - i0 + 1][::-1]
    return A


def _solve_lp(A, M, tl):
    if A.size == 0:
        return None
    n = A.shape[1]
    c = -np.ones(n, dtype=float)
    b = np.full(A.shape[0], float(M), dtype=float)
    bounds = [(0.0, 1000.0)] * n
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": float(tl)},
        )
    except Exception:
        return None
    if res is None or getattr(res, "x", None) is None:
        return None
    g = np.asarray(res.x, dtype=float).ravel()
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.clip(g, 0.0, 1000.0)
    s = float(g.sum())
    if s <= 1e-12:
        return None
    return g / s


def _lp_polish(a0, stop, seed=1):
    best = _normalize(a0)
    if best is None:
        return None, math.inf

    n = best.size
    f = best.copy()
    best_sc = _eval_exact(f)
    rng = np.random.default_rng(seed)

    t = 0.25
    stall = 0
    no_lp = 0
    it = 0

    while time.monotonic() < stop - 5.0 and it < 80:
        y = _conv_fft_y(f)
        m = float(y.max())
        if not math.isfinite(m) or m <= 0.0:
            break

        L = 2 * n - 1
        K = int(min(L, 96 + 32 * min(it // 3, 6)))
        idx = np.argsort(-y)[:K]
        A = _build_A_active(f, idx)

        tl = float(min(5.0, max(0.5, (stop - time.monotonic()) * 0.08)))
        g = _solve_lp(A, m, tl)

        if g is None:
            no_lp += 1
            if no_lp > 6:
                break
            fac = 1.0 + 0.02 * rng.normal(size=n)
            nf = _normalize(f * fac)
            f = nf if nf is not None else best.copy()
            it += 1
            continue

        no_lp = 0
        if float(np.linalg.norm(g - f)) < 1e-8:
            fac = 1.0 + 0.02 * rng.normal(size=n)
            nf = _normalize(f * fac)
            f = nf if nf is not None else best.copy()
            stall += 2
            if stall > 20:
                break
            it += 1
            continue

        cand = []
        base = [0.02, 0.05, 0.10, 0.15, 0.20, 0.30, 0.40, 0.50, 0.60, 0.70, 0.80, 0.90]
        for v in base + [t * 0.5, t * 0.75, t, t * 1.25, t * 1.5, t * 2.0]:
            v = float(v)
            if 1e-4 < v < 1.0 - 1e-4:
                if not any(abs(v - u) < 1e-6 for u in cand):
                    cand.append(v)

        evals = []
        for v in cand:
            h = _normalize((1.0 - v) * f + v * g)
            if h is None:
                continue
            sf = _score_fft(h)
            if math.isfinite(sf):
                evals.append((sf, v, h))

        # Fine line search around the best coarse step.
        if evals:
            evals.sort(key=lambda z: z[0])
            _, bv, _ = evals[0]
            lo = max(1e-3, 0.8 * bv)
            hi = min(1.0 - 1e-3, 1.25 * bv)
            if hi > lo:
                for v in np.linspace(lo, hi, 6):
                    v = float(v)
                    if any(abs(v - z[1]) < 1e-6 for z in evals):
                        continue
                    h = _normalize((1.0 - v) * f + v * g)
                    if h is None:
                        continue
                    sf = _score_fft(h)
                    if math.isfinite(sf):
                        evals.append((sf, v, h))
                evals.sort(key=lambda z: z[0])

        accepted = False
        if evals:
            for _, v, h in evals[:3]:
                sc = _eval_exact(h)
                if math.isfinite(sc) and sc < best_sc - 1e-14:
                    best_sc = sc
                    best = h.copy()
                    f = h.copy()
                    t = float(max(0.02, min(0.98, v)))
                    stall = 0
                    accepted = True
                    break

        if not accepted:
            stall += 1
            if stall % 3 == 0 and evals:
                # Occasionally follow the surrogate best even if exact did not
                # tick yet; line search will protect the saved best.
                _, _, bh = evals[0]
                f = bh.copy()
                t = float(max(0.05, min(0.98, evals[0][1])))

        if stall > 25:
            fac = 1.0 + 0.04 * rng.normal(size=n)
            nf = _normalize(f * fac)
            f = nf if nf is not None else best.copy()
            stall = 0

        it += 1

    for cand in (best, f):
        if cand is not None:
            sc = _eval_exact(cand)
            if math.isfinite(sc) and sc < best_sc - 1e-15:
                best_sc = sc
                best = cand.copy()

    return best, best_sc


def _local_pair_polish(a0, stop, seed=7):
    best = _normalize(a0)
    if best is None:
        return None, math.inf

    n = best.size
    f = best.copy()
    best_sc = _eval_exact(f)
    rng = np.random.default_rng(seed)
    noimp = 0

    while time.monotonic() < stop - 2.0 and noimp < 30:
        y = _conv_fft_y(f)
        k = max(1, min(48, 2 * n - 1))
        idx = np.argsort(y)[-k:]
        p = np.zeros(2 * n - 1, dtype=float)
        p[idx] = 1.0 / k
        g = 2.0 * _raw_from_weights(f, p, n)

        hi = np.argsort(g)[-6:]
        lo = np.argsort(g)[:6]

        improved = False
        for i in hi:
            if improved:
                break
            ai = float(f[i])
            if ai <= 1e-15:
                continue
            for j in lo:
                if i == j:
                    continue
                for r in (0.1, 0.25, 0.5, 0.75, 0.9):
                    h = f.copy()
                    d = r * ai
                    h[i] = ai - d
                    if h[i] < 0.0:
                        continue
                    h[j] = h[j] + d
                    sf = _score_fft(h)
                    if sf < best_sc - 1e-13:
                        sc = _eval_exact(h)
                        if math.isfinite(sc) and sc < best_sc - 1e-14:
                            best_sc = sc
                            best = h.copy()
                            f = h.copy()
                            improved = True
                            break
                if improved:
                    break

        if not improved:
            noimp += 1
            if noimp % 5 == 0:
                fac = 1.0 + 0.01 * rng.normal(size=n)
                nf = _normalize(f * fac)
                if nf is not None:
                    f = nf

    return best, best_sc


def _make_inits(parent, seed):
    n = parent.size
    out = []
    i = np.arange(n, dtype=float)
    central = _central_binomial(n)
    if central is None:
        central = np.ones(n) / n

    rng = np.random.default_rng(seed)

    cands = []
    cands.append(parent.copy())

    for beta in (0.15, 0.35, 0.60, 0.80):
        cands.append((1.0 - beta) * parent + beta * central)

    cands.append(central.copy())
    cands.append(central[::-1].copy())

    for alpha in (0.6, 2.0, 6.0):
        v = 1.0 / np.sqrt(i + alpha) + 1.0 / np.sqrt((n - 1 - i) + alpha)
        cands.append(v)

    for p in (1.5, 3.0):
        v = 0.02 + np.power(1.0 + np.cos(2.0 * math.pi * i / float(n)), p)
        cands.append(v)

    for scale in (0.08, 0.18, 0.32):
        k = max(3, n // 64)
        sigma = max(1.0, k // 3)
        kern = np.exp(-0.5 * (np.arange(-k, k + 1) / sigma) ** 2)
        kern /= kern.sum()
        noise = np.convolve(rng.normal(0.0, 1.0, n), kern, "same")
        cands.append(0.5 * parent + 0.3 * central + scale * np.abs(noise) + 0.01)

    for c in cands:
        a = _normalize(c)
        if a is None:
            continue
        if any(float(np.linalg.norm(a - b)) < 1e-5 for b in out):
            continue
        out.append(a)

    if not out:
        out.append(np.ones(n) / float(n))

    return out


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    opt_deadline = t0 + budget - 18.0
    if opt_deadline < t0 + 60.0:
        opt_deadline = t0 + budget - 8.0

    target = 1.50274349

    # Load the provided parent directly, without embedding its values.
    try:
        parent = np.asarray(height_sequence_1, dtype=float).ravel()
    except Exception:
        parent = np.ones(384, dtype=float)

    parent = _normalize(parent)
    if parent is None:
        parent = np.ones(384, dtype=float) / 384.0

    best_seq = parent.copy()
    best_score = _eval_exact(best_seq)
    if not math.isfinite(best_score):
        best_score = math.inf

    inits = _make_inits(parent, int(seed))
    max_inits = max(2, min(len(inits), int(max(2, budget / 90.0))))

    # Phase 1: smooth / active L-BFGS from diverse starts.
    smooth_limit = min(opt_deadline - 90.0, t0 + 0.58 * budget)
    if smooth_limit < t0 + 25.0:
        smooth_limit = opt_deadline - 25.0

    for idx, init in enumerate(inits[:max_inits]):
        now = time.monotonic()
        if now >= smooth_limit:
            break
        if best_score <= target + 1e-5:
            break

        rem = smooth_limit - now
        remaining = max(1, max_inits - idx)
        per = rem / float(remaining)
        stop = now + max(12.0, min(per, rem))
        if stop > smooth_limit:
            stop = smooth_limit
        if stop - time.monotonic() < 5.0:
            break

        try:
            cand, sc = _optimize_init(init, stop, seed=int(seed) + 7919 * (idx + 1), final=False)
        except Exception:
            cand, sc = None, math.inf
        if cand is not None and math.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(cand, dtype=float).copy()
            best_score = float(sc)

    # Phase 2: LP-guided local improvement on the best state.
    now = time.monotonic()
    if now < opt_deadline - 45.0:
        if best_score > target + 1e-4:
            alloc = max(35.0, min(0.24 * budget, (opt_deadline - now) * 0.40))
        else:
            alloc = max(20.0, min(0.12 * budget, (opt_deadline - now) * 0.20))
        lp_stop = now + alloc
        if lp_stop > opt_deadline - 35.0:
            lp_stop = opt_deadline - 35.0
        try:
            cand, sc = _lp_polish(best_seq, lp_stop, seed=9973)
        except Exception:
            cand, sc = None, math.inf
        if cand is not None and math.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(cand, dtype=float).copy()
            best_score = float(sc)

    # Phase 3: cheap exact pair moves driven by active-constraint subgradients.
    now = time.monotonic()
    if now < opt_deadline - 35.0:
        alloc = max(15.0, min(0.06 * budget, (opt_deadline - now) * 0.30))
        pair_stop = now + alloc
        if pair_stop > opt_deadline - 25.0:
            pair_stop = opt_deadline - 25.0
        try:
            cand, sc = _local_pair_polish(best_seq, pair_stop, seed=313)
        except Exception:
            cand, sc = None, math.inf
        if cand is not None and math.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(cand, dtype=float).copy()
            best_score = float(sc)

    # Phase 4: final high-resolution refine.
    now = time.monotonic()
    if now < opt_deadline - 8.0:
        alloc = max(25.0, min(0.12 * budget, opt_deadline - now - 5.0))
        final_stop = now + alloc
        if final_stop > opt_deadline - 3.0:
            final_stop = opt_deadline - 3.0
        try:
            cand, sc = _optimize_init(best_seq, final_stop, seed=int(seed) + 55511, final=True)
        except Exception:
            cand, sc = None, math.inf
        if cand is not None and math.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(cand, dtype=float).copy()
            best_score = float(sc)

    # Final validation and safe return.
    final_seq = np.asarray(best_seq, dtype=float).ravel()
    if final_seq.size == 0:
        final_seq = parent.copy()
    final_seq = np.where(np.isfinite(final_seq), final_seq, 0.0)
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    if float(final_seq.sum()) < 0.01:
        fallback = np.asarray(parent, dtype=float).ravel()
        fallback = np.where(np.isfinite(fallback), fallback, 0.0)
        fallback = np.clip(fallback, 0.0, 1000.0)
        if float(fallback.sum()) < 0.01:
            fallback = np.ones(384, dtype=float)
        final_seq = fallback

    # One last exact score check; if somehow invalid, fall back to uniform.
    try:
        final_score = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_score = _score_np(final_seq)

    if not math.isfinite(final_score):
        return [1.0]

    return [float(x) for x in final_seq]
```
