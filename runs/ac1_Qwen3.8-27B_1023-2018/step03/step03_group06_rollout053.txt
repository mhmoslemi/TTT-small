```python
import time
import math

def _fftlen(m):

    if m not in _fft_cache:

    return _fft_cache[m]

def _conv(a):
    """Linear convolution a*a, nonnegative output."""
    n = len(a)
    if n < 1: return np.array([0.0])
    L = _fftlen(2*n-1)
    fa = rfft(a, L)
    c = irfft(fa*fa, L)[:2*n-1]
```

```python
def _lbfgs_obj(a, beta):
    """Return (F, dF/dz) where a = softmax(z), F = (1/beta)*log(sum exp(beta*c))"""
    n = len(a)
    c = _cv(a)
    mc = float(c.max())
    x = beta * (c - mc)
    x = np.clip(x, -50, 0)
    exp_x = np.exp(x)
    S = float(exp_x.sum())
    F = mc + math.log(S) / beta
    
    w = beta * exp_x / S  # softmax weights times beta
    
    # dF/da
    r = np.correlate(w, a, mode='valid')  # length n
    dFda = r.copy()
    for i in range(n):
        if 2*i < 2*n-1:
            dFda[i] += w[2*i] * a[i]
    
    return F, dFda
```

```python
def _grad_z(dFda, a):
    """dF/dz[j] = a[j] * (dFda[j] - sum_i a[i]*dFda[i])"""
    mean_a = float(np.dot(a, dFda))
    grad = a * (dFda - mean_a)
    return grad
```

```python
def _lbfgs(z0, beta, stop_time, maxfun=1000):
    n = len(z0)
    
    def fgrad(z):
        if time.monotonic() > stop_time:
            raise _TO
        z = np.clip(z, -50, 50)  # prevent overflow
        zm = z - z.max()
        ez = np.exp(zm)
        Z = ez.sum()
        if Z < 1e-300 or not math.isfinite(Z):
            return 1e100, np.zeros(n)
        a = ez / Z
        
        F, dFda = _lbfgs_obj(a, beta)
        if not math.isfinite(F):
            return 1e100, np.zeros(n)
        
        dFdz = _grad_z(dFda, a)
        dFdz = np.where(np.isfinite(dFdz), dFdz, 0.0)
        return F, dFdz
    
    try:
        res = minimize(fgrad, z0, jac=True, method='L-BFGS-B',
                       options={'maxiter': 300, 'maxfun': maxfun, 'ftol': 1e-14,
                                'gtol': 1e-10, 'maxls': 30})
        if hasattr(res, 'x') and res.x is not None:
            zm = res.x - res.x.max()
            a = np.exp(zm)
            a /= a.sum()
            return a
    except _TO:
        pass
    except Exception:
        pass
    return None
```

```python
def _lp_step(a, max_iters=50, stop_time=None):
    """Iterated LP: minimize t s.t. c_a[j] + 2*(a*delta)[j] <= t, sum(delta)=0, delta >= -a"""
    n = len(a)
    if n < 2: return a, _sc(a)
    
    best = a.copy()
    best_sc = _sc(a)
    rng_local = np.random.default_rng(12345)
    
    for it in range(max_iters):
        if stop_time and time.monotonic() > stop_time:
            break
        
        c = _cv(a)
        M = float(c.max())
        
        # Build LP
        # min t
        # s.t. 2*sum_i a[j-i]*delta[i] - t <= -c[j]  for j=0..2n-2
        #      sum_i delta[i] = 0
        #      delta[i] >= -a[i]
        
        # A_ub: (2n-1) x (n+1)
        # For constraint j: sum_i coeff[j,i]*delta[i] - t <= -c[j]
        # coeff[j,i] = 2*a[j-i] if 0<=j-i<n, else 0
        
        A_ub = np.zeros((2*n-1, n+1))
        for j in range(2*n-1):
            lo = max(0, j-n+1)
            hi = min(n-1, j)
            for i in range(lo, hi+1):
                A_ub[j, i] = 2.0 * a[j-i]
            A_ub[j, n] = -1.0
        
        b_ub = -c  # (linear terms) - t <= -c[j]
        
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = [(-a[i], None) for i in range(n)] + [(0, 2*M)]
        
        obj = np.zeros(n+1)
        obj[n] = 1.0
        
        try:
            res = linprog(obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': max(1, (stop_time or time.monotonic()) - time.monotonic())})
        except:
            break
        
        if res.status != 0 or res.x is None:
            break
        
        delta = res.x[:n]
        delta = np.clip(delta, -a + 1e-15, None)
        
        # Limit step size
        step = float(np.abs(delta).sum()) / max(1.0, float(a.sum()))
        if step > 0.3:
            delta *= 0.3 / step
        
        a_new = a + delta
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s
        
        sc = _sc(a_new)
        if math.isfinite(sc) and sc < best_sc - 1e-14:
            best_sc = sc
            best = a_new.copy()
            a = best
        else:
            # No improvement, try a smaller step
            for frac in [0.5, 0.1, 0.01]:
                a_try = a + frac * delta
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-12: continue
                a_try /= s_try
                sc_try = _sc(a_try)
                if math.isfinite(sc_try) and sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    best = a_try.copy()
                    a = best
                    break
            else:
                # No improvement even with small steps
                # Add small noise
                noise = rng_local.normal(0, 0.001 * a.mean(), n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                a_pert /= a_pert.sum()
                sc_pert = _sc(a_pert)
                if math.isfinite(sc_pert) and sc_pert < best_sc + 0.01:
                    a = a_pert
                if best_sc < 1.50 or it > 30:
                    break
    
    return best, best_sc
```

```python
                else:
                    g[i] = a[j]

            a_new = a

        # Actually, for subgradient, we should sometimes accept non-improving steps
```

import numpy as np
import time, math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize

class _TO(Exception): pass
_FC = {}

def _fl(m):
    m = int(max(1, m))
    if m not in _FC:
        _FC[m] = int(next_fast_len(m))
    return _FC[m]

def _cv(a):
    n = len(a)
    if n < 1: return np.array([0.0])
    L = _fl(2*n-1)
    fa = rfft(a, L)
    return np.maximum(irfft(fa*fa, L)[:2*n-1], 0.0)

def _nm(a):
    a = np.asarray(a, np.float64).ravel()
    a = np.clip(np.nan_to_num(a, nan=0., posinf=1e6, neginf=0.), 0., 1000.)
    s = float(a.sum())
    if s < 1e-15 or not math.isfinite(s):
        n = max(1, len(a))
        return np.ones(n, np.float64) / n
    return a / s

def _sc(a):
    a = _nm(a)
    n = len(a)
    if n < 1: return math.inf
    c = _cv(a)
    mx = float(c.max())
    if not math.isfinite(mx): return math.inf
    return 2.0 * float(n) * mx

def _ex(a):
    try:
        fn = globals().get('evaluate_sequence')
        if fn is None: return math.inf
        return float(fn([float(x) for x in np.asarray(a, np.float64).ravel()]))
    except: return math.inf

def _pt():
    try:
        h = globals()['height_sequence_1']
        if h is None: return None
        p = np.asarray(h, np.float64).ravel()
        if len(p) < 1: return None
        p = np.clip(np.nan_to_num(p, nan=0., posinf=1e6, neginf=0.), 0., 1000.)
        s = float(p.sum())
        if s < 1e-15: return None
        return p / s
    except: return None

# --- L-BFGS-B with softmax ---
def _lf(z, beta):
    """Objective and gradient for L-BFGS-B. z -> a = softmax(z)."""
    n = len(z)
    zm = z - z.max()
    ez = np.exp(np.clip(zm, -50, 50))
    Z = float(ez.sum())
    if Z < 1e-300 or not math.isfinite(Z):
        return 1e100, np.zeros(n)
    a = ez / Z
    c = _cv(a)
    mc = float(c.max())
    x = np.clip(beta * (c - mc), -50, 0)
    ex_ = np.exp(x)
    S = float(ex_.sum())
    F = mc + math.log(S) / beta
    if not math.isfinite(F):
        return 1e100, np.zeros(n)
    w = beta * ex_ / S
    r = np.correlate(w, a, 'valid')
    dFda = r.copy()
    # Add the 2x correction
    two_i = np.arange(0, 2*n-1, 2)
    dFda += w[two_i] * a
    mu = float(np.dot(a, dFda))
    dFdz = a * (dFda - mu)
    dFdz = np.where(np.isfinite(dFdz), dFdz, 0.0)
    return float(F), dFdz

def _lbfgs_run(a0, beta, stop, maxfun=1200):
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-30))
    z0 = (z0 - z0.mean())  # center to avoid large values
    
    def fgrad(z):
        if time.monotonic() > stop:
            raise _TO
        return _lf(z, beta)
    
    try:
        res = minimize(fgrad, z0, jac=True, method='L-BFGS-B',
                       options={'maxiter': 400, 'maxfun': maxfun,
                                'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 30,

            zm = res.x - res.x.max()
            a = np.exp(np.clip(zm, -50, 50))
            s = a.sum()
            if s > 1e-300 and math.isfinite(s):

    except _TO:

    except:

    return None

# --- LP linearization ---
def _lp_refine(a0, stop, rng, max_iter=80):
    a = a0.copy()
    n = len(a)
    if n < 2 or time.monotonic() >= stop: return a, _sc(a)
    best = a.copy()
    best_sc = _sc(a)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() >= stop - 2.0: break
        
        c = _cv(a)
        M = float(c.max())
        
        # Build LP: min t s.t. 2*(a*delta)[j] - t <= -c[j], sum(delta)=0, delta >= -a
        A_ub = np.zeros((2*n-1, n+1), np.float64)
        for j in range(n):
            A_ub[j, :j+1] = 2.0 * a[:j+1][::-1]
            A_ub[j, n] = -1.0
        for j in range(n, 2*n-1):
            lo = j - n + 1
            A_ub[j, lo:] = 2.0 * a[:n-lo][::-1]
            A_ub[j, n] = -1.0
