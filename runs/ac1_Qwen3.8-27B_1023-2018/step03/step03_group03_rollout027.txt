import time
import math
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0 or not np.all(np.isfinite(p)):
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0, 1000)

def _conv(a):
    return np.convolve(a, a)

def _score_exact(a):
    """Matches evaluate_sequence exactly."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    c = np.convolve(a, a)
    m = float(c.max())
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """Build (2n-1) x n matrix where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_step(f, n, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0.
    Returns normalized g (sum=1) or None."""
    A = _build_A(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, 1.0)]*n,  # add upper bound 1 for safety
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-10:
        return None
    return g / gs
