    np.maximum(c, 0.0, out=c)

    g = np.asarray(res.x, dtype=np.float64)

    polish_end = deadline - 5.0
    if time.monotonic() < polish_end - 15.0:
        try:
            best_seq, best_score = _coord_polish(best_seq, min(polish_end - 5, time.monotonic() + (polish_end - time.monotonic()) * 0.6), seed=4242)
        except Exception:
            pass

    # Second polish with different seed
    if time.monotonic() < polish_end - 10.0:
        try:
            best_seq2, sc2 = _coord_polish(best_seq, min(polish_end - 3, time.monotonic() + 60.0), seed=12345)
            if sc2 < best_score - 1e-12:
                best_seq = best_seq2
                best_score = sc2
        except Exception:
            pass

    # Final LP micro-optimization if time permits
    if time.monotonic() < polish_end - 15.0:
        try:
            stop_final = min(polish_end - 4, time.monotonic() + 45.0)
            f_fin, sc_fin = _lp_optimize(best_seq, len(best_seq), stop_final, seed + 31337)
            if f_fin is not None and sc_fin < best_score - 1e-12:
                f_fin = np.clip(f_fin, 0, 1000)
                if f_fin.sum() > 0.01:
                    best_seq = f_fin
                    best_score = sc_fin
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    if final_sc >= best_score - 1e-10:
        best_score = final_sc

    return [float(x) for x in best_seq]
