    best = f.copy()
    best_sc = _sc(f)

        if time.monotonic() > deadline:
            break
        
        # Solve LP
        A = np.zeros((2*n-1, n+1))
        for j in range(n):
            A[j:j+n, j] = f
        A[:, n] = -1.0
        
        res = linprog(
            c=np.array([0]*n + [1.0]),
            A_ub=A, b_ub=np.zeros(2*n-1),
            A_eq=np.array([[1.0]*n + [0.0]]),
            b_eq=[1.0],
            bounds=[(0, None)]*n + [(0, None)],
            method='highs'
        )
        
        if res.status != 0:
            break
        
        g = res.x[:n]
        g = np.clip(g, 0, None)
        gs = g.sum()
        if gs < 1e-12:
            break
        g = g / gs  # ensure sum = 1 (should already be, but just in case)
        
        # Line search
        cf = _conv(f)
        cfs = _conv(f, g)
        cg = _conv(g)

        vals2 = np.array([

Actually, the softmax parameterization with a pinned element is a bit awkward. Let me use a simpler approach: parameterize all n elements with the standard softmax.

```python
def _smooth_opt(a0, beta, deadline, maxfun=2000):
    n = a0.size
    
    def obj_grad(y):
        a = _softmax(y)
        c = _conv(a)
        maxc = c.max()
        x = np.clip(beta*(c - maxc), -30, 0)
        w = np.exp(x)
        S = w.sum()
        fun = beta * maxc + math.log(S)
        d = w / S
        grad_a = 2*beta*_valid_conv(d, a)
        mg = (a*grad_a).sum()
        grad_y = a*(grad_a - mg)
        return fun, grad_y
    
    y0 = np.log(np.maximum(a0, 1e-300))
    y0 = y0 - y0.max()
    
    res = minimize(obj_grad, y0, jac=True, method='L-BFGS-B',
                  options={'maxiter': 200, 'maxfun': maxfun, 'ftol': 1e-14, 'gtol': 1e-10})
    
    if hasattr(res, 'x') and res.x is not None:
        y_sol = res.x
        a_new = _softmax(y_sol)
        return a_new
    return a0

def _softmax(y):
    y = np.asarray(y, dtype=np.float64)
    y = y - y.max()
    e = np.exp(y)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(y) / y.size
    return e / s
```
