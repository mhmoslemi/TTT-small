            a_l = np.clip((1 - best_t) * a_l + best_t * g_l, 0, None)
            sc_now = _score(a_l, L)
            if sc_now < best_score - 1e-13:
                best_a = a_l.copy()
                best_score = sc_now
                n = L
                stall = 0
            else:
                stall += 1

    # --- Phase 5: Final polish on best ---
    final_stop = deadline - 5.0
    if time.monotonic() < final_stop - 15.0:
        n_best = len(best_a)
        rng_final = np.random.default_rng(seed + 12345)
        best_a, sc_f1 = _coord_descent(best_a, n_best, final_stop - 8.0, rng_final, max_passes=50)
        if sc_f1 < best_score - 1e-14:
            best_score = sc_f1

        if time.monotonic() < final_stop - 10.0:
            best_a, sc_f2 = _gradient_polish(best_a, n_best, final_stop - 5.0, rng_final, max_rounds=500)
            if sc_f2 < best_score - 1e-14:
                best_score = sc_f2

    # --- Final validation ---
    best_a = np.clip(np.asarray(best_a, dtype=np.float64), 0, 1000)
    if best_a.size == 0 or best_a.sum() < 0.01:
        best_a = parent.astype(np.float64).copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_a]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Fallback to raw parent
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    return [float(x) for x in best_a]
