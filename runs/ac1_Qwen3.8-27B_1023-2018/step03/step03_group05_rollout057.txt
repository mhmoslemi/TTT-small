    c = irfft(f*f, nfft)[:2*n-1]

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}"""
    n = v.size
    if n == 0:
        return np.ones(1)
    v = np.where(np.isfinite(v), v, 0.0)
    # Standard simplex projection
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho = np.arange(1, n+1, dtype=np.float64)
    cond = (cv - target) / rho < mu
    idx = np.nonzero(cond)[0]
    if idx.size == 0:
        return np.full(n, target/n)
    theta = (cv[idx[-1]] - target) / (idx[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0 and np.isfinite(s):
        return (target / s) * w
    return np.full(n, target/n)

def _project_simplex(v, target=1.0):
    n = v.size
    if n == 0:
        return np.ones(1)
    v = np.where(np.isfinite(v), v, 0.0)
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho = np.arange(1, n+1, dtype=np.float64)
    cond = (cv - target) / rho < mu
    idx = np.nonzero(cond)[0]
    if idx.size == 0:
        return np.full(n, target/n)
    theta = (cv[idx[-1]] - target) / (idx[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0 and np.isfinite(s):
        return (target / s) * w
    return np.full(n, target/n)
