    """

        M = float(conv.max())

        # d conv[k]/da[i] = 2*a[k-i] if 0 <= k-i < n
        subgrad = np.zeros(n)

            for i in range(n):

                if 0 <= idx < n:

        subgrad /= len(active)

        # Gradient step with projection
        f_new = f - lr * subgrad
        f_new = np.clip(f_new, 0, None)

        if fs_new < 1e-10:

        f_new = f_new / fs_new
        sc = _score(f_new)
        if sc < best_sc - 1e-12:
            best_f = f_new.copy()
            best_sc = sc
            f = f_new
            lr = min(0.1, lr * 1.02)
        else:
            lr = max(0.005, lr * 0.7)

    return best_f, best_sc
