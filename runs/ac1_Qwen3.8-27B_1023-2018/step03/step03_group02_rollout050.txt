import time
import math
from scipy.fft import rfft, irfft, next_fast_len

    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:

    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf

def _build_A(f, n):

    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_step(f, n, M, time_limit=15.0, top_k=None):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0"""
    if top_k is not None:
        b = _conv_full(f)
        sorted_idx = np.argsort(-b)[:top_k]
        A_full = _build_A(f, n)
        A_sub = A_full[sorted_idx]
    else:
        A_sub = _build_A(f, n)
    
    c = -np.ones(n)
    b_ub = np.full(A_sub.shape[0], M)
    
    try:
        res = linprog(
            c, A_ub=A_sub, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum

def _polish(a, stop_time, seed=99):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 8
    
    while time.monotonic() < stop_time - 2.0:
        improved = False
        
        # Coordinate descent - evaluate full score
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            if orig > 1e-8:
                mults = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
                vals = orig * mults
            else:
                vals = np.array([0.01, 0.05, 0.1, 0.5, 1.0])
            vals = np.append(vals, 0.0)
            best_val = orig
            best_val_sc = best_sc
            for nv in vals:
                best[i] = nv
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-15:
                    best_val = nv
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-15:
                best_sc = best_val_sc
                improved = True
        
        # Pair updates
        n_pairs = min(80, n)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 2.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            total = oi + oj
            if total < 1e-12:
                continue
            for frac in (0.2, 0.4, 0.5, 0.6, 0.8):
                ni = frac * total
                nj = (1 - frac) * total
                best[i], best[j] = ni, nj
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                    break
            best[i], best[j] = oi, oj
        
        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0
    
    return best, best_sc

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

    best_score = _score(f)
    M = float(_conv_full(f).max())
    stall = 0
    max_stall = 100
    it = 0

    while time.monotonic() < stop_time - 3.0:
        # Choose LP variant
        if it % 5 == 0:
            g_dir = _lp_focused(f, n, M, top_k=100)
        else:
            g_dir = _lp_solve(f, n, M)

        if g_dir is None:
            stall += 4
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
            fs = f.sum()
            f = f / fs if fs > 1e-10 else best_f.copy()
            M = float(_conv_full(f).max())
            continue

        if float(np.linalg.norm(g_dir - f, ord=np.inf)) < 1e-11:
            stall += 4
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
            fs = f.sum()
            f = f / fs if fs > 1e-10 else best_f.copy()
            M = float(_conv_full(f).max())
            continue

        improved = False
        for tt in (1.0, 0.7, 0.5, 0.3, 0.15, 0.08):
            fn = (1.0 - tt) * f + tt * g_dir
            fn = np.clip(fn, 0, None)
            fsn = float(fn.sum())
            if fsn < 1e-10:
                continue
            fn = fn / fsn
            sc = _score(fn)
            if sc < best_score - 1e-14:
                f = fn
                best_f = f.copy()
                best_score = sc
                M = float(_conv_full(f).max())
                stall = 0
                improved = True
                break

        if not improved:
            stall += 1
            if stall >= max_stall:
                strat = rng.integers(5)
                if strat == 0:
                    f = np.clip(f + rng.normal(0, 0.12, n), 0, None)
                elif strat == 1:
                    f2 = f.copy()
                    f2 = (f2 + f2[::-1]) / 2.0
                    f = np.clip(f2 + rng.normal(0, 0.04, n), 0, None)
                elif strat == 2:
                    blk = max(2, n // 8)
                    f2 = f.copy()
                    for _ in range(4):
                        b1 = int(rng.integers(0, n - blk))
                        b2 = int(rng.integers(0, n - blk))
                        tmp = f2[b1:b1+blk].copy()
                        f2[b1:b1+blk] = f2[b2:b2+blk]
                        f2[b2:b2+blk] = tmp
                    f = f2
                elif strat == 3:
                    noise = rng.normal(0, 1, n)
                    kern = np.exp(-0.5 * (np.arange(-8, 9) / 3.0) ** 2)
                    kern /= kern.sum()
                    sm = np.convolve(noise, kern, 'same')
                    f = np.clip(f * (1.0 + 0.2 * sm), 0, None)
                else:
                    f = np.abs(f + rng.normal(0, 0.05, n)) + 0.001
                fs = f.sum()
                f = f / fs if fs > 1e-10 else best_f.copy()
                M = float(_conv_full(f).max())
                stall = 0
        it += 1

    return best_f, best_score


def _subgrad_polish(a, stop_time, seed=77):
    n = len(a)
    a = np.clip(a.copy(), 0, None)
    s = a.sum()
    if s < 0.01:
        return a, _score(a)
    a = a / s
    best_sc = _score(a)
    rng = np.random.default_rng(seed)
    lr = 0.05
    no_imp = 0

    while time.monotonic() < stop_time - 2.0 and no_imp < 30:
        b = _conv_full(a)
        mv = b.max()
        max_idx = np.where(b >= mv - 1e-12 * mv)[0]

        sg = np.zeros(n)
        for k in max_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                sg[i] += 2.0 * a[k - i]
        sg /= len(max_idx)
        sgn = np.linalg.norm(sg)
        if sgn < 1e-14:
            break

        improved = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05):
            lr_t = lr * mult
            a_new = np.clip(a - lr_t * sg / sgn, 0, None)
            sn = a_new.sum()
            if sn < 0.01:
                continue
            a_new = a_new / sn
            sc = _score(a_new)
            if sc < best_sc - 1e-15:
                a = a_new
                best_sc = sc
                improved = True
                no_imp = 0
                break
        if not improved:
            no_imp += 1
            lr = max(0.002, lr * 0.6)

    return a, best_sc


def _coord_polish(a, stop_time, seed=55):
    n = len(a)
    best = np.clip(a.copy(), 0, None)
    fs = best.sum()
    if fs < 0.01:
        return best, _score(best)
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no = 6

    while time.monotonic() < stop_time - 2.0 and no_imp < max_no:
        improved = False

        # Single-coordinate
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            if orig > 1e-9:
                tri_vals = orig * np.array([0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0, 0.0])
            else:
                tri_vals = np.array([0.05, 0.2, 0.5, 1.0, 2.0])
            bv, bvs = orig, best_sc
            for nv in tri_vals:
                best[i] = nv
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < bvs - 1e-15:
                    bv, bvs = nv, sc
            best[i] = bv
            if bvs < best_sc - 1e-15:
                best_sc = bvs
                improved = True

        # Pair moves
        for _ in range(min(60, n)):
            if time.monotonic() > stop_time - 2.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            tot = oi + oj
            if tot < 1e-12:
                continue
            for frac in (0.15, 0.3, 0.5, 0.7, 0.85):
                best[i], best[j] = frac * tot, (1 - frac) * tot
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                    break
            best[i], best[j] = oi, oj

        no_imp = 0 if improved else no_imp + 1

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    n = len(parent)

    fallback = parent.copy()
    try:
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(n) / n
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc
    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    # Build diverse initializations
    inits = []
    inits.append(parent.copy())

    # Cosine taper
    inits.append(0.5 + 0.5 * np.cos(np.pi * i_arr / (n - 1)))

    # Parent + smooth noise
    noise = rng_main.normal(0, 1, n)
    kern = np.exp(-0.5 * (np.arange(-10, 11) / 3.0) ** 2)
    kern /= kern.sum()
    inits.append(np.abs(parent + 0.12 * np.convolve(noise, kern, 'same')) + 0.005)

    # Symmetric
    half = n // 2
    hr = np.abs(rng_main.normal(0.5, 0.3, half)) + 0.02
    inits.append(np.concatenate([hr, hr[::-1]])[:n])

    # Multi-frequency
    sup = 0.2 + 0.1 * np.cos(4 * np.pi * i_arr / n) + 0.08 * np.cos(8 * np.pi * i_arr / n + 0.5)
    inits.append(np.abs(sup) + 0.01)

    # Zipper
    z = np.ones(n)
    z[::2] *= 1.6
    z[1::2] *= 0.5
    inits.append(z)

    # Phase 1: LP optimization (65% budget)
    phase1_end = t0 + budget * 0.65
    time_per = max(20.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        if best_score < 1.5028:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per)
        if time.monotonic() >= stop:
            break
        try:
            res, sc = _lp_optimize(init, n, stop, seed + 1009 * (idx + 1))
            if np.all(np.isfinite(res)) and res.sum() > 0.01:
                rc = np.clip(res, 0, 1000)
                ex = float(evaluate_sequence([float(x) for x in rc]))
                if math.isfinite(ex) and ex < best_score:
                    best_seq = rc.copy()
                    best_score = ex
        except Exception:
            pass

    # Phase 2: Subgradient + coordinate polish (20%)
    phase2_end = t0 + budget * 0.85
    if time.monotonic() < phase2_end - 10.0:
        for po in range(3):
            p_stop = min(time.monotonic() + (phase2_end - time.monotonic()) / 3, phase2_end - 2.0)
            if time.monotonic() >= p_stop - 1.0:
                break
            try:
                best_seq, best_score = _subgrad_polish(best_seq, p_stop, seed=2000 + po * 131)
            except Exception:
                pass
            try:
                best_seq, best_score = _coord_polish(best_seq, min(p_stop + 5, phase2_end - 2.0), seed=3000 + po * 131)
            except Exception:
                pass

    # Phase 3: Another LP round on best + final polish (15%)
    phase3_end = deadline - 5.0
    if time.monotonic() < phase3_end - 30.0:
        lp_stop = min(time.monotonic() + (phase3_end - time.monotonic()) * 0.5, phase3_end - 20.0)
        try:
            res, sc = _lp_optimize(best_seq, n, lp_stop, seed=77777)
            if np.all(np.isfinite(res)) and res.sum() > 0.01:
                rc = np.clip(res, 0, 1000)
                ex = float(evaluate_sequence([float(x) for x in rc]))
                if math.isfinite(ex) and ex < best_score:
                    best_seq = rc.copy()
                    best_score = ex
        except Exception:
            pass

    if time.monotonic() < phase3_end - 10.0:
        try:
            best_seq, best_score = _subgrad_polish(best_seq, phase3_end - 5.0, seed=88888)
        except Exception:
            pass
        try:
            best_seq, best_score = _coord_polish(best_seq, phase3_end - 3.0, seed=99999)
        except Exception:
            pass

    # Final validation
    final_seq = np.clip(best_seq, 0, 1000)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in final_seq]
