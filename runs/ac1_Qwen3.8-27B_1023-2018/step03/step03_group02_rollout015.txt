import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(200, dtype=np.float64)
    if p.size == 0:
        p = np.ones(200, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(200, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, n, M, time_limit=12.0, top_k=None):
    A = _build_A(f, n)
    if top_k is not None and top_k < 2 * n - 1:
        b = _conv_full(f)
        idx = np.argsort(-b)[:top_k]
        A_sub = A[idx]
        b_ub = np.full(top_k, M)
    else:
        A_sub = A
        b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum


def _subgrad_step(f, n, lr, iters=200):
    f = f.copy()
    s = f.sum()
    if s < 1e-10:
        return f
    f = f / s
    for _ in range(iters):
        b = _conv_full(f)
        k_star = int(b.argmax())
        # subgradient of b_k* w.r.t. f_i = f_{k*-i} for valid i
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        if i_min <= i_max:
            idx = np.arange(i_min, i_max + 1)
            sg = np.zeros(n)
            sg[idx] = f[k_star - idx]
            f = np.maximum(0, f - lr * sg / s)
        fs = f.sum()
        if fs < 1e-10:
            f = np.ones(n) / n
        else:
            f = f / fs
    return f

def _polish(a, stop_time, seed=99):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score_fast(best)
    best = best / fs
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 5

    while time.monotonic() < stop_time - 2.0:
        improved = False

        # 1D coordinate descent
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            if orig > 1e-8:
                mults = (0.3, 0.5, 0.7, 0.85, 1.1, 1.3, 1.5, 2.0, 0.0)
            else:
                mults = (0.0, 0.02, 0.05, 0.1, 0.5 / n)
            best_val = orig
            best_val_sc = best_sc
            for mlt in mults:
                nv = orig * mlt if orig > 1e-8 else mlt
                best[i] = nv
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score_fast(best)
                if sc < best_val_sc - 1e-15:
                    best_val = nv
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-15:
                best_sc = best_val_sc
                improved = True

        # 2-coordinate recombination
        n_pairs = min(60, n)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 2.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            total = oi + oj
            if total < 1e-12:
                continue
            for ratio in (0.25, 0.5, 0.75, 0.0, 1.0, 0.4, 0.6):
                best[i] = ratio * total
                best[j] = (1.0 - ratio) * total
                sc = _score_fast(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc
