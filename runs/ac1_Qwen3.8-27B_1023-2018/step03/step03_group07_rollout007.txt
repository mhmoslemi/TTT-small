    n = a.size
    a = a / a.sum()
    best_a = a.copy()
    best_M = np.convolve(a, a).max()
    
    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break
        
        conv = np.convolve(a, a)
        M = conv.max()
        k_star = conv.argmax()
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
        
        # Subgradient
        g = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                g[j] = 2 * a[idx]
        
        # Linear minimization oracle: put mass on argmin g
        j_min = np.argmin(g)
        v = np.zeros(n)
        v[j_min] = 1.0

        t_step = 2.0 / (it + 2)
        
        a_new = (1 - t_step) * a + t_step * v
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        a = a_new
    
    return best_a
