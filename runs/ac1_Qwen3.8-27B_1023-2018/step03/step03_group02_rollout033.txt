import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, n, M, time_limit=15.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum

        t_opt, m_opt = _line_search(f, g_dir, n, M, stop_time)

        if t_opt > 0 and m_opt < M - 1e-14:
            f = (1.0 - t_opt) * f + t_opt * g_dir
            f = np.clip(f, 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            M = float(_conv_full(f).max())
            if M < best_M:
                best_M = M
                best_f = f.copy()
            stall = 0
        else:
            stall += 1

        iter_count += 1

    # Final score
    best_sc = _score(best_f)
    return best_f, best_sc


def _polish(a, stop_time, seed=99):
    """Coordinate descent + multi-coordinate polish."""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel().copy(), 0, None)
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 8

    while time.monotonic() < stop_time - 2.0:
        improved = False

        # 1D coordinate descent
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            if orig > 1e-10:
                mults = (0.0, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0)
            else:
                mults = (0.0, 0.005, 0.01, 0.05, 0.1, 0.5, 1.0)
            best_val = orig
            best_val_sc = best_sc
            for mlt in mults:
                nv = orig * mlt if orig > 1e-10 else mlt
                best[i] = nv
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-15:
                    best_val = nv
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-15:
                best_sc = best_val_sc
                improved = True

        # 2-coordinate moves
        n_pairs = min(150, n * 3)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 2.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            total = oi + oj
            if total < 1e-12:
                best[i], best[j] = oi, oj
                continue
            for frac in (0.0, 0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9, 1.0):
                best[i] = frac * total
                best[j] = (1.0 - frac) * total
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj

        # 3-coordinate moves (fewer)
        n_triples = min(50, n)
        for _ in range(n_triples):
            if time.monotonic() > stop_time - 2.0:
                break
            idx = rng.choice(n, size=3, replace=False)
            i, j, k = int(idx[0]), int(idx[1]), int(idx[2])
            oi, oj, ok = best[i], best[j], best[k]
            total = oi + oj + ok
            if total < 1e-12:
                continue
            for fi in (0.0, 0.3, 0.5):
                for fj in (0.0, 0.3, 0.5):
                    fk = 1.0 - fi - fj
                    if fk < -1e-9:
                        continue
                    fk = max(0.0, fk)
                    s = fi + fj + fk
                    if s < 1e-9:
                        continue
                    fi, fj, fk = fi / s, fj / s, fk / s
                    best[i], best[j], best[k] = fi * total, fj * total, fk * total
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
            best[i], best[j], best[k] = oi, oj, ok

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc


def _block_polish(a, stop_time, seed=123):
    """Polish by blocking consecutive coordinates."""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel().copy(), 0, None)
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    block_sizes = [2, 3, 4, 5, 8]
    no_imp = 0

    while time.monotonic() < stop_time - 2.0 and no_imp < 6:
        improved = False
        for bs in block_sizes:
            n_blocks = n // bs
            for start in range(0, n - bs + 1, bs // 2):
                if time.monotonic() > stop_time - 2.0:
                    break
                if start + bs > n:
                    break
                block = best[start:start + bs].copy()
                total = block.sum()
                if total < 1e-12:
                    continue
                # Try a few random redistributions
                for _ in range(3):
                    w = rng.dirichlet(np.ones(bs) * 2.0)
                    best[start:start + bs] = w * total
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
                # Restore
                best[start:start + bs] = block
        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 25.0  # leave 25s headroom

    # Get parent
    try:
        h = globals()["height_sequence_1"]
        parent = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(384, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(384, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    if parent.size == 0:
        parent = np.ones(384, dtype=np.float64)

    n_p = len(parent)

    # Fallback
    fallback = parent.copy()
    try:
        fallback_sc = _score(fallback)
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(n_p) / n_p
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    # Phase allocation
    phase1_end = t0 + budget * 0.55   # LP optimization at parent length
    phase2_end = t0 + budget * 0.72   # Other lengths
    phase3_end = deadline - 5.0       # Polish

    # Phase 1: LP optimization at parent length
    n = n_p
    rng_main = np.random.default_rng(seed)

    # Initializations to try
    inits = []
    inits.append(("parent", parent.copy()))

    # Cosine taper
    i_arr = np.arange(n, dtype=np.float64)
    taper = 0.5 + 0.5 * np.cos(np.pi * i_arr / (n - 1))
    inits.append(("taper", taper))

    # Parent + structured noise
    kern = np.exp(-0.5 * (np.arange(-6, 7) / 2.5) ** 2)
    kern /= kern.sum()
    smooth_noise = np.convolve(rng_main.normal(0, 1, n), kern, 'same')
    p_noisy = np.abs(parent + 0.15 * smooth_noise)
    inits.append(("parent_noisy", p_noisy))

    # Multi-frequency
    sup = np.ones(n) * 0.1
    for fr in (1, 2, 3, 5, 7):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.3 * np.cos(2 * math.pi * fr * i_arr / n + ph)
    inits.append(("multifreq", np.abs(sup) + 0.001))

    # Zipper
    z = np.ones(n)
    z[::2] *= 1.5
    z[1::2] *= 0.6
    inits.append(("zipper", z))

    time_per_init = max(15.0, (phase1_end - t0) / len(inits))

    for idx, (name, init) in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        if best_score < 1.5028:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        if time.monotonic() >= stop:
            break
        try:
            res, sc = _lp_optimize(init, n, stop, seed + 7919 * (idx + 1),
                                   use_focused_every=(4 if idx > 0 else 0))
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_clipped = np.clip(res, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in res_clipped]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = res_clipped.copy()
                    best_score = exact
        except Exception:
            pass

    # Phase 2: Try other lengths
    other_lengths = sorted(set([max(2, n_p // 2), max(16, n_p * 3 // 4), n_p,
                                min(512, n_p + 64), min(512, n_p * 5 // 4)]))
    time_per_len = max(8.0, (phase2_end - phase1_end) / max(1, len(other_lengths)))

    for L in other_lengths:
        if time.monotonic() > phase2_end - 5.0:
            break
        if best_score < 1.5028:
            break
        stop = min(phase2_end - 3.0, time.monotonic() + time_per_len)
        if time.monotonic() >= stop:
            break

        # Initialize for this length
        if L == n_p:
            init_l = best_seq.copy() if len(best_seq) == L else parent.copy()
        elif len(best_seq) == L and best_score < 2.0:
            init_l = best_seq.copy()
        else:
            # Interpolate parent to length L
            if len(parent) > 1:
                init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(parent)), parent)
            else:
                init_l = np.ones(L)
        
        try:
            res, sc = _lp_optimize(init_l, L, stop, seed + 104729 + L)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_clipped = np.clip(res, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in res_clipped]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = res_clipped.copy()
                    best_score = exact
        except Exception:
            pass

    # Phase 3: Multi-pass polish
    if time.monotonic() < phase3_end - 10.0:
        polish_rounds = 4
        polish_time_each = max(10.0, (phase3_end - time.monotonic()) / polish_rounds)
        for pidx in range(polish_rounds):
            p_stop = time.monotonic() + polish_time_each
            if p_stop > phase3_end:
                p_stop = phase3_end
            if time.monotonic() >= p_stop - 2.0:
                break
            try:
                if pidx % 2 == 0:
                    best_seq, best_score = _polish(best_seq, p_stop, seed=5000 + pidx * 111)
                else:
                    best_seq, best_score = _block_polish(best_seq, p_stop, seed=8000 + pidx * 200)
            except Exception:
                pass

    # Final: one more short coordinate descent
    final_stop = t0 + budget - 8.0
    if time.monotonic() < final_stop - 4.0:
        try:
            best_seq, best_score = _polish(best_seq, final_stop, seed=77777)
        except Exception:
            pass

    # Final validation with exact evaluator
    final_seq = np.clip(best_seq, 0, 1000)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    if final_sc < best_score:
        best_score = final_sc

    return [float(x) for x in final_seq]
