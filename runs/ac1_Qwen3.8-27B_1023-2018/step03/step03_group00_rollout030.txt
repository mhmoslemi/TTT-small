    b0 = _conv(a0)
    tau = 0.3 * max(b0.max() - b0.min(), 1e-8)
    if tau < 1e-8:
        tau = 1e-4

    def obj_grad(x):
        f, g = _smooth_obj_grad(x, tau, n)
        if not math.isfinite(f):
            return 1e10, np.zeros(n)
        return f, np.clip(g, -1e6, 1e6)

    try:
        res = minimize(obj_grad, x0, jac=True, method='L-BFGS-B',
                       options={'maxiter': max_iters, 'gtol': 1e-7, 'maxls': 30})
        x_best = res.x
        a_new = np.exp(x_best - x_best.max())
        a_new = a_new / a_new.sum()
        a_new = S0 * a_new
    except Exception:
        a_new = a.copy()

    sc_new = _score(a_new)
    sc_old = _score(a)
    if sc_new < sc_old:
        return a_new, sc_new
    return a, sc_old


def _lp_optimize(f0, n, stop_time, seed):
    """Main LP-based optimization loop with gradient refinement."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    if f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    if f.sum() < 1e-10:
        f = np.ones(n)

    best_f = f.copy()
    best_score = _score(f)
    iteration = 0
    grad_counter = 0
    no_improve = 0
    max_no_improve = 40

    while time.monotonic() < stop_time - 10.0:
        iteration += 1

        b = _conv(f)
        M = float(b.max())
        if M < 1e-18:
            break

        # Get LP direction
        g = _lp_direction(f, n, M, timeout=6.0)

        if g is not None:
            g_sum = float(g.sum())
            if g_sum > 1e-15:
                g_norm = g * (f.sum() / g_sum)
                dist = float(np.linalg.norm(g_norm - f) / max(1.0, np.linalg.norm(f)))

                if dist > 1e-10:
                    a_new, sc_new = _line_search_best(f, g_norm, n, best_score, stop_time)
                    if a_new is not None and sc_new < best_score - 1e-13:
                        f = a_new
                        best_f = f.copy()
                        best_score = sc_new
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 2

        # Periodic gradient refinement every 8 iterations
        if iteration % 8 == 0 and time.monotonic() < stop_time - 20.0:
            f, sc = _gradient_refine(f, n, stop_time, max_iters=25)
            if sc < best_score - 1e-13:
                best_f = f.copy()
                best_score = sc
                no_improve = 0

        # If stuck, perturb
        if no_improve >= max_no_improve:
            noise_scale = 0.05 * max(1.0, f.mean())
            f = np.maximum(f + rng.normal(0, noise_scale, n), 0)
            if f.sum() > 1e-10:
                f = f / f.sum() * best_f.sum()
            no_improve = 0
            max_no_improve += 20

        # Additional LP with slightly perturbed M
        if iteration % 15 == 0 and time.monotonic() < stop_time - 15.0:
            for M_mult in (0.92, 1.05):
                g2 = _lp_direction(f, n, M * M_mult, timeout=4.0)
                if g2 is not None and g2.sum() > 1e-15:
                    g2n = g2 * (f.sum() / g2.sum())
                    a_new2, sc2 = _line_search_best(f, g2n, n, best_score, stop_time)
                    if a_new2 is not None and sc2 < best_score - 1e-13:
                        f = a_new2
                        best_f = f.copy()
                        best_score = sc2
                        no_improve = 0

    return best_f, best_score


def _coordinate_polish(a, stop_time, seed=99):
    """Coordinate descent + random perturbations."""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    S = best.sum()
    if S < 1e-10:
        return best, _score(best)
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no = 15

    while time.monotonic() < stop_time - 5.0:
        improved = False

        # Single coordinate sweep
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break
            orig = best[i]
            if orig < 1e-14:
                continue
            mults = (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0, 0.0)
            best_i_val = orig
            best_i_sc = best_sc
            for m in mults:
                best[i] = orig * m
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_i_sc - 1e-15:
                    best_i_val = best[i]
                    best_i_sc = sc
            best[i] = best_i_val
            if best_i_sc < best_sc - 1e-15:
                best_sc = best_i_sc
                improved = True

        # Pair coordinate
        for _ in range(200):
            if time.monotonic() > stop_time - 5.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            if oi < 1e-14 and oj < 1e-14:
                continue
            for mi in (0.4, 0.7, 1.4, 2.0, 0.0):
                for mj in (0.4, 0.7, 1.4, 2.0, 0.0):
                    if mi == 1.0 and mj == 1.0:
                        continue
                    best[i] = oi * mi if oi > 1e-14 else 0.0
                    best[j] = oj * mj if oj > 1e-14 else 0.0
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        # Small random perturbations
        if not improved:
            for _ in range(50):
                if time.monotonic() > stop_time - 5.0:
                    break
                k = int(rng.integers(n))
                old = best[k]
                best[k] = max(0.0, old + rng.normal(0, 0.03 * max(1.0, best.mean())))
                if best.sum() < 0.01:
                    best[k] = old
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                else:
                    best[k] = old

        if not improved:
            no_improve += 1
            if no_improve >= max_no:
                break
        else:
            no_improve = 0

    return best, best_sc
