import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _fft_size(n):
    return next_fast_len(2 * n - 1)

def _conv(a):
    n = len(a)
    fs = _fft_size(n)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A_rows(f, k_indices):
    n = len(f)
    A = np.zeros((len(k_indices), n))
    for ii, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    return A

def _lp_dir(f, n, M, top_k=None, tlimit=4.0):
    if top_k is None or top_k >= 2 * n - 1:
        b = _conv(f)
        idx = np.argsort(-b)[:top_k] if top_k else np.arange(2 * n - 1)
    else:
        b = _conv(f)
        idx = np.argsort(-b)[:top_k]
    A = _build_A_rows(f, idx)
    c = -np.ones(n)
    b_ub = np.full(len(idx), M)
    bounds = [(0.0, None)] * n
    try:

                      method='highs',
                      options={'presolve': True, 'time_limit': tlimit})
    except Exception:

    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-12:

    return g / gs

def _golden_search(f, g, n, max_iter=20):
    """Find t in [0, 1.5] minimizing score along (1-t)*f + t*g."""
    phi = (1.0 + math.sqrt(5.0)) / 2.0
    a, b = 0.0, 1.5
    c_ = b - (b - a) / phi
    d_ = a + (b - a) / phi
    fs = f.sum()
    s = 1.0  # normalized
    # Evaluate score at intermediate points
    def sc(t):
        a_new = (1.0 - t) * f + t * g
        a_new = np.clip(a_new, 0, None)
        ss = a_new.sum()
        if ss < 1e-10:
            return math.inf
        a_n = a_new / ss
        return _score(a_n)
    fc_ = sc(c_)
    fd_ = sc(d_)
    for _ in range(max_iter):
        if fc_ < fd_:
            b = d_
            d_ = c_
            fd_ = fc_
            c_ = b - (b - a) / phi
            fc_ = sc(c_)
        else:
            a = c_
            c_ = d_
            fc_ = fd_
            d_ = a + (b - a) / phi
            fd_ = sc(d_)
    t_best = (a + b) / 2.0
    return t_best, min(fc_, fd_)

def _fw_step(f, n):
    """One Frank-Wolfe step. Returns new f."""
    b = _conv(f)
    k_star = int(np.argmax(b))
    # subgradient: s[m] = 2*f[k_star - m] if valid, else 0
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    s_grad = np.zeros(n)
    # s_grad[m] = 2*f[k_star - m] for m in [lo, hi]
    s_grad[lo:hi + 1] = 2.0 * f[k_star - hi:k_star - lo + 1][::-1]
    # FW: min s_grad^T v over simplex -> v = e[argmin s_grad]
    e_idx = int(np.argmin(s_grad))
    return e_idx, k_star

def _polish_coord(a, n, stop_time, seed):
    """Coordinate descent polish."""
    rng = np.random.default_rng(seed)
    best = np.clip(a.astype(np.float64).copy(), 0, None)
    bs = best.sum()
    if bs < 1e-10:
        return best, math.inf
    best = best / bs

    no_improve = 0
    max_no = 6
    while time.monotonic() < stop_time - 1.0 and no_improve < max_no:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            best_sc_i = best_sc
            best_val = orig
            for mlt in (0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 0.0, 0.55, 1.4):
                nv = orig * mlt if orig > 1e-8 else (mlt * 0.01 if mlt > 0 else 0.0)
                best[i] = nv
                ss = best.sum()
                if ss < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_sc_i - 1e-15:
                    best_sc_i = sc
                    best_val = nv
            best[i] = best_val
            if best_sc_i < best_sc - 1e-15:
                best_sc = best_sc_i
                improved = True
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    return best, best_sc
