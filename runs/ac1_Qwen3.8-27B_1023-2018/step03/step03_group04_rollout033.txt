    while time.monotonic() < stop_time - 1.0:
        improved = False

        # 1-coordinate moves (shuffled order)
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                return best, best_sc
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc
            for mlt in multipliers:
                new_val = orig * mlt if orig > 1e-14 else (0.5 if mlt > 0 else 0.0)
                best[i] = new_val
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-14:
                    best_val = new_val
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True

        # 2-coordinate moves
        n_2coord = max(40, n // 4)
        for _ in range(n_2coord):
            if time.monotonic() > stop_time - 1.5:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.0, 0.3, 0.5, 0.7, 1.3, 1.5, 2.0):
                for mj in (0.0, 0.3, 0.5, 0.7, 1.3, 1.5, 2.0):
                    best[i] = oi * mi if oi > 1e-14 else (0.5 if mi > 0 else 0.0)
                    best[j] = oj * mj if oj > 1e-14 else (0.5 if mj > 0 else 0.0)
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        # 3-coordinate moves (fewer, more expensive)
        n_3coord = max(10, n // 16)
        for _ in range(n_3coord):
            if time.monotonic() > stop_time - 2.0:
                break
            i, j, k = int(rng.integers(n)), int(rng.integers(n)), int(rng.integers(n))
            if i == j or j == k or i == k:
                continue
            oi, oj, ok = best[i], best[j], best[k]
            for mi in (0.5, 1.5, 0.0):
                for mj in (0.5, 1.5, 0.0):
                    for mk in (0.5, 1.5, 0.0):
                        best[i] = oi * mi if oi > 1e-14 else (0.3 if mi > 0 else 0.0)
                        best[j] = oj * mj if oj > 1e-14 else (0.3 if mj > 0 else 0.0)
                        best[k] = ok * mk if ok > 1e-14 else (0.3 if mk > 0 else 0.0)
                        if best.sum() < 0.01:
                            best[i], best[j], best[k] = oi, oj, ok
                            continue
                        sc = _score(best)
                        if sc < best_sc - 1e-14:
                            best_sc = sc
                            improved = True
            best[i], best[j], best[k] = oi, oj, ok

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return best, best_sc


def _gradient_refine(f, stop_time, seed=99):
    """Subgradient-based refinement for the max-convolution term."""
    n = len(f)
    f = np.clip(f.copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        return f, math.inf
    f = f / fs
    best_f = f.copy()
    best_sc = _score(f)
    rng = np.random.default_rng(seed)

    lr = 0.08
    for it in range(500):
        if time.monotonic() > stop_time - 2.0:
            break
        conv = _conv_fft(f)
        M = float(conv.max())
        thresh = 0.9 * M
        active = np.where(conv >= thresh)[0]
        if len(active) == 0:
            break

        # Compute subgradient at active points
        subgrad = np.zeros(n)
        for k in active:
            for i in range(n):
                d = abs(k - i)
                if d < n:
                    subgrad[i] += f[d]
        subgrad /= len(active)

        # Normalize and take step
        f_new = f - lr * subgrad
        f_new = np.clip(f_new, 0, None)
        fs_new = f_new.sum()
        if fs_new < 1e-10:
            break
        f_new = f_new / fs_new
        sc = _score(f_new)
        if sc < best_sc - 1e-13:
            best_f = f_new.copy()
            best_sc = sc
            f = f_new
            lr = min(0.15, lr * 1.02)
        else:
            lr = max(0.005, lr * 0.7)

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 20.0  # Leave 20s headroom

    parent = _get_parent()
    n_p = len(parent)

    # Fallback
    fallback = parent.copy()
    try:
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(384)
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _track(f):
        nonlocal best_seq, best_score
        if f is None or len(f) == 0:
            return
        f = np.clip(np.asarray(f, dtype=np.float64), 0, 1000)
        if f.sum() < 0.01 or not np.all(np.isfinite(f)):
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in f]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score:
            best_seq = f.copy()
            best_score = exact

    def _run_lp(init, stop, sd):
        try:
            res, sc = _lp_optimize(init, stop, sd)
            _track(res)
        except Exception:
            pass

    # Phase 1: Main LP optimization from parent (45% of budget)
    phase1_end = t0 + budget * 0.45
    n = n_p

    # Start directly from parent (normalized)
    init = parent / parent.sum()
    _run_lp(init, phase1_end, seed)

    # Phase 2: Continue LP from best + try perturbed restarts (25%)
    phase2_end = t0 + budget * 0.70
    rng_main = np.random.default_rng(seed + 77)

    # Try a few perturbed versions
    for attempt in range(4):
        if time.monotonic() > phase2_end - 5.0:
            break
        pert = np.clip(best_seq / best_seq.sum() + rng_main.normal(0, 0.015, len(best_seq)), 0, None)
        ps = pert.sum()
        if ps < 1e-10:
            continue
        pert = pert / ps
        alloc = (phase2_end - time.monotonic()) / max(1, 4 - attempt)
        stop = min(phase2_end - 3.0, time.monotonic() + alloc)
        if time.monotonic() < stop:
            _run_lp(pert, stop, seed + 10000 + attempt * 1000)

    # Phase 3: Gradient refinement (10%)
    phase3_end = t0 + budget * 0.80
    if time.monotonic() < phase3_end - 10.0:
        try:
            ref_f, ref_sc = _gradient_refine(best_seq, phase3_end - 2.0, seed + 555)
            _track(ref_f)
        except Exception:
            pass

    # Phase 4: Polish (15%)
    phase4_end = deadline - 25.0
    if time.monotonic() < phase4_end - 15.0:
        try:
            polished, pol_sc = _polish(best_seq, phase4_end - 5.0, seed + 111)
            _track(polished)
        except Exception:
            pass

    # Phase 5: Another LP round from polished result (5%)
    if time.monotonic() < deadline - 30.0:
        _run_lp(best_seq / best_seq.sum() if best_seq.sum() > 1e-10 else best_seq,
                deadline - 25.0, seed + 99999)

    # Final polish
    if time.monotonic() < deadline - 15.0:
        try:
            polished, pol_sc = _polish(best_seq, deadline - 10.0, seed + 222)
            _track(polished)
        except Exception:
            pass

    # Final gradient refinement
    if time.monotonic() < deadline - 10.0:
        try:
            ref_f, ref_sc = _gradient_refine(best_seq, deadline - 5.0, seed + 333)
            _track(ref_f)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
