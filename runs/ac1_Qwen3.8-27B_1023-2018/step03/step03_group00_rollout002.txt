def _normalize_prob(a):
    try:
        a = np.asarray(a, dtype=np.float64).ravel()

        return None
    if a.size == 0:
        return None

def _conv_full(a):

        return math.inf
    t = float(q.max())
    if not math.isfinite(t) or t < 0.0:
        t = 0.0
    return float(2.0 * n * t)


def _interpolate(a, L):
    L = int(max(1, L))
    p = _normalize_prob(a)
    if p is None:
        return np.full(L, 1.0 / L)
    n = p.size
    if n == L:
        return p.copy()
    if n == 1:
        return np.full(L, float(p[0]))
    xo = np.arange(n, dtype=np.float64)
    xn = np.linspace(0.0, n - 1.0, L)
    return np.interp(xn, xo, p)


def _smooth_prob(p, frac=0.05):
    p = _normalize_prob(p)
    if p is None:
        return p
    n = p.size
    if n < 12:
        return p
    try:
        f = rfft(p)
        freqs = np.fft.rfftfreq(n) * n
        sigma = max(1.0, n * frac)
        mask = np.exp(-(freqs * freqs) / (2.0 * sigma * sigma))
        g = irfft(f * mask, n=n)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.clip(g, 0.0, None)
        s = float(g.sum())
        if s <= 1e-12:
            return p
        return g / s
    except Exception:
        return p


def _solve_linprog(c, A, b_ub, A_eq=None, b_eq=None, bounds=None, stop=None):
    opts = {}
    if stop is not None:
        rem = max(0.05, float(stop) - time.monotonic())
        opts["time_limit"] = float(min(3.0, max(0.25, rem * 0.25)))

    for meth in ("highs", "interior-point"):
        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method=meth,
                options=opts,
            )
            if res is None or res.x is None:
                continue
            x = np.asarray(res.x, dtype=np.float64)
            if np.all(np.isfinite(x)):
                return x, getattr(res, "status", 0)
        except TypeError:
            try:
                res = linprog(
                    c,
                    A_ub=A,
                    b_ub=b_ub,
                    A_eq=A_eq,
                    b_eq=b_eq,
                    bounds=bounds,
                    method=meth,
                )
                if res is None or res.x is None:
                    continue
                x = np.asarray(res.x, dtype=np.float64)
                if np.all(np.isfinite(x)):
                    return x, getattr(res, "status", 0)
            except Exception:
                pass
        except Exception:
            pass
    return None, None


def _active_indices(q, M, max_active=256):
    q = np.asarray(q, dtype=np.float64)
    nq = q.size
    if nq <= max_active:
        return np.arange(nq, dtype=int)
    gap = max(1e-12, 0.05 * max(0.0, M))
    close = np.flatnonzero(q >= M - gap)
    if close.size <= max_active:
        return close
    top = np.argpartition(q, -max_active)[-max_active:]
    idx = np.unique(np.concatenate((top, close)))
    if idx.size > max_active:
        part = np.argpartition(q[idx], -max_active)[-max_active:]
        idx = idx[part]
    return idx


def _lp_cheby(p, stop_time, active_cap=64):
    p = _normalize_prob(p)
    if p is None:
        return None, None
    n = p.size
    q = _conv_full(p)
    qsize = q.size
    if qsize == 0:
        return None, None
    M = float(q.max())
    if not math.isfinite(M) or M <= 1e-15:
        return None, None

    use_full = False
    if active_cap == 0:
        if (qsize * (n + 1)) <= 1500000:
            use_full = True
        else:
            active_cap = 512

    slack = 1e-10
    invM = 1.0 / M

    if use_full:
        A = np.zeros((qsize, n + 1), dtype=np.float64)
        ps = p * (2.0 * invM)
        for i in range(n):
            A[i : i + n, i] = ps
        A[:, n] = 1.0
        b = 1.0 + q * invM + slack
    else:
        idx = _active_indices(q, M, int(max_active_safe := min(max(8, active_cap), 512)))
        m = idx.size
        A = np.zeros((m, n + 1), dtype=np.float64)
        ps = p * (2.0 * invM)
        for r, k in enumerate(idx):
            A[r, n] = 1.0
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            start = k - hi
            stop = k - lo + 1
            A[r, lo : hi + 1] = ps[start:stop][::-1]
        b = 1.0 + q[idx] * invM + slack

    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0

    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    bounds = [(0.0, 1.0)] * n + [(-2.0 - 1e-9, 0.0)]

    v, _ = _solve_linprog(c, A, b, A_eq=Aeq, b_eq=b_eq, bounds=bounds, stop=stop_time)
    if v is None:
        return None, None

    x = v[:n]
    nu = float(v[n])
    if not np.all(np.isfinite(x)):
        return None, None

    x = np.clip(x, 0.0, 1.0)
    s = float(x.sum())
    if s <= 1e-12:
        return None, None
    x = x / s

    if nu > 0.0:
        nu = 0.0
    mu = nu * M
    return x, mu


def _lp_sum(p, stop_time, active_cap=64):
    p = _normalize_prob(p)
    if p is None:
        return None, None
    n = p.size
    q = _conv_full(p)
    qsize = q.size
    if qsize == 0:
        return None, None
    M = float(q.max())
    if not math.isfinite(M) or M <= 1e-15:
        return None, None

    use_full = False
    if active_cap == 0:
        if (qsize * n) <= 2000000:
            use_full = True
        else:
            active_cap = 512

    slack = 1e-10
    invM = 1.0 / M
    ps = p * invM

    if use_full:
        A = np.zeros((qsize, n), dtype=np.float64)
        for i in range(n):
            A[i : i + n, i] = ps
        b = np.full(qsize, 1.0 + slack, dtype=np.float64)
    else:
        idx = _active_indices(q, M, int(min(max(8, active_cap), 512)))
        m = idx.size
        A = np.zeros((m, n), dtype=np.float64)
        for r, k in enumerate(idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            start = k - hi
            stop = k - lo + 1
            A[r, lo : hi + 1] = ps[start:stop][::-1]
        b = np.full(m, 1.0 + slack, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, 1.0)] * n

    x, _ = _solve_linprog(c, A, b, A_eq=None, b_eq=None, bounds=bounds, stop=stop_time)
    if x is None:
        return None, None
    x = np.asarray(x, dtype=np.float64)
    if not np.all(np.isfinite(x)):
        return None, None
    x = np.clip(x, 0.0, 1.0)
    s = float(x.sum())
    if s <= 1e-12:
        return None, None
    return x / s, s


def _line_search(p, x, current_sc, stop_time):
    x = _normalize_prob(x)
    if x is None:
        return None, None
    p = _normalize_prob(p)
    if p is None or x.size != p.size:
        return None, None

    diff = float(np.linalg.norm(x - p))
    if diff <= 1e-10:
        return None, None

    best_x = None
    best_sc = current_sc
    ts = (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 0.02, 0.08, 0.4, 0.6, 0.8, 0.95, 1.0)

    for t in ts:
        if time.monotonic() > stop_time:
            break
        pn = p + float(t) * (x - p)
        if np.any(pn < 0.0):
            pn = np.clip(pn, 0.0, None)
        s = float(pn.sum())
        if s <= 1e-12:
            continue
        pn = pn / s
        sc = _score_array(pn)
        if math.isfinite(sc) and sc < best_sc - 1e-15:
            best_sc = sc
            best_x = pn.copy()

    if best_x is None:
        return None, None
    return best_x, best_sc


def _lbfgsb_run(p0, stop_time, seed=0, maxfun=None, active_K=3):
    p = _normalize_prob(p0)
    if p is None:
        return p0, _score_array(p0)
    n = p.size
    if n == 0:
        return p, math.inf
    if n > 2500:
        return p, _score_array(p)
    if maxfun is None:
        maxfun = 1200 if n <= 512 else (700 if n <= 1024 else 350)
    if time.monotonic() > stop_time - 10.0:
        return p, _score_array(p)

    logp = np.log(np.maximum(p, 1e-15))
    lm = float(logp.max())
    z0 = logp - (lm + float(np.log(np.sum(np.exp(logp - lm)))))

    q0 = _conv_full(p)
    T0 = float(q0.max()) if q0.size else 0.0
    if not math.isfinite(T0) or T0 < 0.0:
        T0 = 0.0

    state = {"T": T0, "p": p.copy()}

    def softmax(z):
        z = np.asarray(z, dtype=np.float64)
        if z.size != n or not np.all(np.isfinite(z)):
            return None
        z = z - float(z.max())
        e = np.exp(z)
        s = float(e.sum())
        if not math.isfinite(s) or s <= 0.0:
            return None
        return e / s

    def fun(z):
        pz = softmax(z)
        if pz is None:
            return 1e100
        q = _conv_full(pz)
        T = float(q.max()) if q.size else 1e100
        if math.isfinite(T) and T < state["T"] - 1e-15:
            state["T"] = T
            state["p"] = pz.copy()
        if not math.isfinite(T):
            return 1e100
        return T

    def grad(z):
        pz = softmax(z)
        if pz is None:
            return np.zeros(n, dtype=np.float64)
        q = _conv_full(pz)
        if q.size == 0:
            return np.zeros(n, dtype=np.float64)
        M = float(q.max())
        if not math.isfinite(M) or M <= 0.0:
            return np.zeros(n, dtype=np.float64)

        K = int(max(1, min(active_K, q.size)))
        if K == 1 or q.size == 1:
            idx = np.array([int(np.argmax(q))], dtype=int)
        else:
            part = np.argpartition(q, -K)[-K:]
            idx = part[np.argsort(q[part])[-K:]]

        gp = np.zeros(n, dtype=np.float64)
        cnt = max(1, idx.size)
        for k in idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            start = k - hi
            stop = k - lo + 1
            gp[lo : hi + 1] += 2.0 * pz[start:stop][::-1] / cnt

        dot = float(np.dot(gp, pz))
        g = pz * (gp - dot)
        if not np.all(np.isfinite(g)):
            return np.zeros(n, dtype=np.float64)
        return g

    try:
        res = minimize(
            fun,
            z0,
            jac=grad,
            method="L-BFGS-B",
            options={
                "maxiter": 250,
                "maxfun": int(maxfun),
                "ftol": 1e-15,
                "gtol": 1e-12,
                "maxls": 25,
            },
        )
    except Exception:
        res = None

    if res is not None and getattr(res, "x", None) is not None:
        pz = softmax(np.asarray(res.x, dtype=np.float64))
        if pz is not None:
            q = _conv_full(pz)
            T = float(q.max()) if q.size else math.inf
            if math.isfinite(T) and T < state["T"] - 1e-15:
                state["T"] = T
                state["p"] = pz.copy()

    best_p = state["p"]
    T = float(state["T"])
    if not math.isfinite(T) or T < 0.0:
        T = T0
    best_sc = float(2.0 * n * max(0.0, T))
    return best_p, best_sc


def _coordinate_polish(p0, stop_time, seed=1, max_passes=50, do_pairwise=True):
    p = _normalize_prob(p0)
    if p is None:
        return p0, math.inf
    n = p.size
    if n < 2:
        return p, _score_array(p)

    q = _conv_full(p)
    T = float(q.max()) if q.size else 0.0
    if not math.isfinite(T) or T < 0.0:
        T = 0.0
    best_sc = float(2.0 * n * T)
    best_p = p.copy()

    rng = np.random.default_rng(seed)
    multipliers = (0.0, 0.25, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.6, 2.0, 3.0)
    no_imp = 0

    for _pass in range(int(max_passes)):
        if time.monotonic() > stop_time:
            break

        q = _conv_full(p)
        T = float(q.max()) if q.size else 0.0
        if not math.isfinite(T) or T < 0.0:
            T = 0.0
        cur_sc = float(2.0 * n * T)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_p = p.copy()

        improved_pass = 0
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time:
                break
            old = float(p[i])
            mean = float(p.mean()) if n else 0.0
            if old > 1e-14:
                vals = (old * m for m in multipliers)
            else:
                vals = (0.0, mean * 1e-5, mean * 1e-4, mean * 1e-3, mean * 1e-2, mean * 0.05)

            accepted = False
            for new in vals:
                if not math.isfinite(new) or new < 0.0:
                    continue
                if new > 1000.0:
                    continue
                delta = float(new) - old
                if abs(delta) <= 1e-16 * max(1.0, abs(old)):
                    continue

                q2 = q.copy()
                q2[i : i + n] += 2.0 * delta * p
                idx2 = 2 * i
                if idx2 < q2.size:
                    q2[idx2] += delta * delta

                s2 = float(p.sum() + delta)
                if s2 <= 1e-12:
                    continue

                m2 = float(q2.max())
                if not math.isfinite(m2) or m2 < 0.0:
                    m2 = 0.0
                sc2 = float(2.0 * n * (m2 / (s2 * s2)))

                if sc2 < best_sc - 1e-15:
                    p[i] = float(new)
                    inv = 1.0 / s2
                    p *= inv
                    q = q2 * (inv * inv)
                    best_sc = sc2
                    best_p = p.copy()
                    improved_pass += 1
                    accepted = True
                    break
            if accepted:
                continue

        if do_pairwise and n <= 900 and time.monotonic() < stop_time:
            for _trial in range(80):
                if time.monotonic() > stop_time:
                    break
                pair = rng.choice(n, size=2, replace=False)
                i = int(pair[0])
                j = int(pair[1])
                oi = float(p[i])
                oj = float(p[j])
                mean = float(p.mean())

                if oi > 1e-14:
                    vi = (0.0, 0.5 * oi, 0.8 * oi, 1.2 * oi, 1.5 * oi, 2.0 * oi)
                else:
                    vi = (0.0, mean * 1e-4, mean * 1e-3, mean * 1e-2)
                if oj > 1e-14:
                    vj = (0.0, 0.5 * oj, 0.8 * oj, 1.2 * oj, 1.5 * oj, 2.0 * oj)
                else:
                    vj = (0.0, mean * 1e-4, mean * 1e-3, mean * 1e-2)

                found = False
                for ni in vi:
                    if found:
                        break
                    for nj in vj:
                        if not math.isfinite(ni) or not math.isfinite(nj):
                            continue
                        if ni < 0.0 or nj < 0.0:
                            continue
                        di = float(ni) - oi
                        dj = float(nj) - oj
                        if abs(di) + abs(dj) < 1e-16:
                            continue

                        q3 = q.copy()
                        q3[i : i + n] += 2.0 * di * p
                        if 2 * i < q3.size:
                            q3[2 * i] += di * di
                        q3[j : j + n] += 2.0 * dj * p
                        if 2 * j < q3.size:
                            q3[2 * j] += dj * dj
                        cross = i + j
                        if cross < q3.size:
                            q3[cross] += 2.0 * di * dj

                        s3 = float(p.sum() + di + dj)
                        if s3 <= 1e-12:
                            continue

                        m3 = float(q3.max())
                        if not math.isfinite(m3) or m3 < 0.0:
                            m3 = 0.0
                        sc3 = float(2.0 * n * (m3 / (s3 * s3)))

                        if sc3 < best_sc - 1e-15:
                            p[i] = float(ni)
                            p[j] = float(nj)
                            inv = 1.0 / s3
                            p *= inv
                            q = q3 * (inv * inv)
                            best_sc = sc3
                            best_p = p.copy()
                            improved_pass += 1
                            found = True
                            break

        if improved_pass == 0:
            no_imp += 1
            if no_imp >= 3:
                break
        else:
            no_imp = 0

    return best_p, best_sc


def _lp_search(p0, n, stop_time, seed=0):
    n = int(max(1, n))
    p = _normalize_prob(p0)
    if p is None or p.size != n:
        p = _interpolate(p0 if p0 is not None else np.ones(n), n)
    if p is None:
        p = np.full(n, 1.0 / n)

    best_p = p.copy()
    best_sc = _score_array(p)
    rng = np.random.default_rng(seed)
    stall = 0
    it = 0
    n = int(p.size)

    if n <= 1000 and stop_time - time.monotonic() > 35.0:
        cst = min(stop_time - 25.0, time.monotonic() + 25.0)
        if cst > time.monotonic() + 5.0:
            p2, sc2 = _coordinate_polish(
                p, cst, seed=seed + 999, max_passes=2, do_pairwise=(n <= 800)
            )
            if sc2 < best_sc - 1e-15:
                best_p = p2.copy()
                best_sc = sc2
                p = p2.copy()

    while time.monotonic() < stop_time - 5.0:
        it += 1

        if not np.all(np.isfinite(p)):
            p = best_p.copy()
        p = np.clip(p, 0.0, None)
        s = float(p.sum())
        if s <= 1e-12:
            p = best_p.copy()
            s = float(best_p.sum())
        p = p / s

        sc = _score_array(p)
        if math.isfinite(sc) and sc < best_sc - 1e-15:
            best_p = p.copy()
            best_sc = sc

        if sc > best_sc + max(1e-5, 0.05 * max(0.1, best_sc)):
            p = best_p.copy()
            sc = best_sc

        improved = False

        if time.monotonic() < stop_time - 12.0:
            if n <= 800:
                caps = [24, 80, 0]
            else:
                caps = [32, 128, 256]

            for cap in caps:
                if time.monotonic() > stop_time - 10.0:
                    break
                x, _mu = _lp_cheby(p, stop_time, active_cap=cap)
                if x is None:
                    continue
                diff = float(np.linalg.norm(x - p))
                if diff <= 1e-8:
                    continue
                pn, psc = _line_search(p, x, sc, stop_time - 8.0)
                if pn is not None:
                    p = pn
                    sc = psc
                    improved = True
                    break

            if not improved and time.monotonic() < stop_time - 10.0:
                if n <= 800:
                    sum_caps = [64, 0]
                else:
                    sum_caps = [128, 256]
                for cap in sum_caps:
                    if time.monotonic() > stop_time - 8.0:
                        break
                    xs, sx = _lp_sum(p, stop_time, active_cap=cap)
                    if xs is None:
                        continue
                    diff = float(np.linalg.norm(xs - p))
                    if (sx > 1.001 and diff > 1e-7) or diff > 1e-5:
                        pn, psc = _line_search(p, xs, sc, stop_time - 6.0)
                        if pn is not None:
                            p = pn
                            sc = psc
                            improved = True
                            break

        if improved:
            stall = 0
            continue

        stall += 1

        if n <= 1000 and stall in (1, 2, 4, 7, 10) and time.monotonic() < stop_time - 25.0:
            rem = stop_time - time.monotonic()
            ct = time.monotonic() + max(5.0, min(30.0, 0.20 * rem))
            if ct < stop_time - 5.0:
                p2, sc2 = _coordinate_polish(
                    p, ct, seed=seed + 5000 + it, max_passes=2, do_pairwise=(n <= 800)
                )
                if sc2 < sc - 1e-15:
                    p = p2
                    sc = sc2
                    stall = 0
                    continue
                if sc2 < best_sc - 1e-15:
                    best_p = p2.copy()
                    best_sc = sc2
                if sc2 < sc:
                    p = p2
                    sc = sc2

        if stall in (3, 6, 10, 16, 24) and time.monotonic() < stop_time - 60.0:
            rem = stop_time - time.monotonic()
            bt = time.monotonic() + max(10.0, min(45.0, 0.25 * rem))
            if bt < stop_time - 20.0:
                K = (1, 3, 8, 16)[stall % 4]
                p3, sc3 = _lbfgsb_run(
                    p,
                    bt,
                    seed=seed + 7000 + it,
                    maxfun=500 if n <= 600 else 250,
                    active_K=K,
                )
                if sc3 < sc - 1e-15:
                    p = p3
                    sc = sc3
                    stall = 0
                elif sc3 < best_sc - 1e-15:
                    best_p = p3.copy()
                    best_sc = sc3
                    p = p3.copy()
                    sc = sc3

        if stall >= 8:
            p = best_p.copy()
            mode = it % 4
            mean = max(float(p.mean()), 1e-12)
            ar = np.arange(p.size, dtype=np.float64)

            if mode == 0:
                sig = 0.05 * mean
                p = np.clip(p + rng.normal(0.0, sig, p.size), 0.0, None)
            elif mode == 1:
                p = p.copy()
                if p.size > 20:
                    idx = rng.choice(p.size, size=max(1, p.size // 50), replace=False)
                    p[idx] = 0.0
                sig = 0.02 * mean
                p = np.clip(p + rng.normal(0.0, sig, p.size), 0.0, None)
            elif mode == 2:
                rc = 0.5 * (1.0 + np.cos(2.0 * math.pi * ar / p.size))
                p = 0.65 * p + 0.35 * rc
            else:
                f = 0.05 * np.ones(p.size, dtype=np.float64)
                for fr in (1, 2, 3, 5, 8):
                    ph = float(rng.uniform(0.0, 2.0 * math.pi))
                    f += 0.08 * np.cos(2.0 * math.pi * fr * ar / p.size + ph)
                p = 0.65 * p + 0.35 * np.abs(f)

        s = float(p.sum())
        if s <= 1e-12:
            p = best_p.copy()
            s = float(best_p.sum())
        p = np.clip(p, 0.0, None)
        p = p / p.sum()

        if stall > 80:
            break

    return best_p, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    headroom = min(30.0, max(5.0, 0.05 * budget))
    deadline = t0 + max(5.0, budget - headroom)

    parent = _get_parent()
    best_p = _normalize_prob(parent)
    if best_p is None:
        best_p = np.ones(320, dtype=np.float64)
    best_p = np.asarray(best_p, dtype=np.float64).ravel()
    best_sc = _score_array(best_p)

    def consider(p, sc=None):
        nonlocal best_p, best_sc
        if p is None:
            return
        pn = _normalize_prob(p)
        if pn is None:
            return
        if sc is None:
            sc = _score_array(pn)
        if not math.isfinite(sc):
            return
        if sc < best_sc - 1e-14:
            best_p = pn.copy()
            best_sc = float(sc)

    total = max(10.0, deadline - t0)

    # Phase 0: very quick length resampling and smoothing.
    quick_end = t0 + min(35.0, 0.03 * total)
    base_len = int(best_p.size)
    cand_set = set()
    for r in (0.75, 0.80, 0.85, 0.90, 0.95, 1.00, 1.05, 1.10, 1.15, 1.20, 1.30, 1.50, 2.00):
        L = int(round(base_len * r))
        if 64 <= L <= 1600:
            cand_set.add(L)
    for L in (
        128, 160, 192, 224, 256, 288, 320, 352, 384, 416, 432,
        448, 464, 480, 512, 544, 576, 640, 768, 896, 1024, 1280, 1536,
    ):
        cand_set.add(L)

    for L in cand_set:
        if time.monotonic() > quick_end:
            break
        pl = _interpolate(best_p, L)
        consider(pl)

    consider(_smooth_prob(best_p, frac=0.05))
    consider(_smooth_prob(best_p, frac=0.02))

    # Phase 1: primary local LP-based descent at the currently best length.
    phase1_end = t0 + 0.45 * total
    if time.monotonic() < phase1_end - 20.0:
        p1, sc1 = _lp_search(best_p, int(best_p.size), phase1_end, int(seed))
        consider(p1, sc1)

    # Phase 2: multiple L-BFGS-B starts, mainly for nonlinear escape.
    phase2_end = t0 + 0.65 * total
    if time.monotonic() < phase2_end - 20.0:
        n = int(best_p.size)
        starts = []
        starts.append(best_p.copy())
        starts.append(_smooth_prob(best_p, frac=0.05))
        starts.append(_smooth_prob(best_p, frac=0.02))

        pp = _normalize_prob(parent)
        if pp is not None and pp.size != n:
            pp = _interpolate(pp, n)
        if pp is not None:
            starts.append(pp)

        i = np.arange(n, dtype=np.float64)
        starts.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i / max(1, n))))

        rng = np.random.default_rng(int(seed) + 31337)
        for alpha in (0.3, 0.55):
            noise = np.abs(rng.normal(0.0, 1.0, n)) + 0.05
            noise = noise / max(1e-300, float(noise.sum()))
            starts.append((1.0 - alpha) * best_p + alpha * noise)

        uniq = []
        for st in starts:
            stn = _normalize_prob(st)
            if stn is None or stn.size != n:
                continue
            if not any(float(np.linalg.norm(stn - u)) < 1e-6 for u in uniq):
                uniq.append(stn)
        starts = uniq[:8]

        Ks = (1, 3, 8, 16)
        run_id = 0
        for st in starts:
            if time.monotonic() > phase2_end - 25.0:
                break
            for K in Ks:
                if time.monotonic() > phase2_end - 20.0:
                    break
                maxfun = 1200 if n <= 512 else (700 if n <= 1024 else 350)
                pB, scB = _lbfgsb_run(
                    st,
                    min(phase2_end - 15.0, time.monotonic() + max(5.0, 0.08 * (phase2_end - t0))),
                    seed=int(seed) + 9000 + run_id * 17,
                    maxfun=maxfun,
                    active_K=K,
                )
                consider(pB, scB)
                run_id += 1

        for K in (1, 3, 8):
            if time.monotonic() > phase2_end - 10.0:
                break
            pB, scB = _lbfgsb_run(
                best_p,
                phase2_end - 5.0,
                seed=int(seed) + 12000 + K,
                maxfun=2000 if n <= 512 else 1000,
                active_K=K,
            )
            consider(pB, scB)

    # Phase 3: explore other lengths from the current best shape.
    phase3_end = t0 + 0.85 * total
    if time.monotonic() < phase3_end - 60.0:
        base_len = int(best_p.size)
        cand = set()
        for r in (0.80, 0.85, 0.90, 0.95, 1.05, 1.10, 1.15, 1.20, 1.30, 1.40, 1.50, 1.75):
            L = int(round(base_len * r))
            if 64 <= L <= 1600:
                cand.add(L)
        for L in (384, 400, 416, 432, 448, 464, 480, 512, 544, 576, 640, 768, 896, 1024, 1280, 1536):
            if abs(L - base_len) <= max(12, base_len // 4):
                cand.add(L)

        scored = []
        for L in cand:
            if time.monotonic() > phase3_end - 50.0:
                break
            if L == base_len:
                continue
            pL = _interpolate(best_p, L)
            scL = _score_array(pL)
            if math.isfinite(scL) and scL < best_sc + 0.08:
                scored.append((scL, L))

        scored.sort(key=lambda z: z[0])
        chosen = []
        if not scored:
            for L in (base_len + 32, int(base_len * 1.25), int(base_len * 1.5)):
                if 64 <= L <= 1600 and L != base_len:
                    chosen.append((math.inf, L))
        else:
            for scL, L in scored:
                if len(chosen) >= 8:
                    break
                if L != base_len:
                    chosen.append((scL, L))

        for idx, (_, L) in enumerate(chosen):
            if time.monotonic() > phase3_end - 25.0:
                break
            rem = max(5.0, phase3_end - time.monotonic())
            time_per = max(15.0, min(90.0, rem / max(1, len(chosen) - idx)))
            stopL = min(phase3_end - 10.0, time.monotonic() + time_per)
            init = _interpolate(best_p, int(L))
            pL, scL = _lp_search(init, int(L), stopL, int(seed) + 20000 + int(L))
            consider(pL, scL)

    # Phase 4: dedicated final coordinate polish on the best candidate.
    phase4_end = deadline - 5.0
    if time.monotonic() < phase4_end - 25.0 and int(best_p.size) <= 1200:
        pC, scC = _coordinate_polish(
            best_p,
            phase4_end,
            seed=24681,
            max_passes=1000,
            do_pairwise=(int(best_p.size) <= 900),
        )
        consider(pC, scC)

    if time.monotonic() < deadline - 25.0 and int(best_p.size) <= 1200:
        pC2, scC2 = _coordinate_polish(
            best_p,
            deadline - 10.0,
            seed=98765,
            max_passes=1000,
            do_pairwise=(int(best_p.size) <= 900),
        )
        consider(pC2, scC2)

    # If there is still meaningful time, one last short local search.
    if time.monotonic() < deadline - 35.0 and int(best_p.size) <= 1000:
        pR, scR = _lp_search(
            best_p,
            int(best_p.size),
            deadline - 25.0,
            int(seed) + 424242,
        )
        consider(pR, scR)

    # Final cleanup and exact validation with the authoritative evaluator.
    out = np.asarray(best_p, dtype=np.float64).ravel()
    if out.size == 0 or not np.all(np.isfinite(out)):
        out = np.ones(1, dtype=np.float64)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0.0:
        out = np.ones(1, dtype=np.float64)
        s = 1.0
    else:
        out = out / s

    try:
        final_sc = float(evaluate_sequence([float(x) for x in out]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        fb = _normalize_prob(parent)
        if fb is None:
            fb = np.ones(1, dtype=np.float64)
        fb = np.clip(np.asarray(fb, dtype=np.float64).ravel(), 0.0, 1000.0)
        ss = float(fb.sum())
        if ss > 0.0:
            fb = fb / ss
        else:
            fb = np.ones(1, dtype=np.float64)
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fb]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fb]

    return [float(x) for x in out]
