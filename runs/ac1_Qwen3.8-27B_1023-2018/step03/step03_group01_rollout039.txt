                res, sc = lp_loop(init_l, L, stop, np.random.default_rng(seed + L * 100))
                try_update(res)
            except Exception:
                pass
            if best_sc < 1.5028:
                break

    # If we found a better length, polish it
    if len(best_seq) != n_p and time.monotonic() < other_end - 5.0:
        try:
            res, sc = smooth_optimize(best_seq, len(best_seq), other_end - 3.0, initial_sigma=0.1)
            try_update(res)
        except Exception:
            pass

    # ============ PHASE 4: Final polish ============
    polish_end = deadline - 3.0
    if time.monotonic() < polish_end - 10.0:
        try:
            res, sc = smooth_optimize(best_seq, len(best_seq), polish_end - 5.0, initial_sigma=0.05)
            try_update(res)
        except Exception:
            pass

    # Final short LP burst
    if time.monotonic() < polish_end - 3.0:
        try:
            res, sc = lp_loop(best_seq, len(best_seq), polish_end - 2.0,
                              np.random.default_rng(seed + 12345), max_stall=20)
            try_update(res)
        except Exception:
            pass

    # ============ Final validation ============
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        sc = evaluate_sequence(final_list)
        if math.isfinite(sc) and sc > 0:
            return final_list
    except Exception:
        pass

    # Fallback
    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass

    return [1.0] * 384
