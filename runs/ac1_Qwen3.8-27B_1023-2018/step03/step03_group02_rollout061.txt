        res = linprog(c, A_ub=A_use, b_ub=b_ub,
                      bounds=[(0, None)]*n, method='highs',
                      options={'presolve': True, 'time_limit': tl})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
    s = g.sum()
    return g/s if s > 1e-12 else None

def _lp_loop(f0, n, stop_time, seed, fok=4):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0, None)
    sf = f.sum()
    f = f/sf if sf > 1e-10 else np.ones(n)/n
    bf = f.copy(); bs = _score(f)
    bff = _conv_full(f); M = bff.max()
    tv, tlo, thi, st, stm, it = 0.3, 0.05, 0.8, 0, 130, 0
    while time.monotonic() < stop_time - 2:
        g = (_lp_dir(f, n, M, 8, max(50, n//3)) if fok > 1 and it%fok == fok-1
             else _lp_dir(f, n, M, 12))
        if g is None or np.linalg.norm(g - f) < 1e-8:
            st += 5 if g is None else 3
            if st > stm: break
            f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
            s = f.sum()
            f = f/s if s > 1e-10 else bf.copy()
            bff = _conv_full(f); M = bff.max()
            it += 1; continue
        bfg = _conv_cross(f, g); bgg = _conv_full(g)
        Sf, Sg = f.sum(), g.sum()
        imp = False
        for _ in range(16):
            a1, a2 = 1-tv, tv
            bn = a1*a1*bff + 2*a1*a2*bfg + a2*a2*bgg
            np.maximum(bn, 0, out=bn)
            sn = a1*Sf + a2*Sg
            if sn < 0.01:
                tv = max(tlo, tv*0.5); continue
            sc = 2*n*bn.max()/(sn*sn)
            if sc < bs - 1e-15:
                f = (a1*f + a2*g)
                bf, bs = f.copy(), sc
                bff, M = bn, bn.max()
                imp = True; tv = min(thi, tv*1.1); st = 0
                break
            tv = max(tlo, tv*0.5)
        if not imp:
            st += 1
            if st >= stm:
                r = st % 4
                if r == 0: f = np.clip(f + rng.normal(0, 0.1, n), 0, None)
                elif r == 1:
                    bl = n//4
                    f2 = f.copy(); f2[:bl], f2[n-bl:] = f[n-bl:].copy(), f[:bl].copy()
                    f = f2
                elif r == 2:
                    kn = np.exp(-0.5*(np.arange(-4,5)/2.0)**2); kn /= kn.sum()
                    f = np.clip(f*(1+np.convolve(rng.normal(0,0.2,n), kn, 'same')), 0, None)
                else: f = np.clip(f + np.abs(rng.normal(0, 0.08, n)), 0, None)
                s = f.sum()
                f = f/s if s > 1e-10 else bf.copy()
                bff = _conv_full(f); M = bff.max(); tv = 0.3; st = 0
        it += 1
    return bf, bs

def _polish(f, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(f.copy(), 0, None).astype(np.float64)
    s = f.sum()
    f = f/s if s > 1e-10 else np.ones(n)/n
    bs = _score(f)
    ni, mni = 0, 8
    while time.monotonic() < stop_time - 1.5:
        imp = False
        # Coordinate sweep
        for i in rng.permutation(n):
            if time.monotonic() > stop_time - 1.5: break
            o, bv, bvs = f[i], o, bs
            mults = (0.3,0.5,0.7,0.85,1.0,1.15,1.3,1.6,2.0,0.0) if o > 1e-8 \
                     else (0.005,0.02,0.1,0.5,1.0)
            for ml in mults:
                f[i] = o*ml if o > 1e-8 else ml
                if f.sum() < 0.01: f[i] = o; continue
                sc = _score(f)
                if sc < bvs - 1e-15: bvs, bv = sc, f[i]
            f[i] = bv
            if bvs < bs - 1e-15: bs, imp = bvs, True
        # 2-coord
        for _ in range(80):
            if time.monotonic() > stop_time - 1.5: break
            i, j = rng.integers(n), rng.integers(n)
            if i == j: continue
            oi, oj = f[i], f[j]
            for r in (0.3, 0.6, 1.0, 1.5, 0.0, 1.0, 2.0):
                f[i] = r * (oi + oj)
                f[j] = (1-r) * (oi + oj)
                if f.sum() < 0.01: f[i], f[j] = oi, oj; continue
                sc = _score(f)
                if sc < bs - 1e-15: bs, imp = sc, True
            f[i], f[j] = oi, oj
        if not imp:
            ni += 1
            if ni >= mni: break
        else:
            ni = 0
    return f, bs
