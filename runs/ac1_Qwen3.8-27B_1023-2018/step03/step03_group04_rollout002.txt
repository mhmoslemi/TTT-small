def _convolution(p):
    n = int(p.size)
    if n <= 768:
        c = np.convolve(p, p, mode="full")
    elif next_fast_len is not None:
        L = int(next_fast_len(2 * n - 1))

    return c


def _score_pc(p, c):
    p = np.asarray(p, dtype=np.float64).ravel()
    c = np.asarray(c, dtype=np.float64).ravel()
    if p.size == 0 or c.size == 0:
        return math.inf

    if not math.isfinite(s) or s <= 1e-15:
        return math.inf
    m = float(np.max(c))
    if not math.isfinite(m) or m < 0.0:

    sc = (2.0 * float(p.size) * m) / (s * s)
    return float(sc) if math.isfinite(sc) else math.inf


def _apply_change(c, p, idx, old, new, scratch):
    delta = float(new) - float(old)
    if delta == 0.0:
        return
    n = int(p.size)
    base_diag = c[2 * idx]
    np.multiply(p, 2.0 * delta, out=scratch)

    c[2 * idx] = base_diag + delta * (float(new) + float(old))

def _coord_best(p, c, score, i, scratch, s0, mean, sqrtm):
    n = int(p.size)
    old = float(p[i])
    best_x = old
    best_sc = float(score)

            return

            return

    passes = 0

    while time.monotonic() < stop and passes < max_passes:
        s = float(p.sum())
        if not math.isfinite(s) or s <= 1e-15:
            break
        if abs(s - 1.0) > 1e-12:
            p = p / s
            c = c / (s * s)
        score = _score_pc(p, c)
        s0 = 1.0
        mean = 1.0 / float(n)
        sqrtm = math.sqrt(max(float(np.max(c)), 0.0))
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop - 1.0:
                break
            best_x, best_sc = _coord_best(p, c, score, int(i), scratch, s0, mean, sqrtm)
            if math.isfinite(best_sc) and best_sc < score - IMP_TOL:
                old = float(p[i])
                delta = float(best_x) - old
                if delta != 0.0:
                    _apply_change(c, p, int(i), old, best_x, scratch)
                    p[i] = best_x
                    S = 1.0 + delta
                    if S > 1e-12:
                        inv = 1.0 / S
                        p *= inv
                        c *= (inv * inv)
                score = float(best_sc)
                improved = True

        if two > 0 and n >= 2 and time.monotonic() < stop - 3.0:
            cnt = max(1, min(int(two), 40))
            for _ in range(cnt):
                if time.monotonic() > stop - 2.0:
                    break
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii == jj:
                    continue
                oi = float(p[ii])
                oj = float(p[jj])
                vi = _pair_vals(oi, 1.0 / n)
                vj = _pair_vals(oj, 1.0 / n)
                if len(vi) <= 1 and len(vj) <= 1:
                    continue
                best_sc = score
                best_vi, best_vj = oi, oj
                best_c = None
                base_c = c.copy()

                for xi in vi:
                    for xj in vj:
                        if xi == oi and xj == oj:
                            continue
                        S = 1.0 + (xi - oi) + (xj - oj)
                        if S <= 1e-12:
                            continue
                        c2 = base_c.copy()
                        p[ii] = oi
                        p[jj] = oj
                        p[ii] = xi
                        _apply_change(c2, p, ii, oi, xi, scratch)
                        p[jj] = xj
                        _apply_change(c2, p, jj, oj, xj, scratch)
                        mx = float(c2.max())
                        if mx < 0.0 or not math.isfinite(mx):
                            mx = 0.0
                        sc = (2.0 * float(n) * mx) / (S * S)
                        if math.isfinite(sc) and sc < best_sc:
                            best_sc = sc
                            best_vi, best_vj = xi, xj
                            best_c = c2.copy()

                p[ii] = oi
                p[jj] = oj

                if best_c is not None and best_sc < score - IMP_TOL:
                    p[ii] = best_vi
                    p[jj] = best_vj
                    c = best_c.copy()
                    S = 1.0 + (best_vi - oi) + (best_vj - oj)
                    if S > 1e-12:
                        inv = 1.0 / S
                        p *= inv
                        c *= (inv * inv)
                    score = float(best_sc)
                    improved = True

        if improved:
            passes += 1
            if passes % 10 == 0:
                c = _convolution(p)
                score = _score_pc(p, c)
        else:
            break

    return p, c, score


def _perturb(p, sigma, rng):
    q = p * np.exp(rng.normal(0.0, float(sigma), p.size))
    q = np.clip(q, 1e-14, None)
    s = float(q.sum())
    if not math.isfinite(s) or s <= 1e-15:
        return p.copy()
    return q / s


def _mix_search(p, c, score, q, stop):
    n = int(p.size)
    qs = float(q.sum())
    if not math.isfinite(qs) or qs <= 1e-15:
        return p, c, score
    q = np.clip(q, 0.0, None)
    qs = float(q.sum())
    if qs <= 1e-15:
        return p, c, score
    q = q / qs

    best_t = 0.0
    best_p = p.copy()
    best_c = c.copy()
    best_sc = float(score)

    def consider(t):
        nonlocal best_t, best_p, best_c, best_sc
        if not math.isfinite(t) or t < 0.0 or t > 1.0:
            return
        if time.monotonic() > stop - 2.0:
            return
        pt = (1.0 - t) * p + t * q
        ct = _convolution(pt)
        sc = _score_pc(pt, ct)
        if math.isfinite(sc) and sc < best_sc - 1e-14:
            best_sc = sc
            best_t = t
            best_p = pt.copy()
            best_c = ct.copy()

    for t in (0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0):
        consider(t)

    if 0.0 < best_t < 1.0:
        lo = max(0.0, best_t * 0.7)
        hi = min(1.0, best_t * 1.3)
        for t in np.linspace(lo, hi, 7):
            consider(float(t))
    elif best_t == 0.0:
        for t in (0.005, 0.01, 0.02, 0.05):
            consider(t)
    elif best_t == 1.0:
        for t in (0.85, 0.9, 0.95, 1.0):
            consider(t)

    return best_p, best_c, best_sc


def _lp_direction(f, M, A, cvec, b):
    if linprog is None:
        return None
    n = int(f.size)
    if n == 0 or M <= 1e-15 or not math.isfinite(M):
        return None
    try:
        scale = 1.0 / float(M)
        fs = f * scale
        for i in range(n):
            A[i:i + n, i] = fs
        b.fill(1.0 + 1e-10)
        res = linprog(
            cvec,
            A_ub=A,
            b_ub=b,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True}
        )
        if res.status != 0 or getattr(res, "x", None) is None:
            return None
        g = np.asarray(res.x, dtype=np.float64)
        g = np.where(np.isfinite(g), g, 0.0)
        np.maximum(g, 0.0, out=g)
        s = float(g.sum())
        if not math.isfinite(s) or s <= 1e-10:
            return None
        g /= s
        return g
    except Exception:
        return None


def _lp_phase(p, c, score, stop, seed, best=None, max_lp=300):
    n = int(p.size)
    if n > 700 or linprog is None:
        return p, c, score
    rng = np.random.default_rng(seed)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    cvec = -np.ones(n, dtype=np.float64)
    b = np.empty(2 * n - 1, dtype=np.float64)
    stall = 0
    it = 0
    target_P = TARGET / (2.0 * float(n))

    while time.monotonic() < stop and it < max_lp:
        s = float(p.sum())
        if not math.isfinite(s) or s <= 1e-15:
            break
        if abs(s - 1.0) > 1e-12:
            p = p / s
            c = c / (s * s)
        score = _score_pc(p, c)
        M = float(np.max(c))
        if not math.isfinite(M) or M <= 0.0:
            break

        q = None
        if score > TARGET + 1e-15:
            M_goal = target_P * (1.0 + 1e-7)
            if 0.0 < M_goal < M:
                q = _lp_direction(p, M_goal, A, cvec, b)

        if q is None:
            q = _lp_direction(p, M * (1.0 + 1e-9), A, cvec, b)

        if q is None:
            if stall < 2:
                pp = _perturb(p, 0.03, rng)
                cc = _convolution(pp)
                scp = _score_pc(pp, cc)
                if scp < score - IMP_TOL:
                    p, c, score = pp, cc, scp
                    stall = 0
                    it += 1
                    continue
            break

        if float(np.linalg.norm(q - p, ord=1)) < 1e-7:
            stall += 1
            if stall >= 3:
                pp = _perturb(p, 0.05, rng)
                cc = _convolution(pp)
                scp = _score_pc(pp, cc)
                if scp < score - IMP_TOL:
                    p, c, score = pp, cc, scp
                    stall = 0
            if stall >= 8:
                break
            it += 1
            continue

        p2, c2, s2 = _mix_search(p, c, score, q, stop)
        if s2 < score - 1e-11:
            p, c, score = p2, c2, s2
            stall = 0
            quick_stop = min(stop - 25.0, time.monotonic() + 12.0)
            if quick_stop > time.monotonic():
                p, c, score = _coord_polish(p, c, score, quick_stop, rng,
                                            max_passes=3, two=8)
            if best is not None:
                sc = _score_pc(p, c)
                if sc < best[2] - 1e-15:
                    best[0] = p.copy()
                    best[1] = c.copy()
                    best[2] = sc
        else:
            stall += 1
            if stall > 5:
                pp = _perturb(p, 0.08, rng)
                cc = _convolution(pp)
                scp = _score_pc(pp, cc)
                if scp < score - IMP_TOL:
                    p, c, score = pp, cc, scp
                    stall = 2
                else:
                    break
        it += 1

    return p, c, score


def _smooth_phase(p, c, score, stop, seed=31):
    n = int(p.size)
    if n == 0:
        return p, c, score
    p = np.array(p, dtype=np.float64, copy=True)
    c = np.array(c, dtype=np.float64, copy=True)
    s = float(p.sum())
    if s > 1e-15:
        p = p / s
        c = c / (s * s)
    score = _score_pc(p, c)
    best_p = p.copy()
    best_c = c.copy()
    best_sc = score
    eta = 0.8
    no = 0

    for _ in range(80):
        if time.monotonic() > stop - 2.0:
            break
        M = float(np.max(c))
        if not math.isfinite(M) or M <= 0.0:
            break
        low = 0.90 * M
        w = c - low
        np.maximum(w, 0.0, out=w)
        ws = float(w.sum())
        if ws <= 1e-18:
            k = max(1, n // 10)
            idx = np.argsort(-c)[:k]
            w = np.zeros_like(c)
            w[idx] = 1.0
            ws = float(w.sum())
        if ws <= 0.0:
            break
        w /= ws
        grad = 2.0 * np.correlate(w, p, mode="valid").astype(np.float64)
        if not np.all(np.isfinite(grad)):
            break
        gm = float(grad.max())
        if gm <= 1e-15:
            break
        r = p * np.exp(-eta * grad / gm)
        r = np.clip(r, 0.0, None)
        rs = float(r.sum())
        if rs <= 1e-12:
            break
        r /= rs

        local_sc = score
        local_p = None
        local_c = None
        for beta in (0.05, 0.1, 0.2, 0.3, 0.5, 0.75, 1.0):
            if time.monotonic() > stop - 2.0:
                break
            pt = (1.0 - beta) * p + beta * r
            ct = _convolution(pt)
            sc = _score_pc(pt, ct)
            if math.isfinite(sc) and sc < local_sc - 1e-14:
                local_sc = sc
                local_p = pt.copy()
                local_c = ct.copy()

        if local_p is not None and local_sc < score - 1e-12:
            p, c, score = local_p, local_c, local_sc
            if score < best_sc:
                best_sc = score
                best_p = p.copy()
                best_c = c.copy()
            eta = min(2.0, eta * 1.1)
            no = 0
        else:
            eta *= 0.7
            no += 1
            if no > 6 or eta < 0.02:
                break

    return best_p, best_c, best_sc


def _make_valid(arr):
    try:
        arr = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if arr.sum() <= 1e-12:
        return None
    return arr


def _exact_score(arr):
    ev = globals().get("evaluate_sequence")
    if ev is None:
        return _score_pc(arr, _convolution(arr))
    try:
        return float(ev([float(x) for x in arr]))
    except Exception:
        return math.inf


def _finalize(cands):
    best_arr = None
    best_sc = math.inf
    for arr in cands:
        va = _make_valid(arr)
        if va is None:
            continue
        sc = _exact_score(va)
        if math.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_arr = va
    if best_arr is None:
        best_arr = np.ones(384, dtype=np.float64)
    return [float(x) for x in best_arr]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(1.0, min(float(budget_s), 1300.0))
    except Exception:
        budget = 1300.0

    headroom = min(40.0, max(5.0, 0.05 * budget))
    deadline = t0 + budget - headroom

    parent = _parent_array()
    if parent is None:
        parent = np.ones(384, dtype=np.float64)

    p = _normalize(parent)
    if p is None:
        p = np.ones(384, dtype=np.float64)
    c = _convolution(p)
    score = _score_pc(p, c)
    best = [p.copy(), c.copy(), score]

    if deadline <= t0 + 2.0:
        return _finalize([best[0], p, parent])

    n = int(p.size)
    rng = np.random.default_rng(seed)
    two = 30 if n <= 800 else 0
    maxpass = 150 if n <= 800 else 25

    stop = min(deadline - 30.0, t0 + 60.0)
    if stop > time.monotonic():
        p, c, score = _coord_polish(p, c, score, stop, rng,
                                    max_passes=maxpass, two=two)
        sc = _score_pc(p, c)
        if sc < best[2] - 1e-15:
            best = [p.copy(), c.copy(), sc]

    now = time.monotonic()
    rem = deadline - now
    if rem > 40.0:
        lp_dur = min(700.0, max(20.0, 0.55 * rem))
        stop = min(deadline - max(80.0, 0.30 * rem), now + lp_dur)
        if stop > now and n <= 700:
            p, c, score = _lp_phase(p, c, score, stop, int(seed) + 1, best=best)
            sc = _score_pc(p, c)
            if sc < best[2] - 1e-15:
                best = [p.copy(), c.copy(), sc]

    now = time.monotonic()
    rem = deadline - now
    if rem > 60.0:
        stop = min(deadline - max(50.0, 0.25 * rem), now + 60.0)
        if stop > now:
            p, c, score = _smooth_phase(p, c, score, stop, int(seed) + 11)
            sc = _score_pc(p, c)
            if sc < best[2] - 1e-15:
                best = [p.copy(), c.copy(), sc]

    now = time.monotonic()
    rem = deadline - now
    if rem > 30.0:
        stop = min(deadline - 15.0, now + 120.0)
        if stop > now:
            p, c, score = _coord_polish(p, c, score, stop, rng,
                                        max_passes=maxpass, two=two)
            sc = _score_pc(p, c)
            if sc < best[2] - 1e-15:
                best = [p.copy(), c.copy(), sc]

    now = time.monotonic()
    rem = deadline - now
    if rem > 70.0 and n <= 700:
        stop = min(deadline - 25.0, now + 70.0)
        if stop > now:
            p, c, score = _lp_phase(p, c, score, stop, int(seed) + 101, best=best)
            sc = _score_pc(p, c)
            if sc < best[2] - 1e-15:
                best = [p.copy(), c.copy(), sc]

    stop = deadline - 10.0
    if stop > time.monotonic():
        p, c, score = _coord_polish(p, c, score, stop, rng,
                                    max_passes=maxpass, two=max(0, two // 2))
        sc = _score_pc(p, c)
        if sc < best[2] - 1e-15:
            best = [p.copy(), c.copy(), sc]

    return _finalize([best[0], p, parent, np.ones(384, dtype=np.float64)])
