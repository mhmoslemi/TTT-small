def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)


def _make_conv_matrix(f, n):
    A = np.zeros((2*n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    return A


def _lp_step(f, n, bounds, timeout=4.0):
    A = _make_conv_matrix(f, n)
    c = -np.ones(n)
    bl = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=bounds, bounds=bl,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, 1000.0)
    if g.sum() < 1e-15:
        return None
    return g


def _line_search(f, g, b_f, best_score):
    n = len(f)
    s = float(f.sum())
    if s < 1e-10 or s <= 0:
        return 0.0, best_score
    
    b_fg = _conv_cross(f, g)
    b_g = _conv_full(g)
    
    best_t = 0.0
    best_sc = best_score
    
    # Coarse grid
    for t in np.linspace(0.02, 1.0, 35):
        b_t = (1-t)**2 * b_f + 2*t*(1-t) * b_fg + t**2 * b_g
        sc = 2.0 * n * max(0.0, float(b_t.max())) / (s*s)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    
    # Fine grid around best
    if best_t > 0:
        lo = max(0.0, best_t - 0.03)
        hi = min(1.0, best_t + 0.03)
        for t in np.linspace(lo, hi, 25):
            b_t = (1-t)**2 * b_f + 2*t*(1-t) * b_fg + t**2 * b_g
            sc = 2.0 * n * max(0.0, float(b_t.max())) / (s*s)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    
    return best_t, best_sc


def _lp_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n].copy()
    if f.sum() < 1e-10:
        f = np.ones(n) * 0.1
    
    best_f = f.copy()
    best_score = _score_fast(f)
    if not math.isfinite(best_score):
        return f, best_score
    
    stall = 0
    max_stall = 60
    use_tight = True
    iteration = 0
    
    while time.monotonic() < stop_time - 1.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-25:
            break
        
        if use_tight:
            bounds = b * 1.002 + 1e-10
        else:
            bounds = np.full(2*n-1, M * 1.002 + 1e-10)
        
        g = _lp_step(f, n, bounds, timeout=3.0)
        
        if g is None:
            stall += 4
            if stall > max_stall:
                use_tight = not use_tight
                stall = 0
                max_stall += 30
                if max_stall > 200:
                    break
                noise = rng.normal(0, 0.1 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                fs = f.sum()
                if fs > 0.01:
                    f = f / fs * best_f.sum()
            continue
        
        g_sum = g.sum()
        f_sum = f.sum()
        g = g * (f_sum / g_sum)
        
        dist = np.linalg.norm(g - f) / max(1.0, np.linalg.norm(f))
        if dist < 1e-9:
            stall += 2
            if stall > max_stall:
                break
            continue
        
        best_t, best_t_sc = _line_search(f, g, b, best_score)
        
        if best_t > 0 and best_t_sc < best_score - 1e-13:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, None)
            best_f = f.copy()
            best_score = best_t_sc
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                use_tight = not use_tight
                noise = rng.normal(0, 0.15 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                fs = f.sum()
                if fs > 0.01:
                    f = f / fs * best_f.sum()
                stall = 0
                max_stall += 40
        
        iteration += 1
    
    return best_f, best_score

    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 20
    
    b_cache = _conv_full(best)
    
    while time.monotonic() < stop_time - 2.0:
        improved = False
        
        # Single-coordinate scaling
        order = rng.permutation(n)
        for i in order:

                continue

                    best_val_sc = sc
                    best_val = best[i]
            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True
        
        # Two-coordinate
        for _ in range(300):
            if time.monotonic() > stop_time - 2.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i >= j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.0, 0.3, 0.7, 1.5, 3.0):
                for mj in (0.0, 0.3, 0.7, 1.5, 3.0):
                    if abs(mi-1)+abs(mj-1) < 1e-6:
                        continue
                    best[i] = oi * mi
                    best[j] = oj * mj
                    if best.sum() < 0.01:
                        continue
                    sc = _score_fast(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj
        
        # Random perturbation
        if not improved:
            for _ in range(80):
                if time.monotonic() > stop_time - 2.0:
                    break
                idx = rng.integers(n, size=5)
                old = best[idx].copy()
                best[idx] = np.maximum(0.0, best[idx] + rng.normal(0, 0.08*max(1.0, best.mean()), size=5))
                if best.sum() < 0.01:
                    best[idx] = old
                    continue
                sc = _score_fast(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                else:
                    best[idx] = old
        
        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0
    
    return best, best_sc
