        cand = np.maximum((1 - mid) * a + mid * direction, 0.0)
        s = cand.sum()
        if s < 1e-12:
            hi = mid
            continue
        cand /= s
        v = float(np.convolve(cand, cand).max())
        if v < best_val - 1e-15:
            best_val = v
            best_alpha = mid
            hi = mid
        else:
            lo = mid
    cand = np.maximum((1 - best_alpha) * a + best_alpha * direction, 0.0)
    cand /= cand.sum()
    return cand, best_val


def _gradient_dir(a, k):
    """Subgradient of (a*a)[k] w.r.t. a, projected to sum=0."""
    n = a.size
    g = np.zeros(n)
    jlo = max(0, k - n + 1)
    jhi = min(n - 1, k)
    for j in range(jlo, jhi + 1):
        g[j] += 2.0 * a[k - j]
    g -= g.mean()
    return g


def _optimize(a0, stop_time, seed=0, max_iters=200000):
    """LP best-response + gradient optimization."""
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a /= s

    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0
    n_perturb = 0
    lp_count = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 0.5:
            break

        b = np.convolve(a, a)
        M = float(b.max())

        if M < best_M * (1.0 - 5e-14):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 50:
            sc = 0.04 + 0.02 * min(n_perturb, 8)
            a = a * np.exp(rng.normal(0, sc, n))
            a = np.maximum(a, 1e-10)
            a /= a.sum()
            stuck = 0
            n_perturb += 1
            if n_perturb > 60:
                break
            continue

        found_move = False

        # LP best response
        if lp_count < 5000 or (it % 3 == 0):
            A = _conv_matrix(a)
            g = _solve_lp(A, M * (1.0 + 1e-10), tl=4.0)
            lp_count += 1
            if g is not None and not np.allclose(g, a, atol=1e-9):
                cand, val = _line_search(a, g, M)
                if cand is not None and val < M * (1 - 5e-13):
                    a = cand
                    found_move = True

        if found_move:
            continue

        # Try reversed blend
        if it % 7 == 0:
            g_rev = _solve_lp(_conv_matrix(a), M * (1 + 1e-10), tl=3.0)
            if g_rev is not None:
                for w in [0.3, 0.5, 0.7]:
                    blended = w * g_rev + (1 - w) * g_rev[::-1]
                    blended = np.maximum(blended, 0.0)
                    bs = blended.sum()
                    if bs > 1e-12:
                        blended /= bs
                    cand, val = _line_search(a, blended, M)
                    if cand is not None and val < M * (1 - 5e-13):
                        a = cand
                        found_move = True
                        break

        if found_move:
            continue

        # Gradient descent on the max lag
        k_star = int(np.argmax(b))
        # Also check top few lags
        top_lags = np.argsort(b)[-5:]
        for kl in top_lags:
            gdir = _gradient_dir(a, kl)
            gn = np.linalg.norm(gdir)
            if gn < 1e-14:
                continue
            gdir /= gn
            for alpha in [0.015, 0.008, 0.004, 0.002, 0.001]:
                cand = a - alpha * gdir
                cand = np.maximum(cand, 0.0)
                cs = cand.sum()
                if cs < 1e-12:
                    continue
                cand /= cs
                v = float(np.convolve(cand, cand).max())
                if v < M * (1 - 1e-13):
                    a = cand
                    found_move = True
                    break
            if found_move:
                break

        if not found_move:
            # Tiny random kick
            a = a * np.exp(rng.normal(0, 0.015, n))
            a = np.maximum(a, 1e-10)
            a /= a.sum()

    return best_a


def _asymmetric_inits(n, parent):
    """Generate diverse asymmetric initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    t = x / (n - 1)

    # Parent normalized
    p = np.clip(parent, 0, 1000).astype(np.float64)
    ps = p.sum()
    if ps > 1e-10 and p.size == n:
        inits.append(p / ps)

    # Skewed power
    for pwr in [0.4, 0.7, 1.0, 1.5, 2.0, 3.0]:
        seq = (1.0 - 0.9 * t) ** pwr
        seq = np.maximum(seq, 0.01)
        inits.append(seq / seq.sum())

        seq2 = (0.1 + 0.9 * t) ** pwr
        seq2 = np.maximum(seq2, 0.01)
        inits.append(seq2 / seq2.sum())

    # Two-hump asymmetric
    for c1 in [0.15, 0.25, 0.35]:
        for c2 in [0.55, 0.65, 0.75]:
            seq = np.exp(-30 * (t - c1) ** 2) + 0.6 * np.exp(-30 * (t - c2) ** 2)
            seq = np.maximum(seq, 0.005)
            inits.append(seq / seq.sum())

    # Linear ramps with noise
    seq = np.linspace(0.3, 1.0, n)
    inits.append(seq / seq.sum())
    seq = np.linspace(1.0, 0.3, n)
    inits.append(seq / seq.sum())

    # Cosine-based asymmetric
    for shift in [0.0, 0.2, 0.5]:
        seq = 1.0 + 0.8 * np.cos(2 * np.pi * (t + shift))
        seq = np.maximum(seq, 0.02)
        inits.append(seq / seq.sum())

    # Blend parent with uniform (various ratios)
    if ps > 1e-10 and p.size == n:
        pn = p / ps
        for w in [0.2, 0.4, 0.6, 0.8]:
            blend = w * pn + (1 - w) / n
            inits.append(blend / blend.sum())

    return inits


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 25.0

    parent = _get_parent()
    n_p = parent.size

    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(201) / 201.0
        best_score = 2.0

    rng = np.random.default_rng(seed)

    def _try_update(candidate):
        nonlocal best_seq, best_score
        try:
            sc = _score(candidate)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = candidate.copy()
                return True
        except Exception:
            pass
        return False

    # === Phase 1: Optimize from parent (asymmetric n) ===
    a_init = np.clip(parent, 0, 1000).astype(np.float64)
    a_init = np.maximum(a_init, 1e-10)
    a_init /= a_init.sum()

    p1_end = t0 + min(budget * 0.22, 240.0)
    try:
        res = _optimize(a_init, p1_end, seed=1, max_iters=100000)
        _try_update(res)
    except Exception:
        pass

    # === Phase 2: Asymmetric initializations ===
    inits = _asymmetric_inits(n_p, parent)
    # Deduplicate
    unique_inits = []
    for init in inits:
        init = np.clip(init, 0, 1000).astype(np.float64)
        if init.size != n_p:
            continue
        init = np.maximum(init, 1e-10)
        ist = init.sum()
        if ist < 1e-10:
            continue
        init /= ist
        if not any(np.allclose(init, u, atol=1e-8) for u in unique_inits):
            unique_inits.append(init)

    p2_end = t0 + min(budget * 0.50, 520.0)
    remaining = p2_end - time.monotonic()
    if remaining > 20 and len(unique_inits) > 0:
        per_init = max(2.0, remaining / len(unique_inits))
        for idx, init in enumerate(unique_inits):
            if time.monotonic() > p2_end - 5:
                break
            alloc = min(time.monotonic() + per_init, p2_end - 2)
            try:
                res = _optimize(init, alloc, seed=100 + idx, max_iters=50000)
                _try_update(res)
            except Exception:
                pass

    # === Phase 3: Extended polish of best ===
    p3_end = t0 + min(budget * 0.68, 740.0)
    a_best = np.clip(best_seq, 0, 1000).astype(np.float64)
    a_best = np.maximum(a_best, 1e-10)
    a_best /= a_best.sum()
    try:
        res = _optimize(a_best, p3_end, seed=777, max_iters=200000)
        _try_update(res)
    except Exception:
        pass

    # === Phase 4: Explore nearby lengths ===
    if time.monotonic() < deadline - 100:
        for dn in [-8, -6, -4, -3, -2, -1, 0, 1, 2, 3, 4, 6, 8]:
            nt = n_p + dn
            if nt < 50 or nt > 500 or nt == n_p:
                continue
            if time.monotonic() > deadline - 30:
                break
            # Resample parent
            x_old = np.linspace(0, 1, n_p)
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, parent)
            init_new = np.clip(init_new, 0, 1000)
            init_new = np.maximum(init_new, 1e-10)
            init_new /= init_new.sum()
            alloc = min(time.monotonic() + 35, deadline - 15)
            try:
                res = _optimize(init_new, alloc, seed=500 + nt, max_iters=50000)
                _try_update(res)
            except Exception:
                pass

            # Also try a fresh asymmetric init at this length
            x = np.arange(nt, dtype=np.float64)
            t = x / max(nt - 1, 1)
            for pwr in [1.0, 2.0]:
                if time.monotonic() > deadline - 20:
                    break
                seq = np.maximum(1.0 - 0.8 * t, 0.05) ** pwr
                seq /= seq.sum()
                alloc2 = min(time.monotonic() + 20, deadline - 10)
                try:
                    res = _optimize(seq, alloc2, seed=900 + nt + pwr, max_iters=30000)
                    _try_update(res)
                except Exception:
                    pass

    # === Phase 5: Final tight polish ===
    if time.monotonic() < deadline - 10:
        a_final = np.clip(best_seq, 0, 1000).astype(np.float64)
        a_final = np.maximum(a_final, 1e-10)
        a_final /= a_final.sum()
        try:
            res = _optimize(a_final, deadline, seed=999, max_iters=500000)
            _try_update(res)
        except Exception:
            pass

    # === Final validation with exact evaluator ===
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_score = float('inf')

    if not np.isfinite(final_score) or final_score > best_score * 1.01:
        # Fallback: try a simple valid sequence
        fb = np.ones(201, dtype=np.float64) / 201.0
        try:
            fs = float(evaluate_sequence([float(x) for x in fb]))
            if np.isfinite(fs):
                return [float(x) for x in fb]
        except Exception:
            pass
        return [1.0] * 32

    return [float(x) for x in best_seq]
