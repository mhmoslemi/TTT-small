        k_star = int(np.argmax(b))
        M_val = b[k_star]

        # Gradient of (f*f)_{k*} w.r.t. f_j is 2*f_{k*-j} (if valid)
        # To reduce the peak while maintaining sum, we want to:
        # - decrease f_j where f_{k*-j} is large
        # - increase f_j where f_{k*-j} is small (or zero)

        # Single-coordinate descent on the active constraint
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 4.0:
                break
            orig = best[i]
            if orig < 1e-14:
                continue
            # Try multipliers
            best_val_sc = best_sc
            best_val = orig
            for mlt in (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 0.0):
                best[i] = orig * mlt
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-15:
                    best_val = best[i]
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-15:
                best_sc = best_val_sc
                improved = True

        # Pairs that directly affect the k* convolution
        # (f*f)_{k*} = sum_i f_i * f_{k*-i}
        # Pairs (i, k*-i) both contribute
        for _ in range(200):
            if time.monotonic() > stop_time - 4.0:
                break
            i = int(rng.integers(n))
            j = k_star - i
            if j < 0 or j >= n or j == i:
                j = int(rng.integers(n))
                if j == i:
                    continue
            oi, oj = best[i], best[j]
            for mi in (0.4, 0.6, 0.8, 1.2, 1.5, 0.0):
                for mj in (0.4, 0.6, 0.8, 1.2, 1.5, 0.0):
                    if mi == 1.0 and mj == 1.0:
                        continue
                    best[i] = oi * mi if oi > 1e-15 else 0.0
                    best[j] = oj * mj if oj > 1e-15 else 0.0
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        # Random small perturbations
        if not improved:
            for _ in range(50):
                if time.monotonic() > stop_time - 4.0:
                    break
                k = int(rng.integers(n))
                pert = rng.normal(0, 0.03 * max(1.0, best.mean()))
                old = best[k]
                best[k] = max(0.0, old + pert)
                if best.sum() < 0.01:
                    best[k] = old
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                else:
                    best[k] = old

        if not improved:
            no_imp_count += 1
            if no_imp_count >= max_no_imp:
                break
        else:
            no_imp_count = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_p = len(parent)

    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(320) / 320.0
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try(n, init_arr, stop, sd):
        nonlocal best_seq, best_score
        try:
            res, sc = _lp_optimize(init_arr, n, stop, sd)
        except Exception:
            return
        if res is None or len(res) == 0:
            return
        if not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0, 1000)
        if res.sum() < 0.01:
            return
        if sc < best_score - 1e-14:
            try:
                exact = float(evaluate_sequence([float(x) for x in res]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-14:
                best_seq = res.copy()
                best_score = exact

    # Phase 1: LP optimization on parent length with diverse initializations
    n = n_p
    phase1_end = t0 + budget * 0.45
    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    inits = []
    inits.append(parent.copy())

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n)))

    # Multi-frequency
    sup = np.ones(n) * 0.3
    for fr in (1, 2, 3, 5, 7):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.15 * np.cos(2.0 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.05)

    # Random abs-normal
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.05)

    # Parent + noise
    sig = 0.1 * max(1.0, float(np.mean(parent)))
    inits.append(np.abs(parent + rng_main.normal(0, sig, n)) + 0.01)

    # Gaussian bump
    center = (n - 1) / 2.0
    width = n / 3.5
    inits.append(np.exp(-0.5 * ((i_arr - center) / width) ** 2))

    # Two-lobe
    two = np.exp(-0.5 * ((i_arr - n * 0.25) / (n * 0.15)) ** 2) + \
          np.exp(-0.5 * ((i_arr - n * 0.75) / (n * 0.15)) ** 2)
    inits.append(two)

    # Flat with random modulation
    mod = 1.0 + 0.3 * rng_main.uniform(-1, 1, n)
    inits.append(np.abs(mod) + 0.1)

    # Sine wave
    inits.append(0.5 + 0.4 * np.sin(2 * math.pi * 3 * i_arr / n))

    time_per_init = max(10.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 8.0:
            break
        if best_score < 1.45:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        _try(n, init, stop, seed + 1000 * (idx + 1))

    # Phase 2: Try other lengths
    phase2_end = t0 + budget * 0.65
    other_lengths = [384, 400, 416, 432, 448, 464, 480, 500, 512]
    # Filter out ones we already tried
    other_lengths = [L for L in other_lengths if L != n_p]
    # Sort by distance to parent length
    other_lengths.sort(key=lambda L: abs(L - n_p))
    other_lengths = other_lengths[:6]  # Only try 6 alternative lengths

    if time.monotonic() < phase2_end - 10:
        time_per_len = max(8.0, (phase2_end - time.monotonic()) / max(1, len(other_lengths)))
        for L in other_lengths:
            if time.monotonic() > phase2_end - 6.0:
                break
            stop = min(phase2_end - 4.0, time.monotonic() + time_per_len)
            # Interpolate parent to this length
            if L != len(parent):
                init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(parent)), parent)
            else:
                init_l = parent.copy()
            _try(L, init_l, stop, seed + 50000 + L)

    # Phase 3: Second LP pass on best (warm restart)
    phase3_end = t0 + budget * 0.82
    if time.monotonic() < phase3_end - 15.0:
        stop_p3 = min(phase3_end - 4.0, time.monotonic() + (phase3_end - time.monotonic()))
        _try(len(best_seq), best_seq.copy(), stop_p3, seed + 999999)

    # Additional LP pass with perturbed best
    if time.monotonic() < phase3_end - 10.0:
        pert = np.random.default_rng(seed + 777).normal(0, 0.05, len(best_seq))
        pert_init = np.clip(best_seq + pert, 0, None)
        ps = pert_init.sum()
        if ps > 0.01:
            pert_init = pert_init / ps * best_seq.sum()
        stop_p3b = min(phase3_end - 4.0, time.monotonic() + 30.0)
        _try(len(best_seq), pert_init, stop_p3b, seed + 888888)

    # Phase 4: Gradient-focused polishing
    polish_end = deadline - 8.0
    if time.monotonic() < polish_end - 10.0:
        try:
            best_seq, best_score = _gradient_polish(best_seq, polish_end, seed=12345)
        except Exception:
            pass

    # Second polish pass
    if time.monotonic() < deadline - 12.0:
        try:
            best_seq2, sc2 = _gradient_polish(best_seq, deadline - 6.0, seed=54321)
            if sc2 < best_score - 1e-14:
                best_seq = best_seq2
                best_score = sc2
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    if final_sc < best_score - 1e-12:
        best_score = final_sc

    return [float(x) for x in best_seq]
