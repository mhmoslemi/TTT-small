    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je+1] = a[k - je : k - js + 1][::-1]
    return A

def _solve_lp(a, M, deadline):
    n = a.size
    rem = deadline - time.monotonic()
    if rem < 3.0:
        return None, None
    A = _build_A(a)
    b_ub = M * np.ones(2*n-1)
    c_obj = -np.ones(n)
    tl = max(0.5, min(10.0, 0.3 * rem))
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'time_limit': tl, 'presolve': True})
        if res.success:
            x = np.maximum(res.x, 0)
            s = x.sum()
            if s > 1e-12:
                dir_b = x / s
                # Get dual variables for subgradient
                lam = getattr(res, 'ineqlin', None)
                if lam is not None:
                    lam = np.maximum(lam, 0)
                    ls = lam.sum()
                    weights = lam / ls if ls > 0 else None
                else:
                    weights = None
                return dir_b, weights
    except:
        pass
    return None, None

def _weighted_grad(a, w):
    """Compute weighted subgradient of max(conv) w.r.t. a."""
    n = a.size
    m = 2*n - 1
    if w.size != m:

    # g[j] = 2 * sum_{m2=0}^{n-1} a[m2] * w[j+m2]
    # This is cross-correlation
    N = next_fast_len(m + n)  # need at least m+n-1 = 3n-2
    fa = rfft(a, N)
    fw = rfft(w, N)
    # crosscorr: ifft(conj(fw) * fa) gives sum w[m] a[m+j]... 
    # Actually: ifft(fft(w) * conj(fft(a))) gives correlation
    # Let me think: corr[k] = sum_m w[m] * a[m+k]
    # In FFT: corr = ifft(fft(w, N) * conj(fft(a, N)))

    corr = irfft(fw * np.conj(fa), N)
    g = 2.0 * corr[:n]
    return np.maximum(g, 0.0)

def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=np.float64)
    with np.errstate(divide='ignore', invalid='ignore'):
        cond = (css - target) / idx < u
    rho = np.nonzero(cond)[0]
    if rho.size == 0:
        return np.full(n, target/n)
    theta = (css[rho[-1]] - target) / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 1e-300:
        return (w / s) * target
    return np.full(n, target / n)
