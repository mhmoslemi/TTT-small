import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()

        p = np.ones(198, dtype=np.float64)

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(np.max(b))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(f, n, M, time_limit=0.3):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method="highs", options={"presolve": True,
                                               "time_limit": time_limit,
                                               "output_flag": 0})
    except Exception:
        return None, None
    if res.status not in (0, 3) or res.x is None:
        return None, None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-12:
        return None, None
    y = getattr(res, "unevaluated", None)
    dual = res.ineqlin.marginals if hasattr(res, 'ineqlin') else None
    return g / gs, dual


def _lp_optimize(f0, n, stop_time, seed, max_stall=200):

    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()

    else:
        f = f / fsum

    b = _conv_full(f)
    M = float(b.max())
    t = 0.25
    stall = 0
    iters = 0
    t_min, t_max = 0.03, 0.9

    while time.monotonic() < stop_time - 1.0:
        g, _ = _lp_dir(f, n, M)
        if g is None:
            stall += 3
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.04 + 0.04 * stall / max_stall, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            b = _conv_full(f)
            M = float(b.max())

def _polish(a, stop_time, seed=99):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score_fast(best)
    best = best / fs
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 8

    while time.monotonic() < stop_time - 1.0:
        improved = False
        # 1D coordinate descent
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            if orig < 1e-10:
                cands = [0.001, 0.005, 0.01, 0.05, 0.1, 0.5]
            else:
                cands = [0.0, orig * 0.3, orig * 0.6, orig * 0.8, orig * 1.2,
                         orig * 1.5, orig * 2.0, orig * 3.0]
            bv = orig
            bsc = best_sc
            for cv in cands:
                best[i] = cv
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score_fast(best)
                if sc < bsc - 1e-13:
                    bv = cv
                    bsc = sc
            best[i] = bv
            if bsc < best_sc - 1e-13:
                best_sc = bsc
                improved = True
        # 2-coordinate moves
        for _ in range(60):
            if time.monotonic() > stop_time - 1.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            tot = oi + oj
            if tot < 1e-12:
                continue
            for r in (0.1, 0.3, 0.5, 0.7, 0.9):
                best[i] = r * tot
                best[j] = (1 - r) * tot
                sc = _score_fast(best)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj
        # 3-coordinate
        for _ in range(20):
            if time.monotonic() > stop_time - 1.0:
                break
            idx = rng.choice(n, size=3, replace=False)
            i, j, k = int(idx[0]), int(idx[1]), int(idx[2])
            oi, oj, ok = best[i], best[j], best[k]
            tot = oi + oj + ok
            if tot < 1e-12:
                continue
            for r1 in (0.3, 0.7):
                for r2 in (0.2, 0.5, 0.8):
                    rem = 1 - r1 - r2
                    if rem < -0.01:
                        continue
                    best[i], best[j], best[k] = r1 * tot, r2 * tot, max(0, rem) * tot
                    sc = _score_fast(best)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
            best[i], best[j], best[k] = oi, oj, ok
        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0
    return best, best_sc
