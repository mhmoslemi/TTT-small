import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    if n <= 600:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    """A[k,j] = f[k-j] if 0<=k-j<n, else 0. Shape (2n-1, n)."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=6.0):
    """LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    if M <= 1e-18:
        return None
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-15)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
            options={"presolve": True, "time_limit": timeout,
                     "dual_feasibility_tolerance": 1e-12,
                     "primal_feasibility_tolerance": 1e-12,
                     "dual_edge_tap": 1e-8}
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g


def _lp_optimize(f0, n, stop_time, seed):
    """Main LP-based iterative optimization."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:

    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n)
        f_sum = n

    target_sum = math.sqrt(2.0 * n)
    f = f / f_sum * target_sum

    best_f = f.copy()
    best_score = _score(f)

    t = 0.25
    t_min = 0.02
    t_max = 0.85
    stall = 0
    max_stall = 250
    conv_thresh = 1e-11
    iteration = 0
    lp_fail_count = 0

    while time.monotonic() < stop_time - 5.0:
        iteration += 1
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, n, M)
        if g is None:
            lp_fail_count += 1
            stall += 4
            if stall > max_stall:
                f = best_f + rng.normal(0, 0.04 * max(1.0, best_f.mean()), n)
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                else:
                    f = best_f.copy()
                stall = 0
                max_stall += 80
                t = 0.25
            continue

        lp_fail_count = 0
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            stall += 2
            continue
        g = g * (target_sum / g_sum)

        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))

            reverse=True
        )

def _polish(a, stop_time, seed=4242):
    """Aggressive coordinate-descent polishing."""
    n = len(a)
    target_sum = math.sqrt(2.0 * n)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)
    best = best / fs * target_sum
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 20
    round_num = 0

    while time.monotonic() < stop_time - 5.0 and no_imp < max_no_imp:
        round_num += 1
        improved = False

        # Single-coordinate: golden ratio search around current value
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc

            # Try a grid of multipliers
            multipliers = [0.0, 0.05, 0.1, 0.2, 0.35, 0.5, 0.65, 0.8, 1.2, 1.5, 2.0, 3.0, 5.0]
            for mlt in multipliers:
                nv = orig * mlt if mlt > 0 else 0.0
                best[i] = nv
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-15:
                    best_val = nv
                    best_val_sc = sc

            # Refine around best found (if not at boundary)
            if 0 < best_val < 1e-3 * max(1.0, orig):
                for frac in (0.5, 0.75, 1.25, 1.5):
                    nv = best_val * frac
                    best[i] = nv
                    if best.sum() < 0.01:
                        best[i] = best_val
                        continue
                    sc = _score(best)
                    if sc < best_val_sc - 1e-15:
                        best_val = nv
                        best_val_sc = sc
            elif best_val > 0:
                for frac in (0.9, 1.1, 0.8, 1.2, 1.05, 0.95):
                    nv = orig * frac
                    best[i] = nv
                    if best.sum() < 0.01:
                        best[i] = best_val
                        continue
                    sc = _score(best)
                    if sc < best_val_sc - 1e-15:
                        best_val = nv
                        best_val_sc = sc

            best[i] = best_val
            if best_val_sc < best_sc - 1e-15:
                best_sc = best_val_sc
                improved = True

        # Two-coordinate search
        n_pairs = 300 if round_num < 5 else 150
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 5.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            if oi < 1e-14 and oj < 1e-14:
                continue
            for mi in (0.0, 0.25, 0.5, 0.75, 1.25, 1.75, 0.0):
                for mj in (0.0, 0.25, 0.5, 0.75, 1.25, 1.75, 0.0):
                    if mi == 0.0 and mj == 0.0:
                        continue
                    if mi == 1.0 and mj == 1.0:
                        continue
                    ni = oi * mi if (oi > 1e-14 and mi > 0) else 0.0
                    nj = oj * mj if (oj > 1e-14 and mj > 0) else 0.0
                    best[i], best[j] = ni, nj
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        # Small random perturbations
        if not improved:
            for _ in range(80):
                if time.monotonic() > stop_time - 5.0:
                    break
                k = int(rng.integers(n))
                old = best[k]
                sigma = 0.03 * max(1.0, best.mean())
                best[k] = max(0.0, old + rng.normal(0, sigma))
                if best.sum() < 0.01:
                    best[k] = old
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                else:
                    best[k] = old

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc
