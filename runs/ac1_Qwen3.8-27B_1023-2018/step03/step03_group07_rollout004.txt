def _score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:

    if not np.all(np.isfinite(a)):

def _maxconv(a):
    c = np.convolve(a, a)
    return float(c.max())


def _cache(n):
    n = int(n)
    if n in _CACHE:
        return _CACHE[n]

    m = 2 * n - 1
    k = np.arange(m, dtype=np.int32)[:, None]
    j = np.arange(n, dtype=np.int32)[None, :]
    idx = k - j
    valid = (idx >= 0) & (idx < n)

    rows, cols = np.nonzero(valid)
    rows = rows.astype(np.int32, copy=False)
    cols = cols.astype(np.int32, copy=False)
    validx = idx[valid].astype(np.int32, copy=False)

    slices = []
    for kk in range(m):
        s = max(0, kk - n + 1)
        e = min(n - 1, kk) + 1
        r = kk - np.arange(s, e, dtype=np.int32)
        slices.append((s, e, r))

    _CACHE[n] = (rows, cols, validx, slices)
    return _CACHE[n]


def _build_A_norm(a, M):
    n = int(a.size)
    rows, cols, validx, _ = _cache(n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[rows, cols] = a[validx] / M
    return A


def _lp_direction(a, stop_time, lp_time=0.45):
    if time.monotonic() > stop_time - 0.25:
        return None, 0.0

    n = int(a.size)
    if n <= 1:
        return None, 0.0

    a_lp = np.maximum(a, 1e-12)
    s = float(a_lp.sum())
    if s <= 0.0:
        return None, 0.0
    a_lp = a_lp / s

    c = np.convolve(a_lp, a_lp)
    M = float(c.max())

def _soft_direction(a, c, slices, tau, top=64):
    n = int(a.size)
    m = int(c.size)
    if m <= 0:
        return None

    if tau <= 0.0:
        K = min(top, m)
        p = np.zeros(m, dtype=np.float64)
        if K <= 0:
            return None
        idx = np.arange(m) if K == m else np.argpartition(c, -K)[-K:]
        p[idx] = 1.0
    else:
        M = float(c.max())
        if M <= 0.0:
            return None
        z = (c - M) / tau
        ez = np.exp(z)
        P = float(ez.sum())
        if not np.isfinite(P) or P <= 0.0:
            p = np.zeros(m, dtype=np.float64)
            p[int(np.argmax(c))] = 1.0
        else:
            p = ez / P

        K = min(top, m)
        if K < m:
            idx = np.arange(m) if K == m else np.argpartition(p, -K)[-K:]
            mask = np.zeros(m, dtype=bool)
            mask[idx] = True
            p = p * mask
            P2 = float(p.sum())
            if P2 <= 0.0:
                return None
            p = p / P2

    nz = np.nonzero(p > 1e-14)[0]
    if nz.size == 0:
        return None

    grad = np.zeros(n, dtype=np.float64)
    for kk in nz:
        pk = float(p[kk])

        cand = np.maximum(cand, 0.0)
        ss = float(cand.sum())
        evals += 1
        if ss <= 1e-15:
            continue
        cand = cand / ss
        val = _maxconv(cand)
        if val < best_M - 1e-15:
            best_M = val
            best_step = step
            best_cand = cand

    if best_cand is None:
        return None

    if time.monotonic() < stop_time - 0.1 and evals < max_eval:
        lo = max(0.0, best_step / 1.5)
        hi = min(10.0 * base, best_step * 1.5)
        if hi - lo > 1e-9:
            grid = np.linspace(lo, hi, max(5, max_eval - evals), dtype=np.float64)
            for step in grid:
                if evals >= max_eval or time.monotonic() > stop_time - 0.1:
                    break
                step = float(step)
                cand = a - step * g
                cand = np.maximum(cand, 0.0)
                ss = float(cand.sum())
                evals += 1
                if ss <= 1e-15:
                    continue
                cand = cand / ss
                val = _maxconv(cand)
                if val < best_M - 1e-15:
                    best_M = val
                    best_cand = cand

    if best_cand is not None and best_M < Mcur - 1e-15:
        return best_cand
    return None


def _optimize(a0, stop_time, seed=0, lp_every=4, lp_time=0.45):
    a = _normalize(a0)
    if a is None:
        return a0
    if stop_time <= time.monotonic():
        return a

    n = int(a.size)
    _, _, _, slices = _cache(n)

    c = np.convolve(a, a)
    best_M = float(c.max())
    best_a = a.copy()

    rng = np.random.default_rng(int(seed))
    stuck = 0
    it = 0

    while time.monotonic() < stop_time - 0.5:
        c = np.convolve(a, a)
        M = float(c.max())

        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        improved = False

        if time.monotonic() < stop_time - 1.0:
            tau = max(1e-12, M * 0.10)
            dirg = _soft_direction(a, c, slices, tau, top=64)
            if dirg is not None:
                cand = _line_search_step(a, dirg, M, stop_time, max_eval=20)
                if cand is not None:
                    a = cand
                    improved = True

        if not improved and (stuck > 2 or (it % max(1, int(lp_every))) == 0):
            d, _ = _lp_direction(a, stop_time, lp_time=lp_time)
            if d is not None:
                cand = _line_search_convex(a, d, M, stop_time, max_eval=24)
                if cand is None:
                    dsym = 0.5 * d + 0.5 * d[::-1]
                    dsym = np.maximum(dsym, 0.0)
                    if dsym.sum() > 0:
                        cand = _line_search_convex(a, dsym, M, stop_time, max_eval=18)
                if cand is not None:
                    a = cand
                    improved = True

        if improved:
            it += 1
            continue

        if stuck > 12:
            if stuck % 4 == 0:
                t = min(0.20, 0.02 * (stuck // 4))
                a = (1.0 - t) * a + t / float(n)
            else:
                scale = min(0.30, 0.02 + 0.02 * (stuck // 4))
                a = a * np.exp(rng.normal(0.0, scale, size=n))
            a = np.maximum(a, 1e-14)
            a = a / a.sum()
            stuck = 1

        it += 1
        if it > 1000000:
            break

    out = np.maximum(best_a, 0.0)
    s = float(out.sum())
    if s > 0.0:
        out = out / s
    return out


def _softmax_from_v(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size == 0 or not np.all(np.isfinite(v)):
        return None
    m = float(np.min(v))
    e = np.exp(v - m)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    a = e / s
    s2 = float(a.sum())
    if s2 > 0.0:
        a = a / s2
    return a


def _smooth_opt(a0, stop_time, maxfun=120):
    a = _normalize(a0)
    if a is None:
        return a0

    n = int(a.size)
    best_a = a.copy()
    best_M = _maxconv(a)
    _, _, _, slices = _cache(n)
    v = np.log(np.maximum(a, 1e-30))

    for tau_frac in (0.25, 0.12, 0.06, 0.03):
        if time.monotonic() >= stop_time - 5.0:
            break
        Mcur = _maxconv(best_a)
        tau = max(1e-12, Mcur * tau_frac)

        def obj_grad(vv, tau=tau):
            vv = np.asarray(vv, dtype=np.float64).ravel()
            if vv.size != n or not np.all(np.isfinite(vv)):
                return 1e100, np.zeros(n, dtype=np.float64)

            a_local = _softmax_from_v(vv)
            if a_local is None:
                return 1e100, np.zeros(n, dtype=np.float64)

            c_local = np.convolve(a_local, a_local)
            M_local = float(c_local.max())

            z = (c_local - M_local) / tau
            ez = np.exp(z)
            pz = float(ez.sum())
            if not np.isfinite(pz) or pz <= 0.0:
                return 1e100, np.zeros(n, dtype=np.float64)

            p = ez / pz
            F = M_local + float(np.log(pz))

            grad_a = np.zeros(n, dtype=np.float64)
            nz = np.nonzero(p > 1e-14)[0]
            for kk in nz:
                pk = float(p[kk])
                s_idx, e_idx, r = slices[int(kk)]
                grad_a[s_idx:e_idx] += (2.0 * pk) * a_local[r]

            S1 = float(np.dot(grad_a, a_local))
            S0 = float(grad_a.sum())
            grad_v = a_local * (S1 - S0)

            if not np.isfinite(F):
                F = 1e100
            if not np.all(np.isfinite(grad_v)):
                grad_v = np.zeros(n, dtype=np.float64)
            return F, grad_grad_placeholder(grad_v) if False else grad_v

        # The previous line is intentionally simplified here; remove if present in execution.
        # We redefine safely to avoid any accidental placeholder.
        def obj_grad_safe(vv, tau=tau):
            vv = np.asarray(vv, dtype=np.float64).ravel()
            if vv.size != n or not np.all(np.isfinite(vv)):
                return 1e100, np.zeros(n, dtype=np.float64)

            a_local = _softmax_from_v(vv)
            if a_local is None:
                return 1e100, np.zeros(n, dtype=np.float64)

            c_local = np.convolve(a_local, a_local)
            M_local = float(c_local.max())

            z = (c_local - M_local) / tau
            ez = np.exp(z)
            pz = float(ez.sum())
            if not np.isfinite(pz) or pz <= 0.0:
                return 1e100, np.zeros(n, dtype=np.float64)

            p = ez / pz
            F = M_local + float(np.log(pz))

            grad_a = np.zeros(n, dtype=np.float64)
            nz = np.nonzero(p > 1e-14)[0]
            for kk in nz:
                pk = float(p[kk])
                s_idx, e_idx, r = slices[int(kk)]
                grad_a[s_idx:e_idx] += (2.0 * pk) * a_local[r]

            S1 = float(np.dot(grad_a, a_local))
            S0 = float(grad_a.sum())
            grad_v = a_local * (S1 - S0)

            if not np.isfinite(F):
                F = 1e100
            if not np.all(np.isfinite(grad_v)):
                grad_v = np.zeros(n, dtype=np.float64)
            return F, grad_v

        try:
            res = minimize(
                obj_grad_safe,
                v,
                jac=True,
                method="L-BFGS-B",
                options={"maxiter": 80, "maxfun": maxfun, "ftol": 1e-13, "gtol": 1e-10},
                bounds=[(-30.0, 30.0)] * n,
            )
        except Exception:
            res = None

        if res is not None and getattr(res, "x", None) is not None:
            v_new = np.asarray(res.x, dtype=np.float64).ravel()
            if v_new.size == n and np.all(np.isfinite(v_new)):
                a_new = _softmax_from_v(v_new)
                if a_new is not None:
                    M_new = _maxconv(a_new)
                    if M_new < best_M - 1e-14:
                        best_a = a_new.copy()
                        best_M = M_new
                        v = v_new.copy()

    return best_a


def _resample(a, nt):
    a = np.asarray(a, dtype=np.float64).ravel()
    nt = int(nt)
    if a.size == 0 or nt <= 0:
        return None
    if a.size == nt:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, a.size, endpoint=False, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, nt, endpoint=False, dtype=np.float64)
    out = np.interp(x_new, x_old, a)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 0.0:
        out = np.ones(nt, dtype=np.float64) / float(nt)
    else:
        out = out / s
    return out


def _make_init(n, seed=0):
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64) / max(1, float(n))

    rng = np.random.default_rng(int(seed))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    arr = np.ones(n, dtype=np.float64)

    for _ in range(5):
        c = float(rng.uniform(0.0, 1.0))
        w = float(rng.uniform(0.02, 0.15))
        amp = float(rng.uniform(0.5, 2.0))
        arr += amp * np.exp(-0.5 * ((x - c) / max(w, 1e-3)) ** 2)

    arr = np.maximum(arr, 0.05)
    s = float(arr.sum())
    if s <= 0.0:
        arr = np.ones(n, dtype=np.float64)
        s = float(n)
    return arr / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1300.0
    hard_deadline = t0 + max(5.0, budget - 20.0)

    parent = _get_parent()
    if parent is not None:
        base_n = int(parent.size)
        if base_n > 600:
            parent = _resample(parent, 601)
            base_n = int(parent.size)
        if base_n < 4:
            parent = None
            base_n = 201
    else:
        base_n = 201

    best_a = None
    best_score = float("inf")

    def consider(candidate):
        nonlocal best_a, best_score
        if candidate is None:
            return
        arr = np.asarray(candidate, dtype=np.float64).ravel()
        if arr.size == 0:
            return
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if s < 0.01:
            return
        arr = arr / s
        sc = _score_fast(arr)
        if np.isfinite(sc) and sc < best_score - 1e-15:
            best_a = arr.copy()
            best_score = sc

    if parent is not None:
        consider(parent)
        consider(np.ones(base_n, dtype=np.float64) / float(base_n))
    else:
        consider(_make_init(base_n, int(seed) + 1))
        consider(_make_init(base_n, int(seed) + 2))

    if best_a is None:
        best_a = np.ones(base_n, dtype=np.float64) / float(base_n)
        best_score = _score_fast(best_a)

    stop1 = min(hard_deadline - 35.0, t0 + min(0.45 * budget, 550.0))
    if stop1 > time.monotonic() + 5.0:
        res = _optimize(best_a, stop1, seed=1000 + int(seed), lp_time=0.50)
        consider(res)

    stop2 = min(hard_deadline - 25.0, t0 + min(0.65 * budget, 850.0))
    if stop2 > time.monotonic() + 15.0:
        smooth_stop = stop2 - max(35.0, 0.04 * budget)
        if smooth_stop > time.monotonic() + 20.0:
            res = _smooth_opt(best_a, smooth_stop, maxfun=100)
            consider(res)

        if stop2 > time.monotonic() + 12.0:
            res = _optimize(best_a, stop2, seed=2000 + int(seed), lp_time=0.45)
            consider(res)

    stop3 = min(hard_deadline - 20.0, t0 + min(0.85 * budget, 1150.0))
    if stop3 > time.monotonic() + 120.0 and best_a is not None:
        rng_restart = np.random.default_rng(int(seed) + 7777)
        n_best = int(best_a.size)

        extra_inits = []
        for sc, sd in ((0.03, 7), (0.07, 8), (0.12, 9)):
            arr = best_a * np.exp(rng_restart.normal(0.0, sc, size=n_best))
            arr = np.maximum(arr, 1e-15)
            s = float(arr.sum())
            if s > 0.0:
                extra_inits.append((arr / s, 3000 + int(sd)))

        base_for_resample = parent if parent is not None else best_a
        base_len = int(base_for_resample.size) if base_for_resample is not None else n_best
        nt_opts = []
        seen = {n_best}
        for factor in (0.75, 0.9, 1.1, 1.25, 1.5, 2.0):
            nt = int(round(base_len * factor))
            if 24 <= nt <= 512 and nt not in seen:
                seen.add(nt)
                nt_opts.append(nt)

        for nt in nt_opts:
            if best_a is not None and int(best_a.size) == nt:
                continue
            init = _resample(base_for_resample, nt)
            if init is None:
                init = _make_init(nt, 4000 + int(nt))
            extra_inits.append((init, 5000 + int(nt)))

        if extra_inits:
            remaining = stop3 - time.monotonic()
            per = min(130.0, max(25.0, remaining / float(len(extra_inits) + 1)))
            for arr, sd in extra_inits:
                if time.monotonic() > stop3 - 15.0:
                    break
                alloc_stop = min(stop3 - 5.0, time.monotonic() + per)
                if alloc_stop <= time.monotonic() + 5.0:
                    continue
                arr_size = int(arr.size)
                res = _optimize(arr, alloc_stop, seed=int(sd), lp_time=0.35 if arr_size > 350 else 0.45)
                consider(res)
                if best_score < _TARGET - 1e-9:
                    break

    final_stop = hard_deadline - 5.0
    if best_a is not None and final_stop > time.monotonic() + 20.0:
        remaining = final_stop - time.monotonic()
        if remaining > 70.0:
            mid = min(final_stop - max(30.0, 0.35 * remaining), time.monotonic() + 0.65 * remaining)
            res = _smooth_opt(best_a, mid, maxfun=140)
            consider(res)
            if final_stop > time.monotonic() + 15.0:
                res = _optimize(best_a, final_stop - 2.0, seed=9000 + int(seed), lp_time=0.45)
                consider(res)
        elif final_stop > time.monotonic() + 15.0:
            res = _optimize(best_a, final_stop - 2.0, seed=9100 + int(seed), lp_time=0.30)
            consider(res)

    if best_a is None:
        out = [1.0]
    else:
        best_a = np.maximum(np.asarray(best_a, dtype=np.float64).ravel(), 0.0)
        s = float(best_a.sum())
        if s <= 0.0:
            out = [1.0]
        else:
            best_a = best_a / s
            out = [float(x) for x in best_a]

    ev = globals().get("evaluate_sequence", None)
    if ev is not None:
        try:
            final_score = float(ev(out))
        except Exception:
            final_score = float("inf")
    else:
        final_score = _score_fast(out)

    if not np.isfinite(final_score):
        n_fb = int(best_a.size) if best_a is not None and best_a.size > 0 else base_n
        if n_fb < 1:
            n_fb = 1
        out = [1.0 / float(n_fb) for _ in range(n_fb)]

    return out
