    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs


def _line_search(f, g_dir, best_sc, t_init=0.35, stop_time=None, n_iter=14):
    t_lo, t_hi = 0.0, t_init
    best_t = 0.0
    best_fn = None
    for _ in range(n_iter):
        if stop_time and time.monotonic() > stop_time - 1.0:
            break
        t_mid = 0.5 * (t_lo + t_hi)
        fn = (1.0 - t_mid) * f + t_mid * g_dir
        np.maximum(fn, 0.0, out=fn)
        fs = float(fn.sum())
        if fs < 1e-12:
            t_hi = t_mid
            continue
        sc = _score(fn)
        if sc < best_sc - 1e-15:
            t_lo = t_mid
            best_t = t_mid
            best_fn = fn.copy()
        else:
            t_hi = t_mid
        if t_hi - t_lo < 5e-5:
            break
    if best_t < 1e-6:
        return None, None
    return best_t, best_fn


def _lp_loop(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs

    best_f = f.copy()
    best_sc = _score(f)
    t_cur = 0.30
    stall = 0
    max_stall = 35
    it = 0

    while time.monotonic() < stop_time - 2.0:
        it += 1
        b = _conv_fft(f)
        M = float(b.max())
        if M < 1e-15:
            break

        # Adaptive time limit for LP: start generous, reduce if fast
        tl = 10.0 if it < 20 else max(4.0, 8.0 - it * 0.01)
        g_norm = _lp_solve(f, n, M, time_limit=tl)

        if g_norm is None:
            stall += 8
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
            f = f / max(f.sum(), 1e-10)
            t_cur = 0.25
            continue

        dist = float(np.linalg.norm(g_norm - f))
        if dist < 1e-7:
            stall += 4
            if stall > max_stall:
                break
            f = np.clip(f * (1.0 + rng.normal(0, 0.08, n)), 0, None)
            f = f / max(f.sum(), 1e-10)
            t_cur = 0.30
            continue

        res_t, f_new = _line_search(f, g_norm, best_sc,
                                    t_init=min(0.6, t_cur * 1.15),
                                    stop_time=stop_time)
        if f_new is not None:
            sc_new = _score(f_new)
            if sc_new < best_sc - 1e-13:
                f = f_new
                best_f = f.copy()
                best_sc = sc_new
                t_cur = min(0.65, res_t * 1.12)
                stall = 0
            else:
                t_cur = max(0.03, res_t * 0.55)
                stall += 1
        else:
            t_cur = max(0.02, t_cur * 0.4)
            stall += 1

        if stall >= max_stall:
            # Larger perturbation to escape
            f = np.clip(f * (1.0 + rng.normal(0, 0.15, n)) + rng.normal(0, 0.02, n), 0, None)
            f = f / max(f.sum(), 1e-10)
            t_cur = 0.30
            stall = 0

        # Early exit if very good
        if best_sc < 1.5025 and it > 50:
            break

    return best_f, best_sc


def _subgrad_polish(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f = f / max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = _score(f)
    lr = 0.008
    momentum = np.zeros(n)
    beta = 0.82
    no_imp = 0

    while time.monotonic() < stop_time - 2.0:
        b = _conv_fft(f)
        M = float(b.max())
        # Use top-7 lags for richer gradient
        topk = min(7, 2 * n - 1)
        idx = np.argpartition(b, -topk)[-topk:]
        w = b[idx].copy()
        w -= w.min() + 1e-12
        w /= w.sum()

        grad_M = np.zeros(n)
        for wt, k in zip(w, idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                j = k - i
                if 0 <= j < n:
                    grad_M[i] += wt * 2.0 * f[j]

        s = float(f.sum())
        grad_F = (2.0 * n / (s * s)) * grad_M - (4.0 * n * M / (s ** 3)) * np.ones(n)

        momentum = beta * momentum - lr * grad_F
        f_new = f + momentum
        np.maximum(f_new, 0.0, out=f_new)
        f_new = f_new / max(f_new.sum(), 1e-10)

        sc = _score(f_new)
        if sc < best_sc - 1e-14:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            lr = min(0.04, lr * 1.008)
        else:
            no_imp += 1
            lr = max(0.0005, lr * 0.975)
            if no_imp > 250:
                f = f_new.copy()
                no_imp = 0
                lr = 0.006
                beta = 0.7
                momentum = np.zeros(n)

    return best_f, best_sc


def _coord_polish(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f = f / max(f.sum(), 1e-10)
    best_sc = _score(f)
    no_imp = 0

    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                return f, best_sc
            orig = f[i]
            # Multiplicative moves
            for mlt in (0.3, 0.5, 0.7, 1.4, 1.8, 2.5, 0.0):
                f[i] = orig * mlt if orig > 1e-12 else (0.005 if mlt > 0 else 0.0)
                sc = _score(f)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
                    orig = f[i]
                else:
                    f[i] = max(0.0, orig)
            # Fine additive
            if f[i] > 1e-8:
                base = f[i]
                for fr in (0.95, 1.05, 0.9, 1.1):
                    f[i] = base * fr
                    sc = _score(f)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        improved = True
                        base = f[i]
        # Pair moves
        for _ in range(60):
            if time.monotonic() > stop_time - 2.0:
                return f, best_sc
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            for mi, mj in ((1.3, 0.7), (0.7, 1.3), (1.5, 0.8), (0.8, 1.5), (1.2, 0.9), (0.9, 1.2)):
                f[i] = max(0.0, oi * mi)
                f[j] = max(0.0, oj * mj)
                sc = _score(f)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
                    oi, oj = f[i], f[j]
            f[i], f[j] = oi, oj

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 8:
                break

    return f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 25.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = math.inf
    try:
        best_score = _score(best_seq)
    except Exception:
        pass
    if not math.isfinite(best_score):
        best_seq = np.ones(n_p).tolist()
        best_score = _score(np.array(best_seq))

    def _try_update(cand):
        nonlocal best_seq, best_score
        if cand is None:
            return
        cand = np.clip(np.asarray(cand, dtype=np.float64).ravel(), 0, 1000)
        if cand.size == 0 or cand.sum() < 0.01:
            return
        try:
            sc = _score(cand)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_seq = cand.copy()
            best_score = sc

    # Phase 1: LP-based optimization (55% budget)
    lp_end = t0 + budget * 0.55
    n = n_p
    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    inits = []
    inits.append(("parent", parent.copy()))
    # Cosine bump
    inits.append(("cos1", 0.5 + 0.5 * np.cos(2 * math.pi * i_arr / n)))
    # Multi-frequency superposition
    sup = np.full(n, 0.15)
    for fr in (1, 2, 3, 5, 8, 13):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.25 * np.abs(np.cos(2 * math.pi * fr * i_arr / n + ph))
    inits.append(("multifreq", sup))
    # Parent + structured noise
    noise_scale = 0.3 * max(1.0, float(np.mean(parent)))
    inits.append(("pnoise", np.abs(parent + rng_main.normal(0, noise_scale, n)) + 0.01))
    # Random
    inits.append(("rand", np.abs(rng_main.normal(0, 1, n)) + 0.02))

    time_per_init = max(12.0, (lp_end - t0 - 8) / len(inits))
    for idx, (name, init) in enumerate(inits):
        if time.monotonic() > lp_end - 5.0:
            break
        stop = min(lp_end - 4.0, time.monotonic() + time_per_init)
        try:
            res, sc = _lp_loop(init, n, stop, seed + 10000 * (idx + 1))
            _try_update(res)
        except Exception:
            pass

    # Phase 2: Subgradient polish (15% budget)
    sg_end = min(deadline, t0 + budget * 0.72)
    if time.monotonic() < sg_end - 8.0:
        try:
            res, sc = _subgrad_polish(best_seq, len(best_seq), sg_end, seed + 99001)
            _try_update(res)
        except Exception:
            pass

    # Phase 3: Try a few other lengths if still above target
    other_end = min(deadline, t0 + budget * 0.85)
    if time.monotonic() < other_end - 10.0 and best_score > 1.503:
        other_lengths = [256, 320, 448, 512]
        for L in other_lengths:
            if time.monotonic() > other_end - 5.0:
                break
            if L == n_p:
                continue
            stop = min(other_end - 4.0, time.monotonic() + 30.0)
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n_p), parent)
            try:
                res, sc = _lp_loop(init_l, L, stop, seed + 50000 + L)
                _try_update(res)
            except Exception:
                pass

    # Phase 4: Coordinate polish (remaining time)
    polish_end = deadline - 12.0
    if time.monotonic() < polish_end - 8.0:
        try:
            res, sc = _coord_polish(best_seq, len(best_seq), polish_end, seed + 77001)
            _try_update(res)
        except Exception:
            pass

    # Final subgradient burst
    if time.monotonic() < deadline - 10.0:
        try:
            res, sc = _subgrad_polish(best_seq, len(best_seq), deadline - 8.0, seed + 33001)
            _try_update(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        sc = evaluate_sequence(final_list)
        if math.isfinite(sc) and sc > 0:
            return final_list
    except Exception:
        pass

    # Fallback
    fb = [float(max(0.0, min(1000.0, x))) for x in parent]
    try:
        sc = evaluate_sequence(fb)
        if math.isfinite(sc) and sc > 0:
            return fb
    except Exception:
        pass

    return [1.0] * max(1, n_p)
