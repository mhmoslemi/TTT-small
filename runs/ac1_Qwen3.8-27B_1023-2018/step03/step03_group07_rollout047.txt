    # ============ Final validation and return ============
    best_a = np.clip(best_a, 0.0, 1000.0)
    result = [float(v) for v in best_a]
    try:
        fs = evaluate_sequence(result)
    except Exception:
        fs = float('inf')

    if not np.isfinite(fs) or fs > best_score + 1e-10:
        # Fallback to a safe sequence
        result = [1.0] * 32
        fs = evaluate_sequence(result)
        if not np.isfinite(fs):
            result = [1.0 / 8.0] * 8

    return result
