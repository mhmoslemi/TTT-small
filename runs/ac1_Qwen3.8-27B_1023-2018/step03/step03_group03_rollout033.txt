            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 4.0:
                break

            v = f[i]
            S_rest = S - v
            k_start = i
            k_end = min(i + n, 2 * n - 1)
            infl = f[:k_end - k_start]  # f[0], f[1], ..., f[k_end-1-k_start]

            cur_max = float(conv.max())

            # Build candidate x values
            x_cands = [0.0, v]

            # Analytical optimum for the current peak
            k_star = int(np.argmax(conv))
            if k_start <= k_star < k_end:
                fi = f[k_star - i]
                if k_star == 2 * i:
                    # Quadratic: conv[k*] + 2w*fi + w^2, denominator (S-v+w+v)^2 = (S+w)^2
                    # where w = x - v
                    # Minimize (conv[k*] + 2w*fi + w^2) / (S+w)^2
                    # d/dw = [2(fi+w)(S+w) - (conv[k*]+2w*fi+w^2)] / (S+w)^3 = 0
                    # 2fi*S + 2fi*w + 2wS + 2w^2 - conv[k*] - 2w*fi - w^2 = 0
                    # w^2 + 2S*w + 2fi*S - conv[k*] = 0
                    disc = 4 * S * S - 4 * (2 * fi * S - conv[k_star])
                    if disc >= 0:
                        w_opt = -S + math.sqrt(disc) / 2.0
                        # w^2 + 2Sw + (2fi*S - conv[k*]) = 0
                        # w = -S + sqrt(S^2 - 2fi*S + conv[k*])
                        disc2 = S * S - 2.0 * fi * S + conv[k_star]
                        if disc2 >= 0:
                            x_cands.append(v + (-S + math.sqrt(disc2)))
                else:
                    if fi > 1e-15:
                        # Linear: conv[k*] + 2w*fi, denominator (S-v+w)^2 = (S_rest+x)^2
                        # w = x-v, so denominator = (S_rest+v+x-v)^2 = (S_rest+x)^2... 
                        # Actually S_rest + x = S - v + x = S + w
                        # Minimize (conv[k*] + 2(x-v)*fi) / (S-v+x)^2
                        # d/dx: 2fi*(S-v+x)^2 - (conv[k*]+2(x-v)*fi)*2*(S-v+x) = 0
                        # fi*(S-v+x) = conv[k*] + 2(x-v)*fi
                        # fi*S - fi*v + fi*x = conv[k*] + 2fi*x - 2fi*v
                        # fi*S + fi*v - 2fi*x = conv[k*]
                        # x = (fi*(S+v) - conv[k*]) / (2*fi)
                        x_opt = (fi * (S + v) - conv[k_star]) / (2.0 * fi)
                        if x_opt >= 0:
                            x_cands.append(x_opt)

            # Additional candidates
            if v > 1e-8:
                x_cands.extend([0.3 * v, 0.7 * v, 1.3 * v, 1.8 * v])

            # Filter and deduplicate
            x_cands = [max(0.0, min(x, 1000.0)) for x in x_cands]
            x_cands = sorted(set(round(x, 15) for x in x_cands if math.isfinite(x)))

            best_x = v
            best_obj = 2.0 * n * cur_max / (S * S)

            for x in x_cands:
                S_new = S_rest + x
                if S_new < 0.01:
                    continue
                w = x - v
                if abs(w) < 1e-14:
                    m_new = cur_max
                else:
                    conv[k_start:k_end] += 2.0 * w * infl
                    if 2 * i <= 2 * n - 2:
                        conv[2 * i] += w * w
                    m_new = float(conv.max())
                    conv[k_start:k_end] -= 2.0 * w * infl
                    if 2 * i <= 2 * n - 2:
                        conv[2 * i] -= w * w
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_x = x

            # Local refinement via bisection around best_x
            if best_x != v:
                span = max(0.01, abs(best_x) * 0.15)
                x_lo = max(0.0, best_x - span)
                x_hi = best_x + span
                for _ in range(25):
                    if time.monotonic() > stop_time - 4.0:
                        break
                    x_mid = 0.5 * (x_lo + x_hi)
                    S_new = S_rest + x_mid
                    if S_new < 0.01:
                        x_lo = x_mid
                        continue
                    w = x_mid - v
                    conv[k_start:k_end] += 2.0 * w * infl
                    if 2 * i <= 2 * n - 2:
                        conv[2 * i] += w * w
                    m_new = float(conv.max())
                    obj = 2.0 * n * m_new / (S_new * S_new)
                    conv[k_start:k_end] -= 2.0 * w * infl
                    if 2 * i <= 2 * n - 2:
                        conv[2 * i] -= w * w
                    if obj < best_obj:
                        x_hi = x_mid
                        best_obj = obj
                        best_x = x_mid
                    else:
                        x_lo = x_mid

            if best_x != v and best_obj < best_score - 1e-14:
                w = best_x - v
                f[i] = best_x
                S += w
                conv[k_start:k_end] += 2.0 * w * infl
                if 2 * i <= 2 * n - 2:
                    conv[2 * i] += w * w
                best_score = best_obj
                best_f = f.copy()
                improved = True

        if not improved:
            break

    # Final verification
    f = np.clip(best_f, 0, 1000.0)
    final_sc = _score(f)
    return f, final_sc


def _subgradient_polish(f, stop_time, max_iters=5000, seed=12345):
    """Subgradient method for the max of autocorrelation."""
    n = len(f)
    f = np.clip(np.asarray(f, dtype=np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, math.inf
    f = f / S  # normalize to sum 1

    best_f = f.copy()
    best_score = 2.0 * n * _conv_fast(f).max()

    rng = np.random.default_rng(seed)
    step = 1.0 / (n * 0.1)

    for it in range(max_iters):
        if time.monotonic() > stop_time - 3.0:
            break
        conv = _conv_fast(f)
        M = float(conv.max())
        active = np.where(conv >= M * 0.95)[0]
        
        # Subgradient: for each active k, grad_k[j] = f[k-j] + f[j-k] (with 0 for out of range)
        # Average over active set
        grad = np.zeros(n)
        for k in active:
            for j in range(n):
                ki = k - j
                if 0 <= ki < n:
                    grad[j] += f[ki]
                jk = j - k
                if 0 <= jk < n:
                    grad[j] += f[jk]
        grad /= max(1, len(active))

        # Line search
        best_t = 0
        best_sc = best_score
        for t in [step, step * 0.5, step * 0.25, step * 2.0]:
            f_new = f - t * grad
            f_new = np.clip(f_new, 0, None)
            fs = f_new.sum()
            if fs < 0.01:
                continue
            f_new = f_new / fs
            sc = 2.0 * n * _conv_fast(f_new).max()
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_t = t

        if best_t > 0:
            f = np.clip(f - best_t * grad, 0, None)
            fs = f.sum()
            if fs > 0.01:
                f = f / fs
                if best_sc < best_score:
                    best_score = best_sc
                    best_f = f.copy()
            step = min(step * 1.3, 0.5)
        else:
            step *= 0.7
            if step < 1e-8:
                break

    best_f = best_f / best_f.sum() * best_f.sum()  # keep as is
    final_sc = _score(best_f)
    return best_f, final_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    hard_deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    # Fallback
    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(64)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update_best(seq, sc):
        nonlocal best_seq, best_score
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score:
            best_seq = np.array([float(x) for x in seq], dtype=np.float64).copy()
            best_score = exact

    def _try_lp_and_polish(n, init_arr, lp_end, polish_end, sd):
        nonlocal best_seq, best_score
        if time.monotonic() > lp_end:
            return
        try:
            res, sc = _lp_optimize(init_arr, lp_end, sd)
        except Exception:
            res, sc = init_arr, _score(init_arr)
        if res is None or not np.all(np.isfinite(res)) or res.sum() < 0.01:
            return
        res = np.clip(res, 0, 1000.0)
        if sc < best_score:
            _update_best(res, sc)
        if time.monotonic() < polish_end - 10:
            try:
                res2, sc2 = _coord_descent(res, polish_end, max_sweeps=150, seed=sd + 1)
            except Exception:
                res2, sc2 = res, sc
            if np.all(np.isfinite(res2)) and res2.sum() > 0.01:
                _update_best(res2, sc2)

    # === Phase 1: LP + coordinate descent on parent ===
    rng_main = np.random.default_rng(seed)
    n = n_p

    inits = [parent.copy()]
    # Cosine taper
    i_arr = np.arange(n, dtype=np.float64)
    inits.append(np.clip(0.2 + 0.8 * np.cos(np.pi * i_arr / (n - 1)) ** 2, 0, None))
    # Slight asymmetric
    inits.append(np.clip(0.5 + 0.3 * np.cos(2 * np.pi * i_arr / n) + 0.2 * np.cos(4 * np.pi * i_arr / n), 0, None))

    phase1_end = t0 + budget * 0.25
    time_per_init = max(15.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 5.0:
            break
        if best_score < 1.503:
            break
        stop_lp = min(phase1_end - 3.0, time.monotonic() + time_per_init * 0.6)
        stop_pol = min(hard_deadline, stop_lp + budget * 0.10)
        _try_lp_and_polish(n, init, stop_lp, stop_pol, seed + 1000 * (idx + 1))

    # === Phase 2: Intensive coordinate descent ===
    phase2_end = t0 + budget * 0.55
    if time.monotonic() < phase2_end - 20.0:
        try:
            best_seq, best_score = _coord_descent(best_seq, phase2_end, max_sweeps=300, seed=424242)
        except Exception:
            pass
        _update_best(best_seq, best_score)

    # Second round of coordinate descent with different seed (Shuffling)
    if time.monotonic() < phase2_end - 10.0:
        try:
            best_seq, best_score = _coord_descent(best_seq, phase2_end, max_sweeps=200, seed=99999)
        except Exception:
            pass
        _update_best(best_seq, best_score)

    # === Phase 3: Try extended lengths ===
    phase3_end = t0 + budget * 0.75
    other_lengths = sorted(set([192, 256, 320, 384, 448, 512, 576, 640]))
    # Sort by proximity to current best length
    other_lengths.sort(key=lambda L: abs(L - len(best_seq)))

    time_per_len = max(10.0, (phase3_end - time.monotonic()) / max(1, len(other_lengths) - 1))

    for L in other_lengths:
        if time.monotonic() > phase3_end - 10.0:
            break
        if best_score < 1.5028:
            break
        if L == len(best_seq):
            continue
        stop_lp = min(phase3_end - 5.0, time.monotonic() + time_per_len * 0.5)
        stop_pol = min(hard_deadline, stop_lp + time_per_len * 0.8)

        # Interpolate best to new length
        if len(best_seq) > 1:
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
        else:
            init_l = np.ones(L) / L
        init_l = np.clip(init_l, 0, None)
        if init_l.sum() < 1e-6:
            init_l = np.ones(L) / L

        _try_lp_and_polish(L, init_l, stop_lp, stop_pol, seed + 50000 + L)

    # === Phase 4: Final intensive polish ===
    final_end = hard_deadline - 10.0
    if time.monotonic() < final_end - 15.0:
        try:
            best_seq, best_score = _coord_descent(best_seq, final_end, max_sweeps=200, seed=31337)
        except Exception:
            pass
        _update_best(best_seq, best_score)

    # One more pass with different ordering
    if time.monotonic() < final_end - 8.0:
        try:
            best_seq, best_score = _coord_descent(best_seq, final_end, max_sweeps=100, seed=77777)
        except Exception:
            pass
        _update_best(best_seq, best_score)

    # Final validation and return
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
