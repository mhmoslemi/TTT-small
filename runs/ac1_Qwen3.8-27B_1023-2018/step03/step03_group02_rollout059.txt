    cw = 0.5 + 0.5 * np.cos(np.pi * i_arr / (n - 1))
    alt_starts.append(cw / cw.sum())
    tri = 1.0 - np.abs(2.0 * i_arr / (n - 1) - 1.0)
    alt_starts.append((tri + 0.01) / (tri + 0.01).sum())
    bump = np.exp(-((i_arr - n * 0.25) ** 2 + (i_arr - n * 0.75) ** 2) / (2.0 * (n * 0.12) ** 2)) + 0.005
    alt_starts.append(bump / bump.sum())
    # Smooth random
    base_r = rng.normal(0, 1, n)
    kern_r = np.exp(-0.5 * (np.arange(-6, 7) / 2.5) ** 2)
    kern_r /= kern_r.sum()
    smooth_r = np.convolve(base_r, kern_r, 'same')
    alt_starts.append(np.abs(smooth_r) / np.abs(smooth_r).sum() * 0.8 + 0.2 * f0)
    alt_starts[-1] = alt_starts[-1] / alt_starts[-1].sum()

    t2_start = max(p1_end, time.monotonic())
    time_per = max(8.0, (p2_end - t2_start) / max(1, len(alt_starts) - 1))

    for sidx in range(1, len(alt_starts)):
        remaining = p2_end - time.monotonic()
        if remaining < 8.0:
            break
        stop_s = time.monotonic() + min(time_per + 5.0, remaining - 3.0)
        if time.monotonic() >= stop_s:
            break
        try:
            res, sc = _lp_descent(alt_starts[sidx], stop_s, np.random.default_rng(seed + 7000 + sidx * 31))
            if sc < best_score:
                best_f = res.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 3: Coordinate + pair polish (30%)
    p3_end = deadline
    if time.monotonic() < p3_end - 12.0:
        polish_time = p3_end - time.monotonic()
        n_rounds = min(5, max(2, int(polish_time / 80)))
        for r in range(n_rounds):
            seg_time = polish_time / (n_rounds - r)
            p_stop = time.monotonic() + seg_time
            if p_stop > p3_end:
                p_stop = p3_end
            if time.monotonic() >= p_stop - 2.0:
                break
            try:
                best_f, best_score = _coord_polish(best_f, p_stop, np.random.default_rng(seed + 30000 + r * 29))
            except Exception:
                pass

    # Final pair-coordinate refine
    if time.monotonic() < deadline - 8.0:
        try:
            best_f, best_score = _pair_coord(best_f, deadline - 4.0, np.random.default_rng(seed + 99999))
        except Exception:
            pass

    # Final validation
    final_seq = np.clip(best_f, 0, 1000).astype(np.float64)
    final_list = [float(x) for x in final_seq]
    try:
        final_sc = float(evaluate_sequence(final_list))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc) or final_sc > 2.0:
        parent_clipped = np.clip(parent, 0, 1000)
        try:
            psc = float(evaluate_sequence([float(x) for x in parent_clipped]))
        except Exception:
            psc = math.inf
        if math.isfinite(psc):
            return [float(x) for x in parent_clipped]
        return [1.0]

    return final_list
