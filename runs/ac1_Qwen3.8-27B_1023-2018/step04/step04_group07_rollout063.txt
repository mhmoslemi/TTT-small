def _grad_optimize(a0, n, stop_time, seed=777):

    best_sc = _score_fast(a)

    eps_sm = 0.05

    while time.monotonic() < stop_time - 1.0:

        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[c.argmax()] = 1.0
        else:
            w /= w_sum

        A_fft = rfft(a, L_corr)
        W_fft = rfft(w, L_corr)
        grad = 2.0 * irfft(A_fft * np.conj(W_fft), L_corr)[:n].copy()

        # Momentum

        step = lr * momentum / (np.linalg.norm(momentum) + 1e-12)
        a_new = a - step

                best[i] = frac * total
                best[j] = (1.0 - frac) * total
                sc = _score_fast(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc


def _block_polish(a0, stop_time, seed=555):
    """Polish using block redistributions."""
    n = len(a0)
    best = np.clip(np.asarray(a0, dtype=np.float64).ravel().copy(), 0, None)
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    block_sizes = [2, 3, 4, 5, 8, 12]

    while time.monotonic() < stop_time - 1.0 and no_imp < 5:
        improved = False
        for bs in block_sizes:
            step = max(1, bs // 3)
            for start in range(0, n - bs + 1, step):
                if time.monotonic() > stop_time - 1.0:
                    break
                if start + bs > n:
                    break
                block = best[start:start + bs].copy()
                total = block.sum()
                if total < 1e-12:
                    continue
                for trial in range(4):
                    w = rng.dirichlet(np.ones(bs) * 1.5)
                    best[start:start + bs] = w * total
                    sc = _score_fast(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
                best[start:start + bs] = block
        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 30.0  # 30s headroom for return

    # Get parent
    try:
        h = globals()["height_sequence_1"]
        parent = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(384, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(384, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    n = len(parent)

    # Fallback
    fallback = parent.copy()
    try:
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(n)
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))

    best_seq = fallback.copy()
    best_score = fallback_sc

    rng_main = np.random.default_rng(seed)

    # Phase 1: LP optimization from parent (50% of time)
    ph1_end = t0 + budget * 0.50
    init_p = parent.copy()
    init_p = np.clip(init_p, 0, None)
    s = init_p.sum()
    if s > 1e-10:
        init_p /= s

    try:
        res, sc = _lp_optimize(init_p, n, ph1_end, seed + 104729, focused_frac=0)
        if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
            res_c = np.clip(res, 0, 1000)
            exact = float(evaluate_sequence([float(x) for x in res_c]))
            if math.isfinite(exact) and exact < best_score:
                best_seq = res_c.copy()
                best_score = exact
    except Exception:
        pass

    # Phase 2: LP from best + gradient optimization (25% of time)
    ph2_end = t0 + budget * 0.75
    remaining = ph2_end - time.monotonic()
    if remaining > 10 and best_score > 1.5028:
        # Try a few more LP starts
        inits = []
        # Best so far
        inits.append(best_seq.copy())
        # Taper
        i_arr = np.arange(n, dtype=np.float64)
        taper = 0.5 + 0.5 * np.cos(math.pi * i_arr / (n - 1))
        inits.append(taper)
        # Parent with structured perturbation
        kern = np.exp(-0.5 * (np.arange(-8, 9) / 3.0) ** 2)
        kern /= kern.sum()
        smooth = np.convolve(rng_main.normal(0, 1, n), kern, 'same')
        p_n = np.abs(parent + 0.2 * smooth)
        inits.append(p_n)

        time_per = max(8.0, remaining / len(inits))
        for idx, init in enumerate(inits):
            if time.monotonic() > ph2_end - 5:
                break
            if best_score < 1.5028:
                break
            init_c = np.clip(init, 0, None)
            s_init = init_c.sum()
            if s_init > 1e-10:
                init_c /= s_init
            stop = min(ph2_end - 3, time.monotonic() + time_per)
            if time.monotonic() >= stop:
                break
            try:
                res, sc = _lp_optimize(init_c, n, stop, seed + 200000 + idx * 9973)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res_c = np.clip(res, 0, 1000)
                    exact = float(evaluate_sequence([float(x) for x in res_c]))
                    if math.isfinite(exact) and exact < best_score:
                        best_seq = res_c.copy()
                        best_score = exact
            except Exception:
                pass

    # Phase 3: Gradient-based optimization (15% of time)
    ph3_end = t0 + budget * 0.88
    if time.monotonic() < ph3_end - 5 and best_score > 1.5027:
        grad_stop = min(ph3_end, time.monotonic() + (ph3_end - time.monotonic()))
        try:
            best_seq, best_score = _grad_optimize(best_seq, n, grad_stop, seed=4242)
            if not np.all(np.isfinite(best_seq)) or best_seq.sum() < 0.01:
                best_seq, best_score = _grad_optimize(np.clip(best_seq, 0, 1000) / 100, n, grad_stop, seed=4243)
        except Exception:
            pass

    # Phase 4: Coordinate descent polish
    ph4_end = deadline - 2.0
    polish_time = max(5.0, (ph4_end - time.monotonic()) * 0.5)
    if time.monotonic() < ph4_end - 10:
        cs_stop = time.monotonic() + polish_time
        if cs_stop > ph4_end:
            cs_stop = ph4_end
        try:
            best_seq, best_score = _coord_descent(best_seq, cs_stop, seed=88888)
        except Exception:
            pass

    # Phase 5: Block polish
    if time.monotonic() < ph4_end - 5:
        try:
            best_seq, best_score = _block_polish(best_seq, ph4_end, seed=66666)
        except Exception:
            pass

    # Final coordinate descent (short)
    final_stop = t0 + budget - 10.0
    if time.monotonic() < final_stop - 3:
        try:
            best_seq, best_score = _coord_descent(best_seq, final_stop, seed=99999)
        except Exception:
            pass

    # Final validation with exact evaluator
    final_seq = np.clip(best_seq, 0, 1000)
    final_seq = final_seq.astype(np.float64)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        final_seq = fallback

    if math.isfinite(final_sc) and final_sc < best_score:
        best_score = final_sc

    # Ensure valid return
    final_seq = np.clip(np.asarray(final_seq, dtype=np.float64).ravel(), 0, 1000)
    if len(final_seq) == 0:
        return [1.0]
    return [float(x) for x in final_seq]
