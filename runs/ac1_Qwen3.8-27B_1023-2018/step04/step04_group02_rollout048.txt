            if stall > 100:
                break

        return math.inf
    return 2.0 * n * m / (s * s)


def _score_exact(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _lp_direction(f, n, M, timeout=3.0):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0"""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _grad_smooth(f, T=None):
    """Compute gradient of smooth max of conv(f,f). Returns (grad, max_val, weights)."""
    n = len(f)
    b = _conv_fft(f)
    if T is None:
        T = 50.0 / max(float(b.max() - b.min()), 1e-10)
        T = min(T, 500.0)
    Tmax = b.max()
    Tb = T * (b - Tmax)
    w = np.exp(Tb)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(b))
        w[int(np.argmax(b))] = 1.0
    else:
        w /= ws
    # grad[p] = 2 * sum_j w[p+j] * f[j] for p = 0,...,n-1
    grad = np.zeros(n)
    for p in range(n):
        grad[p] = 2.0 * np.dot(w[p:p + n], f)
    return grad, float(Tmax), w


def _lp_flash(f0, stop_time, seed, max_iter=50000):
    """LP Frank-Wolfe optimization."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(f0.astype(np.float64).copy(), 0, None)
    if f.sum() < 1e-10:
        f = np.ones(n)
    
    best_f = f.copy()
    best_sc = _score(f)
    t_step = 0.4
    stall = 0
    it = 0
    
    while time.monotonic() < stop_time - 5 and it < max_iter:
        b = _conv_fft(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        g = _lp_direction(f, n, M, timeout=2.0)
        if g is None:
            stall += 5
            if stall > 100:
                f = np.clip(f + rng.normal(0, 0.02 * max(1.0, f.std()), n), 0, None)
                if f.sum() < 1e-10:
                    f = best_f.copy()
                stall = 0
            it += 1
            continue
        
        g_sum = g.sum()
        g = g * (f.sum() / g_sum)
        
        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < 1e-9:
            stall += 3
            if stall > 150:
                f = np.clip(f + rng.normal(0, 0.03 * max(1.0, f.mean()), n), 0, None)
                if f.sum() < 1e-10:
                    f = best_f.copy()
                stall = 0
                t_step = 0.4
            it += 1
            continue
        
        improved = False
        for alpha in [t_step, t_step * 0.6, t_step * 0.3, t_step * 1.3, t_step * 0.15]:
            f_new = np.clip((1.0 - alpha) * f + alpha * g, 0, None)
            if f_new.sum() < 1e-10:
                continue
            sc = _score(f_new)
            if sc < best_sc - 1e-15:
                f = f_new
                best_f = f.copy()
                best_sc = sc
                t_step = min(0.9, alpha * 1.1)
                stall = 0
                improved = True
                break
            elif sc < _score(f) - 1e-15:
                f = f_new
                t_step = min(0.9, alpha)
                stall = 0
                improved = True
                break
        
        if not improved:
            t_step = max(0.01, t_step * 0.5)
            stall += 1
        
        if stall > 150:
            f = np.clip(best_f + rng.normal(0, 0.04 * max(1.0, best_f.mean()), n), 0, None)
            if f.sum() < 1e-10:
                f = best_f.copy()
            t_step = 0.4
            stall = 0
        
        it += 1
    
    return best_f, best_sc


def _grad_refine(f0, stop_time, seed):
    """Gradient-based refinement using soft-max of convolution."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(f0.astype(np.float64).copy(), 0, None)
    S = f.sum()
    if S < 1e-10:
        f = np.ones(n)
        S = n
    f = f / S
    
    best_f = f.copy()
    best_sc = _score(f)
    
    lr = 0.5
    T = 100.0
    patience = 0
    max_patience = 300
    
    while time.monotonic() < stop_time - 5:
        grad, bmax, w = _grad_smooth(f, T)
        
        # Project gradient to sum=1 hyperplane
        grad -= grad.mean()
        
        # Adaptive learning rate
        fnorm = np.linalg.norm(f)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            # Scale lr so step is at most lr_factor of f
            lr_eff = lr * fnorm / gnorm
        else:
            lr_eff = 0.0
        
        f_new = f - lr_eff * grad / fnorm if fnorm > 1e-15 else f
        f_new = np.clip(f_new, 0, None)
        if f_new.sum() < 1e-10:
            lr *= 0.5
            continue
        f_new = f_new / f_new.sum()
        
        sc = _score(f_new)
        if sc < best_sc - 1e-15:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            patience = 0
            lr = min(1.0, lr * 1.02)
        else:
            patience += 1
            lr *= 0.95
            if patience > max_patience:
                # Anneal temperature
                T = max(5.0, T * 0.7)
                if T <= 5.0:
                    # Add noise
                    f = np.clip(f + rng.normal(0, 0.005, n), 0, None)
                    f = f / f.sum()
                    patience = 0
                    T = 50.0
                    max_patience = min(max_patience + 100, 800)
        
        # Slowly anneal T
        T = max(5.0, T * 0.9995)
    
    return best_f, best_sc


def _cd_polish(f0, stop_time, seed, max_outer=30):
    """Coordinate descent polish."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = f0.copy()
    cur = _score(f)
    best = f.copy()
    best_sc = cur
    
    mults = [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.0]
    
    for outer in range(max_outer):
        if time.monotonic() > stop_time - 3.0:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = f[i]
            best_v = orig
            best_local = cur
            for m in mults:
                v = orig * m if m > 0 else 0.0
                f[i] = v
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_local:
                    best_local = sc
                    best_v = v
            f[i] = best_v
            if best_local < cur - 1e-15:
                cur = best_local
                improved = True
                if cur < best_sc:
                    best_sc = cur
                    best = f.copy()
        if not improved:
            break
    
    return best, best_sc
