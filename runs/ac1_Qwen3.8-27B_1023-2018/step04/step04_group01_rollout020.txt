from scipy.optimize import linprog

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    return p

def _conv_full(a):
    n = len(a)
    if n <= 1024:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """Build constraint matrix for LP: A[k,j] = f[j] if 0<=k-j<n else 0. Shape (2n-1, n)."""
    # A[k, j] = f[j] when 0 <= k-j < n, i.e., k-n+1 <= j <= k
    # Let's build it using the structure
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_direction(f, n, time_limit=5.0):
    """Find LP direction: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    b = _conv_full(f)
    M = float(b.max())
    if M < 1e-15:
        return None, M
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None, M
    if res.status != 0 or res.x is None:
        return None, M
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-15:

    return g / g_sum, M
