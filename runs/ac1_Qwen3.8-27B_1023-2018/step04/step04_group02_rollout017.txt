def _conv(a):

    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    return irfft(F * F, fsize)[:2 * n - 1]

    # Phase 4: Try nearby lengths (7%)
    phase4_end = t0 + budget * 0.94
    if time.monotonic() < phase4_end - 15.0:
        for L in [n - 8, n + 8, n - 16, n + 16]:
            if L < 64 or L > 600 or time.monotonic() > phase4_end - 12.0:
                continue
            try:
                init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
                _try_opt(L, init_l, min(phase4_end - 4.0, time.monotonic() + 30.0), seed + 90000 + L)
            except Exception:
                pass

    # Phase 5: Final CD with remaining time
    final_end = deadline - 8.0
    if time.monotonic() < final_end - 10.0:
        try:
            f_c, sc_c = _cd_polish(best_seq.copy(), final_end, max_outer=100, seed=seed * 7 + 42)
        except Exception:
            f_c, sc_c = best_seq.copy(), best_score
        if sc_c < best_score - 1e-12:
            exact = _exact_score(f_c)
            if math.isfinite(exact) and exact < best_score:
                best_seq = f_c
                best_score = exact

    # Final validation with exact evaluator
    try:
        final_sc = _exact_score(best_seq)
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = _exact_score(fallback)
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0] * 200
        return [float(x) for x in fallback]

    # Sanity check
    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    if not result:
        result = [1.0]
    if any(math.isnan(x) or math.isinf(x) for x in result):
        result = [float(x) for x in fallback]

    return result
