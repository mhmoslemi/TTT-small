    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            v = f[k - j]
            if v > 1e-20:
                rows.append(k)
                cols.append(j)
                vals.append(v)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_direction(f, M, timeout=4.0, use_all=True):
    n = len(f)
    A = _build_lp_matrix(f, n)
    
    if use_all:
        b_ub = np.full(2 * n - 1, M)
        A_use = A
    else:
        c = _conv_full(f)
        thresh = M * 0.7
        keep = c >= thresh
        if keep.sum() < 10:
            idx = np.argsort(c)[-10:]
            keep[idx] = True
        A_use = A[keep, :]
        b_ub = np.full(A_use.shape[0], M)
    
    c_vec = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_vec, A_ub=A_use, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _optimize_lp(f0, stop_time, seed, use_all=True):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n)
        fs = n
    best_f = f.copy()
    best_score = _score(f)

    t = 0.25
    stall = 0
    max_stall = 80
    lp_timeout = 3.0

    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, M, timeout=lp_timeout, use_all=use_all)
        if g is None:
            stall += 5
            if stall > max_stall:
                noise = rng.normal(0, 0.06 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                fs2 = f.sum()
                if fs2 > 1e-10:
                    f = f / fs2 * fs
                t = 0.25
                stall = 0
                max_stall = min(max_stall + 40, 300)
            continue

        g_sum = g.sum()
        f_sum = f.sum()
        if g_sum < 1e-15:
            stall += 3
            continue
        g = g * (f_sum / g_sum)

        dist = np.linalg.norm(g - f) / max(1.0, np.linalg.norm(f))
        if dist < 1e-9:
            stall += 2
            if stall > max_stall:
                break
            continue

        improved = False
        t_trials = sorted(set([
            max(0.01, min(0.95, t)),
            max(0.01, min(0.95, t * 0.5)),
            max(0.01, min(0.95, t * 1.5)),
            max(0.01, min(0.95, t * 0.25)),
            max(0.01, min(0.95, t * 2.0)),
            0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5
        ]), key=lambda x: -abs(x - t))[:10]

        for t_try in t_trials:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.clip(f_new, 0, None)
            if f_new.sum() < 0.01:
                continue
            sc = _score(f_new)
            if sc < best_score - 1e-14:
                f = f_new
                best_f = f.copy()
                best_score = sc
                t = t_try
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            t = max(0.02, t * 0.7)
            if stall >= max_stall:
                noise = rng.normal(0, 0.1 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                fs2 = f.sum()
                if fs2 > 1e-10:
                    f = f / fs2 * best_f.sum()
                t = 0.25
                stall = 0
                max_stall = min(max_stall + 40, 400)

    return best_f, best_score


def _grad_refine(f0, stop_time, seed):
    """Gradient-based refinement using penalty on peak convolution values."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f = f / f.sum()
    best_f = f.copy()
    best_score = _score(f)
    
    lr = 0.02
    iters = 0
    max_iters = 5000
    
    while time.monotonic() < stop_time - 3.0 and iters < max_iters:
        b = _conv_full(f)
        M = float(b.max())
        
        # Use top 20% of peaks
        thresh = M * 0.8
        peak_idx = np.where(b >= thresh)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])
        
        # Compute gradient of max penalty
        grad = np.zeros(n)
        for k in peak_idx:
            excess = b[k] - M * 0.5  # soft penalty
            if excess <= 0:
                continue
            # d c[k] / d a[i] = 2 * a[k-i] if 0 <= k-i < n
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for i in range(j_start, j_end + 1):
                grad[i] += 2.0 * excess * f[k - i]
        
        grad_scale = np.linalg.norm(grad)
        if grad_scale < 1e-15:
            break
        
        # Multiplicative update ( maintains nonnegativity )
        f_new = f * np.exp(-lr * grad / max(1e-10, grad_scale))
        f_new = np.clip(f_new, 1e-15, None)
        f_new = f_new / f_new.sum()
        
        sc = _score(f_new)
        if sc < best_score - 1e-14:
            best_f = f_new.copy()
            best_score = sc
            f = f_new
            lr = min(lr * 1.02, 0.1)
        else:
            # Try smaller step
            f_try = f * (1 - 0.5 * lr * grad / max(1e-10, grad_scale))
            f_try = np.clip(f_try, 1e-15, None)
            f_try = f_try / f_try.sum()
            sc_try = _score(f_try)
            if sc_try < best_score - 1e-14:
                best_f = f_try.copy()
                best_score = sc_try
                f = f_try
            lr *= 0.8
        
        iters += 1
        if iters % 100 == 0:
            # Occasional perturbation
            noise = rng.normal(0, 0.005, n)
            f = np.clip(f + noise, 1e-15, None)
            f = f / f.sum()
    
    return best_f, best_score


def _polish(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    best_sc = _score(f)
    no_imp = 0
    max_no_imp = 20

    mults_coarse = (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 2.5, 0.0)
    mults_fine = (0.9, 1.1, 0.85, 1.15, 0.95, 1.05, 0.92, 1.08)

    while time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            if f[i] < 1e-14:
                f[i] = 0.005 * max(1.0, f.mean())
                sc = _score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                else:
                    f[i] = 0.0
                continue

            orig = f[i]
            best_local = best_sc
            best_val = orig

            for m in mults_coarse:
                if m == 0.0:
                    f[i] = 0.0
                else:
                    f[i] = orig * m
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_val = f[i]

            if best_local < best_sc - 1e-15:
                f[i] = best_val
                orig = best_val
                for m in mults_fine:
                    f[i] = orig * m
                    if f.sum() < 0.01:
                        f[i] = orig
                        continue
                    sc = _score(f)
                    if sc < best_local - 1e-15:
                        best_local = sc
                        best_val = f[i]
                    else:
                        f[i] = orig
                f[i] = best_val
                best_sc = best_local
                improved = True
            else:
                f[i] = best_val

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return f.copy(), best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = _score(best_seq)

    def _update_best(candidate):
        nonlocal best_seq, best_score
        if candidate is None or len(candidate) == 0:
            return
        candidate = np.clip(np.asarray(candidate, dtype=np.float64), 0, 1000)
        if not np.all(np.isfinite(candidate)):
            return
        if candidate.sum() < 0.01:
            return
        sc = _score(candidate)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq = candidate.copy()
            best_score = sc

    rng = np.random.default_rng(seed)
    i_arr = np.arange(n_p, dtype=np.float64)

    # Generate diverse initializations
    inits = [parent.copy()]

    # Noisy versions
    for sig in (0.03, 0.08, 0.15):
        inits.append(np.clip(parent + rng.normal(0, sig * max(parent.std(), 0.1), n_p), 0, None))

    # Structural inits
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / n_p)) + 0.05)
    center = (n_p - 1) / 2.0
    inits.append(np.exp(-0.5 * ((i_arr - center) / (n_p * 0.3)) ** 2) + 0.02)
    inits.append(np.exp(-0.5 * ((i_arr - n_p * 0.3) / (n_p * 0.12)) ** 2) +
                 np.exp(-0.5 * ((i_arr - n_p * 0.7) / (n_p * 0.12)) ** 2) + 0.02)
    inits.append(np.abs(np.sin(np.pi * i_arr / n_p)) ** 0.5 + 0.03)
    # Tapered uniform
    taper = np.minimum(i_arr, n_p - 1 - i_arr)
    t_norm = np.clip(taper / (n_p * 0.1), 0, 1)
    inits.append(t_norm + 0.1)

    # Phase 1: LP optimization with all constraints (45% time)
    phase1_end = t0 + budget * 0.45
    time_avail = phase1_end - t0 - 15.0
    time_per_init = max(20.0, time_avail / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        if time.monotonic() > stop - 5.0:
            continue
        try:
            res, sc = _optimize_lp(init, stop, seed + 1000 * (idx + 1), use_all=True)
            _update_best(res)
        except Exception:
            pass

    # Phase 2: Refined LP on best (15% time)
    phase2_end = t0 + budget * 0.60
    if time.monotonic() < phase2_end - 15.0:
        stop = phase2_end - 5.0
        try:
            res, sc = _optimize_lp(best_seq, stop, seed + 888888, use_all=True)
            _update_best(res)
        except Exception:
            pass
        # Also try with subset of constraints for diversity
        if time.monotonic() < phase2_end - 20.0:
            try:
                res, sc = _optimize_lp(best_seq, phase2_end - 5.0, seed + 777777, use_all=False)
                _update_best(res)
            except Exception:
                pass

    # Phase 3: Gradient refinement (12% time)
    phase3_end = t0 + budget * 0.72
    if time.monotonic() < phase3_end - 10.0:
        try:
            res, sc = _grad_refine(best_seq, phase3_end - 5.0, seed + 666666)
            _update_best(res)
        except Exception:
            pass

    # Phase 4: Another LP pass after gradient (8% time)
    phase4_end = t0 + budget * 0.80
    if time.monotonic() < phase4_end - 10.0:
        try:
            res, sc = _optimize_lp(best_seq, phase4_end - 5.0, seed + 555555, use_all=True)
            _update_best(res)
        except Exception:
            pass

    # Phase 5: Coordinate descent polish (12% time)
    phase5_end = t0 + budget * 0.92
    if time.monotonic() < phase5_end - 10.0:
        try:
            res, sc = _polish(best_seq, phase5_end - 5.0, seed + 444444)
            _update_best(res)
        except Exception:
            pass

    # Phase 6: Final LP + polish on best (6% time)
    phase6_end = deadline
    if time.monotonic() < phase6_end - 20.0:
        try:
            res, sc = _optimize_lp(best_seq, phase6_end - 15.0, seed + 333333, use_all=True)
            _update_best(res)
        except Exception:
            pass
        if time.monotonic() < phase6_end - 10.0:
            try:
                res, sc = _polish(best_seq, phase6_end - 5.0, seed + 222222)
                _update_best(res)
            except Exception:
                pass

    # Try a couple of nearby lengths if time remains
    if time.monotonic() < deadline - 30.0:
        for L in [n_p + 32, n_p - 32]:
            if time.monotonic() > deadline - 20.0:
                break
            if L < 150 or L > 700:
                continue
            try:
                x_old = np.linspace(0, 1, n_p)
                x_new = np.linspace(0, 1, L)
                init_L = np.interp(x_new, x_old, best_seq)
                if init_L.sum() < 0.01:
                    init_L = np.interp(x_new, x_old, parent)
                stop = min(deadline - 15.0, time.monotonic() + 25.0)
                res, sc = _optimize_lp(init_L, stop, seed + L * 100, use_all=True)
                _update_best(res)
            except Exception:
                pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    return [float(x) for x in best_seq]
