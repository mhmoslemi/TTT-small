The issue is that as we go through the sweep, b changes. So when we get to coordinate j (after having already updated coordinate i), the M_j should reflect the current b. If we precomputed M_j before the sweep, it might be stale.

For correctness, we should recompute M_j from the current b when we optimize coordinate j. This costs O(n) per coordinate.

But actually, for the purpose of finding the best x for coordinate j, we need:
score(x) = 2*n*max(M_j, m_j(x)) / (S + x - v_j)^2

where M_j = max of current b[k] for k outside [j, j+n-1], and m_j(x) = max of changed b[k] for k in [j, j+n-1] as a function of x.

If M_j is stale (too low), we might think a candidate x is better than it really is. But since we verify the actual score after making the change, this is just an efficiency issue (we might accept a change that doesn't actually improve the score, but we'll detect it when we recompute the score after the update).

For efficiency, I'll compute M_j on the fly: M_j = max(b[:j], b[j+n:]) (with proper handling of empty ranges). This is O(n) per coordinate but can be done with numpy's max which is fast.

                fine_tm = np.maximum(fine_aff_max, M)
                fine_new_S = 1.0 + fine_w
                fine_scores = 2.0 * n * fine_tm / (fine_new_S * fine_new_S)
                fine_scores[~np.isfinite(fine_scores)] = math.inf
                fj = int(np.argmin(fine_scores))
                if float(fine_scores[fj]) < best_val:
                    best_j = -1  # marker
                    best_val = float(fine_scores[fj])
                    w_best = float(fine_w[fj])
                else:
                    w_best = float(w[best_j])
            else:
                w_best = float(w[best_j])

            if math.isfinite(best_val) and best_val < cur_score - 1e-14:
                p[i] = v + w_best
                S += w_best
                b[start:end] += w_best * infl
                di_abs = 2 * i
                if 0 <= di_abs < Ltot:
                    b[di_abs] += w_best * w_best
                cur_score = best_val
                improved = True

        # Rescale
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            b *= inv * inv
            S = 1.0

        # Track best
        act_score = float(2.0 * n * b.max())
        if act_score < best_score - 1e-15:
            best_score = act_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 5.0:
                    sigma = 0.02 + 0.02 * min(flat // 4, 5)
                    p *= np.exp(rng.normal(0.0, sigma, n, dtype=np.float64))
                    p = np.maximum(p, 0.0)
                    S = float(p.sum())
                    if S > 0:
                        p /= S
                        S = 1.0
                        b = _conv_self(p)
                        cur_score = float(2.0 * n * b.max())
                    flat = 0
                else:
                    break
        sweep += 1

    # Final
    p = np.maximum(p, 0.0)
    S = float(p.sum())
    if S > 1e-300:
        p /= S
    sc = float(2.0 * n * _conv_self(p).max())
    if sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _lbfgs_lse(p, stop_time, consider=None):
    p = np.maximum(p.astype(np.float64), 0.0).copy()
    n = len(p)
    S = float(p.sum())
    if S < 1e-300:
        return p
    p = p / S

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-5 * cmax)
        scaled = (c - cmax) / T
        # Clip for numerical stability
        scaled = np.clip(scaled, -500, 500)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w_ = np.zeros(c.size)
            w_[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * math.log(sum_exp / c.size)
            w_ = exps / sum_exp
        q = xfast(pp, w_)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    # Temperature stages
    stages = [
        (0.05, 0.12),
        (0.02, 0.15),
        (0.008, 0.18),
        (0.003, 0.20),
        (0.001, 0.18),
        (0.0003, 0.17),
    ]

    start = time.monotonic()
    avail = max(0.0, stop_time - start - 5.0)
    if avail < 10:
        return p

    best_p_local = p.copy()
    best_f_local = _score_arr(p)
    total_w = sum(w for _, w in stages)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += wt
        end_stage = min(stop_time - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue

        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        T = max(1e-300, frac * cmax)

        # Multiple L-BFGS restarts
        stale = 0
        while time.monotonic() < end_stage - 1.5:
            opts = {"maxiter": 25, "ftol": 1e-15, "gtol": 1e-10, "maxls": 15, "maxcor": 15, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_f_local:
                    best_f_local = fval
                    u = res.x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if getattr(res, "status", 1) == 0:
                break

        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 0:
            p_cur /= ss
        sc = _score_arr(p_cur)
        if math.isfinite(sc):
            if sc < best_f_local:
                best_f_local = sc
                best_p_local = p_cur.copy()
                p = p_cur.copy()
            else:
                p = p_cur.copy()  # still use as starting point
            u = p_to_u(p)
            if consider is not None:
                try:
                    consider(best_p_local)
                except Exception:
                    pass
        else:
            p = best_p_local.copy()
            u = p_to_u(p)

    return best_p_local


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 15.0
    if hard <= t0 + 2:
        hard = t0 + max(2.0, budget - 1.0)

    parent = _get_parent()
    n = parent.size
    best_arr = parent.copy()
    best_arr = np.maximum(best_arr, 0.0)
    S0 = float(best_arr.sum())
    if S0 < 0.01:
        best_arr = np.ones(64) / 64.0
    best_arr = best_arr / float(best_arr.sum())
    best_score = _score_arr(best_arr)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            sc = _score_arr(a)
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_arr = (a / s).copy()
                best_score = sc
        except Exception:
            pass

    # Phase 1: Quick coordinate descent on parent
    phase1_end = min(hard, t0 + 60.0)
    if time.monotonic() < phase1_end - 5:
        try:
            p1, _ = _coord_descend(best_arr, phase1_end, consider=consider, seed=seed + 1)
            if p1.size == n:
                consider(p1)
        except Exception:
            pass

    # Phase 2: LSE + L-BFGS
    phase2_end = min(hard, t0 + 60.0 + 180.0)
    if time.monotonic() < phase2_end - 10:
        try:
            p2 = _lbfgs_lse(best_arr, phase2_end, consider=consider)
            if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p2 = p2 / p2.sum()
                consider(p2)
        except Exception:
            pass

    # Phase 3: Extended coordinate descent (bulk of the time)
    phase3_end = min(hard - 30.0, t0 + budget - 25.0)
    if time.monotonic() < phase3_end - 5:
        try:
            p3, _ = _coord_descend(best_arr, phase3_end, consider=consider, seed=seed + 999, G=32)
            if p3.size == n:
                consider(p3)
        except Exception:
            pass

    # Phase 4: Fine coordinate descent with more grid points
    phase4_end = min(hard - 10.0, t0 + budget - 20.0)
    if time.monotonic() < phase4_end - 5:
        try:
            p4, _ = _coord_descend(best_arr, phase4_end, consider=consider, seed=seed + 7777, G=48)
            if p4.size == n:
                consider(p4)
        except Exception:
            pass

    # Phase 5: Final LSE polish
    phase5_end = min(hard - 5.0, t0 + budget - 18.0)
    if time.monotonic() < phase5_end - 10:
        try:
            p5 = _lbfgs_lse(best_arr, phase5_end, consider=consider)
            if p5 is not None and p5.size == n and np.all(np.isfinite(p5)) and p5.sum() > 0:
                p5 = p5 / p5.sum()
                consider(p5)
        except Exception:
            pass

    # Final coordinate descent squeeze
    final_end = min(hard - 3.0, t0 + budget - 16.0)
    if time.monotonic() < final_end - 3:
        try:
            p6, _ = _coord_descend(best_arr, final_end, consider=consider, seed=seed + 55555, G=40)
            if p6.size == n:
                consider(p6)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or out.sum() < 0.01:
            out = np.ones(64) / 64.0
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            ls = [float(x) for x in np.ones(64) / 64.0]
            if not math.isfinite(evaluate_sequence(ls)):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
