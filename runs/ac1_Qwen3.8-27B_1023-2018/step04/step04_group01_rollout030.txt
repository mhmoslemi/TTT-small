def _conv_fft(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = next_fast_len(2*n - 1)
    F = rfft(a, size)
    c = irfft(F * F, size)[:2*n-1]
    return c

def _score(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return math.inf
    b = _conv_fft(a)
    return 2.0 * n * b.max() / (S * S)

def _score_direct(a):
    """Score without FFT, for small n or when we already have conv."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return math.inf
    b = np.convolve(a, a)
    return 2.0 * n * b.max() / (S * S)

def _lp_solve(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = f[j] if 0 <= k-j <= n-1, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,

    if res.status != 0 or res.x is None:
        return None
    g = np.array(res.x, dtype=np.float64)
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs

        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-10:
        return None
    return g / gs


def _lp_phase(f0, n, stop_time, seed=42):
    """LP-based optimization with line search."""
    rng = np.random.default_rng(seed)
    f = f0.copy().astype(np.float64)
    S = f.sum()
    f = f / S
    best_f = f.copy()
    best_score = _score_arr(f)

    n_iter = 0
    stall = 0
    max_stall = 30

    while time.monotonic() < stop_time - 5:
        n_iter += 1
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, M, n)
        if g is None:
            stall += 1
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.01, n), 0, None)
            if f.sum() > 1e-10:
                f /= f.sum()
            else:
                f = best_f.copy()
            continue

        dist = float(np.linalg.norm(g - f))
        if dist < 1e-8:
            stall += 1
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
            if f.sum() > 1e-10:
                f /= f.sum()
            continue

        # Line search: try several step sizes
        cur_score = _score_arr(f)
        best_t = 0.0
        best_sc = cur_score
        for t in [0.3, 0.5, 0.7, 0.85, 1.0, 0.15, 0.1]:
            f_try = f + t * (g - f)
            f_try = np.clip(f_try, 0, None)
            ss = f_try.sum()
            if ss < 1e-10:
                continue
            f_try = f_try / ss
            sc = _score_arr(f_try)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_t = t

        if best_t > 1e-6 and best_sc < best_score - 1e-14:
            f_new = f + best_t * (g - f)
            f_new = np.clip(f_new, 0, None)
            f_new /= f_new.sum()
            f = f_new
            best_f = f.copy()
            best_score = best_sc
            stall = 0
        elif best_t > 1e-6 and best_sc < cur_score - 1e-12:
            f_new = f + best_t * (g - f)
            f_new = np.clip(f_new, 0, None)
            f_new /= f_new.sum()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                if f.sum() > 1e-10:
                    f /= f.sum()
                else:
                    f = best_f.copy()
                stall = 0
                if time.monotonic() > stop_time - 10:
                    break

        if n_iter % 200 == 199 and stall > 10:
            f = np.clip(f + rng.normal(0, 0.015, n), 0, None)
            if f.sum() > 1e-10:
                f /= f.sum()
            else:
                f = best_f.copy()
            stall = 5

    return best_f, best_score


def _coord_descent(a0, stop_time, max_sweeps=200, seed=42):
    """Fast coordinate descent exploiting the quadratic structure."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 0.01:
        return a, math.inf
    a = a / S
    conv = _compute_conv(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 3:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3:
                break

            v = a[i]
            k_start = i
            k_end = min(i + n, 2 * n - 1)
            k_self = 2 * i

            if k_start >= len(conv):
                continue

            # M_away: max of conv outside [k_start, k_end-1]
            if k_start == 0 and k_end >= len(conv):
                M_away = -1.0
            elif k_start == 0:
                M_away = float(conv[k_end:].max()) if k_end < len(conv) else -1.0
            elif k_end >= len(conv):
                M_away = float(conv[:k_start].max()) if k_start > 0 else -1.0
            else:
                left_max = float(conv[:k_start].max()) if k_start > 0 else -1.0
                right_max = float(conv[k_end:].max()) if k_end < len(conv) else -1.0
                M_away = max(left_max, right_max)

            # Slice of conv being affected
            seg = conv[k_start:k_end]  # length = k_end - k_start = n (usually)
            seg_len = k_end - k_start

            # Influence: for changing a[i] from v to x (delta = x - v):
            #   seg[k] += 2*delta*a[k] for k != i (relative index)
            #   seg[i] (the self position) additionally += delta^2
            #   (since the true change at self is (x-v)*(x+v) = 2*v*delta + delta^2)

            # Quick evaluation function
            def eval_at(x):
                delta = x - v
                if abs(delta) < 1e-15:
                    return 2.0 * n * conv.max()
                # Compute new max
                new_seg = seg + 2.0 * delta * a
                if k_self >= k_start and k_self < k_end:
                    idx_self = k_self - k_start
                    new_seg[idx_self] += delta * delta
                affected_max = float(new_seg.max())
                new_max = max(M_away, affected_max)
                S_new = 1.0 + delta
                if S_new < 0.01:
                    return math.inf
                return 2.0 * n * new_max / (S_new * S_new)

            # Determine x range
            x_lo = 0.0
            x_hi = max(2.0 * v, 0.05)
            x_hi = min(x_hi, 10.0)

            # Coarse grid
            n_pts = 100
            if x_hi < x_lo + 1e-12:
                x_hi = x_lo + 0.01
            x_grid = np.linspace(x_lo, x_hi, n_pts)

            best_x = v
            best_x_sc = 2.0 * n * conv.max()

            for x in x_grid:
                sc = eval_at(x)
                if sc < best_x_sc - 1e-15:
                    best_x_sc = sc
                    best_x = x

            # Fine grid around best
            if best_x != v:
                dw = x_hi / n_pts
                fl = max(x_lo, best_x - 4 * dw)
                fh = min(x_hi, best_x + 4 * dw)
                x_fine = np.linspace(fl, fh, 30)
                for x in x_fine:
                    if abs(x - best_x) < dw * 0.01:
                        continue
                    sc = eval_at(x)
                    if sc < best_x_sc - 1e-15:
                        best_x_sc = sc
                        best_x = x

            # Check if improvement
            cur_score = 2.0 * n * conv.max()
            if best_x_sc < cur_score - 1e-14 and best_x >= 0:
                x = best_x
                delta = x - v
                # Update conv
                conv[k_start:k_end] += 2.0 * delta * a
                if k_self >= k_start and k_self < k_end:
                    idx_self = k_self - k_start
                    conv[k_self] += delta * delta
                a[i] = x
                improved = True

        # End of sweep: check and possibly renormalize
        S_now = a.sum()
        if S_now > 0.01:
            a = a / S_now
            conv = _compute_conv(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()

        if not improved and sweep > 5:
            # Try perturbation
            f = np.clip(a + rng.normal(0, 0.005, n), 0, None)
            if f.sum() > 0.01:
                a = f / f.sum()
                conv = _compute_conv(a)
            else:
                a = best_a.copy()
                conv = _compute_conv(a)
        elif not improved and sweep > 15:
            break

    best_a = best_a / best_a.sum()
    return best_a, _score_arr(best_a)


def _grad_phase(f0, n, stop_time, seed=42):
    """Gradient descent with smoothed max."""
    rng = np.random.default_rng(seed)
    a = f0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_arr(a)

    alpha = 0.03
    tau = 60.0
    n_iter = 0

    while time.monotonic() < stop_time - 5:
        n_iter += 1
        b = _conv_full(a)
        M = float(b.max())
        if M < 1e-18:
            break

        # Smoothed max gradient
        b_shifted = b / (tau * M)
        b_shifted -= b_shifted.max()
        p = np.exp(b_shifted)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(2 * n - 1)
            p[int(np.argmax(b))] = 1.0
        else:
            p /= p_sum

        # grad of max(conv) w.r.t. a[i] = 2 * conv(p, a_rev)[i + n - 1]
        a_rev = a[::-1]
        size = next_fast_len(2 * n - 1)
        P = rfft(p, size)
        Ar = rfft(a_rev, size)
        conv_p = irfft(P * Ar, size)[:2 * n - 1]
        grad = 2.0 * conv_p[n - 1:2 * n - 1]

        # Score gradient: d(2n*M/S^2)/da = (2n/S^2) * (grad - 2*M/S * a)
        S = a.sum()
        score_grad = (2.0 * n / (S * S)) * (grad - 2.0 * M / S * a)

        # Gradient step with projection onto {a >= 0, sum(a) = 1}
        a_new = a - alpha * score_grad
        a_new = np.clip(a_new, 0, None)
        sn = a_new.sum()
        if sn < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        sc_new = _score_arr(a_new)
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()

        cur_score = _score_arr(a)
        if sc_new < cur_score - 1e-14:
            a = a_new
            alpha = min(0.05, alpha * 1.01)
            tau = max(10.0, tau * 0.995)
        else:
            alpha = max(1e-6, alpha * 0.7)
            # Perturbation to escape
            if alpha < 1e-5:
                a = np.clip(a + rng.normal(0, 0.01, n), 0, None)
                a /= a.sum()
                alpha = 0.01

        if n_iter % 500 == 499:
            a = np.clip(a + rng.normal(0, 0.008, n), 0, None)
            a /= a.sum()
            alpha = 0.015

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0  # 20s headroom

    # Get parent
    try:
        parent_list = list(height_sequence_1)
        parent = np.array(parent_list, dtype=np.float64)
    except Exception:
        parent = np.ones(384, dtype=np.float64)

    if parent.size == 0 or not np.all(np.isfinite(parent)):
        parent = np.ones(384, dtype=np.float64)
    parent = np.clip(parent, 0, 1000)

    n = parent.size
    a_init = parent / parent.sum()

    # Fallback
    best_a = a_init.copy()
    best_score = _score_arr(a_init)

    def try_update(cand):
        nonlocal best_a, best_score
        if cand is None:
            return
        cand = np.asarray(cand, dtype=np.float64).ravel()
        cand = np.clip(cand, 0, 1000)
        if cand.size < 1 or cand.sum() < 0.01:
            return
        sc = _score_arr(cand)
        if not math.isfinite(sc):
            return
        # Verify with exact evaluator
        try:
            exact = float(evaluate_sequence([float(x) for x in cand]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score:
            best_a = cand.copy()
            best_score = exact

    # === Phase 1: LP optimization ===
    t_lp_end = min(t0 + budget * 0.30, deadline - 120)
    if time.monotonic() < t_lp_end - 10:
        try:
            res, sc = _lp_phase(a_init, n, t_lp_end, seed=seed)
            try_update(res)
        except Exception:
            pass

    # === Phase 2: Coord descent on best ===
    t_cd1_end = min(t0 + budget * 0.50, deadline - 80)
    if time.monotonic() < t_cd1_end - 5:
        try:
            res, sc = _coord_descent(best_a, t_cd1_end, max_sweeps=300, seed=seed + 1)
            try_update(res)
        except Exception:
            pass

    # === Phase 3: More LP (from improved point) ===
    t_lp2_end = min(t0 + budget * 0.60, deadline - 60)
    if time.monotonic() < t_lp2_end - 10 and best_score > 1.5028:
        try:
            res, sc = _lp_phase(best_a, n, t_lp2_end, seed=seed + 100)
            try_update(res)
        except Exception:
            pass

    # === Phase 4: Gradient phase ===
    t_grad_end = min(t0 + budget * 0.72, deadline - 50)
    if time.monotonic() < t_grad_end - 5 and best_score > 1.5028:
        try:
            res, sc = _grad_phase(best_a, n, t_grad_end, seed=seed + 200)
            try_update(res)
        except Exception:
            pass

    # === Phase 5: Intensive coord descent ===
    t_cd2_end = min(t0 + budget * 0.88, deadline - 30)
    if time.monotonic() < t_cd2_end - 5:
        try:
            res, sc = _coord_descent(best_a, t_cd2_end, max_sweeps=500, seed=seed + 300)
            try_update(res)
        except Exception:
            pass

    # === Phase 6: Final LP polish ===
    t_lp3_end = deadline - 15
    if time.monotonic() < t_lp3_end - 10 and best_score > 1.5028:
        try:
            res, sc = _lp_phase(best_a, n, t_lp3_end, seed=seed + 500)
            try_update(res)
        except Exception:
            pass

    # === Phase 7: Final coord descent ===
    if time.monotonic() < deadline - 5:
        try:
            res, sc = _coord_descent(best_a, deadline - 3, max_sweeps=100, seed=seed + 700)
            try_update(res)
        except Exception:
            pass

    # Final validation
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.clip(best_a, 0, 1000)
    best_a = np.where(np.isfinite(best_a), best_a, 0.0)
    if best_a.size < 1 or best_a.sum() < 0.01:
        best_a = a_init.copy()

    # Verify final answer
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_a]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in a_init]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in a_init]

    return [float(x) for x in best_a]
