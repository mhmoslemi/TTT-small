- For k = n: n-1 (j=1,...,n-1)

import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def _fft_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fa = rfft(a, size=size)
    return irfft(fa * fa, size=size)

def _score(a):
    """Compute the exact score matching the evaluator."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    c = _fft_conv(a)
    return 2.0 * n * float(c.max()) / (s * s)

def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a)
    return a / s

def _build_A(n, a):
    """Build (2n-1) x n convolution matrix."""
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        lo = j  # k-j >= 0 => k >= j
        hi = min(2*n-1, j+n)  # k-j < n => k < j+n
        A[lo:hi, j] = a[:hi-lo]
    return A

def _lp_solved(A, M, n, time_limit=2.0):
    """Solve max 1^T g s.t. A g <= M, g >= 0. Returns (g, dual_weights) or (None, None)."""
    c_obj = -np.ones(n)
    b = np.full(2*n-1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n,
                      method='highs',
                      options={'time_limit': time_limit,

                               'primal_feasibility_tolerance': 1e-11,
                               'presolve': True})
        if res.status == 0 and res.x is not None:

            # Try to get dual (reduced costs for inequalities)
            dual = None
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:

    fw = rfft(w, size=size)
    corr = irfft(fa * fw, size=size)

    bc_max = bc.max()
    w = np.exp(bc - bc_max)
    w /= w.sum()
    grad = _gradient_from_weights(w, a, n)
    return grad, w

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def _fft_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fa = rfft(a, size=size)
    return irfft(fa * fa, size=size)


def _score_fast(a):
    """Fast score matching the exact evaluator."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    c = _fft_conv(a)
    return 2.0 * n * float(c.max()) / (s * s)


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a)
    return a / s


def _build_A(n, a):
    """Build (2n-1) x n convolution matrix A[k,j] = a[k-j] if valid."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = lo + min(n, 2 * n - 1 - j)
        A[lo:lo + min(n, hi - lo), j] = a[:hi - lo]
    return A


def _lp_solve(A, M, n, tl=2.0):
    """Solve max sum(g) s.t. Ag <= M, g >= 0. Returns (g, dual) or (None, None)."""
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tl,
                               'dual_feasibility_tolerance': 1e-11,
                               'primal_feasibility_tolerance': 1e-11})
        if res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0, None)
            dual = None
            try:
                if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                    mg = res.ineqlin.marginals
                    if mg is not None and len(mg) == 2 * n - 1:
                        dual = np.abs(np.asarray(mg, dtype=np.float64))
            except Exception:
                pass
            return g, dual
    except Exception:
        pass
    return None, None


def _grad_from_weights(w, a, n):
    """Gradient of weighted sum of autocorrelation using FFT."""
    a_rev = a[::-1]
    size = 2 * (2 * n - 1)
    fa = rfft(a_rev, size=size)
    fw = rfft(w, size=size)
    corr = irfft(fa * fw, size=size)
    return 2.0 * corr[n - 1:n - 1 + n]


def _smoothed_grad(a, n, beta):
    """Gradient of (1/beta)*logsumexp(beta*conv(a,a))."""
    c = _fft_conv(a)
    bc = beta * c
    bc_max = bc.max()
    w = np.exp(bc - bc_max)
    w /= w.sum()
    grad = _grad_from_weights(w, a, n)
    return grad, w
