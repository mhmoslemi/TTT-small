import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(416, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    return p.copy()


def _conv(a):
    """Compute full linear convolution a*a."""
    n = len(a)
    size = next_fast_len(2 * n - 1)
    F = rfft(a, size)
    c = irfft(F * F, size)[:2 * n - 1]
    return c


def _score(a):
    """Compute the exact score."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(np.max(b))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _lp_direction(a, b, peak_count=30, timeout=5.0):
    """Find LP direction to flatten the convolution peak."""
    n = len(a)
    M = float(np.max(b))
    if M < 1e-18:
        return None
    
    # Select peak positions
    threshold = np.percentile(b, 100.0 - 100.0 * peak_count / len(b))
    peak_idx = np.where(b >= threshold)[0]
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(b)])
    
    # Build constraint matrix: A[j, i] = a[peak_idx[j] - i] if valid
    # (a*g)[k] = sum_i a[k-i]*g[i] <= M
    rows = []
    cols = []
    vals = []
    for j, k in enumerate(peak_idx):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = a[k - i]
            if v > 1e-20:
                rows.append(j)
                cols.append(i)
                vals.append(v)
    
    if not rows:
        return None
    
    A = csc_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    c_vec = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(len(peak_idx), M)
    
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _line_search(a, g, a_sum, max_iter=30):
    """Find best t in [0, 1] to minimize score((1-t)*a + t*g)."""
    # Scale g to match a's sum
    g_sum = g.sum()
    if g_sum < 1e-15:
        return a.copy(), 0.0
    
    g_scaled = g * (a_sum / g_sum)
    
    # Golden section search
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = lo + (1.0 - gr) * (hi - lo)
    d = lo + gr * (hi - lo)
    
    fc = _score((1.0 - c) * a + c * g_scaled)
    fd = _score((1.0 - d) * a + d * g_scaled)
    
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + (1.0 - gr) * (hi - lo)
            fc = _score((1.0 - c) * a + c * g_scaled)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _score((1.0 - d) * a + d * g_scaled)
    
    t = (lo + hi) / 2.0
    a_new = (1.0 - t) * a + t * g_scaled
    return a_new, t


def _optimize_lp(a0, stop_time, seed, peak_counts=(20, 30, 50, 80)):

    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64).ravel().copy(), 0, None)
    a_sum = a.sum()
    if a_sum < 0.01:
        return a, math.inf
    
    best_a = a.copy()
    best_score = _score(a)
    
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 50
    
    pc_idx = 0
    t_step = 0.3
    
    while time.monotonic() < stop_time - 3.0:
        b = _conv(a)
        M = float(np.max(b))
        
        peak_count = peak_counts[pc_idx % len(peak_counts)]
        g = _lp_direction(a, b, peak_count=peak_count, timeout=4.0)
        
        if g is None:
            stall += 3
            pc_idx += 1
            if stall > max_stall:
                # Add noise to escape
                noise = rng.normal(0, 0.05 * max(1.0, a.mean()), n)
                a = np.clip(a + noise, 0, None)
                a = a / max(a.sum(), 1e-10) * a_sum
                stall = 0
                max_stall += 30
            continue
        
        # Line search
        a_new, t_used = _line_search(a, g, a_sum, max_iter=25)
        a_new = np.clip(a_new, 0, None)
        a_new_sum = a_new.sum()
        if a_new_sum < 0.01:
            stall += 2
            continue
        a_new = a_new / a_new_sum * a_sum  # Renormalize
        
        sc = _score(a_new)
        
        if sc < best_score - 1e-12:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0
            t_step = t_used  # Remember successful step size
        else:

            small_t = t_used * 0.3
            a_small = (1.0 - small_t) * a + small_t * g * (a_sum / g.sum())
            a_small = np.clip(a_small, 0, None)
            a_small = a_small / max(a_small.sum(), 1e-10) * a_sum
            sc_small = _score(a_small)
            
            if sc_small < best_score - 1e-12:
                best_a = a_small.copy()
                best_score = sc_small
                a = a_small
                stall = 0
                t_step = small_t
            else:
                stall += 1
                pc_idx += 1
                if stall > max_stall:
                    noise = rng.normal(0, 0.08 * max(1.0, a.mean()), n)
                    a = np.clip(a + noise, 0, None)
                    a = a / max(a.sum(), 1e-10) * a_sum
                    stall = 0
                    max_stall += 40
                    if max_stall > 200:

def _subgradient_opt(a0, stop_time, seed):
    """Subgradient method for the max-of-quadratics objective."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64).ravel().copy(), 1e-10, None)
    a = a / a.sum()  # Normalize sum to 1
    
    best_a = a.copy()
    best_score = _score(a)
    
    rng = np.random.default_rng(seed)
    lr = 0.1
    stall = 0
    max_stall = 60
    
    while time.monotonic() < stop_time - 3.0:
        b = _conv(a)
        M = float(np.max(b))
        if M < 1e-18:
            break
        
        # Find active peaks (top few)
        peak_threshold = M * 0.99
        peak_idx = np.where(b >= peak_threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])
        
        # Compute subgradient (average of active gradients)
        grad = np.zeros(n)
        for k in peak_idx:
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                grad[i] += a[k - i]
        grad /= len(peak_idx)
        grad *= 2.0  # From the derivative formula
        
        # Step (with projection onto a >= 0, sum(a) = 1)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        a_new = a - lr * grad / grad_norm
        a_new = np.clip(a_new, 0.0, None)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            break
        a_new = a_new / a_new_sum
        
        sc = _score(a_new)
        
        if sc < best_score - 1e-13:
            best_a = a_new.copy()
            best_score = sc
            stall = 0
            lr *= 0.995
        else:
            # Try damped step
            a_damp = 0.5 * a + 0.5 * a_new
            a_damp = np.clip(a_damp, 0, None)
            a_damp = a_damp / max(a_damp.sum(), 1e-10)
            sc_d = _score(a_damp)
            
            if sc_d < best_score - 1e-13:
                best_a = a_damp.copy()
                best_score = sc_d
                a = a_damp
                stall = 0
                lr *= 0.99
            else:
                a = a_new  # Still move (stochastic)
                stall += 1
                lr *= 0.97
                if stall > max_stall:
                    noise = rng.normal(0, 0.02, n)
                    a = np.clip(a + noise, 1e-10, None)
                    a = a / a.sum()
                    stall = 0
                    lr = 0.05
                    max_stall += 30
                    if max_stall > 300:
                        break
    
    return best_a, best_score
