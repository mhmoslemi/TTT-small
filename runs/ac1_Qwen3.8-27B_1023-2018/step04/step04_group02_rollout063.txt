    cur = _score(f)

            if time.monotonic() > stop_time - 3.0:

            break

    return best_f, best_sc


def _scipy_polish(f, n, stop_time):
    """Use scipy.optimize.minimize with analytic gradient."""
    f = np.asarray(f, dtype=np.float64).ravel().copy()

    def obj(x):
        x = np.clip(x, 0, None)
        if x.sum() < 1e-10:
            return 1e10
        return _score(x)

    def obj_grad(x):
        x = np.clip(x, 0, None)
        sc, gr = _score_grad(x)
        if not math.isfinite(sc):
            return 1e10, np.zeros(n)
        # Project gradient (set to 0 where x==0 for boundary)
        gr = np.where(x < 1e-15, np.maximum(gr, 0), gr)
        return sc, gr

    best_f = f.copy()
    best_sc = obj(f)
    deadline_local = stop_time - 5.0

    # Run several L-BFGS-B restarts
    rng = np.random.default_rng(12345)
    for restart in range(6):
        if time.monotonic() > deadline_local:
            break
        if restart == 0:
            x0 = f.copy()
        else:
            x0 = np.clip(f + rng.normal(0, 0.02 * max(1.0, f.mean()), n), 0, None)
            x0 = x0 / x0.sum()

        try:
            res = minimize(obj_grad, x0, method='L-BFGS-B',
                           bounds=[(0, 1000)] * n,
                           options={'maxiter': 500, 'maxls': 50,
                                    'ftol': 1e-14, 'gtol': 1e-10,
                                    'iprint': -1})
        except Exception:
            continue
        if res.x is not None and res.success:
            x_new = np.clip(res.x, 0, 1000)
            sc_new = obj(x_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_f = x_new.copy()

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(400, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(400, dtype=np.float64)
    n = len(parent)

    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    ps = parent.sum()
    if ps < 0.01:
        parent = np.ones(n)
        ps = n
    parent = parent / ps

    # Fallback
    fallback = parent.copy()
    fallback_sc = _score(fallback)
    try:
        fb_exact = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fb_exact = fallback_sc
    if not math.isfinite(fb_exact):
        fallback = np.ones(min(n, 100))
        fallback_sc = _score(fallback)
        fb_exact = fallback_sc

    best_seq = fallback.copy()
    best_score = min(fallback_sc, fb_exact) if math.isfinite(fb_exact) else fallback_sc

    def _update_best(candidate, sc):
        nonlocal best_seq, best_score
        if candidate is None or len(candidate) == 0:
            return
        candidate = np.clip(np.asarray(candidate, dtype=np.float64), 0, 1000)
        if not np.all(np.isfinite(candidate)) or candidate.sum() < 0.01:
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in candidate]))
        except Exception:
            exact = sc
        if math.isfinite(exact) and exact < best_score:
            best_seq = candidate.copy()
            best_score = exact

    # Phase 1: LP from parent (45% of time)
    p1_end = t0 + budget * 0.45
    try:
        f1, sc1 = _lp_optimize(parent, n, p1_end, seed=seed)
        _update_best(f1, sc1)
    except Exception:
        pass

    # Phase 2: CD polish on best (15%)
    p2_end = t0 + budget * 0.60
    if time.monotonic() < p2_end - 5.0:
        try:
            f2, sc2 = _cd_polish(best_seq, len(best_seq), p2_end, seed=seed + 1)
            _update_best(f2, sc2)
        except Exception:
            pass

    # Phase 3: scipy minimize (15%)
    p3_end = t0 + budget * 0.75
    if time.monotonic() < p3_end - 5.0:
        try:
            f3, sc3 = _scipy_polish(best_seq, len(best_seq), p3_end)
            _update_best(f3, sc3)
        except Exception:
            pass

    # Phase 4: LP from diverse initializations (15%)
    p4_end = t0 + budget * 0.90
    if time.monotonic() < p4_end - 10.0:
        i_arr = np.arange(n, dtype=np.float64)
        rng = np.random.default_rng(seed + 777)

        inits = []
        # Raised cosine
        inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / n)))
        # Parabolic
        ctr = (n - 1) / 2.0
        inits.append(np.maximum(0.0, 1.0 - ((i_arr - ctr) / (0.5 * n)) ** 2))
        # Two-lobe Gaussian
        inits.append(np.exp(-0.5 * ((i_arr - 0.25 * n) / (0.12 * n)) ** 2) +
                     np.exp(-0.5 * ((i_arr - 0.75 * n) / (0.12 * n)) ** 2))
        # Parent + noise
        sig = 0.08 * np.mean(parent)
        inits.append(np.abs(parent + rng.normal(0, sig, n)))
        # Broad top-hat with decay
        inits.append(np.exp(-0.5 * ((i_arr - n / 2) / (0.3 * n)) ** 2))

        for ix in inits:
            if time.monotonic() > p4_end - 8.0:
                break
            ix = np.clip(ix, 0, 1000)
            if ix.sum() < 0.01:
                continue
            ix = ix / ix.sum()
            stop_l = min(p4_end - 4.0, time.monotonic() + max(5.0, (p4_end - time.monotonic()) / max(1, len(inits))))
            try:
                fl, scl = _lp_optimize(ix, n, stop_l, seed=seed + int(hash(str(ix[:5])) % 10000))
                _update_best(fl, scl)
            except Exception:
                pass
            # Quick CD
            if time.monotonic() < p4_end - 5.0:
                try:
                    fc, scf = _cd_polish(fl, n, min(p4_end - 3.0, time.monotonic() + 3.0), seed=seed + 555)
                    _update_best(fc, scf)
                except Exception:
                    pass

    # Phase 5: Final CD + scipy polish (remaining time)
    final_end = deadline - 5.0
    if time.monotonic() < final_end - 8.0:
        try:
            fc, scf = _cd_polish(best_seq, len(best_seq), min(final_end - 10.0, time.monotonic() + (final_end - time.monotonic()) * 0.5), seed=seed * 3 + 1)
            _update_best(fc, scf)
        except Exception:
            pass

    if time.monotonic() < final_end - 8.0:
        try:
            fs, scs = _scipy_polish(best_seq, len(best_seq), final_end)
            _update_best(fs, scs)
        except Exception:
            pass

    # Final CD sweep
    if time.monotonic() < final_end:
        try:
            fc, scf = _cd_polish(best_seq, len(best_seq), final_end, seed=seed * 7 + 42)
            _update_best(fc, scf)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0, 1.0, 1.0]
        best_seq = fallback

    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    if not result or any(math.isnan(x) or math.isinf(x) for x in result):
        result = [float(x) for x in fallback]
    return result
