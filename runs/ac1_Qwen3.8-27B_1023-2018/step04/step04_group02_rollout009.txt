import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _lp_dir(f, n, M, timeout=3.0):
    """Find LP direction: max sum(g) s.t. (f*g)[k] <= M, g >= 0"""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _cd_polish(f, stop_time, max_sweeps=200, seed=0):
    """Coordinate descent polishing."""
    n = len(f)
    rng = np.random.default_rng(seed)
    f = f.copy()
    cur = _score(f)
    best = f.copy()
    best_sc = cur
    
    mults = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.0]
    
    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 2.0:
            break
        improved = False
        order = rng.permutation(n)

            best_v = orig
            best_local = cur

                if m == 1.0:

                f[i] = v
                if f.sum() < 0.01:
                    f[i] = orig
                    continue

                if sc < best_local:
                    best_local = sc
                    best_v = v

                for dv in [orig * 0.1, -orig * 0.1]:
                    v = max(0.0, orig + dv)
                    f[i] = v

                        if sc < best_local:
                            best_local = sc
                            best_v = v
            f[i] = best_v
            if best_local < cur - 1e-15:
                cur = best_local
                improved = True

                    best = f.copy()
        if not improved and sweep > 3:

def _lp_loop(f0, n, stop_time, seed):
    """LP-based optimization loop."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    if f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    if f.sum() < 1e-10:
        f = np.ones(n)
    f = f / f.sum()
    
    best_f = f.copy()
    best_score = _score(f)
    t = 0.3
    stall = 0
    max_stall = 300
    it = 0
    
    while time.monotonic() < stop_time - 3.0:
        b = _conv(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        g = _lp_dir(f, n, M, timeout=2.5)
        if g is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.02 * max(1.0, f.std()), n), 0, None)
            if f.sum() < 1e-10:
                f = best_f.copy()
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            stall += 3
            continue
        g = g * (f.sum() / g_sum)
        
        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < 1e-11:
            stall += 3
            if stall > max_stall:
                break
            continue
        
        improved = False
        step_list = [t, t * 0.5, t * 0.25, t * 1.5]
        for ti in step_list:
            f_new = np.clip((1.0 - ti) * f + ti * g, 0, None)
            if f_new.sum() < 1e-10:
                continue
            sc = _score(f_new)
            if sc < best_score - 1e-14:
                f = f_new
                best_f = f.copy()
                best_score = sc
                improved = True
                t = min(0.85, ti * 1.2)
                stall = 0
                break
            elif sc < _score(f) - 1e-14:
                f = f_new
                improved = True
                t = min(0.85, ti)
                stall = 0
                break
            else:
                t = max(0.01, t * 0.5)
        
        if not improved:
            stall += 1
            if stall >= max_stall:
                f = np.clip(best_f + rng.normal(0, 0.05 * max(1.0, np.mean(best_f)), n), 0, None)
                if f.sum() < 1e-10:
                    f = best_f.copy()
                f = f / f.sum()
                t = 0.3
                stall = 0
                max_stall = min(max_stall + 100, 600)

                best_score = sc_c
                best_f = f_c.copy()
                f = f_c.copy()

    return best_f, best_score
