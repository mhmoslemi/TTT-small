def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)

    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
        fp = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
        out = np.interp(fp, xp, src, left=0.0, right=0.0)

def _fft_ops(n):
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2
    conv_size = next_fast_len(conv_len)
    cross_size = next_fast_len(cross_len)

    def self_(x):
        x = np.asarray(x, dtype=np.float64)
        F = rfft(x, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def cross_(p, w):
        p = np.asarray(p, dtype=np.float64)
        w = np.asarray(w, dtype=np.float64)
        F = rfft(p[::-1], cross_size)
        G = rfft(w, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    return self_, cross_


def _softmax(u):
    u = np.asarray(u, dtype=np.float64).ravel()
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    return e / s

                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss <= 1e-300 or not math.isfinite(ss):
                            p = best_p.copy()
                            c = cfast(p)
                            cur_score = best_score
                            S = 1.0
                        else:
                            p = p / ss
                            S = 1.0
                            c = cfast(p)
                            cm = float(c.max())
                            cur_score = float(2.0 * n * cm) if math.isfinite(cm) else math.inf
                            if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
                                ex = float(_score_exact_arr(p))
                                if math.isfinite(ex) and ex < best_score - 1e-14:
                                    best_score = ex
                                    best_p = p.copy()
                                    cur_score = ex
                                    if consider is not None:
                                        try:
                                            consider(best_p, best_score)
                                        except Exception:
                                            pass
                            if (not math.isfinite(cur_score)) or cur_score > best_score * 1.05 + 1e-6:
                                if flat >= 5:
                                    p = best_p.copy()
                                    c = cfast(p)
                                    cur_score = best_score
                                    S = 1.0
                    except Exception:
                        p = best_p.copy()
                        c = cfast(p)
                        cur_score = best_score
                        S = 1.0
                    flat = 0
                else:
                    break

        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300 and math.isfinite(ss):
        p = p / ss
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))

    if budget > 150.0:
        reserve = max(30.0, min(60.0, 0.04 * budget))
    else:
        reserve = max(5.0, 0.12 * budget)

    hard = t0 + budget - reserve
    if hard <= t0 + 2.0:
        hard = t0 + max(2.0, budget - 1.0)

    parent = _get_parent()
    p0 = _as_normalized(parent)
    if p0.size == 0 or float(p0.sum()) < 0.01:
        p0 = np.ones(256, dtype=np.float64)
        p0 /= 256.0

    best_arr = p0.copy()
    best_score = float(_score_exact_arr(best_arr))
    if not math.isfinite(best_score):
        best_arr = np.ones(256, dtype=np.float64)
        best_arr /= 256.0
        best_score = float(_score_exact_arr(best_arr))
        if not math.isfinite(best_score):
            return [1.0]

    def consider(arr, pre=None):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        if a.size == 0:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s < 0.01:
            return

        sc = pre
        if sc is None or not math.isfinite(float(sc)):
            try:
                sc = float(_score_exact_arr(a))
            except Exception:
                return
        sc = float(sc)
        if not math.isfinite(sc):
            return

        if sc < best_score - 1e-14:
            an = a / s
            if np.all(np.isfinite(an)):
                best_arr = an.copy()
                best_score = sc

    usable = max(1.0, hard - t0)

    lse_stop = t0 + max(35.0, min(180.0, 0.12 * usable))
    if lse_stop > t0 + 15.0 and time.monotonic() < lse_stop - 5.0:
        try:
            p1 = _lbfgs_lse(best_arr, lse_stop, consider, fine=False)
            if p1 is not None and p1.size == best_arr.size and float(p1.sum()) > 0.0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    reserve_after_main = max(320.0, 0.25 * usable)
    main_stop = t0 + max(lse_stop + 30.0, hard - reserve_after_main)
    if main_stop > time.monotonic() + 20.0:
        try:
            p2, _ = _coord_optimize(best_arr, main_stop, seed=int(seed + 101), consider=consider)
            if p2 is not None:
                consider(p2)
        except Exception:
            pass

    now = time.monotonic()
    rem = max(0.0, hard - now - 50.0)
    if rem > 280.0 and best_score > _TARGET + 2e-5 and best_arr.size > 0:
        alt_n = 4096
        p_alt = _as_normalized(_resample_mid(best_arr, alt_n))
        if p_alt.size == alt_n:
            try:
                sc_alt = float(_score_fft_arr(p_alt))
            except Exception:
                sc_alt = math.inf
            if math.isfinite(sc_alt) and sc_alt < best_score + 0.05:
                alt_end = min(hard - 50.0, now + min(185.0, 0.62 * rem))
                if alt_end > time.monotonic() + 70.0:
                    try:
                        lse_end = min(alt_end - 70.0, time.monotonic() + min(75.0, 0.30 * (alt_end - time.monotonic())))
                        p_alt = _lbfgs_lse(p_alt, lse_end, consider, fine=False)
                        consider(p_alt)
                        if time.monotonic() < alt_end - 10.0:
                            p_alt, _ = _coord_optimize(p_alt, alt_end, seed=int(seed + 202), consider=consider)
                            consider(p_alt)
                    except Exception:
                        pass

    now = time.monotonic()
    final_reserve = 25.0
    if hard - now - final_reserve > 50.0 and best_arr.size > 0:
        pFin = best_arr.copy()
        if hard - now - final_reserve > 180.0:
            rng = np.random.default_rng(int(seed + 999))
            pFin = pFin * np.exp(rng.normal(0.0, 0.035, size=pFin.size))
            pFin = np.maximum(pFin, 0.0)
            ss = float(pFin.sum())
            if ss > 0.0 and math.isfinite(ss):
                pFin /= ss
            else:
                pFin = best_arr.copy()
            consider(pFin)
            try:
                if _score_fft_arr(pFin) > best_score + 0.05:
                    pFin = best_arr.copy()
            except Exception:
                pFin = best_arr.copy()

        rem_fin = max(0.0, hard - time.monotonic() - final_reserve)
        if rem_fin > 20.0:
            lse_end = min(hard - final_reserve - 15.0, time.monotonic() + max(10.0, min(45.0, 0.25 * rem_fin)))
            if lse_end > time.monotonic() + 8.0:
                try:
                    pFin = _lbfgs_lse(pFin, lse_end, consider, fine=True)
                    consider(pFin)
                    if _score_fft_arr(pFin) > best_score + 0.05:
                        pFin = best_arr.copy()
                except Exception:
                    pFin = best_arr.copy()

        cstop = hard - final_reserve
        if cstop > time.monotonic() + 15.0:
            try:
                pFin, _ = _coord_optimize(
                    pFin,
                    cstop,
                    seed=int(seed + 303),
                    consider=consider,
                    G=18,
                    fine_pts=24,
                )
                consider(pFin)
            except Exception:
                pass

    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = _as_normalized(parent)
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        lst = [float(x) for x in out]
        final_sc = float(evaluate_sequence(lst))
        if not math.isfinite(final_sc):
            lst = [float(x) for x in _as_normalized(parent)]
            final_sc = float(evaluate_sequence(lst))
            if not math.isfinite(final_sc):
                return [1.0]
        return lst
    except Exception:
        return [1.0]
