def _resample(a, new_n):
    """Resample sequence to new length."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


def _subgradient(a, active_lags):
    """Subgradient of max_k (a*a)_k at the given active lags."""
    n = a.size

        js = np.arange(lo, hi)

def _build_conv_mat(a):
    """Build (2n-1) x n matrix A where A[k,j] = a[k-j] if valid."""
    n = a.size

    for j in range(n):
        k_lo = max(0, 0)
        k_hi = min(2 * n - 1, j + n)
        if k_hi > k_lo:
            A[k_lo:k_hi, j] = a[k_lo - j : k_hi - j]
    return A
