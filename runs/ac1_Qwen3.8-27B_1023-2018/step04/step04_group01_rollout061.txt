    n = len(f0)
    f = np.clip(f0.copy(), 0, None)

            # d/d(f[i]) = f[k-i] (from the j=i term: f[i]*f[k-i])
            #           + f[k-i] (from the (k-j)=i term: f[k-i]*f[i])
            # Wait, that's only if the same index appears. Let me think more carefully.
            #
            # conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} f[j]*f[k-j]
            #
            # When we differentiate w.r.t. f[i]:
            # - j = i contributes: f[i]*f[k-i] (if 0<=k-i<n)
            # - k-j = i, i.e., j = k-i contributes: f[k-i]*f[i] (if 0<=k-i<n)
            # These are the same term when j = i and k-j = i, i.e., k = 2i.

            #
            # For k != 2i: if 0 <= k-i < n, then d(conv[k])/d(f[i]) = 2*f[k-i].
            # Otherwise: 0.
            #
            # Second derivative: d2(conv[k])/d(f[i])^2 = 2 if k = 2i, else 0.

            infl = np.zeros(2 * n - 1)
            lo_k = i
            hi_k = min(i + n, 2 * n - 1)

            # where S_new = 1 (since we normalize)
            # Wait, S changes! S_new = 1 + delta (since we add delta to f[i])

            -np.ones(n),

        return None
