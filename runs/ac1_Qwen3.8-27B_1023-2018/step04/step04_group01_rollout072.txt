import numpy as np
import math
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _conv_full(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(1, dtype=np.float64)
    if n <= 900:
        c = np.convolve(a, a, mode="full")
        c = np.asarray(c, dtype=np.float64)
        return c
    L = 2 * n - 1
    sz = next_fast_len(L)
    F = rfft(a, sz)
    c = irfft(F * F, sz)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _score_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_full(a)
    m = float(np.max(c)) if c.size else 0.0
    if not math.isfinite(m):
        return math.inf
    if m < 0.0:
        m = 0.0
    return 2.0 * a.size * m / (s * s)


def _build_A(f):
    f = np.asarray(f, dtype=np.float64).ravel()
    n = f.size
    if n == 0:
        return np.zeros((0, 0), dtype=np.float64)
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _lp_minmax(A, tie, time_limit=1.2):
    A = np.asarray(A, dtype=np.float64)
    if A.ndim != 2 or A.shape[1] < 1:
        return None
    n = A.shape[1]
    L = A.shape[0]

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0
    if tie is not None:
        tie = np.asarray(tie, dtype=np.float64).ravel()
        if tie.size == n and np.all(np.isfinite(tie)):
            c_obj[:n] += tie

    A_ub = np.hstack((A, -np.ones((L, 1), dtype=np.float64)))
    b_ub = np.zeros(L, dtype=np.float64)
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    bounds = [(0.0, None)] * n + [(0.0, None)]

    options_base = {"presolve": True}
    options = dict(options_base)
    if time_limit is not None and time_limit > 0:
        options["time_limit"] = max(0.3, float(time_limit))

    res = None
    try:
        res = linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options=options,
        )
    except Exception:
        try:
            res = linprog(
                c_obj,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options=options_base,
            )
        except Exception:
            return None

    if res is None or res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.clip(g, 0.0, None)
    s = float(g.sum())
    if s <= 1e-15:
        return None
    return g / s


def _quad_line_search(b, c, d, n):
    b = np.asarray(b, dtype=np.float64).ravel()
    c = np.asarray(c, dtype=np.float64).ravel()
    d = np.asarray(d, dtype=np.float64).ravel()
    if b.size == 0:
        return 0.0, math.inf
    m0 = min(b.size, c.size, d.size)
    if m0 < b.size or m0 < c.size or m0 < d.size:
        b = b[:m0]
        c = c[:m0]
        d = d[:m0]
    if b.size == 0:
        return 0.0, math.inf

    np.maximum(b, 0.0, out=b)
    np.maximum(c, 0.0, out=c)
    np.maximum(d, 0.0, out=d)

    def eval_scores(t):
        t = np.asarray(t, dtype=np.float64)
        if t.size == 0:
            return np.array([], dtype=np.float64)
        one = 1.0 - t
        q = (one * one)[:, None] * b[None, :]
        q += (2.0 * one * t)[:, None] * c[None, :]
        q += (t * t)[:, None] * d[None, :]
        q = np.where(np.isfinite(q), q, 0.0)
        mm = np.max(q, axis=1)
        sc = 2.0 * n * mm
        return np.where(np.isfinite(sc), sc, math.inf)

    t_coarse = np.linspace(0.0, 1.0, 33)
    sc = eval_scores(t_coarse)
    if not np.any(np.isfinite(sc)):
        return 0.0, math.inf
    i0 = int(np.argmin(sc))
    best_t = float(t_coarse[i0])
    best_sc = float(sc[i0])

    intervals = []
    if best_t <= 0.0:
        intervals.append((0.0, 0.08))
    elif best_t >= 1.0:
        intervals.append((0.92, 1.0))
    else:
        intervals.append((max(0.0, best_t - 0.10), min(1.0, best_t + 0.10)))

    for lo, hi in intervals:
        if hi <= lo:
            continue
        tr = np.linspace(lo, hi, 25)
        scr = eval_scores(tr)
        valid = np.isfinite(scr)
        if not np.any(valid):
            continue
        j = int(np.argmin(scr))
        if scr[j] < best_sc - 1e-18:
            best_sc = float(scr[j])
            best_t = float(tr[j])

    return best_t, best_sc


def _coord_polish(f0, stop_time, seed, max_sweeps=200,
                  coarse_pts=45, local_pts=35, fine_pts=25,
                  update_cb=None):
    f = _normalize(f0)
    n = f.size
    if n == 0:
        return f, math.inf
    L = 2 * n - 1
    conv = _conv_full(f)
    current_sc = _score_arr(f)
    best = f.copy()
    best_sc = current_sc
    rng = np.random.default_rng(seed)
    last_cb = math.inf

    def eval_grid(xs, conv_arr, infl, v, has_sq, sq):
        xs = np.asarray(xs, dtype=np.float64).ravel()
        if xs.size == 0:
            return v, math.inf
        w = xs - v
        Snew = 1.0 - v + xs
        valid = Snew > 1e-12
        if not np.any(valid):
            return v, math.inf
        wv = w[valid]
        Snv = Snew[valid]
        xsv = xs[valid]
        q = conv_arr + wv[:, None] * infl[None, :]
        if has_sq and 0 <= sq < q.shape[1]:
            q[:, sq] += wv * wv
        mm = np.max(q, axis=1)
        obj = 2.0 * n * mm / (Snv * Snv)
        obj = np.where(np.isfinite(obj), obj, math.inf)
        idx = int(np.argmin(obj))
        return float(xsv[idx]), float(obj[idx])

    for sweep in range(max_sweeps):
        if time.monotonic() >= stop_time - 2.0:
            break
        improved = 0
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() >= stop_time - 2.0:
                break
            v = float(f[i])
            if v < 0.0 and v > -1e-15:
                v = 0.0
                f[i] = 0.0
            if v < -1e-15:
                continue

            x_hi = max(10.0 * max(v, 0.0), 0.35, 30.0 / n)
            x_hi = min(2.0, x_hi)
            if x_hi < v * (1.0 + 1e-6):
                x_hi = min(2.0, max(v * 1.25, v + 1e-4, 0.35))
            if x_hi <= 0.0:
                continue

            infl = np.zeros(L, dtype=np.float64)
            infl[i:i + n] = 2.0 * f
            sq = 2 * i
            has_sq = 0 <= sq < L

            xs_coarse = np.linspace(0.0, x_hi, coarse_pts)
            local_half = max(0.02, min(0.10, 5.0 * x_hi / max(coarse_pts, 1)))
            lo = max(0.0, v - local_half)
            hi = min(x_hi, v + local_half)
            if hi <= lo:
                hi = min(x_hi, lo + 1e-6)
            xs_local = np.linspace(lo, hi, local_pts)
            xs_all = np.concatenate((xs_coarse, xs_local, np.array([v, x_hi], dtype=np.float64)))

            bx, bobj = eval_grid(xs_all, conv, infl, v, has_sq, sq)
            if bobj < current_sc - 1e-16:
                width = min(x_hi, max(0.02, 2.0 * x_hi / max(coarse_pts, 1)))
                fl_lo = max(0.0, bx - width)
                fl_hi = min(x_hi, bx + width)
                if fl_hi > fl_lo:
                    xs_fine = np.linspace(fl_lo, fl_hi, fine_pts)
                    bx2, obj2 = eval_grid(xs_fine, conv, infl, v, has_sq, sq)
                    if obj2 < bobj - 1e-18:
                        bx, bobj = bx2, obj2

            if bobj < current_sc - 1e-13 and math.isfinite(bobj):
                w = bx - v
                new_conv = conv + w * infl
                if has_sq:
                    new_conv[sq] += w * w
                np.maximum(new_conv, 0.0, out=new_conv)
                Snew_val = 1.0 - v + bx
                if Snew_val <= 1e-12:
                    continue

                f[i] = bx
                f = f / Snew_val
                new_conv = new_conv / (Snew_val * Snew_val)
                conv = new_conv

                current_sc = max(0.0, 2.0 * n * float(np.max(conv)))
                if current_sc < best_sc - 1e-14:
                    best_sc = current_sc
                    best = f.copy()
                improved += 1

        conv = _conv_full(f)
        cur2 = _score_arr(f)
        if math.isfinite(cur2) and cur2 < best_sc - 1e-14:
            best_sc = cur2
            best = f.copy()

        if best_sc < last_cb - 1e-14 and update_cb is not None:
            try:
                update_cb(best)
            except Exception:
                pass
            last_cb = best_sc

        if improved == 0 and sweep >= 3:
            break

    final = _normalize(best)
    final_sc = _score_arr(final)
    other_sc = _score_arr(f)
    if other_sc < final_sc:
        final = _normalize(f)
        final_sc = other_sc
    if final_sc < last_cb - 1e-14 and update_cb is not None:
        try:
            update_cb(final)
        except Exception:
            pass
    return final, final_sc


def _lp_optimize(f0, stop_time, seed, update_cb=None):
    f = _normalize(f0)
    n = f.size
    if n == 0:
        return f, math.inf
    current_sc = _score_arr(f)
    best = f.copy()
    best_sc = current_sc
    rng = np.random.default_rng(seed)

    ar = np.arange(n, dtype=np.float64)
    wave = np.sin(np.linspace(0.0, 2.0 * np.pi, n))
    wave2 = np.cos(np.linspace(0.0, 2.0 * np.pi, n))
    tie_patterns = [
        1e-8 * ar,
        -1e-8 * ar,
        1e-8 * wave,
        -1e-8 * wave2,
        rng.normal(0.0, 1.0, n) * 1e-8,
    ]

    it = 0
    stall = 0

    while time.monotonic() < stop_time - 2.0:
        f = _normalize(f)
        b = _conv_full(f)
        if b.size == 0:
            break
        A = _build_A(f)
        M = float(np.max(b))
        if not math.isfinite(M):
            break
        current_sc = max(0.0, 2.0 * n * M)
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best = f.copy()

        g = _lp_minmax(A, tie_patterns[it % len(tie_patterns)],
                       time_limit=0.8 if time.monotonic() > stop_time - 8 else 1.4)

        progressed = False
        if g is not None:
            dist = float(np.linalg.norm(g - f))
            if dist > 1e-8:
                c = A @ g
                if np.all(np.isfinite(c)):
                    d = _conv_full(g)
                    t_best, sc_best = _quad_line_search(b, c, d, n)
                    if t_best > 1e-9 and math.isfinite(sc_best):
                        cands = []
                        cands.append(float(t_best))
                        cands.append(min(1.0, max(0.0, t_best - 0.05)))
                        cands.append(min(1.0, max(0.0, t_best + 0.05)))
                        cands.append(max(0.0, t_best * 0.5))
                        best_trial_t = 0.0
                        best_trial_sc = current_sc
                        for tt in cands:
                            if tt <= 1e-10 or tt >= 1.0 - 1e-12:
                                tt = min(max(tt, 0.0), 1.0)
                            fn = (1.0 - tt) * f + tt * g
                            fn = _normalize(fn)
                            scn = _score_arr(fn)
                            if math.isfinite(scn) and scn < best_trial_sc - 1e-15:
                                best_trial_sc = scn
                                best_trial_t = tt
                        if best_trial_t > 1e-10 and best_trial_sc < current_sc - 1e-15:
                            fn = (1.0 - best_trial_t) * f + best_trial_t * g
                            fn = _normalize(fn)
                            scn = _score_arr(fn)
                            if math.isfinite(scn):
                                f = fn
                                current_sc = scn
                                progressed = True
                                stall = 0
                                if current_sc < best_sc - 1e-14:
                                    best = f.copy()
                                    best_sc = current_sc
                                    if update_cb is not None:
                                        try:
                                            update_cb(f)
                                        except Exception:
                                            pass

        if progressed:
            it += 1
            continue

        stall += 1
        strength = 0.004 + 0.0018 * min(stall, 25)
        fp = f + rng.normal(0.0, strength, n)
        fp = _normalize(fp)
        scp = _score_arr(fp)
        if math.isfinite(scp) and scp < current_sc - 1e-15:
            f = fp
            current_sc = scp
        elif stall > 8:
            f = best.copy()
            current_sc = best_sc
        else:
            f = fp
            current_sc = scp

        if stall > 14 and time.monotonic() < stop_time - 35:
            micro_stop = min(stop_time - 6.0, time.monotonic() + 18.0)
            if micro_stop > time.monotonic() + 3.0:
                try:
                    fm, scm = _coord_polish(f, micro_stop, seed + 100000 + it,
                                            max_sweeps=2, update_cb=None)
                    cscm = _score_arr(fm)
                    if math.isfinite(cscm) and cscm < best_sc - 1e-14:
                        best = fm.copy()
                        best_sc = cscm
                        f = fm.copy()
                        current_sc = cscm
                        if update_cb is not None:
                            try:
                                update_cb(f)
                            except Exception:
                                pass
                    elif math.isfinite(cscm):
                        f = fm.copy()
                        current_sc = cscm
                except Exception:
                    pass
                stall = 0

        if stall > 28:
            alpha = np.maximum(best * 100.0, 1e-3)
            try:
                rdir = rng.dirichlet(alpha)
            except Exception:
                rdir = rng.exponential(size=n)
                rdir = _normalize(rdir)
            for beta in (0.10, 0.25, 0.50):
                cand = (1.0 - beta) * best + beta * rdir
                cand = _normalize(cand)
                sc = _score_arr(cand)
                if math.isfinite(sc) and sc < best_sc - 1e-14:
                    best = cand.copy()
                    best_sc = sc
                    f = cand.copy()
                    current_sc = sc
                    if update_cb is not None:
                        try:
                            update_cb(f)
                        except Exception:
                            pass
            stall = 0

        it += 1

    final = _normalize(best)
    final_sc = _score_arr(final)
    if _score_arr(f) < final_sc:
        final = _normalize(f)
        final_sc = _score_arr(final)
    if update_cb is not None:
        try:
            update_cb(final)
        except Exception:
            pass
    return final, final_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    h = None
    try:
        h = height_sequence_1
    except Exception:
        h = globals().get("height_sequence_1", None)

    if h is None:
        parent = np.ones(400, dtype=np.float64)
    else:
        parent = np.asarray(h, dtype=np.float64).ravel()
    parent = _normalize(parent)

    def exact_score(seq):
        try:
            return float(evaluate_sequence(seq.tolist()))
        except Exception:
            return math.inf

    best_seq = parent.copy()
    best_score = exact_score(best_seq)
    if not math.isfinite(best_score):
        best_score = _score_arr(best_seq)
        if not math.isfinite(best_score):
            best_seq = _normalize(np.ones(400))
            best_score = exact_score(best_seq)
            if not math.isfinite(best_score):
                best_score = _score_arr(best_seq)

    def update(seq):
        nonlocal best_seq, best_score
        seq = _normalize(np.asarray(seq, dtype=np.float64).ravel())
        if seq.size == 0 or float(seq.sum()) < 0.01:
            return
        sc = _score_arr(seq)
        if not math.isfinite(sc):
            return
        esc = exact_score(seq)
        cand = esc if math.isfinite(esc) else sc
        if math.isfinite(cand) and cand < best_score - 1e-12:
            best_seq = seq.copy()
            best_score = cand

    if best_seq.size == 0:
        best_seq = _normalize(np.ones(400))

    n0 = best_seq.size
    rng0 = np.random.default_rng(seed)

    for _ in range(6):
        try:
            alpha = np.maximum(best_seq * 100.0 + 1e-3, 1e-3)
            rdir = rng0.dirichlet(alpha)
        except Exception:
            rdir = _normalize(rng0.standard_normal(n0))
        for beta in (0.02, 0.05, 0.10, 0.20):
            cand = (1.0 - beta) * best_seq + beta * rdir
            update(cand)

    idx = np.arange(n0, dtype=np.float64)
    for w in (1.0 + 0.25 * idx / max(n0 - 1, 1),
              1.0 + 0.25 * (n0 - 1 - idx) / max(n0 - 1, 1),
              1.0 + 0.25 * np.sin(np.pi * idx / max(n0, 1))):
        update(best_seq * np.maximum(w, 1e-3))

    if budget > 200:
        t1 = min(deadline - 35, t0 + 0.16 * budget)
        t2 = min(deadline - 25, t0 + 0.52 * budget)
        t3 = min(deadline - 12, t0 + 0.76 * budget)
        t4 = min(deadline - 6, t0 + 0.88 * budget)
    else:
        t1 = min(deadline - 5, t0 + max(5.0, 0.20 * budget))
        t2 = min(deadline - 5, t0 + max(5.0, 0.50 * budget))
        t3 = min(deadline - 5, t0 + max(5.0, 0.75 * budget))
        t4 = deadline - 3.0

    if time.monotonic() < t1 - 2.0:
        try:
            c, _ = _coord_polish(best_seq, t1, seed + 1, max_sweeps=250, update_cb=update)
            update(c)
        except Exception:
            pass

    if time.monotonic() < t2 - 2.0:
        try:
            l, _ = _lp_optimize(best_seq, t2, seed + 100, update_cb=update)
            update(l)
        except Exception:
            pass

    if time.monotonic() < t3 - 2.0:
        try:
            c, _ = _coord_polish(best_seq, t3, seed + 200, max_sweeps=500, update_cb=update)
            update(c)
        except Exception:
            pass

    if time.monotonic() < t4 - 2.0:
        try:
            l, _ = _lp_optimize(best_seq, t4, seed + 300, update_cb=update)
            update(l)
        except Exception:
            pass

    final_stop = deadline - 5.0
    if time.monotonic() < final_stop - 2.0:
        try:
            c, _ = _coord_polish(best_seq, final_stop, seed + 400, max_sweeps=20000, update_cb=update)
            update(c)
        except Exception:
            pass

    best_seq = _normalize(best_seq)
    fs = exact_score(best_seq)
    if not math.isfinite(fs):
        fallback = _normalize(np.ones(max(1, n0)))
        ffb = exact_score(fallback)
        if not math.isfinite(ffb):
            return [1.0]
        best_seq = fallback

    final_fs = exact_score(best_seq)
    if not math.isfinite(final_fs):
        return [float(x) for x in best_seq]

    return [float(x) for x in best_seq]
