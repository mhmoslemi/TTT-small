- sum(f_t) = (1-t)*sum(f) + t*sum(g) >= sum(f) (since sum(g) >= sum(f) because f is feasible)

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _solve_lp(f, active_k, M, timeout=0.5):
    n = len(f)
    K = len(active_k)
    if K == 0 or K > 2*n-1:
        return None
    A = np.zeros((K, n), dtype=np.float64)
    for idx, k in enumerate(active_k):
        t_min = max(0, k - (n - 1))
        t_max = min(k, n - 1)
        if t_max >= t_min:
            A[idx, t_min:t_max+1] = f[t_max-t_min::-1][::-1]
            # A[idx, t] = f[k-t] for t in [t_min, t_max]
            # So A[idx, t_min] = f[k-t_min], A[idx, t_min+1] = f[k-t_min-1], ...
            # That's f[k-t_min], f[k-t_min-1], ..., f[k-t_max]
            A[idx, t_min:t_max+1] = f[k-t_min : k-t_max-1 : -1] if k-t_min > k-t_max else f[k-t_max:k-t_min+1]
    
    c = -np.ones(n)
    b_ub = np.full(K, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g
