                xx = np.arange(n, dtype=np.float64)
                a = a * (1.0 + sigma * (xx / n - 0.5) * rng.uniform(0.5, 2.0))
            elif mode == 3:
                # Block perturbation
                bl = rng.integers(10, max(20, n // 4))
                st = rng.integers(0, n - bl)
                a[st:st+bl] *= (1.0 + sigma * rng.uniform(-1, 1))
            else:
                a = a * np.exp(rng.normal(0, sigma, n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stuck = 0
            if (stuck // max_stuck) > 100:
                break
            continue

        # Try LP
        made_move = False
        g = _active_set_lp(a, M * (1 + 1e-10), n, tol=0.2, time_limit=4.0)
        
        if g is not None:
            g = g / g.sum()
            if np.max(np.abs(g - a)) > 1e-10:
                cand, val = _line_search(a, g, M, n)
                if cand is not None and val < M * (1 - 1e-13):
                    a = cand
                    made_move = True
        
        if not made_move:
            # Try full LP
            g = _lp_best_response(a, M * (1 + 1e-9), n, time_limit=3.0)
            if g is not None:
                g = g / g.sum()
                if np.max(np.abs(g - a)) > 1e-10:
                    cand, val = _line_search(a, g, M, n)
                    if cand is not None and val < M * (1 - 1e-13):
                        a = cand
                        made_move = True
        
        if not made_move:
            # Subgradient step
            grad, k_star = _subgradient_direction(a, n)
            gn = float(np.linalg.norm(grad))
            if gn > 1e-12:
                for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
                    cand = a - alpha * grad / gn
                    cand = np.maximum(cand, 0.0)
                    cs = cand.sum()
                    if cs < 1e-12:
                        continue
                    cand = cand / cs
                    val = float(np.convolve(cand, cand).max())
                    if val < M * (1 - 1e-13):
                        a = cand
                        made_move = True
                        break

        if not made_move:
            # Small random kick
            a = a * np.exp(rng.normal(0, 0.015, n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()

    return best_a_norm

def _optimize_smoothed(a0, stop_time, T_start=5.0, T_end=100.0, seed=42):
    """Optimize using smoothed (log-sum-exp) objective with gradient descent."""
    a0 = np.asarray(a0, dtype=np.float64)
    n = a0.size
    a = np.clip(a0, 0.0, 1000.0).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())

    rng = np.random.default_rng(seed)
    stuck = 0

    while time.monotonic() < stop_time:
        c_aa = np.convolve(a, a)
        M = float(c_aa.max())
        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 30:
            a = a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stuck = 0
            continue

        # Adaptive T
        progress = 1.0 - max(0, stop_time - time.monotonic()) / max(1, stop_time - (stop_time - 600))
        T = T_start * (T_end / T_start) ** min(progress, 1.0)
        T = max(T_start, T)

        grad, _ = _smoothed_grad(a, T, n)
        grad -= grad.mean()  # Project to sum=0
        gn = float(np.linalg.norm(grad))
        if gn < 1e-14:
            break

        made_move = False
        for alpha in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            cand = a - alpha * grad / gn
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-12:
                continue
            cand = cand / cs
            val = float(np.convolve(cand, cand).max())
            if val < M * (1 - 1e-14):
                a = cand
                made_move = True
                break

        if not made_move:
            # Try smaller steps with combined LP direction
            g = _active_set_lp(a, M * (1 + 1e-10), n, tol=0.3, time_limit=3.0)
            if g is not None:
                g = g / g.sum()
                for t in [0.1, 0.2, 0.3]:
                    cand = (1 - t) * a + t * g
                    cand = np.maximum(cand, 0.0)
                    cs = cand.sum()
                    if cs < 1e-12:
                        continue
                    cand = cand / cs
                    val = float(np.convolve(cand, cand).max())
                    if val < M * (1 - 1e-14):
                        a = cand
                        made_move = True
                        break

        if not made_move:
            a = a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()

    return best_a

def _try_length(a_best, n_target, stop_time):
    """Resample to a different length and optimize briefly."""
    n_orig = len(a_best)
    x_old = np.linspace(0, 1, n_orig)
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_old, a_best)
    a_new = np.clip(a_new, 0, 1000)
    a_new = a_new / a_new.sum()
    return _optimize_lp(a_new, stop_time, seed=n_target, max_stuck=30)

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 20.0  # Safety margin

    parent = _get_parent()
    n_parent = parent.size
    a_parent = np.clip(parent, 0, 1000).astype(np.float64)
    a_parent = a_parent / a_parent.sum()

    best_seq = a_parent.copy()
    best_score = _eval_score(best_seq)

    def try_update(cand):
        nonlocal best_seq, best_score
        try:
            sc = _eval_score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.clip(cand.astype(np.float64), 0, 1000).copy()
        except Exception:
            pass

    # Phase 1: LP optimization from parent (40% of time)
    t1 = t0 + budget * 0.40
    try:
        r = _optimize_lp(a_parent, t1, seed=1, max_stuck=60)
        try_update(r)
    except Exception:
        pass

    # Phase 2: Smoothed gradient from current best (20% of time)
    t2 = t0 + budget * 0.60
    if time.monotonic() < t2 - 5:
        try:
            ab = best_seq / best_seq.sum()
            r = _optimize_smoothed(ab, t2, T_start=5.0, T_end=200.0, seed=2)
            try_update(r)
        except Exception:
            pass

    # Phase 3: LP from reversed parent (10% of time)
    t3 = t0 + budget * 0.70
    if time.monotonic() < t3 - 5:
        a_rev = a_parent[::-1].copy()
        try:
            r = _optimize_lp(a_rev, t3, seed=3, max_stuck=40)
            try_update(r)
        except Exception:
            pass

    # Phase 4: Diverse asymmetric initializations (15% of time)
    t4 = t0 + budget * 0.85
    n = n_parent
    x = np.arange(n, dtype=np.float64)
    inits = []
    pn = best_seq / best_seq.sum()
    
    # Asymmetric linear ramps
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        v = 1.0 + alpha * (x / (n - 1) - 0.5)
        v = np.maximum(v, 0.01)
        inits.append(v / v.sum())
    
    # Power laws (asymmetric)
    for p in [0.5, 1.0, 2.0, 3.0]:
        v = (x / (n - 1) + 0.1) ** p
        inits.append(v / v.sum())
        v2 = ((n - 1 - x) / (n - 1) + 0.1) ** p
        inits.append(v2 / v2.sum())
    
    # Gaussian bumps (off-center)
    for center_frac in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        for width in [0.1, 0.2, 0.3]:
            ctr = center_frac * (n - 1)
            v = np.exp(-0.5 * ((x - ctr) / (width * n)) ** 2)
            inits.append(v / v.sum())
    
    # Blends with parent
    for frac in [0.3, 0.5, 0.7]:
        for init in inits[:6]:
            v = frac * pn + (1 - frac) * init
            v = np.maximum(v, 0.01)
            inits.append(v / v.sum())

    per_init = max(2.0, (t4 - time.monotonic()) / max(1, len(inits)))
    for i, init in enumerate(inits):
        if time.monotonic() > t4 - 5:
            break
        alloc = min(time.monotonic() + per_init, t4 - 3.0)
        try:
            r = _optimize_lp(init, alloc, seed=100 + i, max_stuck=25)
            try_update(r)
        except Exception:
            pass

    # Phase 5: Try nearby lengths (10% of time)
    t5 = t0 + budget * 0.95
    if time.monotonic() < t5 - 5:
        for dn in [-10, -5, -3, -2, -1, 1, 2, 3, 5, 10]:
            if time.monotonic() > t5 - 5:
                break
            nt = n_parent + dn
            if nt < 10 or nt > 600:
                continue
            try:
                r = _try_length(best_seq, nt, time.monotonic() + 15.0)
                try_update(r)
            except Exception:
                pass

    # Phase 6: Final deep polish on best (remaining time)
    if time.monotonic() < deadline - 5:
        ab = best_seq / best_seq.sum()
        try:
            r = _optimize_lp(ab, deadline - 5.0, seed=999, max_stuck=80)
            try_update(r)
        except Exception:
            pass

    # Final validation
    try:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if not np.isfinite(fs):
            raise ValueError("invalid")
    except Exception:
        fallback = [1.0] * 32
        return fallback

    return [float(x) for x in best_seq]
