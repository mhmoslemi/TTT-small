def _conv_full(a):
    n = len(a)

    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fast(a, b=None):

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def _build_lp_matrix(f, n):
    """Build A matrix for LP: A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2 * n - 1, n))
    # Use FFT or direct construction

        end = n

        # so k ranges from j to j+n-1
        A[j:j+n, j] = f
    return A

def _lp_direction(f, n, M, time_limit=5.0):

    A = _build_lp_matrix(f, n)
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': time_limit}
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs  # normalize to sum 1

def _coord_descent(f0, stop_time, max_sweeps=200, seed=12345):

    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, math.inf
    f = f / S
    S = 1.0
    b = _conv_full(f)
    best_score = 2.0 * n * float(b.max())
    best_f = f.copy()
    rng = np.random.default_rng(seed)
    
    conv_size = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 5.0:
            break
        
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break
            
            v = f[i]
            
            # Influence: db[k]/dv = 2*f[k-i] for 0<=k-i<n
            # So delta_b[k] = 2*w*f[k-i] + w^2*delta(k, 2i)
            # where w = new_val - v
            
            infl = np.zeros(conv_size)
            # f[k-i] for 0 <= k-i < n means k ranges from i to i+n-1
            k_start = i
            k_end = min(i + n, conv_size)
            idx = np.arange(k_start, k_end)
            infl[idx] = 2.0 * f[idx - i]
            
            di = 2 * i
            has_diag = di < conv_size
            
            # Search for best w
            # w ranges from -v (make f[i]=0) to some max
            w_max = max(2.0, 5.0 * v) if v > 1e-10 else 2.0
            w_max = min(w_max, 10.0)
            
            # Coarse grid
            n_pts = 80
            w_grid = np.linspace(-v, w_max, n_pts)
            
            best_w = 0.0
            best_sc = best_score
            
            for w in w_grid:
                sc = best_score
                if abs(w) < 1e-15:
                    continue
                # Update b
                b_new = b + w * infl
                if has_diag:
                    b_new[di] += w * w
                m_new = float(b_new.max())
                sc_new = 2.0 * n * m_new  # S=1
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_w = w
            
            # Fine grid around best
            if best_w != 0.0:
                dw = w_max / n_pts
                w_lo = max(-v, best_w - 3*dw)
                w_hi = min(w_max, best_w + 3*dw)
                w_fine = np.linspace(w_lo, w_hi, 40)
                for w in w_fine:
                    if abs(w) < 1e-15:
                        continue
                    b_new = b + w * infl
                    if has_diag:
                        b_new[di] += w * w
                    sc_new = 2.0 * n * float(b_new.max())
                    if sc_new < best_sc:
                        best_sc = sc_new
                        best_w = w
            
            # Accept
            if best_w != 0.0 and best_sc < best_score - 1e-15:
                w = best_w
                f[i] = v + w
                b += w * infl
                if has_diag:
                    b[di] += w * w
                best_score = best_sc
                best_f = f.copy()
                improved = True
        
        if not improved and sweep > 5:
            # Try perturbation
            f = best_f + rng.normal(0, 0.02, n)
            f = np.clip(f, 0, None)
            f /= f.sum()
            b = _conv_full(f)
            best_score = 2.0 * n * float(b.max())
    
    return best_f, best_score

    def _update_best(seq, sc=None):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64)
        seq = np.clip(seq, 0, 1000)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        if sc is None:
            sc = _score_fast(seq)
        if not math.isfinite(sc):
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-15:
            best_seq = seq.copy()
            best_score = exact
    
    def _try_size(n, init_arr, stop_time):
        nonlocal best_seq, best_score
        if time.monotonic() > stop_time - 10:
            return
        init_arr = np.clip(np.asarray(init_arr, dtype=np.float64).ravel(), 0, None)
        while init_arr.size < n:
            init_arr = np.concatenate([init_arr, np.zeros(n - init_arr.size)])
        init_arr = init_arr[:n]
        if init_arr.sum() < 1e-10:
            init_arr = np.ones(n) / n
        init_arr = init_arr / init_arr.sum()
        
        # Phase A: LP optimization
        t_lp = min(stop_time - 30, t0 + (stop_time - t0) * 0.4)
        if time.monotonic() < t_lp - 5:
            try:
                res, sc = _lp_optimize(init_arr, n, t_lp, seed)
                if res is not None:
                    _update_best(res, sc)
            except Exception:
                pass
        
        # Phase B: Coordinate descent
        t_cd = min(stop_time - 10, t0 + (stop_time - t0) * 0.75)
        if time.monotonic() < t_cd - 5:
            try:
                res, sc = _coord_descent(best_seq, t_cd, max_sweeps=100, seed=seed+1)
                _update_best(res, sc)
            except Exception:
                pass
        
        # Phase C: More coordinate descent
        if time.monotonic() < stop_time - 5:
            try:
                res, sc = _coord_descent(best_seq, stop_time - 5, max_sweeps=50, seed=seed+2)
                _update_best(res, sc)
            except Exception:
                pass
    
    # Main phase: optimize at parent length
    main_end = t0 + budget * 0.55
    _try_size(n_p, parent, main_end)
    
    # Try alternative initializations
    if time.monotonic() < main_end + budget * 0.15 and best_score > 1.503:
        i_arr = np.arange(n_p, dtype=np.float64)
        alts = [
            np.clip(1.0 + 0.5*np.cos(2*np.pi*i_arr/(n_p-1)**0.5), 0.01, None),
            np.clip(0.5 + 0.5*np.sin(np.pi*i_arr/(n_p-1))**2, 0.01, None),
            np.clip(np.exp(-0.5*((i_arr - n_p/2)/(n_p/4))**2), 0.001, None),
        ]
        for alt in alts:
            if time.monotonic() > main_end + budget * 0.1:
                break
            if best_score < 1.503:
                break
            alt = alt / alt.sum()
            t_alloc = min(main_end + budget*0.15 - 10, time.monotonic() + budget*0.05)
            try:
                res, sc = _lp_optimize(alt, n_p, t_alloc, seed+100)
                if res is not None:
                    _update_best(res, sc)
            except Exception:
                pass
    
    # Intensive coordinate polish
    polish_end = min(t0 + budget * 0.80, hard_deadline - 60)
    if time.monotonic() < polish_end - 10:
        try:
            res, sc = _coord_descent(best_seq, polish_end, max_sweeps=150, seed=77777)
            _update_best(res, sc)
        except Exception:
            pass
    
    # Try different lengths
    if time.monotonic() < polish_end + 60 and best_score > 1.5025:
        for L in [n_p + 50, n_p + 100, n_p + 200, n_p - 50]:
            if time.monotonic() > hard_deadline - 80:
                break
            if L < 32 or L > 2000:
                continue
            inter = np.interp(np.linspace(0,1,L), np.linspace(0,1,len(best_seq)), best_seq)
            inter = np.clip(inter, 0.0001, None)
            inter /= inter.sum()
            t_alloc = min(hard_deadline - 80, time.monotonic() + 40)
            try:
                res, sc = _coord_descent(inter, t_alloc, max_sweeps=30, seed=seed+L)
                _update_best(res, sc)
            except Exception:
                pass
    
    # Final polish
    if time.monotonic() < hard_deadline - 5:
        try:
            res, sc = _coord_descent(best_seq, hard_deadline - 5, max_sweeps=30, seed=99999)
            _update_best(res, sc)
        except Exception:
            pass
