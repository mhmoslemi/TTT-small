    a = _norm1(a0.astype(np.float64).copy())
    avg = a.copy()
    best_a = a.copy()

    while time.monotonic() < deadline:
        T += 1
        c = np.convolve(a, a)
        M = float(c.max())
        ks = int(c.argmax())

        grad = _cgrad(a, ks)
        # Project to zero-sum (since we need to maintain normalization)
        grad -= grad.mean()
        gnrm = float(np.linalg.norm(grad))
        
        if gnrm > 1e-14:
            eta = eta0 / np.sqrt(T)
            a = np.maximum(a * np.exp(-eta * grad / gnrm), 0)
            a = a / a.sum()

        avg = (avg * (T-1) + a) / T

            best_sc = sc
            best_a = a.copy()
        sc_avg = _score(avg)
        if sc_avg < best_sc:
            best_sc = sc_avg
            best_a = avg.copy()

        if f2 < bf:
            bt, bf = c2, f2
    return bt, bf

def _cgrad(a, k):
    n = len(a)
    m = np.arange(n)
    km = k - m
    v = (km >= 0) & (km < n)
    g = np.zeros(n)
    g[v] = a[km[v]] + a[v]
    return g

def _opt_lp(a0, deadline, seed=42):
    n = len(a0)
    a = _norm1(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stick = 0
    n_pert = 0
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        M = float(c.max())
        ks = int(c.argmax())
        if M < best_M * (1.0 - 1e-14):
            best_M, best_a, stick = M, a.copy(), 0
        else:
            stick += 1
        if stick > 80:
            sigma = min(0.02 + 0.004 * (stick // 80), 0.30)
            a = _norm1(a * np.exp(rng.normal(0.0, sigma, n)))
            stick = 0
            n_pert += 1
            if n_pert > 200:
                break
            continue
        A = _build_A(n, a)
        mv = False
        rem = deadline - time.monotonic()
        if rem > 8:
            for Mf in [1.0, 1.0 + 1e-8, 1.0 + 1e-6, 1.0 + 1e-4, 1.0 - 1e-6]:
                if time.monotonic() > deadline - 6:
                    break
                tl = min(4.0, max(0.5, deadline - time.monotonic() - 10))
                g = _lp_br(A, M * Mf, n, tl)
                if g is None or g.sum() < 1e-15:
                    continue
                gn = g / g.sum()
                if float(np.max(np.abs(gn - a))) < 1e-11:
                    continue
                t, v = _golden(a, gn, M)
                if v < M * (1.0 - 1e-13):
                    a = _norm1((1.0 - t) * a + t * gn)
                    mv = True
                    break
        if not mv and time.monotonic() < deadline - 4:
            grad = _cgrad(a, ks)
            gnrm = float(np.linalg.norm(grad))
            if gnrm > 1e-14:
                for al in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001]:
                    cand = np.maximum(a - al * grad / gnrm, 0.0)
                    cs = cand.sum()
                    if cs < 1e-12:
                        continue
                    cand /= cs
                    if float(np.convolve(cand, cand).max()) < M * (1.0 - 1e-13):
                        a = cand
                        mv = True
                        break
        if not mv:
            a = _norm1(a * np.exp(rng.normal(0.0, 0.012, n)))
    return best_a

def _opt_lsg(a0, deadline, seed=42):
    n = len(a0)
    a = _norm1(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(seed)
    eps_seq = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    for eps in eps_seq:
        if time.monotonic() > deadline - 8:
            break
        def fg(x):
            x = np.maximum(x, 1e-14)
            s = x.sum()
            c = np.convolve(x, x)
            cm = c.max()
            ec = np.exp(eps * (c - cm))
            es = ec.sum()
            w = ec / es
            L = (cm + np.log(es)) / eps
            G = 2.0 * n * L / (s * s)
            gL = np.zeros(n)
            for m in range(n):
                k0 = max(0, m)
                k1 = min(2 * n - 2, m + n - 1)
                if k0 <= k1:
                    gL[m] = np.dot(w[k0:k1 + 1], x[k0 - m:k1 - m + 1] + x[m])
            gG = (2.0 * n / (s * s)) * gL - 4.0 * n * L / (s ** 3)
            return G, gG
        rem = max(2.0, deadline - time.monotonic() - 5)
        try:
            res = minimize(fg, a, method='L-BFGS-B', jac=True,
                           bounds=[(1e-14, 1000.0)] * n,
                           options={'maxiter': max(20, int(rem * 25)),
                                    'ftol': 1e-15, 'maxfun': max(50, int(rem * 80)),
                                    'maxls': 40})
            if res.x is not None:
                an = _norm1(res.x)
                sc = _score(an)
                if sc < best_sc:
                    best_sc, best_a = sc, an.copy()
                a = an
        except Exception:
            pass
    return best_a

def _opt_subgrad(a0, deadline, seed=42):
    n = len(a0)
    a = _norm1(a0.astype(np.float64).copy())
    avg = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    T = 0
    eta0 = 0.08
    while time.monotonic() < deadline - 3:
        T += 1
        c = np.convolve(a, a)
        M = float(c.max())
        ks = int(c.argmax())
        grad = _cgrad(a, ks)
        grad -= grad.mean()
        gnrm = float(np.linalg.norm(grad))
        if gnrm > 1e-14:
            eta = eta0 / np.sqrt(T)
            a = _norm1(a * np.exp(-eta * grad / gnrm))
        avg = (avg * (T - 1) + a) / T
        avg = _norm1(avg)
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
        sc_a = _score(avg)
        if sc_a < best_sc:
            best_sc, best_a = sc_a, avg.copy()
    return best_a

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    dl = t0 + budget_s - 20.0
    parent = _get_parent()
    n = len(parent)
    best = parent.copy()
    best_sc = _score(best)

    def update(cand):
        nonlocal best, best_sc
        try:
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = np.clip(np.asarray(cand, dtype=np.float64), 0, 1000).copy()
        except Exception:
            pass

    # Phase 1: LP refinement of parent
    t1 = t0 + min(280.0, budget_s * 0.22)
    try:
        update(_opt_lp(parent, t1, seed=1))
    except Exception:
        pass

    # Phase 2: LSG L-BFGS-B exploration
    t2 = t0 + min(510.0, budget_s * 0.39)
    try:
        update(_opt_lsg(best, t2, seed=100))
    except Exception:
        pass

    # Phase 3: LP refinement from best
    t3 = t0 + min(760.0, budget_s * 0.58)
    try:
        update(_opt_lp(best, t3, seed=200))
    except Exception:
        pass

    # Phase 4: Subgradient + diverse inits
    if time.monotonic() < dl - 120:
        try:
            update(_opt_subgrad(best, t3 + 60, seed=300))
        except Exception:
            pass
        x = np.arange(n, dtype=np.float64)
        pn = parent / max(parent.sum(), 1e-15)
        inits = []
        for p in [0.3, 0.5, 0.8, 1.2, 2.0, 3.0]:
            v = np.maximum(1.0 - x / max(n - 1, 1), 1e-8) ** p
            inits.append(v / v.sum())
        for al in [0.5, 1.0, 2.0, 4.0]:
            v = np.exp(-al * x / n)
            inits.append(v / v.sum())
        for w in [0.1, 0.2, 0.35]:
            ctr = (n - 1) / 2.0
            v = np.exp(-0.5 * ((x - ctr) / (w * n / 2.0)) ** 2)
            inits.append(v / v.sum())
        for fr in [1, 2, 3]:
            v = np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * fr * x / max(n - 1, 1)), 0.01)
            inits.append(v / v.sum())
        for frac in [0.3, 0.5, 0.7]:
            v = frac * pn + (1.0 - frac) * inits[0]
            inits.append(v / v.sum())
        t4 = dl - 120
        per_init = max(3.0, (t4 - time.monotonic()) / max(1, len(inits)))
        for i, init in enumerate(inits):
            if time.monotonic() > t4 - 3:
                break
            alloc = min(time.monotonic() + per_init, t4 - 2)
            try:
                update(_opt_lp(init, alloc, seed=500 + i))
            except Exception:
                pass

    # Phase 5: Nearby lengths
    if time.monotonic() < dl - 50:
        for dn in [-10, -5, -2, -1, 1, 2, 5, 10, 20]:
            if time.monotonic() > dl - 25:
                break
            nt = n + dn
            if nt < 10 or nt > 500:
                continue
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, parent)
            init_new = np.clip(init_new, 0, 1000)
            sn = init_new.sum()
            if sn < 1e-10:
                continue
            init_new /= sn
            rem = min(25.0, dl - time.monotonic() - 10)
            if rem < 5:
                break
            try:
                update(_opt_lp(init_new, time.monotonic() + rem, seed=1000 + nt))
            except Exception:
                pass

    # Phase 6: Final deep polish
    if time.monotonic() < dl - 8:
        af = _norm1(best.copy())
        try:
            update(_opt_lp(af, dl - 3, seed=9999))
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        fs = float(evaluate_sequence([float(x) for x in best]))
        if not np.isfinite(fs):
            raise ValueError("invalid")
    except Exception:
        fallback = np.ones(32, dtype=np.float64)
        return [float(x) for x in fallback]

    return [float(x) for x in best]
