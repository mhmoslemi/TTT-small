import math

        A[j:end, j] = f[:end - j]

def _lp_dir(f, M, timeout=4.0):

    c = -np.ones(n)

import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _conv(a):
    n = len(a)
    if n <= 2048:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        A[j:end, j] = f[:end - j]
    return A


def _lp_dir(f, M, timeout=3.0):
    n = len(f)
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-12:
        return None
    return g

        _try_opt(len(best_seq), best_seq.copy(), phase2_end, seed + 33333)

    # Phase 3: Aggressive CD (20% of time)
    phase3_end = t0 + budget * 0.85
    if time.monotonic() < phase3_end - 5.0:
        _try_cd(best_seq.copy(), phase3_end, seed * 3 + 1)

    # Phase 4: Try nearby lengths (10% of time)
    phase4_end = deadline - 15.0
    if time.monotonic() < phase4_end - 20.0:
        tried = set()
        for L in [n - 32, n + 32, n - 64, n + 64, n - 16, n + 16]:
            if L < 64 or L in tried or time.monotonic() > phase4_end - 12.0:
                continue
            tried.add(L)
            if len(best_seq) != L:
                xv = np.linspace(0, 1, len(best_seq))
                x_new = np.linspace(0, 1, L)
                init_l = np.interp(x_new, xv, best_seq)
            else:
                init_l = best_seq.copy()
            stop_l = min(phase4_end - 5.0, time.monotonic() + 40.0)
            _try_opt(L, init_l, stop_l, seed + 77000 + L)

    # Phase 5: Final CD polish
    final_end = deadline - 5.0
    if time.monotonic() < final_end - 8.0:
        _try_cd(best_seq.copy(), final_end, seed * 7 + 42)

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0, 1.0, 1.0]
        return [float(x) for x in fallback]

    best_score = min(best_score, final_sc)
    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    if not result:
        result = [1.0]
    if any(math.isnan(x) or math.isinf(x) for x in result):
        return [float(x) for x in fallback]
    return result
