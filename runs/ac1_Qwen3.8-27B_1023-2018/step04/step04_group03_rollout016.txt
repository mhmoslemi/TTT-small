def _lp_direction(f, n, b, peak_frac, timeout):
    M = float(b.max())
    if M < 1e-18:
        return None
    threshold = M * (1.0 - peak_frac)
    peak_idx = np.where(b >= threshold)[0]
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(b)])
    
    # Build constraint matrix: for each peak k, sum_i f[k-i] * g[i] <= M
    # But this is f * g (cross-conv), not g * g
    # Actually, the constraint should be on the NEW sequence's self-convolution
    # But that's quadratic in g. The LP linearizes by using f * g instead.
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(peak_idx):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-20:
                rows.append(idx)
                cols.append(i)
                vals.append(v)
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    c_vec = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(len(peak_idx), M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _optimize(f0, stop_time, seed, peak_frac=0.05):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    best_f = f.copy()
    best_score = _score(f)
    
    t = 0.3
    stall = 0
    max_stall = 60
    
    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        g = _lp_direction(f, n, b, peak_frac=peak_frac, timeout=5.0)
        if g is None:
            stall += 3
            if stall > max_stall:
                # Perturbation
                noise = rng.normal(0, 0.1 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                fs = f.sum()
                if fs < 1e-10:
                    f = best_f.copy()
                else:
                    f = f / fs * best_f.sum()
                t = 0.3
                stall = 0
                max_stall = min(max_stall + 30, 300)
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            stall += 2
            continue
        f_sum = f.sum()
        g = g * (f_sum / g_sum)  # Normalize to same sum
        
        # Check if direction is meaningful
        dist = np.linalg.norm(g - f) / max(1.0, np.linalg.norm(f))
        if dist < 1e-8:
            stall += 2
            if stall > max_stall:
                break
            continue
        
        # Line search
        best_t = t
        best_sc = best_score
        improved = False
        
        for t_try in [t, t*0.5, t*1.5, t*0.25, t*2, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            t_try = max(0.01, min(0.99, t_try))
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.clip(f_new, 0, None)
            if f_new.sum() < 0.01:
                continue
            sc = _score(f_new)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_t = t_try
                improved = True
        
        if improved:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0, None)
            best_f = f.copy()
            best_score = best_sc
            t = best_t
            stall = 0
        else:
            stall += 1
            t *= 0.8
            if stall > max_stall:
                noise = rng.normal(0, 0.15 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * best_f.sum()
                t = 0.3
                stall = 0
                max_stall = min(max_stall + 30, 400)
    
    return best_f, best_score
