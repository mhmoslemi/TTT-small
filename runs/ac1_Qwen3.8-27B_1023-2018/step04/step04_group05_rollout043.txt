When f[j] changes from `old` to `new`, the convolution changes as follows:
- For k where both j and k-j are in [0, n-1]:

The affected k values are from max(0, j) to min(2n-2, j+n-1), which is n values (since j is in [0, n-1], the range is j to j+n-1, which is n values, all within [0, 2n-2]).

`(f*f)[k] = sum_i f[i] * f[k-i]`

When f[j] changes from `old` to `new = old + delta`:
- Terms where i = j: f[j]*f[k-j] changes by delta * f[k-j]
- Terms where k-i = j, i.e., i = k-j: f[k-j]*f[j] changes by f[k-j]*delta
- If k-j = j (i.e., k = 2j): both terms are the same, so total change is f[j]*f[j] -> (f[j]+delta)*(f[j]+delta), change = 2*f[j]*delta + delta²
- If k-j ≠ j: total change is delta*f[k-j] + f[k-j]*delta = 2*delta*f[k-j]

So:
- For k = 2j: `c[k] += 2*old*delta + delta²` (or equivalently `c[k] += new² - old²`)
- For k ≠ 2j (and k-j valid): `c[k] += 2*delta*f[k-j]`

Let me also note that the valid range for k is [0, 2n-2], and for the term f[i]*f[k-i] to be valid, we need 0 <= i < n and 0 <= k-i < n, i.e., max(0, k-n+1) <= i <= min(k, n-1).

For the specific i=j: valid if 0 <= j < n (always true) and 0 <= k-j < n, i.e., j <= k < j+n.
For the specific i=k-j: valid if 0 <= k-j < n (same condition) and 0 <= j < n (always true).

Since j < n, we have j+n-1 < 2n-1, so the upper bound is j+n-1 (since j+n-1 <= 2n-2 iff j <= n-1, which is true).

import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_conv_matrix(f, n):
    """A[k,j] = f[k-j] if 0<=k-j<n, else 0. Shape (2n-1, n)."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=10.0):
    if M <= 1e-18:
        return None
    A = _build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-13)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
            options={"presolve": True, "time_limit": timeout}
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g


def _smooth_gradient_step(f, alpha):
    """Compute one gradient direction for the smooth objective."""
    n = len(f)
    c = _conv_full(f)
    max_c = c.max()
    
    shifted = alpha * (c - max_c)
    shifted = np.clip(shifted, -100, 100)
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    
    # grad[j] = (2/alpha) * sum_{m=0}^{n-1} f[m] * p[j+m]
    idx = np.arange(n)
    P_sub = p[idx[:, None] + idx[None, :]]
    grad = (2.0 / alpha) * (P_sub @ f)
    
    # Project onto sum=1 hyperplane
    grad = grad - grad.mean()
    
    return grad, p, max_c

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=8.0):
    if M <= 1e-18:
        return None
    A = _build_A(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-13)
    bounds = [(0.0, None)] * n
    remaining = timeout
    if remaining < 2.0:
        return None
    try:
        res = linprog(
            c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
            options={"presolve": True, "time_limit": remaining}
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g


def _smooth_grad_step(f, alpha):
    n = len(f)
    c = _conv_full(f)
    max_c = c.max()
    shifted = alpha * (c - max_c)
    shifted = np.clip(shifted, -100, 100)
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    idx = np.arange(n)
    P_sub = p[idx[:, None] + idx[None, :]]
    grad = (2.0 / alpha) * (P_sub @ f)
    grad = grad - grad.mean()
    return grad, p, max_c

    rng = np.random.default_rng(seed)
    n = len(f0)
    f = np.clip(f0.copy(), 0, None)
    f = f / max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = 2.0 * n * float(_conv_full(f).max())
    t = 0.3
    stall = 0
    max_stall = 80
    while time.monotonic() < stop_time - 6.0:
        c = _conv_full(f)
        M = float(c.max())
        if M < 1e-18:
            break
        remaining = stop_time - time.monotonic() - 8.0
        if remaining < 4.0:
            break
        g = _lp_direction(f, n, M, timeout=min(lp_timeout, remaining))
        if g is None:
            stall += 3
            if stall > max_stall:
                f = best_f + rng.normal(0, 0.04 * max(1.0, best_f.mean()), n)
                f = np.clip(f, 0, None)
                f = f / max(f.sum(), 1e-10)
                stall = 0
                t = 0.3
                max_stall += 60
            continue
        stall = 0
        g = g / g.sum()
        dist = float(np.linalg.norm(g - f))
        if dist < 1e-11:
            break
        improved = False
        candidates = sorted({max(0.01, min(1.0, c)) for c in
                             [t, t * 0.5, t * 0.75, t * 1.25, t * 1.5, 0.1, 0.4, 0.7, 1.0]},
                            reverse=True)
        for tc in candidates:
            f_new = (1.0 - tc) * f + tc * g
            f_new = np.clip(f_new, 0, None)
            s_new = f_new.sum()
            if s_new < 0.01:

            f_new = f_new / s_new
            sc_new = 2.0 * n * float(_conv_full(f_new).max())
            if sc_new < best_sc - 1e-13:
                f = f_new
                best_f = f.copy()
                best_sc = sc_new
                t = min(1.0, tc * 1.3)

        if not improved:

            t = max(0.01, t * 0.5)

                f = np.clip(f, 0, None)
