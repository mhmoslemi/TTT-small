import numpy as np

from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _lp_direction(f, n, b_full, peak_frac=0.10, timeout=8.0):
    M = float(b_full.max())
    if M < 1e-18:
        return None
    threshold = M * (1.0 - peak_frac)
    peak_idx = np.where(b_full >= threshold)[0]
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(b_full)])

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(peak_idx):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-15:
                rows.append(idx)
                cols.append(i)
                vals.append(v)

    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    c_vec = -np.ones(n)
    b_ub = np.full(len(peak_idx), M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None

    if res.status not in (0, 3) or res.x is None:
        return None

    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _optimize_subgrad(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-10, None)
    f = f / f.sum()
    best_f = f.copy()
    best_score = _score(f)

    lr = 0.15
    stall = 0
    max_stall = 50

    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        # Use top 10% peaks for subgradient
        threshold = M * 0.95
        peak_idx = np.where(b >= threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])

        grad = np.zeros(n)
        for k in peak_idx:
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                grad[i] += 2.0 * f[k - i]
        grad /= len(peak_idx)

        gscale = grad.max()
        if gscale < 1e-15:
            break

        eff_lr = lr / gscale
        exponent = -eff_lr * grad
        exponent -= exponent.max()
        weights = np.exp(exponent)
        f_new = (f + 1e-15) * weights
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        sc = _score(f_new)
        if sc < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = sc
            stall = 0
            lr *= 0.99
            f = f_new
        else:
            f_damp = 0.5 * f + 0.5 * f_new
            sc_d = _score(f_damp)
            if sc_d < best_score - 1e-12:
                best_f = f_damp.copy()
                best_score = sc_d
                f = f_damp
                stall = 0
                lr *= 0.97
            else:
                stall += 1
                lr *= 0.94
                f = f_new
                if stall > max_stall:
                    noise = rng.normal(0, 0.02, n)

                    f = f / f.sum()
                    stall = 0
                    lr = 0.1
                    max_stall += 20
                    if max_stall > 200:
                        break

def _polish(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)

    best_f = f.copy()

    mean_val = max(f.mean(), 1e-3)

    while time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break

            orig = f[i]

            # Ternary search for optimal f[i]
            lo = 0.0
            hi = max(5.0 * mean_val, 1.0)
            best_local = _score(f)
            best_val = orig

            for _ in range(10):
                d = (hi - lo) / 3.0
                m1 = lo + d
                m2 = hi - d
                f[i] = m1
                s1 = _score(f)
                f[i] = m2
                s2 = _score(f)
                if s1 < s2:
                    hi = m2
                else:
                    lo = m1
                if hi - lo < 1e-6 * max(1.0, mean_val):
                    break

            f[i] = (lo + hi) / 2.0
            sc_mid = _score(f)
            # Also try the original
            f[i] = orig
            sc_orig = _score(f)

            if sc_mid < best_local - 1e-15:
                f[i] = (lo + hi) / 2.0
                best_own = sc_mid
            else:
                f[i] = orig
                best_own = sc_orig

            if best_own < best_sc - 1e-14:
                best_sc = best_own
                best_f = f.copy()
                improved = True

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0
            mean_val = max(f.mean(), 1e-3)

    return best_f, best_sc


def _make_inits(parent, n, seed):
    rng = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)
    inits = [parent.copy()]

    # Perturbed parent
    sig = 0.12 * max(1.0, float(parent.mean()))
    inits.append(np.clip(parent + rng.normal(0, sig, n), 0, None))

    # Multiplicative perturbation
    inits.append(np.clip(parent * np.exp(0.15 * rng.normal(0, 1, n)), 0, None))

    # Ramp (linearly increasing)
    ramp = np.linspace(0.3, 2.5, n)
    inits.append(ramp)

    # Ramp reversed
    inits.append(ramp[::-1].copy())

    # Cosine bump centered
    center = (n - 1) / 2.0
    sigma = n * 0.25
    inits.append(np.exp(-0.5 * ((i_arr - center) / sigma) ** 2) + 0.05)

    # Two bumps
    c1 = n * 0.25
    c2 = n * 0.75
    w = n * 0.12
    inits.append(np.exp(-0.5 * ((i_arr - c1) / w) ** 2) + np.exp(-0.5 * ((i_arr - c2) / w) ** 2) + 0.03)

    # Exponential decay
    r = 1.0 + 1.5 / n
    exp_seq = r ** i_arr
    inits.append(exp_seq / exp_seq.max() + 0.05)

    # Exponential growth
    inits.append(exp_seq[::-1].copy() / (exp_seq.max()) + 0.05)

    # Quadratic bump (parabola)
    inits.append(1.0 - ((i_arr - center) / (n * 0.45)) ** 2)
    inits[-1] = np.clip(inits[-1], 0.02, None)

    # Random smooth
    raw = rng.uniform(0.3, 1.5, n)
    # Smooth with small kernel
    k = 7
    pad = k // 2
    padded = np.pad(raw, pad, mode='edge')
    smooth = np.convolve(padded, np.ones(k) / k, 'valid')[:n]
    inits.append(smooth)

    return inits

    # Phase 3: Coordinate descent polish (25% time)
    phase3_end = t0 + budget * 0.80
    if time.monotonic() < phase3_end - 10.0:
        stop = min(phase3_end - 3.0, time.monotonic() + (phase3_end - time.monotonic() - 3.0))
        try:
            res, sc = _polish(best_seq, stop, seed + 55555)
            _update_best(res)
        except Exception:
            pass

    # Phase 4: LP after polish (8% time)
    phase4_end = t0 + budget * 0.88
    if time.monotonic() < phase4_end - 10.0:
        stop = min(phase4_end - 3.0, time.monotonic() + (phase4_end - time.monotonic() - 3.0))
        try:
            res, sc = _optimize_lp(best_seq, stop, seed + 666666, peak_frac=0.04)
            _update_best(res)
        except Exception:
            pass

    # Phase 5: Try different lengths (5% time)
    phase5_end = min(deadline + 2.0, t0 + budget * 0.95)
    if time.monotonic() < phase5_end - 20.0:
        lengths_to_try = sorted(set([n_p - 32, n_p + 32, n_p - 64, n_p + 64,
                                     n_p - 128, n_p + 128]),
                                key=lambda L: abs(L - n_p))
        x_old = np.linspace(0, 1, n_p)
        for L in lengths_to_try:
            if time.monotonic() > phase5_end - 8.0:
                break
            if L < 80 or L > 1000:
                continue
            x_new = np.linspace(0, 1, L)
            init_L = np.interp(x_new, x_old, best_seq)
            stop = min(phase5_end - 3.0, time.monotonic() + 30.0)
            try:
                res, sc = _optimize_lp(init_L, stop, seed + L * 100, peak_frac=0.10)
                _update_best(res)
            except Exception:
                pass

    # Final polish with remaining time
    remaining = deadline - time.monotonic()
    if remaining > 15.0:
        try:
            res, sc = _polish(best_seq, deadline - 3.0, seed + 99999)
            _update_best(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    return [float(x) for x in best_seq]
