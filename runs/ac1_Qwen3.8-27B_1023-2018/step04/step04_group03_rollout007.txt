def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_from_conv(a, b):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _score(a):
    b = _conv_full(a)
    return _score_from_conv(a, b)


def _lp_direction(f, n, b, peak_frac, timeout=5.0):
    M = float(b.max())
    if M < 1e-18:
        return None
    threshold = M * (1.0 - peak_frac)
    peak_idx = np.where(b >= threshold)[0]
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(b)])

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(peak_idx):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-20:
                rows.append(idx)
                cols.append(i)
                vals.append(v)

    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    c_vec = -np.ones(n)
    b_ub = np.full(len(peak_idx), M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None

    if res.status not in (0, 3) or res.x is None:
        return None

    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _subgrad_optimize(f0, stop_time, seed):

    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-10, None)
    S = f.sum()
    f = f / S  # normalize to sum=1
    best_f = f.copy()
    best_score = _score(f)

    lr = 0.3
    momentum = 0.9
    v = np.zeros(n)
    stall = 0

    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        
        # Get subgradient: average of gradients at active peaks
        peak_threshold = M * 0.999
        peak_idx = np.where(b >= peak_threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])
        
        grad = np.zeros(n)
        for k in peak_idx:
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                grad[i] += f[k - i]
        grad /= len(peak_idx)
        
        # Subgradient descent with momentum
        v = momentum * v - lr * grad
        f_new = f + v
        
        # Projection onto simplex (f >= 0, sum(f) = 1)
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-15:
            break
        f_new = f_new / S_new
        
        sc = _score(f_new)
        if sc < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = sc
            f = f_new
            lr = min(lr * 1.01, 0.5)
            stall = 0

            f = f + 0.5 * v  # half step

                break

            best_local_val = orig

                    continue

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(416, dtype=np.float64)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    return p.copy()


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    b = _conv_full(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * len(a) * m / (s * s)


def _lp_direction(f, n, b, peak_frac, timeout=4.0):
    M = float(b.max())
    if M < 1e-18:
        return None
    threshold = M * (1.0 - peak_frac)
    peak_idx = np.where(b >= threshold)[0]
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(b)])

    rows, cols, vals = [], [], []
    for idx, k in enumerate(peak_idx):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-20:
                rows.append(idx)
                cols.append(i)
                vals.append(v)

    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    c_vec = -np.ones(n)
    b_ub = np.full(len(peak_idx), M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None

    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _optimize_lp(f0, stop_time, seed, peak_frac=0.05):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    best_f = f.copy()
    best_score = _score(f)

    t = 0.2
    stall = 0
    max_stall = 40

    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, n, b, peak_frac=peak_frac, timeout=3.0)
        if g is None:
            stall += 3
            if stall > max_stall:
                noise = rng.normal(0, 0.04 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * best_f.sum()
                t = 0.2
                stall = 0
                max_stall = min(max_stall + 20, 150)
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            stall += 2
            continue
        f_sum = f.sum()
        g = g * (f_sum / g_sum)

        dist = np.linalg.norm(g - f) / max(1.0, np.linalg.norm(f))
        if dist < 1e-10:
            stall += 2
            if stall > max_stall:
                break
            continue

        # Golden-section line search
        best_t_score = best_score
        best_t_val = 0.0
        for t_try in [t, t * 0.5, t * 1.5, t * 0.25, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            t_try = max(0.01, min(0.95, t_try))
            f_new = np.clip((1.0 - t_try) * f + t_try * g, 0, None)
            if f_new.sum() < 0.01:
                continue
            sc = _score(f_new)
            if sc < best_t_score:
                best_t_score = sc
                best_t_val = t_try

        if best_t_score < best_score - 1e-14 and best_t_val > 0:
            f = np.clip((1.0 - best_t_val) * f + best_t_val * g, 0, None)
            best_f = f.copy()
            best_score = best_t_score
            t = best_t_val
            stall = 0
        else:
            stall += 1
            t = max(0.02, t * 0.75)

    return best_f.copy(), best_score


def _subgrad_optimize(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-10, None)
    f = f / f.sum()
    best_f = f.copy()
    best_score = _score(f)

    lr = 0.3
    mo = 0.85
    v = np.zeros(n)
    stall = 0
    total_stall = 0

    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        peak_threshold = M * 0.999
        peak_idx = np.where(b >= peak_threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])

        grad = np.zeros(n)
        for k in peak_idx:
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                grad[i] += f[k - i]
        grad /= len(peak_idx)

        v = mo * v - lr * grad
        f_new = f + v
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-15:
            break
        f_new = f_new / S_new

        sc = _score(f_new)
        if sc < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = sc
            f = f_new
            lr = min(lr * 1.005, 0.4)
            stall = 0
            total_stall = 0
        else:
            f = f + 0.5 * v
            f = np.clip(f, 1e-15, None)
            f = f / f.sum()
            stall += 1
            total_stall += 1
            lr *= 0.98
            if stall > 80:
                f = best_f.copy()
                f = f + rng.normal(0, 0.003, n)
                f = np.clip(f, 1e-10, None)
                f = f / f.sum()
                v = np.zeros(n)
                lr = 0.2
                stall = 0
                if total_stall > 600:
                    break

    return best_f.copy(), best_score


def _coord_descent(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    best_sc = _score(f)
    best_f = f.copy()

    no_imp = 0
    max_no_imp = 15

    ratios_coarse = [0.0, 0.05, 0.1, 0.2, 0.4, 0.6, 0.8, 0.9, 0.95,
                     1.05, 1.1, 1.2, 1.3, 1.5, 1.8, 2.2, 3.0, 5.0]
    ratios_fine = [0.85, 0.9, 0.95, 1.0, 1.05, 1.1, 1.15]

    while time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break

            orig = f[i]
            best_local = best_sc
            best_val = orig

            # Try zero
            f[i] = 0.0
            if f.sum() >= 0.01:
                sc = _score(f)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_val = 0.0

            if orig > 1e-14:
                for r in ratios_coarse:
                    f[i] = orig * r
                    if f.sum() < 0.01:
                        continue
                    sc = _score(f)
                    if sc < best_local - 1e-15:
                        best_local = sc
                        best_val = f[i]

                # Fine search around best_val if improved
                if best_local < best_sc - 1e-14 and best_val > 0:
                    for r in ratios_fine:
                        f[i] = best_val * r
                        if f.sum() < 0.01:
                            continue
                        sc = _score(f)
                        if sc < best_local - 1e-15:
                            best_local = sc
                            best_val = f[i]

                    # Binary search refinement
                    lo = 0.0 if orig > best_val else best_val
                    hi = best_val if orig > best_val else orig
                    for _ in range(12):
                        mid = 0.5 * (lo + hi)
                        f[i] = mid
                        if f.sum() < 0.01:
                            break
                        sc = _score(f)
                        if sc <= best_local:
                            best_local = min(best_local, sc)
                            best_val = mid
                            lo = mid
                        else:
                            hi = mid
                    f[i] = best_val

            if best_local < best_sc - 1e-14:
                f[i] = best_val
                best_sc = best_local
                best_f = f.copy()
                improved = True
            else:
                f[i] = best_val

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                # Random perturbation restart
                f = best_f.copy()
                f = f + rng.normal(0, 0.005 * max(1.0, best_f.mean()), n)
                f = np.clip(f, 0, None)
                if f.sum() > 0.01:
                    sc = _score(f)
                    if sc < best_sc:
                        best_sc = sc
                        best_f = f.copy()
                no_imp = 0
                max_no_imp = min(max_no_imp + 5, 40)
        else:
            no_imp = 0

    return best_f.copy(), best_sc
