def _make_conv_matrix_sparse(n, weights):
    # weights length n, a. Return A row k col j = weights[j] if 0<=k-j<n.
    j = np.arange(n, dtype=np.int32)
    off = np.arange(n, dtype=np.int32)
    rows = (np.repeat(j, n) + np.tile(off, n)).astype(np.int32)
    cols = np.tile(j, n).astype(np.int32)
    data = np.repeat(weights, n).astype(np.float64)
    A = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
    return A

def _fft_conv(a):

        return np.zeros(0, dtype=np.float64)

    np.maximum(c, 0.0, out=c)
    return c


def _fft_cross(p, w):

def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == 1:
        return np.full(n, float(src[0]) / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / float(n), xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _softmax(u):
    u = np.asarray(u, dtype=np.float64).ravel()
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        u = np.where(np.isfinite(u), u, 0.0)
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    return e / s


def _lbfgs_optimize(p, stop_time, consider=None, fine_only=False):

    n = int(p.size)
    if n <= 1 or time.monotonic() > stop_time:
        return p

    conv_len = 2 * n - 1
    cross_len = 3 * n - 2
    conv_size = next_fast_len(conv_len)
    cross_size = next_fast_len(cross_len)

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

        pp = np.maximum(pp, 1e-15)
        uu = np.log(pp, dtype=np.float64)
        uu -= float(np.mean(uu))
        return uu

    def lse_func(uu, T):
        uu = np.asarray(uu, dtype=np.float64).ravel()
        if uu.size != n:
            return 1e100, np.zeros(n, dtype=np.float64)
        if not np.all(np.isfinite(uu)):

        e = np.exp(uu - m, dtype=np.float64)
        s = float(e.sum())
        if not math.isfinite(s) or s <= 0.0:
            pp = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            pp = e / s
        if not np.all(np.isfinite(pp)):
            pp = np.full(n, 1.0 / n, dtype=np.float64)
        ss = float(pp.sum())
        if not math.isfinite(ss) or ss <= 0.0:
            pp = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            pp /= ss

        c = cfast(pp)
        if c.size != conv_len:
            return 1e100, np.zeros(n, dtype=np.float64)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0.0:
            return 1e100, np.zeros(n, dtype=np.float64)

        if not math.isfinite(T) or T <= 0.0:
            T = max(1e-300, 1e-4 * cmax)
        T = max(1e-14, T)

        scaled = (c - cmax) / T
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(conv_len, dtype=np.float64)
            im = int(np.argmax(c))
            w[im] = 1.0
            f = cmax
        else:
            w = exps / sum_exp
            ratio = sum_exp / float(c.size)
            if ratio <= 0.0:
                f = cmax
            else:
                f = cmax + T * float(math.log(ratio))
                if not math.isfinite(f):
                    f = cmax

        q = xfast(pp, w)
        gid = 2.0 * q
        gd = float(np.dot(pp, gid))
        grad = pp * (gid - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    def run_chunk(uu, T, end):
        try:
            f0, _ = lse_func(uu, T)
        except Exception:
            return uu
        if not math.isfinite(f0):
            return uu
        best_u = np.array(uu, dtype=np.float64, copy=True)
        best_f = float(f0)
        stale = 0

        while time.monotonic() < end - 2.0:
            maxit = 14 if n > 2000 else 20
            opts = {
                "maxiter": maxit,
                "gtol": 1e-9,
                "ftol": 1e-14,
                "maxls": 14,
                "maxcor": 15,
                "disp": False,
                "maxdisp": 0,
            }
            try:
                res = minimize(
                    lse_func,
                    best_u,
                    args=(T,),
                    method="L-BFGS-B",
                    jac=True,
                    options=opts,
                )
            except Exception:
                break

            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-13:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 2:
                            break
            if getattr(res, "status", 1) == 0:
                break
        return best_u

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score_fft(p)
    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass

    if fine_only:
        stages = ((0.0010, 0.22), (0.00025, 0.30), (0.00006, 0.48))
    else:
        stages = (
            (0.018, 0.14),
            (0.006, 0.18),
            (0.0016, 0.20),
            (0.00045, 0.20),
            (0.00012, 0.28),
        )

    start = time.monotonic()
    avail = float(stop_time - start - 5.0)
    if avail <= 5.0:
        return best_p

    total_w = float(sum(w for _, w in stages))
    if total_w <= 0.0:
        total_w = 1.0
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 6.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 5.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:
            continue

        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-14, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_new = _softmax(u)
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if math.isfinite(ss) and ss > 1e-300:
            p_new /= ss
        else:
            p_new = best_p.copy()

        sc_new = _score_fft(p_new)
        if math.isfinite(sc_new) and sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_p = p_new.copy()
            p = p_new.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass
        else:
            p = best_p.copy()
        u = p_to_u(p)

    if best_p.size == n:
        return best_p
    return p


def _lp_solve_direction(a, M):
    if linprog is None or sp is None:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n < 32 or n > 600:
        return None
    if not math.isfinite(M) or M <= 0.0:
        return None

    try:
        idx = np.arange(n, dtype=np.int32)
        cols = np.repeat(idx, n)
        off = np.tile(idx, n)
        rows = cols + off
        data = a[off].astype(np.float64)
        A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        cobj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        try:
            res = linprog(
                cobj,
                A_ub=A,
                b_ub=b_ub,
                bounds=(0.0, None),
                method="highs",
            )
        except Exception:
            res = linprog(cobj, A_ub=A, b_ub=b_ub, bounds=(0.0, None))
        if res is not None and getattr(res, "success", False):
            b = np.asarray(res.x, dtype=np.float64).ravel()
            b = np.where(np.isfinite(b), b, 0.0)
            b = np.maximum(b, 0.0)
            if float(b.sum()) > 0.0:
                return b
    except Exception:
        return None
    return None


def _lp_improve_small(src, n_small, stop_time, max_iter=4):
    n_small = int(n_small)
    if n_small <= 16 or n_small > 576:
        return None
    if time.monotonic() > stop_time:
        return None

    a = _resample_mid(src, n_small)
    a = _normalize_nonneg(a)
    best_a = a.copy()
    best_sc = _score_fft(a)
    if not math.isfinite(best_sc):
        return None

    for _ in range(max_iter):
        if time.monotonic() > stop_time - 8.0:
            break
        c = _fft_conv(a)
        M = float(c.max()) if c.size else 0.0
        if not math.isfinite(M) or M <= 0.0:
            break
        b = _lp_solve_direction(a, M)
        if b is None:
            break
        s = float(b.sum())
        if not math.isfinite(s) or s <= 0.0:
            break
        g = b / s
        improved = False
        for t in (0.55, 0.35, 0.20, 0.10, 0.05):
            cand = (1.0 - t) * a + t * g
            cand = np.maximum(cand, 0.0)
            ss = float(cand.sum())
            if not math.isfinite(ss) or ss <= 0.0:
                continue
            cand /= ss
            sc = _score_fft(cand)
            if math.isfinite(sc) and sc < best_sc - 1e-10:
                a = cand
                best_sc = sc
                best_a = cand.copy()
                improved = True
                break
        if not improved:
            break

    if math.isfinite(_score_fft(best_a)):
        return best_a
    return None


def _coord_eval(xs, base, infl_vec, max_const, v, Sloc, n, di):
    xs = np.asarray(xs, dtype=np.float64).ravel()
    if xs.size == 0:
        return np.zeros(0, dtype=np.float64), xs
    w = xs - v
    den = Sloc + w
    valid = den > 1e-14
    aff = base[None, :] + w[:, None] * infl_vec[None, :]
    Ls = base.size
    if 0 <= di < Ls:
        aff[:, di] = base[di] + w * infl_vec[di] + w * w
    np.maximum(aff, max_const, out=aff)
    mv = np.max(aff, axis=1)
    vals = np.full(xs.size, math.inf, dtype=np.float64)
    if np.any(valid):
        vals[valid] = (2.0 * n * mv[valid]) / (den[valid] * den[valid])
    vals = np.where(np.isfinite(vals), vals, math.inf)
    return vals, xs


def _coord_polish(p, stop_time, G=64, seed=12345, consider=None):
    p = _normalize_nonneg(p)
    n = int(p.size)
    if n <= 1:
        return p, math.inf

    c = _fft_conv(p)
    c = np.maximum(c, 0.0)
    S = 1.0
    infl = 2.0 * p
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass

    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < 1000000 and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)

        for pos in range(n):
            if time.monotonic() > stop_time - 0.4:
                break
            if S <= 1e-300:
                break
            if S > 1.20 or S < 0.80:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                infl *= inv
                S = 1.0
                c = np.maximum(c, 0.0)
                cur_score = float(2.0 * n * c.max())

            i = int(order[pos])
            v = float(p[i])
            L = 2 * n - 1
            start = i
            end = min(i + n, L)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end]
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < L:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc
            if not math.isfinite(max_const) or max_const < 0.0:
                max_const = 0.0

            Sloc = float(S)
            xhi = max(20.0 * v, 20.0 * Sloc / float(n), 0.004)
            xhi = min(xhi, max(0.05, v))
            if not math.isfinite(xhi) or xhi <= 0.0:
                xhi = max(1e-8, 10.0 * Sloc / float(n))

            pts = [0.0, v, xhi]

            m1 = max(12, int(G * 0.45))
            minx = max(1e-9, 1e-5 * v, Sloc / (2000.0 * float(n)))
            if minx < xhi:
                pts.extend(np.geomspace(minx, xhi, m1, dtype=np.float64).tolist())

            if v > 1e-12:
                m2 = max(8, int(G * 0.25))
                factors = np.geomspace(0.2, 5.0, m2, dtype=np.float64)
                pts.extend((v * factors).tolist())

            m3 = max(10, int(G * 0.30))
            linhi = min(xhi, max(8.0 * Sloc / float(n), 0.0015))
            if linhi > 0.0:
                pts.extend(np.linspace(0.0, linhi, m3, dtype=np.float64).tolist())

            pts = np.unique(np.clip(np.asarray(pts, dtype=np.float64), 0.0, xhi))
            if pts.size == 0:
                pts = np.array([v], dtype=np.float64)

            di = 2 * i - start
            vals, pts = _coord_eval(pts, base, infl, max_const, v, Sloc, n, di)
            if not np.any(np.isfinite(vals)):
                continue
            im = int(np.argmin(vals))
            best_x = float(pts[im])
            best_val = float(vals[im])

            if best_val < cur_score + max(1e-10, 1e-5 * cur_score):
                if best_x <= 0.0:
                    lo = 0.0
                    hi = max(5.0 * Sloc / float(n), 5.0 * v, 1e-7)
                else:
                    step = max(v, Sloc / float(n), 1e-9)
                    lo = max(0.0, best_x * 0.5, best_x - 2.5 * step)
                    hi = min(xhi, max(best_x * 1.5, best_x + 2.5 * step))
                if hi <= lo:
                    hi = lo + 1e-9
                fine = [best_x]
                fine.extend(np.linspace(lo, hi, 36, dtype=np.float64).tolist())
                if lo > 0.0:
                    fine.extend(np.geomspace(lo, hi, 16, dtype=np.float64).tolist())
                else:
                    fine.extend(np.linspace(0.0, hi, 16, dtype=np.float64).tolist())
                fine = np.unique(np.clip(np.asarray(fine, dtype=np.float64), 0.0, xhi))
                vals2, fine = _coord_eval(fine, base, infl, max_const, v, Sloc, n, di)
                if np.any(np.isfinite(vals2)):
                    im2 = int(np.argmin(vals2))
                    vx2 = float(fine[im2])
                    vv2 = float(vals2[im2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2

            if (
                math.isfinite(best_val)
                and best_val < cur_score - 1e-11
                and 0.0 <= best_x <= xhi
            ):
                d = best_x - v
                if abs(d) > 1e-16:
                    c[start:end] += d * infl
                    diag = 2 * i
                    if 0 <= diag < L:
                        c[diag] += d * d
                    p[i] = best_x
                    infl[i] = 2.0 * best_x
                    S += d
                    cur_score = best_val
                    improved = True

        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            infl *= inv
            S = 1.0
        else:
            break
        c = _fft_conv(p)
        c = np.maximum(c, 0.0)
        desc = float(2.0 * n * c.max())
        if math.isfinite(desc) and desc < best_score - 1e-13:
            best_score = desc
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 25.0:
                    sigmas = (0.035, 0.07, 0.11, 0.16)
                    sigma = sigmas[min(flat - 4, len(sigmas) - 1)]
                    noise = rng.normal(0.0, sigma, n, dtype=np.float64)
                    q = p * np.exp(noise)
                    if sigma >= 0.11:
                        x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
                        amp = 0.20 + 0.35 * float(rng.random())
                        f1 = 1 + int(rng.integers(0, 4))
                        ph1 = 2.0 * math.pi * float(rng.random())
                        q *= np.exp(amp * math.sin(2.0 * math.pi * f1 * x + ph1))
                    q = np.maximum(q, 0.0)
                    qs = float(q.sum())
                    if math.isfinite(qs) and qs > 0.0:
                        p = q / qs
                        c = _fft_conv(p)
                        c = np.maximum(c, 0.0)
                        S = 1.0
                        infl = 2.0 * p
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider is not None:
                                try:
                                    consider(best_p)
                                except Exception:
                                    pass
                    flat = 0
                else:
                    break
        sweep += 1

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(10.0, budget)
    if budget > 300.0:
        reserve = 25.0
    elif budget > 60.0:
        reserve = 12.0
    else:
        reserve = 2.5
    hard = t0 + budget - reserve
    if hard <= t0 + 2.0:
        hard = t0 + max(1.5, budget - 0.5)

    parent = _get_parent()
    n_target = int(parent.size) if parent.size > 0 else 3072

    best_arr = _normalize_nonneg(parent.copy())
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.full(64, 1.0 / 64.0, dtype=np.float64)
        try:
            best_score = float(_score_exact_arr(best_arr))
        except Exception:
            best_score = math.inf

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return
        if arr.size == 0:
            return
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if not math.isfinite(s) or s < 0.01:
            return
        arr = arr / s
        try:
            sc = float(_score_exact_arr(arr))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-13:
            best_arr = arr.copy()
            best_score = sc

    phase_limit = min(hard - 8.0, t0 + max(1.0, budget - max(20.0, 0.04 * budget)))
    rng_top = np.random.default_rng(seed)
    explored = [best_arr.copy()]

    def add_explore(cand):
        cand = _normalize_nonneg(cand)
        if cand.size != n_target:
            cand = _resample_mid(cand, n_target)
        cand = _normalize_nonneg(cand)
        sc = _score_fft(cand)
        base_sc = _score_fft(best_arr)
        if math.isfinite(sc) and math.isfinite(base_sc) and sc < base_sc + 0.06:
            if any(np.allclose(cand, x, rtol=1e-7, atol=1e-9) for x in explored):
                return
            explored.append(cand)

    # Cheap explicit shaped perturbations.
    base = best_arr.copy()
    power_ts = (0.55, 0.80, 1.20, 1.55, 2.10)
    for q in power_ts:
        if time.monotonic() > phase_limit - 60.0:
            break
        cand = np.power(np.maximum(base, 0.0), float(q))
        cand = _normalize_nonneg(cand)
        consider(cand)
        add_explore(cand)

    for _ in range(5):
        if time.monotonic() > phase_limit - 90.0:
            break
        side = 1.0 if rng_top.random() < 0.5 else -1.0
        amp = 0.30 + 0.95 * float(rng_top.random())
        x = (np.arange(n_target, dtype=np.float64) + 0.5) / float(n_target)
        cand = base * np.exp(side * amp * (2.0 * x - 1.0))
        cand = np.maximum(cand, 0.0)
        cand = _normalize_nonneg(cand)
        consider(cand)
        add_explore(cand)

    # Coarse LP-generated shapes, if affordable.
    if budget > 360.0 and time.monotonic() < phase_limit - 130.0:
        small_lens = [256, 384, 512]
        if n_target > 2048:
            small_lens.append(512)
        small_lens = sorted({x for x in small_lens if x < n_target})
        for ns in small_lens:
            if time.monotonic() > phase_limit - 90.0:
                break
            stop = min(phase_limit, time.monotonic() + (30.0 if ns <= 256 else 55.0 if ns <= 384 else 75.0))
            try:
                small = _lp_improve_small(best_arr, ns, stop, max_iter=3)
            except Exception:
                small = None
            if small is not None:
                cand = _resample_mid(small, n_target)
                cand = _normalize_nonneg(cand)
                consider(cand)
                add_explore(cand)

    def run_attempt(p, stop, seed_int, lse_frac=0.16, coord_G=64, fine_lse=False):
        p = _normalize_nonneg(p)
        consider(p)
        now = time.monotonic()
        rem = float(stop - now)
        if rem <= 5.0:
            return p
        if do_lse_possible := (rem > 35.0):
            lse_stop = now + min(
                130.0,
                max(20.0, float(lse_frac) * rem),
            )
            lse_stop = min(lse_stop, stop - 25.0)
            if lse_stop > now + 12.0:
                p = _lbfgs_optimize(
                    p,
                    lse_stop,
                    consider=consider,
                    fine_only=fine_lse,
                )
        coord_stop = min(
            float(stop) - 2.0,
            time.monotonic() + max(20.0, 0.93 * max(0.0, stop - time.monotonic())),
        )
        if coord_stop > time.monotonic() + 12.0:
            p, _ = _coord_polish(
                p,
                coord_stop,
                G=int(coord_G),
                seed=int(seed_int),
                consider=consider,
            )
        consider(p)
        return p

    now = time.monotonic()
    primary_stop = min(hard - 6.0, phase_limit)

    # One alternate start if it is competitive and time permits.
    if len(explored) > 1 and primary_stop - now > 520.0:
        alt = explored[1]
        sc_alt = _score_fft(alt)
        sc_best = _score_fft(best_arr)
        if math.isfinite(sc_alt) and math.isfinite(sc_best) and sc_alt < sc_best + 0.04:
            alt_stop = min(primary_stop, now + 270.0)
            if alt_stop - now > 80.0:
                try:
                    run_attempt(alt, alt_stop, seed + 911, lse_frac=0.20, coord_G=56, fine_lse=False)
                except Exception:
                    pass

    # Main optimization on the current best exact point.
    primary_stop = min(
        hard - 5.0,
        time.monotonic() + max(30.0, phase_limit - time.monotonic()),
    )
    if primary_stop > time.monotonic() + 30.0:
        try:
            p_main = best_arr.copy()
            p_main = run_attempt(
                p_main,
                primary_stop,
                seed + 1729,
                lse_frac=(0.12 if len(explored) > 1 else 0.18),
                coord_G=72,
                fine_lse=False,
            )
            consider(p_main)
        except Exception:
            pass

    # Final high-resolution squeeze if meaningful time remains.
    if time.monotonic() < hard - 150.0 and best_arr.size > 0:
        try:
            p_final = best_arr.copy()
            rem = max(0.0, hard - 6.0 - time.monotonic())
            lse_stop = time.monotonic() + min(90.0, max(25.0, 0.22 * rem))
            lse_stop = min(lse_stop, hard - 35.0)
            if lse_stop > time.monotonic() + 15.0:
                p_final = _lbfgs_optimize(
                    p_final,
                    lse_stop,
                    consider=consider,
                    fine_only=True,
                )
