def _fft_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fast(a):
    """Fast score computation."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

    else:

        i_hi = min(k, n - 1)

def _lp_minimax(f, n, timeout=3.0):
    """Solve: min M s.t. conv(f, a)_k - M/2 <= conv(f,f)_k/2, sum(a)=1, a>=0, M>=0"""
    ff = _fft_conv(f)  # f*f convolution
    # Build constraint matrix
    # Inequality: sum_j a_j * f_{k-j} - 0.5*M <= ff_k / 2
    A_ineq = np.zeros((2*n-1, n+1))
    b_ineq = np.zeros(2*n-1)
    for k in range(2*n-1):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            # A_ineq[k, i] = f[k-i] for i in [i_lo, i_hi]
            idx = np.arange(i_lo, i_hi+1)
            A_ineq[k, i_lo:i_hi+1] = f[k - idx]
        A_ineq[k, n] = -0.5  # coefficient of M
        b_ineq[k] = ff[k] / 2.0
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M
    c = np.zeros(n+1)
    c[n] = 1.0
    
    bounds = [(0.0, None)] * (n+1)
    
    try:
        res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None, None
    
    if res.status != 0 or res.x is None:
        return None, None
    
    a_new = np.clip(res.x[:n], 0, None)
    M_new = max(0.0, float(res.x[n]))
    
    if a_new.sum() < 1e-15:
        return None, None
    
    a_new = a_new / a_new.sum()  # Ensure sum=1 exactly
    return a_new, M_new

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(414, dtype=np.float64)
    if p.size == 0:
        p = np.ones(414, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(414, dtype=np.float64)
    return p

def _fft_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    # Small negative values due to floating point
    np.maximum(c, 0.0, out=c)
    return c

def _score_fast(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _lp_minimax(f, n, timeout=3.0):
    """Sequential linearization: min M s.t. 2*conv(f,a)_k - ff_k <= M, sum(a)=1, a>=0, M>=0"""
    ff = _fft_conv(f)
    
    # Inequality: sum_j a_j * f_{k-j} <= (M + ff_k) / 2
    # Rearranged: sum_j a_j * f_{k-j} - 0.5*M <= ff_k / 2
    n_ineq = 2 * n - 1
    A_ineq = np.zeros((n_ineq, n + 1))
    b_ineq = ff / 2.0
    
    for k in range(n_ineq):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            A_ineq[k, i_lo:i_hi+1] = f[k - idx]
        A_ineq[k, n] = -0.5
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0.0, None)] * (n + 1)
    
    try:
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    a_new = np.clip(res.x[:n], 0.0, None)
    if a_new.sum() < 1e-15:
        return None
    a_new /= a_new.sum()
    return a_new

def _lp_direction(f, n, M, top_frac=0.25, timeout=2.0):
    """Original LP: max sum(g) s.t. conv(f,g)_k <= M, g>=0. Uses top constraints."""
    b = _fft_conv(f)
    
    if top_frac < 1.0:
        n_active = max(1, int(n_ineq * top_frac))
        # Get indices of largest b values
        active = np.argpartition(b, -n_active)[-n_active:]
    else:
        active = np.arange(2 * n - 1)
    
    A = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            idx_arr = np.arange(i_lo, i_hi + 1)
            A[idx, i_lo:i_hi+1] = f[k - idx_arr]
    
    c = -np.ones(n)
    b_ub = M * np.ones(len(active))
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
