def _gradient_descent(a, stop_time, seed=0):

    best_f = f.copy()
    best_score = _score(f)

    no_imp = 0
    max_no_imp = 50
    
    while time.monotonic() < stop_time - 5.0 and no_imp < max_no_imp:
        b = _conv_full(f)
        M = float(b.max())
        S = float(f.sum())
        # Find all indices achieving the max (within tolerance)
        k_star = np.where(b >= M - 1e-10)[0]
        
        # Compute gradient
        grad = np.zeros(n)
        for k in k_star:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += f[j]
        # Average if multiple maxima
        if len(k_star) > 1:
            grad /= len(k_star)
        
        # normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Gradient descent step (we want to minimize M/S^2)
        # d(score)/d(f_i) = (2n/S^2) * grad_i - (2n*M/S^3) * 1
        # = (2n/S^2) * (grad_i - M/S)

        # d_i = -(grad_i * S - 2*M) / S = 2*M/S - grad_i
