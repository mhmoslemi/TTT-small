import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_bc(a, conv):
    """Score from precomputed convolution."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    m = float(conv.max())
    return 2.0 * n * m / (s * s)

def _build_conv_matrix(f, n):
    """Build the (2n-1) x n matrix A where (A@g)[k] = (f*g)[k]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _line_search_max(f, g, M, stop_time):
    """Find t in [0,1] minimizing max_conv((1-t)f + tg)."""
    best_t = 0.0
    best_m = M
    for t in [0.1, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]:
        if time.monotonic() > stop_time:
            break
        ft = (1.0 - t) * f + t * g
        mt = float(_conv_full(ft).max())
        if mt < best_m:
            best_m = mt
            best_t = t
    # Refine
    if 0.05 < best_t < 0.95:
        lo, hi = max(0.0, best_t - 0.1), min(1.0, best_t + 0.1)
        gr = (1.0 + math.sqrt(5.0)) / 2.0
        for _ in range(7):
            if time.monotonic() > stop_time:
                break
            m1 = hi - (hi - lo) / gr
            m2 = lo + (hi - lo) / gr
            v1 = float(_conv_full((1.0 - m1) * f + m1 * g).max())
            v2 = float(_conv_full((1.0 - m2) * f + m2 * g).max())
            if v1 < v2:
                hi, m1 = m2, hi
            else:
                lo, m2 = m1, lo
        tm = (lo + hi) / 2.0
        vm = float(_conv_full((1.0 - tm) * f + tm * g).max())
        if vm < best_m:
            best_m = vm
            best_t = tm
    return best_t, best_m

        A = _build_conv_matrix(f, n)
        # Use top-k constraints
        top_k = min(200, 2 * n - 1)
