def _lp_direction(f, n, M, timeout=5.0):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-15)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"presolve": True, "time_limit": timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g

- For the term where k-j = i, i.e., j = k-i: contribution changes from f[k-i]*f[i] to f[k-i]*(f[i]+d), so delta = d*f[k-i] (valid if 0 <= k-i < n and k-i ≠ i)

Actually wait. When j = i and k-j = i (i.e., k = 2i), the term is f[i]*f[i] = f[i]². After change: (f[i]+d)² = f[i]² + 2d*f[i] + d². So delta = 2*d*f[i] + d².

When j = i but k-j ≠ i: delta = d*f[k-i]
When k-j = i but j ≠ i (i.e., j = k-i ≠ i): delta = d*f[k-i]
When both (k = 2i): delta = 2*d*f[i] + d²

So the total delta for c[k]:
- If k-i is out of range [0, n-1]: delta = 0
- If k = 2i (and 0 <= i < n): delta = 2*d*f[i] + d²
- If k ≠ 2i and 0 <= k-i < n: delta = 2*d*f[k-i]
- If k = 2i is out of range: doesn't apply

Let me also think about the 2-coordinate move. If I change both f[i] and f[j] simultaneously, the change in c[k] is more complex:
- From changing f[i] by di: delta_i[k] = 2*di*f[k-i] (or 2*di*f[i]+di² if k=2i)
- From changing f[j] by dj: delta_j[k] = 2*dj*f[k-j] (or 2*dj*f[j]+dj² if k=2j)
- Cross term: the product di*dj appears when both i and j contribute to c[k], i.e., when k-i = j (k = i+j) or k-j = i (k = i+j). So there's an extra di*dj at k = i+j.

So: delta_c[k] = delta_i[k] + delta_j[k] + di*dj*(indicator(k = i+j))

This is still tractable. The affected range is the union of [i, i+n-1] and [j, j+n-1] plus the point k=i+j.

import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    if n <= 800:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)
    c = c[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    """A[k,j] = f[k-j] if 0<=k-j<n, else 0. Shape (2n-1, n)."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, n, M, timeout=5.0):
    """LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    if M <= 1e-18:
        return None
    A = _build_A(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-12)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs",
                      options={"presolve": True, "time_limit": timeout,
                               "dual_feasibility_tolerance": 1e-10,
                               "primal_feasibility_tolerance": 1e-10})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g


def _lp_optimize(f0, n, stop_time, seed):

    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n)
        f_sum = n

    target_sum = math.sqrt(2.0 * n)
    f = f / f_sum * target_sum

    best_f = f.copy()
    best_score = _score(f)

    t = 0.4
    stall = 0
    max_stall = 80
    iteration = 0

    while time.monotonic() < stop_time - 5.0:
        iteration += 1
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_solve(f, n, M)
        if g is None:
            stall += 5
            if stall > max_stall:
                # Perturb and restart
                f = best_f.copy()
                f += rng.normal(0, 0.05 * max(1.0, best_f.mean()), n)
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                stall = 0
                max_stall = int(max_stall * 1.3)
                t = 0.4
            continue

        g_sum = float(g.sum())
        g = g * (target_sum / g_sum)

        # Check if g is close to f (convergence)
        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < 1e-10:
            stall = 10
            if stall > max_stall:
                f = best_f.copy()
                idx = rng.choice(n, size=min(20, n), replace=False)
                for k in idx:
                    f[k] = max(0.0, f[k] * rng.uniform(0.2, 3.0))
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                stall = 0
                max_stall = int(max_stall * 1.5)
                t = 0.4
            continue

        # Line search: try multiple step sizes
        improved = False
        step_candidates = sorted(set([
            min(1.0, t), min(1.0, t * 1.5), min(1.0, t * 0.5),
            0.1, 0.3, 0.5, 0.7, 1.0
        ]), reverse=True)

        best_t = 0
        best_sc_in_line = best_score

        for tc in step_candidates:
            if tc < 0.01:
                continue
            f_new = (1.0 - tc) * f + tc * g
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 0.01:
                continue
            f_new = f_new / fs_new * target_sum
            sc = _score(f_new)
            if sc < best_sc_in_line - 1e-13:
                best_sc_in_line = sc
                best_t = tc

        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0, None)
            f = f / f.sum() * target_sum
            if best_sc_in_line < best_score - 1e-13:
                best_f = f.copy()
                best_score = best_sc_in_line
            improved = True
            t = min(1.0, best_t * 1.2)
            stall = 0
        else:

                strength = 0.05 + 0.03 * min(stall // max_stall, 5)
                f += rng.normal(0, strength * max(1.0, best_f.mean()), n)
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                stall = 0
                max_stall = int(max_stall * 1.4)
                t = 0.4
