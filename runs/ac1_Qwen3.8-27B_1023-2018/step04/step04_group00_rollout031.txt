        s = float(x.sum())
        if s < 0.01:
            return
        xn = x / s
        sc = exact_score(xn)
        if sc < best_sc - 1e-15:
            best_a = xn.copy()
            best_sc = sc

    # Phase 1: Brief L-BFGS with log-sum-exp to escape potential flat spots
    def lse_objective(u, T):
        pp = np.exp(u - np.max(u))
        pp /= pp.sum()
        c = compute_conv(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(u)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        se = float(exps.sum())
        if not math.isfinite(se) or se <= 0:
            f = cmax
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            f = cmax + T * math.log(se / c.size)
            w = exps / se
        # Gradient: d/dpp of max_c approx = 2*(pp*w)_diag-ish
        # For self-conv: d(b_k)/d(a_i) = a_{k-i} + a_{k-i} (or a_i if k=2i)
        # Gradient of w^T b with respect to pp: sum_k w_k * db_k/dpp_i
        # db_k/dpp_i = 2*pp_{k-i} (for valid indices) or pp_i if k=2i
        # This is: 2 * (pp * w)_cross
        # Use FFT for cross-correlation
        L2 = 3 * n - 2
        sz2 = next_fast_len(L2)
        Fp = rfft(pp[::-1], sz2)
        Fw = rfft(w, sz2)
        xc = irfft(Fp * Fw, sz2)[:L2]
        q = xc[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        grad = 2.0 * pp * q
        tdp = float(np.dot(pp, grad))
        grad = pp * (grad - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return f, grad

    c_init = compute_conv(a)
    cmax_init = float(c_init.max())
    if cmax_init > 0 and time.monotonic() < deadline - 200.0:
        # Convert to log-parameterization
        a_safe = np.maximum(a, 1e-300)
        u0 = np.log(a_safe) - np.mean(np.log(a_safe))
        lse_stop = min(deadline - 180.0, t0 + 120.0)
        for T_frac in (0.1, 0.04, 0.01, 0.003):
            if time.monotonic() > lse_stop:
                break
            T = max(1e-8, T_frac * cmax_init)
            stage_end = min(lse_stop, time.monotonic() + max(20.0, 40.0 * T_frac / 0.1))
            for _ in range(3):
                if time.monotonic() > stage_end - 2.0:
                    break
                try:
                    res = minimize(lse_objective, u0, args=(T,), method='L-BFGS-B',
                                   jac=True, options={'maxiter': 30, 'gtol': 1e-9,
                                                       'ftol': 1e-14, 'maxls': 15,
                                                       'maxcor': 15, 'maxdisp': 0, 'disp': False})
                except Exception:
                    break
                if res.x is not None:
                    x = np.asarray(res.x, dtype=np.float64).ravel()
                    if x.size == n and np.all(np.isfinite(x)):
                        u0 = x
                        pp = np.exp(x - np.max(x))
                        pp /= pp.sum()
                        sc = exact_score(pp)
                        if sc < best_sc:
                            best_a = pp.copy()
                            best_sc = sc
                if getattr(res, 'status', 1) == 0:
                    break
        # Use result if better
        pp = np.exp(u0 - np.max(u0))
        pp /= pp.sum()
        sc_pp = exact_score(pp)
        if sc_pp < best_sc:
            a = pp.copy()
            best_a = pp.copy()
            best_sc = sc_pp
        else:
            a = best_a.copy()

    # Phase 2: Extended coordinate descent
    a = best_a.copy()
    c = compute_conv(a)
    cur_sc = 2.0 * n * float(c.max())
    rng = np.random.default_rng(seed)

    G = 44
    G_fine = 56
    noise_sigma = 0.04
    max_sweeps = 999999

    sweep = 0
    flat = 0

    while time.monotonic() < deadline - 4.0 and sweep < max_sweeps:
        improved = False
        order = rng.permutation(n)

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > deadline - 3.0:
                break

            v = float(a[i])
            start = i
            end = min(i + n, conv_len)
            Ls = end - start
            if Ls <= 0:
                continue

            # Influence vector: 2*a[k-i] for k in [i, i+Ls)
            infl = 2.0 * a[:Ls]
            # Diagonal position
            diag_pos = 2 * i - start  # = i
            # infl[diag_pos] = 2*a[i] = 2v (already correct since a[:Ls][i] = a[i] = v)

            # Max outside influence
            max_const = 0.0
            if start > 0:
                mc = float(np.max(c[:start]))
                if mc > max_const:
                    max_const = mc
            if end < conv_len:
                mc = float(np.max(c[end:]))
                if mc > max_const:
                    max_const = mc

            # Search range
            x_hi = max(30.0 * v, 5.0 / n, 1e-6)
            x_hi = min(x_hi, 10.0 / n * 3)  # cap
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = 2.0 / n

            # Grid evaluation
            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.append(pts, v)
            pts = np.unique(pts)
            t_arr = pts - v
            base = c[start:end]

            # Vectorized: max_aff[j, k] = base[k] + t[j]*infl[k] for all j,k
            max_aff = base[None, :] + t_arr[:, None] * infl[None, :]
            max_aff = np.maximum(max_aff, max_const)
            if 0 <= diag_pos < Ls:
                max_aff[:, diag_pos] += t_arr * t_arr
            max_aff = np.maximum(max_aff, 0.0)
            scores = (2.0 * n * max_aff) / (1.0 + t_arr) ** 2

            finite_mask = np.isfinite(scores)
            if not np.any(finite_mask):
                continue
            imin = int(np.argmin(np.where(finite_mask, scores, np.inf)))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            # Fine search
            if G > 2 and math.isfinite(best_val):
                dw = x_hi / G
                lo_f = max(0.0, best_x - 2.5 * dw)
                hi_f = min(x_hi, best_x + 2.5 * dw)
                if hi_f <= lo_f:
                    hi_f = lo_f + 1e-10
                fine = np.linspace(lo_f, hi_f, G_fine, dtype=np.float64)
                if 0.0 < best_x < x_hi:
                    fine = np.append(fine, best_x)
                fine = np.unique(fine)
                tf = fine - v
                max_aff_f = base[None, :] + tf[:, None] * infl[None, :]
                max_aff_f = np.maximum(max_aff_f, max_const)
                if 0 <= diag_pos < Ls:
                    max_aff_f[:, diag_pos] += tf * tf
                max_aff_f = np.maximum(max_aff_f, 0.0)
                scores_f = (2.0 * n * max_aff_f) / (1.0 + tf) ** 2
                fm = np.isfinite(scores_f)
                if np.any(fm):
                    imin_f = int(np.argmin(np.where(fm, scores_f, np.inf)))
                    if scores_f[imin_f] < best_val:
                        best_x = float(fine[imin_f])
                        best_val = float(scores_f[imin_f])

            # Apply update
            if math.isfinite(best_val) and best_val < cur_sc - 1e-14 and 0.0 <= best_x:
                w = best_x - v
                if abs(w) > 1e-16:
                    a[i] = best_x
                    c[start:end] += w * infl
                    dp = 2 * i
                    if 0 <= dp < conv_len:
                        c[dp] += w * w
                    cur_sc = best_val
                    improved = True

        # End of sweep
        sweep += 1

        # Periodic exact convolution recompute (every 50 sweeps or when flat)
        if sweep % 50 == 0 or flat >= 3:
            c = compute_conv(a)
            np.maximum(c, 0.0, out=c)
            cur_sc = 2.0 * n * float(c.max())

        if cur_sc < best_sc - 1e-15:
            best_sc = cur_sc
            best_a = a.copy()

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 7 and time.monotonic() < deadline - 80.0:
                # Noise restart
                noise = np.exp(rng.normal(0.0, noise_sigma, n, dtype=np.float64))
                a = a * noise
                a = np.maximum(a, 0.0)
                s = float(a.sum())
                if s > 0:
                    a /= s
                c = compute_conv(a)
                cur_sc = 2.0 * n * float(c.max())
                flat = 0
                # If noise made things much worse, revert
                if cur_sc > best_sc * 1.5:
                    a = best_a.copy()
                    c = compute_conv(a)
                    cur_sc = 2.0 * n * float(c.max())

            elif flat >= 15 and time.monotonic() < deadline - 80.0:
                # Stronger restart
                noise = np.exp(rng.normal(0.0, noise_sigma * 2.0, n, dtype=np.float64))
                a = a * noise
                a = np.maximum(a, 0.0)
                s = float(a.sum())
                if s > 0:
                    a /= s
                c = compute_conv(a)
                cur_sc = 2.0 * n * float(c.max())
                flat = 0
                if cur_sc > best_sc * 2.0:
                    a = best_a.copy()
                    c = compute_conv(a)
                    cur_sc = 2.0 * n * float(c.max())

    # Final consideration
    consider(a)
    consider(best_a)

    # Final validation
    out = best_a.copy()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s < 0.01:
        out = a.copy()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if s < 0.01:
            out = np.ones(64, dtype=np.float64) / 64.0

    result = [float(x) for x in out]
    try:
        fs = float(evaluate_sequence(result))
        if not math.isfinite(fs):
            result = [float(x) for x in parent]
    except Exception:
        result = [float(x) for x in parent]

    return result
