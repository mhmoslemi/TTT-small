    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _lp_direction(f, n, M=None):
    if M is None:
        b = _conv_full(f)
        M = float(b.max())
    if M < 1e-15:
        return None
    
    # Build constraint matrix: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_{j: 0<=j<n, 0<=k-j<n} f[k-j] * g[j]
    # So row k of A has A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    
    A = np.zeros((2*n-1, n))
    for j in range(n):
        start = j
        end = min(j + n, 2*n - 1)
        A[start:end, j] = f[:end - start]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.status != 0 or res.x is None:
        return None
    g = res.x
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s  # normalize to sum 1

import numpy as np
import time
import math
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(400, dtype=np.float64)
    if p.size == 0:
        p = np.ones(400, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(400, dtype=np.float64)
    return p


def _conv(a):
    n = len(a)
    if n <= 1024:
        return np.convolve(a, a, mode="full")
    from scipy.fft import rfft, irfft, next_fast_len
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

            return None
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = f

            -np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
            bounds=[(0, None)] * n, method='highs',
            options={'time_limit': tl})
        if res.status != 0 or res.x is None:
            return None
        g = np.asarray(res.x, dtype=np.float64)
        gs = float(g.sum())
        if gs < 1e-15:
            return None
        return g / gs

    def _cd_eval(b, f, i, S, n, x_g):

        Sr = S - v
        k0 = i
        k1 = min(i + n, 2 * n - 1)
        L = k1 - k0

            return np.full(len(x_g), 0.0)

        f_sl = f[:L]
        b_af = b[k0:k1]

        b_um = 0.0
        if k0 > 0:
            b_um = max(b_um, float(b[:k0].max()))
        if k1 < 2 * n - 1:
            b_um = max(b_um, float(b[k1:].max()))

        w = x_g - v
        mods = b_af[None, :] + 2.0 * w[:, None] * f_sl[None, :]
        if i < L:
            mods[:, i] += w * w

        bm = np.maximum(mods.max(axis=1), b_um)
        bm = np.maximum(bm, 0.0)
        Sn = np.maximum(Sr + x_g, 0.01)
        return 2.0 * n * bm / (Sn * Sn)

                bx = xg[bi]

                    xf = np.linspace(xl, xh, 30)

                    # Save old slice for b update
                    k0 = i
                    k1 = min(i + n, 2 * n - 1)

                    if 2 * i < 2 * n - 1 and i < L:

                    np.maximum(b, 0.0, out=b)
                    sc = bs
                    improved = True

            if not improved and sweep > 5:

        final_sc = _score(f)
        if math.isfinite(final_sc) and final_sc < best_sc - 1e-15:
            best_seq = f.copy()
            best_sc = final_sc
        return f

    # === Phase 1: LP optimization (main workhorse) ===
    f = np.clip(parent, 0, None).astype(np.float64)
    f = f / f.sum()
    t_end1 = t0 + budget * 0.52
