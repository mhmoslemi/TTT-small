def _project_simplex(v):
    v = np.asarray(v, dtype=float)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1,n+1)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond): return np.ones(n)/n
    r = int(np.nonzero(cond)[-1][0])
    theta = (cssv[r-1] - 1.0)/float(r)
    return np.maximum(v - theta, 0.0)
```

def _as_sequence(v, n=None):
    arr = np.asarray(v, dtype=np.float64).ravel()
    if n is None:
        n = arr.size

    if n <= 0:
        n = 64
        arr = np.ones(n, dtype=np.float64)

def _conv_full(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)

    if n <= 1024:
        c = np.convolve(a, a, mode="full")
    else:
        fsize = next_fast_len(2 * n - 1)
        fa = rfft(a, fsize)
        c = irfft(fa * fa, fsize)[:2 * n - 1]

    if c.size:
        np.maximum(c, 0.0, out=c)
    return c


def _conv_cross(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    na, nb = a.size, b.size
    if na == 0 or nb == 0:
        return np.empty(0, dtype=np.float64)

    if min(na, nb) <= 1024:
        c = np.convolve(a, b, mode="full")
    else:
        fsize = next_fast_len(na + nb - 1)
        c = irfft(rfft(a, fsize) * rfft(b, fsize), fsize)[:na + nb - 1]

    if c.size:
        np.maximum(c, 0.0, out=c)
    return c


def _score_np(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return math.inf
    s = float(v.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(v)
    if b.size == 0:
        return math.inf
    m = float(b.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _exact_score(v):

        return _score_np(v)
    try:
        arr = _as_sequence(v)

    except Exception:

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.ones(1, dtype=np.float64)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)

    r = int(np.nonzero(cond)[0][-1]) + 1
    theta = (cssv[r - 1] - 1.0) / float(r)
    w = np.maximum(v - theta, 0.0)
    ss = float(w.sum())
    if ss < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return w / ss


def _line_search_quad(v, g, F=None, stop=None):
    v = np.asarray(v, dtype=np.float64).ravel()
    g = np.asarray(g, dtype=np.float64).ravel()
    n = v.size
    L = 2 * n - 1

    if F is None:
        F = _conv_full(v)
    G = _conv_full(g)
    C = _conv_cross(v, g)

    # Ensure consistent lengths in case of numerical/edge issues.
    while F.size < L:
        F = np.concatenate([F, np.zeros(L - F.size, dtype=np.float64)])
    while G.size < L:
        G = np.concatenate([G, np.zeros(L - G.size, dtype=np.float64)])
    while C.size < L:
        C = np.concatenate([C, np.zeros(L - C.size, dtype=np.float64)])
    F = F[:L]
    G = G[:L]
    C = C[:L]

    def eval_ts(ts):
        ts = np.asarray(ts, dtype=np.float64)
        t = ts[:, None]
        u = 1.0 - t
        W = (u * u) * F[None, :] + (2.0 * t * u) * C[None, :] + (t * t) * G[None, :]
        vals = 2.0 * n * W.max(axis=1)
        vals[~np.isfinite(vals)] = math.inf
        return vals

    K = 32
    ts = np.linspace(0.0, 1.0, K)
    vals = eval_ts(ts)
    j = int(np.argmin(vals))
    best_t = float(ts[j])
    best_sc = float(vals[j])

    if stop is None or time.monotonic() < stop - 8.0:
        if 0 < j < K - 1:
            lo = float(ts[j - 1])
            hi = float(ts[j + 1])
            ts2 = np.linspace(lo, hi, 16)
            vals2 = eval_ts(ts2)
            j2 = int(np.argmin(vals2))
            if vals2[j2] < best_sc:
                best_sc = float(vals2[j2])
                best_t = float(ts2[j2])

    return best_t, best_sc


def _lp_direction(f, n, time_limit=5.0):
    f = np.asarray(f, dtype=np.float64).ravel()
    if n < 1:
        return None, 0.0, np.empty(0)

    F = _conv_full(f)
    M = float(F.max()) if F.size else 0.0
    if not math.isfinite(M) or M <= 0.0:
        return None, M, F

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    slack = max(1e-12, 1e-8 * M)
    b_ub = np.full(2 * n - 1, M + slack, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    tl = float(max(0.05, min(60.0, time_limit)))

    # Max-sum LP: maximize sum(g) subject to f*g <= M, g >= 0.
    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": tl},
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
            gs = float(g.sum())
            if gs > 1e-12:
                return g / gs, M, F
    except Exception:
        pass

    # Fallback: min-peak LP with equality sum(g)=1.  Slightly larger but can
    # recover in a few numerical edge cases.
    try:
        A_ub = np.empty((2 * n - 1, n + 1), dtype=np.float64)
        A_ub[:, 0] = -1.0
        A_ub[:, 1:] = A
        b_ub2 = np.zeros(2 * n - 1, dtype=np.float64)
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, 1:] = 1.0
        b_eq = np.ones(1, dtype=np.float64)
        c = np.zeros(n + 1, dtype=np.float64)
        c[0] = 1.0
        bounds2 = [(0.0, None)] * (n + 1)

        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub2,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds2,
            method="highs",
            options={"presolve": True, "time_limit": tl},
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            g = np.maximum(np.asarray(res.x[1:], dtype=np.float64), 0.0)
            gs = float(g.sum())
            if gs > 1e-12:
                return g / gs, M, F
    except Exception:
        pass

    return None, M, F


def _lp_phase(v0, n, stop, seed=0, max_iters=10000):
    v = _as_sequence(v0, n)
    best_v = v.copy()
    best_sc = _score_np(v)
    if not math.isfinite(best_sc):
        return best_v, 0.0

    rng = np.random.default_rng(seed)
    stall = 0
    fails = 0

    for _ in range(max_iters):
        if time.monotonic() >= stop - 5.0:
            break

        rem = max(0.2, stop - time.monotonic())
        if n <= 512:
            tl = min(12.0, max(1.0, 0.12 * rem))
        elif n <= 1024:
            tl = min(30.0, max(2.0, 0.16 * rem))
        else:
            tl = min(60.0, max(3.0, 0.20 * rem))

        g, M, F = _lp_direction(v, n, time_limit=tl)
        if g is None:
            fails += 1
            if fails > 10:
                break
            v = _as_sequence(v + rng.normal(0.0, 0.01, n), n)
            if _score_np(v) > best_sc * 1.25 + 1e-9:
                v = best_v.copy()
            continue

        fails = 0
        cur = _score_np(v)

        if np.linalg.norm(g - v) < 1e-10:
            stall += 2
            v = _as_sequence(v + rng.normal(0.0, 0.02, n), n)
            if _score_np(v) > best_sc * 1.25 + 1e-9:
                v = best_v.copy()
            continue

        best_t, best_line = _line_search_quad(v, g, F=F, stop=stop)

        if best_t > 1e-6 and best_line < cur - 1e-12:
            nv = v + best_t * (g - v)
            nv = _as_sequence(nv, n)
            sc = _score_np(nv)
            if sc < cur - 1e-13:
                v = nv
                cur = sc
                stall = 0
                if cur < best_sc - 1e-13:
                    best_sc = cur
                    best_v = v.copy()
            else:
                stall += 1
        else:
            stall += 1

        if stall >= 7:
            v = _as_sequence(v + rng.normal(0.0, 0.025, n), n)
            if _score_np(v) > best_sc * 1.3 + 1e-9:
                v = best_v.copy()
            stall = 0

    return best_v, best_sc


def _grad_phase(v0, stop, seed=0):
    v = _as_sequence(v0)
    n = v.size
    best_v = v.copy()
    best_sc = _score_np(v)
    if not math.isfinite(best_sc):
        return best_v, 0.0

    rng = np.random.default_rng(seed)
    step = min(0.05, max(1e-4, 2.0 / float(n)))
    tau = 50.0
    stall = 0

    it = 0
    while time.monotonic() < stop - 3.0:
        it += 1
        c = _conv_full(v)
        cur = 2.0 * n * float(c.max())
        if cur < best_sc - 1e-12:
            best_sc = cur
            best_v = v.copy()

        cmax = float(c.max())
        if cmax <= 0.0:
            break

        shifted = tau * (c - cmax)
        # Shift again after cmax to avoid overflow, though already shifted by cmax.
        shifted -= shifted.max()
        p = np.exp(shifted)
        ps = float(p.sum())
        if ps < 1e-300:
            break
        p /= ps

        # If smoothing has collapsed to one active constraint and we are stalled,
        # use an exact subgradient average over all active max lags.
        if stall > 12 and p.max() > 0.85:
            active = c >= (cmax - max(1e-12, 1e-6 * cmax))
            if active.any():
                p = np.zeros_like(c, dtype=np.float64)
                p[active] = 1.0
                p /= p.sum()

        N = next_fast_len(3 * n)
        Fv = rfft(v, N)
        Fp = rfft(p, N)
        # Gradient of c = v*v is 2 * forward cross-correlation with p.
        h1 = irfft(Fv * np.conjugate(Fp), N)[:n]
        grad = 2.0 * h1
        grad -= grad.mean()
        gnorm = float(np.linalg.norm(grad))

        if gnorm < 1e-13:
            v = _as_sequence(v + rng.normal(0.0, max(1e-5, step * 0.1), n), n)
            stall += 1
            if stall > 60:
                break
            continue

        direction = -grad / gnorm
        improved = False
        for fac in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
            if time.monotonic() >= stop - 3.0:
                break
            w = _project_simplex(v + step * fac * direction)
            sc = _score_np(w)
            if sc < cur - 1e-13:
                v = w
                cur = sc
                step = min(0.1, step * 1.05)
                stall = 0
                improved = True
                break

        if not improved:
            step *= 0.5
            stall += 1
            if step < 1e-7:
                step = min(0.05, max(1e-4, 2.0 / float(n)))
                v = _as_sequence(v + rng.normal(0.0, 5.0 / float(n), n), n)
                c = _conv_full(v)
                cur = 2.0 * n * float(c.max())

            if stall % 40 == 0:
                tau = max(5.0, tau * 0.85)
                step = min(0.05, step * 2.0)

        if stall > 400:
            break

    return best_v, best_sc


def _coord_polish(v0, stop, sweeps=60, seed=0):
    v = _as_sequence(v0)
    n = v.size
    if n == 0:
        return v, 0.0

    conv = _conv_full(v)
    cur = 2.0 * n * float(conv.max())
    best_sc = cur
    best_v = v.copy()

    rng = np.random.default_rng(seed)
    avg = 1.0 / float(n)
    L = 2 * n - 1

    for sweep in range(sweeps):
        if time.monotonic() >= stop - 2.0:
            break

        improved = False
        if sweep % 2 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n, dtype=np.int64)[::-1]

        K = 56 if n <= 600 else 40

        for i in order:
            if time.monotonic() >= stop - 2.0:
                break

            old = float(v[i])

            hi = max(old, min(0.5, max(12.0 * avg, 3.0 * old + 6.0 * avg)))
            if hi <= 0.0:
                hi = 5.0 * avg

            xs = np.linspace(0.0, hi, K, dtype=np.float64)
            np.clip(xs, 0.0, None, out=xs)
            xs[int(np.argmin(np.abs(xs - old)))] = old
            xs = np.unique(xs)
            xs = np.clip(xs, 0.0, None)
            if xs.size > 120:
                xs = np.linspace(xs[0], xs[-1], 120, dtype=np.float64)

            D = xs - old
            if not np.any(np.isfinite(D)):
                continue

            d = np.zeros(L, dtype=np.float64)
            d[i:i + n] = 2.0 * v
            idx2 = 2 * i

            base = conv[None, :] + D[:, None] * d[None, :]
            if 0 <= idx2 < L:
                base[:, idx2] += D * D

            S_new = 1.0 + D
            valid = (S_new >= 0.01) & np.isfinite(S_new)
            mx = base.max(axis=1)
            with np.errstate(divide="ignore", invalid="ignore"):
                obj = np.where(valid, (2.0 * n) * mx / (S_new * S_new), math.inf)

            j = int(np.argmin(obj))
            best_x = float(xs[j])
            best_obj = float(obj[j])

            # Local refinement around the coarse winner.
            if math.isfinite(best_obj) and best_x != old and best_obj < cur - 1e-13:
                step_x = max(hi / max(1, K - 1), 1e-9)
                lo = max(0.0, best_x - 2.0 * step_x)
                hi_f = min(hi, best_x + 2.0 * step_x)
                if hi_f > lo:
                    xs2 = np.linspace(lo, hi_f, 24, dtype=np.float64)
                    D2 = xs2 - old
                    base2 = conv[None, :] + D2[:, None] * d[None, :]
                    if 0 <= idx2 < L:
                        base2[:, idx2] += D2 * D2
                    S2 = 1.0 + D2
                    valid2 = (S2 >= 0.01) & np.isfinite(S2)
                    mx2 = base2.max(axis=1)
                    with np.errstate(divide="ignore", invalid="ignore"):
                        obj2 = np.where(valid2, (2.0 * n) * mx2 / (S2 * S2), math.inf)
                    j2 = int(np.argmin(obj2))
                    if obj2[j2] < best_obj:
                        best_obj = float(obj2[j2])
                        best_x = float(xs2[j2])

            if math.isfinite(best_obj) and best_obj < cur - 1e-13:
                delta = best_x - old
                v[i] = best_x
                conv += delta * d
                if 0 <= idx2 < L:
                    conv[idx2] += delta * delta

                S = 1.0 + delta
                if S < 0.01:
                    v = _as_sequence(v, n)
                    conv = _conv_full(v)
                    cur = 2.0 * n * float(conv.max())
                else:
                    inv = 1.0 / S
                    v *= inv
                    conv *= inv * inv

                cur = best_obj
                improved = True
                if cur < best_sc - 1e-13:
                    best_sc = cur
                    best_v = v.copy()

        if not improved:
            if sweep >= 2:
                break
            v = _as_sequence(v + rng.normal(0.0, 5.0 * avg, n), n)
            conv = _conv_full(v)
            cur = 2.0 * n * float(conv.max())
            if cur < best_sc - 1e-13:
                best_sc = cur
                best_v = v.copy()

    return best_v, best_sc


def _pair_polish(v0, stop, seed=0, max_rounds=2, trials=24):
    v = _as_sequence(v0)
    n = v.size
    if n == 0:
        return v, 0.0
    if n > 900:
        trials = min(trials, 12)

    cur = _score_np(v)
    best_sc = cur
    best_v = v.copy()
    rng = np.random.default_rng(seed)
    avg = 1.0 / float(n)
    mult = (0.0, 0.5, 1.0, 1.5, 2.5)

    def cand(old, m):
        if m == 0.0:
            return 0.0
        if old > 1e-12:
            return float(old) * float(m)
        return 0.25 * avg

    for _round in range(max_rounds):
        improved = False
        for _trial in range(trials):
            if time.monotonic() >= stop - 3.0:
                break

            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue

            oi = float(v[i])
            oj = float(v[j])
            if oi < 1e-12 and oj < 1e-12:
                continue

            best_pair = (oi, oj)
            best_sc_local = cur

            for mi in mult:
                ni = cand(oi, mi)
                for mj in mult:
                    if time.monotonic() >= stop - 3.0:
                        break
                    nj = cand(oj, mj)
                    v[i], v[j] = ni, nj
                    ss = float(v.sum())
                    if ss < 0.01:
                        v[i], v[j] = oi, oj
                        continue
                    sc = _score_np(v)
                    if math.isfinite(sc) and sc < best_sc_local - 1e-14:
                        best_sc_local = sc
                        best_pair = (ni, nj)

            v[i], v[j] = best_pair
            ss = float(v.sum())
            if ss > 0.01:
                v /= ss
            sc = _score_np(v)
            if math.isfinite(sc) and sc < cur - 1e-13:
                cur = sc
                improved = True
                if cur < best_sc - 1e-13:
                    best_sc = cur
                    best_v = v.copy()

        if not improved:
            break

    return best_v, best_sc


def _interp_len(v, L):
    v = np.asarray(v, dtype=np.float64).ravel()
    L = int(max(1, L))
    if v.size == L:
        out = v.copy()
    else:
        x = np.linspace(0.0, 1.0, v.size)
        y = np.linspace(0.0, 1.0, L)
        out = np.interp(y, x, v)
    return _as_sequence(out, L)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not math.isfinite(budget) or budget <= 10.0:
        budget = 10.0

    hard = t0 + budget - 25.0
    if hard < t0 + 5.0:
        hard = t0 + 3.0

    parent = _get_parent()
    n0 = int(parent.size)
    if n0 < 16:
        parent = np.ones(400, dtype=np.float64)
        n0 = 400

    best_seq = _as_sequence(parent, n0)
    best_score = _exact_score(best_seq)
    if not math.isfinite(best_score):
        best_seq = np.ones(64, dtype=np.float64) / 64.0
        best_score = _exact_score(best_seq)
        if not math.isfinite(best_score):
            return [1.0]

    def update_best(v):
        nonlocal best_seq, best_score
        try:
            vv = _as_sequence(v)
            if time.monotonic() > hard - 5.0:
                return
            sc_int = _score_np(vv)
            # Exact evaluation is authoritative, but skip hopeless internal
            # candidates to save time and evaluator overhead.
            if math.isfinite(sc_int) and math.isfinite(best_score) and sc_int > best_score + 0.25:
                return
            sc = _exact_score(vv)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_seq = vv.copy()
        except Exception:
            return

    # Quick multi-start: parent plus a few cheap structured perturbations.
    init_pool = [parent]
    if time.monotonic() < hard - 20.0 and n0 >= 16:
        idx = np.arange(n0, dtype=np.float64)
        p = np.asarray(parent, dtype=np.float64)

        try:
            init_pool.append(np.power(p + 1e-12, 0.5))
            init_pool.append(np.power(p + 1e-12, 2.0))
            init_pool.append(0.5 * (p + p[::-1]))
        except Exception:
            pass

        for f in (1, 2, 3, 4, 6, 8):
            try:
                init_pool.append(p * (1.0 + 0.5 * np.cos(2.0 * math.pi * f * idx / float(n0))))
            except Exception:
                pass

        try:
            init_pool.append(np.ones(n0, dtype=np.float64))
            init_pool.append(np.sin(math.pi * idx / float(n0 - 1)) ** 2)
            init_pool.append(1.0 - 0.7 * np.cos(2.0 * math.pi * (idx - n0 / 2.0) / float(n0)))
            init_pool.append(1.0 + 0.35 * np.sin(math.pi * idx / float(n0 - 1)))
        except Exception:
            pass

    for arr in init_pool:
        if time.monotonic() > hard - 30.0:
            break
        try:
            vv = _as_sequence(arr, n0)
            sc_int = _score_np(vv)
            if math.isfinite(sc_int) and (not math.isfinite(best_score) or sc_int < best_score + 0.05):
                update_best(vv)
        except Exception:
            pass

    v_work = _as_sequence(best_seq, best_seq.size)

    # Phase 1: LP-based bilinear direction search.
    stop1 = min(hard - 120.0, t0 + max(60.0, budget * 0.50))
    if time.monotonic() < stop1 - 5.0:
        v_lp, sc_lp = _lp_phase(v_work, v_work.size, stop1, seed=seed, max_iters=20000)
        update_best(v_lp)
        v_work = _as_sequence(best_seq, best_seq.size)

    # Phase 2: Smoothed-max projected gradient.
    stop2 = min(hard - 80.0, t0 + max(40.0, budget * 0.68))
    if time.monotonic() < stop2 - 5.0:
        v_g, sc_g = _grad_phase(v_work, stop2, seed=seed + 1)
        update_best(v_g)
        v_work = _as_sequence(best_seq, best_seq.size)

    # Phase 3: Coordinate descent with quadratic envelope line searches.
    stop3 = min(hard - 45.0, t0 + max(30.0, budget * 0.85))
    if time.monotonic() < stop3 - 5.0:
        v_c, sc_c = _coord_polish(v_work, stop3, sweeps=120, seed=seed + 2)
        update_best(v_c)
        v_work = _as_sequence(best_seq, best_seq.size)

    # Phase 4: Extra local attempts and nearby lengths if useful.
    stop4 = hard - 35.0
    if time.monotonic() < stop4 - 10.0:
        seed_off = 1000

        modified = []
        bv = np.asarray(best_seq, dtype=np.float64)
        try:
            modified.append(np.power(bv + 1e-12, 0.33))
            modified.append(np.power(bv + 1e-12, 2.5))
            modified.append(np.exp(0.35 * np.log(bv + 1e-12)))
        except Exception:
            pass

        rng_mod = np.random.default_rng(seed + 7)
        for _ in range(4):
            try:
                freq = int(rng_mod.integers(1, 10))
                amp = float(rng_mod.uniform(0.08, 0.45))
                ph = float(rng_mod.uniform(0.0, 2.0 * math.pi))
                idx = np.arange(bv.size, dtype=np.float64)
                modified.append(np.clip(bv * (1.0 + amp * np.cos(2.0 * math.pi * freq * idx / float(bv.size) + ph)), 0.0, 1000.0))
            except Exception:
                pass

        for arr in modified:
            if time.monotonic() > stop4 - 18.0:
                break
            vv = _as_sequence(arr, bv.size)
            update_best(vv)

            alloc = max(15.0, min(70.0, stop4 - time.monotonic() - 20.0))
            stop_mid = min(stop4 - 28.0, time.monotonic() + 0.55 * alloc)
            if time.monotonic() < stop_mid - 3.0:
                v2, sc2 = _lp_phase(vv, vv.size, stop_mid, seed=seed + seed_off, max_iters=5)
                update_best(v2)

                if time.monotonic() < stop4 - 20.0:
                    v3, sc3 = _coord_polish(v2, stop4 - 18.0, sweeps=8, seed=seed + seed_off + 1)
                    update_best(v3)
            seed_off += 17

        # Try a few adjacent lengths only if time remains.
        if time.monotonic() < stop4 - 25.0:
            lens = []
            base_n = int(best_seq.size)
            for L in (
                int(round(base_n * 1.25)),
                int(round(base_n * 0.80)),
                base_n + 64,
                base_n + 128,
                max(64, base_n // 2),
                max(128, int(round(base_n * 2.0 / 3.0))),
            ):
                if 64 <= L <= 1024 and L != base_n:
                    lens.append(int(L))

            lens = sorted(set(lens), key=lambda L: abs(L - base_n))
            for L in lens[:3]:
                if time.monotonic() > stop4 - 22.0:
                    break
                init_l = _interp_len(best_seq, L)
                update_best(init_l)

                alloc = max(20.0, min(90.0, stop4 - time.monotonic() - 20.0))
                stop_lp = min(stop4 - 32.0, time.monotonic() + 0.50 * alloc)
                if time.monotonic() < stop_lp - 3.0:
                    v2, sc2 = _lp_phase(init_l, L, stop_lp, seed=seed + 5000 + L, max_iters=6)
                    update_best(v2)

                    if time.monotonic() < stop4 - 18.0:
                        v3, sc3 = _coord_polish(v2, stop4 - 16.0, sweeps=6, seed=seed + 6000 + L)
                        update_best(v3)

    # Phase 5: Final polish on the current exact best.
    final_end = hard - 5.0
    if time.monotonic() < final_end - 4.0:
        try:
            if time.monotonic() < final_end - 40.0 and best_seq.size <= 900:
                v_p, sc_p = _pair_polish(best_seq, final_end - 35.0, seed=seed + 777, max_rounds=2, trials=24)
                update_best(v_p)

            v_f, sc_f = _coord_polish(_as_sequence(best_seq, best_seq.size), final_end, sweeps=40, seed=seed + 999)
            update_best(v_f)
        except Exception:
            pass

    # Final validation and return.
    try:
        out = _as_sequence(best_seq, best_seq.size)
        sc_final = _exact_score(out)
        if not math.isfinite(sc_final):
            out = _as_sequence(parent, n0)
            sc_final = _exact_score(out)
    except Exception:
        out = np.ones(64, dtype=np.float64) / 64.0
        sc_final = _exact_score(out)

    if not math.isfinite(sc_final):
        out = np.ones(64, dtype=np.float64) / 64.0
        sc_final = _exact_score(out)

    if not math.isfinite(sc_final):
        return [1.0]

    return [float(x) for x in out]
