        return u

        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)

        np.maximum(q, 0.0, out=q)
        return q

        c = cfast(pp)
        cmax = float(c.max())

        if T <= 0:

        else:

            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    best_p = p.copy()
    best_u = u.copy()
