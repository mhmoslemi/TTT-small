import time
from scipy.optimize import linprog

            if cand.size == 0 or not np.all(np.isfinite(cand)) or cand.sum() < 1e-12:
                return
            sc = score_of(cand)
            if np.isfinite(sc) and sc < bestsc:
                bestsc = sc
                best = cand.copy()
        except Exception:
            pass

    def make_ctx(n):
        k_idx = np.arange(2 * n - 1)[:, None]
        j_idx = np.arange(n)[None, :]
        d = k_idx - j_idx
        mask = (d >= 0) & (d < n)
        d_vals = d[mask]

        def build_A(a):
            A = np.zeros((2 * n - 1, n), dtype=np.float64)
            A[mask] = a[d_vals]
            return A

        return build_A

    def lp_direction(a, build_A, n, M, tl=4.0):
        A = build_A(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * (1.0 + 1e-9))
        try:

                          bounds=[(0.0, None)] * n, method="highs",

            return None

        if gs < 1e-15:
            return None
        return g / gs
