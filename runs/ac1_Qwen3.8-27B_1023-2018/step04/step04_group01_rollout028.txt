from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    if n <= 512:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_vals = np.nonzero(cond)[0]
    if len(rho_vals) == 0:
        return np.ones(n) / n
    r = rho_vals[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)

        A[j:j+n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None

        return None
    g = np.asarray(res.x, dtype=np.float64)

    if s < 1e-15:
        return None
    return g / s

def _conv_full(a):
    n = len(a)
    if n <= 512:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)

def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    r = idx[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)

            bounds=[(0, None)] * n,
            method="highs",
            options={"time_limit": time_limit, "presolve": True},

            f_pol, S_pol, b_pol, n, rng3, d_scale=0.5, n_coarse=30, n_fine=20
        )
        if imp:
            sc = 2.0 * n * float(b_pol.max()) / (S_pol * S_pol)
            if sc < best_score - 1e-13:
                _update(f_pol.copy())
        else:
            f_pol = np.clip(f_pol + rng3.normal(0, 0.002, n), 0, None)
            S_pol = float(f_pol.sum())
            if S_pol < 0.01:
                break
            f_pol /= S_pol
            S_pol = 1.0
            b_pol = _conv_full(f_pol)

    _update(f_pol)

    # Final validation and return
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = parent.copy()

    final_sc = _eval_exact(best_seq)
    if not math.isfinite(final_sc):
        best_seq = parent.copy()
        final_sc = _eval_exact(best_seq)
        if not math.isfinite(final_sc):
            return [1.0]

    return [float(x) for x in best_seq]
