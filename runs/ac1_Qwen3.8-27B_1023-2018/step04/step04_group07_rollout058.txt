        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """Build the LP constraint matrix (2n-1) x n."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_direction(f, n, M, time_limit=5.0):
    """Solve LP to get direction g."""
    A = _build_A(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs

        hi = min(n - 1, k_idx)

    direction = np.maximum(direction, 0)  # keep non-negative
    ds = float(direction.sum())
    if ds < 1e-12:

    return direction / ds

def _conv_full(a):
    """Full convolution via FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)

    """LP: max sum(g) s.t. f*g <= M, g >= 0. Returns normalized g or None."""
    A = _build_A(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': tlimit})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs

def _suggest_step(f, g, M_cur, n, stop_time):
    """Find best t in [0,1] for f_new = (1-t)f + t*g."""
    best_t, best_m = 0.0, M_cur
    for t in (0.15, 0.3, 0.5, 0.7, 1.0):
        if time.monotonic() > stop_time:
            break
        fn = (1.0 - t) * f + t * g
        fn = np.maximum(fn, 0.0)
        fs = float(fn.sum())
        if fs < 1e-10:
            continue
        fn /= fs
        m = float(_conv_full(fn).max())
        if m < best_m:
            best_m = m
            best_t = t
    return best_t, best_m

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fn(a):

        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
