                denom = 1.0 + w  # since sum(p) = 1
                aff = base[None, :] + w[:, None] * infl[None, :]
                dval = base[i] + w * infl[i] + w * w
                aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                dv = denom
                mv = aff.max(axis=1)
                vals = np.full(pts.shape, math.inf, dtype=np.float64)
                valid = dv > 1e-15
                vals[valid] = (2.0 * n * mv[valid]) / (dv[valid] ** 2)

                if not np.any(np.isfinite(vals)):
                    continue
                imin = int(np.argmin(vals))
                best_x = float(pts[imin])
                best_val = float(vals[imin])

                # Fine search
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)
                if hi > lo:
                    fine = np.linspace(lo, hi, 24, dtype=np.float64)
                    fine = np.unique(np.concatenate((fine, [best_x])))
                    w2 = fine - v
                    aff2 = base[None, :] + w2[:, None] * infl[None, :]
                    dval2 = base[i] + w2 * infl[i] + w2 * w2
                    aff2 = np.maximum(aff2, dval2[:, None])
                    aff2 = np.maximum(aff2, max_const)
                    dv2 = 1.0 + w2
                    mv2 = aff2.max(axis=1)
                    vals2 = np.full(fine.shape, math.inf, dtype=np.float64)
                    valid2 = dv2 > 1e-15
                    vals2[valid2] = (2.0 * n * mv2[valid2]) / (dv2[valid2] ** 2)
                    if np.any(np.isfinite(vals2)):
                        im2 = int(np.argmin(vals2))
                        if float(vals2[im2]) < best_val:
                            best_x = float(fine[im2])
                            best_val = float(vals2[im2])

                if math.isfinite(best_val) and best_val < cur_score - 1e-14:
                    w_update = best_x - v
                    if abs(w_update) > 1e-15:
                        p[i] = best_x
                        c[start:end] += w_update * infl
                        di = 2 * i
                        if 0 <= di < Ltot:
                            c[di] += w_update * w_update
                        cur_score = best_val
                        improved = True

            # Renormalize
            ss = float(p.sum())
            if ss > 1e-300 and math.isfinite(ss):
                p /= ss
                c /= (ss * ss)
            c = np.maximum(c, 0.0)
            new_sc = float(2.0 * n * c.max())
            if new_sc < best_local - 1e-15:
                best_local = new_sc
                best_p = p.copy()

            if improved:
                flat = 0
            else:
                flat += 1
                if flat >= 4:
                    if time.monotonic() < stop_time - 10.0:
                        p = p * np.exp(rng.normal(0.0, 0.04, n))
                        p = np.maximum(p, 0.0)
                        ss2 = p.sum()
                        if ss2 > 0:
                            p /= ss2
                        c = conv_fft(p)
                        c = np.maximum(c, 0.0)
                        cur_score = float(2.0 * n * c.max())
                        flat = 0
                    else:
                        break
            sweep += 1

        ss3 = float(p.sum())
        if ss3 > 1e-300:
            p /= ss3
        return best_p

    # === L-BFGS on log-sum-exp ===
    def lbfgs_smooth(f, stop_time, T_frac=0.005):
        n = f.size
        f = np.maximum(f, 1e-300)
        f = f / f.sum()
        u = np.log(f) - np.log(f).mean()

        def softmax(uu):
            uu = uu - uu.max()
            e = np.exp(uu)
            s = e.sum()
            if s < 1e-300:
                return np.full(n, 1.0 / n)
            return e / s

        def objective(uu):
            a = softmax(uu)
            c = conv_fft(a)
            cmax = float(c.max())
            if not math.isfinite(cmax):
                return 1e100, np.zeros(n)
            T = max(1e-6, T_frac * cmax)
            scaled = (c - cmax) / T
            exps = np.exp(scaled - scaled.max())
            sum_exp = float(exps.sum())
            if sum_exp < 1e-300:
                w = np.zeros(c.size)
                w[int(np.argmax(c))] = 1.0
            else:
                w = (exps / exps.sum())
            fval = cmax + T * math.log(float(sum_exp) / c.size) if sum_exp > 1 else cmax
            # Gradient: dw/da_j = sum_k w_k * a[k-j] (valid)
            # Then grad u_j = a_j * (grad_a_j - <a, grad_a>)
            # cross-convolution of w and a
            Lx = 2 * n - 1
            szx = next_fast_len(Lx)
            Fw = rfft(w[::-1], szx)
            Fa = rfft(a, szx)
            cx = irfft(Fw * Fa, szx)[:Lx]
            grad_a = cx[n - 1:2 * n - 1]
            gnorm = float(np.dot(a, grad_a))
            grad_u = a * (grad_a - gnorm)
            grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
            return float(fval), grad_u

        best_u = u.copy()
        best_f = objective(u)[0]
        stale = 0
        while time.monotonic() < stop_time - 3.0:
            opts = {
                "maxiter": 40,
                "gtol": 1e-12,
                "ftol": 1e-16,
                "maxls": 20,
                "maxcor": 25,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(objective, best_u, method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fv = float(res.fun)
                if math.isfinite(fv) and fv < best_f:
                    best_f = fv
                    best_u = res.x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if getattr(res, "status", 1) == 0:
                break
        return softmax(best_u)

    # === Main optimization ===

    # Phase 1: LP iterations on parent
    f = parent.copy()
    f = f / f.sum()
    try_update(f)

    n_lp = int(f.size)
    if n_lp <= 5000:
        max_lp_iters = 8
    else:
        max_lp_iters = 4

    for it in range(max_lp_iters):
        now = time.monotonic()
        if now > hard - 300:
            break
        # check if score is already at target
        if best_score <= 1.503:
            break
        try:
            f_new = lp_step(f)
            if f_new is not None and np.all(np.isfinite(f_new)) and f_new.sum() > 0:
                f_new = f_new / f_new.sum()
                f = f_new
                try_update(f)
        except Exception:
            break
        # Early stop if no improvement
        c_check = conv_fft(f)
        if float(c_check.max()) * 2 * n_lp < 1.503:
            break

    try_update(f)

    # Phase 2: L-BFGS smoothing
    rem_to_hard = hard - time.monotonic()
    if rem_to_hard > 80 and f.size <= 5000:
        lb_stop = time.monotonic() + min(120.0, 0.4 * rem_to_hard)
        try:
            f_lb = lbfgs_smooth(f, lb_stop, T_frac=0.004)
            if f_lb is not None and np.all(np.isfinite(f_lb)) and f_lb.sum() > 0:
                f_lb = f_lb / f_lb.sum()
                try_update(f_lb)
                if score_fast(f_lb) < score_fast(f):
                    f = f_lb
        except Exception:
            pass
        try_update(f)

    # Phase 3: Coordinate descent
    rem_to_hard = hard - time.monotonic()
    if rem_to_hard > 40 and f.size <= 5000:
        cd_stop = time.monotonic() + min(200.0, 0.5 * rem_to_hard, hard - 100.0)
        try:
            f_cd = coord_descent(f, cd_stop, max_sweeps=30000, seed_cd=int(seed + 777))
            if f_cd is not None and np.all(np.isfinite(f_cd)) and f_cd.sum() > 0:
                f_cd = f_cd / f_cd.sum()
                try_update(f_cd)
                if score_fast(f_cd) < score_fast(f):
                    f = f_cd
        except Exception:
            pass
        try_update(f)

    # Phase 4: Try a different length (resample)
    rem_to_hard = hard - time.monotonic()
    if rem_to_hard > 150:
        for n_try in [n0 * 2, n0 // 2]:
            if time.monotonic() > hard - 80:
                break
            if n_try < 64 or n_try > 6000:
                continue
            # Resample
            xp = (np.arange(n0) + 0.5) / n0
            fp = np.interp((np.arange(n_try) + 0.5) / n_try, xp, parent)
            fp = np.maximum(fp, 0.0)
            ssr = fp.sum()
            if ssr < 1e-300:
                continue
            fp /= ssr
            try_update(fp)

            if n_try <= 4500 and time.monotonic() < hard - 120:
                try:
                    f_try = lp_step(fp)
                    if f_try is not None and np.all(np.isfinite(f_try)) and f_try.sum() > 0:
                        f_try /= f_try.sum()
                        try_update(f_try)
                except Exception:
                    pass

            rem_cd = hard - 60 - time.monotonic()
            if rem_cd > 30 and n_try <= 4500:
                try:
                    f_try2 = coord_descent(fp, time.monotonic() + min(80.0, rem_cd),
                                           max_sweeps=20000, seed_cd=int(seed + 31337 + n_try))
                    if f_try2 is not None and np.all(np.isfinite(f_try2)) and f_try2.sum() > 0:
                        f_try2 /= f_try2.sum()
                        try_update(f_try2)
                except Exception:
                    pass

    # Phase 5: Final coordinate descent on best
    rem_to_hard = hard - time.monotonic()
    if rem_to_hard > 20:
        n_best = best.size
        if n_best <= 5000:
            cd_stop2 = min(hard - 5.0, time.monotonic() + min(120.0, rem_to_hard - 10.0))
            try:
                f_final = coord_descent(best, cd_stop2, max_sweeps=50000, seed_cd=int(seed + 55555))
                if f_final is not None and np.all(np.isfinite(f_final)) and f_final.sum() > 0:
                    f_final /= f_final.sum()
                    try_update(f_final)
            except Exception:
                pass

    # Phase 6: One more LSE pass if time
    rem_to_hard = hard - time.monotonic()
    if rem_to_hard > 40 and best.size <= 4000:
        try:
            f_fin2 = lbfgs_smooth(best, time.monotonic() + min(50.0, rem_to_hard - 5.0), T_frac=0.001)
            if f_fin2 is not None and np.all(np.isfinite(f_fin2)) and f_fin2.sum() > 0:
                f_fin2 /= f_fin2.sum()
                try_update(f_fin2)
        except Exception:
            pass

    # Final validation and return
    try:
        out = np.asarray(best, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or out.sum() < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            return [1.0]
        return ls
    except Exception:
        return [1.0]
