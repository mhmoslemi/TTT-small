def _subgrad_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = _score(f)
    step = 0.05
    no_imp = 0
    momentum = np.zeros(n)
    mu = 0.7

    for it in range(100000):
        if time.monotonic() > stop_time:
            break

        # Compute subgradient of the score
        S = float(np.sum(f))
        b = _fft_conv(f)
        M = float(b.max())
        eps = max(1e-15, 1e-10 * M)
        active = np.flatnonzero(b >= M - eps)
        
        # Subgradient of max(f*f) w.r.t. f
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo <= i_hi:
                idx = k - np.arange(i_lo, i_hi + 1)
                grad[i_lo:i_hi + 1] += 2.0 * f[idx]
        if len(active) > 0:
            grad /= len(active)
        
        # Subgradient of 2n * M / S^2
        # d(2n*M/S^2)/df = 2n * (grad*M_avg * S - 2*M) / S^3  (roughly)
        # Actually: f = 2n * M * S^(-2)
        # df/d(f_i) = 2n * (dM/df_i * S^(-2) + M * (-2) * S^(-3) * 1)
        # = 2n * (dM/df_i / S^2 - 2M / S^3)
        
        f_sub = 2.0 * n * (grad / (S ** 2) - 2.0 * M / (S ** 3))
        
        norm_sg = np.linalg.norm(f_sub)
        if norm_sg < 1e-18:
            break

        momentum = mu * momentum - step * f_sub / norm_sg
        fn = np.clip(f + momentum, 0, None)
        s = fn.sum()
        if s < 0.01:
            step *= 0.5
            continue
        fn = fn / s
        sc = _score(fn)

        if sc < best_sc - 1e-14:
            f = fn
            best_f = f.copy()
            best_sc = sc
            step = min(0.5, step * 1.02)
            no_imp = 0
        else:
            step *= 0.92
            no_imp += 1
            if no_imp > 300:
                f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                momentum = np.zeros(n)
                step = 0.05
                no_imp = 0

    return best_f, best_sc

        S = float(np.sum(f))
        b = _fft_conv(f)
        M = float(b.max())
        eps = max(1e-14, 1e-10 * M)
        active = np.flatnonzero(b >= M - eps)

        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo <= i_hi:
                idx = k - np.arange(i_lo, i_hi + 1)
                grad[i_lo:i_hi + 1] += 2.0 * f[idx]
        if len(active) > 0:
            grad /= len(active)

        f_sub = 2.0 * n * (grad / (S ** 2) - 2.0 * M / (S ** 3))
        norm_sg = np.linalg.norm(f_sub)
        if norm_sg < 1e-18:
            break

        momentum = mu * momentum - step * f_sub / norm_sg
        fn = np.clip(f + momentum, 0, None)
        s_fn = fn.sum()
        if s_fn < 0.01:
            step *= 0.5
            continue
        fn = fn / s_fn
        sc = _score(fn)

        if sc < best_sc - 1e-13:
            f = fn
            best_f = f.copy()
            best_sc = sc
            step = min(0.4, step * 1.015)
            no_imp = 0
        else:
            step *= 0.93
            no_imp += 1
            if no_imp > 400:
                f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                momentum = np.zeros(n)
                step = 0.04
                no_imp = 0

    return best_f, best_sc


def _s滑落psp(f0, n, stop_time, seed):
    from scipy.optimize import minimize
    
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / max(f.sum(), 1e-10)
    best_sc = _score(f)

    def objective(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 0.01:
            return 1e10
        b = _fft_conv(x)
        return float(b.max())

    def gradient(x):
        x = np.maximum(x, 0)
        b = _fft_conv(x)
        M = float(b.max())
        eps = max(1e-12, 1e-9 * M)
        active = np.flatnonzero(b >= M - eps)
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo <= i_hi:
                idx = k - np.arange(i_lo, i_hi + 1)
                grad[i_lo:i_hi + 1] += 2.0 * x[idx]
        if len(active) > 0:
            grad /= len(active)
        return grad

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0, None)] * n

    try:
        res = minimize(objective, f, method='SLSQP', jac=gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 300, 'ftol': 1e-14})
        if hasattr(res, 'x') and res.x is not None:
            f_new = np.clip(res.x, 0, None)
            s_new = f_new.sum()
            if s_new > 0.01:
                f_new = f_new / s_new
                sc_new = _score(f_new)
                if sc_new < best_sc:
                    return f_new, sc_new
    except Exception:
        pass
    return f, best_sc


def _coord_descent(f0, stop_time, seed, max_epochs=300):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    n = len(f)
    best_sc = _score(f)
    best_f = f.copy()

    for epoch in range(max_epochs):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            if orig < 1e-16:
                continue
            best_v = orig
            best_v_sc = best_sc
            for mult in [0.05, 0.2, 0.4, 0.6, 0.75, 0.9, 1.1, 1.3, 1.6, 2.2, 0.0]:
                f[i] = orig * mult
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_v_sc - 1e-16:
                    best_v = f[i]
                    best_v_sc = sc
            f[i] = best_v
            if best_v_sc < best_sc - 1e-15:
                best_sc = best_v_sc
                best_f = f.copy()
                improved = True
        if not improved:
            break

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n = len(parent)
    parent_sum = float(np.sum(parent))

    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(320) / 320.0
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update(a, sc):
        nonlocal best_seq, best_score
        if math.isfinite(sc) and sc < best_score - 1e-12:
            a = np.asarray(a, dtype=np.float64)
            if a.size == 0 or not np.all(np.isfinite(a)):
                return
            if a.sum() < 0.01:
                return
            a_clip = np.clip(a, 0, 1000.0)
            best_seq = a_clip.copy()
            best_score = sc

    # Phase 1: Main LP optimization
    phase1_end = t0 + budget * 0.45
    p_norm = parent / max(parent_sum, 1e-10)
    try:
        res, sc = _lp_optimize(p_norm, n, phase1_end, seed)
        if math.isfinite(sc) and sc < best_score:
            _update(res * parent_sum, sc)
    except Exception:
        pass

    # Phase 1b: LP from perturbed starting points
    if time.monotonic() < t0 + budget * 0.58:
        for idx, (pert, scale) in enumerate([(0.03, 1.0), (0.1, 1.0), (1.0, 0.5)]):
            if time.monotonic() > t0 + budget * 0.58:
                break
            rng_p = np.random.default_rng(seed + 1000 + idx)
            base = best_seq / max(best_seq.sum(), 1e-10)
            if pert < 1.0:
                p2 = np.clip(base + rng_p.normal(0, pert, n), 0, None)
            else:
                mask = rng_p.random(n) < 0.7
                p2 = np.where(mask, base, rng_p.random(n) * 0.1)
            p2 = p2 / max(p2.sum(), 1e-10)
            stop1b = min(t0 + budget * 0.58, time.monotonic() + 80)
            try:
                res2, sc2 = _lp_optimize(p2, n, stop1b, seed + 2000 + idx)
                if math.isfinite(sc2) and sc2 < best_score:
                    _update(res2 * parent_sum, sc2)
            except Exception:
                pass

    # Phase 2: SLSQP refinement
    phase2_end = t0 + budget * 0.68
    if time.monotonic() < phase2_end:
        f_sl = best_seq / max(best_seq.sum(), 1e-10)
        try:
            res_sl, sc_sl = _滑落sps(f_sl, n, phase2_end, seed + 500)
            if math.isfinite(sc_sl) and sc_sl < best_score:
                _update(res_sl, sc_sl)
        except Exception:
            pass

    # Phase 3: Subgradient refinement
    phase3_end = t0 + budget * 0.78
    if time.monotonic() < phase3_end:
        f_sg = best_seq / max(best_seq.sum(), 1e-10)
        try:
            res_sg, sc_sg = _subgrad_optimize(f_sg, n, phase3_end, seed + 700)
            if math.isfinite(sc_sg) and sc_sg < best_score:
                _update(res_sg, sc_sg)
        except Exception:
            pass

    # Phase 4: LP polish
    phase4_end = t0 + budget * 0.85
    if time.monotonic() < phase4_end:
        f_lp = best_seq / max(best_seq.sum(), 1e-10)
        try:
            res_lp, sc_lp = _lp_optimize(f_lp, n, phase4_end, seed + 900)
            if math.isfinite(sc_lp) and sc_lp < best_score:
                _update(res_lp * best_seq.sum(), sc_lp)
        except Exception:
            pass

    # Phase 5: Coordinate descent
    phase5_end = deadline - 15
    if time.monotonic() < phase5_end:
        try:
            res_cd, sc_cd = _coord_descent(best_seq, phase5_end, seed + 1100, max_epochs=500)
            if math.isfinite(sc_cd) and sc_cd < best_score:
                _update(res_cd, sc_cd)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
