    """Coordinate descent with golden section search on each coordinate."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    best_f = f.copy()
    best_score = _score(f)
    no_imp_rounds = 0
    max_no_imp = 6

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    gr = 1.0 - 1.0 / phi

    def _optimize_coord(f_arr, i):
        """Optimize a single coordinate using golden section + coarse scan."""
        orig = f_arr[i]
        S = f_arr.sum()
        n_local = len(f_arr)

        # Determine search range
        if orig < 1e-15:
            lo = 0.0
            hi = max(2.0 * f_arr[f_arr > 0].mean() if np.any(f_arr > 0) else 1.0, 0.5)
        else:
            lo = 0.0
            hi = max(orig * 4.0, 500.0)

        # Coarse scan to find approximate minimum
        n_coarse = 20
        coarse_scores = []
        coarse_vals = np.linspace(lo, hi, n_coarse)
        f_arr[i] = 0  # Will be set in loop
        for v in coarse_vals:
            f_arr[i] = v
            if f_arr.sum() < 0.01:
                f_arr[i] = orig
                continue
            sc = _score(f_arr)
            coarse_scores.append(sc)

        f_arr[i] = orig
        coarse_scores = np.array(coarse_scores)
        valid = np.isfinite(coarse_scores)
        if not np.any(valid):
            return orig, _score(f_arr)

        # Find best coarse point and its neighbors for bracketing
        min_idx = np.argmin(np.where(valid, coarse_scores, np.inf))
        # Expand bracket
        if min_idx == 0:
            b_lo, b_hi = 0.0, coarse_vals[2] if n_coarse > 2 else hi
        elif min_idx == n_coarse - 1:
            b_lo, b_hi = coarse_vals[n_coarse - 3] if n_coarse > 2 else 0.0, hi
        else:
            b_lo = coarse_vals[max(0, min_idx - 2)]
            b_hi = coarse_vals[min(n_coarse - 1, min_idx + 2)]

        # Golden section search within bracket
        if b_hi - b_lo < 1e-10 * max(1.0, hi):
            best_val = coarse_vals[min_idx]
            f_arr[i] = best_val
            best_sc = _score(f_arr)
            f_arr[i] = orig
            return best_val, best_sc

        a_lb, a_ub = b_lo, b_hi
        x1 = a_ub - gr * (a_ub - a_lb)
        x2 = a_lb + gr * (a_ub - a_lb)
        f_arr[i] = x1
        f1 = _score(f_arr) if f_arr.sum() >= 0.01 else math.inf
        f_arr[i] = x2
        f2 = _score(f_arr) if f_arr.sum() >= 0.01 else math.inf

        max_iter = 35
        for _ in range(max_iter):
            if a_ub - a_lb < 1e-8 * max(1.0, a_ub):
                break
            if f1 < f2:
                a_ub = x2
                x2 = x1
                f2 = f1
                x1 = a_ub - gr * (a_ub - a_lb)
                f_arr[i] = x1
                f1 = _score(f_arr) if f_arr.sum() >= 0.01 else math.inf
            else:
                a_lb = x1
                x1 = x2
                f1 = f2
                x2 = a_lb + gr * (a_ub - a_lb)
                f_arr[i] = x2
                f2 = _score(f_arr) if f_arr.sum() >= 0.01 else math.inf

        best_val = (a_lb + a_ub) / 2.0
        f_arr[i] = best_val
        best_sc = _score(f_arr) if f_arr.sum() >= 0.01 else math.inf
        f_arr[i] = orig
        return best_val, best_sc

    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break

            new_val, new_sc = _optimize_coord(f, i)
            if new_sc < best_score - 1e-14:
                f[i] = new_val
                best_score = new_sc
                best_f = f.copy()
                improved = True

        if not improved:
            no_imp_rounds += 1
            if no_imp_rounds >= max_no_imp:
                # Perturbation to escape local minimum
                noise = rng.normal(0, 0.005 * max(1.0, f.mean()), n)
                f_pert = np.clip(f + noise, 0, None)
                if f_pert.sum() >= 0.01:
                    sc_pert = _score(f_pert)
                    if sc_pert < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = sc_pert
                    else:
                        # Try larger perturbation
                        noise2 = rng.normal(0, 0.02 * max(1.0, f.mean()), n)
                        f_pert2 = np.clip(f + noise2, 0, None)
                        if f_pert2.sum() >= 0.01:
                            sc_pert2 = _score(f_pert2)
                            if sc_pert2 < best_score:
                                f = f_pert2
                                best_f = f.copy()
                                best_score = sc_pert2
                            else:
                                f = best_f.copy()
                no_imp_rounds = 0
                max_no_imp = min(max_no_imp + 2, 14)
        else:
            no_imp_rounds = 0

    return best_f, best_score


def _smoothed_grad_optimize(f0, stop_time, seed):
    """Optimize using smoothed max (log-sum-exp) surrogate with gradient descent."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-10, None)
    f = f / f.sum()
    best_f = f.copy()
    best_score = _score(f)

    lr = 0.3
    temperature = 0.05
    stall = 0
    max_stall = 50

    while time.monotonic() < stop_time - 2.0:
        b = _conv_full(f)
        S = f.sum()

        # Soft-max weights
        b_safe = np.maximum(b, 1e-30)
        log_w = b_safe / temperature
        log_w -= log_w.max()
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-15:
            break
        w = w / w_sum

        # Gradient of soft-max(b) w.r.t. f
        grad = np.zeros(n)
        for idx_k, k in enumerate(range(2 * n - 1)):
            wk = w[idx_k]
            if wk < 1e-8:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                grad[i] += wk * f[k - i]

        # Gradient of full objective (approximately, ignoring denominator for direction)
        # Use multiplicative update
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        lr_eff = lr * temperature / grad_norm
        exponent = -lr_eff * grad
        exponent -= exponent.max()
        weights = np.exp(exponent)
        f_new = f * weights
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        sc = _score(f_new)
        if sc < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = sc
            stall = 0
            lr *= 0.995
            temperature *= 0.999
        else:
            f_half = 0.5 * f + 0.5 * f_new
            f_half = np.maximum(f_half, 1e-10)
            sc_half = _score(f_half)
            if sc_half < best_score - 1e-12:
                f = f_half
                best_f = f.copy()
                best_score = sc_half
                stall = 0
                lr *= 0.98
            else:
                stall += 1
                lr *= 0.93
                temperature *= 0.99
                if stall > max_stall:
                    noise = rng.normal(0, 0.003, n)
                    f = np.clip(f + noise, 1e-10, None)
                    f = f / f.sum()
                    stall = 0
                    lr = 0.2
                    temperature = max(0.01, temperature * 0.8)
                    max_stall += 15
                    if max_stall > 200:
                        break

        f = f_new

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = _score(best_seq)

    def _update_best(candidate):
        nonlocal best_seq, best_score
        if candidate is None or len(candidate) == 0:
            return
        candidate = np.clip(np.asarray(candidate, dtype=np.float64), 0, 1000)
        if not np.all(np.isfinite(candidate)):
            return
        if candidate.sum() < 0.01:
            return
        sc = _score(candidate)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq = candidate.copy()
            best_score = sc

    rng = np.random.default_rng(seed)
    i_arr = np.arange(n_p, dtype=np.float64)

    # Generate initializations
    inits = [parent.copy()]

    # Noisy perturbations of parent
    for sig in (0.03, 0.07, 0.12):
        sig_val = sig * max(1.0, float(parent.mean()))
        inits.append(np.clip(parent + rng.normal(0, sig_val, n_p), 0, None))

    # Structured initializations
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n_p)))
    inits.append(0.5 * (1.0 + np.cos(4.0 * math.pi * i_arr / n_p)))

    # Multi-frequency
    sup = np.ones(n_p) * 0.25
    for fr in (1, 2, 3, 5, 7, 11):
        ph = rng.uniform(0, 2 * math.pi)
        sup += 0.1 * np.cos(2.0 * math.pi * fr * i_arr / n_p + ph)
    inits.append(np.abs(sup) + 0.02)

    # Random
    inits.append(rng.uniform(0.2, 1.0, n_p))

    # Phase 1: LP optimization (45% of budget)
    phase1_end = t0 + budget * 0.45
    n_inits = len(inits)
    time_per_init = max(6.0, (phase1_end - t0 - 4.0) / n_inits)

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 6.0:
            break
        stop = min(phase1_end - 2.0, time.monotonic() + time_per_init)
        if time.monotonic() > stop - 1.0:
            continue
        try:
            res, sc = _optimize_lp(init, stop, seed + 1000 * (idx + 1),
                                   peak_frac_start=0.02, peak_frac_end=0.10)
            _update_best(res)
        except Exception:
            pass

    # Focused LP on current best
    if time.monotonic() < phase1_end - 12.0:
        stop = phase1_end - 2.0
        try:
            res, sc = _optimize_lp(best_seq, stop, seed + 888888,
                                   peak_frac_start=0.01, peak_frac_end=0.06)
            _update_best(res)
        except Exception:
            pass

    # Phase 2: Smoothed gradient (12% of budget)
    phase2_end = t0 + budget * 0.57
    if time.monotonic() < phase2_end - 8.0:
        stop = phase2_end - 2.0
        try:
            res, sc = _smoothed_grad_optimize(best_seq, stop, seed + 77777)
            _update_best(res)
        except Exception:
            pass

    # Phase 3: Coordinate descent golden section (25% of budget)
    phase3_end = t0 + budget * 0.82
    if time.monotonic() < phase3_end - 8.0:
        stop = phase3_end - 2.0
        try:
            res, sc = _coord_descent_golden(best_seq, stop, seed + 55555)
            _update_best(res)
        except Exception:
            pass

    # Phase 4: LP refinement after coordinate descent (8% of budget)
    phase4_end = t0 + budget * 0.90
    if time.monotonic() < phase4_end - 8.0:
        stop = phase4_end - 2.0
        try:
            res, sc = _optimize_lp(best_seq, stop, seed + 666666,
                                   peak_frac_start=0.01, peak_frac_end=0.04)
            _update_best(res)
        except Exception:
            pass

    # Phase 5: Final coordinate descent (7% of budget)
    if time.monotonic() < deadline - 8.0:
        stop = deadline - 2.0
        try:
            res, sc = _coord_descent_golden(best_seq, stop, seed + 99999)
            _update_best(res)
        except Exception:
            pass

    # Try a couple of different lengths if time permits
    remaining = deadline - time.monotonic()
    if remaining > 80.0:
        for L in (n_p + 16, n_p - 16):
            if time.monotonic() > deadline - 20.0:
                break
            if L < 100 or L > 800:
                continue
            x_old = np.linspace(0, 1, n_p)
            x_new = np.linspace(0, 1, L)
            init_L = np.interp(x_new, x_old, best_seq)
            stop = min(deadline - 15.0, time.monotonic() + max(20.0, remaining * 0.25))
            try:
                res, sc = _optimize_lp(init_L, stop, seed + L * 137)
                _update_best(res)
            except Exception:
                pass
            # Quick coordinate descent on the result
            if time.monotonic() < deadline - 15.0 and res is not None:
                stop2 = min(deadline - 12.0, time.monotonic() + max(10.0, remaining * 0.1))
                try:
                    res2, sc2 = _coord_descent_golden(res, stop2, seed + L * 211)
                    _update_best(res2)
                except Exception:
                    pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    return [float(x) for x in best_seq]
