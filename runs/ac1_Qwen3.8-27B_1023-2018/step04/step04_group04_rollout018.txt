from scipy.fft import rfft, irfft, fft, ifft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p

def _conv(a):
    n = len(a)
    if n == 1:
        return np.array([a[0]*a[0]])
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(np.max(b))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _symmetrize(a):
    return (a + a[::-1]) / 2.0

def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_dir(f, n, M, timeout=3.0):
    A = _build_A(f, n)
    cvec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(cvec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout,
                               'primal_feasibility_tolerance': 1e-12,
                               'dual_feasibility_tolerance': 1e-12})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-14:
        return None
    return g / g.sum()

def _optimize_lp(f0, n, t_deadline, seed):
    rng = np.random.default_rng(seed)
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
        s = n
    f /= s
    
    best_f = f.copy()
    best_sc = _score(f)
    t_param = 0.3
    stall = 0
    max_stall = 150
    
    while time.monotonic() < t_deadline:
        b = _conv(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        g = _lp_dir(f, n, M, timeout=3.0)
        if g is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.01, n), 0, None)
            f /= max(f.sum(), 1e-10)
            continue
        
        d = np.linalg.norm(g - f)
        if d < 1e-10:
            stall += 2
            if stall > max_stall:
                break
            continue
        
        # Line search
        best_t = 0.0
        best_sc_try = best_sc
        for tt in [t_param, t_param*0.5, t_param*1.5, 0.05, 0.1, 0.2, 0.4, 0.6, 0.8, 0.95]:
            tt = min(1.0, max(0.001, tt))
            fn = np.maximum((1-tt)*f + tt*g, 0)
            sc = _score(fn)
            if sc < best_sc_try - 1e-14:
                best_sc_try = sc
                best_t = tt
        
        if best_sc_try < best_sc - 1e-13:
            fn = np.maximum((1-best_t)*f + best_t*g, 0)
            f = fn
            best_f = f.copy()
            best_sc = best_sc_try
            t_param = min(0.95, max(0.05, best_t * 1.05))
            stall = 0
        else:
            t_param = max(0.01, t_param * 0.5)
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                f /= max(f.sum(), 1e-10)
                t_param = 0.3
                stall = 0
                max_stall += 80
    
    return best_f, best_sc

def _coord_descent(f0, t_deadline, seed):
    rng = np.random.default_rng(seed)
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0).copy()
    n = len(f)
    best_sc = _score(f)
    best_f = f.copy()
    
    # Single-coordinate
    epoch = 0
    while time.monotonic() < t_deadline and epoch < 200:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > t_deadline:
                break
            orig = f[i]
            if orig < 1e-15:
                continue
            best_v = orig
            bv_sc = best_sc
            for mult in [0.01, 0.1, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 4.0, 0.0]:
                f[i] = orig * mult if orig > 1e-15 else 0.0
                sc = _score(f)
                if sc < bv_sc - 1e-15:
                    bv_sc = sc
                    best_v = f[i]
            f[i] = best_v
            if bv_sc < best_sc - 1e-14:
                best_sc = bv_sc
                best_f = f.copy()
                improved = True
        epoch += 1
        if not improved:
            break
    
    return best_f, best_sc

def _subgrad_step(f, n):
    """Compute subgradient of max(a*a) at a"""
    b = _conv(f)
    M = float(b.max())
    eps = max(1e-14, 1e-10 * M)
    active = np.where(b >= M - eps)[0]
    grad = np.zeros(n)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            grad[i] += 2.0 * f[k - i]
    if len(active) > 0:
        grad /= len(active)
    return grad, M

def _subgrad_opt(f0, n, t_deadline, seed):
    rng = np.random.default_rng(seed)
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0).copy()
    f /= max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = _score(f)
    step = 0.05
    mu = 0.7
    mom = np.zeros(n)
    no_imp = 0
    
    while time.monotonic() < t_deadline:
        sg, M = _subgrad_step(f, n)
        sg -= sg.mean()  # Remove mean to stay on simplex
        ns = np.linalg.norm(sg)
        if ns < 1e-16:
            break
        mom = mu * mom - step * sg / ns
        fn = np.maximum(f + mom, 0)
        fn /= max(fn.sum(), 1e-10)
        sc = _score(fn)
        if sc < best_sc - 1e-14:
            f = fn
            best_f = f.copy()
            best_sc = sc
            step = min(0.3, step * 1.02)
            no_imp = 0
        else:
            step *= 0.92
            no_imp += 1
            if no_imp > 300:
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                f /= max(f.sum(), 1e-10)
                mom = np.zeros(n)
                step = 0.05
                no_imp = 0
    
    return best_f, best_sc

    # Try several modified initializations

    # Phase 1: LP optimization on best initializations

        if time.monotonic() > main_end - 30:

        stop_i = main_end - 30
        try:

            if math.isfinite(sc):

                _update(res * parent.sum(), sc)

        alt_lengths = [L for L in alt_lengths if L >= 4]
        time_per = min(30.0, (alt_end - time.monotonic()) / max(1, len(alt_lengths)))
        for L in alt_lengths:
            if time.monotonic() > alt_end - 5:

            try:

                stop_L = min(alt_end, time.monotonic() + time_per)

        except Exception:

    return [float(x) for x in best_seq]

Actually, I realize the line search can be optimized. Since we already computed `b = _conv(f)` and know `M`, we can compute the score of `(1-t)f + tg` analytically:

`((1-t)f + tg) * ((1-t)f + tg)` is a quadratic in `t`:
`= (1-t)^2 (f*f) + 2t(1-t)(f*g) + t^2 (g*g)`

So `max_k` of this is the max of `2n-1` quadratics in `t`. We could compute all three convolutions once (`f*f`, `f*g`, `g*g`) and then evaluate the max of the quadratics for any `t` in O(2n-1) time.

This would make the line search much faster! Let me implement this.

```python
def _conv3(f, g, n):
    """Compute f*f, f*g, g*g simultaneously using FFT"""
    fsize = next_fast_len(2 * n - 1)
    F = rfft(f, fsize)
    G = rfft(g, fsize)
    ff = irfft(F * F, fsize)[:2*n-1]
    fg = irfft(F * G, fsize)[:2*n-1]
    gg = irfft(G * G, fsize)[:2*n-1]
    return np.maximum(ff, 0), np.maximum(fg, 0), np.maximum(gg, 0)

def _score_quad(t, ff, fg, gg, n, s_f, s_g):
    """Score of (1-t)f + tg using precomputed convolutions"""
    s = (1-t)*s_f + t*s_g
    if s < 0.01:
        return math.inf
    c = (1-t)**2 * ff + 2*t*(1-t) * fg + t**2 * gg
    m = float(np.max(c))
    return 2.0 * n * m / (s*s)
```

            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                f /= max(f.sum(), 1e-10)
                tp = 0.3
                stall = 0
                max_stall += 60

    return best_f, best_sc


def _subgrad_opt(f0, n, t_stop, seed):
    rng = np.random.default_rng(seed)
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0).copy()
    f /= max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = _score_norm(f, n)
    step = 0.04
    mu = 0.7
    mom = np.zeros(n)
    no_imp = 0

    while time.monotonic() < t_stop:
        b = _conv(f)
        M = float(b.max())
        eps = max(1e-14, 1e-10 * M)
        active = np.where(b >= M - eps)[0]
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            idx = k - np.arange(i_lo, i_hi + 1)
            grad[i_lo:i_hi + 1] += 2.0 * f[idx]
        if len(active) > 0:
            grad /= len(active)

        grad -= grad.mean()
        ns = np.linalg.norm(grad)
        if ns < 1e-16:
            break

        mom = mu * mom - step * grad / ns
        fn = np.maximum(f + mom, 0.0)
        fn /= max(fn.sum(), 1e-10)
        sc = _score_norm(fn, n)

        if sc < best_sc - 1e-14:
            f = fn
            best_f = f.copy()
            best_sc = sc
            step = min(0.3, step * 1.02)
            no_imp = 0
        else:
            step *= 0.92
            no_imp += 1
            if no_imp > 300:
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                f /= max(f.sum(), 1e-10)
                mom = np.zeros(n)
                step = 0.04
                no_imp = 0

    return best_f, best_sc


def _coord_descent(f0, t_stop, seed):
    rng = np.random.default_rng(seed)
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0).copy()
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return f, _score(f)
    f /= s
    best_sc = _score_norm(f, n)
    best_f = f.copy()

    epoch = 0
    while time.monotonic() < t_stop and epoch < 400:
        improved = False
        for i in rng.permutation(n):
            if time.monotonic() > t_stop:
                break
            orig = f[i]
            if orig < 1e-15:
                continue
            bv = orig
            bsc = best_sc
            for m in [0.01, 0.1, 0.25, 0.5, 0.75, 1.25, 1.5, 2.0, 0.0]:
                f[i] = orig * m
                sc = _score_norm(f, n)
                if sc < bsc - 1e-15:
                    bsc = sc
                    bv = f[i]
            f[i] = bv
            if bsc < best_sc - 1e-14:
                best_sc = bsc
                best_f = f.copy()
                improved = True
        epoch += 1
        if not improved:
            break

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n = len(parent)

    best_seq = parent.copy()
    best_score = _score(best_seq)

    def _upd(a, sc):
        nonlocal best_seq, best_score
        if math.isfinite(sc) and sc < best_score - 1e-12:
            a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000.0)
            if len(a) == 0 or a.sum() < 0.01 or not np.all(np.isfinite(a)):
                return
            best_seq = a.copy()
            best_score = sc

    # --- Quick initialisation ---
    pn = parent / max(parent.sum(), 1e-10)
    cand_list = [parent, pn]
    for sig in [max(1, n // 25), max(1, n // 50), max(1, n // 15)]:
        try:
            cand_list.append(gaussian_filter1d(pn, sigma=sig))
        except Exception:
            pass
    cand_list.append(0.5 * (pn + pn[::-1]))
    rng0 = np.random.default_rng(seed)
    for sd in [0.05, 0.12]:
        p = np.clip(pn + rng0.normal(0, sd, n), 0, None)
        cand_list.append(p / max(p.sum(), 1e-10))

    for c in cand_list:
        _upd(c, _score(c))

    # --- Phase 1: Main LP optimisation (best normalised init) ---
    main_end = t0 + budget * 0.62
    best_init = best_seq / max(best_seq.sum(), 1e-10)
    res, sc = _optimize(best_init, n, main_end, seed)
    if math.isfinite(sc):
        _upd(res, sc)

    # --- Phase 1b: Second LP run with perturbed init ---
    p1b_end = t0 + budget * 0.74
    if time.monotonic() < p1b_end - 10:
        p = np.clip(best_seq / max(best_seq.sum(), 1e-10)
                    + np.random.default_rng(seed + 1).normal(0, 0.03, n), 0, None)
        p /= max(p.sum(), 1e-10)
        try:
            res2, sc2 = _optimize(p, n, p1b_end, seed + 100)
            if math.isfinite(sc2):
                _upd(res2, sc2)
        except Exception:
            pass

    # --- Phase 2: Subgradient refinement ---
    sg_end = t0 + budget * 0.84
    if time.monotonic() < sg_end - 5:
        fsg = best_seq / max(best_seq.sum(), 1e-10)
        try:
            res3, sc3 = _subgrad_opt(fsg, n, sg_end, seed + 200)
            if math.isfinite(sc3):
                _upd(res3, sc3)
        except Exception:
            pass

    # --- Phase 3: Coordinate descent ---
    cd_end = deadline - 15
    if time.monotonic() < cd_end:
        try:
            res4, sc4 = _coord_descent(best_seq, cd_end, seed + 300)
            if math.isfinite(sc4):
                _upd(res4, sc4)
        except Exception:
            pass

    # --- Symmetrisation check ---
    sym = 0.5 * (best_seq + best_seq[::-1])
    ss = _score(sym)
    if math.isfinite(ss):
        _upd(sym, ss)

    # --- Final validation with exact evaluator ---
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            fb = [float(max(0.0, min(1000.0, x))) for x in parent]
            fsc = float(evaluate_sequence(fb))
        except Exception:
            fb = [1.0]
            fsc = math.inf
        if math.isfinite(fsc):
            return fb
        return [1.0]

    # Ensure all values are valid
    out = []
    for x in best_seq:
        v = float(max(0.0, min(1000.0, x)))
        if not math.isfinite(v):
            v = 0.0
        out.append(v)
    if not out or sum(out) < 0.01:
        return [1.0]
    return out
