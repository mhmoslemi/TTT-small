import time
from scipy.optimize import linprog

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * a.size * float(c.max()) / (s * s)

def _max_conv(a):
    c = np.convolve(a, a)
    return c, float(c.max())

def _conv_matrix(a):
    n = a.size
    k = np.arange(2*n-1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    A[mask] = a[d[mask]]
    return A

def _lp_solve(A, M, n, tl=2.0):
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)]*n,
                     method='highs',
                     options={'time_limit': tl})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    x = np.asarray(res.x, dtype=np.float64)
    return np.maximum(x, 0.0)

                a *= 1.0 + smooth_noise

                a = 0.5*a + 0.5*a[::-1]
                a *= np.exp(rng.normal(0, sigma*0.5, n))

import numpy as np
import time
from scipy.optimize import linprog


def _score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * a.size * float(c.max()) / (s * s)


def _maxconv(a):
    c = np.convolve(a, a)
    return c, float(c.max())


def _conv_mat(a):
    n = a.size
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[d[mask]]
    return A


def _lp_br(a, M, n, tl=2.0):
    A = _conv_mat(a)
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tl, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    x = np.asarray(res.x, dtype=np.float64)
    return np.maximum(x, 0.0)

def _try_sym(a):
    """Try symmetrization: a -> 0.5*(a + flip(a))."""
    b = 0.5 * (a + a[::-1])
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s < 1e-12:
        return None
    b /= s
    return b
