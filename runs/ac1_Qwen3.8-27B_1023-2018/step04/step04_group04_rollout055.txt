def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """A[g] = f * g (linear convolution). Shape: (2n-1, n)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_minmax(f, n, timeout=10.0):
    """min t s.t. f*g <= t (all lags), sum(g)=1, g>=0."""
    A = _build_A(f, n)
    m = 2 * n - 1
    c = np.zeros(n + 1); c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
    b_eq = [1.0]
    bounds = [(0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0:
        return None
    g = np.clip(res.x[:n], 0, None)
    gs = g.sum()
    if gs < 1e-12:
        return None
    return g / gs

def _lp_maxsum(f, n, M, timeout=10.0):
    """max sum(g) s.t. f*g <= M, g>=0."""
    A = _build_A(f, n)
    m = 2 * n - 1
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0:
        return None
    g = np.clip(res.x, 0, None)
    if g.sum() < 1e-12:
        return None
    return g / g.sum()

def _optimize_lp(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = _score(f)
    t_param = 0.3
    stall = 0
    max_stall = 300
    
    while time.monotonic() < stop_time:
        if time.monotonic() > stop_time - 2:
            break
        b = _fft_conv(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        # Try both LP formulations
        g1 = _lp_minmax(f, n, timeout=6.0)
        if g1 is None:
            g1 = _lp_maxsum(f, n, M, timeout=6.0)
        if g1 is None:
            f = np.clip(f + rng.normal(0, 0.015, n), 0, None)
            f = f / max(f.sum(), 1e-10)
            stall += 10
            if stall > max_stall * 2:
                break
            continue
        
        dist = np.linalg.norm(g1 - f)
        if dist < 1e-12:
            stall += 5
            if stall > max_stall:
                break
            continue
        
        # Line search
        bt = t_param
        bsc = best_sc
        for tt in [t_param, t_param*0.5, t_param*1.5, 0.05, 0.1, 0.2, 0.4, 0.6, 0.8]:
            tt = min(0.99, max(0.003, tt))
            fn = np.clip((1-tt)*f + tt*g1, 0, None)
            fn = fn / max(fn.sum(), 1e-10)
            sc = _score(fn)
            if sc < bsc - 1e-15:
                bt = tt
                bsc = sc
        
        if bsc < best_sc - 1e-14:
            fn = np.clip((1-bt)*f + bt*g1, 0, None)
            f = fn / max(fn.sum(), 1e-10)
            best_f = f.copy()
            best_sc = _score(f)
            t_param = min(0.95, max(0.05, bt * 1.03))
            stall = 0
        else:
            t_param = max(0.01, t_param * 0.5)
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                t_param = 0.3
                stall = 0
                max_stall = min(1000, max_stall + 100)
    
    return best_f, best_sc

def _subgrad_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = _score(f)
    step = 0.03
    no_imp = 0
    momentum = np.zeros(n)
    mu = 0.5
    
    while time.monotonic() < stop_time:
        if time.monotonic() > stop_time - 1:
            break
        # Subgradient of max_k (f*f)[k]
        b = _fft_conv(f)
        M = float(b.max())
        eps = max(1e-14, 1e-10 * M)
        active = np.flatnonzero(b >= M - eps)
        
        # Gradient of (f*f)[k] w.r.t. f
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            for i in range(i_lo, i_hi + 1):
                grad[i] += 2.0 * f[k - i]
        if len(active) > 0:
            grad /= len(active)
        
        # Gradient of full objective: 2n*max(b)/S^2
        S = float(f.sum())
        # d/d f_i [2n*M/S^2] = 2n*(S*grad_i - 2*M)/S^3
        f_grad = 2.0 * n * (S * grad - 2.0 * M) / (S ** 3)
        f_grad -= np.mean(f_grad)  # project to sum-zero (simplex tangent)
        
        gn = np.linalg.norm(f_grad)
        if gn < 1e-16:
            break
        
        momentum = mu * momentum - step * f_grad / gn
        fn = np.clip(f + momentum, 0, None)
        fn = fn / max(fn.sum(), 1e-10)
        sc = _score(fn)
        
        if sc < best_sc - 1e-14:
            f = fn
            best_f = f.copy()
            best_sc = sc
            step = min(0.3, step * 1.01)
            no_imp = 0
        else:
            step *= 0.92
            no_imp += 1
            if no_imp > 300:
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                momentum = np.zeros(n)
                step = 0.03
                no_imp = 0
    
    return best_f, best_sc

def _coord_descent(f0, stop_time, seed, max_epochs=30):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    n = len(f)
    best_sc = _score(f)
    best_f = f.copy()
    
    for epoch in range(max_epochs):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            if orig < 1e-15:
                continue
            bv = orig
            bvs = best_sc
            for mult in [0.3, 0.6, 0.8, 1.1, 1.3, 1.6, 2.0, 0.0]:
                f[i] = orig * mult if orig > 1e-15 else 0.0
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < bvs - 1e-15:
                    bv = f[i]
                    bvs = sc
            f[i] = bv
            if bvs < best_sc - 1e-15:
                best_sc = bvs
                best_f = f.copy()
                improved = True
        if not improved:
            break
    
    return best_f, best_sc

The matrix A for convolution: A[k, j] = f[k-j] if 0 <= k-j < n, else 0.

For row k (0 <= k <= 2n-2) and column j (0 <= j <= n-1):
A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0.

So for row k, the non-zero columns j range from max(0, k-n+1) to min(k, n-1).
And A[k, j] = f[k-j].

Let me verify: (f*g)[k] = sum_j f[k-j]*g[j] (valid j). Yes, that's right.

                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_f = f.copy()
                    improved = True
            f[i], f[j] = oi, oj

        if not improved:
            break

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n = len(parent)

    best_seq = parent.copy()
    best_score = _score(parent)
    if not math.isfinite(best_score):
        best_seq = np.ones(n)
        best_score = _score(best_seq)

    def _update(a, sc):
        nonlocal best_seq, best_score
        if math.isfinite(sc) and sc < best_score - 1e-15:
            a = np.clip(a, 0.0, 1000.0)
            if a.size < 1 or a.sum() < 0.01 or not np.all(np.isfinite(a)):
                return
            best_seq = a.copy()
            best_score = sc

    # Phase 1: Main LP optimization on parent length (45%)
    p1_end = t0 + budget * 0.45
    p_norm = parent / max(parent.sum(), 1e-10)
    try:
        r1, s1 = _lp_optimize(p_norm, n, p1_end, seed)
        _update(r1, s1)
    except Exception:
        pass

    # Phase 1b: LP with perturbed start (10%)
    if time.monotonic() < t0 + budget * 0.55:
        rng_p = np.random.default_rng(seed + 77)
        p2 = np.clip(p_norm + rng_p.normal(0, 0.12, n), 0, None)
        p2 = p2 / max(p2.sum(), 1e-10)
        try:
            r1b, s1b = _lp_optimize(p2, n, t0 + budget * 0.55, seed + 200)
            _update(r1b, s1b)
        except Exception:
            pass

    # Phase 2: Subgradient refinement (13%)
    if time.monotonic() < t0 + budget * 0.68:
        f2 = best_seq / max(best_seq.sum(), 1e-10)
        try:
            r2, s2 = _subgrad_optimize(f2, n, t0 + budget * 0.68, seed + 300)
            _update(r2, s2)
        except Exception:
            pass

    # Phase 3: Coordinate descent (15%)
    if time.monotonic() < t0 + budget * 0.83:
        try:
            r3, s3 = _coord_descent(best_seq, t0 + budget * 0.83, seed + 400, max_epochs=50)
            _update(r3, s3)
        except Exception:
            pass

    # Phase 4: Try nearby lengths (10%)
    if time.monotonic() < deadline - 15:
        alt_lens = [n - 8, n - 4, n - 1, n + 1, n + 4, n + 8]
        alt_lens = sorted(set(L for L in alt_lens if 8 <= L <= 700))
        rem = deadline - 15 - time.monotonic()
        per_L = max(15.0, rem / max(1, len(alt_lens)))
        for L in alt_lens:
            if time.monotonic() > deadline - 15:
                break
            init_L = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n), parent)
            init_L = init_L / max(init_L.sum(), 1e-10)
            stop_L = min(deadline - 15, time.monotonic() + per_L)
            try:
                rL, sL = _lp_optimize(init_L, L, stop_L, seed + 1000 + L)
                _update(rL, sL)
            except Exception:
                pass

    # Phase 5: Final polish with coordinate descent
    if time.monotonic() < deadline - 5:
        try:
            rf, sf = _coord_descent(best_seq, deadline - 5, seed + 5000, max_epochs=30)
            _update(rf, sf)
        except Exception:
            pass

    # Final validation
    try:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        fs = math.inf

    if not math.isfinite(fs):
        fallback = np.ones(n)
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
