import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2*n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_local(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _simplex_project(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    # Normalize first
    v = v.copy()
    # Standard algorithm
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rng_ = np.arange(1, n+1)
    cond = u * rng_ > (cv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    n = len(a)
    c = _conv_full(a)
    c_max = c.max()

    return g, w, c

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_local(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}"""
    n = len(v)
    if n == 0:
        return v.copy()
    v = v.astype(np.float64).copy()
    # Find theta such that sum(max(v - theta, 0)) = target_sum
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rng_ = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rng_ > (cv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smooth_grad(a, tau):
    """Compute gradient of smooth max of a*a (softmax approximation).
    Returns (gradient array of length n, max value)."""
    n = len(a)
    c = _conv_full(a)
    M = float(c.max())
    # Softmax weights with overflow protection
    c_shifted = tau * (c - M)
    exp_c = np.exp(c_shifted)
    exp_sum = exp_c.sum()
    if exp_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = exp_c / exp_sum
    # g[j] = 2 * sum_{i=0}^{n-1} w[j+i] * a[i]
    g = np.empty(n)
    for j in range(n):
        g[j] = 2.0 * np.dot(w[j:j + n], a)
    return g, M, w


def _smooth_grad_fft(a, tau):
    """FFT-based gradient computation for speed."""
    n = len(a)
    c = _conv_full(a)
    M = float(c.max())
    c_shifted = tau * (c - M)
    exp_c = np.exp(c_shifted)
    exp_sum = exp_c.sum()
    if exp_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = exp_c / exp_sum
    # g[j] = 2 * sum_{i=0}^{n-1} w[j+i] * a[i]
    # This is a cross-correlation; use FFT
    # Pad w and a to length >= 2n-1 + n - 1 = 3n-2
    fsize = next_fast_len(3 * n - 1)
    W = rfft(w, fsize)
    A = rfft(a, fsize)
    # Correlation: corr[j] = sum_i w[j+i] * a[i]
    # = sum_i w[j+i] * a[i] = (w * a_flipped) at appropriate lags
    # Actually: define b[i] = a[i], then corr[j] = sum_i b[i] * w[j+i]
    # This is the cross-correlation (b ⋆ w)[j] = sum_i b[i] * conj(w[i-j])
    # For real: = sum_i b[i] * w[i-j]
    # We want sum_i w[j+i] * a[i], let k = j+i, i = k-j
    # = sum_k w[k] * a[k-j] for k=j..j+n-1
    # This is (a * w)[j + n - 1] if we reverse w... 
    # Actually let's just use the direct method for n=384, it's fast enough.
    g = np.empty(n)
    for j in range(n):
        g[j] = 2.0 * np.dot(w[j:j + n], a)
    return g, M, w


def _gradient_descent_smooth(a0, stop_time, seed=0, tau_init=200.0):
    """Smooth max gradient descent with projection onto simplex."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_M = float(_conv_full(a).max())
    rng = np.random.default_rng(seed)
    tau = tau_init
    eta = 0.05
    stall = 0
    max_stall = 50
    prev_M = best_M

    for it in range(20000):
        if time.monotonic() > stop_time:
            break

        g, M, w = _smooth_grad(a, tau)

        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Backtracking line search
        improved = False
        for trial in range(12):
            step = a - eta * g
            proj = _simplex_project(step, 1.0)
            c_trial = _conv_full(proj)
            M_trial = float(c_trial.max())
            if M_trial < M - 1e-14:
                a = proj
                M = M_trial
                improved = True
                break
            eta *= 0.6
        if not improved:
            eta = max(eta * 0.5, 1e-5)
            # Perturb to escape
            a = a + rng.normal(0, 0.005, n)
            a = _simplex_project(a, 1.0)

        # Increase tau periodically for sharper max
        if it % 30 == 0:
            tau = min(tau * 1.15, 8000.0)

        if M < best_M:
            best_M = M
            best_a = a.copy()

            a = best_a + rng.normal(0, 0.02, n)
            a = _simplex_project(a, 1.0)
            M = float(_conv_full(a).max())
            stall = 0
            max_stall = min(max_stall + 10, 100)
            tau = max(tau_init, tau * 0.7)

    return best_a, best_M


def _lp_direction(f, n, M, time_limit=8.0, top_k=0):
    """Find LP direction: max sum(b) s.t. f*b <= M, b >= 0. Returns normalized b."""
    b_conv = _conv_full(f)
    if top_k > 0:
        idx = np.argsort(-b_conv)[:top_k]
    else:
        idx = np.arange(2 * n - 1)

    # Build constraint matrix A[idx, :] where A[k, j] = f[k-j] if 0<=k-j<n else 0
    A = np.zeros((len(idx), n))
    for li, k in enumerate(idx):
        # A[li, j] = f[k-j] if 0 <= k-j < n
        j_max = min(n - 1, k)
        j_min = max(0, k - n + 1)
        for j in range(j_min, j_max + 1):
            A[li, j] = f[k - j]

    c_obj = -np.ones(n)
    b_ub = np.full(len(idx), M)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": time_limit})
    except Exception:
        return None

    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs


def _lp_optimize(f0, n, stop_time, seed=123, max_iters=30):
    """Original LP-based optimization."""
    rng = np.random.default_rng(seed)
    f = f0.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s

    best_f = f.copy()
    best_M = float(_conv_full(f).max())
    M = best_M
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break

        # Solve LP
        g_dir = _lp_direction(f, n, M, time_limit=6.0, top_k=0)
        if g_dir is None:
            stall += 3
            if stall > 15:
                break
            f = _simplex_project(f + rng.normal(0, 0.05, n), 1.0)
            M = float(_conv_full(f).max())
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-9:
            stall += 2
            continue

        # Line search: try several t values
        best_t = 0.0
        best_val = M
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            if time.monotonic() > stop_time - 2.0:
                break
            f_new = (1.0 - t) * f + t * g_dir
            f_new = np.clip(f_new, 0, None)
            s_new = f_new.sum()
            if s_new < 1e-10:
                continue
            f_new = f_new / s_new
            M_new = float(_conv_full(f_new).max())
            if M_new < best_val:
                best_val = M_new
                best_t = t

        if best_t > 0 and best_val < M - 1e-14:

            f = np.clip(f, 0, None)
            f = f / f.sum()
            M = float(_conv_full(f).max())
            if M < best_M:
                best_M = M
                best_f = f.copy()

        else:
            stall += 1
            f = _simplex_project(f + rng.normal(0, 0.03, n), 1.0)

def _coord_descent_polish(a0, stop_time, seed=99):

    n = len(a0)
    best = a0.copy().astype(np.float64)
    s = best.sum()

        improved = False

        # 1D coordinate descent

        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            total = 1.0  # since normalized
            # Try scaling: new value in [0, min(2*orig+0.01, 1-1e-6)]
            candidates = []
            if orig > 1e-10:
                for mult in (0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0):
                    candidates.append(orig * mult)
            candidates.extend([0.0, 0.001, 0.005, 0.02, 0.1, 0.3, 0.5])
            candidates = [c for c in candidates if 0 <= c <= 1.0 - 1e-9]

            best_c = orig
            best_Mc = best_M
            for cv in candidates:
                best[i] = cv
                # Renormalize
                ss = best.sum()
                if ss < 1e-10:
                    best[i] = orig
                    continue
                best_norm = best / ss
                Mc = float(_conv_full(best_norm).max())
                if Mc < best_Mc - 1e-15:
                    best_Mc = Mc
                    best_c = cv
            if best_Mc < best_M - 1e-14:
                best[i] = best_c
                ss = best.sum()
                best = best / ss
                best_M = best_Mc
                improved = True
            else:
                best[i] = orig

        # 2-coordinated
        n_pairs = min(200, n * 2)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 2.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            total_ij = oi + oj
            if total_ij < 1e-12:
                continue
            for frac in np.linspace(0, 1, 7):
                best[i] = frac * total_ij
                best[j] = (1 - frac) * total_ij
                ss = best.sum()
                if ss < 1e-10:
                    break
                best_norm = best / ss
                Mc = float(_conv_full(best_norm).max())
                if Mc < best_M - 1e-14:
                    best_M = Mc
                    improved = True
                    # Keep this change
                    break
            # Restore if didn't improve in this pair
            if not improved:
                best[i], best[j] = oi, oj
            improved = False

        # 3-coordinated (fewer)
        n_triples = min(50, n // 8)
        for _ in range(n_triples):
            if time.monotonic() > stop_time - 2.0:
                break
            idx = rng.choice(n, size=3, replace=False)
            i, j, k = int(idx[0]), int(idx[1]), int(idx[2])
            oi, oj, ok = best[i], best[j], best[k]
            total3 = oi + oj + ok
            if total3 < 1e-12:
                continue
            for w1 in (0.0, 0.3, 0.5, 0.7, 1.0):
                for w2 in (0.0, 0.3, 0.5, 0.7):
                    w3 = 1.0 - w1 - w2
                    if w3 < -1e-9:
                        continue
                    w3 = max(0.0, w3)
                    best[i], best[j], best[k] = w1 * total3, w2 * total3, w3 * total3
                    ss = best.sum()
                    if ss < 1e-10:
                        break
                    best_norm = best / ss
                    Mc = float(_conv_full(best_norm).max())
                    if Mc < best_M - 1e-14:
                        best_M = Mc
                        improved = True
                        break
        if improved:
            best[i], best[j], best[k] = best[i], best[j], best[k]  # keep
        else:
            best[i], best[j], best[k] = oi, oj, ok

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_M


def _block_polish(a0, stop_time, seed=555):
    """Polish by optimizing blocks of consecutive coordinates."""
    n = len(a0)
    best = a0.copy().astype(np.float64)
    s = best.sum()
    if s < 1e-10:
        return best, math.inf
    best = best / s
    best_M = float(_conv_full(best).max())
    rng = np.random.default_rng(seed)
    no_imp = 0

    block_sizes = [3, 4, 5, 8, 12]
    while time.monotonic() < stop_time - 2.0 and no_imp < 8:
        improved = False
        for bs in block_sizes:
            if time.monotonic() > stop_time - 2.0:
                break
            for start in range(0, n - bs + 1, max(1, bs // 2)):
                if time.monotonic() > stop_time - 3.0:
                    break
                if start + bs > n:
                    break
                block = best[start:start + bs].copy()
                bsum = block.sum()
                if bsum < 1e-12:
                    continue
                # Try several Dirichlet samples
                for trial in range(5):
                    alpha_param = np.ones(bs) * rng.uniform(0.5, 5.0)
                    w = rng.dirichlet(alpha_param)
                    best[start:start + bs] = w * bsum
                    ss = best.sum()
                    if ss < 1e-10:
                        best[start:start + bs] = block
                        continue
                    best_norm = best / ss
                    Mc = float(_conv_full(best_norm).max())
                    if Mc < best_M - 1e-14:
                        best_M = Mc
                        best = best_norm
                        improved = True
                        break
                best[start:start + bs] = block

        if improved:
            no_imp = 0
        else:
            no_imp += 1

    return best, best_M


def _make_initializations(parent, n, seed=42):
    """Create diverse initializations."""
    rng = np.random.default_rng(seed)
    inits = []
    i_arr = np.arange(n, dtype=np.float64)

    # Parent (normalized)
    p = parent.copy()
    ps = p.sum()
    if ps > 1e-10:
        inits.append(("parent", p / ps))
    else:
        inits.append(("parent", np.ones(n) / n))

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * i_arr / n))
    cos_w[cos_w < 0] = 0
    cos_w += 1e-6
    inits.append(("cosine", cos_w / cos_w.sum()))

    # Gaussian-like centered
    gauss = np.exp(-0.5 * ((i_arr - n / 2) / (n / 4)) ** 2)
    gauss += 1e-6
    inits.append(("gauss", gauss / gauss.sum()))

    # Two peaks
    two_peak = np.exp(-0.5 * ((i_arr - n * 0.25) / (n / 8)) ** 2) + \
               np.exp(-0.5 * ((i_arr - n * 0.75) / (n / 8)) ** 2)
    two_peak += 1e-6
    inits.append(("2peak", two_peak / two_peak.sum()))

    # Smoothed random
    noise = rng.normal(0, 1, n)
    kern_w = 5
    kern = np.exp(-0.5 * (np.arange(-kern_w, kern_w + 1) / 2.0) ** 2)
    kern /= kern.sum()
    smooth = np.convolve(noise, kern, 'same')
    smooth = np.abs(smooth) + 0.1
    inits.append(("smooth_rand", smooth / smooth.sum()))

    # Parent + smooth noise blend
    sn = np.convolve(rng.normal(0, 1, n), kern, 'same')
    blend = np.abs(parent * 0.8 + 0.2 * sn) + 1e-4
    inits.append(("parent_blend", blend / blend.sum()))

    # Multi-peak (3 peaks)
    multi = sum(np.exp(-0.5 * ((i_arr - n * f) / (n / 10)) ** 2)
                for f in (0.2, 0.5, 0.8)) + 1e-4
    inits.append(("3peak", multi / multi.sum()))

    # Zipper pattern
    z = np.ones(n)
    z[::2] *= 1.4
    z[1::2] *= 0.7
    inits.append(("zipper", z / z.sum()))

    return inits

                res, _ = _block_polish(best_seq, p_stop, seed=8000 + pidx * 211)
            else:
                # Another round of smooth gradient descent with high tau
                a_init = best_seq.copy()
                s = a_init.sum()
                if s > 1e-10:
                    a_init = a_init / s
                res, _ = _gradient_descent_smooth(a_init, p_stop, seed=10000 + pidx * 331,
                                                  tau_init=2000.0)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_c = np.clip(res, 0, 1000)
                sc = float(evaluate_sequence([float(x) for x in res_c]))
                if math.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = res_c.copy()
        except Exception:
            pass

    # Final quick polish
    final_stop = t0 + budget - 12.0
    if time.monotonic() < final_stop - 5.0:
        try:
            res, _ = _coord_descent_polish(best_seq, final_stop, seed=99999)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_c = np.clip(res, 0, 1000)
                sc = float(evaluate_sequence([float(x) for x in res_c]))
                if math.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = res_c.copy()
        except Exception:
            pass

    # Final validation
    final_seq = np.clip(best_seq, 0, 1000.0)
    final_seq = np.where(np.isfinite(final_seq), final_seq, 0.0)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    if final_sc < best_score:
        best_score = final_sc
        best_seq = final_seq

    return [float(x) for x in final_seq]
