def _conv_full(a):
    n = len(a)
    if n <= 800:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_conv_matrix(f, n):
    """A[k,j] = f[k-j] if 0<=k-j<n, else 0. Shape (2n-1, n)."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=4.0):
    """LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    if M <= 1e-18:
        return None
    A = _build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-14)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
            options={"presolve": True, "time_limit": timeout,
                     "dual_feasibility_tolerance": 1e-12,
                     "primal_feasibility_tolerance": 1e-12}
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g

def _coordinate_descent(a, stop_time, seed=0):
    """Coordinate descent polishing."""
    n = len(a)
    target_sum = math.sqrt(2.0 * n)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)
    best = best / fs * target_sum
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 30
    round_num = 0
    mults = [0.0, 0.1, 0.25, 0.5, 0.75, 1.25, 1.5, 2.0, 3.0, 5.0]

    while time.monotonic() < stop_time - 3.0 and no_imp < max_no_imp:
        round_num += 1
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break

            best_val = orig
            best_val_sc = best_sc
            for mlt in mults:
                nv = orig * mlt if mlt > 0 else 0.0

                    best[i] = orig
                    continue

                    best_val_sc = sc

            no_imp = 0
