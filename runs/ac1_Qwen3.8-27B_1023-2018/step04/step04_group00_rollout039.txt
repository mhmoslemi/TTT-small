        # dLSE/da[i] = sum_m wgt[m] * (a[m-i] + a[i-m])
        # where wgt[m] = softmax((c[m] - max(c))/T)
        # and a[j] = 0 for j outside [0, n-1].
        #
        # So dLSE/da[i] = sum_m wgt[m] * a[m-i] + sum_m wgt[m] * a[i-m]
        # = (wgt * a)[i] + (wgt * reverse(a))[i]  (with appropriate shifts)
        #
        # Actually, sum_m wgt[m] * a[m-i] is the cross-correlation of wgt and a, evaluated at i.
        # And sum_m wgt[m] * a[i-m] is also a cross-correlation.
        #
        # Let me define: for a fixed i, dLSE/da[i] = sum_m wgt[m] * (a[m-i] + a[i-m])
        # where a[j] is interpreted as 0 for j outside [0, n-1].
        #
        # Note: a[i-m] = a[-(m-i)], so this is the same as the cross-correlation of wgt with a,
        # evaluated at lag i, plus the cross-correlation of wgt with a, evaluated at lag -i.
        # But since we sum over all m and a is finite length, let's just compute it directly.
        #
        # A simpler way: dLSE/da[i] = (wgt * a)[i] + (reverse(wgt) * a)[i]
        # where * denotes convolution and we take the appropriate slice.
        #
        # Actually, the cleanest way:
        # grad[i] = sum_m wgt[m] * a[m-i] (with a[j]=0 for j not in [0,n-1])
        #         + sum_m wgt[m] * a[i-m]
        #
        # First term: for a given i, sum_m wgt[m] * a[m-i] = sum_j wgt[j+i] * a[j] (substituting j = m-i)
        # = (wgt_shifted * a)[i] where wgt_shifted is wgt with appropriate zero-padding.
        #
        # This is getting complicated. Let me just use the FFT approach:
        # Define f[i] = sum_m wgt[m] * a[m-i] for i in [0, n-1].
        # This is the cross-correlation of wgt and a, sampled at i.
        # Using FFT: f = ifft(fft(wgt, 2n-1) * conj(fft(a, 2n-1))) shifted...
        #
        # Actually, let me just use a simpler (but still efficient) approach:
        # f[i] = sum_m wgt[m] * a[m-i]
        #     = sum_{m=i}^{i+n-1} wgt[m] * a[m-i]  (since a[j] = 0 outside [0,n-1])
        #     = sum_{j=0}^{n-1} wgt[j+i] * a[j]  (substituting j = m-i, valid for m in [i, i+n-1])
        #     = dot(wgt[i:i+n], a)
        #
        # Wait, that's not right either. Let me be very careful.
        # f[i] = sum_m wgt[m] * a[m-i]
        # a[m-i] is nonzero only when 0 <= m-i <= n-1, i.e., i <= m <= i+n-1.
        # So f[i] = sum_{m=i}^{min(i+n-1, 2n-2)} wgt[m] * a[m-i]
        # Let j = m-i: f[i] = sum_{j=0}^{min(n-1, 2n-2-i)} wgt[j+i] * a[j]
        # For i in [0, n-1]: f[i] = sum_{j=0}^{n-1} wgt[j+i] * a[j] (since 2n-2-i >= n-1 for i <= n-1)
        # So f[i] = dot(wgt[i:i+n], a)
        #
        # Similarly, second term: g[i] = sum_m wgt[m] * a[i-m]
        # a[i-m] is nonzero when 0 <= i-m <= n-1, i.e., i-(n-1) <= m <= i.
        # g[i] = sum_{m=max(0, i-n+1)}^{i} wgt[m] * a[i-m]
        # Let j = i-m: g[i] = sum_{j=0}^{min(i, n-1)} wgt[i-j] * a[j]
        # For i in [0, n-1]: g[i] = sum_{j=0}^{min(i, n-1)} wgt[i-j] * a[j] = dot(wgt[0:i+1][::-1], a[0:i+1])
        #
        # Hmm, this is O(n²) total if done naively. Let me use FFT.
        
        # Better approach: use the fact that:
        # grad[i] = f[i] + g[i]
        # where f[i] = sum_j wgt[i+j] * a[j] and g[i] = sum_j wgt[i-j] * a[j]
        #
        # f is the cross-correlation of wgt and a (with wgt as the "signal" and a as the "template")
        # g is the cross-correlation of a and wgt (reversed roles)
        #
        # Using FFT:
        # f[i] = <wgt[.], shift_i(a)> = <a, shift_{-i}(wgt)>
        # This is the cross-correlation of a and wgt at lag i.
        #
        # In scipy, we can compute this as:
        # cross = np.correlate(wgt, a, mode='full')  # O(n²) for dense, but n=3072 gives 6143 points
        # f = cross[n-1-n+1 : n-1+n]  # the relevant slice
        
        # Actually, for n=3072, np.correlate is O(n²) = 9.4M, which is fine for a single call.
        # But we're calling it in every L-BFGS function evaluation, so we need it to be fast.
        # FFT-based: O(n log n)
        
        # Let me use the FFT approach properly.
        # cross-correlation of x and y: c[k] = sum_n x[n+k] * y[n] for k in [-(N-1), N-1]
        # Using FFT: c = ifft(conj(fft(x, L)) * fft(y, L)) where L >= N+x_N-1... 
        # Actually: c = ifft(fft(x, L) * conj(fft(y, L))) shifted...
        
        # Let me just use a direct FFT computation:
        # f[i] = sum_j wgt[j+i] * a[j] for j in [0, n-1]
        # This is: (wgt ⊛ a)[i] where ⊛ denotes cross-correlation
        # f = ifft(fft(wgt, L) * conj(fft(a, L))) [appropriate slice]
        # where L = 2n-1
        
        # Let L = 2n-1 (or next_fast_len(2n-1))
        # W = fft(wgt, L), A = fft(a, L)
        # c = ifft(W * conj(A), L)
        # Then c[k] = sum_n wgt[n] * conj(a[n-k]) ... hmm, I need to be careful with conventions.
        
        # In numpy/scipy, for real signals:
        # ifft(fft(x, L) * conj(fft(y, L)))[k] = sum_n x[n] * y[(n-k) mod L]
        # For linear (not circular) correlation, we need L >= len(x) + len(y) - 1.
        # Then for k in [0, L-1]:
        #   if k < len(y): result = sum_n x[n] * y[n-k] (with x[n]=0 for n >= len(x), y[n-k]=0 for n-k >= len(y))
        #   if k >= L - len(y) + 1: result = sum_n x[n-(k-(L-len(y)+1))] * y[n] ... this is the wrap-around part.
        
        # This is getting too complicated. Let me just use a simpler approach for n=3072:
        # Since n=3072, the convolution has 6143 elements. Computing the gradient via
        # matrix multiplication or direct summation is O(n * 2n) = O(n²) ≈ 9.4M per evaluation.
        # With FFT it's O(n log n) ≈ 40K. Let me use FFT properly.
        
        # Simplest correct approach:
        # grad[i] = sum_m wgt[m] * (a[m-i] + a[i-m])
        # = sum_m wgt[m] * a[m-i] + sum_m wgt[m] * a[i-m]
        #
        # Let h = wgt (length 2n-1) and a (length n, zero-padded to 2n-1).
        # Define a_ext[j] = a[j] for j in [0, n-1], 0 otherwise.
        # 
        # First term: T1[i] = sum_m h[m] * a_ext[m-i]
        # This is the cross-correlation of h and a_ext, evaluated at lag i.
        # T1 = correlate(h, a_ext, 'full')[shifted]
        # Using FFT: T1 = ifft(fft(h, L) * conj(fft(a_ext, L)))
        # The result is a circular cross-correlation. For linear, we need L >= len(h) + len(a_ext) - 1 = (2n-1) + (2n-1) - 1 = 4n-3.
        
        # Hmm, that's a large FFT. For n=3072: L = 4*3072-3 = 12285, next_fast_len = 16384.
        # That's fine for a single computation.
        
        # Actually, let me think about this differently.
        # 
        # The gradient of (a*a)[m] with respect to a[i] is:
        # - a[m-i] if 0 <= m-i < n (i.e., if a[m-i] exists)
        # - a[i-m] if 0 <= i-m < n (i.e., if a[i-m] exists)
        # (with both terms present if both m-i and i-m are in [0, n-1])
        #
        # So grad[i] = sum_m wgt[m] * (a[m-i] + a[i-m]) (with a[j] = 0 for j outside [0,n-1])
        #
        # Let's define the "interaction kernel" for each i:
        # K_i[m] = a[m-i] + a[i-m] (with a[j] = 0 outside [0,n-1])
        #
        # K_i[m] is nonzero when m-i in [0,n-1] or i-m in [0,n-1], i.e., m in [i, i+n-1] ∪ [i-n+1, i].
        #
        # So grad[i] = sum_{m in [max(0,i-n+1), min(2n-2, i+n-1)]} wgt[m] * K_i[m]
        #
        # For efficiency, note that K_i[m] = a[m-i] + a[i-m].
        # Let u = m-i. Then K_i[m] = a[u] + a[-u].
        # grad[i] = sum_u wgt[i+u] * (a[u] + a[-u]) (with appropriate bounds)
        #
        # = sum_u wgt[i+u] * a[u] + sum_u wgt[i+u] * a[-u]
        #
        # First sum: sum_u wgt[i+u] * a[u] = cross-correlation of wgt and a at lag i
        # Second sum: sum_u wgt[i+u] * a[-u] = sum_u wgt[i+u] * rev_a[u] (where rev_a[u] = a[-u] = a[n-1-u])
        # = cross-correlation of wgt and rev_a at lag i
        #
        # So grad = cross_corr(wgt, a) + cross_corr(wgt, rev_a) (evaluated at lags 0..n-1)
        #
        # Using FFT for cross-correlation:
        # cross_corr(x, y)[k] = sum_n x[n+k] * y[n] (for k in [-(len(y)-1), len(x)-1])
        
        # OK I think the cleanest way is:
        # grad = ifft(fft(wgt, L) * (conj(fft(a, L)) + conj(fft(rev_a, L))))[appropriate slice]
        # where L >= 4n-3 to avoid circular aliasing.
        
        # Actually, I realize this is overcomplicating things. For n=3072, each L-BFGS
        # function evaluation can afford O(n²) = 9.4M operations. Let me just compute the
        # gradient directly.
        
        pass  # Will implement below

    # Let me use a simpler gradient computation that's O(n²) but with good constants.
    # For n=3072, O(n²) = 9.4M per evaluation, and L-BFGS does ~1000 evaluations total,
    # so 9.4G operations total. That's about 10 seconds in numpy. Acceptable.
    
    # Actually, let me use the FFT approach after all. Here's a clean way:
    # 
    # c = a * a (convolution, length 2n-1)
    # dc/da[i] is a vector of length 2n-1 where (dc/da[i])[m] = a[m-i] + a[i-m] (with zeros outside)
    # 
    # grad[i] = dot(wgt, dc/da[i]) = sum_m wgt[m] * (a[m-i] + a[i-m])
    #
    # Now, sum_m wgt[m] * a[m-i]:
    # Let's define b = a zero-padded to length 2n-1 (or just work with a of length n).
    # sum_m wgt[m] * a[m-i] = sum_j wgt[j+i] * a[j] (substituting m = j+i)
    # = dot(wgt[i:i+n], a)  (for i in [0, n-1], since wgt has length 2n-1 >= i+n for i <= n-1)
    #
    # Similarly: sum_m wgt[m] * a[i-m] = sum_j wgt[i-j] * a[j] (substituting m = i-j)
    # = dot(wgt[max(0,i-n+1):i+1][::-1], a[0:min(i+1, n)])
    #
    # Total: O(n²) if done naively. With FFT: O(n log n).
    #
    # Let me use the FFT approach:
    # Define x = wgt (length 2n-1), y = a (length n).
    #
    # First term: f[i] = sum_j wgt[j+i] * a[j] for i in [0, n-1]
    # This is: (wgt ⊛ a)[i] where ⊛ is cross-correlation, evaluated at i.
    # Using FFT: f = ifft(fft(wgt, L) * conj(fft(a, L))) where L >= 3n-2
    # The result f has length L, and f[i] for i in [0, n-1] is what we want.
    #
    # Second term: g[i] = sum_j wgt[i-j] * a[j] for i in [0, n-1]
    # Let k = -i: g[i] = sum_j wgt[k+j+n] * a[j] ... hmm, this is the negative lag part.
    #
    # Actually, g[i] = sum_j wgt[i-j] * a[j] = sum_j wgt[-(j-i)] * a[j]
    # Let l = j-i: g[i] = sum_l wgt[-l] * a[l+i] = sum_l wgt[-l] * a_ext[l+i]
    # where a_ext is a zero-padded to length 2n-1.
    #
    # This is the cross-correlation of wgt and a_ext at lag -(i), or equivalently,
    # the cross-correlation of a_ext and wgt at lag i.
    #
    # Using FFT: g = ifft(fft(a_ext, L) * conj(fft(wgt, L))) where L >= 3n-2
    #
    # So: grad = f[0:n] + g[0:n]
    # where f = ifft(fft(wgt, L) * conj(fft(a, L))) and g = ifft(fft(a_ext, L) * conj(fft(wgt, L)))
    #
    # But wait, a_ext is just a zero-padded to length 2n-1, and wgt already has length 2n-1.
    # So if L = next_fast_len(3*n-2) or larger:
    # F_wgt = rfft(wgt, L)
    # F_a = rfft(a, L)  # a is zero-padded to L automatically
    # f_full = irfft(F_wgt * np.conj(F_a), L)  # this gives cross-correlation of wgt and a
    # g_full = irfft(F_a * np.conj(F_wgt), L)  # this gives cross-correlation of a and wgt
    #
    # But f_full and g_full are related: g_full[k] = f_full[L-k] (by symmetry of cross-correlation).
    # Actually for real signals: ifft(A * conj(B))[k] = sum_n a[n] * b[(n-k) mod L]
    # and ifft(B * conj(A))[k] = sum_n b[n] * a[(n-k) mod L]
    # These are different unless a = b.
    #
    # Let me just compute both.
    
    n = n  # capture
    L_fft = next_fast_len(3 * n)  # enough to avoid aliasing
    
    def compute_grad(a_vec, wgt):
        """Compute gradient of LSE(c, T) w.r.t. a, where c = a*a and wgt = softmax((c-max)/T)."""
        FA = rfft(a_vec, L_fft)
        FW = rfft(wgt, L_fft)
        # f[i] = sum_j wgt[j+i] * a[j] = cross-correlation of wgt and a at lag i
        f = irfft(FW * np.conj(FA), L_fft)
        # g[i] = sum_j wgt[i-j] * a[j] = cross-correlation of a and wgt at lag i
        g = irfft(FA * np.conj(FW), L_fft)
        # grad[i] = f[i] + g[n-1-i] ... hmm I need to get the indexing right.
        # Actually, let me just verify with a small example numerically.
        # For now, let me use the direct computation which is O(n * 2n) = O(n²):
        # grad[i] = sum_m wgt[m] * (a[m-i] + a[i-m])
        
        # Direct computation (O(n²)):
        # For each i, the sum involves m from max(0, i-n+1) to min(2n-2, i+n-1)
        # This is at most 2n-1 terms.
        # Total: O(n * 2n) = O(n²)
        
        # Actually, let me reconsider. For n=3072:
        # Each function eval: O(n²) = 9.4M for gradient + O(n log n) for the LSE value
        # L-BFGS: ~500 iterations × ~10 line search evals = ~5000 function evals
        # Total: 5000 × 9.4M = 47G operations → too slow!
        
        # So I need the FFT approach. Let me work it out properly.
        
        # The issue is getting the right slice from the FFT result.
        # Let me define:
        # x = wgt (length 2n-1)
        # y = a (length n)
        # L >= (2n-1) + n - 1 = 3n-2
        #
        # cross_corr(x, y)[k] = sum_m x[m] * y[m-k] (for k in [-(n-1), 2n-2])
        # Using FFT: cc = ifft(fft(x, L) * conj(fft(y, L)), L)
        # cc[k] = sum_m x[m] * y[(m-k) mod L]
        # For the linear case (no wrap-around), this equals:
        # cc[k] = sum_m x[m] * y[m-k] for k in [0, L-1]
        # where y[j] = 0 for j < 0 or j >= n.
        #
        # So cross_corr(x,y)[k] = sum_{m=k}^{k+n-1} x[m] * y[m-k] (with x[m]=0 for m >= 2n-1)
        # = sum_{j=0}^{n-1} x[j+k] * y[j]
        #
        # For k in [0, 2n-2]: this is valid (no wrap-around since L >= 3n-2 > 2n-2+n = 3n-2).
        # Wait, we need L >= (2n-1) + (n-1) = 3n-2 to avoid wrap-around for all k in [0, 2n-2].
        # Actually, we need: for all k in [0, 2n-2] and all j in [0, n-1], (j+k) mod L = j+k.
        # This requires j+k < L for all valid j, k: max(j+k) = (n-1) + (2n-2) = 3n-3 < L = 3n-2. ✓
        
        # So: cross_corr(wgt, a)[k] = sum_j wgt[j+k] * a[j] for k in [0, 2n-2]
        # This is exactly our first term: f[i] = cross_corr(wgt, a)[i] for i in [0, n-1].
        
        # For the second term: g[i] = sum_j wgt[i-j] * a[j]
        # Let k = -j: sum_k wgt[i+k] * a[-k] = sum_k wgt[i+k] * a_rev[k] where a_rev[k] = a[-k] = a[n-1-k]
        # = cross_corr(wgt, a_rev)[i]
        #
        # So: grad[i] = cross_corr(wgt, a)[i] + cross_corr(wgt, a_rev)[i] for i in [0, n-1]
        #
        # Using FFT:
        # cc1 = ifft(fft(wgt, L) * conj(fft(a, L)), L)[:2n-1]  # cross_corr(wgt, a)
        # a_rev = a[::-1]
        # cc2 = ifft(fft(wgt, L) * conj(fft(a_rev, L)), L)[:2n-1]  # cross_corr(wgt, a_rev)
        # grad = cc1[:n] + cc2[:n]
        
        a_rev = a_vec[::-1]
        FA = rfft(a_vec, L_fft)
        FAR = rfft(a_rev, L_fft)
        FW = rfft(wgt, L_fft)
        cc1 = irfft(FW * np.conj(FA), L_fft)[:2*n-1]
        cc2 = irfft(FW * np.conj(FAR), L_fft)[:2*n-1]
        return cc1[:n] + cc2[:n]

    # Now the LSE function:
    def lse_func(u, T):
        """u is the log-parameterization, p = softmax(u)."""
        # Convert u to p
        u_shifted = u - u.max()
        p = np.exp(u_shifted)
        s = p.sum()
        if s < 1e-300:
            return 1e100, np.zeros_like(u)
        p = p / s
        
        # Compute convolution
        c = conv_fast(p)
        if c.size == 0:
            return 1e100, np.zeros_like(u)
        
        cmax = float(c.max())
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            wgt = np.zeros_like(c)
            wgt[int(np.argmax(c))] = 1.0
            fval = cmax
        else:
            fval = cmax + T * math.log(sum_exp / c.size)
            if not math.isfinite(fval):
                fval = cmax
            wgt = exps / sum_exp
        
        # Compute gradient
        grad_p = compute_grad(p, wgt)
        # Transform to u-space: du_j = p_j * (grad_p_j - sum(p * grad_p))
        tdp = float(np.dot(p, grad_p))
        grad_u = p * (grad_p - tdp)
        grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
        
        return fval, grad_u

    # Initial u
    p_init = _softmax(p)
    u = np.log(np.maximum(p_init, 1e-300)) - np.mean(np.log(np.maximum(p_init, 1e-300)))

    # Multi-stage optimization
    stages_T = [0.2, 0.06, 0.018, 0.005, 0.0015, 0.0004, 0.0001]
    best_u = u.copy()
    best_f = None

    try:
        best_f = lse_func(u, stages_T[0])[0]
    except:
        pass

    for T in stages_T:
        now = time.monotonic()
        if now > stop_time - 10:
            break
        chunk_time = min(80.0, stop_time - 15 - now)
        if chunk_time < 10:
            break
        chunk_end = now + chunk_time
        
        while time.monotonic() < chunk_end - 2:
            opts = {
                "maxiter": 30,
                "gtol": 1e-10,
                "ftol": 1e-15,
                "maxls": 20,
                "maxcor": 20,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except:
                break
            
            if res.x is not None and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and (best_f is None or fval < best_f):
                    best_f = fval
                    best_u = res.x.copy()
                    # Check time
                    if time.monotonic() > chunk_end - 5:
                        break
                    p_check = _softmax(res.x)
                    consider(p_check)
            if getattr(res, "status", 1) == 0:
                break

    return _softmax(best_u)


def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0/u.size)
    return e / s


def _coord_descent(a, stop_time, consider, seed=12345):
    """Coordinate descent with exact scoring."""
    n = a.size
    rng = np.random.default_rng(seed)
    
    # Normalize to sum=1
    a = a / a.sum()
    c = _conv_fft(a)
    S = 1.0
    cur_score = float(2.0 * n * c.max())
    
    best_p = a.copy()
    best_score = cur_score
    
    G = 24  # grid points
    
    sweep = 0
    flat = 0
    
    while time.monotonic() < stop_time - 2:
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time - 0.5:
                break
            
            v = float(a[i])
            start = i
            end = min(i + n, 2 * n - 1)
            Ls = end - start
            if Ls <= 0:
                continue
            
            base = c[start:end].copy()
            # infl[k] = 2 * a[k + i - i] = 2 * a[k] for k in [0, Ls-1]
            # Wait: the change in c[j] when a[i] changes by w is:
            # c[j] += 2*w*a[j-i] for j in [i, i+n-1] where j-i in [0, n-1]
            # So for j = start + k (k in [0, Ls-1]): c[j] += 2*w*a[k]
            # infl[k] = 2*a[k]
            infl = 2.0 * a[:Ls]
            
            # max outside the affected region
            max_const = 0.0
            if start > 0:
                max_const = max(max_const, float(c[:start].max()))
            if end < 2 * n - 1:
                max_const = max(max_const, float(c[end:].max()))
            
            # Grid search
            # x_range: from 0 to some upper bound
            x_hi = max(10.0 * v, 50.0 / n, 0.02)
            x_hi = min(x_hi, 5.0)  # cap
            
            pts = np.linspace(0.0, x_hi, G + 1)
            w_grid = pts - v
            
            # Vectorized evaluation
            # mat[k, m] = base[k] + w_grid[m] * infl[k]
            mat = base[None, :] + w_grid[:, None] * infl[None, :]  # shape (G+1, Ls)
            max_in_region = mat.max(axis=1)
            
            # Diagonal correction: at position 2i, add w²
            diag_idx = 2 * i - start
            if 0 <= diag_idx < Ls:
                max_in_region = np.maximum(max_in_region, base[diag_idx] + w_grid * infl[diag_idx] + w_grid**2)
            
            max_total = np.maximum(max_in_region, max_const)
            denom = (S + w_grid)**2
            scores = (2.0 * n * max_total) / denom
            
            valid = np.isfinite(scores) & (denom > 1e-30)
            if not np.any(valid):
                continue
            
            scores[~valid] = math.inf
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])
            
            # Fine refinement
            if G > 1 and best_x > 0 and best_x < x_hi:
                dw = x_hi / G
                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)
                fine_pts = np.linspace(lo, hi, 16)
                fine_w = fine_pts - v
                fine_mat = base[None, :] + fine_w[:, None] * infl[None, :]
                fine_max = fine_mat.max(axis=1)
                if 0 <= diag_idx < Ls:
                    fine_max = np.maximum(fine_max, base[diag_idx] + fine_w * infl[diag_idx] + fine_w**2)
                fine_total = np.maximum(fine_max, max_const)
                fine_denom = (S + fine_w)**2
                fine_scores = (2.0 * n * fine_total) / fine_denom
                valid_f = np.isfinite(fine_scores) & (fine_denom > 1e-30)
                if np.any(valid_f):
                    fine_scores[~valid_f] = math.inf
                    imin_f = int(np.argmin(fine_scores))
                    if fine_scores[imin_f] < best_val:
                        best_x = float(fine_pts[imin_f])
                        best_val = float(fine_scores[imin_f])
            
            if math.isfinite(best_val) and best_val < cur_score - 1e-12:
                w = best_x - v
                a[i] = best_x
                S += w
                c[start:end] += w * infl
                if 0 <= 2*i < 2*n-1:
                    c[2*i] += w * w
                cur_score = best_val
                improved = True
        
        # Rescale
        if S > 1e-300 and (S > 1.05 or S < 0.95):
            inv = 1.0 / S
            a *= inv
            c *= inv * inv
            S = 1.0
            cur_score = float(2.0 * n * c.max())
        
        # Periodic recompute for numerical stability
        if sweep % 50 == 0 and sweep > 0:
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss > 0:
                a /= ss
                S = 1.0
            c = _conv_fft(a)
            cur_score = float(2.0 * n * c.max())
        
        if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = a.copy()
            consider(a.copy())
        
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3 and time.monotonic() < stop_time - 20:
                # Perturbation
                try:
                    a = a * np.exp(rng.normal(0.0, 0.03, n))
                    a = np.maximum(a, 0.0)
                    ss = float(a.sum())
                    if ss > 0:
                        a /= ss
                        S = 1.0
                    c = _conv_fft(a)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                except:
                    break
            elif flat >= 5:
                break
        
        sweep += 1
    
    # Final
    a = np.maximum(a, 0.0)
    ss = float(a.sum())
    if ss > 0:
        a /= ss
    c = _conv_fft(a)
    sc = float(2.0 * n * c.max())
    if sc < best_score:
        best_score = sc
        best_p = a.copy()
    
    return best_p
