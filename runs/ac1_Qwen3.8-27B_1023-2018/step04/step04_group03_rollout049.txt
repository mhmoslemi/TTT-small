import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _get_parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(416, dtype=np.float64)

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)  # numerical artifacts
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_conv_matrix(f, n):
    """Build sparse matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0"""
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        j = i_start
        for j in range(i_start, i_end):
            v = f[k - j]
            if v > 1e-20:
                rows.append(k)
                cols.append(j)
                vals.append(v)
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def _solve_lp(f, n, M, timeout=5.0):
    """Solve max sum(g) s.t. f*g[k] <= M for all k, g >= 0"""
    A = _build_conv_matrix(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n

                      method='highs', options={'time_limit': timeout, 'presolve': True})

        return None

        return None

    return g

        f_new = np.clip(f_new, 0, None)
        scs[i] = _score(f_new)
    
    best_idx = np.argmin(scs)

        else:

    if sc_star < best_sc:

    return best_t, best_sc

                stall = 0

        f_sum = f.sum()
        if g_sum < 1e-15:
            stall += 5

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(416, dtype=np.float64)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_conv_matrix(f, n):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for j in range(i_start, i_end):
            v = f[k - j]
            if v > 1e-20:
                rows.append(k)
                cols.append(j)
                vals.append(v)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _solve_lp_full(f, n, M, timeout=3.0):
    A = _build_conv_matrix(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _line_search(f, g, n_evals=25):
    best_t = 0.0
    best_sc = _score(f)
    ts = np.linspace(0.0, 1.0, n_evals)
    scs = np.empty(n_evals)
    for i in range(n_evals):
        t = ts[i]
        f_new = (1.0 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        scs[i] = _score(f_new)
    best_idx = int(np.argmin(scs))
    best_t = ts[best_idx]
    best_sc = scs[best_idx]
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(len(ts) - 1, best_idx + 1)]
    for _ in range(10):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.clip((1.0 - m1) * f + m1 * g, 0, None)
        f2 = np.clip((1.0 - m2) * f + m2 * g, 0, None)
        s1 = _score(f1)
        s2 = _score(f2)
        if s1 < s2:
            hi = m2
        else:
            lo = m1
    t_star = (lo + hi) / 2.0
    f_star = np.clip((1.0 - t_star) * f + t_star * g, 0, None)
    sc_star = _score(f_star)
    if sc_star < best_sc:
        best_t = t_star
        best_sc = sc_star
    return best_t, best_sc


def _optimize_lp_main(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f_sum = float(f.sum())
    best_f = f.copy()
    best_score = _score(f)

    stall = 0
    max_stall = 60
    iter_count = 0

    while time.monotonic() < stop_time - 5.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _solve_lp_full(f, n, M, timeout=3.0)
        iter_count += 1

        if g is None:
            stall += 10
            if stall > max_stall:
                noise = rng.normal(0, 0.06 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                if f.sum() > 1e-10:
                    f = f * (f_sum / f.sum())
                stall = 0
                max_stall = min(max_stall + 40, 300)
            continue

        g_sum = float(g.sum())
        if g_sum < 1e-15:
            stall += 5
            if stall > max_stall:
                break
            continue

        g = g * (f_sum / g_sum)

        dist = np.linalg.norm(g - f)
        if dist < 1e-9 * max(1.0, np.linalg.norm(f)):
            stall += 2
            if stall > max_stall:
                # Try reversed
                f_rev = f[::-1].copy()
                sc_rev = _score(f_rev)
                if sc_rev < best_score:
                    f = f_rev
                    best_f = f.copy()
                    best_score = sc_rev
                    stall = 0
                else:
                    break
            continue

        best_t, new_sc = _line_search(f, g, n_evals=20)

        if new_sc < best_score - 1e-13:
            f = np.clip((1.0 - best_t) * f + best_t * g, 0, None)
            f_sum = float(f.sum())
            if f_sum < 0.01:
                f = best_f.copy()
                f_sum = float(f.sum())
                continue
            best_f = f.copy()
            best_score = new_sc
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                # Perturbation strategies
                strategies = [
                    lambda: np.clip(f + rng.normal(0, 0.1 * max(1.0, f.mean()), n), 0, None),
                    lambda: np.clip(f * (1.0 + 0.1 * rng.normal(0, 1, n)), 0, None),
                    lambda: f[::-1].copy(),
                    lambda: np.clip(f + 0.05 * max(1.0, f.mean()) * rng.standard_normal(n), 0, None),
                ]
                s_idx = int(stall % len(strategies))
                f_new = strategies[s_idx]()
                if f_new.sum() > 1e-10:
                    f_new = f_new * (f_sum / f_new.sum())
                    sc_new = _score(f_new)
                    if sc_new < best_score * 1.02:
                        f = f_new
                        f_sum = float(f.sum())
                stall = 0
                max_stall = min(max_stall + 40, 300)

    return best_f, best_score


def _coordinate_descent(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    best_sc = _score(f)
    no_imp_rounds = 0
    max_no_imp = 8

    multipliers = (0.2, 0.4, 0.6, 0.8, 1.05, 1.15, 1.3, 1.6, 2.0, 0.05)
    fine_multipliers = (0.9, 1.1, 0.95, 1.05, 0.85, 1.15)

    while time.monotonic() < stop_time - 5.0:
        improved = False
        order = list(rng.permutation(n))

        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break
            orig = f[i]
            if orig < 1e-14:
                val = 0.01 * max(1.0, float(f.mean()))
                f[i] = val
                sc = _score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                else:
                    f[i] = 0.0
                continue

            best_local = best_sc
            best_val = orig

            for mlt in multipliers:
                f[i] = orig * mlt
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_val = f[i]

            if best_local < best_sc - 1e-15:
                f[i] = best_val
                orig = best_val
                for mlt in fine_multipliers:
                    f[i] = orig * mlt
                    if f.sum() < 0.01:
                        f[i] = orig
                        continue
                    sc = _score(f)
                    if sc < best_local - 1e-15:
                        best_local = sc
                        best_val = f[i]
                    else:
                        f[i] = orig
                f[i] = best_val
                best_sc = best_local
                improved = True
            else:
                f[i] = best_val

        if not improved:
            no_imp_rounds += 1
            if no_imp_rounds >= max_no_imp:
                break
        else:
            no_imp_rounds = 0

    return f.copy(), best_sc


def _subgradient_phase(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-10, None)
    f = f / f.sum()
    best_f = f.copy()
    best_score = _score(f)
    lr = 0.3
    stall = 0
    max_stall = 100

    while time.monotonic() < stop_time - 5.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        # Use top few peaks for gradient
        top_k = max(3, min(20, int(0.03 * len(b))))
        peak_idx = np.argsort(b)[-top_k:]
        if b[peak_idx[0]] < 0.8 * M:
            peak_idx = np.array([np.argmax(b)])

        grad = np.zeros(n)
        for k in peak_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):
                grad[j] += f[k - j]
        grad /= len(peak_idx)
        grad *= 2.0

        gmax = grad.max()
        if gmax < 1e-15:
            break

        # Multiplicative update
        f_new = f * np.exp(-lr * grad / gmax)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        sc = _score(f_new)
        if sc < best_score - 1e-13:
            best_f = f_new.copy()
            best_score = sc
            stall = 0
            lr *= 0.995
            f = f_new
        else:
            # Try damped
            f_damp = 0.7 * f + 0.3 * f_new
            f_damp = np.clip(f_damp, 1e-10, None)
            f_damp = f_damp / f_damp.sum()
            sc_d = _score(f_damp)
            if sc_d < best_score - 1e-13:
                best_f = f_damp.copy()
                best_score = sc_d
                f = f_damp
                stall = 0
                lr *= 0.98
            else:
                stall += 1
                lr *= 0.93
                f = f_new
                if stall > max_stall:
                    noise = rng.normal(0, 0.02, n)
                    f = np.clip(f + noise, 1e-10, None)
                    f = f / f.sum()
                    stall = 0
                    lr = 0.2
                    max_stall = min(max_stall + 50, 400)

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    # Normalize parent
    f_init = np.clip(parent, 0, 1000).astype(np.float64).copy()
    if f_init.sum() < 0.01:
        f_init = np.ones(n_p)
    target_sum = np.sqrt(2.0 * n_p)
    f_init = f_init * (target_sum / f_init.sum())

    best_seq = f_init.copy()
    best_score = _score(best_seq)

    def _update_best(candidate):
        nonlocal best_seq, best_score
        if candidate is None or len(candidate) == 0:
            return
        candidate = np.clip(np.asarray(candidate, dtype=np.float64), 0, 1000)
        if not np.all(np.isfinite(candidate)):
            return
        if candidate.sum() < 0.01:
            return
        sc = _score(candidate)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq = candidate.copy()
            best_score = sc

    rng = np.random.default_rng(seed)

    # === Phase 1: Main LP optimization (55% time) ===
    phase1_end = t0 + budget * 0.55

    # Multiple initializations for LP
    inits = [f_init.copy()]

    # Noisy versions
    for sig in (0.05, 0.1, 0.2):
        inits.append(np.clip(f_init + rng.normal(0, sig * f_init.mean(), n_p), 0, None) * (target_sum / max(1e-10, np.clip(f_init + rng.normal(0, sig * f_init.mean(), n_p), 0, None).sum())))

    # Reversed
    inits.append(f_init[::-1].copy())

    # Phase-shifted cosines
    i_arr = np.arange(n_p, dtype=np.float64)
    center = (n_p - 1) / 2.0
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * (i_arr - center) / n_p * 2)))
    # Gaussian
    inits.append(np.exp(-0.5 * ((i_arr - center) / (n_p * 0.25)) ** 2))
    # Two peaks
    inits.append(np.exp(-0.5 * ((i_arr - n_p * 0.3) / (n_p * 0.1)) ** 2) +
                 np.exp(-0.5 * ((i_arr - n_p * 0.7) / (n_p * 0.1)) ** 2))

    # Normalize all inits
    norm_inits = []
    for init in inits:
        init = np.clip(init, 0, None)
        s = init.sum()
        if s > 1e-10:
            init = init * (target_sum / s)
        norm_inits.append(init)

    time_per_init = max(15.0, (phase1_end - t0) / len(norm_inits))

    for idx, init in enumerate(norm_inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        if stop - time.monotonic() < 5.0:
            break
        try:
            res, sc = _optimize_lp_main(init, stop, seed + 1000 * (idx + 1))
            _update_best(res)
        except Exception:
            pass

    # === Phase 2: Second LP pass on best with finer line search (15% time) ===
    phase2_end = t0 + budget * 0.70
    if time.monotonic() < phase2_end - 15.0:
        stop = phase2_end - 5.0
        try:
            res, sc = _optimize_lp_main(best_seq, stop, seed + 555555)
            _update_best(res)
        except Exception:
            pass

    # === Phase 3: Subgradient (10% time) ===
    phase3_end = t0 + budget * 0.80
    if time.monotonic() < phase3_end - 10.0:
        stop = phase3_end - 5.0
        try:
            res, sc = _subgradient_phase(best_seq, stop, seed + 66666)
            _update_best(res)
        except Exception:
            pass

    # === Phase 4: LP again after subgradient (10% time) ===
    phase4_end = t0 + budget * 0.90
    if time.monotonic() < phase4_end - 10.0:
        stop = phase4_end - 5.0
        try:
            res, sc = _optimize_lp_main(best_seq, stop, seed + 777777)
            _update_best(res)
        except Exception:
            pass

    # === Phase 5: Coordinate descent polish (8% time) ===
    phase5_end = deadline + 2.0
    if time.monotonic() < phase5_end - 15.0:
        stop = phase5_end - 5.0
        try:
            res, sc = _coordinate_descent(best_seq, stop, seed + 88888)
            _update_best(res)
        except Exception:
            pass

    # === Phase 6: Final LP + coordinate on best (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 20.0:
        half_rem = remaining * 0.5
        if half_rem > 10.0:
            try:
                res, sc = _optimize_lp_main(best_seq, t0 + budget - 20.0, seed + 99999)
                _update_best(res)
            except Exception:
                pass
        rem2 = deadline - time.monotonic()
        if rem2 > 15.0:
            try:
                res, sc = _coordinate_descent(best_seq, deadline - 3.0, seed + 111111)
                _update_best(res)
            except Exception:
                pass

    # Final validation with exact evaluator
    try:
        final_list = [float(x) for x in best_seq]
        final_sc = float(evaluate_sequence(final_list))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):

            final_sc = float(evaluate_sequence(final_list))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]

    return [float(x) for x in best_seq]
