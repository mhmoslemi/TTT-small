import numpy as np

from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _conv_cross(a, b):
    n = len(a)
    m = len(b)
    fsize = next_fast_len(n + m - 1)
    FA = rfft(a, fsize)
    FB = rfft(b, fsize)
    c = irfft(FA * FB, fsize)[:n + m - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:

    return 2.0 * n * m / (s * s)


def _build_A(active_idx, f, n):
    K = len(active_idx)
    A = np.zeros((K, n))
    for i, k in enumerate(active_idx):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(k_int, n - 1)
        if hi >= lo:
            A[i, lo:hi + 1] = f[k_int - hi:k_int - lo + 1][::-1]
    return A


def _lp_solve(f, n, M, active_idx, time_limit=8.0):
    K = len(active_idx)

    try:

            c_obj, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': time_limit}
        )
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)

    gs = float(g.sum())

        return None
    return g / gs


def _line_search_quad(ff, fg, gg):
    """Golden section on max of convex quadratics in t."""
    def obj(t):
        val = (1.0 - t) ** 2 * ff + 2.0 * t * (1.0 - t) * fg + t * t * gg
        return float(val.max())

    lo, hi = 0.0, 1.0
    best_t, best_val = 0.0, obj(0.0)

    m1 = hi - (hi - lo) / gr
    m2 = lo + (hi - lo) / gr
    f1, f2 = obj(m1), obj(m2)

    for _ in range(45):
        if (hi - lo) < 1e-12:
            break
        if f1 <= f2:
            hi, f2, m2 = m2, f1, m1
            m1 = hi - (hi - lo) / gr
            f1 = obj(m1)
        else:
            lo, f1, m1 = m1, f2, m2
            m2 = lo + (hi - lo) / gr
            f2 = obj(m2)
        tm = (lo + hi) / 2.0
        fm = obj(tm)
        if fm < best_val:
            best_val = fm
            best_t = tm

    return best_t, best_val


def _one_lp_iteration(f, n, K_active, time_limit, rng):
    ff = _conv_full(f)
    M = float(ff.max())
    top_k = min(K_active, len(ff))

    if g is None:
        return None, M
    return g, M
