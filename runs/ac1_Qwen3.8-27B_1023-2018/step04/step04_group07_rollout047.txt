```python
def _proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.where(u * np.arange(1, n+1) > (cssv - 1), np.arange(1, n+1)))
    theta = (cssv[rho-1] + 1) / rho
    return np.maximum(v - theta, 0)
```

```python
    lr = lr0
    for it in range(n_steps):
        if stop_t is not None and time.monotonic() > stop_t:
            break
        b = _conv_full(a)
        M = float(b.max())
        if M < best_m:
            best_m = M
            best_a = a.copy()
        # Subgradient at argmax lags
        peak_thresh = M * 0.95
        peaks = np.where(b >= peak_thresh)[0]
        grad = np.zeros(n)
        for k in peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n, k + 1)
            for j in range(js, je):
                grad[j] += a[k - j]
        grad /= max(len(peaks[:8]), 1)
        # Projected step onto simplex
        a_new = a - lr * 2.0 * grad
        # Project onto {x>=0, sum(x)=1}
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u)
        rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
        if len(rho) == 0:
            a_new = a
        else:
            rho = rho[-1] + 1
            theta = (cssv[rho - 1] + 1) / rho
            a_new = np.maximum(a_new - theta, 0)
        a = a_new
        # Decay learning rate
        lr *= 0.998
        # Occasional perturbation
        if it % 200 == 199:
            a = _norm(np.clip(a + rng.normal(0, 0.01 * best_m, n), 0, 100))
    return best_a, best_m


def _coord_descent(a, n, stop_t, seed=777, max_rounds=50):
    """Fast coordinate descent with 1D search per coordinate."""
    a = np.clip(a.copy(), 0, 1000)
    a = _norm(a)
    best_sc = float(_conv_full(a).max())
    rng = np.random.default_rng(seed)
    no_imp = 0
    for rnd in range(max_rounds):
        if time.monotonic() > stop_t:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_t:
                break
            orig = a[i]
            osum = a.sum()
            # Try golden section on a[i] in [0, 5*avg]
            avg = osum / n
            hi = max(orig * 3.0, 5.0 * avg)
            if hi > 100:
                hi = 100
            lo = 0.0
            best_v = orig
            bm = best_sc
            # Coarse scan
            for tv in np.linspace(0.05 * hi, hi, 8):
                a[i] = tv
                m = float(_conv_full(a).max())
                if m < bm:
                    bm = m
                    best_v = tv
            if best_v != orig:
                # Fine search around best_v
                lo2 = max(0.0, best_v - 0.15 * hi)
                hi2 = min(hi, best_v + 0.15 * hi)
                for _ in range(5):
                    m1 = lo2 + 0.382 * (hi2 - lo2)
                    m2 = lo2 + 0.618 * (hi2 - lo2)
                    a[i] = m1
                    v1 = float(_conv_full(a).max())
                    a[i] = m2
                    v2 = float(_conv_full(a).max())
                    if v1 < v2:
                        hi2 = m2
                    else:
                        lo2 = m1
                    tv = (lo2 + hi2) / 2
                    a[i] = tv
                    tv_m = float(_conv_full(a).max())
                    if tv_m < bm:
                        bm = tv_m
                        best_v = tv
            a[i] = best_v
            a = _norm(a)
            m = float(_conv_full(a).max())
            if m < best_sc - 1e-14:
                best_sc = m
                improved = True
            else:
                a[i] = orig
                a = _norm(a)
        if not improved:
            no_imp += 1
            if no_imp >= 3:
                break
        else:
            no_imp = 0
    return a, best_sc


def _block_cd(a, n, stop_t, seed=999):
    """2-3 coordinate block descent."""
    a = np.clip(_norm(a.copy()), 0, 1000)
    best_m = float(_conv_full(a).max())
    rng = np.random.default_rng(seed)
    no_imp = 0
    while time.monotonic() < stop_t and no_imp < 8:
        improved = False
        n_pairs = min(300, 2 * n)
        for _ in range(n_pairs):
            if time.monotonic() > stop_t:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or abs(i - j) > n:
                continue
            oi, oj = a[i], a[j]
            tot = oi + oj
            if tot < 1e-12:
                continue
            for frac in np.linspace(0.05, 0.95, 7):
                a[i] = frac * tot
                a[j] = (1 - frac) * tot
                m = float(_conv_full(a).max())
                if m < best_m - 1e-14:
                    best_m = m
                    improved = True
                    break
        a[i], a[j] = oi, oj
        if not improved:
            no_imp += 1
        else:
            no_imp = 0
    return a, _norm(a)


def _lp_optimize(f0, n, stop_t, seed, use_focused=100):
    rng = np.random.default_rng(seed)
    a = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while a.size < n:
        a = np.concatenate([a, np.zeros(n - a.size)])
    a = a[:n]
    a = _norm(a)
    M = float(_conv_full(a).max())
    best_a = a.copy()
    best_M = M
    stall = 0
    it = 0
    prev_M = M
    while time.monotonic() < stop_t - 1.0:
        if M < prev_M - 1e-13:
            stall = max(0, stall - 2)
        else:
            stall += 1
        prev_M = M
        if stall > 60:
            # Perturb
            strat = stall % 4
            if strat == 0:
                a = np.clip(a + rng.normal(0, 0.05, n), 0, None)
            elif strat == 1:
                a = 0.6 * best_a + 0.4 * np.abs(rng.normal(1, 0.3, n))
            elif strat == 2:
                blk = max(1, n // 4)
                a2 = a.copy()
                a2[:blk], a2[-blk:] = a[-blk:].copy(), a[:blk].copy()
                a = _norm(np.clip(a2, 0, None))
            else:
                a = _norm(np.clip(a * (1 + 0.1 * rng.normal(0, 1, n)), 0, None))
            a = _norm(a)
            M = float(_conv_full(a).max())
            stall = 0
            continue
        # LP
        if use_focused > 0 and it % 3 != 0:
            g = _lp_dir(a, n, M, top_k=use_focused, tl=3.0)
        else:
            g = _lp_dir(a, n, M, top_k=-1, tl=4.0)
        if g is None:
            stall += 3
            a = _norm(np.clip(a + rng.normal(0, 0.03, n), 0, None))
            M = float(_conv_full(a).max())
            continue
        d = float(np.linalg.norm(g - a))
        if d < 1e-10:
            stall += 3
            a = _norm(np.clip(a + rng.normal(0, 0.05, n), 0, None))
            M = float(_conv_full(a).max())
            continue
        t_opt, m_opt = _line_search(a, g, n, M, stop_t)
        if t_opt > 0 and m_opt < M - 1e-14:
            a = _norm((1 - t_opt) * a + t_opt * g)
            M = float(_conv_full(a).max())
            if M < best_M:
                best_M = M
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
        it += 1
    return best_a, float(_score(best_a))


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 18.0

    # Get parent
    try:
        h = globals()["height_sequence_1"]
        parent = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(384)
    if parent.size == 0:
        parent = np.ones(384)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    n = len(parent)
    if n < 4:
        parent = np.ones(384)
        n = 384

    best_seq = parent.copy()
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        best_score = math.inf
    if not math.isfinite(best_score):
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))

    # Phase 1: LP optimization
    p1_end = t0 + budget * 0.60
    rng_main = np.random.default_rng(seed)

    inits = []
    inits.append(("parent", parent.copy()))
    i_arr = np.arange(n, dtype=np.float64)
    taper = 0.5 + 0.5 * np.cos(np.pi * i_arr / (n - 1))
    inits.append(("taper", taper))
    kern = np.exp(-0.5 * (np.arange(-4, 5) / 2.0) ** 2)
    kern /= kern.sum()
    smooth = np.convolve(rng_main.normal(0, 1, n), kern, 'same')
    inits.append(("perturbed_parent", np.clip(parent + 0.2 * smooth, 0, None)))
    # Multi-peak
    mp = 0.2 * np.ones(n)
    for fr in (2, 3, 5):
        ph = rng_main.uniform(0, 2 * math.pi)
        mp += 0.4 * np.abs(np.cos(2 * math.pi * fr * i_arr / n + ph))
    inits.append(("multipk", mp + 0.01))

    time_per_init = max(10.0, (p1_end - t0) / max(len(inits), 1))
    for idx, (name, init) in enumerate(inits):
        if time.monotonic() > p1_end - 8.0:
            break
        stop = min(p1_end - 4.0, time.monotonic() + time_per_init)
        if time.monotonic() >= stop:
            break
        try:
            res, sc = _lp_optimize(init, n, stop, seed + 7919 * (idx + 1),
                                   use_focused=120)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                rc = np.clip(res, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in rc]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = rc.copy()
                    best_score = exact
        except Exception:
            pass

    # Phase 2: Subgradient refinement
    p2_end = t0 + budget * 0.80
    if time.monotonic() < p2_end - 5.0:
        try:
            s_stop = min(p2_end - 3.0, time.monotonic() + (p2_end - p1_end) * 0.7)
            res, _ = _subgrad_steps(best_seq, n, n_steps=5000,
                                    lr0=0.015, stop_t=s_stop)
            res = np.clip(res, 0, 1000)
            exact = float(evaluate_sequence([float(x) for x in res]))
            if math.isfinite(exact) and exact < best_score:
                best_seq = res.copy()
                best_score = exact
        except Exception:
            pass

    # Phase 3: Coordinate descent polish
    p3_end = deadline - 8.0
    acd_stop = min(p3_end, time.monotonic() + (p3_end - max(p2_end, time.monotonic())) * 0.6)
    if time.monotonic() < acd_stop - 3.0:
        try:
            res, _ = _coord_descent(best_seq, n, acd_stop, seed=5555)
            res = np.clip(res, 0, 1000)
            exact = float(evaluate_sequence([float(x) for x in res]))
            if math.isfinite(exact) and exact < best_score:
                best_seq = res.copy()
                best_score = exact
        except Exception:
            pass

    # Final short LP from best
    if time.monotonic() < p3_end - 10.0:
        try:
            stop_lp = time.monotonic() + max(5.0, (p3_end - time.monotonic()) * 0.5)
            res, sc = _lp_optimize(best_seq, n, stop_lp, seed=99999, use_focused=80)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                rc = np.clip(res, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in rc]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = rc.copy()
                    best_score = exact
        except Exception:
            pass

    # Final coordinate polish
    if time.monotonic() < deadline - 5.0:
        try:
            res, _ = _coord_descent(best_seq, n, deadline - 5.0, seed=77777,
                                    max_rounds=30)
            res = np.clip(res, 0, 1000)
            exact = float(evaluate_sequence([float(x) for x in res]))
            if math.isfinite(exact) and exact < best_score:
                best_seq = res.copy()
                best_score = exact
        except Exception:
            pass

    # Final validation
    final_seq = np.clip(best_seq, 0, 1000)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf
    if not math.isfinite(final_sc):
        return [1.0]
    return [float(x) for x in final_seq]
```
