def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def _lp_direction(f, n, b, M, peak_frac=0.1, timeout=5.0):
    """Find LP direction to maximize sum(g) s.t. (f*g)_k <= M for peak k."""
    threshold = M * (1.0 - peak_frac)
    peak_idx = np.where(b >= threshold)[0]
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(b)])
    
    # Build constraint matrix: (f*g)_k = sum_i f_i * g_{k-i}
    # For each peak lag k, the constraint is sum_{valid i} f_{k-i} * g_i <= M
    # Note: (f*g)_k = sum_i f_i * g_{k-i} = sum_i f_{k-i} * g_i
    # So the coefficient of g_i in constraint k is f_{k-i} (when 0 <= k-i < n)
    
    A = np.zeros((len(peak_idx), n))
    for r, k in enumerate(peak_idx):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            A[r, i] = f[k - i]
    
    c_vec = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(len(peak_idx), M)
    bounds = [(0.0, None)] * n
    
    try:

                      method='highs', options={'time_limit': timeout})
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

                t = 0.2
                stall = 0

            if f_new.sum() < 0.01:

            sc = _score(f_new)
            if sc < best_score - 1e-13:

                best_score = sc
