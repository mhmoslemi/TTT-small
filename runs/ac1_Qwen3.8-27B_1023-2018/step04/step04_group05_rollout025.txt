    M = b.max()
    if M <= 0:
        return np.array([b.argmax()])
    return np.where(b >= M * (1.0 - tol_frac))[0]


def _subgrad(f, active, n, lam=None):
    """
    Compute subgradient of max_k (f*f)_k at point f.
    Returns the gradient vector (not projected).
    """
    if lam is None:
        lam = np.ones(len(active)) / len(active)
    g = np.zeros(n)
    for k, lk in zip(active, lam):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            # g[j] += 2 * lk * f[k-j] for j in [j_lo, j_hi]
            # f indices: k-j_lo, k-j_lo-1, ..., k-j_hi
            f_idx = np.arange(k - j_lo, k - j_hi - 1, -1)
            g[j_lo:j_hi+1] += 2.0 * lk * f[f_idx]
    return g
