    fs = best.sum()

        return best, _score(best)

    best_sc = _score(best)
    rng = np.random.default_rng(seed)

    max_no_imp = 15

        round_num += 1
        improved = False

        c = _conv_full(best)
        M = float(c.max())
        peak_positions = np.where(c >= M * 0.999)[0]

        # For peak at position k, coordinate i affects it via a[k-i] (gradient = 2*a[k-i] if in range)
        coord_importance = np.zeros(n)
        for k in peak_positions:

                if 0 <= k - i < n:
                    coord_importance[i] += best[k - i]

        return None
    return g


def _lp_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n)
        f_sum = n

    target_sum = math.sqrt(2.0 * n)
    f = f / f_sum * target_sum

    best_f = f.copy()
    best_score = _score(f)
    best_g = f.copy()

    t = 0.25
    t_min = 0.01
    t_max = 1.0
    stall = 0
    max_stall = 200
    iteration = 0

    while time.monotonic() < stop_time - 2.0:
        iteration += 1
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, n, M, x0=best_g)
        if g is None:
            stall += 5
            if stall > max_stall:
                f = best_f + rng.normal(0, 0.04 * max(1.0, best_f.mean()), n)
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                else:
                    f = best_f.copy()
                stall = 0
                max_stall += 80
                t = 0.25
            continue

        g_sum = float(g.sum())
        if g_sum < 1e-15:
            stall += 2
            continue
        g = g * (target_sum / g_sum)
        best_g = g.copy()

        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < 1e-12:
            stall += 3
            if stall > max_stall:
                f = best_f.copy()
                idx = rng.choice(n, size=min(15, n), replace=False)
                for k in idx:
                    f[k] = max(0.0, f[k] * rng.uniform(0.3, 2.5))
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                stall = 0
                max_stall += 60
                t = 0.25
            continue

        improved = False
        candidates = sorted(
            {max(t_min, min(t_max, c)) for c in
             [t, t * 0.6, t * 0.8, t * 1.2, t * 1.5, 0.05, 0.1, 0.2, 0.4, 0.7, 1.0]},
            reverse=True
        )
        for tc in candidates:
            f_new = (1.0 - tc) * f + tc * g
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 0.01:
                continue
            f_new = f_new / fs_new * target_sum
            sc = _score(f_new)
            if sc < best_score - 1e-14:
                f = f_new.copy()
                best_f = f.copy()
                best_score = sc
                improved = True
                t = min(t_max, tc * 1.3)
                stall = 0
                break

        if not improved:
            stall += 1
            t = max(t_min, t * 0.55)
            if stall >= max_stall:
                f = best_f.copy()
                strength = 0.04 + 0.01 * min(stall // max_stall, 10)
                f += rng.normal(0, strength * max(1.0, best_f.mean()), n)
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                else:
                    f = best_f.copy()
                stall = 0
                max_stall += 60
                t = 0.25

    return best_f, best_score


def _gradient_polish_step(a, n):
    """Compute gradient of score w.r.t. each coordinate (for active peak)."""
    c = _conv_full(a)
    M = float(c.max())
    S = float(a.sum())
    if S < 0.01 or M < 1e-30:
        return np.zeros(n), c, M
    # Find active peaks (within 0.1% of max)
    thresh = M * 0.999
    active_peaks = np.where(c >= thresh)[0]
    
    # Gradient: d(score)/da[i] = 4n/S * (a[k*-i] - M/S) for active peak k*
    # For multiple peaks, average
    grad = np.zeros(n)
    for k_ in active_peaks:
        for i in range(n):
            diff = k_ - i
            if 0 <= diff < n:
                grad[i] += a[diff]
    if len(active_peaks) > 0:
        grad /= len(active_peaks)
        grad = grad - M / S
    return grad, c, M


def _polish(a, stop_time, seed=4242):
    n = len(a)
    target_sum = math.sqrt(2.0 * n)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)
    best = best / fs * target_sum
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 12
    round_num = 0

    while time.monotonic() < stop_time - 2.0 and no_imp < max_no_imp:
        round_num += 1
        improved = False

        # Get gradient info for ordering
        try:
            grad, c_arr, M_val = _gradient_polish_step(best, n)
        except Exception:
            grad = np.zeros(n)
        
        # Process coordinates sorted by |gradient| (most promising first)
        order = np.argsort(-np.abs(grad))

        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc

            if orig < 1e-14:
                for nv in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0):
                    best[i] = nv
                    if best.sum() < 0.01:
                        best[i] = orig
                        continue
                    sc = _score(best)
                    if sc < best_val_sc - 1e-14:
                        best_val = nv
                        best_val_sc = sc
            else:
                # Grid search around current value
                multipliers = [0.0, 0.1, 0.2, 0.35, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0, 3.0]
                for mlt in multipliers:
                    nv = orig * mlt if mlt > 0 else 0.0
                    best[i] = nv
                    if best.sum() < 0.01:
                        best[i] = orig
                        continue
                    sc = _score(best)
                    if sc < best_val_sc - 1e-14:
                        best_val = nv
                        best_val_sc = sc

                # Fine refinement
                if best_val > 0:
                    for frac in (0.88, 0.93, 0.97, 1.03, 1.07, 1.12):
                        nv = best_val * frac
                        best[i] = nv
                        if best.sum() < 0.01:
                            best[i] = best_val
                            continue
                        sc = _score(best)
                        if sc < best_val_sc - 1e-14:
                            best_val = nv
                            best_val_sc = sc

            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True

        # Pair updates
        n_pairs = 200 if round_num < 3 else 80
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 2.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            if oi < 1e-14 and oj < 1e-14:
                continue
            local_best = best_sc
            found = False
            for mi in (0.0, 0.4, 0.7, 1.3, 1.7):
                for mj in (0.0, 0.4, 0.7, 1.3, 1.7):
                    ni = oi * mi if (oi > 1e-14 and mi > 0) else 0.0
                    nj = oj * mj if (oj > 1e-14 and mj > 0) else 0.0
                    best[i], best[j] = ni, nj
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < local_best - 1e-14:
                        local_best = sc
                        found = True
            if found:
                best_sc = local_best
                improved = True
            else:
                best[i], best[j] = oi, oj

        # Small random kicks
        if not improved:
            for _ in range(40):
                if time.monotonic() > stop_time - 2.0:
                    break
                k = int(rng.integers(n))
                old = best[k]
                sigma = 0.015 * max(1.0, best.mean())
                best[k] = max(0.0, old + rng.normal(0, sigma))
                if best.sum() < 0.01:
                    best[k] = old
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                else:
                    best[k] = old

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc
