```python
def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
    """Project v onto the simplex {x : sum(x)=1, x>=0}."""
```

```python
        if lo <= hi:

    g /= len(S)
    return g


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _grad_polish(f, stop_time, max_iter=5000):
    n = len(f)
    f = f.copy()
    s = f.sum()
    f = f / s
    cur_score = _score(f)
    best_f = f.copy()
    best_score = cur_score
    lr = 0.5
    stagnant = 0

    for it in range(max_iter):
        if time.monotonic() > stop_time - 2.0:
            break
        q = _conv_full(f)
        g = _subgradient(f, q)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm

        improved = False
        for t in [lr, lr * 0.5, lr * 2.0, lr * 0.25, lr * 0.75]:
            f_new = _proj_simplex(f - t * g)
            sc = _score(f_new)
            if sc < cur_score - 1e-14:
                f = f_new
                cur_score = sc
                if sc < best_score:
                    best_score = sc
                    best_f = f.copy()
                    stagnant = 0
                else:
                    stagnant += 1
                improved = True
                lr = min(2.0, t * 1.3)
                break
        if not improved:
            lr *= 0.5
            stagnant += 1
            if lr < 1e-8 or stagnant > 200:
                break

    return best_f * s, best_score


def _cd_polish(f, stop_time, seed=0):
    n = len(f)
    rng = np.random.default_rng(seed)
    f = f.copy()
    cur = _score(f)
    best = f.copy()
    best_sc = cur
    mults = [0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 0.0]

    for outer in range(500):
        if time.monotonic() > stop_time - 2.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = f[i]
            best_v = orig
            best_local = cur
            for m in mults:
                if m == 1.0:
                    continue
                v = max(0.0, orig * m) if orig > 1e-12 else (0.3 * f.mean() if m >= 1.0 else 0.0)
                f[i] = v
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_local:
                    best_local = sc
                    best_v = v
            f[i] = best_v
            if best_local < cur - 1e-15:
                cur = best_local
                improved = True
                if cur < best_sc:
                    best_sc = cur
                    best = f.copy()
        if not improved:
            if outer >= 3:
                break
            # 2-coordinate moves
            for _ in range(100):
                if time.monotonic() > stop_time - 2.0:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                oi, oj = f[i], f[j]
                bi, bj = oi, oj
                bsc = cur
                for mi in [0.5, 0.8, 1.2, 1.5]:
                    for mj in [0.5, 0.8, 1.2, 1.5]:
                        fi = max(0.0, oi * mi) if oi > 1e-15 else 0.0
                        fj = max(0.0, oj * mj) if oj > 1e-15 else 0.0
                        f[i], f[j] = fi, fj
                        if f.sum() < 0.01:
                            f[i], f[j] = oi, oj
                            continue
                        sc = _score(f)
                        if sc < bsc:
                            bsc = sc
                            bi, bj = fi, fj
                f[i], f[j] = bi, bj
                if bsc < cur - 1e-15:
                    cur = bsc
                    best_sc = min(best_sc, cur)
                    if cur <= best_sc + 1e-15:
                        best = f.copy()
                        best_sc = cur
    return best, best_sc


def _run_lp_loop(f0, n, stop_time, seed, update_fn):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    if f.sum() < 1e-10:
        f = np.ones(n)

    best_f = f.copy()
    best_score = _score(f)
    t = 0.3
    stall = 0
    max_stall = 150
    it = 0

    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, n, M, sum_f=f.sum())

        if g is None:
            stall += 4
            if stall > max_stall:
                f = np.clip(best_f + rng.normal(0, 0.03 * max(1.0, np.mean(best_f)), n), 0, None)
                if f.sum() < 1e-10:
                    f = best_f.copy()
                t = 0.3
                stall = 0
                max_stall = min(max_stall + 80, 400)
            continue

        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < 1e-12:
            stall += 3
            if stall > max_stall:
                f = np.clip(best_f + rng.normal(0, 0.04 * max(1.0, np.mean(best_f)), n), 0, None)
                if f.sum() < 1e-10:
                    f = best_f.copy()
                t = 0.3
                stall = 0
            continue

        improved = False
        t_candidates = [t, t * 0.5, t * 0.25, t * 1.5, t * 0.75, t * 2.0]
        best_t = 0.0
        best_sc = _score(f)
        best_f_try = None

        for ti in t_candidates:
            f_new = np.clip(f + ti * (g - f), 0, None)
            if f_new.sum() < 1e-10:
                continue
            sc = _score(f_new)
            if sc < best_sc:
                best_sc = sc
                best_t = ti
                best_f_try = f_new

        if best_f_try is not None:
            if best_sc < best_score - 1e-14:
                f = best_f_try
                best_f = f.copy()
                best_score = best_sc
                t = min(0.9, best_t * 1.15)
                stall = 0
                improved = True
            elif best_sc < _score(f) - 1e-14:
                f = best_f_try
                t = min(0.9, best_t)
                stall = 0
                improved = True
            else:
                t = max(0.02, t * 0.5)

        if not improved:
            stall += 1
            if stall >= max_stall:
                f = np.clip(best_f + rng.normal(0, 0.05 * max(1.0, np.mean(best_f)), n), 0, None)
                if f.sum() < 1e-10:
                    f = best_f.copy()
                t = 0.3
                stall = 0
                max_stall = min(max_stall + 100, 600)

        if it % 25 == 24 and time.monotonic() < stop_time - 12.0:
            cd_f, cd_sc = _cd_polish(f, min(stop_time - 6.0, time.monotonic() + 6.0), seed=seed + it)
            if cd_sc < best_score - 1e-14:
                best_score = cd_sc
                best_f = cd_f.copy()
                f = cd_f.copy()

        update_fn(best_f, best_score)
        it += 1

    update_fn(best_f, best_score)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(400, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(400, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    n = len(parent)

    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(200)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_update(seq, sc):
        nonlocal best_seq, best_score
        if seq is None or len(seq) == 0 or not np.all(np.isfinite(seq)):
            return
        seq = np.clip(seq, 0, 1000)
        if seq.sum() < 0.01:
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-14:
            best_seq = seq.copy()
            best_score = exact

    # Phase 1: LP optimization with diverse inits (55%)
    phase1_end = t0 + budget * 0.55
    rng = np.random.default_rng(seed)

    i_arr = np.arange(n, dtype=np.float64)
    inits = [parent.copy()]
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / n)))
    inits.append(np.maximum(0.0, 1.0 - (2.0 * i_arr / n - 1.0) ** 2))
    inits.append(np.exp(-0.5 * ((i_arr - 0.5 * n) / (0.15 * n)) ** 2))
    sig = 0.15 * max(1.0, np.mean(parent))
    inits.append(np.abs(parent + rng.normal(0, sig, n)) + 1e-4)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.15)
    inits.append(alt * (0.5 + 0.5 * np.cos(2.0 * np.pi * i_arr / n)))
    inits.append(np.maximum(0, np.cos(np.pi * i_arr / n)))

    inits_dedup = []
    seen = set()
    for ix in inits:
        ix = np.clip(ix, 0, None)
        if ix.sum() < 1e-10:
            continue
        key = ix.round(3).tobytes()
        if key not in seen:
            seen.add(key)
            inits_dedup.append(ix)

    time_per_init = max(12.0, (phase1_end - t0 - 8.0) / len(inits_dedup))
    for idx, init in enumerate(inits_dedup):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        _run_lp_loop(init, n, stop, seed + 100 * (idx + 1), _try_update)

    # Phase 2: Gradient polish (15%)
    phase2_end = t0 + budget * 0.70
    if time.monotonic() < phase2_end - 12.0:
        gd_f, gd_sc = _grad_polish(best_seq, phase2_end - 5.0, max_iter=4000)
        _try_update(gd_f, gd_sc)

    # Phase 3: LP warm restart (15%)
    phase3_end = t0 + budget * 0.85
    if time.monotonic() < phase3_end - 12.0:
        _run_lp_loop(best_seq.copy(), len(best_seq), phase3_end - 5.0, seed + 77777, _try_update)

    # Phase 4: CD polish (10%)
    phase4_end = deadline - 5.0
    if time.monotonic() < phase4_end - 5.0:
        cd_f, cd_sc = _cd_polish(best_seq, phase4_end, seed=seed * 7 + 42)
        _try_update(cd_f, cd_sc)

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0, 1.0, 1.0]
        return [float(x) for x in fallback]

    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    if not result or any(math.isnan(x) or math.isinf(x) for x in result):
        return [float(x) for x in fallback]
    return result
```
