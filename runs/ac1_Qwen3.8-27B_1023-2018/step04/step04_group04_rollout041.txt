import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _fft_conv2(a, b):
    n = len(a)
    m = len(b)
    fsize = next_fast_len(n + m - 1)
    FA = rfft(a, fsize)
    FB = rfft(b, fsize)
    c = irfft(FA * FB, fsize)[:n + m - 1]
    return c


def _full_score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, n, M, time_limit=8.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _line_search_fast(f, g, f2, fg, g2):
    """Fast line search: minimize max_k [(1-t)^2 f2_k + 2t(1-t) fg_k + t^2 g2_k]
    assuming sum(f)=sum(g)=1 so denominator is always 1."""
    n = len(f)
    sf = f.sum()
    sg = g.sum()
    best_t = 0.0
    best_score = 2.0 * n * f2.max() / (sf * sf)

    for t in np.linspace(0.01, 1.0, 50):
        denom = (1 - t) * sf + t * sg
        if denom < 1e-10:
            continue
        val = (1 - t) * (1 - t) * f2 + 2.0 * t * (1 - t) * fg + t * t * g2
        peak = float(val.max())
        sc = 2.0 * n * peak / (denom * denom)
        if sc < best_score - 1e-14:
            best_score = sc
            best_t = t

    if best_t > 0:
        step = 0.025
        for _ in range(5):
            improved = False
            for dt in (-step, step):
                t_try = best_t + dt
                if t_try < 0.001 or t_try > 1.0:
                    continue
                denom = (1 - t_try) * sf + t_try * sg
                if denom < 1e-10:
                    continue
                val = (1 - t_try)**2 * f2 + 2.0 * t_try * (1 - t_try) * fg + t_try**2 * g2
                peak = float(val.max())
                sc = 2.0 * n * peak / (denom * denom)
                if sc < best_score - 1e-15:
                    best_score = sc
                    best_t = t_try
                    improved = True
            step *= 0.4
            if not improved:
                break

    return best_t, best_score


def _lp_iteration(f0, n, stop_time, max_iters=5000):
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s

    f2 = _fft_conv(f)
    best_f = f.copy()
    best_score = 2.0 * n * float(f2.max())
    stall = 0
    rng = np.random.default_rng(n * 7 + 13)

    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break

        M = float(f2.max())
        if M < 1e-18:
            break

        g = _lp_solve(f, n, M, time_limit=8.0)
        if g is None:
            stall += 5
            if stall > 120:
                # Perturb to escape
                f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
                f = f / f.sum()
                f2 = _fft_conv(f)
                stall = 0
                if stall > 200:
                    break
            continue

        g = g / g.sum()

        dist = np.linalg.norm(g - f)
        if dist < 1e-8:
            break

        fg = _fft_conv2(f, g)
        g2 = _fft_conv(g)

        best_t, _ = _line_search_fast(f, g, f2, fg, g2)

        if best_t > 0.001:
            f = np.clip((1 - best_t) * f + best_t * g, 0, None)
            fs = f.sum()
            if fs < 1e-10:
                break
            f = f / fs
            f2 = _fft_conv(f)
            sc = 2.0 * n * float(f2.max())
            if sc < best_score - 1e-13:
                best_score = sc
                best_f = f.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break

    return best_f, best_score


def _coord_descent(f0, stop_time, seed, max_sweeps=300):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    n = len(f)
    best_f = f.copy()
    best_sc = _full_score(f)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]

            if orig < 1e-15:
                best_v = 0.0
                best_v_sc = best_sc
                for v in (0.3 / n, 1.0 / n, 2.0 / n, 5.0 / n, 10.0 / n):
                    f[i] = v
                    sc = _full_score(f)
                    if sc < best_v_sc - 1e-15:
                        best_v_sc = sc
                        best_v = v
                f[i] = best_v
                if best_v_sc < best_sc - 1e-14:
                    best_sc = best_v_sc
                    best_f = f.copy()
                    improved = True
                continue

            best_v = orig
            best_v_sc = best_sc
            for mult in (0.0, 0.1, 0.25, 0.5, 0.75, 0.9, 1.1, 1.3, 1.6, 2.0, 3.0):
                f[i] = orig * mult
                sc = _full_score(f)
                if sc < best_v_sc - 1e-15:
                    best_v_sc = sc
                    best_v = orig * mult
            f[i] = best_v
            if best_v_sc < best_sc - 1e-14:
                best_sc = best_v_sc
                best_f = f.copy()
                improved = True

        if not improved and sweep > 10:
            break

    return best_f, best_sc


def _two_coord_descent(f0, stop_time, seed, max_rounds=500):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    n = len(f)
    best_f = f.copy()
    best_sc = _full_score(f)
    no_imp = 0

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        oi, oj = f[i], f[j]
        if oi < 1e-15 and oj < 1e-15:
            continue

        found = False
        for mi in (0.0, 0.3, 0.6, 0.9, 1.3, 1.7, 2.5):
            for mj in (0.0, 0.3, 0.6, 0.9, 1.3, 1.7, 2.5):
                if mi == oj / oj if oj > 0 else False:
                    pass
                vi = oi * mi if oi > 1e-15 else 0.0
                vj = oj * mj if oj > 1e-15 else 0.0
                f[i], f[j] = vi, vj
                sc = _full_score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_f = f.copy()
                    found = True
                    break
            if found:
                break
        if not found:
            f[i], f[j] = oi, oj
            no_imp += 1
        else:
            no_imp = 0

        if no_imp > 200:
            break

    return best_f, best_sc
