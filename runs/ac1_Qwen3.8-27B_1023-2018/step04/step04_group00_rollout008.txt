def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    """Full self-convolution using FFT for large n."""
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_arr(a):
    """Compute AC1 score = 2*n*max_b/(sum_a)^2."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = c.max()
    return float(2.0 * n * m / (s * s))


def _resample(src, n):
    """Resample src to length n using midpoint interpolation, normalized to sum=1."""
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.full(n, 1.0/n)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m) + 0.5) / m
        fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.full(n, 1.0/n)
    return out / s
