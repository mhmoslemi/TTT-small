def _subgrad_step(f, n, step_size):
    """One subgradient step for the normalized problem"""
    b = _fft_conv(f)
    m = float(b.max())
    eps = max(1e-15, 1e-10 * m)
    active = np.flatnonzero(b >= m - eps)
    
    # Subgradient of max_k (f*f)_k w.r.t. f
    grad = np.zeros(n)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            # (f*f)_k = sum_j f[j] * f[k-j], d/d f[i] = f[k-i] + f[i] (if k-i != i)
            # Actually d/d f[i] of sum_j f[j]f[k-j] = f[k-i] (from term j=i) + f[i] (from term k-j=i, i.e., j=k-i)
            # If k-i == i, i.e., 2i=k, then it's 2*f[i]
            j = k - i
            if 0 <= j < n:
                if j == i:
                    grad[i] += 2.0 * f[j]
                else:
                    grad[i] += f[j]
    if len(active) > 0:
        grad /= len(active)
    
    # Subgradient of log(sum f) term: we're minimizing 2n*max/(sum)^2
    # d/d f_i [max/(sum)^2] = (grad_i * sum - 2*max * 1) / sum^3
    S = float(np.sum(f))
    f_sub = 2.0 * n * (grad * S - 2.0 * m) / (S**3)
    
    # Normalize to unit sum for the step

import numpy as np

import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    """Sparse convolution matrix: (A @ g)_k = (f*g)_k"""
    idx_n = np.arange(n)
    rows = np.concatenate([idx_n + j for j in range(n)])
    cols = np.repeat(idx_n, n)
    vals = np.tile(f, n)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_solve(f, n, M, time_limit=2.5):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True,
                               'dual_feasibility_tolerance': 1e-8})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    gs = g.sum()
    if gs < 1e-14:
        return None
    return g / gs

def _subgrad_dir(f, n):
    """Compute normalized subgradient of the objective"""
    b = _fft_conv(f)
    m = float(b.max())
    eps = max(1e-14, 1e-9 * m)
    active = np.flatnonzero(b >= m - eps)
    if len(active) == 0:
        return np.zeros(n), m

    grad = np.zeros(n)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            j = k - i
            if 0 <= j < n:
                grad[i] += (2.0 * f[j] if j == i else f[j])
    grad /= len(active)

    S = float(np.sum(f))
    f_sub = 2.0 * n * (grad * S - 2.0 * m) / (S ** 3)
    f_sub -= np.mean(f_sub)
    return f_sub, m

            stop3 = min(t0 + budget * 0.52, time.monotonic() + 30)
            res3, sc3, _ = _lp_optimize(p3, len(p3), stop3, seed + 100, lp_timeout=1.5)
            if math.isfinite(sc3):
                _update(res3 * best_seq.sum(), sc3)

    # === Phase 3: Subgradient refinement (12%) ===
    if time.monotonic() < t0 + budget * 0.64:
        f_sg = best_seq / max(best_seq.sum(), 1e-10)
        stop3 = t0 + budget * 0.64
        res_sg, sc_sg = _subgrad_optimize(f_sg, len(f_sg), stop3, seed + 200)
        if math.isfinite(sc_sg):
            _update(res_sg * best_seq.sum(), sc_sg)

    # === Phase 4: Coordinate descent polish (18%) ===
    if time.monotonic() < t0 + budget * 0.82:
        stop4 = t0 + budget * 0.82
        res_cd, sc_cd = _coord_descent(best_seq, stop4, seed + 400, max_sweeps=400)
        if math.isfinite(sc_cd):
            _update(res_cd, sc_cd)

    # === Phase 5: Try different lengths (12%) ===
    phase5_start = time.monotonic()
    phase5_end = deadline - 20
    if phase5_end > phase5_start + 30:
        alt_lengths = []
        for L in [n_parent - 6, n_parent - 4, n_parent - 2, n_parent + 2,
                  n_parent + 4, n_parent + 6, 256, 320, 512, 384, 448]:
            if L >= 8 and L != n_parent:
                alt_lengths.append(L)
        alt_lengths = sorted(set(alt_lengths))
        n_alts = len(alt_lengths)
        time_per = max(10.0, (phase5_end - phase5_start) / n_alts)

        for L in alt_lengths:
            if time.monotonic() > phase5_end - 10:
                break
            init_L = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n_parent), parent)
            init_L = init_L / max(init_L.sum(), 1e-10)
            stop_L = min(phase5_end, time.monotonic() + time_per)
            try:
                rL, sL, _ = _lp_optimize(init_L, L, stop_L, seed + 2000 + L,
                                         lp_timeout=1.0)
                if math.isfinite(sL):
                    _update(rL * parent.sum(), sL)
            except Exception:
                pass

    # === Phase 6: Final polish (remaining) ===
    final_stop = deadline - 5
    if time.monotonic() < final_stop:
        res_fin, sc_fin = _coord_descent(best_seq, final_stop, seed + 9000, max_sweeps=500)
        if math.isfinite(sc_fin):
            _update(res_fin, sc_fin)

        # Two-coordinate refinement
        if time.monotonic() < final_stop - 5:
            rng_fin = np.random.default_rng(seed + 9999)
            nf, sc_nf = _two_coord(best_seq, len(best_seq), final_stop - 5,
                                   rng_fin, n_pairs=500)
            if math.isfinite(sc_nf):
                _update(nf, sc_nf)

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
