    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method="highs",
                      options={"presolve": True, "time_limit": timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-14:
        return None
    return g


def _lp_max_sum(f, n, M, timeout=4.0):
    """LP: max sum(g) s.t. (f*g)_k <= M, g>=0. Returns g or None."""
    if M <= 1e-18:
        return None
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-14)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"presolve": True, "time_limit": timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-14:
        return None
    return g


def _lp_iterate(f0, n, stop_time, seed, use_min_peak=True):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n)
        f_sum = n
    target_sum = math.sqrt(2.0 * n)
    f = f / f_sum * target_sum

    best_f = f.copy()
    best_score = _score(f)
    t = 0.35
    t_min, t_max = 0.03, 1.0
    stall = 0
    max_stall = 150
    conv_thresh = 5e-10

    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break
        S = float(f.sum())

        if use_min_peak:
            g = _lp_min_peak(f, n, S)
        else:
            g = _lp_max_sum(f, n, M)

        if g is None:
            stall += 4
            if stall > max_stall:
                f = best_f * (1.0 + rng.normal(0, 0.08, n))
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                else:
                    f = best_f.copy()
                stall = 0
                max_stall += 60
                t = 0.35
            continue

        g_sum = float(g.sum())
        if g_sum < 1e-14:
            stall += 2
            continue
        g = g / g_sum * target_sum

        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < conv_thresh:
            stall += 3
            if stall > max_stall:
                f = best_f.copy()
                # Structural perturbation
                strategy = stall % 4
                if strategy == 0:
                    sigma = 0.1
                    f += rng.normal(0, sigma * max(1.0, best_f.mean()), n)
                elif strategy == 1:
                    idx = rng.permutation(n)[:n // 4]
                    f[idx] *= rng.uniform(0.3, 3.0, len(idx))
                elif strategy == 2:
                    f = f[::-1].copy()
                else:
                    f = best_f * (0.5 + rng.uniform(0.2, 1.3, n))
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                else:
                    f = best_f.copy()
                stall = 0
                max_stall += 60
                t = 0.35
            continue

        improved = False
        candidates = [t, t * 0.5, t * 1.5, 0.15, 0.3, 0.5, 0.7, 1.0]
        candidates = sorted({max(t_min, min(t_max, c)) for c in candidates}, reverse=True)
        for tc in candidates:
            if time.monotonic() > stop_time - 3.0:
                break
            f_new = (1.0 - tc) * f + tc * g
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 0.01:
                continue
            f_new = f_new / fs_new * target_sum
            sc = _score(f_new)
            if sc < best_score - 1e-13:
                f = f_new.copy()
                best_f = f.copy()
                best_score = sc
                improved = True
                t = min(t_max, tc * 1.3)
                stall = 0
                break

        if not improved:
            t = max(t_min, t * 0.6)
            stall += 1
            if stall >= max_stall:
                f = best_f.copy()
                f += rng.normal(0, 0.05 * max(1.0, best_f.mean()), n)
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                stall = 0
                t = 0.35

    return best_f, best_score


def _coord_descent(a, stop_time, seed=0, max_rounds=50):
    """Coordinate descent with analytical convolution updates."""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    target_sum = math.sqrt(2.0 * n)
    best = best / fs * target_sum
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 3.0:
            break
        improved_round = False
        order = rng.permutation(n)
        b = _conv_full(best)  # current convolution
        S = float(best.sum())

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            old_val = best[i]
            if old_val < 1e-16 and b[i] > 1e-16:
                # still try setting it
                pass

            # The affected k range: [i, i+n-1]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            # Find max outside affected range
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                outer_max = float(b[mask].max()) if mask.any() else 0.0
            else:
                outer_max = 0.0

            f_affected = best[k_lo - i:k_hi - i + 1]  # f_{k-i} for k in [k_lo, k_hi]
            b_affected = b[k_lo:k_hi + 1].copy()

            def _eval_coord(x):
                new_b_aff = b_affected.copy()
                # for k in [i, i+n-1]: new_b[k] = b[k] + (x - old_val)*f_{k-i}
                # but if k = 2i, it's x^2 instead of 2*x*f_i... 
                # Actually: b[k] = sum_j f_j * f_{k-j}
                # The terms with f_i: j=i gives f_i*f_{k-i}, j=k-i gives f_{k-i}*f_i
                # If k-i != i: contribution is 2*f_i*f_{k-i}, change is 2*(x-old)*f_{k-i}
                # If k-i == i (k=2i): contribution is f_i^2, change is x^2 - old^2
                new_b_aff = b_affected + 2.0 * (x - old_val) * f_affected
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    idx_2i = 2 * i - k_lo
                    new_b_aff[idx_2i] = b_affected[idx_2i] - old_val * old_val + x * x
                new_max = max(outer_max, float(new_b_aff.max()))
                new_S = S - old_val + x
                if new_S < 0.01:
                    return math.inf
                return 2.0 * n * new_max / (new_S * new_S)

            # Golden section search on [0, 5*old_val] or [0, 3*mean]
            lo = 0.0
            hr = max(1.0, best.mean())
            hi = max(5.0 * old_val if old_val > 0 else 0.0, 2.0 * hr)
            if hi < 1e-3:
                hi = 0.5 * hr

            gr = (math.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = _eval_coord(c1)
            f2 = _eval_coord(c2)
            for _ in range(30):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = _eval_coord(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = _eval_coord(c2)
                if hi - lo < 1e-10 * max(1.0, hi):
                    break
            x_opt = (lo + hi) / 2.0
            f_opt = _eval_coord(x_opt)
            # Also try x=0
            f_zero = _eval_coord(0.0)
            if f_zero < f_opt:
                x_opt = 0.0
                f_opt = f_zero

            if f_opt < best_sc - 1e-14:
                best[i] = x_opt
                best_sc = f_opt
                S = float(best.sum())
                b = _conv_full(best)
                improved_round = True

        if not improved_round:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

    return best, best_sc


def _polish_two_coord(a, stop_time, seed=0):
    """Two-coordinate descent for additional refinement."""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    S = float(best.sum())

    for _ in range(200):
        if time.monotonic() > stop_time - 3.0:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        oi, oj = best[i], best[j]
        if oi < 1e-14 and oj < 1e-14:
            continue
        best_found = (oi, oj, best_sc)
        # Try a grid of perturbations
        for di in (0.0, 0.5, 1.5, 2.5, -0.5):
            for dj in (0.0, 0.5, 1.5, 2.5, -0.5):
                if di == 0.0 and dj == 0.0:
                    continue
                ni = max(0.0, oi * di) if di > 0 else 0.0
                nj = max(0.0, oj * dj) if dj > 0 else 0.0
                best[i], best[j] = ni, nj
                if best.sum() < 0.01:
                    best[i], best[j] = oi, oj
                    continue
                sc = _score(best)
                if sc < best_found[2] - 1e-14:
                    best_found = (ni, nj, sc)
        best[i], best[j] = best_found[0], best_found[1]
        best_sc = best_found[2]

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not math.isfinite(best_score):
        best_seq = np.ones(100)
        best_score = _score(best_seq)

    def _try_update(seq, sc):
        nonlocal best_seq, best_score
        if seq is None or len(seq) == 0:
            return
        if not np.all(np.isfinite(seq)):
            return
        seq = np.clip(seq, 0.0, 1000.0)
        if seq.sum() < 0.01:
            return
        if sc < best_score - 1e-14:
            try:
                exact = float(evaluate_sequence([float(x) for x in seq]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-15:
                best_seq = seq.copy()
                best_score = exact

    # Build diverse initializations
    rng_main = np.random.default_rng(seed)
    n = n_p
    i_arr = np.arange(n, dtype=np.float64)

    inits = [parent.copy()]
    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / n)))
    # U-shaped (higher at edges)
    inits.append(1.0 + 0.8 * (2.0 * i_arr / (n - 1) - 1.0) ** 2)
    # Multi-frequency
    sup = np.ones(n) * 0.5
    for fr in (1, 2, 3, 5, 8, 13):
        ph = rng_main.uniform(0, 2 * np.pi)
        sup += 0.2 * np.cos(2.0 * np.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.02)
    # Parent * smooth random
    smooth = np.exp(-0.5 * ((i_arr - n/2) / (n/4)) ** 2)
    inits.append(parent * (0.8 + 0.4 * smooth))
    # Random smooth (low frequency noise)
    lows = np.zeros(n)
    for k in range(1, 8):
        lows += rng_main.normal(0, 1.0 / k) * np.cos(2 * np.pi * k * i_arr / n + rng_main.uniform(0, 2*np.pi))
    inits.append(np.abs(lows) + 0.1)
    # Product of two cosines
    inits.append(np.abs(np.cos(np.pi * i_arr / n) * np.cos(2 * np.pi * i_arr / n)) + 0.05)
    # Parent perturbed
    inits.append(np.abs(parent + rng_main.normal(0, 0.15 * max(1, parent.mean()), n)) + 0.01)

    # Phase 1: LP iteration on parent length
    phase1_end = t0 + budget * 0.50
    n_inits = len(inits)
    time_per_init = max(8.0, (phase1_end - t0) / n_inits)

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        try:
            res, sc = _lp_iterate(init, n, stop, seed + 1000 * (idx + 1), use_min_peak=(idx % 2 == 0))
            if res is not None and sc < best_score - 1e-14:
                _try_update(res, sc)
        except Exception:
            pass

    # Phase 2: Second LP pass on best (warm start, longer time)
    phase2_end = t0 + budget * 0.68
    if time.monotonic() < phase2_end - 30.0:
        stop_p2 = min(phase2_end - 5.0, time.monotonic() + (phase2_end - time.monotonic()))
        try:
            res, sc = _lp_iterate(best_seq.copy(), len(best_seq), stop_p2, seed + 77777, use_min_peak=True)
            if res is not None and sc < best_score - 1e-14:
                _try_update(res, sc)
        except Exception:
            pass

    # Phase 3: Try alternative lengths
    phase3_end = t0 + budget * 0.78
    other_lengths = [400, 448, 384, 480, 320, 512]
    time_per_len = max(5.0, (phase3_end - phase3_end) / 1) if False else max(5.0, (phase3_end - max(phase2_end, time.monotonic())) / len(other_lengths))
    for L in other_lengths:
        if time.monotonic() > phase3_end - 8.0:
            break
        stop = min(phase3_end - 5.0, time.monotonic() + time_per_len)
        init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
        try:
            res, sc = _lp_iterate(init_l, L, stop, seed + 50000 + L, use_min_peak=True)
            if res is not None and sc < best_score - 1e-14:
                _try_update(res, sc)
        except Exception:
            pass

    # Phase 4: Coordinate descent polishing
    polish_end = deadline - 15.0
    if time.monotonic() < polish_end - 10.0:
        try:
            b_seq, b_sc = _coord_descent(best_seq, polish_end, seed=4242, max_rounds=100)
            if b_sc < best_score - 1e-14:
                _try_update(b_seq, b_sc)
        except Exception:
            pass

    # Phase 5: Another LP pass on best after polish
    if time.monotonic() < deadline - 25.0:
        stop_p5 = min(deadline - 20.0, time.monotonic() + 40.0)
        try:
            res, sc = _lp_iterate(best_seq.copy(), len(best_seq), stop_p5, seed + 31337, use_min_peak=True)
            if res is not None and sc < best_score - 1e-14:
                _try_update(res, sc)
        except Exception:
            pass

    # Phase 6: Final coordinate descent
    if time.monotonic() < deadline - 10.0:
        try:
            b_seq, b_sc = _coord_descent(best_seq, deadline - 8.0, seed=88888, max_rounds=80)
            if b_sc < best_score - 1e-14:
                _try_update(b_seq, b_sc)
        except Exception:
            pass

    # Phase 7: Two-coordinate polish
    if time.monotonic() < deadline - 8.0:
        try:
            b_seq, b_sc = _polish_two_coord(best_seq, deadline - 6.0, seed=55555)
            if b_sc < best_score - 1e-14:
                _try_update(b_seq, b_sc)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        fallback = np.ones(100)
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
