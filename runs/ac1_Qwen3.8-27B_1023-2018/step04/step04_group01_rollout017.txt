    c_ff = _conv_full(f)
    c_fg = _cross_conv(f, g)
    c_gg = _conv_full(g)

- The j=i term contributes: f'[i] * f'[k-i] - f[i] * f[k-i]
- The k-j=i term (j=k-i) contributes: f'[k-i] * f'[i] - f[k-i] * f[i]

If k ≠ 2i (i.e., i ≠ k-i):
Total change = δ*f[k-i] + δ*f[i] = δ(f[k-i] + f[i])

If k = 2i:
These are the same term (j=i and k-i=i are the same).
Change = f'[i]*f'[i] - f[i]*f[i] = (v+δ)^2 - v^2 = 2vδ + δ^2

For k outside [i, i+n-1]: no change.

So the update is:
- For k in [i, i+n-1], k ≠ 2i: c[k] += δ*(f[k-i] + f[i])
- For k = 2i (if in range): c[k] += 2*v*δ + δ^2

And the sum changes by δ (since we changed one element by δ).

This is very efficient! For each coordinate, we only need to update n entries of c.

For the 1D optimization in coordinate i:
score = 2n * max(c_new) / (S+δ)^2

    if M < 1e-18:
        return None
    A = _build_tmat(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-15 or not np.all(np.isfinite(g)):
        return None
    return g / gs


def _score_fast(a, conv=None):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    if conv is None:
        conv = _convolve_full(a)
    m = float(conv.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (S * S)


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    ws = w.sum()
    if ws < 1e-30:
        return np.ones(n) / n
    return w / ws


def _lp_phase(f, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f, dtype=np.float64).astype(np.float64).ravel().copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs

    best_f = f.copy()
    best_sc = _score_fast(f)

    conv = _convolve_full(f)
    M = float(conv.max())
    stall = 0
    max_stall = 25
    iteration = 0

    while time.monotonic() < stop_time - 8:
        iteration += 1
        if iteration > 3000:
            break

        # Solve LP
        g = _solve_lp(f, M, n, time_limit=8.0)
        if g is None:
            stall += 4
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
            fs = f.sum()
            if fs < 0.01:
                f = best_f.copy()
            else:
                f /= fs
            conv = _convolve_full(f)
            M = float(conv.max())
            continue

        g = g / g.sum()

        # Check convergence
        dist = float(np.linalg.norm(g - f))
        if dist < 1e-8:
            stall += 3
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            fs = f.sum()
            if fs < 0.01:
                f = best_f.copy()
            else:
                f /= fs
            conv = _convolve_full(f)
            M = float(conv.max())
            continue

        # Precompute for line search
        c_ff = conv
        c_gg = _convolve_full(g)
        c_fg = _cross_conv(f, g)

        # Line search with precomputed convs
        cur_sc = 2.0 * n * M  # S=1
        best_t = 0.0
        best_trial_sc = cur_sc

        # Quick scan
        t_candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for t_try in t_candidates:
            f_try = f + t_try * (g - f)
            if f_try.min() < -1e-12:
                continue
            S_try = f_try.sum()
            if S_try < 0.01:
                continue
            c_try = c_ff + 2.0 * t_try * c_fg + t_try * t_try * c_gg
            c_try = np.maximum(c_try, 0.0)
            sc_try = 2.0 * n * float(c_try.max()) / (S_try * S_try)
            if sc_try < best_trial_sc:
                best_trial_sc = sc_try
                best_t = t_try

        # Refine
        if best_t > 0:
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.05)
            for _ in range(10):
                t_try = (lo + hi) / 2.0
                f_try = f + t_try * (g - f)
                if f_try.min() < -1e-12:
                    hi = t_try
                    continue
                S_try = f_try.sum()
                if S_try < 0.01:
                    hi = t_try
                    continue
                c_try = c_ff + 2.0 * t_try * c_fg + t_try * t_try * c_gg
                c_try = np.maximum(c_try, 0.0)
                sc_try = 2.0 * n * float(c_try.max()) / (S_try * S_try)
                if sc_try < best_trial_sc:
                    best_trial_sc = sc_try
                    best_t = t_try
                    lo = t_try
                else:
                    hi = t_try

        # Apply update
        if best_t > 5e-4:
            f_new = f + best_t * (g - f)
            f_new = np.clip(f_new, 0, None)
            S_new = f_new.sum()
            if S_new < 0.01:
                stall += 1
            else:
                f_new /= S_new
                conv = _convolve_full(f_new)
                M = float(conv.max())
                sc_new = 2.0 * n * M
                f = f_new
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_f = f_new.copy()
                stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                fs = f.sum()
                if fs < 0.01:
                    f = best_f.copy()
                else:
                    f /= fs
                conv = _convolve_full(f)
                M = float(conv.max())
                stall = 0
                if time.monotonic() > stop_time - 30:
                    break

    # Return best found
    sc_best = _score_fast(best_f)
    sc_cur = _score_fast(f)
    if sc_cur < sc_best:
        return f, sc_cur
    return best_f, sc_best


def _grad_phase(f, stop_time, seed):
    n = len(f)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f, dtype=np.float64).ravel().copy(), 0, None)
    f = f / f.sum()
    best_f = f.copy()
    best_sc = _score_fast(f)

    alpha = 0.04
    alpha_min = 1e-12
    tau = 60.0
    tau_decay = 0.998
    it = 0

    while time.monotonic() < stop_time - 5:
        it += 1
        if it > 100000:
            break

        conv = _convolve_full(f)
        S = 1.0
        M = float(conv.max())
        sc = 2.0 * n * M

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_f = f.copy()

        # Softmax weights for smoothed max
        shifted = conv - M
        tau_conv = tau * shifted
        tau_conv -= tau_conv.max()
        p = np.exp(tau_conv)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(2 * n - 1)
            p[int(np.argmax(conv))] = 1.0
            p_sum = 1.0
        p /= p_sum

        # Gradient: d(sc)/d(f[i]) = 2n * sum_k p[k] * d(conv[k])/d(f[i]) / S^2 - 2*sc/S
        # d(conv[k])/d(f[i]) = f[k-i] if 0<=k-i<n, plus f[i] if 0<=k-i<n (symmetric)
        # = (p * f_rev)[i + n - 1] * 2 for the symmetric part
        f_rev = f[::-1]
        conv_size = 2 * n - 1
        fsize = next_fast_len(conv_size)
        P = rfft(p, fsize)
        Fr = rfft(f_rev, fsize)
        conv_p_f = irfft(P * Fr, fsize)[:conv_size]
        # grad[i] = 2n * (conv_p_f[i+n-1] + 2*p[2i]*f[i] if 2i<n... )
        # Actually: d(conv[k])/d(f[i]) for k in [i, i+n-1] is f[k-i] + (1 if k-i=i else 0)*...
        # Simpler: d(conv[k])/d(f[i]) = f[k-i] + f[i-... 

        # Let me use the cross-conv formula:
        # sum_k p[k] * d(conv[k])/d(f[i]) = sum_k p[k] * (f[k-i] if valid) + ...
        # = (p * f)[i] (cross correlation) 
        # Actually the gradient of conv[k] w.r.t. f[i] is:
        # f[k-i] (from the f[j] factor with j=i) + f[k-i] (from the f[k-j] factor with k-j=i, j=k-i)
        # But if i = k-i (i.e., k=2i), it's 2*f[i]
        # So sum_k p[k]*d(conv[k])/d(f[i]) = 2 * sum_k p[k]*f[k-i] (over valid range)
        # which is 2 * (p reversed convolved with f)[...]
        
        # Using the precomputed conv_p_f = ifft(fft(p) * fft(f_rev)):
        # conv_p_f[m] = sum_k p[k] * f_rev[m-k] = sum_k p[k] * f[n-1-(m-k)]
        # We want sum_k p[k]*f[k-i]. Let j = k-i, then sum_j p[j+i]*f[j]
        # = sum_j f[j]*p[j+i]
        # This is the cross-correlation of f and p at lag i.
        # corr[i] = sum_j f[j]*p[j+i] = (f * p_rev)[i] where p_rev = p[::-1]
        # Or: it's the convolution of f with reversed p.
        
        p_rev = p[::-1]
        fsize2 = next_fast_len(2 * n - 1)
        Ff = rfft(f, fsize2)
        Fp = rfft(p_rev, fsize2)
        corr = irfft(Ff * Fp, fsize2)[:2*n-1]
        
        # corr[i] = sum_j f[j] * p_rev[?] ... let me just use the direct formula
        # Actually, let me use: grad[i] ∝ sum_k p[k] * (f[k-i] if 0<=k-i<n)
        # This equals the i-th "cross-correlation" coefficient.
        # corr_full[i] = sum_j f[j] * p[j+i] for j=0..n-1-i (if i>=0)
        # But p has length 2n-1, so j+i can go up to 2n-2.
        # We need: for each i in 0..n-1, sum over k in [i, i+n-1] of p[k]*f[k-i]
        # = sum over j in 0..n-1 of p[j+i]*f[j]
        
        # Use FFT: this is the cross-correlation C[i] = <f, shift_i(p)>
        # C = ifft(fft(f) * conj(fft(p))) ... in real case: ifft(fft(f) * fft(p))[n-1:2n-1]
        # No wait, let me think differently.
        
        # f * p_rev (convolution) gives: (f*p_rev)[m] = sum_j f[j]*p_rev[m-j] = sum_j f[j]*p[2n-2-(m-j)]
        # I want: C[i] = sum_j f[j]*p[j+i]
        # Let m = 2n-2-i: (f*p_rev)[2n-2-i] = sum_j f[j]*p[2n-2-(2n-2-i)-j] = sum_j f[j]*p[i-j]
        # That's not quite right either.
        
        # Simplest: just use the direct computation with the cross conv I already have.
        # C[i] = sum_{j=0}^{n-1} f[j] * p[j+i] (where p[j+i]=0 if j+i >= 2n-1)
        # This is the (n-1-i)-th coefficient of the cross-correlation... 
        
        # Let me just compute it directly for n=400 (it's O(n^2) = 160000, fine)
        C = np.zeros(n)
        for i in range(n):
            # C[i] = sum_{j=0}^{n-1} f[j] * p[j+i]
            idx = np.arange(i, i + n)
            idx = idx[idx < 2 * n - 1]
            j_idx = idx - i
            C[i] = np.dot(f[j_idx], p[idx])
        
        grad = 2.0 * n * (2.0 * C)  # factor of 2 for symmetry, 2n for score
        # Subtract the S-gradient term (but S=1 fixed, so this is like a multiplier)
        # Actually for the projected gradient on the simplex, we just use grad and project.
        
        g_raw = f - alpha * grad
        f_new = _project_simplex(g_raw)
        
        sc_new = _score_fast(f_new)
        if sc_new < sc - 1e-15:
            f = f_new
            alpha = min(0.2, alpha * 1.02)
        else:
            alpha = max(alpha_min, alpha * 0.7)
            if alpha < alpha_min * 10:
                f = np.clip(f + rng.normal(0, 0.003, n), 0, None)
                f /= f.sum()
                alpha = 0.01
                best_f = f.copy()
                best_sc = _score_fast(f)
        
        # Periodic adaptive tau
        if it % 500 == 499:
            tau = max(50, tau * tau_decay)
        
        if it % 2000 == 1999:
            f = np.clip(f + rng.normal(0, 0.01, n), 0, None)
            f /= f.sum()
            alpha = 0.02

    sc_final = _score_fast(f)
    if sc_final < best_sc:
        return f, sc_final
    return best_f, best_sc


def _coord_polish(f, stop_time, seed, max_sweeps=100):
    n = len(f)
    f = np.clip(np.asarray(f, dtype=np.float64).ravel().copy(), 0, None)
    S = f.sum()
    if S < 0.01:
        return f, math.inf
    f = f / S
    S = 1.0
    conv = _convolve_full(f)
    best_sc = 2.0 * n * float(conv.max())
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 3:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3:
                break

            v = f[i]
            S_rest = S - v

            # Precompute influence
            # For k in [i, i+n-1]: delta contribution = 2*delta*f[k-i] + delta^2 (if k=2i)
            # Precompute the "base" max excluding the affected region
            lo_k = i
            hi_k = min(i + n - 1, 2 * n - 2)
            
            if lo_k > 0 or hi_k < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[lo_k:hi_k + 1] = False
                base_max = float(conv[mask].max()) if mask.any() else 0.0
            else:
                base_max = 0.0

            # Affected entries: for k in [lo_k, hi_k], k != 2i:
            #   new_c[k] = conv[k] + 2*delta*f[k-i]
            # For k = 2i (if in range):
            #   new_c[2i] = conv[2i] + 2*v*delta + delta^2

            # Build the affected linear coefficients
            aff_idx = np.arange(lo_k, hi_k + 1)
            if 2 * i in aff_idx and 2 * i < 2 * n - 1:
                # Separate the quadratic point
                aff_lin_idx = np.delete(aff_idx, aff_idx == 2 * i)
                quad_k = 2 * i
            else:
                aff_lin_idx = aff_idx
                quad_k = -1

            if len(aff_lin_idx) > 0:
                aff_convs = conv[aff_lin_idx]
                aff_slopes = 2.0 * f[np.clip(aff_lin_idx - i, 0, n - 1)]
            else:
                aff_convs = np.array([])
                aff_slopes = np.array([])

            quad_conv_val = conv[quad_k] if quad_k >= 0 else 0.0

            x_max = max(3.0 * v + 0.01, 0.1) if v > 1e-8 else 0.5

            # Grid search for delta (new value x = v + delta, so delta = x - v)
            # x in [0, x_max], so delta in [-v, x_max - v]
            n_pts = 40
            x_grid = np.linspace(0.0, x_max, n_pts)

            best_x = v
            best_obj = best_sc

            for x in x_grid:
                delta = x - v
                if abs(delta) < 1e-14:
                    m_new = float(conv.max())
                else:
                    # Compute max of updated conv
                    if len(aff_lin_idx) > 0:
                        aff_new = aff_convs + delta * aff_slopes
                        aff_max = float(aff_new.max())
                    else:
                        aff_max = base_max
                    if quad_k >= 0:
                        quad_new = conv[quad_k] + 2.0 * v * delta + delta * delta
                        quad_max = quad_new
                    else:
                        quad_max = -1e30
                    m_new = max(base_max, aff_max, quad_max)

                S_new = S_rest + x
                if S_new < 0.01:
                    continue
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-16:
                    best_obj = obj
                    best_x = x

            # Refine
            if best_x != v:
                dw = x_max / n_pts
                x_lo = max(0.0, best_x - 2 * dw)
                x_hi = min(x_max, best_x + 2 * dw)
                x_fine = np.linspace(x_lo, x_hi, 20)
                for x in x_fine:
                    delta = x - v
                    if abs(delta) < 1e-14:
                        continue
                    if len(aff_lin_idx) > 0:
                        aff_new = aff_convs + delta * aff_slopes
                        aff_max = float(aff_new.max())
                    else:
                        aff_max = base_max
                    if quad_k >= 0:
                        quad_new = conv[quad_k] + 2.0 * v * delta + delta * delta
                        quad_max = quad_new
                    else:
                        quad_max = -1e30
                    m_new = max(base_max, aff_max, quad_max)
                    S_new = S_rest + x
                    if S_new < 0.01:
                        continue
                    obj = 2.0 * n * m_new / (S_new * S_new)
                    if obj < best_obj - 1e-16:
                        best_obj = obj
                        best_x = x

            if best_x < v * 0.5 and v > 0.01:
                best_x = v  # don't reduce too aggressively
            if best_x != v and best_obj < best_sc - 1e-15:
                delta = best_x - v
                if len(aff_lin_idx) > 0:
                    conv[aff_lin_idx] += delta * aff_slopes
                if quad_k >= 0:
                    conv[quad_k] += 2.0 * v * delta + delta * delta
                f[i] = best_x
                S += delta
                best_sc = best_obj
                improved = True

        if not improved and sweep > 5:
            break

    # Final score
    f = np.clip(f, 0, None)
    final_sc = _score_fast(f)
    if final_sc < best_sc:
        best_sc = final_sc
    return f, final_sc
