import time
import math
from scipy.optimize import linprog

    np.maximum(c, 0, out=c)  # numerical safety

    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _lp_direction(f, n, M, timeout=5.0):
    """Solve max sum(g) s.t. (f*g)[k] <= M, g >= 0"""
    # Build A: A[k,j] = f[k-j] if 0 <= k-j < n
    # Using broadcasting
    idx = np.arange(2*n-1)[:,None] - np.arange(n)[None,:]
    mask = (idx >= 0) & (idx < n)
    A = np.zeros((2*n-1, n))
    A[mask] = f[idx[mask]]

    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout})
    except Exception:
        return None, None
    
    if res.status != 0 or res.x is None:
        return None, None
    
    g = np.clip(res.x, 0, None)
    if g.sum() < 1e-15:
        return None, None
    
    # Also get dual info if available
    return g, res

    """Coordinate descent polish"""
    n = len(f)
    rng = np.random.default_rng(seed)
    f = f.copy()
    cur = _score(f)
    best = f.copy()
    best_sc = cur
    
    mults = [0.0, 0.1, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0]

    n = len(a)
    if n <= 512:
        return np.convolve(a, a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)


def _lp_dir(f, n, M, timeout=5.0):
    idx = np.arange(2 * n - 1)[:, None] - np.arange(n)[None, :]
    mask = (idx >= 0) & (idx < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = f[idx[mask]]
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    if g.sum() < 1e-15:
        return None
    return g

                best_sc = sc_c
                best_f = f_c.copy()
                f = f_c.copy()

def _cd_polish(f, stop_time, max_passes=20, seed=0):
    n = len(f)
    rng = np.random.default_rng(seed)
    f = f.copy()
    cur = _score(f)
    best = f.copy()
    best_sc = cur
    
    mults = [0.0, 0.1, 0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0, 5.0]
    
    for p in range(max_passes):
        if time.monotonic() > stop_time - 3.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = f[i]
            best_v = orig
            best_local = cur
            for m in mults:
                if m == 1.0:
                    continue
                if orig > 1e-15:
                    v = orig * m
                else:
                    v = m * 0.01 * max(f.mean(), 0.1)
                if abs(v - orig) < 1e-15:
                    continue
                f[i] = v
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_v = v
            f[i] = best_v
            if best_local < cur - 1e-14:
                cur = best_local
                improved = True
                if cur < best_sc:
                    best_sc = cur
                    best = f.copy()
        if not improved:
            break
    
    return best, best_sc
