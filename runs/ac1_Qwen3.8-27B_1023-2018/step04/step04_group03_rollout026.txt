import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len

def _conv_full(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0]*a[0]])
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    # Numerical errors might give tiny negative values
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    """Compute the AC1 upper bound."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)
