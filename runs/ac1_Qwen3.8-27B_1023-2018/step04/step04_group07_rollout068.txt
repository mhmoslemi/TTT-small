import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


# ----------------- fast convolution / scoring -----------------

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _nm(a):
    s = float(a.sum())
    if s < 1e-12:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


# ----------------- LP machinery -----------------

def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_full(f, n, M, time_limit=4.0):
    A = _build_A(f, n)
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method="highs",
                      options={"presolve": True, "time_limit": time_limit})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0.0)
    gs = float(g.sum())
    return g / gs if gs > 1e-12 else None


def _lp_focused(f, n, M, top_k=160, time_limit=3.0):
    b = _conv_full(f)
    idx = np.argsort(-b)[:top_k]
    A = _build_A(f, n)
    try:
        res = linprog(-np.ones(n), A_ub=A[idx], b_ub=np.full(idx.size, M),
                      bounds=[(0, None)] * n, method="highs",
                      options={"presolve": True, "time_limit": time_limit})
    except Exception:
        return None

        return None
    g = np.asarray(res.x, dtype=np.float64)

    gs = float(g.sum())
    return g / gs if gs > 1e-12 else None

def _line_search(f, g, n, M, stop_time):
    best_t = 0.0
    best_val = M
    for t in (1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1):
        if time.monotonic() > stop_time:
            break
        f_new = (1.0 - t) * f + t * g
        m_new = float(_conv_full(f_new).max())
        if m_new < best_val:
            best_val, best_t = m_new, t

        lo = max(0.0, best_t - 0.18)
        hi = min(1.0, best_t + 0.18)

                break

def _subgrad_polish(a, stop_time):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel().copy(), 0.0, None)
    best = _nm(best)

        c = _conv_full(best)
        kstar = int(np.argmax(c))
        grad = np.zeros(n, dtype=np.float64)
        for i in range(n):
            j = kstar - i
            if 0 <= j < n:
                grad[i] = 2.0 * best[j]
        a_new = np.clip(best - eta * grad, 0.0, None)
        a_new = _nm(a_new)
        sc = _score(a_new)
        if sc < best_sc - 1e-15:

            ok += 1
            eta = min(eta * 1.15, 0.4)
        else:
            eta *= 0.7
            if eta < 0.002:
                break
    return best, best_sc

                r2 = _polish(best_seq, t0 + budget - 12.0, seed=88888)
                maybe_update(r2[0])
            except Exception:
                pass

    # ---- final validation ----
    try:
        out = np.clip(np.asarray(best_seq, dtype=np.float64).ravel(), 0.0, 1000.0)
        out_sc = float(evaluate_sequence([float(x) for x in out]))
    except Exception:
        out, out_sc = fallback, fallback_sc
    if not math.isfinite(out_sc):
        out, out_sc = fallback, fallback_sc
    if not math.isfinite(out_sc):
        return [1.0]
    return [float(v) for v in out]
