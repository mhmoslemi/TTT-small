    # Helpers
    def conv_full(a):
        n = len(a)
        sz = 2 * n - 1
        if n <= 1024:
            return np.convolve(a, a, mode='full')
        fsize = next_fast_len(sz)
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:sz]
        np.maximum(c, 0.0, out=c)
        return c
    
    def score_fast(a, conv=None):
        n = len(a)
        s = float(a.sum())
        if s < 0.01:
            return math.inf
        if conv is None:
            conv = conv_full(a)
        m = float(conv.max())
        return 2.0 * n * m / (s * s)
    
    # Track best
    best_seq = parent.copy()
    best_score = score_fast(best_seq)
    
    def update_best(seq, sc=None):
        nonlocal best_seq, best_score
        if sc is None:
            sc = score_fast(seq)
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_seq = seq.copy()
            best_score = sc

        f = np.clip(f_init.copy(), 0, None)
        s = f.sum()

            f = np.ones(n) / n
        else:

                )
