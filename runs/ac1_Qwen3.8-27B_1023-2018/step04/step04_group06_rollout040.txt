```python
def _score(a):
    """Fast score computation."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fftconvolve(a, a)

def _normalize(a):
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _conv_matrix(n):
    """Build the convolution matrix structure (indices)."""
    # For a given a, A @ b = a * b (full convolution)
    # A has shape (2n-1, n)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This depends on a, so we can't precompute it fully
    # But we can precompute the index structure
    rows, cols = [], []
    for k in range(2*n-1):
        for j in range(n):
            d = k - j
            if 0 <= d < n:
                rows.append(k)
                cols.append(j)
                # A[k,j] = a[d]
    return np.array(rows), np.array(cols)

def _build_A(n, a):
    """Build convolution matrix: A @ b = full_convolve(a, b)."""
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = a[k - j]
    return A
```

```python
    aa = fftconvolve(a, a)
    ag = fftconvolve(a, g)
    gg = fftconvolve(g, g)

    # We want to minimize f(t) for t in [0, 1]

            best_t = t
```

```python
def _lp_min_M(a):
    """Solve min M s.t. (a*b)_k <= M for all k, sum(b)=1, b>=0."""
    n = len(a)
    # Variables: [b_0, ..., b_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # A_ub: (2n-1) x (n+1)
    # Row k: [A[k,0], ..., A[k,n-1], -1] where A[k,j] = a[k-j] if valid
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    k_idx = np.arange(2*n-1)
    j_idx = np.arange(n)
    d = k_idx[:, None] - j_idx[None, :]
    mask = (d >= 0) & (d < n)
    
    for k in range(2*n-1):
        A_ub[k, mask[k, :]] = a[d[k, mask[k, :]]]
        A_ub[k, -1] = -1.0
    
    # A_eq: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success and res.x is not None:
        b = np.clip(res.x[:n], 0, None)
        M_opt = res.x[-1]
        return b, M_opt
    
    return None, None
```

```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.signal import fftconvolve


def _score_fast(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fftconvolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _build_conv_A(a):
    """Build matrix A such that A @ b = full_convolve(a, b). Shape (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    d = k_idx - j_idx
    mask = (d >= 0) & (d < n)
    A[mask] = a.ravel()[d[mask]]
    return A


def _lp_min_M(a, timeout=3.0):
    """Solve min M s.t. (a*b)_k <= M for all k, sum(b)=1, b>=0.
    Returns (b_opt, M_opt) or (None, None)."""
    n = len(a)
    A = _build_conv_A(a)
    
    # Variables: [b_0,...,b_{n-1}, M], total n+1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A @ b - M <= 0 for each row
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.success and res.x is not None:
            b_opt = np.clip(res.x[:n], 0, None)
            b_opt = _normalize(b_opt)
            M_opt = float(res.x[-1])
            return b_opt, M_opt
    except Exception:
        pass
    return None, None


def _lp_max_sum(a, M, timeout=3.0):
    """Solve max sum(b) s.t. (a*b)_k <= M, b>=0. Returns b or None."""
    n = len(a)
    A = _build_conv_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.success and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None


def _line_search(a, g, M_cur):
    """Find best t in [0,1] minimizing max(((1-t)a+tg)*((1-t)a+tg)).
    Both a and g normalized to sum 1."""
    aa = fftconvolve(a, a)
    ag = fftconvolve(a, g)
    gg = fftconvolve(g, g)
    
    best_t = 0.0
    best_val = float(aa.max())
    
    for t in np.linspace(0.0, 1.0, 41):
        val = float(np.max((1-t)**2 * aa + 2*t*(1-t) * ag + t**2 * gg))
        if val < best_val:
            best_val = val
            best_t = t
    
    if best_t > 0.005:
        lo = max(0.0, best_t - 0.02)
        hi = min(1.0, best_t + 0.02)
        for t in np.linspace(lo, hi, 81):

            if val < best_val:

        return None, best_val
    
    a_new = (1 - best_t) * a + best_t * g
    a_new = np.maximum(a_new, 0)
    a_new = _normalize(a_new)
    return a_new, best_val


def _flat_ac_init(n):
    """Generate approximate flat-autocorrelation sequence."""
    a = np.zeros(n)
    a[0] = 1.0
    C = 1.0
    for k in range(1, n):
        s_rest = 0.0
        for j in range(1, k):

    return a / a.sum()

                x_new = np.linspace(0, 1, nt)
                init_new = np.interp(x_new, x_old, parent)
            else:
                init_new = parent.copy()
            init_new = np.clip(init_new, 0, 1000)
            init_new = np.maximum(init_new, 1e-8)
            init_new /= init_new.sum()
            
            # Also try flat AC for this length
            flat_init = _flat_ac_init(nt)
            
            alloc = min(30.0, deadline - time.monotonic() - 10)
            if alloc < 5:
                break
            try:
                r = _optimize(init_new, time.monotonic() + alloc / 2, seed=500 + nt)
                try_update(r)
            except Exception:
                pass
            if time.monotonic() < deadline - 10:
                try:
                    r = _optimize(flat_init, deadline - 10, seed=600 + nt)
                    try_update(r)
                except Exception:
                    pass
    
    # ===== Phase 4: Deep polish on best =====
    if time.monotonic() < deadline - 10:
        ab = _normalize(best_a)
        try:
            r = _deep_polish(ab, deadline, seed=9999)
            try_update(r)
        except Exception:
            pass
    
    # ===== Final validation =====
    try:
        fs = float(evaluate_sequence([float(x) for x in best_a]))
        if not np.isfinite(fs):
            raise ValueError("invalid")
    except Exception:
        return [1.0] * 32
    
    return [float(x) for x in best_a]
```
