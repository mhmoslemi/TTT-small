    def _lp_run(init, n_init, stop, sd):
        if time.monotonic() > stop - 2:
            return
        try:
            res, sc = _lp_optimize(init, n_init, stop, sd)
        except Exception:
            return
        if res is not None and len(res) > 0:
            _update(res, sc)

    # Phase 1: LP on parent length with diverse inits (55%)
    n = n_p
    phase1_end = t0 + total * 0.50
    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    inits = [parent.copy()]
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n)))
    sup = np.ones(n) * 0.3
    for fr in (1, 2, 3, 5, 7, 11, 13):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.12 * np.cos(2.0 * math.pi * fr * i_arr / n + ph)
    inits.append(np.maximum(sup, 0.02))
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.03)
    sig = 0.15 * max(1.0, float(np.mean(parent)))
    inits.append(np.abs(parent + rng_normal(0, sig, n)) + 0.01)
    center = (n - 1) / 2.0
    inits.append(np.exp(-0.5 * ((i_arr - center) / (n / 5.0)) ** 2))
    inits.append(np.ones(n))
    inits.append(1.0 + 0.5 * np.cos(math.pi * i_arr / (n - 1)))
    inits.append(np.maximum(parent * 1.1 + rng_main.normal(0, 0.05, n), 0.01))

    n_inits = len(inits)
    time_per_init = max(8.0, (phase1_end - t0) / n_inits) * 0.95

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 6.0:
            break
        stop_i = min(phase1_end - 3.0, time.monotonic() + time_per_init)
        _lp_run(init, n, stop_i, seed + 1000 * (idx + 1))

    # Phase 2: Re-run LP on best (warm start, 10%)
    phase2_end = t0 + total * 0.62
    if time.monotonic() < phase2_end - 15:
        stop_p2 = min(phase2_end - 3, time.monotonic() + (phase2_end - time.monotonic()))
        _lp_run(best_seq.copy(), len(best_seq), stop_p2, seed + 777777)

    # Phase 3: Try other lengths (10%)
    phase3_end = t0 + total * 0.72
    other_lens = [384, 400, 320, 448]
    for L in other_lens:
        if time.monotonic() > phase3_end - 6:
            break
        stop_l = min(phase3_end - 3, time.monotonic() + 18.0)
        init_l = np.interp(np.linspace(0, 1, L),
                           np.linspace(0, 1, len(best_seq)), best_seq)
        _lp_run(init_l, L, stop_l, seed + 5000 + L)

    # Phase 4: Gradient polish (15%)
    phase4_end = t0 + total * 0.87
    if time.monotonic() < phase4_end - 10:
        try:
            ps, psc = _gradient_polish(best_seq, phase4_end - 3, seed=12345)
            _update(ps, psc)
        except Exception:
            pass

    # Phase 5: Coordinate polish (13%)
    phase5_end = deadline - 5.0
    if time.monotonic() < phase5_end - 10:
        try:
            cs, csc = _coord_polish(best_seq, phase5_end, seed=4242)
            _update(cs, csc)
        except Exception:
            pass

    # Final second polish if time allows
    if time.monotonic() < deadline - 8.0:
        try:
            cs2, csc2 = _coord_polish(best_seq, deadline - 5.0, seed=99999)
            _update(cs2, csc2)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf
    if not math.isfinite(final_sc):
        return [1.0]
    return [float(x) for x in best_seq]


def rng_normal(mean, std, size):
    return np.random.default_rng(12345).normal(mean, std, size)
