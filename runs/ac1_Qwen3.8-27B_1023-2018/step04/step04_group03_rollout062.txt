import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(416, dtype=np.float64)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _lp_dir(f, n, b, peak_count, timeout=8.0):
    M = float(b.max())
    if M < 1e-18:
        return None
    
    # Select peak positions (top by value, but at least peak_count)
    sorted_idx = np.argsort(b)[::-1]
    top = sorted_idx[:peak_count]
    # Filter: only include peaks >= 90% of max
    thr = M * 0.90
    sel = top[b[top] >= thr]
    if len(sel) == 0:
        sel = np.array([np.argmax(b)])
    
    pcnt = len(sel)
    rows = []
    cols = []
    vals = []
    for ri, k in enumerate(sel):
        klo = max(0, k - n + 1)
        khi = min(n, k + 1)
        for j in range(klo, khi):
            v = f[k - j]
            if v > 1e-20:
                rows.append(ri)
                cols.append(j)
                vals.append(v)
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(pcnt, n))
    cvec = -np.ones(n)
    bub = np.full(pcnt, M)
    
    try:
        res = linprog(cvec, A_ub=A, b_ub=bub, bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0, None)
    if g.sum() < 1e-15:
        return None
    return g

    f = f0.copy()

    max_ni = 20
    
    mults_c = (0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.1, 0.0)
    mults_f = (0.9, 1.1, 0.95, 1.05, 0.85, 1.15)
    
    for rnd in range(max_rounds):
        if time.monotonic() > stop_t:
            break
        improved = False
        order = rng.permutation(n)

def _lp_solve(f, b, npk, tl=8.0):
    n = len(f)
    M = b.max()
    if M < 1e-18:
        return None
    idx = np.argsort(b)[::-1]
    sel = idx[b[idx] >= M * 0.90]
    sel = sel[:max(npk, len(sel))]
    if len(sel) == 0:
        sel = np.array([np.argmax(b)])
    r, c, v = [], [], []
    for ri, k in enumerate(sel):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            x = f[k - j]
            if x > 1e-18:
                r.append(ri); c.append(j); v.append(x)
    if not r:
        return None
    A = csr_matrix((v, (r, c)), shape=(len(sel), n))
    try:

                      bounds=[(0, None)] * n, method='highs',

    except Exception:

    return g

def _conv(a):
    n = len(a)
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    return irfft(F * F, fs)[:2 * n - 1]


def _eval(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())

def _lp_solve(f, b, npk, tl=8.0):

        sel = np.array([int(np.argmax(b))])
    r, c, v = [], [], []
    for ri, k in enumerate(sel):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            x = f[k - j]
            if x > 1e-18:
                r.append(ri)
                c.append(j)
                v.append(x)

        return None
