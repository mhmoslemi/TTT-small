from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _conv_fft(a):
    n = len(a)
    if n <= 256:
        c = np.convolve(a, a, mode="full")
    else:
        fsize = next_fast_len(2 * n - 1)
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_fft(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)
