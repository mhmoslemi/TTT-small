    if res.status not in (0, 3):
        return None
    if res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _lp_optimize(f_init, n, stop_time, seed=0):
    rng = np.random.default_rng(seed)
    f = np.asarray(f_init, dtype=np.float64).ravel().copy()
    f = np.clip(f, 0, None)
    S = f.sum()
    if S < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / S

    best_f = f.copy()
    best_score = _score_norm(f)
    t = 0.4
    stall = 0
    max_stall = 150
    it = 0
    prev_score = best_score

    while time.monotonic() < stop_time - 1.0:
        b = _conv_fft(f)
        M = float(b.max())
        cur_score = 2.0 * n * M  # sum(f)=1

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_f = f.copy()

        if M < 1e-15:
            break

        g = _lp_dir(f, M)
        if g is None:
            stall += 5
            if stall > max_stall:
                sigma = 0.01 + 0.02 * (stall / max_stall)
                f = np.clip(best_f + rng.normal(0, sigma, n), 0, None)
                f /= f.sum()
                stall = 0
                max_stall = min(max_stall + 100, 800)
                t = 0.4
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-13:
            stall += 2
            if stall > max_stall:
                f = np.clip(best_f + rng.normal(0, 0.02, n), 0, None)
                f /= f.sum()
                stall = 0
                t = 0.4
            continue

        # Line search: try steps from large to small
        steps = sorted([t, t * 1.5, t * 0.6, 0.9, 0.1], reverse=True)
        steps = [s for s in steps if 0.02 < s < 0.98]
        if not steps:
            steps = [0.5]

        improved = False
        for ti in steps:
            f_new = (1.0 - ti) * f + ti * g
            sc = _score_norm(f_new)
            if sc < cur_score - 1e-13:
                f = f_new
                improved = True
                t = min(0.9, ti * 1.15)
                stall = 0
                prev_score = sc
                break

        if not improved:
            t = max(0.02, t * 0.5)
            stall += 1
            if stall >= max_stall:
                f = np.clip(best_f + rng.normal(0, 0.03, n), 0, None)
                f /= f.sum()
                t = 0.4
                stall = 0
                max_stall = min(max_stall + 200, 1500)

        it += 1

    return best_f, best_score


def _lse_polish(f, stop_time, alpha_start=30.0, lr_start=0.05, seed=0):
    """Polish using log-sum-exp gradient descent."""
    rng = np.random.default_rng(seed)
    f = f.copy()
    best_f = f.copy()
    best_score = _score_norm(f)
    alpha = alpha_start
    lr = lr_start
    stalled = 0

    while time.monotonic() < stop_time - 1.0:
        b = _conv_fft(f)
        bm = float(b.max())
        cur_score = 2.0 * len(f) * bm / (f.sum() ** 2)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_f = f.copy()
            stalled = 0
        else:
            stalled += 1
            if stalled > 50:
                break

        # Compute LSE gradient
        w = np.exp(alpha * (b - bm))
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        grad = 2.0 * np.correlate(w, f, mode='valid')

        # Projected gradient step
        f_new = f - lr * grad
        f_new = np.clip(f_new, 0, None)
        if f_new.sum() < 1e-10:
            lr *= 0.5
            continue
        f_new = f_new / f_new.sum()

        sc_new = _score_norm(f_new)
        if sc_new < cur_score - 1e-15:
            f = f_new
            lr = min(0.5, lr * 1.1)
            alpha = min(200.0, alpha * 1.05)
        else:
            lr = max(0.001, lr * 0.6)
            alpha = max(10.0, alpha * 0.9)

    return best_f, best_score
