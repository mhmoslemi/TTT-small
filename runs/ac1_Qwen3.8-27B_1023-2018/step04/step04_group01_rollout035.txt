import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(400, dtype=np.float64)
    if p.size == 0:
        p = np.ones(400, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(400, dtype=np.float64)
    return p.copy()


def _conv(a):
    n = len(a)
    size = next_fast_len(2 * n - 1)
    F = rfft(a, size)
    c = irfft(F * F, size)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    return float(2.0 * n * _conv(a).max() / (s * s))


def _lp_direction(f, n, M, time_limit=5.0):
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    s = float(g.sum())
    if s < 1e-10:
        return None
    return g / s


def _golden_section(f, g, n, A, B, C, iters=45):
    """Find t in [0,1] minimizing max_k (C_k t^2 + B_k t + A_k) where A_k=(1-t)^2*(f*f)[k] etc.
    Actually we parameterize: F(t) = max_k [(1-t)^2*A[k] + 2t(1-t)*B[k] + t^2*C[k]]
    Simplify: = max_k [A[k] + t*(2B[k]-2A[k]) + t^2*(A[k]-2B[k]+C[k])]
    """
    a_coeff = A - 2 * B + C
    b_coeff = 2 * B - 2 * A
    c_coeff = A

    def obj(t):
        vals = a_coeff * t * t + b_coeff * t + c_coeff
        return vals.max()

    gr = (math.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = obj(c_pt)
    fd = obj(d_pt)
    for _ in range(iters):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = obj(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = obj(d_pt)
    return (lo + hi) / 2.0


def _lp_optimize(f0, n, stop_time, seed, max_iters=500):

    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n) / n
        fs = 1.0
    else:
        f = f / fs
        fs = 1.0

    best_f = f.copy()
    best_score = _score(f)
    cur_score = best_score

    stall = 0
    max_stall = 60
    t0_inner = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() > stop_time - 5.0:
            break

        b = _conv(f)
        M = float(b.max())
        cur_score = 2.0 * n * M

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_f = f.copy()

        g = _lp_direction(f, n, M, time_limit=8.0)
        if g is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
            f /= f.sum()
            continue

        dist = float(np.linalg.norm(g - f))
        if dist < 1e-8:
            stall += 4

                break
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            f /= f.sum()
            continue

        # Precompute for golden section
        fg = np.convolve(f, g)
        gg = _conv(g)
        ff = b  # already computed

        t_opt = _golden_section(ff, fg, gg, n, iters=40)

        if t_opt < 1e-6:
            stall += 1
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
            f /= f.sum()
            continue

        f_new = (1.0 - t_opt) * f + t_opt * g
        f_new = np.clip(f_new, 0, None)
        f_new /= f_new.sum()

        new_score = _score(f_new)
        if new_score < cur_score - 1e-13:
            f = f_new
            stall = 0
        else:
            # Try smaller step
            t_try = t_opt * 0.5
            f_try = (1.0 - t_try) * f + t_try * g
            f_try = np.clip(f_try, 0, None)

            sc_try = _score(f_try)

                stall = 0

                if stall > max_stall // 2:
                    f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                    f /= f.sum()
                    stall = 0
                if stall > max_stall:
                    break

    return best_f, best_score


def _coord_descent_exact(f0, stop_time, max_sweeps=200, seed=123):
    """Coordinate descent with exact 1D golden-section minimization."""
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, _score(f)
    f = f / S
    conv = _conv(f)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    # Precompute structure
    # For coordinate i, changing f[i] by delta affects conv[k] for k in [i, i+n-1]
    # Delta conv[k] = 2*delta*f[k-i] + delta^2 * (1 if k==2i else 0)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break

            v = f[i]
            if v < 1e-15:
                x_lo, x_hi = 0.0, max(0.1, 2.0 * (S / n))
            else:
                x_lo, x_hi = 0.0, max(5.0 * v, 2.0 * (S / n))
                x_hi = min(x_hi, 2.0 * S)

            if x_lo == x_hi:
                continue

            # Precompute the base conv and the influence vector
            # conv_affected[k] = conv[k] for k in [i, min(i+n-1, 2n-2)]
            k_start = i
            k_end = min(i + n, 2 * n - 1)
            influence = 2.0 * f[k_start - i:k_end - i]  # 2*f[j] for j = k - i
            base_affected = conv[k_start:k_end].copy()

            # Max outside affected range
            if k_start > 0:
                max_out = float(conv[:k_start].max()) if k_start < 2 * n - 1 else 0.0
                max_out = max(max_out, float(conv[k_end:].max())) if k_end < 2 * n - 1 else max_out
            else:
                max_out = float(conv[k_end:].max()) if k_end < 2 * n - 1 else 0.0

            # Check if 2i is in affected range
            has_quad = (0 <= 2 * i < 2 * n - 1) and (k_start <= 2 * i < k_end)
            quad_idx = (2 * i) - k_start if has_quad else -1

            S_rest = S - v

            def eval_score_delta(delta):
                """Evaluate score when f[i] changes by delta (from v to v+delta)."""
                new_S = S_rest + (v + delta)
                if new_S < 0.01:
                    return math.inf
                # New affected conv values
                new_affected = base_affected + delta * influence
                if has_quad:
                    new_affected[quad_idx] += delta * delta
                new_M = max(float(new_affected.max()), max_out)
                return 2.0 * n * new_M / (new_S * new_S)

            # Golden section search
            gr = (math.sqrt(5.0) - 1.0) / 2.0
            a_g, b_g = 0.0, x_hi - v
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            fc_g = eval_score_delta(c_g)
            fd_g = eval_score_delta(d_g)

            for _ in range(35):
                if fc_g < fd_g:
                    b_g = d_g
                    d_g = c_g
                    fd_g = fc_g
                    c_g = b_g - gr * (b_g - a_g)
                    fc_g = eval_score_delta(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc_g = fd_g
                    d_g = a_g + gr * (b_g - a_g)
                    fd_g = eval_score_delta(d_g)

            best_delta = (a_g + b_g) / 2.0
            best_sc = eval_score_delta(best_delta)

            # Also check delta = 0 (don't change)
            sc_0 = 2.0 * n * M
            if best_sc < sc_0 - 1e-14:
                new_val = v + best_delta
                if new_val < 0:
                    new_val = 0.0
                delta_final = new_val - v
                f[i] = new_val
                S += delta_final
                conv[k_start:k_end] += delta_final * influence
                if has_quad:
                    conv[2 * i] += delta_final * delta_final
                M = max(float(conv.max()), 0.0)
                best_score = 2.0 * n * M
                best_f = f.copy()
                improved = True

        if not improved:
            break

    f = np.clip(f, 0, 1000.0)
    final_score = _score(f)
    if final_score < best_score:
        best_score = final_score
        best_f = f.copy()
    return best_f, best_score

def _pair_moves(f0, stop_time, max_rounds=30, seed=777):
    """Try random pair modifications to escape local optima."""

    f = f / f.sum()
    best_f = f.copy()
    best_score = _score(f)
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        for _ in range(80):
            if time.monotonic() > stop_time - 5.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            multipliers = (0.3, 0.5, 0.7, 1.5, 2.0, 3.0, 0.0)
            best_sc = best_score
            best_vals = (oi, oj)
            for mi in multipliers:
                ni = oi * mi if oi > 1e-12 else (0.01 * f.sum() / n if mi > 0 else 0.0)
                for mj in multipliers:
                    nj = oj * mj if oj > 1e-12 else (0.01 * f.sum() / n if mj > 0 else 0.0)
                    f[i], f[j] = ni, nj
                    fs = f.sum()
                    if fs < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        best_vals = (ni, nj)
            f[i], f[j] = best_vals
            if best_sc < best_score - 1e-14:
                best_score = best_sc
                best_f = f.copy()
                improved = True
    return best_f, best_score

    best_score = init_score

        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64)
        seq = np.clip(seq, 0, 1000.0)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or float(seq.sum()) < 0.01:
            return
        if sc is None:
            sc = _score(seq)
        if not math.isfinite(sc):
            return
        if sc < best_score - 1e-14:
            best_score = sc
            best_seq = seq.copy()

    # === Phase 1: LP Optimization (45% of budget) ===
    t_phase1_end = t0 + budget * 0.45
    try:
        res, sc = _lp_optimize(f_init, n, min(t_phase1_end, hard_deadline - 60), seed)
        if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
            res_norm = res / res.sum()
            _update_best(res_norm * fs, sc)
    except Exception:
        pass

    # === Phase 2: Coordinate Descent (30% of budget) ===
    t_phase2_end = t0 + budget * 0.75
    if time.monotonic() < t_phase2_end - 10:
        try:
            f_cur = best_seq / best_seq.sum()
            res, sc = _coord_descent_exact(f_cur, min(t_phase2_end, hard_deadline - 40),
                                           max_sweeps=150, seed=seed + 1)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_norm = res / res.sum()
                _update_best(res_norm * fs, sc)
        except Exception:
            pass

    # === Phase 3: More LP with perturbed start (10% of budget) ===
    t_phase3_end = t0 + budget * 0.85
    if time.monotonic() < t_phase3_end - 10 and best_score > 1.503:
        rng = np.random.default_rng(seed + 999)
        f_cur = best_seq / best_seq.sum()
        f_pert = np.clip(f_cur + rng.normal(0, 0.05, n), 0, None)
        f_pert /= f_pert.sum()
        try:
            res, sc = _lp_optimize(f_pert, n, min(t_phase3_end, hard_deadline - 30), seed + 500,
                                   max_iters=100)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_norm = res / res.sum()
                _update_best(res_norm * fs, sc)
        except Exception:
            pass

    # === Phase 4: Fine coordinate polish (7% of budget) ===
    t_phase4_end = t0 + budget * 0.92
    if time.monotonic() < t_phase4_end - 10:
        try:
            f_cur = best_seq / best_seq.sum()
            res, sc = _coord_polish_cold(f_cur, min(t_phase4_end, hard_deadline - 20),
                                         max_sweeps=100, seed=seed + 2000)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_norm = res / res.sum()
                _update_best(res_norm * fs, sc)
        except Exception:
            pass

    # === Phase 5: Pair moves (3% of budget) ===
    t_phase5_end = hard_deadline - 15.0
    if time.monotonic() < t_phase5_end - 5:
        try:
            f_cur = best_seq / best_seq.sum()
            res, sc = _pair_moves(f_cur, t_phase5_end, max_rounds=20, seed=seed + 3000)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_norm = res / res.sum()
                _update_best(res_norm * fs, sc)
        except Exception:
            pass

    # === Final coordinate polish in remaining time ===
    if time.monotonic() < hard_deadline - 8:
        try:
            f_cur = best_seq / best_seq.sum()
            res, sc = _coord_polish_cold(f_cur, hard_deadline - 5, max_sweeps=50, seed=seed + 4000)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_norm = res / res.sum()
                _update_best(res_norm * fs, sc)
        except Exception:
            pass

    # Final validation with exact evaluator
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000.0)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or float(best_seq.sum()) < 0.01:
        best_seq = np.ones(n)
        best_seq = best_seq / n

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Fallback: try parent
        try:
            return [float(x) for x in parent]
        except Exception:
            return [1.0] * n

    return [float(x) for x in best_seq]
