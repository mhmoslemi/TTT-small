import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(f, n, M, timeout=2.5):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': timeout,
                               'primal_feasibility_tolerance': 1e-12,
                               'dual_feasibility_tolerance': 1e-12,
                               'mip_rel_gap': 1e-12})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

    best_score = _score(f)
    t_step = 0.3
    stall = 0
    max_stall = 200
    it = 0

        M = float(b.max())
        if M < 1e-18:
            break

        if g is None:
            stall += 5
            if stall > max_stall:
                break

            f_new = np.clip(f + noise, 0, 1000)
            if f_new.sum() < 0.01:

            f = f_new
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            stall += 3
            continue
        g = g * (f.sum() / g_sum)

            f_c, sc_c = _cd_polish(best_seq, phase3_end, max_outer=60, seed=seed * 3 + 1)
        except Exception:
            f_c, sc_c = best_seq.copy(), best_score
        if sc_c < best_score - 1e-13:
            try:
                exact = float(evaluate_sequence([float(x) for x in f_c]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-13:
                best_seq = f_c.copy()
                best_score = exact

    # Phase 4: Try neighboring lengths (~5% time)
    phase4_end = t0 + budget * 0.92
    if time.monotonic() < phase4_end - 15.0:
        for L in [n - 20, n + 20, n - 40, n + 40]:
            if L < 64 or L > 1000 or time.monotonic() > phase4_end - 12.0:
                continue
            init_l = np.interp(np.linspace(0, 1, L),
                               np.linspace(0, 1, len(best_seq)),
                               best_seq)
            stop_l = min(phase4_end - 4.0, time.monotonic() + 20.0)
            _try_optimize(L, init_l, stop_l, seed + 90000 + L)

    # Phase 5: Final CD polish with remaining time
    final_end = deadline - 8.0
    if time.monotonic() < final_end - 10.0:
        try:
            f_c, sc_c = _cd_polish(best_seq, final_end, max_outer=100, seed=seed * 7 + 42)
        except Exception:
            f_c, sc_c = best_seq.copy(), best_score
        if sc_c < best_score - 1e-13:
            try:
                exact = float(evaluate_sequence([float(x) for x in f_c]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-13:
                best_seq = f_c.copy()
                best_score = exact

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0] * 100
        best_seq = fallback
        best_score = final_sc

    # Ensure all values valid
    result = []
    for x in best_seq:
        x = float(x)
        if not math.isfinite(x):
            x = 0.0
        x = max(0.0, min(1000.0, x))
        result.append(x)

    if not result or sum(result) < 0.01:
        result = [float(x) for x in fallback] if fallback.sum() >= 0.01 else [1.0]

    return result
