def _base_seed(seed):

        s = int(seed)
    except Exception:
        s = 42

def _next_seed(base, offset):
    return (base + int(offset)) % (2 ** 32)


def _nan_clean(a):
    return np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(32, dtype=np.float64)
    a = _nan_clean(a).copy()
    np.clip(a, 0.0, 1000.0, out=a)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        a = np.ones(max(int(a.size), 32), dtype=np.float64)
        s = float(a.sum())
    a /= s
    return a


def _maxconv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return 1e100
    c = np.convolve(a, a)
    m = float(c.max())
    if not np.isfinite(m):
        return 1e100
    return m


def _score_seq(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = _nan_clean(a).copy()
    np.clip(a, 0.0, 1000.0, out=a)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return float("inf")
    m = _maxconv(a)
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * a.size * m / (s * s))

    try:
        return (
            np.isfinite(newM)
            and np.isfinite(oldM)
            and newM < oldM
            and (oldM - newM) > (1e-14 + 1e-12 * max(1.0, float(oldM)))

    try:
        x = np.asarray(x, dtype=np.float64).ravel()
        y = np.asarray(y, dtype=np.float64).ravel()
        if x.size != y.size:
            return float("inf")
        d = float(np.linalg.norm(x - y))
        return d if np.isfinite(d) else float("inf")

def _get_idx(n):
    n = int(n)
    if n in _IDX_CACHE:
        return _IDX_CACHE[n]
    m = 2 * n - 1
    k_arr = np.arange(m, dtype=np.int64)
    lengths = np.minimum(k_arr + 1, m - k_arr).astype(np.int64)
    rows = np.repeat(k_arr, lengths)
    lo = np.maximum(0, k_arr - n + 1).astype(np.int64)
    total = int(lengths.sum())

        c = np.zeros(n + 1, dtype=np.float64)
        c[n] = 1.0
        bounds = [(0.0, None)] * n + [(0.0, None)]
        res = _linprog_highs(
            c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, tl=tl

        if s < 1e-15:
            return None, None
        g = g / s
        q = float(x[n])
        if not np.isfinite(q):
            q = float(np.max(A @ g)) if g.sum() > 0 else 1e100
        return g, q
    except Exception:
        return None, None


def _blend_M(a, d, t, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    d = np.asarray(d, dtype=np.float64).ravel()
    if t <= 0.0:
        return _maxconv(a)
    if t >= 1.0:
        return _maxconv(d)
    c = a + float(t) * (d - a)
    if c.min() < -1e-13:
        c = np.maximum(c, 0.0)
    s = float(c.sum())
    if s < 1e-15:
        return 1e100
    if abs(s - 1.0) > 1e-14:
        c = c / s
    return _maxconv(c)


def _line_search(a, d, n, current_M):
    a = np.asarray(a, dtype=np.float64).ravel()
    d = np.asarray(d, dtype=np.float64).ravel()
    if _dist(a, d) < 1e-9:
        return a.copy(), current_M

    best_t = 0.0
    best_M = current_M
    grid = (0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0)
    for t in grid:
        Mv = _blend_M(a, d, t, n)
        if _better(Mv, best_M):
            best_M = Mv
            best_t = float(t)

    if best_t == 0.0:
        return a.copy(), current_M

    lo = max(0.0, best_t - 0.12)
    hi = min(1.0, best_t + 0.12)
    for _ in range(60):
        if hi - lo < 1e-9:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = _blend_M(a, d, m1, n)
        v2 = _blend_M(a, d, m2, n)
        if v1 < v2:
            if _better(v1, best_M):
                best_M = v1
                best_t = m1
            hi = m2
        else:
            if _better(v2, best_M):
                best_M = v2
                best_t = m2
            lo = m1

    c = a + best_t * (d - a)
    c = np.maximum(c, 0.0)
    s = float(c.sum())
    if s < 1e-15:
        return a.copy(), current_M
    c = c / s
    Mv = _maxconv(c)
    if _better(Mv, best_M):
        return c, Mv
    return a.copy(), current_M


def _perturb(a, rng, sigma):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = a.size
    if n <= 0:
        return a
    if n > 1:
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        f1 = float(rng.uniform(0.3, 4.0))
        f2 = float(rng.uniform(0.3, 6.0))
        p1 = float(rng.uniform(0.0, 2.0 * np.pi))
        p2 = float(rng.uniform(0.0, 2.0 * np.pi))
        wave = (
            float(rng.uniform(0.25, 1.0)) * np.sin(2.0 * np.pi * f1 * x + p1)
            + float(rng.uniform(0.15, 0.8)) * np.cos(2.0 * np.pi * f2 * x + p2)
        )
        noise = 0.25 * rng.normal(0.0, 1.0, n)
        tilt = float(rng.uniform(-0.6, 0.6)) * (2.0 * x - 1.0)
        mult = np.exp(max(1e-8, float(sigma)) * (wave + noise + tilt))
        a = a * mult
        if rng.random() < 0.12:
            a = a[::-1].copy()
    else:
        a = np.array([1.0], dtype=np.float64)
    np.clip(a, 1e-14, None, out=a)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones_like(a) / max(a.size, 1)
    return a / s


def _active_gradient(a, c, n, rng=None, subset_factor=1.0, weights=None):
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
        c = np.asarray(c, dtype=np.float64).ravel()
        n = int(n)
        M = float(c.max())
        if M <= 0.0 or not np.isfinite(M):
            return None

        act = np.where(c >= M * (1.0 - 1e-5))[0]
        if act.size == 0:
            act = np.array([int(np.argmax(c))], dtype=np.int64)

        if act.size > 16:
            act = act[np.argsort(c[act])[-16:]]

        w = None
        if weights is not None:
            try:
                weights = np.asarray(weights, dtype=np.float64).ravel()
                if weights.size == c.size:
                    w = np.zeros(act.size, dtype=np.float64)
                    for i, k in enumerate(act):
                        if 0 <= k < weights.size:
                            w[i] = max(0.0, float(weights[k]))
                    if w.sum() > 1e-14 and subset_factor < 1.0:
                        pos = np.where(w > 1e-14)[0]
                        if pos.size > 2 and rng is not None:
                            ksel = max(1, int(round(pos.size * max(0.05, subset_factor))))
                            if ksel < pos.size:
                                sel = rng.choice(pos, size=ksel, replace=False)
                                w2 = np.zeros_like(w)
                                w2[sel] = w[sel]
                                w = w2
                    if w.sum() > 1e-14:
                        w = w / float(w.sum())
                    else:
                        w = None
            except Exception:
                w = None

        if w is None:
            w = np.clip(c[act] / max(M, 1e-300), 0.0, 1.0) ** 4
            if w.sum() < 1e-14:
                w = np.ones_like(w)
            w = w / float(w.sum())

        grad = np.zeros(n, dtype=np.float64)
        for k, wk in zip(act, w):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            js = np.arange(lo, hi, dtype=np.int64)
            idxs = k - js
            grad[lo:hi] += float(wk) * 2.0 * a[idxs]
        grad -= grad.mean()
        return grad
    except Exception:
        return None


def _gradient_step(a, c, M, n, rng, weights=None):
    try:
        attempts = (
            (1.0, None),
            (0.6, rng),
            (0.35, rng),
        )
        steps = (0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001)
        for factor, r in attempts:
            grad = _active_gradient(a, c, n, rng=r, subset_factor=factor, weights=weights)
            if grad is None:
                continue
            gn = float(np.linalg.norm(grad))
            if gn < 1e-14:
                continue
            for step in steps:
                cand = a - float(step) * grad / gn
                cand = np.maximum(cand, 0.0)
                s = float(cand.sum())
                if s < 1e-15:
                    continue
                cand = cand / s
                Mv = _maxconv(cand)
                if _better(Mv, M):
                    return cand
        return None
    except Exception:
        return None


def _optimize(a0, stop_time, seed=42, max_stuck=80):

        a = _normalize(a0)
        n = int(a.size)
        if n < 2 or n > 800:
            return a, _maxconv(a)

        stop_time = float(stop_time)
        best = a.copy()
        best_M = _maxconv(a)
        rng = np.random.default_rng(seed)
        stuck = 0
        perturb_level = 0
        it = 0
        dual_grad = None

        while True:
            now = time.monotonic()
            if now >= stop_time:
                break

            c = np.convolve(a, a)
            M = float(c.max())
            if not np.isfinite(M):
                break

            if _better(M, best_M):
                best_M = M
                best = a.copy()
                stuck = 0
            else:
                stuck += 1

            rem = stop_time - time.monotonic()
            if rem < 0.35:
                break

            if stuck > max_stuck:
                sigma = min(0.1 + 0.05 * float(perturb_level), 0.9)
                a = _perturb(a, rng, sigma)
                perturb_level += 1
                stuck = 0
                if perturb_level > 90:
                    break
                continue

            A = _build_A(n, a)
            made = False
            tl = min(0.85, max(0.04, rem * 0.35))

            g, dual = _lp_maxsum(A, M * (1.0 + 1e-9) + 1e-12, n, tl=tl)
            if dual is not None:
                dual_grad = dual
            if g is not None and _dist(g, a) > 1e-8:
                cand, candM = _line_search(a, g, n, M)
                if _better(candM, M):
                    a = cand
                    made = True

            if not made and rem > 0.65:
                for factor in (1e-7, 1e-5, 1e-4):
                    if time.monotonic() > stop_time - 0.35:
                        break
                    Mf = M * (1.0 + float(factor)) + 1e-12
                    g, _ = _lp_maxsum(A, Mf, n, tl=min(0.55, max(0.03, rem * 0.18)))
                    if g is None:
                        continue
                    if _dist(g, a) < 1e-8:
                        continue
                    cand, candM = _line_search(a, g, n, M)
                    if _better(candM, M):
                        a = cand
                        made = True
                        break

            if not made and (stuck >= 5 or it % 20 == 0) and rem > 0.85:
                g, q = _lp_minq(A, n, tl=min(1.05, max(0.05, rem * 0.35)))
                if g is not None:
                    if (np.isfinite(q) and q < M * (1.0 - 1e-9)) or _dist(g, a) > 1e-8:
                        cand, candM = _line_search(a, g, n, M)
                        if _better(candM, M):
                            a = cand
                            made = True

            if not made:
                cand = _gradient_step(a, c, M, n, rng, weights=dual_grad)
                if cand is not None:
                    a = cand
                    made = True

            if not made and stuck > 2:
                dirs = [a[::-1].copy()]
                if stuck > 6:
                    dirs.append(np.roll(a, 1).copy())
                    dirs.append(np.roll(a, -1).copy())
                for d in dirs:
                    cand, candM = _line_search(a, d, n, M)
                    if _better(candM, M):
                        a = cand
                        made = True
                        break

            if not made:
                sigma = 0.018 if stuck < 10 else 0.03
                a = _perturb(a, rng, sigma)

            it += 1

        best = _normalize(best)
        return best, _maxconv(best)
    except Exception:
        return _normalize(a0), 1e100


def _resample(seq, n):
    try:
        n = int(n)
        if n < 2:
            return np.ones(32, dtype=np.float64)
        a = _normalize(seq)
        if a.size < 2:
            return np.ones(n, dtype=np.float64)
        xo = np.linspace(0.0, 1.0, a.size, dtype=np.float64)
        xn = np.linspace(0.0, 1.0, n, dtype=np.float64)
        out = np.interp(xn, xo, a)
        return _normalize(out)
    except Exception:
        return np.ones(max(int(n), 32), dtype=np.float64)


def _length_list(base_n):
    base = int(base_n)
    candidates = [
        base - 45, base - 35, base - 25, base - 15, base - 10, base - 6,
        base - 3, base + 3, base + 6, base + 10, base + 15, base + 25,
        base + 35, base + 50, base + 80, base + 130,
        int(round(base * 0.75)), int(round(base * 1.15)), int(round(base * 1.35)),
    ]
    out = []
    seen = set()
    for nn in candidates:
        if 24 <= nn <= 700 and nn not in seen:
            seen.add(nn)
            out.append(nn)
    out.sort(key=lambda x: (abs(x - base), x))
    return out[:16]


def _make_n0_trials(base_seq, rng):
    N = int(base_seq.size)
    if N < 10:
        N = 100
    x = np.linspace(0.0, 1.0, N, dtype=np.float64)
    b = _normalize(base_seq)
    trials = []

    trials.append((b[::-1].copy(), 22.0))

    for tilt in (0.35, 0.8, 1.4):
        trials.append((_normalize(b * (1.0 + tilt * x)), 18.0))
        trials.append((_normalize(b * (1.0 + tilt * (1.0 - x))), 18.0))

    params = (
        (2, 1, 0.65),
        (1, 3, 0.35),
        (3, 1, 1.4),
        (1, 1, 0.9),
        (2, 2, 1.0),
        (5, 1, 0.2),
    )
    for p, q, cc in params:
        v = np.maximum((1.0 - x) ** p + float(cc) * (x ** q), 1e-9)
        trials.append((_normalize(v), 16.0))

    for i in range(2):
        trials.append((_perturb(b, rng, 0.25 + 0.15 * float(i)), 16.0))

    return trials


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    base = _base_seed(seed)

    try:
        total = float(budget_s)
    except Exception:
        total = 1300.0
    if not np.isfinite(total) or total < 20.0:
        total = 1300.0

    if total > 500.0:
        reserve = min(180.0, max(90.0, total * 0.13))
    else:
        reserve = min(60.0, max(20.0, total * 0.25))

    deadline = t0 + max(10.0, total - 25.0)
    exploration_deadline = max(t0 + 5.0, deadline - reserve)

    parent_raw = _get_parent()
    parent_norm = _normalize(parent_raw)

    best_seq = parent_norm.copy()
    best_score = _score_seq(best_seq)

    def consider(seq):
        nonlocal best_seq, best_score
        try:
            cand = _normalize(seq)
            if cand.size == 0:
                return
            sc = _score_seq(cand)
            if np.isfinite(sc) and (not np.isfinite(best_score) or sc < best_score - 1e-15):
                best_score = sc
                best_seq = cand.copy()
        except Exception:
            pass

    def exact_safe(seq):
        nonlocal best_score
        try:
            vals = [float(x) for x in np.asarray(seq, dtype=np.float64).ravel()]
            if not vals:
                return float("inf")
            if not all(np.isfinite(v) for v in vals):
                return float("inf")
            return float(evaluate_sequence(vals))
        except Exception:
            return float("inf")

    parent_exact = exact_safe(parent_norm)
    if np.isfinite(parent_exact) and (not np.isfinite(best_score) or parent_exact < best_score):
        best_score = parent_exact
        best_seq = parent_norm.copy()

    # Primary polish of the supplied parent construction.
    parent_window = max(
        15.0,
        min(
            760.0,
            (exploration_deadline - t0) * 0.68,
            exploration_deadline - t0,
        ),
    )
    parent_stop = min(exploration_deadline, t0 + parent_window)
    if parent_stop > time.monotonic() + 5.0:
        try:
            seq, _ = _optimize(parent_norm, parent_stop, seed=_next_seed(base, 11), max_stuck=90)
            consider(seq)
        except Exception:
            pass

    # Diverse short ensembles at the current preferred length.
    trial_deadline = min(exploration_deadline, deadline - reserve)
    if time.monotonic() < trial_deadline - 8.0:
        N0 = int(best_seq.size) if best_seq.size >= 8 else int(parent_norm.size)
        if N0 >= 8:
            rng_trials = np.random.default_rng(_next_seed(base, 1001))
            for i, (init, quota) in enumerate(_make_n0_trials(best_seq, rng_trials)):
                now = time.monotonic()
                if now >= trial_deadline - 4.0:
                    break
                q = min(float(quota), max(4.0, trial_deadline - now - 0.5))
                if q < 4.0:
                    break
                stop = now + q
                try:
                    seq, _ = _optimize(init, stop, seed=_next_seed(base, 2000 + i), max_stuck=55)
                    consider(seq)
                except Exception:
                    pass

    # Nearby length re-samplings, using the best sequence found so far.
    if time.monotonic() < trial_deadline - 12.0:
        base_len = int(best_seq.size) if best_seq.size >= 8 else int(parent_norm.size)
        for j, nn in enumerate(_length_list(base_len)):
            now = time.monotonic()
            if now >= trial_deadline - 4.0:
                break
            q = min(24.0, max(4.0, trial_deadline - now - 0.5))
            if q < 4.0:
                break
            init = _resample(best_seq, nn)
            stop = now + q
            try:
                seq, _ = _optimize(init, stop, seed=_next_seed(base, 5000 + j), max_stuck=60)
                consider(seq)
            except Exception:
                pass

    # Final polish on the best exact/fast candidate.
    final_stop = max(time.monotonic() + 0.5, deadline - 1.0)
    if time.monotonic() < final_stop - 3.5:
        try:
            seq, _ = _optimize(best_seq, final_stop, seed=_next_seed(base, 99999), max_stuck=120)
            consider(seq)
        except Exception:
            pass

    best_seq = _normalize(best_seq)
    final_list = [float(x) for x in best_seq]
    final_score = exact_safe(final_list)

    if np.isfinite(parent_exact) and (not np.isfinite(final_score) or parent_exact < final_score - 1e-15):
        final_list = [float(x) for x in parent_norm]
        final_score = parent_exact

    if not np.isfinite(final_score):
        fb = _normalize(parent_norm)
        final_list = [float(x) for x in fb]
        final_score = exact_safe(final_list)

    if not np.isfinite(final_score):
        final_list = [1.0] * 32

    return final_list
