def _peak_flatten(a, eta, n_steps):
    n = len(a)
    a = a / a.sum()
    L = 2*n - 1
    size = next_fast_len(L)
    
    for step in range(n_steps):
        c = irfft(rfft(a, size)**2, size)[:L]
        cmax = c.max()
        cmin = c.mean()  # = 1/(2n-1) for sum=1
        # Weights: emphasize lags near the peak
        margin = 0.1 * (cmax - cmin)
        w = np.where(c > cmax - margin, 1.0, 0.0)

        # This is a cross-correlation of w and a

def _score_fft(a):
    """Fast score using FFT (may have tiny numerical errors)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0: return math.inf
    s = float(a.sum())
    if s < 0.01: return math.inf
    L = 2*n - 1
    S = next_fast_len(L)
    c = irfft(rfft(a, S)**2, S)[:L]
    c = np.maximum(c, 0.0)
    m = float(c.max())
    if not math.isfinite(m): return math.inf
    return 2.0*n*m/(s*s)

    """Exact score using np.convolve."""
    a = np.asarray(a, dtype=np.float64).ravel()

    s = float(a.sum())
    if s < 0.01: return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m): return math.inf
    return 2.0*a.size*m/(s*s)

def _resample(src, n):
    """Resample src to length n using midpoint interpolation."""
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.full(n, 1.0/n)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m) + 0.5) / m
        fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300: return np.full(n, 1.0/n)
    return out / s

def _mul_subgrad(a, n_iters, eta, deadline, verbose=False):
    """Multiplicative subgradient on the convolution max."""
    n = a.size
    a = np.asarray(a, dtype=np.float64).copy()
    s = float(a.sum())
    a /= s  # normalize to sum 1
    
    L = 2*n - 1
    conv_S = next_fast_len(L)
    grad_S = next_fast_len(3*n - 1)  # for gradient computation
    flip_a_base = None  # will update
    
    best_a = a.copy()
    best_cmax = float('inf')
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution
        F = rfft(a, conv_S)
        c = irfft(F * F, conv_S)[:L]
        cmax = float(c.max())
        
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
        
        # Compute weights: soft emphasis on peak lags
        cmean = float(c.mean())  # should be ≈ 1/(2n-1) for sum=1
        spread = cmax - cmean
        if spread < 1e-300:
            break
        # Temperature for soft weight
        T = 0.08 * spread
        w = np.exp((c - cmax) / T)
        w /= w.sum()  # normalize weights
        
        # Compute gradient: g[i] = 2 * sum_j a[j] * w[i+j]
        flip_a = a[::-1]
        Fflip = rfft(flip_a, grad_S)
        Fw = rfft(w, grad_S)
        conv_grad = irfft(Fflip * Fw, grad_S)[:3*n-2]
        g = 2.0 * conv_grad[n-1:2*n-1]
        
        # Multiplicative update with damping
        gmax = float(g.max())
        if gmax < 1e-300:
            break
        step = eta * g / gmax
        # Damped update: a *= (1 - step) to keep some mass
        a *= (1.0 - step)
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s < 1e-300:
            break
        a /= s
    
    return best_a, best_cmax
