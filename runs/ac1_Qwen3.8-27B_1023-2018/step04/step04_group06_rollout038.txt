    """LP with constraint generation (lazy constraints)."""
    n_local = len(a)
    c = np.convolve(a, a)
    active = [int(c.argmax())]
    
    t_start = time.monotonic()
    g = None
    
    for it in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break
        
        # Build LP with only active constraints
        k_list = np.array(active)
        A_rows = np.zeros((len(k_list), n_local))
        for idx, k in enumerate(k_list):
            j_idx = np.arange(n_local)
            d = k - j_idx
            mask = (d >= 0) & (d < n_local)
            A_rows[idx, mask] = a[d[mask]]
        
        c_obj = -np.ones(n_local)
        b_ub = np.full(len(k_list), M)
        bounds = [(0, None)] * n_local
        
        try:
            res = linprog(c_obj, A_ub=A_rows, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': 2.0})
        except:
            break
        
        if res.status != 0:
            break
        
        g = np.clip(res.x, 0, None)
        if g.sum() < 1e-15:
            break
        
        # Check all constraints
        ag = np.convolve(a, g)
        violations = np.where(ag > M * (1 + 1e-9))[0]
        if len(violations) == 0:
            break  # All constraints satisfied
        
        # Add most violated
        worst = int(violations[np.argmax(ag[violations])])
        if worst not in active:
            active.append(worst)
        # Limit active set size
        if len(active) > 20:
            # Keep only the most violated
            active = [int(w) for w in violations[:15]] + [int(c.argmax())]
    
    if g is not None and g.sum() > 1e-15:
        return g / g.sum()
    return None

        return float(np.convolve(a, a).max())

        return 2.0 * n * M_of(a)

        a = np.maximum(a, 0.0)
        s = a.sum()

            return

            best_M = m

        kl = np.asarray(k_list)

        for i, k in enumerate(kl):
            j = np.arange(nloc)
            d = k - j

            A[i, mask] = a[d[mask]]
        return A
