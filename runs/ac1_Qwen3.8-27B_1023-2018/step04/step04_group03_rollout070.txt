import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


_LP_CACHE = {}


def _get_parent():
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0:
                p = np.where(np.isfinite(p), p, 0.0)
                p = np.clip(p, 0.0, 1000.0)
                if p.size > 0:
                    return p
        except Exception:
            pass
    return np.ones(416, dtype=np.float64)


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(416, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _conv_full_fft(a):
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    m = next_fast_len(2 * n - 1)
    F = rfft(a, m)
    c = irfft(F * F, m)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _conv_np(a):
    return np.convolve(a, a)


def _score_np(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_np(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _exact_score(a):
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            return math.inf
        a = np.clip(a, 0.0, 1000.0)
        if not np.all(np.isfinite(a)) or float(a.sum()) < 0.01:
            return math.inf
        return float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        return math.inf


def _flat_autocorr_init(n):
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64) / max(1, n)
    a = np.zeros(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        s = float(np.dot(a[1:k + 1], a[k - 1::-1]))
        a[k] = max(0.0, 1.0 - s) * 0.5
    s = float(a.sum())
    if s < 1e-12:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _make_inits(n, parent, seed):
    rng = np.random.default_rng(seed)
    n = int(n)
    inits = []

    def add(a):
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
            if a.size != n:
                if a.size == 0:
                    a = np.ones(n, dtype=np.float64)
                elif a.size == 1:
                    a = np.full(n, float(a[0]), dtype=np.float64)
                else:
                    a = np.interp(
                        np.linspace(0.0, 1.0, n),
                        np.linspace(0.0, 1.0, a.size),
                        a
                    )
            a = _normalize(a)
            if a.size == n and np.all(np.isfinite(a)) and float(a.sum()) > 0.01:
                inits.append(a)
        except Exception:
            pass

    if parent is not None and len(parent) > 0:
        add(parent)

    add(_flat_autocorr_init(n))

    i = np.arange(n, dtype=np.float64)
    x = i / float(max(1, n - 1))

    for alpha in (0.5, 1.0, 1.5, 2.0):
        add(np.power(np.maximum(n - 1.0 - i, 1e-12), alpha))
        add(np.power(np.maximum(i, 1e-12), alpha))

    for p in (1.0, 2.0, 4.0):
        add(np.power(np.cos(0.5 * math.pi * x), p) + 1e-8)
        add(np.power(np.cos(0.5 * math.pi * (1.0 - x)), p) + 1e-8)

    for _ in range(2):
        add(rng.random(n) + 0.05)

    for _ in range(2):
        c = float(n) * (0.25 + 0.5 * rng.random())
        w = float(n) * (0.08 + 0.25 * rng.random())
        add(np.exp(-0.5 * ((i - c) / max(1e-8, w)) ** 2) + 1e-8)

    if parent is not None and len(parent) == n and float(parent.mean()) > 0:
        add(parent + rng.normal(0.0, 0.08 * max(1.0, float(parent.mean())), n))

    return inits


class _LPCache(object):
    def __init__(self, n):
        self.n = int(n)
        m = 2 * self.n - 1

        K_list = []
        J_list = []
        for k in range(m):
            i0 = max(0, k - self.n + 1)
            i1 = min(k, self.n - 1) + 1
            cnt = i1 - i0
            if cnt > 0:
                K_list.extend([k] * cnt)
                J_list.extend(range(i0, i1))

        K = np.asarray(K_list, dtype=np.int32)
        J = np.asarray(J_list, dtype=np.int32)
        off = K - J
        non = int(off.size)

        Kt = np.arange(m, dtype=np.int32)
        Jt = np.full(m, self.n, dtype=np.int32)

        rows = np.concatenate((K, Kt))
        cols = np.concatenate((J, Jt))
        tags = np.arange(non + m, dtype=np.float64)

        A = csr_matrix((tags, (rows, cols)), shape=(m, self.n + 1))
        pattern = A.data.astype(np.int64).copy()

        self.is_f = pattern < non
        self.f_tags = pattern[self.is_f]
        self.t_mask = ~self.is_f

        self.A = A
        new_data = np.empty(pattern.size, dtype=np.float64)
        new_data[self.is_f] = 0.0
        new_data[self.t_mask] = -1.0
        self.A.data = new_data

        self.zeros = np.zeros(m, dtype=np.float64)
        self.c = np.zeros(self.n + 1, dtype=np.float64)
        self.c[-1] = 1.0
        self.A_eq = np.ones(self.n + 1, dtype=np.float64)
        self.A_eq[-1] = 0.0
        self.base_bounds = [(0.0, None)] * (self.n + 1)

    def update(self, f):
        data = self.A.data
        data[self.is_f] = f[self.f_tags]
        data[self.t_mask] = -1.0


def _get_lp_cache(n):
    n = int(n)
    if n not in _LP_CACHE:
        if n <= 1:
            _LP_CACHE[n] = None
        else:
            try:
                _LP_CACHE[n] = _LPCache(n)
            except Exception:
                _LP_CACHE[n] = None
    return _LP_CACHE[n]


def _lp_direction(f, cache, timeout=2.0):
    if cache is None:
        return None
    try:
        cache.update(f)
        bounds = cache.base_bounds.copy()
        res = linprog(
            cache.c,
            A_ub=cache.A,
            b_ub=cache.zeros,
            A_eq=cache.A_eq,
            b_eq=1.0,
            bounds=bounds,
            method='highs',
            options={'time_limit': max(0.05, float(timeout)), 'presolve': True}
        )
        if res.status not in (0, 3) or res.x is None:
            return None
        g = np.asarray(res.x[:cache.n], dtype=np.float64).copy()
        tval = float(res.x[cache.n])
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        s = float(g.sum())
        if not math.isfinite(tval) or s < 1e-10:
            return None
        return g / s
    except Exception:
        return None


def _line_search_peak(bff, bfg, bgg):
    def pk(t):
        u = 1.0 - t
        return float(np.max((u * u) * bff + (2.0 * t * u) * bfg + (t * t) * bgg))

    best_val = pk(0.0)
    best_t = 0.0

    ts = np.linspace(0.0, 1.0, 31)
    extra = (0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.0)
    for t0 in extra:
        if 0.0 <= t0 <= 1.0:
            ts = np.concatenate((ts, [t0]))

    for t0 in ts:
        t0 = float(t0)
        if t0 < 0.0:
            t0 = 0.0
        elif t0 > 1.0:
            t0 = 1.0
        v = pk(t0)
        if v < best_val:
            best_val = v
            best_t = t0

    if 0.0 < best_t < 1.0:
        delta = 0.025
        for _ in range(18):
            old = best_val
            for t0 in (best_t - delta, best_t + delta,
                       best_t * 0.8, best_t * 1.25,
                       best_t * 0.6, best_t * 1.7,
                       best_t - 0.5 * delta, best_t + 0.5 * delta):
                if t0 < 0.0 or t0 > 1.0:
                    continue
                v = pk(float(t0))
                if v < best_val:
                    best_val = v
                    best_t = float(t0)
            if not (best_val < old):
                delta *= 0.5
            if delta < 1e-6:
                break

    gr = (math.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = pk(c)
    fd = pk(d)
    for _ in range(30):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = pk(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = pk(d)
    t0 = (a + b) * 0.5
    v = pk(t0)
    if v < best_val:
        best_val = v
        best_t = t0

    return float(best_t), float(best_val)


def _optimize_lp(f0, stop_time, seed, timeout=2.0):
    f0 = _normalize(f0)
    n = f0.size
    if n == 0:
        return np.ones(416, dtype=np.float64), math.inf
    if n == 1:
        return f0, 2.0

    f = f0.copy()
    cache = _get_lp_cache(n)

    b = _conv_full_fft(f)
    best_peak = float(b.max())
    best_f = f.copy()

    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 45
    last_update = time.monotonic()

    while time.monotonic() < stop_time - 4.0:
        rem = stop_time - time.monotonic()
        if rem < 6.0:
            break

        tmo = min(max(0.25, timeout), max(0.2, rem * 0.25))
        b = _conv_full_fft(f)
        cur_peak = float(b.max())

        if cur_peak < best_peak - 1e-12:
            best_peak = cur_peak
            best_f = f.copy()
            last_update = time.monotonic()

        g = _lp_direction(f, cache, tmo)
        if g is None:
            stall += 2
        else:
            dist = float(np.linalg.norm(g - f))
            if dist < 1e-7:
                stall += 3
            else:
                bfg = _conv_full_fft(f, g)
                bgg = _conv_full_fft(g)
                t_new, new_peak = _line_search_peak(b, bfg, bgg)
                if t_new > 1e-5 and new_peak < cur_peak - 1e-12:
                    f_new = (1.0 - t_new) * f + t_new * g
                    f_new = _normalize(f_new)
                    bv = _conv_full_fft(f_new)
                    pv = float(bv.max())
                    if pv < cur_peak:
                        f = f_new
                        cur_peak = pv
                    if pv < best_peak - 1e-12:
                        best_peak = pv
                        best_f = f.copy()
                        last_update = time.monotonic()
                    stall = 0
                else:
                    stall += 1

        if stall >= max_stall or time.monotonic() - last_update > 120.0:
            i = np.arange(n, dtype=np.float64)
            choice = int(rng.integers(0, 5))
            if choice == 0:
                shape = _flat_autocorr_init(n)
            elif choice == 1:
                shape = _normalize(np.power(np.maximum(n - 1.0 - i, 1e-10), 1.0))
            elif choice == 2:
                shape = _normalize(np.power(np.maximum(i, 1e-10), 1.0))
            elif choice == 3:
                xx = i / float(max(1, n - 1))
                shape = _normalize(np.cos(0.5 * math.pi * xx) ** 2 + 1e-8)
            else:
                shape = _normalize(rng.random(n) + 0.01)
            w = float(rng.uniform(0.15, 0.55))
            if rng.random() < 0.65:
                noise = rng.normal(0.0, 0.04 * max(float(np.mean(best_f)), 1e-3), n)
            else:
                noise = 0.0
            f = _normalize((1.0 - w) * best_f + w * shape + noise)
            stall = 0
            max_stall = min(max_stall + 35, 260)
            last_update = time.monotonic()

    return _normalize(best_f), float(best_peak)


def _polish_cd(f0, stop_time, seed):
    f = _normalize(f0)
    n = f.size
    if n == 0:
        return np.ones(416, dtype=np.float64)
    if n == 1:
        return f

    b = _conv_np(f)
    best_score = 2.0 * n * float(b.max())
    best_f = f.copy()

    rng = np.random.default_rng(seed)
    no_imp = 0
    total = 2 * n - 1
    multipliers = (0.125, 0.25, 0.5, 0.75, 0.9, 1.1, 1.25, 1.5, 2.0, 4.0)

    while time.monotonic() < stop_time - 3.0 and no_imp < 4:
        improved_any = False
        scale = max(float(np.max(f)) if f.size else 1.0, 1.0 / n, 1e-6)
        xmax = max(2.0, 10.0 * scale)
        grid = np.geomspace(1e-10, xmax, 18)

        if no_imp % 2 == 0:
            order = np.arange(n)
        else:
            order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break

            x = float(f[i])
            s_tot = float(f.sum())
            s_rest = s_tot - x
            if s_rest <= 1e-14:
                continue

            ks = i + np.arange(n, dtype=np.int64)
            old = 2.0 * x * f
            old_i = x * x
            old[i] = old_i
            base = b[ks] - old
            lin = 2.0 * f.copy()
            lin[i] = 0.0

            unaff = 0.0
            if i > 0:
                unaff = max(unaff, float(np.max(b[:i])))
            if i + n < total:
                unaff = max(unaff, float(np.max(b[i + n:])))

            base_i = float(base[i])

            def sc_x(xv, base=base, lin=lin, base_i=base_i, unaff=unaff, s_rest=s_rest):
                S = s_rest + xv
                if S <= 1e-14:
                    return math.inf
                arr = base + lin * xv
                m_arr = float(np.max(arr))
                ai = base_i + xv * xv
                mb = unaff
                if m_arr > mb:
                    mb = m_arr
                if ai > mb:
                    mb = ai
                if mb < 0.0:
                    mb = 0.0
                return (2.0 * n * mb) / (S * S)

            candidates = [0.0, x,
                          scale * 1e-3, scale * 0.1, scale * 0.5,
                          scale, 2.0 * scale]
            candidates.extend(grid.tolist())
            if x > 0.0:
                for mlt in multipliers:
                    candidates.append(x * float(mlt))

            best_x = x
            best_sc = best_score
            for xv in candidates:
                xv = float(xv)
                if xv < 0.0 or xv > 1000.0:
                    continue
                sc = sc_x(xv)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_x = xv

            xv = best_x
            for _ in range(8):
                old_sc = best_sc
                step = max(1e-6, 0.05 * scale, 0.1 * xv if xv > 0.0 else 0.0)
                trials = [xv - step, xv + step]
                if xv > 0.0:
                    trials.extend([xv * 0.8, xv * 1.25, xv * 0.65, xv * 1.5])
                else:
                    trials.extend([1e-8, step, 5.0 * step, 0.1 * scale])
                for xt in trials:
                    xt = float(xt)
                    if xt < 0.0 or xt > 1000.0:
                        continue
                    sc = sc_x(xt)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        xv = xt
                if not (best_sc < old_sc):
                    break

            if best_sc < best_score - 1e-12:
                old_f = f.copy()
                f[i] = best_x
                S = float(f.sum())
                if S < 1e-14:
                    f = old_f
                    continue
                f = f / S
                new_b = _conv_np(f)
                new_score = 2.0 * n * float(new_b.max())
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_f = f.copy()
                    b = new_b
                    improved_any = True

        if improved_any:
            no_imp = 0
        else:
            no_imp += 1

    return _normalize(best_f)


def _resample(a, L):
    L = int(L)
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == L:
        return a.copy()
    if a.size == 0 or L <= 0:
        return np.ones(max(1, L), dtype=np.float64)
    if a.size == 1:
        return np.full(max(1, L), float(a[0]), dtype=np.float64)
    return np.interp(
        np.linspace(0.0, 1.0, L),
        np.linspace(0.0, 1.0, a.size),
        a
    )


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    reserve = max(10.0, 0.02 * budget)
    deadline = t0 + budget - reserve

    parent = _get_parent()
    parent_n = _normalize(parent)
    n0 = parent_n.size

    exact_best = parent_n.copy()
    exact_best_score = _exact_score(exact_best)

    def _consider(a):
        nonlocal exact_best, exact_best_score
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
            if a.size == 0:
                return
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if float(a.sum()) < 0.01:
                return
            sc = _exact_score(a)
            if math.isfinite(sc) and sc < exact_best_score - 1e-15:
                exact_best = a.copy()
                exact_best_score = sc
        except Exception:
            return

    inits = _make_inits(n0, parent_n, seed)
    for init in inits:
        _consider(init)

    if not math.isfinite(exact_best_score):
        exact_best = parent_n.copy()
        exact_best_score = _exact_score(exact_best)

    work = max(10.0, deadline - t0)

    phase1_end = t0 + work * 0.55
    if phase1_end - time.monotonic() > 25.0:
        try:
            res, _ = _optimize_lp(exact_best, phase1_end, seed, timeout=2.0)
            _consider(res)
        except Exception:
            pass

    phase2_end = t0 + work * 0.72
    if phase2_end - time.monotonic() > 25.0:
        try:
            res = _polish_cd(exact_best, phase2_end, seed + 101)
            _consider(res)
        except Exception:
            pass

    phase3_end = t0 + work * 0.80
    if phase3_end - time.monotonic() > 25.0:
        try:
            res, _ = _optimize_lp(exact_best, phase3_end, seed + 202, timeout=1.6)
            _consider(res)
        except Exception:
            pass

    phase4_end = t0 + work * 0.86
    if phase4_end - time.monotonic() > 20.0:
        try:
            res = _polish_cd(exact_best, phase4_end, seed + 303)
            _consider(res)
        except Exception:
            pass

    tried_lengths = {int(exact_best.size)}
    base_len = int(n0) if int(n0) > 50 else 416
    for delta in (32, -32, 64, 96, 128, -64, 160):
        L = base_len + delta
        if L < 120 or L > 900 or L in tried_lengths:
            continue
        rem = deadline - time.monotonic()
        if rem < 45.0:
            break
        tried_lengths.add(L)
        try:
            init_L = _normalize(_resample(exact_best, L))
            _consider(init_L)
            stop1 = time.monotonic() + min(35.0, max(10.0, rem * 0.5))
            if stop1 > deadline - 5.0:
                stop1 = deadline - 5.0
            res, _ = _optimize_lp(init_L, stop1, seed + L, timeout=1.3)
            _consider(res)
            rem = deadline - time.monotonic()
            if rem > 18.0:
                stop2 = time.monotonic() + min(18.0, rem * 0.4)
                if stop2 > deadline - 3.0:
                    stop2 = deadline - 3.0
                res = _polish_cd(res, stop2, seed + L + 1000)
                _consider(res)
        except Exception:
            pass

    rem = deadline - time.monotonic()
    if rem > 25.0:
        try:
            res = _polish_cd(exact_best, deadline - 3.0, seed + 999)
            _consider(res)
        except Exception:
            pass

    final_seq = np.asarray(exact_best, dtype=np.float64).ravel()
    if final_seq.size == 0:
        final_seq = parent_n.copy()

    ok = (
        final_seq.size > 0
        and np.all(np.isfinite(final_seq))
        and float(np.clip(final_seq, 0.0, 1000.0).sum()) >= 0.01
        and math.isfinite(_exact_score(final_seq))
    )
    if not ok:
        for cand in (parent_n, np.ones(416, dtype=np.float64)):
            cand = _normalize(cand)
            if math.isfinite(_exact_score(cand)):
                final_seq = cand
                break

    final_seq = np.clip(np.asarray(final_seq, dtype=np.float64).ravel(), 0.0, 1000.0)
    if final_seq.size == 0 or float(final_seq.sum()) < 0.01 or not np.all(np.isfinite(final_seq)):
        final_seq = np.ones(416, dtype=np.float64)

    return [float(x) for x in final_seq]
