import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(414, dtype=np.float64)
    if p.size == 0:
        p = np.ones(414, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(414, dtype=np.float64)
    return p


def _fft_conv(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    fsize = next_fast_len(n + m - 1)
    F = rfft(a, fsize)
    G = rfft(b, fsize)
    c = irfft(F * G, fsize)[:n + m - 1]
    return c


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, time_limit=1.5):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

    it = 0

    while it < max_iters:
        it += 1

            break

        b = _fft_conv(f)

                f = np.clip(f + rng.normal(0, 0.015, n), 0, None)

def _subgrad_optimize(f0, n, stop_time, seed, max_iters=100000):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = _score(f)
    step = 0.03
    no_imp = 0
    momentum = np.zeros(n)
    mu = 0.6

    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break

        b = _fft_conv(f)
        M = float(b.max())
        eps = max(1e-15, 1e-10 * M)
        active = np.flatnonzero(b >= M - eps)
        if len(active) == 0:
            break

        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo <= i_hi:
                idx = k - np.arange(i_lo, i_hi + 1)
                grad[i_lo:i_hi + 1] += 2.0 * f[idx]
        grad /= len(active)

        S = float(np.sum(f))
        obj_sub = 2.0 * n * (grad * S - 2.0 * M) / (S ** 3)
        obj_sub -= np.mean(obj_sub)
        norm_sg = np.linalg.norm(obj_sub)

            momentum *= 0.5
            continue
