import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = None
    if p is None or p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_conv_matrix(f, n):
    """Build matrix A where (f@g)[k] = A[k,:] @ g."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=10.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A = _build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    if np.sum(g) < 1e-15:
        return None
    return g


def _lp_optimize(f0, n, stop_time, seed, max_iter=3000):
    """Iterative LP-based optimization minimizing max(f*f) with sum(f)=1."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n)
    else:
        f = f / fs

    best_f = f.copy()
    best_sc = _score(f)
    t_param = 0.15
    stall = 0
    max_stall = 150
    prev_M = None
    consecutive_no_lp = 0

    for it in range(max_iter):
        if time.monotonic() > stop_time:
            break

        b = _fft_conv(f)
        M = float(b.max())

        if prev_M is not None and abs(M - prev_M) < 1e-14 * max(1, M):
            stall += 1
        else:
            if prev_M is not None and M < prev_M:
                stall = max(0, stall - 5)
            prev_M = M

        if M < 1e-18:
            break

        g = _lp_direction(f, n, M, timeout=8.0)
        if g is None:
            consecutive_no_lp += 1
            stall += 3
            if stall > max_stall:
                if consecutive_no_lp > 10:
                    break
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                stall = max_stall // 3
                consecutive_no_lp = 0
            continue
        consecutive_no_lp = 0

        gs = g.sum()
        g = g / gs

        diff = g - f
        dnorm = np.linalg.norm(diff)
        f_norm = np.linalg.norm(f)
        if dnorm < 1e-11 * max(1.0, f_norm):
            stall += 8
            if stall > max_stall * 1.5:
                break
            continue

        # Line search with multiple trial points
        best_t = t_param
        best_sc_try = best_sc
        trials = sorted(set([t_param, t_param * 0.4, t_param * 0.7,
                             t_param * 1.3, t_param * 1.8,
                             0.03, 0.08, 0.15, 0.25, 0.4, 0.6, 0.8]))
        for tt in trials:
            tt = min(0.99, max(0.005, tt))
            fn = (1.0 - tt) * f + tt * g
            fn = np.clip(fn, 0, None)
            if fn.sum() < 1e-10:
                continue
            sc_try = _score(fn)
            if sc_try < best_sc_try - 1e-15:
                best_sc_try = sc_try
                best_t = tt

        if best_sc_try < best_sc - 1e-13:
            fn = (1.0 - best_t) * f + best_t * g
            fn = np.clip(fn, 0, None)
            f = fn
            best_f = f.copy()
            best_sc = best_sc_try
            t_param = min(0.9, best_t * 1.15)
            stall = 0
        else:
            t_param = max(0.01, t_param * 0.5)
            stall += 1
            if stall >= max_stall:
                # Perturb to escape
                f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                t_param = 0.15
                stall = 0
                max_stall = min(max_stall + 50, 600)
                prev_M = None

    return best_f, best_sc


def _autoconv_subgrad(f, active_lags):
    """Compute subgradient of max_k (f*f)[k] at given active lags."""
    n = len(f)
    grad = np.zeros(n)
    for k in active_lags:
        # Term 1: f[k-m] for m where 0 <= k-m < n
        lo1 = max(0, k - n + 1)
        hi1 = min(k, n - 1)
        if lo1 <= hi1:
            idx1 = k - np.arange(lo1, hi1 + 1)
            grad[lo1:hi1 + 1] += f[idx1]
        # Term 2: f[m-k] for m where 0 <= m-k < n
        lo2 = max(k, 0)
        hi2 = min(k + n - 1, n - 1)
        if lo2 <= hi2:
            idx2 = np.arange(lo2, hi2 + 1) - k
            grad[lo2:hi2 + 1] += f[idx2]
    grad /= len(active_lags)
    return grad


def _subgrad_optimize(f0, n, stop_time, seed, max_iter=80000):
    """Subgradient + momentum optimization."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / max(f.sum(), 1e-10)

    best_f = f.copy()
    best_sc = _score(f)
    step = 0.02
    momentum = np.zeros(n)
    mu = 0.4
    no_imp = 0
    patience = 3000

    for it in range(max_iter):
        if time.monotonic() > stop_time:
            break

        b = _fft_conv(f)
        M = float(b.max())
        eps = max(1e-13, 1e-9 * M)
        active = np.flatnonzero(b >= M - eps)
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        grad = _autoconv_subgrad(f, active)
        # Project: remove mean to stay on simplex
        g_proj = grad - np.mean(grad)
        g_norm = np.linalg.norm(g_proj)
        if g_norm < 1e-16:
            break

        momentum = mu * momentum - step * g_proj / g_norm
        fn = f + momentum
        fn = np.clip(fn, 0, None)
        s = fn.sum()
        if s < 1e-10:
            step *= 0.5
            momentum *= 0.5
            continue
        fn = fn / s

        sc = _score(fn)
        if sc < best_sc - 1e-14:
            f = fn
            best_f = f.copy()
            best_sc = sc
            step = min(0.25, step * 1.015)
            no_imp = 0
        else:
            step *= 0.93
            no_imp += 1
            if no_imp > patience:
                f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                momentum = np.zeros(n)
                step = 0.02
                no_imp = 0
                patience = min(patience + 500, 8000)
                if patience > 8000 and it > 50000:
                    break

    return best_f, best_sc


def _polish(f0, n, stop_time, seed, max_rounds=200):
    """Local polish: coordinate and pair adjustments."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    best_sc = _score(f)
    best_f = f.copy()

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time:
            break
        improved_this_round = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            if orig < 1e-15:
                continue
            best_v = orig
            best_vsc = best_sc
            for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.4, 1.8, 0.0]:
                f[i] = orig * mult if mult > 0 else 0.0
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_vsc - 1e-15:
                    best_v = f[i]
                    best_vsc = sc
            f[i] = best_v
            if best_vsc < best_sc - 1e-15:
                best_sc = best_vsc
                best_f = f.copy()
                improved_this_round = True

        # Pair moves
        n_pairs = min(100, n)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-15 and oj < 1e-15:
                continue
            for mi in [0.5, 1.5, 0.0]:
                for mj in [0.5, 1.5, 0.0]:
                    if mi == 1.0 and mj == 1.0:
                        continue
                    f[i] = oi * mi if oi > 1e-15 and mi > 0 else 0.0
                    f[j] = oj * mj if oj > 1e-15 and mj > 0 else 0.0
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        best_f = f.copy()
                        improved_this_round = True
            f[i], f[j] = oi, oj

        if not improved_this_round:
            break

    return best_f, best_sc

                best_seq = a_clip.copy()

    # Phase 1: Primary LP optimization (50% of budget)
