    # A[k,i] = 2*f[k-i] if 0<=k-i<n else 0

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]

def _lp_descent(f, n, timeout=10.0):
    """Find direction d that reduces peak of (f+d)*(f+d) via linearized LP."""
    B = _conv_full(f)
    M = B.max()
    if M < 1e-30:
        return None
    
    # Build A_ub: (2n-1) x (n+1)
    # Row k: A[k,i] = 2*f[k-i] for valid i (0≤i<n, 0≤k-i<n)
    # A[k,n] = -1
    # b_ub[k] = -B[k]

    for i in range(n):
        ks = slice(i, i+n)
        A_ub[ks, i] = 2.0 * f
    A_ub[:, n] = -1.0
    b_ub = -B.copy()
    
    # Equality: sum d_i = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: d_i ∈ [-f_i, inf), M' ∈ [0, inf)
    bounds = [(-f[i], None) for i in range(n)] + [(0.0, None)]
    
    # Objective: minimize M'
    c = np.zeros(n+1)
    c[n] = 1.0

                      bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    
    if res.status not in (0, 3):
        return None
    
    d = res.x[:n].copy()
    if np.max(np.abs(d)) < 1e-15:
        return None
    return d

def _subgradient_step(f, n, step=0.5):
    """Move f to reduce peak autocorrelation using subgradient of max B_k."""
    B = _conv_full(f)
    M = B.max()
    # Subgradient of max: uniform over argmax lags
    peak_idx = np.where(B >= M * 0.999)[0]
    # Gradient of B_k w.r.t. f_i is f_{k-i} (symmetric)
    grad = np.zeros(n)
    for k in peak_idx:
        for i in range(max(0, k-n+1), min(n, k+1)):
            grad[i] += f[k-i]
    grad /= len(peak_idx)
    
    # Update: f -= step * grad, projected to [0, inf), normalized
    f_new = f - step * grad / max(1, grad.max())
    f_new = np.clip(f_new, 0, None)
    s = f_new.sum()
    if s > 0:
        f_new /= s
    return f_new

Let me also think about a **completely different approach**: 

**Riemannian optimization on the simplex**: Since the problem is scale-invariant, we can work on the simplex {a_i ≥ 0, sum(a_i) = 1}. The objective is 2n * max_k (a*a)_k. We can use Riemannian trust-region or Gauss-Newton methods on this manifold.

One more idea: **Use the fact that the problem is symmetric under reversal**. a_i and a_{n-1-i} give the same autocorrelation. So we can fix a_i = a_{n-1-i} and reduce the problem size. But this might exclude asymmetric optima. Let me not do this for now.

Let me also consider **n values**. The score is 2n * max_b / S^2. For the same "shape" of a, the max_b scales differently with n. Let me try n from 300 to 600 and see which gives the best score.

Actually, for a given shape (say, a normalized hat function), as n increases:
- S stays the same (normalized)
- max_b: the peak of the convolution. For a smooth function, max_b scales as 1/n (since the convolution of two functions on a finer grid has a lower peak for the same L1 norm).
- So score = 2n * (c/n) / 1 = 2c, which is constant.

Wait, that's not right. Let me think more carefully. If a has length n and sum=1, and we think of it as a sample of a continuous function f on [0,1] with ∫f = 1, then:
- (a*a)_k ≈ (n * ∫ f(x) f(x + k/n) dx) * (1/n) = (1/n) * (f*f)_cont(k/n)
- max_k (a*a)_k ≈ max_x (f*f)_cont(x) / n
- score = 2n * (1/n) * max(f*f) / 1 = 2 * max(f*f)

So the score is independent of n for a fixed continuous shape! The only way to improve is to change the shape.

But for discrete effects (e.g., a sequence that's exactly uniform), the score is exactly 2 regardless of n. For a "perfect" sequence, the score approaches 2n/(2n-1) ≈ 1 as n→∞.

So for practical purposes, n=416 is fine. Changing n by ±50 won't help much unless the optimization finds a fundamentally different shape.

    n = len(a)

    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
