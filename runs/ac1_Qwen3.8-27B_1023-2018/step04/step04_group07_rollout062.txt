import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_impl(a):
    """Compute the exact score from the evaluator formula."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

**Alternative: Penalty method**

Instead of the LP, I can use a penalty/soft-max approach:
1. Define G(a) = log(sum_k exp(β * (a*a)_k - β * M_0)) where M_0 is a reference

            jlo = max(0, k-n+1)
            jhi = min(n-1, k)
            if jhi >= jlo:
                grad[jlo:jhi+1] += 2.0 * f[k-jhi:k-jlo+1][::-1]
        grad /= float(len(S))
        
        alpha = a0 / math.sqrt(it+1)
        fn = f - alpha*grad
        fn = np.maximum(fn, 0.0)
        s = float(fn.sum())
        if s < 1e-10: break
        fn /= s
        
        nsc = _score_impl(fn)
        if nsc < sc:
            f, sc = fn, nsc
            if sc < best_sc:
                best_sc = sc
                best = f.copy()
        it += 1

    # ====== Phase 3: Final validation ======
    fin = np.clip(best, 0.0, 1000.0)
    try:
        fsc = float(evaluate_sequence([float(x) for x in fin]))
    except Exception:
        fsc = math.inf
    if not math.isfinite(fsc):
        return [float(x) for x in P]
    return [float(x) for x in fin]
