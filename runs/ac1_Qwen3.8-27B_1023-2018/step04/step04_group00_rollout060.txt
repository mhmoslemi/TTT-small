            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    def run_chunk(uu, T, end_stage):
        try:
            f0, _ = lse_func(uu, T)
        except Exception:
            return uu
        if not math.isfinite(f0):
            return uu
        best_u = np.array(uu, dtype=np.float64, copy=True)
        best_f = float(f0)
        stale = 0

        while time.monotonic() < end_stage - 1.5:
            maxit = 30 if n > 2000 else 40
            opts = {
                "maxiter": maxit,
                "gtol": 1e-12,
                "ftol": 1e-15,
                "maxls": 20,
                "maxcor": 30,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(
                    lse_func, best_u, args=(T,),
                    method="L-BFGS-B", jac=True, options=opts,
                )
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == uu.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, "status", 1) == 0:
                break
        return best_u

    if fine_only:
        stages = ((0.002, 0.30), (0.0006, 0.40), (0.00015, 0.30))
    else:
        stages = (
            (0.15, 0.10),
            (0.05, 0.12),
            (0.015, 0.15),
            (0.004, 0.18),
            (0.001, 0.22),
            (0.0003, 0.18),
            (0.00008, 0.12),
            (0.00002, 0.13),
        )

    start = time.monotonic()
    avail = float(stop_time - start - 10.0)
    if avail <= 5.0:
        return p

    best_p = p.copy()
    best_sc = _score_fft(p)
    total_w = float(sum(w for _, w in stages))
    if total_w <= 0.0:
        total_w = 1.0

    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 4.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        sc_cur = _score_fft(p_cur)
        if not math.isfinite(sc_cur):
            p = best_p.copy()
            u = p_to_u(p)
            continue
        if sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif sc_cur > best_sc + max(0.001, 1e-4 * best_sc):
            p = best_p.copy()
        else:
            p = p_cur.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, max_sweeps=200000, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p, math.inf
    p = p / s

    if n <= 450:
        G = 48
    elif n <= 1200:
        G = 36
    elif n <= 4096:
        G = 28
    else:
        G = 22

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue
            base = c[start_i:end_i].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start_i > 0:
                ml = float(c[:start_i].max())
                if ml > max_const:
                    max_const = ml
            if end_i < Ltot:
                mr = float(c[end_i:].max())
                if mr > max_const:
                    max_const = mr

            avg = S / float(n)
            x_hi = max(25.0 * v, 80.0 * avg, 0.04 * S, 1e-12)
            x_hi = min(x_hi, 15.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, avg)

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 <= v <= x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag = i - start_i  # always i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                dval = base[diag] + w * infl[diag] + w * w
                maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if G > 4:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.5 * dw)
                hi = min(x_hi, best_x + 2.5 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 40, dtype=np.float64)
                if 0.0 <= best_x <= x_hi:
                    fine = np.concatenate((fine, [best_x]))
                    fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 15.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start_i:end_i] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        np.maximum(c, 0.0, out=c)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 8.0:
                    try:
                        p = p * np.exp(rng.normal(0.0, 0.025, n, dtype=np.float64))
                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss > 0.0:
                            p /= ss
                        c = _conv_self(p)
                        S_total = 1.0
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider is not None:
                                consider(best_p)
                        flat = 0
                    except Exception:
                        break
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _lp_step(f, stop_time):
    """LP-based direction: maximize sum(g) s.t. f*g <= M, g >= 0."""
    n = int(f.size)
    if n < 10:
        return None
    M = float(_conv_self(f).max())
    if M <= 0 or not math.isfinite(M):
        return None

    # Build constraint matrix A: A[k,j] = f[j] for j <= k <= j+n-1
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f[j]

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        constraints = [A @ g <= b]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        t_start = time.monotonic()
        solver = None
        for slv in [cp.SCS, cp.ECOS, cp.CLP]:
            try:
                prob.solve(solver=slv, verbose=False)
                solver = slv
                break
            except Exception:
                continue
            if time.monotonic() > stop_time - 5:
                break
        if g.value is None:
            return None
        g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        gv = float(g_val.sum())
        if gv < 1e-10 or not math.isfinite(gv):
            return None
        return g_val / gv
    except Exception:
        return None


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 20.0
    if hard <= t0 + 3.0:
        hard = t0 + max(2.0, budget - 1.0)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64)
        best_arr /= 64.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 5e-14:
            best_arr = a.copy()
            best_score = sc

    n_par = int(best_arr.size)

    # === Phase 1: LSE optimization on the parent length ===
    phase1_end = min(hard - 60.0, t0 + 0.28 * budget)
    if phase1_end > t0 + 30.0 and n_par >= 128:
        p1 = best_arr.copy()
        p1 = p1 / float(p1.sum())
        try:
            p1 = _lbfgs_optimize(p1, phase1_end, consider=consider, fine_only=False)
            if p1 is not None and p1.size == n_par and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    # Use the best found so far
    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 2: Coordinate descent ===
    phase2_end = min(hard - 80.0, t0 + 0.52 * budget)
    if phase2_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord_polish(work, phase2_end, max_sweeps=300000,
                                    seed=int(seed + 7777), consider=consider)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 3: LP step for diversity ===
    lp_end = min(hard - 60.0, time.monotonic() + 50.0)
    if lp_end > time.monotonic() + 15.0 and np.all(np.isfinite(work)):
        g_dir = _lp_step(work, lp_end)
        if g_dir is not None and g_dir.size == n_par:
            for t_alpha in [0.25, 0.15, 0.08]:
                blended = (1.0 - t_alpha) * work + t_alpha * g_dir
                blended = np.maximum(blended, 0.0)
                bs = float(blended.sum())
                if bs > 0:
                    blended /= bs
                    consider(blended)

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 4: More coordinate descent ===
    phase4_end = min(hard - 40.0, t0 + 0.80 * budget)
    if phase4_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord_polish(work, phase4_end, max_sweeps=300000,
                                    seed=int(seed + 123456), consider=consider)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 5: Fine LSE + coordinate polish ===
    phase5_end = min(hard - 25.0, t0 + budget - 30.0)
    if phase5_end > time.monotonic() + 30.0:
        lse5_end = min(phase5_end - 30.0, time.monotonic() + 0.35 * (phase5_end - time.monotonic()))
        if lse5_end > time.monotonic() + 15.0:
            try:
                p5 = _lbfgs_optimize(work, lse5_end, consider=consider, fine_only=True)
                if p5 is not None and p5.size == n_par and np.all(np.isfinite(p5)) and p5.sum() > 0:
                    p5 = p5 / float(p5.sum())
                    consider(p5)
                    work = best_arr.copy()
                    work = work / float(work.sum())
            except Exception:
                pass

        coord5_end = phase5_end
        if coord5_end > time.monotonic() + 15.0:
            try:
                work, _ = _coord_polish(work, coord5_end, max_sweeps=200000,
                                        seed=int(seed + 987654), consider=consider)
            except Exception:
                pass

    # === Phase 6: Try a slightly different length if time permits ===
    rem = hard - 20.0 - time.monotonic()
    if rem > 120.0 and n_par != 2048:
        n2 = min(2048, max(512, int(n_par * 0.67)))
        # Resample best to n2
        xp = (np.arange(n_par, dtype=np.float64) + 0.5) / float(n_par)
        fp = np.interp((np.arange(n2, dtype=np.float64) + 0.5) / float(n2), xp, best_arr)
        p2 = np.maximum(fp, 0.0)
        p2s = float(p2.sum())
        if p2s > 0:
            p2 /= p2s
            consider(p2)
            if rem > 100.0:
                p2_end = min(hard - 25.0, time.monotonic() + rem - 30.0)
                if p2_end > time.monotonic() + 20.0:
                    try:
                        p2, _ = _coord_polish(p2, p2_end, max_sweeps=100000,
                                              seed=int(seed + 424242), consider=consider)
                    except Exception:
                        pass

    # Final squeeze on best
    fin_end = min(hard - 15.0, t0 + budget - 10.0)
    if fin_end > time.monotonic() + 20.0:
        work = best_arr.copy()
        work = work / float(work.sum())
        try:
            work, _ = _coord_polish(work, fin_end, max_sweeps=100000,
                                    seed=int(seed + 555555), consider=consider)
        except Exception:
            pass

    # Final validation and return
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
