       timer = 0.0
       start_time = time.time()
       history_best = [best_fitness]

       # Evolution loop
       for gen in range(gens):
           # Calculate maximum timer allowance for this gen
           if timer + time.time() >= start_time + budget_s:
               break
           timer = time.time() - start_time
           
           # Dynamic parameters adaptation
           f = np.clip(0.5 + 0.3 * gen/gens, 0.5, 0.8)  # Mutation factor
           cr = np.clip(0.05 + 0.02 * (gen % 3), 0.05, 0.2)  # Crossover rate
           pop_size = max(16, min(64, int(pop_size * (1-gen/gens)**0.5)))  # Adaptive population size

           # Roulette-wheel selection based on fitness
           fitness_norm = (fitness_values + 1e6) / (np.sum(fitness_values) + 1e6)
           parents_idx = np.random.choice(range(pop_size), size=pop_size, p=fitness_norm)
           selected = initial_population[parents_idx]

           next_population = []
           # Local repair for individuals with low fitness
           mask = fitness_values < history_best[-1] * 0.75
           for i, x in enumerate(selected):
               if mask[i] and np.random.random() < 0.7:
                   # Repair through coordinatewise optimization
                   grad = np.diff(np.convolve(x, x, mode='full'), 1, prepend=0) * (-1)
                   repair_sol = x - 0.05 * grad / np.max(np.abs(grad)) * np.sign(np.random.randn(n) - 0.5)
                   repair_sol = np.clip(repair_sol, 0, 1000)
                   repair_sol /= np.sum(repair_sol) * np.sum(x) / 5
                   if evaluate_fitness(repair_sol) > fitness_values[i] * 0.95:
                       selected[i] = repair_sol
                       fitness_values[i] = evaluate_fitness(selected[i])

           # Replace worst with repair solutions if necessary
           worst_idx = np.argmax(-fitness_values)
           if evaluate_fitness(best_candidate) < fitness_values[worst_idx]*0.9:
               repair_sol = repair_coarse(x = x_best, ...)
               if repair_sol and evaluate_fitness(repair_sol) < fitness_values[worst_idx]:
                   selected[worst_idx] = repair_sol
                   fitness_values[worst_idx] = evaluate_fitness(repair_sol)

           # Generate new children through crossover and mutation
           next_population = []
           for i in range(0, pop_size, 2):
               # Choose two parents
               p_idx1, p_idx2 = np.random.choice(range(pop_size), 2, replace=False)
               p1, p2 = selected[p_idx1], selected[p_idx2]

               # Mutation
               mutant = mutation(p1, f, cr)
               
               # Crossover
               child = crossover(p1, mutant)
               
               # Clipping and normalization
               total_child = np.sum(child)
               if total_child < 0.01:
                   child *= 0.01 / total_child
               next_population.append(child)

           try:
               # Local improvement - get 10 best and do gradient descent
               elite_idx = np.argsort(fitness_values)[-min(10, pop_size):]
               x_elite = next_population[i] # Use actual children
               # Limit time for gradient steps
               for _ in range(2):
                   grad = compute_gradient(x_elite)
                   x_elite = projected_gradient_descent(
                       x_elite, grad, step_size=0.05, max_iter=5,
                       constraints={'bounds': [(0,1000)]*n}
                   )
                   if evaluate_fitness(x_elite) < history_best[-1]*0.99:
                       break
               best_fitness_new = evaluate_fitness(x_elite)
               if best_fitness_new > history_best[-1]:
                   best_candidate = x_elite
                   history_best.append(best_fitness_new)
           except:
               pass  # Skip if gradient calculation fails

           # Population update
           initial_population = np.array(next_population)
           fitness_values = np.array([evaluate_fitness(x) for x in initial_population])
           timer += 2.0  # Account for next eval

           # Early termination heuristic
           if len(history_best) > 10 and np.max(history_best) < best_fitness * (1 - tol):
               break

       try:
           # Precondition and normalization
           best_candidate = best_candidate / np.sum(best_candidate) * 0.01
           best_candidate = np.clip(best_candidate, 0, 1000)
           score = evaluate_score(best_candidate)
           if not np.isfinite(score) or score >= 1.50275 or score <= history_best[-1]:
               # Fallback to parent sequence
               best_candidate = parent_normalized.copy()
               best_candidate = best_candidate / np.sum(best_candidate) * 0.01
               np.clip(best_candidate, 0, 1000)
               score = evaluate_score(best_candidate)
           return [float(x) for x in best_candidate]
       except:
           return [float(x) for x in parent_normalized]
