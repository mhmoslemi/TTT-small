def _conv(a):
    n = len(a)
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    c = irfft(F*F, fsize)[:2*n-1]
    return c

def _score_norm(a):
    """Score with sum(a)=1 normalization: 2n*max(conv)/S^2"""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    return 2.0 * n * m / (s*s)

def _conv_at_k(a, k):
    """Compute (conv(a,a))_k efficiently."""
    n = len(a)
    if k < 0 or k >= 2*n-1:
        return 0.0
    lo = max(0, k-n+1)
    hi = min(k, n-1)
    return float(np.dot(a[lo:hi+1], a[k-hi:k-lo+1][::-1]))

import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    c = irfft(F*F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)  # clip tiny negative values from FFT
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s*s)

    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_chebyshev(f, n, time_limit=8.0):
    """min t s.t. (f*g)_k <= t, sum(g)=1, g>=0"""
    A = _build_A(f, n)
    c_vec = np.zeros(n+1)
    c_vec[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)

    b_eq = np.array([1.0])
    bounds = [(0, None)]*n + [(None, None)]
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})

        sup += 0.3 * np.cos(2 * math.pi * fr * i_arr / n + rng_main.uniform(0, 2 * math.pi))
    inits.append(("mfreq", np.abs(sup) + 0.001))
    z = np.ones(n)
    z[::2] *= 1.5
    z[1::2] *= 0.6
    inits.append(("zipper", z))

    tp_i = max(12.0, (phase1_end - t0) / len(inits))
    for idx, (name, init) in enumerate(inits):
        if time.monotonic() > phase1_end - 8.0:
            break
        if best_score < 1.5028:
            break
        stop = min(phase1_end - 4.0, time.monotonic() + tp_i)
        if time.monotonic() >= stop:
            break
        try:
            res, sc = _lp_optimize(init, n, stop, seed + 7919 * (idx + 1),
                                   use_cheb=(idx == 0))
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                rc = np.clip(res, 0, 1000)
                ex = float(evaluate_sequence([float(x) for x in rc]))
                if math.isfinite(ex) and ex < best_score:
                    best_seq = rc.copy()
                    best_score = ex
        except Exception:
            pass

    # Phase 2: Other lengths
    phase2_end = t0 + budget * 0.65
    oth = sorted(set([max(2, n_p // 2), max(16, int(n_p * 0.75)), n_p,
                      min(512, n_p + 64), min(576, int(n_p * 1.25))]))
    tp_l = max(8.0, (phase2_end - phase1_end) / max(1, len(oth)))
    for L in oth:
        if time.monotonic() > phase2_end - 5.0:
            break
        stop = min(phase2_end - 3.0, time.monotonic() + tp_l)
        if time.monotonic() >= stop:
            break
        if L == n_p:
            init_l = best_seq.copy() if len(best_seq) == L else parent.copy()
        else:
            src = parent if len(parent) > 1 else np.ones(1)
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(src)), src)
        try:
            res, sc = _lp_optimize(init_l, L, stop, seed + 104729 + L)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                rc = np.clip(res, 0, 1000)
                ex = float(evaluate_sequence([float(x) for x in rc]))
                if math.isfinite(ex) and ex < best_score:
                    best_seq = rc.copy()
                    best_score = ex
        except Exception:
            pass

    # Phase 3: Intensive polishing
    phase3_end = deadline - 5.0
    if time.monotonic() < phase3_end - 15.0:
        nr = max(1, int((phase3_end - time.monotonic()) // 3))
        nr = min(nr, 8)
        pe = max(10.0, (phase3_end - time.monotonic()) / nr)
        for pidx in range(nr):
            ps = time.monotonic() + pe
            if ps > phase3_end:
                ps = phase3_end
            if time.monotonic() >= ps - 2.0:
                break
            try:
                best_seq, best_score = _polish(best_seq, ps, seed=5000 + pidx * 111)
            except Exception:
                pass
            # Evaluate exact score on polished result
            try:
                bs = np.clip(best_seq, 0, 1000)
                es = float(evaluate_sequence([float(x) for x in bs]))
                if math.isfinite(es) and es < best_score:
                    best_score = es
            except Exception:
                pass

    # Final short polish
    fs = t0 + budget - 10.0
    if time.monotonic() < fs - 4.0:
        try:
            best_seq, best_score = _polish(best_seq, fs, seed=77777)
        except Exception:
            pass

    # Final validation
    fseq = np.clip(best_seq, 0, 1000)
    try:
        fsc = float(evaluate_sequence([float(x) for x in fseq]))
    except Exception:
        fsc = math.inf
    if not math.isfinite(fsc):
        try:
            fsc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            fsc = math.inf
        if not math.isfinite(fsc):
            return [1.0]
        return [float(x) for x in fallback]
    if fsc < best_score:
        fseq = best_seq.copy()
    return [float(x) for x in fseq]
