    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)

    b = _fft_conv(a)
    m = float(np.max(b))

    return 2.0 * n * m / (s * s)
