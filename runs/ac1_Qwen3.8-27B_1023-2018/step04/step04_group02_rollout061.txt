import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _cd_polish(a, stop_time, max_outer=50, seed=42):

def _lp_direction(f, n, M, timeout=6.0):
    """Find LP direction: max sum(g) s.t. ||f*g||∞ ≤ M, g ≥ 0."""
    # Build A matrix: A[k, j] = f[(k-j) mod n]... 
    # Actually, (f*g)[k] = sum_j f[j]*g[k-j]
    # A[k, j] = f[k-j] if 0 ≤ k-j ≤ n-1, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, 
                      bounds=[(0, None)]*n,
                      method='highs',
                      options={'time_limit': timeout,

                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0, None)
            if g.sum() > 1e-15:

    except:

    return None

def _gradient_step(a, lr, n):
    """One step of projected gradient descent."""
    s = float(a.sum())
    if s < 1e-15:
        return a
    b = _conv(a)
    k = int(np.argmax(b))
    M = float(b[k])
    
    # grad[i] = 4n*(a[k-i]*s - M) / s^3 if 0<=k-i<n, else -4n*M/s^3
    grad = np.full(n, -4.0*n*M/(s**3))
    lo = max(0, k-n+1)
    hi = min(n-1, k)
    for i in range(lo, hi+1):
        grad[i] = 4.0*n*a[k-i]/(s*s) - 4.0*n*M/(s**3)
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    return a_new, k, M
