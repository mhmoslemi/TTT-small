    a is normalized to sum=1, we minimize max_k c[k].
    """

            best_a = a.copy()
        
        # Adaptive temperature

        T = T_start * (T_end / T_start) ** (1.0 - progress)
        T = max(T, T_start)

        c_shifted = T * (c - M)  # shift so max is 0, avoid overflow
        w = np.exp(c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-15:

            w[M.argmax()] = 1.0

                if 0 <= idx < n:
                    grad[j] += w[k] * a[idx]
                    if idx != j:
                        grad[idx] += w[k] * a[j]

        grad -= grad.mean()

        for alpha in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05]:

                a = np.maximum(a, 1e-15)
                a /= a.sum()
                lr = 0.01
        
        # Check deadline more frequently
        if time.monotonic() > stop_time:
            break
    
    # Final normalization
    best_a = np.maximum(best_a, 0.0)
    best_a /= best_a.sum()
    return best_a

def _grad_max_conv(a, T):
    """
    Compute gradient of softmax approximation to max(conv(a,a)).
    Returns (gradient, soft_max_value, active_weights)
    """
    n = len(a)
    c = np.convolve(a, a)
    M = float(c.max())
    
    # Softmax weights
    c_shift = T * (c - M)
    w = np.exp(c_shift)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(2*n-1)
        w[int(c.argmax())] = 1.0
    else:
        w /= w_sum

    k = np.arange(2*n-1)[:, None]
    i_idx = np.arange(n)[None, :]
    d = k - i_idx
    valid = (d >= 0) & (d < n)

    # Wait, k is (2n-1,1) and i_idx is (1,n), so k.squeeze() is (2n-1,) and i_idx.squeeze() is (n,)
    # The comparison k==2*i creates a (2n-1,n) array
    diag = valid & (k == 2 * i_idx)

    grad -= grad.mean()

def _build_conv_matrix(n, a):
    """Build (2n-1) x n convolution matrix for a."""
    k = np.arange(2*n-1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2*n-1, n))
    A[mask] = a[d[mask]]
    return A

def _lp_best_response(a, M, n, time_limit=3.0):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    A = _build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                      method='highs', options={'time_limit': time_limit})
        if res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0, None)
            if g.sum() > 1e-15:
                g /= g.sum()
                return g
    except Exception:
        pass
    return None

def _blend_line_search(a, g, M_cur, n_steps=20):
    """Find best blend (1-t)*a + t*g."""
    best_t = 0.0
    best_val = M_cur
    ts = np.linspace(0.0, 1.0, n_steps)
    for t in ts:
        cand = (1-t)*a + t*g
        cand = np.maximum(cand, 1e-15)
        cs = cand.sum()
        if cs < 1e-12:
            continue
        cand /= cs
        val = float(np.convolve(cand, cand).max())
        if val < best_val:
            best_val = val
            best_t = t
    
    # Refine around best_t
    if best_t > 0:
        lo = max(0, best_t - 0.05)
        hi = min(1, best_t + 0.05)
        for _ in range(15):
            if hi - lo < 1e-6:
                break
            mid = 0.5*(lo+hi)
            cand = (1-mid)*a + mid*g
            cand = np.maximum(cand, 1e-15)
            cand /= cand.sum()
            val = float(np.convolve(cand, cand).max())
            if val < best_val:
                best_val = val
                best_t = mid
                hi = mid
            else:
                lo = mid
    
    if best_t <= 0:
        return None, M_cur
    cand = (1-best_t)*a + best_t*g
    cand = np.maximum(cand, 1e-15)
    cand /= cand.sum()
    return cand, best_val

def _eval_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')

    return 2.0 * len(a) * float(c.max()) / (s * s)

def _softmax_grad_step(a, T):
    """One gradient step on softmax approximation of max(conv(a,a)).
    a is normalized to sum=1. Returns new_a or None if no improvement."""
    n = a.size
    c = np.convolve(a, a)
    M = float(c.max())
    
    # Softmax weights
    c_shift = T * (c - M)
    w = np.exp(c_shift)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(2 * n - 1)
        w[int(c.argmax())] = 1.0
    else:
        w /= ws
    
    # Gradient: grad[i] = sum_k w[k] * J[k,i]
    # J[k,i] = 2*a[k-i] if 0<=k-i<n, except J[2i,i]=a[i] (diagonal correction)
    k = np.arange(2 * n - 1)[:, None]
    ii = np.arange(n)[None, :]
    d = k - ii
    valid = (d >= 0) & (d < n)
    
    # Efficient: grad[i] = 2*sum_j a[j]*w[i+j] - w[2i]*a[i] (bounds permitting)
    # Use correlation trick:
    # s1[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i+j in [0, 2n-2]
    # This is cross-correlation of a and w
    # np.correlate(w, a, 'full')[k] = sum_j w[j+k]*a[j] for valid j
    # We want sum_j a[j]*w[i+j] = sum_j a[j]*w[j+i]
    # = np.correlate(a, w, 'full')... let me just use the matrix for n=281
    
    J = np.zeros((2 * n - 1, n))
    J[valid] = 2.0 * a[d[valid]]
    diag = valid & (k == 2 * ii)
    J[diag] = a[d[diag]]
    
    grad = J.T @ w
    grad -= grad.mean()  # project to sum-zero
    
    return grad, M, w

def _build_A(n, a):
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n))

    return A


def _lp_solve(A, M, n, tl=3.0):
    c_obj = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': tl})
        if res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0, None)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None
