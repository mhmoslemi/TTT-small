def _fft_conv_sq(a):
    """Full autocorrelation a*a via FFT."""
    n = len(a)
    if n <= 200:
        return np.convolve(a, a)
    m = next_fast_len(2 * n - 1)
    F = rfft(a, m)
    c = irfft(F * F, m)[:2 * n - 1]
    return c


def _score(a):

    return 2.0 * len(a) * m / (s * s)


def _build_A(f, n):
    """A[k,j] = f[j] if 0<=k-j<n else 0. Shape (2n-1, n)."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)

        A[j:j + n, j] = f
    return A


def _lp_dir(f, n, timeout=3.0):
