import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = f[:end - j]
    return A


def _lp_dir(f, n, M, timeout=5.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _subgradient_dir(a, b, max_k_list):
    """Compute average subgradient direction from active constraints."""
    n = len(a)
    grad = np.zeros(n)
    for k in max_k_list:
        # gradient of b_k w.r.t. a: component j is 2*a[k-j] if 0<=k-j<n
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            kj = k - j
            if 0 <= kj < n:
                grad[j] += 2.0 * a[kj]
    grad /= max(1, len(max_k_list))
    return grad


def _cd_polish(a, stop_time, max_outer=30, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    a = a.copy()
    cur = _score_fast(a)
    best = a.copy()
    best_sc = cur

    mults = [0.2, 0.35, 0.5, 0.65, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0, 0.0, 3.0, 0.05]

    for outer in range(max_outer):
        if time.monotonic() > stop_time - 2.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = a[i]
            best_v = orig
            best_local = cur
            for m in mults:
                if orig > 1e-15:
                    v = orig * m
                else:
                    v = 0.0 if m == 0 else 0.005 * m
                a[i] = v
                if a.sum() < 0.01:
                    a[i] = orig
                    continue
                sc = _score_fast(a)
                if sc < best_local:
                    best_local = sc
                    best_v = v
            a[i] = best_v
            if best_local < cur - 1e-14:
                cur = best_local
                improved = True
                if cur < best_sc:
                    best_sc = cur
                    best = a.copy()

        if not improved and outer >= 1:
            # Try 2-coordinate moves
            for _ in range(300):
                if time.monotonic() > stop_time - 2.0:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                oi, oj = a[i], a[j]
                bi, bj = oi, oj
                bsc = cur
                for mi in [0.3, 0.6, 1.0, 1.5, 2.5, 0.0]:
                    for mj in [0.3, 0.6, 1.0, 1.5, 2.5, 0.0]:
                        if mi == 1.0 and mj == 1.0:
                            continue
                        a[i] = oi * mi if oi > 1e-15 else 0.0
                        a[j] = oj * mj if oj > 1e-15 else 0.0

                            a[i], a[j] = oi, oj
                            continue

                            bsc = sc
                            bi, bj = a[i], a[j]
                a[i], a[j] = bi, bj
                if bsc < cur - 1e-14:
                    cur = bsc
                    improved = True
                    if cur < best_sc:
                        best_sc = cur
                        best = a.copy()
            if not improved:

def _lp_optimize(f0, n, stop_time, rng, verbose=False):
    """Main LP-based optimization loop."""
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    if len(f) < n:
        f = np.concatenate([f, np.zeros(n - len(f))])
    f = f[:n]
    if f.sum() < 1e-10:
        f = np.ones(n)
    f = f / f.sum()

    best_f = f.copy()
    best_score = _score_fast(f)
    t = 0.3
    stall = 0
    max_stall = 300
    it = 0

    while time.monotonic() < stop_time - 3.0:
        b = _conv(f)
        M = float(b.max())
        if M < 1e-18:
            break

        # Find active set (near-maximal b_k)
        thresh = M * 0.995
        active = np.where(b >= thresh)[0].tolist()
        if not active:
            active = [int(np.argmax(b))]

        g = _lp_dir(f, n, M, timeout=4.0)

        if g is None:
            # Subgradient fallback
            grad = _subgradient_dir(f, b, active)
            # Projected negative gradient: minimize grad . d s.t. d >= 0 feasible direction
            # Simple: d = -grad projected, then normalize
            d = -grad
            d = np.maximum(d, 0)
            d_sum = d.sum()
            if d_sum < 1e-15:
                stall += 3
                f = np.clip(f + rng.normal(0, 0.02 * max(1.0, f.std()), n), 0, None)
                f = f / max(1e-10, f.sum())
                if stall > max_stall:
                    break
                continue
            g = d / d_sum * f.sum()
            stall += 1

        g_sum = g.sum()
        if g_sum < 1e-15:
            stall += 3
            continue
        g = g * (f.sum() / g_sum)

        # Binary search for best step
        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < 1e-12:
            stall += 2
            if stall > max_stall:
                # Perturb and restart
                f = np.clip(best_f + rng.normal(0, 0.05 * max(1.0, np.mean(best_f)), n), 0, None)
                f = f / max(1e-10, f.sum())
                t = 0.3
                stall = 0
                max_stall = min(max_stall + 200, 800)
                continue

        # Try a range of step sizes, pick best
        step_candidates = sorted(set([t, t*0.5, t*0.25, t*1.5, t*2.0, 0.05, 0.1, 0.15]))
        improved = False
        best_new_sc = _score_fast(f)
        best_new_f = f.copy()
        for ti in step_candidates:
            f_new = np.clip((1.0 - ti) * f + ti * g, 0, None)
            if f_new.sum() < 0.01:
                continue
            sc = _score_fast(f_new)
            if sc < best_new_sc - 1e-14:
                best_new_sc = sc
                best_new_f = f_new.copy()

        if best_new_sc < best_score - 1e-14:
            f = best_new_f
            best_f = f.copy()
            best_score = best_new_sc
            improved = True
            stall = 0
            # Adapt step
            if best_new_sc < _score_fast((1.0-t)*f+ t*g) - 1e-14:
                t = min(0.7, t * 1.25)
            else:
                t = max(0.05, t * 0.8)
        elif best_new_sc < best_score:
            f = best_new_f
            stall = 0
        else:
            t = max(0.03, t * 0.6)
            stall += 1
            if stall >= max_stall:
                # Big restart from best with perturbation
                f = np.clip(best_f + rng.normal(0, 0.08 * max(1.0, np.mean(best_f)), n), 0, None)
                f = f / max(1e-10, f.sum())
                t = 0.25
                stall = 0
                max_stall = min(max_stall + 200, 1000)

        it += 1
        # Periodic CD polish
        if it % 40 == 0 and time.monotonic() < stop_time - 20.0:
            f_c, sc_c = _cd_polish(f, min(stop_time - 10.0, time.monotonic() + 8.0),
                                     max_outer=5, rng=rng)
            if sc_c < best_score:
                best_score = sc_c
                best_f = f_c.copy()
                f = f_c.copy()

    # Final CD
    if time.monotonic() < stop_time - 5.0:
        f_c, sc_c = _cd_polish(best_f, stop_time - 2.0, max_outer=15, rng=rng)
        if sc_c <= best_score + 1e-14:
            best_f = f_c
            best_score = sc_c

    return best_f, best_score
