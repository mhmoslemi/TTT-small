def _lp_dir(f, n, M, timeout=3.0):
    # Build constraint matrix: (f*g)[k] = sum_j f[j]*g[k-j] <= M
    # (f*g)[k] for k=0,...,2n-2
    # (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j]*g[k-j]
    # Let i = k-j, then g[i] appears with coefficient f[k-i]
    # So (f*g)[k] = sum_i g[i]*f[k-i] (where 0<=k-i<=n-1)
    # A[k,i] = f[k-i] if 0<=k-i<=n-1 else 0
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        # A[k, lo:hi+1] = f[k-lo : k-hi+n] ... 
        # Actually A[k,i] = f[k-i] for i in [lo, hi]
        # So for i=lo,...,hi: A[k,i] = f[k-i]
        idx = np.arange(lo, hi+1)
        A[k, idx] = f[k - idx]
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g
