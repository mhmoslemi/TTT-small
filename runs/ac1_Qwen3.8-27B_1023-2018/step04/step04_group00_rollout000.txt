        try:
            sc = float(_score(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-13:
            best_arr = a.copy()
            best_score = sc
    
    n = int(best_arr.size)
    
    # Phase 1: L-BFGS optimization (global structure)
    def _lbfgs(p, stop_time, consider_fn):
        p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
        nn = p.size
        ss = float(p.sum())
        if ss < 1e-300:
            return p
        p = p / ss
        
        conv_size = next_fast_len(2 * nn - 1)
        conv_len = 2 * nn - 1
        cross_size = next_fast_len(3 * nn - 2)
        cross_len = 3 * nn - 2
        
        def cfast(a):
            F = rfft(a, conv_size)
            c = irfft(F * F, conv_size)[:conv_len]
            np.maximum(c, 0.0, out=c)
            return c
        
        def xfast(pp, ww):
            F = rfft(pp[::-1], cross_size)
            G = rfft(ww, cross_size)
            cc = irfft(F * G, cross_size)[:cross_len]
            q = cc[nn-1:2*nn-1]
            np.maximum(q, 0.0, out=q)
            return q
        
        def softmax(u):
            m = float(np.max(u))
            e = np.exp(u - m)
            s = float(e.sum())
            return e / s if s > 0 else np.full(u.size, 1.0/u.size)
        
        def p_to_u(pp):
            uu = np.log(np.maximum(pp, 1e-300))
            return uu - float(np.mean(uu))
        
        u = p_to_u(p)
        
        def lse_func(uu, T):
            pp = softmax(uu)
            c = cfast(pp)
            cmax = float(c.max())
            if T <= 0 or not math.isfinite(T):
                T = max(1e-300, 1e-4 * cmax)
            scaled = (c - cmax) / T
            exps = np.exp(scaled)
            sum_exp = float(exps.sum())
            if sum_exp <= 0 or not math.isfinite(sum_exp):
                w = np.zeros(c.size)
                w[int(np.argmax(c))] = 1.0
                f = cmax
            else:
                f = cmax + T * float(np.log(sum_exp / float(c.size)))
                w = exps / sum_exp
            q = xfast(pp, w)
            gdp = 2.0 * q
            tdp = float(np.dot(pp, gdp))
            grad = pp * (gdp - tdp)
            grad = np.where(np.isfinite(grad), grad, 0.0)
            # Add small regularization to ensure gradient is well-defined
            return float(f), grad
        
        # Temperature stages
        c0 = cfast(p)
        cmax0 = float(c0.max()) if c0.size else 1.0
        stages = (0.1, 0.04, 0.012, 0.004, 0.0012, 0.0004)
        
        start = time.monotonic()
        avail = float(stop_time - start)
        if avail < 30:
            return p
        
        # Allocate time proportionally, more to later stages
        weights = [0.08, 0.12, 0.18, 0.22, 0.22, 0.18]
        total_w = sum(weights)
        
        best_p_local = p.copy()
        best_f_local = 1e30
        
        cum = 0
        for si, (frac, wt) in enumerate(zip(stages, weights)):
            now = time.monotonic()
            if now > stop_time - 5:
                break
            cum += wt
            end_stage = min(float(stop_time) - 3, start + avail * (cum / total_w))
            if end_stage <= now + 5:
                continue
            T = max(1e-300, frac * cmax0)
            
            # Run multiple L-BFGS iterations
            stale = 0
            while time.monotonic() < end_stage - 1:
                maxit = 30 if nn > 2000 else 50
                try:
                    res = minimize(lse_func, u, args=(T,), method='L-BFGS-B', jac=True,
                                  options={'maxiter': maxit, 'gtol': 1e-11, 'ftol': 1e-15,
                                         'maxls': 20, 'maxcor': 20, 'disp': False})
                except Exception:
                    break
                if res.x is not None and np.all(np.isfinite(res.x)):
                    fval = float(res.fun)
                    if math.isfinite(fval) and fval < best_f_local:
                        best_f_local = fval
                        u = res.x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
                    if getattr(res, 'status', 1) == 0:
                        break
                else:
                    break
            
            pp = softmax(u)
            pp = np.maximum(pp, 0.0)
            ss2 = float(pp.sum())
            if ss2 > 0:
                pp = pp / ss2
            sc = float(_score(pp))
            if math.isfinite(sc) and sc < float(_score(best_p_local)):
                best_p_local = pp.copy()
            u = p_to_u(pp)
            if consider_fn:
                consider_fn(pp)
        
        return best_p_local if np.all(np.isfinite(best_p_local)) else p

                p2 = p2 / float(p2.sum())
                consider(p2)
        except Exception:
            pass

                consider(p3)
