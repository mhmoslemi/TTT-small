def _lp_best_response(A, M, n, tl=3.0):
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                     method='highs',
                     options={'time_limit': tl})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    return np.clip(res.x, 0.0, None)

def _line_search(a, g, M_cur):
    a_n = a / a.sum()
    g_n = g / g.sum()
    
    def f(t):
        cand = (1.0-t)*a_n + t*g_n
        cand = np.maximum(cand, 0.0)
        cs = cand.sum()
        if cs < 1e-15:
            return M_cur
        cand /= cs
        return float(np.convolve(cand, cand).max())
    
    # Coarse grid
    ts = np.array([0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 0.95])
    vals = np.array([f(t) for t in ts])
    best_i = int(np.argmin(vals))
    best_t = ts[best_i]
    best_val = float(vals[best_i])
    
    if best_val >= M_cur * (1 - 1e-13):
        return None, M_cur
    
    # Refine
    lo = ts[max(0, best_i-1)]
    hi = ts[min(len(ts)-1, best_i+1)]
    
    for _ in range(40):
        if hi - lo < 1e-9:
            break
        m1 = lo + 0.382*(hi-lo)
        m2 = lo + 0.618*(hi-lo)
        f1 = f(m1)
        f2 = f(m2)
        if f1 <= f2:
            hi = m2
            if f1 < best_val:
                best_val = f1
                best_t = m1
        else:
            lo = m1
            if f2 < best_val:
                best_val = f2
                best_t = m2
    
    cand = (1.0-best_t)*a_n + best_t*g_n
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    return cand, best_val

def _optimize_core(a0, stop_time, seed=42, max_stuck=50):
    """Core LP + gradient optimization loop."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = a.size
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a /= s
    
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0
    pert_level = 0
    
    while time.monotonic() < stop_time:
        c_full = np.convolve(a, a)
        M = float(c_full.max())
        k_star = int(c_full.argmax())
        
        if M < best_M - 1e-14 * max(best_M, 1.0):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > max_stuck:
            sigma = min(0.02 + 0.015 * pert_level, 0.6)
            xx = np.arange(n, dtype=np.float64)
            mode = pert_level % 4
            
            if mode == 0:
                a *= np.exp(rng.normal(0, sigma, n))
            elif mode == 1:
                # Multi-frequency sine perturbation
                nf = rng.integers(1, 4)
                mod = np.ones(n)
                for _ in range(nf):
                    fr = rng.uniform(0.5, 6.0)
                    ph = rng.uniform(0, 2*np.pi)
                    mod *= (1 + sigma*0.4*np.sin(2*np.pi*fr*xx/n + ph))
                a *= mod
            elif mode == 2:
                # Block perturbation
                bl = n // rng.integers(3, 8)
                for _ in range(rng.integers(1, 3)):
                    start = rng.integers(0, n - bl)
                    a[start:start+bl] *= np.exp(rng.normal(0, sigma, bl))
            else:
                # Antisymmetric perturbation (breaks symmetry)
                half = n // 2
                pert = rng.normal(0, sigma, half)
                a[:half] *= np.exp(pert)
                if n % 2 == 0:
                    a[half:] *= np.exp(pert[::-1] * 0.5)
                
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            stuck = 0
            pert_level += 1
            if pert_level > 80:
                break
            continue
        
        made_move = False
        
        # LP best response
        A = _build_A(n, a)
        g = _lp_best_response(A, M*(1+1e-10), n, tl=2.5)
        
        if g is not None:
            gs = g.sum()
            if gs > 1e-15:
                g /= gs
                if np.max(np.abs(g - a)) > 1e-10:
                    cand, val = _line_search(a, g, M)
                    if cand is not None and val < M - 1e-15:
                        a = cand
                        made_move = True
        
        # Subgradient step
        if not made_move:
            grad = np.zeros(n)
            for j in range(n):
                idx = k_star - j
                if 0 <= idx < n:
                    grad[j] = a[idx]
            grad -= grad.mean()
            gn = np.linalg.norm(grad)
            if gn > 1e-14:
                for alpha in [0.03, 0.01, 0.005, 0.002, 0.001]:
                    cand = a - alpha * grad / gn
                    cand = np.maximum(cand, 0.0)
                    cs = cand.sum()
                    if cs < 1e-12:
                        continue
                    cand /= cs
                    if float(np.convolve(cand, cand).max()) < M - 1e-15:
                        a = cand
                        made_move = True
                        break
        
        if not made_move:
            a *= np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 1e-12)
            a /= a.sum()
    
    return best_a
