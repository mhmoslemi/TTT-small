def _lp_refine(f0, stop_time, max_iters=15, seed=42):
    """LP-based direction finding with line search."""
    n = len(f0)
    f = np.asarray(f0, dtype=np.float64).copy()
    S = float(f.sum())
    f = f / S  # normalize to sum=1
    S = 1.0
    c = np.convolve(f, f, mode='full')
    best_score = 2.0 * n * float(c.max())
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    # Build constraint matrix once (A is fixed for given f, but f changes)
    for it in range(max_iters):
        if time.monotonic() > stop_time - 5:
            break

        M = float(c.max())
        if M < 1e-15:
            break

        # Build A: A[k,j] = f[j] if 0 <= k-j < n else 0
        # A[k,j] = f[k-j] for max(0,k-n+1) <= j <= min(k, n-1)
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = f

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        try:
            time_limit = max(2.0, stop_time - time.monotonic() - 10)
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'time_limit': min(time_limit, 30.0)})
        except Exception:
            break

        if res.status != 0 or res.x is None:
            break

        g = np.asarray(res.x, dtype=np.float64)
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Line search: f_new = (1-t)*f + t*g
        f_curr = f.copy()
        best_t = 0.0
        best_sc = best_score
        lo, hi = 0.0, 1.0
        for _ in range(20):
            mid = (lo + hi) / 2.0
            f_try = (1 - mid) * f_curr + mid * g
            f_try = np.clip(f_try, 0, None)
            s_try = float(f_try.sum())
            if s_try < 0.01:
                hi = mid
                continue
            f_try = f_try / s_try
            c_try = np.convolve(f_try, f_try, mode='full')
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_t = mid
                lo = mid
            else:
                hi = mid

        if best_t > 1e-4 and best_sc < best_score - 1e-12:
            f = (1 - best_t) * f_curr + best_t * g
            f = np.clip(f, 0, None)
            fs = f.sum()
            if fs < 0.01:
                f = best_f.copy()
                break
            f = f / fs
            c = np.convolve(f, f, mode='full')
            best_score = best_sc
            best_f = f.copy()
        else:
            # Try perturbation
            f = np.clip(f + rng.normal(0, 0.01, n), 0, None)
            f /= f.sum()
            c = np.convolve(f, f, mode='full')
            sc = 2.0 * n * float(c.max())
            if sc < best_score:
                best_score = sc
                best_f = f.copy()

    return best_f, best_score


def _grad_descent(f0, stop_time, max_iters=5000, seed=99):
    """Gradient descent with softmax-smoothed max."""
    n = len(f0)
    f = np.asarray(f0, dtype=np.float64).copy()
    S = float(f.sum())
    f = f / S
    c = np.convolve(f, f, mode='full')
    best_score = 2.0 * n * float(c.max())
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    alpha = 0.02
    tau = 50.0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 3:
            break

        M = float(c.max())
        if M < 1e-18:
            break

        # Softmax weights
        b_shift = c - M
        p = np.exp(b_shift * tau)
        p /= p.sum()

        # Gradient of smoothed max: grad[i] = 2 * sum_k p[k] * x[k-i] (valid range)
        # = 2 * (p * f_flip)[i + n - 1]
        from scipy.fft import rfft, irfft, next_fast_len
        fsize = next_fast_len(2 * n - 1)
        P = rfft(p, fsize)
        Fr = rfft(f[::-1], fsize)
        conv_val = irfft(P * Fr, fsize)[:2 * n - 1]
        grad = 2.0 * conv_val[n - 1:2 * n - 1]

        # Gradient of score = 2n * M_smooth (since sum=1)
        # Just use grad as direction to decrease M_smooth
        f_new = f - alpha * grad
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 0.01:
            f_new = f + rng.normal(0, 0.005, n)
            f_new = np.maximum(f_new, 0)
            s_new = f_new.sum()
        f_new = f_new / s_new

        c_new = np.convolve(f_new, f_new, mode='full')
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_f = f_new.copy()
            f = f_new
            c = c_new
            alpha = min(0.05, alpha * 1.05)
        else:
            alpha = max(0.001, alpha * 0.7)
            if alpha < 0.002:
                f = np.clip(f + rng.normal(0, 0.01, n), 0, None)
                f /= f.sum()
                c = np.convolve(f, f, mode='full')
                alpha = 0.01

    return best_f, best_score

                if sc_norm < best_score:
                    best, best_score = res_norm, sc_norm
        except Exception:
            pass

    # Phase 4: More coordinate descent (another round)
    end4 = t0 + budget_s * 0.70
    if time.monotonic() < end4 - 10:
        try:
            res, sc = _coord_descent(best, end4, max_sweeps=200, n_pts=20, seed=seed + 3)
            if sc < best_score:
                best, best_score = res, sc
        except Exception:
            pass

    # Phase 5: LP refinement
    end5 = t0 + budget_s * 0.85
    if time.monotonic() < end5 - 20:
        try:
            res, sc = _lp_refine(best, end5, max_iters=8, seed=seed + 4)
            res_s = float(res.sum())
            if res_s >= 0.01:
                res_norm = res / res_s
                sc_norm = _score(res_norm)
                if sc_norm < best_score:
                    best, best_score = res_norm, sc_norm
        except Exception:
            pass

    # Phase 6: Final coordinate polish
    end6 = deadline - 5.0
    if time.monotonic() < end6 - 5:
        try:
            res, sc = _coord_descent(best, end6, max_sweeps=100, n_pts=16, seed=seed + 5)
            if sc < best_score:
                best, best_score = res, sc
        except Exception:
            pass

    # Final random-order polish
    if time.monotonic() < deadline - 3:
        try:
            res, sc = _coord_descent_random(best, deadline - 2, max_sweeps=30, n_pts=12, seed=seed + 6)
            if sc < best_score:
                best, best_score = res, sc
        except Exception:
            pass

    # Cleanup and validate
    best = np.asarray(best, dtype=np.float64)
    best = np.clip(best, 0, 1000)
    best = np.where(np.isfinite(best), best, 0.0)
    if best.size == 0 or best.sum() < 0.01:
        best = parent.copy()

    # Final exact evaluation
    try:
        final_score = float(evaluate_sequence([float(x) for x in best]))
    except Exception:
        final_score = math.inf

    if not math.isfinite(final_score):
        try:
            fb_score = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            fb_score = math.inf
        if math.isfinite(fb_score):
            return [float(x) for x in parent]
        return [1.0]

    # Also check parent as fallback
    try:
        parent_score = float(evaluate_sequence([float(x) for x in parent]))
    except Exception:
        parent_score = math.inf

    if math.isfinite(parent_score) and parent_score < final_score:
        return [float(x) for x in parent]

    return [float(x) for x in best]
