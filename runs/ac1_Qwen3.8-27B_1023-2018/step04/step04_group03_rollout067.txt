def _get_parent():
    try:

        p = np.asarray(h, dtype=np.float64).ravel()

def _ramp(n, p=1.0):
    i = np.arange(1, n + 1, dtype=np.float64)
    f = (i / n) ** p
    return f


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    a = np.clip(a, 0.0, 1000.0)
    if not np.all(np.isfinite(a)):
        return math.inf
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _lp_direction(f, n, b, peak_frac, timeout=6.0):
    M = float(b.max())
    if M < 1e-18:
        return None
    threshold = M * (1.0 - peak_frac)
    peak_idx = np.where(b >= threshold)[0]
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(b)])
    # keep the densest cluster of peaks to limit row count
    if len(peak_idx) > 60:
        order = np.argsort(b[peak_idx])[::-1]
        peak_idx = peak_idx[order[:60]]

    rows, cols, vals = [], [], []
    for idx, k in enumerate(peak_idx):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-20:
                rows.append(idx)
                cols.append(i)
                vals.append(v)
    if not rows:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    c_vec = -np.ones(n)
    b_ub = np.full(len(peak_idx), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _optimize_subgrad(f0, stop_time, seed):

def _polish(f0, stop_time, seed):

    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)

                sc = _score(f)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_val = f[i]
            if best_local < best_sc - 1e-15:
                f[i] = best_val
                o2 = best_val
                for mlt in mults_fine:
                    f[i] = o2 * mlt
                    sc = _score(f)
                    if sc < best_local - 1e-15:
                        best_local = sc
                        best_val = f[i]
                    else:
                        f[i] = o2
                f[i] = best_val
                best_sc = best_local
                improved = True
            else:
                f[i] = best_val
        if not improved:
            no_imp_rounds += 1
            if no_imp_rounds >= max_no_imp:
                break
        else:
            no_imp_rounds = 0
    return f.copy(), best_sc

    # ramps centered at the parent length but also smaller/larger for extra margin
    for n in [max(64, n_p // 2), n_p, min(1000, n_p + 128)]:
        for p in (1.0, 1.3, 1.7, 2.3):
            inits.append(_ramp(n, p=p))
            inits.append(_ramp(n, p=p)[::-1].copy())  # mirrored ramp
    # parent (clipped) and parent + ramp blend
    inits.append(parent.copy())
    inits.append(0.5 * parent / max(parent.max(), 1e-9) + 0.5 * _ramp(n_p, 1.0))
    # parent perturbed
    sig = 0.1 * max(1.0, float(parent.mean()))
    inits.append(np.clip(parent + rng.normal(0, sig, n_p), 0, None))

    # score all inits cheaply
    init_records = []
    for arr in inits:
        arr = np.clip(np.asarray(arr, dtype=np.float64), 0, 1000)
        sc = _score(arr)
        if math.isfinite(sc):
            init_records.append((sc, arr.copy()))
            _update_best(arr)
    init_records.sort(key=lambda x: x[0])

    # ---- Phase 1: LP refinement of the best initializations.
    phase1_end = t0 + budget * 0.50
    # focus LP on the best few, and on parent-length ones (matrix size)
    ranked = []
    seen_len_seen = set()
    for sc, arr in init_records:
        L = len(arr)
        # keep at most 2 per length, prefer best scores and moderate lengths
        key = (L,)
        cnt = sum(1 for _, a in ranked if len(a) == L)
        if L == n_p and cnt < 3:
            ranked.append((sc, arr))
        elif L != n_p and cnt < 1:
            ranked.append((sc, arr))
        if len([x for x in ranked]) >= 12:
            break
    ranked = sorted(set((len(a), tuple(round(v, 6) for v in a)) for _, a in
                        [(s, a) for s, a in ranked])) and ranked,
                   key=lambda x: x[0])
    # ensure at least the global best is included
    if init_records:
        ranked.append((init_records[0][0], init_records[0][1]))
    # de-duplicate by exact length+rounded values
    ded = []
    seen = set()
    for sc, arr in ranked:
        key = (len(arr),)
        if key in seen:
            continue
        seen.add(key)
        ded.append((sc, arr))
    ranked = ded[:10]

    per_lp = max(12.0, (phase1_end - t0) / max(1, len(ranked)))
    for idx, (sc, arr) in enumerate(ranked):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + per_lp)
        if time.monotonic() > stop - 3.0:
            continue
        try:
            res, rs = _optimize_lp(arr, stop, seed + 1000 * (idx + 1), peak_frac=0.03)
            _update_best(res)
        except Exception:
            pass

    # second LP pass on the current best with a tighter peak set
    if best_seq is not None and time.monotonic() < phase1_end - 20.0:
        try:
            res, rs = _optimize_lp(best_seq, phase1_end - 5.0, seed + 8888, peak_frac=0.02)
            _update_best(res)
        except Exception:
            pass

    # ---- Phase 2: subgradient on the best.
    phase2_end = t0 + budget * 0.66
    if best_seq is not None and time.monotonic() < phase2_end - 12.0:
        try:
            res, rs = _optimize_subgrad(best_seq, phase2_end - 5.0, seed + 77777)
            _update_best(res)
        except Exception:
            pass

    # ---- Phase 3: coordinate-descent polish.
    phase3_end = t0 + budget * 0.86
    if best_seq is not None and time.monotonic() < phase3_end - 12.0:
        try:
            res, rs = _polish(best_seq, phase3_end - 5.0, seed + 55555)
            _update_best(res)
        except Exception:
            pass

    # ---- Phase 4: one more LP + polish on the polished result.
    if best_seq is not None and time.monotonic() < deadline - 40.0:
        try:
            res, rs = _optimize_lp(best_seq, time.monotonic() + min(120.0, deadline - time.monotonic() - 60.0),
                                   seed + 66666, peak_frac=0.02)
            _update_best(res)
        except Exception:
            pass
        if time.monotonic() < deadline - 30.0:
            try:
                res, rs = _polish(best_seq, deadline - 15.0, seed + 99999)
                _update_best(res)
            except Exception:
                pass

    # ---- Final validation with the exact evaluator.
    if best_seq is None:
        # guaranteed-valid fallback
        best_seq = _ramp(n_p, 1.0)

    final_list = [float(x) for x in best_seq]
    try:
        final_sc = float(evaluate_sequence(final_list))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # try parent as fallback
        try:
            pl = [float(x) for x in parent]
            if math.isfinite(float(evaluate_sequence(pl))):
                return pl
        except Exception:
            pass
        return [1.0]

    return final_list
