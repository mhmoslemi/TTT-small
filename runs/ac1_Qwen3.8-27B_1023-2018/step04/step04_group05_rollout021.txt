import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize, linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    if n <= 800:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _exact_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _pnorm_obj_grad(a, p):
    """Compute p-norm score and its gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf, np.zeros(n)
    b = _conv_full(a)
    b = np.maximum(b, 0.0)
    b_p = float((b ** p).sum())
    if b_p < 1e-300:
        return 0.0, np.zeros(n)
    norm_p = b_p ** (1.0 / p)
    obj = 2.0 * n * norm_p / (s * s)
    # Gradient
    v = (b ** (p - 1)) / (norm_p ** (p - 1))
    coeff = 2.0 * n / (s * s)
    scalar = -4.0 * n * norm_p / (s ** 3)
    conv_va = np.convolve(v, a)
    ac = np.convolve(a, v[::-1])
    grad = coeff * (conv_va[:n] + ac[n - 1:2 * n - 1][::-1]) + scalar
    return obj, grad


def _pnorm_liter(a0, n, p_list, stop_time, max_iter_per_p=200):
    """Continuation L-BFGS-B on p-norm score."""
    a = np.maximum(np.asarray(a0, dtype=np.float64).ravel()[:n].copy(), 0.0)
    while a.size < n:
        a = np.concatenate([a, np.zeros(n - a.size)])
    a = a[:n]
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    best = a.copy()
    best_sc = _exact_score(a)
    bounds = [(0.0, 1000.0)] * n

    for p in p_list:
        if time.monotonic() > stop_time - 5.0:
            break

        def fun(x):
            v, _ = _pnorm_obj_grad(x, p)
            return v

        def fgrad(x):
            _, g = _pnorm_obj_grad(x, p)
            return g

        try:
            res = minimize(
                fun, a, jac=fgrad, method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": max_iter_per_p, "maxfun": 400,
                         "gtol": 1e-9, "ftol": 1e-12}
            )
            a_new = np.maximum(res.x, 0.0)
            sc = _exact_score(a_new)
            if sc < best_sc - 1e-14:
                a = a_new
                best = a.copy()
                best_sc = sc
            else:
                a = a_new  # still use as starting point for next p
        except Exception:
            pass
        a = np.clip(a, 0.0, 1000.0)

    return best.copy(), best_sc


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)

def _lp_direction(f, n, M, timeout=5.0):

                      options={"presolve": True, "time_limit": timeout,
                               "primal_feasibility_tolerance": 1e-12,
                               "dual_feasibility_tolerance": 1e-12})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g

def _make_inits(n, parent, seed):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    i = np.arange(n, dtype=np.float64)
    inits = []
    # Parent (resampled to n if needed)
    if len(parent) >= n:
        inits.append(parent[:n].astype(np.float64).copy())
    else:
        inits.append(np.interp(i, np.linspace(0, n - 1, len(parent)), parent))
    # Raised cosine (Hanning)
    inits.append(0.5 * (1 + np.cos(2 * np.pi * i / n)))
    # Tukey-like
    a_t = 0.4
    tu = np.ones(n)
    edge = int(n * a_t / 2)
    if edge > 0:
        for k in range(edge):
            tu[k] = 0.5 * (1 + np.cos(np.pi * (2 * k / a_t / n + 1) / (a_t / n)))
            tu[n - 1 - k] = tu[k]
    inits.append(tu)
    # Multi-frequency
    sup = 0.3 * np.ones(n)
    for fr, amp in [(1, 0.3), (2, 0.2), (3, 0.15), (5, 0.1), (7, 0.08), (11, 0.05)]:
        ph = rng.uniform(0, 2 * np.pi)
        sup += amp * np.cos(2 * np.pi * fr * i / n + ph)
    inits.append(np.abs(sup) + 0.01)
    # Gaussian double-lobe
    inits.append(np.exp(-0.5 * ((i - 0.3 * n) / (0.08 * n)) ** 2) +
                 np.exp(-0.5 * ((i - 0.7 * n) / (0.08 * n)) ** 2) + 0.02)
    # |Normal|
    inits.append(np.abs(rng.normal(0, 1, n)) + 0.02)
    # Uniform
    inits.append(np.ones(n))
    # Parent + noise
    sig = 0.15 * max(1.0, parent.mean())
    inits.append(np.abs(parent[:n] + rng.normal(0, sig, n)) + 0.01 if len(parent) >= n
                 else np.abs(np.interp(i, np.linspace(0, n-1, len(parent)), parent) +
                             rng.normal(0, sig, n)) + 0.01)
    # Flat-top window (approx)
    x = i / (n - 1) - 0.5
    ft = 1.0
    ft -= 0.262 * np.ones(n)
    ft += 0.231 * np.cos(2 * np.pi * x)
    ft -= 0.082 * np.cos(4 * np.pi * x)
    ft += 0.014 * np.cos(6 * np.pi * x)
    inits.append(np.maximum(ft, 0.0) + 0.005)
    return inits

                _update_best(res2)
            except Exception:
                pass

    # Phase 4: Try alternate lengths (brief)
    phase4_end = deadline - 30.0
    other_ns = [n - 32, n + 32, n - 64, n + 64]
    other_ns = [x for x in other_ns if 100 <= x <= 600]
    if other_ns and time.monotonic() < phase4_end - 20.0:
        time_per_len = max(5.0, (phase4_end - time.monotonic()) / len(other_ns))
        for L in other_ns:
            if time.monotonic() > phase4_end - 10.0:
                break
            stop_l = min(phase4_end - 5.0, time.monotonic() + time_per_len)
            try:
                init_l = np.interp(np.linspace(0, 1, L),
                                   np.linspace(0, 1, len(best_seq)), best_seq)
                res_l, sc_l = _lp_iterate(init_l, L, stop_l, seed + 50000 + L, max_iter=80)
                _update_best(res_l)
            except Exception:
                pass

    # Phase 5: Final short polish
    if time.monotonic() < deadline - 8.0:
        try:
            res_f, sc_f = _polish(best_seq, deadline - 6.0, seed=27182)
            _update_best(res_f)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            return [1.0]
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    best_score = final_sc if final_sc < best_score else best_score
    return [float(x) for x in best_seq]
