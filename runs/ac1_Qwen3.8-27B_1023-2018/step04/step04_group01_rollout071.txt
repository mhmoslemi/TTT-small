def coord_sweeps(f, conv, S, stop, sweeps, rng, max_per_sweep=None, grid=32, fine=20):
    n=len(f); conv_size=2*n-1
    f=f.copy(); conv=conv.copy(); S=S
    score=2*n*conv.max()/(S*S)
    best_f=f.copy(); best_conv=conv.copy(); best_S=S; best_score=score
    no_improve=0
    for sweep in range(sweeps):
       if time > stop: break
       current_score=score
       if max_per_sweep and max_per_sweep<n:
           idxs = rng.choice(n, size=max_per_sweep, replace=False)
       else: idxs = rng.permutation(n)
       improved=False
       for i in idxs:
           if time > stop: break
           v=f[i]
           S_rest=S-v
           start=i; end=min(i+n, conv_size); L=end-start
           if L<=0: continue
           base=conv[start:end]
           infl=2.0*f[:L]
           # unaffected C
           if start>0:
               C=conv[:start].max()
           else:
               C=0.0
           if end<conv_size:
               C2=conv[end:].max()
               if C2>C: C=C2
           # x grid
           x_hi = min(10.0, max(4.0*v, 1.0, 12.0/n, 0.05*S))
           # also ensure at least old v
           xs = np.linspace(0.0, x_hi, grid)
           if v < xs[0] or v > xs[-1]: xs = np.append(xs, v)
           xs = np.unique(xs)
           w = xs - v
           S_new = S_rest + xs
           mask = S_new > max(0.01, 1e-12)
           if not np.any(mask): continue
           xs_m = xs[mask]; w_m=w[mask]; S_m=S_new[mask]
           # base vector length L
           vals = base[None,:] + w_m[:,None] * infl[None,:]
           idx2 = 2*i
           if start <= idx2 < end:
               j = idx2-start
               # replace with max of (base + linear) and base + linear + w^2 (same as values at idx2 + w^2)
               vals[:, j] = np.maximum(vals[:, j], base[j] + 2.0*w_m*f[i] + w_m*w_m)
           mvals = vals.max(axis=1)
           if C>0:
               mvals = np.maximum(mvals, C)
           objs = 2*n*mvals/(S_m*S_m)
           # choose min; handle nan
           finite = np.isfinite(objs)
           if not np.any(finite): continue
           jbest = np.argmin(np.where(finite,objs,np.inf))
           new_sc = objs[jbest]
           if new_sc < score - 1e-13:
               xbest=xs_m[jbest]; wbest=w_m[jbest]
               f[i]=xbest; S=S_rest+xbest
               conv[start:end] = base + wbest*infl
               if start <= idx2 < end:
                   conv[idx2] += wbest*wbest
               score = new_sc
               improved=True
               if score < best_score - 1e-14:
                   best_score = score; best_f=f.copy(); best_S=S; best_conv=conv.copy()
       # end sweep
       # normalize to reduce drift
       if S <= 0: break
       f = f / S
       conv = conv / (S*S)
       S = 1.0
       score = 2*n*conv.max()
       if score < best_score - 1e-14:
           best_score = score; best_f=f.copy(); best_conv=conv.copy(); best_S=1.0
       if improved:
           no_improve=0
       else:
           no_improve += 1
       if no_improve >= 3 and sweep > 2:
           # small random perturb to escape?
           if best_score == score:
              f = best_f.copy(); conv=best_conv.copy(); S=1
           # optional perturb: f += small noise? But normalized and nonnegative.
           f = np.maximum(f + (rng.normal(0, 0.01/n, n)), 0)
           f /= f.sum()
           conv = convfun(f)
           score=2*n*conv.max()
    return best_f, best_conv, best_S, best_score

import numpy as np
import time
import math

try:
    from scipy.fft import rfft, irfft, next_fast_len
except Exception:
    import numpy.fft as _nfft
    rfft = _nfft.rfft
    irfft = _nfft.irfft

    def next_fast_len(x):
        p = 2
        while p < max(1, int(x)):
            p *= 2
        return max(16, p)

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _clean_seq(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    return a


def _conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n <= 1536 or rfft is None:
        c = np.convolve(a, a, mode="full")
    else:
        size = next_fast_len(2 * n - 1)
        fa = rfft(a, size)
        c = irfft(fa * fa, size)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _score_from(f, conv=None):
    s = float(np.sum(f))
    if s < 0.01:
        return math.inf
    if conv is None:
        conv = _conv(f)
    if conv.size == 0:
        return math.inf
    m = float(np.max(conv))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * f.shape[0] * m / (s * s)


def _normalize_scored(a):
    a = _clean_seq(a)
    if a.size == 0:
        a = np.ones(1, dtype=np.float64)
    s = float(np.sum(a))
    if s < 0.01:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.size)
    else:
        a = a / s
    conv = _conv(a)
    if conv.size:
        m = float(np.max(conv))
        if not math.isfinite(m):
            m = 0.0
    else:
        m = 0.0
    return a, conv, 1.0, 2.0 * a.size * m


def _interp_length(seq, L):
    seq = _clean_seq(seq)
    L = int(max(1, L))
    if seq.size == 0:
        return np.ones(L, dtype=np.float64)
    x0 = np.linspace(0.0, 1.0, seq.size)
    y = np.linspace(0.0, 1.0, L)
    out = np.interp(y, x0, seq)
    out = _clean_seq(out)
    s = float(out.sum())
    if s < 0.01:
        out = np.ones(L, dtype=np.float64)
    else:
        out = out / s
    return out


def _make_window(L, kind=0, rng=None):
    L = int(max(1, L))
    x = np.linspace(0.0, 1.0, L)
    if kind == 0:
        w = np.ones(L, dtype=np.float64)
    elif kind == 1:
        w = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    elif kind == 2:
        w = 0.42 - 0.5 * np.cos(2.0 * np.pi * x) - 0.08 * np.cos(4.0 * np.pi * x)
    elif kind == 3:
        w = np.exp(-8.0 * (x - 0.5) * (x - 0.5))
    elif kind == 4:
        w = 1.0 - np.abs(2.0 * x - 1.0)
    elif kind == 5:
        w = np.exp(-10.0 * (x - 0.25) * (x - 0.25)) + np.exp(-10.0 * (x - 0.75) * (x - 0.75))
    elif kind == 6:
        w = 1.0 + 0.7 * np.cos(2.0 * np.pi * x)
    elif kind == 7:
        if rng is None:
            rng = np.random.default_rng(7)
        w = rng.dirichlet(np.full(L, 1.0, dtype=np.float64))
    elif kind == 8:
        if rng is None:
            rng = np.random.default_rng(11)
        w = rng.dirichlet(np.full(L, 0.25, dtype=np.float64))
    else:
        w = np.ones(L, dtype=np.float64)

    w = _clean_seq(w)
    s = float(w.sum())
    if s < 0.01:
        w = np.ones(L, dtype=np.float64)
    else:
        w = w / s
    return w


def _lp_direction(f, time_limit=5.0):
    if linprog is None:
        return None
    f = _clean_seq(f)
    n = f.size
    if n < 2 or n > 600:
        return None
    conv = _conv(f)
    if conv.size == 0:
        return None
    M = float(np.max(conv))
    if M <= 1e-15 or not math.isfinite(M):
        return None

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    time_limit = max(1.0, float(time_limit))

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=[(0.0, None)] * n,
                method="highs",
                options={"presolve": True},
            )
        except Exception:
            return None

    if res is None or getattr(res, "x", None) is None:
        return None

    g = np.asarray(res.x, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.clip(g, 0.0, None)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs


def _subgradient(f, conv, S, max_active=32):
    f = np.asarray(f, dtype=np.float64)
    conv = np.asarray(conv, dtype=np.float64)
    n = f.size
    if n == 0 or conv.size == 0:
        return None
    S = float(S)
    if S <= 1e-12:
        return None
    M = float(np.max(conv))
    if M <= 1e-15 or not math.isfinite(M):
        return None

    tol = max(1e-14, 1e-10 * M)
    if n <= 1024:
        act = np.nonzero(conv >= M - tol)[0]
        if act.size > max_active:
            vals = conv[act]
            top = np.argsort(vals)[::-1][:max_active]
            act = act[top]
    else:
        order = np.argsort(conv)[::-1][:max_active]
        act = order[conv[order] >= M - tol]

    if act.size == 0:
        act = np.array([int(np.argmax(conv))], dtype=int)

    grad = np.zeros(n, dtype=np.float64)
    for k in act:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += f[k - hi:k - lo + 1][::-1]

    grad *= 2.0 / float(act.size)
    invS = 1.0 / S
    invS2 = invS * invS
    invS3 = invS2 * invS
    return (2.0 * n * invS2) * grad - (4.0 * n * M * invS3) * np.ones(n, dtype=np.float64)


def _subgrad_steps(a, conv, S, stop, max_steps=30, alpha=0.1):
    f, conv, S, score = _normalize_scored(a)
    f = f.copy()
    conv = conv.copy()
    S = 1.0
    n = f.size
    best_f = f.copy()
    best_conv = conv.copy()
    best_score = float(score)
    current = best_score

    for _ in range(max_steps):
        if time.monotonic() > stop:
            break
        sg = _subgradient(f, conv, S)
        if sg is None:
            break

        accepted = False
        for a_try in (alpha, 0.5 * alpha, 0.25 * alpha, 0.1 * alpha):
            cand = f - a_try * sg
            cand = np.maximum(cand, 0.0)
            cs = float(cand.sum())
            if cs < 0.01:
                continue
            cconv = _conv(cand)
            sc = 2.0 * n * float(np.max(cconv)) / (cs * cs)
            if math.isfinite(sc) and sc < current - 1e-13:
                f = cand
                conv = cconv
                S = cs
                current = sc
                alpha = min(1.0, a_try * 1.15)
                accepted = True
                break

        if not accepted:
            alpha = max(1e-5, alpha * 0.5)
            if alpha < 1e-4:
                break

        if current < best_score - 1e-13:
            best_score = current
            best_f = f.copy()
            best_conv = conv.copy()

    bf, bc, bs, bsc = _normalize_scored(best_f)
    return bf, bc, bs, bsc


def _coord_sweeps(a, conv, S, stop, sweeps=20, rng=None, max_per_sweep=None, grid=32):
    f, conv, S, score = _normalize_scored(a)
    f = f.copy()
    conv = conv.copy()
    S = 1.0
    n = f.size
    conv_size = 2 * n - 1

    best_f = f.copy()
    best_conv = conv.copy()
    best_S = 1.0
    best_score = score
    current = score

    if rng is None:
        rng = np.random.default_rng(0)

    no_improve = 0

    for sweep in range(max(1, int(sweeps))):
        if time.monotonic() > stop:
            break

        if max_per_sweep is None:
            if n <= 800:
                max_per_sweep = n
            else:
                max_per_sweep = max(200, int(600000 // max(1, n)))

        if max_per_sweep >= n:
            idxs = rng.permutation(n)
        else:
            max_per_sweep = int(max(1, min(n, max_per_sweep)))
            idxs = rng.choice(n, size=max_per_sweep, replace=False)

        sweep_improved = False

        for i in idxs:
            if time.monotonic() > stop:
                break
            i = int(i)
            v = float(f[i])
            S_rest = S - v

            start = i
            end = min(i + n, conv_size)
            L = end - start
            if L <= 0:
                continue

            base = conv[start:end]
            infl = 2.0 * f[:L]

            C = 0.0
            if start > 0 and start < conv_size:
                C = max(C, float(np.max(conv[:start])))
            if end > 0 and end < conv_size:
                C = max(C, float(np.max(conv[end:])))

            x_hi = max(4.0 * abs(v), 1.0, 12.0 / max(1, n), 0.1 * abs(S))
            x_hi = min(10.0, max(0.25, x_hi))

            if x_hi <= 1e-12:
                xs = np.array([0.0, v, 0.1], dtype=np.float64)
            else:
                xs1 = np.linspace(0.0, x_hi, max(8, int(grid)))
                small_hi = min(0.35 * x_hi, 0.5)
                xs2 = np.linspace(0.0, small_hi, max(6, int(grid * 0.5)))
                xs = np.unique(np.concatenate((xs1, xs2, np.array([v, 0.0, x_hi], dtype=np.float64))))

            xs = _clean_seq(xs)
            if xs.size == 0:
                continue

            w = xs - v
            S_new = S_rest + xs
            mask = S_new > max(1e-12, 0.01 / max(1, n))
            if not np.any(mask):
                continue

            xs = xs[mask]
            w = w[mask]
            S_new = S_new[mask]
            if xs.size == 0:
                continue

            vals = base[None, :] + w[:, None] * infl[None, :]
            idx2 = 2 * i
            if start <= idx2 < end:
                j2 = idx2 - start
                vals[:, j2] = np.maximum(vals[:, j2], base[j2] + 2.0 * w * v + w * w)

            mvals = vals.max(axis=1)
            if C > 0.0:
                np.maximum(mvals, C, out=mvals)

            objs = (2.0 * n * mvals) / (S_new * S_new)
            if not np.all(np.isfinite(objs)):
                finite = np.isfinite(objs)
                if not np.any(finite):
                    continue
                objs = np.where(finite, objs, np.inf)

            if objs.size == 0:
                continue

            jbest = int(np.argmin(objs))
            new_sc = float(objs[jbest])

            if math.isfinite(new_sc) and new_sc < current - 1e-13:
                xbest = float(xs[jbest])
                wbest = float(w[jbest])

                f[i] = xbest
                S = S_rest + xbest
                conv[start:end] = base + wbest * infl
                if start <= idx2 < end:
                    conv[idx2] += wbest * wbest

                current = new_sc
                sweep_improved = True

                if current < best_score - 1e-13:
                    best_score = current
                    best_f = f.copy()
                    best_conv = conv.copy()
                    best_S = S

        if S > 0.0:
            f = f / S
            conv = conv / (S * S)
            S = 1.0
            current = 2.0 * n * float(np.max(conv))
            if current < best_score - 1e-13:
                best_score = current
                best_f = f.copy()
                best_conv = conv.copy()
                best_S = 1.0

        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 4:
            break

    return best_f, best_conv, best_S, best_score


def _random_moves(a, stop, rng, max_trials=None):
    f, conv, S, score = _normalize_scored(a)
    f = f.copy()
    n = f.size
    best_f = f.copy()
    best_score = float(score)

    if max_trials is None:
        max_trials = max(120, min(2500, int(1000000 // max(1, n))))
    max_trials = int(max_trials)

    for _ in range(max_trials):
        if time.monotonic() > stop:
            break
        if n < 3:
            break

        cand = f.copy()
        kmax = max(1, min(60, n // 20 + 2))
        k = int(rng.integers(1, kmax + 1))
        idx = rng.choice(n, size=k, replace=False)
        factors = rng.uniform(0.2, 3.0, size=k)
        cand[idx] *= factors
        cand = _clean_seq(cand)
        cs = float(cand.sum())
        if cs < 0.01:
            continue
        cand = cand / cs
        cconv = _conv(cand)
        sc = 2.0 * n * float(np.max(cconv))
        if math.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_f = cand.copy()

    bf, bc, bs, bsc = _normalize_scored(best_f)
    return bf, bc, bs, bsc


def _quick_opt(a, stop, use_lp=False, rng=None, max_lps=6, cd_sweeps=25, sub_steps=15,
               max_per_sweep=None, grid=32):
    if rng is None:
        rng = np.random.default_rng(0)

    f, conv, S, score = _normalize_scored(a)
    f = f.copy()
    conv = conv.copy()
    S = 1.0
    n = f.size

    best_f = f.copy()
    best_conv = conv.copy()
    best_S = 1.0
    best_score = float(score)

    if use_lp and n <= 600 and time.monotonic() < stop - 45:
        lp_stop = min(stop - 90, time.monotonic() + max(20.0, 0.35 * (stop - time.monotonic())))
        for _ in range(int(max_lps)):
            if time.monotonic() > lp_stop - 5:
                break
            tl = max(1.0, min(8.0, 0.65 * max(0.0, lp_stop - time.monotonic())))
            g = _lp_direction(f, tl)
            if g is None:
                continue

            current = 2.0 * n * float(np.max(conv))
            for t in (1.0, 0.85, 0.70, 0.50, 0.35, 0.20, 0.10):
                cand = (1.0 - t) * f + t * g
                cand = _clean_seq(cand)
                csum = float(cand.sum())
                if csum < 0.01:
                    continue
                cand = cand / csum
                cconv = _conv(cand)
                sc = 2.0 * n * float(np.max(cconv))
                if math.isfinite(sc) and sc < current - 1e-13:
                    f = cand
                    conv = cconv
                    S = 1.0
                    current = sc
                    break

            if current < best_score - 1e-13:
                best_score = current
                best_f = f.copy()
                best_conv = conv.copy()
                best_S = 1.0

    if time.monotonic() < stop - 18:
        coord_stop = min(stop - 40, time.monotonic() + max(15.0, 0.45 * (stop - time.monotonic())))
        cf, cc, cs, cs_sc = _coord_sweeps(
            f, conv, S, coord_stop,
            sweeps=cd_sweeps, rng=rng,
            max_per_sweep=max_per_sweep, grid=grid
        )
        if float(cs_sc) < best_score - 1e-13:
            best_score = float(cs_sc)
            best_f = cf
            best_conv = cc
            best_S = cs

    if time.monotonic() < stop - 30 and n <= 1600:
        sub_stop = min(stop - 25, time.monotonic() + max(5.0, 0.20 * (stop - time.monotonic())))
        sf, scv, ss, ss_sc = _subgrad_steps(best_f, best_conv, best_S, sub_stop, max_steps=sub_steps)
        if float(ss_sc) < best_score - 1e-13:
            best_score = float(ss_sc)
            best_f = sf
            best_conv = scv
            best_S = ss

    if time.monotonic() < stop - 10:
        cf, cc, cs, fs_sc = _coord_sweeps(
            best_f, best_conv, best_S, max(time.monotonic() + 0.1, stop - 5),
            sweeps=max(5, cd_sweeps // 2), rng=rng,
            max_per_sweep=max_per_sweep, grid=grid
        )
        if float(fs_sc) < best_score - 1e-13:
            best_score = float(fs_sc)
            best_f = cf
            best_conv = cc
            best_S = cs

    bf, bc, bs, bsc = _normalize_scored(best_f)
    return bf, bc, bs, bsc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(60.0, float(budget_s))
    buffer = min(30.0, max(5.0, 0.05 * budget))
    deadline = t0 + max(1.0, budget - buffer)
    rng = np.random.default_rng(int(seed) if seed is not None else 42)

    raw = globals().get("height_sequence_1", None)
    if raw is None:
        raw = np.ones(400, dtype=np.float64)

    parent, _, _, _ = _normalize_scored(raw)
    parent_n = parent.size

    fallback, _, _, _ = _normalize_scored(parent)
    fallback = fallback.copy()

    best_seq = fallback.copy()
    best_score = math.inf

    def consider(a):
        nonlocal best_seq, best_score
        a = _clean_seq(a)
        if a.size == 0 or float(a.sum()) < 0.01:
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in a]))
        except Exception:
            sc = math.inf
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()

    consider(parent)
    if not math.isfinite(best_score):
        best_seq = np.ones(400, dtype=np.float64)
        consider(best_seq)
    if not math.isfinite(best_score):
        return [1.0]

    # Phase 1: primary optimization on parent length.
    phase1_stop = min(deadline, t0 + 0.42 * budget)
    if time.monotonic() < phase1_stop - 20:
        grid = 40 if parent_n <= 600 else 28
        try:
            res = _quick_opt(
                parent, phase1_stop,
                use_lp=True,
                rng=rng,
                max_lps=24,
                cd_sweeps=55,
                sub_steps=25,
                grid=grid,
            )
            consider(res[0])
        except Exception:
            pass

    # Phase 2: alternative initializations at parent length.
    phase2_stop = min(deadline, t0 + 0.56 * budget)
    if time.monotonic() < phase2_stop - 35:
        alt_inits = []
        for kind in (2, 1, 3, 5, 4, 7, 8, 6):
            try:
                alt_inits.append(_make_window(parent_n, kind, rng))
            except Exception:
                pass

        try:
            pert = parent + rng.normal(0.0, 0.02 / max(1, parent_n), parent_n)
            pert = _clean_seq(pert)
            ps = float(pert.sum())
            if ps > 0.01:
                alt_inits.append(pert / ps)
        except Exception:
            pass

        alt_inits = alt_inits[:5]
        for idx, w in enumerate(alt_inits):
            if time.monotonic() > phase2_stop - 30:
                break
            rem = max(0.0, phase2_stop - time.monotonic())
            alloc = max(20.0, min(70.0, rem / max(1, len(alt_inits) - idx)))
            stop = min(phase2_stop - 12, time.monotonic() + alloc)
            try:
                res = _quick_opt(
                    w, stop,
                    use_lp=(idx < 2 and parent_n <= 500),
                    rng=rng,
                    max_lps=4,
                    cd_sweeps=22,
                    sub_steps=10,
                    grid=32,
                )
                consider(res[0])
            except Exception:
                pass

    # Phase 3: nearby and larger lengths.
    phase3_stop = min(deadline, t0 + 0.84 * budget)
    if time.monotonic() < phase3_stop - 50:
        lengths = []
        candidates = [
            parent_n,
            int(parent_n * 0.75),
            int(parent_n * 0.875),
            int(parent_n * 1.125),
            int(parent_n * 1.25),
            int(parent_n * 1.5),
            512, 768, 1024, 1280,
        ]
        for L in candidates:
            L = int(L)
            if 64 <= L <= 2048:
                lengths.append(L)
        lengths = sorted(set(lengths), key=lambda x: abs(x - parent_n))
        lengths = [L for L in lengths if L != parent_n][:5]

        for idx, L in enumerate(lengths):
            if time.monotonic() > phase3_stop - 35:
                break
            rem = max(0.0, phase3_stop - time.monotonic())
            alloc = max(25.0, min(95.0, rem / max(1, len(lengths) - idx)))
            stop = min(phase3_stop - 18, time.monotonic() + alloc)
            use_lp = L <= 500
            grid = 32 if L <= 800 else 24

            try:
                init = _interp_length(best_seq, L)
                res = _quick_opt(
                    init, stop,
                    use_lp=use_lp,
                    rng=rng,
                    max_lps=5 if use_lp else 0,
                    cd_sweeps=20,
                    sub_steps=10,
                    grid=grid,
                )
                consider(res[0])
            except Exception:
                pass

            if time.monotonic() < phase3_stop - 35:
                try:
                    stop2 = min(phase3_stop - 12, time.monotonic() + max(15.0, 0.25 * (phase3_stop - time.monotonic())))
                    w = _make_window(L, 2 if L < 1024 else 3, rng)
                    res2 = _quick_opt(
                        w, stop2,
                        use_lp=use_lp,
                        rng=rng,
                        max_lps=2 if use_lp else 0,
                        cd_sweeps=14,
                        sub_steps=6,
                        grid=max(20, grid - 6),
                    )
                    consider(res2[0])
                except Exception:
                    pass

    # Phase 4: final intensive polish on the current best.
    final_stop = deadline - 4.0
    if time.monotonic() < final_stop - 15:
        try:
            f, conv, S, _ = _normalize_scored(best_seq)
            grid = 36 if best_seq.size <= 800 else 24
            cf, cc, cs, fs = _coord_sweeps(
                f, conv, S, final_stop - 8,
                sweeps=120, rng=rng, grid=grid
            )
            consider(cf)
        except Exception:
            pass

    if time.monotonic() < final_stop - 10:
        try:
            trials = max(120, min(1800, int(800000 // max(1, best_seq.size))))
            rf, rc, rs, _ = _random_moves(best_seq, final_stop - 4, rng, max_trials=trials)
            consider(rf)
        except Exception:
            pass

    best_seq = _clean_seq(best_seq)
    s = float(best_seq.sum())
    if best_seq.size == 0 or s < 0.01:
        best_seq = fallback.copy()
        s = float(best_seq.sum())

    if s > 0.0:
        best_seq = best_seq / s

    try:
        out = [float(x) for x in best_seq]
        final_sc = float(evaluate_sequence(out))
    except Exception:
        out = None
        final_sc = math.inf

    if out is None or not math.isfinite(final_sc):
        try:
            out = [float(x) for x in fallback]
            sc2 = float(evaluate_sequence(out))
        except Exception:
            return [1.0]
        if not math.isfinite(sc2):
            return [1.0]

    return out
