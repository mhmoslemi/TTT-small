    cur = _score_fast(f)
    if not math.isfinite(cur):
        return f, cur
    best_f = f.copy()
    best_sc = cur

    mults = [0.0, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5, 4.0]
    no_improve = 0

    for p in range(max_passes):
        if time.monotonic() > deadline - 2.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline - 2.0:
                break
            orig = f[i]
            cands = set()
            if orig > 1e-12:
                for m in mults:
                    cands.add(orig * m)
            else:
                base = max(1e-3, 0.5 * f.mean()) if f.mean() > 1e-12 else 1e-3
                for m in [0.1, 0.3, 1.0, 3.0, 10.0]:
                    cands.add(base * m)
            cands.add(orig)

            best_v = orig
            best_l = cur
            for v in cands:
                f[i] = v
                s = f.sum()
                if s < 0.01:
                    f[i] = orig
                    continue
                sc = _score_fast(f)
                if sc < best_l - 1e-14:
                    best_l = sc
                    best_v = v
            f[i] = best_v
            if best_l < cur - 1e-13:
                cur = best_l
                improved = True
                if cur < best_sc - 1e-14:
                    best_sc = cur
                    best_f = f.copy()

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

    return best_f, best_sc


def _cd_2coord(f, deadline, n_tries=500, seed=1):
    n = len(f)
    rng = np.random.default_rng(seed)
    f = np.asarray(f, dtype=np.float64).copy()
    cur = _score_fast(f)
    best_f = f.copy()
    best_sc = cur

    for _ in range(n_tries):
        if time.monotonic() > deadline - 2.0:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        oi, oj = f[i], f[j]
        bi, bj = oi, oj
        bsc = cur
        for mi in [0.0, 0.3, 0.7, 1.3, 2.0, 1.0]:
            for mj in [0.0, 0.3, 0.7, 1.3, 2.0, 1.0]:
                if mi == 1.0 and mj == 1.0:
                    continue
                f[i] = oi * mi if oi > 1e-15 else 0.0
                f[j] = oj * mj if oj > 1e-15 else 0.0
                if f.sum() < 0.01:
                    f[i], f[j] = oi, oj
                    continue
                sc = _score_fast(f)
                if sc < bsc - 1e-13:
                    bsc = sc
                    bi, bj = f[i], f[j]
        f[i], f[j] = bi, bj
        if bsc < cur - 1e-13:
            cur = bsc
            if cur < best_sc:
                best_sc = cur
                best_f = f.copy()

    return best_f, best_sc


def _lp_loop(f0, n, deadline, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    if f.sum() < 1e-10:
        f = np.ones(n)
    f = f / f.sum()

    best_f = f.copy()
    best_sc = _score_fast(f)
    t = 0.25
    stall = 0
    max_stall = 300
    it = 0
    use_relaxed = False

    while time.monotonic() < deadline - 4.0:
        b = _fast_conv(f)
        M = float(b.max())
        if M < 1e-18:
            break

        # Try LP (full or relaxed)
        if not use_relaxed:
            g = _lp_dir(f, n, M, timeout=4.0)
            if g is None:
                use_relaxed = True
        else:
            n_active = min(80, 2 * n - 1)
            g = _lp_dir_relaxed(f, n, M, n_active, timeout=2.5)
            if g is None:
                g = _lp_dir(f, n, M, timeout=3.0)

        if g is None:
            stall += 5
            if stall > max_stall:
                break
            # Perturb
            f = np.clip(best_f + rng.normal(0, 0.02, n), 0, None)
            if f.sum() < 1e-10:
                f = best_f.copy()
            f = f / f.sum()
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            stall += 3
            continue
        g = g * (f.sum() / g_sum)

        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < 1e-11:
            stall += 3
            if stall > max_stall // 2:
                # Try CD instead
                fc, sc_c = _cd_polish(f, min(deadline - 5, time.monotonic() + 8),
                                       max_passes=5, seed=seed + it)
                if sc_c < best_sc:
                    best_sc, best_f = sc_c, fc.copy()
                    f = fc.copy()
                    stall = 0
            if stall > max_stall:
                break
            continue

        improved = False
        step_list = [t, t * 0.6, t * 0.3, t * 1.5, t * 0.15]
        for ti in step_list:
            f_new = np.clip((1.0 - ti) * f + ti * g, 0, None)
            if f_new.sum() < 0.01:
                continue
            sc = _score_fast(f_new)
            if sc < best_sc - 1e-13:
                f = f_new
                best_f = f_new.copy()
                best_sc = sc
                improved = True
                t = min(0.85, ti * 1.12)
                stall = 0
                break
            elif sc < _score_fast(f) - 1e-13:
                f = f_new
                improved = True
                t = min(0.85, ti)
                stall = 0
                break
            else:
                t = max(0.02, t * 0.5)

        if not improved:
            stall += 1
            if stall >= 20 and stall % 40 == 0:
                fc, sc_c = _cd_polish(f, min(deadline - 5, time.monotonic() + 6),
                                       max_passes=4, seed=seed + it * 7)
                if sc_c < best_sc:
                    best_sc, best_f = sc_c, fc.copy()
                    f = fc.copy()
                    stall = 0
            if stall >= max_stall:
                f = np.clip(best_f + rng.normal(0, 0.04 * max(1.0, best_f.mean()), n), 0, None)
                if f.sum() < 1e-10:
                    f = best_f.copy()
                f = f / f.sum()
                t = 0.25
                stall = 0
                max_stall = min(max_stall + 100, 600)
                use_relaxed = False

        it += 1

    # Final CD
    if time.monotonic() < deadline - 4.0:
        fc, sc_c = _cd_polish(best_f, deadline - 2, max_passes=15, seed=seed + 9999)
        if sc_c <= best_sc + 1e-13:
            best_f, best_sc = fc, sc_c

    return best_f, best_sc


def _make_inits(parent, n, rng):
    inits = []
    inits.append(parent[:n].copy() if len(parent) >= n else np.pad(parent, (0, max(0, n - len(parent)))))
    i_arr = np.arange(n, dtype=np.float64)
    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / n)))
    # Parabolic
    ctr = (n - 1) / 2.0
    inits.append(np.maximum(0.0, 1.0 - ((i_arr - ctr) / (0.5 * n)) ** 2))
    # Two-lobe
    inits.append(np.exp(-0.5 * ((i_arr - 0.25 * n) / (0.08 * n)) ** 2) +
                 np.exp(-0.5 * ((i_arr - 0.75 * n) / (0.08 * n)) ** 2))
    # Constant
    inits.append(np.ones(n))
    # Parent with noise
    sig = 0.15 * max(1.0, np.mean(parent[:n]) if len(parent) >= n else 1.0)
    p = parent[:n].copy() if len(parent) >= n else np.pad(parent, (0, n - len(parent)))
    inits.append(np.abs(p + rng.normal(0, sig, n)) + 1e-5)
    # Broad Gaussian
    inits.append(np.exp(-0.5 * ((i_arr - n / 2) / (n / 3)) ** 2))
    # Cosine envelope
    inits.append(np.abs(np.cos(2 * np.pi * i_arr / (n // 2))))
    # Filter
    filtered = []
    for ix in inits:
        ix = np.clip(np.asarray(ix, dtype=np.float64), 0, None)
        if ix.sum() < 1e-10:
            continue
        filtered.append(ix)
    return filtered


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(60.0, float(budget_s))
    deadline = t0 + budget - 15.0

    # Parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(400, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(400, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    n_parent = len(parent)

    # Fallback
    fallback = parent.copy()
    try:
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(100)
        fallback_sc = _score_fast(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_optimize(init, n, stop, sd):
        nonlocal best_seq, best_score
        try:
            res, sc = _lp_loop(init, n, stop, sd)
        except Exception:
            return
        if res is None or len(res) == 0:
            return
        res = np.clip(np.asarray(res, dtype=np.float64), 0, 1000)
        if res.sum() < 0.01:
            return
        if sc < best_score - 1e-12:
            try:
                exact = float(evaluate_sequence([float(x) for x in res]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-12:
                best_seq = res.copy()
                best_score = exact

    rng = np.random.default_rng(seed)
    inits = _make_inits(parent, n_parent, rng)

    # Phase 1: Main LP on parent length (45%)
    phase1_end = t0 + budget * 0.45
    time_per = max(10.0, (phase1_end - t0) / max(1, len(inits)))
    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop_i = min(phase1_end - 5.0, time.monotonic() + time_per)
        _try_optimize(init.copy(), n_parent, stop_i, seed + 1000 * (idx + 1))

    # Phase 2: Warm restart on best (15%)
    phase2_end = t0 + budget * 0.60
    if time.monotonic() < phase2_end - 15.0:
        n_best = len(best_seq)
        stop_p2 = min(phase2_end - 4.0, time.monotonic() + (phase2_end - time.monotonic()) * 0.85)
        _try_optimize(best_seq.copy(), n_best, stop_p2, seed + 55555)

    # Phase 3: CD polish (15%)
    phase3_end = t0 + budget * 0.75
    if time.monotonic() < phase3_end - 5.0:
        try:
            fc, sc_c = _cd_polish(best_seq, phase3_end - 2, max_passes=60, seed=seed * 3)
        except Exception:
            fc, sc_c = best_seq.copy(), best_score
        if sc_c < best_score - 1e-12:
            try:
                exact = float(evaluate_sequence([float(x) for x in fc]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-12:
                best_score = exact
                best_seq = np.clip(fc, 0, 1000).copy()

        # 2-coord CD
        if time.monotonic() < phase3_end - 3.0:
            try:
                fc2, sc_c2 = _cd_2coord(best_seq, phase3_end - 2, n_tries=800, seed=seed + 77)
            except Exception:
                fc2, sc_c2 = best_seq.copy(), best_score
            if sc_c2 < best_score - 1e-12:
                try:
                    exact = float(evaluate_sequence([float(x) for x in fc2]))
                except Exception:
                    exact = math.inf
                if math.isfinite(exact) and exact < best_score - 1e-12:
                    best_score = exact
                    best_seq = np.clip(fc2, 0, 1000).copy()

    # Phase 4: Try different lengths (15%)
    phase4_end = t0 + budget * 0.90
    if time.monotonic() < phase4_end - 15.0:
        n_best = len(best_seq)
        for L in [n_best - 50, n_best + 50, n_best // 2, n_best * 2]:
            if L < 50 or time.monotonic() > phase4_end - 12.0:
                continue
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n_best), best_seq)
            stop_l = min(phase4_end - 5.0, time.monotonic() + max(8.0, (phase4_end - time.monotonic()) * 0.4))
            _try_optimize(init_l, L, stop_l, seed + 90000 + L)

    # Phase 5: Final CD + LP refinement (10%)
    final_end = deadline - 5.0
    if time.monotonic() < final_end - 10.0:
        n_f = len(best_seq)
        stop_f = min(final_end - 3.0, time.monotonic() + (final_end - time.monotonic()) * 0.6)
        _try_optimize(best_seq.copy(), n_f, stop_f, seed + 123456 + 7)

        if time.monotonic() < deadline - 5.0:
            try:
                fc, sc_c = _cd_polish(best_seq, deadline - 3, max_passes=40, seed=seed * 7 + 42)
            except Exception:
                fc, sc_c = best_seq.copy(), best_score
            if sc_c < best_score - 1e-12:
                try:
                    exact = float(evaluate_sequence([float(x) for x in fc]))
                except Exception:
                    exact = math.inf
                if math.isfinite(exact) and exact < best_score - 1e-12:
                    best_score = exact
                    best_seq = np.clip(fc, 0, 1000).copy()

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0, 1.0, 1.0]
        best_seq = fallback
        best_score = final_sc

    best_score = min(best_score, final_sc)

    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    if not result:
        result = [1.0]
    if any(math.isnan(x) or math.isinf(x) for x in result):
        result = [float(x) for x in fallback]

    return result
