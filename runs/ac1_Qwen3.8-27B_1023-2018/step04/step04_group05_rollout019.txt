        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, 1e6)

        return None

                f, best_f, best_sc = fn, fn.copy(), sc
                ok = True
                t = min(0.9, tc * 1.25)
                stall = 0
                break
        if not ok:
            stall += 1
            t = max(0.01, t * 0.55)
            if stall >= ms:
                f = np.clip(best_f + rng.normal(0, 0.04 * max(1, best_f.mean()), n), 0, None)
                f = f / max(f.sum(), 1e-10) * ts
                stall, ms, t = 0, ms + 60, 0.3
    return best_f, best_sc

def _grad_polish(a, stop, seed=0):
    n = len(a)
    f = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    best = f.copy()
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    lr = 0.005
    for _ in range(20000):
        if time.monotonic() > stop - 4:
            break
        S = f.sum()
        if S < 0.01:
            break
        grad_M, M = _subgrad_M(f)
        grad = 2.0 * n * grad_M / (S * S) - 4.0 * n * M / (S * S * S) * np.ones(n)
        f_new = f - lr * grad
        f_new = np.clip(f_new, 0, None)
        if f_new.sum() < 0.01:
            lr = max(lr * 0.5, 1e-8)
            f = best.copy()
            continue
        sc = _score(f_new)
        if sc < best_sc - 1e-14:
            best, best_sc = f_new.copy(), sc
            f = f_new
            lr = min(lr * 1.08, 0.5)
        else:
            lr = max(lr * 0.5, 1e-9)
            if lr < 1e-8:
                break
            f = best.copy()
            if rng.random() < 0.3:
                f = np.clip(best + rng.normal(0, 0.02, n), 0, None)
    return best, best_sc

def _polish(a, stop, seed=0):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    best = best / max(best.sum(), 1e-10) * math.sqrt(2.0 * n)
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    ni = 0
    max_rounds = 30

    while time.monotonic() < stop - 5 and ni < max_rounds:
        imp = False
        # Single-coordinate golden section
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop - 5:
                break
            old = best[i]
            bv, bvs = old, best_sc
            # Coarse grid
            for m in (0.0, 0.05, 0.1, 0.15, 0.25, 0.4, 0.55, 0.7, 0.85, 1.0, 1.15, 1.35, 1.6, 2.0, 3.0, 5.0):
                v = 0.0 if m == 0.0 else old * m
                best[i] = v
                if best.sum() < 0.01:
                    best[i] = old
                    continue
                sc = _score(best)
                if sc < bvs - 1e-14:
                    bv, bvs = v, sc
            # Golden section refine
            if bv > 1e-12:
                lo, hi = bv * 0.75, bv * 1.25
                if hi - lo < 1e-8:
                    hi = lo + 0.01
                for _ in range(25):
                    if time.monotonic() > stop - 5:
                        break
                    m1 = lo + (hi - lo) * 0.381966
                    m2 = lo + (hi - lo) * 0.618034
                    best[i] = m1
                    s1 = _score(best) if best.sum() >= 0.01 else math.inf
                    best[i] = m2
                    s2 = _score(best) if best.sum() >= 0.01 else math.inf
                    if s1 < s2:
                        hi = m2
                    else:
                        lo = m1
                    bmf = min(s1, s2)
                    if bmf < bvs - 1e-15:
                        bvs = bmf
                        bv = m1 if s1 < s2 else m2
                    if hi - lo < 1e-9 * max(1.0, bv):
                        break
            best[i] = bv
            if bvs < best_sc - 1e-14:
                best_sc = bvs
                imp = True

        # Pair-coordinate search
        npairs = 400 if ni < 8 else 200
        for _ in range(npairs):
            if time.monotonic() > stop - 5:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            if oi < 1e-14 and oj < 1e-14:
                continue
            for mi in (0.0, 0.2, 0.4, 0.6, 0.8, 1.2, 1.6, 2.5):
                for mj in (0.0, 0.2, 0.4, 0.6, 0.8, 1.2, 1.6, 2.5):
                    if mi == 1.0 and mj == 1.0:
                        continue
                    best[i] = oi * mi if (mi > 0 and oi > 0) else 0.0
                    best[j] = oj * mj if (mj > 0 and oj > 0) else 0.0
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        imp = True
            best[i], best[j] = oi, oj

        # Random single-coordinate perturbations
        if not imp:
            for _ in range(80):
                if time.monotonic() > stop - 5:
                    break
                k = int(rng.integers(n))
                ol = best[k]
                sig = 0.03 * max(1.0, float(best.mean()))
                best[k] = max(0.0, ol + rng.normal(0, sig))
                if best.sum() >= 0.01:
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        imp = True
                    else:
                        best[k] = ol
                else:
                    best[k] = ol

        ni += 0 if imp else 1
    return best, best_sc

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, min(float(budget_s), 1300.0))
    dl = t0 + budget - 14.0

    parent = _get_parent()
    np_ = len(parent)

    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not math.isfinite(best_score):
        best_seq = np.ones(100)
        best_score = _score(best_seq)

    def _try(n, init, stop, sd):
        nonlocal best_seq, best_score
        if time.monotonic() > stop - 3:
            return
        try:
            res, sc = _lp_run(init, n, stop, sd)
        except Exception:
            return
        if res is None or len(res) == 0 or not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0.0, 1000.0)
        if res.sum() < 0.01:
            return
        if sc < best_score - 1e-12:
            try:
                ex = float(evaluate_sequence([float(x) for x in res]))
            except Exception:
                ex = math.inf
            if math.isfinite(ex) and ex < best_score - 1e-14:
                best_seq, best_score = res.copy(), ex

    rng = np.random.default_rng(seed)
    i_n = np.arange(np_, dtype=np.float64)

    # Phase 1: LP on parent length with diverse initializations
    p1_end = t0 + budget * 0.44
    inits = []
    inits.append(parent.copy())
    inits.append(np.ones(np_))
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i_n / np_)))
    inits.append(np.exp(-0.5 * ((i_n - (np_ - 1) / 2.0) / (np_ / 5.0)) ** 2))
    inits.append(np.abs(rng.normal(0, 1, np_)) + 0.02)
    inits.append(np.power(i_n + 1, -0.25) + 0.01)
    inits.append(1.0 + 0.5 * np.cos(np.pi * i_n / max(1, np_ - 1)))
    inits.append(np.clip(parent + rng.normal(0, 0.12, np_), 0, None) + 0.01)
    two = (np.exp(-0.5 * ((i_n - np_ * 0.3) / (np_ * 0.09)) ** 2) +
           np.exp(-0.5 * ((i_n - np_ * 0.7) / (np_ * 0.09)) ** 2))
    inits.append(two)
    x_c = np.clip(np.abs(i_n - (np_ - 1) / 2.0) / (np_ / 4.0), 0, 1)
    inits.append((1 - x_c) ** 3 + 0.01)

    tpi = max(12.0, (p1_end - t0) / len(inits))
    for idx, ini in enumerate(inits):
        if time.monotonic() > p1_end - 8:
            break
        stop = min(p1_end - 5, time.monotonic() + tpi)
        _try(np_, ini, stop, seed + 7777 * (idx + 1))

    # Phase 2: Try a few other lengths
    p2_end = t0 + budget * 0.62
    other_lens = [256, 320, 384, 448, 512]
    tpi2 = max(12.0, (p2_end - t0) / len(other_lens))
    for L in other_lens:
        if time.monotonic() > p2_end - 8:
            break
        init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
        stop = min(p2_end - 5, time.monotonic() + tpi2)
        _try(L, init_l, stop, seed + 50000 + L)

    # Phase 3: Re-optimize the best so far (warm restart)
    p3_end = t0 + budget * 0.78
    if time.monotonic() < p3_end - 20:
        _try(len(best_seq), best_seq.copy(), p3_end - 5, seed + 999999)

    # Phase 4: Gradient polish
    p4_end = t0 + budget * 0.86
    if time.monotonic() < p4_end - 12:
        try:
            g_seq, g_sc = _grad_polish(best_seq, p4_end, seed=31337)
            if g_sc < best_score - 1e-14:
                ex = float(evaluate_sequence([float(x) for x in g_seq]))
                if math.isfinite(ex) and ex < best_score - 1e-14:
                    best_seq, best_score = g_seq, ex
        except Exception:
            pass

    # Phase 5: Coordinate-descent polish
    p5_end = dl - 12
    if time.monotonic() < p5_end - 15:
        try:
            pl_seq, pl_sc = _polish(best_seq, p5_end, seed=4242)
            if pl_sc < best_score - 1e-14:
                ex = float(evaluate_sequence([float(x) for x in pl_seq]))
                if math.isfinite(ex) and ex < best_score - 1e-14:
                    best_seq, best_score = pl_seq, ex
        except Exception:
            pass

    # Phase 6: Second gradient polish + mini-polish
    if time.monotonic() < dl - 10:
        try:
            g2, g2s = _grad_polish(best_seq, dl - 8, seed=77701)
            if g2s < best_score - 1e-14:
                ex = float(evaluate_sequence([float(x) for x in g2]))
                if math.isfinite(ex) and ex < best_score - 1e-14:
                    best_seq, best_score = g2, ex
        except Exception:
            pass
        if time.monotonic() < dl - 8:
            try:
                pl2, pl2s = _polish(best_seq, dl - 6, seed=88888)
                if pl2s < best_score - 1e-14:
                    ex = float(evaluate_sequence([float(x) for x in pl2]))
                    if math.isfinite(ex) and ex < best_score - 1e-14:
                        best_seq, best_score = pl2, ex
            except Exception:
                pass

    # Final validation
    try:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        fs = math.inf
    if not math.isfinite(fs):
        try:
            fs = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            fs = math.inf
        if not math.isfinite(fs):
            return [1.0]
        return [float(x) for x in parent]

    if not math.isfinite(best_score) or best_score > fs:
        best_score = fs
    return [float(x) for x in best_seq]
