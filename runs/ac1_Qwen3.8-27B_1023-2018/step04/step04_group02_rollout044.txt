import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _conv(a):
    n = len(a)
    if n <= 128:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _lp_direction(f, n, M, timeout=3.0):
    """Find g maximizing sum(g) s.t. (f*g)_k <= M, g >= 0"""
    # Build constraint matrix A: A[k,j] = f[j] if max(0,k-n+1) <= j <= min(n-1,k)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        fj = f[j]
        if fj < 1e-15:
            continue
        # A[k,j] = f[j] for k in [j, j+n-1]
        A[j:j + n, j] = fj
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _subgrad_step(a, alpha, stop_time):
    """One subgradient step to reduce max(a*a)"""
    n = len(a)
    b = _conv(a)
    M = b.max()
    # Find all active peaks (within 1% of max)
    threshold = M * 0.99
    active = np.where(b >= threshold)[0]
    
    # Subgradient: average of 2*a[k-j] over active k's
    grad = np.zeros(n)
    for k in active:
        # grad[j] += 2 * a[k-j] for valid k-j
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * a[k - j]
    grad /= len(active)
    
    # Normalize gradient
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a.copy(), M
    
    # Subgradient step
    a_new = a - alpha * grad / gnorm
    
    # Project onto {a >= 0, sum(a) = sum(a)}
    target_sum = a.sum()
    a_new = np.maximum(a_new, 0)
    if a_new.sum() > 1e-15:
        a_new = a_new * (target_sum / a_new.sum())
    else:
        a_new = a / n  # reset to uniform
