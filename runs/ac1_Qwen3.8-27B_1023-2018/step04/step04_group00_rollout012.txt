    n = a.size
    L = 2 * n - 1
    sz = next_fast_len(L)
    F = rfft(a, sz)
    c = irfft(F * F, sz)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    m = float(u.max())
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    return e / s


def _lise_optimize(p_init, n, stop_time, seed=0):
    p = np.maximum(np.asarray(p_init, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    u = np.log(np.maximum(p, 1e-300), dtype=np.float64)
    u -= float(u.mean())

    conv_sz = next_fast_len(2 * n - 1)
    conv_ln = 2 * n - 1
    cross_sz = next_fast_len(3 * n - 2)
    cross_ln = 3 * n - 2

    def lse_obj(uu, T):
        pp = _softmax(uu)
        F = rfft(pp, conv_sz)
        c = irfft(F * F, conv_sz)[:conv_ln]
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu, dtype=np.float64)
        sc_ = (c - cmax) / T
        ex = np.exp(sc_, dtype=np.float64)
        sx = float(ex.sum())
        if not math.isfinite(sx) or sx <= 0:
            f = cmax
            w = np.zeros(c.size, dtype=np.float64)
            w[int(c.argmax())] = 1.0
        else:
            f = cmax + T * math.log(sx / c.size)
            w = ex / sx

        F2 = rfft(pp[::-1], cross_sz)
        G2 = rfft(w, cross_sz)
        gc = irfft(F2 * G2, cross_sz)[:cross_ln]
        q = gc[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)

        gpp = 2.0 * q
        tdp = float(pp @ gpp)
        gu = pp * (gpp - tdp)
        gu = np.where(np.isfinite(gu), gu, 0.0)
        return float(f), gu

    stages = [
        (0.03, 25),
        (0.008, 22),
        (0.002, 20),
        (0.0005, 18),
        (0.00015, 15),
    ]

    for T_frac, maxit in stages:
        if time.monotonic() > stop_time - 5.0:
            break
        c = _conv_self(p)
        cmax = float(c.max()) if c.size else 1.0
        if cmax <= 0 or not math.isfinite(cmax):
            break
        T = max(1e-12, T_frac * cmax)
        try:
            res = minimize(
                lse_obj, u, args=(T,), method="L-BFGS-B", jac=True,
                options={"maxiter": maxit, "gtol": 1e-11, "ftol": 1e-15,
                         "maxls": 20, "maxcor": 18, "maxdisp": 0, "disp": False}
            )
            if np.all(np.isfinite(res.x)) and res.fun < 1e99:
                u = res.x
                p = _softmax(u)
                ps = float(p.sum())
                if ps > 1e-300:
                    p /= ps
        except Exception:
            pass

    return p


def _coord_descent(p_init, stop_time, seed=12345, verbose=False):
    n = int(p_init.size)
    p = np.maximum(np.asarray(p_init, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    L = 2 * n - 1
    c = _conv_self(p)
    S = 1.0
    cur_peak = float(c.max())
    best_peak = cur_peak
    best_p = p.copy()

    rng = np.random.default_rng(seed)
    G = 56  # grid points
    sweep = 0
    flat = 0
    max_flat = 10

    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)

        for ii in order:
            i = int(ii)
            if time.monotonic() > stop_time - 0.5:
                break
            if S < 1e-300:
                break

            if S > 1.06 or S < 0.94:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0

            v = float(p[i])
            start = i
            end = min(i + n, L)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end]
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start > 0:
                mc1 = float(c[:start].max())
                if mc1 > max_const:
                    max_const = mc1
            if end < L:
                mc2 = float(c[end:].max())
                if mc2 > max_const:
                    max_const = mc2

            x_hi = max(15.0 * v, 50.0 * S / float(n), 0.03)
            x_hi = min(x_hi, 8.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = S / float(n)

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if v > 0.0:
                pts = np.unique(np.concatenate([pts, [v]]), )
            P = pts.size

            w = pts - v
            denom = S + w
            vals = np.full(P, math.inf, dtype=np.float64)
            valid = denom > 1e-14
            if not np.any(valid):
                continue

            vw = w[valid]
            vd = denom[valid]

            # Vectorized: (Pv, Ls) matrix
            aff = base[None, :] + vw[:, None] * infl[None, :]
            maff = np.max(aff, axis=1)

            # Diagonal correction at k=2i -> position diag_idx in base
            diag_idx = 2 * i - start
            if 0 <= diag_idx < Ls:
                dval = base[diag_idx] + vw * infl[diag_idx] + vw * vw
                maff = np.maximum(maff, dval)

            if max_const > 0:
                maff = np.maximum(maff, max_const)

            vals[valid] = (2.0 * n * maff) / (vd * vd)

            if not np.any(np.isfinite(vals)):
                continue

            imin = int(np.argmin(vals))
            bx = float(pts[imin])
            bv = float(vals[imin])

            # Fine refinement around best
            dw = x_hi / G
            lo = max(0.0, bx - 2.5 * dw)
            hi = min(x_hi, bx + 2.5 * dw)
            if hi <= lo:
                hi = lo + 1e-12
            fine = np.linspace(lo, hi, 32, dtype=np.float64)
            fine = np.unique(np.concatenate([fine, [bx]]))
            Pf = fine.size
            wf = fine - v
            df = S + wf
            vals_f = np.full(Pf, math.inf, dtype=np.float64)
            validf = df > 1e-14
            if np.any(validf):
                vwf = wf[validf]
                vdf = df[validf]
                afff = base[None, :] + vwf[:, None] * infl[None, :]
                maf = np.max(afff, axis=1)
                if 0 <= diag_idx < Ls:
                    dvf2 = base[diag_idx] + vwf * infl[diag_idx] + vwf * vwf
                    maf = np.maximum(maf, dvf2)
                if max_const > 0:
                    maf = np.maximum(maf, max_const)
                vals_f[validf] = (2.0 * n * maf) / (vdf * vdf)
            if np.any(np.isfinite(vals_f)):
                imf = int(np.argmin(vals_f))
                if vals_f[imf] < bv:
                    bx = float(fine[imf])
                    bv = float(vals_f[imf])

            if math.isfinite(bv) and bv < cur_peak - 1e-14:
                d = bx - v
                if abs(d) > 1e-15:
                    p[i] = bx
                    S += d
                    c[start:end] += d * infl
                    if 0 <= 2 * i < L:
                        c[2 * i] += d * d
                    cur_peak = bv
                    improved = True

        # Sweep done: re-normalize
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        # Recompute to kill drift
        c = _conv_self(p)
        cur_peak = float(c.max())
        if math.isfinite(cur_peak) and cur_peak < best_peak - 1e-14:
            best_peak = cur_peak
            best_p = p.copy()

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= max_flat:
                if time.monotonic() < stop_time - 30.0:
                    # Perturb to escape local minimum
                    scale = 0.02 + 0.015 * float(flat / max_flat)
                    p *= np.exp(rng.normal(0.0, scale, n, dtype=np.float64))
                    p = np.maximum(p, 0.0)
                    ps = float(p.sum())
                    if ps > 1e-300:
                        p /= ps
                        S = 1.0
                        c = _conv_self(p)
                        cur_peak = float(c.max())
                        flat = 0
                    else:
                        p = best_p.copy()
                        c = _conv_self(p)
                        S = 1.0
                        cur_peak = float(c.max())
                        flat = 0
                else:
                    break
        sweep += 1

    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 25.0

    parent = _get_parent()
    best = parent.copy()
    try:
        best_sc = float(_score_exact(best))
    except Exception:
        best_sc = math.inf

    if not math.isfinite(best_sc) or best.size == 0:
        best = np.ones(64, dtype=np.float64) / 64.0
        best_sc = _score_exact(best)

    def consider(a):
        nonlocal best, best_sc
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_sc:
            best = (a / s).copy()
            best_sc = sc

    n = int(best.size)
    rng = np.random.default_rng(seed)

    # Phase 1: LSE optimization (if n is manageable, ~100s)
    if n <= 4096 and time.monotonic() < deadline - 250.0:
        lse_end = min(deadline - 200.0, t0 + 110.0)
        try:
            p_lse = _lise_optimize(best, n, lse_end, seed=int(seed) + 7)
            if p_lse is not None and p_lse.size == n and np.all(np.isfinite(p_lse)):
                consider(p_lse)
        except Exception:
            pass

    # Phase 2: Coordinate descent on the current best
    p_cd_start = best.copy()
    p_cd_start = p_cd_start / float(p_cd_start.sum())

    cd_end = deadline - 15.0
    if time.monotonic() < cd_end - 10.0:
        try:
            p_cd = _coord_descent(p_cd_start, cd_end, seed=int(seed) + 999)
            if p_cd is not None and p_cd.size == n:
                consider(p_cd)
        except Exception:
            pass

    # Phase 3: If time remains, second CD pass with different seed
    rem = deadline - time.monotonic()
    if rem > 60.0 and n <= 3500:
        p2 = best.copy() / float(best.sum())
        cd2_end = min(deadline - 10.0, time.monotonic() + rem * 0.7)
        try:
            p_cd2 = _coord_descent(p2, cd2_end, seed=int(seed) + 55555)
            if p_cd2 is not None and p_cd2.size == n:
                consider(p_cd2)
        except Exception:
            pass

    # Phase 4: Brief final LSE + CD squeeze if time allows
    rem = deadline - time.monotonic()
    if rem > 80.0 and n <= 3500:
        lse2_end = min(deadline - 50.0, time.monotonic() + rem * 0.4)
        try:
            p_lse2 = _lise_optimize(best, n, lse2_end, seed=int(seed) + 4242)
            if p_lse2 is not None and p_lse2.size == n:
                consider(p_lse2)
        except Exception:
            pass
        rem = deadline - time.monotonic()
        if rem > 30.0:
            try:
                p_f = best.copy() / float(best.sum())
                p_cd3 = _coord_descent(p_f, deadline - 8.0, seed=int(seed) + 777777)
                if p_cd3 is not None and p_cd3.size == n:
                    consider(p_cd3)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
