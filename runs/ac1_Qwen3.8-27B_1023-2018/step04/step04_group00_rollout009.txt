import numpy as np
import time
import math

def _exact_score(a):
    """Compute the exact evaluator score."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = a.sum()
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    mb = b.max()
    return 2.0 * n * mb / (s * s)

    n = a.size
    L = 2*n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F*F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c
