def _build_A(n, a):

    k = np.arange(2 * n - 1)[:, None].astype(np.int64)

    d = k - j
    mask = (d >= 0) & (d < n)

    A[mask] = a[d[mask]]
    return A


def _lp_solve(A, M, n, tl=3.0):
    """Solve max 1^T g s.t. A g <= M, g >= 0."""
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tl,
                               'dual_feasibility_tolerance': 1e-11,
                               'primal_feasibility_tolerance': 1e-11})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    return np.clip(res.x, 0.0, None)
