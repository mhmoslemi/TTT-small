```python
def _project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    # Adjust to get exact sum
    diff = S - w.sum()
    if abs(diff) > 1e-12:
        w += diff / n
        w = np.maximum(w, 0)
        # Re-normalize
        s = w.sum()
        if s > 0:
            w = w / s * S
    return w
```

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _fft_conv(a):
    n = len(a)
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    c[c < 0] = 0
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(b.max())
    if not math.isfinite(m) or m <= 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.empty((2 * n - 1, n), dtype=np.float64)
    A[:] = 0
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(f, n, M, tmo=3.0):
    if M < 1e-16:
        return None
    A = _build_A(f, n)
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M + 1e-13),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': tmo,
                           'dual_feasibility_tolerance': 1e-11,
                           'primal_feasibility_tolerance': 1e-11})
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, 1e6)
    if g.sum() < 1e-14:
        return None
    return g

    f = f[:n]
    fs = f.sum()
    if fs < 1e-10:

    else:
        f /= fs
    f *= math.sqrt(2.0 * n)
    best_f = f.copy()

        g = g * (f.sum() / g.sum())
