One more idea: instead of a single LP, try LPs with different "weight" vectors in the objective. E.g., max w^T g for different w. This could find different "best responses" and give more diverse directions.

But this adds complexity. Let me keep it simple and focus on making the core loop robust.

def _mscore(a):
    """Fast score computation."""
    a = np.ascontiguousarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * a.size * float(c.max()) / (s * s)

    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def _apply_lp(a, M, n, tl=5.0):
    A = _A(n, a)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    try:
        r = linprog(c, A_ub=A, b_ub=b,
                    bounds=[(0.0, None)]*n,
                    method='highs',

                             'dual_feasibility_tolerance': 1e-12,
                             'primal_feasibility_tolerance': 1e-12})
    except Exception:
        return None
    if r.status != 0 or r.x is None:
        return None
    g = np.maximum(r.x, 0)
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s

def _blend(a, g, M):

def _grad_active(a, k, n):
    """Gradient of (a*a)[k] w.r.t. a, projected to sum=0."""
    gm = np.zeros(n)
    v = (k - np.arange(n))
    mask = v >= 0
    v_valid = v[mask]
    gm[mask] = 2.0 * a[v_valid]
    gm -= gm.mean()
    return gm

def _fw_step(a, k, n):
    """Frank-Wolfe step for min c_k over simplex."""
    gm = np.zeros(n)
    v = k - np.arange(n)
    mask = v >= 0
    v_valid = v[mask]
    gm[mask] = 2.0 * a[v_valid]
    # FW: move towards vertex with min gradient

    # Line search for best step

    return (1-best_t)*a + best_t*e
