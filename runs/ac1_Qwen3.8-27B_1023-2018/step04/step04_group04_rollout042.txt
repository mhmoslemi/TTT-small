                    best_v = f[i]
                    best_v_sc = sc
            f[i] = best_v
            if best_v_sc < best_sc - 1e-15:
                best_sc = best_v_sc
                best_f = f.copy()
                improved = True

        # Two-coordinate moves
        for _ in range(min(300, n)):
            if time.monotonic() > stop_time:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-15 and oj < 1e-15:
                continue
            for mi in [0.3, 0.6, 1.4, 2.0, 0.0]:
                for mj in [0.3, 0.6, 1.4, 2.0, 0.0]:
                    if mi == 1.0 and mj == 1.0:
                        continue
                    f[i] = oi * mi if oi > 1e-15 else 0.0
                    f[j] = oj * mj if oj > 1e-15 else 0.0
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        best_f = f.copy()
                        improved = True
            f[i], f[j] = oi, oj

        if not improved:
            break

    return best_f, best_sc


def _subgrad_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    f = f / max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = _score(f)
    step = 0.08
    no_imp = 0
    momentum = np.zeros(n)
    mu = 0.6

    for it in range(100000):
        if time.monotonic() > stop_time:
            break

        b = _fft_conv(f)
        m = float(b.max())
        eps = max(1e-14, 1e-10 * m)
        active = np.flatnonzero(b >= m - eps)

        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo <= i_hi:
                idx = k - np.arange(i_lo, i_hi + 1)
                grad[i_lo:i_hi + 1] += 2.0 * f[idx]
        if len(active) > 0:
            grad /= len(active)

        S = float(f.sum())
        obj_sg = 2.0 * n * (grad * S - 2.0 * m) / (S ** 3)
        obj_sg -= np.mean(obj_sg)
        nsg = np.linalg.norm(obj_sg)
        if nsg < 1e-16:
            break

        momentum = mu * momentum - step * obj_sg / nsg
        fn = np.clip(f + momentum, 0, None)
        s = fn.sum()
        if s < 0.01:
            step *= 0.5
            momentum *= 0.5
            continue
        fn = fn / s
        sc = _score(fn)

        if sc < best_sc - 1e-14:
            f = fn
            best_f = f.copy()
            best_sc = sc
            step = min(0.4, step * 1.02)
            no_imp = 0
        else:
            step *= 0.92
            no_imp += 1
            if no_imp > 800:
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                momentum = np.zeros(n)
                step = 0.08
                no_imp = 0

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 30.0

    parent = _get_parent()
    n_par = len(parent)

    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(100) / 100.0
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update(a, sc):
        nonlocal best_seq, best_score
        if sc < best_score - 1e-13:
            if not np.all(np.isfinite(a)) or a.sum() < 0.01:
                return
            a_c = np.clip(a, 0, 1000.0)
            best_seq = a_c.copy()
            best_score = sc

    # Phase 1: LP on parent (largest allocation)
    phase1_end = t0 + budget * 0.40
    p_norm = parent / max(parent.sum(), 1e-10)
    res1, sc1 = _lp_optimize(p_norm, n_par, phase1_end, seed)
    if math.isfinite(sc1):
        _update(res1 * parent.sum(), sc1)

    # Phase 2: LP with perturbed parent
    if time.monotonic() < t0 + budget * 0.52:
        rng2 = np.random.default_rng(seed + 77)
        p2 = np.clip(p_norm + rng2.normal(0, 0.06, n_par), 0, None)
        p2 = p2 / max(p2.sum(), 1e-10)
        stop2 = min(t0 + budget * 0.52, deadline - 60)
        res2, sc2 = _lp_optimize(p2, n_par, stop2, seed + 200)
        if math.isfinite(sc2):
            _update(res2 * parent.sum(), sc2)

    # Phase 3: Subgradient refinement
    if time.monotonic() < t0 + budget * 0.62:
        f_sg = best_seq / max(best_seq.sum(), 1e-10)
        stop3 = min(t0 + budget * 0.62, deadline - 45)
        res3, sc3 = _subgrad_optimize(f_sg, n_par, stop3, seed + 500)
        if math.isfinite(sc3):
            _update(res3 * best_seq.sum(), sc3)

    # Phase 4: Try nearby lengths
    if time.monotonic() < deadline - 60:
        alt_n = sorted(set([n_par - 6, n_par - 4, n_par - 2, n_par - 1,
                            n_par + 1, n_par + 2, n_par + 4, n_par + 6]))
        alt_n = [L for L in alt_n if L >= 8]
        remaining = deadline - 60 - time.monotonic()
        per_alt = min(35.0, remaining / max(1, len(alt_n)))

        for L in alt_n:
            if time.monotonic() > deadline - 60:
                break
            if per_alt < 5.0:
                break
            init_L = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n_par), parent)
            init_L = init_L / max(init_L.sum(), 1e-10)
            stop_L = min(deadline - 60, time.monotonic() + per_alt)
            try:
                rL, sL = _lp_optimize(init_L, L, stop_L, seed + 1000 + L)
                if math.isfinite(sL):
                    _update(rL * best_seq.sum(), sL)
            except Exception:
                pass

    # Phase 5: Coordinate descent polish
    if time.monotonic() < deadline - 10:
        stop5 = deadline - 10
        res5, sc5 = _coord_descent(best_seq, stop5, seed + 5000, max_epochs=80)
        if math.isfinite(sc5):
            _update(res5, sc5)

    # Phase 6: Final short LP polish on best
    if time.monotonic() < deadline - 5:
        n_best = len(best_seq)
        fb = best_seq / max(best_seq.sum(), 1e-10)
        stop6 = deadline - 5
        res6, sc6 = _lp_optimize(fb, n_best, stop6, seed + 9999)
        if math.isfinite(sc6):
            _update(res6 * best_seq.sum(), sc6)

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
