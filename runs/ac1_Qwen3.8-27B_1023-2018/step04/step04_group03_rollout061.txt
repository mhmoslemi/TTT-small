Let me also consider: instead of golden section for each coordinate, I can compute the analytical gradient and do a Newton step. For a single coordinate `f[i]`, the score as a function of `x = f[i]` is:

`score(x) = 2*n * max_k b_k(x) / (sum_a(x))^2`

where `sum_a(x) = S + (x - f[i])` (S = sum of all other coordinates) and `b_k(x)` is the autocorrelation at lag k with `f[i]` replaced by `x`.

The peak lag `k*` is the one achieving the max. Around `x = f[i]`, the score is:
`score(x) ≈ 2*n * b_{k*}(x) / (S + x - f[i])^2`

`b_{k*}(x) = b_{k*}(f[i]) + (x - f[i]) * db_k*[i] + (x - f[i])^2 * (1 if 2i = k* else 0)`

where `db_k*[i]` is the partial derivative of `b[k*]` w.r.t. `f[i]`.

The optimal `x` can be found by solving `d score / dx = 0`, which gives a quadratic equation. This is much faster than golden section!

But I need to handle the case where the peak lag changes. For a small perturbation, the peak lag should stay the same. So:
1. Identify peak lag `k*`
2. Compute `b_{k*}` and `db_{k*}/df[i]`
3. Solve quadratic for optimal `x`
4. Verify the peak lag hasn't changed (evaluate score)

This is O(n) to compute the derivative (need `f[k*-i]` and `f[k*-i]` again), and O(1) to solve the quadratic.

def _conv_full(a, fft_len=None):
    n = len(a)

        fft_len = next_fast_len(2 * n - 1)
    F = rfft(a, fft_len)
    c = irfft(F * F, fft_len)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

        if len(peak_idx) == 0:
            peak_idx = np.array([int(np.argmax(b))])
        
        grad = np.zeros(n)
        for k in peak_idx:
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                grad[i] += f[k - i]
        grad /= len(peak_idx)
        
        gscale = grad.max()
        if gscale < 1e-18:
            break
        lr_eff = lr / gscale
        
        exponent = -lr_eff * grad
        exponent -= exponent.max()
        weights = np.exp(exponent)
        f_new = f * weights
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        
        sc = _score(f_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            stall = 0
            lr *= 0.995
        else:
            f_mix = 0.3 * f + 0.7 * f_new
            f_mix = np.clip(f_mix, 1e-12, None)
            f_mix = f_mix / f_mix.sum()
            sc_mix = _score(f_mix)
            if sc_mix < best_score - 1e-12:
                best_score = sc_mix
                best_f = f_mix.copy()
                f = f_mix
                stall = 0
                lr *= 0.99
            else:
                f = f_new
                stall += 1
                lr *= 0.94
                if stall > 100:
                    noise = rng.normal(0, 0.01, n)
                    f = np.clip(f + noise, 1e-12, None)
                    f = f / f.sum()
                    lr = 0.12
                    stall = 0
                    if lr < 0.005:
                        break
    
    return best_f, best_score

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 15.0
    
    parent = _get_parent()
    n_p = len(parent)
    
    best_seq = parent.copy()
    best_score = _score(best_seq)
    
    def _try_update(candidate):
        nonlocal best_seq, best_score
        if candidate is None or len(candidate) == 0:
            return
        c = np.clip(np.asarray(candidate, dtype=np.float64).ravel(), 0, 1000)
        if not np.all(np.isfinite(c)) or c.sum() < 0.01:
            return
        sc = _score(c)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_seq = c.copy()
    
    rng = np.random.default_rng(seed)
    
    # Phase 1: LP optimization from parent (55% of time)
    phase1_end = t0 + budget * 0.55
    if time.monotonic() < phase1_end - 10.0:
        stop = min(phase1_end - 3.0, time.monotonic() + (phase1_end - time.monotonic() - 5.0))
        try:
            res, sc = _optimize_lp(parent, stop, seed + 1, min_frac_init=0.0)
            _try_update(res)
        except Exception:
            pass
    
    # Phase 2: Second LP pass on current best (10%)
    phase2_end = t0 + budget * 0.65
    if time.monotonic() < phase2_end - 10.0:
        stop = min(phase2_end - 3.0, time.monotonic() + 100.0)
        try:
            res, sc = _optimize_lp(best_seq, stop, seed + 100, min_frac_init=0.0)
            _try_update(res)
        except Exception:
            pass
    
    # Phase 3: Subgradient refinement (10%)
    phase3_end = t0 + budget * 0.75
    if time.monotonic() < phase3_end - 10.0:
        stop = min(phase3_end - 3.0, time.monotonic() + 100.0)
        try:
            res, sc = _subgrad_step(best_seq, stop, seed + 200)
            _try_update(res)
        except Exception:
            pass
    
    # Phase 4: LP after subgradient (7%)
    phase4_end = t0 + budget * 0.82
    if time.monotonic() < phase4_end - 10.0:
        stop = min(phase4_end - 3.0, time.monotonic() + 80.0)
        try:
            res, sc = _optimize_lp(best_seq, stop, seed + 300, min_frac_init=0.0)
            _try_update(res)
        except Exception:
            pass
    
    # Phase 5: Coordinate descent polish (11%)
    phase5_end = t0 + budget * 0.93
    if time.monotonic() < phase5_end - 10.0:
        stop = min(phase5_end - 3.0, time.monotonic() + (phase5_end - time.monotonic() - 5.0))
        try:
            res, sc = _polish_coord(best_seq, stop, seed + 400)
            _try_update(res)
        except Exception:
            pass
    
    # Phase 6: Final LP + polish (5%)
    if time.monotonic() < hard_deadline - 20.0:
        stop_lp = min(hard_deadline - 10.0, time.monotonic() + 50.0)
        try:
            res, sc = _optimize_lp(best_seq, stop_lp, seed + 500, min_frac_init=0.0)
            _try_update(res)
        except Exception:
            pass
    
    if time.monotonic() < hard_deadline - 10.0:
        stop_polish = min(hard_deadline - 5.0, time.monotonic() + 40.0)
        try:
            res, sc = _polish_coord(best_seq, stop_polish, seed + 600)
            _try_update(res)
        except Exception:
            pass
    
    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf
    
    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]
    
    return [float(x) for x in best_seq]
