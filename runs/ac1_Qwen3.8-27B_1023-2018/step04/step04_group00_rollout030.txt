where $C_i = \max(c_0, ..., c_{i-1}, c_{i+n}, ..., c_{2n-2})$ is the max of the unaffected entries.

This is clean! And the computation per evaluation is O(n) (computing the max of the base + x*infl array).

2. Evaluate at a few points: $x = \pm \delta$ for small $\delta$, and $x = x_{max}$
3. If the minimum is interior, do golden section

Actually, since $a_i \geq 0$ is required, we need $x \geq -v$ where $v = a_i$. But in practice, we'll usually want $x \geq 0$ (increasing the coordinate) or a small negative $x$ (decreasing). ```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    """Compute full self-convolution of a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n <= 256:
        c = np.convolve(a, a, mode="full")
    else:
        L = 2 * n - 1
        size = next_fast_len(L)
        F = rfft(a, size)
        c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _score_arr(a):
    """Compute the exact score for array a (assumed sum=1)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    if c.size == 0:
        return math.inf
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _score_exact_eval(a):
    """Use the exact evaluator for final validation."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if a.size == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * a.size * m / (s * s))


def _lse_optimize(p, stop_time, consider=None):
    """LSE-smoothed optimization using L-BFGS-B in log/softmax space."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2

    def softmax(uu):
        m = float(np.max(uu))
        e = np.exp(uu - m)
        ss = float(e.sum())
        if ss <= 0 or not math.isfinite(ss):
            return np.full(n, 1.0 / n)
        return e / ss

    def lse_func(uu, T):
        pp = softmax(uu)
        F = rfft(pp, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros(n)
        if T <= 1e-300:
            T = 1e-6 * cmax
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * np.log(sum_exp / conv_len)
            w = exps / sum_exp
        # Cross-correlation for gradient
        Fp = rfft(pp[::-1], cross_size)
        G = rfft(w, cross_size)
        cc = irfft(Fp * G, cross_size)[:cross_len]
        q = cc[n - 1:2 * n - 1].copy()
        np.maximum(q, 0.0, out=q)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    best_u = u.copy()
    best_sc = _score_arr(p)

    # Adaptive temperature schedule
    c = _conv_self(p)
    cmax = float(c.max())
    stages = [
        (0.10, 0.15),
        (0.04, 0.15),
        (0.012, 0.20),
        (0.004, 0.20),
        (0.001, 0.20),
        (0.0003, 0.10),
    ]

    start = time.monotonic()
    avail = max(0.0, stop_time - start - 5.0)
    if avail < 10:
        return p

    cum_w = 0.0
    total_w = sum(w for _, w in stages)

    for frac, wt in stages:
        if time.monotonic() > stop_time - 8.0:
            break
        cum_w += wt
        end_stage = min(stop_time - 4.0, start + avail * (cum_w / total_w))
        if end_stage <= time.monotonic() + 5.0:
            continue
        T = max(1e-300, frac * cmax)

        stale = 0
        while time.monotonic() < end_stage - 1.0 and stale < 3:
            opts = {
                "maxiter": 50 if n > 1500 else 80,
                "gtol": 5e-9,
                "ftol": 1e-14,
                "maxls": 30,
                "maxcor": 25,
                "disp": False,
            }
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                x = res.x
                xx = softmax(x)
                sc = _score_arr(xx)
                if math.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_u = x.copy()
                    p = xx.copy()
                    stale = 0
                    if consider:
                        consider(p.copy())
                else:
                    stale += 1
            else:
                break
            if time.monotonic() > stop_time - 8.0:
                break

    return p


def _coord_descent(p, stop_time, consider=None, G=30, seed=12345):
    """Aggressive coordinate descent on the true objective."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s  # normalize to sum=1

    c = _conv_self(p)
    Ltot = 2 * n - 1
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    infl = 2.0 * p  # influence vector (same for all coordinates)

    rng = np.random.default_rng(seed)
    sweep = 0
    flat_count = 0
    S_total = 1.0

    def eval_1d(xs, base, max_const, diag_idx, v, S):
        """Evaluate g(x) for an array of x values."""
        xs = np.asarray(xs, dtype=np.float64)
        w = xs - v
        denom = S + w
        vals = np.full(xs.shape, math.inf, dtype=np.float64)
        valid = denom > 1e-15
        if not np.any(valid):
            return vals
        aff = base[None, :] + w[:, None] * infl[None, :]
        maff = np.max(aff, axis=1)
        # Add diagonal correction
        dval = base[diag_idx] + w * infl[diag_idx] + w * w
        maff = np.maximum(maff, dval)
        maff = np.maximum(maff, max_const)
        dv = denom[valid]
        mv = maff[valid]
        vals[valid] = (2.0 * n * mv) / (dv * dv)
        return vals

    max_sweeps = 100000

    while sweep < max_sweeps and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 1.0:
                break
            if S_total <= 1e-300:
                break
            # Renormalize periodically
            if S_total > 1.35 or S_total < 0.70:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                infl *= inv
                S_total = 1.0

            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end]
            diag = 2 * i - start  # position of diagonal in base

            max_const = 0.0
            if start > 0:
                max_const = max(max_const, float(c[:start].max()))
            if end < Ltot:
                max_const = max(max_const, float(c[end:].max()))

            # Search range
            x_hi = max(20.0 * v, 6.0 * S / n, 0.03 * S, 1e-13)
            x_hi = min(x_hi, 8.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = S / n

            # Coarse grid
            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if v > 0:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            vals = eval_1d(pts, base, max_const, diag, v, S)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Golden section refinement around best point
            if 0 < imin < len(pts) - 1:
                lo = float(pts[max(0, imin - 1)])
                hi = float(pts[min(len(pts) - 1, imin + 1)])
                # Golden section
                gr = (math.sqrt(5.0) - 1.0) / 2.0
                a_gs = lo
                b_gs = hi
                c_gs = b_gs - gr * (b_gs - a_gs)
                d_gs = a_gs + gr * (b_gs - a_gs)
                fc = float(np.min(eval_1d([c_gs], base, max_const, diag, v, S)))
                fd = float(np.min(eval_1d([d_gs], base, max_const, diag, v, S)))
                for _ in range(24):
                    if fc < fd:
                        b_gs = d_gs
                        d_gs = c_gs
                        fd = fc
                        c_gs = b_gs - gr * (b_gs - a_gs)
                        fc = float(np.min(eval_1d([c_gs], base, max_const, diag, v, S)))
                    else:
                        a_gs = c_gs
                        c_gs = d_gs
                        fc = fd
                        d_gs = a_gs + gr * (b_gs - a_gs)
                        fd = float(np.min(eval_1d([d_gs], base, max_const, diag, v, S)))
                gs_x = (a_gs + b_gs) / 2.0
                gs_val = float(np.min(eval_1d([gs_x], base, max_const, diag, v, S)))
                if gs_val < best_val:
                    best_x = gs_x
                    best_val = gs_val

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl[:Ls]
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    infl[i] += 2.0 * w
                    cur_score = best_val
                    improved = True

        # End of sweep: renormalize
        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            infl *= inv
        else:
            break
        S_total = 1.0

        # Recompute to correct drift every few sweeps
        if sweep % 10 == 9:
            c = _conv_self(p)
            infl = 2.0 * p
            new_score = 2.0 * n * float(c.max())
            cur_score = new_score

        if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            if consider:
                consider(best_p.copy())

        if improved:
            flat_count = 0
        else:
            flat_count += 1
            if flat_count >= 4:
                if time.monotonic() < stop_time - 15.0:
                    # Perturb to escape
                    scale = 0.02 + 0.03 * min(flat_count / 10.0, 1.0)
                    p = p * np.exp(rng.normal(0.0, scale, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    S_total = 1.0
                    c = _conv_self(p)
                    infl = 2.0 * p
                    cur_score = 2.0 * n * float(c.max())
                    flat_count = 0
                else:
                    break
        sweep += 1

    # Final
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_self(p)
    sc = 2.0 * n * float(c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 20.0  # safety margin

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = _score_exact_eval(best_arr)

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(128)
        best_score = _score_exact_eval(best_arr)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            sc = 2.0 * a.size * float(_conv_self(a).max()) / (s * s)
            if math.isfinite(sc) and sc < best_score - 1e-13:
                best_arr = a.copy()
                best_score = sc
        except Exception:
            pass

    n = int(best_arr.size)

    # Phase 1: LSE optimization
    lse_stop = min(hard_deadline, t0 + 0.30 * (budget - 20.0))
    if lse_stop > t0 + 30.0:
        try:
            p_lse = _lse_optimize(best_arr, lse_stop, consider=consider)
            if p_lse is not None and p_lse.size == n and np.all(np.isfinite(p_lse)) and p_lse.sum() > 0:
                p_cur = p_lse / p_lse.sum()
                consider(p_cur)
                if _score_arr(p_cur) < _score_arr(best_arr):
                    best_arr = p_cur.copy()
        except Exception:
            pass

    # Phase 2: Coordinate descent (main workhorse)
    cd_stop = min(hard_deadline, t0 + 0.80 * (budget - 20.0))
    if cd_stop > time.monotonic() + 20.0:
        try:
            p_cd, _ = _coord_descent(best_arr, cd_stop, consider=consider, G=28, seed=int(seed) + 777)
            if p_cd is not None and p_cd.size == n and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                p_cd = p_cd / p_cd.sum()
                consider(p_cd)
        except Exception:
            pass

    # Phase 3: Second round of coordinate descent with different seed
    cd2_stop = min(hard_deadline - 10.0, t0 + (budget - 30.0))
    if cd2_stop > time.monotonic() + 15.0:
        try:
            p_cd2, _ = _coord_descent(best_arr, cd2_stop, consider=consider, G=20, seed=int(seed) + 99999)
            if p_cd2 is not None and p_cd2.size == n and np.all(np.isfinite(p_cd2)) and p_cd2.sum() > 0:
                p_cd2 = p_cd2 / p_cd2.sum()
                consider(p_cd2)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
