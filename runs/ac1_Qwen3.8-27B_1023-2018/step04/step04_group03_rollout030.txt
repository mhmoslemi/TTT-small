                noise = rng.normal(0, noise_scale, n)
                f = np.clip(f + noise, 0, None)
                fs2 = f.sum()
                if fs2 > 0.01:
                    f = f / fs2
                stall = 0
                frac = 1.0
                max_stall = min(80, max_stall + 15)
            continue

        # Check if direction is meaningfully different
        dist = np.linalg.norm(g - f)
        if dist < 1e-8:
            stall += 3
            if stall > max_stall:
                noise = rng.normal(0, 0.08, n)
                f = np.clip(f + noise, 0, None)
                fs2 = f.sum()
                if fs2 > 0.01:
                    f = f / fs2
                stall = 0
                max_stall = min(80, max_stall + 15)
            continue

        # Line search
        f_new, sc, alpha = _line_search(f, g, best_score)

        if sc < best_score - 1e-13:
            f = f_new
            best_f = f.copy()
            best_score = sc
            stall = 0
            # Narrow constraints as we progress
            if best_score < last_best_score * 0.99:
                frac = max(0.4, frac * 0.98)
                last_best_score = best_score
        else:
            stall += 1
            if stall > 5:
                # Try a smaller step directly
                f_try = f + 0.01 * (g - f)
                np.clip(f_try, 0, None, out=f_try)
                sc_try = _score(f_try)
                if sc_try < best_score - 1e-13:
                    f = f_try
                    best_f = f.copy()
                    best_score = sc_try
                    stall = 0
            if stall > max_stall:
                noise = rng.normal(0, 0.1, n)
                f = np.clip(f + noise, 0, None)
                fs2 = f.sum()
                if fs2 > 0.01:
                    f = f / fs2
                stall = 0
                frac = 1.0
                max_stall = min(100, max_stall + 20)

    return best_f, best_score


def _coord_descent(f0, stop_time, seed):
    """Coordinate descent with per-coordinate golden section search."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    best_sc = _score(f)
    no_imp = 0
    max_no_imp = 20

    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break

            orig = f[i]
            f_mean = f.sum() / n

            # Candidate values for f[i]
            cands = [0.0,
                     orig * 0.2, orig * 0.4, orig * 0.6, orig * 0.8,
                     orig * 1.0, orig * 1.2, orig * 1.5, orig * 2.0,
                     orig * 0.1, orig * 3.0,
                     0.1 * f_mean, 0.5 * f_mean, 1.0 * f_mean,
                     2.0 * f_mean, 0.01 * f_mean]
            cands = [max(0.0, c) for c in cands]

            best_local = best_sc
            best_x = orig

            for x in cands:
                f[i] = x
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_x = x

            if best_local < best_sc - 1e-14:
                # Refine
                for _ in range(3):
                    lo = max(0.0, best_x * 0.85)
                    hi = best_x * 1.15
                    for frac_point in [0.3, 0.5, 0.7]:
                        x_try = lo + (hi - lo) * frac_point
                        f[i] = x_try
                        if f.sum() < 0.01:
                            f[i] = best_x
                            continue
                        sc = _score(f)
                        if sc < best_local - 1e-14:
                            best_local = sc
                            best_x = x_try
                f[i] = best_x
                best_sc = best_local
                improved = True
            else:
                f[i] = orig

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return f.copy(), best_sc


def _spectral_init(n, seed):
    """Generate a spectral (pseudo-random phase) initialization."""
    rng = np.random.default_rng(seed)
    fsize = next_fast_len(n)
    # Random phases for positive frequencies
    n_pos = fsize // 2 + 1
    phases = rng.uniform(0, 2 * np.pi, n_pos)
    F = np.zeros(fsize, dtype=complex)
    mags = np.ones(n_pos) * 0.5
    mags[0] = 1.0  # DC
    F[:n_pos] = mags * np.exp(1j * phases)
    # Hermitian symmetry
    if fsize % 2 == 1:
        F[fsize - (np.arange(n_pos)[1:])] = np.conj(F[1:n_pos])
    else:
        F[fsize // 2:] = np.conj(F[1:fsize // 2][::-1])
    f = np.real(np.fft.ifft(F))[:n]
    f = np.abs(f) + 0.005
    return f / f.sum()


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    # Normalize parent
    f_parent = parent.copy()
    ps = f_parent.sum()
    if ps > 0.01:
        f_parent_norm = f_parent / ps
    else:
        f_parent_norm = np.ones(n_p) / n_p

    best_seq = parent.copy()
    best_score = _score(parent)

    def _update_best(candidate):
        nonlocal best_seq, best_score
        if candidate is None or len(candidate) == 0:
            return
        candidate = np.asarray(candidate, dtype=np.float64).ravel()
        candidate = np.clip(candidate, 0, 1000)
        if not np.all(np.isfinite(candidate)):
            return
        if candidate.sum() < 0.01:
            return
        sc = _score(candidate)
        if math.isfinite(sc) and sc < best_score - 1e-12:
            best_seq = candidate.copy()
            best_score = sc

    rng = np.random.default_rng(seed)
    i_arr = np.arange(n_p, dtype=np.float64)

    # Generate diverse initializations
    inits = [f_parent_norm.copy()]

    # Noisy perturbations of parent
    for sig in [0.08, 0.15, 0.25]:
        noisy = np.clip(f_parent_norm + rng.normal(0, sig, n_p), 0, None)
        if noisy.sum() > 0.01:
            inits.append(noisy / noisy.sum())

    # Spectral init
    inits.append(_spectral_init(n_p, seed + 1))
    inits.append(_spectral_init(n_p, seed + 2))

    # Cosine window
    cos_init = 0.5 * (1 + np.cos(2 * np.pi * i_arr / n_p)) + 0.01
    inits.append(cos_init / cos_init.sum())

    # Multi-frequency
    sup = np.ones(n_p) * 0.3
    for fr in [1, 2, 3, 5, 8]:
        ph = rng.uniform(0, 2 * np.pi)
        sup += 0.08 * np.cos(2 * np.pi * fr * i_arr / n_p + ph)
    sup = np.abs(sup) + 0.01
    inits.append(sup / sup.sum())

    # Gaussian bump
    center = (n_p - 1) / 2.0
    gauss = np.exp(-0.5 * ((i_arr - center) / (n_p * 0.2)) ** 2)
    inits.append(gauss / gauss.sum())

    # Two bumps
    c1, c2 = n_p * 0.25, n_p * 0.75
    two_bump = np.exp(-0.5 * ((i_arr - c1) / (n_p * 0.1)) ** 2) + \
               np.exp(-0.5 * ((i_arr - c2) / (n_p * 0.1)) ** 2)
    inits.append(two_bump / two_bump.sum())

    # Sine magnitude
    sine = np.abs(np.sin(2 * np.pi * 3 * i_arr / n_p)) + 0.05
    inits.append(sine / sine.sum())

    time_per_init = max(12.0, (budget * 0.48) / len(inits))

    # Phase 1: LP on multiple inits (48% time)
    phase1_end = t0 + budget * 0.48
    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop = min(phase1_end - 3.0, time.monotonic() + time_per_init)
        if time.monotonic() > stop - 3.0:
            continue
        try:
            res, sc = _optimize_lp_loop(init, stop, seed + 1000 * (idx + 1))
            _update_best(res)
        except Exception:
            pass

    # Phase 2: Continued LP on best (22% time)
    phase2_end = t0 + budget * 0.70
    if time.monotonic() < phase2_end - 10.0:
        cur = best_seq.copy()
        cs = cur.sum()
        if cs > 0.01:
            cur = cur / cs
        else:
            cur = f_parent_norm.copy()
        try:
            res, sc = _optimize_lp_loop(cur, phase2_end - 3.0, seed + 77777)
            _update_best(res)
        except Exception:
            pass

    # Phase 3: Coordinate descent (15% time)
    phase3_end = t0 + budget * 0.85
    if time.monotonic() < phase3_end - 10.0:
        try:
            res, sc = _coord_descent(best_seq, phase3_end - 3.0, seed + 55555)
            _update_best(res)
        except Exception:
            pass

    # Phase 4: LP on polished (6% time)
    phase4_end = t0 + budget * 0.91
    if time.monotonic() < phase4_end - 10.0:
        cur = best_seq.copy()
        cs = cur.sum()
        if cs > 0.01:
            cur = cur / cs
        try:
            res, sc = _optimize_lp_loop(cur, phase4_end - 3.0, seed + 66666)
            _update_best(res)
        except Exception:
            pass

    # Phase 5: Final coordinate descent (4% time)
    phase5_end = deadline + 2.0
    if time.monotonic() < phase5_end - 5.0:
        try:
            res, sc = _coord_descent(best_seq, phase5_end - 2.0, seed + 99999)
            _update_best(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Fallback to parent
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            return [1.0]
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    return [float(x) for x in best_seq]
