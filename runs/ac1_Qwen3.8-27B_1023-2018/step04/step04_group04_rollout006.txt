def _subgrad_optimize(f0, n, stop_time, seed):

    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / max(f.sum(), 1e-10)
    best_f = f.copy()

        M = float(np.max(b))
        eps = max(1e-14, 1e-10 * M)
        active = np.flatnonzero(b >= M - eps)

        grad = np.zeros(n)

        obj_grad -= obj_grad.mean()
        ng = np.linalg.norm(obj_grad)
        if ng < 1e-16:
            break

        momentum = mu * momentum - step * obj_grad / ng
        fn = np.clip(f + momentum, 0, None)
        s = fn.sum()
        if s < 0.01:
            step *= 0.5
            momentum *= 0.5
            continue
        fn = fn / s
        sc = _score(fn)

        if sc < best_sc - 1e-14:
            f = fn
            best_f = f.copy()
            best_sc = sc
            step = min(0.4, step * 1.02)
            no_imp = 0
        else:
            step *= 0.92
            no_imp += 1
            if no_imp > 300:
                f = np.clip(f + rng.normal(0, 0.025, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                momentum = np.zeros(n)
                step = 0.08
                no_imp = 0
                if no_imp > 5000:
                    break

    return best_f, best_sc


def _coord_descent(f0, stop_time, seed, max_rounds=200):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    n = len(f)
    best_sc = _score(f)
    best_f = f.copy()
    consecutive_stalls = 0

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            if orig < 1e-15 and rng.random() < 0.95:
                continue
            best_v = orig
            best_v_sc = best_sc
            # Try multiple multipliers
            mults = [0.1, 0.25, 0.4, 0.6, 0.8, 1.1, 1.3, 1.6, 2.0, 0.0]
            if orig > 1e-15:
                mults = [0.05 * (orig / max(orig, 1e-15)) / orig if orig > 0 else 0] + mults
            for mult in mults:
                f[i] = orig * mult
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_v_sc - 1e-15:
                    best_v = f[i]
                    best_v_sc = sc
            f[i] = best_v
            if best_v_sc < best_sc - 1e-15:
                best_sc = best_v_sc
                best_f = f.copy()
                improved = True

        # Pair coordinate moves
        n_pairs = min(300, n)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-15 and oj < 1e-15:
                continue
            for mi in [0.3, 0.7, 1.4]:
                for mj in [0.3, 0.7, 1.4]:
                    f[i] = oi * mi if oi > 1e-15 else 0.0
                    f[j] = oj * mj if oj > 1e-15 else 0.0
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        best_f = f.copy()
                        improved = True
                        f[i], f[j] = oi, oj
        if improved:
            f[i], f[j] = oi, oj
        else:
            f[i], f[j] = oi, oj

        # Actually restore after pair
        f[i], f[j] = oi, oj

        if not improved:
            consecutive_stalls += 1
            if consecutive_stalls >= 3:
                break
        else:
            consecutive_stalls = 0

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n = len(parent)

    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(256)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update(a, sc):
        nonlocal best_seq, best_score
        if math.isfinite(sc) and sc < best_score - 1e-12:
            a = np.asarray(a, dtype=np.float64).copy()
            a = np.clip(a, 0, 1000.0)
            if a.sum() >= 0.01 and np.all(np.isfinite(a)):
                best_seq = a
                best_score = sc

    # Phase 1: LP optimization (main engine) - 50% budget
    phase1_end = t0 + budget * 0.50
    p_norm = parent / max(parent.sum(), 1e-10)
    res, sc = _lp_optimize(p_norm, n, phase1_end, seed)
    if math.isfinite(sc):
        _update(res * parent.sum(), sc)

    # Phase 1b: LP with perturbed init
    if time.monotonic() < t0 + budget * 0.56:
        rng_p = np.random.default_rng(seed + 7)
        p2 = np.clip(p_norm + rng_p.normal(0, 0.1, n), 0, None)
        p2 = p2 / max(p2.sum(), 1e-10)
        stop1b = min(t0 + budget * 0.56, deadline - 200)
        res2, sc2 = _lp_optimize(p2, n, stop1b, seed + 200)
        if math.isfinite(sc2):
            _update(res2 * parent.sum(), sc2)

    # Phase 2: Subgradient refinement - 12%
    phase2_end = t0 + budget * 0.68
    if time.monotonic() < phase2_end:
        f_sg = best_seq / max(best_seq.sum(), 1e-10)
        res_sg, sc_sg = _subgrad_optimize(f_sg, n, phase2_end, seed + 500)
        if math.isfinite(sc_sg):
            _update(res_sg * best_seq.sum(), sc_sg)

    # Phase 3: Coordinate descent - 15%
    phase3_end = t0 + budget * 0.83
    if time.monotonic() < phase3_end:
        res_cd, sc_cd = _coord_descent(best_seq, phase3_end, seed + 800, max_rounds=150)
        if math.isfinite(sc_cd):
            _update(res_cd, sc_cd)

    # Phase 4: Length exploration - 10%
    phase4_end = deadline - 20
    if time.monotonic() < phase4_end:
        alt_lengths = [n - 6, n - 4, n - 2, n + 2, n + 4, n + 8]
        alt_lengths = [L for L in alt_lengths if L >= 16 and L <= 600]
        time_per_alt = min(30.0, (phase4_end - time.monotonic()) / max(1, len(alt_lengths)))
        for L in alt_lengths:
            if time.monotonic() > phase4_end - 5:
                break
            try:
                init_L = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n), parent)
                init_L = np.clip(init_L, 0, None)
                init_L = init_L / max(init_L.sum(), 1e-10)
                stop_L = min(phase4_end, time.monotonic() + time_per_alt)
                rL, sL = _lp_optimize(init_L, L, stop_L, seed + 2000 + L)
                if math.isfinite(sL):
                    _update(rL * parent.sum(), sL)
            except Exception:
                pass

    # Phase 5: Final coordinate descent polish
    if time.monotonic() < deadline - 8:
        res_final, sc_final = _coord_descent(best_seq, deadline - 8, seed + 5000, max_rounds=80)
        if math.isfinite(sc_final):
            _update(res_final, sc_final)

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
