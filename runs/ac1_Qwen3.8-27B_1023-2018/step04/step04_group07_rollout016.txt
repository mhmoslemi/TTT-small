```python
def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_normalized(a):
    """Assumes sum(a) = 1, returns 2n * max(a*a)"""
    n = len(a)
    c = _conv_full(a)
    return 2.0 * n * c.max()

def _subgradient(a, active_mask):
    """Compute subgradient of max_k (a*a)_k w.r.t. a"""
    n = len(a)
    h = np.zeros(2 * n - 1)
    h[active_mask] = 1.0

    for i in range(n):

    return g
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _lp_direction(f, n, M, top_k=None, time_limit=5.0):
    """Find LP descent direction"""
    if top_k is None or top_k >= 2 * n - 1:
        indices = np.arange(2 * n - 1)
    else:
        c = _conv_full(f)
        indices = np.argsort(-c)[:top_k]
    
    # Build constraint matrix
    A = np.zeros((len(indices), n))
    for idx, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for m in range(lo, hi):
            A[idx, m] = f[k - m]
    
    c_obj = -np.ones(n)
    b_ub = np.full(len(indices), M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs

def _line_search(f, g, M, stop_time):
    """Find best mixing parameter t"""
    best_t = 0.0
    best_val = M
    
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        if time.monotonic() > stop_time:
            break
        f_new = (1.0 - t) * f + t * g
        m_new = float(_conv_full(f_new).max())
        if m_new < best_val:
            best_val = m_new
            best_t = t
    
    if best_t > 0:
        # Refine
        lo = max(0.0, best_t - 0.1)
        hi = min(1.0, best_t + 0.1)
        for _ in range(6):
            if time.monotonic() > stop_time:

            m1 = lo + (hi - lo) / 3
            m2 = hi - (hi - lo) / 3
            v1 = float(_conv_full((1 - m1) * f + m1 * g).max())

                hi = m2

                lo = m1
            tm = (lo + hi) / 2
            vm = float(_conv_full((1 - tm) * f + tm * g).max())
            if vm < best_val:
                best_val = vm
                best_t = tm
    
    return best_t, best_val

            pert = min(0.3, 0.05 + 0.01 * (stall - 30))
            strategy = rng.integers(5)
            if strategy == 0:
                f = np.clip(f + rng.normal(0, pert * f.mean(), n), 0, None)
            elif strategy == 1:
                blk = max(1, n // 4)
                f2 = f.copy()
                f2[:blk], f2[n-blk:] = f[n-blk:].copy(), f[:blk].copy()
                f = f2
            elif strategy == 2:
                mask = rng.random(n) < 0.3
                f[mask] = rng.uniform(0, 2 * f[mask].max())
            elif strategy == 3:
                f = 0.7 * best_f + 0.3 * np.abs(rng.normal(1, 0.5, n))
            else:
                # Smooth noise
                kern = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
                kern /= kern.sum()
                f = np.clip(f * (1 + pert * np.convolve(rng.normal(0, 1, n), kern, 'same')), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            M = float(_conv_full(f).max())
            stall = 0
            continue
        
        # Determine LP parameters
        if stall < 10:
            top_k = 50
        elif stall < 20:
            top_k = 100
        else:
            top_k = 200
        
        tl = max(1.0, min(8.0, 3.0 + 0.1 * stall))
        g_dir = _lp_direction(f, n, M, top_k=top_k, time_limit=tl)
        
        if g_dir is None:
            stall += 3
            f = np.clip(f + rng.normal(0, 0.03 * f.mean(), n), 0, None)
            f = f / f.sum()
            M = float(_conv_full(f).max())
            continue
        
        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-8:
            stall += 2
            continue
        
        t_opt, m_opt = _line_search(f, g_dir, M, stop_time - 2.0)
        
        if t_opt > 0 and m_opt < M - 1e-14:
            f = (1 - t_opt) * f + t_opt * g_dir
            f = np.clip(f, 0, None)
            fs = f.sum()
            f = f / fs
            M = float(_conv_full(f).max())
            stall = 0
        else:
            stall += 1
        
        iter_count += 1
    
    return best_f, 2.0 * n * best_M
```

```python
def _polish(a, stop_time, seed=99):
    n = len(a)
    best = np.clip(a.astype(np.float64).copy(), 0, None)
    fs = best.sum()
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    
    no_imp = 0
    max_no_imp = 10
    
    while time.monotonic() < stop_time - 2.0 and no_imp < max_no_imp:
        improved = False
        
        # Single coordinate
        for i in rng.permutation(n):
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            total = best.sum()
            best_val = orig
            best_sc_val = best_sc
            for nv in [0, orig * 0.5, orig * 0.8, orig * 1.2, orig * 1.5, 
                       total * 0.01, total * 0.05]:
                if nv < 0 or nv > 1000:
                    continue
                best[i] = nv
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_sc_val - 1e-14:
                    best_sc_val = sc
                    best_val = nv
            best[i] = best_val
            if best_sc_val < best_sc - 1e-14:
                best_sc = best_sc_val
                improved = True
        
        # Pair moves
        for _ in range(min(200, n * 2)):
            if time.monotonic() > stop_time - 2.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            tot = oi + oj
            if tot < 1e-12:
                continue
            for fr in [0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]:
                best[i], best[j] = fr * tot, (1 - fr) * tot
                sc = _score(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj
        
        if improved:
            no_imp = 0
        else:
            no_imp += 1
    
    return best, best_sc
```

Let me add one more technique: **proximal gradient** for the smooth part. The max-convolution objective can be approximated by the log-sum-exp:

`f_smooth(a) = (1/beta) * log(sum_k exp(beta * c_k))`

As `beta → ∞`, this approaches `max_k c_k`. The gradient of `f_smooth` is:
`grad = (2/beta) * sum_k exp(beta*c_k)/sum_j exp(beta*c_j) * T_k * a`

where `T_k * a` is the gradient of `c_k` with respect to `a` (which we computed as `2 * (1_{[k-i in 0..n-1]} a_{k-i})` summed over i).

Actually, `T_k a = 2 * slide(a, k)` where `slide` is the appropriate cross-correlation.

This gives a smooth approximation that's easier to optimize with standard methods.

```python
def _smooth_grad(a, n, beta):
    c = _conv_full(a)
    cmax = c.max()
    e = np.exp(beta * (c - cmax))
    w = e / e.sum()
    # g[i] = 2 * sum_k w[k] * a[k-i] (where k-i in [0,n-1])
    # = 2 * (w_weighted_convolution with a)
    # For each i: g[i] = 2 * sum_{k=i}^{i+n-1} w[k] * a[k-i]
    # = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    g = np.array([2.0 * np.dot(a, w[i:i+n]) for i in range(n)])
    return g
```

This is the same structure as the subgradient computation but with weighted entries. For n=384, it's 384 dot products of length 384, so ~295k flops. Very fast.
