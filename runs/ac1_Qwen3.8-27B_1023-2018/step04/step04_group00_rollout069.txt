    u = np.asarray(u, dtype=np.float64)

    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:

def _pair_move(p, rng):
    n = p.size
    k = int(np.argmax(_conv(p)))
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo >= hi:
        return p
    cand = p[lo:hi + 1]
    i_min = lo + int(np.argmin(cand))
    i_max = lo + int(np.argmax(cand))
    if i_min == i_max:
        return p
    best = p.copy()
    best_sc = _score_fft(p)
    base = p.copy()
    pm = float(p[i_max])
    if pm <= 0.0:
        return p
    for tv in (0.03, 0.06, 0.10, 0.16, 0.24, 0.35, 0.5):
        tt = tv * pm
        if tt > pm:
            continue
        q = base.copy()
        q[i_max] = pm - tt
        q[i_min] = p[i_min] + tt
        q = np.maximum(q, 0.0)
        sc = _score_fft(q)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best = q.copy()
    return best

            lstop = min(final_end - max(30.0, 0.5 * rem), time.monotonic() + 0.45 * rem)
            try:
                p2 = _lbfgs(cur, lstop, consider=consider)
                if p2 is not None and p2.size == nb:
                    cur = np.maximum(p2, 0.0)
                    ss = float(cur.sum())
                    if ss > 1e-300:
                        cur = cur / ss
            except Exception:
                pass
        cstop = final_end - 6.0
        if cstop > time.monotonic() + 12.0:
            try:
                p3, _ = _coord_desc(cur, cstop, seed=int(seed + 987654321),
                                    consider=consider, sigma0=0.02)
                if p3 is not None and p3.size == nb:
                    cur = np.maximum(p3, 0.0)
                    ss = float(cur.sum())
                    if ss > 1e-300:
                        cur = cur / ss
            except Exception:
                pass

    consider(cur)
    consider(best_arr)

    # Validate with exact evaluator and return.
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(384, 1.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            f2 = parent
            ls2 = [float(x) for x in np.clip(np.asarray(f2, dtype=np.float64), 0, 1000)]
            final_sc = float(evaluate_sequence(ls2))
            if not math.isfinite(final_sc):
                return [1.0]
            ls = ls2
        return ls
    except Exception:
        return [1.0]
