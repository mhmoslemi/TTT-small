def optimize_smooth(parent, tau, max_iter=200, tol=1e-12):
    n = len(parent)
    a0 = parent / parent.sum()  # normalize to sum=1
    b0 = np.log(a0 + 1e-30)  # initial softmax parameter
    # Actually, better to work directly with a in the simplex using a different parameterization
    
    # Or: use scipy.optimize.minimize with constraints
    # Actually, the softmax parameterization is clean. Let's use it.
    
    def objective(b):
        a = softmax(b)
        conv = np.convolve(a, a)
        return (1/tau) * (tau * conv.max() + np.log(np.sum(np.exp(tau * conv - tau * conv.max()))))
    
    def gradient(b):
        a = softmax(b)
        conv = np.convolve(a, a)
        shifted = tau * conv - tau * conv.max()

        t1 = np.correlate(p, a, mode='valid')
