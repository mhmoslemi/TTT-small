def _lp_direction(f, n, M, timeout=3.0):
    """Find g maximizing sum(g) s.t. conv(f,g) <= M, g >= 0."""
    # conv(f, g)[j] = sum_i f[i]*g[j-i] for valid i

    # Actually A[j, k] = f[j-k] if 0 <= j-k < n, else 0

        return None

        return None
    g = np.clip(res.x, 0, None)
    if g.sum() < 1e-15:
        return None
    return g

import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv(a):
    n = len(a)
    if n <= 4096:
        c = np.convolve(a, a)
    else:
        fsize = next_fast_len(2*n-1)
        F = rfft(a, fsize)
        c = irfft(F*F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    return 2.0*n*m/(s*s)

def _project_simplex(v):
    """Project v onto {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Subgradient of y^T P_{j_star} y w.r.t. y, for the conv problem."""
    n = len(y)
    sg = np.zeros(n)
    for i in range(n):
        k = j_star - i
        if 0 <= k < n:
            sg[i] = 2.0 * y[k]
    return sg

    """One FW step: move towards the vertex that maximizes the subgradient."""
    n = len(y)
    sg = _fw_subgrad(y, j_star)
    k_star = int(np.argmax(sg))
    y_new = (1.0 - t) * y
    y_new[k_star] += t
    return y_new

    """Projected subgradient step."""
    y_new = y - t * sg
    return _project_simplex(y_new)

def _lp_dir(f, n, M, timeout=3.0):
    A = np.zeros((2*n-1, n))
    for k in range(n):
        A[k:k+n, k] = f
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:

                      bounds=bounds, method='highs',
                      options={'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    return g if g.sum() > 1e-15 else None

    """Main optimization loop combining FW, SGD, LP, and CD."""
    n = len(f0)
    x = f0 / f0.sum()

    best = x.copy()
    best_sc = cur

    sgd_lr = 0.01

        m = float(b.max())
        j_star = int(np.argmax(b))
        cur = 2.0 * n * m  # since sum(x)=1
        
        if cur < best_sc:
            best_sc = cur
            best = x.copy()
        
        # FW step
        t = 2.0 / (2.0 + it)
        x_fw = _fw_step(x, t, j_star)
        
        # SGD step
        sg = _fw_subgrad(x, j_star)
        sgd_lr = 0.05 / (1.0 + 0.01 * it)
        x_sgd = _sgd_step(x, sg, sgd_lr)
        
        # LP step (less frequently)
        if use_lp and it % 10 == 5:
            M = float(b.max())
            g = _lp_dir(x, n, M, timeout=2.0)
            if g is not None:
                g = g / g.sum()  # normalize to simplex
                t_lp = 0.3
                x_lp = (1-t_lp)*x + t_lp*g
                x_lp = np.clip(x_lp, 0, None)
                if x_lp.sum() > 0:
                    x_lp = x_lp / x_lp.sum()
                if _score(x_lp) < _score(x):
                    x = x_lp
                    it += 1
                    continue
        
        # Choose between FW and SGD based on immediate score
        sc_fw = _score(x_fw)
        sc_sgd = _score(x_sgd)
        
        if min(sc_fw, sc_sgd) < cur:
            if sc_fw < sc_sgd:
                x = x_fw
            else:
                x = x_sgd
        elif sc_sgd < cur:
            x = x_sgd
        else:
            # Try occasional random perturbation
            if it % 50 == 0:
                noise = rng.normal(0, 0.02, n)
                x_pert = np.clip(x + noise, 0, None)
                if x_pert.sum() > 0:
                    x_pert = x_pert / x_pert.sum()
                    if _score(x_pert) < cur:
                        x = x_pert
            
        it += 1
    
    return best, best_sc

    lo1 = max(0, j_star - n + 1)
    hi1 = min(j_star, n - 1)
    if lo1 <= hi1:
        ks = np.arange(lo1, hi1 + 1)
        sg[ks] += y[j_star - ks]
    # sg[k] += y[k-j*] if 0<=k-j*<n
    lo2 = max(0, j_star)
    hi2 = min(j_star + n - 1, n - 1)
    if lo2 <= hi2:
        ks = np.arange(lo2, hi2 + 1)
        sg[ks] += y[ks - j_star]
    return sg


def _fw_step(x, t, j_star):
    """Frank-Wolfe step: move toward argmin of subgradient."""
    n = len(x)
    sg = _subgrad_conv_fast(x, j_star)
    k_star = int(np.argmin(sg))
    x_new = (1.0 - t) * x
    x_new[k_star] += t
    return x_new


def _lp_direction(f, n, M, timeout=3.0):
    A = np.zeros((2 * n - 1, n))
    for k in range(n):
        A[k:k + n, k] = f
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _cd_polish(f, stop_time, max_sweeps=200, seed=0):
    """Coordinate descent on simplex-normalized sequence."""
    n = len(f)
    x = f / f.sum()
    cur = _score(x)
    best = x.copy()
    best_sc = cur
    rng = np.random.default_rng(seed)
    b = _conv(x)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n) if sweep > 0 else np.arange(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = x[i]
            # Generate candidates around current value
            cands = set()
            cands.add(orig)
            cands.add(0.0)
            if orig > 1e-12:
                for m in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
                    cands.add(orig * m)
            else:
                base = max(1e-4, 0.05 / n)
                for m in [0.5, 1.0, 2.0, 5.0, 10.0]:
                    cands.add(base * m)

            best_v = orig
            best_local = cur
            for v in cands:
                v = max(0.0, min(1000.0, v))
                x[i] = v
                s = x.sum()
                if s < 0.01:
                    x[i] = orig
                    continue
                sc = _score(x)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_v = v
            x[i] = best_v
            if best_local < cur - 1e-15:
                cur = best_local
                improved = True
                if cur < best_sc:
                    best_sc = cur
                    best = x.copy()
        if not improved:
            if sweep >= 3:
                break
            # Try 2-coord moves
            for _ in range(100):
                if time.monotonic() > stop_time:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                oi, oj = x[i], x[j]
                bi, bj, bsc = oi, oj, cur
                for mi in [0.3, 0.6, 1.4, 2.0, 0.0]:
                    for mj in [0.3, 0.6, 1.4, 2.0, 0.0]:
                        if (mi, mj) == (1.0, 1.0):
                            continue
                        x[i] = oi * mi if oi > 1e-15 else 0.0
                        x[j] = oj * mj if oj > 1e-15 else 0.0
                        if x.sum() < 0.01:
                            x[i], x[j] = oi, oj
                            continue
                        sc = _score(x)
                        if sc < bsc - 1e-15:
                            bsc = sc
                            bi, bj = x[i], x[j]
                x[i], x[j] = bi, bj
                if bsc < cur - 1e-15:
                    cur = bsc
                    if cur < best_sc:
                        best_sc = cur
                        best = x.copy()
    return best, best_sc


def _main_optimize(f0, stop_time, seed=0):
    """Combined FW + LP + SGD optimization on simplex."""
    n = len(f0)
    x = f0.copy()
    x = np.clip(x, 0, None)
    if x.sum() < 1e-10:
        x = np.ones(n)
    x = x / x.sum()

    cur = _score(x)
    best = x.copy()
    best_sc = cur
    rng = np.random.default_rng(seed)

    it = 0
    t_fw = 1.0
    lr_sgd = 0.08
    lp_interval = max(5, n // 50)
    stall = 0
    max_stall = 300

    while time.monotonic() < stop_time - 3.0:
        b = _conv(x)
        m = float(b.max())
        j_star = int(np.argmax(b))
        cur = 2.0 * n * m  # sum(x)=1 so score = 2n*max_conv

        if cur < best_sc - 1e-14:
            best_sc = cur
            best = x.copy()
            stall = 0
        else:
            stall += 1

        # FW step
        t_fw = 2.0 / (2.0 + it)
        x_fw = _fw_step(x, t_fw, j_star)
        sc_fw = _score(x_fw)

        # SGD step
        sg = _subgrad_conv_fast(x, j_star)
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-15:
            lr_adapt = lr_sgd / (1.0 + 0.005 * it)
            x_sgd = _project_simplex(x - lr_adapt * sg)
            sc_sgd = _score(x_sgd)
        else:
            x_sgd = x.copy()
            sc_sgd = cur

        # LP step (periodically)
        sc_lp = cur
        x_lp = None
        if it % lp_interval == lp_interval // 2:
            g = _lp_direction(x, n, m, timeout=2.0)
            if g is not None:
                g = g / g.sum()
                for t_lp in [0.3, 0.5, 0.8, 0.2]:
                    x_try = (1 - t_lp) * x + t_lp * g
                    x_try = np.clip(x_try, 0, None)
                    if x_try.sum() > 0:
                        x_try = x_try / x_try.sum()
                        sc_try = _score(x_try)
                        if sc_try < sc_lp:
                            sc_lp = sc_try
                            x_lp = x_try

        # Choose best move
        moves = [(sc_fw, x_fw), (sc_sgd, x_sgd)]
        if x_lp is not None:
            moves.append((sc_lp, x_lp))

        best_move_sc = min(m[0] for m in moves)
        if best_move_sc < cur - 1e-15:
            for sc_m, x_m in moves:
                if sc_m == best_move_sc:
                    x = x_m
                    break
        elif stall > max_stall // 2:
            # Perturbation to escape
            noise = rng.normal(0, 0.05 * np.mean(x) + 1e-4, n)
            x_p = np.clip(x + noise, 0, None)
            if x_p.sum() > 0:
                x_p = x_p / x_p.sum()
                sc_p = _score(x_p)
                if sc_p < cur + 0.01:
                    x = x_p
            if stall > max_stall:
                x = best.copy()
                x = np.clip(x + rng.normal(0, 0.1 * np.mean(x) + 1e-4, n), 0, None)
                if x.sum() > 0:
                    x = x / x.sum()
                stall = 0
                max_stall = int(max_stall * 1.3)
                lr_sgd *= 1.5

        it += 1

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = None
    if parent is None or len(parent) == 0:
        n = 400
        parent = np.ones(n)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    if parent.sum() < 0.01:
        parent = np.ones(400)
    n_parent = len(parent)
    n = n_parent

    # Fallback
    fallback = parent.copy()
    try:
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(200)
        fallback_sc = 2.0

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_update(seq, sc):
        nonlocal best_seq, best_score
        if seq is None or len(seq) == 0:
            return
        seq = np.clip(np.asarray(seq, dtype=np.float64), 0, 1000)
        if not np.all(np.isfinite(seq)) or seq.sum() < 0.01:
            return
        if sc < best_score - 1e-14:
            try:
                exact = float(evaluate_sequence([float(x) for x in seq]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-14:
                best_seq = seq.copy()
                best_score = exact

    # Generate initializations
    i_arr = np.arange(n, dtype=np.float64)
    ctr = (n - 1) / 2.0
    inits = []

    # Parent
    inits.append(parent.copy())

    # Raised cosine (Hann window)
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / n)))
    # Flatter top
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * (i_arr - n * 0.15) / (n * 0.7))))
    # Parabolic
    inits.append(np.maximum(0.0, 1.0 - ((i_arr - ctr) / (0.5 * n)) ** 2))
    # Sinc-like (approximate)
    x_sinc = np.sin(np.pi * (i_arr - ctr) / (0.35 * n)) / (np.pi * (i_arr - ctr) / (0.35 * n) + 1e-10)
    inits.append(np.maximum(0.01, x_sinc))
    # Two-lobe Gaussian
    inits.append(np.exp(-0.5 * ((i_arr - 0.3 * n) / (0.12 * n)) ** 2) +
                 np.exp(-0.5 * ((i_arr - 0.7 * n) / (0.12 * n)) ** 2))
    # Flat with tapered edges
    taper = 0.3
    flat = np.ones(n)
    edge = int(taper * n)
    ramp = np.linspace(0, 1, edge)
    flat[:edge] = ramp
    flat[-edge:] = ramp[::-1]
    inits.append(flat)
    # Random
    rng_init = np.random.default_rng(seed)
    inits.append(rng_init.random(n) + 0.01)
    # Parent + noise
    inits.append(np.abs(parent + rng_init.normal(0, 0.1 * np.mean(parent), n)) + 1e-4)
    # Cosine with phase
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * (i_arr + n * 0.05) / n)))

    # Deduplicate and normalize
    inits_dedup = []
    for ix in inits:
        ix = np.clip(ix, 0, None).astype(np.float64)
        if ix.sum() < 1e-10:
            ix = np.ones(n) * 0.01
        key = ix.tobytes()
        if all(key != k for k in [y.tobytes() for y in inits_dedup]):
            inits_dedup.append(ix)

    # Phase 1: Main optimization on each init (~55% time)
    phase1_end = t0 + budget * 0.55
    time_per_init = max(10.0, (phase1_end - t0) / len(inits_dedup))

    for idx, init in enumerate(inits_dedup):
        if time.monotonic() > phase1_end - 15.0:
            break
        if best_score < 1.42:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init * 1.2)
        try:
            res, sc = _main_optimize(init, stop, seed=seed + 100 * (idx + 1))
        except Exception:
            continue
        _try_update(res, sc)

    # Phase 2: LP-focused refinement on best (~20% time)
    phase2_end = t0 + budget * 0.75
    if time.monotonic() < phase2_end - 10.0:
        stop_p2 = min(phase2_end - 3.0, time.monotonic() + (phase2_end - time.monotonic()) * 0.85)
        x0 = best_seq / best_seq.sum()
        # Run LP loop
        f = x0.copy()
        it = 0
        while time.monotonic() < stop_p2 - 2.0:
            b = _conv(f)
            M = float(b.max())
            if M < 1e-18:
                break
            g = _lp_direction(f, n, M, timeout=3.0)
            if g is None:
                break
            g = g / g.sum()
            improved = False
            for t in [0.3, 0.5, 0.7, 0.2, 0.8, 0.1]:
                f_new = (1 - t) * f + t * g
                f_new = np.clip(f_new, 0, None)
                if f_new.sum() < 0.01:
                    continue
                f_new = f_new / f_new.sum()
                sc_new = _score(f_new)
                if sc_new < _score(f) - 1e-14:
                    f = f_new
                    improved = True
                    break
            if not improved:
                # Add small noise
                f = np.clip(f + np.random.default_rng(42 + it).normal(0, 0.02, n), 0, None)
                f = f / f.sum()
            it += 1
        _try_update(f, _score(f))

    # Phase 3: CD polishing (~20% time)
    phase3_end = t0 + budget * 0.92
    if time.monotonic() < phase3_end - 5.0:
        try:
            f_c, sc_c = _cd_polish(best_seq, phase3_end, max_sweeps=100, seed=seed + 555)
        except Exception:
            f_c, sc_c = best_seq.copy(), best_score
        _try_update(f_c, sc_c)

    # Phase 4: Try nearby lengths (~5% time)
    phase4_end = deadline - 10.0
    if time.monotonic() < phase4_end - 15.0:
        for L in [n - 8, n + 8, n - 16]:
            if L < 50 or time.monotonic() > phase4_end - 8.0:
                continue
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n), best_seq)
            stop_l = min(phase4_end - 4.0, time.monotonic() + max(5.0, 15.0))
            try:
                res_l, sc_l = _main_optimize(init_l, stop_l, seed=seed + 80000 + L)
            except Exception:
                continue
            _try_update(res_l, sc_l)

    # Phase 5: Final CD polish
    final_end = deadline - 5.0
    if time.monotonic() < final_end - 8.0:
        try:
            f_c, sc_c = _cd_polish(best_seq, final_end, max_sweeps=150, seed=seed + 777)
        except Exception:
            f_c, sc_c = best_seq.copy(), best_score
        _try_update(f_c, sc_c)

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0, 1.0, 1.0]
        return [float(x) for x in fallback]

    best_score = min(best_score, final_sc)

    result = [max(0.0, min(1000.0, float(x))) for x in np.asarray(best_seq)]
    if not result:
        result = [1.0]
    if any(math.isnan(x) or math.isinf(x) for x in result):
        result = [float(x) for x in fallback]

    return result
