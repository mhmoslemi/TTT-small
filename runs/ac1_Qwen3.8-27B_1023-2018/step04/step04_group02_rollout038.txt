import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len

def _conv(a):
    n = len(a)
    if n <= 2048:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    c = irfft(F*F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_dir(f, n, M, weights=None, timeout=5.0):
    """Find LP direction: maximize sum(g) s.t. conv(f,g) <= M*w, g>=0."""
    A = _build_A(f, n)
    c = -np.ones(n)
    if weights is not None:
        b_ub = M * weights
    else:
        b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _gradient_step(a, n, lr, num_active=3):
    """Take a gradient descent step on the score."""
    conv_a = _conv(a)
    S = float(a.sum())
    B = float(conv_a.max())
    score = 2.0 * n * B / (S * S)
    
    # Find active positions (within tolerance of max)
    tol = 1e-8 * B + 1e-15
    active = np.where(conv_a >= B - tol)[0]
    if len(active) > num_active:
        # Keep only the top ones
        active = active[np.argsort(conv_a[active])[-num_active:]]
    
    # Compute gradient
    # For each active position p, d(conv_a[p])/d(a[i]) = a[p-i] if 0<=p-i<n
    grad_conv = np.zeros(n)
    for p in active:
        for i in range(n):
            j = int(p) - i
            if 0 <= j < n:
                grad_conv[i] += a[j]
    grad_conv /= len(active)
    
    # d(score)/d(a[i]) = 2n/S^2 * grad_conv[i] - 2*score*a[i]/S
    grad = (2.0 * n / (S * S)) * grad_conv - (2.0 * score / S) * a
    
    # Projected gradient step
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    
    # Normalize to keep sum the same (optional)
    if a_new.sum() > 0:
        # Don't normalize - the score is scale-invariant
        pass
    
    return a_new, score
