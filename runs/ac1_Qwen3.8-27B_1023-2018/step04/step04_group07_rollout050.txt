    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    """Full convolution using FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    """Compute the exact score (scale-invariant)."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _build_lp_matrix(a, n, constraint_indices=None):
    """
    Build the constraint matrix for the LP:
    (a*g)[k] - t <= 0 for each k in constraint_indices.
    
    Returns A matrix of shape (len(constraint_indices), n+1)
    where columns 0..n-1 are for g[0..n-1] and column n is for t.
    """
    if constraint_indices is None:
        constraint_indices = np.arange(2 * n - 1)
    
    m = len(constraint_indices)
    A = np.zeros((m, n + 1))
    for row, k in enumerate(constraint_indices):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            j = k - i  # index into g
            A[row, j] = a[i]
        A[row, n] = -1.0  # -t
    
    return A

def _lp_direction(a, n, time_limit=8.0, all_constraints=True):
    """
    Solve: minimize t s.t. (a*g)[k] <= t, g >= 0, sum(g) = 1
    Returns (g_normalized, t_opt) or (None, None).
    """
    n_cs = 2 * n - 1
    if all_constraints:
        idx = np.arange(n_cs)
    else:
        # Focus on the most violated / most active constraints
        b = _conv_full(a)
        M = b.max()
        # Include constraints where b is close to M, plus some spread
        thr = M * 0.5
        idx = np.where(b >= thr)[0]
        if len(idx) < 50:
            idx = np.argsort(-b)[:100]
    
    A = _build_lp_matrix(a, n, idx)
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    bounds = [(0, None)] * (n + 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.zeros(len(idx)),
                      A_eq=A_eq, b_eq=np.array([1.0]),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception as e:
        return None, None
    
    if res.status not in (0, 3) or res.x is None:
        return None, None
    
    g = np.clip(res.x[:n], 0, None)
    g_sum = g.sum()
    if g_sum < 1e-12:
        return None, None
    g = g / g_sum
    return g, float(res.x[n])
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _lp_direction(a, n, time_limit=8.0):
    """Solve min t s.t. (a*g)[k] <= t, g>=0, sum(g)=1."""
    n_cs = 2 * n - 1
    
    A = np.zeros((n_cs, n + 1))
    for k in range(n_cs):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            A[k, k - i] = a[i]
        A[k, n] = -1.0
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.zeros(n_cs),
                      A_eq=A_eq, b_eq=np.array([1.0]),
                      bounds=[(0, None)] * (n + 1),
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None, None
    
    if res.status not in (0, 3) or res.x is None:
        return None, None
    
    g = np.clip(res.x[:n], 0, None)
    g_sum = g.sum()
    if g_sum < 1e-12:
        return None, None
    return g / g_sum, float(res.x[n])

def _line_search(a, g, stop_time, max_evals=30):
    """Find best t in [0,1] for a_t = normalize((1-t)*a + t*g)."""
    best_t = 0.0
    best_sc = _score(a)
    
    # Coarse scan
    ts = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    for t in ts:
        if time.monotonic() > stop_time or max_evals <= 0:
            break
        a_t = (1 - t) * a + t * g
        a_t = np.clip(a_t, 0, None)
        s = a_t.sum()
        if s < 1e-10:
            continue
        a_t = a_t / s
        sc = _score(a_t)
        max_evals -= 1
        if sc < best_sc:
            best_sc = sc
            best_t = t
    
    # Golden section around best_t
    if 0 < best_t < 1 and max_evals > 8:
        if best_t == 0.05:
            lo, hi = 0.0, 0.15
        elif best_t == 1.0:
            lo, hi = 0.5, 1.0
        else:
            lo = max(0, best_t - 0.1)
            hi = min(1, best_t + 0.1)
        
        gr = (1 + math.sqrt(5)) / 2
        for _ in range(min(8, max_evals)):
            if time.monotonic() > stop_time:
                break
            m1 = hi - (hi - lo) / gr
            m2 = lo + (hi - lo) / gr
            a1 = np.clip((1-m1)*a + m1*g, 0, None)
            s1 = a1.sum()
            sc1 = _score(a1/s1) if s1 > 1e-10 else math.inf
            a2 = np.clip((1-m2)*a + m2*g, 0, None)
            s2 = a2.sum()
            sc2 = _score(a2/s2) if s2 > 1e-10 else math.inf
            if sc1 < sc2:
                hi = m2
            else:
                lo = m1
            t_mid = (lo + hi) / 2
            a_mid = np.clip((1-t_mid)*a + t_mid*g, 0, None)
            s_mid = a_mid.sum()
            sc_mid = _score(a_mid/s_mid) if s_mid > 1e-10 else math.inf
            if sc_mid < best_sc:
                best_sc = sc_mid
                best_t = t_mid
    
    return best_t, best_sc
```

**Alternative: Direct convex minimization (not actually convex but close)**

The score f(a) = 2n * max_k (a*a)[k] / (sum a)².

If I fix the sum (normalize to 1), I minimize 2n * max_k (a*a)[k]. This is the max of n*(2n-1) quadratic functions, which is convex on the nonnegative orthant? Let me check.

(a*a)[k] = sum_i a[i]*a[k-i]. The Hessian with respect to a is:
H[j,l] = ∂²(a*a)[k]/(∂a[j]∂a[l])

This is nonzero only when {j,l} = {k-j', k-i'} for some valid pairs. Specifically:
H[j,l] = 1 if j+l = k (and j≠l, both valid)
H[j,j] = 1 if 2j = k (j valid) [from the a[j]² term]
H[j,l] = 0 otherwise

Wait, let me compute more carefully:
(a*a)[k] = sum_i a[i]*a[k-i]

∂/∂a[j]: a[k-j] (+ a[k-j] if j≠k-j, giving 2a[k-j])

Actually I already computed this: ∂(a*a)[k]/∂a[j] = 2*a[k-j] (if 0<=k-j<n) else 0.

∂²/∂a[j]∂a[l]: ∂(2a[k-j])/∂a[l] = 2*δ(k-j, l) = 2*δ(j+l, k)

So H[j,l] = 2*δ(j+l, k). This is a "Hankel" matrix (constant on anti-diagonals).

For k = n-1 (the central coefficient): H[j,l] = 2*δ(j+l, n-1). This is a persymmetric matrix.

Is this PSD? The eigenvalues of the matrix with H[j,l] = 2*δ(j+l, n-1) for j,l in {0,...,n-1}:
This is a permutation matrix (reversed) times 2. Its eigenvalues are 2*unit_circle_roots. Specifically, the reversed identity matrix has eigenvalues that are ±1 (or complex for even n). So it's NOT PSD in general.

This means (a*a)[n-1] is NOT convex as a function of a! (It's not even concave.) So maximizing or minimizing it is non-convex.

Wait, but on the nonnegative orthant? Let me check with n=2: (a*a)[1] = 2*a[0]*a[1]. The Hessian is [[0,2],[2,0]], eigenvalues 2 and -2. Not PSD. So even on the positive orthant, it's not convex (the Hessian is indefinite).

So the problem is genuinely non-convex, and the LP-based approach (which is a kind of block coordinate method) is appropriate.
