            score_new = float(2.0 * n * val_new)
            if math.isfinite(score_new) and score_new < best_score:
                best_val = val_new
                best_score = score_new
                best_x = x_new.copy()
                if consider is not None:
                    try:
                        consider(best_x)
                    except Exception:
                        pass
            # Adaptive L
            if (it + 1) % 200 == 0:
                c_x = _conv_self(x)
                val_x = float(c_x.max())
                if val_new > val_x * 1.001 + 1e-15:
                    L *= 1.3
                elif val_new < val_x * 0.999:
                    L = max(1.0, L * 0.92)

        x = x_new

    return best_x, best_score


def _coord_polish(p, stop_time, max_sweeps=50000, seed=999, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p, math.inf
    p = p / s

    G = 40 if n <= 2048 else 32
    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 1.0:
            break
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            S = 1.0
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end]
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(20.0 * v, 40.0 / float(n), 0.02, 1e-12)
            x_hi = min(x_hi, 5.0)
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, 1.0 / float(n))

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.concatenate((pts, [v]))
            pts = np.unique(pts)

            diag = i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                aff = base[None, :] + w[:, None] * infl[None, :]
                dval = base[diag] + w * infl[diag] + w * w
                maff = np.maximum(aff, dval[:, None])
                maff = np.maximum(maff, max_const)
                vals = (2.0 * n * maff).astype(np.float64)
                return vals

            try:
                vals = eval_xs(pts)
                if not np.any(np.isfinite(vals)):
                    continue
                imin = int(np.argmin(vals))
                best_x = float(pts[imin])
                best_val_s = float(vals[imin])

                dw = x_hi / float(G)
                lo = max(0.0, best_x - 3.0 * dw)
                hi = min(x_hi, best_x + 3.0 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 48, dtype=np.float64)
                fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    if float(vals2[imin2]) < best_val_s:
                        best_x = float(fine[imin2])
                        best_val_s = float(vals2[imin2])

                if math.isfinite(best_val_s) and best_val_s < cur_score - 1e-12:
                    w = best_x - v
                    if abs(w) > 1e-15:
                        p[i] = best_x
                        c[start:end] += w * infl
                        di = 2 * i
                        if 0 <= di < Ltot:
                            c[di] += w * w
                        cur_score = best_val_s
                        improved = True
            except Exception:
                continue

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4 and time.monotonic() < stop_time - 8.0:
                try:
                    p = p * np.exp(rng.normal(0.0, 0.03, n, dtype=np.float64))
                    p = np.maximum(p, 0.0)
                    ps = float(p.sum())
                    if ps > 0:
                        p /= ps
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                except Exception:
                    break
            elif flat >= 8:
                break

        sc = float(2.0 * n * c.max())
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_score = sc
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

    return best_p, best_score


def _lse_optimize(a_init, stop_time, consider=None):
    a = np.maximum(np.asarray(a_init, dtype=np.float64).ravel(), 0.0).copy()
    n = a.size
    s = float(a.sum())
    if s < 1e-300:
        return a
    a = a / s

    # Softmax parameterization
    eps = 1e-300
    u = np.log(np.maximum(a, eps), dtype=np.float64)
    u -= float(np.mean(u))

    L = 2 * n - 1
    size = next_fast_len(L)

    def lse_func(uu, T):
        aa = np.exp(uu - np.max(uu))
        ss = float(aa.sum())
        if ss < eps:
            return 1e100, np.zeros_like(uu)
        aa = aa / ss
        F = rfft(aa, size)
        c = irfft(F * F, size)[:L]
        np.maximum(c, 0.0, out=c)
        cmax = float(c.max())
        if not math.isfinite(cmax):
            return 1e100, np.zeros_like(uu)
        T_eff = max(eps, T)
        scaled = (c - cmax) / T_eff
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            f = cmax
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            f = cmax + T_eff * math.log(sum_exp / float(L))
            w = exps / sum_exp
        # Gradient via cross-correlation
        # grad of f w.r.t. a_i -> chain rule to u
        # df/da_i = 2 * sum_k w[k] * (a[k-i] + a[i-k+... ]) / L ... 
        # Simplified: use numeric or FFT-based
        # g_a[i] = 2 * (a * w)[i] in appropriate sense
        # Actually: d c[k]/d a[i] = 2 a[k-i] (if valid)
        # So d max / da[i] = sum_k w[k] * 2 * a[k-i]
        # This is 2 * (w_reversed * a) cross-correlation
        size_x = next_fast_len(3 * n - 2)
        Fw = rfft(w[::-1], size_x)
        Fa = rfft(aa, size_x)
        xc = ir.fft(Fw * Fa, size_x)[:3*n-2] if False else None
        # Use simpler: for each i, g_a[i] = 2 * sum_k w[k]*a[k-i]
        # = 2 * (w convolved with reversed a) at appropriate offset
        # Let's just compute directly for small n or use FFT
        Fw2 = rfft(w, size)
        Fa2 = rfft(aa, size)
        temp = irfft(Fw2 * Fa2, size)[:L]
        # g_a[i] = 2 * temp[i] (this is w * a at offset i, which is sum_j w[j]a[j-i])
        # Actually we need sum_k w[k] * a[k-i] which is the cross-correlation
        Fwr = rfft(aa[::-1], size)
        temp2 = irfft(Fwr * Fw2, size)[:L]
        # temp2[j] = sum_i aa[i] w[j-i] ... this is aa * w (convolution)
        # We want: for fixed i, sum_k w[k] * aa[k-i] = (w * aa)[i + n - 1] shifted
        # Let me just compute as: g_a[i] = 2 * sum_k w[k] * aa[k-i]
        # = 2 * cross_corr(w, aa)[i] where cross_corr uses appropriate shift
        # Using FFT: cross_corr(w, aa) = ifft(conj(FFT(w)) * FFT(aa)) -- but w is real
        # Actually in the n-lag sense: g_a[i] = 2 * sum_{k:max(0,i)<=k<n} w[k] * aa[k-i]
        # This equals 2 * (w * aa_padded)[i + (n-1)] in the (n-1)-shifted convolution
        # Let me just do it naively with FFT convolution of w and reversed aa
        size_cc = next_fast_len(2 * n - 1)
        Fwr3 = rfft(aa[::-1], size_cc)  # reversed a
        Fw3 = rfft(w, size_cc)
        cc = irfft(Fwr3 * Fw3, size_cc)[:2*n-1]
        # cc[j] = sum_i aa[n-1-i] * w[j-i] for valid indices
        # Hmm this is getting complex. Let me just use the fact that:
        # g_a[i] = 2 * sum_k w[k] * aa[k-i] (k from max(0,i) to n-1, and 0<=k-i<n)
        # = 2 * (aa * w_reversed)[something]
        # For simplicity, compute directly if n <= 4096
        if n <= 5000:
            g_a = np.zeros(n, dtype=np.float64)
            for i in range(n):
                k_lo = max(0, i)
                k_hi = min(n, n + i)  # k-i < n means k < n+i; k>=0
                k_hi = min(n, k_hi)
                k_lo2 = max(0, 0)  # k >= 0
                k_lo = max(k_lo2, i)  # k >= i (so k-i >= 0)
                if k_lo < k_hi:
                    g_a[i] = 2.0 * float(np.dot(w[k_lo:k_hi], aa[k_lo - i:k_hi - i]))
        else:
            g_a = np.zeros(n, dtype=np.float64)

        # Chain rule: u -> a via softmax
        # da_i/du_j = a_i*(delta_ij - a_j)
        # du/d下的 f = sum_i (df/da_i) * (da_i/du_j) = sum_i g_a_i * a_i*(delta_ij - a_j)
        # = a_j * (g_a_j - sum_i g_a_i * a_i)
        tad = float(np.dot(aa, g_a))
        grad_u = aa * (g_a - tad)
        grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
        return float(f), grad_u

    from scipy.optimize import minimize

    stages = [(0.15, 0.20), (0.05, 0.20), (0.015, 0.20), (0.005, 0.20), (0.0015, 0.20)]
    start = time.monotonic()
    avail = stop_time - start - 5.0
    if avail <= 5.0:
        return a

    best_u = u.copy()
    total_w = sum(wt for _, wt in stages)
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        cum += wt
        end_stage = min(stop_time - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 4.0:
            continue
        c = _conv_self(a)
        cmax = float(c.max()) if c.size else 1.0
        T = max(eps, frac * cmax)
        maxit = 30
        opts = {"maxiter": maxit, "gtol": 1e-12, "ftol": 1e-15, "maxls": 15, "maxcor": 15, "disp": False}
        for round_i in range(3):
            if time.monotonic() > end_stage - 2.0:
                break
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                xu = np.asarray(res.x, dtype=np.float64).ravel()
                if np.all(np.isfinite(xu)):
                    f0, _ = lse_func(best_u, T)
                    f1, _ = lse_func(xu, T)
                    if f1 < f0:
                        best_u = xu.copy()

        aa_new = np.exp(best_u - np.max(best_u))
        ss = float(aa_new.sum())
        if ss > eps:
            aa_new /= ss
            s_new = _score_fft(aa_new)
            if math.isfinite(s_new):
                a = aa_new

    return a


def _make_init_cosine(n, center=0.5, width=0.4):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x - center) / max(width, 0.01)))
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s > 0:
        a /= s
    return a


def _make_init_gaussian(n, mu=0.5, sigma=0.25):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s > 0:
        a /= s
    return a


def _resample_to(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    m = src.size
    if m == n:
        return src.copy()
    xp = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
    fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / float(n), xp, src)
    fp = np.maximum(fp, 0.0)
    s = float(fp.sum())
    if s > 1e-300:
        fp /= s
    else:
        fp = np.full(n, 1.0 / n, dtype=np.float64)
    return fp


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 20.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(384, dtype=np.float64)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    parent = np.clip(parent, 0.0, 1000.0)
    if float(parent.sum()) < 0.01:
        parent = np.ones(384, dtype=np.float64)
        parent /= 384.0

    n_parent = int(parent.size)
    best_arr = parent.copy()
    best_score = _score_exact(best_arr)
    if not math.isfinite(best_score):
        best_arr = np.ones(64, dtype=np.float64) / 64.0
        best_score = _score_exact(best_arr)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0 or float(a.sum()) < 0.01:
                return
            sc = _score_exact(a)
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_arr = a.copy()
                best_score = sc
        except Exception:
            pass

    # Primary optimization at parent length
    n_primary = n_parent
    a_primary = parent.copy()
    s_pr = float(a_primary.sum())
    if s_pr > 0:
        a_primary = a_primary / s_pr

    # Phase 1: LSE optimization
    p1_end = min(hard_deadline - (budget * 0.5), t0 + budget * 0.25)
    if p1_end > t0 + 10.0:
        try:
            a_lse = _lse_optimize(a_primary, p1_end, consider=consider)
            if a_lse is not None and a_lse.size == n_primary:
                s_lse = float(a_lse.sum())
                if s_lse > 0:
                    a_primary = a_lse / s_lse
                    consider(a_primary)
        except Exception:
            pass

    # Phase 2: FISTA (subgradient with Nesterov)
    p2_end = min(hard_deadline - (budget * 0.3), t0 + budget * 0.60)
    if p2_end > time.monotonic() + 20.0:
        try:
            a_fista, sc_fista = _fista_optimize(a_primary, p2_end, L_init=3.0, consider=consider)
            if a_fista.size == n_primary:
                s_f = float(a_fista.sum())
                if s_f > 0:
                    a_primary = a_fista / s_f
                    consider(a_primary)
        except Exception:
            pass

    # Phase 3: Coordinate descent polish
    p3_end = min(hard_deadline - 30.0, t0 + budget * 0.82)
    if p3_end > time.monotonic() + 20.0:
        try:
            a_cd, sc_cd = _coord_polish(a_primary, p3_end, max_sweeps=50000, seed=seed + 777, consider=consider)
            if a_cd.size == n_primary:
                consider(a_cd)
        except Exception:
            pass

    # Phase 4: Second FISTA pass on the best found
    p4_end = min(hard_deadline - 15.0, t0 + budget * 0.93)
    if p4_end > time.monotonic() + 15.0:
        try:
            a_fista2, _ = _fista_optimize(best_arr, p4_end, L_init=2.0, max_iters=200000, consider=consider)
            if a_fista2.size == best_arr.size:
                consider(a_fista2)
        except Exception:
            pass

    # Phase 5: Final coordinate descent
    p5_end = hard_deadline - 10.0
    if p5_end > time.monotonic() + 10.0 and best_arr.size <= 5000:
        try:
            a_final_cd, _ = _coord_polish(best_arr, p5_end, max_sweeps=20000, seed=seed + 8888, consider=consider)
            if a_final_cd.size == best_arr.size:
                consider(a_final_cd)
        except Exception:
            pass

    # Validate and return
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = evaluate_sequence(ls)
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
