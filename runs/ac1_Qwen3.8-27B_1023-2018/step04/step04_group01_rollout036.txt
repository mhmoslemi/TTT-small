    A = np.zeros((2*n-1, n))

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)

    """Gradient-based optimization using smoothed max."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum
    best_f = f.copy()
    best_score = _score(f)

    alpha = 0.03
    alpha_min = 1e-12
    tau = 50.0
    conv_size = 2 * n - 1

    it = 0
    while time.monotonic() < stop_time - 5.0:
        if it % 500 == 499 and alpha < 1e-8:
            # Perturb to escape
            f = np.clip(f + rng.normal(0, 0.01, n), 0, None)
            f /= f.sum()
            alpha = 0.01
            it += 1
            continue

        b = _conv_full(f)
        S = float(f.sum())
        M = float(b.max())
        if M < 1e-18:
            break
        score = 2.0 * n * M / (S * S)

        if score < best_score - 1e-14:
            best_score = score
            best_f = f.copy()

        # Smoothed max gradient
        b_shifted = b - M
        tau_b = tau * b_shifted
        tau_b -= tau_b.max()
        p = np.exp(tau_b)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(conv_size)
            p[np.argmax(b)] = 1.0
            p_sum = 1.0
        p /= p_sum

        # grad[i] = 2 * sum_j p[i+j] * f[j] = 2 * (p * f_rev)[i + n - 1]
        f_rev = f[::-1]
        P = rfft(p, conv_size)
        Fr = rfft(f_rev, conv_size)
        conv_val = irfft(P * Fr, conv_size)[:conv_size]
        grad = 2.0 * conv_val[n - 1:conv_size]

        # Score gradient (accounting for normalization)
        score_grad = (2.0 * n / (S * S)) * (grad - 2.0 * M / S)

        # Gradient step with projection
        g_raw = f - alpha * score_grad
        f_new = _project_simplex(g_raw)

        sc_new = _score(f_new)
        if sc_new < score - 1e-15:
            f = f_new
            alpha = max(alpha_min, alpha * 1.01)
        else:
            alpha = max(alpha_min, alpha * 0.5)
            if alpha < alpha_min:
                f = np.clip(f + rng.normal(0, 0.005, n), 0, None)
                f /= f.sum()
                alpha = 0.01

        it += 1

    return best_f, best_score


def _coord_polish(f0, stop_time, max_sweeps=100, seed=777):
    """Coordinate descent with exact 1D optimization."""
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, _score(f)
    f = f / S
    S = 1.0
    conv = _conv_full(f)
    best_score = 2.0 * n * conv.max()
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break

            v = f[i]
            S_rest = S - v

            # Influence vector: how conv changes when f[i] changes by dw
            # conv[k] changes by dw * (sum of f[j] where the cross term appears) + dw^2 (at k=2i... wait)
            # Actually: conv[k] = sum_j f[j]*f[k-j]
            # If we change f[i] by dw:
            #   conv[k] += dw * (f[k-i] + f[i-k]) for k != 2i... 
            #   Actually for k: the terms involving f[i] are f[i]*f[k-i] and f[k-i+... 
            #   Let me think more carefully.
            #   conv[k] = sum_j f[j] * f[k-j]
            #   Terms with f[i]: when j=i: f[i]*f[k-i]; when k-j=i, i.e., j=k-i: f[k-i]*f[i]
            #   If i != k-i (i.e., k != 2i), both terms present: dw*f[k-i] + f[k-i]*dw = 2*dw*f[k-i]
            #   If i == k-i (i.e., k = 2i): only one term f[i]*f[i] = f[i]^2, so change is 2*f[i]*dw + dw^2
            #   Wait, that's not right either. Let me be more careful.
            #   conv[k] = sum_j f[j] * f[k-j]
            #   If I change f[i] -> f[i] + dw:
            #   New conv[k] = sum_j (f[j] + dw*delta_{j,i}) * (f[k-j] + dw*delta_{k-j,i})
            #   = sum_j f[j]*f[k-j] + dw * sum_j delta_{j,i}*f[k-j] + dw * sum_j f[j]*delta_{k-j,i} + dw^2 * delta_{j,i}*delta_{k-j,i}
            #   = conv[k] + dw*f[k-i] + dw*f[i] * [1 if k-i valid else 0]... 
            #   
            #   Hmm, this is getting complicated. Let me just use the direct approach.
            #   The coefficient of dw in new conv[k] - conv[k] is:
            #   (a) f[k-i] if 0 <= k-i < n (from the first factor change)
            #   (b) f[i] * [1 if 0 <= k-i < n else 0]... no.
            #   
            #   Let me redo. conv[k] = sum_{j=0}^{n-1} f[j] * f[k-j] (where f[out-of-range] = 0)
            #   Change f[i] by dw.
            #   New conv[k] = conjugate... no, it's just:
            #   = sum_j (f[j] + dw*δ(j,i)) * f[k-j] + (correction for the second factor)
            #   
            #   Actually, since f appears in both slots of the convolution, changing f[i] by dw:
            #   Δconv[k] = dw * f[k-i] + dw * f[i-k] + dw^2 * δ(k, 2i) * δ(0 <= k-i < n)
            #   
            #   Wait: the first term comes from changing the first f[j] (j=i): contributes dw * f[k-i]
            #   The second term comes from changing the second f[k-j] (k-j=i, j=k-i): contributes f[k-i] * dw... 
            #   No, that's the same term if k-i = i, i.e., k = 2i.
            #   
            #   Let me just do it directly:
            #   conv[k] = sum_j f[j] * f[k-j]
            #   d(conv[k])/d(f[i]) = f[k-i] + f[i-(k-i)]... no.
            #   d(conv[k])/d(f[i]) = d/d(f[i]) [sum_j f[j] * f[k-j]]
            #   = f[k-i] (from j=i term: f[i]*f[k-i])
            #   + f[k-i] (from the j where k-j = i, i.e., j = k-i: f[k-i]*f[i])
            #   
            #   Wait, these are the same derivative. Let me be very explicit:
            #   conv[k] = f[0]*f[k] + f[1]*f[k-1] + ... + f[k]*f[0] + ... (only valid indices)
            #   d(conv[k])/d(f[i]) = f[k-i] (if 0<=k-i<n) + f[k-i] (if 0<=k-i<n)
            #   = 2*f[k-i] if 0<=k-i<n and i != k-i
            #   = 2*f[i] if k = 2i (only one term but it's f[i]*f[i], derivative is 2*f[i])
            #   
            #   Actually for k = 2i: the term is f[i]*f[i], and d/d(f[i])[f[i]^2] = 2*f[i]. ✓
            #   For k != 2i: the terms are f[i]*f[k-i] and f[k-i]*f[i] (if both in range), 
            #   and d/d(f[i]) = f[k-i] + f[k-i] = 2*f[k-i]. ✓
            #   
            #   So: infl[k] = 2*f[k-i] if 0 <= k-i < n, else 0.
            #   And: Δconv[k] = dw * infl[k] + dw^2 (if k = 2i and 0 <= 2i < 2n-1)
            #   
            #   The dw^2 term only appears at k=2i (the "self" term).

            infl = np.zeros(2 * n - 1)
            start = i  # k-i >= 0 means k >= i
            end = min(i + n, 2 * n - 1)  # k-i < n means k < i+n
            infl[start:end] = 2.0 * f[:end - start]

            # Self term at k=2i
            self_idx = 2 * i
            has_self = (0 <= self_idx < 2 * n - 1)

            # Search for best value of f[i]
            x_max = max(10.0 * v, 1.0) if v > 1e-8 else 1.0
            x_max = min(x_max, 100.0)

            # Grid search
            n_pts = 80
            x_grid = np.linspace(0.0, x_max, n_pts)

            best_x = v
            best_obj = best_score

            for x in x_grid:
                S_new = S_rest + x
                if S_new < 0.01:
                    continue
                w = x - v
                if abs(w) < 1e-14:
                    m_new = conv.max()
                else:
                    # Update conv efficiently
                    # We need max of conv + w*infl + w^2*δ(k,2i)
                    # Instead of copying, compute max analytically
                    # For most k, new conv[k] = conv[k] + w*infl[k]
                    # For k=2i, new conv[2i] = conv[2i] + w*infl[2i] + w^2
                    tmp = conv + w * infl
                    if has_self:
                        tmp[self_idx] += w * w
                    m_new = float(tmp.max())
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-16:
                    best_obj = obj
                    best_x = x

            # Fine grid around best
            if best_x != v:
                dw_step = x_max / n_pts
                x_lo = max(0.0, best_x - 3 * dw_step)
                x_hi = min(x_max, best_x + 3 * dw_step)
                x_fine = np.linspace(x_lo, x_hi, 40)
                for x in x_fine:
                    S_new = S_rest + x
                    if S_new < 0.01:
                        continue
                    w = x - v
                    if abs(w) < 1e-14:
                        continue
                    tmp = conv + w * infl
                    if has_self:
                        tmp[self_idx] += w * w
                    m_new = float(tmp.max())
                    obj = 2.0 * n * m_new / (S_new * S_new)
                    if obj < best_obj - 1e-16:
                        best_obj = obj
                        best_x = x

            # Accept
            if best_x != v and best_obj < best_score - 1e-15:
                w = best_x - v
                f[i] = best_x
                S += w
                conv += w * infl
                if has_self:
                    conv[self_idx] += w * w
                best_score = best_obj
                improved = True

        if not improved and sweep > 5:
            break

    f = np.clip(f, 0, 1000)
    final_score = _score(f)
    return f, final_score


def _multi_start_lp(f_best, n, stop_time, seed):
    """Try multiple perturbations of the best found and run short LP optimization."""
    rng = np.random.default_rng(seed)
    best_f = f_best.copy()
    best_score = _score(f_best)
    
    n_starts = 10
    time_per_start = max(2.0, (stop_time - time.monotonic() - 10) / n_starts)
    
    for s in range(n_starts):
        if time.monotonic() > stop_time - 10:
            break
        
        # Different perturbation scales
        scales = [0.1, 0.05, 0.2, 0.01, 0.3, 0.02, 0.15, 0.08, 0.25, 0.005]
        pert = rng.normal(0, scales[s % len(scales)], n)
        f_init = np.clip(f_best + pert, 0, None)
        fs = f_init.sum()
        if fs < 1e-10:
            continue
        f_init = f_init / fs
        
        stop_this = min(stop_time - 5, time.monotonic() + time_per_start)
        try:
            res, sc = _lp_optimize(f_init, n, stop_this, seed + s * 1000, max_iters=50)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res = res / res.sum()
                sc = _score(res)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_f = res.copy()
        except Exception:
            pass
    
    return best_f, best_score
