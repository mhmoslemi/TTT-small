```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _subgrad_step(a, step_size):
    """One subgradient step to reduce max convolution."""
    n = len(a)
    b = _conv_full(a)
    k_star = int(b.argmax())
    # Compute subgradient
    sg = np.zeros(n)
    for i in range(n):
        idx = k_star - i
        val = 0.0
        if 0 <= idx < n:
            val += a[idx]
        idx2 = i - k_star
        if 0 <= idx2 < n:
            val += a[idx2]
        sg[i] = val
    # Project to maintain sum
    sg = sg - sg.mean()
    a_new = a - step_size * sg
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new

def _focused_lp(f, n, M, top_k=150, time_limit=10.0):
    b = _conv_full(f)
    sorted_idx = np.argsort(-b)[:top_k]
    # Build constraint matrix for these rows
    A = np.zeros((top_k, n))
    for row, k in enumerate(sorted_idx):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[row, j] = f[idx]
    c = -np.ones(n)
    b_ub = np.full(top_k, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": time_limit})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum
```

Actually, for coordinate descent, changing one element a[i] by delta changes conv(a,a) in a structured way. Specifically, conv(a,a)[k] changes by:
- delta * a[k-i] (from the a[i] * a[k-i] term)
- delta * a[i-k] (from the a[k-i] * a[i] term, same thing)

So the change in conv at lag k is 2 * delta * a[k-i] (with proper bounds).

This means for a 1D coordinate descent, I can compute the new max convolution in O(n) by only updating the affected lags.

Let me implement this:

```python
def _coord_update_conv(b, a, i, delta):
    """Update conv array when a[i] changes by delta. Returns new conv."""
    n = len(a)
    b_new = b.copy()
    old_ai = a[i]
    new_ai = old_ai + delta
    # conv[k] = sum_j a[j]*a[k-j]
    # Terms involving a[i]:
    # j=i: a[i]*a[k-i], valid for k in [i, i+n-1]
```

```python
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    return 2.0 * len(a) * _mc(a) / (s * s)


def _buildA(f, n, kidx):
    k = np.asarray(kidx, dtype=np.int64)
    j = np.arange(n, dtype=np.int64)
    d = k[:, None] - j[None, :]
    m = (d >= 0) & (d < n)
    A = np.zeros((len(kidx), n), dtype=np.float64)
    A[m] = f[d[m]]
    return A


def _fLP(f, n, M, tk=150, tl=5.0):
    b = _conv_full(f)
    idx = np.argpartition(-b, min(tk, 2 * n - 1))[:tk]
    A = _buildA(f, n, idx)
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(tk, M),
                    bounds=[(0, None)] * n, method='highs',
                    options={'presolve': True, 'time_limit': tl})
    except Exception:
        return None
    if r.status not in (0, 3) or r.x is None:
        return None
    g = np.asarray(r.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs


def _sgd(f, step):
    n = len(f)
    b = _conv_full(f)
    k = int(np.argmax(b))
    j = np.arange(n)
    sg = np.zeros(n)
    i1 = k - j
    m1 = (i1 >= 0) & (i1 < n)
    sg[m1] += f[i1[m1]]
    i2 = j - k
    m2 = (i2 >= 0) & (i2 < n)
    sg[m2] += f[i2[m2]]
    sg -= sg.mean()
    f2 = np.clip(f - step * sg, 0, None)
    s = f2.sum()
    return f2 / s if s > 1e-10 else f


def _ls(f, g, stop):
    bolt, bm = 0.0, math.inf
    for t in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0):
        if time.monotonic() > stop:
            break
        fn = np.clip((1 - t) * f + t * g, 0, None)
        ss = fn.sum()
        if ss < 1e-10:
            continue
        fn /= ss
        m = _mc(fn)
        if m < bm:
            bm, bolt = m, t
    if 0.05 < bolt < 0.9:
        lo, hi = max(0.0, bolt - 0.08), min(1.0, bolt + 0.08)
        gr = (1.0 + math.sqrt(5.0)) / 2.0
        for _ in range(5):
            if time.monotonic() > stop:
                break
            a = hi - (hi - lo) / gr
            b = lo + (hi - lo) / gr
            va = _mc(np.clip((1 - a) * f + a * g, 0, None))
            vb = _mc(np.clip((1 - b) * f + b * g, 0, None))
            if va < vb:
                hi = b
            else:
                lo = a
            tm = (lo + hi) / 2.0
            vm = _mc(np.clip((1 - tm) * f + tm * g, 0, None))
            if vm < bm:
                bm, bolt = vm, tm
    return bolt, bm


def _lp_loop(f0, n, stop, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None)
    if f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    if f.sum() < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f.sum()
    bf = f.copy()
    bM = _mc(f)
    M = bM
    prev = bM
    stall = 0
    it = 0
    while time.monotonic() < stop - 1.0:
        if M < prev - 1e-14:
            stall = max(0, stall - 3)
        else:
            stall += 1
        prev = M

        if stall > 35:
            s = stall % 4
            if s == 0:
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            elif s == 1:
                blk = max(2, n // 4)
                f2 = f.copy()
                f2[:blk], f2[n-blk:] = f[n-blk:].copy(), f[:blk].copy()
                f = f2
            elif s == 2:
                f = np.clip(f * (1 + 0.1 * rng.normal(0, 1, n)), 0, None)
            else:
                f = 0.5 * bf + 0.5 * np.abs(rng.normal(0, 1, n))
            sm = f.sum()
            f = f / sm if sm > 1e-10 else bf.copy()
            M = _mc(f)
            stall = 0
            continue

        tk = 150 if stall < 15 else 100
        g = _fLP(f, n, M, tk=tk, tl=5.0)
        if g is None:
            f = _sgd(f, 0.02)
            M = _mc(f)
            stall += 2
        else:
            d = np.linalg.norm(g - f)
            if d < 1e-8:
                f = _sgd(f, 0.015)
                M = _mc(f)
                stall += 1
            else:
                t, m = _ls(f, g, stop)
                if t > 0 and m < M - 1e-15:
                    f = np.clip((1 - t) * f + t * g, 0, None)
                    sm = f.sum()
                    f = f / sm if sm > 1e-10 else f
                    M = _mc(f)
                else:
                    f = _sgd(f, 0.01)
                    M = _mc(f)
                    stall += 1

        if M < bM:
            bM = M
            bf = f.copy()

        # Extra subgradient steps
        if it % 4 == 3 and time.monotonic() < stop - 2.0:
            f2 = _sgd(f, 0.01)
            m2 = _mc(f2)
            if m2 < M - 1e-15:
                f = f2
                M = m2
                if M < bM:
                    bM = M
                    bf = f.copy()
        it += 1
    return bf, bM


def _coord(a, stop, seed):
    n = len(a)
    best = np.clip(a.copy(), 0, None)
    sm = best.sum()
    best = best / sm if sm > 1e-10 else np.ones(n) / n
    bM = _mc(best)
    best_sc = _sc(best)
    rng = np.random.default_rng(seed)
    noimp = 0

    while time.monotonic() < stop - 1.0 and noimp < 15:
        imp = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop - 1.0:
                break
            orig = best[i]
            tot = best.sum()
            mults_i = (0.0, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0) if orig > 1e-12 else (0.0, 0.01, 0.05, 0.1, 0.5)
            bv, bm = orig, best_sc
            for ml in mults_i:
                best[i] = orig * ml if orig > 1e-12 else ml
                ss = best.sum()
                if ss < 1e-10:
                    continue
                sc = _sc(best)
                if sc < bm - 1e-15:
                    bm, bv = sc, best[i]
            best[i] = bv
            if bm < best_sc - 1e-15:
                best_sc = bm
                bM = _mc(best)
                imp = True

        # Pair moves
        for _ in range(min(200, n * 2)):
            if time.monotonic() > stop - 1.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            tv = oi + oj
            if tv < 1e-12:
                continue
            bv, bm = None, best_sc
            for fr in (0.0, 0.2, 0.35, 0.5, 0.65, 0.8, 1.0):
                best[i] = fr * tv
                best[j] = (1 - fr) * tv
                ss = best.sum()
                if ss < 1e-10:
                    continue
                sc = _sc(best)
                if sc < bm - 1e-15:
                    bm, bv = sc, (best[i], best[j])
            if bv is not None:
                best[i], best[j] = bv
                best_sc = bm
                bM = _mc(best)
                imp = True
            else:
                best[i], best[j] = oi, oj

        noimp = 0 if imp else noimp + 1

    return best, best_sc
```
