    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    """Build the matrix A such that (A @ g)[k] = (f*g)[k] for cross-correlation"""
    # (f*g)[k] = sum_j f[j]*g[k-j] for valid j
    # We want A such that (A@x)[k] = sum_j A[k,j] * x[j] = sum_j f[j]*x[k-j]
    # So A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(f, n, M, timeout=5.0):
    """Find direction g that maximizes sum(g) subject to f*g <= M, g >= 0"""
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _lp_dir2(f, n, M, timeout=5.0):
    """Alternative LP: minimize max deviation. Find g close to f with f*g <= M, maximizing sum(g)"""
    # Same as _lp_dir but with an additional soft constraint on ||g - f||
    # Actually, let's try a different approach: minimize sum(g) - f sum subject to f*g <= M
    # This pushes g towards zero, which is bad. Let's stick with the original.
    # Instead, let's try: maximize sum(g) subject to max(f*g) <= M and g_i <= f_i * 2 (box constraint)
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 2.0 * max(f[i], 1e-10)) for i in range(n)]
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _subgradient_step(a, n, step_size=0.01):
    """Take a subgradient step to minimize max(b)"""
    b = _conv(a)
    k_star = int(np.argmax(b))
    M = b[k_star]
    # Gradient of b[k_star] w.r.t. a[i] = 2*a[k_star - i] if 0 <= k_star - i < n
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    # Also consider other near-active lags for robustness
    threshold = M * 0.99
    active = np.where(b >= threshold)[0]
    if len(active) > 1:
        grad_sum = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad_sum[i] += 2.0 * a[j]
        grad = grad_sum / len(active)
    
    # Step in negative gradient direction, projected onto a >= 0, sum(a) = sum(a)
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0)
    # Renormalize to keep sum the same
    if a_new.sum() > 0:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new

            best_local = cur

            best_score = sc_c

    return best_f, best_score

    n_parent = len(parent)

    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(200)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update(res, sc):
        nonlocal best_seq, best_score
        if res is None or sc is None or not math.isfinite(sc):
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in res]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-14:
            best_seq = np.asarray(res, dtype=np.float64).copy()
            best_score = exact

    rng_main = np.random.default_rng(seed)
    n = n_parent
    i_arr = np.arange(n, dtype=np.float64)

    # Build diverse initializations
    inits = [parent.copy()]
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / n)))
    ctr = (n - 1) / 2.0
    inits.append(np.maximum(0.0, 1.0 - ((i_arr - ctr) / (0.45 * n)) ** 2))
    inits.append(np.exp(-0.5 * ((i_arr - 0.3 * n) / (0.12 * n)) ** 2) +
                 np.exp(-0.5 * ((i_arr - 0.7 * n) / (0.12 * n)) ** 2))
    sig = 0.15 * max(1.0, np.mean(parent))
    inits.append(np.abs(parent + rng_main.normal(0, sig, n)) + 1e-4)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.4)
    inits.append(alt * (0.5 + 0.5 * np.cos(2.0 * np.pi * i_arr / n)))
    inits.append(np.ones(n) * 0.5 + 0.5 * np.cos(4.0 * np.pi * i_arr / n + 0.3))
    inits.append(np.maximum(0.0, np.cos(2.0 * np.pi * np.arange(n) * 3.0 / n) ** 2))

    # Deduplicate
    seen = set()
    inits_dedup = []
    for ix in inits:
        ix = np.clip(ix, 0, 1000)
        if ix.sum() < 1e-10:
            continue
        key = hash(ix.tobytes())
        if key not in seen:
            seen.add(key)
            inits_dedup.append(ix)

    # Phase 1: LP optimization on parent length (~55% of budget)
    phase1_end = t0 + budget * 0.52
    time_per_init = max(10.0, (phase1_end - t0) / max(1, len(inits_dedup)))

    for idx, init in enumerate(inits_dedup):
        if time.monotonic() > phase1_end - 10.0:
            break
        if best_score < 1.45:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        if stop <= time.monotonic() + 5.0:
            break
        res, sc = _try_len(n, init, stop, seed + 10000 * (idx + 1), best_score)
        _update(res, sc)

    # Phase 2: Warm restart on best found (~13%)
    phase2_end = t0 + budget * 0.68
    if time.monotonic() < phase2_end - 15.0:
        stop_p2 = phase2_end - 5.0
        res, sc = _try_len(len(best_seq), best_seq.copy() + rng_main.normal(0, 0.01, len(best_seq)),
                           stop_p2, seed + 77777, best_score)
        _update(res, sc)

    # Phase 3: CD polish on best (~12%)
    phase3_end = t0 + budget * 0.80
    if time.monotonic() < phase3_end - 5.0:
        try:
            f_c, sc_c = _cd_polish(best_seq, phase3_end, max_outer=40, seed=seed * 3 + 1)
            _update(f_c, sc_c)
        except Exception:
            pass

    # Phase 4: Try other lengths (~10%)
    phase4_end = t0 + budget * 0.90
    if time.monotonic() < phase4_end - 12.0:
        other_lens = [n - 16, n + 16, n - 32, n + 32, n - 64, n + 64]
        other_lens = [L for L in other_lens if 64 <= L <= 1200 and L != n]
        for L in other_lens[:4]:
            if time.monotonic() > phase4_end - 10.0:
                break
            if len(best_seq) >= L:
                init_l = best_seq[:L].copy()
            else:
                init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
            stop_l = min(phase4_end - 5.0, time.monotonic() + max(8.0, (phase4_end - time.monotonic()) * 0.4))
            res, sc = _try_len(L, init_l, stop_l, seed + 90000 + L, best_score)
            _update(res, sc)

    # Phase 5: Final heavy CD polish (~8%)
    final_end = deadline - 5.0
    if time.monotonic() < final_end - 8.0:
        try:
            f_c, sc_c = _cd_polish(best_seq, final_end, max_outer=60, seed=seed * 7 + 42)
            _update(f_c, sc_c)
        except Exception:
            pass

    # Ultra-final: one more short LP pass if time remains
    if time.monotonic() < deadline - 15.0:
        stop_f = deadline - 8.0
        res, sc = _try_len(len(best_seq), best_seq.copy(), stop_f, seed + 123456, best_score)
        _update(res, sc)

    # Final validation
    try:
        result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
        final_sc = float(evaluate_sequence(result))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc) or final_sc > 100:
        result = [float(x) for x in fallback]

    if not result:
        result = [1.0]

    return result
