import math

from scipy.sparse import csr_matrix

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(416, dtype=np.float64)
    if p.size == 0:

    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:

    return p

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _fw_lp_direction(f, n):
    """Solve the Frank-Wolfe LP to find the best direction.
    Minimize t subject to (f*g)_k <= t for all k, g >= 0, sum(g) = 1.
    Returns g* (the optimal vertex) or None.
    """
    # Build the constraint matrix
    # Variables: g_0, ..., g_{n-1}, t (total n+1)
    # For constraint k: sum_j f_j * g_{k-j} <= t
    #   => sum_j f_j * g_{k-j} - t <= 0
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            gi = k - j  # index of g in the product
            if 0 <= gi < n:
                rows.append(k)
                cols.append(gi)
                vals.append(f[j])
        # -t term
        rows.append(k)
        cols.append(n)  # t is variable n
        vals.append(-1.0)
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    # Objective: minimize t
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # g_i >= 0, t >= 0
    
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 10.0, 'presolve': True})
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    
    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.clip(g, 0.0, None)
    gs = g.sum()
    if gs < 1e-15:
        return None
    g = g / gs
    return g

            best_score = sc
            best_gamma = g

    f_new = (1 - best_gamma) * f + best_gamma * g_star
    f_new = np.clip(f_new, 0.0, None)
    if f_new.sum() < 1e-10:
        return f, 0.0, _score_fast(f)
    f_new = f_new / f_new.sum()
    return f_new, best_gamma, best_score


def _frank_wolfe(f0, stop_time, max_iters=5000):
    """Frank-Wolfe optimization with exact LP oracle."""
    n = len(f0)
    f = f0 / f0.sum()
    best_f = f.copy()
    best_score = _score_fast(f)
    iter_count = 0
    no_imp = 0

    while time.monotonic() < stop_time - 3.0 and iter_count < max_iters:
        ff = _conv_full(f)
        g_star = _fw_lp_direction(f, n, timeout=5.0)
        if g_star is None:
            no_imp += 10
            if no_imp > 200:
                break
            continue

        fg = _cross_conv(f, g_star)
        gg = _conv_full(g_star)

        f_new, gamma, sc = _fw_line_search(f, g_star, n, ff, fg, gg)
        iter_count += 1

        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f_new.copy()
            if gamma > 0:
                f = f_new
            no_imp = 0
        else:
            # Try a smaller step
            g_smaller = g_star * 0.5
            g_smaller = g_smaller / g_smaller.sum()
            fg2 = _cross_conv(f, g_smaller)
            gg2 = _conv_full(g_smaller)
            f_new2, gamma2, sc2 = _fw_line_search(f, g_smaller, n, ff, fg2, gg2)
            if sc2 < best_score - 1e-12:
                best_score = sc2
                best_f = f_new2.copy()
                if gamma2 > 0:
                    f = f_new2
                no_imp = 0
            else:
                no_imp += 1
                if no_imp > 300:
                    break

    return best_f, best_score


def _mirror_descent(f0, stop_time, seed):
    """Multiplicative weights / mirror descent on normalized simplex."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = f0.copy()
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    best_f = f.copy()
    best_score = _score_fast(f)

    lr = 0.5
    no_imp = 0
    max_no_imp = 200

    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-20:
            break

        # Find all peak indices
        threshold = M * 0.99
        peak_idx = np.where(b >= threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])

        # Subgradient: for each peak k, grad[i] += f[k-i]
        grad = np.zeros(n)
        for k in peak_idx:
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                gi = k - i
                if 0 <= gi < n:
                    grad[i] += f[gi]
        grad /= len(peak_idx)

        if grad.max() < 1e-15:
            break

        # Multiplicative update
        grad_scale = grad.max()
        lr_eff = lr / grad_scale
        exponent = -lr_eff * grad
        exponent -= exponent.max()
        weights = np.exp(exponent)
        f_new = f * weights
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        sc = _score_fast(f_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            no_imp = 0
            lr *= 0.995
        else:
            # Damped step
            f_damped = 0.5 * f + 0.5 * f_new
            f_damped = f_damped / f_damped.sum()
            sc_d = _score_fast(f_damped)
            if sc_d < best_score - 1e-13:
                best_score = sc_d
                best_f = f_damped.copy()
                f = f_damped
                no_imp = 0
                lr *= 0.98
            else:
                no_imp += 1
                lr *= 0.95
                if no_imp > max_no_imp:
                    # Random perturbation
                    noise = rng.normal(0, 0.05, n)
                    f = np.maximum(f + noise * f, 1e-15)
                    f = f / f.sum()
                    no_imp = 0
                    lr = 0.3
                    max_no_imp += 100

    return best_f, best_score


def _coordinate_descent(f0, stop_time, seed):
    """Cyclic coordinate descent polish."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = f0.copy()
    best_sc = _score_fast(f)
    no_imp_rounds = 0
    max_no_imp = 20

    mults = (0.1, 0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0)
    fine_mults = (0.9, 1.1, 0.8, 1.25)

    while time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = f[i]
            if orig < 1e-15:
                continue

            best_local = best_sc
            best_val = orig
            for mlt in mults:
                f[i] = orig * mlt
                sc = _score_fast(f)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_val = f[i]

            if best_local < best_sc - 1e-15:
                f[i] = best_val
                orig2 = best_val
                for mlt in fine_mults:
                    f[i] = orig2 * mlt
                    sc = _score_fast(f)
                    if sc < best_local - 1e-15:
                        best_local = sc
                        best_val = f[i]
                    else:
                        f[i] = orig2
                f[i] = best_val
                best_sc = best_local
                improved = True
            else:
                f[i] = orig

        if not improved:
            no_imp_rounds += 1
            if no_imp_rounds >= max_no_imp:
                break
        else:
            no_imp_rounds = 0

    return f.copy(), best_sc


def _make_inits(parent, n, seed):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    inits = [parent.copy()]
    i_arr = np.arange(n, dtype=np.float64)

    # Gaussian noise
    sig = 0.1 * max(1.0, float(parent.mean()))
    inits.append(np.clip(parent + rng.normal(0, sig, n), 0, None))

    # Multiplicative noise
    inits.append(np.clip(parent * (1.0 + 0.2 * rng.normal(0, 1, n)), 0, None))

    # Cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n)))

    # Multi-frequency
    sup = np.ones(n) * 0.3
    for fr in (2, 3, 5, 7, 13):
        ph = rng.uniform(0, 2 * math.pi)
        sup += 0.1 * np.cos(2.0 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.05)

    # Gaussian bump
    center = (n - 1) / 2.0
    for width in (n * 0.15, n * 0.25, n * 0.4):
        inits.append(np.exp(-0.5 * ((i_arr - center) / width) ** 2))

    # Two bumps
    c1, c2 = n * 0.25, n * 0.75
    w = n * 0.1
    inits.append(np.exp(-0.5 * ((i_arr - c1) / w) ** 2) +
                 np.exp(-0.5 * ((i_arr - c2) / w) ** 2))

    # Power law
    inits.append((1.0 + i_arr / n) ** 2)
    inits.append((1.0 + (n - 1 - i_arr) / n) ** 2)

    # Random sparse peaks
    m = max(10, n // 8)
    sp = np.zeros(n)
    for _ in range(m):
        pos = rng.integers(0, n)
        sp[pos] += rng.uniform(0.5, 2.0)
    inits.append(sp)

    # Filter valid inits
    valid_inits = []
    for ini in inits:
        ini = np.asarray(ini, dtype=np.float64).clip(0, None)
        if ini.sum() > 1e-6:
            valid_inits.append(ini / ini.sum())
    return valid_inits


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n = len(parent)

    best_f = parent / parent.sum()
    best_score = _score_fast(best_f)

    def _try_update(candidate):
        nonlocal best_f, best_score
        if candidate is None:
            return
        candidate = np.asarray(candidate, dtype=np.float64).clip(0, None)
        if candidate.size != n or candidate.sum() < 1e-10:
            return
        cs = candidate / candidate.sum()
        sc = _score_fast(cs)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_f = cs
            best_score = sc

    # Phase 1: Frank-Wolfe from multiple inits (50% time)
    phase1_end = t0 + budget * 0.50
    inits = _make_inits(parent, n, seed)
    time_per_init = max(5.0, (phase1_end - t0) / max(1, len(inits)))

    for idx, ini in enumerate(inits):
        if time.monotonic() > phase1_end - 5.0:
            break
        stop = min(phase1_end - 3.0, time.monotonic() + time_per_init)
        if time.monotonic() > stop - 1.0:
            continue
        try:
            res, sc = _frank_wolfe(ini, stop)
            _try_update(res)
        except Exception:
            pass

    # Extra FW from best (if time remains in phase 1)
    if time.monotonic() < phase1_end - 10.0:
        stop = phase1_end - 3.0
        try:
            res, sc = _frank_wolfe(best_f, stop)
            _try_update(res)
        except Exception:
            pass

    # Phase 2: More FW refinement on best (15% time)
    phase2_end = t0 + budget * 0.65
    if time.monotonic() < phase2_end - 10.0:
        stop = phase2_end - 3.0
        try:
            res, sc = _frank_wolfe(best_f, stop, max_iters=10000)
            _try_update(res)
        except Exception:
            pass

    # Phase 3: Mirror descent (10% time)
    phase3_end = t0 + budget * 0.75
    if time.monotonic() < phase3_end - 10.0:
        stop = phase3_end - 3.0
        try:
            res, sc = _mirror_descent(best_f, stop, seed + 12345)
            _try_update(res)
        except Exception:
            pass

    # Phase 4: Coordinate descent polish (15% time)
    phase4_end = t0 + budget * 0.90
    if time.monotonic() < phase4_end - 10.0:
        stop = phase4_end - 3.0
        try:
            res, sc = _coordinate_descent(best_f, stop, seed + 54321)
            _try_update(res)
        except Exception:
            pass

    # Phase 5: Try other lengths + final polish (10% time)
    phase5_end = deadline + 2.0
    remaining_for_lengths = phase5_end - time.monotonic() - 20.0
    if remaining_for_lengths > 20.0:
        x_old = np.linspace(0, 1, n)
        for L in (n - 32, n + 32, n - 64, n + 64):
            if time.monotonic() > phase5_end - 25.0:
                break
            if L < 80 or L > 1000:
                continue
            x_new = np.linspace(0, 1, L)
            init_L = np.interp(x_new, x_old, parent)
            stop_len = min(phase5_end - 20.0, time.monotonic() + max(10.0, remaining_for_lengths / 4))
            try:
                res_L, sc_L = _frank_wolfe(init_L / init_L.sum(), stop_len)
                if math.isfinite(sc_L) and sc_L < best_score - 1e-10:
                    # Found a better length!
                    best_f = res_L
                    best_score = sc_L
                    n = L
            except Exception:
                pass

    # Final coordinate descent on best
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        try:
            res, sc = _coordinate_descent(best_f, deadline - 5.0, seed + 99999)
            _try_update(res)
        except Exception:
            pass

    # Final FW if time allows
    if deadline - time.monotonic() > 15.0:
        try:
            res, sc = _frank_wolfe(best_f, deadline - 5.0, max_iters=2000)
            _try_update(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    # Scale to reasonable values (mean ~ 1)
    result = best_f * max(1.0, float(best_f.size))
    result = np.clip(result, 0.0, 1000.0)

    try:
        final_sc = float(evaluate_sequence([float(x) for x in result]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Fallback: parent
        parent_scaled = np.clip(parent, 0.0, 1000.0)
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent_scaled]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent_scaled]

    return [float(x) for x in result]
