    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(100)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_optimize(n, init_arr, stop, sd):
        nonlocal best_seq, best_score
        if time.monotonic() > stop - 4.0:
            return
        try:
            res, sc = _lp_optimize(init_arr, n, stop, sd)
        except Exception:
            return
        if res is None or len(res) == 0:
            return
        if not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0.0, 1000.0)
        if res.sum() < 0.01:
            return
        if sc < best_score - 1e-13:
            try:
                exact = float(evaluate_sequence([float(x) for x in res]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-14:
                best_seq = res.copy()
                best_score = exact

    # Phase 1: LP on parent length with diverse initializations
    n = n_p
    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    inits = [parent.copy()]

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n)))
    # Multi-frequency
    sup = np.ones(n) * 0.3
    for fr in (1, 2, 3, 5, 7, 11, 13):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.12 * np.cos(2.0 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.01)
    # |Normal|
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.02)
    # Parent + noise
    sig = 0.2 * max(1.0, float(np.mean(parent)))
    inits.append(np.abs(parent + rng_main.normal(0, sig, n)) + 0.01)
    # Gaussian bump
    center = (n - 1) / 2.0
    width = n / 5.0
    inits.append(np.exp(-0.5 * ((i_arr - center) / width) ** 2))
    # Two-lobe
    two = (np.exp(-0.5 * ((i_arr - n * 0.3) / (n * 0.12)) ** 2) +
           np.exp(-0.5 * ((i_arr - n * 0.7) / (n * 0.12)) ** 2))
    inits.append(two)
    # Flat
    inits.append(np.ones(n))
    # Boundary-weighted
    inits.append(1.0 + 0.5 * np.cos(math.pi * i_arr / max(1, n - 1)))
    # Sqrt shape
    inits.append(np.sqrt(i_arr + 1.0))

    n_inits = len(inits)
    phase1_end = t0 + budget * 0.50
    time_per_init = max(12.0, (phase1_end - t0 - 10.0) / n_inits)

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop = min(phase1_end - 8.0, time.monotonic() + time_per_init)
        _try_optimize(n, init, stop, seed + 7777 * (idx + 1))

    # Phase 2: Second LP pass on best (warm-start, longer time)
    phase2_end = t0 + budget * 0.68
    if time.monotonic() < phase2_end - 30.0:
        stop_p2 = min(phase2_end - 8.0, time.monotonic() + (phase2_end - time.monotonic() - 10.0))
        _try_optimize(len(best_seq), best_seq.copy(), stop_p2, seed + 999999)

    # Phase 3: Try a different length
    phase3_end = t0 + budget * 0.80
    other_lengths = [400, 480, 384]
    time_per_len = max(8.0, (phase3_end - phase2_end) / max(1, len(other_lengths)))
    for L in other_lengths:
        if time.monotonic() > phase3_end - 8.0:
            break
        stop = min(phase3_end - 6.0, time.monotonic() + time_per_len)
        init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
        _try_optimize(L, init_l, stop, seed + 50000 + L)

    # Phase 4: Polish the best candidate
    polish_end = deadline - 15.0
    if time.monotonic() < polish_end - 15.0:
        try:
            b_seq, b_sc = _coordinate_polish(best_seq, polish_end, seed=4242)
            if math.isfinite(b_sc) and b_sc < best_score - 1e-14:
                b_seq = np.clip(b_seq, 0, 1000)
                if b_seq.sum() >= 0.01:
                    exact = float(evaluate_sequence([float(x) for x in b_seq]))
                    if math.isfinite(exact) and exact < best_score - 1e-14:
                        best_seq = b_seq
                        best_score = exact
        except Exception:
            pass

    # Phase 5: Another polish with different seed
    if time.monotonic() < deadline - 18.0:
        try:
            b_seq2, b_sc2 = _coordinate_polish(best_seq, deadline - 15.0, seed=88888)
            if math.isfinite(b_sc2) and b_sc2 < best_score - 1e-14:
                b_seq2 = np.clip(b_seq2, 0, 1000)
                if b_seq2.sum() >= 0.01:
                    exact = float(evaluate_sequence([float(x) for x in b_seq2]))
                    if math.isfinite(exact) and exact < best_score - 1e-14:
                        best_seq = b_seq2
                        best_score = exact
        except Exception:
            pass

    # Phase 6: Final LP micro-pass if time allows
    if time.monotonic() < deadline - 25.0:
        try:
            b_seq3, b_sc3 = _lp_optimize(best_seq, len(best_seq), deadline - 20.0, seed + 1234567)
            if b_sc3 < best_score - 1e-14:
                b_seq3 = np.clip(b_seq3, 0, 1000)
                if b_seq3.sum() >= 0.01:
                    exact = float(evaluate_sequence([float(x) for x in b_seq3]))
                    if math.isfinite(exact) and exact < best_score - 1e-14:
                        best_seq = b_seq3
                        best_score = exact
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
