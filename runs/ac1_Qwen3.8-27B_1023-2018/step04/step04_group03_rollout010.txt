from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _lp_direction(f, n, b, peak_frac=0.05, timeout=10.0, n_extra=0):
    M = float(b.max())
    if M < 1e-18:
        return None
    threshold = M * (1.0 - peak_frac)
    peak_idx = np.where(b >= threshold)[0]
    if n_extra > 0:
        sort_order = np.argsort(b)[::-1]
        for idx in sort_order[len(peak_idx):len(peak_idx) + n_extra]:
            if idx not in peak_idx:
                peak_idx = np.append(peak_idx, idx)
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(b)])

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(peak_idx):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-22:
                rows.append(idx)
                cols.append(i)
                vals.append(v)

    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    c_vec = -np.ones(n)
    b_ub = np.full(len(peak_idx), M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None

    if res.status not in (0, 3) or res.x is None:
        return None

    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _optimize_lp(f0, stop_time, seed, peak_frac=0.05):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    best_f = f.copy()
    best_score = _score(f)

    t = 0.3
    stall = 0
    max_stall = 60

    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, n, b, peak_frac=peak_frac, timeout=5.0)
        if g is None:
            stall += 5
            if stall > max_stall:
                noise = rng.normal(0, 0.1 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                fs = f.sum()
                if fs < 1e-10:
                    f = best_f.copy()
                else:
                    f = f / fs * best_f.sum()
                t = 0.3
                stall = 0
                max_stall = min(max_stall + 30, 250)
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            stall += 3
            continue
        f_sum = f.sum()
        g = g * (f_sum / g_sum)

        dist = np.linalg.norm(g - f) / max(1.0, np.linalg.norm(f))
        if dist < 1e-9:
            stall += 2
            if stall > max_stall:
                break
            continue

        improved = False
        t_trials = sorted(set([
            t, t * 0.5, t * 1.5, t * 0.25, t * 2.0,

            sc = _score(f_new)

                best_f = f.copy()

                improved = True
                stall = 0
                break

def _optimize_subgrad(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-12, None)

    lr = 0.15
    stall = 0
    max_stall = 60
    avg_f = f.copy()

    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        peak_threshold = M * 0.99
        peak_idx = np.where(b >= peak_threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])

        grad = np.zeros(n)
        for k in peak_idx:
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                grad[i] += f[k - i]
        grad /= len(peak_idx)
        grad *= 2.0

        grad_scale = grad.max()
        if grad_scale < 1e-15:
            break
        lr_eff = lr / grad_scale

        exponent = -lr_eff * grad
        exponent -= exponent.max()
        weights = np.exp(exponent)
        f_new = (f + 1e-15) * weights
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        avg_f = 0.5 * avg_f + 0.5 * f_new
        avg_f = avg_f / avg_f.sum()

        sc = _score(f_new)
        sc_avg = _score(avg_f)
        best_eval = min(sc, sc_avg)
        if sc_avg < sc:
            f_new = avg_f
            sc = sc_avg

        if sc < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = sc
            stall = 0
            lr *= 0.98

            lr *= 0.94

                noise = rng.normal(0, 0.02, n)

                f = f / f.sum()
                stall = 0

        if not improved:

def _pair_polish(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    best_sc = _score(f)
    no_imp = 0

    while time.monotonic() < stop_time - 3.0 and no_imp < 5:
        no_imp += 1
        rounds_this_pass = 0
        for _ in range(50):
            if time.monotonic() > stop_time - 3.0:
                break
            i, j = rng.choice(n, size=2, replace=False)
            if f[i] < 1e-12 and f[j] < 1e-12:
                continue
            orig_i, orig_j = f[i], f[j]
            total = orig_i + orig_j
            if total < 1e-12:
                continue
            ratios = [0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9]
            best_r = 0.5
            best_s = best_sc
            orig_ratio = orig_i / total if total > 1e-15 else 0.5
            all_ratios = sorted(set([best_r, best_r * 0.5, best_r * 1.5] +
                                    [r for r in ratios if abs(r - orig_ratio) < 0.4]))
            for r in all_ratios:
                r = max(0.001, min(0.999, r))
                f[i] = total * r
                f[j] = total * (1 - r)
                sc = _score(f)
                if sc < best_s - 1e-15:

                    best_r = r
            f[i] = total * best_r
            f[j] = total * (1 - best_r)
            if best_s < best_sc - 1e-14:
                best_sc = best_s
                no_imp = 0
                rounds_this_pass += 1
            else:
                f[i] = orig_i
                f[j] = orig_j
