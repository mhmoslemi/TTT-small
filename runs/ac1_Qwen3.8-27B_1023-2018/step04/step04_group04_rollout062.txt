        fb = np.ones(320) / 320.0
        fbsc = _sc_a(fb)

    best_seq, best_sc = fb.copy(), fbsc

    def _upd(a, sc):
        nonlocal best_seq, best_sc
        if sc < best_sc - 1e-11 and np.all(np.isfinite(a)) and a.sum() >= 0.01:
            best_seq = np.clip(a, 0.0, 1000.0).copy()
            best_sc = sc

    pn = parent / psum

    # P1: main LP
    e1 = t0 + budget * 0.40
    r, s = _lp_opt(pn, n, e1, seed)
    if math.isfinite(s):
        _upd(r * psum, s)

    # P2: perturbed LP
    e2 = t0 + budget * 0.52
    if time.monotonic() < e2:
        rp = np.random.default_rng(seed + 1)
        p2 = np.clip(pn + rp.normal(0, 0.12, n), 0, None)
        p2 /= max(p2.sum(), 1e-10)
        r2, s2 = _lp_opt(p2, n, e2, seed + 100)
        if math.isfinite(s2):
            _upd(r2 * psum, s2)

    # P3: LP from best
    e3 = t0 + budget * 0.62
    if time.monotonic() < e3:
        r3, s3 = _lp_opt(best_seq / max(best_seq.sum(), 1e-10), n, e3, seed + 200)
        if math.isfinite(s3):
            _upd(r3 * psum, s3)

    # P4: coordinate descent
    e4 = t0 + budget * 0.78
    if time.monotonic() < e4:
        r4, s4 = _cd(best_seq / max(best_seq.sum(), 1e-10), e4, seed + 300, 400)
        if math.isfinite(s4):
            _upd(r4 * psum, s4)

    # P5: nearby lengths
    e5 = dl - 18
    if time.monotonic() < e5:
        alts = sorted(set(L for L in (n - 4, n - 2, n - 1, n + 1, n + 2, n + 4, n + 8) if L >= 4))
        tp = max(8.0, min(35.0, (e5 - time.monotonic()) / max(1, len(alts))))
        for L in alts:
            if time.monotonic() > e5 - 5:
                break
            init_L = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n), parent)
            init_L /= max(init_L.sum(), 1e-10)
            try:
                rL, sL = _lp_opt(init_L, L, min(e5, time.monotonic() + tp), seed + 1000 + L)
                if math.isfinite(sL):
                    _upd(rL * psum, sL)
            except Exception:
                pass

    # P6: final polish
    if time.monotonic() < dl - 5:
        r6, s6 = _cd(best_seq / max(best_seq.sum(), 1e-10), dl - 5, seed + 5000, 200)
        if math.isfinite(s6):
            _upd(r6 * psum, s6)

    # validate
    try:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        fs = math.inf
    if not math.isfinite(fs):
        try:
            float(evaluate_sequence([float(x) for x in fb]))
            return [float(x) for x in fb]
        except Exception:
            return [1.0]
    return [float(x) for x in best_seq]
