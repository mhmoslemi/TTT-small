            f_new /= f_new.sum()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                scale = rng.choice([0.5, 0.8, 1.2, 1.5])
                f = np.clip(f * scale + rng.normal(0, 0.03, n), 0, None)
                fs = f.sum()
                f /= fs if fs > 1e-10 else 1
                stall = 0
                if time.monotonic() > stop_time - 15:
                    break
        iteration += 1

    return best_f, best_score


def _coord_descent(f0, stop_time, n_sweeps=200, seed=777):
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, math.inf
    f = f / S
    S = 1.0
    conv = _conv(f).copy()
    best_score = 2.0 * n * float(conv.max())
    rng = np.random.default_rng(seed)
    no_improve_sweeps = 0

    for sweep in range(n_sweeps):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break
            v = f[i]
            S_rest = S - v
            start = i
            end = min(i + n, 2 * n - 1)
            infl = np.zeros(2 * n - 1)
            infl[start:end] = 2.0 * f[: end - start]

            # Search range
            x_max = max(5.0 * v + 0.1, 0.3)
            x_max = min(x_max, 5.0)

            # Coarse grid
            n_pts = 80
            x_grid = np.linspace(0.0, x_max, n_pts)
            best_x = v
            best_obj = best_score

            for x in x_grid:
                S_new = S_rest + x
                if S_new < 0.01:
                    continue
                w = x - v
                if abs(w) < 1e-14:
                    m_new = float(conv.max())
                else:
                    tmp = conv.copy()
                    tmp[start:end] += w * infl[start:end]
                    if 0 <= 2 * i <= 2 * n - 2:
                        tmp[2 * i] += w * w
                    m_new = float(tmp.max())
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_x = x

            # Fine grid around best
            if best_x != v:
                dw = x_max / n_pts
                x_lo = max(0.0, best_x - 2 * dw)
                x_hi = min(x_max, best_x + 2 * dw)
                x_fine = np.linspace(x_lo, x_hi, 40)
                for x in x_fine:
                    S_new = S_rest + x
                    if S_new < 0.01:
                        continue
                    w = x - v
                    if abs(w) < 1e-14:
                        continue
                    tmp = conv.copy()
                    tmp[start:end] += w * infl[start:end]
                    if 0 <= 2 * i <= 2 * n - 2:
                        tmp[2 * i] += w * w
                    m_new = float(tmp.max())
                    obj = 2.0 * n * m_new / (S_new * S_new)
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_x = x

            if best_x != v and best_obj < best_score - 1e-14:
                w = best_x - v
                f[i] = best_x
                S += w
                conv[start:end] += w * infl[start:end]
                if 0 <= 2 * i <= 2 * n - 2:
                    conv[2 * i] += w * w
                best_score = best_obj
                improved = True

        if improved:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 8:
                # Perturb to escape
                idx = rng.choice(n, size=max(1, n // 20), replace=False)
                f[idx] *= rng.uniform(0.5, 1.5, size=len(idx))
                f = np.clip(f, 0, None)
                S = f.sum()
                if S > 1e-10:
                    f /= S
                    S = 1.0
                    conv = _conv(f)
                    best_score = 2.0 * n * float(conv.max())
                no_improve_sweeps = 0

    f = np.clip(f, 0, 1000)
    fs = f.sum()
    if fs > 1e-10:
        f /= fs
    final_score = _score(f)
    return f, final_score


def _grad_optimize(f0, n, stop_time, seed, max_iters=5000):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = f.sum()
    f = f / fs if fs > 1e-10 else np.ones(n) / n
    best_f = f.copy()
    best_score = _score(f)

    alpha = 0.03
    alpha_min = 1e-10
    tau = 50.0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 5.0:
            break
        b = _conv(f)
        S = float(f.sum())
        M = float(b.max())
        if M < 1e-18:
            break
        score = 2.0 * n * M / (S * S)
        if score < best_score - 1e-13:
            best_score = score
            best_f = f.copy()

        b_shifted = b - M
        tau_b = tau * b_shifted
        tau_b -= tau_b.max()
        p = np.exp(tau_b)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(2 * n - 1)
            p[int(np.argmax(b))] = 1.0
            p_sum = 1.0
        p /= p_sum

        f_rev = f[::-1]
        csz = 2 * n - 1
        P = rfft(p, csz)
        Fr = rfft(f_rev, csz)
        conv_val = irfft(P * Fr, csz)[:csz]
        grad = 2.0 * conv_val[n - 1 : 2 * n - 1]

        score_grad = (2.0 * n / (S * S)) * (grad - 2.0 * M / S)
        g_raw = f - alpha * score_grad
        f_new = _project_simplex(g_raw)

        sc_new = _score(f_new)
        if sc_new < score - 1e-16:
            f = f_new
            alpha = max(alpha_min, alpha * 1.02)
        else:
            alpha = max(alpha_min, alpha * 0.5)
            if alpha < alpha_min:
                f = np.clip(f + rng.normal(0, 0.008, n), 0, None)
                f /= f.sum()
                alpha = 0.01
                tau = max(10.0, tau * 0.5)

        if it % 500 == 499:
            f = np.clip(f + rng.normal(0, 0.01, n), 0, None)
            f /= f.sum()
            alpha = 0.02
            tau = 60.0

    return best_f, best_score


def _pair_search(f0, stop_time, n_trials=200, seed=99999):
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f /= f.sum()
    best_score = _score(f)
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for _ in range(n_trials):
        if time.monotonic() > stop_time - 5.0:
            break
        i, j = int(rng.integers(n)), int(rng.integers(n))
        if i == j:
            continue
        oi, oj = f[i], f[i]
        if oi < 1e-12 and oj < 1e-12:
            continue

        factors = (0.3, 0.5, 0.7, 1.0, 1.4, 2.0, 0.0)
        bp_sc = best_score
        bp = (oi, oj)
        for mi in factors:
            ni = oi * mi if oi > 1e-12 else (0.01 if mi > 0 else 0.0)
            for mj in factors:
                nj = oj * mj if oj > 1e-12 else (0.01 if mj > 0 else 0.0)
                f[i], f[j] = ni, nj
                if f.sum() < 0.01:
                    continue
                sc = _score(f)
                if sc < bp_sc - 1e-14:
                    bp_sc = sc
                    bp = (ni, nj)

        f[i], f[j] = bp
        if bp_sc < best_score - 1e-14:
            best_score = bp_sc
            best_f = f.copy()

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    hard_deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)
    n_p = max(32, min(n_p, 1024))

    # Normalized parent
    p_norm = parent.copy()
    p_norm = np.clip(p_norm, 0, None)
    ps = p_norm.sum()
    p_norm = p_norm / ps if ps > 1e-10 else np.ones(n_p) / n_p

    # Fallback
    fallback = p_norm.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(n_p) / n_p
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_update(seq, sc=None):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64).ravel()
        seq = np.clip(seq, 0, 1000)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        if sc is None:
            sc = _score(seq)
        if not math.isfinite(sc):
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-14:
            best_seq = seq.copy()
            best_score = exact

    def _pipeline(init, n, stop, lbl=""):
        nonlocal best_seq, best_score
        if time.monotonic() > stop - 10:
            return
        init = np.clip(np.asarray(init, dtype=np.float64).ravel().copy(), 0, None)
        while init.size < n:
            init = np.concatenate([init, np.zeros(n - init.size)])
        init = init[:n]
        ist = init.sum()
        init = init / ist if ist > 1e-10 else np.ones(n) / n

        # LP phase
        t_lp = min(stop - 20, t0 + (stop - t0) * 0.5)
        if time.monotonic() < t_lp - 5:
            try:
                res, sc = _lp_optimize(init, n, t_lp, seed + hash(lbl) % 10000)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res /= res.sum()
                    _try_update(res, sc)
            except Exception:
                pass

        # Gradient phase
        t_gr = min(stop - 15, t0 + (stop - t0) * 0.7)
        if time.monotonic() < t_gr - 5 and best_score > 1.503:
            try:
                res, sc = _grad_optimize(best_seq, len(best_seq), t_gr, seed + 1)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    _try_update(res, sc)
            except Exception:
                pass

        # Coord descent
        t_cd = min(stop - 5, stop)
        if time.monotonic() < t_cd - 5:
            try:
                res, sc = _coord_descent(best_seq, t_cd, n_sweeps=100, seed=seed + 2)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    _try_update(res, sc)
            except Exception:
                pass

    # Phase 1: Main LP optimization on parent
    p1_end = t0 + budget * 0.40
    _pipeline(p_norm, n_p, p1_end, "phase1")

    # Phase 2: Alternative initializations
    p2_end = min(t0 + budget * 0.55, hard_deadline - 60)
    if time.monotonic() < p2_end - 10 and best_score > 1.5028:
        i_arr = np.arange(n_p, dtype=np.float64)
        alts = [
            np.clip(0.3 + 0.7 * np.cos(np.pi * i_arr / (n_p - 1)) ** 2, 0, None),
            np.clip(0.5 + 0.5 * np.cos(2 * np.pi * i_arr / n_p), 0, None),
            np.clip(1.0 - 0.6 * (i_arr / (n_p - 1) - 0.5) ** 2, 0, None),
            np.clip(0.2 + 0.8 / (1 + 0.1 * i_arr), 0, None),
        ]
        alts = [a / a.sum() for a in alts if a.sum() > 0.01]
        t_alloc_each = max(20.0, (p2_end - time.monotonic()) / max(1, len(alts)))
        for idx, alt in enumerate(alts[:3]):
            if time.monotonic() > p2_end - 10:
                break
            if best_score < 1.5028:
                break
            t_stop = min(p2_end - 5, time.monotonic() + t_alloc_each)
            _pipeline(alt, n_p, t_stop, f"alt{idx}")

    # Phase 3: Intensive coordinate descent on best
    p3_end = min(t0 + budget * 0.72, hard_deadline - 50)
    if time.monotonic() < p3_end - 10:
        try:
            res, sc = _coord_descent(best_seq, p3_end, n_sweeps=150, seed=424242)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _try_update(res, sc)
        except Exception:
            pass

    # Phase 4: Pair search for local refinements
    p4_end = min(t0 + budget * 0.82, hard_deadline - 40)
    if time.monotonic() < p4_end - 10:
        try:
            res, sc = _pair_search(best_seq, p4_end, n_trials=300, seed=55555)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _try_update(res, sc)
        except Exception:
            pass

    # Phase 5: Try nearby lengths
    p5_end = min(t0 + budget * 0.88, hard_deadline - 50)
    if time.monotonic() < p5_end - 20 and best_score > 1.5028:
        other_lens = []
        for L in [n_p - 32, n_p - 16, n_p + 16, n_p + 32, n_p * 2, n_p // 2]:
            if 32 <= L <= 1024 and L != n_p:
                other_lens.append(L)
        other_lens.sort(key=lambda x: abs(x - n_p))
        tper = max(15.0, (p5_end - time.monotonic()) / max(1, len(other_lens[:3])))
        for L in other_lens[:3]:
            if time.monotonic() > p5_end - 10:
                break
            if best_score < 1.5028:
                break
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
            init_l = np.clip(init_l, 0, None)
            ls = init_l.sum()
            init_l = init_l / ls if ls > 0.01 else np.ones(L) / L
            _pipeline(init_l, L, min(p5_end - 5, time.monotonic() + tper), f"len{L}")

    # Phase 6: Final intense polish
    fin_end = hard_deadline - 10.0
    if time.monotonic() < fin_end - 8:
        try:
            res, sc = _coord_descent(best_seq, fin_end - 8, n_sweeps=60, seed=31337)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _try_update(res, sc)
        except Exception:
            pass

    if time.monotonic() < fin_end - 5:
        try:
            res, sc = _pair_search(best_seq, fin_end - 3, n_trials=100, seed=77777)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _try_update(res, sc)
        except Exception:
            pass

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
