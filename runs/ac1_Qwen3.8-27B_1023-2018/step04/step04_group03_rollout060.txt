    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for k in peak_indices"""
    n = len(f)

    for idx, k in enumerate(peak_indices):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-15:
                rows.append(idx)
                cols.append(i)
                vals.append(v)
    if not rows:
        return None

    c_vec = -np.ones(n)
    b_ub = np.full(len(peak_indices), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None

        f_try = np.clip(f_try, 0, None)
        if f_try.sum() < 0.01:

    # Second pass: fine around best_t

    t_hi = min(0.99, best_t + 0.1)

                improved = True
            else:
                f[i] = orig  # restore (in case we modified it)
                f[i] = best_val
        if not improved:
            break
    return f, best_sc

But copying b every time is O(n). With 8 trials per coordinate and 416 coords: 8 * 416 * 416 = 1.4M for copies + computation. Still fast.

Actually, I can avoid the copy by just tracking the max analytically. The max of b_try is:
max( max of b_base[k] for k not in k_range,
      max over k in k_range of (b_base[k] + delta_k(x)) )

where delta_k(x) = x^2 (if k=2i) or 2*x*f[k-i] (otherwise).

The "max over k not in k_range" is a constant (call it b_other). The "max over k in k_range" is max of:
- b_base[2i] + x^2 (one quadratic)
- b_base[k] + 2*x*f[k-i] for k in k_range, k≠2i (n-1 lines)
- b_other (constant)

So max_b(x) = max(b_other, b_base[2i] + x^2, max_k (b_base[k] + 2*x*f[k-i]))

The "max_k (b_base[k] + 2*x*f[k-i])" is the max of n-1 lines in x. Since all slopes (2*f[k-i]) are non-negative, this is a non-decreasing function of x (piecewise linear, convex).

I can precompute this: for each trial x, just evaluate the max. Or I can find the upper envelope of the lines.

For simplicity, let me just evaluate at the trial x values. With 8 trials and O(n) max computation each, it's 8*416 = 3328 operations per coordinate, total 416*3328 = 1.4M per pass. Fast.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(416, dtype=np.float64)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a, b=None):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    if b is None:
        b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _solve_lp(f, peak_indices, M, timeout=4.0):
    n = len(f)
    rows, cols, vals = [], [], []
    for idx, k in enumerate(peak_indices):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-14:
                rows.append(idx)
                cols.append(i)
                vals.append(v)
    if not rows:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_indices), n))
    c_vec = -np.ones(n)
    b_ub = np.full(len(peak_indices), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-14:
        return None
    return g


def _line_search(f, g, best_score):
    f_sum = float(f.sum())
    g_sum = float(g.sum())
    if g_sum < 1e-14:
        return f, best_score
    g_norm = g * (f_sum / g_sum)
    best_t = 0.0
    best_sc = best_score
    t_coarse = np.array([0.05, 0.08, 0.1, 0.12, 0.15, 0.2, 0.25, 0.3,
                         0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 0.95])
    for t in t_coarse:
        f_try = (1.0 - t) * f + t * g_norm
        f_try = np.clip(f_try, 0.0, None)
        if f_try.sum() < 0.01:
            continue
        sc = _score_fast(f_try)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        return f, best_score
    t_lo = max(0.01, best_t - 0.12)
    t_hi = min(0.99, best_t + 0.12)
    t_fine = np.linspace(t_lo, t_hi, 25)
    for t in t_fine:
        f_try = (1.0 - t) * f + t * g_norm
        f_try = np.clip(f_try, 0.0, None)
        if f_try.sum() < 0.01:
            continue
        sc = _score_fast(f_try)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    f_new = (1.0 - best_t) * f + best_t * g_norm
    f_new = np.clip(f_new, 0.0, None)
    return f_new, best_sc


def _coord_descent(f, b, stop_time, max_passes=5):
    n = len(f)
    best_sc = _score_fast(f, b)
    f = f.copy()
    b = b.copy()
    S = float(f.sum())
    for _pass in range(max_passes):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = np.random.default_rng(_pass + 1).permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            x_old = f[i]
            S_base = S - x_old
            if S_base < 0.01 and x_old < 0.01:
                continue
            k_lo = max(0, i)
            k_hi = min(2 * n - 1, i + n - 1)
            k_range = range(k_lo, k_hi + 1)
            b_base = b.copy()
            for k in k_range:
                ki = k - i
                if ki == i:
                    b_base[k] -= x_old * x_old
                else:
                    b_base[k] -= 2.0 * x_old * f[ki]
            b_other = 0.0
            for k in range(2 * n - 1):
                if k < k_lo or k > k_hi:
                    if b_base[k] > b_other:
                        b_other = b_base[k]
            if k_lo <= 2 * i <= k_hi:
                b_q_base = b_base[2 * i]
            else:
                b_q_base = -1.0
            trial_xs = [0.0]
            if x_old > 1e-12:
                trial_xs += [0.25 * x_old, 0.5 * x_old, 0.75 * x_old,
                             x_old, 1.25 * x_old, 1.5 * x_old,
                             2.0 * x_old, 3.0 * x_old, 5.0 * x_old]
            best_x = x_old
            best_local_sc = math.inf
            for x in trial_xs:
                if x < 0:
                    continue
                S_new = S_base + x
                if S_new < 0.01:
                    continue
                max_b = b_other
                for k in k_range:
                    ki = k - i
                    if ki == i:
                        val = b_base[k] + x * x
                    else:
                        val = b_base[k] + 2.0 * x * f[ki]
                    if val > max_b:
                        max_b = val
                if max_b < 0:
                    max_b = 0.0
                sc = 2.0 * n * max_b / (S_new * S_new)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_x = x
            if best_x != x_old and best_local_sc < best_sc - 1e-15:
                for k in k_range:
                    ki = k - i
                    if ki == i:
                        b[k] += best_x * best_x - x_old * x_old
                    else:
                        b[k] += 2.0 * (best_x - x_old) * f[ki]
                f[i] = best_x
                S += best_x - x_old
                best_sc = best_local_sc
                improved = True
            elif best_x != x_old:
                for k in k_range:
                    ki = k - i
                    if ki == i:
                        b[k] += best_x * best_x - x_old * x_old
                    else:
                        b[k] += 2.0 * (best_x - x_old) * f[ki]
                f[i] = best_x
                S += best_x - x_old
                bsc = _score_fast(f, b)
                if bsc < best_sc:
                    best_sc = bsc
                    improved = True
        if not improved:
            break
    return f, best_sc


def _lp_optimize(f0, stop_time, seed, peak_frac_start=0.03, peak_frac_end=0.01):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0.0, None)
    if f.sum() < 0.01:
        f = _get_parent().copy()
    b = _conv_full(f)
    best_f = f.copy()
    best_score = _score_fast(f, b)
    iter_count = 0
    stall = 0
    max_stall = 40
    t_prev = 0.3
    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break
        frac = peak_frac_start * (peak_frac_end / peak_frac_start) ** (min(iter_count, 100.0) / 100.0)
        frac = max(frac, 0.005)
        thresh = M * (1.0 - frac)
        peak_idx = np.where(b >= thresh)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])
        if len(peak_idx) > 150:
            peak_idx = peak_idx[np.argsort(-b[peak_idx])[:150]]
        g = _solve_lp(f, peak_idx.tolist(), M, timeout=3.0)
        if g is None:
            stall += 5
            if stall > max_stall:
                noise = rng.normal(0, 0.05 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0.0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * best_f.sum()
                b = _conv_full(f)
                stall = 0
                max_stall = min(max_stall + 50, 200)
            continue
        f, sc = _line_search(f, g, best_score)
        if sc < best_score - 1e-14:
            best_f = f.copy()
            best_score = sc
            stall = 0
            iter_count += 1
        else:
            stall += 1
            if stall >= max_stall:
                noise = rng.normal(0, 0.03 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0.0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * best_f.sum()
                stall = 0
                max_stall = min(max_stall + 50, 300)
                if max_stall > 500:
                    break
        if iter_count % 15 == 0 and time.monotonic() < stop_time - 10.0:
            f, sc_cd = _coord_descent(f, _conv_full(f), time.monotonic() + 8.0, max_passes=2)
            if sc_cd < best_score - 1e-14:
                best_f = f.copy()
                best_score = sc_cd
                f = f.copy()
            else:

    return best_f, best_score


def _mirror_descent(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-10, None)
    f = f / f.sum()
    best_f = f.copy()
    best_score = _score_fast(f)
    lr = 0.15
    stall = 0
    max_stall = 60
    for it in range(2000):
        if time.monotonic() > stop_time - 3.0:
            break
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break
        peak_thresh = M * 0.97
        peak_idx = np.where(b >= peak_thresh)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])
        if len(peak_idx) > 50:
            peak_idx = peak_idx[np.argsort(-b[peak_idx])[:50]]
        grad = np.zeros(n)
        for k in peak_idx:
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                grad[i] += f[k - i]
        grad /= len(peak_idx)
        gs = float(grad.max())
        if gs < 1e-15:
            break
        exponent = -lr * grad / gs
        exponent -= exponent.max()
        weights = np.exp(exponent)
        f_new = (f + 1e-12) * weights
        s = f_new.sum()
        if s < 1e-12:
            break
        f_new = f_new / s
        sc = _score_fast(f_new)
        if sc < best_score - 1e-13:
            best_f = f_new.copy()
            best_score = sc
            stall = 0
            lr *= 0.995
        else:
            f_damp = 0.5 * f + 0.5 * f_new
            sc_d = _score_fast(f_damp)
            if sc_d < best_score - 1e-13:
                f_new = f_damp
                best_f = f_new.copy()
                best_score = sc_d
                stall = 0
                lr *= 0.99
            else:
                stall += 1
                lr *= 0.96
                if stall > max_stall:
                    noise = rng.normal(0, 0.01, n)
                    f = np.clip(f + noise, 1e-10, None)
                    f = f / f.sum()
                    stall = 0
                    lr = 0.08
                    max_stall = min(max_stall + 40, 300)
                    if max_stall > 400:
                        break
        f = f_new
    return best_f, best_score

            pass

    # Phase 6: Final coordinate descent polish (3% time)
    ph6_end = hard_deadline
    if time.monotonic() < ph6_end - 3.0:
        stop = ph6_end - 1.0
        try:
            b_cur = _conv_full(best_seq)
            res, sc = _coord_descent(best_seq, b_cur, stop, max_passes=20)
            _try_update(res)
        except Exception:
            pass

    # Try alternate lengths if time remains
    remaining = hard_deadline - time.monotonic()
    if remaining > 30.0:
        for L in [n_p - 16, n_p + 16, n_p - 32, n_p + 32]:
            if time.monotonic() > hard_deadline - 15.0:
                break
            if L < 80 or L > 700:
                continue
            x_old = np.linspace(0, 1, n_p)
            x_new = np.linspace(0, 1, L)
            init_L = np.interp(x_new, x_old, parent)
            stop = min(hard_deadline - 12.0, time.monotonic() + 20.0)
            try:
                res, sc = _lp_optimize(init_L, stop, seed + L * 137,
                                       peak_frac_start=0.03, peak_frac_end=0.01)
                if res is not None:
                    _try_update(res)
            except Exception:
                pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf
    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]
    return [float(x) for x in best_seq]
