    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    u = np.log(np.maximum(p, 1e-300))
    u -= u.mean()

    def func_grad(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if T <= 1e-30:
            T = 1e-4 * max(cmax, 1e-10)
        scaled = (c - cmax) / T
        scaled = np.minimum(scaled, 50.0)
        exps = np.exp(scaled, dtype=np.float64)
        s_exp = float(exps.sum())
        if s_exp > 0 and math.isfinite(s_exp):
            f = cmax + T * math.log(s_exp / float(c.size))
            w = exps / s_exp
        else:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    c0 = cfast(p)
    cmax_ref = float(c0.max()) if c0.size > 0 else 1.0

    stages = [
        (0.12, 0.10),
        (0.04, 0.15),
        (0.012, 0.18),
        (0.004, 0.20),
        (0.0012, 0.18),
        (0.0004, 0.14),
        (0.0001, 0.10),
        (0.00003, 0.08),
    ]

    start_time = time.monotonic()
    avail = max(0.0, stop_time - start_time - 10.0)
    if avail < 15.0:
        return p

    total_w = float(sum(wt for _, wt in stages))
    best_p = p.copy()
    best_sc = _score_fft(p)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        cum += wt
        end_stage = min(start_time + avail * (cum / total_w), stop_time - 5.0)
        if end_stage < now + 8.0:
            break
        T = max(1e-30, frac * cmax_ref)
        best_u = u.copy()
        best_f = float('inf')
        stale = 0
        while time.monotonic() < end_stage - 2.0:
            maxit = 25
            opts = {
                "maxiter": maxit,
                "gtol": 1e-11,
                "ftol": 1e-15,
                "maxls": 20,
                "maxcor": 20,
                "disp": False,
            }
            try:
                res = minimize(func_grad, best_u, args=(T,),
                               method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fv = float(res.fun) if res.fun is not None else float('inf')
                if math.isfinite(fv) and fv < best_f:
                    best_f = fv
                    best_u = np.asarray(res.x, dtype=np.float64).copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if getattr(res, "status", 1) == 0:
                break

        u = best_u
        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur):
            if sc_cur < best_sc:
                best_p = p_cur.copy()
                best_sc = sc_cur
            if consider:
                consider(p_cur)
        p = p_cur.copy()
        cmax_ref = float(cfast(p).max()) if cmax_ref > 0 else 1.0

    return best_p


def _build_conv_matrix(a):
    """Build the (2n-1) x n matrix A where (A@g)[m] = conv(a,g)[m]."""
    n = a.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _lp_best_response(a, stop_time, max_iter=80, consider=None):
    """LP-based best response iterations."""
    n = a.size
    if n == 0 or n > 5000:
        return a

    a = np.maximum(a, 0.0).astype(np.float64)
    s = float(a.sum())
    if s < 1e-300:
        return a
    a = a / s

    best = a.copy()
    best_sc = float(_score_fft(best))
    rng = np.random.default_rng(99999)

    # Precompute A matrix
    A = _build_conv_matrix(a)

    for it in range(max_iter):
        if time.monotonic() > stop_time - 3.0:
            break

        c = _conv_self(a)
        M = float(c.max())
        if M <= 0 or not math.isfinite(M):
            break

        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                          method='highs', options={'presolve': True, 'time_limit': 15})
        except Exception:
            break

        if res.status != 0 or res.x is None:
            break

        g = np.maximum(res.x, 0.0)
        sg = float(g.sum())
        if sg < 1e-10:
            break
        g = g / sg

        # Check if g is close to a (convergence)
        diff = float(np.linalg.norm(g - a))
        if diff < 1e-5:
            break

        # Try several step sizes
        best_candidate = a.copy()
        best_candidate_sc = best_sc
        for t in [0.8, 0.5, 0.3, 0.15, 0.07, 0.03]:
            trial = (1.0 - t) * a + t * g
            trial = np.maximum(trial, 0.0)
            st = float(trial.sum())
            if st < 1e-10:
                continue
            trial = trial / st
            sc = float(_score_fft(trial))
            if math.isfinite(sc) and sc < best_candidate_sc:
                best_candidate_sc = sc
                best_candidate = trial.copy()

        if math.isfinite(best_candidate_sc) and best_candidate_sc < best_sc - 1e-14:
            a = best_candidate.copy()
            best = a.copy()
            best_sc = best_candidate_sc
            # Update A matrix
            A = _build_conv_matrix(a)
            if consider:
                consider(a)
        else:
            # No improvement from LP, try a perturbation
            if time.monotonic() < stop_time - 10.0:
                p = a * np.exp(rng.normal(0.0, 0.02, n))
                p = np.maximum(p, 0.0)
                sp = float(p.sum())
                if sp > 0:
                    p /= sp
                    sc_p = float(_score_fft(p))
                    if math.isfinite(sc_p) and sc_p < best_sc:
                        a = p.copy()
                        best = a.copy()
                        best_sc = sc_p
                        A = _build_conv_matrix(a)
                        if consider:
                            consider(a)
                    # Also try random restart with larger perturbation every 10 iters
                    if it % 12 == 11:
                        p2 = a * np.exp(rng.normal(0.0, 0.08, n))
                        p2 = np.maximum(p2, 0.0)
                        sp2 = float(p2.sum())
                        if sp2 > 0:
                            p2 /= sp2
                            sc2 = float(_score_fft(p2))
                            if math.isfinite(sc2) and sc2 < best_sc:
                                a = p2.copy()
                                best = a.copy()
                                best_sc = sc2
                                A = _build_conv_matrix(a)
                                if consider:
                                    consider(a)
            else:
                break

    return best


def _coord_polish(p, stop_time, consider=None, seed=12345):
    """Coordinate descent with incremental convolution updates."""
    n = int(p.size)
    if n == 0:
        return p, math.inf
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).astype(np.float64)
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    G = 24 if n <= 2000 else 18

    sweep = 0
    flat = 0
    max_sweeps = 200000

    while sweep < max_sweeps and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break

            S = 1.0  # p is normalized to sum 1
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]

            # Max outside affected region
            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max()) if start > 0 else 0.0
                max_const = max(max_const, ml)
            if end < Ltot:
                mr = float(c[end:].max()) if end < Ltot else 0.0
                max_const = max(max_const, mr)

            x_hi = max(20.0 * v, 60.0 / n, 5e-5, 1e-12)
            x_hi = min(x_hi, 5.0 / n * 10)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, 1.0 / n)

            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 <= v <= x_hi:
                pts = np.concatenate((pts, [v]))
            pts = np.unique(pts)

            diag = i  # index in base where the w^2 term applies

            def eval_xs(xs):
                w = xs - v
                aff = base[None, :] + w[:, None] * infl[None, :]
                dval = base[diag] + w * infl[diag] + w * w
                aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                vals = (2.0 * n * mv) / (1.0)  # S=1, S+w normalized later
                # Actually need to account for the normalization change
                # score = 2n * max_conv / (S+w)^2
                denom = 1.0 + w
                valid = denom > 1e-15
                vals = np.full(xs.shape, math.inf)
                vals[valid] = (2.0 * n * mv[valid]) / (denom[valid] ** 2)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Fine search
            dw = x_hi / G
            lo = max(0.0, best_x - 2.0 * dw)
            hi = min(x_hi, best_x + 2.0 * dw)
            if hi <= lo:
                hi = lo + 1e-15
            fine = np.linspace(lo, hi, 24)
            if 0.0 <= best_x <= x_hi:
                fine = np.concatenate((fine, [best_x]))
            fine = np.unique(fine)
            vals2 = eval_xs(fine)
            if np.any(np.isfinite(vals2)):
                imin2 = int(np.argmin(vals2))
                vx2 = float(fine[imin2])
                vv2 = float(vals2[imin2])
                if vv2 < best_val:
                    best_x = vx2
                    best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-14:
                w = best_x - v
                p[i] = best_x
                c[start:end] += w * infl
                di = 2 * i
                if 0 <= di < Ltot:
                    c[di] += w * w
                # Renormalize
                S_new = 1.0 + w
                if S_new > 1e-10:
                    p /= S_new
                    c /= (S_new * S_new)
                cur_score = best_val * (S_new * S_new) / (S_new * S_new)
                cur_score = float(2.0 * n * c.max())
                improved = True

        # Verify and track best
        np.maximum(c, 0.0, out=c)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
            if consider:
                consider(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 10.0:
                    p = p * np.exp(rng.normal(0.0, 0.03, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider:
                            consider(best_p)
                    flat = 0
                else:
                    break
        sweep += 1

    np.maximum(p, 0.0, out=p)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    c = _conv_self(p)
    sc = float(2.0 * n * c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 20.0  # 20s reserve

    parent = _get_parent()
    n_parent = parent.size

    best_arr = parent.copy()
    best_score = float(evaluate_sequence([float(x) for x in best_arr]))
    if not math.isfinite(best_score):
        best_score = math.inf

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in a]))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-13:
            best_arr = a.copy()
            best_score = sc

    # ============ Phase 1: LSE optimization ============
    lse_stop = min(hard_deadline - 300.0, t0 + 400.0)
    if lse_stop > t0 + 30.0:
        try:
            p_lse = _lse_optimize(parent, lse_stop, consider=consider)
            if p_lse is not None and p_lse.size == n_parent:
                consider(p_lse)
        except Exception:
            pass

    # ============ Phase 2: LP best response ============
    lp_stop = min(hard_deadline - 150.0, t0 + 850.0)
    if lp_stop > time.monotonic() + 60.0:
        try:
            p_lp = _lp_best_response(best_arr, lp_stop, max_iter=100, consider=consider)
            if p_lp is not None and p_lp.size == n_parent:
                consider(p_lp)
        except Exception:
            pass

    # ============ Phase 3: Second LSE pass (fine temperatures) ============
    lse2_stop = min(hard_deadline - 120.0, t0 + 1050.0)
    if lse2_stop > time.monotonic() + 60.0:
        try:
            p_lse2 = _lse_optimize(best_arr, lse2_stop, consider=consider)
            if p_lse2 is not None and p_lse2.size == n_parent:
                consider(p_lse2)
        except Exception:
            pass

    # ============ Phase 4: Coordinate descent polish ============
    coord_stop = min(hard_deadline - 30.0, t0 + budget - 35.0)
    if coord_stop > time.monotonic() + 30.0:
        try:
            p_cd, _ = _coord_polish(best_arr, coord_stop, consider=consider,
                                    seed=int(seed) + 777)
            if p_cd is not None and p_cd.size == n_parent:
                consider(p_cd)
        except Exception:
            pass

    # ============ Final validation ============
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        try:
            return [float(x) for x in parent]
        except Exception:
            return [1.0]
