        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p

def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _cross_corr(p, w):
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    L = 3 * n - 2
    size = next_fast_len(L)
    F = rfft(p[::-1], size)
    G = rfft(w, size)
    c = irfft(F * G, size)[:L]
    q = c[n - 1:2 * n - 1]
    np.maximum(q, 0.0, out=q)
    return q

def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    return e / s

def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))

def _score_exact_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    return float(2.0 * a.size * m / (s * s))

def _lse_optimize(p, stop_time, consider=None, seed=42):
    """Optimize p using LSE-smoothed max autocorrelation with L-BFGS-B."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    
    def to_u(pp):
        pp = np.maximum(pp, 1e-300)
        uu = np.log(pp)
        return uu - float(np.mean(uu))
    
    u = to_u(p)
    
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c
    
    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q
    
    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros(n)
        
        T = max(T, 1e-300)
        scaled = (c - cmax) / T
        # Clip to avoid overflow
        scaled = np.clip(scaled, -50, 0)
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        
        if sum_exp > 0 and math.isfinite(sum_exp):
            f = cmax + T * math.log(sum_exp / conv_len)
            w = exps / sum_exp
        else:
            f = cmax
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
        
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad
    
    # Temperature schedule
    c0 = cfast(p)
    cmax0 = float(c0.max()) if c0.size > 0 else 1.0
    
    # Stages: (fraction_of_cmax, time_fraction)
    stages = [
        (0.10, 0.06),
        (0.04, 0.08),
        (0.015, 0.10),
        (0.005, 0.12),
        (0.002, 0.14),
        (0.0008, 0.12),
        (0.0003, 0.10),
        (0.0001, 0.08),
        (0.00003, 0.06),
        (0.00001, 0.05),
        (0.000003, 0.05),
        (0.000001, 0.04),
    ]
    
    start = time.monotonic()
    avail = max(0.0, float(stop_time) - start - 10.0)
    if avail < 20:
        return p
    
    best_p = p.copy()
    best_sc = _score_fft(p)
    
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 8:
            break
        cum += wt
        end_stage = min(stop_time - 6, start + avail * cum)
        if end_stage <= now + 5:
            continue
        
        T = max(1e-300, frac * cmax0)
        
        # Run L-BFGS-B for this temperature stage
        best_u_here = u.copy()
        best_f_here, _ = lse_func(u, T)
        stale = 0
        
        while time.monotonic() < end_stage - 2:
            maxit = 30
            opts = {
                "maxiter": maxit,
                "gtol": 1e-12,
                "ftol": 1e-14,
                "maxls": 25,
                "maxcor": 30,
                "maxdisp": 0
            }
            try:
                res = minimize(lse_func, best_u_here, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            
            if res.x is not None and np.all(np.isfinite(res.x)):
                try:
                    fval = float(res.fun)
                except:
                    fval = math.inf
                if math.isfinite(fval) and fval < best_f_here:
                    best_f_here = fval
                    best_u_here = np.array(res.x, dtype=np.float64)
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if getattr(res, "status", -1) == 0:
                break
        
        p_cur = _softmax(best_u_here)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        
        sc_cur = _score_fft(p_cur)
        if not math.isfinite(sc_cur):
            u = to_u(best_p)
            continue
        
        if sc_cur < best_sc:
            best_sc = sc_cur
            best_p = p_cur.copy()
            p = p_cur.copy()
            u = to_u(p)
            
            if consider is not None:
                try:
                    consider(best_p)
                except:
                    pass
        else:
            # Keep the current p for continuity, but don't update best
            p = p_cur.copy()
            u = to_u(p)
    
    return best_p
