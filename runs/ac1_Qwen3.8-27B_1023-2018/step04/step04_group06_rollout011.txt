```python
def _lp_with_dual(a, M, n, time_limit=5):
    """Solve LP and return primal, dual, and objective."""
    k_arr = np.arange(2*n-1)[:, None]
    j_arr = np.arange(n)[None, :]
    d = k_arr - j_arr
    mask = (d >= 0) & (d < n)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    A[mask] = a[d[mask]]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs',
                     options={'time_limit': time_limit,
                              'dual_feasibility_tolerance': 1e-12,
                              'primal_feasibility_tolerance': 1e-12})
        if res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0, None)
            dual = res.ub  # dual variables for A_ub constraints
            return g, dual, -res.fun
    except:
        pass
    return None, None, None
```
