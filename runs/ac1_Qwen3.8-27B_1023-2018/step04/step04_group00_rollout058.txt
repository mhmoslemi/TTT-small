def _lp_iteration(f, n, stop_time, consider):
    """One LP-based iteration."""
    conv = _conv_self(f)
    M = float(conv.max())
    if M <= 0 or not math.isfinite(M):
        return f
    
    # Select active constraints (top 15% of conv values)
    thresh = M * 0.85
    active = np.where(conv >= thresh)[0]
    if len(active) < 20:
        active = np.argsort(conv)[-100:]
    if len(active) > 200:
        active = np.argsort(conv)[::-1][:200]
    
    # Build sparse constraint matrix
    n_active = len(active)
    A_rows = np.zeros(n_active * n, dtype=np.int32)
    A_cols = np.zeros(n_active * n, dtype=np.int32)
    A_vals = np.zeros(n_active * n, dtype=np.float64)
    nnz = 0
    b_ub = np.zeros(n_active, dtype=np.float64)
    
    for idx, k in enumerate(active):
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        m_count = j_hi - j_lo + 1
        if m_count <= 0:
            nnz = -1  # signal error
            break
        A_rows[nnz:nnz+m_count] = idx
        A_cols[nnz:nnz+m_count] = range(j_lo, j_hi+1)
        A_vals[nnz:nnz+m_count] = f[k-j_hi:k-j_lo+1]
        nnz += m_count
        b_ub[idx] = M
    
    if nnz < 0 or nnz == 0:
        return f
    
    A = csr_matrix((A_vals[:nnz], (A_rows[:nnz], A_cols[:nnz])), shape=(n_active, n))
    
    t_clamp = time.monotonic() + 45.0
    if t_clamp > stop_time - 5:
        return f
    
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)]*n, method='highs',
                      options={'time_limit': 40, 'output_flag': 0})
    except Exception:
        return f
    
    if res is None or not getattr(res, 'success', False):
        return f
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0)
    sg = float(g.sum())
    if sg < 1e-10:
        return f
    g = g / sg
    
    # Line search: try several step sizes
    best_f = f.copy()
    best_score = _score_fft(f)
    for t in (0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
        f_trial = (1-t)*f + t*g
        f_trial = np.maximum(f_trial, 0)
        s = float(f_trial.sum())
        if s < 1e-10:
            continue
        f_trial = f_trial / s
        sc = _score_fft(f_trial)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_trial.copy()
            consider(best_f)
    
    return best_f

def _lbfgs_optimize(p, stop_time, consider=None, fine_only=False):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    
    conv_size = next_fast_len(2*n - 1)
    cross_size = next_fast_len(3*n - 2)
    conv_len = 2*n - 1
    cross_len = 3*n - 2
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F*F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c
    
    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F*G, cross_size)[:cross_len]
        q = c[n-1:2*n-1]
        np.maximum(q, 0.0, out=q)
        return q
    
    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if T <= 0:
            T = max(1e-300, 1e-4*cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * math.log(sum_exp / c.size)
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad
    
    u = np.log(np.maximum(p, 1e-300))
    u = u - u.mean()
    best_u = u.copy()
    best_f = lse_func(u, 0.01)[0]
    
    if fine_only:
        stages = ((0.003, 0.30), (0.0008, 0.45), (0.0002, 0.25))
    else:
        stages = (
            (0.20, 0.10), (0.06, 0.15), (0.018, 0.20),
            (0.005, 0.20), (0.0015, 0.25), (0.0004, 0.10),
        )
    
    start = time.monotonic()
    avail = max(0.0, stop_time - start - 12.0)
    if avail < 10:
        return p
    
    total_w = sum(w for _, w in stages)
    cum = 0.0
    p_cur = p.copy()
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 8:
            break
        cum += wt
        end_stage = min(stop_time - 6, start + avail * (cum/total_w))
        if end_stage <= now + 5:
            continue
        
        c = cfast(p_cur)
        cmax = float(c.max()) if c.size else 0
        T = max(1e-300, frac * cmax)
        
        stale = 0
        while time.monotonic() < end_stage - 2:
            maxit = 30 if n > 1500 else 40
            try:
                res = minimize(lse_func, best_u, args=(T,), method='L-BFGS-B',
                              jac=True, options={'maxiter': maxit, 'gtol': 1e-10,
                                                 'ftol': 1e-15, 'maxls': 20, 'maxcor': 20, 'disp': False})
            except Exception:
                break
            if res.x is not None and res.x.size == n:
                xv = np.asarray(res.x, dtype=np.float64)
                if np.all(np.isfinite(xv)):
                    fv = float(res.fun)
                    if math.isfinite(fv) and fv < best_f:
                        best_f = fv
                        best_u = xv.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, 'status', 1) == 0:
                break
        
        p_cur = _softmax(best_u)
        p_cur = np.maximum(p_cur, 0)
        s = float(p_cur.sum())
        if s > 0:
            p_cur /= s
        sc = _score_fft(p_cur)
        if math.isfinite(sc) and consider:
            consider(p_cur)
    
    return p_cur
