def _lp_minimax(A, n, time_limit=5.0):
    """Solve: min M s.t. A@g <= M*1, sum(g) = 1, g >= 0.
    Variables: g (n), M (1) -> total n+1 variables.
    Constraints: A@g - M <= 0 (2n-1 constraints), sum(g) = 1, g >= 0."""
    # Using linprog: min c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds
    # x = [g_0, ..., g_{n-1}, M]
    c = np.zeros(n+1)  # minimize M, so c[-1] = 1
    c[-1] = 1.0
    
    # A_ub: [A, -1] * [g; M] <= 0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    # A_eq: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)]*n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
    except Exception:
        return None
    
    if res.status != 0:
        return None
    return res.x[:n], res.x[n]

def _subgradient_step(a, k_star, alpha, n):
    """One subgradient step on -conv(a,a)_{k_star} (minimizing it).
    Project onto {a >= 0, sum(a) = 1}."""
    grad = np.zeros(n)
    jlo = max(0, k_star - n + 1)
    jhi = min(n, k_star + 1)
    for j in range(jlo, jhi):
        grad[j] = 2.0 * a[k_star - j]
    # Projected gradient step: a_new = proj(a - alpha * grad)
    a_new = a - alpha * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new /= s
    return a_new
