```python
    n = len(a)

    weights /= weights.sum()

    # grad_q_j(a)_m = a[m-j] + a[m+j] (with bounds)

            # a[m+j] contribution  
            if 0 <= m+j < n:

    # Project to zero-sum (since we're on the simplex)
    grad -= grad.mean()
```

```python
def subgradient_step(a, step_candidates=None):
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    # Find active set
    active = c >= M - 1e-6 * M
    # Compute subgradient
    sg = np.zeros(n)
    for j in range(2*n-1):
        if not active[j]:
            continue
        w = 1.0 / np.sum(active)
        for m in range(n):
            if 0 <= m-j < n:
                sg[m] += w * a[m-j]
            if 0 <= m+j < n:
                sg[m] += w * a[m+j]
    # Project to zero-mean (for simplex constraint)
    sg -= sg.mean()
    
    # Backtracking line search
    best_a = a.copy()
    best_val = M
    base = best_val
    for alpha in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
        cand = a - alpha * sg
        cand = np.maximum(cand, 0)
        cs = cand.sum()
        if cs < 1e-12:
            continue
        cand /= cs
        val = np.convolve(cand, cand).max()
        if val < best_val:
            best_val = val
            best_a = cand
    return best_a, best_val
```
