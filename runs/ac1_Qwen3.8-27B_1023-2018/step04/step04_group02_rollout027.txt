We want to minimize `max(conv(b,b))` subject to `sum(b) = 1`, `b >= 0`.

This is a non-convex problem (the max of linear functions is convex, so minimizing it is non-convex). But we can use a "peak reduction" approach:

1. Start with some `b`
2. Find the lag `k*` where `conv(b,b)` is maximized
3. Solve: minimize `conv(b,b)[k*]` subject to `conv(b,b)[k] <= conv(b,b)[k*]` for all `k`, `sum(b) = 1`, `b >= 0`

This is a quadratic programming problem (minimize a quadratic subject to quadratic inequalities), which is hard in general. But if we linearize the constraints (use first-order approximations), it becomes an LP.

Alternatively, we can use the "iterative proportion" approach:
1. Find `k*` = argmax of `conv(b,b)`
2. Compute the "pressure" at each coordinate: `p[i] = conv(b,b)[k*]`'s sensitivity to `b[i]`
3. Update: `b[i] *= exp(-eta * p[i])` (or some similar multiplicative update)
4. Renormalize

This is similar to the "water-filling" or "apportionment" methods used in peak reduction.

import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len

def _conv(a):
    n = len(a)
    if n <= 4096:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _trim(a, tol=1e-10):
    n = len(a)
    start = 0
    while start < n and a[start] < tol:
        start += 1
    end = n
    while end > start and a[end-1] < tol:
        end -= 1
    return a[start:end]

def _score_exact(a):
    """Use the exact evaluator."""
    try:
        return float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        return math.inf

def _score(a):
    n = len(a)
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """Build the convolution constraint matrix. A[m, k] = f[m-k] if valid."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(n):
        # A[m, k] = f[m-k] for m such that 0 <= m-k < n, i.e., k <= m < k+n
        m_start = k
        m_end = min(2 * n - 1, k + n)
        # m-k ranges from 0 to m_end-1-k
        A[m_start:m_end, k] = f[:m_end - m_start]
    return A

def _lp_direction(f, n, M, timeout=3.0):
    """Find direction g that maximizes sum(g) subject to f*g <= M, g >= 0."""
    A = _build_A(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, 
                     method='highs', 
                     options={'time_limit': timeout,
                              'dual_feasibility_tolerance': 1e-10,
                              'primal_feasibility_tolerance': 1e-10})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _gradient_direction(f, n):
    """Compute subgradient of max(conv(f,f)) and return a descent direction."""
    b = _conv(f)
    k_star = int(np.argmax(b))
    M = b[k_star]
    # d(conv(f,f)[k_star])/d(f[i])
    # = 2*f[k_star - i] if 0 <= k_star - i < n, adjusted for i == k_star - i
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            if i == j:
                grad[i] = f[i]
            else:
                grad[i] = 2.0 * f[j]
    # We want to decrease M, so move in -grad direction
    # But also want to maintain sum(f)
    # Simple: g = f - eta * grad, clipped to >= 0
    return grad, k_star, M

def _cd_polish(f, stop_time, max_outer=50, seed=0, tol=1e-10):
    """Coordinate descent polish with trimming."""
    n = len(f)
    rng = np.random.default_rng(seed)
    f = np.array(f, dtype=np.float64).copy()
    cur = _score(f)
    best = f.copy()
    best_sc = cur
    
    mults = [0.1, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.7, 2.0, 3.0, 0.0]
    
    for outer in range(max_outer):
        if time.monotonic() > stop_time - 3.0:
            break
        improved_any = False
        order = rng.permutation(len(f))
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            if i >= len(f):
                continue
            orig = f[i]
            f[i] = 0.0
            f = _trim(f, tol)
            # Now f might be shorter
            n_now = len(f)
            if i >= n_now:
                # We trimmed past i, restore
                f = best.copy()
                cur = best_sc
                break
            
            best_v = f[i]
            best_local = _score(f)
            
            f[i] = orig
            for m in mults:
                v = max(0.0, orig * m) if orig > 1e-12 else (m * 0.5 * max(0.1, f.mean()) if m > 0 else 0.0)
                f[i] = v
                sc = _score(f)
                if sc < best_local:
                    best_local = sc
                    best_v = v
            
            f[i] = best_v
            f = _trim(f, tol)
            
            if best_local < cur - 1e-14:
                cur = best_local
                improved_any = True
                if cur < best_sc:
                    best_sc = cur
                    best = f.copy()
        
        if not improved_any:
            if outer >= 3:
                break
    
    return best, best_sc

import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len

def _conv(a):
    n = len(a)
    if n <= 4096:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _trim(a, tol=1e-9):
    if len(a) == 0:
        return a
    start = 0
    end = len(a)
    while start < end and a[start] < tol:
        start += 1
    while end > start and a[end - 1] < tol:
        end -= 1
    return a[start:end]

def _score(a):
    n = len(a)

        return math.inf

    m = float(b.max())

        m_end = min(2 * n - 1, k + n)
        A[m_start:m_end, k] = f[:m_end - m_start]
    return A

    n = len(f)

    best_score = _score(f)
    t = t_init
    stall = 0
    max_stall = 100
    it = 0
    
    while time.monotonic() < stop_time - 5.0:
        if max_iters is not None and it >= max_iters:

        n = len(f)
        b = _conv(f)

        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)

        timeout = min(2.0, max(0.5, stop_time - 10.0 - time.monotonic()))
        try:

            stall += 5

            f = _trim(f, 1e-9)

            continue

            continue
        g = g * (f.sum() / g_sum)

                f = _trim(f, 1e-9)
                if len(f) < 4 or f.sum() < 1e-10:
                    f = best_f.copy()
                t = 0.2
                stall = 0
                max_stall = min(max_stall + 50, 300)
        it += 1
    
    return best_f, best_score
