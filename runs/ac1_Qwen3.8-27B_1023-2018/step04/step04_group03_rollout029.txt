        if s < 1e-15:
            break
        f_new = f_new / s

        sc = _score(f_new)
        if sc < best_score - 1e-13:
            best_f = f_new.copy()
            best_score = sc
            stall = 0
            lr *= 0.995
        else:
            damp = f_new.copy()
            for frac in [0.5, 0.3, 0.1]:
                damp = (1 - frac) * f + frac * f_new
                dc = _score(damp)
                if dc < best_score - 1e-13:
                    f_new = damp
                    best_f = f_new.copy()
                    best_score = dc
                    stall = 0
                    lr *= 0.98
                    break
            else:
                stall += 1
                lr *= 0.93
                if stall > max_stall:
                    noise = rng.normal(0, 0.02, n)
                    f = np.clip(f + noise, 1e-15, None)
                    f = f / f.sum()
                    stall = 0
                    lr = 0.3
                    max_stall += 40
                    if max_stall > 400:
                        break

        f = f_new

    return best_f, best_score


def _polish(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    best_sc = _score(f)
    no_imp = 0
    max_no_imp = 15

    mults_coarse = (0.2, 0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 3.0, 0.0)
    mults_fine = (0.9, 1.1, 0.85, 1.15, 0.95, 1.05)

    while time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            if f[i] < 1e-14:
                f[i] = 0.01 * max(1.0, f.mean() if f.mean() > 0 else 1.0)
                sc = _score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                else:
                    f[i] = 0.0
                continue

            orig = f[i]
            best_local = best_sc
            best_val = orig

            for mlt in mults_coarse:
                f[i] = orig * mlt
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_val = f[i]

            if best_local < best_sc - 1e-15:
                f[i] = best_val
                orig = best_val
                for mlt in mults_fine:
                    f[i] = orig * mlt
                    if f.sum() < 0.01:
                        f[i] = orig
                        continue
                    sc = _score(f)
                    if sc < best_local - 1e-15:
                        best_local = sc
                        best_val = f[i]
                    else:
                        f[i] = orig
                f[i] = best_val
                best_sc = best_local
                improved = True
            else:
                f[i] = best_val

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return f.copy(), best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = _score(best_seq)

    def _update_best(candidate):
        nonlocal best_seq, best_score
        if candidate is None or len(candidate) == 0:
            return
        candidate = np.clip(np.asarray(candidate, dtype=np.float64), 0, 1000)
        if not np.all(np.isfinite(candidate)):
            return
        if candidate.sum() < 0.01:
            return
        sc = _score(candidate)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq = candidate.copy()
            best_score = sc

    rng = np.random.default_rng(seed)
    i_arr = np.arange(n_p, dtype=np.float64)

    # Generate diverse initializations
    inits = [parent.copy()]

    sig = 0.1 * max(1.0, float(parent.mean()))
    inits.append(np.clip(parent + rng.normal(0, sig, n_p), 0, None))
    inits.append(np.clip(parent * (1.0 + 0.2 * rng.normal(0, 1, n_p)), 0, None))

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n_p)) + 0.01)

    # Multi-harmonic
    sup = np.ones(n_p) * 0.3
    for fr in (1, 2, 3, 5, 7, 11, 13):
        ph = rng.uniform(0, 2 * math.pi)
        sup += 0.08 * np.cos(2.0 * math.pi * fr * i_arr / n_p + ph)
    inits.append(np.abs(sup) + 0.02)

    # Gaussian center
    center = (n_p - 1) / 2.0
    for width in (n_p * 0.15, n_p * 0.25, n_p * 0.35):
        inits.append(np.exp(-0.5 * ((i_arr - center) / width) ** 2))

    # Two Gaussians
    c1, c2 = n_p * 0.3, n_p * 0.7
    w = n_p * 0.1
    inits.append(np.exp(-0.5 * ((i_arr - c1) / w) ** 2) +
                 np.exp(-0.5 * ((i_arr - c2) / w) ** 2))

    # Geometric (asymmetric - key for reducing center peak)
    for r in (0.98, 0.99, 0.995):
        geo = r ** i_arr
        inits.append(geo / geo.sum())

    # Anti-symmetric geometric (concentrated at one end)
    for r in (0.95, 0.97):
        geo = r ** i_arr
        f_asym = geo / geo.sum()
        inits.append(f_asym)
        inits.append(f_asym[::-1])

    # W-shaped
    f_w = np.exp(-0.5 * ((i_arr - n_p * 0.25) / (n_p * 0.08)) ** 2) + \
          np.exp(-0.5 * ((i_arr - n_p * 0.75) / (n_p * 0.08)) ** 2) + \
          np.ones(n_p) * 0.1
    inits.append(f_w)

    # Phase 1: LP optimization on multiple initializations (50% budget)
    phase1_end = t0 + budget * 0.50
    time_per_init = max(15.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        if time.monotonic() > stop - 3.0:
            continue
        try:
            res, sc = _optimize_lp(init, stop, seed + 1000 * (idx + 1),
                                   use_full=True, peak_frac=0.05)
            _update_best(res)
        except Exception:
            pass

    # Second LP pass on best (full constraints)
    if time.monotonic() < phase1_end - 10.0:
        try:
            res, sc = _optimize_lp(best_seq, phase1_end - 5.0, seed + 888888,
                                   use_full=True, peak_frac=0.03)
            _update_best(res)
        except Exception:
            pass

    # Phase 2: Subgradient on best (12% budget)
    phase2_end = t0 + budget * 0.62
    if time.monotonic() < phase2_end - 10.0:
        try:
            res, sc = _subgradient_optimize(best_seq, phase2_end - 5.0, seed + 77777)
            _update_best(res)
        except Exception:
            pass

    # Phase 3: LP again on result of subgradient (15% budget)
    phase3_end = t0 + budget * 0.77
    if time.monotonic() < phase3_end - 10.0:
        try:
            res, sc = _optimize_lp(best_seq, phase3_end - 5.0, seed + 666666,
                                   use_full=True, peak_frac=0.02)
            _update_best(res)
        except Exception:
            pass

    # Phase 4: Coordinate descent polish (10% budget)
    phase4_end = t0 + budget * 0.87
    if time.monotonic() < phase4_end - 10.0:
        try:
            res, sc = _polish(best_seq, phase4_end - 5.0, seed + 55555)
            _update_best(res)
        except Exception:
            pass

    # Phase 5: Final LP + polish combo (7% budget)
    phase5_end = deadline + 5.0
    if time.monotonic() < phase5_end - 15.0:
        try:
            res, sc = _optimize_lp(best_seq, phase5_end - 10.0, seed + 444444,
                                   use_full=True, peak_frac=0.01)
            _update_best(res)
        except Exception:
            pass

    if time.monotonic() < phase5_end - 10.0:
        try:
            res, sc = _polish(best_seq, phase5_end - 5.0, seed + 333333)
            _update_best(res)
        except Exception:
            pass

    # Try a few different lengths
    if time.monotonic() < deadline - 30.0:
        for L in [n_p - 32, n_p + 32, n_p - 64, n_p + 64]:
            if time.monotonic() > deadline - 20.0:
                break
            if L < 64 or L > 800:
                continue
            x_old = np.linspace(0, 1, n_p)
            x_new = np.linspace(0, 1, L)
            init_L = np.interp(x_new, x_old, parent)
            stop = min(deadline - 15.0, time.monotonic() + 20.0)
            if time.monotonic() > stop - 3.0:
                continue
            try:
                res, sc = _optimize_lp(init_L, stop, seed + L * 7,
                                       use_full=True, peak_frac=0.05)
                _update_best(res)
            except Exception:
                pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    return [float(x) for x in best_seq]
