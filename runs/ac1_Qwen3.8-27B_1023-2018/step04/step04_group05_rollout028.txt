For a sequence f of length n with sum(f) = S, the "average" autocorrelation is (1/(2n-1)) * sum_k (f*f)_k = (sum_k (f*f)_k) / (2n-1).

Now sum_k (f*f)_k = sum_k sum_j f_j * f_{k-j} = sum_j sum_k f_j * f_{k-j} = sum_j f_j * sum_k f_{k-j} = sum_j f_j * S = S^2.

So the average autocorrelation value is S^2/(2n-1) ≈ S^2/(2n) for large n.

The objective is 2n * max_b / S^2. If the autocorrelation were flat (all values equal to the average), then max_b = S^2/(2n-1) ≈ S^2/(2n), and the objective would be 2n * S^2/(2n) / S^2 = 1.

But of course, the autocorrelation can't be perfectly flat for a nonnegative sequence. The value at lag 0 (or rather, at the "center" of the full convolution) is related to the "energy" of the sequence.

Actually, for the full linear convolution, the center value (at k = n-1) is:
b[n-1] = sum_j f_j * f_{n-1-j}

This is the cross-correlation of f with its reverse at zero lag. If f is palindromic (f[j] = f[n-1-j]), then b[n-1] = sum_j f_j^2 = ||f||_2^2.

For a non-palindromic f, b[n-1] can be different from ||f||_2^2.

The maximum of b can be at any position, not just the center.

For a "good" sequence, we want all b[k] to be as small and equal as possible. The constraint is that sum of all b[k] = S^2.

If we could make all b[k] = S^2/(2n-1), the objective would be 1. The question is how close to 1 we can get.

The target of 1.5027 corresponds to max_b = 1.5027 * S^2 / (2n).
The "flat" value would be S^2/(2n-1) ≈ S^2/(2n).
So max_b / flat_value ≈ 1.5027. 

We need the peak to be only 50% above the average. That's quite achievable.

For comparison, a random nonnegative sequence would have a peak that's O(sqrt(n)) times the average (by concentration of measure), giving an objective of O(sqrt(n)) which is much larger.

A constant sequence f = [c, c, ..., c] has:
b[k] = c^2 * min(k+1, n, 2n-1-k) for k = 0, ..., 2n-1
max_b = c^2 * n (at k = n-1)
S = nc
Objective = 2n * c^2 * n / (nc)^2 = 2n * c^2 * n / (n^2 * c^2) = 2

def _conv_full(a):
    n = len(a)
    if n <= 800:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=5.0):
    if M <= 1e-18:
        return None
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-14)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"presolve": True, "time_limit": timeout,
                               "dual_feasibility_tolerance": 1e-12,
                               "primal_feasibility_tolerance": 1e-12})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g


def _softmax_gradient(f, n, tau):
    """Compute gradient of softmax approximation of max_k (f*f)_k."""
    b = _conv_full(f)
    # Numerically stable softmax
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    Z = exp_b.sum()
    if Z < 1e-300:
        return None
    p = exp_b / Z  # weights over lags
    # Gradient: grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * f[j] for valid i+j
    grad = np.zeros(n)
    for i in range(n):
        valid_end = min(2 * n - 1, i + n)
        if i >= 2 * n - 1:
            break
        seg_p = p[i:valid_end]
        seg_f = f[:len(seg_p)]
        grad[i] = 2.0 * np.dot(seg_p, seg_f)
    return grad, b, b.max()


def _adam_step(f, grad, lr, m_cache, v_cache, beta1, beta2, t, eps=1e-8):
    m_cache[:] = beta1 * m_cache + (1 - beta1) * grad
    v_cache[:] = beta2 * v_cache + (1 - beta2) * grad * grad
    m_hat = m_cache / (1 - beta1 ** t)
    v_hat = v_cache / (1 - beta2 ** t)
    return f - lr * m_hat / (np.sqrt(v_hat) + eps)


def _project(f, target_sum):
    f = np.clip(f, 0.0, None)
    s = f.sum()
    if s < 1e-15:
        return np.full_like(f, target_sum / len(f))
    return f * (target_sum / s)

            pass
        # Try a random init at this length too
        if _time_left() > 30:
            stop2 = min(phase3_end - 5.0, time.monotonic() + time_per_len * 0.7)
            init_rand = np.abs(rng_main.normal(0, 1, L)) + 0.01
            try:
                res_r, sc_r = _lp_optimize(init_rand, L, stop2, seed + 90000 + L)
                _update_best(res_r, sc_r)
            except Exception:
                pass

    # Phase 4: Second LP pass on global best (longer time)
    if _time_left() > 80:
        stop_p4 = min(phase3_end + budget * 0.05, deadline - 60.0)
        try:
            res4, sc4 = _lp_optimize(best_seq, len(best_seq), stop_p4, seed + 999999)
            _update_best(res4, sc4)
        except Exception:
            pass

    # Phase 5: Polishing (remaining time)
    polish_end = deadline - 10.0
    if _time_left() > 20:
        try:
            b_seq, b_sc = _polish(best_seq, polish_end, seed=4242)
            _update_best(b_seq, b_sc)
        except Exception:
            pass

    # Second polish with different seed
    if _time_left() > 15:
        try:
            b_seq2, b_sc2 = _polish(best_seq, deadline - 8.0, seed=88888)
            _update_best(b_seq2, b_sc2)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    if final_sc < best_score:
        best_score = final_sc

    return [float(x) for x in best_seq]
