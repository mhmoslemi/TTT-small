import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _fft_conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _fft_conv(a, b):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    G = rfft(b, fsize)
    c = irfft(F * G, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _fft_conv_full(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _solve_1d_minimax(A, B, C, t_lo=0.0, t_hi=1.0):
    """
    Find t in [t_lo, t_hi] that minimizes max_k (A[k]*t^2 + B[k]*t + C[k]).
    A, B, C are arrays of the same length.
    """
    # Evaluate at several points first
    candidates = np.linspace(t_lo, t_hi, 21)
    best_t = t_lo
    best_val = math.inf
    
    for t in candidates:
        vals = A * t * t + B * t + C
        v = float(vals.max())
        if v < best_val:
            best_val = v
            best_t = t
    
    # Refine with golden section
    gr = (1.0 + math.sqrt(5)) / 2.0
    lo = max(t_lo, best_t - (t_hi - t_lo) * 0.15)
    hi = min(t_hi, best_t + (t_hi - t_lo) * 0.15)
    
    prev_val = best_val
    for _ in range(30):
        m1 = hi - (hi - lo) / gr
        m2 = lo + (hi - lo) / gr
        v1 = float((A * m1 * m1 + B * m1 + C).max())
        v2 = float((A * m2 * m2 + B * m2 + C).max())
        if v1 < v2:
            hi = m2
        else:
            lo = m1
        # Check midpoint
        tm = (lo + hi) / 2.0
        vm = float((A * tm * tm + B * tm + C).max())
        if vm < best_val:
            best_val = vm
            best_t = tm
    
    return best_t, best_val

def _build_A_matrix(f, n):
    """Build the convolution matrix A such that A @ g = f * g (full conv)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_direction(f, n, M, time_limit=2.0):
    """Solve LP: maximize sum(g) subject to g >= 0, (f*g)[k] <= M for all k."""
    A = _build_A_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                     bounds=[(0, None)] * n,
                     method='highs',
                     options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum

def _lp_direction_focused(f, n, M, top_k=200, time_limit=1.0):
    """Solve LP with only the top constraints (where f*f is largest)."""
    ff = _fft_conv_full(f)
    idx = np.argsort(-ff)[:top_k]
    A = _build_A_matrix(f, n)
    A_sub = A[idx]
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A_sub, b_ub=np.full(len(idx), M),
                     bounds=[(0, None)] * n,
                     method='highs',
                     options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum

def _lp_direction_min_cross(f, n, time_limit=2.0):
    """Solve: minimize max_k (f*g)[k] subject to g >= 0, sum(g) = 1."""
    a = _build_A_matrix(f, n)
    c = np.zeros(n + 1)
    c[-1] = 1  # minimize t
    
    # Constraints: (f*g)[k] - t <= 0 for all k
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:2*n-1, :n] = a
    A_ub[:2*n-1, n] = -1
    b_ub = np.zeros(2 * n - 1)
    
    # sum(g) = 1 -> -sum(g) <= 0 and sum(g) <= 1
    # Actually, let's just add it as equality... but linprog doesn't handle eq in A_ub
    # Use A_eq
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x[:n], dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum

    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum
    
    best_f = f.copy()
    best_M = float(_fft_conv_full(f).max())
    
    iter_count = 0
    stall = 0
    no_improve = 0
    consecutive_no_imp = 0
    
    while time.monotonic() < stop_time - 1.0:
        ff = _fft_conv_full(f)
        M = float(ff.max())
        
        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            consecutive_no_imp = 0
        else:
            consecutive_no_imp += 1
        
        if consecutive_no_imp > 30:
            break

        use_focused = (iter_count % 3 == 1)
        if use_focused:
            g_dir = _lp_direction_focused(f, n, M, top_k=300, time_limit=1.5)
            if g_dir is None:
                g_dir = _lp_direction(f, n, M, time_limit=2.0)
        else:
            g_dir = _lp_direction(f, n, M, time_limit=2.0)
            if g_dir is None:

            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            stall += 3
            iter_count += 1
            continue

        if dist < 1e-8:
            f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            stall += 2
            iter_count += 1

        # = t^2 * (gg[k] - 2*fg[k] + ff[k]) + t * (2*fg[k] - 2*ff[k]) + ff[k]
        A_coef = gg - 2 * fg + ff
        B_coef = 2 * fg - 2 * ff
        C_coef = ff
        
        t_opt, m_opt = _solve_1d_minimax(A_coef, B_coef, C_coef)

            f = (1.0 - t_opt) * f + t_opt * g_dir
            f = np.clip(f, 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            stall = 0
        else:
            stall += 1
            # Try smaller step
            for t_try in [0.01, 0.02, 0.05]:

                f_try = np.clip(f_try, 0, None)
                fs_try = f_try.sum()
                if fs_try < 1e-10:
                    continue
                f_try = f_try / fs_try
                m_try = float(_fft_conv_full(f_try).max())
                if m_try < M - 1e-14:
                    f = f_try
                    stall = 0
                    break
            else:
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)

            fs = f.sum()

Let me restructure the coordinate descent:

```python
for i in order:
    old_val = a[i]
    for candidate in candidates:
        d = candidate - old_val
        # Apply
        _conv_delta(conv, a, i, d, n)  # uses a (which has old_val at position i)
        # Evaluate
        m = conv.max()
        # Revert
        _conv_delta_revert(conv, a, i, d, n)  # same formula, subtract
        a[i] = old_val  # restore (shouldn't be needed if we don't modify a during trial)
    # Apply best
    d = best_candidate - old_val
    _conv_delta(conv, a, i, d, n)
    a[i] = best_candidate
```

Wait, I don't modify `a` during the trial! I only modify `conv`. So `a[i]` is still `old_val` during the trial. The formula uses `a[k-i]` which for `k-i == i` gives `a[i] = old_val`. 

So the revert is just: `_conv_delta(conv, a, i, -d, n)` where `a` still has `old_val` at position i. Let me verify:
`_conv_delta(conv, a, i, -d, n)` does:
`conv[k] += 2*(-d)*a[k-i] + (-d)^2*[k==2i] = -2d*a[k-i] + d^2*[k==2i]`

And the correct revert is:
`conv[k] -= 2*d*a_old[k-i] + d^2*[k==2i]`

Since `a[i] = old_val` (we haven't changed `a`), `a[k-i] = a_old[k-i]`. So:
`-2d*a[k-i] + d^2*[k==2i]` vs `-2d*a_old[k-i] - d^2*[k==2i]`

These differ in the sign of the `d^2` term! The first has `+d^2` and the second has `-d^2`.

So the simple "call with -d" does NOT correctly revert. I need a separate revert function:

```python
def _conv_delta_apply(conv, a, i, d, n):
    """Apply: change a[i] by d (a is pre-change)."""
    k_start = i
    k_end = min(2*n-2, i + n - 1)
    for k in range(k_start, k_end + 1):
        conv[k] += 2.0 * d * a[k - i]
    if 2 * i <= 2 * n - 2:
        conv[2*i] += d * d

def _conv_delta_revert(conv, a, i, d, n):
    """Revert: undo the change of a[i] by d (a is still pre-change)."""
    k_start = i
    k_end = min(2*n-2, i + n - 1)
    for k in range(k_start, k_end + 1):
        conv[k] -= 2.0 * d * a[k - i]
    if 2 * i <= 2 * n - 2:
        conv[2*i] -= d * d
```

This is correct: apply adds `2d*a[k-i] + d^2*[k==2i]` and revert subtracts the same.

OK so this approach works. Let me now put together the full program.

Actually, for n=384, the loop `for k in range(k_start, k_end+1)` has at most 384 iterations. With 10 candidate values and 384 coordinates, that's 384*10*384 = ~1.5M operations per sweep. In Python, this would be slow. Let me vectorize:

```python
def _conv_delta_apply_vec(conv, a, i, d, n):
    k_start = i
    k_end = min(2*n-2, i + n - 1)
    j_range = np.arange(k_start, k_end + 1) - i  # = arange(0, k_end-k_start+1)
    conv[k_start:k_end+1] += 2.0 * d * a[j_range]
    conv[2*i] += d * d
```

This is a single NumPy operation on a slice, which is fast.

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _fft_conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _fft_conv(a, b):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    G = rfft(b, fsize)
    c = irfft(F * G, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _fft_conv_full(a)
    m = float(b.max())
    if m < 0:
        m = 0.0
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve_max_sum(f, n, M, time_limit=3.0):
    """maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for all k"""
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                     bounds=[(0, None)] * n,
                     method='highs',
                     options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum


def _lp_solve_min_cross(f, n, time_limit=3.0):
    """minimize t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1"""
    A_conv = _build_A(f, n)
    # Variables: g (n vars) + t (1 var) = n+1 vars
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # A_ub: (f*g)[k] - t <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_conv
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x[:n], dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum


def _solve_1d_minimax(A_coef, B_coef, C_coef):
    """Find t in [0,1] minimizing max_k (A[k]*t^2 + B[k]*t + C[k])."""
    # Coarse grid
    ts = np.linspace(0, 1, 51)
    vals = np.array([float((A_coef * t**2 + B_coef * t + C_coef).max()) for t in ts])
    best_idx = int(np.argmin(vals))
    best_t = ts[best_idx]
    best_val = vals[best_idx]
    
    # Golden section refinement
    gr = (1.0 + math.sqrt(5)) / 2.0
    span = 2.0 / 50.0  # width of coarse grid cell
    lo = max(0.0, best_t - span)
    hi = min(1.0, best_t + span)
    
    for _ in range(40):
        if hi - lo < 1e-10:
            break
        m1 = hi - (hi - lo) / gr
        m2 = lo + (hi - lo) / gr
        v1 = float((A_coef * m1**2 + B_coef * m1 + C_coef).max())
        v2 = float((A_coef * m2**2 + B_coef * m2 + C_coef).max())
        if v1 < v2:
            hi = m2
        else:
            lo = m1
        tm = (lo + hi) / 2.0
        vm = float((A_coef * tm**2 + B_coef * tm + C_coef).max())
        if vm < best_val:
            best_val = vm
            best_t = tm
    
    return best_t, best_val

    rng = np.random.default_rng(seed)

    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum
    
    ff = _fft_conv_full(f)
    M = float(ff.max())
    best_f = f.copy()
    best_M = M
    
    no_improve = 0
    stall = 0

    while time.monotonic() < stop_time - 1.0:

        if iter_count % 2 == 0:
