from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _cross_conv(a, b):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    FA = rfft(a, fsize)
    FB = rfft(b, fsize)
    c = irfft(FA * FB, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_full(a)
    m = float(c.max())
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, n, M, time_limit=8.0, top_k=None):

    c_vec = -np.ones(n)
    if top_k is not None and top_k < 2 * n - 1:
        b_self = _conv_full(f)
        idx = np.argsort(-b_self)[:top_k]
        A_sub = A[idx]
        b_ub = np.full(len(idx), M)
    else:
        A_sub = A
        b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c_vec, A_ub=A_sub, b_ub=b_ub,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs

                continue
            for fr in (0.0, 0.15, 0.3, 0.5, 0.7, 0.85, 1.0):
                best[i] = fr * tot
                best[j] = (1.0 - fr) * tot
                sc = _score_fast(best)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj

        # Blocks
        nb = min(60, n // 3)
        for _ in range(nb):
            if time.monotonic() > stop_time - 2.0:
                break
            bs = int(rng.integers(2, 6))
            if n < bs:
                continue
            st = int(rng.integers(0, n - bs + 1))
            blk = best[st:st + bs].copy()
            tot = blk.sum()
            if tot < 1e-12:
                continue
            for _ in range(5):
                w = rng.dirichlet(np.ones(bs) * 2.5)
                best[st:st + bs] = w * tot
                sc = _score_fast(best)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
            best[st:st + bs] = blk

        if not improved:
            no_imp += 1
        else:
            no_imp = 0
    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    deadline = t0 + budget - 12.0

    # Get parent
    try:
        h = globals()["height_sequence_1"]
        parent = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(384, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(384, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    n_p = len(parent)
    ps = parent.sum()
    if ps < 1e-10:
        parent = np.ones(n_p) / n_p
    else:
        parent = parent / ps

    best_seq = parent.copy()
    best_score = float(evaluate_sequence([float(x) for x in best_seq]))
    if not math.isfinite(best_score):
        best_seq = np.ones(n_p)
        best_score = float(evaluate_sequence([1.0] * n_p))

    n = n_p
    rng = np.random.default_rng(seed)

    # Phase 1: LP optimization - 68% of budget
    phase1_end = t0 + budget * 0.68
    # Build diverse initializations
    inits = [parent.copy()]
    # Smoothed noise
    kern = np.exp(-0.5 * (np.arange(-8, 9) / 3.0) ** 2)
    kern /= kern.sum()
    noisy = np.convolve(rng.normal(0, 0.5, n), kern, 'same')
    inits.append(np.clip(parent + noisy, 0, None))
    # Mirror blend
    inits.append(0.5 * (parent + parent[::-1]))
    # Taper × parent
    taper = 0.5 + 0.5 * np.cos(np.pi * np.arange(n, dtype=float) / (n - 1))
    inits.append(parent * taper + 0.01)
    # Two-scale
    ref = np.interp(np.linspace(0, 1, max(2, n // 8)), 0, 1, parent)
    coarse = np.interp(np.linspace(0, 1, n), 0, 1, ref)
    inits.append(0.6 * parent + 0.4 * coarse)

    for k in range(len(inits)):
        inits[k] = np.clip(inits[k], 0, None)
        s = inits[k].sum()
        inits[k] = inits[k] / s if s > 1e-10 else np.ones(n) / n

    time_per_init = max(15.0, (phase1_end - t0) / len(inits))

    for ii, init in enumerate(inits):
        if time.monotonic() > phase1_end - 8.0:
            break
        stop = min(phase1_end - 4.0, time.monotonic() + time_per_init)
        if time.monotonic() >= stop - 2.0:
            break

        f = init.copy()
        M = float(_conv_full(f).max())
        stall = 0
        max_stall = 55
        it = 0

        while time.monotonic() < stop - 1.0:
            if stall > max_stall:
                st = stall % 5
                if st == 0:
                    f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
                elif st == 1:
                    f = 0.7 * f + 0.3 * rng.uniform(0.1, 1.0, n)
                elif st == 2:
                    f = np.clip(f + 0.15 * np.gradient(f), 0, None)
                elif st == 3:
                    f = 0.5 * best_seq + 0.5 * np.abs(rng.normal(0, 1, n))
                else:
                    f = parent.copy()
                    M_cur = float(_conv_full(f).max())
                    if M_cur > M * 1.1:
                        f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                fs = f.sum()
                f = f / fs if fs > 1e-10 else parent.copy()
                M = float(_conv_full(f).max())
                stall = 0
                it += 1
                continue

            # LP solve
            use_full = (it < 12) or (it % 10 == 0)
            tl = 8.0 if use_full else 5.0
            tk = None if use_full else min(70, 2 * n - 1)
            g_dir = _lp_solve(f, n, M, time_limit=tl, top_k=tk)

            if g_dir is None:
                stall += 4
                f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                fs = f.sum()
                f = f / fs if fs > 1e-10 else parent.copy()
                M = float(_conv_full(f).max())
                it += 1
                continue

            dist = float(np.linalg.norm(g_dir - f))
            if dist < 1e-9:
                stall += 3
                f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
                fs = f.sum()
                f = f / fs if fs > 1e-10 else parent.copy()
                M = float(_conv_full(f).max())
                it += 1
                continue

            t_opt, m_opt = _line_search_quad(f, g_dir)

            if t_opt > 0.005 and m_opt < M - 1e-14:
                f_new = (1.0 - t_opt) * f + t_opt * g_dir
                f_new = np.clip(f_new, 0, None)
                fs = f_new.sum()
                f = f_new / fs if fs > 1e-10 else f
                M_new = float(_conv_full(f).max())
                if M_new < M:
                    M = M_new
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

            # Track best
            if it % 4 == 0:
                sc = _score_fast(f)
                if math.isfinite(sc) and sc < best_score - 1e-7:
                    best_score = sc
                    best_seq = f.copy()

            it += 1

        # End of init: update best
        sc = _score_fast(f)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f.copy()

    # Phase 1b: Another round of LP on best_seq if time permits
    if time.monotonic() < phase1_end - 20.0 and phase1_end - time.monotonic() > 25.0:
        f = best_seq.copy()
        M = float(_conv_full(f).max())
        stall = 0
        it = 0
        stop2 = min(phase1_end - 3.0, time.monotonic() + (phase1_end - time.monotonic()))
        while time.monotonic() < stop2 - 1.0 and stall < 50:
            tk = None if it < 8 else min(60, 2 * n - 1)
            g_dir = _lp_solve(f, n, M, time_limit=6.0, top_k=tk)
            if g_dir is None:
                stall += 4
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                fs = f.sum()
                f = f / fs if fs > 1e-10 else best_seq.copy()
                M = float(_conv_full(f).max())
                it += 1
                continue
            dist = float(np.linalg.norm(g_dir - f))
            if dist < 1e-9:
                stall += 2
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
                fs = f.sum()
                f = f / fs if fs > 1e-10 else best_seq.copy()
                M = float(_conv_full(f).max())
                it += 1
                continue
            t_opt, m_opt = _line_search_quad(f, g_dir)
            if t_opt > 0.003 and m_opt < M - 1e-15:
                f_new = (1.0 - t_opt) * f + t_opt * g_dir
                f_new = np.clip(f_new, 0, None)
                fs = f_new.sum()
                f = f_new / fs if fs > 1e-10 else f
                M = float(_conv_full(f).max())
                stall = 0
            else:
                stall += 1
            sc = _score_fast(f)
            if math.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = f.copy()
            it += 1

    # Phase 2: Polish - 25% of budget
    phase2_end = t0 + budget * 0.90
    if time.monotonic() < phase2_end - 8.0:
        n_passes = max(2, int((phase2_end - time.monotonic()) / 30.0))
        n_passes = min(n_passes, 5)
        pt = (phase2_end - time.monotonic() - 4.0) / n_passes
        for pp in range(n_passes):
            pst = time.monotonic() + pt
            if pst > phase2_end - 2.0:
                pst = phase2_end - 2.0
            if time.monotonic() >= pst - 1.0:
                break
            try:
                best_seq, best_score = _polish(best_seq, pst, seed=5000 + pp * 131)
            except Exception:
                pass

    # Phase 3: Final short polish
    final_stop = deadline - 4.0
    if time.monotonic() < final_stop - 8.0:
        try:
            best_seq, best_score = _polish(best_seq, final_stop, seed=99999)
        except Exception:
            pass

    # Final validation
    fin = np.clip(best_seq, 0.0, 1000.0)
    fin = np.where(np.isfinite(fin), fin, 0.0)
    try:
        fs2 = float(evaluate_sequence([float(x) for x in fin]))
    except Exception:
        fs2 = math.inf
    if not math.isfinite(fs2) or fs2 > best_score + 0.01:
        # Fallback to parent
        try:
            fs2 = float(evaluate_sequence([float(x) for x in np.clip(parent, 0, 1000)]))
        except Exception:
            fs2 = math.inf
        if math.isfinite(fs2):
            fin = np.clip(parent, 0, 1000)
        else:
            fin = np.ones(1)

    return [float(x) for x in fin]
