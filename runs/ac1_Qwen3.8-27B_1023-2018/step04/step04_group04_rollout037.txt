def smooth_max_conv(a, tau):
    """Compute smooth approximation of max auto-convolution."""
    c = fft_conv(a)  # O(n log n)
    # log-sum-exp for numerical stability
    max_c = c.max()
    log_sum_exp = max_c + np.log(np.sum(np.exp(tau * (c - max_c))))
    return (log_sum_exp) / tau

def smooth_max_conv_grad(a, tau):
    """Compute gradient of smooth max auto-convolution."""
    c = fft_conv(a)
    max_c = c.max()
    weights = np.exp(tau * (c - max_c))
    weights /= weights.sum()
    # grad of c[k] w.r.t. a[i] is a[k-i] (for valid indices)

    return grad

    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _lp_optimize(f0, n, stop_time, seed, max_iters=10000):
    """LP-based optimization with adaptive step and constraint tightening."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs

    best_f = f.copy()
    best_sc = _score(f)
    t_param = 0.3
    stall = 0
    delta = 0.0

    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break

        b = _fft_conv(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, n, M, timeout=3.0, delta=delta)
        if g is None:
            stall += 10
            if stall > 200:
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                stall = 0
                delta = 0.0
                t_param = 0.3
            continue

        gs = g.sum()
        if gs < 1e-15:
            stall += 5
            continue
        g = g / gs

        dist = np.linalg.norm(g - f)
        if dist < 1e-10:
            stall += 3
            delta = min(0.15, delta + 0.01)
            if stall > 300:
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                stall = 0
                delta = 0.0
                t_param = 0.3
            continue

        best_t = t_param
        best_sc_try = best_sc
        candidates_t = [t_param, t_param * 0.5, t_param * 1.4, 0.05, 0.15, 0.3, 0.5, 0.7, 0.9, 0.98]
        for tt in candidates_t:
            tt = min(0.99, max(0.005, tt))
            fn = np.clip((1 - tt) * f + tt * g, 0, None)
            sn = fn.sum()
            if sn < 0.01:
                continue
            fn = fn / sn
            sc_try = _score(fn)
            if sc_try < best_sc_try - 1e-15:
                best_t = tt
                best_sc_try = sc_try

        if best_sc_try < best_sc - 1e-13:
            fn = np.clip((1 - best_t) * f + best_t * g, 0, None)
            fn = fn / max(fn.sum(), 1e-10)
            f = fn
            best_f = f.copy()
            best_sc = _score(f)
            t_param = min(0.95, best_t * 1.08)
            stall = 0
            delta = max(0.0, delta * 0.85)
        else:
            t_param = max(0.03, t_param * 0.55)
            delta = min(0.15, delta + 0.008)
            stall += 1
            if stall > 80:
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                t_param = 0.3
                delta = 0.0
                stall = 0
                if time.monotonic() > stop_time:
                    break

    return best_f, best_sc


def _subgrad_optimize(f0, n, stop_time, seed):
    """Subgradient method with momentum for the max auto-convolution objective."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / max(f.sum(), 1e-10)

    best_f = f.copy()
    best_sc = _score(f)
    step = 0.08
    momentum = np.zeros(n)
    mu = 0.75
    no_imp = 0

    for it in range(200000):
        if time.monotonic() > stop_time:
            break

        c = _fft_conv(f)
        M = float(c.max())
        eps = max(1e-14, 1e-10 * M)
        active = np.flatnonzero(c >= M - eps)

        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo <= i_hi:
                idx = k - np.arange(i_lo, i_hi + 1)
                grad[i_lo:i_hi + 1] += 2.0 * f[idx]
        if len(active) > 0:
            grad /= len(active)

        S = float(f.sum())
        f_sub = 2.0 * n * (grad * S - 2.0 * M) / (S ** 3)
        f_sub -= np.mean(f_sub)
        norm_sg = np.linalg.norm(f_sub)
        if norm_sg < 1e-16:
            break

        momentum = mu * momentum - step * f_sub / norm_sg
        fn = np.clip(f + momentum, 0, None)
        s = fn.sum()
        if s < 0.01:
            step *= 0.5
            momentum *= 0.5
            continue
        fn = fn / s
        sc = _score(fn)

        if sc < best_sc - 1e-14:
            f = fn
            best_f = f.copy()
            best_sc = sc
            step = min(0.4, step * 1.03)
            no_imp = 0
        else:
            step *= 0.92
            no_imp += 1
            if no_imp > 300:
                f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                momentum = np.zeros(n)
                step = 0.08
                no_imp = 0

    return best_f, best_sc


def _coord_descent(f0, stop_time, seed, max_epochs=100):
    """Coordinate descent with adaptive perturbation."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    n = len(f)
    best_sc = _score(f)
    best_f = f.copy()

    for epoch in range(max_epochs):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            if orig < 1e-14:
                continue
            best_v = orig
            best_v_sc = best_sc
            for mult in [0.1, 0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0, 0.0]:
                f[i] = orig * mult
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_v_sc - 1e-15:
                    best_v = f[i]
                    best_v_sc = sc
            f[i] = best_v
            if best_v_sc < best_sc - 1e-15:
                best_sc = best_v_sc
                best_f = f.copy()
                improved = True

        # Pairwise moves
        n_pairs = min(300, n * 2)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-14 and oj < 1e-14:
                continue
            for mi in [0.2, 0.5, 1.5, 0.0]:
                for mj in [0.2, 0.5, 1.5, 0.0]:
                    if mi == 1.0 and mj == 1.0:
                        continue
                    f[i] = oi * mi
                    f[j] = oj * mj
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        best_f = f.copy()
                        improved = True
            f[i], f[j] = oi, oj

        if not improved:
            break

    return best_f, best_sc
