    return 2.0 * n * m / (s * s)


def _build_A_toeplitz(f, n):
    """Build constraint matrix for LP: A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(f, n, M, timeout=3.0):
    A = _build_A_toeplitz(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _subgrad_dir(f, n):
    """Compute a descent direction based on subgradient of the max-convolution."""
    b = _conv(f)
    k_max = int(np.argmax(b))
    # subgradient of c[k_max] w.r.t. f[i]
    sg = np.zeros(n)
    for i in range(n):
        v1 = f[k_max - i] if 0 <= k_max - i < n else 0.0
        v2 = f[i - k_max] if 0 <= i - k_max < n else 0.0
        sg[i] = v1 + v2
    # normalize
    if sg.max() < 1e-15:
        return None
    sg = sg / sg.max()
    return sg


def _cd_polish(f, stop_time, max_outer=40, seed=0):
    n = len(f)
    rng = np.random.default_rng(seed)
    f = np.asarray(f, dtype=np.float64).copy()
    cur = _score(f)
    best = f.copy()
    best_sc = cur

    m_small = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.0, 3.0]
    m_med = [0.2, 0.5, 0.8, 1.2, 1.5, 2.5, 0.0, 4.0]

    for outer in range(max_outer):
        if time.monotonic() > stop_time - 3.0:
            break
        improved_any = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = f[i]
            cands = {orig, 0.0}
            if orig > 1e-12:
                for m in m_small:
                    cands.add(max(0.0, orig * m))
            else:
                base = max(1e-3, 0.3 * np.mean(f[f > 0])) if np.any(f > 0) else 1e-3
                for m in [0.3, 0.7, 1.0, 2.0, 5.0]:
                    cands.add(base * m)

            best_v = orig
            best_local = cur
            for v in cands:
                f[i] = v
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_local:
                    best_local = sc
                    best_v = v
            f[i] = best_v
            if best_local < cur - 1e-15:
                cur = best_local
                improved_any = True
                if cur < best_sc:
                    best_sc = cur
                    best = f.copy()

        if not improved_any:
            # Try multi-coordinate moves
            for _ in range(300):
                if time.monotonic() > stop_time - 3.0:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                oi, oj = f[i], f[j]
                bi, bj = oi, oj
                bsc = cur
                for mi in m_med:
                    for mj in m_med:
                        if mi == 1.0 and mj == 1.0:
                            continue
                        f[i] = max(0.0, oi * mi) if oi > 1e-15 else (0.0 if mi != 0 else 0.0)
                        f[j] = max(0.0, oj * mj) if oj > 1e-15 else (0.0 if mj != 0 else 0.0)
                        if f[i] == 0.0 and oi > 1e-15:
                            f[i] = 0.0
                        if f[j] == 0.0 and oj > 1e-15:
                            f[j] = 0.0
                        if f.sum() < 0.01:
                            f[i], f[j] = oi, oj
                            continue
                        sc = _score(f)
                        if sc < bsc - 1e-15:
                            bsc = sc
                            bi, bj = f[i], f[j]
                f[i], f[j] = bi, bj
                if bsc < cur - 1e-15:
                    cur = bsc
                    improved_any = True
                    if cur < best_sc:
                        best_sc = cur
                        best = f.copy()
            if outer >= 3:
                break

    return best, best_sc


def _lp_loop(f0, n, stop_time, seed, verbose=False):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0.0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    if f.sum() < 1e-10:
        f = np.ones(n)
    f = f / f.sum()

    best_f = f.copy()
    best_score = _score(f)
    t = 0.3
    stall = 0
    max_stall = 300
    it = 0

    while time.monotonic() < stop_time - 3.0:
        b = _conv(f)
        M = float(b.max())
        if M < 1e-18:
            break

        # Try LP with slightly relaxed M for more diverse directions
        M_raises = [M, M * 1.01, M * 0.99]
        g = None
        for Mr in M_raises:
            if time.monotonic() > stop_time - 3.0:
                break
            g = _lp_dir(f, n, Mr, timeout=2.5)
            if g is not None:
                break

        if g is None:
            # Fall back to subgradient direction
            sg = _subgrad_dir(f, n)
            if sg is None:
                stall += 10
                if stall > max_stall:
                    break
                f = np.clip(f + rng.normal(0, 0.02 * max(1.0, f.std()), n), 0, None)
                if f.sum() < 1e-10:
                    f = best_f.copy()
                continue
            g = sg / sg.sum() * f.sum()

        g_sum = g.sum()
        if g_sum < 1e-15:
            stall += 3
            continue
        g = g * (f.sum() / g_sum)

        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < 1e-12:
            stall += 2
            if stall > max_stall:
                break
            continue

        improved = False
        # Adaptive step size search
        steps = [t, t * 0.7, t * 0.4, t * 1.4, t * 0.2, t * 2.0, t * 0.1]
        for ti in steps:
            f_new = np.clip((1.0 - ti) * f + ti * g, 0.0, None)
            if f_new.sum() < 1e-10:
                continue
            sc = _score(f_new)
            if sc < best_score - 1e-14:
                f = f_new
                best_f = f.copy()
                best_score = sc
                improved = True
                t = min(0.9, ti * 1.2)
                stall = 0
                break
            elif sc < _score(f) - 1e-14:
                f = f_new
                improved = True
                t = min(0.9, ti)
                stall = 0
                break
            else:
                t = max(0.005, t * 0.5)

        if not improved:
            stall += 1
            if stall >= max_stall:
                # Restart with perturbation
                noise_scale = 0.05 * max(1.0, np.mean(best_f[best_f > 0]))
                f = np.clip(best_f + rng.normal(0, noise_scale, n), 0, None)
                if f.sum() < 1e-10:
                    f = best_f.copy()
                f = f / f.sum()
                t = 0.3
                stall = 0
                max_stall = min(max_stall + 100, 600)

        it += 1
        # Periodic CD polish
        if it % 50 == 0 and time.monotonic() < stop_time - 30.0:
            f_c, sc_c = _cd_polish(f, min(stop_time - 25.0, time.monotonic() + 20.0),
                                   max_outer=4, seed=seed + it)
            if sc_c < best_score:
                best_score = sc_c
                best_f = f_c.copy()
                f = f_c.copy()

    # Final CD
    if time.monotonic() < stop_time - 5.0:
        f_c, sc_c = _cd_polish(best_f, stop_time - 2.0, max_outer=15, seed=seed + 9999)
        if sc_c <= best_score + 1e-14:
            best_f = f_c
            best_score = sc_c

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(400, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(400, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    n_parent = len(parent)

    # Fallback
    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(200)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _check_and_update(seq, sc):
        nonlocal best_seq, best_score
        if seq is None or len(seq) == 0:
            return
        seq = np.asarray(seq, dtype=np.float64)
        if not np.all(np.isfinite(seq)):
            return
        seq = np.clip(seq, 0.0, 1000.0)
        if seq.sum() < 0.01:
            return
        if sc < best_score - 1e-14:
            try:
                exact = float(evaluate_sequence([float(x) for x in seq]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-14:
                best_seq = seq.copy()
                best_score = exact

    # Phase 1: LP optimization on parent length (~55% time)
    n = n_parent
    phase1_end = t0 + budget * 0.55
    rng_main = np.random.default_rng(seed)

    i_arr = np.arange(n, dtype=np.float64)
    inits = [parent.copy()]

    # Raised cosine variants
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / n)))
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * (i_arr + n * 0.15) / n)))
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * (i_arr + n * 0.3) / n)))

    # Parabolic
    ctr = (n - 1) / 2.0
    inits.append(np.maximum(0.0, 1.0 - ((i_arr - ctr) / (0.6 * n)) ** 2))

    # Edge-weighted (anti-palindromic tendency)
    inits.append(np.concatenate([np.linspace(1, 0.1, n // 2), np.linspace(0.1, 1, n - n // 2)]))

    # Two-lobe
    inits.append(np.exp(-0.5 * ((i_arr - 0.25 * n) / (0.12 * n)) ** 2) +
                 np.exp(-0.5 * ((i_arr - 0.75 * n) / (0.12 * n)) ** 2))

    # Parent with noise
    sig = 0.15 * max(1.0, np.mean(parent))
    inits.append(np.abs(parent + rng_main.normal(0, sig, n)) + 1e-4)

    # Alternating modulated
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.3)
    mod = 0.7 + 0.3 * np.cos(2.0 * np.pi * i_arr / (n // 8))
    inits.append(alt * mod)

    # Deduplicate
    seen = set()
    inits_dedup = []
    for ix in inits:
        if ix.sum() < 1e-10:
            continue
        key = tuple(np.round(ix / max(ix.max(), 1e-10), 3))
        if key not in seen:
            seen.add(key)
            inits_dedup.append(ix)

    time_per_init = max(10.0, (phase1_end - t0) / max(1, len(inits_dedup)))
    for idx, init in enumerate(inits_dedup):
        if time.monotonic() > phase1_end - 10.0:
            break
        if best_score < 1.45:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        try:
            res, sc = _lp_loop(init, n, stop, seed + 1000 * (idx + 1))
            _check_and_update(res, sc)
        except Exception:
            pass

    # Phase 2: Continued LP on best found (~12% time)
    phase2_end = t0 + budget * 0.68
    if time.monotonic() < phase2_end - 20.0:
        try:
            res, sc = _lp_loop(best_seq, len(best_seq), phase2_end - 3.0, seed + 77777)
            _check_and_update(res, sc)
        except Exception:
            pass

    # Phase 3: CD polish (~13% time)
    phase3_end = t0 + budget * 0.81
    if time.monotonic() < phase3_end - 5.0:
        try:
            f_c, sc_c = _cd_polish(best_seq, phase3_end - 2.0, max_outer=60, seed=seed * 3 + 1)
            _check_and_update(f_c, sc_c)
        except Exception:
            pass

    # Phase 4: Try nearby lengths (~9% time)
    phase4_end = t0 + budget * 0.90
    if time.monotonic() < phase4_end - 15.0:
        for L in [n - 8, n + 8, n - 16, n + 16]:
            if L < 64 or time.monotonic() > phase4_end - 12.0:
                continue
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
            stop_l = min(phase4_end - 4.0, time.monotonic() + max(8.0, (phase4_end - time.monotonic()) * 0.35))
            try:
                res, sc = _lp_loop(init_l, L, stop_l, seed + 90000 + L)
                _check_and_update(res, sc)
            except Exception:
                pass

    # Phase 5: Final extended CD with remaining time
    final_end = deadline - 8.0
    if time.monotonic() < final_end - 10.0:
        try:
            f_c, sc_c = _cd_polish(best_seq, final_end, max_outer=100, seed=seed * 7 + 42)
            _check_and_update(f_c, sc_c)
        except Exception:
            pass

    # One more LP burst if time allows
    if time.monotonic() < deadline - 20.0:
        try:
            res, sc = _lp_loop(best_seq, len(best_seq), deadline - 10.0, seed + 55555)
            _check_and_update(res, sc)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0, 1.0, 1.0]
        best_seq = fallback

    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    if not result:
        result = [1.0]
    if any(math.isnan(x) or math.isinf(x) for x in result):
        result = [float(x) for x in fallback]

    return result
