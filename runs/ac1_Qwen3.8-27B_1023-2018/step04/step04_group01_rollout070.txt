def _project_simplex(v):
    v=np.asarray(v,float).ravel()
    n=v.size
    v=np.where(np.isfinite(v),v,0.0)
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    rho=np.nonzero(u*np.arange(1,n+1) > (cssv - 1.0))[0]
    if rho.size==0:
       return np.ones(n)/n
    idx=rho[-1]
    theta=(cssv[idx]-1.0)/(idx+1.0)
    w=np.maximum(v-theta,0.0)
    s=w.sum()
    if s<1e-12: return np.ones(n)/n
    return w/s

def _conv_full(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n <= 512:
        c = np.convolve(a, a, mode="full")
    else:
        L = next_fast_len(2 * n - 1)
        F = rfft(a, L)
        c = irfft(F * F, L)[: 2 * n - 1]
    c = np.asarray(c, dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    np.maximum(c, 0.0, out=c)

def _norm1(f):
    f = np.asarray(f, dtype=np.float64).ravel()
    f = np.where(np.isfinite(f), f, 0.0)

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if rho.size == 0:
        return np.ones(n, dtype=np.float64) / n
    idx = int(rho[-1])
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s < 1e-12:
        return np.ones(n, dtype=np.float64) / n
    return w / s


def _conv_gradient(f, p):
    f = np.asarray(f, dtype=np.float64).ravel()
    p = np.asarray(p, dtype=np.float64).ravel()
    n = f.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)

    fr = f[::-1]
    y = irfft(rfft(p, L) * rfft(fr, L), L)[: 3 * n - 2]
    g = 2.0 * y[n - 1 : 2 * n - 1]

def _lp_direction(f, n, K=80, slack=0.25, time_limit=4.0, gcap=100.0):

    b = M * (1.0 + float(slack)) - float(slack) * c[top]

            best_sc = sc
            best_t = float(t)

    if best_t <= 1e-7 or not math.isfinite(best_sc) or best_sc >= float(current_score) - 1e-14:
        return f.copy(), float(current_score), False

    h = f + best_t * (g - f)
    h = np.clip(np.where(np.isfinite(h), h, 0.0), 0.0, None)
    h = _norm1(h)
    return h, float(best_sc), True


def _lp_phase(f, n, stop_time, seed, K=None, max_blank=35):
    f = _norm1(f)
    best = f.copy()
    best_sc = _score_internal(f)
    local = f.copy()
    local_sc = best_sc
    rng = np.random.default_rng(seed)
    if K is None:
        if n <= 512:
            K = 80
        elif n <= 800:
            K = 120
        else:
            K = 160
    slacks = [0.0, 0.5, 1.0, 0.5, 0.75]
    blank = 0
    it = 0

    while time.monotonic() < stop_time - 6:
        it += 1
        remaining = stop_time - time.monotonic()
        if remaining < 15:
            break
        slack = slacks[it % len(slacks)]
        tl = min(8.0, max(1.0, remaining * 0.18))
        g = _lp_direction(local, n, K=K, slack=slack, time_limit=tl)

        if g is None:
            blank += 3
        else:
            if np.linalg.norm(g - local, ord=1) < 1e-6:
                blank += 2
            else:
                h, sc, ok = _blend_accept(local, g, local_sc)
                if ok:
                    local = h
                    local_sc = sc
                    blank = 0
                    if local_sc < best_sc - 1e-13:
                        best_sc = local_sc
                        best = local.copy()
                else:
                    blank += 1

        if blank > 5:
            scale = rng.choice([0.008, 0.02, 0.04, 0.07])
            cand = np.clip(local + rng.normal(0.0, float(scale), n), 0.0, None)
            cand = _norm1(cand)
            sc = _score_internal(cand)
            if math.isfinite(sc):
                local = cand
                local_sc = sc
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best = cand.copy()
            blank = 0

        if blank > max_blank:
            break

    if local_sc < best_sc:
        best = local.copy()
        best_sc = local_sc
    return best, best_sc


def _grad_polish(f, stop_time, seed=0, max_iters=5000):
    f = _norm1(f)
    n = f.size
    best = f.copy()
    best_sc = _score_internal(f)
    local = f.copy()
    local_sc = best_sc
    rng = np.random.default_rng(seed)
    alpha = float(rng.uniform(0.01, 0.04))
    c0 = _conv_full(local)
    tau = max(1e-12, 0.05 * float(c0.max()))
    stall = 0
    no_grad = 0

    for _ in range(max_iters):
        if time.monotonic() > stop_time - 4:
            break

        c = _conv_full(local)
        M = float(c.max())
        if not math.isfinite(M) or M < 1e-14:
            break

        w = (c - M) / tau
        w -= float(w.max())
        expw = np.exp(np.clip(w, -50.0, 50.0))
        ps = float(expw.sum())
        if ps < 1e-30:
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
        else:
            p = expw / ps

        grad = 2.0 * n * tau * _conv_gradient(local, p)
        if not np.all(np.isfinite(grad)) or float(np.sum(np.abs(grad))) < 1e-18:
            no_grad += 1
            if no_grad > 20:
                break
            tau = max(1e-12, tau * 0.7)
            continue
        no_grad = 0

        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1):
            h = _project_simplex(local - alpha * float(mult) * grad)
            sc = _score_internal(h)
            if math.isfinite(sc) and sc < local_sc - 1e-13:
                local = h
                local_sc = sc
                accepted = True
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best = h.copy()
                break

        if accepted:
            alpha = min(0.25, max(alpha * 1.03, 1e-9))
            tau = max(1e-12, tau * 0.99)
            stall = 0
        else:
            alpha = max(1e-10, alpha * 0.5)
            tau = max(1e-13, tau * 0.75)
            stall += 1
            if stall > 25:
                local = _norm1(np.clip(best + rng.normal(0.0, 0.02, n), 0.0, None))
                local_sc = _score_internal(local)
                alpha = 0.02
                cnow = _conv_full(local)
                tau = max(1e-12, 0.05 * float(cnow.max()))
                stall = 0

    if local_sc < best_sc:
        best = local.copy()
        best_sc = local_sc
    return best, best_sc


def _coord_polish(f, stop_time, max_sweeps=10000, seed=42, grid_size=180):
    f = np.asarray(f, dtype=np.float64).ravel()
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.clip(f, 0.0, None)
    n = f.size
    if n == 0:
        return np.ones(1, dtype=np.float64), math.inf

    S = float(f.sum())
    if S < 1e-8:
        f = np.ones(n, dtype=np.float64)
        S = 1.0
    else:
        f = f / S
        S = 1.0

    c = _conv_full(f)
    curr_score = float(2.0 * n * c.max())
    best_score = curr_score
    best_f = f.copy()
    curr_f = f.copy()
    rng = np.random.default_rng(seed)
    conv_len = 2 * n - 1

    if n <= 600:
        pts = int(max(80, grid_size))
    elif n <= 800:
        pts = int(max(70, min(120, grid_size)))
    else:
        pts = int(max(50, min(90, grid_size)))

    sweep = 0
    no_improve = 0
    restarts = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 3:
        sweep += 1
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3:
                break

            v = float(f[i])
            shifted = np.zeros(conv_len, dtype=np.float64)
            end = min(conv_len, i + n)
            if end > i:
                shifted[i:end] = f[: end - i]

            x_max = max(5.0 * v, 0.5 * S, 0.2)
            if v > x_max:
                x_max = max(v * 1.2, v + 1e-4)
            x_max = float(np.clip(x_max, 0.0, 500.0 * S + 1.0))
            if x_max <= 0.0:
                continue

            xg = np.linspace(0.0, x_max, pts)
            d = xg - v
            c_new = c[None, :] + 2.0 * d[:, None] * shifted[None, :]
            if 0 <= 2 * i <= conv_len - 1:
                c_new[:, 2 * i] += d * d

            cmax_new = c_new.max(axis=1)
            Snew = S + d
            valid = Snew > 1e-8
            if not valid.any():
                continue

            obj = np.where(valid, (2.0 * n * cmax_new) / (Snew * Snew), math.inf)
            idx = int(np.argmin(obj))
            if not math.isfinite(obj[idx]):
                continue

            if obj[idx] < curr_score - 1e-14:
                dw = x_max / max(1, pts)
                lo = max(0.0, float(xg[idx]) - 3.0 * dw)
                hi = min(x_max, float(xg[idx]) + 3.0 * dw)
                xf = np.linspace(lo, hi, max(10, pts // 2))
                df = xf - v
                cf = c[None, :] + 2.0 * df[:, None] * shifted[None, :]
                if 0 <= 2 * i <= conv_len - 1:
                    cf[:, 2 * i] += df * df
                cmf = cf.max(axis=1)
                Snf = S + df
                vf = Snf > 1e-8
                of = np.where(vf, (2.0 * n * cmf) / (Snf * Snf), math.inf)
                idxf = int(np.argmin(of))
                if math.isfinite(of[idxf]) and of[idxf] < curr_score - 1e-14:
                    dval = float(df[idxf])
                    f[i] = v + dval
                    c += 2.0 * dval * shifted
                    if 0 <= 2 * i <= conv_len - 1:
                        c[2 * i] += dval * dval
                    S += dval

                    if not math.isfinite(S) or S <= 1e-8:
                        f = np.ones(n, dtype=np.float64)
                        S = 1.0
                        c = _conv_full(f)
                        curr_score = float(2.0 * n * c.max())
                        if curr_score < best_score:
                            best_score = curr_score
                            best_f = f.copy()
                            curr_f = f.copy()
                        break

                    if S < 0.1 or S > 10.0:
                        a = 1.0 / S
                        f *= a
                        c *= a * a
                        S = 1.0

                    if not np.all(np.isfinite(c)):
                        f = _norm1(f)
                        S = float(f.sum())
                        c = _conv_full(f)

                    sc = float(2.0 * n * c.max() / (S * S))
                    if math.isfinite(sc):
                        curr_score = sc
                        if sc < best_score - 1e-14:
                            best_score = sc
                            best_f = f.copy()
                    improved = True

        if sweep % 8 == 0:
            S = float(f.sum())
            if S > 1e-8:
                c = _conv_full(f)
                sc = float(2.0 * n * c.max() / (S * S))
                if math.isfinite(sc):
                    curr_score = sc
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_f = f.copy()
            else:
                break

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                no_improve = 0
                if restarts < 5:
                    restarts += 1
                    scale = 0.004 * (1.0 + 0.35 * restarts)
                    f = _norm1(np.clip(best_f + rng.normal(0.0, float(scale), n), 0.0, None))
                    S = 1.0
                    c = _conv_full(f)
                    curr_score = float(2.0 * n * c.max())
                    curr_f = f.copy()
                else:
                    break
            else:
                if restarts < 5:
                    restarts = 0
        else:
            if no_improve > 0 and improved:
                no_improve = max(0, no_improve - 1)

    final_f = np.clip(np.where(np.isfinite(best_f), best_f, 0.0), 0.0, None)
    final_sc = _score_internal(final_f)
    if not math.isfinite(final_sc):
        final_f = _norm1(parent) if 'parent' in dir() else np.ones(64, dtype=np.float64)
        final_sc = _score_internal(final_f)
    return final_f, final_sc


def _refine(f0, n, stop_time, seed, do_lp=True, do_grad=True, grid=180, lp_frac=0.35):
    f = _resize(f0, n)
    best = f.copy()
    best_sc = _score_internal(f)
    now = time.monotonic()
    remaining = stop_time - now
    if remaining < 8:
        return best, best_sc

    if do_lp and remaining > 70:
        t_lp = min(stop_time - 80, now + max(25.0, remaining * float(lp_frac)))
        if time.monotonic() < t_lp - 12:
            try:
                res, sc = _lp_phase(f, n, t_lp, seed)
            except Exception:
                res, sc = f, best_sc
            if res is not None:
                res = _norm1(res)
                sc = _score_internal(res)
                if math.isfinite(sc) and sc < best_sc - 1e-14:
                    best = res.copy()
                    best_sc = sc
            f = best.copy()

    if do_grad and time.monotonic() < stop_time - 45:
        t_grad = min(stop_time - 40, time.monotonic() + max(15.0, (stop_time - time.monotonic()) * 0.18))
        if time.monotonic() < t_grad - 12:
            try:
                res, sc = _grad_polish(f, t_grad, seed + 7)
            except Exception:
                res, sc = f, best_sc
            if res is not None:
                res = _norm1(res)
                sc = _score_internal(res)
                if math.isfinite(sc) and sc < best_sc - 1e-14:
                    best = res.copy()
                    best_sc = sc
            f = best.copy()

    if time.monotonic() < stop_time - 6:
        try:
            res, sc = _coord_polish(f, stop_time - 2, grid_size=grid, seed=seed + 23)
        except Exception:
            res, sc = best, best_sc
        if res is not None:
            res = _norm1(res)
            sc = _score_internal(res)
            if math.isfinite(sc) and sc < best_sc - 1e-14:
                best = res.copy()
                best_sc = sc

    return best, best_sc


def _make_alt_inits(n, current_best, seed, limit=4):
    inits = []
    rng = np.random.default_rng(seed)
    if n <= 1:
        inits.append(np.ones(n, dtype=np.float64) / max(1, n))
        return inits

    t = np.linspace(0.0, 1.0, n)

    def add(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size != n:
            a = _resize(a, n)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, None)
        s = float(a.sum())
        if s > 1e-10:
            a = a / s
            inits.append(a)

    p = _resize(current_best, n)
    add(p * (1.0 + rng.normal(0.0, 0.05, n)))
    add(np.exp(-1.25 * t) + np.exp(-1.25 * (1.0 - t)))
    add(np.exp(-2.8 * t) + np.exp(-2.8 * (1.0 - t)))
    add(np.exp(-1.8 * t))
    add(np.exp(-1.8 * (1.0 - t)))
    add(np.clip(1.0 + 0.8 * np.cos(2.4 * np.pi * t + rng.uniform(0.0, 6.28)), 0.0, None))
    add(rng.dirichlet(np.full(n, 0.15)))
    add(rng.dirichlet(np.full(n, 0.04)))

    seen = []
    out = []
    for a in inits:
        key = tuple(np.round(a, 10))
        if key not in seen:
            seen.append(key)
            out.append(a)
        if len(out) >= limit:
            break
    return out


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(1.0, float(budget_s))
    hard_deadline = t0 + budget - max(3.0, budget * 0.02)

    parent = _get_parent()
    parent = _norm1(parent)
    n_parent = parent.size

    best_seq = parent.copy()
    best_exact = math.inf
    try:
        best_exact = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        best_exact = math.inf

    if not math.isfinite(best_exact) or best_seq.size == 0 or float(best_seq.sum()) < 0.01:
        best_seq = np.ones(64, dtype=np.float64) / 64.0
        try:
            best_exact = float(evaluate_sequence([float(x) for x in best_seq]))
        except Exception:
            best_exact = math.inf

    target = 1.50274349

    def update_best(seq):
        nonlocal best_seq, best_exact
        arr = np.asarray(seq, dtype=np.float64).ravel()
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size == 0 or float(arr.sum()) < 0.01:
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in arr]))
        except Exception:
            sc = math.inf
        if math.isfinite(sc) and sc < best_exact:
            best_exact = sc
            best_seq = arr.copy()

    # Primary optimization at the provided construction length.
    if budget > 30:
        t_main = min(hard_deadline - 90, t0 + max(60.0, budget * 0.46))
        if time.monotonic() < t_main - 25:
            try:
                res, _ = _refine(parent, n_parent, t_main, seed, do_lp=True, do_grad=True, grid=210, lp_frac=0.38)
                update_best(res)
            except Exception:
                pass

    # Alternative initializations at the same length.
    if budget > 120:
        t_alt = min(hard_deadline - 100, t0 + max(30.0, budget * 0.62))
        if time.monotonic() < t_alt - 45 and best_exact > target + 1e-8:
            alts = _make_alt_inits(n_parent, best_seq, seed + 101, limit=3)
            for idx, alt in enumerate(alts):
                if time.monotonic() > t_alt - 35:
                    break
                if best_exact <= target + 1e-8:
                    break
                remaining_slice = t_alt - time.monotonic()
                alloc = max(35.0, min(120.0, remaining_slice / max(1, len(alts) - idx)))
                stop = min(t_alt - 25 - idx * 2, time.monotonic() + alloc)
                if stop <= time.monotonic() + 25:
                    continue
                try:
                    res, _ = _refine(alt, n_parent, stop, seed + 1000 + idx, do_lp=True, do_grad=False, grid=170, lp_frac=0.28)
                    update_best(res)
                except Exception:
                    pass

    # Explore nearby and slightly larger lengths. Longer sequences can lower the
    # asymptotic bound because more convolution lags share the total energy.
    if budget > 220:
        t_len = min(hard_deadline - 70, t0 + max(60.0, budget * 0.82))
        if time.monotonic() < t_len - 60 and best_exact > target + 1e-8:
            cand_lengths = []
            for L in [n_parent + 32, n_parent + 64, n_parent + 96, n_parent + 128,
                      n_parent + 160, n_parent + 192, n_parent + 256,
                      max(64, n_parent // 2)]:
                if 64 <= int(L) <= 1024 and int(L) not in cand_lengths:
                    cand_lengths.append(int(L))
            for L in [448, 512, 576, 640, 704, 768, 832, 896, 960, 1024]:
                if L not in cand_lengths:
                    cand_lengths.append(L)

            preferred = [512, 640, 768, 576, 704, 896, 448, 576, 832, 960]
            ordered = []
            for L in preferred:
                if L in cand_lengths and L not in ordered:
                    ordered.append(L)
            for L in sorted(cand_lengths):
                if L not in ordered:
                    ordered.append(L)

            tried = 0
            for L in ordered:
                if time.monotonic() > t_len - 35:
                    break
                if best_exact <= target + 1e-8:
                    break
                if tried >= 4:
                    break
                tried += 1
                remaining_len = t_len - time.monotonic()
                alloc = max(30.0, min(120.0, remaining_len / max(1, 4 - tried + 1)))
                stop = min(t_len - 25, time.monotonic() + alloc)
                if stop <= time.monotonic() + 20:
                    continue
                init = _resize(best_seq, L)
                try:
                    grid = 170 if L <= 640 else (130 if L <= 800 else 100)
                    res, _ = _refine(
                        init,
                        L,
                        stop,
                        seed + 50000 + L,
                        do_lp=(stop - time.monotonic() > 55),
                        do_grad=False,
                        grid=grid,
                        lp_frac=0.30,
                    )
                    update_best(res)
                except Exception:
                    pass

    # Final high-resolution coordinate polish on the globally best candidate.
    if time.monotonic() < hard_deadline - 12:
        try:
            res, _ = _coord_polish(
                best_seq,
                hard_deadline - 8,
                max_sweeps=50000,
                seed=31337,
                grid_size=230 if best_seq.size <= 600 else (170 if best_seq.size <= 800 else 120),
            )
            update_best(res)
        except Exception:
            pass

    # Very short final fallback polish if any time remains.
    if time.monotonic() < hard_deadline - 4:
        try:
            res, _ = _coord_polish(
                best_seq,
                hard_deadline - 2,
                max_sweeps=200,
                seed=99999,
                grid_size=140,
            )
            update_best(res)
        except Exception:
            pass

    best_seq = np.asarray(best_seq, dtype=np.float64).ravel()
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.size == 0 or float(best_seq.sum()) < 0.01:
        best_seq = np.ones(64, dtype=np.float64) / 64.0

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        fallback = np.ones(64, dtype=np.float64) / 64.0
        try:
            fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            fallback_sc = math.inf
        if math.isfinite(fallback_sc):
            return [float(x) for x in fallback]
        return [1.0]

    return [float(x) for x in best_seq]
