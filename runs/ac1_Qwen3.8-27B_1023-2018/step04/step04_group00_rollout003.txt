Actually, I realize I should be more careful about the coordinate descent. The key insight is:

When we change p[i] from v to v+δ, the convolution changes as:
c_new[k] = c_old[k] + δ*(p[k-i] + p[i]) + δ^2 (if k-i=i, i.e., k=2i)
= c_old[k] + δ*(2p[k-i]) + δ^2 (if k=2i)

Wait, more carefully:
c[k] = sum_j p[j] p[k-j]
If p[i] changes by δ, then c[k] changes by:
δ * p[k-i] (from term j=i) + δ * p[k-i] (from term k-j=i, i.e., j=k-i)
= 2δ * p[k-i] (if 0 <= k-i < n and k-i ≠ i)
= 2δ * p[i] (if k-i = i, i.e., k = 2i)
= 0 (if k-i is out of range)

Wait, if k-i = i (i.e., k = 2i):
c[2i] includes the term p[i]*p[i], and its derivative w.r.t. p[i] is 2*p[i].
So c_new[2i] = c_old[2i] + 2δ*p[i] + δ^2.

For k ≠ 2i: c_new[k] = c_old[k] + 2δ*p[k-i] (if 0 <= k-i < n).

So the change in the max is:
max_k (c_old[k] + 2δ*p[k-i] + [δ^2 if k=2i])

This is a piecewise linear (or piecewise quadratic at k=2i) function of δ. The minimum can be found by checking the breakpoints where the identity of the argmax changes.

For a one-dimensional search, the function is:
f(δ) = max_k (c[k] + 2δ*p[k-i] * I(0<=k-i<n) + δ^2 * I(k=2i))

This is the max of a set of linear (plus one quadratic) functions, so it's convex in δ. The minimum is where the subgradient contains 0, or at a boundary.

For the coordinate descent, we want to minimize:
2n * max_k (c[k] + Δ_k(δ)) / (S + δ)^2

where S = sum p, Δ_k(δ) = 2δ*p[k-i]*I + δ^2*I(k=2i), and the max is over k.

This is more complex but still 1D. The previous code uses a grid search which is O(G * n) per coordinate. For n=3072 and G=20, that's about 60K operations per coordinate, and 3072 coordinates, so ~180M per sweep. That's a lot.

Optimization: Instead of evaluating the max over all 2n-1 points for each δ, we can precompute:
- For each k in the range [i, i+n-1] (the affected lags), the value c[k] and the slope 2*p[k-i].
- The unaffected lags have a constant max = max(c[0:i]) ∪ max(c[i+n:]).

So the affected lags are k ∈ {i, i+1, ..., i+n-1} ∩ {0, ..., 2n-2}.
The slope for lag k is 2*p[k-i] (with special case for k=2i adding δ^2).

For the 1D optimization, we need to find the minimum of:
g(δ) = max(max_unaffected, max_{k affected} (c[k] + 2δ*p[k-i] + δ^2*I(k=2i))) / (S+δ)^2

The numerator is convex in δ (max of linear/quadratic functions). The denominator (S+δ)^2 is increasing for δ > -S. So the ratio is not necessarily convex, but for small |δ| it's well-behaved.

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)

    return p


def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(c, 0.0)


def _cross_corr(p, w):
    """Cross-correlation: result[i] = sum_k w[k] * p[k-i] for valid indices."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    L = 3 * n - 2
    size = next_fast_len(L)
    F = rfft(p[::-1], size)
    G = rfft(w, size)
    c = irfft(F * G, size)[:L]
    q = c[n - 1:2 * n - 1]
    return np.maximum(q, 0.0)


def _softmax(u):
    u = np.asarray(u, dtype=np.float64).ravel()
    if u.size == 0:
        return u
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    return e / s


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * a.size * m / (s * s))


def _lse_lbf(p, stop_time, consider=None):
    """L-BFGS with log-sum-exp smoothing, multi-stage."""
    n = int(p.size)
    if n == 0:
        return p
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)

    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return np.maximum(c, 0.0)

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        return np.maximum(c[n - 1:2 * n - 1], 0.0)

    def obj_grad(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros(n)
        scaled = (c - cmax) / T
        max_s = float(np.max(scaled))
        if max_s > 500:
            scaled = scaled - max_s + 500
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * math.log(sum_exp / float(2 * n - 1))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    stages = (
        (0.15, 0.08),
        (0.05, 0.12),
        (0.015, 0.18),
        (0.004, 0.20),
        (0.001, 0.22),
        (0.0003, 0.15),
        (0.00008, 0.05),
    )

        return p

    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 10.0:
            break
        cum += wt
        end_stage = min(stop_time - 8.0, start + avail * (cum / total_w) + 15.0)
        if end_stage <= now + 5.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        T = max(1e-30, frac * cmax)

        stale = 0
        for _ in range(8):
            if time.monotonic() > end_stage - 2.0:
                break
            maxit = 30 if n <= 2000 else 20
            opts = {
                "maxiter": maxit,
                "gtol": 1e-12,
                "ftol": 1e-15,
                "maxls": 25,
                "maxcor": 25,
                "maxdisp": 0,
            }
            try:
                res = minimize(obj_grad, u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < 1e90:
                    p_new = _softmax(res.x)
                    p_new = np.maximum(p_new, 0.0)
                    ss = float(p_new.sum())
                    if ss > 1e-300:
                        p_new /= ss
                    sc_new = _score_fft(p_new)
                    if math.isfinite(sc_new) and sc_new < best_sc:
                        best_sc = sc_new
                        best_p_out = p_new.copy()
                        p = p_new.copy()
                        u = p_to_u(p)
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 2:
                            break
            else:
                break
            if getattr(res, "status", 1) == 0:
                break

        if consider is not None:

            except Exception:
                pass

    return best_p_out


def _coord_descent(p, stop_time, seed=12345, consider=None, G=None):

    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s
    S = 1.0

    if G is None:
        G = 30 if n <= 2000 else 22 if n <= 4000 else 16
    G = int(max(12, G))

    c = _conv_self(p)
    Ltot = 2 * n - 1
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    sweep = 0
    flat_count = 0
    max_flat = 5

    while sweep < 50000 and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S > 1.3 or S < 0.7:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0

            v = float(p[i])
            start_k = i
            end_k = min(i + n, Ltot)
            Ls = end_k - start_k
            if Ls <= 0:

            infl = 2.0 * p[:Ls]
            diag_ii = 2 * i

            max_const = 0.0
            if start_k > 0:
                mc = float(c[:start_k].max())
                if mc > max_const:
                    max_const = mc
            if end_k < Ltot:
                mc = float(c[end_k:].max())
                if mc > max_const:
                    max_const = mc

            x_hi = max(25.0 * v, 80.0 / n, 0.05)
            x_hi = min(x_hi, 12.0 / n * 10)
            if not math.isfinite(x_hi) or x_hi <= 0:

            pts = np.linspace(0.0, x_hi, G + 1)

            denom = S + w
            valid = denom > 1e-14

            aff = np.maximum(aff, max_const)
            maff = np.max(aff, axis=1)

            vals = np.full(G + 2, math.inf)
            dv = denom[valid]
            vw = maff[valid]

            best_v = float(vals[imin])

            # Fine search
            if G > 2:

                lo = max(0.0, best_x - 2.5 * dw)

                    hi = lo + 1e-12

                denomf = S + wf

                    afff = np.maximum(afff, max_const)
                    mafff = np.max(afff, axis=1)
                    vf = denomf[validf]
                    vv = mafff[validf]
                    valsf = np.full(fine.shape, math.inf)

                    im2 = int(np.argmin(valsf))
                    if valsf[im2] < best_v:

            if math.isfinite(best_v) and best_v < cur_score - 1e-13:

                s1 = float(p1.sum())
                if s1 > 1e-300:
                    p = p1 / s1
            else:
                p = p.copy()
        except Exception:
            pass

    # Phase 2: Coordinate descent
    phase2_end = t0 + 0.85 * (hard - t0)
    if phase2_end > time.monotonic() + 30.0:
        try:
            p2, _ = _coord_descent(p, phase2_end, seed=int(seed + 777),
                                   consider=consider)
            if p2 is not None and p2.size == n and np.all(np.isfinite(p2)):
                s2 = float(p2.sum())
                if s2 > 1e-300:
                    p = p2 / s2
                    consider(p)
        except Exception:
            pass

    # Phase 3: Final LSE refinement
    if time.monotonic() < hard - 40.0:
        p3_end = min(hard - 25.0, time.monotonic() + 0.4 * (hard - time.monotonic()))
        try:
            p3 = _lse_lbf(p, p3_end, consider=consider)
            if p3 is not None and p3.size == n and np.all(np.isfinite(p3)):
                s3 = float(p3.sum())
                if s3 > 1e-300:
                    p = p3 / s3
        except Exception:
            pass

    # Phase 4: Second coordinate descent
    if time.monotonic() < hard - 30.0:
        p4_end = min(hard - 20.0, time.monotonic() + 0.5 * (hard - time.monotonic()))
        try:
            p4, _ = _coord_descent(p, p4_end, seed=int(seed + 9999),
                                   consider=consider, G=26)
            if p4 is not None and p4.size == n and np.all(np.isfinite(p4)):
                s4 = float(p4.sum())
                if s4 > 1e-300:
                    p = p4 / s4
                    consider(p)
        except Exception:
            pass

    # Phase 5: Active set polish
    if time.monotonic() < hard - 15.0:
        as_end = hard - 12.0
        try:
            p5 = _active_set_polish(p, as_end, consider=consider)
            if p5 is not None and p5.size == n and np.all(np.isfinite(p5)):
                s5 = float(p5.sum())
                if s5 > 1e-300:
                    p = p5 / s5
                    consider(p)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = evaluate_sequence(ls)
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
