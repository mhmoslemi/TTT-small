import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(416, dtype=np.float64)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _score_from_parts(b, n, S):
    if S < 0.01:
        return math.inf
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (S * S)


def _build_A(f, n):
    """Sparse (2n-1 x n) matrix: (A*g)[k] = (f*g)[k]."""
    rows_list, cols_list, vals_list = [], [], []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n, k + 1)
        for i in range(i_lo, i_hi):
            v = f[k - i]
            if v > 1e-18:
                rows_list.append(k)
                cols_list.append(i)
                vals_list.append(v)
    if not rows_list:
        return csr_matrix((2 * n - 1, n))
    return csr_matrix((vals_list, (rows_list, cols_list)), shape=(2 * n - 1, n))


def _lp_dir(f, n, M, timeout=8.0):
    """LP: max sum(g) s.t. (f*g)_k <= M, g>=0. Returns g or None."""
    A = _build_A(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
    if g.sum() <= f.sum() * (1.0 + 1e-9):
        return None
    return g


def _exp_subgrad_step(f, b, n, lr=0.4):
    """Multiplicative weight subgradient step on peak lags."""
    M = float(b.max())
    if M < 1e-18:
        return f.copy()
    thresh = M * 0.99
    peak_idx = np.where(b >= thresh)[0]
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(b)])
    grad = np.zeros(n)
    for k in peak_idx:
        i_lo = max(0, k - n + 1)
        i_hi = min(n, k + 1)
        for i in range(i_lo, i_hi):
            grad[i] += f[k - i]
    grad /= len(peak_idx)
    gmax = grad.max()
    if gmax < 1e-18:
        return f.copy()
    lr_eff = lr / gmax
    exp_arg = -lr_eff * grad
    exp_arg -= exp_arg.max()
    w = np.exp(np.clip(exp_arg, -50, 50))
    f_new = f * w
    s = f_new.sum()
    if s < 0.01:
        return f.copy()
    return f_new / s


def _lp_optimize(f0, stop_time, seed):
    """LP-based iterative optimization. Returns (best_f, best_score)."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.maximum(np.asarray(f0, dtype=np.float64).ravel().copy(), 0.0)
    s0 = f.sum()
    if s0 < 0.01:
        f[:] = 1.0
    f = f / f.sum()  # normalize to unit sum

    b = _conv_full(f)
    best_f = f.copy()
    best_score = _score_from_parts(b, n, 1.0)
    stall = 0
    max_stall = 300

    t0 = time.monotonic()
    while time.monotonic() < stop_time - 2.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_dir(f, n, M, timeout=6.0)
        if g is None:
            stall += 1
            if stall > max_stall:
                # Subgradient kick
                f = _exp_subgrad_step(f, b, n, lr=0.5)
                s = f.sum()
                if s < 0.01:
                    f = best_f.copy()
                else:
                    f = f / s
                stall = 0
                max_stall = min(max_stall + 50, 600)
            elif stall > max_stall // 2:
                # Small noise
                f = np.maximum(f + rng.normal(0, 0.02, n), 0.0)
                s = f.sum()
                if s > 0.01:
                    f /= s
                else:
                    f = best_f.copy()
                stall = 0
            continue

        g_sum = g.sum()
        g = g / g_sum  # normalize g to unit sum

        # Line search
        best_t = 0.0
        best_sc = best_score
        for t in (0.15, 0.3, 0.5, 0.7, 0.85, 1.0, 0.05, 0.2):
            f_try = (1.0 - t) * f + t * g
            f_try = np.maximum(f_try, 0.0)
            s_try = f_try.sum()
            if s_try < 0.01:
                continue
            f_try = f_try / s_try
            b_try = _conv_full(f_try)
            sc = _score_from_parts(b_try, n, 1.0)
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * g
            f = np.maximum(f, 0.0)
            s = f.sum()
            f = f / s
            if best_sc < best_score:
                best_f = f.copy()
                best_score = best_sc
            stall = 0

            stall += 1
            if stall > max_stall:
                f = _exp_subgrad_step(f, b, n, lr=0.6)
                s = f.sum()
                if s < 0.01:
                    f = best_f.copy()
                else:
                    f /= s
                stall = 0
                max_stall = min(max_stall + 40, 600)

def _coord_descent(f0, stop_time):
    """Coordinate descent with incremental convolution updates."""
    n = len(f0)
    f = np.maximum(np.asarray(f0, dtype=np.float64).ravel().copy(), 0.0)
    S = float(f.sum())
    if S < 0.01:
        return f0.copy(), _score(f0)
    f = f / S
    S = 1.0
    b = _conv_full(f)
    best_f = f.copy()
    best_score = _score_from_parts(b, n, S)

    mults_coarse = (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 2.5, 0.0)
    mults_fine = (0.92, 0.96, 1.04, 1.08, 0.88, 1.12)

    no_imp = 0
    max_no_imp = 4

    while time.monotonic() < stop_time - 2.0:
        improved = False

        for i in range(n):
            if time.monotonic() > stop_time - 2.0:
                break
            orig = f[i]
            best_m = 1.0
            best_local = best_score

            # Coarse search
            for m in mults_coarse:
                nv = orig * m
                d = nv - orig
                if abs(d) < 1e-16:
                    continue
                nS = S + d
                if nS < 0.01:
                    continue
                # Compute new max autocorrelation
                new_max = 0.0
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        if j == i:
                            nbk = b[k] + 2.0 * orig * d + d * d
                        else:
                            nbk = b[k] + 2.0 * d * f[j]
                        nbk = max(0.0, nbk)
                    else:
                        nbk = b[k]
                    if nbk > new_max:
                        new_max = nbk
                # Also check unaffected peaks
                for k in range(2 * n - 1):
                    if k < k_lo or k > k_hi:
                        if b[k] > new_max:
                            new_max = b[k]

                sc = 2.0 * n * new_max / (nS * nS)
                if sc < best_local:
                    best_local = sc
                    best_m = m

            # Fine search around best
            if best_m != 1.0:
                for m in mults_fine:
                    nm = best_m * m
                    if nm < 0:
                        nm = 0.0
                    nv = orig * nm
                    d = nv - orig
                    if abs(d) < 1e-16:
                        continue
                    nS = S + d
                    if nS < 0.01:
                        continue
                    new_max = 0.0
                    k_lo = max(0, i)
                    k_hi = min(2 * n - 2, i + n - 1)
                    for k in range(k_lo, k_hi + 1):
                        j = k - i
                        if 0 <= j < n:
                            if j == i:
                                nbk = b[k] + 2.0 * orig * d + d * d
                            else:
                                nbk = b[k] + 2.0 * d * f[j]
                            nbk = max(0.0, nbk)
                        else:
                            nbk = b[k]
                        if nbk > new_max:
                            new_max = nbk
                    for k in range(2 * n - 1):
                        if k < k_lo or k > k_hi:
                            if b[k] > new_max:
                                new_max = b[k]
                    sc = 2.0 * n * new_max / (nS * nS)
                    if sc < best_local:
                        best_local = sc
                        best_m = nm

            if best_m != 1.0:
                # Apply update
                nv = orig * best_m
                d = nv - orig
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        if j == i:
                            b[k] = max(0.0, b[k] + 2.0 * orig * d + d * d)
                        else:
                            b[k] = max(0.0, b[k] + 2.0 * d * f[j])
                f[i] = nv
                S += d
                best_score = best_local
                best_f = f.copy()
                improved = True

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
            # Small random perturbation to escape
            f = np.maximum(f + np.random.default_rng(hash((n, no_imp))).normal(0, 0.01, n), 0.0)
            s = f.sum()
            if s > 0.01:
                f /= s
            else:
                f = best_f.copy()
            b = _conv_full(f)
            S = 1.0
            best_score = _score_from_parts(b, n, S)
        else:
            no_imp = 0

    # Rescale to unit sum for return
    s_final = best_f.sum()
    if s_final > 0.01:
        best_f = best_f / s_final
    return best_f, best_score
