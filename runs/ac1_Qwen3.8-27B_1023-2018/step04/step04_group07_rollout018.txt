def _lp_solve_v2(f, n, time_limit=10.0):
    """Minimize M s.t. (f*g)_k <= M, sum g = 1, g >= 0."""
    from scipy.optimize import linprog
    
    # Variables: g_0, ..., g_{n-1}, M (total n+1 variables)
    # Objective: minimize M -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    # Constraints: (f*g)_k - M <= 0 for all k
    # A_ub[k, j] = f[k-j] (if valid), A_ub[k, n] = -1
    # b_ub[k] = 0
    
    A = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = f[k-j]
        A[k, n] = -1.0
    
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum g = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # g_i >= 0, M free
    
    res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    
    if res.status in (0, 3) and res.x is not None:
        g = res.x[:n].copy()
        M_opt = res.x[n]
        return g, M_opt
    return None, None

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_conv_matrix(f, n):
    """Build the constraint matrix A where (f*g)_k = A[k,:] @ g[:n]"""
    A = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n, k + 1)
        for j in range(jmin, jmax):
            A[k, j] = f[k - j]
        A[k, n] = -1.0
    return A

def _lp_minimize_peakh(f, n, time_limit=8.0):
    """
    Minimize M subject to (f*g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    A = _build_conv_matrix(f, n)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None, None
    
    if res.status not in (0, 3) or res.x is None:
        return None, None
    
    g = np.asarray(res.x[:n], dtype=np.float64)
    M = float(res.x[n])
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None, None
    g = g / g_sum  # Ensure exact normalization
    return g, M
```

                f = best_f.copy()

            stall = 0

    return best_f, _score(best_f)

        n_pairs = min(200, n * 2)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 3.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue

Let me also consider: after finding g from the LP, the step (1-t)f + tg should reduce the self-correlation peak. But the LP only controls the cross-correlation f*g, not the self-correlation of the new point. The step is:

new = (1-t)f + tg

The peak of new*new is bounded by:
max_k [(1-t)^2 (f*f)_k + 2t(1-t) (f*g)_k + t^2 (g*g)_k]

We know (f*g)_k <= M (from LP), and (f*f)_k <= M_old (current peak). But (g*g)_k could be large!

So the line search is essential - we need to find t such that the new peak is actually reduced.

If t is small (first-order approximation):
new*new ≈ f*f + 2t(f*g - f*f)

For this to reduce the peak, we need (f*g)_k < (f*f)_k at the peak location k*. The LP ensures (f*g)_k <= M_lp where M_lp is the minimum possible peak of f*g. If M_lp < M_old = max(f*f), then a small step will help.

In fact, the LP gives M_lp = min max_k (f*g)_k over all probability vectors g. If this is less than the current peak M_old, we can make progress.

At the optimum (where no further improvement is possible), we'd have M_lp >= M_old, meaning the minimum cross-correlation peak is at least the current self-correlation peak. This would be the stopping condition.

Let me also add a "subgradient polish" step: after each LP iteration, take a few small gradient steps.

The subgradient of max_k b_k at point a (where k* = argmax b_k) is:
s_i = 2 * a_{k* - i} if 0 <= k* - i <= n-1, else 0

The projected subgradient step:
a_new = a - alpha * s
a_new = max(0, a_new)  (project to nonnegative)
a_new = a_new / sum(a_new)  (project to simplex)

The step size alpha should be small (like 0.01/n or adaptive).

This is a subgradient method which converges slowly but can help escape local minima.

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    """Fast score computation (may differ slightly from evaluator due to clip)."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    """Build constraint matrix for LP: A[k, j] = f[k-j] if valid, A[k, n] = -1"""
    A = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n, k + 1)
        for j in range(jmin, jmax):
            A[k, j] = f[k - j]
        A[k, n] = -1.0
    return A


def _lp_solve(f, n, time_limit=8.0):
    """Minimize M s.t. (f*g)_k <= M, sum g = 1, g >= 0."""
    A = _build_A(f, n)
    c = np.zeros(n + 1)
    c[n] = 1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    
    g = np.asarray(res.x[:n], dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    g = g / g_sum
    return g


def _subgrad_step(a, alpha=0.005):
    """One projected subgradient step on max(b)."""
    n = len(a)
    b = _conv_full(a)
    k_star = int(np.argmax(b))
    # Gradient of b_{k_star} w.r.t. a:
    # s_i = 2 * a[k_star - i] if 0 <= k_star - i <= n-1
    s = np.zeros(n)
    valid = (k_star - np.arange(n)) >= 0
    valid2 = (k_star - np.arange(n)) < n
    mask = valid & valid2
    s[mask] = 2.0 * a[k_star - np.arange(n)[mask]]
    
    a_new = a - alpha * s
    a_new = np.maximum(a_new, 0.0)
    a_sum = a_new.sum()
    if a_sum > 1e-10:
        a_new = a_new / a_sum
    else:
        a_new = a.copy()
    return a_new


def _lp_optimize(f0, n, stop_time, seed):

    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)

        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum
    
    best_f = f.copy()
    best_M = float(_conv_full(f).max())
    M = best_M
    stall = 0
    iter_count = 0
    
    while time.monotonic() < stop_time - 3.0:
        remaining = stop_time - time.monotonic()
        if remaining < 5.0:
            break
        
        # Solve LP
        tl = min(6.0, max(1.0, remaining * 0.3))
        g = _lp_solve(f, n, time_limit=tl)
        
        if g is None:
            stall += 3
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:

            else:

            M = float(_conv_full(f).max())
            continue
        
        dist = float(np.linalg.norm(g - f))
