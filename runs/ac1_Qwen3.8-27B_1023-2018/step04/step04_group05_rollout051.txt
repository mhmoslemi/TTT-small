I recall that for the problem of minimizing the peak sidelobe level in radar/communications, sequences like the Lorentzian (1/(1+x^2)) or similar are used. But those are for real (not necessarily nonnegative) sequences.

For nonnegative sequences, I think the optimal might be close to a circular shape or a "wrapping" distribution.

Actually, let me think about it differently. If we think of p as a probability density on a circle (wrapping around), then the circular autocorrelation would be:
`(p⊛p)_k = sum_i p_i p_{(k-i) mod n}`

For a uniform distribution, this is 1/n for all k. For a point mass, it's 1 at k=0 and 0 elsewhere.

The linear (non-circular) autocorrelation is related but different. The linear autocorrelation has the "edge effects" where there are fewer terms in the sum.

For a sequence that's been "wrapped" (i.e., the values at the edges are small), the linear and circular autocorrelations are similar in the middle.

I think the key is to have a distribution that's approximately uniform but with some structure that flattens the autocorrelation. The parent solution (score 1.509) is already doing this.

def _coord_descent(p, stop_time, seed=123, max_rounds=50):

    rng = np.random.default_rng(seed)

                    best_M_try = M_try

                    nv = best_val * frac

def _subgradient_step(p, b, eta):
    n = len(p)
    M = float(b.max())
    active_frac = 0.02
    active = np.where(b >= M * (1.0 - active_frac))[0]
    if len(active) == 0:
        active = np.where(b == M)[0]

    g = np.zeros(n)
    for k in active:
        # d/dp_j (p*p)_k = p_{k-j} if 0<=k-j<n plus p_{j-k} if 0<=j-k<n
        # Since k-j and j-k have opposite signs (or both zero), at most one is in [0,n-1]
        dk = k
        for j in range(n):
            diff = k - j
            if 0 <= diff < n:
                g[j] += p[diff]
            elif 0 <= (j - k) < n:
                g[j] += p[j - k]

    g /= max(len(active), 1)
    p_new = p - eta * g
    p_new = _proj_simplex(p_new)
    return p_new


def _sg_refine(p, stop_time, seed=999, n_steps=5000):
    n = len(p)
    rng = np.random.default_rng(seed)
    p = np.clip(p.copy(), 0, None)
    S = p.sum()
    if S < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / S

    best_p = p.copy()
    best_M = float(_autocorr(p).max())
    eta = 0.05 / n
    stall = 0
    max_stall = 500
    it = 0

    while time.monotonic() < stop_time - 3.0 and it < n_steps:
        it += 1
        b = _autocorr(p)
        M = float(b.max())

        p_new = _subgradient_step(p, b, eta)
        M_new = float(_autocorr(p_new).max())

        if M_new < best_M - 1e-14:
            p = p_new
            best_p = p.copy()
            best_M = M_new
            eta = min(eta * 1.05, 0.5 / n)
            stall = 0
        else:
            if M_new < M:
                p = p_new
            eta *= 0.95
            stall += 1

        if stall > max_stall:
            p = best_p.copy()
            p += rng.normal(0, 0.01 * p.mean(), n)
            p = np.clip(p, 0, None)
            S = p.sum()
            p = p / max(S, 1e-15)
            eta = 0.05 / n
            stall = 0
            max_stall += 200

    return best_p, 2.0 * n * best_M


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    n_p = len(parent)

    # Normalize parent to probability
    S_p = parent.sum()
    if S_p < 1e-15:
        parent_prob = np.ones(n_p) / n_p
    else:
        parent_prob = np.clip(parent, 0, None) / S_p

    best_seq = parent.copy()
    best_score = _score_raw(parent)
    if not math.isfinite(best_score):
        best_seq = np.ones(100)
        best_score = _score_raw(best_seq)

    def _update_best(seq, sc):
        nonlocal best_seq, best_score
        if sc < best_score - 1e-13:
            best_seq = seq.copy()
            best_score = sc

    def _try_lp(n, init_p, stop, sd):
        if time.monotonic() > stop - 3.0:
            return
        try:
            res_p, res_sc = _lp_refine(init_p, stop, seed=sd)
        except Exception:
            return
        if res_p is None or len(res_p) == 0 or not np.all(np.isfinite(res_p)):
            return
        # Convert back to raw sequence (scale doesn't matter for score, but keep reasonable)
        S = res_p.sum()
        if S < 1e-15:
            return
        raw = res_p * math.sqrt(2.0 * n)
        raw = np.clip(raw, 0, 1000)
        sc = 2.0 * n * float(_autocorr(res_p).max())
        if math.isfinite(sc) and sc < best_score - 1e-12:
            exact = float(evaluate_sequence([float(x) for x in raw]))
            if math.isfinite(exact) and exact < best_score - 1e-13:
                best_seq = raw.copy()
                best_score = exact

    # Phase 1: LP refinement from parent (45% of time)
    phase1_end = t0 + budget * 0.45
    _try_lp(n_p, parent_prob, phase1_end, seed)

    # Phase 2: LP from several diverse initializations (20% of time)
    phase2_end = t0 + budget * 0.65
    rng = np.random.default_rng(seed + 1)
    i_arr = np.arange(n_p, dtype=np.float64)

    inits = []
    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n_p)) + 0.01)
    # |Normal|
    inits.append(np.abs(rng.normal(0, 1, n_p)) + 0.02)
    # Multi-structure
    sup = np.ones(n_p) * 0.3
    for fr in (2, 5, 13):
        ph = rng.uniform(0, 2 * math.pi)
        sup += 0.15 * np.cos(2.0 * math.pi * fr * i_arr / n_p + ph)
    inits.append(np.abs(sup) + 0.01)
    # Parent + stronger noise
    sig = 0.25 * parent_prob.mean()
    inits.append(np.clip(parent_prob + rng.normal(0, sig, n_p), 0, None) + 0.01)
    # Wrap-like (higher at edges)
    inits.append(0.5 + 0.5 * np.sin(math.pi * i_arr / (n_p - 1)) ** 2 + 0.01)

    # Normalize inits to probability
    inits_norm = []
    for init in inits:
        S_i = init.sum()
        if S_i > 1e-15:
            inits_norm.append(init / S_i)

    time_per_init = max(5.0, (phase2_end - t0) / max(1, len(inits_norm)))
    for idx, ip in enumerate(inits_norm):
        if time.monotonic() > phase2_end - 8.0:
            break
        stop = min(phase2_end - 3.0, time.monotonic() + time_per_init)
        _try_lp(n_p, ip, stop, seed + 30000 * (idx + 1))

    # Phase 3: Subgradient refinement on best (15% of time)
    phase3_end = t0 + budget * 0.80
    if time.monotonic() < phase3_end - 15.0:
        S_b = best_seq.sum()
        if S_b > 1e-10:
            p_b = np.clip(best_seq, 0, None) / S_b
            try:
                res_p, res_sc = _sg_refine(p_b, phase3_end, seed=seed + 777, n_steps=8000)
                if res_sc < best_score - 1e-12:
                    raw = res_p * math.sqrt(2.0 * len(res_p))
                    raw = np.clip(raw, 0, 1000)
                    exact = float(evaluate_sequence([float(x) for x in raw]))
                    if math.isfinite(exact) and exact < best_score - 1e-13:
                        best_seq = raw.copy()
                        best_score = exact
            except Exception:
                pass

    # Phase 4: Coordinate descent polishing (10% of time)
    phase4_end = t0 + budget * 0.90
    if time.monotonic() < phase4_end - 10.0:
        S_b = best_seq.sum()
        if S_b > 1e-10:
            p_b = np.clip(best_seq, 0, None) / S_b
            try:
                res_p, res_sc = _coord_descent(p_b, phase4_end, seed=seed + 55555, max_rounds=80)
                if res_sc < best_score - 1e-12:
                    raw = res_p * math.sqrt(2.0 * len(res_p))
                    raw = np.clip(raw, 0, 1000)
                    exact = float(evaluate_sequence([float(x) for x in raw)])
                    if math.isfinite(exact) and exact < best_score - 1e-13:
                        best_seq = raw.copy()
                        best_score = exact
            except Exception:
                pass

    # Phase 5: Second LP pass (warm start from best, 8% of time)
    phase5_end = deadline - 15.0
    if time.monotonic() < phase5_end - 15.0:
        S_b = best_seq.sum()
        if S_b > 1e-10:
            p_b = np.clip(best_seq, 0, None) / S_b
            _try_lp(len(p_b), p_b, phase5_end, seed + 999999)

    # Phase 6: Final coordinate descent polish
    if time.monotonic() < deadline - 8.0:
        S_b = best_seq.sum()
        if S_b > 1e-10:
            p_b = np.clip(best_seq, 0, None) / S_b
            try:
                res_p, res_sc = _coord_descent(p_b, deadline - 5.0, seed=seed + 111111, max_rounds=30)
                if res_sc < best_score - 1e-12:
                    raw = res_p * math.sqrt(2.0 * len(res_p))
                    raw = np.clip(raw, 0, 1000)
                    exact = float(evaluate_sequence([float(x) for x in raw)])
                    if math.isfinite(exact) and exact < best_score - 1e-13:
                        best_seq = raw.copy()
                        best_score = exact
            except Exception:
                pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Fallback: return parent
        try:
            fb_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            fb_sc = math.inf
        if math.isfinite(fb_sc):
            return [float(x) for x in parent]
        return [1.0]

    return [float(x) for x in best_seq]
