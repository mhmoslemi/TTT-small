    b = _conv_full(f)
    S = float(f.sum())

    b_scaled = b / tau

    sum_exp = exp_vals.sum()
    smooth_max = tau * (b_max_val + np.log(sum_exp))

    b = _conv_full(f)
    M = float(b.max())
    
    rows, cols, vals = [], [], []
    for k in range(2*n-1):
        i_start = max(0, k-n+1)
        i_end = min(n, k+1)
        for i in range(i_start, i_end):
            v = f[k-i]

                rows.append(k)

                vals.append(v)

        return None

    # Minimize t (last variable)
    c_vec = np.zeros(n+1)
    c_vec[-1] = 1.0

    A_full = A.copy()

    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_lp_matrix(f, n, n_constraints=None):
    """Build sparse LP constraint matrix for (f*g)[k] <= M."""
    if n_constraints is None:
        n_constraints = 2 * n - 1
    rows, cols, vals = [], [], []
    for k in range(n_constraints):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-20:
                rows.append(k)
                cols.append(i)
                vals.append(v)
    if not rows:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(n_constraints, n))
    return A


def _lp_direction(f, n, peak_frac=None, timeout=5.0):
    """Solve LP to find best direction g."""
    b = _conv_full(f)
    M = float(b.max())
    if M < 1e-18:
        return None, None

    if peak_frac is not None:
        threshold = M * (1.0 - peak_frac)
        peak_idx = np.where(b >= threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])
        # Build sub-matrix for only peak constraints
        rows, cols, vals = [], [], []
        for idx, k in enumerate(peak_idx):
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                v = f[k - i]
                if v > 1e-20:
                    rows.append(idx)
                    cols.append(i)
                    vals.append(v)
        if not rows:
            return None, None
        A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
        b_ub = np.full(len(peak_idx), M)
    else:
        A = _build_lp_matrix(f, n)
        if A is None:
            return None, None
        b_ub = np.full(2 * n - 1, M)

    c_vec = -np.ones(n)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout})
    except Exception:
        return None, None

    if res.status not in (0, 3) or res.x is None:
        return None, None

    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-15:
        return None, None
    return g, M


def _line_search(f, g, n, max_evals=25):
    """Find best step size t for f_new = (1-t)*f + t*g."""
    f_sum = f.sum()
    g_sum = g.sum()
    if g_sum < 1e-15:
        return 0.0, _score_direct(f)
    g = g * (f_sum / g_sum)

    best_score = _score_direct(f)
    best_t = 0.0
    evals = 0

    candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.85]
    for t in candidates:
        f_new = np.maximum((1.0 - t) * f + t * g, 0.0)
        if f_new.sum() < 0.01:
            continue
        sc = _score_direct(f_new)
        evals += 1
        if sc < best_score and math.isfinite(sc):
            best_score = sc
            best_t = t
        if evals >= max_evals:
            break

    if best_t > 0 and evals < max_evals:
        lo = max(0.0, best_t - 0.15)
        hi = min(0.98, best_t + 0.15)
        for _ in range(5):
            if evals >= max_evals:
                break
            t1 = lo + 0.382 * (hi - lo)
            t2 = lo + 0.618 * (hi - lo)
            f1 = np.maximum((1.0 - t1) * f + t1 * g, 0.0)
            f2 = np.maximum((1.0 - t2) * f + t2 * g, 0.0)
            sc1 = _score_direct(f1)
            sc2 = _score_direct(f2)
            evals += 2
            if sc1 < sc2:
                hi = t2
            else:
                lo = t1
            t_mid = (lo + hi) / 2.0
            fm = np.maximum((1.0 - t_mid) * f + t_mid * g, 0.0)
            scm = _score_direct(fm)
            evals += 1
            if scm < best_score:
                best_score = scm
                best_t = t_mid

    return best_t, best_score


def _optimize_lp_phase(f0, stop_time, seed, peak_frac=None):
    """LP-based optimization phase."""
    n = len(f0)
    rng = np.random.default_mgr = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0.0, None)
    best_f = f.copy()
    best_score = _score_direct(f)

    stall = 0
    max_stall = 50
    iter_count = 0

    while time.monotonic() < stop_time - 3.0:
        iter_count += 1
        g, M = _lp_direction(f, n, peak_frac=peak_frac, timeout=4.0)
        if g is None:
            stall += 4
            if stall > max_stall:
                noise = rng.normal(0, 0.08 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                if f.sum() < 0.01:
                    f = best_f.copy()
                stall = 0
                max_stall += 20
                if max_stall > 200:
                    break
            continue

        t, sc = _line_search(f, g, n, max_evals=20)
        if t > 0.01 and sc < best_score - 1e-14 and math.isfinite(sc):
            f = np.maximum((1.0 - t) * f + t * g * (f.sum() / g.sum()), 0.0)
            best_f = f.copy()
            best_score = sc
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                noise = rng.normal(0, 0.12 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0.0, None)
                if f.sum() < 0.01:
                    f = best_f.copy()
                stall = 0
                max_stall = min(max_stall + 30, 300)

    return best_f, best_score


def _smooth_objective(f, tau=0.005):
    """Smooth approximation of the score using log-sum-exp."""
    n = len(f)
    S = float(f.sum())
    if S < 0.01:
        return 1e10, np.ones(n)
    b = _conv_full(f)
    M = float(b.max())

    b_scaled = b / tau
    b_max = b_scaled.max()
    exp_vals = np.exp(b_scaled - b_max)
    sum_exp = float(exp_vals.sum())
    if sum_exp < 1e-300:
        return 1e10, np.ones(n)
    smooth_max = tau * (b_max + np.log(sum_exp))
    weights = exp_vals / sum_exp

    # Gradient of smooth_max w.r.t. f
    grad_sm = np.zeros(n)
    for k in range(2 * n - 1):
        w = float(weights[k])
        if w < 1e-10:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            grad_sm[i] += 2.0 * w * f[k - i]

    score = 2.0 * n * smooth_max / (S * S)
    grad = 2.0 * n * (S * grad_sm - 2.0 * smooth_max) / (S ** 3)
    return score, grad


def _optimize_smooth_lbfgs(f0, stop_time, seed):
    """L-BFGS with smooth log-sum-exp objective."""
    n = len(f0)
    f = np.clip(f0.copy(), 1e-10, None)

    tau = 0.005
    eval_count = 0
    start_time = time.monotonic()

    def fun_grad(x):
        nonlocal eval_count
        eval_count += 1
        if time.monotonic() > stop_time - 5.0:
            raise TimeoutError()
        sc, gr = _smooth_objective(x, tau=tau)
        if not np.all(np.isfinite(gr)):
            gr = np.zeros(len(x))
        return sc, gr

    best_x = f.copy()
    best_sc = _score_direct(f)

    try:
        res = minimize(fun_grad, f, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, 1000.0)] * n,
                       options={'maxiter': 300, 'maxfun': 5000,
                                'ftol': 1e-14, 'gtol': 1e-10})
        if res.x is not None:
            f_new = np.clip(res.x, 0.0, None)
            sc = _score_direct(f_new)
            if math.isfinite(sc) and sc < best_sc:
                best_x = f_new
                best_sc = sc
    except (TimeoutError, Exception):
        pass

    # If time remains, try with smaller tau for more accurate gradient
    remaining = stop_time - time.monotonic() - 5.0
    if remaining > 5.0:
        tau = 0.001
        eval_count = 0
        f_start = best_x.copy()

        def fun_grad2(x):
            nonlocal eval_count
            eval_count += 1
            if time.monotonic() > stop_time - 5.0:
                raise TimeoutError()
            sc, gr = _smooth_objective(x, tau=tau)
            if not np.all(np.isfinite(gr)):
                gr = np.zeros(len(x))
            return sc, gr

        try:
            res = minimize(fun_grad2, f_start, method='L-BFGS-B', jac=True,
                           bounds=[(0.0, 1000.0)] * n,
                           options={'maxiter': 200, 'maxfun': 3000,
                                    'ftol': 1e-14, 'gtol': 1e-11})
            if res.x is not None:
                f_new = np.clip(res.x, 0.0, None)
                sc = _score_direct(f_new)
                if math.isfinite(sc) and sc < best_sc:
                    best_x = f_new
                    best_sc = sc
        except (TimeoutError, Exception):
            pass

    return best_x, best_sc


def _optimize_coord_fast(f0, stop_time, seed):
    """Fast coordinate descent with O(n) per-eval convolution update."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0.0, None)
    S = float(f.sum())
    b = _conv_full(f)
    best_score = 2.0 * n * float(b.max()) / (S * S)
    best_f = f.copy()

    no_imp = 0
    max_no_imp = 40
    pass_count = 0

    while time.monotonic() < stop_time - 3.0 and no_imp < max_no_imp:
        pass_count += 1
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break

            f_i = f[i]
            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            num_aff = k_max - k_min + 1

            # Unaffected max
            b_unaff = 0.0
            if k_min > 0:
                b_unaff = max(b_unaff, float(b[:k_min].max()))
            if k_max < 2 * n - 1:
                b_unaff = max(b_unaff, float(b[k_max + 1:].max()))

            affected = b[k_min:k_max + 1].copy()
            mults = f[:num_aff]  # f[0], f[1], ..., f[num_aff-1]

            has_sq = (2 * i >= k_min and 2 * i <= k_max)
            sq_idx = (2 * i - k_min) if has_sq else -1

            currently_best = 2.0 * n * max(b_unaff, float(affected.max())) / (S * S)

            best_d = 0.0
            best_s = currently_best

            if f_i > 1e-15:
                fracs = (-0.6, -0.4, -0.25, -0.12, -0.06, -0.03, -0.01,
                         0.01, 0.03, 0.06, 0.1, 0.2, 0.35, 0.6)
                for fr in fracs:
                    d = f_i * fr
                    new_fi = f_i + d
                    if new_fi < 0.0:
                        continue
                    S_new = S + d
                    if S_new < 0.01:
                        continue
                    inner = affected + 2.0 * d * mults
                    if has_sq:
                        inner[sq_idx] += d * d
                    im = float(inner.max())
                    max_b = max(b_unaff, im)
                    s = 2.0 * n * max_b / (S_new * S_new)
                    if s < best_s:
                        best_s = s
                        best_d = d
            else:
                for val in (0.005, 0.02, 0.05, 0.1, 0.3) * max(1.0, f.mean() if f.mean() > 0 else 1.0):
                    d = val
                    S_new = S + d
                    if S_new < 0.01:
                        continue
                    inner = affected + 2.0 * d * mults
                    if has_sq:
                        inner[sq_idx] += d * d
                    im = float(inner.max())
                    max_b = max(b_unaff, im)
                    s = 2.0 * n * max_b / (S_new * S_new)
                    if s < best_s:
                        best_s = s
                        best_d = d

            if best_d != 0.0 and best_s < best_score - 1e-15:
                f[i] += best_d
                S += best_d
                b[k_min:k_max + 1] = affected + 2.0 * best_d * mults
                if has_sq:
                    b[2 * i] += best_d * best_d
                best_score = best_s
                best_f = f.copy()
                improved = True

        if not improved:
            no_imp += 1
            # Try a different random order with a small perturbation
            if no_imp % 5 == 0 and no_imp < max_no_imp:
                noise = rng.normal(0, 0.005 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                S = float(f.sum())
                b = _conv_full(f)
                no_imp = max(0, no_imp - 2)

    return best_f, best_score


def _optimize_coord_fine(f0, stop_time, seed):
    """Finer coordinate descent with local search around current values."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0.0, None)
    S = float(f.sum())
    b = _conv_full(f)
    best_score = 2.0 * n * float(b.max()) / (S * S)
    best_f = f.copy()
    no_imp = 0

    fine_fracs = (-0.08, -0.05, -0.03, -0.02, -0.01, -0.005,
                  0.005, 0.01, 0.02, 0.03, 0.05, 0.08)

    while time.monotonic() < stop_time - 3.0 and no_imp < 60:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            f_i = f[i]
            if f_i < 1e-14:
                continue

            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            num_aff = k_max - k_min + 1

            b_unaff = 0.0
            if k_min > 0:
                b_unaff = max(b_unaff, float(b[:k_min].max()))
            if k_max < 2 * n - 1:
                b_unaff = max(b_unaff, float(b[k_max + 1:].max()))

            affected = b[k_min:k_max + 1]
            mults = f[:num_aff]
            has_sq = (2 * i >= k_min and 2 * i <= k_max)
            sq_idx = (2 * i - k_min) if has_sq else -1

            best_d = 0.0
            best_s = 2.0 * n * max(b_unaff, float(affected.max())) / (S * S)

            for fr in fine_fracs:
                d = f_i * fr
                new_fi = f_i + d
                if new_fi < 0:
                    continue
                S_new = S + d
                if S_new < 0.01:
                    continue
                inner = affected + 2.0 * d * mults
                if has_sq:
                    inner[sq_idx] += d * d
                im = float(inner.max())
                max_b = max(b_unaff, im)
                s = 2.0 * n * max_b / (S_new * S_new)
                if s < best_s:
                    best_s = s
                    best_d = d

            if best_d != 0.0 and best_s < best_score - 1e-14:
                f[i] += best_d
                S += best_d
                b[k_min:k_max + 1] = affected + 2.0 * best_d * mults
                if has_sq:
                    b[2 * i] += best_d * best_d
                best_score = best_s
                best_f = f.copy()
                improved = True

        if not improved:
            no_imp += 1

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = _score_direct(best_seq)

    def _update_best(candidate):
        nonlocal best_seq, best_score
        if candidate is None or len(candidate) == 0:
            return
        candidate = np.clip(np.asarray(candidate, dtype=np.float64), 0.0, 1000.0)
        if not np.all(np.isfinite(candidate)):
            return
        if candidate.sum() < 0.01:
            return
        sc = _score_direct(candidate)
        if math.isfinite(sc) and sc < best_score - 1e-16:
            best_seq = candidate.copy()
            best_score = sc

    rng = np.random.default_rng(seed)
    i_arr = np.arange(n_p, dtype=np.float64)

    # Build initializations
    inits = [parent.copy()]

    # Perturbed parent
    for sig_factor in (0.05, 0.1, 0.15):
        sig = sig_factor * max(1.0, float(parent.mean()))
        inits.append(np.clip(parent + rng.normal(0, sig, n_p), 0, None))

    # Cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / n_p)))

    # Superposition of cosines
    sup = np.ones(n_p) * 0.3
    for fr in (1, 2, 3, 5, 7, 11, 13, 17):
        ph = rng.uniform(0, 2 * np.pi)
        sup += 0.08 * np.cos(2.0 * np.pi * fr * i_arr / n_p + ph)
    inits.append(np.abs(sup) + 0.02)

    # Gaussian
    center = (n_p - 1) / 2.0
    inits.append(np.exp(-0.5 * ((i_arr - center) / (n_p * 0.25)) ** 2))

    # Two bumps
    c1, c2 = n_p * 0.25, n_p * 0.75
    w = n_p * 0.1
    inits.append(np.exp(-0.5 * ((i_arr - c1) / w) ** 2) + np.exp(-0.5 * ((i_arr - c2) / w) ** 2))

    # Symmetric pattern
    sym = np.ones(n_p)
    for i in range(n_p):
        sym[i] *= 0.5 + 0.5 * np.cos(np.pi * i / n_p)
    inits.append(sym)

    # Truncate
    inits = inits[:8]

    # Phase 1: LP optimization on multiple inits (40% of budget)
    phase1_end = t0 + budget * 0.40
    time_per_init = max(12.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        if time.monotonic() > stop - 3.0:
            continue
        try:
            res, sc = _optimize_lp_phase(init, stop, seed + 1000 * (idx + 1),
                                         peak_frac=0.05)
            _update_best(res)
        except Exception:
            pass

    # Continue LP on best with tighter constraints
    if time.monotonic() < phase1_end - 20.0:
        stop = phase1_end - 5.0
        try:
            res, sc = _optimize_lp_phase(best_seq, stop, seed + 999001,
                                         peak_frac=0.03)
            _update_best(res)
        except Exception:
            pass
        if time.monotonic() < stop - 10.0:
            try:
                res, sc = _optimize_lp_phase(best_seq, stop, seed + 999002,
                                             peak_frac=None)
                _update_best(res)
            except Exception:
                pass

    # Phase 2: Smooth L-BFGS (18% of budget)
    phase2_end = t0 + budget * 0.58
    if time.monotonic() < phase2_end - 10.0:
        stop = phase2_end - 5.0
        try:
            res, sc = _optimize_smooth_lbfgs(best_seq, stop, seed + 555001)
            _update_best(res)
        except Exception:
            pass

    # Phase 3: Fast coordinate descent (22% of budget)
    phase3_end = t0 + budget * 0.80
    if time.monotonic() < phase3_end - 10.0:
        stop = phase3_end - 5.0
        try:
            res, sc = _optimize_coord_fast(best_seq, stop, seed + 444001)
            _update_best(res)
        except Exception:
            pass

    # Phase 4: Fine coordinate descent (10% of budget)
    phase4_end = t0 + budget * 0.90
    if time.monotonic() < phase4_end - 10.0:
        stop = phase4_end - 5.0
        try:
            res, sc = _optimize_coord_fine(best_seq, stop, seed + 333001)
            _update_best(res)
        except Exception:
            pass

    # Phase 5: Another LP + coord cycle (8% of budget)
    phase5_end = deadline + 5.0
    if time.monotonic() < phase5_end - 30.0:
        # Quick LP
        stop_lp = min(phase5_end - 20.0, time.monotonic() + 40.0)
        if time.monotonic() < stop_lp - 5.0:
            try:
                res, sc = _optimize_lp_phase(best_seq, stop_lp, seed + 222001,
                                             peak_frac=0.02)
                _update_best(res)
            except Exception:
                pass
        # Quick coord
        stop_cd = phase5_end - 5.0
        if time.monotonic() < stop_cd - 10.0:
            try:
                res, sc = _optimize_coord_fast(best_seq, stop_cd, seed + 111001)
                _update_best(res)
            except Exception:
                pass

    # Final: very short fine coord
    if time.monotonic() < deadline - 5.0:
        stop = deadline - 3.0
        try:
            res, sc = _optimize_coord_fine(best_seq, stop, seed + 770001)
            _update_best(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    final_candidate = [float(x) for x in best_seq]
    try:
        final_sc = float(evaluate_sequence(final_candidate))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Fallback
        fallback = [float(x) for x in parent]
        try:
            fb_sc = float(evaluate_sequence(fallback))
        except Exception:
            fb_sc = math.inf
        if math.isfinite(fb_sc):
            return fallback
        return [1.0] * n_p

    return final_candidate
