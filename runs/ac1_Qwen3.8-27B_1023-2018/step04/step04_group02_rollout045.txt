def _build_A(f, n):
    """Build the (2n-1) x n constraint matrix for f*g <= M."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):

    return A


def _lp_dir(f, n, M, timeout=3.0):
    """Solve LP to find direction g maximizing sum(g) s.t. f*g <= M, g >= 0."""
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g
