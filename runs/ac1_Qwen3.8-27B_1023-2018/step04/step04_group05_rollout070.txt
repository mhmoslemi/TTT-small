def _proj_simplex(v):
  x=np.asarray(v,dtype)
  n=len(x)
  if n==1: return np.maximum(x,0)
  u=np.sort(x)[::-1]
  cssv=np.cumsum(u)
  rho=np.nonzero(u * np.arange(1,n+1) > (cssv - 1.0))[0]
  if len(rho)==0: return np.zeros_like(x)
  r=rho[-1]
  theta=(cssv[r]-1.0)/(r+1)
  return np.maximum(x-theta,0)

def _proj_simplex(v):

import numpy as np
import time
import math
from scipy.optimize import linprog


def _norm(f):
    f = np.asarray(f, dtype=np.float64).ravel()
    if f.size == 0:
        return np.ones(1, dtype=np.float64)
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.clip(f, 0.0, 1000.0)
    s = float(f.sum())
    if s < 1e-12:
        return np.ones(f.size, dtype=np.float64) / float(f.size)
    return f / s


def _conv(a, b=None):
    if b is None:
        return np.convolve(a, a, mode="full")
    return np.convolve(a, b, mode="full")


def _score(f):
    f = np.asarray(f, dtype=np.float64).ravel()
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.clip(f, 0.0, 1000.0)
    n = len(f)
    if n == 0:
        return math.inf
    s = float(f.sum())
    if s < 0.01:
        return math.inf
    b = _conv(f)
    m = float(b.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return 2.0 * float(n) * m / (s * s)


def _proj_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = len(v)
    if n == 0:
        return np.array([1.0])
    if n == 1:
        return np.array([max(0.0, float(v[0]))], dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cand = u * np.arange(1, n + 1) - (cssv - 1.0)
    idx = np.nonzero(cand > 0.0)[0]
    if idx.size == 0:
        return np.zeros(n, dtype=np.float64)
    r = int(idx[-1])
    theta = (cssv[r] - 1.0) / float(r + 1)
    return np.maximum(v - theta, 0.0)


def _subgrad_step(f, b, beta):
    n = len(f)
    if n <= 1:
        return f.copy()
    K = 2 * n - 1
    w = np.exp(beta * (b - float(b.max())))
    ws = float(w.sum())
    if ws <= 0.0 or not math.isfinite(ws):
        return f.copy()
    w = w / ws

    grad = np.zeros(n, dtype=np.float64)
    for i in range(n):
        seg = min(n, K - i)
        if seg > 0:
            grad[i] = 2.0 * float(np.dot(f[:seg], w[i:i + seg]))

    if not np.all(np.isfinite(grad)):
        return f.copy()

    best = f.copy()
    best_sc = 2.0 * n * float(b.max())

    for alpha in (0.05, 0.1, 0.25, 0.5, 1.0):
        y = f - alpha * grad
        y = _proj_simplex(y)
        y = _norm(y)
        by = _conv(y)
        sc = 2.0 * n * float(by.max())
        if math.isfinite(sc) and sc < best_sc:
            best = y
            best_sc = sc
    return best


def _perturb(f, rng, strength=0.3):
    n = len(f)
    if n <= 1:
        f = np.ones(1, dtype=np.float64)
        b = _conv(f)
        return f, b, 2.0 * float(b.max())

    choice = int(rng.integers(4))
    if choice == 0:
        out = f + rng.normal(0.0, strength / float(n), n)
    elif choice == 1:
        out = f * np.exp(rng.normal(0.0, 0.6 * strength, n))
    elif choice == 2:
        out = f.copy()
        k = max(5, n // 20)
        idx = rng.choice(n, size=k, replace=False)
        out[idx] *= rng.uniform(0.2, 3.0, k)
    else:
        b = _conv(f)
        out = _subgrad_step(f, b, rng.uniform(5.0, 25.0))

    out = _norm(out)
    if not np.all(np.isfinite(out)):
        out = f.copy()
    b = _conv(out)
    sc = 2.0 * float(n) * float(b.max())
    return out, b, sc


def _vals(x, abs_vals, ratios):
    out = [float(x), 0.0]
    if x > 1e-15:
        for r in ratios:
            out.append(float(x) * float(r))
    out.extend(float(z) for z in abs_vals)
    arr = np.asarray(out, dtype=np.float64)
    arr = arr[np.isfinite(arr)]
    arr = np.clip(arr, 0.0, 100.0)
    return np.unique(arr)


def _line_search(f, g, ff, fg, gg, n, cur_sc, t_guess):
    ts = {1.0, 0.9, 0.8, 0.7, 0.6, 0.55, 0.5, 0.45, 0.4, 0.35, 0.3,
          0.25, 0.2, 0.15, 0.12, 0.09, 0.07, 0.05, 0.035, 0.025, 0.015, 0.008}
    if t_guess is not None and math.isfinite(t_guess):
        for m in (0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0):
            ts.add(float(np.clip(t_guess * m, 0.0, 1.0)))
    best_t = 0.0
    best_sc = cur_sc
    for t in sorted(ts, reverse=True):
        if t <= 0.0:
            continue
        wt = 1.0 - t
        mt = (wt * wt) * ff + (2.0 * t * wt) * fg + (t * t) * gg
        m = float(np.max(mt))
        sc = 2.0 * float(n) * m
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_t = t
    return best_t


def _lp_direction(f, n, timeout=2.0):
    K = 2 * n - 1
    A = np.zeros((K, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0
    A_ub = np.empty((K, n + 1), dtype=np.float64)
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(K, dtype=np.float64)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"time_limit": float(timeout)},
        )
    except Exception:
        return None

    if res is None or getattr(res, "x", None) is None:
        return None
    status = int(getattr(res, "status", 1))
    if status not in (0, 3):
        return None
    x = np.asarray(res.x, dtype=np.float64)
    if not np.all(np.isfinite(x)):
        return None

    g = np.clip(x[:n], 0.0, None)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    g /= gs
    return g


def _local_polish(f0, b0, stop, seed=0, max_sweeps=500, pair_count=None, fine_after=2):
    tnow = time.monotonic()
    f = _norm(f0)
    n = len(f)
    if n <= 1:
        f = np.ones(1, dtype=np.float64)
        b = _conv(f)
        return f, b, 2.0 * float(b.max())

    if pair_count is None:
        pair_count = 90 if n <= 512 else (40 if n <= 900 else 16)

    b = _conv(f)
    sc = 2.0 * n * float(b.max())
    best_f = f.copy()
    best_b = b.copy()
    best_sc = sc

    rng = np.random.default_rng(seed)
    K = 2 * n - 1

    inv_n = 1.0 / float(n)
    abs_coarse = inv_n * np.array(
        [1e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 1e-1, 3e-1,
         1.0, 3.0, 10.0, 30.0, 100.0, 300.0], dtype=np.float64)
    abs_fine = inv_n * np.array(
        [1e-6, 1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 1e-1,
         1.0, 3.0], dtype=np.float64)
    ratios_coarse = np.array(
        [0.0, 0.2, 0.5, 0.75, 0.9, 1.1, 1.3, 2.0, 5.0], dtype=np.float64)
    ratios_fine = np.array(
        [0.5, 0.8, 0.95, 0.99, 1.01, 1.05, 1.2], dtype=np.float64)
    pair_abs = inv_n * np.array([0.0, 3.0, 10.0, 30.0, 100.0], dtype=np.float64)
    pair_ratios = np.array([0.3, 1.0, 2.0], dtype=np.float64)

    tol = 1e-14
    sweeps = 0
    no_imp = 0

    while time.monotonic() < stop and sweeps < max_sweeps:
        sweeps += 1
        improved = False
        use_fine = sweeps >= fine_after
        abs_vals = abs_fine if use_fine else abs_coarse
        ratios = ratios_fine if use_fine else ratios_coarse
        order = rng.permutation(n) if sweeps > 1 else np.arange(n)

        f2 = 2.0 * f

        for ii in order:
            if time.monotonic() >= stop:
                break
            i = int(ii)
            x = float(f[i])
            vals = _vals(x, abs_vals, ratios)
            if vals.size <= 1:
                continue
            d = vals - x
            if not np.any(np.abs(d) > 1e-15):
                continue

            out_max = 0.0
            if i > 0:
                m1 = float(np.max(b[:i]))
                if m1 > out_max:
                    out_max = m1
            if i + n < K:
                m2 = float(np.max(b[i + n:]))
                if m2 > out_max:
                    out_max = m2

            T = b[i:i + n][None, :] + d[:, None] * f2[None, :]
            pos = i
            T[:, pos] += d[:, None] * d[:, None]

            s = 1.0 + d
            valid = s > 1e-12
            if not valid.all():
                T[~valid] = np.inf
            mmax = np.maximum(T.max(axis=1), out_max)
            sc_try = 2.0 * n * mmax / (s * s)
            sc_try[~valid] = math.inf

            idx = int(np.argmin(sc_try))
            if valid[idx] and sc_try[idx] < sc - tol and abs(vals[idx] - x) > 1e-15:
                dval = float(d[idx])
                sval = float(s[idx])
                vval = float(vals[idx])
                two = 2 * i

                new_b = b.copy()
                new_b[i:i + n] += dval * f2
                new_b[two] += dval * dval
                inv_s2 = 1.0 / (sval * sval)
                new_b *= inv_s2

                new_f = f / sval
                new_f[i] = vval / sval

                f = new_f
                b = new_b
                sc = float(sc_try[idx])
                f2 = 2.0 * f
                improved = True

                if sc < best_sc - tol:
                    best_sc = sc
                    best_f = f.copy()
                    best_b = b.copy()

        if time.monotonic() + 0.02 < stop:
            f2 = 2.0 * f
            pc = max(8, pair_count // 3 if use_fine else pair_count)
            for _ in range(pc):
                if time.monotonic() >= stop:
                    break

                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                xi = float(f[i])
                xj = float(f[j])
                if xi + xj < 1e-14 and rng.random() < 0.75:
                    continue

                base_i = np.zeros(K, dtype=np.float64)
                base_i[i:i + n] = f2
                base_j = np.zeros(K, dtype=np.float64)
                base_j[j:j + n] = f2

                vi_vals = _vals(xi, pair_abs, pair_ratios)
                vj_vals = _vals(xj, pair_abs, pair_ratios)
                pair_acc = False

                for vi in vi_vals:
                    if pair_acc:
                        break
                    di = float(vi) - xi
                    for vj in vj_vals:
                        dj = float(vj) - xj
                        if abs(di) <= 1e-15 and abs(dj) <= 1e-15:
                            continue
                        pub = 1.0 + di + dj
                        if pub <= 1e-12:
                            continue

                        T = b + di * base_i + dj * base_j
                        ti = 2 * i
                        tj = 2 * j
                        ij = i + j
                        T[ti] += di * di
                        T[tj] += dj * dj
                        T[ij] += 2.0 * di * dj

                        mmax = float(T.max())
                        sc_try = 2.0 * n * mmax / (pub * pub)
                        if sc_try < sc - tol:
                            inv_s2 = 1.0 / (pub * pub)
                            new_b = T * inv_s2
                            new_f = f / pub
                            new_f[i] = float(vi) / pub
                            new_f[j] = float(vj) / pub

                            f = new_f
                            b = new_b
                            sc = float(sc_try)
                            improved = True
                            pair_acc = True
                            if sc < best_sc - tol:
                                best_sc = sc
                                best_f = f.copy()
                                best_b = b.copy()
                            break

        b = _conv(f)
        sc = 2.0 * n * float(b.max())
        if sc < best_sc - tol:
            best_sc = sc
            best_f = f.copy()
            best_b = b.copy()

        if improved:
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 2:
            break

    best_f = _norm(best_f)
    best_b = _conv(best_f)
    best_sc = 2.0 * n * float(best_b.max())
    return best_f, best_b, best_sc


def _lp_search(f0, b0, stop, seed=0, lp_timeout=2.0, coord_interval=10):
    f = _norm(f0)
    n = len(f)
    if n <= 1:
        f = np.ones(1, dtype=np.float64)
        b = _conv(f)
        return f, b, 2.0 * float(b.max())

    b = _conv(f)
    sc = 2.0 * n * float(b.max())
    best_f = f.copy()
    best_b = b.copy()
    best_sc = sc

    rng = np.random.default_rng(seed)
    t_step = 0.35
    stall = 0
    it = 0
    tol = 1e-14

    while time.monotonic() < stop:
        it += 1
        if it % coord_interval == 0:
            local_stop = min(stop, time.monotonic() + 3.5)
            f, b, sc = _local_polish(
                f, b, local_stop,
                seed=seed + it,
                max_sweeps=2,
                pair_count=20 if n <= 512 else 8,
                fine_after=0,
            )
            if sc < best_sc - tol:
                best_sc = sc
                best_f = f.copy()
                best_b = b.copy()

        rem = stop - time.monotonic()
        if rem < 0.8:
            break

        M = float(b.max())
        if not math.isfinite(M) or M <= 1e-18:
            break

        to = min(lp_timeout, rem)
        g = _lp_direction(f, n, timeout=to)
        if g is None:
            stall += 2
            if stall > 8:
                f2, b2, sc2 = _perturb(f, rng, strength=0.35 + 0.05 * min(stall // 3, 12))
                if math.isfinite(sc2) and sc2 < 10.0 * best_sc + 1.0:
                    f, b, sc = f2, b2, sc2
                    if sc < best_sc - tol:
                        best_sc = sc
                        best_f = f.copy()
                        best_b = b.copy()
                    stall = 0
                    t_step = 0.35
                else:
                    f = best_f.copy()
                    b = best_b.copy()
                    sc = best_sc
                    stall = 0
            continue

        fg = _conv(f, g)
        fgmax = float(fg.max())
        if not math.isfinite(fgmax) or fgmax > M + max(1e-10, 1e-4 * M):
            stall += 1
            if stall > 6:
                f2, b2, sc2 = _perturb(f, rng, strength=0.3 + 0.05 * min(stall // 3, 10))
                if math.isfinite(sc2) and sc2 < 10.0 * best_sc + 1.0:
                    f, b, sc = f2, b2, sc2
                    if sc < best_sc - tol:
                        best_sc = sc
                        best_f = f.copy()
                        best_b = b.copy()
                    stall = 0
                    t_step = 0.35
                else:
                    f = best_f.copy()
                    b = best_b.copy()
                    sc = best_sc
                    stall = 0
            continue

        gg = _conv(g)
        tb = _line_search(f, g, b, fg, gg, n, sc, t_step)

        if tb > 1e-6:
            old_sc = sc
            fnew = (1.0 - tb) * f + tb * g
            fnew = np.clip(fnew, 0.0, 1000.0)
            f = _norm(fnew)
            b = _conv(f)
            sc = 2.0 * n * float(b.max())

            if sc < old_sc - tol:
                t_step = min(1.0, tb * 1.25)
                stall = 0
                if sc < best_sc - tol:
                    best_sc = sc
                    best_f = f.copy()
                    best_b = b.copy()
            else:
                t_step = max(0.02, t_step * 0.6)
                stall += 1
        else:
            stall += 1

        if stall > 10:
            f2, b2, sc2 = _perturb(f, rng, strength=0.35 + 0.05 * min(stall // 3, 12))
            if math.isfinite(sc2) and sc2 < 10.0 * best_sc + 1.0:
                f, b, sc = f2, b2, sc2
                if sc < best_sc - tol:
                    best_sc = sc
                    best_f = f.copy()
                    best_b = b.copy()
                stall = 0
                t_step = 0.35
            else:
                f = best_f.copy()
                b = best_b.copy()
                sc = best_sc
                stall = 0

    best_f = _norm(best_f)
    best_b = _conv(best_f)
    best_sc = 2.0 * n * float(best_b.max())
    return best_f, best_b, best_sc


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(352, dtype=np.float64)
    if p.size == 0:
        p = np.ones(352, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s < 1e-12:
        p = np.ones(p.size if p.size > 0 else 352, dtype=np.float64)
        s = float(p.size)
    return p / s


def _make_inits(L, parent, rng):
    L = int(L)
    idx = np.arange(L, dtype=np.float64)
    denom = max(1.0, float(L - 1))
    inits = []

    def add(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size != L:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if float(a.sum()) <= 1e-12:
            return
        inits.append(a)

    if parent is not None and len(parent) >= 2:
        xold = np.linspace(0.0, 1.0, len(parent), dtype=np.float64)
        xnew = np.linspace(0.0, 1.0, L, dtype=np.float64)
        p = np.interp(xnew, xold, parent)
        add(p)
        add(p[::-1])
        add(p * (1.0 + 0.5 * idx / denom))
        add(p * (1.0 - 0.4 * idx / denom))
        add(p * (1.0 + 0.5 * np.cos(2.0 * np.pi * idx / float(L) + 0.7)))

    if parent is not None and len(parent) == L:
        add(parent)

    for c, r in ((0.35, 2.0), (0.45, 1.6), (0.55, 1.3)):
        a = np.ones(L, dtype=np.float64)
        m = int(round(c * L))
        m = max(1, min(L - 1, m))
        a[:m] = r
        add(a)

    for g in (1.5, 3.0):
        add(np.exp(-g * idx / float(L)))
        add(np.exp(g * idx / float(L)))

    add(1.0 + 0.5 * np.sin(np.pi * idx / float(L)))
    return inits


def _try_length(L, parent, stop, seed, current_best_sc):
    if time.monotonic() > stop:
        return None
    L = int(L)
    rng = np.random.default_rng(seed)
    inits = _make_inits(L, parent, rng)

    fbest = None
    bbest = None
    bsc = math.inf

    for a in inits:
        if time.monotonic() > stop:
            break
        f = _norm(a)
        b = _conv(f)
        sc = 2.0 * L * float(b.max())
        if math.isfinite(sc) and sc < bsc:
            bsc = sc
            fbest = f
            bbest = b

    if fbest is None or not math.isfinite(bsc):
        return None
    if bsc > current_best_sc + 0.03:
        return None

    if bsc < current_best_sc and time.monotonic() > stop - 2.0:
        return fbest, bbest, bsc

    f, b, sc = _local_polish(
        fbest, bbest, stop,
        seed=seed + 1000003 + L,
        max_sweeps=30,
        pair_count=50 if L <= 512 else 18,
        fine_after=2,
    )
    if math.isfinite(sc) and (sc < current_best_sc or bsc < current_best_sc):
        return f, b, sc
    return None


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    deadline = t0 + budget - 18.0

    parent = _get_parent()
    best_f = parent.copy()
    best_b = _conv(best_f)
    best_sc = 2.0 * len(best_f) * float(best_b.max())

    rng_global = np.random.default_rng(seed)

    if budget < 60:
        if time.monotonic() < deadline - 2:
            best_f, best_b, best_sc = _local_polish(
                best_f, best_b, min(deadline, time.monotonic() + budget - 10),
                seed=seed, max_sweeps=100, pair_count=30, fine_after=1,
            )
        try:
            exact = float(evaluate_sequence([float(x) for x in best_f]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact):
            best_sc = exact
            return [float(x) for x in best_f]
        return [1.0]

    # Initial local polish of parent to give a better shape before LP/length trials.
    stop_parent0 = min(deadline, t0 + max(20.0, 0.02 * budget))
    if time.monotonic() < stop_parent0 - 2.0:
        best_f, best_b, best_sc = _local_polish(
            best_f, best_b, stop_parent0,
            seed=seed + 123,
            max_sweeps=80,
            pair_count=70,
            fine_after=2,
        )

    # Warmed-up LP optimization on the parent length.
    parent_lp_stop = min(deadline, time.monotonic() + max(45.0, 0.07 * budget))
    if time.monotonic() < parent_lp_stop - 10.0:
        best_f, best_b, best_sc = _lp_search(
            best_f, best_b, parent_lp_stop,
            seed=seed + 45678,
            lp_timeout=1.4 if len(best_f) <= 512 else 2.2,
            coord_interval=10,
        )

    # Explore nearby and larger lengths using the current best shape as template.
    lengths = [336, 384, 416, 448, 480, 512, 576, 640, 768, 1024]
    alt_stop = min(deadline, time.monotonic() + max(70.0, 0.18 * budget))
    seen = {len(best_f)}
    for L in lengths:
        if time.monotonic() > alt_stop - 8.0:
            break
        if int(L) in seen:
            continue
        seen.add(int(L))
        per_stop = min(alt_stop, time.monotonic() + 14.0)
        cand = _try_length(L, best_f, per_stop, seed=seed + 909101 + int(L), current_best_sc=best_sc)
        if cand is not None:
            cf, cb, cs = cand
            if math.isfinite(cs) and cs < best_sc - 1e-14:
                best_f, best_b, best_sc = cf, cb, cs

    # Deep optimization of the best candidate.
    deep_stop = min(deadline - 110.0, time.monotonic() + max(100.0, 0.62 * budget))
    if time.monotonic() < deep_stop - 20.0:
        nbest = len(best_f)
        best_f, best_b, best_sc = _lp_search(
            best_f, best_b, deep_stop,
            seed=seed + 777777 + nbest,
            lp_timeout=1.3 if nbest <= 512 else (1.9 if nbest <= 800 else 2.6),
            coord_interval=8 if nbest <= 512 else 12,
        )

    # Final concise recheck on slightly different nearby lengths if ample time remains.
    if time.monotonic() < deadline - 160.0:
        cand_lens = [len(best_f) // 2, len(best_f), len(best_f) + 64, len(best_f) + 128]
        for L in cand_lens:
            if time.monotonic() > deadline - 70.0:
                break
            L = int(L)
            if L <= 8 or seen & {L}:
                continue
            seen.add(L)
            per_stop = min(deadline - 70.0, time.monotonic() + 12.0)
            cand = _try_length(L, best_f, per_stop, seed=seed + 555888 + L, current_best_sc=best_sc)
            if cand is not None:
                cf, cb, cs = cand
                if math.isfinite(cs) and cs < best_sc - 1e-14:
                    best_f, best_b, best_sc = cf, cb, cs

    # Final polish until just before deadline.
    final_stop = min(deadline - 6.0, t0 + budget - 8.0)
    if time.monotonic() < final_stop - 3.0:
        best_f, best_b, best_sc = _local_polish(
            best_f, best_b, final_stop,
            seed=seed + 9999999,
            max_sweeps=5000,
            pair_count=110 if len(best_f) <= 512 else (45 if len(best_f) <= 900 else 18),
            fine_after=1,
        )

    # If time remains, try one perturbation-rescue polish to escape the final local basin.
    rescue_stop = min(deadline - 6.0, t0 + budget - 8.0)
    if time.monotonic() < rescue_stop - 25.0:
        rng2 = np.random.default_rng(seed + 20240521)
        pf, pb, psc = _perturb(best_f, rng2, strength=0.55)
        if math.isfinite(psc) and psc < 2.5 * best_sc:
            rf, rb, rsc = _local_polish(
                pf, pb, rescue_stop,
                seed=seed + 31337,
                max_sweeps=3000,
                pair_count=90 if len(pf) <= 512 else 30,
                fine_after=1,
            )
            if math.isfinite(rsc) and rsc < best_sc - 1e-14:
                best_f, best_b, best_sc = rf, rb, rsc

    best_f = _norm(best_f)
    out = [float(min(1000.0, max(0.0, x))) for x in best_f]
    if not out or any(not math.isfinite(x) for x in out):
        return [1.0]

    try:
        exact = float(evaluate_sequence(out))
    except Exception:
        exact = math.inf

    if not math.isfinite(exact):
        fallback = [1.0] * len(best_f)
        try:
            exact2 = float(evaluate_sequence(fallback))
        except Exception:
            exact2 = math.inf
        if math.isfinite(exact2):
            return fallback
        return [1.0]

    return out
