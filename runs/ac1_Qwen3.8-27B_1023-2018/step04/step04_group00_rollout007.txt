def _coord_adjust(p, c, i, n, S):
    """Find optimal delta for coordinate i."""
    v = p[i]
    # Affected range: k from i to min(i+n-1, 2n-2)
    start = i
    end = min(i + n, 2*n - 1)  # exclusive
    Ls = end - start
    if Ls <= 0:
        return v, 0.0
    
    # Cross-correlation for affected positions
    # c[k] changes by 2*delta*a[k-i] + delta^2 if k==2i
    # a[k-i] for k in [start, end): a[0], a[1], ..., a[Ls-1]
    infl = p[:Ls]  # a[0], a[1], ..., a[Ls-1]
    base = c[start:end].copy()
    
    # Unaffected max
    max_const = 0.0
    if start > 0:
        max_const = max(max_const, c[:start].max())
    if end < 2*n - 1:
        max_const = max(max_const, c[end:].max())
    
    # Diagonal index within affected range
    diag_local = 2*i - start  # position of 2i within [start, end)
    has_diag = 0 <= diag_local < Ls
    
    # For a given delta, new max in affected range:
    # base[j] + 2*delta*infl[j] + delta^2 * (j == diag_local)
    # score = 2n * max(max_const, max_affected) / (S + delta)^2
    
    def new_max(delta):
        vals = base + 2*delta*infl
        if has_diag:
            vals[diag_local] += delta*delta
        m = vals.max()
        if max_const > m:
            m = max_const
        return m
    
    def obj(delta):
        m = new_max(delta)
        s = S + delta
        if s <= 0:
            return np.inf
        return 2*n*m / (s*s)
    
    # Current value
    cur = obj(0.0)
    
    # Find optimal delta using golden section or bisection
    # Search in [0, x_hi] (we only increase or allow decrease to 0)
    # Actually delta can be negative (decreasing v) but v+delta >= 0
    lo = -v + 1e-15  # v + delta >= 0
    x_hi = max(20*v, 60*S/n, 0.05*S)
    x_hi = min(x_hi, 10*S)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    a_g = lo
    b_g = x_hi
    c_g = b_g - gr*(b_g - a_g)
    d_g = a_g + gr*(b_g - a_g)
    fc = obj(c_g)
    fd = obj(d_g)
    
    for _ in range(40):
        if b_g - a_g < 1e-12 * max(1, abs(c_g), abs(d_g)):
            break
        if fc < fd:
            b_g = d_g
            d_g = c_g
            fd = fc
            c_g = b_g - gr*(b_g - a_g)
            fc = obj(c_g)
        else:
            a_g = c_g
            c_g = d_g
            fc = fd
            d_g = a_g + gr*(b_g - a_g)
            fd = obj(d_g)
    
    best_delta = (a_g + b_g) / 2
    best_val = obj(best_delta)
    
    if best_val < cur - 1e-15:
        new_v = v + best_delta
        return new_v, best_delta
    return v, 0.0

                    dval = base[diag_local] + w[valid] * infl[diag_local] + w[valid] * w[valid]
                    maff = np.maximum(maff, dval)

                maff = np.maximum(maff, max_const)
                vals = (2.0 * n_c * maff) / (denom[valid] ** 2)

                if not np.any(np.isfinite(vals)):
                    continue
                imin = int(np.argmin(vals))
                best_x = float(pts[valid][imin])
                best_val = float(vals[imin])

                # Parabolic refinement around best
                if G > 2 and imin > 0 and imin < len(pts[valid]) - 1:
                    p1 = float(pts[valid][imin - 1])
                    p2 = float(pts[valid][imin])
                    p3 = float(pts[valid][imin + 1])
                    v1 = float(vals[imax - 1]) if False else float(vals[imax - 1]) if False else None
                    # Get actual values
                    v1v = float(vals[imax - 1]) if imax >= 1 else best_val
                    # Simpler: just use the grid
                    pass

                # Quick 1D refinement: 8 points around best
                dw_ft = x_hi / G
                lo_r = max(0.0, best_x - 2.0 * dw_ft)
                hi_r = min(x_hi, best_x + 2.0 * dw_ft)
                if hi_r > lo_r:
                    fine_pts = np.linspace(lo_r, hi_r, 16, dtype=np.float64)
                    fine_pts = np.unique(np.concatenate((fine_pts, [best_x])))
                    wf = fine_pts - v
                    df = S + wf
                    vf = df > 1e-14
                    if np.any(vf):
                        afff = base[None, :] + wf[vf, None] * infl[None, :]
                        mafff = np.max(afff, axis=1)
                        if 0 <= diag_local < Ls:
                            dv = base[diag_local] + wf[vf] * infl[diag_local] + wf[vf] ** 2
                            mafff = np.maximum(mafff, dv)
                        mafff = np.maximum(mafff, max_const)
                        valsf = (2.0 * n_c * mafff) / (df[vf] ** 2)
                        if np.any(np.isfinite(valsf)):
                            im2 = int(np.argmin(valsf))
                            if float(valsf[im2]) < best_val:
                                best_x = float(fine_pts[vf][im2])
                                best_val = float(valsf[im2])

                if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 10.0 * S:
                    w_upd = best_x - v
                    if abs(w_upd) > 1e-15:
                        p_c[i] = best_x
                        S += w_upd
                        c[start:end] += w_upd * infl
                        di = 2 * i
                        if 0 <= di < 2 * n_c - 1:
                            c[di] += w_upd * w_upd
                        cur_score = best_val
                        improved = True

            # Renormalize
            if S > 1e-300:
                inv = 1.0 / S
                p_c *= inv
                c *= inv * inv
            S = 1.0

            # Recompute from scratch periodically
            if sweep % 10 == 9:
                c = _fft_conv(p_c)
                np.maximum(c, 0.0, out=c)

            new_sc = float(2.0 * n_c * c.max())
            if math.isfinite(new_sc) and new_sc < best_local_score - 1e-14:
                best_local_score = new_sc
                best_local = p_c.copy()
                consider(best_local)

            if improved:
                flat_count = 0
            else:
                flat_count += 1
                if flat_count >= 4:
                    # Perturb
                    p_c = p_c * np.exp(rng.normal(0, 0.03, n_c))
                    p_c = np.maximum(p_c, 0)
                    ss2 = p_c.sum()
                    if ss2 > 0:
                        p_c /= ss2
                    c = _fft_conv(p_c)
                    cur_score = float(2.0 * n_c * c.max())
                    flat_count = 0

        return best_local

    # --- Main optimization loop ---
    p_cur = p.copy()

    # Phase 1: Coarse L-BFGS
    stop1 = t0 + 60.0
    if time.monotonic() < stop1 - 5:
        p_cur = _lbfgs(p_cur, 0.06, 30, stop1)

    # Phase 2: Coordinate descent
    stop2 = t0 + 250.0
    p_cur = _coord_descent(p_cur, stop2, max_sweeps=80, seed_cd=int(seed))
    p_cur = best_arr.copy()

    # Phase 3: Fine L-BFGS
    stop3 = t0 + 350.0
    if time.monotonic() < stop3 - 5:
        p_cur = _lbfgs(p_cur, 0.015, 25, stop3)

    # Phase 4: Coordinate descent
    stop4 = t0 + 550.0
    p_cur = _coord_descent(p_cur, stop4, max_sweeps=100, seed_cd=int(seed + 1))
    p_cur = best_arr.copy()

    # Phase 5: Very fine L-BFGS
    stop5 = t0 + 650.0
    if time.monotonic() < stop5 - 5:
        p_cur = _lbfgs(p_cur, 0.004, 30, stop5)

    # Phase 6: Coordinate descent
    stop6 = t0 + 850.0
    p_cur = _coord_descent(p_cur, stop6, max_sweeps=120, seed_cd=int(seed + 2))
    p_cur = best_arr.copy()

    # Phase 7: Fourier mode exploration
    stop7 = t0 + 1000.0
    for k_mode in [1, 2, 3, 5, 7, 11, 13, 17, 23]:
        if time.monotonic() > stop7 - 30:
            break
        p_f = best_arr.copy()
        i_arr = np.arange(n, dtype=np.float64)
        for sign in [1, -1]:
            for fr in [0.02, 0.05]:
                mode = sign * fr * np.cos(2.0 * math.pi * k_mode * i_arr / n)
                p_try = np.maximum(p_f + mode, 0.0)
                ss3 = p_try.sum()
                if ss3 > 0.01:
                    p_try /= ss3
                    consider(p_try)
                    if _score_fft(p_try) < _score_fft(best_arr) + 0.01:
                        # Short coord descent on perturbed
                        stop_short = min(stop7 - 10, time.monotonic() + 40)
                        if time.monotonic() < stop_short:
                            p_td = _coord_descent(p_try, stop_short, max_sweeps=20,
                                                  seed_cd=int(seed + k_mode * 100))
                            consider(p_td)

    # Phase 8: Final fine L-BFGS + coordinate descent
    stop8 = deadline - 10.0
    if time.monotonic() < stop8 - 20 and best_arr.size == n:
        p_cur = _lbfgs(best_arr, 0.001, 20, min(stop8 - 30, time.monotonic() + 50))
        p_cur = _coord_descent(p_cur, stop8, max_sweeps=60, seed_cd=int(seed + 999))

    # --- Final validation ---
    try:
        out_arr = np.clip(best_arr, 0.0, 1000.0).ravel()
        out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
        if out_arr.size == 0 or out_arr.sum() < 0.01:
            out_arr = np.ones(64, dtype=np.float64) / 64.0
        out_list = [float(x) for x in out_arr]
        final_sc = evaluate_sequence(out_list)
        if not math.isfinite(final_sc):
            out_list = [float(x) for x in np.ones(64) / 64.0]
            final_sc = evaluate_sequence(out_list)
        return out_list
    except Exception:
        return [1.0]
