                    g_res2, g_sc2 = _grad_refine(pert, stop_p2, seed=seed + 60000, max_iter=15000)
                except Exception:
                    g_res2, g_sc2 = pert, _score(pert)
                _try_improve(g_res2, g_sc2, stop_p2)
    
    # Phase 3: Additional LP on best (10% time)
    phase3_end = t0 + budget * 0.90
    if time.monotonic() < phase3_end - 10.0:
        stop_p3 = min(phase3_end - 5.0, deadline - 30.0)
        try:
            res3, sc3 = _lp_loop(best_seq.copy(), n, stop_p3, seed + 77777)
        except Exception:
            res3, sc3 = best_seq, best_score
        _try_improve(res3, sc3, stop_p3)
    
    # Phase 4: CD polish (10% time)
    phase4_end = deadline - 10.0
    if time.monotonic() < phase4_end - 5.0:
        try:
            f_c, sc_c = _cd_polish(best_seq, phase4_end, max_outer=50, seed=seed * 3 + 1)
        except Exception:
            f_c, sc_c = best_seq.copy(), best_score
        _try_improve(f_c, sc_c, phase4_end)
        
        # One more gradient pass
        if time.monotonic() < phase4_end - 10.0:
            try:
                g_final, g_sc_final = _grad_refine(best_seq, phase4_end, seed=seed + 88888, max_iter=10000)
            except Exception:
                g_final, g_sc_final = best_seq, best_score
            _try_improve(g_final, g_sc_final, phase4_end)
    
    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf
    
    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0] * 10
        return [float(max(0.0, min(1000.0, x))) for x in fallback]
    
    best_score = min(best_score, final_sc)
    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    if not result:
        result = [1.0]
    if any(math.isnan(x) or math.isinf(x) for x in result):
        return [float(max(0.0, min(1000.0, x))) for x in fallback]
    
    return result
