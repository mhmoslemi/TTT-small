            p /= ps
        consider(p)

        lse_end = min(phase_end - 60.0, time.monotonic() + 0.45 * (phase_end - time.monotonic()))
        if lse_end > time.monotonic() + 20.0:
            try:
                p = _lse_optimize(p, lse_end, consider=consider)
            except Exception:
                pass

        if do_lp and n_target <= 4096:
            lp_end = min(phase_end - 30.0, time.monotonic() + 0.25 * (phase_end - time.monotonic()))
            if lp_end > time.monotonic() + 15.0:
                try:
                    p = _lp_phase(p, lp_end, max_iters=25, consider=consider)
                except Exception:
                    pass

        coord_end = min(phase_end - 5.0, time.monotonic() + 0.30 * (phase_end - time.monotonic()))
        if coord_end > time.monotonic() + 15.0:
            try:
                p = _coord_polish(p, coord_end, consider=consider, seed=sd, G=56)
            except Exception:
                pass
        consider(p)

    n_parent = int(best_arr.size)

    # Phase 1: Optimize parent length
    phase1_end = min(hard, t0 + 400.0)
    if phase1_end > t0 + 60.0:
        run_phase(best_arr, n_parent, phase1_end, int(seed + 100), do_lp=True)

    # Phase 2: Try 2x parent length
    n2 = min(2 * n_parent, 4096)
    if n2 > n_parent:
        phase2_end = min(hard, time.monotonic() + 350.0)
        if phase2_end > time.monotonic() + 80.0:
            p2_init = _resample(best_arr, n2)
            rng2 = np.random.default_rng(int(seed + 200))
            p2_init = p2_init * np.exp(rng2.normal(0.0, 0.02, n2))
            p2_init = np.maximum(p2_init, 0.0)
            p2_s = float(p2_init.sum())
            if p2_s > 0:
                p2_init /= p2_s
            run_phase(p2_init, n2, phase2_end, int(seed + 300), do_lp=True)

    # Phase 3: Try 8192
    n3 = 8192
    phase3_end = min(hard, time.monotonic() + 280.0)
    if phase3_end > time.monotonic() + 80.0:
        p3_init = _resample(best_arr, n3)
        rng3 = np.random.default_rng(int(seed + 400))
        p3_init = p3_init * np.exp(rng3.normal(0.0, 0.015, n3))
        p3_init = np.maximum(p3_init, 0.0)
        p3_s = float(p3_init.sum())
        if p3_s > 0:
            p3_init /= p3_s
        run_phase(p3_init, n3, phase3_end, int(seed + 500), do_lp=False)

    # Phase 4: Final polish on best
    final_end = min(hard - 5.0, time.monotonic() + 120.0)
    if final_end > time.monotonic() + 30.0:
        n_best = int(best_arr.size)
        p_final = best_arr.copy()
        lse_e = min(final_end - 30.0, time.monotonic() + 50.0)
        if lse_e > time.monotonic() + 15.0 and n_best <= 8192:
            try:
                p_final = _lse_optimize(p_final, lse_e, consider=consider)
            except Exception:
                pass
        coord_e = min(final_end - 3.0, time.monotonic() + 60.0)
        if coord_e > time.monotonic() + 10.0:
            try:
                p_final = _coord_polish(p_final, coord_e, consider=consider, seed=int(seed + 999), G=48)
            except Exception:
                pass
        consider(p_final)

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        sc = float(evaluate_sequence(ls))
        if not math.isfinite(sc):
            ls = [1.0]
        return ls
    except Exception:
        return [1.0]
