import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _conv_cross(a, b):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    FA = rfft(a, fsize)
    FB = rfft(b, fsize)
    c = irfft(FA * FB, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _build_A(active_idx, f, n):
    K = len(active_idx)
    A = np.zeros((K, n))
    for i, k in enumerate(active_idx):
        # A[i, l] = f[k-l] if 0 <= k-l < n
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for l in range(lo, hi + 1):
            A[i, l] = f[k - l]
    return A

def _lp_solve(f, n, M, active_idx, time_limit=5.0):
    K = len(active_idx)
    A = _build_A(active_idx, f, n)
    c = -np.ones(n)
    b_ub = np.full(K, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                     bounds=[(0, None)] * n,
                     method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs

def _line_search_quad(f, g, n, stop_time):
    """Using quadratic structure for exact line search."""
    ff = _conv_full(f)
    fg = _conv_cross(f, g)
    gg = _conv_full(g)
    
    def obj(t):
        val = (1.0 - t)**2 * ff + 2.0 * t * (1.0 - t) * fg + t**2 * gg
        return float(val.max())
    
    # Golden section
    gr = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0

        else:
            lo = m1
        tm = (lo + hi) / 2.0
