import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len

def _conv(a):
    """Fast full convolution using FFT."""
    n = len(a)
    fsize = next_fast_len(2*n - 1)
    F = rfft(a, fsize)
    c = irfft(F*F, fsize)[:2*n-1]
    return c

def _score(a):
    """Compute the exact score."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s*s)

def _build_A(f, n):
    """Build the constraint matrix for the LP.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    for k = 0, ..., 2n-2
    """
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_direction(f, n, M, A=None, timeout=3.0):
    """Find LP direction: max sum(g) s.t. Ag <= M, g >= 0."""
    if A is None:

    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:

                     bounds=bounds, method='highs',
                     options={'time_limit': timeout})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:  # 0=optimal, 3=time limit
        return None
    g = np.maximum(res.x, 0)
    if g.sum() < 1e-15:
        return None
    return g

                delta = f[k] - (f[k] if k not in range(n) else 0)

    return best_f, best_sc


def _lp_iteration(f, n, deadline):
    c = _conv(f)
    M = float(c.max())
    s = float(f.sum())
    cur = 2.0 * n * M / (s * s)
    if M < 1e-18:
        return f, cur, 1
    g = _lp_dir(f, n, M)
    if g is None:
        return f, cur, 0
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return f, cur, 0
    g = g * (s / g_sum)
    dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
    if dist < 1e-12:
        return f, cur, 0
    # line search
    steps = [0.35, 0.2, 0.5, 0.1, 0.7, 0.05]
    best_f_new = f.copy()
    best_sc = cur
    for t in steps:
        fn = np.clip((1 - t) * f + t * g, 0, None)
        if fn.sum() < 0.01:
            continue
        sc = _score(fn)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_f_new = fn
    return best_f_new, best_sc, 1


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    deadline = t0 + budget - 15.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(400, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(400, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    n_parent = len(parent)

    # Fallback
    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(200)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc
    rng = np.random.default_rng(seed)

    def _try_update(cand, sc):
        nonlocal best_seq, best_score
        if cand is None or len(cand) == 0:
            return
        cand = np.clip(np.asarray(cand, dtype=np.float64).ravel(), 0, 1000)
        if cand.sum() < 0.01 or not np.all(np.isfinite(cand)):
            return
        if sc < best_score - 1e-13:
            try:
                exact = float(evaluate_sequence([float(x) for x in cand]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-13:
                best_seq = cand.copy()
                best_score = exact

    # === Phase 1: LP-guided descent on parent length ===
    n = n_parent
    phase1_end = t0 + budget * 0.55

    # Initializations
    i_arr = np.arange(n, dtype=np.float64)
    ctr = (n - 1) / 2.0
    inits = [
        parent.copy(),
        0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / n)),
        np.maximum(0.0, 1.0 - ((i_arr - ctr) / (0.45 * n)) ** 2),
        np.exp(-0.5 * ((i_arr - 0.2 * n) / (0.12 * n)) ** 2) + np.exp(-0.5 * ((i_arr - 0.8 * n) / (0.12 * n)) ** 2),
        np.clip(parent + rng.normal(0, 0.08 * max(1.0, np.mean(parent)), n), 0, None) + 1e-5,
        np.where(i_arr % 3 == 0, 1.5, 0.6) * (0.5 + 0.5 * np.cos(2.0 * np.pi * i_arr / (n // 3))),
    ]

    # Deduplicate
    inits_dedup = []
    seen_bytes = []
    for ix in inits:
        if ix.sum() < 0.01:
            continue
        b = ix.tobytes()
        if b not in seen_bytes:
            seen_bytes.append(b)
            inits_dedup.append(ix)

    time_per_init = max(12.0, (phase1_end - t0) / max(1, len(inits_dedup)))
    for idx, init in enumerate(inits_dedup):
        if time.monotonic() > phase1_end - 10.0:
            break
        if best_score < 1.50:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        f = np.clip(init, 0, 1000).copy()
        if f.sum() < 0.01:
            continue
        f = f / f.sum()

        # LP loop
        t_lp = 0.25
        stall = 0
        max_stall = 200
        for _it in range(5000):
            if time.monotonic() > stop:
                break
            f, sc, moved = _lp_iteration(f, n, stop)
            if sc < best_score - 1e-13:
                _try_update(f, sc)
            if not moved:
                stall += 1
                if stall >= 10:
                    # CD burst
                    if time.monotonic() < stop - 2.0:
                        f_c, sc_c = _cd_polish(f, stop - 1.0, max_sweeps=5, seed=seed + idx * 100 + _it)
                        if sc_c < best_score - 1e-13:
                            _try_update(f_c, sc_c)
                        f = f_c if sc_c < _score(f) else f
                    if stall >= max_stall:
                        break
                if stall >= 50:
                    # perturb
                    f = np.clip(f + rng.normal(0, 0.02 * max(1.0, np.mean(f)), n), 0, None)
                    if f.sum() < 0.01:
                        f = best_seq.copy()
                    f = f / f.sum()
                    t_lp = 0.25
                    stall = 0
                    max_stall = min(max_stall + 100, 500)
            else:
                stall = 0
                t_lp = min(0.5, t_lp * 1.02)

        # CD after LP
        if time.monotonic() < stop:
            f_c, sc_c = _cd_polish(f, stop, max_sweeps=15, seed=seed + 500 + idx)
            _try_update(f_c, sc_c)

    # === Phase 2: Extended CD on best found ===
    phase2_end = t0 + budget * 0.72
    if time.monotonic() < phase2_end - 5.0 and len(best_seq) == n:
        f_c, sc_c = _cd_polish(best_seq, phase2_end - 2.0, max_sweeps=200, seed=seed * 3 + 1)
        _try_update(f_c, sc_c)

    # === Phase 3: More LP iterations on best ===
    phase3_end = t0 + budget * 0.82
    if time.monotonic() < phase3_end - 10.0 and _endure_check(deadline):
        f = best_seq.copy() / max(best_seq.sum(), 1e-10)
        stall = 0
        for _it in range(3000):
            if time.monotonic() > phase3_end - 3.0:
                break
            f, sc, moved = _lp_iteration(f, n, phase3_end)
            if sc < best_score - 1e-13:
                _try_update(f, sc)
            if not moved:
                stall += 1
                if stall >= 30:
                    f_c, sc_c = _cd_polish(f, phase3_end - 1.0, max_sweeps=8, seed=seed + 9000 + _it)
                    if sc_c < best_score:
                        _try_update(f_c, sc_c)
                        f = f_c
                    stall = 0
                    if _it % 100 == 0:
                        f = np.clip(f + rng.normal(0, 0.015, n), 0, None)
                        f = f / f.sum()
            else:
                stall = 0

    # === Phase 4: Try nearby lengths ===
    phase4_end = t0 + budget * 0.90
    for dl in [-8, 8, -16, 16, 32]:
        L = n + dl
        if L < 100 or time.monotonic() > phase4_end - 8.0:
            continue
        init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
        stop_l = min(phase4_end - 3.0, time.monotonic() + 20.0)
        f = init_l / max(init_l.sum(), 1e-10)
        for _it in range(500):
            if time.monotonic() > stop_l:
                break
            f, sc, moved = _lp_iteration(f, L, stop_l)
            if sc < best_score - 1e-13:
                _try_update(f, sc)
            if not moved and _it > 50:
                break
        if time.monotonic() < stop_l:
            f_c, sc_c = _cd_polish(f, stop_l, max_sweeps=10, seed=seed + L)
            _try_update(f_c, sc_c)

    # === Phase 5: Final intensive CD ===
    final_end = deadline - 4.0
    if time.monotonic() < final_end - 3.0:
        f_c, sc_c = _cd_polish(best_seq, final_end, max_sweeps=500, seed=seed * 7 + 42)
        _try_update(f_c, sc_c)

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0, 1.0, 1.0]
        best_seq = fallback

    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    if not result or any(math.isnan(x) or math.isinf(x) for x in result):
        return [float(x) for x in fallback]

    return result
