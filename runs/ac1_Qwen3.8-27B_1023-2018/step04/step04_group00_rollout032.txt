def _subgradient_polish(p, stop_time, consider=None, max_iters=5000):

    n = int(p.size)

    if s < 1e-300:
        return p
    p = p / s
    
    Ltot = 2 * n - 1
    best_p = p.copy()
    best_score = float(2.0 * n * _conv_self(p).max())
    step_decay = 0.999
    base_step = 0.01
    
    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break
        
        c = _conv_self(p)
        cmax = float(c.max())
        # Find active lags (within 1% of max)
        threshold = cmax * 0.995
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            break

        # For convolution: (a*a)[k] = sum_i a[i]*a[k-i]
        # d/dx_i (a*a)[k] = a[k-i] + a[k-i] = ...

        # More carefully: (a*a)[k] = sum_i a[i] * a[k-i]

                # (a*a)[k] = sum_i a[i]*a[k-i] where 0<=i<n and 0<=k-i<n
                # = sum_i a[i]*a[k-i] where max(0,k-n+1) <= i <= min(k, n-1)
                # d/dx_j: appears when i=j (gives a[k-j]) or when k-i=j i.e. i=k-j (gives a[j])
                # But we need 0<=k-j<n and 0<=j<n

                if 0 <= k-j < n and (k-j) != j:

                elif (k-j) == j and 0 <= j < n:
                    dval += p[j]  # count only once? No, the two terms are separate
                    # Actually when i=j and i=k-j (same term), we get 2*a[j]
                    # Let me redo: (a*a)[k] = sum_i a[i]*a[k-i]

def _get_parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()

def _conv_self(a):
    n = a.size
    if n <= 512:
        c = np.convolve(a, a, mode="full")
    else:
        L = 2 * n - 1
        size = next_fast_len(L)
        F = rfft(a, size)
        c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _cross_corr(p, w):
    n = p.size
    L = 3 * n - 2
    size = next_fast_len(L)
    F = rfft(p[::-1], size)
    G = rfft(w, size)
    c = irfft(F * G, size)[:L]
    q = c[n - 1:2 * n - 1]
    np.maximum(q, 0.0, out=q)
    return q


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _score_fft(a):

        return math.inf
    c = _conv_self(a)
    m = float(c.max())

    return float(2.0 * n * m / (s * s))

def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == 1:
        return np.full(n, float(src[0]) / n)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m) + 0.5) / m
        fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s

    n = int(p.size)
    if n == 0:
        return p

    if s < 1e-300:
        return p

        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)

            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0

        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))

            except Exception:

                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, "status", 1) == 0:

    if fine_only:

    else:

            (0.12, 0.08),

def _subgradient_polish(p, stop_time, consider=None, max_iters=3000):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    best_p = p.copy()
    best_score = float(2.0 * n * _conv_self(p).max())
    base_step = 0.005
    min_step = 1e-8

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break

        c = _conv_self(p)
        cmax = float(c.max())
        threshold = cmax * 0.999
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Compute gradient of max (subgradient = average over active lags)
        grad = np.zeros(n)
        for k in active:
            # d/dx_j (a*a)[k] = a[k-j] + a[j] if 0 <= k-j < n
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                contrib = p[kj[valid]] + p[valid]
                grad[valid] += contrib
        grad /= len(active)

        step = base_step / (1.0 + 0.02 * it)
        if step < min_step:
            break

        for scale in (1.0, 0.5, 0.25, 0.1):
            p_cand = p - scale * step * grad
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = float(2.0 * n * _conv_self(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-16:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                break
        else:
            base_step *= 0.9
            if base_step < 1e-12:
                break

        if it % 50 == 0 and consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p

            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    def make_skew(src, n, sd):
        p = _resample_mid(src, n)
        rng = np.random.default_rng(sd)
        x = (np.arange(n) + 0.5) / n
        side = 1.0 if rng.random() < 0.5 else -1.0
        amp = 0.4 + 0.8 * float(rng.random())
        q = p * np.exp(side * amp * (2.0 * x - 1.0))
        q = np.maximum(q, 0.0)
        s = float(q.sum())
        if s > 0.0:
            q /= s
        else:
            q = p
        return q

    def run_attempt(n, src_arr, end_time, sd, do_coord=True):
        nonlocal best_arr, best_score
        n = int(max(1, n))
        p = _resample_mid(src_arr, n)
        if p.size != n or not np.all(np.isfinite(p)) or p.sum() <= 0.0:
            p = np.full(n, 1.0 / n)
        p = p / float(p.sum())
        consider(p)

        phase_len = max(0.0, float(end_time) - time.monotonic())
        if phase_len <= 10.0:
            return

        lbfgs_stop = min(float(end_time) - 10.0, time.monotonic() + 0.75 * phase_len)
        if lbfgs_stop > time.monotonic() + 10.0:
            try:
                p2 = _lbfgs_optimize(p, n, lbfgs_stop, consider=consider, fine_only=False)
                if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0.0:
                    p = p2 / float(p2.sum())
            except Exception:
                pass

        rem = float(end_time) - time.monotonic()
        if do_coord and rem > 15.0:
            coord_stop = min(
                float(end_time) - 1.5,
                time.monotonic() + max(15.0, min(120.0, 0.30 * phase_len)),
            )
            try:
                p3, _ = _coord_polish(p, coord_stop, max_sweeps=100000, seed=sd, consider=consider)
                if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and p3.sum() > 0.0:
                    p = p3
            except Exception:
                pass

        if p is not None and p.size == n and np.all(np.isfinite(p)) and float(p.sum()) > 0.0:
            consider(p)

    # Determine lengths to try
    parent_len = int(best_arr.size) if best_arr.size > 0 else 3072
    lengths = [parent_len]
    # Also try slightly different lengths for diversity
    if parent_len > 1024:
        lengths.append(parent_len // 2)
    lengths = sorted(set(lengths))

    # Allocate time: main length gets 85%, secondary gets 15%
    if len(lengths) == 1:
        weights = [1.0]
    else:
        weights = [0.85, 0.15]

    rng_top = np.random.default_rng(seed)
    phase_limit = min(phase_limit, hard - 12.0)

    for idx, L in enumerate(lengths):
        now = time.monotonic()
        if now >= phase_limit - 5.0:
            break
        rem_to_end = max(0.0, phase_limit - now)
        if rem_to_end <= 12.0:
            break
        w_rem = float(sum(weights[idx:]))
        if w_rem <= 0:
            w_rem = 1.0
        end_phase = min(phase_limit, now + (weights[idx] / w_rem) * rem_to_end)
        if end_phase <= now + 10.0:
            break

        src = best_arr.copy() if best_arr.size > 0 else parent.copy()

        if idx == 0 and (end_phase - now) > 200.0:
            v0 = src
            v1 = make_skew(src, L, int(seed + 777))
            mid = min(end_phase, now + 0.55 * (end_phase - now))
            run_attempt(L, v0, mid, int(seed + 100), do_coord=True)
            if time.monotonic() < end_phase - 10.0:
                run_attempt(L, v1, end_phase, int(seed + 200), do_coord=True)
        else:
            run_attempt(L, src, end_phase, int(seed + 500 + L), do_coord=True)

    # Final squeeze on the best solution found
    final_end = min(hard - 8.0, t0 + budget - 5.0)
    now = time.monotonic()
    if final_end > now + 30.0 and best_arr.size > 0:
        nbest = int(best_arr.size)
        rem = final_end - now

        # LSE fine-tune
        if rem > 60.0 and nbest <= 4000:
            lstop = min(final_end - 40.0, now + 0.45 * rem)
            try:
                p2 = _lbfgs_optimize(best_arr, nbest, lstop, consider=consider, fine_only=True)
                if p2 is not None and p2.size == nbest and np.all(np.isfinite(p2)) and p2.sum() > 0.0:
                    p2 = p2 / float(p2.sum())
                    consider(p2)
            except Exception:
                pass

        # Subgradient polish
        now = time.monotonic()
        sg_stop = min(final_end - 10.0, now + max(15.0, 0.20 * (final_end - now)))
        if sg_stop > now + 10.0:
            try:
                p_sg = _subgradient_polish(best_arr, sg_stop, consider=consider, max_iters=3000)
                if p_sg is not None and p_sg.size == nbest and np.all(np.isfinite(p_sg)) and p_sg.sum() > 0.0:
                    p_sg = p_sg / float(p_sg.sum())
                    consider(p_sg)
            except Exception:
                pass

        # Final coordinate polish
        now = time.monotonic()
        cstop = min(final_end - 3.0, now + max(10.0, 0.35 * (final_end - now)))
        if cstop > now + 8.0:
            try:
                p3, _ = _coord_polish(best_arr, cstop, max_sweeps=100000, seed=int(seed + 999999), consider=consider)
                if p3 is not None and p3.size == nbest and np.all(np.isfinite(p3)) and p3.sum() > 0.0:
                    p3 = p3 / float(p3.sum())
                    consider(p3)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
