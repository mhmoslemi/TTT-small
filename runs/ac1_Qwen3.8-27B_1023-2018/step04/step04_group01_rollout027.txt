import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(400, dtype=np.float64)
    if p.size == 0:
        p = np.ones(400, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0 or p.sum() < 0.01:
        p = np.ones(400, dtype=np.float64)
    return p


def _conv(a):
    n = len(a)
    size = next_fast_len(2 * n - 1)
    F = rfft(a, size)
    c = irfft(F * F, size)[: 2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _lp_solve(f, n, M, time_limit=5.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit, "mip_rel_gap": 0.0},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs


def _step_size(f, g, n, max_sc, bisection_iters=20):
    """Binary search for best step t in [0,1] minimizing score of (1-t)f + t*g."""
    best_t = 0.0
    best_sc = max_sc
    lo, hi = 0.0, 1.0
    for _ in range(bisection_iters):
        mid = (lo + hi) / 2.0
        f_new = f + mid * (g - f)
        f_new = np.maximum(f_new, 0.0)
        fs = f_new.sum()
        if fs < 1e-15:
            hi = mid
            continue
        f_new /= fs
        sc = _score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_t = mid
            lo = mid
        else:
            hi = mid
    # Refine around best_t
    if best_t > 0:
        delta = (hi - lo) / 2.0
        for t in [best_t]:
            f_new = f + t * (g - f)
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-15:
                continue
            f_new /= fs
            sc = _score(f_new)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    return best_t, best_sc


def _lp_phase(f0, n, stop_time, seed, max_iters=5000):
    rng = np.random.default_rng(seed)
    f = np.asarray(f0, dtype=np.float64).copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()
    best_f = f.copy()
    best_sc = _score(f)

    stall = 0
    max_stall = 150
    it = 0

    while time.monotonic() < stop_time - 3.0 and it < max_iters:
        it += 1
        if it % 200 == 0 and time.monotonic() > stop_time - 5:
            break

        b = _conv(f)
        M = float(b.max())
        sc = 2.0 * n * M  # sum(f)=1
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_f = f.copy()

        g = _lp_solve(f, n, M, time_limit=3.0)
        if g is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.maximum(f + rng.normal(0, 0.02, n), 0)
            f /= f.sum()
            continue

        dist = float(np.abs(g - f).max())
        if dist < 1e-9:
            stall += 3
            if stall > max_stall:
                break
            f = np.maximum(f + rng.normal(0, 0.03, n), 0)
            f /= f.sum()
            continue

        t_opt, sc_new = _step_size(f, g, n, best_sc)

        if t_opt > 1e-6 and sc_new < best_sc - 1e-14:
            f_new = f + t_opt * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new /= f_new.sum()
            f = f_new
            stall = 0
        elif sc_new < sc - 1e-12:
            # Accept even if not global best (progressive)
            f_new = f + t_opt * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new /= f_new.sum()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                # Perturb and reset stall
                scale = 0.05 if stall < max_stall * 2 else 0.1
                f = np.maximum(f + rng.normal(0, scale, n), 0)
                f /= f.sum()
                stall = 0
                if time.monotonic() > stop_time - 20:
                    break

    return best_f, best_sc


def _mirror_descent(f0, n, stop_time, seed, max_iters=50000):
    """Mirror descent on the simplex for min max(f*f)."""
    rng = np.random.default_rng(seed)
    f = np.asarray(f0, dtype=np.float64).copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()
    best_f = f.copy()
    best_sc = _score(f)

    tau = 100.0
    lr = 0.1
    lr_min = 1e-8

    for it in range(max_iters):
        if time.monotonic() > stop_time - 3.0:
            break

        b = _conv(f)
        M = float(b.max())
        sc = 2.0 * n * M

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_f = f.copy()

        # Smoothed max gradient
        b_shift = b - M
        tb = tau * b_shift
        tb -= tb.max()
        p = np.exp(tb)
        ps = p.sum()
        if ps < 1e-30:
            p = np.zeros(2 * n - 1)
            p[int(np.argmax(b))] = 1.0
            ps = 1.0
        else:
            p /= ps

        # grad of LSE: grad[i] = sum_k p[k] * (f shifted)
        # (p * f_rev)[i + n - 1] gives the gradient of the smoothed max w.r.t. f[i]
        f_rev = f[::-1]
        csize = 2 * n - 1
        P = rfft(p, csize)
        Fr = rfft(f_rev, csize)
        conv_val = irfft(P * Fr, csize)[:csize]
        grad = conv_val[n - 1: 2 * n - 1] * 2.0

        # Mirror descent update: f_new[i] prop to f[i] * exp(-lr * grad[i])
        update = -lr * grad
        update -= update.max()  # numerical stability
        weights = np.exp(update)
        f_new = f * weights
        f_new = np.maximum(f_new, 0.0)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = f + rng.normal(0, 0.01, n)
            f_new = np.maximum(f_new, 0)
            f_new /= f_new.sum()
        else:
            f_new /= fs

        sc_new = _score(f_new)
        if sc_new < sc:
            f = f_new
            lr = max(lr_min, lr * 0.99)
        else:
            lr = max(lr_min, lr * 0.5)
            if lr < lr_min * 2:
                f = np.maximum(f + rng.normal(0, 0.005, n), 0)
                f /= f.sum()
                lr = 0.02

        if it % 5000 == 4999:
            tau = max(20.0, tau * 0.8)
            f = np.maximum(f + rng.normal(0, 0.01, n), 0)
            f /= f.sum()
            lr = 0.03

    return best_f, best_sc


def _coord_polish(f0, stop_time, max_sweeps=100, seed=12345):
    """Coordinate descent with exact 1D line search."""
    n = len(f0)
    f = np.asarray(f0, dtype=np.float64).copy()
    f = np.maximum(f, 0.0)
    S = f.sum()
    if S < 1e-15:
        return f, math.inf
    f /= S
    conv = _conv(f)
    best_sc = 2.0 * n * conv.max()
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 3.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break

            v = f[i]
            S_rest = 1.0 - v  # since sum(f)=1

            # Influence: changing f[i] by delta changes conv[k] by delta*(2*f[k-i] + delta)
            # for k in [i, i+n-1], and conv[2i] additionally by delta^2
            start = i
            end = min(i + n, 2 * n - 1)
            infl = 2.0 * f[: end - start]

            # 1D minimization: minimize 2n * max(conv + delta*infl + delta^2*e_{2i}) / (1+delta)^2
            # where e_{2i} has 1 at position 2i
            # Coarse scan
            v_min = 0.0
            v_max = max(5.0 * v, 0.5)
            v_max = min(v_max, 1.0)  # since f is on simplex, max value is 1

            n_grid = 60
            x_grid = np.linspace(v_min, v_max, n_grid)
            best_x = v
            best_obj = best_sc

            for x in x_grid:
                dx = x - v
                if abs(dx) < 1e-14:
                    m_new = conv.max()
                else:
                    m_new = conv.max()
                    # Check the affected region
                    reg = conv[start:end].copy()
                    reg += dx * infl
                    if 2 * i <= 2 * n - 2:
                        reg[2 * i - start] += dx * dx
                    m_affected = reg.max()
                    m_unaffected = float(np.delete(np.delete(conv, slice(start, end), axis=0), max(0, min(2 * n - 2, 2 * i if start <= 2 * i < end else -1))) if False else 0)
                    # Simpler: just compare affected max with overall max excluding affected
                    m_new = max(m_affected, float(np.concatenate([conv[:start], conv[end:]]).max()) if (start > 0 or end < 2*n-1) else m_affected)
                S_new = S_rest + x
                if S_new < 1e-15:
                    continue
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x

            # Fine grid around best
            if best_x != v:
                span = (v_max - v_min) / n_grid
                x_lo = max(0.0, best_x - 3 * span)
                x_hi = min(v_max, best_x + 3 * span)
                x_fine = np.linspace(x_lo, x_hi, 30)
                for x in x_fine:
                    dx = x - v
                    if abs(dx) < 1e-14 or x < 0:
                        continue
                    reg = conv[start:end].copy()
                    reg += dx * infl
                    if 2 * i <= 2 * n - 2:
                        reg[2 * i - start] += dx * dx
                    m_affected = reg.max()
                    other = np.concatenate([conv[:start], conv[end:]])
                    m_unaff = other.max() if other.size > 0 else 0.0
                    m_new = max(m_affected, m_unaff)
                    S_new = S_rest + x
                    if S_new < 1e-15:
                        continue
                    obj = 2.0 * n * m_new / (S_new * S_new)
                    if obj < best_obj:
                        best_obj = obj
                        best_x = x

            # Apply
            if best_x != v and best_obj < best_sc - 1e-15:
                dx = best_x - v
                f[i] = best_x
                conv[start:end] += dx * infl
                if 2 * i <= 2 * n - 2:
                    conv[2 * i] += dx * dx
                best_sc = best_obj
                best_f = f.copy()
                improved = True

        if not improved and sweep > 2:
            break

    return best_f, best_sc


def _structured_perturb(f, n, rng, strength=0.1):
    """Add structured perturbations to escape local optima."""
    f_new = f.copy()
    strategy = rng.integers(5)
    i = np.arange(n)
    if strategy == 0:
        # Sine perturbation
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * math.pi)
        f_new += strength * np.sin(freq * 2 * math.pi * i / n + phase)
    elif strategy == 1:
        # Block perturbation
        block = n // 4
        f_new[:block] += strength
        f_new[block:2*block] -= strength
    elif strategy == 2:
        # Gaussian
        f_new += strength * rng.normal(0, 1, n)
    elif strategy == 3:
        # Edge emphasis
        w = np.abs((i - (n-1)/2) / (n/2))
        f_new += strength * (w - 0.5)
    else:
        # Random sparse kick
        idx = rng.choice(n, size=n//5, replace=False)
        f_new[idx] += strength * rng.choice([-1, 1], size=n//5)
    f_new = np.maximum(f_new, 0.0)
    f_new /= f_new.sum()
    return f_new


def _update_best(best_seq, best_score, candidate, exact_eval):
    """Update best if candidate is better (verified by exact evaluator)."""
    candidate = np.asarray(candidate, dtype=np.float64)
    candidate = np.clip(candidate, 0, 1000)
    candidate = np.where(np.isfinite(candidate), candidate, 0.0)
    if candidate.size == 0 or candidate.sum() < 0.01:
        return best_seq, best_score
    try:
        sc = float(exact_eval([float(x) for x in candidate]))
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score - 1e-14:
        return candidate.copy(), sc
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 12.0

    parent = _get_parent()
    n = len(parent)

    # Fallback
    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(n) / n
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _exact_best(cand):
        nonlocal best_seq, best_score
        best_seq, best_score = _update_best(best_seq, best_score, cand, evaluate_sequence)

    # Normalize parent
    f = parent.copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()

    # === Phase 1: LP iteration (45% of budget) ===
    t1_end = t0 + budget * 0.45
    if time.monotonic() < t1_end - 10:
        try:
            f, sc = _lp_phase(f, n, t1_end, seed)
            _exact_best(f)
        except Exception:
            pass

    # === Phase 2: Mirror descent refinement (15% of budget) ===
    t2_end = t0 + budget * 0.60
    if time.monotonic() < t2_end - 5 and best_score > 1.5027:
        try:
            f, sc = _mirror_descent(best_seq, n, t2_end, seed + 1)
            _exact_best(f)
        except Exception:
            pass

    # === Phase 3: Coordinate descent (20% of budget) ===
    t3_end = t0 + budget * 0.80
    if time.monotonic() < t3_end - 5:
        try:
            f, sc = _coord_polish(best_seq, t3_end, max_sweeps=150, seed=seed + 2)
            _exact_best(f)
        except Exception:
            pass

    # === Phase 4: Restart with perturbations (15% of budget) ===
    t4_end = min(t0 + budget * 0.95, hard_deadline - 15)
    if time.monotonic() < t4_end - 10 and best_score > 1.5028:
        rng = np.random.default_rng(seed + 99)
        n_restarts = 3
        per_restart = (t4_end - time.monotonic()) / n_restarts
        for r in range(n_restarts):
            if time.monotonic() > t4_end - 10:
                break
            f_pert = _structured_perturb(best_seq, n, rng, strength=0.08)
            stop_lp = min(t4_end - 5, time.monotonic() + per_restart * 0.5)
            stop_cd = min(t4_end - 3, time.monotonic() + per_restart * 0.8)
            try:
                f, sc = _lp_phase(f_pert, n, stop_lp, seed + 1000 + r, max_iters=2000)
                _exact_best(f)
                if time.monotonic() < stop_cd - 3:
                    f, sc = _coord_polish(best_seq, stop_cd, max_sweeps=50, seed=seed + 2000 + r)
                    _exact_best(f)
            except Exception:
                pass

    # === Phase 5: Final intensive coordinate polish ===
    if time.monotonic() < hard_deadline - 8:
        try:
            f, sc = _coord_polish(best_seq, hard_deadline - 6, max_sweeps=200, seed=77777)
            _exact_best(f)
        except Exception:
            pass

    # === Final validation ===
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            fallback_sc2 = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            fallback_sc2 = math.inf
        if not math.isfinite(fallback_sc2):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
