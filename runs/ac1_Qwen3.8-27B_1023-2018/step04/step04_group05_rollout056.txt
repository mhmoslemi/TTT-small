    F = rfft(a, fsize)

    np.maximum(c, 0, out=c)

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = None
    if p is None or p.size == 0:
        p = np.ones(352, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(352, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    if n <= 512:
        b = np.convolve(a, a, mode='full')
        return np.maximum(b, 0.0)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(np.max(b))
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, n, M, timeout=4.0):
    if M <= 1e-20:
        return None
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-14)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12,
                               'dual_edge_tap': 1e-8})
        if res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0, 1e6)
            if g.sum() > 1e-14:
                return g
    except Exception:
        pass
    return None


def _optimize_lp(f0, n, stop, seed, max_iter=8000):

    ts = math.sqrt(2.0 * n)

    f = f[:n]
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
        s = n
    f = f / s * ts

    # Symmetrize initial
    fs = (f + f[::-1]) / 2.0
    if fs.sum() > 0.01:
        sc_sym = _score(fs)
        if sc_sym < _score(f):
            f = np.clip(fs, 0, None)
            f = f / f.sum() * ts

    best_f = f.copy()
    best_sc = _score(f)

    t = 0.3
    stall = 0
    max_stall = 80
    it = 0
    last_improve_it = 0

    while time.monotonic() < stop and it < max_iter:
        it += 1
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-30:
            break

        g = _lp_solve(f, n, M, timeout=3.0)
        if g is None:
            stall += 3
            if stall > max_stall:
                f = best_f + rng.normal(0, 0.04 * max(1.0, best_f.mean()), n)
                f = np.clip(f, 0, None)
                if f.sum() > 0.01:
                    f = f / f.sum() * ts
                else:
                    f = best_f.copy()
                stall = 0
                t = 0.3
            continue
        lp_fail_reset = True

        g = g / g.sum() * ts

def _coord_descent(f0, n, stop, seed, max_rounds=300000):
    rng = np.random.default_rng(seed)
    ts = math.sqrt(2.0 * n)
    f = np.clip(f0.astype(np.float64).copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    s = f.sum()
    if s < 1e-10:
        return f, math.inf
    f = f / s * ts

    best_f = f.copy()
    b = _conv_full(f)
    best_sc = 2.0 * n * float(b.max()) / (float(f.sum()) ** 2)
    no_imp = 0
    max_no_imp = 150

    for rn in range(max_rounds):
        if time.monotonic() > stop:
            break

        S = float(f.sum())
        improved = False

        order = rng.permutation(n) if rn % 4 == 0 else np.arange(n)

        for i in order:
            if time.monotonic() > stop:
                break
            orig = f[i]
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)

            # Unaffected max
            max_unaff = 0.0
            if k_lo > 0:
                max_unaff = max(max_unaff, float(b[:k_lo].max()))
            if k_hi < 2 * n - 2:
                max_unaff = max(max_unaff, float(b[k_hi + 1:].max()))

            # Precompute kis
            kis = np.arange(k_lo, k_hi + 1) - i

            if orig > 1e-12:
                fracs = [0.0, -0.9, -0.7, -0.5, -0.3, -0.15, -0.05, 0.1, 0.3, 0.5]
            else:
                fracs = [0.01, 0.03, 0.1, 0.2]

            best_d = 0.0
            best_s = best_sc

            for fr in fracs:
                d = orig * fr if orig > 1e-12 else max(1.0, f.mean()) * fr
                nv = orig + d
                if nv < 0 or nv > 1000:
                    continue
                nS = S + d
                if nS < 0.01:
                    continue

                new_b_seg = b[k_lo:k_hi + 1] + 2.0 * d * f[kis]
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    new_b_seg[2 * i - k_lo] += d * d
                max_new = max(float(new_b_seg.max()), max_unaff)
                sc = 2.0 * n * max_new / (nS * nS)
                if sc < best_s:
                    best_s = sc
                    best_d = d

            if best_d != 0.0:
                f[i] = orig + best_d
                S = S + best_d
                b[k_lo:k_hi + 1] += 2.0 * best_d * f[kis]
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    b[2 * i] += best_d * best_d
                improved = True

        if improved:
            if rn % 80 == 0:
                b = _conv_full(f)
                sc = _score(f)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_f = f.copy()
                    no_imp = 0
                else:
                    no_imp += 1
            else:

                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_f = f.copy()
                    no_imp = 0
                else:
                    no_imp += 1
        else:
            no_imp += 1
            if no_imp > max_no_imp:
                break

    sc_final = _score(f)
    if sc_final < best_sc - 1e-15:
        best_sc = sc_final
        best_f = f.copy()
    return best_f, best_sc


def _try_symmetrize(f):
