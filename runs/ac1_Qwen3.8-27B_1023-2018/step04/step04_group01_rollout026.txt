def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: 2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def _build_lp_matrix(f, n):
    """Build A for: (f*g)[k] <= M, A has shape (2n-1, n). A[k,j] = f[j] if 0<=k-j<n else 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        if f[j] == 0:
            continue
        A[j : j + n, j] = f
    return A


def _lp_direction(f, n, M_target, time_limit=12.0):
    """LP: max sum(g) s.t. (f*g)[k] <= M_target, g >= 0. Returns g/sum(g) or None."""
    A = _build_lp_matrix(f, n)
    # Remove zero rows for speed
    row_sums = A.sum(axis=1)
    nonzero_rows = np.where(row_sums > 1e-15)[0]
    if len(nonzero_rows) < n:
        return None
    A_reduced = A[nonzero_rows, :]
    b_ub = np.full(len(nonzero_rows), M_target)
    c = -np.ones(n)
    try:
        res = linprog(
            c, A_ub=A_reduced, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit, "dual_feasibility_tolerance": 1e-9},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum

def _coord_descent(f0, stop_time, max_sweeps=200, seed=123):

                break

            infl = 2.0 * f  # linear coefficient for (x-v)

            for xg in x_grid:

            # Evaluate all candidates
            best_x = v
            best_obj = v_score
            for xc in candidates:
                if xc < 0:
                    xc = 0.0
                sc = obj_1d(xc)
                if sc < best_obj - 1e-15:
                    best_obj = sc
                    best_x = xc

                dw = x_max / n_grid

                        if sc < best_move_sc - 1e-12:
                            best_move_sc = sc
                            best_move = f_new.copy()

        if best_move is not None:
            f = best_move
            f = f / f.sum()
            if best_move_sc < best_score - 1e-12:
                best_score = best_move_sc
                best_f = f.copy()
        else:
            # Try random perturbation focused on peak-reducing direction
            if it % 20 == 19:
                f = np.clip(f + rng.normal(0, 0.01, n), 0, None)
                f /= f.sum()
            else:
                break

    return best_f, best_score


def _multi_start_lp(f0, n, stop_time, seed):
    """Multiple LP starts with different M targets."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs

    best_f = f.copy()
    best_score = _score_fast(f)

    while time.monotonic() < stop_time - 10:
        conv = _conv_full(f)
        M = float(conv.max())
        
        # Try several M targets
        found = False
        for m_frac in [0.99, 0.97, 0.95, 0.93, 0.90]:
            M_target = M * m_frac
            g_dir = _lp_direction(f, n, M_target, time_limit=8.0)
            if g_dir is None:
                continue
            
            # Evaluate
            for t_try in [0.5, 0.8, 1.0]:
                f_new = np.clip(f + t_try * (g_dir - f), 0, None)
                fs_new = f_new.sum()
                if fs_new < 0.01:
                    continue
                f_new = f_new / fs_new
                sc = _score_fast(f_new)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_f = f_new.copy()
                    f = f_new
                    found = True
                    break
            if found:
                break
        
        if not found:
            # Perturb and continue
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f /= fs
            else:
                f = best_f.copy()
            if time.monotonic() > stop_time - 15:
                break

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    hard_deadline = t0 + budget - 20.0

    parent = _get_parent()
    n = len(parent)

    # Fallback
    fallback = parent.copy()
    fallback = np.clip(fallback, 0, 1000)
    fallback = np.where(np.isfinite(fallback), fallback, 0.0)
    if fallback.size == 0 or fallback.sum() < 0.01:
        fallback = np.ones(n) / n
    fallback_score = _score_fast(fallback)

    best_seq = fallback.copy()
    best_score = fallback_score

    def _update(seq, sc=None):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64)
        seq = np.clip(seq, 0, 1000)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        if sc is None:
            sc = _score_fast(seq)
        if not math.isfinite(sc):
            return
        if sc < best_score - 1e-14:
            best_seq = seq.copy()
            best_score = sc

    # Start from parent
    f = parent.copy()
    f = np.clip(f, 0, None)
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs

    # === Phase 1: LP optimization (main) ===
    t1_end = t0 + budget * 0.35
    try:
        res, sc = _multi_start_lp(f, n, t1_end, seed)
        _update(res, sc)
    except Exception:
        pass

    # === Phase 2: L-BFGS on smoothed objective ===
    t2_end = min(t0 + budget * 0.55, hard_deadline - 80)
    if time.monotonic() < t2_end - 10 and best_score > 1.5027:
        try:
            res, sc = _grad_lbfgs(best_seq, n, t2_end, seed + 100)
            _update(res, sc)
        except Exception:
            pass

    # === Phase 3: Coordinate descent ===
    t3_end = min(t0 + budget * 0.75, hard_deadline - 50)
    if time.monotonic() < t3_end - 10:
        try:
            res, sc = _coord_descent(best_seq, t3_end, max_sweeps=300, seed=seed + 200)
            _update(res, sc)
        except Exception:
            pass

    # === Phase 4: Peak flattening ===
    t4_end = min(t0 + budget * 0.88, hard_deadline - 30)
    if time.monotonic() < t4_end - 10:
        try:
            res, sc = _peak_flatten(best_seq, t4_end, seed=seed + 300)
            _update(res, sc)
        except Exception:
            pass

    # === Phase 5: Final coordinate polish ===
    t5_end = hard_deadline - 10
    if time.monotonic() < t5_end - 5:
        try:
            res, sc = _coord_descent(best_seq, t5_end, max_sweeps=100, seed=seed + 400)
            _update(res, sc)
        except Exception:
            pass

    # Final LP boost if time remains
    if time.monotonic() < hard_deadline - 20 and best_score > 1.5027:
        try:
            res, sc = _multi_start_lp(best_seq, n, hard_deadline - 5, seed + 500)
            _update(res, sc)
        except Exception:
            pass

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
