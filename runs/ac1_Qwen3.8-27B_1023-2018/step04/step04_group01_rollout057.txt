                improved += 1
        if improved == 0:
            f = np.clip(f + rng.normal(0, 0.01, n), 0, None)
            fs2 = f.sum()
            if fs2 > 1e-10:
                f /= fs2
                S = float(f.sum())
                conv = _conv_full(f)
                best_obj = 2.0 * n * float(conv.max())
                best_f = f.copy()
            if sweep > 5:
                break

    return best_f, best_obj

def _eval_1d(x, v, S_r, conv, st, en, infl, di, n):
    S_new = S_r + x
    if S_new < 0.01:
        return math.inf
    w = x - v
    if abs(w) < 1e-14:
        m = float(conv.max())
    else:
        m = float(conv.max())
        if w > 0:
            cur_max = float(conv[st:en].max())
            new_seg = cur_max + w * infl[st:en].max()
            m = max(m, new_seg)
        else:
            new_seg = conv[st:en] + w * infl[st:en]
            idx = np.argmax(new_seg)
            val = float(new_seg[idx])
            if val > m:
                m = val
    if 0 <= di <= 2 * n - 2:
        m = max(m, float(conv[di] + w * infl[di] + w * w))
    return 2.0 * n * m / (S_new * S_new)

def _pair_polish(f0, stop_t, seed, max_rounds=40):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    fs = float(f.sum())
    if fs < 1e-10:
        return f, math.inf
    f /= fs
    best_sc = _score(f)
    best_f = f.copy()
    mults = (0.1, 0.3, 0.5, 0.7, 1.0, 1.4, 2.0, 3.0)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_t - 3:
            break
        n_trials = 40
        for _ in range(n_trials):
            if time.monotonic() > stop_t - 3:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-10 and oj < 1e-10:
                continue
            best_pair = (oi, oj)
            best_ps = best_sc
            for mi in mults:
                ni = oi * mi if oi > 1e-10 else (0.01 if mi > 0 else 0.0)
                for mj in mults:
                    nj = oj * mj if oj > 1e-10 else (0.01 if mj > 0 else 0.0)
                    if float(f.sum()) - oi - oj + ni + nj < 0.01:
                        continue
                    f[i], f[j] = ni, nj
                    sc = _score(f)
                    if sc < best_ps - 1e-13:
                        best_ps = sc
                        best_pair = (ni, nj)
            f[i], f[j] = best_pair
            if best_ps < best_sc - 1e-13:
                best_sc = best_ps
                best_f = f.copy()

    return best_f, best_sc

def _lp_optimize(f0, n, stop_t, seed, max_iters=50):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f /= fs
    f = f / f.sum()
    best = f.copy()
    best_sc = _score(f)

    for _ in range(max_iters):
        if time.monotonic() > stop_t - 5:
            break
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-15:
            break
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = f
        try:
            res = linprog(
                -np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                bounds=[(0, None)] * n, method="highs",
                options={"presolve": True, "time_limit": 5.0}
            )
        except Exception:
            break
        if res.status != 0 or res.x is None:
            break
        g = np.asarray(res.x, dtype=np.float64)
        gs = float(g.sum())
        if gs < 1e-15:
            break
        g /= gs
        dist = float(np.linalg.norm(g - f))
        if dist < 1e-7:
            break
        f_cs = _score(f)
        lo, hi, bt, bsc = 0.0, 1.0, 0.0, f_cs
        for _ in range(16):
            mid = (lo + hi) / 2.0
            fn = np.clip(f + mid * (g - f), 0, None)
            s = fn.sum()
            if s < 1e-10:
                hi = mid
                continue
            fn /= s
            sc = _score(fn)
            if sc < bsc - 1e-14:
                bsc, bt = sc, mid
                lo = mid
            else:
                hi = mid
        if bt > 1e-4 and bsc < best_sc - 1e-12:
            fn = np.clip(f + bt * (g - f), 0, None)
            fn /= fn.sum()
            f, best, best_sc = fn, fn.copy(), bsc
        elif bt <= 1e-4:
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            f = f / f.sum()

    return best, best_sc

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_dl = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)
    fallback = parent.copy() / max(parent.sum(), 1.0)
    try:
        fb_sc = _score(fallback)
    except Exception:
        fb_sc = math.inf
    if not math.isfinite(fb_sc):
        fallback = np.ones(32) / 32.0
        fb_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fb_sc

    def update(seq, sc=None):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64)
        seq = np.clip(seq, 0, 1000.0)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        if sc is None:
            sc = _score(seq)
        if not math.isfinite(sc):
            return
        try:
            ex = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            ex = math.inf
        if math.isfinite(ex) and ex < best_score - 1e-14:
            best_seq = seq.copy()
            best_score = ex

    # === Phase 1: Gradient descent on parent (40%) ===
    p1_end = t0 + budget * 0.40
    update(parent, _score(parent))
    if time.monotonic() < p1_end - 10:
        try:
            r, s = _grad_optimize(parent, p1_end, seed)
            update(r, s * (r.sum() ** 2) if s is not None else None)
        except Exception:
            pass

    # === Phase 2: Coordinate descent (25%) ===
    p2_end = t0 + budget * 0.65
    if time.monotonic() < p2_end - 10:
        try:
            r, s = _coord_polish(best_seq, p2_end, seed + 100, max_sweeps=80)
            r2 = r / r.sum()
            update(r2, _score(r2))
        except Exception:
            pass

    # === Phase 3: Pair polish (10%) ===
    p3_end = t0 + budget * 0.75
    if time.monotonic() < p3_end - 10:
        try:
            r, s = _pair_polish(best_seq, p3_end, seed + 200, max_rounds=40)
            r2 = r / r.sum()
            update(r2, _score(r2))
        except Exception:
            pass

    # === Phase 4: LP direction on current best (10%) ===
    p4_end = t0 + budget * 0.85
    if time.monotonic() < p4_end - 15 and best_score > 1.503:
        try:
            r, s = _lp_optimize(best_seq, len(best_seq), p4_end, seed + 300, max_iters=20)
            r = r / r.sum()
            update(r, _score(r))
        except Exception:
            pass

    # === Phase 5: Alternative initializations (10%) ===
    p5_end = hard_dl - 40
    if time.monotonic() < p5_end - 20 and best_score > 1.503:
        n = len(best_seq)
        ia = np.arange(n, dtype=np.float64)
        inits = [
            np.clip(0.1 + 0.9 * np.cos(np.pi * ia / n) ** 4, 0, None),
            np.clip(0.3 + 0.7 * np.cos(2 * np.pi * ia / n), 0, None),
            np.clip(1.0 - 0.6 * ((ia / n) - 0.5) ** 4, 0, None),
            np.clip(0.5 + 0.5 * np.sin(np.pi * ia / n), 0, None),
        ]
        inits = [x / x.sum() for x in inits if x.sum() > 0.01]
        per_init = max(10.0, (p5_end - time.monotonic()) / max(1, len(inits)))
        for idx, ini in enumerate(inits):
            if time.monotonic() > p5_end - 10:
                break
            if best_score < 1.503:
                break
            t_stop = min(p5_end - 5, time.monotonic() + per_init * 0.6)
            t_stop2 = min(p5_end - 3, t_stop + per_init * 0.4)
            try:
                r1, _ = _grad_optimize(ini, t_stop, seed + 500 + idx)
                r2, _ = _coord_polish(r1, t_stop2, seed + 600 + idx, max_sweeps=15)
                r2 = r2 / r2.sum()
                update(r2, _score(r2))
            except Exception:
                pass

    # === Phase 6: Try different lengths (if time) ===
    p6_end = hard_dl - 30
    if time.monotonic() < p6_end - 30 and best_score > 1.5028:
        for L in [n_p + 20, n_p - 20, n_p + 40, n_p - 40, n_p + 8]:
            if time.monotonic() > p6_end - 15:
                break
            if best_score < 1.5028:
                break
            if L < 16 or L > 1500:
                continue
            init_l = np.interp(
                np.linspace(0, 1, L),
                np.linspace(0, 1, len(best_seq)),
                best_seq
            )
            init_l = np.clip(init_l, 0, None)
            if init_l.sum() < 0.01:
                init_l = np.ones(L) / L
            init_l /= init_l.sum()
            t_s1 = min(p6_end - 10, time.monotonic() + 30)
            t_s2 = min(p6_end - 5, t_s1 + 30)
            try:
                r1, _ = _grad_optimize(init_l, t_s1, seed + 9000 + L, max_iter=5000)
                r2, _ = _coord_polish(r1, t_s2, seed + 9100 + L, max_sweeps=10)
                r2 = r2 / r2.sum()
                update(r2, _score(r2))
            except Exception:
                pass

    # === Phase 7: Final intensive polish ===
    fin_end = hard_dl - 8
    if time.monotonic() < fin_end - 8:
        try:
            r, s = _coord_polish(best_seq, fin_end - 5, seed + 7777, max_sweeps=40)
            r = r / r.sum()
            update(r, _score(r))
        except Exception:
            pass

    if time.monotonic() < hard_dl - 4:
        try:
            r, s = _pair_polish(best_seq, hard_dl - 3, seed + 8888, max_rounds=20)
            r = r / r.sum()
            update(r, _score(r))
        except Exception:
            pass

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()
    try:
        fv = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        fv = math.inf
    if not math.isfinite(fv):
        try:
            fv = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            fv = math.inf
        if not math.isfinite(fv):
            return [1.0]
        return [float(x) for x in fallback]
    return [float(x) for x in best_seq]
