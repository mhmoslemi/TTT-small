The max of updated_b is max(unchanged_max, max_{k∈[i,i+n-1]} (b[k] + 2w*a[k-i] + w^2*δ_{k,2i}))

For a given w, this max is a convex piecewise-linear (plus one quadratic) function of w. The score is this max divided by (S+w)^2, times 2n.

The minimum of score(w) for w in [0, x_max] (since a_i must be ≥ 0) can be found by:
1. Checking w = 0
2. For each k in [i, i+n-1], the stationary point of (b[k] + 2w*a[k-i])/(S+w)^2 is at w_k = S - b[k]/a[k-i] (if a[k-i] > 0)
3. The stationary point of (b[2i] + 2w*a[i] + w^2)/(S+w)^2
4. The upper bound w = x_max

But there are n candidate stationary points, and we need to check which one is feasible (w ≥ 0, the corresponding constraint is actually active at that point). This is O(n) per coordinate, same as the grid.

However, we can be smarter: only the "top few" constraints matter. If we precompute the sorted order of b[k] values, we only need to check the top ~5-10 constraints that could be active near the current point.

            best_w = w_vals[imin]
            best_val = scores[imin]

            if G > 1:

                lo = max(-v, best_w - 2*dw)
                hi = min(x_hi - v, best_w + 2*dw)
                if hi > lo:

                    w_fine = fine_pts

                        best_val = scores_f[imin_f]

                p[i] = new_v

                if 2*i < Ltot:
                    c[2*i] += best_w**2
                cur_score = best_val
                improved = True

            inv = 1.0 / S
            p *= inv

            S = 1.0

                best_score = cur_score
                best_p = p.copy()

        stages = [(0.15, 25), (0.04, 30), (0.01, 35), (0.003, 35), (0.0008, 30), (0.0002, 25)]

    c_init = cfast(p)
    cmax_init = float(c_init.max()) if c_init.size else 1.0

    for frac, maxit in stages:
        if time.monotonic() > stop_time - 4.0:
            break
        T = max(1e-300, frac * cmax_init)
        for _ in range(4):
            if time.monotonic() > stop_time - 4.0:
                break
            opts = {"maxiter": maxit, "gtol": 1e-11, "ftol": 1e-15,
                    "maxls": 25, "maxcor": 25, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and res.x.size == best_u.size and np.all(np.isfinite(res.x)):
                fv = float(res.fun)
                if math.isfinite(fv) and fv < best_f - 1e-15:
                    best_f = fv
                    best_u = res.x.copy()
        p = _softmax(best_u)
        p = np.maximum(p, 0.0)
        ss = float(p.sum())
        if ss > 1e-300:
            p /= ss
        else:
            break
        if consider:
            try:
                consider(p)
            except Exception:
                pass

    return p


def _coord_descent(p, stop_time, seed=42, G=None, consider=None, perturb_sd=0.03):
    n = int(p.size)
    p = np.maximum(p.astype(np.float64).ravel(), 0.0).copy()
    S = float(p.sum())
    if S < 1e-300:
        return p, math.inf
    p = p / S
    S = 1.0

    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()

    if G is None:
        if n <= 500:
            G = 32
        elif n <= 1200:
            G = 28
        elif n <= 2500:
            G = 24
        else:
            G = 20
    G = int(max(12, G))

    rng = np.random.default_rng(seed)
    no_improve = 0
    sweep = 0

    while sweep < 500000 and time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 1.5:
                break
            if S <= 1e-300:
                break

            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            infl = 2.0 * p[:Ls]
            base = c[start:end]

            um = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > um:
                    um = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > um:
                    um = mr

            x_hi = max(20.0 * v, 40.0 / float(n), 0.02 / float(n), 1e-12)
            x_hi = min(x_hi, 5.0)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, 1.0 / float(n))

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.concatenate((pts, [v]))
            # Add a few smart points
            smart = []
            if v > 0:
                smart.append(v * 0.5)
                smart.append(v * 1.5)
                smart.append(v * 0.8)
                smart.append(v * 1.2)
            for sp in smart:
                if 0.0 < sp < x_hi:
                    pts = np.concatenate((pts, [sp]))
            pts = np.unique(pts)

            w_vals = pts - v
            aff = base[None, :] + w_vals[:, None] * infl[None, :]
            diag_idx = 2 * i
            if start <= diag_idx < end:
                aff[diag_idx - start, :] += w_vals ** 2
            maff = np.maximum(aff, um)
            denom = (S + w_vals) ** 2
            scores = 2.0 * n * maff / denom
            scores = np.where(np.isfinite(scores), scores, 1e30)

            imin = int(np.argmin(scores))
            best_w = float(w_vals[imin])
            best_val = float(scores[imin])

            # Fine refinement
            if len(pts) > 2:
                dw = x_hi / float(G)
                lo = max(-v + 1e-15, best_w - 2.5 * dw)
                hi = min(x_hi - v - 1e-15, best_w + 2.5 * dw)
                if hi > lo + 1e-15:
                    fine = np.linspace(lo, hi, 48, dtype=np.float64)
                    aff_f = base[None, :] + fine[:, None] * infl[None, :]
                    if start <= diag_idx < end:
                        aff_f[diag_idx - start, :] += fine ** 2
                    maff_f = np.maximum(aff_f, um)
                    denom_f = (S + fine) ** 2
                    scores_f = 2.0 * n * maff_f / denom_f
                    scores_f = np.where(np.isfinite(scores_f), scores_f, 1e30)
                    imin_f = int(np.argmin(scores_f))
                    if scores_f[imin_f] < best_val - 1e-14:
                        best_w = float(fine[imin_f])
                        best_val = float(scores_f[imin_f])

            if best_val < cur_score - 1e-13 and best_w > -v - 1e-14:
                new_v = v + best_w
                if new_v < 0:
                    new_v = 0.0
                    best_w = -v
                p[i] = new_v
                S += best_w
                if best_w != 0.0:
                    c[start:end] += best_w * infl
                    if diag_idx < Ltot:
                        c[diag_idx] += best_w ** 2
                cur_score = best_val
                improved = True

        # Renormalize periodically
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
            cur_score = 2.0 * n * float(c.max())

        if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
            if consider:
                try:
                    consider(best_p)
                except Exception:
                    pass
        else:
            no_improve += 1

        if no_improve >= 4:
            if time.monotonic() < stop_time - 20.0:
                sd = perturb_sd * (1.0 + 0.3 * min(no_improve // 10, 3.0))
                p = p * np.exp(rng.normal(0.0, sd, n, dtype=np.float64))
                p = np.maximum(p, 0.0)
                ss = float(p.sum())
                if ss > 1e-300:
                    p /= ss
                    S = 1.0
                else:
                    p = np.full(n, 1.0 / n, dtype=np.float64)
                    S = 1.0
                c = _conv_self(p)
                cur_score = 2.0 * n * float(c.max())
                if math.isfinite(cur_score) and cur_score < best_score:
                    best_score = cur_score
                    best_p = p.copy()
                no_improve = 0
            else:
                break

        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    try:
        c2 = _conv_self(p)
        sc2 = float(2.0 * n * c2.max())
    except Exception:
        sc2 = math.inf
    if math.isfinite(sc2) and sc2 < best_score:
        best_score = sc2
        best_p = p.copy()

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 30.0
    if hard < t0 + 10.0:
        hard = t0 + max(5.0, budget - 1.0)

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = math.inf
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        pass
    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(128, dtype=np.float64) / 128.0
        best_score = float(evaluate_sequence([float(x) for x in best_arr]))
        if not math.isfinite(best_score):
            best_score = _score_exact_arr(best_arr)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            sc = float(_score_exact_arr(a))
            if math.isfinite(sc) and sc < best_score - 1e-14:
                a = a / s
                best_arr = a.copy()
                best_score = sc
        except Exception:
            pass

    n = int(best_arr.size)
    # Phase 1: LSE optimization to refine the parent
    p = best_arr / float(best_arr.sum())
    lse_stop = min(hard - 200.0, t0 + min(120.0, 0.12 * budget))
    if lse_stop > t0 + 15.0:
        try:
            p2 = _lse_optimize(p, lse_stop, consider=consider)
            if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p = p2 / float(p2.sum())
                consider(p)
        except Exception:
            pass

    # Phase 2: Main coordinate descent (bulk of time)
    coord_stop = hard - 80.0
    if coord_stop > time.monotonic() + 30.0:
        try:
            p3, sc3 = _coord_descent(p, coord_stop, seed=int(seed), consider=consider, G=22)
            if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and p3.sum() > 0:
                p = p3 / float(p3.sum())
                consider(p)
        except Exception:
            pass

    # Phase 3: Another LSE round at low temperature
    rem = hard - time.monotonic() - 40.0
    if rem > 40.0:
        lse_stop2 = time.monotonic() + min(rem * 0.35, 60.0)
        try:
            p4 = _lse_optimize(p, lse_stop2, consider=consider,
                              stages=[(0.005, 30), (0.001, 35), (0.0003, 35), (0.0001, 30)])
            if p4 is not None and p4.size == n and np.all(np.isfinite(p4)) and p4.sum() > 0:
                p = p4 / float(p4.sum())
                consider(p)
        except Exception:
            pass

    # Phase 4: Final coordinate polish
    cstop2 = hard - 30.0
    if cstop2 > time.monotonic() + 15.0:
        try:
            p5, sc5 = _coord_descent(p, cstop2, seed=int(seed + 77777),
                                     consider=consider, G=28, perturb_sd=0.015)
            if p5 is not None and p5.size == n and np.all(np.isfinite(p5)) and p5.sum() > 0:
                p = p5 / float(p5.sum())
                consider(p)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
