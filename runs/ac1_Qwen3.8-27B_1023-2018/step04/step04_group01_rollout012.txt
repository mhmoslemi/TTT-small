    """Solve: min M' s.t. (f*g)[k] <= M', sum(g) = sum(f), g >= 0"""
    n = len(f)

    # (f*g)[k] = sum_j A[k,j] * g[j] where A[k,j] = f[k-j] if 0<=k-j<n else 0
    A = np.zeros((2*n-1, n))

    b_eq = np.array([f.sum()])

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    return None

    def coord_descent(f, stop_time, max_sweeps=2000, grid_n=60, seed_off=0):

    def lp_step(f, stop_time, max_iters=200):

        best_fl = f.copy()

            A = np.zeros((2 * n - 1, n))

                A[j:j + n, j] = f
            
            # LP: min M' s.t. A@g - M' <= 0, sum(g) = s, g >= 0, M' >= 0
            # Variables: [g (n), M' (1)]
            c = np.zeros(n + 1)
            c[-1] = 1.0  # minimize M'
            
            A_ub = np.hstack([A, -np.eye(2 * n - 1)])
            b_ub = np.zeros(2 * n - 1)
            
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([s])
            
            bounds = [(0, None)] * n + [(0, None)]
            
            try:
                res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs',
                             options={'presolve': True, 'time_limit': 5.0})
            except Exception:
                break
            
            if res.status != 0 or res.x is None:
                break
            
            g = res.x[:n]
            g_sum = float(g.sum())
            if g_sum < 1e-10:
                break
            
            # Check if g is meaningfully different from f
            dist = float(np.linalg.norm(g - f))
            if dist < 1e-8 * max(1, s):
                break

            f_cur_score = compute_score(f)
            best_t = 0.0
            best_sc = f_cur_score
            lo, hi = 0.0, 1.0
            for _ in range(15):

                fn = f + mid * (g - f)
                fn = np.maximum(fn, 0)
                sc = compute_score(fn)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_t = mid
                    lo = mid
                else:

                f = fn
                best_fl = fn.copy()
                best_sl = best_sc
            elif best_t > 1e-6:
                fn = np.maximum(f + best_t * (g - f), 0)
                f = fn
                # Don't update best, just move
            else:
                # No progress, try perturbation
                f = np.maximum(f + np.random.default_rng(it).normal(0, 0.01*s/n, n), 0)
        
        return best_fl, best_sl
    
    # Generate initializations
    inits = [parent.copy()]

    i_arr = np.arange(n, dtype=np.float64)
    rc = 0.5 * (1 + np.cos(np.pi * i_arr / n))
    rc = np.maximum(rc, 0.01)
    rc /= rc.sum()
    inits.append(rc)
    
    # Gaussian-like
    center = np.arange(n) - n / 2
    gc = np.exp(-center**2 / (2 * (n/4)**2))
    gc /= gc.sum()
    inits.append(gc)
    
    # Linear ramp + cosine
    lr = np.linspace(0.5, 1.0, n) * 0.5 + 0.5 * np.cos(2 * np.pi * i_arr / n)
    lr = np.maximum(lr, 0.01)
    lr /= lr.sum()
    inits.append(lr)
    
    # Process each initialization
    time_per_init = (deadline - t0) / max(1, len(inits))
    
    for idx, init in enumerate(inits):

        if remaining < 30:
            break
        
        # Allocate time: LP first, then coord descent
        lp_budget = min(remaining * 0.3, 200.0)
        cd_budget = remaining * 0.7
        
        # LP phase
        if lp_budget > 20:
            try:
                res, sc = lp_step(init, t0 + time.monotonic() - t0 + lp_budget, max_iters=100)
                if sc < best_score:
                    best_f = res
                    best_score = sc
            except Exception:

        if time.monotonic() < stop_cd - 10:

            pass

    return [float(x) for x in best_f]
