    # A[k, j] = f[j] if 0 <= k-j <= n-1, else 0

    cols = []
    vals = []

        j_end = min(n - 1, k)

                rows.append(k)
                cols.append(j)
                vals.append(v)

        return None

def _vconv(a):

    if s < 1e-300:

                    best_sc = sc
                    best = p.copy()
                    improved = True
                    break
                p[target] = orig
            if improved:
                break
        if not improved:
            # Try bumping a cold spot
            cold = int(c.argmin())
            bump = 0.005 / n
            p[cold] += bump
            s = p.sum()
            p = p / s
            sc = _score_p(p)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best = p.copy()
            else:
                p = best.copy()

    return best, best_sc


def _cd_polish(p, stop_time, seed=0, max_rounds=80):
    """Coordinate descent on the simplex."""
    n = len(p)
    rng = np.random.default_rng(seed)
    p = p.copy()
    best = p.copy()
    best_sc = _score_p(p)
    cur = best_sc

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 2.0:
            break
        improved_any = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = p[i]
            if orig < 1e-10:
                continue
            # Try scaling down
            for frac in [0.5, 0.7, 0.85, 0.3]:
                p[i] = orig * frac
                s = p.sum()
                if s < 1e-15:
                    p[i] = orig
                    continue
                pt = p / s
                sc = _score_p(pt)
                if sc < cur - 1e-14:
                    p = pt
                    cur = sc
                    improved_any = True
                    break
            p[i] = p[i]  # keep current
            # Also try setting to 0
            p[i] = 0.0
            s = p.sum()
            if s > 1e-15:
                pt = p / s
                sc = _score_p(pt)
                if sc < cur - 1e-14:
                    p = pt
                    cur = sc
                    improved_any = True
        if not improved_any:
            # Try bumping zeros
            zeros = np.where(p < 1e-10)[0]
            if len(zeros) > 0:
                zi = zeros[rng.integers(len(zeros))]
                for bump in [0.01/n, 0.05/n, 0.1/n]:
                    p = best if p.sum() < 1e-15 else p / p.sum()
                    p[zi] += bump
                    s = p.sum()
                    pt = p / s
                    sc = _score_p(pt)
                    if sc < cur - 1e-14:
                        p = pt
                        cur = sc
                        improved_any = True
                        break
        if cur < best_sc:
            best_sc = cur
            best = p.copy()
        if not improved_any and rnd > 10:
            break

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 10.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(400, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(400, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    n = len(parent)

    ps = parent.sum()
    if ps < 1e-15:
        p_init = np.ones(n) / n
        ps = 1.0
    else:
        p_init = parent / ps

    best_p = p_init.copy()
    best_score = _score_p(p_init)

    try:
        exact = float(evaluate_sequence([float(x) for x in parent]))
        if math.isfinite(exact):
            best_score = min(best_score, exact)
    except Exception:
        pass

    rng = np.random.default_rng(seed)

    def _try_update(p_c, sc_c):
        nonlocal best_p, best_score
        if sc_c < best_score - 1e-13:
            best_p = p_c.copy()
            best_score = sc_c

    # === Phase 1: Mirror descent with LSE (35% time) ===
    phase1_end = t0 + budget * 0.35
    p = p_init.copy()
    lr = 0.8
    tau = 0.003
    no_improve = 0

    while time.monotonic() < phase1_end - 3.0:
        grad = _grad_lse(p, tau)
        p_new = _mirror_step(p, grad, lr)
        sc_new = _score_p(p_new)
        if sc_new < best_score - 1e-13:
            _try_update(p_new, sc_new)
            no_improve = 0
        else:
            no_improve += 1
        p = p_new
        lr *= 0.9995
        tau = max(0.0005, tau * 0.9997)
        if no_improve > 500:
            # Restart with perturbation
            p = _mirror_step(best_p, rng.normal(0, 1, n), 0.02)
            lr = 0.5
            tau = 0.002
            no_improve = 0

    # === Phase 2: LP direction finding (25% time) ===
    phase2_end = t0 + budget * 0.60
    p = best_p.copy()
    cur_sc = _score_p(p)
    t_mix = 0.12
    stall = 0

    while time.monotonic() < phase2_end - 3.0:
        g = _lp_direction(p, timeout=2.0)
        if g is None:
            stall += 3
            if stall > 40:
                p = _mirror_step(best_p, rng.normal(0, 1, n), 0.015)
                cur_sc = _score_p(p)
                stall = 0
                t_mix = 0.12
            continue

        improved = False
        for alpha in [t_mix, t_mix * 0.5, t_mix * 2.0, t_mix * 0.25]:
            p_try = np.maximum((1 - alpha) * p + alpha * g, 0)
            s = p_try.sum()
            if s < 1e-15:
                continue
            p_try = p_try / s
            sc_try = _score_p(p_try)
            if sc_try < cur_sc - 1e-14:
                p = p_try
                cur_sc = sc_try
                improved = True
                t_mix = min(0.7, alpha * 1.2)
                break
            elif sc_try < _score_p(p) - 1e-14:
                p = p_try
                improved = True
                t_mix = min(0.7, alpha)
                break
        if improved:
            stall = 0
            if cur_sc < best_score:
                _try_update(p, cur_sc)
        else:
            stall += 1
            t_mix = max(0.02, t_mix * 0.6)
            if stall > 80:
                p = _mirror_step(best_p, rng.normal(0, 1, n), 0.02)
                cur_sc = _score_p(p)
                stall = 0
                t_mix = 0.12

    # === Phase 3: Peak shaving (15% time) ===
    phase3_end = t0 + budget * 0.75
    if time.monotonic() < phase3_end - 5.0:
        p_ps, sc_ps = _peak_shave(best_p, phase3_end, seed=seed + 100, rounds=8000)
        if sc_ps < best_score:
            _try_update(p_ps, sc_ps)

    # === Phase 4: CD polish (15% time) ===
    phase4_end = t0 + budget * 0.90
    if time.monotonic() < phase4_end - 5.0:
        p_cd, sc_cd = _cd_polish(best_p, phase4_end, seed=seed + 200, max_rounds=100)
        if sc_cd < best_score:
            _try_update(p_cd, sc_cd)

    # === Phase 5: Final mirror descent with tiny tau (5% time) ===
    final_end = deadline - 5.0
    if time.monotonic() < final_end - 5.0:
        p = best_p.copy()
        lr = 0.1
        tau = 0.0005
        for it in range(20000):
            if time.monotonic() > final_end - 2.0:
                break
            grad = _grad_lse(p, tau)
            p = _mirror_step(p, grad, lr)
            sc = _score_p(p)
            if sc < best_score:
                _try_update(p, sc)
            lr *= 0.9998

    # Final validation with exact evaluator
    # Try multiple scalings (score is scale-invariant so any works)
    result_candidates = [best_p, best_p * ps]
    best_final_sc = math.inf
    best_final = None
    for rc in result_candidates:
        rc = np.clip(rc, 0, 1000)
        if rc.sum() < 0.01:
            continue
        try:
            sc = float(evaluate_sequence([float(x) for x in rc]))
            if math.isfinite(sc) and sc < best_final_sc:
                best_final_sc = sc
                best_final = rc.copy()
        except Exception:
            pass

    if best_final is None:
        # Fallback to parent
        try:
            sc = float(evaluate_sequence([float(x) for x in parent]))
            if math.isfinite(sc):
                return [float(x) for x in parent]
        except Exception:
            pass
        return [1.0] * n

    return [max(0.0, min(1000.0, float(x))) for x in best_final]
