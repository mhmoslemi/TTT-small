def _subgrad_step(f, n, step):
    B = _fft_conv2(f)
    S = f.sum()
    M = B.max()
    
    # Find active set (top values)
    active = np.where(B >= M - 1e-10 * M)[0]
    
    # Subgradient of max: average of gradients at active points
    # grad[k][j] = 2 * f[k-j] if 0 <= k-j < n else 0
    # Subgradient of F = (1/|K|) * sum_{k in K} grad_k
    
    grad = np.zeros(n)
    for k in active:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += 2.0 * f[k-j_hi : k-j_lo+1]
    grad /= len(active)
    
    # Score gradient (with S = sum(f)):
    # dS/df_j = (2n/S^2) * (grad_f_j - 2*M/S)
    # = (2n/S^2) * grad_f_j - 4n*M/S^3
    
    score_grad = (2.0 * n / (S*S)) * (grad - 2.0*M/S)
    
    # Step
    f_new = f - step * score_grad
    f_new = _project_simplex(f_new)
    
    return f_new
