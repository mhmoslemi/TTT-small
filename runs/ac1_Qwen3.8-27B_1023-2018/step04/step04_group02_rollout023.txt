import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _conv(a):
    """Compute full convolution a * a."""
    n = len(a)
    if n <= 1024:
        return np.convolve(a, a, mode='full')
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    """Compute the AC1 score."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _lp_direction(f, n, M, timeout=3.0):
    """Find LP direction: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    # Build the constraint matrix A where (A @ g)_k = conv(f, g)_k
    # conv(f, g)_k = sum_j f_j * g_{k-j} for valid j, k-j
    # This is a Toeplitz-like matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.maximum(res.x, 0.0)
    if g.sum() < 1e-15:
        return None
    return g

    best_v = f[i]
    best_sc = _score(f)

    orig = f[i]

Let me also think about whether there's a fundamentally better approach.

**Alternative: Direct QP/SOCP formulation**

The problem is: minimize max_k (a^T B_k a) subject to a ≥ 0, sum(a) = 1, where B_k is the matrix such that a^T B_k a = conv(a,a)_k.

This is a non-convex QP (because B_k may not be PSD). However, for a fixed "active" constraint (say conv(a,a)_{k*} = M and it's the maximum), the problem becomes:
minimize a^T B_{k*} a subject to a ≥ 0, sum(a) = 1, conv(a,a)_k ≤ a^T B_{k*} a for all k.

Still non-convex.

**Alternative: Interior point on the max**

Introduce variable M and minimize M subject to conv(a,a)_k ≤ M for all k, a ≥ 0, sum(a) = 1. This is a QCQP. Can be solved with general non-convex solvers, but within the time budget and for n=400, it might be too slow.

**Alternative: Iterative reweighting**

At each iteration, identify the active set S = {k : conv(a,a)_k ≈ max}. Then solve the LP:
max sum(a) subject to conv(a, a)_k ≤ M for k in S, a ≥ 0, sum(a) = 1.

But conv(a,a)_k ≤ M is quadratic, not linear. So this doesn't work directly.

import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _conv_full(a):
    n = len(a)
    if n <= 1024:
        return np.convolve(a, a, mode='full')
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)


def _build_constraint_matrix(f, n):
    """Build A where (A@g)_k = conv(f, g)_k."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, active_set=None, timeout=3.0):
    """LP: max sum(g) s.t. conv(f,g)_k <= M (for active k), g >= 0."""
    if active_set is not None:
        # Only constrain active set (faster LP)
        A_rows = np.zeros((len(active_set), n), dtype=np.float64)
        for idx, k in enumerate(active_set):
            for j in range(n):
                if 0 <= k - j < n:
                    A_rows[idx, j] = f[j]
        b_ub = np.full(len(active_set), M)
    else:
        A_full = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A_full[j:j + n, j] = f
        A_rows = A_full
        b_ub = np.full(2 * n - 1, M)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_rows, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.maximum(np.asarray(res.x), 0.0)
    if g.sum() < 1e-15:
        return None
    return g

    best = f.copy()

            if time.monotonic() > stop_time - 2.0:

            cands.add(0.0)
            cands.add(orig)

    return best, best_sc

        res = np.asarray(res, dtype=np.float64).ravel()
        if not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0.0, 1000.0)
        if res.sum() < 0.01:
            return
        if sc < best_score - 1e-10:
            try:
                exact = float(evaluate_sequence([float(x) for x in res]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-10:
                best_seq = res.copy()
                best_score = exact

    # Build initializations
    ia = np.arange(n, dtype=np.float64)
    rng0 = np.random.default_rng(seed)
    inits = []
    inits.append(parent / max(parent.sum(), 1e-10))
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * ia / n)) + 0.05)
    ctr = (n - 1) / 2.0
    inits.append(np.maximum(0.1, 1.0 - ((ia - ctr) / (0.48 * n)) ** 2))
    inits.append(1.0 + 0.35 * np.cos(4.0 * np.pi * ia / n) + 0.15 * np.cos(6.0 * np.pi * ia / n))
    inits.append(np.exp(-0.5 * (np.minimum(ia, n - ia) / (0.18 * n)) ** 2) + 0.25)
    step_m = np.tile([1.0, 0.65, 0.4, 0.25, 0.55], n // 5 + 1)[:n]
    inits.append(step_m)
    pert = np.abs(parent / max(parent.sum(), 1e-10) + rng0.normal(0, 0.12, n)) + 0.02
    inits.append(pert)
    # Checkerboard modulation
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.45)
    inits.append(alt * (0.8 + 0.2 * np.cos(2.0 * np.pi * ia / n)))

    # Deduplicate
    uniq = [inits[0]]
    for ix in inits[1:]:
        if ix.sum() < 1e-6:
            continue
        if not any(np.allclose(ix, u, atol=0.05) for u in uniq):
            uniq.append(ix)
    inits = uniq

    # Phase 1: LP on each initialization (~48% time)
    p1_end = t0 + budget * 0.48
    time_per = max(12.0, (p1_end - t0) / max(1, len(inits)))
    for idx, init in enumerate(inits):
        if time.monotonic() > p1_end - 12.0:
            break
        stop = min(p1_end - 4.0, time.monotonic() + time_per)
        _try(lambda i=init.copy(), s=stop, d=seed + 500 * (idx + 1): _lp_loop(i, s, d),
             stop, seed + idx)

    # Phase 2: Further LP on best found (~12% time)
    p2_end = t0 + budget * 0.62
    if time.monotonic() < p2_end - 18.0:
        norm_bw = best_seq / max(best_seq.sum(), 1e-10)
        _try(lambda: _lp_loop(norm_bw, p2_end - 3.0, seed + 99999), p2_end, seed)

    # Phase 3: Finer LP warm restart (~8% time)
    p3_end = t0 + budget * 0.70
    if time.monotonic() < p3_end - 12.0:
        # Slight perturbation of best
        rng_p = np.random.default_rng(seed + 31337)
        pert_best = np.clip(best_seq + rng_p.normal(0, 0.03 * max(1.0, best_seq.mean()), n), 0, 1000)
        pert_best = pert_best / max(pert_best.sum(), 1e-10)
        _try(lambda: _lp_loop(pert_best, p3_end - 2.0, seed + 44444), p3_end, seed)

    # Phase 4: Intensive CD polish (~25% time)
    p4_end = min(t0 + budget * 0.95, deadline - 8.0)
    if time.monotonic() < p4_end - 10.0:
        norm_bw = best_seq / max(best_seq.sum(), 1e-10)
        try:
            fc, sc = _cd_pass(norm_bw, p4_end, seed + 55555, max_passes=500)
            if sc < best_score - 1e-10:
                try:
                    exact = float(evaluate_sequence([float(x) for x in fc]))
                except Exception:
                    exact = math.inf
                if math.isfinite(exact) and exact < best_score - 1e-10:
                    best_seq = fc.copy()
                    best_score = exact
        except Exception:
            pass

    # Phase 5: Final CD with more passes on the best
    final_cd_end = min(deadline - 5.0, p4_end + max(0, (deadline - 10.0) - p4_end) * 0.6)
    if time.monotonic() < final_cd_end - 5.0:
        norm_bw2 = best_seq / max(best_seq.sum(), 1e-10)
        try:
            fc, sc = _cd_pass(norm_bw2, final_cd_end, seed + 88888, max_passes=400)
            if sc < best_score - 1e-10:
                try:
                    exact = float(evaluate_sequence([float(x) for x in fc]))
                except Exception:
                    exact = math.inf
                if math.isfinite(exact) and exact < best_score - 1e-10:
                    best_seq = fc.copy()
                    best_score = exact
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0] * 100
        best_seq = fallback

    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    if not result or any(math.isnan(x) or math.isinf(x) for x in result):
        result = [float(x) for x in fallback]

    return result
