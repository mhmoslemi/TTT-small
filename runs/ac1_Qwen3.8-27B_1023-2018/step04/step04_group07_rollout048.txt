```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    """Full auto-convolution using FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fast(a):
    """Fast score computation."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_toep(f, n):
    """Build the convolution matrix A where A[k,j] = f[k-j] if valid."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_direction(f, n, M, time_limit=10.0, focused_k=0):
    """Solve LP to find direction g."""
    A = _build_toep(f, n)
    c = -np.ones(n)
    if focused_k > 0:
        b_conv = _conv_full(f)
        top_idx = np.argsort(-b_conv)[:focused_k]
        A_sub = A[top_idx]
        b_ub = np.full(len(top_idx), M)
    else:
        A_sub = A
        b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs

def _golden_search_1d(f, a, b, iters=40):
    """Golden section search for minimum of f on [a, b]."""
    gr = (1.0 + math.sqrt(5.0)) / 2.0
    c = b - (b - a) / gr
    d = a + (b - a) / gr
    fc = f(c)
    fd = f(d)
    for _ in range(iters):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - (b - a) / gr
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + (b - a) / gr
            fd = f(d)
    return (a + b) / 2.0
```

```python
            x_hi = max(x_hi, x_lo + 1e-8)

            if new_sc < cur_sc - 1e-15:
                delta = x_opt - ai
                a[i] = x_opt
                S += delta
                # Update B (convolution)
                # B[k] changes for k in [i, i+n-1]
                # For k in [i, i+n-1], k != 2i: B[k] += 2*delta*a[k-i]
                # For k = 2i: B[2i] += (x_opt^2 - ai^2) = (x_opt-ai)(x_opt+ai)
                ks2 = np.arange(max(0, i), min(2*n-1, i+n))
                ks2_neq = ks2[ks2 != 2*i]
                if len(ks2_neq) > 0:
                    B[ks2_neq] += 2.0 * delta * a[ks2_neq - i]
                if has_quad:
                    B[2*i] += x_opt*x_opt - ai*ai
                ai = x_opt  # update for next iterations
                best_score = new_sc if new_sc < best_score else best_score
                improved = True
        
        sweeps += 1
        if not improved and sweeps > 3:
            break
    
    return a, best_score, sweeps
```

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a, B=None):

        return math.inf

        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
