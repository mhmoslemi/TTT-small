import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(414, dtype=np.float64)
    if p.size == 0:
        p = np.ones(414, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.sum() < 0.01:
        p = np.ones(414, dtype=np.float64)
    return p


def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _grad_maxconv(a):
    n = len(a)
    b = _fft_conv(a)
    m = float(np.max(b))
    eps = max(1e-14, 1e-10 * m)
    active = np.flatnonzero(b >= m - eps)
    grad = np.zeros(n)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += 2.0 * a[k - idx]
    if len(active) > 0:
        grad /= len(active)
    return grad, m, active


def _lp_dir(f, n, M, A=None, timeout=6.0):
    if A is None:
        A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _line_search(f, g, cur_sc):
    best_t = 0.0
    best_sc_t = cur_sc
    # Check g endpoint
    s1 = _score(g) if g.sum() > 0.01 else math.inf
    if s1 < best_sc_t:
        best_sc_t, best_t = s1, 1.0
    hi = 1.3 if s1 < cur_sc else 1.0
    lo = 0.0
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    for _ in range(22):
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        t1, t2 = c1, c2
        fn1 = np.maximum((1 - t1) * f + t1 * g, 0)
        fn2 = np.maximum((1 - t2) * f + t2 * g, 0)
        s1 = _score(fn1) if fn1.sum() > 0.01 else math.inf
        s2 = _score(fn2) if fn2.sum() > 0.01 else math.inf
        if s1 < s2:
            hi = c1
            if s1 < best_sc_t:
                best_sc_t, best_t = s1, c1
        else:
            lo = c2
            if s2 < best_sc_t:
                best_sc_t, best_t = s2, c2
        if hi - lo < 1e-8:
            break
    return best_t, best_sc_t

def _lp_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0, None)
    f = f / max(f.sum(), 1e-15)
    best_f = f.copy()
    best_sc = _score(f)
    stall = 0
    max_stall = 500
    A = _build_A(f, n)
    rebuild_every = 15
    
    for it in range(50000):
        if time.monotonic() > stop_time:
            break
        b = _fft_conv(f)
        M = float(np.max(b))
        if M < 1e-20:
            break
        g = _lp_dir(f, n, M, A=A, timeout=5.0)
        if g is None:
            # Fallback to subgradient
            sg, _, _ = _grad_maxconv(f)
            sg -= np.mean(sg)
            g = np.maximum(-sg, 0)
            gs = g.sum()
            if gs < 1e-15:
                stall += 10
                if stall > max_stall:
                    break
                f = np.maximum(f + rng.normal(0, 0.04, n), 0)
                f /= max(f.sum(), 1e-15)
                A = _build_A(f, n)
                continue
            g /= gs
        else:
            gs = g.sum()
            if gs < 1e-15:
                stall += 5
                continue
            g /= gs
        dist = np.linalg.norm(g - f)
        if dist < 1e-12:
            stall += 3
            if stall > max_stall:
                break
            continue
        bt, bsc = _line_search(f, g, best_sc)
        if bt > 1e-8 and bsc < best_sc - 1e-14:
            fn = np.maximum((1 - bt) * f + bt * g, 0)
            fn /= max(fn.sum(), 1e-15)
            sc = _score(fn)
            if sc < best_sc:
                best_sc = sc
                best_f = fn.copy()
            f = fn
            stall = 0
            if (it + 1) % rebuild_every == 0:
                A = _build_A(f, n)
        else:
            stall += 1
            if stall >= max_stall:
                f = np.maximum(f + rng.normal(0, 0.06, n), 0)
                f /= max(f.sum(), 1e-15)
                A = _build_A(f, n)
                stall = 0
                max_stall += 100
    return best_f, best_sc

def _subgrad_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0, None)
    f = f / max(f.sum(), 1e-15)
    best_f = f.copy()
    best_sc = _score(f)
    alpha = 0.5
    k = 0
    no_imp = 0
    while time.monotonic() < stop_time:
        k += 1
        t = alpha / math.sqrt(k)
        sg, M, _ = _grad_maxconv(f)
        sg -= np.mean(sg)
        ng = np.linalg.norm(sg)
        if ng < 1e-16:
            break
        fn = np.maximum(f - t * sg / (ng * ng), 0)
        fn = fn / max(fn.sum(), 1e-15)
        sc = _score(fn)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_f = fn.copy()
            f = fn
            alpha = min(2.0, alpha * 1.02)
            no_imp = 0
        else:
            alpha = max(0.01, alpha * 0.95)
            no_imp += 1
            if no_imp > 3000:
                f = np.maximum(f + rng.normal(0, 0.03, n), 0)
                f /= max(f.sum(), 1e-15)
                alpha = 0.5
                no_imp = 0
            if no_imp > 10000:
                break
    return best_f, best_sc

def _coord_descent(f0, stop_time, seed, max_epochs=60):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0, None)
    n = len(f)
    best_sc = _score(f)
    best_f = f.copy()
    for epoch in range(max_epochs):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            if orig < 1e-15:
                continue
            bv, bvs = orig, best_sc
            for mult in [0.2, 0.5, 0.7, 0.9, 1.1, 1.3, 1.6, 2.0, 0.0]:
                f[i] = orig * mult
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < bvs - 1e-15:
                    bv, bvs = f[i], sc
            f[i] = bv
            if bvs < best_sc - 1e-15:
                best_sc = bvs
                best_f = f.copy()
                improved = True
        # 2D moves
        for _ in range(min(100, n)):
            if time.monotonic() > stop_time:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-14 and oj < 1e-14:
                continue
            for mi in [0.4, 0.7, 1.4, 2.0]:
                for mj in [0.4, 0.7, 1.4, 2.0]:
                    f[i] = oi * mi if oi > 1e-14 else 0.0
                    f[j] = oj * mj if oj > 1e-14 else 0.0
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        best_f = f.copy()
                        improved = True
            f[i], f[j] = oi, oj
        if not improved:
            break
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 25.0
    
    parent = _get_parent()
    n = len(parent)
    if n < 2:
        return [float(x) for x in parent] if parent.size else [1.0]
    
    psum = float(parent.sum())
    pnorm = parent / max(psum, 1e-15)
    parent_sc = _score(parent)
    best_seq = parent.copy()
    best_score = parent_sc
    
    def _upd(seq, sc):
        nonlocal best_seq, best_score
        if (math.isfinite(sc) and sc < best_score - 1e-12):
            if np.all(np.isfinite(seq)) and seq.sum() >= 0.01 and len(seq) > 0:
                best_seq = np.clip(seq, 0, 1000.0).copy()
                best_score = sc
    
    # Phase 1: Main LP (55%)
    p1 = t0 + 0.55 * budget
    if time.monotonic() < p1:
        r, s = _lp_optimize(pnorm, n, p1, seed)
        if math.isfinite(s):
            _upd(r * psum, s)
    
    # Phase 2: LP from perturbations (12%)
    p2 = t0 + 0.67 * budget
    if time.monotonic() < p2:
        for pi in range(2):
            if time.monotonic() > p2:
                break
            rng_p = np.random.default_rng(seed + 1000 + pi)
            p2_init = np.maximum(pnorm + rng_p.normal(0, 0.1, n), 0)
            p2_init /= max(p2_init.sum(), 1e-15)
            stop = min(p2 - 2, time.monotonic() + (budget * 0.05))
            r2, s2 = _lp_optimize(p2_init, n, stop, seed + 2000 + pi)
            if math.isfinite(s2):
                _upd(r2 * psum, s2)
    
    # Phase 3: Subgradient (10%)
    p3 = t0 + 0.77 * budget
    if time.monotonic() < p3:
        fsg = best_seq / max(best_seq.sum(), 1e-15)
        r3, s3 = _subgrad_optimize(fsg, len(fsg), p3, seed + 3000)
        if math.isfinite(s3):
            _upd(r3 * max(best_seq.sum(), 1e-15), s3)
    
    # Phase 4: Coordinate descent (10%)
    p4 = t0 + 0.87 * budget
    if time.monotonic() < p4:
        r4, s4 = _coord_descent(best_seq, p4, seed + 4000, max_epochs=50)
        if math.isfinite(s4):
            _upd(r4, s4)
    
    # Phase 5: Final polish (to deadline)
    if time.monotonic() < deadline:
        r5, s5 = _coord_descent(best_seq, deadline, seed + 5000, max_epochs=40)
        if
