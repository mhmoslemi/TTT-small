import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize_scalar


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(400, dtype=np.float64)
    if p.size == 0:
        p = np.ones(400, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(400, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    if n <= 1024:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_bfpw(f, n, stop_time, seed, max_iters=600):
    """Frank-Wolfe with LP direction finding and exact convex line search."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f, dtype=np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / S
    best_f = f.copy()
    best_score = _score(f)

    prev_dir = None
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break

        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        A = _build_A(f, n)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method="highs",
                          options={"presolve": True, "time_limit": 5.0})
        except Exception:
            break

        if res.status != 0 or res.x is None:
            break

        g = np.asarray(res.x, dtype=np.float64)
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g_dir = g / g_sum

        dir_vec = g_dir - f
        dir_norm = float(np.linalg.norm(dir_vec))

        if prev_dir is not None and prev_dir is not False:
            pn = float(np.linalg.norm(prev_dir))
            if pn > 1e-12 and dir_norm > 1e-12:
                cos_sim = float(np.dot(dir_vec, prev_dir) / (dir_norm * pn))
                if cos_sim > 0.9995 and dir_norm < 1e-5:
                    stall += 1
                else:
                    stall = 0
            else:
                stall += 1
        prev_dir = dir_vec.copy()

        if dir_norm < 1e-12:
            break
        if stall > 50:
            break

        # Precompute cross-correlations for line search
        conv_f = b
        conv_fd = np.convolve(f, g_dir, mode="full")
        conv_gd = _conv_full(g_dir)

        def line_obj(t):
            a1 = (1.0 - t) * (1.0 - t)
            a2 = 2.0 * t * (1.0 - t)
            a3 = t * t
            cf = a1 * conv_f + a2 * conv_fd + a3 * conv_gd
            return float(cf.max())

        try:
            res_ls = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                     options={'xatol': 1e-12})
            t_opt = float(res_ls.x)
        except Exception:
            t_opt = 0.5

        f_new = (1.0 - t_opt) * f + t_opt * g_dir
        f_new = np.clip(f_new, 0, None)
        fns = float(f_new.sum())
        if fns < 1e-12:
            break
        f_new = f_new / fns

        sc_new = _score(f_new)
        if sc_new < best_score - 1e-16:
            best_f = f_new.copy()
            best_score = sc_new
            f = f_new
        elif sc_new <= best_score + 1e-14:
            f = f_new

    return best_f, best_score


def _cd_optimize(f0, stop_time, seed, max_sweeps=2000):
    """Intensive coordinate descent with exact 1D minimization."""
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 1e-15:
        f = np.ones(n) / n
        S = 1.0
    else:
        f = f / S
        S = 1.0

    conv = _conv_full(f)
    best_score = 2.0 * n * float(conv.max())
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time:
                return best_f, best_score

            v = f[i]
            start = i
            end = min(i + n, 2 * n - 1)
            seg_len = end - start
            x_aff = f[:seg_len]
            conv_base_aff = conv[start:end].copy()
            two_i = 2 * i
            has_quad = two_i < 2 * n - 1 and two_i >= start and two_i < end
            quad_offset = two_i - start if has_quad else -1

            # Compute unchanged max
            if start > 0:
                m_left = float(conv[:start].max())
            else:
                m_left = 0.0
            if end < 2 * n - 1:
                m_right = float(conv[end:].max())
            else:
                m_right = 0.0
            m_unchanged = max(m_left, m_right)

            def obj_1d(w):
                new_seg = conv_base_aff + 2.0 * w * x_aff
                if has_quad:
                    new_seg[quad_offset] += w * w
                m_new = max(m_unchanged, float(new_seg.max()))
                S_new = S + w
                if S_new < 1e-8:
                    return 1e10
                return 2.0 * n * m_new / (S_new * S_new)

            # Coarse grid
            w_lo = max(-0.99 * S, -5.0)
            w_hi = max(5.0 * v, 0.5) + 0.5
            w_hi = min(w_hi, 10.0)
            n_grid = 40
            w_grid = np.linspace(w_lo, w_hi, n_grid)
            best_w = v
            best_obj = obj_1d(0.0)

            for wg in w_grid:
                o = obj_1d(wg)
                if o < best_obj:
                    best_obj = o
                    best_w = wg

            # Fine refinement with Brent
            dw = (w_hi - w_lo) / n_grid
            lo_b = max(w_lo, best_w - 2.0 * dw)
            hi_b = min(w_hi, best_w + 2.0 * dw)
            if hi_b - lo_b > 1e-12:
                try:
                    res_b = minimize_scalar(obj_1d, bounds=(lo_b, hi_b),
                                            method='bounded', options={'xatol': 1e-12})
                    w_b = float(res_b.x)
                    o_b = obj_1d(w_b)
                    if o_b < best_obj:
                        best_obj = o_b
                        best_w = w_b
                except Exception:
                    pass

            # Check if improved
            if best_obj < best_score - 1e-15 and best_w != 0.0:
                w = best_w
                f[i] = v + w
                S = S + w
                conv[start:end] += 2.0 * w * x_aff
                if has_quad:
                    conv[two_i] += w * w
                best_score = best_obj
                improved = True

        if not improved and sweep > 5:
            # Try perturbation
            if sweep % 20 == 0:
                f = np.clip(f + rng.normal(0, 0.005, n), 0, None)
                fs = float(f.sum())
                if fs > 1e-12:
                    f = f / fs
                    S = 1.0
                    conv = _conv_full(f)
                    best_score_cur = 2.0 * n * float(conv.max())
                    if best_score_cur < best_score:
                        best_score = best_score_cur
                        best_f = f.copy()
                improved = True

        # Track best
        cur_score = 2.0 * n * float(conv.max()) / (S * S)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

    return best_f, best_score


def _grad_descent(f0, n, stop_time, seed, max_iters=20000):
    """Gradient descent with smoothed max for polishing."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / S
    best_f = f.copy()
    best_score = _score(f)

    alpha = 0.03
    tau = 50.0

    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break

        b = _conv_full(f)
        S = float(f.sum())
        M = float(b.max())
        if M < 1e-18:
            break
        score = 2.0 * n * M / (S * S)

        if score < best_score:
            best_score = score
            best_f = f.copy()

        # Softmax smoothed max
        tau_b = tau * (b - M)
        tau_b -= tau_b.max()
        p = np.exp(tau_b)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(2 * n - 1)
            p[int(np.argmax(b))] = 1.0
            p_sum = 1.0
        p /= p_sum

        # Gradient of smoothed max w.r.t. f
        f_rev = f[::-1]
        P = rfft(p, 2 * n - 1)
        Fr = rfft(f_rev, 2 * n - 1)
        conv_val = irfft(P * Fr, 2 * n - 1)[:2 * n - 1]
        grad_max = 2.0 * conv_val[n - 1:2 * n - 1]

        # Gradient of score = 2n*M/S^2
        score_grad = (2.0 * n / (S * S)) * (grad_max - 2.0 * M * f / S)

        # Projected gradient step
        f_new = f - alpha * score_grad
        # Project to simplex {x >= 0, sum(x) = 1}
        u = np.sort(f_new)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        rho = np.nonzero(cond)[0]
        if len(rho) == 0:
            f_new = np.ones(n) / n
        else:
            rho = rho[-1]
            theta = (cssv[rho] - 1) / (rho + 1.0)
            f_new = np.maximum(f_new - theta, 0)

        sc_new = _score(f_new)
        if sc_new < score - 1e-15:
            f = f_new
            alpha = min(alpha * 1.05, 0.1)
        else:
            alpha = max(alpha * 0.5, 1e-8)
            if alpha < 1e-7:
                f = np.clip(f + rng.normal(0, 0.003, n), 0, None)
                f /= f.sum()
                alpha = 0.01

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
