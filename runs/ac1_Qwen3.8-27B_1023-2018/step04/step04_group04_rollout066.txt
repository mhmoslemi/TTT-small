import numpy as np

from scipy.optimize import minimize, linprog

def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _subgrad_maxconv(a):
    n = len(a)
    b = _fft_conv(a)
    m = float(np.max(b))
    eps = max(1e-15, 1e-11 * m)
    active = np.flatnonzero(b >= m - eps)
    grad = np.zeros(n)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            idx = k - np.arange(i_lo, i_hi + 1)
            grad[i_lo:i_hi + 1] += 2.0 * a[idx]
    if len(active) > 0:
        grad /= len(active)
    return grad, m, active


def _grad_lse(a, w):
    """Return 2*sum_m w[m]*a[m-j] for j=0..n-1 via one FFT cross-correlation."""
    n = len(a)
    w_r = w[::-1]
    size = next_fast_len(3 * n)
    W = rfft(w_r, size)
    A = rfft(a, size)
    conv = irfft(W * A, size)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def _smoothed_obj_grad(a, eps):
    n = len(a)
    S = float(np.sum(a))

        return 1e12, np.zeros(n)
    c = _fft_conv(a)

    lse = m + float(np.log(np.sum(np.exp(c_scaled - m))))
    w = np.exp(c_scaled - lse)

    grad_lse = _grad_lse(a, w)
    grad = 2.0 * n / (S ** 3) * (S * grad_lse - 2.0 * eps * lse)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return obj, grad


def _lp_direction_fast(f, n, M, timeout=3.0):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _lse_lbfgs(f0, n, stop_time, eps_fracs):

def _coord_descent(f0, stop_time, seed):

            if time.monotonic() > stop_time:
                break
            orig = f[i]
            cands = []
            if orig > 1e-15:
                cands = [0.0, orig * 0.4, orig * 0.7, orig * 1.3, orig * 1.7]

                best_f = f.copy()
                no_imp = 0
