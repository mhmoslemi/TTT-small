import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except:
        p = np.ones(200)
    if p.size == 0:
        p = np.ones(200)
    p = np.clip(p, 0, 1000)
    p = np.where(np.isfinite(p), p, 0.0)
    return p

def _cconv(a, b):
    n = max(len(a), len(b))
    size = next_fast_len(2*n - 1)
    A = rfft(a, size)
    B = rfft(b, size)
    c = irfft(A * B, size)[:2*n-1]
    return np.maximum(c, 0)

def _score_fast(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return math.inf
    b = _cconv(a, a)
    return 2.0 * n * b.max() / (s * s)

def _lp_dir(f, n, M, rows, timeout=4.0):
    # Build constraint matrix
    A = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        jlo = max(0, k - n + 1)
        jhi = min(k + 1, n)
        if jhi > jlo:
            A[idx, jlo:jhi] = f[k-jhi+1:k-jlo+1][::-1]
    
    c = -np.ones(n)
    b_ub = np.full(len(rows), M + 1e-12)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except:
        return None
    if res.status != 0:
        return None
    g = np.clip(res.x, 0, None)
    return g if g.sum() > 1e-15 else None

def _optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    target_sum = math.sqrt(2.0 * n)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
        s = n
    f = f / s * target_sum

    best_sc = _score_fast(f)
    
    t = 0.3
    stall = 0
    max_stall = 100
    all_rows = list(range(2*n - 1))
    active_rows = all_rows[:]

        b = _cconv(f, f)
        M = b.max()

        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:

def _cconv(a, b):
    """Cross-convolution using FFT."""
    n = max(len(a), len(b))
    size = next_fast_len(2 * n - 1)
    A = rfft(a, size)
    B = rfft(b, size)
    c = irfft(A * B, size)[:2 * n - 1]
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _cconv(a, a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)
