                        best_val_sc = sc
            
            best[i] = best_val
            if best_val_sc < best_sc - 1e-15:
                best_sc = best_val_sc

        n_pairs = min(200, n)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 8.0:

            if oi < 1e-14 and oj < 1e-14:

            best_pair_sc = best_sc
            bi, bj = oi, oj
            for mi in (0.5, 0.75, 1.25, 1.5):
                for mj in (0.5, 0.75, 1.25, 1.5):

                    nj = oj * mj

                    sc = _score(best)
                    if sc < best_pair_sc - 1e-15:

                        bi, bj = ni, nj
            best[i], best[j] = bi, bj
            if best_pair_sc < best_sc - 1e-15:
                best_sc = best_pair_sc
                improved = True
        
        if not improved:
            # Random perturbations
            for _ in range(100):
                if time.monotonic() > stop_time - 8.0:
                    break
                k = int(rng.integers(n))
                old = best[k]
                sigma = 0.02 * max(1.0, best.mean())
                best[k] = max(0.0, old + rng.normal(0, sigma))
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                else:
                    best[k] = old
            
            if not improved:
                no_imp_rounds += 1
        else:
            no_imp_rounds = 0
    
    # Final normalization
    fs = best.sum()
    if fs > 1e-10:
        best = best / fs * target_sum
        best_sc = _score(best)

def _conv_full(a):
    n = len(a)
    if n <= 800:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, n, M, A=None, timeout=3.0):
    if M <= 1e-18:
        return None
    if A is None:
        A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-14)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"presolve": True, "time_limit": timeout,
                               "dual_feasibility_tolerance": 1e-11,
                               "primal_feasibility_tolerance": 1e-11})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g


def _lp_dual_solve(f, n, M, A=None, timeout=3.0):
    """Solve dual: min M * sum(y) s.t. A^T y >= 1, y >= 0."""
    if M <= 1e-18:
        return None
    if A is None:
        A = _build_A(f, n)
    AT = A.T
    c_dual = np.full(2 * n - 1, M)
    A_ub_dual = -AT
    b_ub_dual = -np.ones(n)
    bounds_dual = [(0.0, None)] * (2 * n - 1)
    try:
        res = linprog(c_dual, A_ub=A_ub_dual, b_ub=b_ub_dual, bounds=bounds_dual,
                      method="highs", options={"presolve": True, "time_limit": timeout,
                                               "dual_feasibility_tolerance": 1e-11,
                                               "primal_feasibility_tolerance": 1e-11})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    y = np.asarray(res.x, dtype=np.float64)
    y = np.clip(y, 0.0, None)
    return y

def _polish_cd(a, stop_time, seed=4242):

    n = len(a)

    rng = np.random.default_rng(seed)

    max_no_imp = 12

        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 8.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc

            # Coarse grid
            multipliers = [0.0, 0.05, 0.1, 0.2, 0.35, 0.5, 0.65, 0.8,
                           0.9, 1.1, 1.2, 1.4, 1.7, 2.2, 3.0, 5.0]
            for mlt in multipliers:
                nv = orig * mlt if mlt > 0 else 0.0
                best[i] = nv
                sc = _score(best)
                if sc < best_val_sc - 1e-15:
                    best_val = nv
                    best_val_sc = sc

            # Refine
            if best_val > 1e-12:
                for _ in range(3):
                    lo = best_val * 0.85
                    hi = best_val * 1.15
                    gr = (1 + math.sqrt(5)) / 2

                    x2 = lo + (hi - lo) / gr
                    best[i] = x1; s1 = _score(best)

                        break
                    best[i] = x2; s2 = _score(best)
                    for _ in range(8):
                        if s1 < s2:
                            hi = x2; x2 = x1; s2 = s1

                            best[i] = x1; s1 = _score(best)
                        else:
                            lo = x1; x1 = x2; s1 = s2

                    if sv < best_val_sc - 1e-15:

            if best_val_sc < best_sc - 1e-15:

def _kkt_refine(f0, n, stop_time, seed):
    """KKT-based active set refinement."""
    rng = np.random.default_rng(seed)
    target_sum = math.sqrt(2.0 * n)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel(), 0, None).copy()
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    fs = f.sum()
    if fs < 1e-10:
        return f0, _score(f0)
    f = f / fs * target_sum
    best_f = f.copy()
    best_sc = _score(f)

    # Repeatedly solve the LP dual to get lambda, then construct new f from lambda
    for iteration in range(200):
        if time.monotonic() > stop_time - 8.0:
            break
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-14:
            break

        # Get dual variables
        A = _build_A(f, n)
        y = _lp_dual_solve(f, n, M, A=A, timeout=2.0)
        if y is None or y.sum() < 1e-14:
            break

        # Compute h[j] = sum_k y[k] * f[k-j] (valid terms only)
        # h[j] = sum_{k: 0<=k-j<n} y[k] * f[k-j] = (y * f)[j] in convolution sense
        # Actually: h[j] = sum_k y[k] * f[k-j] where k-j in [0,n-1]
        # This is the cross-correlation of y and f
        # h = np.convolve(y, f, mode='full') gives h_full[m] = sum_k y[k] f[m-k]
        # We want h[j] = sum_k y[k] f[k-j] = h_full[n-1-j + something]...
        # Let me just compute it directly
        h = np.zeros(n)
        for j in range(n):
            # h[j] = sum_k y[k] * f[k-j] for 0 <= k-j < n
            # k ranges from j to j+n-1, but k < 2n-1
            k_min = j
            k_max = min(2 * n - 2, j + n - 1)
            if k_min <= k_max:
                idx = slice(k_min, k_max + 1)
                f_idx = slice(0, k_max - k_min + 1)  # f[k-j] for k from k_min to k_max
                h[j] = np.dot(y[idx], f[f_idx])

        h_max = h.max()
        if h_max < 1e-14:
            break

        # New f: proportional to max(0, h - threshold)
        # Choose threshold so that sum(f_new) = target_sum
        # f_new[j] = max(0, h[j] - alpha) for some alpha
        # sum_j max(0, h[j] - alpha) = target_sum
        # Binary search for alpha
        h_sorted = np.sort(h)[::-1]
        # sum of top-m values minus m*alpha = target_sum
        # Try different support sizes
        best_alpha = None
        for m in range(1, n + 1):
            s_top = h_sorted[:m].sum()
            alpha = (s_top - target_sum) / m
            if alpha < h_sorted[m - 1] - 1e-15 and (m == n or alpha >= h_sorted[m] + 1e-15):
                best_alpha = alpha
                break
        if best_alpha is None:
            best_alpha = h.min() - 1e-3 * h_max

        f_new = np.maximum(0.0, h - best_alpha)
        fns = f_new.sum()
        if fns < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / fns * target_sum

        sc = _score(f_new)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_f = f_new.copy()
            f = f_new.copy()
        else:
            # Try a blend
            for tt in (0.3, 0.5, 0.7):
                f_blend = (1 - tt) * f + tt * f_new
                f_blend = np.clip(f_blend, 0, None)
                fbs = f_blend.sum()
                if fbs > 1e-10:
                    f_blend = f_blend / fbs * target_sum
                    sc_b = _score(f_blend)
                    if sc_b < best_sc - 1e-15:
                        best_sc = sc_b
                        best_f = f_blend.copy()
                        f = f_blend.copy()
                        break
            else:
                if iteration > 50:
                    break

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n = len(parent)

    # Initial score
    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not math.isfinite(best_score):
        best_seq = np.ones(n)
        best_score = _score(best_seq)

    def _try_update(candidate, sc):
        nonlocal best_seq, best_score
        if candidate is None or len(candidate) == 0:
            return
        cand = np.clip(np.asarray(candidate, dtype=np.float64).ravel(), 0, 1000)
        if cand.sum() < 0.01:
            return
        if not np.all(np.isfinite(cand)):
            return
        if sc < best_score - 1e-14:
            try:
                exact = float(evaluate_sequence([float(x) for x in cand]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-15:
                best_seq = cand.copy()
                best_score = exact

    # Phase 1: Accelerated FW on parent (main phase)
    phase1_end = min(deadline - 60.0, t0 + budget * 0.40)
    if time.monotonic() < phase1_end - 10.0:
        try:
            res, sc = _accelerated_fw(parent, n, phase1_end, seed=seed)
            _try_update(res, sc)
        except Exception:
            pass

    # Phase 2: Standard FW (different landscape)
    phase2_end = min(deadline - 50.0, t0 + budget * 0.55)
    if time.monotonic() < phase2_end - 10.0:
        try:
            res, sc = _fw_standard(parent, n, phase2_end, seed=seed + 1)
            _try_update(res, sc)
        except Exception:
            pass

    # Phase 3: KKT-based refinement
    phase3_end = min(deadline - 40.0, t0 + budget * 0.70)
    if time.monotonic() < phase3_end - 10.0:
        try:
            res, sc = _kkt_refine(best_seq, len(best_seq), phase3_end, seed=seed + 2)
            _try_update(res, sc)
        except Exception:
            pass

    # Phase 4: Coordinate descent polishing (most important for final squeeze)
    phase4_end = min(deadline - 20.0, t0 + budget * 0.90)
    if time.monotonic() < phase4_end - 10.0:
        try:
            res, sc = _polish_cd(best_seq, phase4_end, seed=seed + 100)
            _try_update(res, sc)
        except Exception:
            pass

    # Phase 5: Second polish pass
    phase5_end = deadline - 10.0
    if time.monotonic() < phase5_end - 5.0:
        try:
            res, sc = _polish_cd(best_seq, phase5_end, seed=seed + 200)
            _try_update(res, sc)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Fallback to parent
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            return [1.0]
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    best_score = min(best_score, final_sc)
    return [float(x) for x in best_seq]
