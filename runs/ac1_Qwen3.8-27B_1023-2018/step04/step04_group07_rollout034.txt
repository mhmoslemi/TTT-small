import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):

        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_conv_matrix(f, n):

    return A


def _lp_direction(f, n, M, top_k=0, time_limit=15.0):
    """Find g >= 0 maximizing sum(g) s.t. f*g <= M (element-wise)."""
    b = _conv_full(f)
    if top_k > 0 and top_k < 2 * n - 1:
        idx = np.argsort(-b)[:top_k]
        # Ensure we include the max
        if np.argmax(b) not in idx:
            idx[0] = np.argmax(b)
    else:
        idx = np.arange(2 * n - 1)
    
    A_full = _build_conv_matrix(f, n)
    A = A_full[idx]
    
    c = -np.ones(n)
    b_ub = np.full(len(idx), M)

        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': time_limit}
        )
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    g[np.abs(g) < 1e-12] = 0
    gs = g.sum()
    if gs < 1e-12:
        return None

For the LP phase, the key is to:

The LP is: maximize sum(g) subject to (f*g)_k <= M for all k, g >= 0.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_conv_matrix(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, top_k=0, time_limit=12.0):
    b = _conv_full(f)
    if top_k > 0 and top_k < 2 * n - 1:
        idx = np.argsort(-b)[:top_k]
        max_idx = np.argmax(b)
        if max_idx not in idx:
            idx[0] = max_idx
    else:
        idx = np.arange(2 * n - 1)
    A_full = _build_conv_matrix(f, n)
    A = A_full[idx]
    c = -np.ones(n)
    b_ub = np.full(len(idx), M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': time_limit}
        )
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g[np.abs(g) < 1e-12] = 0.0
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs


def _line_search(f, g, n, current_max, stop_time):

    best_val = current_max
    ts = np.linspace(0, 1, 25)
    for t in ts:

        f_new = (1.0 - t) * f + t * g
        m_new = float(_conv_full(f_new).max())
        if m_new < best_val:
            best_val = m_new
            best_t = t
    if 0.02 < best_t < 0.98:
        step = np.diff(ts)[0]
        lo = max(0.0, best_t - step)
        hi = min(1.0, best_t + step)
        gr = (1.0 + math.sqrt(5.0)) / 2.0
        for _ in range(14):
            if time.monotonic() > stop_time:
                break
            m1 = hi - (hi - lo) / gr
            m2 = lo + (hi - lo) / gr
            v1 = float(_conv_full((1.0 - m1) * f + m1 * g).max())
            v2 = float(_conv_full((1.0 - m2) * f + m2 * g).max())
            if v1 < v2:
                hi = m2
            else:
                lo = m1
            mid = (lo + hi) / 2.0

def _coord_descent_fast(a, stop_time, max_passes=5000, seed=42):
    n = len(a)
    a = np.clip(np.asarray(a, dtype=np.float64).ravel()[:n].copy(), 0, None)
    s = float(a.sum())
    if s < 1e-10:
        return a, math.inf

    b = _conv_full(a)
    best_score = 2.0 * n * float(b.max()) / (s * s)
    best_a = a.copy()
    best_s = s

    no_improve = 0
    max_no_improve = 15
    rng = np.random.default_rng(seed)

    for pass_idx in range(max_passes):
        if time.monotonic() > stop_time or no_improve >= max_no_improve:
            break

        if pass_idx % 100 == 0:
            b = _conv_full(a)

        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time:
                break

            orig = a[i]
            best_local = best_score
            best_nu = orig

            if orig > 1e-10:
                mults = (0.0, 0.2, 0.4, 0.6, 0.8, 0.9, 1.0, 1.1, 1.25, 1.5, 2.0, 3.0, 5.0)
            else:
                mults = (0.0, 0.002, 0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0, 5.0)

            for m in mults:
                nu = orig * m if orig > 1e-10 else m
                delta = nu - orig
                if abs(delta) < 1e-15:
                    continue

                new_s = s - orig + nu
                if new_s < 1e-10:
                    continue

                new_max = 0.0
                for k in range(2 * n - 1):
                    val = b[k]
                    if k == 2 * i:
                        val = val - orig * orig + nu * nu
                    else:
                        j = k - i
                        if 0 <= j < n:
                            val = val + 2.0 * delta * a[j]
                    if val > new_max:
                        new_max = val

                sc = 2.0 * new_max / (new_s * new_s)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_nu = nu

            if best_nu != orig:
                delta = best_nu - orig
                for k in range(2 * n - 1):
                    if k == 2 * i:
                        b[k] = b[k] - orig * orig + best_nu * best_nu
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] = b[k] + 2.0 * delta * a[j]
                a[i] = best_nu
                s = s - orig + best_nu
                np.maximum(b, 0, out=b)
                cur_score = 2.0 * float(b.max()) / (s * s)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                    best_s = s
                    improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1

    return best_a, best_score


def _pair_optimize(a, stop_time, seed=99):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel()[:n].copy(), 0, None)
    s = float(best.sum())
    if s < 1e-10:
        return best, math.inf
    best = best / s
    best_score = 2.0 * n * float(_conv_full(best).max())
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() < stop_time and no_improve < max_no_improve:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        oi, oj = best[i], best[j]
        total = oi + oj
        if total < 1e-12:
            no_improve += 1
            continue

        best_local = best_score
        best_frac = oi / total if total > 1e-15 else 0.5
        fracs = np.linspace(0, 1, 31)
        for frac in fracs:
            best[i] = frac * total
            best[j] = (1.0 - frac) * total
            sc = float(_conv_full(best).max()) * 2.0 * n
            if sc < best_local - 1e-14:
                best_local = sc
                best_frac = frac

        if best_frac != oi / total if total > 1e-15 else True:
            best[i] = best_frac * total
            best[j] = (1.0 - best_frac) * total
            if best_local < best_score - 1e-14:
                best = best / best.sum()
                best_score = best_local
                no_improve = 0
            else:
                best[i], best[j] = oi, oj
                no_improve += 1
        else:
            no_improve += 1

    return best, best_score


def _anneal_optimize(a, stop_time, seed=777):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel()[:n].copy(), 0, None)

    curr_score = best_score
    rng = np.random.default_rng(seed)

    t_total = max(1.0, stop_time - t_start)
    T0 = 0.001

        i = int(rng.integers(n))
        noise = rng.normal(0, 0.05, 1)
        new_val = max(0.0, curr[i] + noise[0])
        old_val = curr[i]
        if abs(new_val - old_val) < 1e-12:

        if ns < 1e-10:

            if curr_score < best_score:

        else:

    return best, best_score
