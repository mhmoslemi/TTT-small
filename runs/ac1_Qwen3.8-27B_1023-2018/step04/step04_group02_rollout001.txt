```python
def _conv(a):

    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)
```

Now for the LP step, I need to build the convolution matrix. For a sequence f of length n, the matrix A has shape (2n-1, n) where A[k, j] = f[k-j] if 0 ≤ k-j < n, else 0. This is a banded (actually, lower-Hessenberg-like) matrix.

For n=400: 799 × 400 = 319,600 entries. About 2.5 MB. Fine for dense.

Let me build it efficiently:
```python
def _build_A(f, n):
    # A[k, j] = f[k - j] if 0 <= k-j < n else 0
    # k ranges 0 to 2n-2, j ranges 0 to n-1
    A = np.zeros((2*n-1, n))
    for j in range(n):
        valid_k = range(j, j+n)  # k from j to j+n-1
        A[valid_k, j] = f
    return A
```

Actually, this is a Toeplitz-like matrix (but not exactly Toeplitz since f isn't periodic). Let me just build it directly.

For the LP, I'll use HiGHS with a time limit. The problem:
- Maximize: -sum(g) (minimize sum(g) with linprog which minimizes)
  Actually, linprog minimizes c^T x. To maximize sum(g), set c = -ones.
- Constraints: A @ g <= M (upper bounds)
- Bounds: g >= 0

```python
def _LP_direction(f, n, M, time_limit=5.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n,
                  method='highs', options={'time_limit': time_limit})
    if res.status != 0:
        return None
    g = np.clip(res.x, 0, None)
    if g.sum() < 1e-15:
        return None
    return g
```

```python
def _optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.astype(float).copy(), 0, None)
    if f.size != n:
        # interpolate or truncate
        ...

        f = np.ones(n)
    
    best = f.copy()
    best_sc = _score(f)
    t_step = 0.3
    stall = 0
    
    while time.monotonic() < stop_time - 5:
        b = _conv(f)
        M = float(b.max())
        
        g = _LP_direction(f, n, M, time_limit=4.0)
        if g is None:
            # Fallback: random perturbation
            f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
            stall += 5
            if stall > 100:
                f = best.copy()
                stall = 0
            continue
        
        g = g / g.sum() * f.sum()  # normalize to same sum
        
        # Line search
        best_local = _score(f)
        best_t = 0
        for t in [t_step, t_step*0.5, t_step*0.25, t_step*1.5, 0.05]:
            f_new = np.clip((1-t)*f + t*g, 0, None)
            sc = _score(f_new)
            if sc < best_local:
                best_local = sc
                best_t = t
        
        if best_t > 0:
            f = (1-best_t)*f + best_t*g
            f = np.clip(f, 0, None)
            sc = _score(f)
            if sc < best_sc - 1e-12:
                best = f.copy()
                best_sc = sc
            t_step = min(0.8, best_t * 1.2)
            stall = 0
        else:
            stall += 1
            t_step = max(0.02, t_step * 0.5)
            if stall > 80:
                # Restart from best with perturbation
                f = np.clip(best + rng.normal(0, 0.05 * max(1, best.mean()), n), 0, None)
                t_step = 0.3
                stall = 0
    
    return best, best_sc
```

```python
def _cd_polish(f, stop_time, seed, max_outer=100):
    n = len(f)
    rng = np.random.default_rng(seed)
    f = f.astype(float).copy()
    cur = _score(f)
    best = f.copy()
    best_sc = cur
    
    multipliers = [0.1, 0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0, 0.0]
    
    for outer in range(max_outer):
        if time.monotonic() > stop_time - 3:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3:
                break
            orig = f[i]
            best_v = orig
            best_local = cur
            for m in multipliers:
                v = orig * m if m > 0 else 0.0
                if m == 0 and orig < 1e-3:
                    continue
                f[i] = v
                sc = _score(f)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_v = v
            f[i] = best_v
            if best_local < cur - 1e-15:
                cur = best_local
                improved = True
                if cur < best_sc:
                    best_sc = cur
                    best = f.copy()
        if not improved:
            break
    return best, best_sc
```
