# LP-based structured direction:  max sum(g)  s.t.  (a*g)[k] <= M,  g >= 0.

def _lp_direction(a, n, M, time_limit=5.0):
    if linprog is None:
        return None
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    try:
        res = linprog(
            -np.ones(n),
            A_ub=A,
            b_ub=np.full(2 * n - 1, M),
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": float(time_limit)},
        )
    except Exception:
        return None
    if res is None or getattr(res, "status", 1) != 0 or getattr(res, "x", None) is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-12:
        return None
    return g / g.sum()  # normalized to sum 1

                break
            if best_score < 1.5030:
                break
            init = _alt_init(n, kind)
            # seed from current best where useful: blend
            if best_seq is not None and best_seq.size == n:
                init = (init + best_seq / best_seq.sum()) / 2.0
            try:
                r, sc = _try_lengths_pipeline(n, init, min(alt_end - 10.0, time.monotonic() + budget * 0.12),
                                              seed + 1000 + kind)
                _consider(r, sc)
            except Exception:
                pass

    # ---- Nearby lengths (interpolate best) --------------------------------
    if best_seq is not None and best_score > 1.5028 and time.monotonic() < hard_deadline - 30.0:
        cands_L = []
        for L in (n - 8, n + 8, n - 16, n + 16, max(32, n // 2 * 2), n * 2):
            if 16 <= L <= 1024 and L not in cands_L and L != n:
                cands_L.append(L)
        cands_L.sort(key=lambda L: abs(L - n))
        pool_end = min(hard_deadline - 30.0, time.monotonic() + (hard_deadline - time.monotonic()) * 0.35)
        for L in cands_L[:3]:
            if time.monotonic() > pool_end:
                break
            if best_score < 1.5028:
                break
            init = np.interp(np.linspace(0.0, 1.0, L), np.linspace(0.0, 1.0, best_seq.size), best_seq)
            init = np.clip(init, 0.0, None)
            if init.sum() < 0.01:
                init = np.ones(L) / L
            try:
                r, sc = _try_lengths_pipeline(L, init, min(pool_end - 8.0, time.monotonic() + budget * 0.10),
                                              seed + 5000 + L)
                _consider(r, sc)
            except Exception:
                pass

    # ---- Final polish on the best found -----------------------------------
    if best_seq is not None and time.monotonic() < hard_deadline - 25.0:
        try:
            r, sc = _cd(best_seq, hard_deadline - 22.0, 4000, seed + 991)
            _consider(r, sc)
        except Exception:
            pass

    if best_seq is not None and time.monotonic() < hard_deadline - 15.0:
        try:
            r, sc = _pair_polish(best_seq, hard_deadline - 14.0, seed + 7777)
            _consider(r, sc)
        except Exception:
            pass

    if best_seq is not None and time.monotonic() < hard_deadline - 8.0:
        try:
            r, sc = _cd(best_seq, hard_deadline - 7.0, 2000, seed + 4242)
            _consider(r, sc)
        except Exception:
            pass

    # ---- Final exact validation -------------------------------------------
    if best_seq is None:
        best_seq = fallback
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.size == 0 or float(best_seq.sum()) < 0.01:
        best_seq = fallback

    try:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        fs = math.inf
    if not math.isfinite(fs):
        try:
            fs = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            fs = math.inf
        if not math.isfinite(fs):
            return [1.0] * max(1, n)
        best_seq = fallback

    return [float(x) for x in best_seq]
