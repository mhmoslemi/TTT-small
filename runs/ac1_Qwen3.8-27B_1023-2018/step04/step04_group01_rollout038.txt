def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    n = len(f)

    b_shifted = (b - b.max()) / tau
    exp_b = np.exp(b_shifted)
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)

        n_valid = k_end - j
        grad[j] = 2.0 * np.dot(p[j:k_end], f[:n_valid])
    return grad

def _grad_descent(f_init, n, stop_time, tau_init, tau_final, seed):
    """Projected gradient descent on smoothed max."""
    rng = np.random.default_rng(seed)
    f = _project_simplex(f_init)

                cands = [0.0, 0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 1.0 / n, 2.0 / n, 5.0 / n]
            else:
                cands = [
                    0.0, v * 0.1, v * 0.25, v * 0.5, v * 0.75,
                    v, v * 1.25, v * 1.5, v * 2.0, v * 3.0, v * 5.0,
                    1.0 / n, 2.0 / n, 5.0 / n, 0.01, 0.05, 0.1, 0.5, 1.0,
                ]
            # Deduplicate and filter
            cands = sorted(set(min(max(x, 0.0), 100.0) for x in cands))

            for cv in cands:
                obj = _eval(cv)
                if obj < best_obj:
                    best_obj = obj
                    best_val = cv

            # Golden section refinement around best
            if best_val != v:
                lo = max(0.0, best_val * 0.5) if best_val > 0 else 0.0
                hi = best_val * 2.0 if best_val > 0 else best_val + 0.01
                hi = min(hi, max(hi, best_val + 0.005))
                for _ in range(12):
                    m1 = hi - (hi - lo) * 0.382
                    m2 = lo + (hi - lo) * 0.382
                    f1 = _eval(m1)
                    f2 = _eval(m2)
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                    if _eval((lo + hi) / 2) < best_obj:
                        best_obj = _eval((lo + hi) / 2)
                        best_val = (lo + hi) / 2

            if best_val != v and best_obj < best_score - 1e-14:
                delta = best_val - v
                f[i] = best_val
                S += delta
                c[k_start:k_end] += delta * (2.0 * f[:n_aff] - delta * (f[i] - best_val) * 0)
                # Recompute influence with old f for correctness
                # Actually, influ was 2*f_old[:n_aff], and we need to add delta * influ_old
                # But we already modified f[i], so let me redo this properly
                # The correct update: c[k] += 2*delta*f_old[k-i] + delta^2*[k=2i]
                # We stored influ = 2*f_old[:n_aff] before the update
                # So c[k_start:k_end] should get delta*influ added (but we already partially did)
                # Let me just recompute to be safe for the delta^2 part
                if 0 <= idx_2i < n_aff:
                    c[idx_2i] += delta * delta
                # Note: the linear part delta*influ was NOT added yet (I removed it above)
                # Let me fix: I need to add delta * influ
                # But influ was computed from f BEFORE the update, which is correct
                # However, I already modified f[i], so I need to use the stored influ
                # Let me redo: the update should be c[k_start:k_end] += delta * influ_old
                # where influ_old = 2 * f_old[0:n_aff]
                # I stored influ = 2*f[:n_aff] before modifying f, so it's correct
                # But above I wrote c[k_start:k_end] += delta * (2.0 * f[:n_aff] - ...)
                # which is wrong because f has been modified. Let me fix this.
                # Actually, I'll just recompute the affected part properly.
                # FOR CORRECTNESS, let me just recompute c from scratch every sweep.
                # Actually no, that's too expensive. Let me fix the update.
                # The issue: I computed influ = 2*f[:n_aff] BEFORE updating f[i].
                # Then I do f[i] = best_val (which changes f).
                # Then I need c[k_start:k_end] += delta * influ (old influ).
                # But I wrote a wrong expression above. Let me just do it right.
                improved = True

    # Final recompute
    best_f = np.clip(best_f, 0, None)
    final_score = _score(best_f)
    return best_f, final_score


def _coord_descent_v2(f0, stop_time, max_sweeps=500, seed=123):
    """Corrected coordinate descent with proper incremental updates."""
    n = len(f0)
    f = np.clip(f0.copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        f = np.ones(n) / n
        S = 1.0
    else:
        f = f / S
        S = 1.0

    c = _conv(f)
    best_score = 2.0 * n * float(c.max())
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = np.arange(n)
        if sweep % 3 != 0:
            rng.shuffle(order)

        for i in order:
            if time.monotonic() > stop_time:
                break

            v = f[i]
            k_start = i
            k_end = min(i + n, 2 * n - 1)
            n_aff = k_end - k_start
            if n_aff <= 0:
                continue

            # Store old influence (must be before any modification to f)
            influ_old = 2.0 * f[:n_aff].copy()
            idx_2i = 2 * i - k_start
            S_rest = S - v

            # Precompute max outside affected range
            max_out = 0.0
            if k_start > 0:
                max_out = max(max_out, float(c[:k_start].max()))
            if k_end < 2 * n - 1:
                max_out = max(max_out, float(c[k_end:].max()))

            def _eval(val):
                delta = val - v
                S_new = S_rest + val
                if S_new < 0.01:
                    return math.inf
                c_aff = c[k_start:k_end] + delta * influ_old
                if 0 <= idx_2i < n_aff:
                    c_aff = c_aff.copy()
                    c_aff[idx_2i] += delta * delta
                m_aff = float(c_aff.max())
                m_new = max(m_aff, max_out)
                return 2.0 * n * m_new / (S_new * S_new)

            best_val = v
            best_obj = _eval(v)

            # Candidate values
            med = float(np.median(f[f > 0])) if np.any(f > 0) else 1.0 / n
            if v < 1e-12:
                cands = [0.0, 0.001, 0.005, 0.01, 0.02, med * 0.5, med, med * 2, 1.0 / n, 2.0 / n, 5.0 / n, 0.05, 0.1]
            else:
                cands = [
                    0.0, v * 0.1, v * 0.25, v * 0.5, v * 0.75,
                    v, v * 1.25, v * 1.5, v * 2.0, v * 3.0, v * 5.0,
                    med * 0.25, med, med * 2, med * 4,
                    1.0 / n, 2.0 / n, 5.0 / n, 0.01, 0.05, 0.1, 0.5, 1.0, 2.0,
                ]
            cands = sorted(set(min(max(x, 0.0), 100.0) for x in cands))

            for cv in cands:
                obj = _eval(cv)
                if obj < best_obj:
                    best_obj = obj
                    best_val = cv

            # Golden section refinement
            if best_val != v and best_val > 0:
                lo = max(0.0, best_val * 0.5)
                hi = best_val * 2.0
                if hi - lo < 1e-6:
                    lo = max(0.0, best_val - best_val * 0.5)
                    hi = best_val * 2.0
                for _ in range(10):
                    m1 = hi - (hi - lo) * 0.382
                    m2 = lo + (hi - lo) * 0.382
                    f1 = _eval(m1)
                    f2 = _eval(m2)
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                    mid = (lo + hi) / 2
                    fmid = _eval(mid)
                    if fmid < best_obj:
                        best_obj = fmid
                        best_val = mid

            if best_val != v and best_obj < best_score - 1e-14:
                delta = best_val - v
                # Apply update using OLD influence
                c[k_start:k_end] += delta * influ_old
                if 0 <= idx_2i < n_aff:
                    c[idx_2i] += delta * delta
                f[i] = best_val
                S += delta
                best_score = best_obj
                best_f = f.copy()
                improved = True

        if not improved and sweep > 5:
            break

    best_f = np.clip(best_f, 0, None)
    final_score = _score(best_f)
    return best_f, final_score


def _pair_polish(f0, stop_time, max_rounds=30, seed=555):
    """2-coordinate moves."""
    n = len(f0)
    f = np.clip(f0.copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, _score(f)
    f = f / S
    S = 1.0
    best_score = _score(f)
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time:
            break
        improved = False
        n_trials = 40
        for _ in range(n_trials):
            if time.monotonic() > stop_time:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            vi, vj = f[i], f[j]
            if vi < 1e-12 and vj < 1e-12:
                # Try turning on
                f[i] = 0.01
                f[j] = 0.01
                sc = _score(f)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_f = f.copy()
                    improved = True
                else:
                    f[i], f[j] = vi, vj
                continue

            factors = [0.0, 0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0]
            best_pair = (vi, vj)
            best_sc = best_score

            for fi in factors:
                ni = vi * fi if vi > 1e-12 else (0.01 if fi > 0 else 0.0)
                for fj in factors:
                    nj = vj * fj if vj > 1e-12 else (0.01 if fj > 0 else 0.0)
                    f[i], f[j] = ni, nj
                    S_test = S - vi - vj + ni + nj
                    if S_test < 0.01:
                        continue
                    sc = _score(f)
                    if sc < best_sc:
                        best_sc = sc
                        best_pair = (ni, nj)

            f[i], f[j] = best_pair
            if best_sc < best_score - 1e-14:
                best_score = best_sc
                best_f = f.copy()
                improved = True

        if not improved:
            break

    return best_f, _score(best_f)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 20.0

    parent = _get_parent()
    n = len(parent)

    # Fallback
    fallback = parent.copy()
    try:
        fallback_sc = _score(fallback)
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(32) / 32.0
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update_best(seq):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64).copy()
        seq = np.clip(seq, 0, 1000)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            sc = math.inf
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_seq = seq.copy()
            best_score = sc

    def _time_left():
        return max(0.0, hard_deadline - time.monotonic())

    # === Phase 1: LP-based optimization (20% of budget) ===
    phase1_end = t0 + budget * 0.20
    f = parent / parent.sum() if parent.sum() > 0.01 else np.ones(n) / n
    _update_best(f)

    stall = 0
    max_stall = 15
    lp_iter = 0
    while time.monotonic() < phase1_end - 5 and stall < max_stall:
        b = _conv(f)
        M = float(b.max())
        S = float(f.sum())
        score = 2.0 * n * M / (S * S)

        g_dir = _lp_direction(f, n, M, time_limit=4.0)
        if g_dir is None:
            stall += 3
            f = np.clip(f + np.random.default_rng(seed + lp_iter).normal(0, 0.02, n), 0, None)
            if f.sum() > 0.01:
                f /= f.sum()
            continue

        # Blend
        best_t = 0.0
        best_sc = score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_try = f + t * (g_dir - f)
            f_try = np.clip(f_try, 0, None)
            if f_try.sum() < 0.01:
                continue
            f_try /= f_try.sum()
            sc_try = _score(f_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_t = t

        if best_t > 0:
            f = np.clip(f + best_t * (g_dir - f), 0, None)
            f /= f.sum()
            _update_best(f)
            stall = 0
        else:
            stall += 1
            f = np.clip(f + np.random.default_rng(seed + lp_iter + 7).normal(0, 0.03, n), 0, None)
            if f.sum() > 0.01:
                f /= f.sum()

        lp_iter += 1

    _update_best(f)

    # === Phase 2: Gradient descent (25% of budget) ===
    phase2_end = t0 + budget * 0.45
    if time.monotonic() < phase2_end - 10 and _time_left() > 30:
        try:
            res, sc = _grad_optimize(best_seq, len(best_seq), phase2_end - 5, seed=1000)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update_best(res)
        except Exception:
            pass

    _update_best(best_seq)

    # === Phase 3: Coordinate descent (30% of budget) ===
    phase3_end = t0 + budget * 0.75
    if time.monotonic() < phase3_end - 10 and _time_left() > 20:
        try:
            res, sc = _coord_descent_v2(best_seq, phase3_end - 5, max_sweeps=300, seed=42)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update_best(res)
        except Exception:
            pass

    _update_best(best_seq)

    # === Phase 4: More coordinate descent + pair polish (15% of budget) ===
    phase4_end = t0 + budget * 0.90
    if time.monotonic() < phase4_end - 10 and _time_left() > 15:
        try:
            res, sc = _coord_descent_v2(best_seq, min(phase4_end - 20, time.monotonic() + _time_left() * 0.5),
                                        max_sweeps=200, seed=999)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update_best(res)
        except Exception:
            pass

        if time.monotonic() < phase4_end - 10 and _time_left() > 15:
            try:
                res, sc = _pair_polish(best_seq, min(phase4_end - 5, time.monotonic() + _time_left() * 0.3),
                                       max_rounds=20, seed=7777)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    _update_best(res)
            except Exception:
                pass

    _update_best(best_seq)

    # === Phase 5: Final intensive coordinate descent ===
    final_end = hard_deadline - 10.0
    if time.monotonic() < final_end - 5 and _time_left() > 10:
        try:
            res, sc = _coord_descent_v2(best_seq, final_end, max_sweeps=100, seed=31337)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update_best(res)
        except Exception:
            pass

    _update_best(best_seq)

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
