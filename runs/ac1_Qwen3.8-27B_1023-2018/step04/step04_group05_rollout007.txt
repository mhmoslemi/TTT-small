    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _gradient_score(f):
    """Compute score and subgradient for active max positions."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-15:
        return math.inf, np.zeros(n)
    b = _conv_full(f)
    M = float(b.max())
    score = 2.0 * n * M / (s * s)
    # Find active positions (within 1e-10 of max)
    active = np.where(b >= M - 1e-10 * max(1.0, M))[0]
    # Subgradient: average over active positions
    g = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        # dc_k/df_i = 2*f[k-i] if 0<=k-i<n else 0
        lo_i = max(0, k - (n - 1))
        hi_i = min(n - 1, k)
        if lo_i <= hi_i:
            idx = np.arange(lo_i, hi_i + 1)
            g[idx] += 2.0 * f[k - idx]
    g /= max(1, len(active))
    # d(score)/df_i = (2n/s^2)*g_i - (4n*M/s^3)*1
    coeff1 = 2.0 * n / (s * s)
    coeff2 = 4.0 * n * M / (s * s * s)
    grad = coeff1 * g - coeff2
    return score, grad


def _grad_optimize(f0, n, stop_time, seed=42):
    """Projected gradient descent with adaptive step size."""
    rng = np.random.default_rng(seed)
    f = np.asarray(f0, dtype=np.float64).ravel()[:n].copy()
    if f.size < n:
        pad = np.zeros(n - f.size)
        f = np.concatenate([f, pad])
    f = np.clip(f, 0, None)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n)
        return f, _score(f)
    f = f / s  # normalize sum to 1

    best_f = f.copy()
    best_score = _score(f)
    alpha = 0.01
    alpha_min = 1e-6
    stall = 0
    max_stall = 50

    for _ in range(50000):
        if time.monotonic() > stop_time - 3.0:
            break
        score, grad = _gradient_score(f)
        if not math.isfinite(score):
            break
        if score < best_score - 1e-12:
            best_score = score
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            # Perturb to escape
            f = best_f + rng.normal(0, 0.02 * max(1.0, best_f.mean()), n)
            f = np.clip(f, 0, None)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            stall = 0
            max_stall += 30
            alpha = 0.01
            continue

        # Gradient step with backtracking
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            stall += 5
            if stall > max_stall:
                f = best_f + rng.normal(0, 0.01, n)
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-15:
                    f = f / fs
                stall = 0
            continue

        # Skip the constant component (it doesn't help with scale-invariant problem)
        grad_centered = grad - grad.mean()
        gc_norm = np.linalg.norm(grad_centered)
        if gc_norm < 1e-12:
            stall += 5
            continue

        improved = False
        for _backtrack in range(25):
            f_new = f - alpha * grad_centered
            f_new = np.clip(f_new, 0, None)
            fs = f_new.sum()
            if fs < 1e-15:
                alpha *= 0.5
                continue
            f_new = f_new / fs
            sc_new = _score(f_new)
            if sc_new < score - 1e-12:
                f = f_new
                improved = True
                alpha = min(0.5, alpha * 1.15)
                break
            alpha *= 0.5
            if alpha < alpha_min:
                break

        if not improved:
            if stall % 10 == 0:
                # Reset to best and try different alpha
                f = best_f.copy()
                alpha = max(alpha_min, 0.005 * rng.uniform(0.5, 3.0))
            if stall > max_stall:
                f = best_f + rng.normal(0, 0.015 * best_f.std(), n)
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-15:
                    f = f / fs
                stall = 0
                max_stall += 40
                alpha = 0.005

    return best_f, best_score


def _lp_direction(f, n, M, timeout=8.0):
    """LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    if M <= 1e-18:
        return None
    # Build constraint matrix A of shape (2n-1, n): A[k,j] = f[k-j] if valid
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-14)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs",
                      options={"presolve": True, "time_limit": timeout,
                               "dual_feasibility_tolerance": 1e-12,
                               "primal_feasibility_tolerance": 1e-12})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g


def _lp_optimize(f0, n, stop_time, seed=42):
    """LP-based iterative optimization."""
    rng = np.random.default_rng(seed)
    f = np.asarray(f0, dtype=np.float64).ravel()[:n].copy()
    if f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = np.clip(f, 0, None)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n)
        return f, _score(f)
    f = f / s

    best_f = f.copy()
    best_score = _score(f)
    t = 0.3
    t_min = 0.02
    t_max = 0.9
    stall = 0
    max_stall = 200
    conv_thresh = 1e-12

    for _ in range(3000):
        if time.monotonic() > stop_time - 3.0:
            break
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, n, M, timeout=6.0)
        if g is None:
            stall += 5
            if stall > max_stall:
                f = best_f + rng.normal(0, 0.03 * max(1.0, best_f.mean()), n)
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-15:
                    f = f / fs
                stall = 0
                max_stall += 50
                t = 0.3
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            stall += 3
            continue
        g = g / g_sum  # normalize

        dist = np.linalg.norm(g - f) / max(1.0, np.linalg.norm(f))
        if dist < conv_thresh:
            stall += 3
            if stall > max_stall:
                f = best_f.copy()
                idx = rng.choice(n, size=min(n, 20), replace=False)
                for k in idx:
                    f[k] = max(0.0, f[k] * rng.uniform(0.2, 3.0))
                fs = f.sum()
                if fs > 1e-15:
                    f = f / fs
                stall = 0
                max_stall += 60
                t = 0.3
            continue

        stall = 0
        improved = False
        for tc in sorted({max(t_min, min(t_max, c)) for c in
                          [t, t * 0.5, t * 0.7, t * 1.3, 0.1, 0.3, 0.5, 0.7, 1.0]},
                         reverse=True):
            f_new = (1.0 - tc) * f + tc * g
            f_new = np.clip(f_new, 0, None)
            fs = f_new.sum()
            if fs < 1e-15:
                continue
            f_new = f_new / fs
            sc = _score(f_new)
            if sc < best_score - 1e-12:
                f = f_new
                best_f = f.copy()
                best_score = sc
                improved = True
                t = min(t_max, tc * 1.2)
                break

        if not improved:
            t = max(t_min, t * 0.5)
            stall += 1
            if stall >= max_stall:
                f = best_f + rng.normal(0, 0.05 * best_f.std(), n)
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-15:
                    f = f / fs
                stall = 0
                max_stall += 60
                t = 0.3

    return best_f, best_score


def _coord_polish(a, stop_time, seed=4242):
    """Coordinate descent polishing."""
    n = len(a)
    best = np.asarray(a, dtype=np.float64).ravel().copy()
    best = np.clip(best, 0, None)
    fs = best.sum()
    if fs < 1e-15:
        return best, _score(best)
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 15

    while time.monotonic() < stop_time - 3.0 and no_imp < max_no_imp:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = best[i]
            best_i_val = orig
            best_i_sc = best_sc
            # Try multipliers
            for mlt in [0.0, 0.1, 0.2, 0.35, 0.5, 0.65, 0.8, 1.2, 1.5, 2.0, 3.0, 5.0]:
                nv = orig * mlt if mlt > 0 else 0.0
                best[i] = nv
                if best.sum() < 0.01:
                    best[i] = best_i_val
                    continue
                sc = _score(best)
                if sc < best_i_sc - 1e-14:
                    best_i_val = nv
                    best_i_sc = sc
            # Refine
            if 0 < best_i_val < orig:
                for frac in (1.25, 0.8, 1.5, 0.6):
                    nv = best_i_val * frac
                    best[i] = nv
                    if best.sum() < 0.01:
                        best[i] = best_i_val
                        continue
                    sc = _score(best)
                    if sc < best_i_sc - 1e-14:
                        best_i_val = nv
                        best_i_sc = sc
            elif orig > 0:
                for frac in (1.05, 0.95, 1.1, 0.9, 1.2, 0.8):
                    nv = orig * frac
                    best[i] = nv
                    if best.sum() < 0.01:
                        best[i] = best_i_val
                        continue
                    sc = _score(best)
                    if sc < best_i_sc - 1e-14:
                        best_i_val = nv
                        best_i_sc = sc
            best[i] = best_i_val
            if best_i_sc < best_sc - 1e-14:
                best_sc = best_i_sc
                improved = True

        # Two-coordinate moves
        n_pairs = 200
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 3.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            if oi < 1e-14 and oj < 1e-14:
                continue
            for mi in (0.0, 0.5, 0.75, 1.25, 1.75):
                for mj in (0.0, 0.5, 0.75, 1.25, 1.75):
                    if mi == 0.0 and mj == 0.0:
                        continue
                    ni = oi * mi if (oi > 1e-14 and mi > 0) else 0.0
                    nj = oj * mj if (oj > 1e-14 and mj > 0) else 0.0
                    best[i], best[j] = ni, nj
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        # Random perturbations
        if not improved:
            for _ in range(50):
                if time.monotonic() > stop_time - 3.0:
                    break
                k = int(rng.integers(n))
                old = best[k]
                sigma = 0.02 * max(1.0 / n, best.std())
                best[k] = max(0.0, old + rng.normal(0, sigma))
                if best.sum() < 0.01:
                    best[k] = old
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                else:
                    best[k] = old

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)
    n = n_p  # Use parent length as primary

    # Fallback: uniform sequence (score = 2)
    uniform = np.ones(n)
    fallback = uniform.copy()
    fallback_sc = _score(fallback)

    # Track best
    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_update(seq, score):
        nonlocal best_seq, best_score
        if seq is None or len(seq) == 0:
            return
        if not np.all(np.isfinite(seq)):
            return
        seq = np.clip(seq, 0, 1000)
        if seq.sum() < 0.01:
            return
        if score < best_score - 1e-13:
            try:
                exact = float(evaluate_sequence([float(x) for x in seq]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-14:
                best_seq = seq.copy()
                best_score = exact

    # Evaluate parent
    if np.all(np.isfinite(parent)) and parent.sum() > 0.01:
        p_sc = _score(parent)
        if math.isfinite(p_sc):
            _try_update(parent, p_sc)

    rng_main = np.random.default_rng(seed)

    # Generate diverse initializations
    inits = []
    inits.append(parent.copy())
    inits.append(np.ones(n))  # uniform

    # Raised cosine
    idx = np.arange(n, dtype=np.float64)
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * idx / max(1, n - 1))))

    # Flat-top (zeros at edges)
    ft = np.ones(n)
    edge = max(1, n // 12)
    ft[:edge] = 0.0
    ft[-edge:] = 0.0
    inits.append(ft + 1e-6)

    # Gaussian bump
    center = (n - 1) / 2.0
    width = n / 4.0
    inits.append(np.exp(-0.5 * ((idx - center) / width) ** 2) + 0.01)

    # Two-lobe
    two = np.exp(-0.5 * ((idx - n * 0.25) / (n * 0.12)) ** 2) + \
          np.exp(-0.5 * ((idx - n * 0.75) / (n * 0.12)) ** 2)
    inits.append(two + 0.01)

    # Multi-frequency
    sup = np.ones(n) * 0.3
    for fr in (1, 2, 3, 5, 7):
        ph = rng_main.uniform(0, 2 * np.pi)
        sup += 0.2 * np.cos(2.0 * np.pi * fr * idx / n + ph)
    inits.append(np.abs(sup) + 0.01)

    # Parent + noise
    sig = 0.15 * max(1.0 / n, float(np.std(parent)))
    inits.append(np.abs(parent + rng_main.normal(0, sig, n)) + 1e-4)

    # |Normal|
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.01)

    # Boundary-weighted
    inits.append(1.0 + 0.5 * np.cos(np.pi * idx / max(1, n - 1)))

    # Tricube
    x = np.clip(np.abs(idx - (n - 1) / 2.0) / max(1, n / 3.5), 0, 1)
    inits.append((1 - x) ** 3 + 0.005)

    # Phase 1: Gradient optimization from diverse inits
    phase1_end = t0 + budget * 0.40
    n_inits = len(inits)
    time_per_init = max(8.0, (phase1_end - t0) / n_inits)

    for idx_i, init in enumerate(inits):
        if time.monotonic() > phase1_end - 5.0:
            break
        stop = min(phase1_end - 3.0, time.monotonic() + time_per_init)
        if time.monotonic() > stop - 2.0:
            continue
        try:
            res, sc = _grad_optimize(init, n, stop, seed + 1000 * (idx_i + 1))
            if res is not None and len(res) == n:
                _try_update(res, sc)
        except Exception:
            pass

    # Phase 2: LP-based optimization on best and a few inits
    phase2_end = t0 + budget * 0.65
    time_per_lp = max(15.0, (phase2_end - phase1_end) / 4)

    lp_inits = [best_seq.copy(), inits[0], inits[2], inits[4]]
    for idx_l, init in enumerate(lp_inits):
        if time.monotonic() > phase2_end - 10.0:
            break
        stop = min(phase2_end - 5.0, time.monotonic() + time_per_lp)
        if time.monotonic() > stop - 5.0:
            continue
        try:
            res, sc = _lp_optimize(init, n, stop, seed + 5000 * (idx_l + 1))
            if res is not None and len(res) == n:
                _try_update(res, sc)
        except Exception:
            pass

    # Phase 3: Alternative lengths
    phase3_end = t0 + budget * 0.78
    other_lengths = [384, 400, 448, 480, 512, 320, 256]
    time_per_len = max(8.0, (phase3_end - phase2_end) / max(1, len(other_lengths)))

    for L in other_lengths:
        if time.monotonic() > phase3_end - 8.0:
            break
        stop = min(phase3_end - 4.0, time.monotonic() + time_per_len)
        if time.monotonic() > stop - 3.0:
            continue
        # Interpolate best to new length
        init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
        try:
            res, sc = _grad_optimize(init_l, L, stop, seed + 20000 + L)
            if res is not None and len(res) == L:
                _try_update(res, sc)
        except Exception:
            pass

    # Phase 4: Combined gradient + LP on best
    phase4_end = t0 + budget * 0.88
    if time.monotonic() < phase4_end - 20.0:
        stop_p4 = min(phase4_end - 5.0, time.monotonic() + (phase4_end - time.monotonic() - 5.0))
        n_best = len(best_seq)
        try:
            res, sc = _grad_optimize(best_seq, n_best, stop_p4, seed + 999001)
            if res is not None:
                _try_update(res, sc)
        except Exception:
            pass

        if time.monotonic() < stop_p4 - 15.0:
            stop_p4b = min(phase4_end - 5.0, time.monotonic() + 60.0)
            try:
                res, sc = _lp_optimize(best_seq, len(best_seq), stop_p4b, seed + 999002)
                if res is not None:
                    _try_update(res, sc)
            except Exception:
                pass

    # Phase 5: Coordinate descent polish
    polish_end = deadline - 8.0
    if time.monotonic() < polish_end - 10.0:
        try:
            b_seq, b_sc = _coord_polish(best_seq, polish_end, seed=4242)
            if b_sc < best_score - 1e-14:
                _try_update(b_seq, b_sc)
        except Exception:
            pass

    # Phase 6: Second polish
    if time.monotonic() < deadline - 15.0:
        try:
            b_seq2, b_sc2 = _coord_polish(best_seq, deadline - 10.0, seed=88888)
            if b_sc2 < best_score - 1e-14:
                _try_update(b_seq2, b_sc2)
        except Exception:
            pass

    # Final validation
    final_list = [float(x) for x in best_seq]
    try:
        final_sc = float(evaluate_sequence(final_list))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Fallback to uniform
        return [1.0] * n_p

    if final_sc >= best_score:
        best_score = final_sc

    return final_list
