    # Phase 2: LP with perturbed inits from current best
    p2_end = t0 + budget * 0.56
    for attempt in range(4):
        if time.monotonic() > p2_end:
            break
        rng_p = np.random.default_rng(seed + 7000 * (attempt + 1))
        pert = 0.08 + 0.04 * attempt
        base = best_seq / max(best_seq.sum(), 1e-10)
        p2 = np.clip(base + rng_p.normal(0, pert, n), 0, None)
        p2 = p2 / max(p2.sum(), 1e-10)
        stop2 = min(p2_end, time.monotonic() + budget * 0.04)
        res2, sc2 = _lp_refine(p2, n, stop2, seed + 40000 * (attempt + 1))
        if math.isfinite(sc2):
            _update(res2 * parent_sum, sc2)

    # Phase 3: Subgradient refinement
    p3_end = t0 + budget * 0.67
    if time.monotonic() < p3_end:
        f_sg = best_seq / max(best_seq.sum(), 1e-10)
        res_sg, sc_sg = _subgrad_refine(f_sg, n, p3_end, seed + 90000)
        if math.isfinite(sc_sg):
            _update(res_sg * parent_sum, sc_sg)

    # Phase 4: Coordinate descent polish
    p4_end = t0 + budget * 0.80
    if time.monotonic() < p4_end:
        res_cd, sc_cd = _coord_descent(best_seq, p4_end, seed + 100000, max_epochs=120)
        if math.isfinite(sc_cd):
            _update(res_cd, sc_cd)

    # Phase 5: Try alternative lengths
    p5_end = deadline - 35
    if time.monotonic() < p5_end:
        alt_lengths = sorted(set([L for L in [n-6, n-4, n-2, n-1, n+1, n+2, n+4, n+6, n+10, n+20] if L >= 16]))
        time_per = min(28.0, (p5_end - time.monotonic()) / max(1, len(alt_lengths)))
        for L in alt_lengths:
            if time.monotonic() > p5_end - 5:
                break
            init_L = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n), parent)
            init_L = np.clip(init_L, 0, None)
            init_L = init_L / max(init_L.sum(), 1e-10)
            stop_L = min(p5_end - 5, time.monotonic() + time_per)
            try:
                rL, sL = _lp_refine(init_L, L, stop_L, seed + 200000 + L)
                if math.isfinite(sL):
                    _update(rL * parent_sum, sL)
            except Exception:
                pass

    # Phase 6: Final LP micro-refinement on best
    if time.monotonic() < deadline - 15:
        f_fin = best_seq / max(best_seq.sum(), 1e-10)
        n_fin = len(f_fin)
        stop_fin = deadline - 15
        res_fin, sc_fin = _lp_refine(f_fin, n_fin, stop_fin, seed + 300000)
        if math.isfinite(sc_fin):
            _update(res_fin * parent_sum, sc_fin)

    # Phase 7: Final coordinate descent
    if time.monotonic() < deadline - 8:
        res_final, sc_final = _coord_descent(best_seq, deadline - 8, seed + 400000, max_epochs=60)
        if math.isfinite(sc_final):
            _update(res_final, sc_final)

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
