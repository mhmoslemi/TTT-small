import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel().copy()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s < 1e-12:
        p = np.ones(384, dtype=np.float64)
        s = 384.0
    p = p / s
    return p


def _conv_fft(a, size=None):
    n = a.size
    L = 2 * n - 1
    if size is None:
        size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_unnorm(p):
    """Score for unit-sum p: 2n * max(p*p)."""
    n = p.size
    if n == 0:
        return math.inf
    c = _conv_fft(p)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m


def _score_exact_list(a):
    """Exact score matching the evaluator."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _subgrad_fun(u, n, fft_size, smooth_T=None):
    """
    Objective and subgradient for L-BFGS on log-parameterization.
    p = exp(u - max(u)) / sum(exp(u - max(u)))  (softmax for stability)
    Returns (f, grad_wrt_u).
    """
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s < 1e-300 or not math.isfinite(s):
        return 1e100, np.zeros(n)
    p = e / s
    c = _conv_fft(p, fft_size)
    cm = float(c.max())
    if not math.isfinite(cm):
        return 1e100, np_zeros
    if smooth_T is not None and smooth_T > 1e-15:
        # Smooth max via log-sum-exp
        scaled = (c - cm) / smooth_T
        exps = np.exp(scaled)
        se = float(exps.sum())
        if not math.isfinite(se) or se <= 0:
            f = cm
            i_star = int(np.argmax(c))
        else:
            f = cm + smooth_T * math.log(se / c.size)
            i_star = None  # not used in smooth case
    else:
        f = cm
        i_star = int(np.argmax(c))

    score = 2.0 * n * f

    if i_star is not None:
        # Subgradient: d(score)/d(u_l) = 4n * p[l] * (p[i*-l] - C[i*])
        # where p[i*-l] = 0 if not (0 <= i*-l < n)
        diff = i_star - np.arange(n)
        # Build vector p[i* - l] for valid indices
        pcorr = np.zeros(n, dtype=np.float64)
        valid = (diff >= 0) & (diff < n)
        pcorr[valid] = p[diff[valid]]
        grad = 4.0 * n * p * (pcorr - cm)
    else:
        # Smooth case: gradient of log-sum-exp
        w = exps / se
        # d(max_smooth)/dp[k] = sum_j w[j] * dC[j]/dp[k] = sum_j w[j] * 2*p[j-k]
        # = 2 * (w * p)[k] in the convolution sense
        # dC[j]/dp[k] = 2*p[j-k] (if 0<=j-k<n)
        # So d(f)/dp[k] = sum_j w[j]*2*p[j-k] = 2*(w reversed * p)[k]... 
        # Actually: sum_j w[j]*p[j-k] = (w * p)[k] in cross-correlation
        # Let's compute: for each k, sum_{j: 0<=j-k<n} w[j]*p[j-k]
        # This is the cross-correlation of w and p at lag k
        Lx = 2 * n - 1
        sx = next_fast_len(Lx)
        Fw = rfft(w, sx)
        Fp = rfft(p[::-1], sx)
        cc = irfft(Fw * Fp, sx)[:Lx]
        dcp = 2.0 * cc[n - 1:2 * n - 1]  # length n
        # d(f)/du_l = d(f)/dp_l * p_l * (1 - p_l) ... no
        # d(p_l)/du_l = p_l*(1-p_l), d(p_k)/du_l = -p_k*p_l for k!=l
        # d(f)/du_l = sum_k d(f)/dp_k * dp_k/du_l = d(f)/dp_l * p_l - p_l * sum_k d(f)/dp_k * p_k
        dfdp = dcp  # already includes the factor of 2
        t = float(np.dot(dfdp, p))
        grad = score * (dfdp - t) * p / (4.0 * n) * (4.0 * n)
        # Simplify: grad = 2n * f factor... let me just compute directly
        # score = 2n*f, d(score)/du_l = 2n * d(f)/du_l
        # d(f)/du_l = d(f)/dp_l * p_l - p_l * sum_k d(f)/dp_k * p_k
        grad = 2.0 * n * (dfdp - t) * p

    grad = np.where(np.isfinite(grad), grad, 0.0)
    return score, grad


def _lbfgs_optimize(p, stop_time, max_iter_total=500, seed=42):
    """L-BFGS on log-parameterization with subgradient."""
    n = p.size
    p = np.maximum(p, 1e-300).copy()
    s = p.sum()
    if s < 1e-300:
        return p
    p = p / s

    u = np.log(p) - np.log(p).mean()
    fft_size = next_fast_len(2 * n - 1)

    f0, _ = _subgrad_fun(u, n, fft_size)
    best_u = u.copy()
    best_f = f0
    best_p = p.copy()

    time_left_check = stop_time - 5.0
    stage_frac = 1.0
    # Run multiple stages with decreasing smooth_T
    stages = [(None, 1.0), (1e-4, 0.6), (1e-6, 0.3)]
    cum = 0.0
    avail = stop_time - time.monotonic() - 10.0
    if avail < 20:
        return best_p

    for si, (T, frac) in enumerate(stages):
        now = time.monotonic()
        if now > time_left_check:
            break
        end_stage = min(time_left_check, now + frac * avail)
        if end_stage - now < 15:
            continue

        maxit = 80 if si == 0 else 60
        opts = {
            "maxiter": maxit,
            "gtol": 1e-10,
            "ftol": 1e-15,
            "maxls": 30,
            "maxcor": 40,
            "maxdisp": 0,
            "disp": False,
        }
        try:
            res = minimize(
                _subgrad_fun, best_u,
                args=(n, fft_size, T),
                method="L-BFGS-B",
                jac=True,
                options=opts,
            )
        except Exception:
            break

        if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
            try:
                fv = float(res.fun)
            except Exception:
                fv = math.inf
            if math.isfinite(fv) and fv < best_f:
                best_f = fv
                best_u = res.x.copy()
                pp = _softmax(best_u)
                best_p = pp

        u = best_u.copy()

    p_out = _softmax(best_u)
    sc = _score_unnorm(p_out)
    if not math.isfinite(sc):
        return best_p
    if sc < _score_unnorm(best_p):
        best_p = p_out
    return best_p


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _coord_descend(p, stop_time, G=None, noise_level=0.03, max_flat=5, seed=123):
    """Coordinate descent on p (unit sum)."""
    n = p.size
    p = np.maximum(p, 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p.copy()
    p = p / s

    if G is None:
        if n <= 512:
            G = 48
        elif n <= 1024:
            G = 36
        elif n <= 2048:
            G = 28
        else:
            G = 24
    G = int(G)

    Ltot = 2 * n - 1
    c = _conv_fft(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < 500000 and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for ii in range(n):
            i = int(order[ii])
            if time.monotonic() > stop_time - 1.0:
                break
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            # Influence vector: d[j-start] = 2*p[j-i] for j in [start, end)
            idx = np.arange(start, end)
            d = 2.0 * p[idx - i]  # length Ls
            base = c[start:end].copy()
            diag_off = 2 * i - start  # index within [start,end) where j=2i, or -1

            # Unaffected max
            if start > 0:
                max_unaff = float(c[:start].max())
            else:
                max_unaff = 0.0
            if end < Ltot:
                mu2 = float(c[end:].max())
                if mu2 > max_unaff:
                    max_unaff = mu2

            # Grid of w values
            x_hi = max(15.0 * v, 40.0 / float(n), 1e-12)
            x_hi = min(x_hi, 8.0 / float(n))
            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if v > 0 and v < x_hi:
                pts = np.unique(np.concatenate([pts, [v]]))
            w = pts - v  # changes

            # Evaluate score at each w
            # For each w_k: new_c[j] = base[j] + w_k * d[j] for j in affected
            # Plus quadratic term at diag if valid
            aff_max = base[None, :] + w[:, None] * d[None, :]  # shape (len(pts), Ls)
            if 0 <= diag_off < Ls:
                aff_max[:, diag_off] += w * w
            aff_max_row = aff_max.max(axis=1)  # shape (len(pts),)
            total_max = np.maximum(aff_max_row, max_unaff)

            denom = (1.0 + w)  # since sum=1, new sum = 1+w
            valid = denom > 1e-15
            scores = np.full(pts.size, math.inf)
            scores[valid] = 2.0 * n * total_max[valid] / (denom[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_w = float(w[imin])
            best_s = float(scores[imin])

            # Refine around best
            if G > 4:
                dw = x_hi / float(G)
                lo = max(-v + 1e-15, best_w - 2.0 * dw)
                hi = min(x_hi, best_w + 2.0 * dw)
                if hi > lo:
                    fine_w = np.linspace(lo, hi, 24, dtype=np.float64)
                    if 0.0 <= v - fine_w.min() and v - fine_w.max() < hi:
                        pass
                    aff2 = base[None, :] + fine_w[:, None] * d[None, :]
                    if 0 <= diag_off < Ls:
                        aff2[:, diag_off] += fine_w * fine_w
                    aff2_row = aff2.max(axis=1)
                    total2 = np.maximum(aff2_row, max_unaff)
                    den2 = 1.0 + fine_w
                    sc2 = np.full(fine_w.size, math.inf)
                    v2 = den2 > 1e-15
                    sc2[v2] = 2.0 * n * total2[v2] / (den2[v2] ** 2)
                    if np.any(np.isfinite(sc2)):
                        im2 = int(np.argmin(sc2))
                        if sc2[im2] < best_s:
                            best_w = float(fine_w[im2])
                            best_s = float(sc2[im2])

            if math.isfinite(best_s) and best_s < cur_score - 1e-14:
                p[i] = v + best_w
                # Update c
                c[start:end] = base + best_w * d
                if 0 <= diag_off < Ls:
                    c[2 * i] += best_w * best_w
                cur_score = best_s
                improved = True

        # After sweep, recompute c to avoid drift
        c = _conv_fft(p)
        ns = float(2.0 * n * c.max())
        if math.isfinite(ns) and ns < best_score - 1e-14:
            best_score = ns
            best_p = p.copy()

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 15.0:
                    # Perturb and restart
                    p = p * np.exp(rng.normal(0.0, noise_level, n))
                    p = np.maximum(p, 0.0)
                    ss = p.sum()
                    if ss > 1e-300:
                        p /= ss
                    c = _conv_fft(p)
                    cur_score = float(2.0 * n * c.max())
                    noise_level *= 0.8
                    flat = 0
                else:
                    break
        sweep += 1

    ss = p.sum()
    if ss > 1e-300:
        p = p / ss
    sc = _score_unnorm(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p


def _exp_family(n, c_val):
    """Exponential decay initialization: a[i] = exp(-c*i/n), normalized to sum=1."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-c_val * x / float(n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return a / s


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m) + 0.5) / float(m)
        fp = np.interp((np.arange(n) + 0.5) / float(n), xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_stop = t0 + budget - 20.0  # 20s reserve for final validation

    parent = _get_parent()
    parent_n = parent.size

    # Track best
    best_arr = parent.copy()
    best_score = _score_exact_list(best_arr)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            sc = _score_exact_list(a)
            if math.isfinite(sc) and sc < best_score - 1e-14:
                a = a / s
                best_arr = a.copy()
                best_score = sc
        except Exception:
            pass

    # Define attempts: (length, init_array)
    attempts = []

    # Parent
    attempts.append((parent_n, parent.copy()))

    # Exponential family at parent length
    for c_val in [0.8, 1.2, 1.6, 2.2]:
        attempts.append((parent_n, _exp_family(parent_n, c_val)))

    # Other lengths
    for n2 in [1024, 2048]:
        if n2 != parent_n:
            attempts.append((n2, _exp_family(n2, 1.2)))
            attempts.append((n2, _resample(parent, n2)))

    # Sort by initial score (best first)
    attempts.sort(key=lambda x: _score_exact_list(x[1]))
    # Keep top attempts
    max_attempts = 6
    attempts = attempts[:max_attempts]

    # Time allocation
    now = time.monotonic()
    total_avail = max(5.0, hard_stop - now - 30.0)
    k = len(attempts)

    # Weights: give more time to top candidates
    weights = np.ones(k)
    if k > 1:
        for i in range(1, k):
            weights[i] = 0.6 ** i
    weights /= weights.sum()

    remaining = total_avail
    for idx, (n, init) in enumerate(attempts):
        now = time.monotonic()
        if now > hard_stop - 25.0:
            break
        # Time for this attempt
        if idx < k - 1:
            t_alloc = weights[idx] * remaining
        else:
            t_alloc = remaining

        # Minimum time to bother
        if t_alloc < 30.0:
            break

        remaining -= t_alloc
        end_phase = min(hard_stop - 25.0, now + t_alloc)

        p = init.copy()
        consider(p)

        # Phase 1: L-BFGS (60% of time)
        lbfgs_end = min(end_phase, now + 0.55 * t_alloc)
        if lbfgs_end - now > 15.0:
            try:
                p2 = _lbfgs_optimize(p, lbfgs_end, seed=int(seed + idx * 100))
                if p2 is not None and p2.size == n and np.all(np.isfinite(p2)):
                    p2 = p2 / p2.sum()
                    consider(p2)
            except Exception:
                pass
            if time.monotonic() > end_phase - 5.0:
                continue
            # Reload best if L-BFGS regressed
            if best_arr.size == n:
                if _score_exact_list(best_arr) < _score_exact_list(p) - 0.001:
                    p = best_arr.copy()

        # Phase 2: Coordinate descent (remaining time)
        coord_end = end_phase
        if coord_end - time.monotonic() > 15.0:
            try:
                p3 = _coord_descend(p, coord_end, seed=int(seed + 999 + idx * 7))
                if p3 is not None and p3.size == n and np.all(np.isfinite(p3)):
                    p3 = p3 / p3.sum()
                    consider(p3)
            except Exception:
                pass

    # Final polish on global best
    now = time.monotonic()
    final_end = hard_stop - 10.0
    if final_end - now > 60.0 and best_arr.size > 0:
        nb = best_arr.size
        pb = best_arr.copy()
        consider(pb)

        # Quick L-BFGS
        if final_end - now > 120.0 and nb <= 4000:
            lstop = min(final_end - 60.0, now + 0.5 * (final_end - now))
            try:
                p2 = _lbfgs_optimize(pb, lstop, seed=int(seed + 77777))
                if p2 is not None and p2.size == nb and np.all(np.isfinite(p2)):
                    p2 = p2 / p2.sum()
                    consider(p2)
            except Exception:
                pass

        # Final coordinate descent
        if time.monotonic() < final_end - 15.0:
            try:
                p3 = _coord_descend(best_arr, final_end, G=20,
                                    noise_level=0.02, seed=int(seed + 55555))
                if p3 is not None and p3.size == nb and np.all(np.isfinite(p3)):
                    p3 = p3 / p3.sum()
                    consider(p3)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                ls = [1.0]
        return ls
    except Exception:
        return [1.0]
