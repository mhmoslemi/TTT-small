                if fs > 1e-10:
                    f = f / fs * target_sum
                stall = 0
                max_stall += 50
            continue

        g_sum = float(g.sum())
        g = g * (target_sum / g_sum)

        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < 1e-10:
            stall += 2
            if stall > max_stall:
                f = best_f.copy()
                idx = rng.choice(n, size=max(5, n//10), replace=False)
                for k in idx:
                    f[k] = max(0.0, f[k] * rng.uniform(0.3, 2.0))
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                stall = 0
                max_stall += 50
            continue

        # Line search
        improved = False
        for tc in [t, t*0.7, t*0.5, t*0.3, 0.1, 0.05]:
            tc = max(t_min, min(t_max, tc))
            f_new = np.clip((1.0 - tc) * f + tc * g, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 0.01:
                continue
            f_new = f_new / fs_new * target_sum
            sc = _score(f_new)
            if sc < best_score - 1e-14:
                f = f_new.copy()
                best_f = f.copy()
                best_score = sc
                improved = True
                t = min(t_max, tc * 1.2)
                stall = 0
                break

        if not improved:
            stall += 1
            t = max(t_min, t * 0.7)
            if stall >= max_stall:
                f = best_f.copy()
                f += rng.normal(0, 0.06 * max(1.0, best_f.mean()), n)
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                stall = 0
                max_stall += 50
                t = 0.25

    return best_f, best_score

def _gradient_step(f, n, target_sum, lr):
    """One projected gradient step on max(a*a)."""
    b = _conv_full(f)
    M = float(b.max())
    # Active set
    active = np.where(b > 0.9 * M)[0]
    
    # Gradient of sum_{k in active} (f*f)[k]
    grad = np.zeros(n)
    for k in active:
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                grad[j] += f[kj]
                if kj != j:
                    grad[j] += f[kj]  # Wait, this is wrong
    
    # Actually let me compute gradient properly using convolution
    # d/d a_j [(a*a)[k]] = a[k-j] if 0<=k-j<n, plus a[k-j] if k-j != j (from the other factor)
    # Actually: (a*a)[k] = sum_i a[i]*a[k-i]
    # d/d a_j: a[k-j] (from a_i term) + a[k-j] (from a[k-i] term where i=k-j)
    # = 2*a[k-j] if 0 <= k-j < n, else 0
    # Wait no. If j != k-j, then:
    #   The term a[j]*a[k-j] contributes a[k-j] to d/d a_j
    #   The term a[k-j]*a[j] contributes a[j] to d/d a_j (when i = k-j)
    #   Total: a[k-j] + a[j]... but wait, a[j] and a[k-j] are different values.
    #   Actually no: the sum is over i from 0 to n-1.
    #   Term i=j: a[j]*a[k-j], derivative wrt a[j] is a[k-j]
    #   Term i=k-j: a[k-j]*a[j], derivative wrt a[j] is a[k-j]... no, derivative of a[k-j]*a[j] wrt a[j] is a[k-j].
    #   So total is 2*a[k-j] if k-j is valid and j != k-j.
    #   If k-j = j (k=2j): term is a[j]^2, derivative is 2*a[j].
    #   So in all cases: grad_j = 2*a[k-j] if 0<=k-j<n, else 0.
    #   Wait that's only if we're summing over all valid i. Let me verify.
    #   (a*a)[k] = sum_{i=0}^{n-1} a[i] * a[k-i] (with a[outside] = 0)
    #   d/d a_j = a[k-j] (from i=j) + [if 0<=k-j<n] a[k-j] (from i=k-j, the a[j] factor)
    #   Hmm wait: from i=k-j: the term is a[k-j]*a[j], and d/d a_j of this is a[k-j].
    #   From i=j: the term is a[j]*a[k-j], and d/d a_j of this is a[k-j].
    #   So total: 2*a[k-j] if 0<=k-j<n AND i=k-j is valid (0<=k-j<n) AND i=j is valid (always).
    #   But we also need i=k-j to be != i=j, i.e., k-j != j, i.e., k != 2j.
    #   If k = 2j: only one term a[j]^2, derivative 2*a[j]. And 2*a[k-j] = 2*a[j]. Same!
    #   So: grad_j[(a*a)[k]] = 2*a[k-j] if 0 <= k-j < n, else 0.
    
    # Let me redo:
    grad = np.zeros(n)
    for k in active:
        # For each k, grad_j += 2*a[k-j] if 0<=k-j<n
        j_lo = max(0, k - (n-1))
        j_hi = min(n-1, k)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += 2 * f[k - j_hi : k - j_lo + 1][::-1]
    
    # Normalize gradient
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return f, M
    
    # Projected gradient step
    f_new = f - lr * grad / gnorm
    f_new = np.clip(f_new, 0, None)
    fs = f_new.sum()
    if fs < 0.01:
        return f, M
    f_new = f_new / fs * target_sum
    return f_new, M

def _polish(a, stop_time, seed=42):
    """Coordinate descent polishing."""
    n = len(a)
    target_sum = math.sqrt(2.0 * n)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, _score(best)
    best = best / fs * target_sum
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 30

    while time.monotonic() < stop_time - 5.0 and no_imp < max_no_imp:
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break
            orig = best[i]
            best_i = orig
            best_i_sc = best_sc
            
            # Golden-section-like search
            lo, hi = 0.0, max(10 * orig if orig > 0 else 1.0, best.max() * 2)
            # Try logarithmic grid
            if orig > 1e-10:
                grid = np.geomspace(max(1e-10, orig * 0.01), orig * 10, 20)
            else:
                grid = np.linspace(0, best.max() * 0.5, 15)
            
            for v in grid:
                best[i] = v
                s = best.sum()
                if s < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_i_sc - 1e-15:
                    best_i = v
                    best_i_sc = sc
            
            # Refine
            if best_i > 1e-10:
                for frac in (0.9, 0.95, 1.05, 1.1, 0.8, 1.2):
                    v = best_i * frac
                    best[i] = v
                    s = best.sum()
                    if s < 0.01:
                        best[i] = best_i
                        continue
                    sc = _score(best)
                    if sc < best_i_sc - 1e-15:
                        best_i = v
                        best_i_sc = sc
            
            best[i] = best_i
            if best_i_sc < best_sc - 1e-15:
                best_sc = best_i_sc
                improved = True
        
        # Random pair perturbations
        n_pairs = 200
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 5.0:
                break
            i, j = rng.integers(n, 2)
            if i == j:
                continue
            oi, oj = best[i], best[j]
            for mi in (0.0, 0.5, 1.0, 1.5, 2.0):
                for mj in (0.0, 0.5, 1.0, 1.5, 2.0):
                    if mi == 1.0 and mj == 1.0:
                        continue
                    best[i] = oi * mi if (oi > 1e-14 and mi > 0) else 0.0
                    best[j] = oj * mj if (oj > 1e-14 and mj > 0) else 0.0
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj
        
        # Small random perturbations
        if not improved:
            for _ in range(100):
                if time.monotonic() > stop_time - 5.0:
                    break
                k = int(rng.integers(n))
                old = best[k]
                sigma = 0.02 * max(1.0, best.mean())
                best[k] = max(0.0, old + rng.normal(0, sigma))
                if best.sum() < 0.01:
                    best[k] = old
                    continue
                sc = _score(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                else:
                    best[k] = old
        
        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc
