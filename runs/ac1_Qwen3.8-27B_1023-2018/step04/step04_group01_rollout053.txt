                if sc < best_sc:
                    best_sc = sc
                    best_t = tv

            # Refine
            if best_t > 0:
                lo = max(0.0, best_t - 0.1)
                hi = min(1.0, best_t + 0.1)
                for _ in range(8):
                    mid = 0.5 * (lo + hi)
                    fn = f_curr + mid * (g - f_curr)
                    np.maximum(fn, 0, out=fn)
                    ss = fn.sum()
                    if ss < 1e-10:
                        hi = mid
                        continue
                    fn /= ss
                    sc = my_score(fn)
                    if sc < best_sc:
                        best_sc = sc
                        best_t = mid
                        lo = mid
                    else:
                        hi = mid

            if best_t > 1e-4 and best_sc < curr_sc - 1e-13:
                f_curr = f_curr + best_t * (g - f_curr)
                np.maximum(f_curr, 0, out=f_curr)
                ss = f_curr.sum()
                f_curr = f_curr / ss if ss > 1e-10 else cage.copy()
                if best_sc < cage_sc:
                    cage_sc = best_sc
                    cage = f_curr.copy()
                stall = 0
            else:
                stall += 1
                if stall > max_stall:
                    f_curr = np.clip(f_curr + rng.normal(0, 0.06, n), 0, None)
                    ss = f_curr.sum()
                    f_curr = f_curr / ss if ss > 1e-10 else cage.copy()
                    stall = 0
                    if time.monotonic() > stop_time - 20:
                        break

        update_best(cage)
        return cage, cage_sc

    # Gradient-based with softmax smoothing
    def grad_optimize(f_init, stop_time, max_iters=5000, seed_val=137):
        f_curr = np.clip(f_init, 0, None).copy()
        f_curr /= f_curr.sum()
        cage = f_curr.copy()
        cage_sc = my_score(cage)
        rng = np.random.default_rng(seed_val)
        alpha = 0.08
        tau = 60.0
        cs = 2 * n - 1
        fft_len = next_fast_len(cs)

        for it in range(max_iters):
            if time.monotonic() > stop_time - 5:
                break

            b = np.convolve(f_curr, f_curr, 'full')
            M = float(b.max())
            sc = 2.0 * n * M  # sum=1

            if sc < cage_sc - 1e-12:
                cage_sc = sc
                cage = f_curr.copy()

            # Softmax weights
            bs = b - M
            tb = tau * bs
            tb -= tb.max()
            p = np.exp(tb)
            ps = p.sum()
            if ps < 1e-30:
                p = np.zeros(cs)
                p[np.argmax(b)] = 1.0
            else:
                p /= ps

            # Gradient via FFT: grad[i] = 2 * sum_k p[k] * f[k-i] (circular, cropped)
            f_rev = f_curr[::-1]
            P = rfft(p, fft_len)
            Fr = rfft(f_rev, fft_len)
            conv_grad = irfft(P * Fr, fft_len)[:cs]
            grad = 2.0 * conv_grad[n - 1:2 * n - 1]

            # Projected gradient step on simplex
            f_new = f_curr - alpha * grad
            np.maximum(f_new, 0, out=f_new)
            ss = f_new.sum()
            if ss < 1e-10:
                f_new = f_curr.copy()
            else:
                f_new /= ss

            sc_new = my_score(f_new)
            if sc_new < sc - 1e-14:
                f_curr = f_new
                alpha = min(alpha * 1.01, 0.4)
            else:
                alpha = max(alpha * 0.6, 1e-9)
                if alpha < 1e-7:
                    f_curr = np.clip(f_curr + rng.normal(0, 0.008, n), 0, None)
                    ss = f_curr.sum()
                    f_curr /= ss if ss > 1e-10 else None
                    alpha = 0.04

            if it % 800 == 700:
                tau = max(15.0, tau * 0.75)
            if it % 2000 == 1999:
                f_curr = np.clip(f_curr + rng.normal(0, 0.015, n), 0, None)
                f_curr /= f_curr.sum()
                alpha = 0.06

        update_best(cage)
        return cage, cage_sc

    # Coordinate descent with incremental convolution
    def coord_descent(f_init, stop_time, max_sweeps=120, seed_val=777):
        f_curr = np.clip(f_init, 0, None).copy()
        f_curr /= f_curr.sum()
        conv = np.convolve(f_curr, f_curr, 'full').astype(np.float64)
        cage = f_curr.copy()
        cs = 2 * n - 1
        cage_sc = 2.0 * n * conv.max()  # sum=1
        rng = np.random.default_rng(seed_val)

        for sweep in range(max_sweeps):
            if time.monotonic() > stop_time - 3:
                break
            improved = False
            order = rng.permutation(n)

            for i in order:
                if time.monotonic() > stop_time - 2:
                    break

                v = f_curr[i]

                # Grid for x
                if v < 1e-7:
                    x_grid = np.array([0.0, 0.005, 0.01, 0.02, 0.05])
                else:
                    x_lo = max(0.0, v * 0.4)
                    x_hi = min(3.0 * v + 0.03, 0.25)
                    if x_hi <= x_lo:
                        x_hi = x_lo + 0.02
                    x_grid = np.linspace(x_lo, x_hi, 25)

                best_x = v
                best_obj = math.inf
                k_lo = i
                k_hi = min(i + n, cs)
                seg_len = k_hi - k_lo

                for x in x_grid:
                    w = x - v
                    if abs(w) < 1e-14:
                        m_new = conv.max()
                    else:
                        conv[k_lo:k_hi] += 2.0 * w * f_curr[0:seg_len]
                        conv[2 * i] += w * w
                        m_new = conv.max()
                        conv[k_lo:k_hi] -= 2.0 * w * f_curr[0:seg_len]
                        conv[2 * i] -= w * w
                    obj = 2.0 * n * m_new
                    if obj < best_obj:
                        best_obj = obj
                        best_x = x

                if best_x != v and best_obj < cage_sc - 1e-13:
                    w = best_x - v
                    k_lo = i
                    k_hi = min(i + n, cs)
                    conv[k_lo:k_hi] += 2.0 * w * f_curr[0:k_hi - k_lo]
                    conv[2 * i] += w * w
                    f_curr[i] = best_x
                    cage_sc = best_obj
                    cage = f_curr.copy()
                    improved = True

            if not improved and sweep > 3:
                # Perturbation to escape
                idx = rng.integers(n)
                f_curr[idx] = max(0.0, f_curr[idx] * (0.3 + 1.5 * rng.random()))
                f_curr = np.clip(f_curr, 0, None)
                ss = f_curr.sum()
                if ss < 1e-10:
                    f_curr = cage.copy()
                else:
                    f_curr /= ss
                    conv = np.convolve(f_curr, f_curr, 'full')
                    cage_sc = 2.0 * n * conv.max()

        update_best(cage)
        return cage, cage_sc

    # Pair coordinate moves for extra refinement
    def pair_polish(f_init, stop_time, max_rounds=30, seed_val=999):
        f_curr = np.clip(f_init, 0, None).copy()
        f_curr /= f_curr.sum()
        cage = f_curr.copy()
        cage_sc = my_score(cage)
        rng = np.random.default_rng(seed_val)
        cs = 2 * n - 1
        conv = np.convolve(f_curr, f_curr, 'full').astype(np.float64)

        for rnd in range(max_rounds):
            if time.monotonic() > stop_time - 3:
                break
            improved = False
            for _ in range(40):
                if time.monotonic() > stop_time - 2:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                oi, oj = f_curr[i], f_curr[j]
                if oi < 1e-10 and oj < 1e-10:
                    continue

                best_pair = (oi, oj)
                best_sc = cage_sc

                for mi in (0.0, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
                    ni = oi * mi if oi > 1e-10 else (0.01 if mi > 0 else 0.0)
                    for mj in (0.0, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
                        nj = oj * mj if oj > 1e-10 else (0.01 if mj > 0 else 0.0)
                        f_curr[i], f_curr[j] = ni, nj
                        ss = f_curr.sum()
                        if ss < 0.01:
                            f_curr[i], f_curr[j] = oi, oj
                            continue
                        sc = 2.0 * n * float(np.convolve(f_curr, f_curr, 'full').max()) / (ss * ss)
                        if sc < best_sc - 1e-13:
                            best_sc = sc
                            best_pair = (ni, nj)

                f_curr[i], f_curr[j] = best_pair
                if best_sc < cage_sc - 1e-13:
                    cage_sc = best_sc
                    cage = f_curr.copy()
                    improved = True

            if not improved:
                break

        update_best(cage)
        return cage, cage_sc

    # === MAIN OPTIMIZATION ===

    # Phase 1: LP optimization from parent (largest chunk)
    p1_end = t0 + budget * 0.35
    if time.monotonic() < p1_end - 10:
        try:
            best_f, best_score = lp_optimize(f, p1_end, max_iters=300, seed_val=42)
        except Exception:
            pass

    # Phase 2: Gradient refinement
    p2_end = t0 + budget * 0.50
    if time.monotonic() < p2_end - 5:
        try:
            best_f, best_score = grad_optimize(best_f, p2_end, max_iters=4000, seed_val=200)
        except Exception:
            pass

    # Phase 3: More LP refinement
    p3_end = t0 + budget * 0.65
    if time.monotonic() < p3_end - 10:
        try:
            best_f, best_score = lp_optimize(best_f, p3_end, max_iters=200, seed_val=4242)
        except Exception:
            pass

    # Phase 4: Coordinate descent
    p4_end = min(t0 + budget * 0.80, hard_deadline - 80)
    if time.monotonic() < p4_end - 5:
        try:
            best_f, best_score = coord_descent(best_f, p4_end, max_sweeps=80, seed_val=3333)
        except Exception:
            pass

    # Phase 5: Try alternative initializations
    if time.monotonic() < hard_deadline - 120 and best_score > 1.5028:
        i_arr = np.arange(n, dtype=np.float64)
        alts = [
            np.clip(1.0 + 0.6 * np.cos(2 * np.pi * i_arr / n), 0.01, None),
            np.clip(1.0 + 0.4 * np.sin(4 * np.pi * i_arr / n), 0.01, None),
            np.clip(1.0 + 0.8 * np.cos(np.pi * i_arr / n) ** 2, 0.01, None),
            np.clip(0.5 + 0.5 * np.cos(3 * np.pi * i_arr / n), 0.01, None),
        ]
        alts = [a / a.sum() for a in alts if a.sum() > 0.01]
        for idx, alt in enumerate(alts[:3]):
            tr = hard_deadline - 60 - time.monotonic()
            if tr < 25:
                break
            if best_score < 1.5028:
                break
            t_stop = time.monotonic() + min(30.0, tr * 0.25)
            try:
                res, sc = lp_optimize(alt, t_stop, max_iters=80, seed_val=50000 + idx)
                update_best(res)
            except Exception:
                pass
            if time.monotonic() < t_stop + 15:
                try:
                    res2, sc2 = coord_descent(res, t_stop + 10, max_sweeps=30, seed_val=60000 + idx)
                    update_best(res2)
                except Exception:
                    pass

    # Phase 6: Final LP + coordinate polish
    p6_end = min(hard_deadline - 30, t0 + budget * 0.92)
    if time.monotonic() < p6_end - 10:
        try:
            best_f, best_score = lp_optimize(best_f, p6_end, max_iters=80, seed_val=88888)
        except Exception:
            pass

    if time.monotonic() < hard_deadline - 20:
        try:
            best_f, best_score = coord_descent(best_f, hard_deadline - 20, max_sweeps=40, seed_val=55555)
        except Exception:
            pass

    # Phase 7: Pair polish
    if time.monotonic() < hard_deadline - 12:
        try:
            best_f, best_score = pair_polish(best_f, hard_deadline - 12, max_rounds=15, seed_val=2718)
        except Exception:
            pass

    # Final micro coordinate pass
    if time.monotonic() < hard_deadline - 5:
        try:
            best_f, best_score = coord_descent(best_f, hard_deadline - 5, max_sweeps=10, seed_val=31415)
        except Exception:
            pass

    # Validate and return
    best_f = np.clip(np.asarray(best_f, dtype=np.float64), 0, 1000)
    best_f = np.where(np.isfinite(best_f), best_f, 0.0)
    if best_f.size == 0 or best_f.sum() < 0.01:
        best_f = parent.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_f]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            return [1.0]
        best_f = parent

    return [float(x) for x in best_f]
