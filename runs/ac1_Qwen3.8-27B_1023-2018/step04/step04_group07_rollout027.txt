import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv(a):
    """Full autocorrelation a*a via FFT."""
    n = len(a)
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _sc(a):
    """Score = 2n * max(conv) / sum^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return math.inf
    m = _conv(a).max()
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _nm(a):
    """Normalize to sum=1."""
    s = a.sum()
    if s < 1e-12:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


def _build_A(f, n):
    """Convolution matrix: result[k] = sum_j A[k,j]*g[j] = (f*g)[k]."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(f, n, M, tl=5.0):
    """LP: max 1'g s.t. A@g <= M, g >= 0. Returns normalized g or None."""
    A = _build_A(f, n)

        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                    bounds=[(0, None)] * n, method='highs',
                    options={'time_limit': tl, 'presolve': True})
    except Exception:
        return None
    if r.status not in (0, 3) or r.x is None:
        return None
    g = np.maximum(r.x, 0.0)
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s

def _fw_iterate(f, n, deadline, seed):

    rng = np.random.default_rng(seed)
    f = np.maximum(f.astype(np.float64), 0.0)

    M = float(_conv(f).max())

    max_stall = 40
    iters = 0

                M = float(_conv(f).max())
                stall = 0
