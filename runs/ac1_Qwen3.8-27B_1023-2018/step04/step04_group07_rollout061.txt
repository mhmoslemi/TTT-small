import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """Build the (2n-1) x n convolution matrix."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_solve(f, n, M, time_limit=5.0, top_k=None):
    """Solve the LP: max sum(g) s.t. A@g <= M, g >= 0"""
    b = _conv_full(f)
    if top_k is not None:

        A = _build_A(f, n)[sorted_idx]
        b_ub = np.full(len(sorted_idx), M)

        A = _build_A(f, n)
        b_ub = np.full(2 * n - 1, M)

    return g / g_sum

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, n, M, time_limit=3.0):
    A = _build_A(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method="highs",
                      options={"time_limit": time_limit, "presolve": True})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum

def _smoothed_grad_descent(a0, n, stop_time, seed):
    """Smoothed max-convolution minimization via log-sum-exp gradient."""
    rng = np.random.default_rng(seed)
    a = np.clip(np.asarray(a0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while a.size < n:
        a = np.concatenate([a, np.zeros(n - a.size)])
    a = a[:n]
    fs = a.sum()
    if fs > 1e-10:
        a = a / fs
    else:
        a = np.ones(n, dtype=np.float64) / n

    best_a = a.copy()
    best_M = float(_conv_full(a).max())

    fsize = next_fast_len(2 * n)
    A_pad = np.zeros(fsize)
    W = np.zeros(fsize)

    mu0 = 0.005
    for t in range(50000):
        if time.monotonic() > stop_time - 1.0:
            break
        conv = _conv_full(a)
        M_cur = float(conv.max())
        if M_cur < best_M - 1e-15:
            best_M = M_cur
            best_a = a.copy()

        mu = max(0.0005, mu0 / (1.0 + 0.001 * t))
        w = np.exp((conv - M_cur) / mu)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        A_pad[:n] = a
        W[:2*n-1] = w
        AF = rfft(A_pad, fsize)
        WF = rfft(W, fsize)
        grad_full = irfft(WF * AF, fsize)[:2*n-1] + irfft(np.conj(WF) * AF, fsize)[:2*n-1]
        grad = grad_full[:n]

        # Backtracking
        eta = 0.002
        improved = False
        for _ in range(8):
            a_new = np.clip(a - eta * grad, 0, None)
            s_new = a_new.sum()
            if s_new < 1e-10:
                eta *= 0.5
                continue
            a_new = a_new / s_new
            M_new = float(_conv_full(a_new).max())
            if M_new < M_cur - 1e-15:
                a = a_new
                improved = True
                break
            eta *= 0.5

        if not improved:
            # Small random perturbation
            a = np.clip(a + rng.normal(0, 0.005, n), 0, None)
            fs2 = a.sum()
            if fs2 > 1e-10:
                a = a / fs2

    return best_a, _score(best_a)


def _coord_descent(a, stop_time, seed=42, max_rounds=10):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel().copy(), 0, None)
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0

    while time.monotonic() < stop_time - 1.0 and no_imp < max_rounds:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            if orig > 1e-10:
                mults = (0.0, 0.1, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0)
            else:
                mults = (0.0, 0.01, 0.05, 0.1, 0.5, 1.0, 5.0)
            best_val = orig
            best_sc_local = best_sc
            for mlt in mults:
                nv = orig * mlt if orig > 1e-10 else mlt
                best[i] = nv
                s = best.sum()
                if s < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_sc_local - 1e-16:
                    best_sc_local = sc
                    best_val = nv
            best[i] = best_val
            if best_sc_local < best_sc - 1e-16:
                best_sc = best_sc_local
                improved = True

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc


def _pair_descent(a, stop_time, seed=99):
    """Coordinate descent on pairs."""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel().copy(), 0, None)
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0

    while time.monotonic() < stop_time - 1.0 and no_imp < 4:
        improved = False
        n_pairs = min(200, n * 2)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 1.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            total = oi + oj
            if total < 1e-12:
                continue
            fracs = (0.0, 0.05, 0.2, 0.4, 0.5, 0.6, 0.8, 0.95, 1.0)
            for fr in fracs:
                best[i] = fr * total
                best[j] = (1.0 - fr) * total
                sc = _score(best)
                if sc < best_sc - 1e-16:
                    best_sc = sc
                    improved = True
        best[i], best[j] = oi, oj
        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc

            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_c = np.clip(res, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in res_c]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = res_c.copy()
                    best_score = exact
        except Exception:
            pass

    # Phase 2: LP again on best + try other lengths
    if time.monotonic() < phase3_end - 30.0:
        lp_stop = min(phase3_end - 25.0, time.monotonic() + 200.0)
        try:
            res, sc = _lp_optimize(best_seq, n, lp_stop, seed + 31337)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_c = np.clip(res, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in res_c]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = res_c.copy()
                    best_score = exact
        except Exception:
            pass

    # Phase 2b: Try other lengths briefly
    if time.monotonic() < phase3_end - 20.0 and best_score > 1.5028:
        for L in [n_p - 32, n_p + 32]:
            if time.monotonic() > phase3_end - 15.0:
                break
            if L < 16 or L > 600:
                continue
            stop = min(phase3_end - 12.0, time.monotonic() + 40.0)
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n), best_seq)
            try:
                res, sc = _lp_optimize(init_l, L, stop, seed + L * 7)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res_c = np.clip(res, 0, 1000)
                    exact = float(evaluate_sequence([float(x) for x in res_c]))
                    if math.isfinite(exact) and exact < best_score:
                        best_seq = res_c.copy()
                        best_score = exact
            except Exception:
                pass

    # Phase 3: Coordinate descent polish
    if time.monotonic() < phase3_end - 5.0:
        remaining = phase3_end - 5.0 - time.monotonic()
        n_rounds = max(2, int(remaining / 30))
        n_rounds = min(n_rounds, 8)
        per_round = remaining / n_rounds

        for r in range(n_rounds):
            r_stop = time.monotonic() + per_round
            if r_stop > phase3_end:
                r_stop = phase3_end
            if time.monotonic() >= r_stop - 2.0:
                break
            try:
                if r % 3 == 0:
                    best_seq, sc = _coord_descent(best_seq, r_stop, seed=20000 + r * 71)
                elif r % 3 == 1:
                    best_seq, sc = _pair_descent(best_seq, r_stop, seed=40000 + r * 83)
                else:
                    best_seq, sc = _coord_descent(best_seq, r_stop, seed=60000 + r * 97)
                if math.isfinite(sc) and sc < best_score:
                    best_score = sc
            except Exception:
                pass

    # Final smooth + LP micro-optimization
    final_stop = t0 + budget - 8.0
    if time.monotonic() < final_stop - 15.0:
        micro_stop = min(final_stop - 5.0, time.monotonic() + 30.0)
        try:
            res, sc = _smoothed_grad_descent(best_seq, len(best_seq), micro_stop, seed + 88888)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_c = np.clip(res, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in res_c]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = res_c.copy()
                    best_score = exact
        except Exception:
            pass

    # One final coordinate descent
    if time.monotonic() < final_stop - 3.0:
        try:
            best_seq, sc = _coord_descent(best_seq, final_stop, seed=99999)
            if math.isfinite(sc) and sc < best_score:
                best_score = sc
        except Exception:
            pass

    # Final validation
    final_seq = np.clip(best_seq, 0, 1000)
    final_seq = final_seq[~np.isnan(final_seq)]
    if final_seq.size == 0:
        final_seq = np.ones(384) / 384
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        final_seq = np.ones(n_p, dtype=np.float64) / n_p
        try:
            final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
        except Exception:
            return [1.0]

    return [float(x) for x in final_seq]
