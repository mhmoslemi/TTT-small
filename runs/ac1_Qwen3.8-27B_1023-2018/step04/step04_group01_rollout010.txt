def _conv_full(a):
    n = len(a)
    if n <= 512:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):

        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):

        return None, M, b

def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def _grad_optimize(f0, n, stop_time, seed, max_iters=50000):

    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f_sum = float(f.sum())

        f = np.ones(n) / n
    else:
        f = f / f_sum
    best_f = f.copy()
    best_score = _score(f)

    alpha = 0.04
    alpha_min = 1e-10
    tau = 60.0
    tau_decay = 0.9995

    for it in range(max_iters):
        if time.monotonic() > stop_time - 6.0:
            break

        b = _conv_full(f)
        S = float(f.sum())
        M = float(b.max())
        if M < 1e-18:
            break
        score = 2.0 * n * M / (S * S)

        if score < best_score - 1e-14:
            best_score = score
            best_f = f.copy()

        # Smooth max gradient
        b_shifted = b - M
        tau_v = tau * b_shifted
        tau_v -= tau_v.max()
        p = np.exp(tau_v)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(2 * n - 1)
            p[np.argmax(b)] = 1.0
            p_sum = 1.0
        p /= p_sum

        # grad[i] = 2 * sum_k p[k] * a[k-i] (for valid k-i)
        # = 2 * (p * a_rev)[i + n - 1]
        f_rev = f[::-1]
        conv_size = 2 * n - 1

        Fr = rfft(f_rev, conv_size)
        conv_val = irfft(P * Fr, conv_size)[:conv_size]
        grad = 2.0 * conv_val[n - 1:2 * n - 1]

        # Score gradient (with respect to f, on simplex)
        score_grad = (2.0 * n / (S * S)) * (grad - 2.0 * M / S)

        # Gradient step
        g_raw = f - alpha * score_grad
        f_new = _project_simplex(g_raw)

        sc_new = _score(f_new)
        if sc_new < score - 1e-15:
            f = f_new
            alpha = max(alpha_min, alpha * 1.01)
            tau *= tau_decay
        else:
            alpha = max(alpha_min, alpha * 0.5)
            if alpha < alpha_min * 10:
                f = np.clip(f + rng.normal(0, 0.003, n), 0, None)

                alpha = 0.02

            f = np.clip(f + rng.normal(0, 0.008, n), 0, None)
            f /= f.sum()
            alpha = 0.03
            tau = 50.0

                    # Use in-place for speed
                    tmp = conv + w * infl
                    if 2 * i < 2 * n - 1:
                        tmp[2 * i] += w * w
                    m_new = float(tmp.max())
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-16:
                    best_obj = obj
                    best_x = x

            # Fine grid around best
            if best_x != v:
                dw = x_max / n_pts
                x_lo = max(0.0, best_x - 2 * dw)
                x_hi = min(x_max, best_x + 2 * dw)
                x_fine = np.linspace(x_lo, x_hi, 40)
                for x in x_fine:
                    S_new = S_rest + x
                    if S_new < 0.01:
                        continue
                    w = x - v
                    if abs(w) < 1e-14:
                        continue
                    tmp = conv + w * infl
                    if 2 * i < 2 * n - 1:
                        tmp[2 * i] += w * w
                    m_new = float(tmp.max())
                    obj = 2.0 * n * m_new / (S_new * S_new)
                    if obj < best_obj - 1e-16:
                        best_obj = obj
                        best_x = x

            # Accept
            if best_x != v and best_obj < best_score - 1e-15:
                w = best_x - v
                f[i] = best_x
                S += w
                conv += w * infl
                if 2 * i < 2 * n - 1:
                    conv[2 * i] += w * w
                best_score = best_obj
                best_f = f.copy()
                improved = True

        if not improved and sweep > 5:
            break

    f = np.clip(best_f, 0, 1000)
    final_score = _score(f)
    return f, final_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard_deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_p = len(parent)

    fallback = parent.copy()
    try:
        fallback_sc = _score(fallback)
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(400) / 400.0
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update_best(seq, sc=None):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64).ravel()
        seq = np.clip(seq, 0, 1000)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        if sc is None:
            sc = _score(seq)
        if not math.isfinite(sc):
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-15:
            best_seq = seq.copy()
            best_score = exact

    def _run_pipeline(n, init_arr, stop_time, label=""):
        nonlocal best_seq, best_score
        if time.monotonic() > stop_time - 15:
            return
        init_arr = np.clip(np.asarray(init_arr, dtype=np.float64).ravel(), 0, None)
        while init_arr.size < n:
            init_arr = np.concatenate([init_arr, np.zeros(n - init_arr.size)])
        init_arr = init_arr[:n]
        if init_arr.sum() < 1e-10:
            init_arr = np.ones(n) / n
        init_arr = init_arr / init_arr.sum()

        # LP phase
        t_lp_end = min(stop_time - 20, time.monotonic() + (stop_time - time.monotonic()) * 0.45)
        if time.monotonic() < t_lp_end - 5:
            try:
                res, sc = _lp_optimize(init_arr, n, t_lp_end, seed)
            except Exception:
                res, sc = init_arr, _score(init_arr)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res = res / res.sum()
                _update_best(res, sc)

        # Gradient phase
        t_grad_end = min(stop_time - 12, time.monotonic() + (stop_time - time.monotonic()) * 0.2)
        if time.monotonic() < t_grad_end - 3 and best_score > 1.502:
            try:
                res, sc = _grad_optimize(best_seq, len(best_seq), t_grad_end, seed + 1)
            except Exception:
                res, sc = best_seq, best_score
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update_best(res, sc)

        # Coord descent
        t_cd_end = min(stop_time - 6, time.monotonic() + (stop_time - time.monotonic()) * 0.25)
        if time.monotonic() < t_cd_end - 3:
            try:
                res, sc = _coord_polish(best_seq, t_cd_end, max_sweeps=60, seed=seed + 2)
            except Exception:
                res, sc = best_seq, best_score
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update_best(res, sc)

    # Phase 1: Main optimization on parent
    phase1_end = t0 + budget * 0.40
    _run_pipeline(n_p, parent, phase1_end, "main")

    # Phase 2: Alternative initializations
    if time.monotonic() < t0 + budget * 0.52 and best_score > 1.503:
        i_arr = np.arange(n_p, dtype=np.float64)
        alt_inits = []
        # Symmetric cosine
        alt_inits.append(np.clip(0.3 + 0.7 * np.cos(np.pi * (i_arr / n_p - 0.5)) ** 2, 0, None))
        # Linear ramp (with zero at ends)
        alt_inits.append(np.clip(np.sin(np.pi * i_arr / (n_p - 1)), 0, None))
        # Two peaks
        alt_inits.append(np.clip(
            0.5 + 0.4 * np.cos(2 * np.pi * i_arr / (n_p // 2)), 0, None
        ))
        # Triangular
        mid = (n_p - 1) / 2.0
        alt_inits.append(np.clip(1.0 - abs(i_arr - mid) / mid, 0, None))
        # Geometric (one-sided)
        alt_inits.append(np.clip(np.exp(-i_arr / (n_p * 0.3)), 0, None))

        for alt in alt_inits:
            if time.monotonic() > t0 + budget * 0.52:
                break
            if alt.sum() < 0.01:
                continue
            alt = alt / alt.sum()
            # Quick LP + coord
            t_alloc = min(t0 + budget * 0.52 - 10, time.monotonic() + 40.0)
            try:
                res, sc = _lp_optimize(alt, n_p, t_alloc, seed + 1000)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res = res / res.sum()
                    _update_best(res, sc)
            except Exception:
                pass

    # Phase 3: Intensive coord polish on best
    t_cd_end = min(t0 + budget * 0.72, hard_deadline - 50)
    if time.monotonic() < t_cd_end - 10:
        try:
            res, sc = _coord_polish(best_seq, t_cd_end, max_sweeps=80, seed=424242)
            _update_best(res, sc)
        except Exception:
            pass

    # Phase 4: Gradient polish
    t_grad_end = min(t0 + budget * 0.82, hard_deadline - 40)
    if time.monotonic() < t_grad_end - 8 and best_score > 1.502:
        try:
            res, sc = _grad_optimize(best_seq, len(best_seq), t_grad_end, seed + 9999)
            _update_best(res, sc)
        except Exception:
            pass

    # Phase 5: Try slightly different lengths
    t_len_end = min(t0 + budget * 0.90, hard_deadline - 40)
    if time.monotonic() < t_len_end - 25 and best_score > 1.5028:
        other_lengths = []
        for L in [n_p - 8, n_p - 16, n_p + 8, n_p + 16, n_p * 2]:
            if 50 <= L <= 800 and L != n_p and L not in other_lengths:
                other_lengths.append(L)
        other_lengths.sort(key=lambda L: abs(L - n_p))

        time_per = max(12.0, (t_len_end - time.monotonic()) / max(1, len(other_lengths[:3])))
        for L in other_lengths[:3]:
            if time.monotonic() > t_len_end - 10:
                break
            if best_score < 1.5028:
                break
            init_l = np.interp(
                np.linspace(0, 1, L),
                np.linspace(0, 1, len(best_seq)),
                best_seq
            )
            init_l = np.clip(init_l, 0, None)
            if init_l.sum() < 0.01:
                init_l = np.ones(L) / L
            else:
                init_l /= init_l.sum()

            t_stop = min(t_len_end - 8, time.monotonic() + time_per)
            try:
                res, sc = _lp_optimize(init_l, L, t_stop, seed + 50000 + L)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res = res / res.sum()
                    _update_best(res, sc)
            except Exception:
                pass

    # Phase 6: Final coord polish
    final_end = hard_deadline - 12.0
    if time.monotonic() < final_end - 8:
        try:
            res, sc = _coord_polish(best_seq, final_end - 8, max_sweeps=50, seed=31337)
            _update_best(res, sc)
        except Exception:
            pass

    # Phase 7: One more gradient pass
    if time.monotonic() < hard_deadline - 6:
        try:
            res, sc = _grad_optimize(best_seq, len(best_seq), hard_deadline - 4, seed + 77777)
            _update_best(res, sc)
        except Exception:
            pass

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
