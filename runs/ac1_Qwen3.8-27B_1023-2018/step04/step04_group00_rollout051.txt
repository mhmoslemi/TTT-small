                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                dval = base[diag] + w * infl[diag] + w * w
                maff = np.maximum(aff, dval[:, None])
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Fine refinement around the best
            dw = x_hi / float(G)
            lo = max(0.0, best_x - 2.5 * dw)
            hi = min(x_hi, best_x + 2.5 * dw)
            if hi <= lo:
                hi = lo + 1e-12
            fine = np.linspace(lo, hi, 48, dtype=np.float64)
            if 0.0 <= best_x <= x_hi:
                fine = np.unique(np.concatenate((fine, [v])))
            vals2 = eval_xs(fine)
            if np.any(np.isfinite(vals2)):
                imin2 = int(np.argmin(vals2))
                vx2 = float(fine[imin2])
                vv2 = float(vals2[imin2])
                if vv2 < best_val:
                    best_x = vx2
                    best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-14:
                w_new = best_x - v
                if abs(w_new) > 1e-15:
                    p[i] = best_x
                    S_total += w_new
                    c[start_i:end_i] += w_new * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w_new * w_new
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_fft(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-16:
            best_score = new_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 8.0:
                    # Perturb to escape local minimum
                    sigma = 0.02 + 0.01 * min(flat // 4, 5)
                    p = p * np.exp(rng.normal(0.0, sigma, n, dtype=np.float64))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0.0:
                        p /= ss
                    c = _conv_fft(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
                        best_score = cur_score
                        best_p = p.copy()
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    c = _conv_fft(p)
    sc = float(2.0 * n * c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 20.0
    if hard <= t0 + 5.0:
        hard = t0 + max(5.0, budget * 0.9)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact_a(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64)
        best_score = 2.0

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in a]))
        except Exception:
            try:
                sc = float(_score_exact_a(a))
            except Exception:
                return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    # Determine lengths to try
    parent_len = int(best_arr.size) if best_arr.size > 0 else 384
    lengths = [parent_len]
    for cl in [parent_len // 2, parent_len * 2, parent_len + 512]:
        if cl >= 64 and cl not in lengths:
            lengths.append(cl)
    lengths = sorted(set(lengths))
    if len(lengths) > 4:
        lengths = lengths[-4:]

    # Time allocation: 55% for first (parent) length, rest split
    phase_end = hard - 20.0
    k = len(lengths)
    if k == 1:
        allocs = [phase_end - t0]
    else:
        wts = [0.55] + [0.45 / (k - 1)] * (k - 1)
        sw = sum(wts)
        wts = [w / sw for w in wts]
        total_avail = phase_end - t0
        allocs = [w * total_avail for w in wts]

    rng_top = np.random.default_rng(seed)

    for idx, L in enumerate(lengths):
        now = time.monotonic()
        end_phase = t0 + sum(allocs[:idx + 1])
        end_phase = min(end_phase, phase_end - 10.0)
        if now > end_phase - 15.0:
            break
        if end_phase <= now + 15.0:
            continue

        src = resample_p = _resample(best_arr, L)
        if src.sum() <= 0:
            src = np.full(L, 1.0 / L)
        src = src / float(src.sum())
        consider(src)

        # Phase 1: LSE optimization
        lse_stop = min(end_phase - 30.0, now + 0.60 * (end_phase - now))
        if lse_stop > now + 10.0:
            try:
                p_lse = _lbfgs_opt(src, lse_stop, consider=consider, fine_only=False)
                if p_lse is not None and p_lse.size == L and np.all(np.isfinite(p_lse)) and p_lse.sum() > 0:
                    p_lse = p_lse / float(p_lse.sum())
                    if _score_fft_a(p_lse) < _score_fft_a(src) - 1e-6:
                        src = p_lse
            except Exception:
                pass

        # Phase 2: Fine LSE
        now = time.monotonic()
        if now < end_phase - 40.0:
            fine_stop = min(end_phase - 35.0, now + 0.20 * (end_phase - now))
            if fine_stop > now + 10.0:
                try:
                    p_fine = _lbfgs_opt(src, fine_stop, consider=consider, fine_only=True)
                    if p_fine is not None and p_fine.size == L and np.all(np.isfinite(p_fine)) and p_fine.sum() > 0:
                        p_fine = p_fine / float(p_fine.sum())
                        if _score_fft_a(p_fine) < _score_fft_a(src) - 1e-8:
                            src = p_fine
                except Exception:
                    pass

        # Phase 3: Coordinate polish
        now = time.monotonic()
        coord_stop = min(end_phase - 2.0, now + max(20.0, (end_phase - now)))
        if coord_stop > now + 10.0:
            try:
                p_c, _ = _coord_polish(src, coord_stop, seed=int(seed + 777 + L), consider=consider, G=None)
                if p_c is not None and p_c.size == L and np.all(np.isfinite(p_c)) and p_c.sum() > 0:
                    p_c = p_c / float(p_c.sum())
                    consider(p_c)
            except Exception:
                pass

    # Final squeeze on the saved best
    now = time.monotonic()
    final_stop = min(hard - 5.0, t0 + budget - 8.0)
    if final_stop > now + 30.0 and best_arr.size > 0:
        nb = int(best_arr.size)
        try:
            p_c, _ = _coord_polish(best_arr, min(final_stop - 5.0, now + final_stop - now - 10.0),
                                   seed=int(seed + 424242), consider=consider, G=56)
            if p_c is not None and p_c.size == nb and np.all(np.isfinite(p_c)) and p_c.sum() > 0:
                p_c = p_c / float(p_c.sum())
                consider(p_c)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.ones(64, dtype=np.float64)
        ls = [float(x) for x in out]
        sc = evaluate_sequence(ls)
        if not math.isfinite(sc):
            ls = [float(x) for x in np.ones(64, dtype=np.float64)]
        return ls
    except Exception:
        return [1.0]
