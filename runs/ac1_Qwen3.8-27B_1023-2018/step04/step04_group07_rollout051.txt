```python
    """LP-based optimization loop."""
    rng = np.random.default_rng(seed)
    f = np.clip(f_init.copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs
    
    best_f = f.copy()
    best_M = float(_conv_full(f).max())
    M = best_M
    
    stall = 0
    iter_count = 0
    
    while time.monotonic() < stop_time - 1.0 and stall < 50:
        M = float(_conv_full(f).max())
        
        # Build constraint matrix
        A = _build_conv_mat(f, n)
        
        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method="highs",
                          options={"presolve": True, "time_limit": 5.0})
        except Exception:
            stall += 10
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            f /= f.sum()
            continue
        
        if res.status not in (0, 3) or res.x is None:
            stall += 5
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            f /= f.sum()
            continue
        
        g = np.asarray(res.x, dtype=np.float64)
        g_sum = float(g.sum())
        if g_sum < 1e-12:
            stall += 10
            continue
        
        g = g / g_sum
        
        # Check if g is meaningfully different from f
        if float(np.linalg.norm(g - f)) < 1e-8:
            stall += 5
            f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
            f /= f.sum()
            continue
        
        # Precompute convolutions for line search
        cff = _conv_full(f)
        cfg = _conv_pair(f, g)
        cgg = _conv_full(g)
        
        # Line search
        t_opt, m_opt = _line_search(cff, cfg, cgg)
        
        if m_opt < M - 1e-12 and t_opt > 0:
            f = (1.0 - t_opt) * f + t_opt * g
            f = np.clip(f, 0, None)
            fs = f.sum()
            if fs < 1e-10:
                f = best_f.copy()
            else:
                f = f / fs
            M = float(_conv_full(f).max())
            if M < best_M - 1e-14:
                best_M = M
                best_f = f.copy()
            stall = 0
        else:
            # Try smaller step
            small_t = 0.1
            f_try = (1.0 - small_t) * f + small_t * g
            f_try = np.clip(f_try, 0, None)
            f_try /= f_try.sum()
            m_try = float(_conv_full(f_try).max())
            if m_try < M - 1e-12:
                f = f_try
                M = m_try
                if M < best_M - 1e-14:
                    best_M = M
                    best_f = f.copy()
                stall = 0
            else:
                stall += 1
        
        # Perturbation for stall
        if 0 < stall < 30:
            f = f + rng.normal(0, 0.02, n)
            f = np.clip(f, 0, None)
            f /= f.sum()
        
        iter_count += 1
    
    return best_f, _score_normalized(best_f)

    fs = f.sum()

            if time.monotonic() > stop_time - 1.0:
                break

            best_d = 0.0
            best_max = best_M  # current max without change

                if orig > 1e-12:
                    d = orig * (mlt - 1.0)
                else:
                    d = mlt * 0.01  # small positive if currently zero
                
                if orig + d < -1e-12:
                    continue
                
                # Update C: C[k] += 2*d*a[k-i] + d^2 (if k=2i) for k in [i, i+n-1]
                k_start = max(0, i)
                k_end = min(2 * n - 2, i + n - 1)

                valid_k = np.arange(k_start, k_end + 1)
                valid_j = valid_k - i  # = k-i, ranges from max(0,-i) to min(n-1, 2n-2-i)

                # Recompute score (sum changes by d)
                # Actually, since score = 2n*max_conv/(sum)^2, and we normalize at the end...
                # Let's just track max_conv and normalize later

                    best_d = d

                f[i] += best_d
                f[i] = max(0.0, f[i])
                C = best_C
                if best_max < best_M - 1e-14:
                    best_M = best_max
                    best_f = f.copy()
                    improved = True
            
        # Renormalize
        fs = f.sum()
        if fs < 1e-10:
            f = best_f.copy()
            C = _conv_full(f)
            best_M = float(C.max())
        else:
            f = f / fs
            C = _conv_full(f)
            best_M = float(C.max())
            if best_M < float(best_f.sum() and _conv_full(best_f).max()):
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _conv_pair(a, b):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    Fa = rfft(a, fsize)
    Fb = rfft(b, fsize)
    c = irfft(Fa * Fb, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_internal(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _line_search(cff, cfg, cgg):
    def obj(t):
        return float(((1-t)**2 * cff + 2*t*(1-t)*cfg + t*t*cgg).max())
    
    lo, hi = 0.0, 1.0
    gr = (1.0 + math.sqrt(5.0)) / 2.0
    for _ in range(80):
        m1 = hi - (hi - lo) / gr
        m2 = lo + (hi - lo) / gr
        if obj(m1) < obj(m2):
            hi = m2
        else:
            lo = m1
    t_opt = (lo + hi) / 2.0
    return t_opt, obj(t_opt)

def _lp_iterate(f, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(f.copy(), 0, None)
    f /= f.sum()
    
    best_f = f.copy()
    best_M = float(_conv_full(f).max())
    M = best_M
    stall = 0
    
    while time.monotonic() < stop_time - 1.0 and stall < 60:
        # Build LP
        A = _build_A(f, n)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method="highs",
                          options={"presolve": True, "time_limit": 5.0})
        except Exception:
            stall += 5
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            f /= f.sum()
            M = float(_conv_full(f).max())
            continue
        
        if res.status not in (0, 3) or res.x is None:
            stall += 5
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            f /= f.sum()
            M = float(_conv_full(f).max())
            continue
        
        g = np.asarray(res.x, dtype=np.float64)
        g_sum = float(g.sum())
        if g_sum < 1e-12:
            stall += 10
            break
        
        g = g / g_sum
        
        # Check difference
        diff = float(np.linalg.norm(g - f))
        if diff < 1e-8:
            stall += 5
            f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
            f /= f.sum()
            M = float(_conv_full(f).max())
            continue
        
        # Line search
        cff = _conv_full(f)
        cfg = _conv_pair(f, g)
        cgg = _conv_full(g)
        t_opt, m_opt = _line_search(cff, cfg, cgg)
        
        if t_opt > 0 and m_opt < M - 1e-12:
            f_new = (1.0 - t_opt) * f + t_opt * g
            f_new = np.clip(f_new, 0, None)
            fs = f_new.sum()
            if fs > 1e-10:
                f_new /= fs
                f = f_new
                M = float(_conv_full(f).max())
                if M < best_M - 1e-14:
                    best_M = M
                    best_f = f.copy()
                stall = 0
            else:
                stall += 1
        else:
            # Try small step
            for ts in [0.05, 0.1, 0.2]:
                f_try = (1-ts)*f + ts*g
                f_try = np.clip(f_try, 0, None)
                f_try /= f_try.sum()
                m_try = float(_conv_full(f_try).max())
                if m_try < M - 1e-12:
                    f = f_try
                    M = m_try
                    if M < best_M - 1e-14:
                        best_M = M
                        best_f = f.copy()
                    stall = 0
                    break
            else:
                stall += 1
        
        # Perturb on mild stall
        if 5 < stall < 40:
            sigma = 0.02 * (stall / 10)
            f = np.clip(f + rng.normal(0, sigma, n), 0, None)
            f /= f.sum()
            M = float(_conv_full(f).max())
    
    return best_f, 2.0 * n * float(_conv_full(best_f).max())
```
