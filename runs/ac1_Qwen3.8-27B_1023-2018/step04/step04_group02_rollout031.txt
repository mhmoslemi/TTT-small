```python
import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len

def _conv(a):
    n = len(a)
    if n <= 1024:
        return np.convolve(a, a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """Build the (2n-1) x n matrix A such that (A @ g)[k] = (f*g)[k]"""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_direction(f, n, M, timeout=2.0):
    """Find direction g maximizing sum(g) subject to (f*g)_k <= M, g >= 0"""
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g
```

```python
                break
            elif sc < _score(f) - 1e-14:
                f = f_new
                improved = True
                t = min(0.9, max(0.05, ti))
                stall = 0
                break
            else:
                t = max(0.02, t * 0.6)

        if not improved:
            stall += 1
            if stall >= max_stall:
                # Restart from best with perturbation
                f = np.clip(best_f + rng.normal(0, 0.03 * max(1.0, np.mean(best_f)), n), 0, None)
                if f.sum() < 1e-10:
                    f = best_f.copy()
                f = f / f.sum()
                t = 0.3
                stall = 0
                max_stall = min(max_stall + 100, 500)
                # CD polish before restarting
                if time.monotonic() < stop_time - 15.0:
                    f_c, sc_c = _cd_polish(f, stop_time - 10.0, max_outer=3, seed=seed + it)
                    if sc_c < best_score:
                        best_score = sc_c
                        best_f = f_c.copy()
                        f = f_c.copy()

        it += 1
        # Periodic CD polish
        if it % 20 == 0 and time.monotonic() < stop_time - 20.0:
            f_c, sc_c = _cd_polish(f, stop_time - 10.0, max_outer=2, seed=seed + it)
            if sc_c < best_score:
                best_score = sc_c
                best_f = f_c.copy()
                f = f_c.copy()
                stall = 0

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(400, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(400, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    n = len(parent)

    # Fallback
    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(200)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_opt(init, stop, sd, n_init=None):
        nonlocal best_seq, best_score
        if n_init is None:
            n_init = len(init)
        try:
            res, sc = _lp_optimize(init, n_init, stop, sd)
        except Exception:
            return
        if res is None or len(res) == 0 or not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0, 1000)
        if res.sum() < 0.01:
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in res]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-14:
            best_seq = res.copy()
            best_score = exact

    # === Phase 1: Multi-start LP optimization (50% of time) ===
    phase1_end = t0 + budget * 0.50
    rng_main = np.random.default_rng(seed)

    i_arr = np.arange(n, dtype=np.float64)
    inits = [parent.copy() / max(parent.sum(), 1e-10)]

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / n)))
    # Shifted raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * (i_arr + n * 0.05) / n)))
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * (i_arr + n * 0.15) / n)))
    # Parabolic
    ctr = (n - 1) / 2.0
    inits.append(np.maximum(0.0, 1.0 - ((i_arr - ctr) / (0.55 * n)) ** 2))
    # Two-lobe
    inits.append(np.exp(-0.5 * ((i_arr - 0.25 * n) / (0.12 * n)) ** 2) +
                 np.exp(-0.5 * ((i_arr - 0.75 * n) / (0.12 * n)) ** 2))
    # Bump
    inits.append(np.exp(-0.5 * ((i_arr - 0.5 * n) / (0.2 * n)) ** 2))
    # Sine-squared
    inits.append(np.sin(np.pi * i_arr / n) ** 2)
    # Parent with noise
    sig = 0.1 * max(1.0, np.mean(parent))
    inits.append(np.abs(parent + rng_main.normal(0, sig, n)) + 1e-4)
    # Parent with different noise
    inits.append(np.abs(parent + rng_main.normal(0, 2 * sig, n)) + 1e-4)
    # Alternating with envelope
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.6)
    inits.append(alt * (0.3 + 0.7 * np.cos(np.pi * i_arr / n) ** 2))
    # Uniform (baseline)
    inits.append(np.ones(n))
    # Triangular (inverted V)
    inits.append(1.0 - 2.0 * np.abs(i_arr / n - 0.5))

    # Deduplicate and normalize
    seen_hashes = set()
    inits_dedup = []
    for ix in inits:
        ix = np.clip(ix, 0, None)
        if ix.sum() < 1e-10:
            continue
        ix = ix / ix.sum()
        h = hash(ix.tobytes())
        if h not in seen_hashes:
            seen_hashes.add(h)
            inits_dedup.append(ix)

    # Run LP on each init with time allocation
    time_per_init = max(5.0, (phase1_end - t0) / len(inits_dedup))
    for idx, init in enumerate(inits_dedup):
        if time.monotonic() > phase1_end - 10.0:
            break
        if best_score < 1.45:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init * 1.5)
        _try_opt(init, stop, seed + 1000 * (idx + 1))

    # === Phase 2: Warm restart on best (15% of time) ===
    phase2_end = t0 + budget * 0.68
    if time.monotonic() < phase2_end - 15.0 and len(best_seq) == n:
        stop_p2 = min(phase2_end - 5.0, time.monotonic() + (phase2_end - time.monotonic()) * 0.85)
        _try_opt(best_seq.copy(), stop_p2, seed + 77777)

    # === Phase 3: CD polish (15% of time) ===
    phase3_end = t0 + budget * 0.83
    if time.monotonic() < phase3_end - 5.0:
        try:
            f_c, sc_c = _cd_polish(best_seq, phase3_end, max_outer=80, seed=seed * 3 + 1)
        except Exception:
            f_c, sc_c = best_seq.copy(), best_score
        if sc_c < best_score - 1e-14:
            try:
                exact = float(evaluate_sequence([float(x) for x in f_c]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-14:
                best_seq = f_c.copy()
                best_score = exact

    # === Phase 4: Try nearby lengths (10% of time) ===
    phase4_end = t0 + budget * 0.93
    if time.monotonic() < phase4_end - 12.0:
        for L in [n + 8, n - 8, n + 16, n - 16, n + 32, n - 32]:
            if L < 100 or time.monotonic() > phase4_end - 10.0:
                continue
            if len(best_seq) != n:
                init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
            else:
                init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
            stop_l = min(phase4_end - 4.0, time.monotonic() + max(5.0, (phase4_end - time.monotonic()) * 0.4))
            _try_opt(init_l, stop_l, seed + 90000 + L, n_init=L)

    # === Phase 5: Final intensive CD + LP (remaining time) ===
    final_end = deadline - 5.0
    if time.monotonic() < final_end - 10.0:
        # More LP iterations
        if len(best_seq) == n and time.monotonic() < final_end - 30.0:
            stop_p5 = min(final_end - 25.0, time.monotonic() + (final_end - time.monotonic()) * 0.5)
            _try_opt(best_seq.copy(), stop_p5, seed + 123456)

        # Final CD
        if time.monotonic() < final_end - 8.0:
            try:
                f_c, sc_c = _cd_polish(best_seq, final_end, max_outer=100, seed=seed * 7 + 42)
            except Exception:
                f_c, sc_c = best_seq.copy(), best_score
            if sc_c < best_score - 1e-14:
                try:
                    exact = float(evaluate_sequence([float(x) for x in f_c]))
                except Exception:
                    exact = math.inf
                if math.isfinite(exact) and exact < best_score - 1e-14:
                    best_seq = f_c.copy()
                    best_score = exact

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0, 1.0, 1.0]
        return [float(x) for x in fallback]

    best_score = min(best_score, final_sc)

    # Ensure valid output
    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    if not result:
        result = [1.0]
    if any(math.isnan(x) or math.isinf(x) for x in result):
        result = [float(x) for x in fallback]
    if sum(result) < 0.01:
        result = [float(x) for x in fallback]

    return result
```
