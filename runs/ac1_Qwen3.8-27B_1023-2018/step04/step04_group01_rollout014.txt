```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv(a):
    """Full convolution of a with itself."""
    n = len(a)
    sz = next_fast_len(2*n-1)
    F = rfft(a, sz)
    c = irfft(F*F, sz)[:2*n-1]
    np.maximum(c, 0, out=c)  # clamp negative to 0 (numerical)
    return c

def _score_normalized(f):
    """Score for sum-normalized f (sum=1): 2n * max(f*f)."""
    n = len(f)
    b = _conv(f)
    M = b.max()
    return 2.0 * n * M, M, b

def _lp_direction(f, M_target, time_limit=15):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M_target for k=0..2n-2, g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    # Build A matrix: A[k, j] = f[k-j] if 0<=k-j<n else 0
    # This is a banded Toeplitz-like matrix
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M_target)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)]*n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0 and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def _gradient_exact(f, b, M, active_thresh=1e-10):
    """Exact subgradient of max(b) where b = f*f."""
    n = len(f)
    active = np.where(b >= M - active_thresh)[0]
    p = np.zeros(2*n-1)
    p[active] = 1.0
    # grad[i] = 2 * sum_j f[j] * p[i+j]
    # = 2 * (f * p_rev)[2n-2-i]
    p_rev = p[::-1]
    conv = np.convolve(f, p_rev, mode='full')
    grad = 2.0 * conv[2*n-2 : 2*n-2-(n-1) : -1]
    return grad, len(active)

def _project_simplex(v):
    """Project v onto {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
