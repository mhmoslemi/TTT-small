        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = 2.0 * f[k - i]
            if v > 1e-15:
                rows.append(k)
                cols.append(i)
                vals.append(v)
        # M column
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))
    b_ub = ff  # 2*(f*u)[k] - M <= (f*f)[k]
    
    # Equality: sum(u) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M
    c = np.zeros(n + 1)
    c[n] = 1.0  # M coefficient
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': 10.0, 'presolve': True})
    
    if res.status == 0 and res.x is not None:
        u = res.x[:n]
        u = np.clip(u, 0, None)
        s = u.sum()
        if s > 1e-10:
            u = u / s  # renormalize (LP might have small numerical error)
        return u, res.x[n]
    return None, None

    s0 = f.sum()

        return f0.copy(), math.inf
