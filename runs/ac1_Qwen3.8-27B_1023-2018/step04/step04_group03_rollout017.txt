So score = 2n * max(conv(a_unnorm, a_unnorm)) / (S_new^2 * S_new^2)... wait no.

Let me define: a_old has sum S. We change a[i] to x, getting a_temp with sum S_new = S - a_old[i] + x. Then a_new = a_temp / S_new.

conv(a_new, a_new)[k] = (1/S_new^2) * conv(a_temp, a_temp)[k]

score = 2n * max_k conv(a_new, a_new)[k] / (sum a_new)^2
     = 2n * max_k conv(a_temp, a_temp)[k] / S_new^2 / 1
     = 2n * max_k conv(a_temp, a_temp)[k] / S_new^2

And conv(a_temp, a_temp)[k] = conv(a_old, a_old)[k] + 2*(x - a_old[i]) * a_old[k-i] (for k ≠ 2i)
or + (x^2 - a_old[i]^2) for k = 2i.

This is getting complicated. Let me just do a numerical 1D search for each coordinate.

For coordinate i, try x in a log-spacing around the current value:
x ∈ {0, orig*0.1, orig*0.3, orig*0.5, orig*0.7, orig*0.85, orig*0.95, orig, orig*1.05, orig*1.15, orig*1.3, orig*1.5, orig*3, orig*5, orig*10}

Evaluate the score for each, keep the best. Then do a local refinement.

This costs O(15 * n log n) per coordinate, O(15 * n^2 log n) per pass. For n=416, that's about 15 * 416^2 * 9 ≈ 23 million operations per pass. Should be fast enough for many passes.

Actually, each _score call involves an FFT of size ~832, which is very fast. Let me just go with this approach.

Let me now write the final program, focusing on efficiency and effectiveness:

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
```


```python
def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(416, dtype=np.float64)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    return p


def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _lp_direction(f, n, b, peak_frac=0.02, max_peaks=60, timeout=3.5):
    M = float(b.max())
    if M < 1e-18:
        return None
    threshold = M * (1.0 - peak_frac)
    peak_idx = np.where(b >= threshold)[0]
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(b)])
    if len(peak_idx) > max_peaks:
        pv = b[peak_idx]
        top = np.argsort(-pv)[:max_peaks]
        peak_idx = peak_idx[top]

    rows, cols, vals = [], [], []
    for idx, k in enumerate(peak_idx):
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        for i in range(i0, i1):
            v = f[k - i]
            if v > 1e-20:
                rows.append(idx)
                cols.append(i)
                vals.append(v)
    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    c_vec = -np.ones(n)
    b_ub = np.full(len(peak_idx), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _opt_lp(f0, stop_time, seed, peak_frac=0.02):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        return f, math.inf
    f = f / fs
    best_f = f.copy()
    best_sc = _score(f)

    t = 0.12
    stall = 0
    max_stall = 50
    it = 0

    while time.monotonic() < stop_time - 3.0 and it < 2000:
        it += 1
        b = _conv(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, n, b, peak_frac=peak_frac, max_peaks=50, timeout=3.0)
        if g is None:
            stall += 2
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                fs2 = f.sum()
                f = f / fs2 if fs2 > 1e-10 else best_f.copy()
                stall = 0
                t = 0.12
                max_stall = min(max_stall + 25, 200)
            continue

        g = g / g.sum()
        improved = False
        t_trials = sorted(set(round(max(0.01, min(0.9, tt)), 4) for tt in
                              [t, t*0.5, t*1.5, t*0.25, t*2, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3]),
                         key=lambda x: -abs(x - t))

        for tt in t_trials:
            fn = (1.0 - tt) * f + tt * g
            fn = np.clip(fn, 0, None)
            fn_s = fn.sum()
            if fn_s < 1e-10:
                continue
            fn = fn / fn_s
            sc = _score(fn)
            if sc < best_sc - 1e-16:
                f = fn
                best_f = f.copy()
                best_sc = sc
                t = tt
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            t = max(0.02, t * 0.85)
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.025, n), 0, None)
                fs2 = f.sum()
                f = f / fs2 if fs2 > 1e-10 else best_f.copy()
                stall = 0
                t = 0.12
                max_stall = min(max_stall + 25, 250)

    return best_f, best_sc


def _opt_mult(f0, stop_time, seed):
    """Multiplicative weight / exponential gradient method."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    a = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-12, None)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.3
    stall = 0
    max_stall = 80

    while time.monotonic() < stop_time - 3.0:
        b = _conv(a)
        M = float(b.max())
        if M < 1e-18:
            break
        thr = M * 0.98
        pk = np.where(b >= thr)[0]
        if len(pk) == 0:
            pk = np.array([np.argmax(b)])

        grad = np.zeros(n)
        for k in pk:
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            for i in range(i0, i1):
                grad[i] += a[k - i]
        grad /= len(pk)
        grad *= 2.0

        gs = grad.max()
        if gs < 1e-15:
            break
        eff = lr / gs

        expo = -eff * grad
        expo -= expo.max()
        w = np.exp(expo)
        a_new = (a + 1e-15) * w
        s2 = a_new.sum()
        if s2 < 1e-15:
            break
        a_new = a_new / s2

        sc = _score(a_new)
        if sc < best_sc - 1e-14:
            best_a = a_new.copy()
            best_sc = sc
            stall = 0
            lr *= 0.995
        else:
            a_d = 0.5 * a + 0.5 * a_new
            sc_d = _score(a_d)
            if sc_d < best_sc - 1e-14:
                a_new = a_d
                best_a = a_new.copy()
                best_sc = sc_d
                stall = 0
                lr *= 0.97
            else:
                stall += 1
                lr *= 0.93
                if stall > max_stall:
                    a = np.clip(a + rng.normal(0, 0.008, n), 1e-12, None)
                    a = a / a.sum()
                    stall = 0
                    lr = 0.15
                    max_stall += 30
                    if max_stall > 300:
                        break

        a = a_new

    return best_a, best_sc


def _coord_polish(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    a = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    s = a.sum()
    if s < 1e-10:
        return a, math.inf
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_imp = 0
    max_no_imp = 4

    while time.monotonic() < stop_time - 3.0:
        if no_imp >= max_no_imp:
            break
        improved = False
        order = list(rng.permutation(n))

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = a[i]

            if orig < 1e-14:
                a[i] = 0.005 / n
                sc = _score(a)
                if sc < best_sc - 1e-16:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = 0.01 / n
                    sc = _score(a)
                    if sc < best_sc - 1e-16:
                        best_sc = sc
                        best_a = a.copy()
                        improved = True
                    else:
                        a[i] = 0.0
                continue

            cands = [orig * m for m in (0.3, 0.5, 0.7, 0.8, 0.9, 0.95,
                                         1.05, 1.1, 1.2, 1.4, 1.7, 2.5, 0.0)]
            best_loc = best_sc
            best_v = orig
            for v in cands:
                a[i] = v
                ns = a.sum()
                if ns < 1e-10:
                    a[i] = orig
                    continue
                sc = _score(a)
                if sc < best_loc - 1e-17:
                    best_loc = sc
                    best_v = v
            if best_v != orig:
                a[i] = best_v
                orig2 = best_v
                for v2 in [orig2 * m for m in (0.9, 1.1, 0.85, 1.15, 0.95, 1.05)]:
                    a[i] = v2
                    ns = a.sum()
                    if ns < 1e-10:
                        a[i] = orig2
                        continue
                    sc = _score(a)
                    if sc < best_loc - 1e-17:
                        best_loc = sc
                        best_v = v2
                    else:
                        a[i] = orig2
                a[i] = best_v
            else:
                a[i] = orig

            if best_sc != orig or True:
                if _score(a) < best_sc - 1e-16:
                    pass
            sc_now = _score(a)
            if sc_now < best_sc - 1e-16:
                best_sc = sc_now
                best_a = a.copy()
                improved = True

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 18.0

    parent = _get_parent()
    n = len(parent)

    best_seq = parent.copy()
    best_score = _score(best_seq)

    def _update(cand):
        nonlocal best_seq, best_score
        if cand is None or len(cand) == 0:
            return
        cand = np.clip(np.asarray(cand, dtype=np.float64), 0, 1000)
        if not np.all(np.isfinite(cand)) or cand.sum() < 0.01:
            return
        sc = _score(cand)
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_seq = cand.copy()
            best_score = sc

    # Phase 1: LP optimization (50% time)
    p1_end = t0 + budget * 0.48
    rng = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    inits = [parent.copy()]
    sig = 0.08 * max(1.0, float(parent.mean()))
    inits.append(np.clip(parent + rng.normal(0, sig, n), 0, None))
    inits.append(np.clip(parent * (1.0 + 0.1 * rng.normal(0, 1, n)), 0, None))
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / n)))

    sup = np.ones(n) * 0.25
    for fr in (1, 2, 3, 5, 7, 11, 13):
        ph = rng.uniform(0, 2 * np.pi)
        sup += 0.1 * np.cos(2.0 * np.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.02)

    center = (n - 1) / 2.0
    for w in (n * 0.15, n * 0.25, n * 0.4):
        inits.append(np.exp(-0.5 * ((i_arr - center) / w) ** 2))

    two_c = n * 0.28
    two_w = n * 0.1
    inits.append(np.exp(-0.5 * ((i_arr - two_c) / two_w) ** 2) +
                 np.exp(-0.5 * ((i_arr - (n - two_c - 1)) / two_w) ** 2))

    # Deduplicate and normalize inits
    inits_dedup = [inits[0]]
    for ii in inits[1:]:
        s = ii.sum()
        if s < 0.1:
            continue
        inits_dedup.append(ii / s)

    time_per_init = max(8.0, (p1_end - t0 - 5.0) / len(inits_dedup))
    for idx, init in enumerate(inits_dedup):
        if time.monotonic() > p1_end - 6.0:
            break
        stop = min(p1_end - 3.0, time.monotonic() + time_per_init)
        if time.monotonic() > stop - 1.0:
            continue
        try:
            res, sc = _opt_lp(init, stop, seed + 1000 * (idx + 1), peak_frac=0.02)
            _update(res)
        except Exception:
            pass

    # Second LP pass on current best
    if time.monotonic() < p1_end - 12.0:
        try:
            res, sc = _opt_lp(best_seq, p1_end - 2.0, seed + 777777, peak_frac=0.015)
            _update(res)
        except Exception:
            pass

    # Phase 2: Multiplicative weight update (15% time)
    p2_end = t0 + budget * 0.63
    if time.monotonic() < p2_end - 8.0:
        try:
            res, sc = _opt_mult(best_seq, p2_end - 3.0, seed + 33333)
            _update(res)
        except Exception:
            pass

    # Phase 3: LP again on result (12% time)
    p3_end = t0 + budget * 0.75
    if time.monotonic() < p3_end - 10.0:
        try:
            res, sc = _opt_lp(best_seq, p3_end - 3.0, seed + 555555, peak_frac=0.01)
            _update(res)
        except Exception:
            pass

    # Phase 4: Coordinate descent polish (20% time)
    p4_end = deadline
    if time.monotonic() < p4_end - 10.0:
        try:
            res, sc = _coord_polish(best_seq, p4_end - 5.0, seed + 99999)
            _update(res)
        except Exception:
            pass

    # Final LP quick pass if time
    if time.monotonic() < p4_end - 15.0:
        try:
            res, sc = _opt_lp(best_seq, p4_end - 5.0, seed + 111111, peak_frac=0.01)
            _update(res)
        except Exception:
            pass

    # Another short polish
    rem = p4_end - time.monotonic()
    if rem > 12.0:
        try:
            res, sc = _coord_polish(best_seq, p4_end - 5.0, seed + 222222)
            _update(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_sc = float(evaluate_sequence(final_list))
    except Exception:
        final_sc = math.inf
        final_list = [float(x) for x in parent]

    if not math.isfinite(final_sc):
        try:
            final_list = [float(max(0.0, min(1000.0, x))) for x in parent]
            final_sc = float(evaluate_sequence(final_list))
        except Exception:
            final_list = [1.0]
            final_sc = float(evaluate_sequence(final_list))

    if not math.isfinite(final_sc):
        return [1.0]

    return final_list
```
