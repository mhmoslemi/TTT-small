    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _subgrad(a):
    n = len(a)
    b = _fft_conv(a)
    m = float(np.max(b))
    if m < 1e-18:
        return np.zeros(n), m
    eps = max(1e-14, 1e-10 * m)
    active = np.flatnonzero(b >= m - eps)
    grad = np.zeros(n)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        idx = k - np.arange(i_lo, i_hi + 1)
        grad[i_lo:i_hi + 1] += 2.0 * a[idx]
    grad /= len(active)
    return grad, m


def _project_simplex(v):
    n = len(v)
    sv = np.sum(v)
    if sv < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _lp_direction(f, M, threshold=0.80, timeout=2.0):
    n = len(f)
    ff = _fft_conv(f)
    active_k = np.flatnonzero(ff > threshold * M)
    if len(active_k) == 0:
        active_k = np.flatnonzero(ff >= ff.max() - 1e-12)
    if len(active_k) == 0:
        active_k = np.array([int(np.argmax(ff))])

    A = np.zeros((len(active_k), n))
    for idx, k in enumerate(active_k):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        j = np.arange(i_lo, i_hi + 1)
        A[idx, j] = f[k - j]

    c = -np.ones(n)
    b_ub = np.full(len(active_k), M + 1e-12)
    bounds = [(0.0, float(n))] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': timeout})
        if res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0.0, None)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None


def _lp_optimize(f0, stop_time, seed, max_iters=100000):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0.0, None)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(len(f)) / len(f)
        s = 1.0
    else:
        f = f / s
    n = len(f)
    f = f + 1e-12
    f = f / f.sum()

    best_f = f.copy()
    best_sc = _score(f)
    t_param = 0.3
    stall = 0
    max_stall = 300
    threshold = 0.75

    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break

        b = _fft_conv(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, M, threshold=threshold, timeout=1.5)
        if g is None:
            g = _lp_direction(f, M, threshold=0.0, timeout=2.5)
            if g is None:
                stall += 5
                if stall > max_stall:
                    f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                    f = f + 1e-12
                    f = f / f.sum()
                    t_param = 0.3
                    stall = 0
                    max_stall = min(800, max_stall + 100)
                    if max_stall > 800:
                        break
                    continue
                continue

        gs = g.sum()
        if gs < 1e-12:
            stall += 3
            continue
        g = g / gs

        dist = np.linalg.norm(g - f) / max(1.0, np.linalg.norm(f))
        if dist < 1e-9:
            stall += 3
            if stall > 50:
                break
            continue

        best_t = t_param
        best_sc_try = best_sc
        candidates_t = sorted(set([
            t_param, t_param * 0.5, t_param * 1.4,
            0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9
        ]))
        for tt in candidates_t:
            tt = min(0.99, max(0.005, tt))
            fn = (1 - tt) * f + tt * g
            fn = np.clip(fn, 0, None)
            sfn = fn.sum()
            if sfn < 0.01:
                continue
            fn = fn / sfn
            sc = _score(fn)
            if sc < best_sc_try - 1e-14:
                best_t = tt
                best_sc_try = sc

        if best_sc_try < best_sc - 1e-13:
            fn = (1 - best_t) * f + best_t * g
            fn = np.clip(fn, 0, None)
            fn = fn + 1e-12
            fn = fn / fn.sum()
            f = fn
            best_f = f.copy()
            best_sc = _score(f)
            t_param = min(0.95, best_t * 1.08)
            threshold = min(0.95, threshold * 1.02)
            stall = 0
        else:
            t_param = max(0.02, t_param * 0.5)
            stall += 1
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
                f = f + 1e-12
                f = f / f.sum()
                t_param = 0.3
                threshold = 0.7
                stall = 0
                max_stall = min(800, max_stall + 100)

    return best_f, best_sc


def _subgrad_optimize(f0, stop_time, seed, max_iters=200000):
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = np.clip(f0.copy(), 0, None)
    f = f / max(f.sum(), 1e-10)

    best_f = f.copy()
    best_sc = _score(f)
    step = 0.1
    momentum = np.zeros(n)
    mu = 0.75
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break

        grad, M = _subgrad(f)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        momentum = mu * momentum - step * grad / gnorm
        fn = np.clip(f + momentum, 0, None)
        s = fn.sum()
        if s < 0.01:
            step *= 0.5
            momentum *= 0.3
            continue
        fn = fn / s
        sc = _score(fn)

        if sc < best_sc - 1e-14:
            f = fn
            best_f = f.copy()
            best_sc = sc
            step = min(1.0, step * 1.02)
            no_imp = 0
        else:
            step *= 0.93
            no_imp += 1
            if no_imp > 200:
                f = np.clip(best_f + rng.normal(0, 0.04, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                momentum = np.zeros(n)
                step = 0.1
                no_imp = 0
            if no_imp > 5000:
                break

    return best_f, best_sc


def _coord_descent(f0, stop_time, seed, max_epochs=300):
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0, None)
    n = len(f)
    best_sc = _score(f)
    best_f = f.copy()

    for epoch in range(max_epochs):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            if orig < 1e-14:
                continue
            best_v = orig
            best_vsc = best_sc
            for mult in [0.3, 0.6, 0.8, 1.2, 1.5, 2.0, 0.0]:
                f[i] = orig * mult
                s = f.sum()
                if s < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_vsc - 1e-15:
                    best_v = f[i]
                    best_vsc = sc
            f[i] = best_v
            if best_vsc < best_sc - 1e-14:
                best_sc = best_vsc
                best_f = f.copy()
                improved = True

        # Two-coordinate steps
        n_two = min(50, n // 4)
        for _ in range(n_two):
            if time.monotonic() > stop_time:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-14 and oj < 1e-14:
                continue
            for mi in [0.4, 0.7, 1.4, 0.0]:
                for mj in [0.4, 0.7, 1.4, 0.0]:
                    if mi == 1.0 and mj == 1.0:
                        continue
                    f[i] = oi * mi if oi > 1e-14 else 0.0
                    f[j] = oj * mj if oj > 1e-14 else 0.0
                    s = f.sum()
                    if s < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        best_f = f.copy()
                        improved = True
            f[i], f[j] = oi, oj

        if not improved:
            f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
            f = f / max(f.sum(), 1e-10)
            if epoch > 50:
                break

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 25.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except (NameError, TypeError, ValueError):
        parent = np.ones(414)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    if parent.size < 2 or parent.sum() < 0.01:
        parent = np.ones(414)

    n = len(parent)
    f0 = parent / max(parent.sum(), 1e-10)

    best_seq = parent.copy()
    best_score = _score(parent)

    def try_update(a, sc):
        nonlocal best_seq, best_score
        if not math.isfinite(sc) or sc >= best_score - 1e-12:
            return
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size < 1:
            return
        a = np.clip(a, 0, 1000.0)
        a = np.where(np.isfinite(a), a, 0.0)
        if a.sum() < 0.01:
            return
        best_seq = a.copy()
        best_score = sc

    # Phase 1: Main LP optimization
    p1_end = t0 + budget * 0.45
    res1, sc1 = _lp_optimize(f0, p1_end, seed)
    if math.isfinite(sc1):
        try_update(res1 * parent.sum(), sc1)

    # Phase 1b: LP with symmetric init
    if time.monotonic() < t0 + budget * 0.55:
        f_sym = 0.5 * (f0 + f0[::-1])
        f_sym = f_sym / max(f_sym.sum(), 1e-10)
        stop_sym = min(t0 + budget * 0.55, time.monotonic() + 60)
        res_s, sc_s = _lp_optimize(f_sym, stop_sym, seed + 100)
        if math.isfinite(sc_s):
            try_update(res_s * parent.sum(), sc_s)

    # Phase 1c: LP with perturbed init
    if time.monotonic() < t0 + budget * 0.62:
        rng_p = np.random.default_rng(seed + 200)
        f_pert = np.clip(f0 + rng_p.normal(0, 0.15, n), 0, None)
        f_pert = f_pert / max(f_pert.sum(), 1e-10)
        stop_p = min(t0 + budget * 0.62, time.monotonic() + 50)
        res_p, sc_p = _lp_optimize(f_pert, stop_p, seed + 300)
        if math.isfinite(sc_p):
            try_update(res_p * parent.sum(), sc_p)

    # Phase 2: Subgradient refinement
    if time.monotonic() < t0 + budget * 0.75:
        p2_end = t0 + budget * 0.75
        f_cur = best_seq / max(best_seq.sum(), 1e-10)
        res_sg, sc_sg = _subgrad_optimize(f_cur, p2_end, seed + 400)
        if math.isfinite(sc_sg):
            try_update(res_sg * best_seq.sum(), sc_sg)

    # Phase 3: Coordinate descent
    if time.monotonic() < t0 + budget * 0.87:
        p3_end = t0 + budget * 0.87
        res_cd, sc_cd = _coord_descent(best_seq, p3_end, seed + 500, max_epochs=200)
        if math.isfinite(sc_cd):
            try_update(res_cd, sc_cd)

    # Phase 4: Length exploration
    if time.monotonic() < deadline - 20:
        alt_lengths = sorted(set([n - 6, n - 3, n - 1, n + 1, n + 3, n + 6, n + 12]))
        alt_lengths = [L for L in alt_lengths if L >= 8]
        remaining = deadline - 20 - time.monotonic()
        time_per = max(3.0, min(25.0, remaining / max(1, len(alt_lengths))))
        for L in alt_lengths:
            if time.monotonic() > deadline - 20 or time_per < 3:
                break
            init_L = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n), f0)
            init_L = init_L / max(init_L.sum(), 1e-10)
            stop_L = time.monotonic() + time_per
            try:
                rL, sL = _lp_optimize(init_L, stop_L, seed + 1000 + L, max_iters=10000)
                if math.isfinite(sL):
                    scale = best_seq.sum() / max(rL.sum(), 1e-10)
                    try_update(rL * scale, sL)
            except Exception:
                pass

    # Phase 5: Final LP on best
    if time.monotonic() < deadline - 15:
        f_final = best_seq / max(best_seq.sum(), 1e-10)
        res_f, sc_f = _lp_optimize(f_final, deadline - 15, seed + 5000, max_iters=50000)
        if math.isfinite(sc_f):
            try_update(res_f * best_seq.sum(), sc_f)

    # Phase 6: Final coordinate polish
    if time.monotonic() < deadline - 5:
        try:
            res_polish, sc_polish = _coord_descent(best_seq, deadline - 5, seed + 9000, max_epochs=80)
            if math.isfinite(sc_polish):
                try_update(res_polish, sc_polish)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            return [1.0]
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    return [float(x) for x in best_seq]
