import numpy as np

from scipy.optimize import linprog


def _fe(a):
    """Fast internal score matching the exact evaluator (for tracking only)."""
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if not np.isfinite(s) or s < 0.01:
            return float("inf")
        c = np.convolve(a, a)
        m = float(c.max())
        return float(2.0 * a.size * m / (s * s))
    except Exception:
        return float("inf")

def _cm(n, a):
    """(2n-1) x n convolution matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    m = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[m] = a[d[m]]
    return A


def _lp(A, M, n, tl=4.0):
    """max 1^T g  s.t.  A g <= M,  g >= 0.  Returns g (clipped >=0) or None."""
    if M <= 0.0:
        return None
    try:
        r = linprog(
            -np.ones(n),
            A_ub=A,
            b_ub=np.full(A.shape[0], float(M)),
            bounds=[(0.0, None)] * n,
            method="highs",
            options={
                "time_limit": tl,
                "dual_feasibility_tolerance": 1e-13,
                "primal_feasibility_tolerance": 1e-13,
            },
        )
        if r.status != 0 or r.x is None:
            return None
        return np.clip(r.x, 0.0, None)
    except Exception:
        return None


def _blend(a, g, Mcur):
    """Best blend t in [0,1] of x=(1-t)a+t*g that maximally cuts max-conv."""
    n = a.size
    bt, bv = 0.0, Mcur
    for t in np.array([0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]):
        x = (1.0 - t) * a + t * g
        s = x.sum()
        if s < 1e-15:
            continue
        x = x / s
        v = float(np.convolve(x, x).max())
        if v < bv:
            bv, bt = v, t
    if bt == 0.0:
        return None, Mcur
    lo, hi = max(0.0, bt - 0.07), min(1.0, bt + 0.07)
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        x = (1.0 - mid) * a + mid * g
        s = x.sum()
        if s < 1e-15:
            hi = mid
            continue
        x = x / s
        v = float(np.convolve(x, x).max())

            bv, bt, hi = v, mid, mid
        else:
            lo = mid
        if hi - lo < 1e-11:
            break
    x = np.maximum((1.0 - bt) * a + bt * g, 0.0)
    x = x / max(x.sum(), 1e-15)
    return x, bv

def _inits(parent, n):

    out = [pn.copy()]

        v = np.maximum(1.0 + 0.35 * np.cos(2.0 * np.pi * fr * x / max(n - 1, 1)), 0.01)

            v = v / max(v.sum(), 1e-15)

    out.append(pn[::-1].copy())
    return out
