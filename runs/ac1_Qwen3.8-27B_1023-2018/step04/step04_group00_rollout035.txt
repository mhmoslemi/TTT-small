def lp_step(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    # Build constraint matrix
    # (f*g)[k] <= M for k = 0,...,2n-2
    constraints = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        # (f*g)[k] = sum_{j=j_start}^{j_end} f[j] * g[k-j]
        # The index into g is k-j, which ranges from k-j_end to k-j_start

        # = max(0, k-n+1) to min(k, n-1)
        # Wait, let me just build it directly
        expr = sum(f[j] * g[k-j] for j in range(j_start, j_end + 1))
        constraints.append(expr <= M)

    prob = cp.Problem(objective, constraints)

def _conv_self(a):

    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)

    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    return float(2.0 * n * m / (s * s))

def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0/n)
    m = src.size
    if m == n:
        out = src.copy()
    elif m == 1:
        return np.full(n, float(src[0])/n)
    else:
        xp = (np.arange(m) + 0.5) / m
        fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s > 0:
        out /= s
    else:
        out = np.full(n, 1.0/n)
    return out

    Given f (normalized, sum=1) and target M = max(f*f),

    if res.status == 0 and res.x is not None:

        g = np.maximum(g, 0.0)
        s = float(g.sum())
        if s > 1e-12:
            g /= s
            return g
    return None

    n = a.size
    if n == 0:
        return np.zeros(0)

    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F*F, size)[:L]
    np.maximum(c, 0.0, out=c)

    s = float(a.sum())
    if s < 0.01: return math.inf

    m = float(c.max())
    if not math.isfinite(m): return math.inf

        return np.full(n, 1.0/n)
    m = src.size
    if m == n: return src.copy()
    xp = (np.arange(m)+0.5)/m
    out = np.interp((np.arange(n)+0.5)/n, xp, src)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s > 0: out /= s
    else: out = np.full(n, 1.0/n)
    return out

# LP iteration
def _lp_step(f, M, time_limit=30):
    n = len(f)
    L = 2*n - 1
    # Build A: A[k,j] = f[j] for j <= k <= j+n-1, k < L
    k_idx = np.arange(L)[:, None]
    j_idx = np.arange(n)[None, :]
    mask = (j_idx <= k_idx) & (k_idx < j_idx + n)
    A = np.where(mask, f[None, :], 0.0)
    
    c = -np.ones(n)
    b_ub = np.full(L, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.status == 0 and res.x is not None:
        g = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
        s = float(g.sum())
        if s > 1e-12:
            return g/s
    return None

# Coordinate descent
def _coord_polish(p, stop_time, consider=None, seed=42):
    n = len(p)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300: return p
    p = p/s
    S = 1.0
    Ltot = 2*n-1
    c = _conv_self(p)
    cur_score = 2.0*n*float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    
    G = 40  # grid points
    
    flat = 0
    sweep = 0
    while sweep < 50000 and time.monotonic() < stop_time - 2:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5: break
            if S <= 1e-300 or S > 1.25 or S < 0.75:
                inv = 1.0/S
                p *= inv; c *= inv*inv
                S = 1.0
            S = float(S)
            v = float(p[i])
            start = i
            end = min(i+n, Ltot)
            Ls = end-start
            if Ls <= 0: continue
            base = c[start:end]
            infl = 2.0*p[:Ls]
            
            max_const = 0.0
            if start > 0: max_const = max(max_const, float(c[:start].max()))
            if end < Ltot: max_const = max(max_const, float(c[end:].max()))
            
            x_hi = max(20.0*v, 60.0*S/n, 0.05*S, 1e-12)
            x_hi = min(x_hi, 10.0*S)
            if not math.isfinite(x_hi) or x_hi <= 0: x_hi = S/n
            
            pts = np.linspace(0.0, x_hi, G+1)
            if 0 <= v <= x_hi:
                pts = np.concatenate([pts, [v]])
            pts = np.unique(pts)
            
            def eval_xs(xs):
                xs = np.asarray(xs)
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf)
                valid = denom > 1e-15
                if not np.any(valid): return vals
                aff = base[None,:] + w[:,None]*infl[None,:]
                maff = np.max(aff, axis=1)
                dval = base[i-start] + w*infl[i-start] + w*w
                maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = 2.0*n*mv/(dv*dv)
                return vals
            
            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)): continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])
            
            # Fine search
            dw = x_hi/G
            lo = max(0.0, best_x - 2*dw)
            hi = min(x_hi, best_x + 2*dw)
            if hi > lo:
                fine = np.linspace(lo, hi, 24)
                if 0 <= best_x <= x_hi:
                    fine = np.concatenate([fine, [best_x]])
                fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    if float(vals2[imin2]) < best_val:
                        best_val = float(vals2[imin2])
                        best_x = float(fine[imin2])
            
            if math.isfinite(best_val) and best_val < cur_score - 1e-14:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S += w
                    c[start:end] += w*infl
                    di = 2*(i-start)  # offset in c
                    if 0 <= i-start < Ls:  # always true since i-start = 0
                        # The diagonal term: c[2i] += w^2
                        diag_c = 2*i
                        if 0 <= diag_c < Ltot:
                            c[diag_c] += w*w
                    cur_score = best_val
                    improved = True
        
        # Normalize
        if S > 1e-300:
            inv = 1.0/S
            p *= inv; c *= inv*inv
            S = 1.0
        c = _conv_self(p)
        new_score = 2.0*n*float(c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
            if consider: consider(best_p)
        
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 10:
                    p = p * np.exp(rng.normal(0, 0.03, n))
                    p = np.maximum(p, 0)
                    ps = float(p.sum())
                    if ps > 0: p /= ps
                    c = _conv_self(p)
                    S = 1.0
                    cur_score = 2.0*n*float(c.max())
                    if cur_score < best_score:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider: consider(best_p)
                    flat = 0
                else:
                    break
        sweep += 1
    
    p = np.maximum(p, 0)
    ps = float(p.sum())
    if ps > 0: p /= ps
    return best_p
