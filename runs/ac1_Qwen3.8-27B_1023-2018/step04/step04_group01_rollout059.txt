```python
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, math.inf
    f = f / S
    S = 1.0
    conv = _conv_full(f)
    best_score = 2.0 * n * conv.max()
    best_f = f.copy()
    rng = np.random.default_rng(seed)
    m = 2 * n - 1  # conv size

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break

            v = f[i]
            start = i
            end = min(i + n, m)
            region_len = end - start
            quad_pos = 2 * i

            if region_len <= 0:
                continue

            # Precompute for this coordinate
            infl = 2.0 * f[:region_len]  # linear influence in affected region
            if start > 0:
                max_left = conv[:start].max()
            else:
                max_left = 0.0
            if end < m:
                max_right = conv[end:].max()
            else:
                max_right = 0.0
            max_unaff = max(max_left, max_right)

            # Search range
            x_max = max(5.0 * v + 0.01, 0.1)
            x_max = min(x_max, 50.0)
            
            n_pts = 60
            x_grid = np.linspace(0.0, x_max, n_pts)
            deltas = x_grid - v
            S_new = S + deltas
            valid = S_new > 0.01

            if not valid.any():
                continue

            # Vectorized computation
            # conv in affected region: conv[start:end] + delta * infl
            base = conv[start:end]  # shape (region_len,)
            # Add quadratic term at quad_pos if in range
            has_quad = (start <= quad_pos < end)
            
            conv_aff = base[:, None] + deltas[None, :] * infl[:, None]  # (region_len, n_pts)
            if has_quad:
                conv_aff[quad_pos - start, :] += deltas**2
            
            max_aff = conv_aff.max(axis=0)  # (n_pts,)
            new_max = np.maximum(max_aff, max_unaff)
            
            with np.errstate(divide='ignore', invalid='ignore'):
                obj = 2.0 * n * new_max / (S_new * S_new)
            obj[~valid] = math.inf

            best_idx = int(obj.argmin())
            best_x = x_grid[best_idx]
            best_obj = obj[best_idx]

            if best_x != v and best_obj < best_score - 1e-10:
                # Fine search
                dw = x_max / n_pts
                x_lo = max(0.0, best_x - 2 * dw)
                x_hi = min(x_max, best_x + 2 * dw)
                x_fine = np.linspace(x_lo, x_hi, 30)
                deltas_f = x_fine - v
                S_nf = S + deltas_f
                valid_f = S_nf > 0.01
                if valid_f.any():
                    conv_aff_f = base[:, None] + deltas_f[None, :] * infl[:, None]
                    if has_quad:
                        conv_aff_f[quad_pos - start, :] += deltas_f**2
                    max_aff_f = conv_aff_f.max(axis=0)
                    new_max_f = np.maximum(max_aff_f, max_unaff)
                    with np.errstate(divide='ignore', invalid='ignore'):
                        obj_f = 2.0 * n * new_max_f / (S_nf * S_nf)
                    obj_f[~valid_f] = math.inf
                    best_f_idx = int(obj_f.argmin())
                    if obj_f[best_f_idx] < best_obj:
                        best_x = x_fine[best_f_idx]
                        best_obj = obj_f[best_f_idx]

                # Apply
                delta = best_x - v
                f[i] = best_x
                S += delta
                conv[start:end] += delta * infl
                if 0 <= quad_pos < m:
                    conv[quad_pos] += delta * delta
                best_score = best_obj
                best_f = f.copy()
                improved = True

        if not improved and sweep > 5:
            break

    f = np.clip(best_f, 0, 1000)
    final_score = _score(f)
    return f, final_score
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(400, dtype=np.float64)
    if p.size == 0:
        p = np.ones(400, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(400, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    """A[k, i] = f[k-i] if 0 <= k-i < n and 0 <= k < 2n-1, else 0."""
    m = 2 * n - 1
    A = np.zeros((m, n))
    for i in range(n):
        ln = min(n, m - i)
        A[i:i + ln, i] = f[:ln]
    return A


def _lp_direction(f, n, M, time_limit=8.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-8,
                               'primal_feasibility_tolerance': 1e-8})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g


def _lp_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs

    best_f = f.copy()
    best_score = _score(f)
    stall = 0
    max_stall = 40
    iteration = 0

    while time.monotonic() < stop_time - 5.0:
        if iteration > 2000:
            break
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g_dir = _lp_direction(f, n, M, time_limit=6.0)
        if g_dir is None:
            stall += 3
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            f /= f.sum()
            iteration += 1
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-7:
            stall += 2
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            f /= f.sum()
            iteration += 1
            continue

        # Line search (binary)
        curr_score = _score(f)
        best_t = 0.0
        best_sc = curr_score
        lo, hi = 0.0, 1.0
        for _ in range(18):
            mid = (lo + hi) / 2
            f_new = np.clip(f + mid * (g_dir - f), 0, None)
            fs_new = f_new.sum()
            if fs_new < 1e-10:
                hi = mid
                continue
            f_new /= fs_new
            sc = _score(f_new)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_t = mid
                lo = mid
            else:
                hi = mid

        if best_t > 1e-4 and best_sc < best_score - 1e-12:
            f_new = np.clip(f + best_t * (g_dir - f), 0, None)
            f_new /= f_new.sum()
            f = f_new
            best_f = f.copy()
            best_score = best_sc
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
                f /= f.sum()
                stall = 0

        iteration += 1

    return best_f, best_score

def _pair_polish(f0, stop_time, max_rounds=50, seed=9999):
    """Try random pair moves."""
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / f.sum()
    best_score = _score(f)
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        for _ in range(40):
            if time.monotonic() > stop_time - 5.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-12 and oj < 1e-12:
                continue

            best_pair = (oi, oj)
            best_pair_sc = best_score

            for mi in (0.0, 0.3, 0.6, 1.0, 1.5, 2.0, 4.0):
                ni = oi * mi if oi > 1e-12 else (0.01 if mi > 0 else 0.0)
                for mj in (0.0, 0.3, 0.6, 1.0, 1.5, 2.0):
                    nj = oj * mj if oj > 1e-12 else (0.01 if mj > 0 else 0.0)
                    f[i], f[j] = ni, nj
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_pair_sc - 1e-12:
                        best_pair_sc = sc
                        best_pair = (ni, nj)

            f[i], f[j] = best_pair
            if best_pair_sc < best_score - 1e-12:
                best_score = best_pair_sc
                best_f = f.copy()
                improved = True

        if not improved:
            break

    return best_f, best_score
```

```python
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(400, dtype=np.float64)

    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    m = 2 * n - 1
    A = np.zeros((m, n))
    for i in range(n):
        ln = min(n, m - i)
        A[i:i + ln, i] = f[:ln]
    return A


def _lp_direction(f, n, M, time_limit=8.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g


def _lp_optimize(f0, n, stop_time, seed):

def _coord_polish(f0, stop_time, max_sweeps=500, seed=777):
    n = len(f0)
    m = 2 * n - 1
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, math.inf
    f = f / S
    S = 1.0
    conv = _conv_full(f)
    best_score = 2.0 * n * conv.max()
    best_f = f.copy()
    rng = np.random.default_rng(seed)
    no_improve_sweeps = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break

            v = f[i]
            start = i
            end = min(i + n, m)
            region_len = end - start
            quad_pos = 2 * i

            if region_len <= 0:
                continue

            infl = 2.0 * f[:region_len]
            max_unaff = 0.0
            if start > 0:
                ml = float(conv[:start].max())
                if ml > max_unaff:
                    max_unaff = ml
            if end < m:
                mr = float(conv[end:].max())
                if mr > max_unaff:
                    max_unaff = mr

            # Adaptive search range
            if v > 1e-6:
                x_max = min(max(8.0 * v, 0.02), 100.0)
            else:
                x_max = 2.0

            n_pts = 50
            x_grid = np.linspace(0.0, x_max, n_pts)
            deltas = x_grid - v
            S_new = S + deltas
            valid = S_new > 0.01

            if not valid.any():
                continue

            base = conv[start:end]
            has_quad = (start <= quad_pos < end)

            conv_aff = base[:, None] + deltas[None, :] * infl[:, None]
            if has_quad:
                conv_aff[quad_pos - start, :] += deltas * deltas

            max_aff = conv_aff.max(axis=0)
            new_max = np.maximum(max_aff, max_unaff)

            with np.errstate(divide='ignore', invalid='ignore'):
                obj = 2.0 * n * new_max / (S_new * S_new)
            obj[~valid] = math.inf

            best_idx = int(np.argmin(obj))
            best_x = float(x_grid[best_idx])
            best_obj = float(obj[best_idx])

            if best_x != v and best_obj < best_score - 1e-9:
                # Fine grid
                dw = x_max / n_pts
                x_lo = max(0.0, best_x - 2.0 * dw)
                x_hi = min(x_max, best_x + 2.0 * dw)
                x_fine = np.linspace(x_lo, x_hi, 40)
                deltas_f = x_fine - v
                S_nf = S + deltas_f
                valid_f = S_nf > 0.01
                if valid_f.any():
                    conv_aff_f = base[:, None] + deltas_f[None, :] * infl[:, None]
                    if has_quad:
                        conv_aff_f[quad_pos - start, :] += deltas_f * deltas_f
                    max_aff_f = conv_aff_f.max(axis=0)
                    new_max_f = np.maximum(max_aff_f, max_unaff)
                    with np.errstate(divide='ignore', invalid='ignore'):
                        obj_f = 2.0 * n * new_max_f / (S_nf * S_nf)
                    obj_f[~valid_f] = math.inf
                    fi = int(np.argmin(obj_f))
                    if float(obj_f[fi]) < best_obj:
                        best_x = float(x_fine[fi])
                        best_obj = float(obj_f[fi])

                # Apply
                delta = best_x - v
                f[i] = best_x
                S += delta
                conv[start:end] += delta * infl
                if 0 <= quad_pos < m:
                    conv[quad_pos] += delta * delta
                best_score = best_obj
                best_f = f.copy()
                improved = True

        if improved:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps > 8:
                # Random restart perturbation
                f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
                S = float(f.sum())
                if S < 0.01:
                    f = best_f.copy()
                    S = 1.0
                else:
                    f /= S
                    S = 1.0
                    conv = _conv_full(f)
                    best_score = 2.0 * n * conv.max()
                no_improve_sweeps = 0

    f = np.clip(best_f, 0, 1000)
    final_score = _score(f)
    return f, final_score


def _pair_polish(f0, stop_time, max_rounds=40, seed=9999):
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f /= f.sum()
    best_score = _score(f)
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        n_trials = 50
        for _ in range(n_trials):
            if time.monotonic() > stop_time - 5.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-12 and oj < 1e-12:
                continue

            best_pair = (oi, oj)
            best_pair_sc = best_score

            for mi in (0.0, 0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0):
                ni = oi * mi if oi > 1e-12 else (0.005 if mi > 0 else 0.0)
                for mj in (0.0, 0.25, 0.5, 0.75, 1.0, 1.5, 2.0):
                    nj = oj * mj if oj > 1e-12 else (0.005 if mj > 0 else 0.0)
                    f[i], f[j] = ni, nj
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_pair_sc - 1e-11:
                        best_pair_sc = sc
                        best_pair = (ni, nj)

            f[i], f[j] = best_pair
            if best_pair_sc < best_score - 1e-11:
                best_score = best_pair_sc
                best_f = f.copy()
                improved = True

        if not improved:
            break

    return best_f, best_score


def _tspike_polish(f0, stop_time, seed=12345):
```
