def _as_finite_nonneg(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0: return np.array([], dtype=np.float64)

def _resample_array(a, n):
    if n <= 0: return np.array([])
    arr = _as_finite_nonneg(a)
    if arr.size == n:
       out=arr.copy()
    elif arr.size == 0:
       out=np.ones(n)
    elif arr.size == 1:
       out=np.full(n, arr[0])
    else:
       x_old=np.linspace(0.0,1.0,arr.size)
       x_new=np.linspace(0.0,1.0,n)
       out=np.interp(x_new, x_old, arr)
    out=np.maximum(out,0.0)
    s=out.sum()
    if s <=1e-14: out=np.ones(n)/n
    else: out/=s
    return out

def _convolution(a):
   arr=np.asarray(a,dtype=np.float64).ravel()
   n=arr.size
   if n == 0: return np.array([], dtype=np.float64)
   if n == 1:
      b=np.array([float(arr[0])*float(arr[0])], dtype=np.float64)
   elif n <= 256:
      b=np.convolve(arr, arr, mode='full')
      b=np.asarray(b,dtype=np.float64)
   else:
      fsize=next_fast_len(2*n-1)
      F=rfft(arr, fsize)
      b=irfft(F*F, fsize)
      need=2*n-1
      if b.size > need: b=b[:need]
      elif b.size < need: # pad?
         b=np.pad(b,(0,need-b.size))
      b=np.asarray(b,dtype=np.float64)
   np.maximum(b,0.0,out=b)
   return b

    if res is None or getattr(res,'x',None) is None: return None

    g=np.asarray(res.x,dtype=np.float64).copy()

import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _as_finite_nonneg(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return arr.copy()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    return arr


def _normalize_array(a):
    arr = _as_finite_nonneg(a)
    if arr.size == 0:
        return None
    s = float(arr.sum())
    if s <= 1e-14:
        arr = np.ones(arr.size, dtype=np.float64)
        s = float(arr.size)
    arr /= s
    return arr


def _resample_array(a, n):
    if n <= 0:
        return np.array([], dtype=np.float64)
    arr = _as_finite_nonneg(a)
    if arr.size == 0:
        out = np.ones(n, dtype=np.float64)
    elif arr.size == n:
        out = arr.copy()
    elif arr.size == 1:
        out = np.full(n, float(arr[0]), dtype=np.float64)
    else:
        x_old = np.linspace(0.0, 1.0, arr.size)
        x_new = np.linspace(0.0, 1.0, n)
        out = np.interp(x_new, x_old, arr)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-14:
        out = np.ones(n, dtype=np.float64) / float(n)
    else:
        out /= s
    return out


def _convolution(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    n = int(arr.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        b = np.array([float(arr[0]) * float(arr[0])], dtype=np.float64)
    elif n <= 256:
        b = np.asarray(np.convolve(arr, arr, mode="full"), dtype=np.float64)
    else:
        need = 2 * n - 1
        fsize = next_fast_len(need)
        F = rfft(arr, fsize)
        b = irfft(F * F, fsize)
        if b.size > need:
            b = b[:need]
        elif b.size < need:
            b = np.pad(b, (0, need - b.size))
        b = np.asarray(b, dtype=np.float64)
    np.maximum(b, 0.0, out=b)
    return b


def _score(a):
    arr = _as_finite_nonneg(a)
    n = int(arr.size)
    if n == 0:
        return math.inf
    s = float(arr.sum())
    if s <= 1e-14:
        return math.inf
    b = _convolution(arr)
    if b.size == 0:
        return math.inf
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _assemble_A(f):
    n = int(f.size)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, Mw, limit):
    n = int(f.size)
    if n <= 0 or not math.isfinite(Mw) or Mw <= 1e-30:
        return None
    A = _assemble_A(f)
    b_ub = np.full(2 * n - 1, float(Mw), dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    options = {
        "time_limit": float(max(0.1, limit)),
        "primal_feasibility_tolerance": 1e-9,
        "dual_feasibility_tolerance": 1e-9,
    }
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options=options)
    except Exception:
        return None
    if res is None or getattr(res, "x", None) is None:
        return None
    status = int(getattr(res, "status", 1))
    if status not in (0, 3):
        return None
    g = np.asarray(res.x, dtype=np.float64).copy()
    g[~np.isfinite(g)] = 0.0
    g = np.maximum(g, 0.0)
    if status == 3:
        try:
            viol = float(np.max(A.dot(g) - b_ub))
            if viol > 1e-5 * max(1.0, float(Mw)) + 1e-7:
                return None
        except Exception:
            return None
    s = float(g.sum())
    if s <= 1e-14:
        return None
    g /= s
    return g


def _cd2d_step(f, b, s, tmp, stop_time, rng, max_pairs):
    n = int(f.size)
    if n < 2:
        cur = float(2.0 * n * float(b.max()) / (s * s)) if s > 0 else math.inf
        return False, cur, f, b, s
    cur = float(2.0 * n * float(b.max()) / (s * s))
    two_f = 2.0 * f
    meanv = float(f.mean())
    maxv = float(f.max())
    base = max(meanv, 0.05 * maxv, 1e-6 / max(1, n))
    mults = (0.0, 0.25, 0.5, 1.0, 2.0, 4.0, 8.0)
    best_local = cur
    best_move = None
    for _ in range(int(max_pairs)):
        if time.monotonic() >= stop_time - 2.0:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        oi = float(f[i])
        oj = float(f[j])
        if oi > 1e-14:
            vals_i = [oi * m if m > 0.0 else 0.0 for m in mults]
        else:
            vals_i = [0.0] + [base * m for m in mults[1:]]
        if oj > 1e-14:
            vals_j = [oj * m if m > 0.0 else 0.0 for m in mults]
        else:
            vals_j = [0.0] + [base * m for m in mults[1:]]
        for v1 in vals_i:
            d1 = float(v1) - oi
            if abs(d1) < 1e-15:
                continue
            for v2 in vals_j:
                d2 = float(v2) - oj
                if abs(d2) < 1e-15:
                    continue
                ns = s + d1 + d2
                if ns <= 1e-12:
                    continue
                tmp[:] = b
                tmp[i:i + n] += d1 * two_f
                tmp[2 * i] += d1 * d1
                tmp[j:j + n] += d2 * two_f
                tmp[2 * j] += d2 * d2
                tmp[i + j] += 2.0 * d1 * d2
                sc = float(2.0 * n * float(tmp.max()) / (ns * ns))
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_move = (i, j, float(v1), float(v2))
    if best_move is None:
        return False, cur, f, b, s
    i, j, v1, v2 = best_move
    d1 = float(v1) - float(f[i])
    d2 = float(v2) - float(f[j])

    b[i + j] += 2.0 * d1 * d2

    f[j] = float(v2)
    s += d1 + d2
    if s > 2.0 or s < 0.5:
        inv = 1.0 / s
        f *= inv
        b *= inv * inv
        s = 1.0
    return True, float(best_local), f, b, s

            if sc_new < best_sc_cur - 1e-13:
                best_sc_cur = sc_new
                best_t = st
                best_f_new = f_new

        if best_f_new is not None:
            f = best_f_new.copy()
            cur = best_sc_cur
            if cur < best_sc - 1e-13:
                best_sc = cur
                best = f.copy()
            t = float(min(0.65, max(0.04, best_t * 1.1)))
            stall = 0
        else:
            t = max(0.02, t * 0.6)
            stall += 1

        if it % 5 == 4 and time.monotonic() < stop_time - 25.0:
            cur_f = _score(f)
            f2, sc2 = _cd_polish(
                f,
                min(stop_time - 5.0, time.monotonic() + 6.0),
                seed=int(seed) + it * 17 + 2,
                max_outer=2,
                use_2d=(it % 10 == 4),
                max_2d_pairs=35,
            )
            if f2 is not None:
                if sc2 < best_sc:
                    best_sc = sc2
                    best = f2.copy()
                if sc2 < cur_f - 1e-13:
                    f = f2.copy()

        it += 1

    if time.monotonic() < stop_time - 4.0:
        f2, sc2 = _cd_polish(
            best,
            stop_time - 2.0,
            seed=int(seed) + 999,
            max_outer=10,
            use_2d=True,
            max_2d_pairs=60,
        )
        if f2 is not None and sc2 <= best_sc + 1e-12:
            best = f2.copy()
            best_sc = sc2
    return best, float(best_sc)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(max(10.0, budget_s))
    hard_deadline = t0 + budget - 20.0
    if hard_deadline <= t0 + 5.0:
        hard_deadline = t0 + max(2.0, budget - 2.0)

    seed_int = int(seed) % (2 ** 63)

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent_arr = np.ones(400, dtype=np.float64)

    parent_arr = _as_finite_nonneg(parent_arr)
    if parent_arr.size < 8 or float(parent_arr.sum()) <= 1e-12:
        parent_arr = np.ones(400, dtype=np.float64)
    if parent_arr.size > 1200:
        parent_arr = _resample_array(parent_arr, 400)

    ref_len = int(parent_arr.size)
    parent = _resample_array(parent_arr, ref_len)

    best_seq = parent.copy()
    fallback_seq = parent.copy()
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        best_score = math.inf
    if not math.isfinite(best_score):
        best_seq = np.ones(400, dtype=np.float64) / 400.0
        fallback_seq = best_seq.copy()
        ref_len = 400
        try:
            best_score = float(evaluate_sequence([float(x) for x in best_seq]))
        except Exception:
            best_score = 2.0
    fallback_score = best_score

    rng = np.random.default_rng(seed_int)

    def _consider(cand, approx_sc=math.inf, force=False):
        nonlocal best_seq, best_score
        if cand is None:
            return
        f = _as_finite_nonneg(cand)
        if f.size == 0:
            return
        fs = float(f.sum())
        if fs <= 1e-14:
            return
        f = f / fs
        if (
            not force
            and math.isfinite(approx_sc)
            and math.isfinite(best_score)
            and approx_sc >= best_score + 1e-7
        ):
            return
        try:
            val = float(evaluate_sequence([float(x) for x in f]))
        except Exception:
            val = math.inf
        if math.isfinite(val) and (not math.isfinite(best_score) or val < best_score - 1e-12):
            best_score = val
            best_seq = f.copy()

    # Quick short/medium length exploration using the fast incremental coordinate descent.
    explore_stop = min(hard_deadline - 220.0, t0 + 50.0)
    if hard_deadline - t0 > 120.0:
        explore_lengths = [24, 32, 48, 64, 96, 128, 160, 192, 256, 320, 384]
        for L in explore_lengths:
            if time.monotonic() > explore_stop - 25.0:
                break
            for mode in (0, 1):
                if time.monotonic() > explore_stop - 10.0:
                    break
                if mode == 0:
                    init = _resample_array(best_seq, L)
                else:
                    if L <= 96:
                        init = np.exp(rng.normal(0.0, 0.6, L))
                    else:
                        base = _resample_array(best_seq, L)
                        sigma = 0.2 * max(float(base.std()), float(base.mean()), 1e-6)
                        init = np.maximum(0.0, base + rng.normal(0.0, sigma, L))
                stopL = min(explore_stop - 5.0, time.monotonic() + 6.0)
                if stopL <= time.monotonic() + 1.0:
                    continue
                f, sc = _cd_polish(
                    init,
                    stopL,
                    seed=seed_int * 1000 + L * 10 + mode,
                    max_outer=4,
                    use_2d=(L <= 128),
                    max_2d_pairs=35,
                )
                if f is not None:
                    _consider(f, sc, force=True)

    # Main LP-guided optimization on a small set of high-value tasks.
    main_stop = min(hard_deadline - 140.0, t0 + budget * 0.72)
    if main_stop <= explore_stop:
        main_stop = min(hard_deadline - 60.0, explore_stop + 100.0)

    tasks = []
    cur_len = int(best_seq.size)
    if 8 <= cur_len <= 1200:
        tasks.append((cur_len, best_seq.copy()))
    if len(tasks) == 0 or tasks[0][0] != ref_len:
        tasks.append((ref_len, parent.copy()))

    pn = parent.copy()
    sigma = 0.2 * max(float(pn.std()), 1e-4)
    pn2 = np.clip(pn + rng.normal(0.0, sigma, ref_len), 0.0, None)
    pn2 /= float(pn2.sum())
    tasks.append((ref_len, pn2))

    iarr = np.arange(ref_len, dtype=np.float64)
    ini = 0.5 * (1.0 + np.cos(2.0 * np.pi * iarr / ref_len)) + 1e-8
    ini /= float(ini.sum())
    tasks.append((ref_len, ini))

    ctr = 0.5 * (ref_len - 1)
    ini = np.maximum(0.0, 1.0 - ((iarr - ctr) / (0.55 * ref_len)) ** 2) + 1e-8
    ini /= float(ini.sum())
    tasks.append((ref_len, ini))

    ini = np.exp(rng.normal(0.0, 0.5, ref_len))
    ini /= float(ini.sum())
    tasks.append((ref_len, ini))

    if cur_len != ref_len and 8 <= cur_len <= 1200:
        tasks.append((ref_len, _resample_array(best_seq, ref_len)))

    tasks = tasks[:6]

    task_idx = 0
    while task_idx < len(tasks) and time.monotonic() < main_stop - 20.0:
        n_t, init = tasks[task_idx]
        if n_t < 8 or n_t > 1200:
            task_idx += 1
            continue
        rem_tasks = max(1, len(tasks) - task_idx)
        per = max(25.0, (main_stop - time.monotonic() - 15.0) / float(rem_tasks))
        stop = min(main_stop - 10.0, time.monotonic() + per)
        if stop - time.monotonic() < 15.0:
            task_idx += 1
            continue
        try:
            f, sc = _lp_loop(init, n_t, stop, seed=seed_int * 10000 + 7 + task_idx * 31)
        except Exception:
            f, sc = _cd_polish(init, min(stop, time.monotonic() + 5.0), seed=seed_int, max_outer=3)
        if f is not None:
            _consider(f, sc, force=True)
        task_idx += 1

    # Extra length experimentation if time remains.
    extra_stop = min(hard_deadline - 70.0, t0 + budget * 0.85)
    if time.monotonic() < extra_stop - 60.0:
        L_best = int(best_seq.size)
        extra_lengths = [L for L in (256, 384, 512, 640) if L != L_best]
        for L in extra_lengths[:3]:
            if time.monotonic() > extra_stop - 40.0:
                break
            if L < 8 or L > 1200:
                continue
            init = _resample_array(best_seq, L)
            stop = min(extra_stop - 20.0, time.monotonic() + max(30.0, 0.25 * (extra_stop - time.monotonic())))
            if stop <= time.monotonic() + 10.0:
                continue
            try:
                f, sc = _lp_loop(init, L, stop, seed=seed_int + 88888 + L)
            except Exception:
                f, sc = _cd_polish(init, min(stop, time.monotonic() + 5.0), seed=seed_int + L, max_outer=5)
            if f is not None:
                _consider(f, sc, force=True)

    # Final polishing. If the score is still above the target, try one more focused LP boost.
    final_stop = hard_deadline - 5.0
    if time.monotonic() < final_stop - 8.0:
        try:
            f, sc = _cd_polish(
                best_seq,
                final_stop,
                seed=seed_int * 7 + 101,
                max_outer=80,
                use_2d=True,
                max_2d_pairs=80,
            )
            if f is not None:
                _consider(f, sc, force=True)
        except Exception:
            pass

    if time.monotonic() < hard_deadline - 35.0 and math.isfinite(best_score) and best_score > 1.5028:
        L = int(best_seq.size)
        if 8 <= L <= 1200:
            stop = hard_deadline - 20.0
            try:
                f, sc = _lp_loop(best_seq, L, stop, seed=seed_int + 777777)
                if f is not None:
                    _consider(f, sc, force=True)
            except Exception:
                pass
            if time.monotonic() < hard_deadline - 8.0:
                try:
                    f, sc = _cd_polish(
                        best_seq,
                        hard_deadline - 5.0,
                        seed=seed_int + 99,
                        max_outer=20,
                        use_2d=True,
                        max_2d_pairs=80,
                    )
                    if f is not None:
                        _consider(f, sc, force=True)
                except Exception:
                    pass

    # Final validation and safe return.
    final_seq = _as_finite_nonneg(best_seq)
    fs = float(final_seq.sum())
    if final_seq.size == 0 or fs < 0.01:
        final_seq = _as_finite_nonneg(fallback_seq)
        fs = float(final_seq.sum())
    if final_seq.size > 0 and 0.0 < fs < 0.01:
        final_seq /= fs

    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        final_seq = _as_finite_nonneg(fallback_seq)
        fs = float(final_seq.sum())
        if final_seq.size > 0 and 0.0 < fs < 0.01:
            final_seq /= fs
        try:
            final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]

    if math.isfinite(fallback_score) and final_sc > fallback_score + 1e-10:
        final_seq = _as_finite_nonneg(fallback_seq)
        fs = float(final_seq.sum())
        if final_seq.size > 0 and 0.0 < fs < 0.01:
            final_seq /= fs

    result = [max(0.0, min(1000.0, float(x))) for x in final_seq]
    if not result:
        result = [1.0]

    try:
        last_val = float(evaluate_sequence(result))
    except Exception:
        last_val = math.inf
    if not math.isfinite(last_val):
        return [1.0]

    return result
