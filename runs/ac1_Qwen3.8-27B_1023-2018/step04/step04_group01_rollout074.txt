def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v.copy()

        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    return np.maximum(v - theta, 0.0)

    f = np.clip(f, 0.0, 1000.0)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    if f.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if f.size == n:
        return f
    xold = np.linspace(0.0, 1.0, f.size)
    xnew = np.linspace(0.0, 1.0, n)
    out = np.interp(xnew, xold, f)
    return np.clip(out, 0.0, 1000.0)


def _smooth_grad(b, f, tau):
    b = np.asarray(b, dtype=np.float64).ravel()
    f = np.asarray(f, dtype=np.float64).ravel()
    n = f.size
    if b.size == 0 or n == 0:
        return 0.0, np.zeros_like(f, dtype=np.float64)

    M = float(b.max())
    if not math.isfinite(M) or M <= 0.0:
        return M, np.zeros_like(f, dtype=np.float64)

    if tau <= 1e-18:
        p = np.zeros_like(b, dtype=np.float64)
        p[int(np.argmax(b))] = 1.0
        smax = M
    else:
        y = (b - M) / tau
        c = float(np.max(y))
        z = y - c
        e = np.exp(z)
        s = float(np.sum(e))
        if s <= 1e-300:
            p = np.zeros_like(b, dtype=np.float64)
            p[int(np.argmax(b))] = 1.0
            smax = M
        else:
            p = e / s
            smax = M + tau * (math.log(s) + c)

    if sliding_window_view is not None and b.size >= n:
        try:
            q = sliding_window_view(p, n).dot(f)
        except Exception:
            q = np.empty(n, dtype=np.float64)
            for i in range(n):
                q[i] = float(np.dot(f, p[i:i + n]))
    else:
        q = np.empty(n, dtype=np.float64)
        for i in range(n):
            q[i] = float(np.dot(f, p[i:i + n]))

    return smax, 2.0 * q


def _line_search(f, g):
    f = np.asarray(f, dtype=np.float64).ravel()
    g = np.asarray(g, dtype=np.float64).ravel()
    n = f.size
    if n == 0:
        return 0.0, math.inf, math.inf, math.inf

    b0 = _conv(f)
    b1 = _conv(g)
    b01 = _conv(f, g)

    score0 = 2.0 * n * float(b0.max())
    score1 = 2.0 * n * float(b1.max())

    best_t = 0.0
    best_sc = score0
    if score1 < best_sc:
        best_t = 1.0
        best_sc = score1

    def val(t):
        if t <= 0.0:
            return score0
        if t >= 1.0:
            return score1
        u = 1.0 - t
        arr = (u * u) * b0 + (2.0 * t * u) * b01 + (t * t) * b1
        return 2.0 * n * float(arr.max())

    lo, hi = 0.0, 1.0
    gr = (1.0 + math.sqrt(5.0)) / 2.0
    x1 = hi - (hi - lo) / gr
    x2 = lo + (hi - lo) / gr
    y1 = val(x1)
    y2 = val(x2)

    if y1 < best_sc:
        best_sc, best_t = y1, x1
    if y2 < best_sc:
        best_sc, best_t = y2, x2

    for _ in range(30):
        if hi - lo < 1e-8:
            break
        if y1 < y2:
            hi = x2
            x2 = x1
            y2 = y1
            x1 = hi - (hi - lo) / gr
            y1 = val(x1)
        else:
            lo = x1
            x1 = x2
            y1 = y2
            x2 = lo + (hi - lo) / gr
            y2 = val(x2)

        if y1 < best_sc:
            best_sc, best_t = y1, x1
        if y2 < best_sc:
            best_sc, best_t = y2, x2

    tm = 0.5 * (lo + hi)
    ym = val(tm)
    if ym < best_sc:
        best_sc, best_t = ym, tm

    return best_t, best_sc, score0, score1


def _lp_minimax(f, time_limit=4.0):
    f = np.asarray(f, dtype=np.float64).ravel()
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.clip(f, 0.0, 1000.0)
    s = float(f.sum())
    n = f.size
    if n == 0 or s <= 1e-12:
        return None, None
    f = f / s
    if n == 1:
        return f.copy(), float(f[0])

    A = np.zeros((2 * n - 1, n + 1), dtype=np.float64)
    A[:, -1] = -1.0
    for j in range(n):
        A[j:j + n, j] = f

    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0
    b_ub = np.zeros(2 * n - 1, dtype=np.float64)
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    bounds = [(0.0, None)] * (n + 1)

    tl = max(1.0, float(time_limit))
    res = None
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": tl},
        )
    except Exception:
        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
        except Exception:
            return None, None

    if res is None or getattr(res, "status", -1) != 0 or res.x is None:
        return None, None

    g = np.asarray(res.x[:n], dtype=np.float64)
    z = float(res.x[n]) if np.size(res.x) > n else 0.0
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.clip(g, 0.0, None)
    sg = float(g.sum())
    if sg < 1e-12:
        return None, z
    return g / sg, z


def _coord_descent(f, stop_time, max_sweeps=1000, seed=0, n_grid=32):
    f = np.asarray(f, dtype=np.float64).ravel().copy()
    if f.size == 0:
        return f, math.inf
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.clip(f, 0.0, 1000.0)
    S = float(f.sum())
    n = f.size
    if S < 1e-12:
        f = np.ones(n, dtype=np.float64) / float(n)
        S = 1.0
    else:
        f = f / S
        S = 1.0

    b = _conv(f)
    best_f = f.copy()
    best_score = 2.0 * n * float(b.max()) / (S * S)
    rng = np.random.default_rng(seed)
    stuck = 0
    perturb_count = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time:
            break

        if abs(S - 1.0) > 0.25:
            f = f / S
            b = b / (S * S)
            S = 1.0

        cur_m = float(b.max())
        cur_score = 2.0 * n * cur_m / (S * S)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_f = f.copy()

        if sweep > 0 and sweep % 40 == 0:
            b = _conv(f)
            S = float(f.sum())
            if S < 1e-12:
                break
            f = f / S
            b = b / (S * S)
            S = 1.0
            cur_score = 2.0 * n * float(b.max())
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_f = f.copy()

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() > stop_time:
                break

            v = float(f[i])
            Srest = S - v
            if Srest <= 1e-12 and v <= 0.0:
                continue

            lo = i
            hi = i + n
            seg = b[lo:hi]

            outside = -np.inf
            if i > 0:
                outside = float(np.max(b[:i]))
            if hi < 2 * n - 1:
                vout = float(np.max(b[hi:2 * n - 1]))
                if vout > outside:
                    outside = vout

            b2i = float(b[2 * i])
            x_max = max(10.0 * v, 0.1 * S, 1e-5)
            x_max = min(x_max, 4.0 * S)
            if x_max < v * 1.5 + 1e-8:
                x_max = v * 1.5 + 1e-8

            xs = np.linspace(0.0, x_max, n_grid)
            w = xs - v
            Snew = Srest + xs
            safe_S = np.where(Snew > 1e-12, Snew, 1.0)
            valid = Snew > 1e-12
            if not valid.any():
                continue

            with np.errstate(divide="ignore", invalid="ignore", over="ignore"):
                mseg = np.max(seg[None, :] + 2.0 * w[:, None] * f, axis=1)
                mquad = b2i + 2.0 * w * f[i] + w * w
                m = np.maximum(mseg, mquad)
                if np.isfinite(outside):
                    m = np.maximum(m, outside)
                m = np.maximum(m, 0.0)
                obj = 2.0 * n * m / (safe_S * safe_S)
            obj[~valid] = math.inf

            k = int(np.argmin(obj))
            x_best = float(xs[k])
            obj_best = float(obj[k])

            if x_best != v and math.isfinite(obj_best):
                if k == 0:
                    x_lo, x_hi = 0.0, float(xs[1])
                elif k == n_grid - 1:
                    x_lo, x_hi = float(xs[-2]), x_max
                else:
                    x_lo, x_hi = float(xs[k - 1]), float(xs[k + 1])
                if x_hi < x_lo:
                    x_lo, x_hi = x_hi, x_lo

                def eval_scalar(x):
                    Sn = Srest + x
                    if Sn <= 1e-12:
                        return math.inf
                    ww = x - v
                    ms = float(np.max(seg + 2.0 * ww * f))
                    mq = b2i + 2.0 * ww * f[i] + ww * ww
                    mm = max(ms, mq)
                    if np.isfinite(outside) and outside > mm:
                        mm = outside
                    if mm < 0.0:
                        mm = 0.0
                    return 2.0 * n * mm / (Sn * Sn)

                gr = (1.0 + math.sqrt(5.0)) / 2.0
                x1 = x_hi - (x_hi - x_lo) / gr
                x2 = x_lo + (x_hi - x_lo) / gr
                y1 = eval_scalar(x1)
                y2 = eval_scalar(x2)

                for _ in range(22):
                    if x_hi - x_lo < 1e-12:
                        break
                    if y1 < y2:
                        x_hi = x2
                        x2 = x1
                        y2 = y1
                        x1 = x_hi - (x_hi - x_lo) / gr
                        y1 = eval_scalar(x1)
                    else:
                        x_lo = x1
                        x1 = x2
                        y1 = y2
                        x2 = x_lo + (x_hi - x_lo) / gr
                        y2 = eval_scalar(x2)

                xc = 0.5 * (x_lo + x_hi)
                yc = eval_scalar(xc)
                if yc < obj_best:
                    x_best, obj_best = xc, yc

            if x_best < 0.0:
                x_best = 0.0

            if obj_best < cur_score - 1e-13:
                wb = x_best - v
                if abs(wb) > 1e-15:
                    b[lo:hi] = b[lo:hi] + 2.0 * wb * f
                    b[2 * i] = b[2 * i] + wb * wb
                    f[i] = x_best
                    S = Srest + x_best
                cur_score = obj_best
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_f = f.copy()
                    improved = True

        if not improved:
            stuck += 1
            if stuck <= 2 and perturb_count < 8:
                perturb_count += 1
                if stuck % 2 == 1 or rng.random() < 0.5:
                    f_new = _block_perturb(best_f, rng, frac=min(0.4, 0.08 + 0.02 * stuck))
                else:
                    f_new = _perturb(f, rng, 0.02 + 0.01 * stuck)
                f_new = _normalize(f_new)
                sc_new = _score_local(f_new)
                if math.isfinite(sc_new) and sc_new < best_score - 1e-14:
                    best_score = sc_new
                    best_f = f_new.copy()
                f = f_new
                S = float(f.sum())
                if S < 1e-12:
                    break
                f = f / S
                S = 1.0
                b = _conv(f)
                stuck = 0
            else:
                break

    f = best_f
    return f, _score_local(f)


def _lp_rounds(f, stop_time, seed=0, max_iters=5000):
    f = _normalize(f)
    best_f = f.copy()
    best_score = _score_local(f)
    rng = np.random.default_rng(seed)
    it = 0
    stall = 0
    perturb_count = 0

    while it < max_iters and time.monotonic() < stop_time - 3.0:
        t_left = stop_time - time.monotonic()
        if t_left < 12.0:
            break

        lp_time = min(5.0, max(1.0, 0.1 * t_left))
        g, _ = _lp_minimax(f, lp_time)
        moved = False

        if g is not None:
            f0 = _normalize(f)
            g0 = _normalize(g)
            t_best, sc_best, sc0, sc1 = _line_search(f0, g0)

            if t_best > 1e-5 and sc_best < max(best_score, sc0) - 1e-13:
                f_new = (1.0 - t_best) * f0 + t_best * g0
                f_new = _normalize(f_new)
                sc_new = _score_local(f_new)
                if math.isfinite(sc_new) and sc_new < best_score - 1e-14:
                    best_score = sc_new
                    best_f = f_new.copy()
                    f = f_new
                    stall = 0
                    moved = True
                elif math.isfinite(sc_new) and sc_new < sc0 - 1e-14:
                    f = f_new
                    stall = 0
                    moved = True

            if not moved and math.isfinite(sc1) and sc1 < max(best_score, sc0) - 1e-14:
                f = g0
                if sc1 < best_score - 1e-14:
                    best_score = sc1
                    best_f = f.copy()
                stall = 0
                moved = True

            if moved and time.monotonic() < stop_time - 8.0:
                cst = min(stop_time - 5.0, time.monotonic() + 1.5)
                if cst > time.monotonic() + 0.5:
                    f2, sc2 = _coord_descent(f, cst, max_sweeps=4, seed=it * 7 + 1, n_grid=20)
                    if math.isfinite(sc2) and sc2 < best_score - 1e-14:
                        best_score = sc2
                        best_f = f2.copy()
                        f = f2

        if not moved:
            stall += 1
            if stall > 8:
                if perturb_count > 30:
                    break
                perturb_count += 1
                if rng.random() < 0.5:
                    f = _block_perturb(
                        best_f,
                        rng,
                        frac=min(0.5, 0.05 + 0.02 * min(10, stall)),
                    )
                else:
                    f = _perturb(f, rng, 0.02 + 0.005 * min(10, stall))
                f = _normalize(f)
                if time.monotonic() < stop_time - 6.0:
                    cst = min(stop_time - 4.0, time.monotonic() + 3.0)
                    f, sc = _coord_descent(f, cst, max_sweeps=6, seed=it * 11 + 2, n_grid=24)
                    sc = _score_local(f)
                    if math.isfinite(sc) and sc < best_score - 1e-14:
                        best_score = sc
                        best_f = f.copy()
                stall = 0

        it += 1

    return best_f, best_score


def _grad_rounds(f, stop_time, seed=0, max_iters=5000):
    f = _normalize(f)
    best_f = f.copy()
    best_score = _score_local(f)
    rng = np.random.default_rng(seed)

    b = _conv(f)
    m0 = float(b.max()) if b.size else 1.0
    tau = max(1e-8, 0.25 * m0)
    tau_min = max(1e-10, 0.01 * m0)
    alpha = 0.05
    stall = 0
    perturb_count = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break

        cur_score = _score_local(f)
        if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
            best_score = cur_score
            best_f = f.copy()

        b = _conv(f)
        if b.size == 0:
            break
        _, grad = _smooth_grad(b, f, tau)
        grad = np.asarray(grad, dtype=np.float64)
        grad -= float(np.mean(grad))
        gn = float(np.linalg.norm(grad))
        if not math.isfinite(gn) or gn < 1e-18:
            break
        grad /= gn

        moved = False
        for _ in range(4):
            f_new = _project_simplex(f - alpha * grad)
            sc_new = _score_local(f_new)
            if math.isfinite(sc_new) and sc_new < best_score - 1e-14:
                f = f_new
                best_score = sc_new
                best_f = f_new.copy()
                moved = True
                alpha = min(0.25, alpha * 1.05)
                break
            if math.isfinite(sc_new) and sc_new < cur_score - 1e-14:
                f = f_new
                moved = True
                alpha = min(0.25, alpha * 1.02)
                break
            alpha = max(1e-6, alpha * 0.6)

        if not moved:
            stall += 1
            if stall > 12:
                if perturb_count > 30:
                    break
                perturb_count += 1
                if rng.random() < 0.3:
                    f = _block_perturb(best_f, rng, frac=0.15)
                else:
                    f = _perturb(f, rng, 0.015)
                f = _normalize(f)
                tau = max(tau_min, tau * 0.7)
                alpha = 0.03
                stall = 0
        else:
            if (it + 1) % 100 == 0:
                tau = max(tau_min, tau * 0.92)

    return best_f, best_score


def _alternative_inits(n, base, seed):
    rng = np.random.default_rng(seed)
    out = []
    base = np.asarray(base, dtype=np.float64).ravel()
    if base.size != n:
        base = _resize(base, n)
    base = _normalize(base)

    out.append(base)
    out.append(_perturb(base, rng, 0.05))
    out.append(_block_perturb(base, rng, 0.30))

    x = np.linspace(0.0, 1.0, n)
    pats = [
        np.ones(n, dtype=np.float64),
        0.2 + 0.8 * np.cos(np.pi * x) ** 2,
        0.2 + 0.8 * np.cos(2.0 * np.pi * x) ** 2,
        0.3 + 0.7 * np.exp(-6.0 * (x - 0.5) ** 2),
        np.exp(-5.0 * (x - 0.35) ** 2) + 0.8 * np.exp(-5.0 * (x - 0.65) ** 2),
        np.maximum(0.0, 1.0 - 1.8 * np.abs(x - 0.5)) + 0.02,
    ]
    for p in pats:
        p = np.asarray(p, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.sum() > 1e-6:
            out.append(p / p.sum())
        if len(out) >= 10:
            break

    try:
        out.append(rng.dirichlet(np.full(n, 1.0)))
    except Exception:
        pass
    try:
        out.append(rng.dirichlet(np.full(n, 0.4)))
    except Exception:
        pass

    res = []
    for a in out:
        a = _normalize(a)
        if a.size == n and np.all(np.isfinite(a)):
            res.append(a)
    return res[:10]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not math.isfinite(budget) or budget < 10.0:
        budget = 10.0

    hard_deadline = t0 + budget
    deadline = hard_deadline - min(25.0, budget * 0.15)
    if deadline <= t0:
        deadline = t0 + max(2.0, budget * 0.8)

    def now():
        return time.monotonic()

    def phase_end(frac, reserve):
        e = min(deadline - reserve, t0 + budget * frac)
        if e < t0:
            e = t0 + 0.5
        return e

    parent = _get_parent()
    if parent is None or parent.size == 0:
        best_seq = _fallback_sequence(400)
    else:
        best_seq = np.asarray(parent, dtype=np.float64).ravel().copy()

    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = _fallback_sequence(400)

    best_score = _exact_score(best_seq)
    if not math.isfinite(best_score):
        best_seq = _fallback_sequence(400)
        best_score = _exact_score(best_seq)
        if not math.isfinite(best_score):
            return [1.0]

    def update(f):
        nonlocal best_seq, best_score
        if f is None:
            return
        f = np.asarray(f, dtype=np.float64).ravel()
        f = np.where(np.isfinite(f), f, 0.0)
        f = np.clip(f, 0.0, 1000.0)
        if f.size == 0 or f.sum() < 0.01:
            return
        loc = _score_local(f)
        if not math.isfinite(loc):
            return
        if loc > best_score + 1e-8:
            return
        ex = _exact_score(f)
        if math.isfinite(ex) and ex < best_score - 1e-15:
            best_seq = f.copy()
            best_score = ex

    f = _normalize(best_seq)

    end1 = phase_end(0.06, 30.0)
    if now() + 2.0 < end1:
        f, _ = _coord_descent(f, end1, max_sweeps=5000, seed=1000, n_grid=32)
        update(f)
        f = _normalize(best_seq)

    end2 = phase_end(0.30, 400.0)
    if now() + 10.0 < end2 and best_score > TARGET - 1e-7:
        f, _ = _lp_rounds(f, end2, seed=2000, max_iters=10000)
        update(f)
        f = _normalize(best_seq)

    end3 = phase_end(0.42, 300.0)
    if now() + 5.0 < end3:
        f, _ = _grad_rounds(f, end3, seed=3000, max_iters=10000)
        update(f)
        f = _normalize(best_seq)

    end4 = phase_end(0.62, 200.0)
    if now() + 5.0 < end4:
        f, _ = _coord_descent(f, end4, max_sweeps=20000, seed=4000, n_grid=40)
        update(f)
        f = _normalize(best_seq)

    end5 = phase_end(0.74, 150.0)
    if now() + 10.0 < end5 and best_score > TARGET - 1e-7:
        f, _ = _lp_rounds(f, end5, seed=5000, max_iters=5000)
        update(f)
        f = _normalize(best_seq)

    end6 = phase_end(0.88, 100.0)
    rseed = 6000
    while now() + 20.0 < end6:
        cstop = min(end6 - 20.0, now() + 12.0)
        if cstop > now() + 3.0:
            f, _ = _coord_descent(f, cstop, max_sweeps=5000, seed=rseed, n_grid=36)
            update(f)
            f = _normalize(best_seq)
            rseed += 1

        gstop = min(end6 - 10.0, now() + 8.0)
        if gstop > now() + 3.0:
            f, _ = _grad_rounds(f, gstop, seed=rseed + 5000, max_iters=3000)
            update(f)
            f = _normalize(best_seq)
            rseed += 1

        if best_score <= TARGET - 1e-6 and now() + 60.0 > end6:
            break

    alt_end = phase_end(0.93, 60.0)
    if now() + 20.0 < alt_end and best_score > TARGET - 1e-5:
        base_n = best_seq.size
        alts = _alternative_inits(base_n, best_seq, seed=7000)
        accepted = []
        for a in alts:
            a = _normalize(a)
            if a.size != base_n:
                continue
            if np.linalg.norm(a - best_seq) < 1e-5:
                continue
            if any(np.linalg.norm(a - u) < 0.12 for u in accepted):
                continue
            accepted.append(a)
            if len(accepted) >= 6:
                break

        for idx, a in enumerate(accepted):
            if now() + 15.0 >= alt_end:
                break
            alloc = min(alt_end - now() - 5.0, 35.0)
            if alloc < 12.0:
                break
            a = _normalize(a)
            if base_n <= 512 and alloc > 18.0:
                lstop = min(alt_end - 5.0, now() + 0.40 * alloc)
                if lstop > now() + 3.0:
                    a, _ = _lp_rounds(a, lstop, seed=8000 + idx, max_iters=50)
                    update(a)
                    a = _normalize(best_seq)
            cstop = min(alt_end - 5.0, now() + max(3.0, 0.50 * alloc))
            if cstop > now() + 2.0:
                a, _ = _coord_descent(a, cstop, max_sweeps=200, seed=9000 + idx, n_grid=28)
                update(a)
                a = _normalize(a)
        f = _normalize(best_seq)

    len_end = phase_end(0.97, 35.0)
    if now() + 20.0 < len_end:
        base_n = best_seq.size
        lengths = []
        for L in (base_n - 32, base_n - 16, base_n + 16, base_n + 32, 320, 352, 384, 416, 448, 480, 512, 640):
            if 32 <= L <= 1000 and L != base_n and L not in lengths:
                lengths.append(L)
        lengths.sort(key=lambda L: abs(L - base_n))

        for idx, L in enumerate(lengths[:6]):
            if now() + 12.0 >= len_end:
                break
            alloc = min(len_end - now() - 5.0, 25.0)
            if alloc < 10.0:
                break
            a = _resize(_normalize(best_seq), L)
            a = _normalize(a)
            if L <= 512 and alloc > 18.0:
                lstop = min(len_end - 5.0, now() + 0.30 * alloc)
                if lstop > now() + 3.0:
                    a, _ = _lp_rounds(a, lstop, seed=10000 + L, max_iters=40)
                    update(a)
                    a = _normalize(a)
            cstop = min(len_end - 5.0, now() + max(3.0, 0.70 * alloc))
            if cstop > now() + 2.0:
                a, _ = _coord_descent(a, cstop, max_sweeps=200, seed=11000 + L, n_grid=28)
                update(a)
                a = _normalize(a)
        f = _normalize(best_seq)

    final_c1 = min(deadline - 8.0, max(now() + 1.0, t0 + budget))
    f, _ = _coord_descent(f, final_c1, max_sweeps=20000, seed=12345, n_grid=48)
    update(f)
    f = _normalize(best_seq)

    if now() < deadline - 6.0:
        gend = min(deadline - 4.0, now() + max(2.0, min(15.0, 0.20 * max(0.0, deadline - now()))))
        if gend > now() + 2.0:
            f, _ = _grad_rounds(f, gend, seed=54321, max_iters=3000)
            update(f)
            f = _normalize(best_seq)

    final_c2 = min(deadline - 1.0, max(now() + 0.5, hard_deadline - 5.0))
    if final_c2 > now() + 0.5:
        f, _ = _coord_descent(f, final_c2, max_sweeps=5000, seed=65432, n_grid=36)
        update(f)

    best_seq = np.asarray(best_seq, dtype=np.float64).ravel()
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = _fallback_sequence(400)

    final_score = _exact_score(best_seq)
    if not math.isfinite(final_score):
        fb = _fallback_sequence(400)
        fb_score = _exact_score(fb)
        if math.isfinite(fb_score):
            best_seq = fb
        else:
            return [1.0]

    return [float(x) for x in best_seq]
