def _binom_init(n):
    """Initialize with binomial coefficients: f_k = C(2k,k)/4^k."""
    f = np.zeros(n)
    f[0] = 1.0
    for k in range(1, n):
        f[k] = f[k-1] * (2*k-1) / (2*k)  # C(2k,k)/4^k = C(2k-2,k-1)/4^{k-1} * (2k-1)/(2k)
    s = f.sum()
    if s > 0:
        f /= s
    return f

def _conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = next_fast_len(2*n-1)
    F = rfft(a, size)
    c = irfft(F * F, size)[:2*n-1]
    np.maximum(c, 0, out=c)  # Clip small negatives from FFT
    return c

def _score(f):
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return math.inf
    b = _conv(f)
    m = b.max()
    return 2.0 * n * m / (s * s)

def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    r = rho_idx[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)

def _smoothed_max_grad(f, tau):
    """Compute gradient of smoothed max of convolution."""
    n = len(f)
    b = _conv(f)
    M = b.max()
    # Softmax weights
    shifted = (b - M) / tau
    shifted -= shifted.max()
    p = np.exp(shifted)
    p /= p.sum()
    # Gradient: grad_j = 2 * sum_k p[k] * f[k-j] (with valid indices)
    # This is 2 * (p * f_rev)[n-1 : 2n-1] where * is convolution
    f_rev = f[::-1]
    size = next_fast_len(2*n-1)
    P = rfft(p, size)
    Fr = rfft(f_rev, size)
    conv_val = irfft(P * Fr, size)[:2*n-1]
    grad = 2.0 * conv_val[n-1:2*n-1]
    return grad, M, b

def _grad_optimize(f0, n, stop_time, seed, max_iters=50000):
    """Smoothed-max gradient descent."""
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0, None)
    f = f / f.sum()
    best_f = f.copy()
    best_score = _score(f)
    
    tau = 2.0  # smoothing parameter
    alpha = 0.01  # step size
    alpha_min = 1e-8
    
    for it in range(max_iters):
        if time.monotonic() > stop_time - 3:
            break
        
        grad, M, b = _smoothed_max_grad(f, tau)
        s = f.sum()
        score = 2.0 * n * M / (s * s)
        
        if score < best_score - 1e-14:
            best_score = score
            best_f = f.copy()
        
        # Score gradient (accounting for normalization)
        # d(score)/d(f_j) = (2n/s^2) * (grad_j - 2*M/s)
        score_grad = (2.0 * n / (s * s)) * (grad - 2.0 * M / s)
        
        # Gradient step
        f_new = f - alpha * score_grad
        f_new = _project_simplex(f_new)
        
        sc_new = _score(f_new)
        if sc_new < score - 1e-15:
            f = f_new
            alpha = min(alpha * 1.05, 0.1)
            tau = max(tau * 0.999, 0.1)
        else:
            alpha *= 0.5
            if alpha < alpha_min:
                f = f + rng.normal(0, 0.002, n)
                f = np.maximum(f, 0)
                f /= f.sum()
                alpha = 0.005
                tau = 1.0
        
        # Periodic perturbation
        if it % 5000 == 4999:
            f = f + rng.normal(0, 0.005, n)
            f = np.maximum(f, 0)
            f /= f.sum()
            alpha = 0.003
    
    return best_f, best_score

def _coord_polish(f0, stop_time, max_sweeps=50, seed=777):
    """Coordinate descent with 1D line search."""
    n = len(f0)
    f = np.clip(f0.copy(), 0, None)
    f = f / f.sum()
    conv = _conv(f)
    best_score = 2.0 * n * conv.max()
    best_f = f.copy()
    rng = np.random.default_rng(seed)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 3:
            break
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time - 3:
                break
            
            v = f[i]
            # Influence on convolution
            # changing f[i] by delta changes conv[k] by delta*(f[k-i] + f[i-k]) + delta^2*(if 2i=k)
            infl = np.zeros(2*n-1)
            for k in range(2*n-1):
                # terms: f[i]*f[k-i] and f[k-i]*f[i] (same thing)
                # conv[k] = ... + f[i]*f[k-i] + f[k-i]*f[i] + ... (if i != k-i)
                # conv[k] = ... + f[i]*f[k-i] + ... (if i == k-i, i.e., k=2i)
                j1 = k - i  # f[i] * f[j1]
                j2 = i - (k - i)  # f[j2] * f[i] where j2 + i = k → j2 = k-i (same!)
                # Actually conv[k] = sum_a f[a]*f[k-a]
                # Terms involving f[i]: a=i (f[i]*f[k-i]) or k-a=i (a=k-i, f[k-i]*f[i])
                # If i != k-i: two terms f[i]*f[k-i] + f[k-i]*f[i] = 2*f[i]*f[k-i]
                # If i == k-i (k=2i): one term f[i]*f[i] = f[i]^2
                if 0 <= k-i < n and k-i != i:
                    infl[k] += 2.0 * f[k-i]
                elif k-i == i and 0 <= i < n:
                    infl[k] += 2.0 * f[i]  # f[i]^2 term, derivative is 2*f[i]
            
            # Search for best new value
            # Objective: 2*n*max(conv + delta*infl + delta^2*(1 if k=2i else 0)) / (sum+delta-something)^2
            # This is complex; use a simple grid search
            delta_range = np.linspace(-v*0.8, v*0.8, 31) if v > 1e-8 else np.linspace(-0.01, 0.01, 31)
            best_d = 0.0
            best_obj = best_score
            S_rest = f.sum() - v
            
            for d in delta_range:
                new_v = v + d
                if new_v < 0:
                    continue
                S_new = S_rest + new_v
                if S_new < 0.01:
                    continue
                new_conv = conv + d * infl
                if 2*i < 2*n-1:
                    new_conv[2*i] += d * d  # f[i]^2 contribution
                m_new = new_conv.max()
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_d = d
            
            if best_d != 0.0 and best_obj < best_score - 1e-14:
                f[i] += best_d
                conv += best_d * infl
                if 2*i < 2*n-1:
                    conv[2*i] += best_d * best_d
                best_score = best_obj
                best_f = f.copy()
                improved = True
        
        if not improved and sweep > 5:
            break
    
    return best_f, best_score

def _lp_direction(f, n, M, time_limit=5.0):
    """Find LP direction: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    # Build constraint matrix A where A[k, j] such that (A @ g)[k] = (f*g)[k]
    # (f*g)[k] = sum_j f[j] * g[k-j] (valid indices)
    # For fixed f, this is linear in g.
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Also: A[k, j] should account for both f[j]*g[k-j] and f[k-j]*g[j]
    # Wait, (f*g)[k] = sum_i f[i]*g[k-i] which is linear in g.
    # So A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # (f*g)[k] includes f[k-j]*g[j] for valid k-j
        # k-j ranges from 0 to n-1, so k ranges from j to j+n-1
        A[j:j+n, j] = f
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                     bounds=[(0, None)]*n,
                     method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0 and res.x is not None:
            g = res.x
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None
