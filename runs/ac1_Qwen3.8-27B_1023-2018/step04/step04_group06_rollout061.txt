```python
from scipy.optimize import linprog

def _score(a):
    """Compute AC1 score matching the evaluator exactly."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return float('inf')
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)

def _lp_best_response(a, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(a)
    # Build convex matrix A (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n else 0
    k_idx = np.arange(2*n-1)
    A_rows = []
    for k in range(2*n-1):
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            row[j] = a[k - j]
        A_rows.append(row)
    A = np.array(A_rows)
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    
    if res.status != 0:
        return None
    return np.clip(res.x, 0, None)
```

```python
def _build_conv_matrix(a):
    n = len(a)
    k = np.arange(2*n-1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2*n-1, n))
    A[mask] = a[d[mask]]
    return A
```

```python
def _line_search(a, g, M_cur, max_iter=50):
    """Find best t in [0,1] to minimize max(conv((1-t)a + t*g, ...))."""
    n = len(a)
    a_norm = a / a.sum()
    g_norm = g / g.sum()
    
    best_t = 0.0
    best_val = M_cur
    
    # Coarse grid
    ts = np.linspace(0, 1, 21)
    for t in ts:
        cand = (1-t)*a_norm + t*g_norm
        cand = np.clip(cand, 0, None)
        cs = cand.sum()
        if cs < 1e-15:
            continue
        cand /= cs
        val = np.convolve(cand, cand).max()
        if val < best_val:
            best_val = val
            best_t = t
    
    if best_t == 0:
        return None, M_cur
    
    # Fine search (bisection)
    lo = max(0, best_t - 0.05)
    hi = min(1, best_t + 0.05)
    for _ in range(max_iter):
        mid = 0.5*(lo+hi)
        cand = (1-mid)*a_norm + mid*g_norm
        cand = np.clip(cand, 0, None)
        cs = cand.sum()
        if cs < 1e-15:
            hi = mid
            continue
        cand /= cs
        val = np.convolve(cand, cand).max()
        if val <= best_val:
            best_val = val
            best_t = mid
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-10:
            break
    
    cand = (1-best_t)*a_norm + best_t*g_norm
    cand = np.clip(cand, 0, None)
    cs = cand.sum()
    if cs < 1e-15:
        return None, M_cur
    return cand/cs, best_val
```

```python
def _subgradient_step(a, tol=1e-8):
    """One step of projected subgradient descent on max(conv(a,a))."""
    n = len(a)
    a = a / a.sum()
    conv = np.convolve(a, a)
    M = conv.max()
    
    # Active set: where conv is close to M
    active = np.where(conv >= M * (1 - tol))[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    # Subgradient: (1/|S|) * sum_{k in S} 2*a[k-m]
    subgrad = np.zeros(n)
    for k in active:
        for m in range(n):
            idx = k - m
            if 0 <= idx < n:
                subgrad[m] += 2 * a[idx]
    subgrad /= len(active)
    
    # Projected subgradient step (project onto simplex)
    for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
        a_new = a - alpha * subgrad
        a_new = _project_simplex(a_new)
        if np.convolve(a_new, a_new).max() < M * (1 - 1e-12):
            return a_new
    return None

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
import numpy as np
import time
from scipy.optimize import linprog


def _score_fast(a):
    """Fast score matching the evaluator (assumes a is numpy array of valid values)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return float('inf')
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    return 2.0 * n * float(conv.max()) / (s * s)


def _build_conv_matrix(a):
    """Build (2n-1) x n convolution matrix."""
    n = len(a)
    k = np.arange(2 * n - 1)[:, None].astype(np.int64)
    j = np.arange(n)[None, :].astype(np.int64)
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[d[mask]]
    return A


def _lp_solve(a, M, time_limit=4.0):
    """Solve max 1^T g s.t. A@g <= M, g >= 0."""
    n = len(a)
    A = _build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    return np.clip(res.x, 0.0, None)


def _line_search(a, g, M_cur, n_fine=40):
    """Golden-section + bisection line search on blend fraction."""
    a_n = a / a.sum()
    g_n = g / g.sum()
    best_t = 0.0
    best_val = M_cur

    # Coarse grid
    ts = np.linspace(0.0, 1.0, 21)
    for t in ts:
        cand = (1.0 - t) * a_n + t * g_n
        cand = np.maximum(cand, 0.0)
        cs = cand.sum()
        if cs < 1e-15:
            continue
        cand /= cs
        val = float(np.convolve(cand, cand).max())
        if val < best_val:
            best_val = val
            best_t = t

    if best_t == 0.0:
        return None, M_cur

    # Refine with bisection
    lo = max(0.0, best_t - 0.06)
    hi = min(1.0, best_t + 0.06)
    for _ in range(n_fine):
        if hi - lo < 1e-10:
            break
        mid = 0.5 * (lo + hi)
        cand = (1.0 - mid) * a_n + mid * g_n
        cand = np.maximum(cand, 0.0)
        cs = cand.sum()
        if cs < 1e-15:
            hi = mid
            continue
        cand /= cs
        val = float(np.convolve(cand, cand).max())
        if val <= best_val:
            best_val = val
            best_t = mid
            hi = mid
        else:
            lo = mid

    cand = (1.0 - best_t) * a_n + best_t * g_n
    cand = np.maximum(cand, 0.0)
    cs = cand.sum()
    if cs < 1e-15:
        return None, M_cur
    return cand / cs, best_val


def _project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgradient_step(a, tol=1e-7, max_active=30):
    """One projected subgradient step on max(conv(a,a)). Returns new a or None."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(conv.max())

    # Active set
    thresh = M * (1.0 - tol)
    active = np.where(conv >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    if len(active) > max_active:
        order = np.argsort(conv[active])[::-1][:max_active]
        active = active[order]

    # Compute subgradient: (1/|S|) * sum_{k in S} grad_k where grad_k[m] = 2*a[k-m] if valid
    subgrad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for m in range(lo, hi):
            subgrad[m] += 2.0 * a[k - m]
    subgrad /= len(active)

    # Try several step sizes
    for alpha in [0.03, 0.02, 0.01, 0.005, 0.003, 0.001]:
        a_new = a - alpha * subgrad
        a_new = _project_simplex(a_new)
        if float(np.convolve(a_new, a_new).max()) < M * (1.0 - 1e-12):
            return a_new
    return None


def _peak_target_step(a, k_star, eps_list=None):
    """Reduce a[j] where j and k_star-j both contribute heavily to the peak."""
    n = len(a)
    if eps_list is None:
        eps_list = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003]

    conv = np.convolve(a, a)
    M = float(conv.max())

    # Contribution to (a*a)_{k_star}: a[j]*a[k_star-j]
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    j_range = np.arange(lo, hi)
    k_range = k_star - j_range
    contributions = a[j_range] * a[k_range]
    total = contributions.sum()
    if total < 1e-15:
        return None

    # Fractional contribution
    frac = contributions / total

    for eps in eps_list:
        a_new = a.copy()
        for i, j in enumerate(j_range):
            a_new[j] *= (1.0 - eps * frac[i] * 2.0)  # Reduce both endpoints
        a_new = np.maximum(a_new, 0.0)
        cs = a_new.sum()
        if cs < 1e-15:
            continue
        a_new /= cs
        if float(np.convolve(a_new, a_new).max()) < M * (1.0 - 1e-12):
            return a_new
    return None


def _optimize_core(a0, stop_time, seed=42, lp_timeout=3.0, max_stuck_lp=40, max_stuck_total=200):
    """Main optimization loop combining LP, subgradient, and peak targeting."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a /= s

    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())

    rng = np.random.default_rng(seed)
    stuck_lp = 0
    stuck_total = 0
    pert_level = 0

    while time.monotonic() < stop_time:
        conv = np.convolve(a, a)
        M = float(conv.max())
        k_star = int(np.argmax(conv))

        if M < best_M * (1.0 - 1e-13):
            best_M = M
            best_a = a.copy()
            stuck_lp = 0
            stuck_total = 0
        else:
            stuck_lp += 1
            stuck_total += 1

        # Check for major stagnation
        if stuck_total > max_stuck_total:
            # Large perturbation
            sigma = min(0.05 + 0.02 * (pert_level // 3), 0.3)
            xx = np.arange(n, dtype=np.float64)
            freq = rng.uniform(1.0, 4.0)
            phase = rng.uniform(0, 2 * np.pi)
            a *= np.exp(sigma * (rng.normal(0, 1, n) + 0.3 * np.sin(2 * np.pi * freq * xx / n + phase)))
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            stuck_total = 0
            stuck_lp = 0
            pert_level += 1
            if pert_level > 30:
                break
            continue

        made_move = False

        # LP best response (primary method)
        if stuck_lp < max_stuck_lp:
            g = _lp_solve(a, M * (1.0 + 1e-9), time_limit=lp_timeout)
            if g is not None and g.sum() > 1e-15:
                g /= g.sum()
                if float(np.max(np.abs(g - a))) > 1e-9:
                    cand, val = _line_search(a, g, M)
                    if cand is not None and val < M * (1.0 - 1e-11):
                        a = cand
                        made_move = True
                        stuck_lp = 0

        # Subgradient refinement (fast, no LP needed)
        if not made_move and stuck_lp < max_stuck_lp * 2:
            a_new = _subgradient_step(a)
            if a_new is not None:
                a = a_new
                made_move = True

        # Peak targeting

            a_new = _peak_target_step(a, k_star)
            if a_new is not None:
                a = a_new
                made_move = True

        # Small perturbation when stuck

            a *= np.exp(rng.normal(0, sigma, n))
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            stuck_lp += 1  # Count as non-progress for LP

        if time.monotonic() > stop_time - 5.0:
            break

    return best_a
```
