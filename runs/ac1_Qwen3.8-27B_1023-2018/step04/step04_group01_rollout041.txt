import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    return p

def _convolve(a, b):
    """Full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    fsize = next_fast_len(n)
    return irfft(rfft(a, fsize) * rfft(b, fsize), fsize)[:n]

def _autocorr(a):
    """Full autocorrelation (a * a)."""
    n = len(a)
    fsize = next_fast_len(2*n - 1)
    F = rfft(a, fsize)
    return irfft(F * F, fsize)[:2*n-1]

def _score(a):
    """Compute the AC1 upper bound score."""
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    b = _autocorr(a)
    M = float(b.max())
    return 2.0 * n * M / (S * S)

def _build_conv_matrix(a):
    """Build matrix A where (A @ g)[k] = (a * g)[k] for all k."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def _lp_direction(a, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = _build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0 and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_vals = np.nonzero(cond)[0]
    if len(rho_vals) == 0:
        return np.ones(n) / n
    r = rho_vals[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)

    def _update(seq, sc=None):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64)
        seq = np.clip(seq, 0, 1000)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        if sc is None:
            sc = _score(seq)
        if not math.isfinite(sc):
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-14:
            best_seq = seq.copy()
            best_score = exact

    # === Phase 1: Quick initialization exploration (15% time) ===
    p1_end = t0 + budget * 0.15
    if time.monotonic() < p1_end:
        inits = [parent]
        if n >= 32:
            inits.append(_make_init_cosine(n))
            inits.append(_make_init_gaussian(n))
            inits.append(_make_init_poly(n, 2))
            inits.append(_make_init_poly(n, 3))

        time_per_init = max(5.0, (p1_end - time.monotonic()) / max(1, len(inits)))
        for idx, init in enumerate(inits):
            sc = _score(init)
            if sc < best_score:
                _update(init, sc)
            if time.monotonic() > p1_end - 5:
                break

    # === Phase 2: LP optimization on best (40% time) ===
    p2_end = t0 + budget * 0.55
    if time.monotonic() < p2_end - 10:
        try:
            res, sc = _lp_optimize(best_seq, p2_end, seed=seed, max_iters=150)
            _update(res, sc)
        except Exception:
            pass

        # Second LP run with different seed if time remains
        if time.monotonic() < p2_end - 30 and best_score > 1.503:
            try:
                res, sc = _lp_optimize(best_seq, p2_end, seed=seed + 1000, max_iters=80)
                _update(res, sc)
            except Exception:
                pass

    # === Phase 3: Gradient refinement (15% time) ===
    p3_end = t0 + budget * 0.70
    if time.monotonic() < p3_end - 10 and best_score > 1.502:
        try:
            res, sc = _grad_optimize(best_seq, p3_end, seed=seed + 2000, max_iters=3000)
            _update(res, sc)
        except Exception:
            pass

    # === Phase 4: Intensive coordinate descent (20% time) ===
    p4_end = t0 + budget * 0.90
    if time.monotonic() < p4_end - 10:
        try:
            res, sc = _coord_polish(best_seq, p4_end, max_sweeps=150, seed=seed + 3000)
            _update(res, sc)
        except Exception:
            pass

    # === Phase 5: Try nearby lengths (5% time) ===
    p5_end = hard_deadline - 30
    if time.monotonic() < p5_end - 20 and best_score > 1.5025:
        other_ns = []
        for L in [n - 16, n - 8, n + 8, n + 16, n // 2, n * 2]:
            if 16 <= L <= 1024 and L != n and L not in other_ns:
                other_ns.append(L)
        other_ns.sort(key=lambda L: abs(L - n))

        time_per_len = max(10.0, (p5_end - time.monotonic()) / max(1, len(other_ns[:3])))
        for L in other_ns[:3]:
            if time.monotonic() > p5_end - 10:
                break
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
            init_l = np.clip(init_l, 0, None)
            if init_l.sum() < 0.01:
                init_l = np.ones(L) / L
            init_l = init_l / init_l.sum()

            stop_cd = min(p5_end - 5, time.monotonic() + time_per_len)
            try:
                res, sc = _coord_polish(init_l, stop_cd, max_sweeps=30, seed=seed + L)
                _update(res, sc)
            except Exception:
                pass

    # === Phase 6: Final polish ===
    final_end = hard_deadline - 10
    if time.monotonic() < final_end - 10:
        try:
            res, sc = _coord_polish(best_seq, final_end, max_sweeps=50, seed=99999)
            _update(res, sc)
        except Exception:
            pass

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
