                break
    
    return best, best_sc

def _coord_polish(a, stop_time, seed=0):
    """Coordinate descent polishing."""
    rng = np.random.default_rng(seed)
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return a.copy(), math.inf
    best = np.clip(a, 0, None).copy()
    best = best / best.sum() * S
    best_sc = _score_fast(best)
    
    order = rng.permutation(n)
    multipliers = np.array([0.0, 0.1, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0])
    no_imp_rounds = 0
    
    for rnd in range(50):
        if time.monotonic() > stop_time:
            break
        if no_imp_rounds > 5:
            break
        round_improved = False
        for idx in order:
            if time.monotonic() > stop_time:
                break
            orig = best[idx]
            if orig < 1e-12:
                # Try small positive values
                for v in (0.001, 0.01, 0.1 * S / n):
                    best[idx] = v
                    sc = _score_fast(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        round_improved = True
                        break
                else:
                    best[idx] = 0.0
                continue
            
            best_val = orig
            best_val_sc = best_sc
            for m in multipliers:
                nv = orig * m if m > 0 else 0.0
                best[idx] = nv
                if best.sum() < 0.01:
                    best[idx] = orig
                    continue
                sc = _score_fast(best)
                if sc < best_val_sc - 1e-14:
                    best_val = nv
                    best_val_sc = sc
            
            # Fine search around best
            if 0 < best_val < orig * 10:
                for frac in (0.8, 0.9, 1.1, 1.2, 0.7, 1.3):
                    nv = best_val * frac
                    best[idx] = nv
                    if best.sum() < 0.01:
                        best[idx] = best_val
                        continue
                    sc = _score_fast(best)
                    if sc < best_val_sc - 1e-14:
                        best_val_sc = sc
                        best_val = nv
            
            best[idx] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                round_improved = True
        
        if not round_improved:
            no_imp_rounds += 1
        else:
            no_imp_rounds = 0
        
        # Reshuffle
        order = rng.permutation(n)
    
    return best, best_sc

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0  # headroom
    
    parent = _get_parent()
    n_p = len(parent)
    
    best_seq = parent.copy()
    best_score = _score_fast(best_seq)
    if not math.isfinite(best_score):
        best_seq = np.ones(100)
        best_score = _score_fast(best_seq)
    
    def _update(a, sc):
        nonlocal best_seq, best_score
        if a is None or len(a) == 0:
            return
        a = np.asarray(a, dtype=np.float64)
        if not np.all(np.isfinite(a)):
            return
        a = np.clip(a, 0, 1000)
        if a.sum() < 0.01:
            return
        # Verify with exact evaluator
        try:
            exact = float(evaluate_sequence([float(x) for x in a]))
        except Exception:
            return
        if math.isfinite(exact) and exact < best_score - 1e-14:
            best_seq = a.copy()
            best_score = exact
    
    def _run_lp(init, n, stop, sd):
        try:
            res, sc = _lp_optimize(init, n, stop, sd)
            _update(res, sc)
        except Exception:
            pass
    
    # Phase 1: LP optimization at parent length (60% of time)
    phase1_end = t0 + budget * 0.55
    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n_p, dtype=np.float64)
    
    inits = []
    inits.append(parent.copy())
    
    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n_p)))
    
    # Linear ramp
    inits.append(np.linspace(0.1, 1.0, n_p))
    
    # Exponential
    inits.append(np.exp(-i_arr / (n_p * 0.3)) + 0.01)
    
    # Two-sided exponential
    center = (n_p - 1) / 2.0
    inits.append(np.exp(-0.8 * np.abs(i_arr - center) / (n_p * 0.2)) + 0.01)
    
    # Multi-frequency
    sup = np.ones(n_p) * 0.3
    for fr in (1, 2, 3, 5, 8):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.12 * np.cos(2.0 * math.pi * fr * i_arr / n_p + ph)
    inits.append(np.abs(sup) + 0.02)
    
    # Random
    inits.append(np.abs(rng_main.normal(0, 1, n_p)) + 0.01)
    
    # Parent + noise
    sig = 0.2 * max(1.0, float(np.mean(parent)))
    inits.append(np.abs(parent + rng_main.normal(0, sig, n_p)) + 0.01)
    
    # Quadratic
    inits.append(1.0 + np.sin(math.pi * i_arr / (n_p - 1)) ** 2)
    
    # Sawtooth
    inits.append(np.abs(2.0 * (i_arr / n_p - np.floor(i_arr / n_p + 0.5))) + 0.05)
    
    # Sparse-ish: blocks
    blocks = np.zeros(n_p)
    for bstart in range(0, n_p, max(1, n_p // 8)):
        b = min(bstart + n_p // 10, n_p)
        blocks[bstart:b] = 1.0
    inits.append(blocks + 0.05)
    
    n_inits = len(inits)
    time_per_init = max(15.0, (phase1_end - t0) / n_inits)
    
    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        _run_lp(init, n_p, stop, seed + 1000 * (idx + 1))
    
    # Phase 2: Continue LP on best (20% of time)
    phase2_end = t0 + budget * 0.75
    if time.monotonic() < phase2_end - 30.0:
        stop_p2 = min(phase2_end - 5.0, time.monotonic() + (phase2_end - time.monotonic()))
        _run_lp(best_seq.copy(), len(best_seq), stop_p2, seed + 777777)
    
    # Phase 3: Try different lengths (10% of time)
    phase3_end = t0 + budget * 0.85
    other_lengths = [350, 355, 360, 345, 400, 300]
    time_per_len = max(8.0, (phase3_end - max(phase2_end, time.monotonic())) / max(1, len(other_lengths)))
    for L in other_lengths:
        if time.monotonic() > phase3_end - 5.0:
            break
        stop = min(phase3_end - 3.0, time.monotonic() + time_per_len)
        init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
        _run_lp(init_l, L, stop, seed + 50000 + L)
    
    # Phase 4: Gradient polish (5% of time)
    phase4_end = t0 + budget * 0.90
    if time.monotonic() < phase4_end - 15.0:
        try:
            gp, gps = _grad_polish(best_seq, min(phase4_end - 5.0, time.monotonic() + (phase4_end - time.monotonic())), seed=1234)
            _update(gp, gps)
        except Exception:
            pass
