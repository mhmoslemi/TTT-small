import numpy as np
import time
from scipy.optimize import linprog


def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * len(a) * c.max() / (s * s))


def _build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a[j]
    return A


def _lp(A, M, tl=5.0):
    n = A.shape[1]
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(A.shape[0], M),
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': tl, 'presolve': True,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    x = np.clip(res.x, 0.0, None)
    if x.sum() < 1e-12:
        return None
    return x


def _blend(a, b, M_ref):
    """Line search for best (1-t)*a + t*b."""
    best_t, best_v = 0.0, M_ref
    for t in np.linspace(0.08, 1.0, 12):
        c = np.maximum((1 - t) * a + t * b, 0.0)
        cs = c.sum()
        if cs < 1e-12:
            continue
        c /= cs
        v = float(np.convolve(c, c).max())
        if v < best_v:
            best_v, best_t = v, t
    if best_t == 0.0:
        return None, M_ref
    lo, hi = max(0.0, best_t - 0.1), min(1.0, best_t + 0.1)
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        c = np.maximum((1 - mid) * a + mid * b, 0.0)
        cs = c.sum()
        if cs < 1e-12:
            hi = mid
            continue
        c /= cs
        v = float(np.convolve(c, c).max())
        if v < best_v:
            best_v, best_t = v, mid
            hi = mid
        else:
            lo = mid
    c = np.maximum((1 - best_t) * a + best_t * b, 0.0)
    c /= c.sum()
    return c, best_v


def _subgrad(a, k_star):
    """Subgradient step on active constraint k_star."""
    n = len(a)
    g = np.zeros(n)
    for j in range(max(0, k_star - n + 1), min(n, k_star + 1)):
        idx = k_star - j
        if 0 <= idx < n:
            g[j] += a[idx]
    g -= g.mean()
    gn = float(np.linalg.norm(g))
    if gn < 1e-14:
        return None
    cur = float(np.convolve(a, a).max())
    best_c = None
    for alpha in (0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005):
        c = np.maximum(a - alpha * g / gn, 1e-12)
        c /= c.sum()
        v = float(np.convolve(c, c).max())
        if v < cur - 1e-15:
            best_c = c
            break
    return best_c


def _perturb(a, rng, level):

    elif m == 2:

        a[lo:hi] *= (1.0 + sig * rng.normal(0, 1, hi - lo))
    elif m == 3:
        a *= 1.0 + sig * rng.uniform(-1.0, 1.0, n)
    elif m == 4:
        j1, j2 = rng.integers(0, n, 2)
        d = sig * a.mean()
        a[j1] = max(a[j1] + d, 1e-12)

    else:
        a[:n//2] *= np.exp(rng.normal(0, sig, n//2))
        a[n//2:] *= np.exp(rng.normal(0, sig, n - n//2))
    a = np.maximum(a, 1e-12)


def _lp_optimize(a0, stop_time, seed=42, max_stuck_total=500):
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0).copy()
    sn = a.sum()
    if sn < 1e-15:
        a = np.ones_like(a)
    a /= a.sum()
    n = len(a)
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())

    stuck = 0
    total_stuck = 0

    while time.monotonic() < stop_time:
        conv = np.convolve(a, a)
        M = float(conv.max())
        k_star = int(conv.argmax())

        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
            total_stuck += 1

        if total_stuck > max_stuck_total:
            break

        if stuck > 30:
            _perturb(a, rng, stuck // 30)
            a /= a.sum()
            stuck = 0
            continue

        A = _build_A(a)
        moved = False

        # Primary LP at current M
        g = _lp(A, M * (1.0 + 1e-10), tl=4.0)
        if g is not None:
            g /= g.sum()
            cand, val = _blend(a, g, M)
            if cand is not None and val < M - 1e-14:
                a = cand
                moved = True

        # Try reduced M
        if not moved:
            for frac in (0.999, 0.997, 0.994):
                g2 = _lp(A, M * frac, tl=3.0)
                if g2 is not None:
                    g2 /= g2.sum()
                    cand, val = _blend(a, g2, M)
                    if cand is not None and val < M - 1e-14:
                        a = cand
                        moved = True
                        break

        # Subgradient
        if not moved:
            cand = _subgrad(a, k_star)
            if cand is not None:
                a = cand
                moved = True

        # Kick
        if not moved:
            a *= np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 1e-12)
            a /= a.sum()

    return best_a

def _blend_fast(aa, ab, bb, a, b, M_ref):
    """Fast line search using precomputed convolutions.
    aa = conv(a,a), ab = conv(a,b), bb = conv(b,b)
    """
    best_t, best_v = 0.0, M_ref
    # Precompute the quadratic coefficients for each conv index k
    # (1-t)^2 * aa[k] + 2t(1-t) * ab[k] + t^2 * bb[k]
    # = aa[k](1-2t+t^2) + ab[k](2t-2t^2) + bb[k]t^2
    # = aa[k] + t(2ab[k]-2aa[k]) + t^2(aa[k]-2ab[k]+bb[k])
    c0 = aa.copy()  # constant term
    c1 = 2.0 * (ab - aa)  # linear term
    c2 = aa - 2.0 * ab + bb  # quadratic term
    
    for t in np.linspace(0.08, 1.0, 12):
        vals = c0 + t * c1 + t * t * c2
        v = float(vals.max())
        if v < best_v:
            best_v, best_t = v, t
    
    if best_t == 0.0:
        return None, M_ref
    
    # Bisection on the max of c0 + t*c1 + t^2*c2
    lo, hi = max(0.0, best_t - 0.1), min(1.0, best_t + 0.1)
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = c0 + mid * c1 + mid * mid * c2
        v = float(vals.max())
        if v < best_v:
            best_v, best_t = v, mid
            hi = mid
        else:
            lo = mid
    
    # Construct the blended sequence
    c = np.maximum((1 - best_t) * a + best_t * b, 0.0)
    c /= c.sum()
    return c, best_v
