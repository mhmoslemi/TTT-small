import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(414, dtype=np.float64)
    if p.size == 0:
        p = np.ones(414, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(414, dtype=np.float64)
    return p.copy()

def _fc(a):
    n = len(a)
    fs = next_fast_len(2 * n - 1)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _sc(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _fc(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _solve_lp(f, n, M, tlim=2.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    try:
        r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                    bounds=[(0.0, None)] * n, method='highs',
                    options={'presolve': True, 'time_limit': tlim})
    except Exception:
        return None
    if r is None or r.status != 0 or r.x is None:
        return None
    g = np.maximum(r.x, 0.0)
    return g if g.sum() > 1e-14 else None

def _solve_lp_min(f, n, tlim=2.0):
    """Minimize M' s.t. A_f @ g <= M', sum(g)=1, g>=0."""
    A = _build_A(f, n)
    nvar = n + 1
    c = np.zeros(nvar)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, nvar))
    A_ub[:, :n] = A
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=[(0.0, None)] * nvar, method='highs',
                    options={'presolve': True, 'time_limit': tlim})
    except Exception:
        return None, None
    if r is None or r.status != 0 or r.x is None:
        return None, None
    g = np.maximum(r.x[:n], 0.0)
    gp = g.sum()
    if gp < 1e-14:
        return None, None
    return g / gp, float(r.x[-1])

def _line_search(f, g, sc0, deadline):
    bt, bs = 0.0, sc0
    for t in (0.04, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.65, 0.8, 0.9, 1.0):
        if time.monotonic() > deadline:
            break
        v = (1.0 - t) * f + t * g
        s = _sc(v)
        if s < bs:
            bs, bt = s, t
    if 0.02 < bt < 0.98:
        lo = max(0.0, bt - 0.08)
        hi = min(1.0, bt + 0.08)
        for t in np.linspace(lo, hi, 11):
            if time.monotonic() > deadline:
                break
            v = (1.0 - t) * f + t * g
            s = _sc(v)
            if s < bs:
                bs, bt = s, t
    return bt, bs

def _lp_opt(f0, n, stop, seed):
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.copy(), 0.0)
    f /= max(float(f.sum()), 1e-10)
    bf, bs = f.copy(), _sc(f)
    stall = 0
    dM = 0.0
    use_alt = False

    for it in range(50000):
        if time.monotonic() > stop - 3:
            break
        b = _fc(f)
        M = float(b.max())

        if use_alt:
            g, Malt = _solve_lp_min(f, n, tlim=2.0)
            use_alt = stall < 10
        else:
            g = _solve_lp(f, n, M * (1.0 + dM), tlim=2.0)
            use_alt = stall > 20 and stall < 40

        if g is None:
            stall += 5
        else:
            d = np.linalg.norm(g - f)
            if d < 1e-10:
                stall += 2
            else:
                tb, sb = _line_search(f, g, bs, stop - 3)
                if sb < bs - 1e-14:
                    fn = np.maximum((1.0 - tb) * f + tb * g, 0.0)
                    fn /= max(float(fn.sum()), 1e-10)
                    f = fn
                    bf = fn.copy()
                    bs = sb
                    stall = 0
                    dM = max(0.0, dM * 0.8)
                    continue
                stall += 1

        if stall > 40:
            amp = 0.01 + 0.004 * min(float(stall) / 40.0, 4.0)
            f = np.maximum(f + rng.normal(0, amp, n), 0.0)
            f /= max(float(f.sum()), 1e-10)
            dM = min(0.08, 1e-5 * stall)
            stall = 0
            if bs > 1.51 and it > 200:
                fr = rng.dirichlet(np.full(n, 0.6))
                f = 0.5 * f + 0.5 * fr
                f /= f.sum()

    return bf, bs

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _sg_opt(f0, n, stop, seed):
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.copy(), 0.0)
    f /= max(float(f.sum()), 1e-10)
    bf, bs = f.copy(), _sc(f)
    step = 0.015
    mom = np.zeros(n)
    mu = 0.7
    no_imp = 0

    for it in range(200000):
        if time.monotonic() > stop - 3:
            break
        b = _fc(f)
        M = float(b.max())
        eps = max(1e-14, 1e-10 * M)
        act = np.flatnonzero(b >= M - eps)
        sg = np.zeros(n)
        for k in act:
            jlo = max(0, k - n + 1)
            jhi = min(k, n - 1)
            idx_y = k - np.arange(jlo, jhi + 1)
            sg[jlo:jhi + 1] += 2.0 * f[idx_y]
        val = max(len(act), 1)
        sg /= val
        sg -= float(sg.mean())
        ns = float(np.linalg.norm(sg))
        if ns < 1e-15:
            break
        mom = mu * mom - step * sg / ns
        fn = _proj_simplex(f + mom)
        sc = _sc(fn)
        if sc < bs - 1e-14:
            f = fn
            bf = f.copy()
            bs = sc
            step = min(0.15, step * 1.01)
            no_imp = 0
        else:
            step *= 0.94
            no_imp += 1
            if no_imp > 200:
                f = _proj_simplex(f + rng.normal(0, 0.015, n))
                mom[:] = 0
                step = 0.015
                no_imp = 0

    return bf, bs

def _cd_opt(f0, stop, seed, max_passes=300):
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = np.maximum(f0.copy(), 0.0)
    f /= max(float(f.sum()), 1e-10)
    bf, bs = f.copy(), _sc(f)
    multipliers = [0.0, 0.25, 0.5, 0.7, 0.85, 1.15, 1.35, 1.6, 2.0]

    for ep in range(max_passes):
        if time.monotonic() > stop - 3:
            break
        imp = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() > stop - 3:
                break
            ov = f[i]
            if ov < 1e-13:
                continue
            bv, bvs = ov, bs
            for m in multipliers:
                f[i] = ov * m
                sc = _sc(f)
                if sc < bvs:
                    bvs, bv = sc, ov * m
            f[i] = bv
            if bvs < bs - 1e-15:
                bs = bvs
                bf = f.copy()
                imp = True

        imp2 = False
        n2d = min(80, n // 3)
        for _ in range(n2d):
            if time.monotonic() > stop - 3:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-13 and oj < 1e-13:
                continue
            for mi in (0.4, 0.7, 1.4):
                for mj in (0.4, 0.7, 1.4):
                    f[i], f[j] = oi * mi, oj * mj
                    sc = _sc(f)
                    if sc < bs - 1e-15:
                        bs = sc
                        bf = f.copy()
                        imp2 = True
            f[i], f[j] = oi, oj

        if not imp and not imp2:
            f = np.maximum(f + rng.normal(0, 0.003, n), 0.0)
            f /= max(float(f.sum()), 1e-10)
            if ep > 100:
                break

    return bf, bs
