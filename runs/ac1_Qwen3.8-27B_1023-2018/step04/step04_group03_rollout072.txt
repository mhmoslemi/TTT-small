def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-12:
       return None
    return a / s

def _fast_conv(a):
    n = len(a)
    size = next_fast_len(2*n - 1)
    F = rfft(a, size)
    b = irfft(F*F, size)[:2*n-1]
    if b.size == 0: return np.array([])
    return np.maximum(b, 0.0)

def _fast_score(a):
   n=len(a); s=float(a.sum())
   if s < 0.01: return math.inf
   b=_fast_conv(a); m=float(b.max()) if b.size else math.inf
   if not math.isfinite(m): return math.inf
   return 2.0*n*m/(s*s)

def _exact_score(a):
   n=len(a); s=float(np.sum(a))
   if s < 0.01: return math.inf
   b=np.convolve(a,a)
   m=float(np.max(b))
   if not math.isfinite(m): return math.inf

import numpy as np

from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals().get("height_sequence_1", None)
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(416, dtype=np.float64)
    if p.size == 0:

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(416, dtype=np.float64)
    return p


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-12:
        return None
    return a / s


def _fast_conv(a):
    n = len(a)
    size = next_fast_len(2 * n - 1)
    F = rfft(a, size)
    b = irfft(F * F, size)[:2 * n - 1]
    if b.size == 0:
        return np.array([])
    return np.maximum(b, 0.0)


def _fast_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _fast_conv(a)
    if b.size == 0:
        return math.inf
    m = float(b.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _exact_score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(np.max(b))
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _top_indices(b, K):
    L = len(b)
    K = int(max(1, K))
    if K >= L:
        return np.arange(L)
    return np.argpartition(b, -K)[-K:]


def _build_active_matrix(a, active):
    n = len(a)
    A = np.zeros((len(active), n), dtype=np.float64)
    idx = np.arange(n)
    for j, k in enumerate(active):
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i1 > i0:
            A[j, i0:i1] = a[k - idx[i0:i1]]
    return A


def _lp_gradient_direction(a, b, K=80, timeout=4.0):
    n = len(a)
    M = float(b.max())
    if M <= 1e-300:
        return None
    active = _top_indices(b, K)
    A = _build_active_matrix(a, active)
    if A.size == 0 or not np.any(A):
        return None

    var_n = n + 1
    c = np.zeros(var_n, dtype=np.float64)
    c[-1] = -1.0

    A_ub = np.column_stack((A, np.ones(A.shape[0], dtype=np.float64)))
    b_ub = np.zeros(A.shape[0], dtype=np.float64)

    A_eq = np.zeros((1, var_n), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0], dtype=np.float64)

    ub_d = max(10.0 / max(1, n), 10.0 * float(np.max(a)), 1e-6)
    bounds = []
    for x in a:
        lb = -0.9 * x if x > 1e-14 else 0.0
        bounds.append((float(lb), float(ub_d)))
    bounds.append((-1.0, 1.0))

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"time_limit": max(0.5, float(timeout)), "presolve": True},
        )
    except Exception:
        return None

    if res.status != 0 or res.x is None:
        return None
    d = np.asarray(res.x[:n], dtype=np.float64)
    d -= float(d.mean())
    mabs = float(np.max(np.abs(d)))
    if mabs <= 1e-14:
        return None
    d /= mabs
    return d

def _apply_gradient_step(a, d):

    mabs = float(np.max(np.abs(d)))
    if mabs <= 1e-14:

    neg = d < -1e-14

    if tmax <= 1e-8:

    ts = ts[(ts > 1e-9) & (ts <= tmax + 1e-12)]

    best_an = None

            sc = _fast_score(an)
            if sc < best_sc:

                best_an = an

        return a, math.inf

    if best_t > 0.0:
        lo = max(0.0, 0.4 * best_t)
        hi = min(1.0, 1.6 * best_t)
        if hi <= lo:
            hi = min(1.0, best_t + 0.05)
        if lo < hi:
            for t in np.linspace(lo, hi, 24):
                t = float(t)
                an = (1.0 - t) * a + t * g
                an = np.clip(an, 0.0, None)
                s = float(an.sum())
                if s < 1e-9:
                    continue
                an = an / s
                sc = _fast_score(an)
                if sc < best_sc:
                    best_sc = sc
                    best_an = an

    return best_an, best_sc


def _run_lp_phase(a0, stop_time, seed):
    a = _normalize(a0)
    if a is None:
        a = np.ones(416, dtype=np.float64) / 416.0
    n = len(a)

    best = a.copy()
    best_sc = _fast_score(a)

    rng = np.random.default_rng(seed)
    Klist = [20, 40, 80, 160, 320, 512]
    ki = 0
    stall = 0

    while time.monotonic() < stop_time - 3.0:
        b = _fast_conv(a)
        improved = False

        for _ in range(3):
            if time.monotonic() > stop_time - 3.0:
                break
            K = Klist[ki % len(Klist)]
            ki += 1
            d = _lp_gradient_direction(a, b, K=K, timeout=3.5)
            if d is not None:
                an, sc = _apply_gradient_step(a, d)
                if an is not None and math.isfinite(sc) and sc < best_sc - 1e-13:
                    best = an.copy()
                    best_sc = sc
                    a = an
                    improved = True
                    stall = 0
                    break

        if not improved and time.monotonic() < stop_time - 8.0:
            g = _lp_nonnegative_direction(a, b, K=160, timeout=3.5)
            if g is not None:
                an, sc = _apply_nonnegative_step(a, g)
                if an is not None and math.isfinite(sc) and sc < best_sc - 1e-13:
                    best = an.copy()
                    best_sc = sc
                    a = an
                    stall = 0
                    continue

        stall += 1
        if stall % 5 == 0:
            sigma = 0.01 if stall < 20 else 0.005
            a *= np.exp(rng.normal(0.0, sigma, n))
            nnorm = _normalize(a)
            if nnorm is None:
                a = best.copy()
                break
            a = nnorm
            sc = _fast_score(a)
            if math.isfinite(sc) and sc < best_sc - 1e-13:
                best = a.copy()
                best_sc = sc
            stall = 0

        if stall > 80:
            break

    return best


def _coord_objective(xs, base, d, special, o):
    xs = np.asarray(xs, dtype=np.float64).ravel()
    out = np.full(xs.shape, math.inf, dtype=np.float64)
    if xs.size == 0:
        return out

    xs = np.where(np.isfinite(xs), xs, -1.0)
    valid = xs >= 0.0
    valid_idx = np.flatnonzero(valid)
    if valid_idx.size == 0:
        return out

    xv = xs[valid_idx]
    denom = 1.0 - o + xv
    keep = np.flatnonzero(denom > 1e-12)
    if keep.size == 0:
        return out

    valid_idx = valid_idx[keep]
    xv = xv[keep]
    denom = denom[keep]

    b = base[None, :] + (xv[None, :] - o) * d[None, :]
    if special >= 0:
        b[:, special] += (xv * xv - o * o)[None, :]

    peak = b.max(axis=1)
    vals = peak / (denom * denom)

    good = np.isfinite(vals)
    if np.any(good):
        out[valid_idx[good]] = vals[good]

    return out


def _refine_coordinate(x0, base, d, special, o):
    n = (len(base) + 1) // 2

    def eval_xs(xs):
        objs = _coord_objective(xs, base, d, special, o)
        if not np.any(np.isfinite(objs)):
            return x0, math.inf
        i = int(np.argmin(objs))
        return float(xs[i]), float(objs[i])

    if x0 <= 0.0:
        hi = max(10.0 / max(1, n), 5e-4)
        hi = min(1.0, hi)
        x0, best = eval_xs(np.linspace(0.0, hi, 80))
        if x0 >= 0.99 * hi and hi < 1.0:
            hi2 = min(1.0, max(2.0 * hi, 20.0 / max(1, n)))
            xs = np.linspace(0.0, hi2, 80) if x0 <= 0 else np.linspace(0.5 * max(x0, 1e-12), hi2, 80)
            x1, b1 = eval_xs(xs)
            if b1 < best:
                x0, best = x1, b1
    else:
        lo = max(1e-15, 0.45 * x0)
        hi = min(1.0, 2.2 * x0)
        if hi <= lo:
            hi = min(1.0, lo * 3.0 + 1e-12)
        xs = np.geomspace(lo, max(hi, lo * 1.01), 90)
        x0, best = eval_xs(xs)

    if x0 > 0.0:
        lo = max(1e-15, 0.955 * x0)
        hi = min(1.0, 1.045 * x0)
        if hi > lo:
            x1, b1 = eval_xs(np.linspace(lo, hi, 60))
            if b1 < best:
                x0, best = x1, b1

        lo = max(1e-15, 0.9955 * x0)
        hi = min(1.0, 1.0045 * x0)
        if hi > lo:
            x2, b2 = eval_xs(np.linspace(lo, hi, 50))
            if b2 < best:
                x0, best = x2, b2
    else:
        hi = min(1.0, max(1e-6, 2.0 / max(1, n)))
        x1, b1 = eval_xs(np.linspace(0.0, hi, 50))
        if b1 < best:
            x0, best = x1, b1

    return x0, best


def _run_coordinate_phase(a0, stop_time, seed):
    a = _normalize(a0)
    if a is None:
        a = np.ones(416, dtype=np.float64) / 416.0
    n = len(a)
    L = 2 * n - 1

    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(416, dtype=np.float64) / 416.0
        S = 1.0
    else:
        a = a / S

    base = np.convolve(a, a)
    base = np.maximum(base, 0.0)
    best = a.copy()
    best_obj = float(base.max())

    rng = np.random.default_rng(seed)

    coarse_factors = np.array(
        [
            0.0,
            0.05,
            0.15,
            0.25,
            0.35,
            0.5,
            0.65,
            0.8,
            0.9,
            0.95,
            0.98,
            1.0,
            1.02,
            1.05,
            1.1,
            1.25,
            1.5,
            2.0,
            3.0,
            5.0,
            8.0,
        ],
        dtype=np.float64,
    )

    no_improve_rounds = 0
    rounds = 0

    while time.monotonic() < stop_time - 3.0:
        improved_round = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            i = int(i)
            o = float(a[i])

            d = np.zeros(L, dtype=np.float64)
            d[i : i + n] = 2.0 * a[:n]
            special = 2 * i
            if 0 <= special < L:
                d[special] = 0.0

            cur = float(base.max())

            if o < 1e-8:
                scale = 1.0 / float(n)
                xs = np.array(
                    [
                        0.0,
                        scale * 0.05,
                        scale * 0.15,
                        scale * 0.35,
                        scale * 0.75,
                        scale * 1.25,
                        scale * 2.0,
                        scale * 3.5,
                        scale * 6.0,
                        scale * 12.0,
                    ],
                    dtype=np.float64,
                )
            else:
                xs = np.clip(o * coarse_factors, 0.0, 1.0)

            objs = _coord_objective(xs, base, d, special, o)
            if not np.any(np.isfinite(objs)):
                continue

            amin = float(np.min(objs))
            aix = int(np.argmin(objs))
            ax = float(xs[aix])

            if amin < cur - 1e-14:
                rx, robj = _refine_coordinate(ax, base, d, special, o)
                if robj < amin:
                    amin = robj
                    ax = rx

            if amin < cur - 1e-12 and ax >= 0.0:
                dx = float(ax - o)
                if dx != 0.0:
                    # Exact O(n) update of the convolution under changing a[i].
                    base[i : i + n] += 2.0 * dx * a[:n]
                    if 0 <= special < L:
                        base[special] += dx * dx

                    ax = max(0.0, min(1.0, ax))
                    a[i] = ax

                    S = 1.0 - o + ax
                    if S <= 1e-9:
                        a = np.clip(a, 0.0, None)
                        S2 = float(a.sum())
                        if S2 < 1e-12:
                            a = best.copy()
                            base = np.convolve(a, a)
                            best_obj = float(base.max())
                            break
                        a = a / S2
                        base = np.convolve(a, a)
                        base = np.maximum(base, 0.0)
                    else:
                        scale = 1.0 / S
                        a *= scale
                        base *= scale * scale
                        base = np.maximum(base, 0.0)

                    cur2 = float(base.max())
                    if cur2 < best_obj - 1e-14:
                        best_obj = cur2
                        best = a.copy()
                    improved_round = True

        rounds += 1

        if time.monotonic() < stop_time - 12.0:
            a = np.clip(a, 0.0, None)
            S = float(a.sum())
            if S < 1e-12:
                a = best.copy()
                S = 1.0
            else:
                a = a / S
            base = np.convolve(a, a)
            base = np.maximum(base, 0.0)
            cur2 = float(base.max())
            if cur2 < best_obj - 1e-14:
                best_obj = cur2
                best = a.copy()

            if cur2 > best_obj * 1.08:
                a = best.copy()
                base = np.convolve(a, a)
                base = np.maximum(base, 0.0)
                improved_round = False

        if not improved_round:
            no_improve_rounds += 1
            if no_improve_rounds >= 3:
                if rounds > 80 and no_improve_rounds >= 10:
                    break
                sigma = 0.008 if no_improve_rounds < 6 else 0.004
                a *= np.exp(rng.normal(0.0, sigma, n))
                a = np.clip(a, 0.0, None)
                S = float(a.sum())
                if S < 1e-12:
                    a = best.copy()
                    S = 1.0
                else:
                    a = a / S
                base = np.convolve(a, a)
                base = np.maximum(base, 0.0)
                cur2 = float(base.max())
                if cur2 < best_obj - 1e-14:
                    best_obj = cur2
                    best = a.copy()
                if cur2 > best_obj * 1.12:
                    a = best.copy()
                    base = np.convolve(a, a)
                    base = np.maximum(base, 0.0)
                no_improve_rounds = 0
        else:
            no_improve_rounds = 0

    return best


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(20.0, min(float(budget_s), 1300.0))
    except Exception:
        budget = 1300.0

    # Leave headroom for final validation and exit.
    deadline = t0 + budget - 30.0
    if deadline < t0 + 30.0:
        deadline = t0 + max(10.0, 0.9 * budget)

    parent = _get_parent()
    a0 = _normalize(parent)
    if a0 is None:
        a0 = np.ones(416, dtype=np.float64) / 416.0

    best = a0.copy()
    best_score = _exact_score(best)
    if not math.isfinite(best_score):
        best = np.ones(416, dtype=np.float64) / 416.0
        best_score = _exact_score(best)
        if not math.isfinite(best_score):
            best = np.ones(1, dtype=np.float64)
            best_score = 2.0

    def consider(candidate):
        nonlocal best, best_score
        y = _normalize(candidate)
        if y is None:
            return
        sc = _exact_score(y)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best = y.copy()
            best_score = sc

    rng = np.random.default_rng(seed)
    cur = best.copy()
    lp_counter = 0
    next_lp = t0 + 45.0

    while time.monotonic() < deadline:
        if time.monotonic() >= next_lp and time.monotonic() < deadline - 55.0:
            lp_stop = min(deadline - 55.0, time.monotonic() + 55.0)
            prev_best = best_score
            if time.monotonic() < lp_stop - 10.0:
                cand = _run_lp_phase(cur, lp_stop, seed + 100000 * (lp_counter + 1))
                consider(cand)
                cur = best.copy()
            lp_counter += 1
            if best_score < prev_best - 1e-12:
                next_lp = time.monotonic() + 25.0
            else:
                next_lp = time.monotonic() + 85.0 + rng.uniform(0.0, 30.0)

        if time.monotonic() < deadline - 22.0:
            coord_stop = min(deadline - 22.0, time.monotonic() + 28.0)
            if time.monotonic() < coord_stop - 5.0:
                cand = _run_coordinate_phase(cur, coord_stop, seed + 700000 + lp_counter)
                consider(cand)
                cur = best.copy()

    # Final precise polishing attempt.
    if time.monotonic() < deadline - 5.0:
        cand = _run_coordinate_phase(best, deadline - 5.0, seed + 987654321)
        consider(cand)

    # Use a modest positive scale. The objective is scale-invariant.
    out = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    out = np.where(np.isfinite(out), out, 0.0) * 10.0
    if float(out.sum()) < 0.01:
        out = (np.ones(416, dtype=np.float64) / 416.0) * 10.0

    try:
        final_list = [float(x) for x in out]
        fs = evaluate_sequence(final_list)
        if not math.isfinite(fs):
            final_list = [float(x) for x in best]
            fs = evaluate_sequence(final_list)
            if not math.isfinite(fs):
                return [10.0]
    except Exception:
        return [float(x) for x in best]

    return final_list
