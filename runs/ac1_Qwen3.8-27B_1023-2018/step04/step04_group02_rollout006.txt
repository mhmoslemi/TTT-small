import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _fft_conv(a):
    """Fast self-convolution via FFT."""
    n = len(a)
    if n <= 256:
        return np.convolve(a, a, mode='full')
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(np.sum(a))

    m = float(np.max(b))
    return 2.0 * n * m / (s * s)


def _build_conv_matrix(f, n):
    """A[k, j] = f[k-j] if 0<=k-j<n else 0. Shape: (2n-1, n)"""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=10.0):
    """max sum(g) s.t. A@g <= M, g >= 0 where A = conv matrix of f."""
    A = _build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': timeout, 'presolve': True,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    return g if g.sum() > 1e-15 else None
