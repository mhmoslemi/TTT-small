import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():

def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_conv_matrix(f, n):
    """Build the convolution matrix A where (A@x)[k] = (f*x)[k]."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_direction(f, n, M, timeout=5.0):
    """Find LP direction: max sum(g) s.t. (f*g) <= M, g >= 0."""
    A = _build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _subgrad_optimize(f0, n, stop_time, seed, max_iters=50000):
    """Subgradient optimization with momentum."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = _score(f)
    step = 0.03
    no_imp = 0
    momentum = np.zeros(n)
    mu = 0.6

    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break

        # Subgradient of max(conv(f,f)) / (sum f)^2 * 2n
        b = _fft_conv(f)
        m = float(b.max())
        S = float(np.sum(f))
        eps = max(1e-15, 1e-10 * m)
        active = np.flatnonzero(b >= m - eps)
        
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo <= i_hi:
                idx = k - np.arange(i_lo, i_hi + 1)
                grad[i_lo:i_hi + 1] += 2.0 * f[idx]
        if len(active) > 0:
            grad /= len(active)
        
        # Full subgradient of score = 2n * m / S^2
        # d(score)/d f = 2n * (grad_m * S - 2*m) / S^3
        f_sub = 2.0 * n * (grad * S - 2.0 * m) / (S ** 3)
        f_sub -= np.mean(f_sub)  # Remove mean to preserve sum
        
        norm_sg = np.linalg.norm(f_sub)
        if norm_sg < 1e-16:
            break

        momentum = mu * momentum - step * f_sub / norm_sg
        fn = np.clip(f + momentum, 0, None)
        s = fn.sum()
        if s < 0.01:
            step *= 0.5
            momentum *= 0.5
            continue
        fn = fn / s
        sc = _score(fn)

        if sc < best_sc - 1e-14:
            f = fn
            best_f = f.copy()
            best_sc = sc
            step = min(0.3, step * 1.02)
            no_imp = 0
        else:
            step *= 0.92
            no_imp += 1
            if no_imp > 800:
                f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                momentum = np.zeros(n)
                step = 0.03
                no_imp = 0

    return best_f, best_sc

def _coord_descent(f0, stop_time, seed, max_epochs=100):
    """Coordinate descent with random restarts."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    n = len(f)
    best_sc = _score(f)
    best_f = f.copy()

    for epoch in range(max_epochs):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            if orig < 1e-14:
                continue
            best_v = orig
            best_v_sc = best_sc
            for mult in [0.1, 0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.0]:

                    continue
                sc = _score(f)
                if sc < best_v_sc - 1e-16:
                    best_v = f[i]
                    best_v_sc = sc
            f[i] = best_v
            if best_v_sc < best_sc - 1e-15:
                best_sc = best_v_sc
                best_f = f.copy()
                improved = True

        if not improved:
            break

    return best_f, best_sc

def _smooth_score_grad(f, tau=None):
    """Compute smooth approximation of score and its gradient."""
    n = len(f)
    S = float(np.sum(f))
    b = _fft_conv(f)
    if tau is None:
        tau = max(1e-4 * max(b.max(), 1e-10), 1e-10)
    
    # Softmax of b/tau
    b_scaled = b / tau
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    sum_exp = np.sum(exp_b)
    softmax = exp_b / sum_exp
    
    m_smooth = tau * (np.log(sum_exp) + b_max)
    
    # Score
    score = 2.0 * n * m_smooth / (S * S)
    
    # Gradient
    grad = np.zeros(n)
    for k in range(len(b)):
        if softmax[k] < 1e-15:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            idx = k - np.arange(i_lo, i_hi + 1)
            grad[i_lo:i_hi + 1] += 2.0 * f[idx] * softmax[k]
    
    grad = 2.0 * n * (grad / (S * S) - 2.0 * m_smooth / (S * S * S))
    
    return score, grad, m_smooth

Actually, one more important thought: the LP direction. Let me verify the formulation.

Given current f, we want to find g >= 0 such that (f * g)[k] <= M for all k, where M = max(f*f). This g represents a "feasible direction" - if we move from f towards g, the cross term (f*g) is bounded by M, which means the new convolution (f_new * f_new) won't increase at the peaks (for small step sizes).

The LP: max sum(g) s.t. A_f * g <= M * 1, g >= 0, where (A_f * g)[k] = (f*g)[k].

This LP is always feasible (g = 0 is feasible) and bounded (sum(g) <= n * M / min(f) if min(f) > 0, or just bounded by the constraints).

The optimal g is an extreme point of the polytope, which has at most 2n-1 active constraints. This means g has a specific structure: it's determined by which constraints are tight.

Now, the direction from f to g is d = g - f (after normalization). The new point is f + t*d for t in [0,1].

The new convolution:
(f + t*d)(f + t*d) = (f*f) + 2t(f*d) + t^2(d*d)

At the peak lags: (f*f)[k] + 2t(f*d)[k] + t^2(d*d)[k]
= (f*f)[k] + 2t((f*g)[k] - (f*f)[k]) + t^2((d*d)[k])

For the active lags where (f*f)[k] = M:
= M + 2t((f*g)[k] - M) + t^2((d*d)[k])
= M - 2t(M - (f*g)[k]) + t^2((d*d)[k])

Since (f*g)[k] <= M, the linear term is <= 0, which is good. But the quadratic term t^2*(d*d)[k] could be positive.

For small t, the linear term dominates and the peak decreases. For large t, the quadratic term might increase the peak.

This is why the line search is important: we need to find the optimal t.
