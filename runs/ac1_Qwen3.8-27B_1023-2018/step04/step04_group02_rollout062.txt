```python
import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len

def _conv(a):
    n = len(a)
    if n <= 2048:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """A[i,j] = f[i-j] if 0<=i-j<n, else 0; shape (2n-1, n)"""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    idx = np.arange(n)
    for i in range(2*n-1):
        j_lo = max(0, i - n + 1)
        j_hi = min(n - 1, i)
        if j_lo <= j_hi:
            j_range = np.arange(j_lo, j_hi + 1)
            A[i, j_range] = f[i - j_range]
    return A

def _lp_dir(f, n, M, timeout=5.0):
    """Find g >= 0 maximizing sum(g) s.t. f*g <= M everywhere."""
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
    except Exception:
        return None
    if res.status not in (0, 1, 2, 3) or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g
```

This means changing a[i] affects n convolution entries (from j=i to j=i+n-1), specifically:
- (a*a)[i] += 2δ*a[0]
- (a*a)[i+1] += 2δ*a[1]
- ...
- (a*a)[2i] += 2δ*a[i] + δ²
- ...
- (a*a)[i+n-1] += 2δ*a[n-1]

This is very useful for CD! Given the current convolution b = a*a and the current value a[i], we can compute the new max convolution in O(n) time by just updating the n affected entries.

Actually, for CD we need to find the best δ for each i. The new max is:
max(max(b[j] for j with Δ=0), max(b[j] + 2δ*a[j-i] for affected j ≠ 2i), b[2i] + 2δ*a[i] + δ²)

The first term is the max over unaffected entries, which is some constant C (the max of b at entries not in [i, i+n-1]).

For the affected entries, we need to find the δ that minimizes the max. This is a 1D problem.

For each affected j (j ≠ 2i, i ≤ j ≤ i+n-1):
  b[j] + 2δ*a[j-i]

This is linear in δ (since a[j-i] ≥ 0, it's non-decreasing in δ if a[j-i] > 0).

For j = 2i: b[2i] + 2δ*a[i] + δ² (quadratic in δ, always increasing for δ > 0 since a[i] ≥ 0).

So the new max is the max of several non-decreasing functions of δ (for δ > 0) or could decrease for δ < 0.

For δ < 0 (decreasing a[i]):
  b[j] + 2δ*a[j-i] decreases if a[j-i] > 0
  b[2i] + 2δ*a[i] + δ² could go either way

The optimal δ is where the maximum is minimized. This is a piecewise-linear-plus-quadratic optimization in 1D.

For a simpler approach: just try a grid of δ values (relative to a[i]) and pick the best. This is what the parent program does but with multipliers.

For each i, try δ = a[i] * (m - 1) for m in a set of multipliers {0, 0.1, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 5.0, 10.0}

```python
def _cd_step_inplace(a, b, i, multipliers):
    """Try changing a[i], return best new value and updated b."""
    n = len(a)
    orig = a[i]
    best_v = orig
    best_max = b.max()
    
    # Precompute which entries are affected
    # Affected: j from i to i+n-1 (clamped to [0, 2n-2])
    j_lo = i
    j_hi = min(2*n-2, i+n-1)
    
    # Unaffected max
    # b[j] for j < j_lo or j > j_hi
    unaffected_max = 0.0
    if j_lo > 0:
        unaffected_max = max(unaffected_max, b[:j_lo].max())
    if j_hi < 2*n-1:
        unaffected_max = max(unaffected_max, b[j_hi+1:].max())
    
    # For each candidate v = a[i] * m:
    for m in multipliers:
        v = max(0.0, orig * m) if orig > 1e-15 else (0.0 if m <= 1.0 else m * 0.01)
        if v == orig:
            continue
        delta = v - orig
        # Compute new max at affected positions
        affected_max = unaffected_max
        for j in range(j_lo, j_hi+1):
            if j == 2*i:
                new_val = b[j] + 2*delta*orig + delta*delta
            else:
                k = j - i  # 0 <= k < n
                new_val = b[j] + 2*delta*a[k]
            if new_val > affected_max:
                affected_max = new_val
        if affected_max < best_max:
            best_max = affected_max
            best_v = v
    
    # Apply best
    if best_v != orig:
        delta = best_v - orig
        a[i] = best_v
        for j in range(j_lo, j_hi+1):
            if j == 2*i:
                b[j] += 2*delta*orig + delta*delta
            else:
                k = j - i
                b[j] += 2*delta*a[k]
    
    return best_v, best_max
```
