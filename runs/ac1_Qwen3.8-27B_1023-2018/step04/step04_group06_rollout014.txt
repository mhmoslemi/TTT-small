        b = np.convolve(a, a)
        return float(b.max()), int(b.argmax())

        j = np.arange(n)[None, :]
        d = k - j

        A[mask] = a[d[mask]]
        return A

                          bounds=[(0.0, None)]*n, method='highs',

            return None

import numpy as np
import time
from scipy.optimize import linprog


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 25.0

    # --- helpers ---
    def maxconv(a):
        b = np.convolve(a, a)
        return float(b.max()), int(b.argmax())

    def conv_mat(a):
        n = len(a)
        k = np.arange(2 * n - 1, dtype=np.int64)[:, None]
        j = np.arange(n, dtype=np.int64)[None, :]
        d = k - j
        mask = (d >= 0) & (d < n)
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        A[mask] = a[d[mask]]
        return A

    def solve_lp(a, M, n, tl=5.0):
        A = conv_mat(a)
        c_obj = -np.ones(n)
        b_ub = np.full(A.shape[0], M)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'time_limit': tl})
        except Exception:
            return None
        if res.status not in (0, 3) or res.x is None:
            return None
        return np.maximum(res.x, 0.0)

    def line_search(a, g, M_cur):
        n = len(a)
        best_t, best_v = 0.0, M_cur
        for t in np.linspace(0.01, 1.0, 40):
            c = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = c.sum()
            if s < 1e-14:
                continue
            c /= s
            v, _ = maxconv(c)
            if v < best_v:
                best_v, best_t = v, t
        if best_t == 0.0:
            return None, M_cur
        lo, hi = max(0.0, best_t - 0.07), min(1.0, best_t + 0.07)
        for _ in range(50):
            if hi - lo < 1e-11:
                break
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            c1 = np.maximum((1.0 - m1) * a + m1 * g, 0.0)
            c2 = np.maximum((1.0 - m2) * a + m2 * g, 0.0)
            s1, s2 = c1.sum(), c2.sum()
            v1 = maxconv(c1 / s1)[0] if s1 > 1e-14 else M_cur
            v2 = maxconv(c2 / s2)[0] if s2 > 1e-14 else M_cur
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        best_t = (lo + hi) / 2.0
        c = np.maximum((1.0 - best_t) * a + best_t * g, 0.0)
        c /= c.sum()
        best_v, _ = maxconv(c)
        return c, best_v

    def grad_active(a, k_list):
        n = len(a)
        grad = np.zeros(n)
        for k in k_list:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * a[idx]
        grad -= grad.mean()
        return grad
