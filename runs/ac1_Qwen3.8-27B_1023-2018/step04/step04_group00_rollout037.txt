import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    if c.size == 0:
        return math.inf
    m = float(c.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    return e / s


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    m = src.size
    if m == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if m == n:
        out = src.copy()
    else:
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        out = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _lse_optimize(p, stop_time, consider, seed=42):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p
    p /= s

    u = np.log(np.maximum(p, 1e-300), dtype=np.float64)
    u -= float(np.mean(u))

    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -60, 60)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if sum_exp <= 0 or not math.isfinite(sum_exp):
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * math.log(sum_exp / c.size)
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    c0 = cfast(p)
    cmax0 = float(c0.max()) if c0.size else 1.0
    if not math.isfinite(cmax0) or cmax0 <= 0:
        return p

    best_u = u.copy()
    try:
        best_f = lse_func(best_u, cmax0 * 0.05)[0]
    except Exception:
        return p
    if not math.isfinite(best_f):
        best_f = cmax0

    stages = [
        (0.08, 0.10),
        (0.04, 0.12),
        (0.015, 0.14),
        (0.006, 0.15),
        (0.002, 0.16),
        (0.0006, 0.16),
        (0.0002, 0.17),
    ]

    t_start = time.monotonic()
    total_avail = max(0.0, stop_time - t_start)
    if total_avail < 20:
        return p

    for frac, tf in stages:
        now = time.monotonic()
        if now > stop_time - 10:
            break
        T = max(1e-300, frac * cmax0)
        stage_budget = tf * total_avail
        max_iters = max(3, int(stage_budget / 2.0))

        for _ in range(max_iters):
            if time.monotonic() > stop_time - 8:
                break
            try:
                res = minimize(
                    lse_func, best_u, args=(T,),
                    method='L-BFGS-B', jac=True,
                    options={'maxiter': 30, 'gtol': 1e-11, 'ftol': 1e-15,
                             'maxcor': 25, 'maxls': 25, 'disp': False}
                )
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_f - 1e-15:
                    best_f = fval
                    best_u = res.x.copy()
            if getattr(res, 'status', 1) == 0:
                break

        p_new = _softmax(best_u)
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if ss > 1e-300:
            p_new /= ss
            consider(p_new)

    p_final = _softmax(best_u)
    p_final = np.maximum(p_final, 0.0)
    ss = float(p_final.sum())
    if ss > 1e-300:
        p_final /= ss
    return p_final


def _coord_descent(p, stop_time, G=44, consider=None, seed=0):

    if n == 0:
        return p
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p /= s

    L = 2 * n - 1
    c = _conv_self(p).copy()
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()

    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    max_sweeps = 50000

    while sweep < max_sweeps and time.monotonic() < stop_time - 0.5:
        improved = False
        order = rng.permutation(n)

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break

            v = float(p[i])
            start = i
            end = min(i + n, L)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end]
            infl = 2.0 * p[:Ls]
            diag_off = 2 * i - start

            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < L:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

            x_lo = max(0.0, v - 3.0 * v)
            x_hi = max(v + 4.0 * v, 15.0 / n, 1e-12)
            x_hi = min(x_hi, 5.0 * s / n)

            if x_hi <= x_lo:
                x_hi = x_lo + 1e-12

            pts = np.linspace(x_lo, x_hi, G + 1, dtype=np.float64)
            if 0.0 <= v <= x_hi:
                pts = np.concatenate([pts, [v]])
            pts = np.unique(pts)

            ws = pts - v
            denoms = 1.0 + ws  # s=1 after normalization
            valid = denoms > 1e-14

            if not np.any(valid):
                continue

            aff = base[None, :] + ws[:, None] * infl[None, :]
            maff = np.max(aff, axis=1)

            if 0 <= diag_off < Ls:
                dval = aff[:, diag_off] + ws * ws
                maff = np.maximum(maff, dval)

            if max_const > 0:
                maff = np.maximum(maff, max_const)

            vals = np.full(pts.shape, math.inf, dtype=np.float64)
            vals[valid] = 2.0 * n * maff[valid] / (denoms[valid] ** 2)

            if np.any(np.isfinite(vals)):
                imin = int(np.argmin(vals))
                best_x = float(pts[imin])
                best_val = float(vals[imin])

                # Fine search around best

                    lo2 = max(0.0, best_x - 2.0 * dw)
                    hi2 = min(x_hi, best_x + 2.0 * dw)
                    if hi2 > lo2:
                        fine = np.linspace(lo2, hi2, 24, dtype=np.float64)
                        if 0.0 <= best_x <= hi2:
                            fine = np.concatenate([fine, [best_x]])
                        fine = np.unique(fine)
                        ws2 = fine - v
                        den2 = 1.0 + ws2
                        val2 = den2 > 1e-14
                        if np.any(val2):
                            aff2 = base[None, :] + ws2[:, None] * infl[None, :]
                            ma2 = np.max(aff2, axis=1)
                            if 0 <= diag_off < Ls:
                                ma2 = np.maximum(ma2, aff2[:, diag_off] + ws2 * ws2)
                            if max_const > 0:
                                ma2 = np.maximum(ma2, max_const)
                            svals = np.full(fine.shape, math.inf)
                            svals[val2] = 2.0 * n * ma2[val2] / (den2[val2] ** 2)
                            if np.any(np.isfinite(svals)):
                                im2 = int(np.argmin(svals))
                                if svals[im2] < best_val:
                                    best_x = float(fine[im2])
                                    best_val = float(svals[im2])

                if best_val < cur_score - 1e-13 and 0.0 <= best_x:
                    w = best_x - v
                    if abs(w) > 1e-14:
                        p[i] = best_x

        new_score = float(2.0 * n * c.max())
        if new_score < best_score - 1e-14:

            flat = 0
        else:

                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break

            if p_fin is not None and p_fin.size == best_arr.size and np.all(np.isfinite(p_fin)):
                consider(p_fin)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.ones(1, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            return [1.0]
        return ls
    except Exception:
        return [1.0]
