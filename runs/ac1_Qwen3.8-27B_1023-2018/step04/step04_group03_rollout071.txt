    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n <= 0: return v
    # if all equal? handle
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0/n)
    rho = np.nonzero(cond)[0][-1]
    tau = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - tau, 0.0)
    s = out.sum()
    if s <= 0: return np.full(n,1.0/n)
    out /= s
    return out
```

        return None
    d = np.asarray(res.x[:n], dtype=np.float64)
    d = np.where(np.isfinite(d), d, 0.0)
    d[np.abs(d) < 1e-12] = 0.0
    if not np.all(np.isfinite(d)):
        return None
    if float(np.max(np.abs(d))) < 1e-12:
        return None
    return d


def _best_blend(f, g, current_sc=None):
    f = np.asarray(f, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    if current_sc is None or not math.isfinite(current_sc):
        current_sc = _score_internal(f)
    if not math.isfinite(current_sc):
        return None, math.inf

    best_t = 0.0
    best_sc = current_sc

    base_ts = (0.01, 0.02, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5,
               0.65, 0.8, 0.9, 1.0)
    for t in base_ts:
        a = (1.0 - t) * f + t * g
        a = np.where(np.isfinite(a), a, 0.0)
        s = float(a.sum())
        if not math.isfinite(s) or s <= 1e-12:
            continue
        if abs(s - 1.0) > 1e-8:
            a = a / s
        sc = _score_internal(a)
        if math.isfinite(sc) and sc < best_sc - 1e-15:
            best_sc = sc
            best_t = t

    if best_t > 0.0:
        lo = max(0.0, best_t * 0.5)
        hi = min(1.0, best_t * 1.5)
        if hi <= lo:
            lo = max(0.0, best_t - 0.1)
            hi = min(1.0, best_t + 0.1)
        refine = np.linspace(lo, hi, 16)
    else:
        refine = np.array([0.004, 0.008, 0.014, 0.022, 0.035,
                           0.055, 0.085, 0.13])

    for t in refine:
        t = float(t)
        if t <= 0.0 or t > 1.0:
            continue
        a = (1.0 - t) * f + t * g
        a = np.where(np.isfinite(a), a, 0.0)
        s = float(a.sum())
        if not math.isfinite(s) or s <= 1e-12:
            continue
        if abs(s - 1.0) > 1e-8:
            a = a / s
        sc = _score_internal(a)
        if math.isfinite(sc) and sc < best_sc - 1e-15:
            best_sc = sc
            best_t = t

    if best_t <= 0.0:
        return None, current_sc
    a = (1.0 - best_t) * f + best_t * g
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(a.sum())
    if s <= 1e-12:
        return None, current_sc
    a = a / s
    return a, best_sc


def _best_dir(f, d, current_sc=None):
    f = np.asarray(f, dtype=np.float64)
    d = np.asarray(d, dtype=np.float64)
    d = np.where(np.isfinite(d), d, 0.0)

    if current_sc is None or not math.isfinite(current_sc):
        current_sc = _score_internal(f)
    if not math.isfinite(current_sc):
        return None, math.inf

    t_max = 1.0
    neg = d < -1e-14
    if np.any(neg):
        ratios = f[neg] / (-d[neg])
        ratios = ratios[np.isfinite(ratios)]
        if ratios.size:
            t_max = min(t_max, float(ratios.min()) * 0.999)
    if t_max < 1e-5:
        return None, current_sc

    best_t = 0.0
    best_sc = current_sc

    for t in np.linspace(0.01, min(1.0, t_max), 14):
        a = f + float(t) * d
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if not math.isfinite(s) or s <= 1e-12:
            continue
        a = a / s
        sc = _score_internal(a)
        if math.isfinite(sc) and sc < best_sc - 1e-15:
            best_sc = sc
            best_t = t

    if best_t > 0.0:
        lo = max(0.01, best_t * 0.5)
        hi = min(1.0, t_max, best_t * 1.5)
        if hi <= lo:
            lo = max(0.01, best_t - 0.05)
            hi = min(1.0, t_max, best_t + 0.05)
        refine = np.linspace(lo, hi, 16)
    else:
        refine = np.linspace(0.005, min(0.2, t_max), 12)

    for t in refine:
        t = float(t)
        if t <= 0.0 or t > t_max:
            continue
        a = f + t * d
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if not math.isfinite(s) or s <= 1e-12:
            continue
        a = a / s
        sc = _score_internal(a)
        if math.isfinite(sc) and sc < best_sc - 1e-15:
            best_sc = sc
            best_t = t

    if best_t <= 0.0:
        return None, current_sc
    a = f + best_t * d
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-12:
        return None, current_sc
    a = a / s
    return a, best_sc


def _proj_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1, dtype=np.float64)) > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = int(np.nonzero(cond)[0][-1])
    tau = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - tau, 0.0)
    s = float(out.sum())
    if not math.isfinite(s) or s <= 1e-15:
        return np.full(n, 1.0 / n, dtype=np.float64)
    out /= s
    return out


def _weighted_grad(a, w):
    n = len(a)
    L = 2 * n - 1
    wrev = w[::-1]
    N = next_fast_len(n + L - 1)
    fa = rfft(a, N)
    fw = rfft(wrev, N)
    conv = irfft(fa * fw, N)[:n + L - 1]
    grad = conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    np.maximum(grad, 0.0, out=grad)
    return grad


def _smooth_move(a, lam, current_sc=None):
    a = np.asarray(a, dtype=np.float64)
    if current_sc is None or not math.isfinite(current_sc):
        current_sc = _score_internal(a)
    if not math.isfinite(current_sc):
        return a, current_sc

    C = _conv_full(a)
    if C.size == 0:
        return a, current_sc
    M = float(C.max())
    mC = float(C.min())
    if not math.isfinite(M) or M <= 1e-18:
        return a, current_sc
    gap = max(M - mC, 1e-18)

    z = float(lam) * (C - M)
    e = np.exp(z)
    e_sum = float(e.sum())
    if e_sum <= 1e-300:
        ws = np.zeros_like(C)
        ws[np.argmax(C)] = 1.0
    else:
        ws = e / e_sum

    top = C >= (M - 0.05 * gap)
    if top.sum() > 0:
        w = 0.55 * ws + 0.45 * (top / float(top.sum()))
    else:
        w = ws
    w = np.where(np.isfinite(w), w, 0.0)
    s = float(w.sum())
    if s <= 1e-300:
        return a, current_sc
    w /= s

    grad = _weighted_grad(a, w)
    gmax = float(grad.max())
    if not math.isfinite(gmax) or gmax <= 1e-18:
        return a, current_sc
    gn = grad / gmax

    best_a = a
    best_sc = current_sc

    for eta in (0.01, 0.03, 0.1, 0.3, 1.0, 3.0, 10.0):
        a_new = a * np.exp(-eta * gn) + 1e-12
        a_new = np.where(np.isfinite(a_new), a_new, 0.0)
        a_new = np.maximum(a_new, 0.0)
        sk = float(a_new.sum())
        if not math.isfinite(sk) or sk <= 1e-12:
            continue
        a_new = a_new / sk
        sc = _score_internal(a_new)
        if math.isfinite(sc) and sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a_new

    for eta in (1e-6, 3e-6, 1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 5e-2):
        a_new = _proj_simplex(a - eta * gn)
        sc = _score_internal(a_new)
        if math.isfinite(sc) and sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a_new

    return best_a, best_sc


def _perturb(a, rng, strength=0.2):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    x = np.linspace(0.0, 1.0, n)
    mode = int(rng.integers(0, 4))

    if mode == 0:
        scale = strength * max(1.0 / n, float(np.std(a)))
        a2 = a + rng.normal(0.0, scale, n)
    elif mode == 1:
        a2 = a * np.exp(rng.normal(0.0, strength, n))
    elif mode == 2:
        base = np.ones(n)
        for fr in (1, 2, 3, 5, 7):
            base += 0.25 * float(rng.normal()) * np.cos(2.0 * math.pi * fr * x + rng.uniform(0.0, 2.0 * math.pi))
        a2 = 0.5 * a + 0.5 * np.abs(base)
    else:
        c1 = float(rng.uniform(0.10, 0.40))
        c2 = float(rng.uniform(0.60, 0.90))
        w = float(rng.uniform(0.08, 0.25))
        base = np.exp(-0.5 * ((x - c1) / w) ** 2)
        base = base + 1.4 * np.exp(-0.5 * ((x - c2) / w) ** 2)
        a2 = 0.30 * a + 0.70 * base

    return _renorm(a2)


def _optimize_mix(a0, stop_time, seed):
    a = _renorm(a0)
    current_sc = _score_internal(a)
    best = a.copy()
    best_sc = current_sc
    if not math.isfinite(best_sc):
        return best, best_sc

    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 18
    step = 0
    lams = (100, 300, 700, 1500, 3000, 7000)
    lam_idx = 0

    while time.monotonic() < stop_time:
        step += 1
        remaining = stop_time - time.monotonic()
        if remaining <= 2.0:
            break

        if stall > max_stall // 2:
            method = 2
        elif step % 3 == 1:
            method = 0
        elif step % 3 == 2:
            method = 1
        else:
            method = 2

        if method in (0, 1) and remaining <= 12.0:
            method = 2

        accepted = False

        if method == 0:
            g = _lp_cross(a, timeout=min(6.0, max(0.5, remaining - 2.0)))
            if g is not None:
                a_new, sc = _best_blend(a, g, current_sc)
                if a_new is not None and math.isfinite(sc) and sc < current_sc - 1e-14:
                    a = a_new
                    current_sc = sc
                    accepted = True
        elif method == 1:
            d = _lp_symmetric(a, timeout=min(6.0, max(0.5, remaining - 2.0)))
            if d is not None:
                a_new, sc = _best_dir(a, d, current_sc)
                if a_new is not None and math.isfinite(sc) and sc < current_sc - 1e-14:
                    a = a_new
                    current_sc = sc
                    accepted = True
        else:
            lam = lams[lam_idx % len(lams)]
            lam_idx += 1
            a_new, sc = _smooth_move(a, lam, current_sc)
            if math.isfinite(sc) and sc < current_sc - 1e-14:
                a = a_new
                current_sc = sc
                accepted = True

        if accepted:
            if math.isfinite(current_sc) and current_sc < best_sc - 1e-14:
                best_sc = current_sc
                best = a.copy()
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                if math.isfinite(best_sc) and math.isfinite(current_sc) and current_sc > best_sc * 1.0005:
                    a = _perturb(best, rng, 0.25)
                else:
                    a = _perturb(a, rng, 0.12)
                current_sc = _score_internal(a)
                stall = 0
                max_stall = min(max_stall + 6, 60)
                if max_stall > 50 and step > 150:
                    break

        if (step % 5 == 0 and math.isfinite(best_sc) and math.isfinite(current_sc)
                and current_sc > best_sc * 1.01):
            a = _perturb(best, rng, 0.30)
            current_sc = _score_internal(a)

    return best, best_sc


def _optimize_smooth(a0, stop_time, seed):
    a = _renorm(a0)
    current_sc = _score_internal(a)
    best = a.copy()
    best_sc = current_sc
    if not math.isfinite(best_sc):
        return best, best_sc

    rng = np.random.default_rng(seed)
    lams = (80, 200, 500, 1200, 2500, 6000, 12000)
    lam_idx = 0
    stall = 0
    max_stall = 10
    step = 0

    while time.monotonic() < stop_time:
        step += 1
        remaining = stop_time - time.monotonic()
        if remaining <= 1.0:
            break

        lam = lams[lam_idx % len(lams)]
        lam_idx += 1
        a_new, sc = _smooth_move(a, lam, current_sc)
        if math.isfinite(sc) and sc < current_sc - 1e-14:
            a = a_new
            current_sc = sc
            if current_sc < best_sc - 1e-14:
                best_sc = current_sc
                best = a.copy()
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                a = _perturb(best, rng, 0.15)
                current_sc = _score_internal(a)
                stall = 0
                max_stall = min(max_stall + 4, 28)
                if max_stall > 24 and step > 120:
                    break
    return best, best_sc


def _polish(a0, stop_time, seed):
    a = _renorm(a0)
    current_sc = _score_internal(a)
    best = a.copy()
    best_sc = current_sc
    if not math.isfinite(best_sc):
        return best, best_sc

    rng = np.random.default_rng(seed)
    n = len(a)
    no_imp = 0
    max_no_imp = 4

    mults = (0.0, 0.05, 0.1, 0.2, 0.35, 0.55, 0.75, 0.9,
             1.1, 1.3, 1.6, 2.0, 2.7, 4.0, 6.0)
    fine = (0.85, 0.92, 0.96, 1.04, 1.08, 1.15)

    while time.monotonic() < stop_time:
        improved = False
        mean = 1.0 / n

        for i in rng.permutation(n):
            if time.monotonic() >= stop_time:
                break

            orig = float(a[i])
            best_val = orig
            local_sc = current_sc

            vals = [0.0]
            if orig > 1e-14:
                for m in mults:
                    vals.append(orig * m)
                for m in (0.66, 0.8, 1.25, 1.5, 1.9, 2.4, 3.3):
                    vals.append(orig * m)
            for m in (0.25, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0):
                vals.append(mean * m)
            vals.append(orig)

            for v in vals:
                if not math.isfinite(v) or v < 0.0:
                    continue
                v = min(1000.0, float(v))
                if a[i] == v:
                    continue
                a[i] = v
                sc = _score_internal(a)
                if math.isfinite(sc) and sc < local_sc - 1e-15:
                    local_sc = sc
                    best_val = v

            a[i] = best_val
            if best_val != orig:
                current_sc = local_sc
                improved = True

                if best_val > 1e-14:
                    for m in fine:
                        v = best_val * m
                        if not math.isfinite(v) or v < 0.0:
                            continue
                        v = min(1000.0, float(v))
                        a[i] = v
                        sc = _score_internal(a)
                        if math.isfinite(sc) and sc < local_sc - 1e-15:
                            local_sc = sc
                            best_val = v
                    a[i] = best_val
                    current_sc = local_sc

            if math.isfinite(current_sc) and current_sc < best_sc - 1e-14:
                best_sc = current_sc
                best = a.copy()

        a = _renorm(a)
        current_sc = _score_internal(a)
        if not math.isfinite(current_sc):
            a = best.copy()
            current_sc = best_sc
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            if current_sc < best_sc - 1e-14:
                best_sc = current_sc
                best = a.copy()
            if not improved:
                no_imp += 1
                if no_imp >= max_no_imp:
                    break
                if rng.random() < 0.5:
                    a = best.copy()
                    current_sc = best_sc
            else:
                no_imp = 0

    return best, best_sc


def _make_inits(n, parent, seed):
    inits = []
    x = np.linspace(0.0, 1.0, n)
    parent = np.asarray(parent, dtype=np.float64)

    inits.append(_renorm(parent))
    inits.append(np.ones(n, dtype=np.float64))

    y = np.ones(n)
    for fr in (1, 2, 3):
        y += 0.5 * np.cos(2.0 * math.pi * fr * x)
    inits.append(_renorm(np.abs(y) + 0.1))

    y = 0.2 + 0.8 * np.abs(np.cos(math.pi * x))
    inits.append(_renorm(y))

    eps = 1e-7
    xa = np.clip(x, eps, 1.0 - eps)
    for p, q in ((1.0, 3.0), (3.0, 1.0), (0.7, 2.5), (2.5, 0.7), (1.5, 4.0)):
        y = (xa ** p) * ((1.0 - xa) ** q)
        inits.append(_renorm(y + 1e-10))

    for c1, c2, w, amp in ((0.20, 0.80, 0.12, 0.8),
                           (0.30, 0.70, 0.16, 1.2),
                           (0.25, 0.75, 0.22, 0.7),
                           (0.15, 0.85, 0.18, 1.0)):
        y = np.exp(-0.5 * ((x - c1) / w) ** 2)
        y = y + amp * np.exp(-0.5 * ((x - c2) / w) ** 2)
        inits.append(_renorm(y))

    for alpha in (2.0, 4.0):
        y = np.exp(-alpha * x) + 0.7 * np.exp(-alpha * (1.0 - x))
        inits.append(_renorm(y))

    rng = np.random.default_rng(seed)
    for beta in (1.5, 2.5, 3.5):
        s = rng.choice([-1.0, 1.0], size=n)
        y = np.exp(beta * s)
        inits.append(_renorm(y))

    for _ in range(4):
        y = np.ones(n)
        for fr in (1, 2, 3, 4, 6):
            y += 0.25 * float(rng.normal()) * np.cos(2.0 * math.pi * fr * x + rng.uniform(0.0, 2.0 * math.pi))
        y = np.abs(y) + 0.05
        inits.append(_renorm(y))

    out = []
    seen = 0.0
    for arr in inits:
        arr = _renorm(arr)
        if arr.size != n:
            continue
        h = hash((tuple(np.round(arr, 10)), n))
        if len(out) > 0 and abs(h - seen) < 2:
            continue
        seen = h
        out.append(arr)
    return out[:18]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 5.0
    work_deadline = hard_deadline - 10.0
    if work_deadline < t0 + 20.0:
        work_deadline = hard_deadline - 2.0

    parent = _get_parent()
    best_arr = _renorm(parent)
    best_score = _score_internal(best_arr)
    top = [(best_score, best_arr.copy())] if math.isfinite(best_score) else []

    def _consider(arr):
        nonlocal best_arr, best_score, top
        if arr is None:
            return
        a = np.asarray(arr, dtype=np.float64).copy()
        if a.size == 0:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if not math.isfinite(s) or s < 0.01:
            return
        sc = _score_internal(a)
        if not math.isfinite(sc):
            return
        if sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()
        if len(top) < 12 or sc < top[-1][0]:
            top.append((sc, a.copy()))
            top.sort(key=lambda z: z[0])
            if len(top) > 12:
                top.pop()

    try:
        n0 = len(best_arr)
        inits = _make_inits(n0, best_arr, int(seed))
        init_scored = []
        for arr in inits:
            sc = _score_internal(arr)
            if math.isfinite(sc):
                init_scored.append((sc, arr))
        init_scored.sort(key=lambda z: z[0])
        for _, arr in init_scored:
            _consider(arr)

        total = max(1.0, work_deadline - t0)
        alloc_main = max(30.0, min(650.0, total * 0.50))
        stop_main = min(work_deadline - 15.0, t0 + alloc_main)
        if stop_main > t0 + 8.0:
            try:
                res, _ = _optimize_mix(best_arr, stop_main, int(seed) + 1)
                _consider(res)
            except Exception:
                pass

        if best_score > _TARGET and time.monotonic() < work_deadline - 80.0:
            diverse = [arr for _, arr in init_scored[1:5]]
            for i, arr in enumerate(diverse):
                now = time.monotonic()
                if now > work_deadline - 70.0:
                    break
                left = work_deadline - now
                per = min(95.0, max(35.0, 0.25 * left))
                if left < per + 50.0:
                    break
                stop = now + per
                try:
                    res, _ = _optimize_mix(arr, stop, int(seed) + 1000 * (i + 1))
                    _consider(res)
                except Exception:
                    pass

        if best_score > _TARGET and time.monotonic() < work_deadline - 100.0:
            lengths = []
            for L in (n0 - 64, n0 - 32, n0 + 32, n0 + 64):
                if 220 <= L <= 700 and L != n0:
                    lengths.append(L)
            lengths = sorted(set(lengths), key=lambda L: abs(L - n0))
            for idx, L in enumerate(lengths[:2]):
                now = time.monotonic()
                if now > work_deadline - 80.0:
                    break
                left = work_deadline - now
                per = min(85.0, max(30.0, 0.30 * left))
                if left < per + 60.0:
                    break
                stop = now + per
                try:
                    initL = _interp_length(best_arr, L)
                    res, _ = _optimize_mix(initL, stop, int(seed) + 9173 * (idx + 1))
                    _consider(res)
                except Exception:
                    pass

        if time.monotonic() < work_deadline - 70.0:
            now = time.monotonic()
            left = work_deadline - now
            smooth_time = min(220.0, max(30.0, 0.35 * left))
            stop_smooth = min(work_deadline - 25.0, now + smooth_time)
            if stop_smooth > now + 20.0:
                try:
                    res, _ = _optimize_smooth(best_arr, stop_smooth, int(seed) + 555555)
                    _consider(res)
                except Exception:
                    pass

        polish_stop = work_deadline - 2.0
        if polish_stop > time.monotonic() + 15.0:
            try:
                res, _ = _polish(best_arr, polish_stop, int(seed) + 999999)
                _consider(res)
            except Exception:
                pass

    except Exception:
        pass

    final_seq = best_arr
    final_score = _exact_score(final_seq)

    for _, arr in top:
        sc = _exact_score(arr)
        if math.isfinite(sc) and (not math.isfinite(final_score) or sc < final_score):
            final_score = sc
            final_seq = arr

    if not math.isfinite(final_score):
        fb = _renorm(parent)
        fs = _exact_score(fb)
        if math.isfinite(fs):
            final_seq = fb
            final_score = fs

    if not math.isfinite(final_score):
        return [1.0]

    out = []
    for v in final_seq:
        try:
            v = float(v)
        except Exception:
            v = 0.0
        if not math.isfinite(v):
            v = 0.0
        v = max(0.0, min(1000.0, v))
        out.append(v)
    if not out or float(np.sum(out)) < 0.01:
        return [1.0]
    return out
