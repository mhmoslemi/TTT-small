            stall = 0
            lr = max(0.01, lr * 0.995)
        else:
            f_d = 0.7 * f + 0.3 * f_new
            f_d = np.clip(f_d, 1e-10, None)
            f_d = f_d / f_d.sum()
            sc_d = _score(f_d)
            if sc_d < best_score - 1e-13:
                f_new = f_d
                best_f = f_new.copy()
                best_score = sc_d
                stall = 0
                lr = max(0.01, lr * 0.98)
            else:
                stall += 1
                lr *= 0.92
                if stall > max_stall:
                    noise = rng.normal(0, 0.015, n)
                    f = np.clip(f + noise, 1e-10, None)
                    f = f / f.sum()
                    stall = 0
                    lr = 0.12
                    max_stall += 30
                    if max_stall > 300:
                        break

        f = f_new

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 25.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = _score(best_seq)

    def _update_best(candidate):
        nonlocal best_seq, best_score
        if candidate is None or len(candidate) == 0:
            return
        candidate = np.clip(np.asarray(candidate, dtype=np.float64), 0, 1000)
        if not np.all(np.isfinite(candidate)):
            return
        if candidate.sum() < 0.01:
            return
        sc = _score(candidate)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq = candidate.copy()
            best_score = sc

    # Phase 1: LP optimization from multiple initializations (35% time)
    phase1_end = t0 + budget * 0.35
    rng = np.random.default_rng(seed)
    i_arr = np.arange(n_p, dtype=np.float64)

    inits = [parent.copy()]

    # Noise perturbation
    sig = 0.06 * max(1.0, float(parent.mean()))
    inits.append(np.clip(parent + rng.normal(0, sig, n_p), 0, None))

    # Cosine window
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n_p)))

    # Multi-frequency
    sup = np.ones(n_p) * 0.15
    for fr in (1, 2, 3, 5, 7, 11):
        ph = rng.uniform(0, 2 * math.pi)
        sup += 0.1 * np.cos(2.0 * math.pi * fr * i_arr / n_p + ph)
    inits.append(np.abs(sup) + 0.02)

    # Gaussian bumps
    center = (n_p - 1) / 2.0
    inits.append(np.exp(-0.5 * ((i_arr - center) / (n_p * 0.2)) ** 2))
    inits.append(np.exp(-0.5 * ((i_arr - center) / (n_p * 0.35)) ** 2))

    # Two bumps
    c1, c2 = n_p * 0.25, n_p * 0.75
    w = n_p * 0.1
    inits.append(np.exp(-0.5 * ((i_arr - c1) / w) ** 2) + np.exp(-0.5 * ((i_arr - c2) / w) ** 2))

    time_per_init = max(12.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        if time.monotonic() > stop - 3.0:
            continue
        try:
            res, sc = _optimize_lp(init, stop, seed + 1000 * (idx + 1), peak_frac=0.08)
            _update_best(res)
        except Exception:
            pass

    # LP refinement on best
    if time.monotonic() < phase1_end - 12.0:
        stop = phase1_end - 4.0
        try:
            res, sc = _optimize_lp(best_seq, stop, seed + 777777, peak_frac=0.05)
            _update_best(res)
        except Exception:
            pass

    # Phase 2: Subgradient (15% time)
    phase2_end = t0 + budget * 0.52
    if time.monotonic() < phase2_end - 10.0:
        stop = phase2_end - 4.0
        try:
            res, sc = _subgradient(best_seq, stop, seed + 55555)
            _update_best(res)
        except Exception:
            pass

    # Phase 3: Coordinate descent (28% time) - this is critical for fine-tuning
    phase3_end = t0 + budget * 0.82
    if time.monotonic() < phase3_end - 10.0:
        stop = phase3_end - 4.0
        try:
            res, sc = _coordinate_descent(best_seq, stop, seed + 33333)
            _update_best(res)
        except Exception:
            pass

    # Phase 4: LP after coordinate descent (5% time)
    phase4_end = t0 + budget * 0.88
    if time.monotonic() < phase4_end - 8.0:
        stop = phase4_end - 3.0
        try:
            res, sc = _optimize_lp(best_seq, stop, seed + 444444, peak_frac=0.03)
            _update_best(res)
        except Exception:
            pass

    # Phase 5: More coordinate descent (5% time)
    phase5_end = t0 + budget * 0.94
    if time.monotonic() < phase5_end - 8.0:
        stop = phase5_end - 3.0
        try:
            res, sc = _coordinate_descent(best_seq, stop, seed + 66666)
            _update_best(res)
        except Exception:
            pass

    # Phase 6: Final LP with tight peaks (3% time)
    phase6_end = t0 + budget * 0.97
    if time.monotonic() < phase6_end - 5.0:
        stop = phase6_end - 2.0
        try:
            res, sc = _optimize_lp(best_seq, stop, seed + 888888, peak_frac=0.02)
            _update_best(res)
        except Exception:
            pass

    # Final coordinate descent in remaining time
    remaining = deadline - time.monotonic()
    if remaining > 15.0:
        try:
            res, sc = _coordinate_descent(best_seq, deadline - 5.0, seed + 99999)
            _update_best(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            return [1.0]
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    return [float(x) for x in best_seq]
