    cols = []
    vals = []

            v = f[k - i]
            if v > 1e-20:
                rows.append(idx)
                cols.append(i)
                vals.append(v)

        return None

    b_ub = np.full(len(peak_idx), M)

            stall += 3

                f_new = np.clip(f + noise, 0, None)

                    if sc < best_score + 0.1:  # Allow slightly worse
                        f = f_new
                else:
                    f = best_f.copy()
                t = 0.2

def _optimize_multi_update(f0, stop_time, seed):
    """Multiplicative/subgradient update focusing on peak reduction."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-10, None)
    S0 = f.sum()
    f = f / S0  # Normalize to sum=1
    best_f = f.copy() * S0  # Keep original scale for evaluation
    best_score = _score(best_f)
    
    lr = 0.3
    stall = 0
    
    while time.monotonic() < stop_time - 5.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        # Find peak indices
        threshold = M * 0.99
        peak_idx = np.where(b >= threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])
        
        # Compute gradient: for each peak k, d(conv)[k]/da[i] = f[k-i]
        grad = np.zeros(n)
        for k in peak_idx:
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                grad[i] += f[k - i]
        grad /= len(peak_idx)
        
        if grad.max() < 1e-15:
            break
        
        # Multiplicative update: f_new[i] = f[i] * exp(-lr * grad[i])
        # Then normalize
        scale = lr / grad.max()
        log_mult = -scale * grad
        log_mult -= log_mult.max()  # Numerical stability
        mult = np.exp(log_mult)
        
        f_new = f * mult
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        
        sc = _score(f_new)
        if sc < best_score - 1e-12:
            best_f = f_new * S0
            best_score = sc
            stall = 0
            lr *= 0.995
        else:
            # Try damped version
            f_damp = 0.7 * f + 0.3 * f_new
            f_damp = f_damp / f_damp.sum()
            sc_d = _score(f_damp)
            if sc_d < best_score - 1e-12:
                best_f = f_damp * S0
                best_score = sc_d
                stall = 0
                lr *= 0.98
            else:
                stall += 1
                lr *= 0.93
                if stall > 100:
                    # Restart with perturbation
                    f = np.clip(f + rng.normal(0, 0.005, n), 1e-10, None)
                    f = f / f.sum()
                    lr = 0.2
                    stall = 0
        
        f = f_new
    
    return best_f, best_score


def _polish(f0, stop_time, seed):

    n = len(f0)

    max_no_imp = 15

    while time.monotonic() < stop_time - 5.0:
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break
            
            orig = float(f[i])
            best_local = best_sc
            best_val = orig
            
            for mlt in mults:
                f[i] = orig * mlt
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_val = f[i]
                else:
                    f[i] = orig
            
            if best_local < best_sc - 1e-15:
                f[i] = best_val
                best_sc = best_local
                improved = True
            else:
                f[i] = orig
        
        if not improved:
            no_imp_rounds += 1
            if no_imp_rounds >= max_no_imp:
                break
        else:
            no_imp_rounds = 0
    
    return f.copy(), best_sc


def _try_length(parent, L, stop_time, seed):

    n = len(a)

        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-20:
                rows.append(idx)
                cols.append(i)
                vals.append(v)

    mults_coarse = (0.15, 0.3, 0.5, 0.7, 0.85, 1.1, 1.3, 1.6, 2.2, 0.0)
    mults_fine = (0.9, 1.1, 0.8, 1.2, 0.95, 1.05, 0.92, 1.08)

    while time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = float(f[i])
            best_local = best_sc
            best_val = orig

            for mlt in mults_coarse:
                f[i] = orig * mlt
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_val = f[i]
                else:
                    f[i] = orig

            if best_local < best_sc - 1e-15:
                f[i] = best_val
                best_sc = best_local
                improved = True
                # Fine search
                for mlt in mults_fine:
                    f[i] = best_val * mlt
                    if f.sum() < 0.01:
                        f[i] = best_val
                        continue
                    sc = _score(f)
                    if sc < best_local - 1e-15:
                        best_local = sc
                    else:
                        f[i] = best_val
                best_sc = best_local
            else:
                f[i] = orig

        if not improved:
            no_imp_rounds += 1
            if no_imp_rounds >= max_no_imp:
                break
        else:
            no_imp_rounds = 0

    return f.copy(), best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 25.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = _score(best_seq)

    def _update_best(candidate):
        nonlocal best_seq, best_score
        if candidate is None or len(candidate) == 0:
            return
        candidate = np.clip(np.asarray(candidate, dtype=np.float64), 0, 1000)
        if not np.all(np.isfinite(candidate)):
            return
        if candidate.sum() < 0.01:
            return
        sc = _score(candidate)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq = candidate.copy()
            best_score = sc

    rng = np.random.default_rng(seed)

    # Phase 1: LP from parent (30% budget)
    p1_end = t0 + budget * 0.30
    if time.monotonic() < p1_end - 10:
        stop = min(p1_end - 5, time.monotonic() + (p1_end - time.monotonic() - 5))
        try:
            res, sc = _optimize_lp(parent, stop, seed + 1,
                                   peak_fracs=(0.05, 0.03, 0.08))
            _update_best(res)
        except Exception:
            pass

    # Phase 2: LP from perturbed parent (12% budget)
    p2_end = t0 + budget * 0.42
    if time.monotonic() < p2_end - 10:
        sigma = 0.12 * max(1.0, float(parent.mean()))
        perturbed = np.clip(parent + rng.normal(0, sigma, n_p), 0, None)
        stop = min(p2_end - 5, time.monotonic() + (p2_end - time.monotonic() - 5))
        try:
            res, sc = _optimize_lp(perturbed, stop, seed + 2,
                                   peak_fracs=(0.07, 0.04, 0.1))
            _update_best(res)
        except Exception:
            pass

    # Phase 3: LP on current best with tight constraints (18% budget)
    p3_end = t0 + budget * 0.60
    if time.monotonic() < p3_end - 10:
        stop = min(p3_end - 5, time.monotonic() + (p3_end - time.monotonic() - 5))
        try:
            res, sc = _optimize_lp(best_seq, stop, seed + 3,
                                   peak_fracs=(0.02, 0.03, 0.01, 0.04))
            _update_best(res)
        except Exception:
            pass

    # Phase 4: Multiplicative update (12% budget)
    p4_end = t0 + budget * 0.72
    if time.monotonic() < p4_end - 10:
        stop = min(p4_end - 5, time.monotonic() + (p4_end - time.monotonic() - 5))
        try:
            res, sc = _optimize_multi(best_seq, stop, seed + 4)
            _update_best(res)
        except Exception:
            pass

    # Phase 5: Coordinate descent polish (18% budget)
    p5_end = t0 + budget * 0.90
    if time.monotonic() < p5_end - 10:
        stop = min(p5_end - 5, time.monotonic() + (p5_end - time.monotonic() - 5))
        try:
            res, sc = _polish(best_seq, stop, seed + 5)
            _update_best(res)
        except Exception:
            pass

    # Phase 6: Final LP refinement (5% budget)
    p6_end = t0 + budget * 0.95
    if time.monotonic() < p6_end - 10:
        stop = min(p6_end - 5, time.monotonic() + (p6_end - time.monotonic() - 5))
        try:
            res, sc = _optimize_lp(best_seq, stop, seed + 6,
                                   peak_fracs=(0.01, 0.02, 0.03))
            _update_best(res)
        except Exception:
            pass

    # Final coordinate polish with whatever time remains
    if time.monotonic() < deadline - 8:
        stop = deadline - 5
        try:
            res, sc = _polish(best_seq, stop, seed + 7)
            _update_best(res)
        except Exception:
            pass

    # Validate with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    return [float(x) for x in best_seq]
