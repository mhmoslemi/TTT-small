import time
from scipy.optimize import linprog

def _eval_score(a):
    """Fast evaluation matching the exact evaluator."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)

def _get_parent():
    try:
        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(201, dtype=np.float64)
    p = np.clip(p, 0, 1000)
    if p.size == 0 or not np.all(np.isfinite(p)):
        p = np.ones(201, dtype=np.float64)
    return p

def _build_conv_matrix(n, a):
    """Build the (2n-1) x n matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        length = min(n, 2*n-1-j)
        A[j:j+length, j] = a[:length]
    return A

def _solve_best_response(a, M, n, tl=5.0):
    """Solve max sum(g) s.t. conv(a,g)_k <= M for all k, g >= 0."""
    A = _build_conv_matrix(n, a)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)]*n,
                      method='highs',
                      options={'time_limit': tl, 'presolve': True})
    except Exception:
        return None
    if res.status not in (0, 3):  # 0=optimal, 3=time limit with feasible
        return None
    if res.x is None:
        return None
    return np.clip(res.x, 0.0, None)

def _line_search(a, g, max_iters=30):
    """Golden section search for best blend parameter t."""
    # a and g both have sum = 1
    # cand(t) = (1-t)*a + t*g, which also has sum = 1
    def obj(t):
        cand = (1.0 - t) * a + t * g
        return float(np.convolve(cand, cand).max())
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc, fd = obj(c), obj(d)
    
    for _ in range(max_iters):
        if fc < fd:
            hi = d
            d = c; fd = fc
            c = hi - gr * (hi - lo)
            fc = obj(c)
        else:
            lo = c
            c = d; fc = fd
            d = lo + gr * (hi - lo)
            fd = obj(d)
        if hi - lo < 1e-10:
            break

            a = np.maximum(a, 1e-15)
            a /= a.sum()

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_parent = parent.size

    best_seq = parent.copy()
    best_score = _eval_score(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(32)
        best_score = _eval_score(best_seq)

    def try_update(cand):
        nonlocal best_seq, best_score
        try:
            sc = _eval_score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.clip(np.asarray(cand, dtype=np.float64), 0, 1000).copy()
        except Exception:
            pass

    # Phase 1: Main optimization on parent (60% of budget)
    t1 = t0 + budget * 0.60
    try:
        r = _optimize_deep(parent, t1, seed=1)
        try_update(r)
    except Exception:
        pass

    # Phase 2: Continue with standard optimizer on current best (15%)
    t2 = t0 + budget * 0.75
    if time.monotonic() < t2:
        try:
            r = _optimize(best_seq, t2, seed=777, max_stuck=100, tl_lp=4.0)
            try_update(r)
        except Exception:
            pass

    # Phase 3: Diverse initializations (15%)
    t3 = t0 + budget * 0.90
    if time.monotonic() < t3:
        pn = parent / max(parent.sum(), 1e-15)
        x = np.arange(n_parent, dtype=np.float64)
        ctr = (n_parent - 1) / 2.0

        inits = []
        # Symmetric triangle
        tri = 1.0 - np.abs(x - ctr) / max(ctr, 1)
        inits.append(tri / tri.sum())
        # Gaussian
        gauss = np.exp(-0.5 * ((x - ctr) / (n_parent * 0.25)) ** 2)
        inits.append(gauss / gauss.sum())
        # Bump
        bump = np.minimum(1.0 - np.abs(x - ctr) / (n_parent * 0.4), 0.5)
        bump = np.maximum(bump, 0.01)
        inits.append(bump / bump.sum())
        # Blend with parent
        for frac in [0.3, 0.5, 0.7]:
            inits.append((frac * pn + (1 - frac) * tri) / (frac * pn + (1 - frac) * tri).sum())
            inits.append((frac * pn + (1 - frac) * gauss) / (frac * pn + (1 - frac) * gauss).sum())
        # Cosine window
        cosw = 0.5 * (1 + np.cos(np.pi * x / max(n_parent - 1, 1)))
        inits.append(cosw / cosw.sum())
        # Two-lobe
        lobe1 = np.exp(-0.5 * ((x - n_parent * 0.25) / (n_parent * 0.15)) ** 2)
        lobe2 = np.exp(-0.5 * ((x - n_parent * 0.75) / (n_parent * 0.15)) ** 2)
        inits.append((lobe1 + lobe2) / (lobe1 + lobe2).sum())

        per_init = max(5.0, (t3 - time.monotonic()) / max(1, len(inits)))
        for i, init in enumerate(inits):
            if time.monotonic() > t3 - 3:
                break
            alloc = min(time.monotonic() + per_init, t3 - 2)
            try:
                r = _optimize(init, alloc, seed=2000 + i, max_stuck=60, tl_lp=3.0)
                try_update(r)
            except Exception:
                pass

    # Phase 4: Final deep polish on best (remaining)
    if time.monotonic() < deadline - 5:
        try:
            r = _optimize_deep(best_seq, deadline, seed=9999)
            try_update(r)
        except Exception:
            pass

    # Final validation
    try:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if not np.isfinite(fs):
            raise ValueError("invalid final")
    except Exception:
        best_seq = np.ones(32)

    return [float(x) for x in best_seq]
