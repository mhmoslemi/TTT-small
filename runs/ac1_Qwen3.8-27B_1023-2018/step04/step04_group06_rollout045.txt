import time
import math
from scipy.optimize import linprog


def _eval_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)


def _get_parent():

        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()

def _convolve(a):
    return np.convolve(a, a)


def _build_A(n, a):
    """(2n-1) x n Toeplitz-like convolution matrix."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_best_response(A, M, n, time_limit=5.0):
    """max 1^T g s.t. A@g <= M, g >= 0"""
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-13,
                               'primal_feasibility_tolerance': 1e-13,
                               'dual_stopepsilon': 1e-13,
                               'primal_stopepsilon': 1e-13})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    return np.clip(res.x, 0.0, None)


def _line_search(a, g, M_cur, max_iter=50):
    """Golden-section search for best blend t minimizing max conv."""
    a_n = a / a.sum()
    g_n = g / g.sum()
    best_t = 0.0
    best_val = M_cur
    gr = (math.sqrt(5) - 1) / 2.0
    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        if hi - lo < 1e-11:
            break
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        c1 = (1 - m1) * a_n + m1 * g_n
        v1 = float(_convolve(c1).max())
        c2 = (1 - m2) * a_n + m2 * g_n
        v2 = float(_convolve(c2).max())
        if v1 < v2:
            hi = m2
        else:
            lo = m1
        if v1 < best_val:
            best_val, best_t = v1, m1
        if v2 < best_val:
            best_val, best_t = v2, m2
    return best_t, best_val


def _subgradient(a, k_star, n):
    """Subgradient of (a*a)_{k_star} w.r.t. a."""
    grad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] = 2.0 * a[idx]
    return grad


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
