import numpy as np
import time
from scipy.optimize import linprog

def _fe(a):
    """Fast evaluation (matching exact evaluator)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * a.size * c.max() / (s * s))

def _gp():
    """Get parent sequence."""
    try:
        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(201, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    if p.size == 0 or not np.all(np.isfinite(p)):
        p = np.ones(201, dtype=np.float64)
    return p

def _cm(n, a):
    """Convolution matrix (2n-1) x n."""
    k = np.arange(2*n-1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    m = (d >= 0) & (d < n)
    A = np.zeros((2*n-1, n))
    A[m] = a[d[m]]
    return A

def _lp(A, M, n, tl=4.0):
    """Solve max 1^T g s.t. A g <= M, g >= 0."""
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(A.shape[0], M),
                    bounds=[(0,None)]*n, method='highs',
                    options={'time_limit': tl,
                             'dual_feasibility_tolerance': 1e-14,
                             'primal_feasibility_tolerance': 1e-14})
        if r.status != 0 or r.x is None:
            return None
        return np.clip(r.x, 0.0, None)
    except Exception:
        return None

def _blend(a, g, Mcur):
    """Optimal blend of a and g."""
    bt, bv = 0.0, Mcur
    for t in np.array([0.05,0.1,0.15,0.2,0.3,0.4,0.5,0.7,1.0]):
        x = (1-t)*a + t*g
        s = x.sum()
        if s < 1e-15: continue
        x /= s
        v = np.convolve(x,x).max()
        if v < bv: bv, bt = v, t
    if bt == 0.0: return None, Mcur
    lo, hi = max(0.0, bt-0.07), min(1.0, bt+0.07)
    for _ in range(40):
        mid = 0.5*(lo+hi)
        x = (1-mid)*a + mid*g
        s = x.sum()
        if s < 1e-15: hi = mid; continue
        x /= s
        v = np.convolve(x,x).max()
        if v < bv: bv, bt, hi = v, mid, mid
        else: lo = mid
        if hi-lo < 1e-11: break
    x = np.maximum((1-bt)*a + bt*g, 0.0)
    x /= max(x.sum(), 1e-15)
    return x, bv

def _opt(a0, stop, seed=42):

    a = np.asarray(a0, dtype=np.float64).copy()
    n = a.size
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15: a = np.ones(n)/n
    a /= s
    ba, bM = a.copy(), float(np.convolve(a,a).max())
    rng = np.random.default_rng(seed)

    while time.monotonic() < stop:

        M = float(cv.max())
        ks = int(cv.argmax())

        else:

            sig = min(0.015 + 0.005*(stuck//80), 0.35)
            mode = stuck % 5
            xx = np.arange(n, dtype=np.float64)
            if mode == 0:

            elif mode == 1:

            elif mode == 2:

                a *= 1.0 + sig*(np.sin(2*np.pi*f1*xx/n) + 0.5*np.cos(2*np.pi*f2*xx/n))
            elif mode == 3:

                a[idx] *= np.exp(rng.normal(0, 2*sig, len(idx)))
            else:

                b0 = rng.integers(0, n//2)
                bl = rng.integers(n//10, n//3)
                a[b0:b0+bl] = a[b0:b0+bl][::-1]
            a = np.maximum(a, 1e-12)
            a /= max(a.sum(), 1e-15)
            stuck = 0

        A = _cm(n, a)
        mv = False
        
        for mf in (1+1e-13, 1+1e-11, 1+1e-9, 1+1e-7, 1+1e-5, 1.001):

            a = np.maximum(a, 1e-12)

    return ba
