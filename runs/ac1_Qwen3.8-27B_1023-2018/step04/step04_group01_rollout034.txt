    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 5:
            break

        improved = False
        order = rng.permutation(n)

        for i in range(n):
            if time.monotonic() > stop_time - 5:
                break

            j = order[i]
            v = f[j]

            start = j
            end = min(j + n, 2 * n - 1)
            length = end - start
            if length <= 0:
                continue

            infl = 2.0 * f[:length]

            # Unchanged max
            if start == 0 and end == 2 * n - 1:
                unchanged_max = 0.0
            else:
                parts = []
                if start > 0:
                    parts.append(conv[:start])
                if end < 2 * n - 1:
                    parts.append(conv[end:])
                unchanged_max = float(np.max(np.concatenate(parts))) if parts else 0.0

            # Search range
            lo = 0.0
            hi = max(5.0 * v if v > 1e-10 else 0.1, S * 0.5 / n)
            hi = min(hi, 50.0)

            n_grid = 50
            grid = np.linspace(lo, hi, n_grid)

            best_val = v
            best_obj = 2.0 * n * conv_max

            for x in grid:
                d = x - v
                if abs(d) < 1e-14:
                    continue
                S_new = S + d
                if S_new < 0.01:
                    continue
                affected = conv[start:end] + d * infl
                if 2 * j >= start and 2 * j < end:
                    idx = 2 * j - start
                    affected[idx] += d * d
                max_aff = float(np.max(affected))
                new_max = max(max_aff, unchanged_max)
                obj = 2.0 * n * new_max / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_val = x

            if best_val != v and best_obj < best_score - 1e-13:
                d = best_val - v
                conv[start:end] += d * infl
                if 2 * j >= start and 2 * j < end:
                    conv[2 * j] += d * d
                f[j] = best_val
                S += d
                conv_max = float(conv.max())
                best_score = 2.0 * n * conv_max / (S * S)
                best_f = f.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 4:
                break
        else:
            no_improve = 0

    final_score = _eval_score(f)
    if final_score < _eval_score(best_f):
        best_f = f.copy()
    return best_f, final_score


def _refined_cd(f0, stop_time, seed, max_sweeps=200):
    """More aggressive coordinate descent with wider search."""
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = np.clip(f0.astype(np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, math.inf
    f = f / S
    S = 1.0

    conv = _conv_full(f)
    conv_max = float(conv.max())
    best_score = 2.0 * n * conv_max
    best_f = f.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 5:
            break

        improved = False
        order = rng.permutation(n)

        for i in range(n):
            if time.monotonic() > stop_time - 5:
                break

            j = order[i]
            v = f[j]

            start = j
            end = min(j + n, 2 * n - 1)
            length = end - start
            if length <= 0:
                continue

            infl = 2.0 * f[:length]

            if start == 0 and end == 2 * n - 1:
                unchanged_max = 0.0
            else:
                parts = []
                if start > 0:
                    parts.append(conv[:start])
                if end < 2 * n - 1:
                    parts.append(conv[end:])
                unchanged_max = float(np.max(np.concatenate(parts))) if parts else 0.0

            # Wider search
            lo = 0.0
            hi = max(10.0 * v if v > 1e-10 else 0.2, S * 1.0 / n)
            hi = min(hi, 100.0)

            # Two-stage: coarse then fine
            n_coarse = 60
            grid = np.linspace(lo, hi, n_coarse)

            best_val = v
            best_obj = 2.0 * n * conv_max

            for x in grid:
                d = x - v
                if abs(d) < 1e-14:
                    continue
                S_new = S + d
                if S_new < 0.01:
                    continue
                affected = conv[start:end] + d * infl
                if 2 * j >= start and 2 * j < end:
                    idx = 2 * j - start
                    affected[idx] += d * d
                max_aff = float(np.max(affected))
                new_max = max(max_aff, unchanged_max)
                obj = 2.0 * n * new_max / (S_new * S_new)
                if obj < best_obj:
                    best_obj = obj
                    best_val = x

            # Fine search around best
            if best_val != v:
                step = (hi - lo) / n_coarse
                fl = max(0.0, best_val - 3 * step)
                fh = min(hi, best_val + 3 * step)
                grid_fine = np.linspace(fl, fh, 30)
                for x in grid_fine:
                    d = x - v
                    if abs(d) < 1e-14:
                        continue
                    S_new = S + d
                    if S_new < 0.01:
                        continue
                    affected = conv[start:end] + d * infl
                    if 2 * j >= start and 2 * j < end:
                        idx = 2 * j - start
                        affected[idx] += d * d
                    max_aff = float(np.max(affected))
                    new_max = max(max_aff, unchanged_max)
                    obj = 2.0 * n * new_max / (S_new * S_new)
                    if obj < best_obj:
                        best_obj = obj
                        best_val = x

            if best_val != v and best_obj < best_score - 1e-13:
                d = best_val - v
                conv[start:end] += d * infl
                if 2 * j >= start and 2 * j < end:
                    conv[2 * j] += d * d
                f[j] = best_val
                S += d
                conv_max = float(conv.max())
                best_score = 2.0 * n * conv_max / (S * S)
                best_f = f.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

    final_score = _eval_score(f)
    if final_score < _eval_score(best_f):
        best_f = f.copy()
    return best_f, final_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    hard_deadline = t0 + budget - 12.0

    # Get parent
    try:
        parent = np.asarray(globals()['height_sequence_1'], dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(400, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    if parent.size == 0:
        parent = np.ones(400, dtype=np.float64)

    n = len(parent)

    # Fallback
    best_seq = parent.copy()
    best_score = _eval_score(parent)

    def _try_update(seq, score=None):
        nonlocal best_seq, best_score
        seq = np.clip(np.asarray(seq, dtype=np.float64), 0, 1000)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or float(seq.sum()) < 0.01:
            return
        if score is None:
            score = _eval_score(seq)
        if math.isfinite(score) and score < best_score - 1e-14:
            best_seq = seq.copy()
            best_score = score

    # === Phase 1: LP optimization (main workhorse) ===
    t1_end = t0 + budget * 0.50
    try:
        res, sc = _lp_main_loop(parent, t1_end, seed=42, max_iters=300)
        _try_update(res, sc)
    except Exception:
        pass

    # === Phase 2: Coordinate descent polish ===
    t2_end = min(t0 + budget * 0.72, hard_deadline - 80)
    if time.monotonic() < t2_end - 10:
        try:
            res, sc = _coord_descent(best_seq, t2_end, seed=123, max_sweeps=300)
            _try_update(res, sc)
        except Exception:
            pass

    # === Phase 3: Second LP round with perturbation ===
    t3_end = min(t0 + budget * 0.82, hard_deadline - 60)
    if time.monotonic() < t3_end - 10 and best_score > 1.5028:
        rng = np.random.default_rng(seed + 777)
        for trial in range(3):
            if time.monotonic() > t3_end - 10:
                break
            pert = np.clip(best_seq + rng.normal(0, 0.03, n), 0, None)
            if pert.sum() > 0.01:
                pert /= pert.sum()
            else:
                continue
            try:
                res, sc = _lp_main_loop(pert, t3_end - time.monotonic() * 0.0, seed=999 + trial, max_iters=80)
                _try_update(res, sc)
            except Exception:
                pass

    # === Phase 4: Refined coordinate descent ===
    t4_end = min(t0 + budget * 0.90, hard_deadline - 30)
    if time.monotonic() < t4_end - 5:
        try:
            res, sc = _refined_cd(best_seq, t4_end, seed=555, max_sweeps=200)
            _try_update(res, sc)
        except Exception:
            pass

    # === Phase 5: Final quick LP + CD ===
    t5_end = hard_deadline - 15
    if time.monotonic() < t5_end - 15:
        try:
            res, sc = _lp_main_loop(best_seq, t5_end - 5, seed=4242, max_iters=40)
            _try_update(res, sc)
        except Exception:
            pass
    if time.monotonic() < t5_end - 5:
        try:
            res, sc = _coord_descent(best_seq, t5_end, seed=9999, max_sweeps=100)
            _try_update(res, sc)
        except Exception:
            pass

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or float(best_seq.sum()) < 0.01:
        return [1.0]

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            fb = parent.copy()
            fb = np.clip(fb, 0, 1000)
            fb = np.where(np.isfinite(fb), fb, 0.0)
            fb_sc = float(evaluate_sequence([float(x) for x in fb]))
            if math.isfinite(fb_sc):
                return [float(x) for x in fb]
        except Exception:
            pass
        return [1.0]

    return [float(x) for x in best_seq]
