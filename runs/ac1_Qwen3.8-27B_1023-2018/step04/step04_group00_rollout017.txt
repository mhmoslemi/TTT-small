def _get_parent():

def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n <= 256:
        return np.convolve(a, a, mode="full")
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    if c.size == 0:
        return math.inf
    return float(2.0 * n * c.max() / (s * s))


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)

        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    return float(2.0 * n * m / (s * s))


def _lbfqs_phase(p, stop_time, best_holder):
    n = p.size
    if n < 10 or n > 5000:
        return p
    p = np.maximum(p.astype(np.float64).copy(), 0.0)
    s = float(p.sum())
    if s < 1e-30:
        return p
    p = p / s

    def to_u(pp):
        pp = np.maximum(pp, 1e-300)
        uu = np.log(pp)
        return uu - float(np.mean(uu))

    u = to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1

    def objective(uu):
        m = float(np.max(uu))
        e = np.exp(uu - m)
        ss = float(e.sum())
        if not math.isfinite(ss) or ss <= 0:
            return 1e100, np.zeros_like(uu)
        pp = e / ss
        F = rfft(pp, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        cmax = float(c.max())
        if not math.isfinite(cmax):
            return 1e100, np.zeros_like(uu)
        T = max(1e-4 * cmax, 1e-8)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
        else:
            w = exps / sum_exp
        f = 2.0 * n * (cmax + T * math.log(sum_exp / float(c.size)))

        # This is the cross-correlation of p and w (shifted)

        cross_len = 3 * n - 2
        Fp = rfft(pp[::-1], cross_size)
        Fw = rfft(w, cross_size)

        q = cc[n - 1:2 * n - 1]
