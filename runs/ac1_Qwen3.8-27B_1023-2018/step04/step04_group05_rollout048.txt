    rng = np.random.default_rng(seed)
    f = np.clip(f0[:n].copy() if len(f0) >= n else np.pad(f0, (0, n-len(f0))), 0, None)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n); s = n
    ts = math.sqrt(2*n)
    f /= s; f *= ts
    
    best = f.copy()
    bs = _score(f)
    t = 0.3; stall = 0; max_stall = 60
    
    while time.monotonic() < stop - 2:
        b = _conv(f)
        M = b.max()
        if M < 1e-18: break
        
        g = _lp_solve(f, n, M)
        if g is None:
            stall += 3
            if stall > max_stall:
                f = best + rng.normal(0, 0.06*max(1, best.mean()), n)
                f = np.clip(f, 0, None)
                if f.sum() > 1e-10: f /= f.sum(); f *= ts
                else: f = best.copy()
                stall = 0; max_stall += 40; t = 0.3
            continue
        
        gs = g.sum()
        if gs < 1e-15:
            stall += 2; continue
        g *= ts / gs
        
        if np.linalg.norm(g-f) / max(1, np.linalg.norm(f)) < 1e-10:
            stall += 2
            if stall > max_stall:
                f = best.copy()
                idx = rng.choice(n, min(15, n), replace=False)
                for k in idx: f[k] = max(0, f[k]*rng.uniform(0.2, 4.0))
                if f.sum() > 1e-10: f /= f.sum(); f *= ts
                stall = 0; max_stall += 40; t = 0.3
            continue
        
        improved = False
        for tc in sorted({max(0.05, min(1.0, x)) for x in [t, t*.5, t*.75, t*1.25, t*1.5, .1, .2, .4, .6, .8, 1.0]}, reverse=True):
            fn = np.clip((1-tc)*f + tc*g, 0, None)
            if fn.sum() < 0.01: continue
            fn /= fn.sum(); fn *= ts
            sc = _score(fn)
            if sc < bs - 1e-13:
                f = fn; best = f.copy(); bs = sc
                improved = True; t = min(1.0, tc*1.3); stall = 0
                break
        
        if not improved:
            stall += 1; t = max(0.05, t*0.6)
            if stall >= max_stall:
                f = best.copy()
                f += rng.normal(0, (0.05+0.02*stall//max_stall)*max(1,best.mean()), n)
                f = np.clip(f, 0, None)
                if f.sum() > 1e-10: f /= f.sum(); f *= ts
                else: f = best.copy()
                stall = 0; max_stall += 40; t = 0.3
    
    return best, bs

def _polish(a, stop, seed=42):
    n = len(a)
    ts = math.sqrt(2*n)
    a = np.clip(a.copy(), 0, None)
    s = a.sum()
    if s < 1e-10: return a, _score(a)
    a /= s; a *= ts
    bs = _score(a)
    rng = np.random.default_rng(seed)
    noimp = 0
    
    while time.monotonic() < stop - 2 and noimp < 15:
        imp = False
        # Coordinate descent: for each coordinate, try several multipliers
        for i in range(n):
            if time.monotonic() > stop - 2: break
            orig = a[i]
            if orig < 1e-14: 
                # Try adding a small value
                for mv in [0.01*ts, 0.05*ts, 0.1*ts]:
                    a[i] = mv
                    sc = _score(a)
                    if sc < bs - 1e-14: bs = sc; orig = mv; imp = True
                a[i] = orig if orig < 1e-14 else orig
                if not imp: a[i] = 0
                continue
            
            best_mv = orig; best_sc = bs
            for m in [0, 0.1, 0.3, 0.5, 0.7, 1.2, 1.5, 2.0, 3.0]:
                a[i] = orig * m if m > 0 else 0.0
                sc = _score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc; best_mv = orig*m if m > 0 else 0.0
            if best_mv != orig:
                a[i] = best_mv; bs = best_sc; imp = True
            else:
                a[i] = orig
        
        # Pair search
        for _ in range(200):
            if time.monotonic() > stop - 2: break
            i, j = rng.integers(n), rng.integers(n)
            if i == j: continue
            oi, oj = a[i], a[j]
            for mi in [0, 0.5, 1, 1.5]:
                for mj in [0, 0.5, 1, 1.5]:
                    if mi == 1 and mj == 1: continue
                    a[i] = oi*mi if mi > 0 and oi > 1e-14 else 0.0
                    a[j] = oj*mj if mj > 0 and oj > 1e-14 else 0.0
                    if a.sum() < 0.01:
                        a[i], a[j] = oi, oj; continue
                    sc = _score(a)
                    if sc < bs - 1e-14: bs = sc; imp = True
            a[i], a[j] = oi, oj
        
        # Small perturbations
        for _ in range(50):
            if time.monotonic() > stop - 2: break
            k = rng.integers(n)
            old = a[k]
            a[k] = max(0, old + rng.normal(0, 0.03*max(1, a.mean())))
            if a.sum() < 0.01: a[k] = old; continue
            sc = _score(a)
            if sc < bs - 1e-14: bs = sc; imp = True
            else: a[k] = old
        
        noimp = 0 if imp else noimp + 1
    
    return a, bs

from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    c = irfft(F*F, fsize)[:2*n-1]
    np.maximum(c, 0, out=c)
    return c

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01: return math.inf
    b = _conv(a)
    return 2.0*n*b.max()/(s*s)

def _lp_dir(f, n, M, timeout=2.0):
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c = -np.ones(n)
    try:
        r = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M+1e-12),
                    bounds=[(0,None)]*n, method='highs',
                    options={'presolve':True,'time_limit':timeout})
    except: return None
    if r.status != 0: return None
    g = np.clip(r.x, 0, 1e6)
    return g if g.sum() > 1e-15 else None
