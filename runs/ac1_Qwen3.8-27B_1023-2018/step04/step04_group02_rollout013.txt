def _build_A(f, n):
    """Build the convolution matrix A where (A @ g)[k] = (f*g)[k]."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(f, n, M, timeout=5.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-11,
                               'primal_feasibility_tolerance': 1e-11})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _grad_step(a, stop_time, alpha=0.01, n_steps=20):
    """Do a few subgradient descent steps."""
    n = len(a)
    s = a.sum()

        b = _conv(a)

import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _conv(a):
    n = len(a)
    if n <= 1024:
        return np.convolve(a, a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_conv_matrix(f, n):
    """A such that A @ g = f * g (full convolution)."""
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    return A


def _lp_direction(f, n, M, timeout=5.0):
    try:
        A = _build_conv_matrix(f, n)
    except Exception:
        return None
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _peak(f):
    b = _conv(f)

def _cd_round(f, stop_time, seed=0, n_rounds=20):
    """Cyclic coordinate descent."""

        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            fm = f.mean()
            
            # Generate candidate values
            cands = [0.0, orig, fm * 0.25, fm * 0.5, fm, fm * 2.0]
            if orig > 1e-12:
                for m in mults:
                    cands.append(max(0.0, orig * m))
            
            best_v = orig
            best_sc_i = cur
            for v in cands:
                if v < 0:
                    v = 0.0
                f[i] = v
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_sc_i - 1e-14:
                    best_sc_i = sc
                    best_v = v
            
            f[i] = best_v
            if best_sc_i < cur - 1e-15:
                cur = best_sc_i
                improved = True
                if cur < best_sc:
