                a = np.maximum(a, 1e-12)
            elif mode == 3:
                a *= np.exp(sigma * rng.normal(0, 1, n))
            else:
                # Random block flip
                block = max(n // 10, 1)
                start = rng.integers(0, n - block)
                a[start:start + block] = a[start + block:start + 2 * block] if start + 2 * block <= n else a[start:start + block]
                a[start:start + block] = np.maximum(a[start:start + block], 1e-12)
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            stuck = 0
            pert_level += 1
            if pert_level > 60:
                break
            continue

        # Build LP matrix
        A = _build_conv_mat_fast(n, a)
        moved = False

        # LP best response
        g = _lp_solve(A, M * (1 + 1e-10), n, tl=3.0)
        if g is not None and g.sum() > 1e-12:
            g = g / g.sum()
            if float(np.max(np.abs(g - a))) > 1e-10:
                cand, val = _line_search(a, g, M)
                if cand is not None and val < M * (1 - 1e-10):
                    a = cand
                    moved = True

        if not moved:
            # Try with slightly relaxed constraint
            g2 = _lp_solve(A, M * (1 + 1e-4), n, tl=2.0)
            if g2 is not None and g2.sum() > 1e-12:
                g2 = g2 / g2.sum()
                if float(np.max(np.abs(g2 - a))) > 1e-10:
                    cand, val = _line_search(a, g2, M)
                    if cand is not None and val < M * (1 - 1e-10):
                        a = cand
                        moved = True

        if not moved:
            # Subgradient step: reduce contribution at active lags
            active_lags = np.where(c_full >= M * 0.9999)[0]
            grad = np.zeros(n)
            for k_star in active_lags:
                lo = max(0, int(k_star) - n + 1)
                hi = min(n, int(k_star) + 1)
                for j in range(lo, hi):
                    idx = int(k_star) - j
                    if 0 <= idx < n:
                        grad[j] += 2.0 * a[idx]
            # Project to zero-mean
            grad = grad - grad.mean()
            gn = float(np.linalg.norm(grad))
            if gn > 1e-14:
                for alpha in [0.03, 0.01, 0.005, 0.002]:
                    cand = a - alpha * grad / gn
                    cand = np.maximum(cand, 0.0)
                    cs = cand.sum()
                    if cs < 1e-12:
                        continue
                    cand = cand / cs
                    v = float(np.convolve(cand, cand).max())
                    if v < M * (1 - 1e-10):
                        a = cand
                        moved = True
                        break

        if not moved:
            a *= np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 1e-12)
            a = a / a.sum()

    return best_a


def _optimize_multi(a0, deadline, seed=42):
    """Multi-strategy optimization: alternate LP, gradient, and local search."""
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a)
    a = a / s
    n = a.size
    best_a = a.copy()
    best_v = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0
    phase = 0

    while time.monotonic() < deadline:
        c_full = np.convolve(a, a)
        M = float(c_full.max())

        if M < best_v * (1 - 1e-13):
            best_v = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 180:
            sigma = min(0.03 + 0.015 * (stuck // 180), 0.4)
            xx = np.arange(n, dtype=np.float64)
            a *= np.exp(sigma * (rng.normal(0, 1, n) +
                                  0.4 * np.sin(2 * np.pi * rng.uniform(1, 5) * xx / n) +
                                  0.3 * np.cos(2 * np.pi * rng.uniform(1, 5) * xx / n)))
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            stuck = 0
            if sigma > 0.35:
                break
            continue

        A = _build_conv_mat_fast(n, a)
        moved = False

        # Strategy 1: LP best response
        g = _lp_solve(A, M * (1 + 1e-10), n, tl=2.5)
        if g is not None and g.sum() > 1e-12:
            g = g / g.sum()
            if float(np.max(np.abs(g - a))) > 1e-10:
                cand, val = _line_search(a, g, M)
                if cand is not None and val < M * (1 - 1e-10):
                    a = cand
                    moved = True

        # Strategy 2: LP with averaging direction
        if not moved and g is not None and g.sum() > 1e-12:
            g_avg = 0.5 * g + 0.5 * a
            g_avg = np.maximum(g_avg, 0)
            gs = g_avg.sum()
            if gs > 1e-12:
                g_avg = g_avg / gs
                cand, val = _line_search(a, g_avg, M)
                if cand is not None and val < M * (1 - 1e-10):
                    a = cand
                    moved = True

        # Strategy 3: Coordinate descent on active lags
        if not moved:
            active = np.where(c_full >= M * 0.999)[0]
            for k_star in active[:5]:
                # Find pair (i,j) with i+j = k_star that maximizes a[i]*a[j]
                lo = max(0, int(k_star) - n + 1)
                hi = min(n, int(k_star) + 1)
                best_i, best_j, best_prod = -1, -1, 0
                for i in range(lo, hi):
                    j = int(k_star) - i
                    if 0 <= j < n:
                        prod = a[i] * a[j]
                        if prod > best_prod:
                            best_prod = prod
                            best_i, best_j = i, j
                if best_i >= 0 and best_i != best_j:
                    # Try to redistribute mass between best_i and best_j
                    delta = 0.1 * min(a[best_i], a[best_j])
                    for frac in [0.3, 0.5, 0.7]:
                        d = frac * delta
                        for sign in [1, -1]:
                            cand = a.copy()
                            cand[best_i] -= sign * d
                            cand[best_j] += sign * d
                            cand = np.maximum(cand, 0)
                            cs = cand.sum()
                            if cs < 1e-12:
                                continue
                            cand = cand / cs
                            v = float(np.convolve(cand, cand).max())
                            if v < M * (1 - 1e-10):
                                a = cand
                                moved = True
                                break
                        if moved:
                            break
                if moved:
                    break

        if not moved:
            # Small random perturbation
            a *= np.exp(rng.normal(0, 0.015, n))
            a = np.maximum(a, 1e-12)
            a = a / a.sum()

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_parent = parent.size

    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(100)
        best_score = _score(best_seq)

    def try_update(cand):
        nonlocal best_seq, best_score
        try:
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.clip(np.asarray(cand, dtype=np.float64), 0, 1000).copy()
        except Exception:
            pass

    total_inits = 0

    # ===== Phase 1: Optimize parent =====
    t1 = t0 + min(250.0, budget * 0.20)
    try:
        r = _optimize_multi(parent, t1, seed=1)
        try_update(r)
    except Exception:
        pass

    # ===== Phase 2: Diverse initializations =====
    x = np.linspace(0, 1, n_parent)
    pn = parent / max(parent.sum(), 1e-15)

    inits = []
    # Power law decay from center
    ctr = (n_parent - 1) / 2.0
    for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.7, 1.0]:
        v = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(v / v.sum())
    # Symmetric peaks
    for w in [0.05, 0.1, 0.15, 0.2, 0.3]:
        v = np.exp(-0.5 * ((x - w) / 0.08) ** 2) + np.exp(-0.5 * ((x - (1 - w)) / 0.08) ** 2)
        inits.append(v / v.sum())
    # Cosine windows
    for m in [1, 2, 3, 4]:
        v = np.maximum(np.cos(m * np.pi * (x - 0.5)) ** 2 + 0.01, 0)
        inits.append(v / v.sum())
    # Random (seeded)
    rng_init = np.random.default_rng(12345)
    for _ in range(8):
        v = np.abs(rng_init.normal(1, 0.5, n_parent)) + 0.01
        inits.append(v / v.sum())
    # Blends with parent
    for frac in [0.3, 0.5, 0.7]:
        for init in inits[:5]:
            v = frac * pn + (1 - frac) * init
            inits.append(v / v.sum())

    t2 = t0 + min(600.0, budget * 0.50)
    elapsed = time.monotonic() - t0
    remaining = max(t2 - time.monotonic(), 1.0)
    per_init = max(3.0, remaining / max(1, len(inits)))

    for i, init in enumerate(inits):
        if time.monotonic() > t2 - 5.0:
            break
        alloc = min(time.monotonic() + per_init, t2 - 3.0)
        try:
            r = _optimize(init, alloc, seed=500 + i)
            try_update(r)
        except Exception:
            pass

    # ===== Phase 3: Deep polish on best =====
    t3 = t0 + min(900.0, budget * 0.75)
    ab = np.clip(best_seq, 0, 1000).astype(np.float64)
    ab = ab / ab.sum()
    try:
        r = _optimize_multi(ab, t3, seed=999)
        try_update(r)
    except Exception:
        pass

    # ===== Phase 4: Try different lengths =====
    if time.monotonic() < deadline - 120:
        dn_list = [-50, -30, -20, -10, -5, 5, 10, 20, 30, 50, 80, 100]
        for dn in dn_list:
            if time.monotonic() > deadline - 30:
                break
            nt = n_parent + dn
            if nt < 10 or nt > 800:
                continue
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, parent)
            init_new = np.clip(init_new, 0, 1000)
            sn = init_new.sum()
            if sn < 1e-10:
                continue
            init_new = init_new / sn
            alloc = min(time.monotonic() + 30, deadline - 20)
            try:
                r = _optimize(init_new, alloc, seed=2000 + nt)
                try_update(r)
            except Exception:
                pass

    # ===== Phase 5: Final polish =====
    if time.monotonic() < deadline - 10:
        af = np.clip(best_seq, 0, 1000).astype(np.float64)
        af = af / af.sum()
        try:
            r = _optimize_multi(af, deadline, seed=7777)
            try_update(r)
        except Exception:
            pass

    # ===== Final validation =====
    try:
        fs = float(evaluate_sequence([float(v) for v in best_seq]))
        if not np.isfinite(fs):
            raise ValueError("invalid")
    except Exception:
        fallback = [1.0] * 100
        return fallback

    return [float(v) for v in best_seq]
