def _conv_full(a):
    n = len(a)
    if n <= 1024:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)

    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if m < 0:
        m = 0
    return 2.0 * n * m / (s * s)

def _project_nonneg(v):
    return np.maximum(v, 0)

def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho_val = rho_idx[-1] + 1
    theta = (cssv[rho_idx[-1]] - 1) / rho_val
    return np.maximum(v - theta, 0)

def _lp_direction(f, n, M, time_limit=10.0):
    """Solve LP: max sum(g) s.t. (f*g)[k]<=M, g>=0."""
    # Build sparse-ish A: A[k,j] = f[k-j] if 0<=k-j<n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j: j + n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs

                scores_f = np.where(valid_f, 2.0 * n * max_total_f / (S_new_f * S_new_f), np.inf)
                bf = np.argmin(scores_f)
                if scores_f[bf] < best_sc_candidate:
                    best_w = w_fine[bf]
                    best_sc_candidate = scores_f[bf]

            if best_sc_candidate < best_score - 1e-14 and best_w > 1e-12:
                w = best_w - v
                a[i] = v + w  # = best_w
                S += w
                # Update b
                b[k_lo:k_hi + 1] += 2.0 * w * infl + w * 2.0 * (a[i - 1: i - 1] if False else 0)
                # Actually: b[k_lo+k] was base_seg[k], new is base_seg[k] + 2w*infl[k] + w^2*diag
                # But we already have the updated values; let's just recompute the segment
                b_new_seg = base_seg + 2.0 * (best_w - v) * infl
                if has_diag:
                    b_new_seg[di] += (best_w - v) ** 2
                b[k_lo:k_hi + 1] = b_new_seg
                best_score = best_sc_candidate
                best_a = a.copy()
                improved = True

        if not improved and sweep > 5:
            # Try perturbation
            a = np.clip(a + rng.normal(0, 0.005 / n, n), 0, None)
            S = a.sum()
            b = _conv(a)
            best_score = 2.0 * n * b.max() / (S * S)
            if best_score < best_score * 0.999:
                best_a = a.copy()
            improved = False

    final_sc = _score(a)
    if final_sc < _score(best_a):
        return a, final_sc
    return best_a, best_sc


def _coord_descent_v2(a0, stop_time, max_sweeps=300, seed=9999):
    """Second CD variant: focused on reducing the top few b_k."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64).copy(), 0, None)
    S = float(a.sum())
    b = _conv(a)
    best_score = 2.0 * n * b.max() / (S * S)
    best_a = a.copy()
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 5.0:
            break
        # Focus on coordinates that affect the top b_k values
        top_b = np.argpartition(b, -10)[-10:]
        # Coordinates that contribute to these b_k:
        cand_coords = set()
        for k in top_b:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for m in range(lo, hi + 1, max(1, (hi - lo + 1) // 10)):
                cand_coords.add(m)
        # Also include some random
        for _ in range(20):
            cand_coords.add(int(rng.integers(n)))
        order = list(cand_coords)
        rng.shuffle(order)

        improved = False
        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break
            v = a[i]
            S_rest = S - v
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            seg_len = k_hi - k_lo + 1
            infl = a[:seg_len].copy()
            di = 2 * i - k_lo
            has_diag = 0 <= di < seg_len

            if k_lo > 0:
                unchanged_max = b[:k_lo].max()
            else:
                unchanged_max = 0.0
            if k_hi < 2 * n - 2:
                unchanged_max = max(unchanged_max, b[k_hi + 1:].max())

            base_seg = b[k_lo:k_hi + 1].copy()
            w_max = max(3.0 * v + 0.02 / n, 0.05 / n)
            w_max = min(w_max, 10.0 / n)

            Q = 50
            w_grid = np.linspace(0.0, w_max, Q)
            new_b_seg = base_seg[:, None] + 2.0 * w_grid[None, :] * infl[:, None]
            if has_diag:
                new_b_seg[di, :] += w_grid ** 2
            max_total = np.maximum(new_b_seg.max(axis=0), unchanged_max)
            S_new = S_rest + w_grid
            valid = S_new > 0.01
            scores = np.where(valid, 2.0 * n * max_total / (S_new * S_new), np.inf)
            bi = np.argmin(scores)
            best_w = w_grid[bi]
            best_cand = scores[bi]

            if best_cand < best_score - 1e-15 and best_w > 1e-12:
                w = best_w - v
                a[i] = best_w
                S += w
                b_new_seg = base_seg + 2.0 * w * infl
                if has_diag:
                    b_new_seg[di] += w * w
                b[k_lo:k_hi + 1] = b_new_seg
                best_score = best_cand
                best_a = a.copy()
                improved = True

        if not improved and sweep > 10:
            a = best_a.copy()
            S = a.sum()
            b = _conv(a)
            break

    final_sc = _score(a)
    if final_sc < _score(best_a):
        return a, final_sc
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 20.0

    parent = _get_parent()
    n = len(parent)

    # Fallback
    fallback = parent.copy()
    try:
        fb_sc = _score(fallback)
    except Exception:
        fb_sc = math.inf
    if not math.isfinite(fb_sc):
        fallback = np.ones(400) / 400.0
        fb_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fb_sc

    def _update_best(seq, sc=None):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64).copy()
        seq = np.clip(seq, 0, 1000)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        if sc is None:
            sc = _score(seq)
        if not math.isfinite(sc):
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = sc
        if math.isfinite(exact) and exact < best_score:
            best_seq = seq.copy()
            best_score = exact

    # Normalize parent
    a_init = np.clip(parent, 0, None).astype(np.float64)
    s_init = a_init.sum()
    if s_init > 0.01:
        a_init = a_init / s_init
    else:
        a_init = np.ones(n) / n

    # === Phase 1: Gradient optimization (40% of time) ===
    t1_end = t0 + budget * 0.38
    if time.monotonic() < t1_end - 10:
        try:
            res, sc = _gradient_optimize(a_init, n, t1_end, seed, max_iters=80000)
            _update_best(res, sc)
        except Exception:
            pass

    # === Phase 2: LP-based refinement (15% of time) ===
    t2_start = time.monotonic()
    t2_end = min(t0 + budget * 0.53, hard_deadline - 100)
    if t2_start < t2_end - 5 and best_score > 1.5028:
        try:
            res, sc = _lp_refine(best_seq, len(best_seq), t2_end, seed + 100, max_rounds=12)
            _update_best(res, sc)
        except Exception:
            pass

    # === Phase 3: Coordinate descent (30% of time) ===
    t3_start = time.monotonic()
    t3_end = min(t0 + budget * 0.82, hard_deadline - 80)
    if t3_start < t3_end - 5:
        try:
            res, sc = _coord_descent(best_seq, t3_end, max_sweeps=200, seed=424242)
            _update_best(res, sc)
        except Exception:
            pass

    # Additional CD from best
    t3b_end = min(t0 + budget * 0.90, hard_deadline - 40)
    if time.monotonic() < t3b_end - 5:
        try:
            res, sc = _coord_descent_v2(best_seq, t3b_end, max_sweeps=200, seed=55555)
            _update_best(res, sc)
        except Exception:
            pass

    # === Phase 4: More gradient from current best (if time) ===
    t4_end = min(t0 + budget * 0.94, hard_deadline - 20)
    if time.monotonic() < t4_end - 10:
        try:
            res, sc = _gradient_optimize(best_seq, len(best_seq), t4_end, seed + 200, max_iters=30000)
            _update_best(res, sc)
        except Exception:
            pass

    # === Phase 5: Final CD polish ===
    t5_end = hard_deadline - 5.0
    if time.monotonic() < t5_end - 3:
        try:
            res, sc = _coord_descent(best_seq, t5_end, max_sweeps=50, seed=12345)
            _update_best(res, sc)
        except Exception:
            pass

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
