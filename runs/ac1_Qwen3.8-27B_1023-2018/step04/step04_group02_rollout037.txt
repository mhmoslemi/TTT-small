import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _conv(a):
    """Compute full convolution of a with itself using FFT for large n."""
    n = len(a)
    if n <= 1024:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    """Compute the AC1 score."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    """Build the constraint matrix for the LP.
    The constraint is (f*g)_k <= M for k = 0, ..., 2n-2.
    (f*g)_k = sum_j f_j * g_{k-j} where g_{k-j} is indexed so that
    the valid range is 0 <= k-j < n, i.e., max(0, k-n+1) <= j < min(n, k+1).
    
    Actually, (f*g)_k = sum_{j=0}^{n-1} f_j * g_{k-j} where g_{k-j} = 0 if k-j < 0 or k-j >= n.
    So (f*g)_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f_j * g_{k-j}
    
    The matrix A has A[k,j] = f_j if max(0,k-n+1) <= j <= min(n-1,k), else 0.
    But wait, that's not right either. Let me be more careful.
    
    (f*g)_k = sum_{j} f_j * g_{k-j}
    For this to be nonzero, we need 0 <= k-j < n AND j >= 0 (since g is length n).
    So j ranges from max(0, k-n+1) to min(n-1, k).
    And g_{k-j} has index (k-j) which ranges from max(0, k-n+1) to min(n-1, k) -> k - j.
    
    So A[k, j] = f_j if max(0, k-n+1) <= j <= min(n-1, k), else 0.
    
    Wait, that means each row k has non-zero entries in columns j from max(0,k-n+1) to min(n-1,k).
    And the value is f_j.
    
    This is correct. The matrix is banded with bandwidth n.
    """
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(f, n, M, timeout=3.0):
    """Solve the LP to find improving direction g.
    Maximize sum(g) subject to A*g <= M, g >= 0.
    """
    A = _build_A(f, n)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _line_search(f, g, n, cur_score, max_iters=30):
    """Find optimal step size t in [0,1] for f + t*(g - f).
    Uses golden section search on the convex function h(t) = max(((1-t)f + tg)*(...)).
    Actually, since the score is scale-invariant, we just need max(conv) for the normalized sequence.
    """
    # h(t) = score((1-t)*f + t*g)
    # This is the max of 2n-1 quadratic functions of t, hence convex.
    # Use golden section search.
    gr = (math.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _score((1 - c) * f + c * g)
    fd = _score((1 - d) * f + d * g)
    
    for _ in range(max_iters):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _score((1 - c) * f + c * g)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _score((1 - d) * f + d * g)
    
    t_opt = (lo + hi) / 2
    return t_opt

def _lp_loop(f0, n, stop_time, seed):

    max_stall = 200

import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _conv(a):
    n = len(a)
    if n <= 2048:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(f, n, M, timeout=5.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _line_search(f, g, max_iters=30):
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    d1 = lo + gr * (hi - lo)
    fc = _score((1 - c1) * f + c1 * g)
    fd = _score((1 - d1) * f + d1 * g)
    for _ in range(max_iters):
        if fc < fd:
            hi = d1
            d1 = c1
            fd = fc
            c1 = hi - gr * (hi - lo)
            fc = _score((1 - c1) * f + c1 * g)
        else:
            lo = c1
            c1 = d1
            fc = fd
            d1 = lo + gr * (hi - lo)
            fd = _score((1 - d1) * f + d1 * g)

    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    if f.sum() < 1e-10:
        f = np.ones(n) / n

    best_f = f.copy()
    best_score = _score(f)
    last_best_score = best_score
    stagnation_iter = 0
    stall = 0
    max_stall = 100
    it = 0

    while time.monotonic() < stop_time:
        b = _conv(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_dir(f, n, M, timeout=5.0)
        if g is None:
            stall += 3
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.02 * max(1.0, f.std()), n), 0, None)
            if f.sum() < 1e-10:
                f = best_f.copy()
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            stall += 2
            continue
        g = g * (f.sum() / g_sum)

        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < 1e-12:
            stall += 3
            if stall > max_stall:
                break
            continue

        t_opt = _line_search(f, g, max_iters=25)

        f_new = np.clip((1 - t_opt) * f + t_opt * g, 0, None)
        sc_new = _score(f_new)

        if sc_new < best_score - 1e-14:
            f = f_new
            best_f = f.copy()
            best_score = sc_new
            stall = 0
            if best_score < last_best_score - 1e-8:
                last_best_score = best_score
                stagnation_iter = 0
            else:
                stagnation_iter += 1
        elif sc_new < _score(f) - 1e-14:
            f = f_new
            stall = 0
        else:
            stall += 1
            stagnation_iter += 1

        it += 1

        # Periodic CD polish
        if it % 15 == 0 and time.monotonic() < stop_time - 8.0:
            f_c, sc_c = _cd_polish(f, time.monotonic() + 4.0, max_outer=3, seed=seed + it)
            if sc_c < best_score:
                best_score = sc_c
                best_f = f_c.copy()
                f = f_c.copy()
                if best_score < last_best_score - 1e-8:
                    last_best_score = best_score
                    stagnation_iter = 0

        # Stagnation handling
        if stagnation_iter > 80 and time.monotonic() < stop_time - 20.0:
            f = np.clip(best_f + rng.normal(0, 0.04 * max(1.0, np.mean(best_f)), n), 0, None)
            if f.sum() < 1e-10:
                f = best_f.copy()
            else:
                f = f * (best_f.sum() / f.sum())
            stall = 0
            stagnation_iter = 0
            last_best_score = best_score
            max_stall = min(max_stall + 30, 300)

        if stall >= max_stall:
            break

    # Final CD
    if time.monotonic() < stop_time - 3.0:
        f_c, sc_c = _cd_polish(best_f, stop_time, max_outer=15, seed=seed + 9999)
        if sc_c <= best_score + 1e-14:
            best_f = f_c
            best_score = sc_c

    return best_f, best_score


def _peak_focused_cd(f, stop_time, max_outer=100, seed=0):
    """CD that prioritizes coordinates contributing to the peak autocorrelation."""
    n = len(f)
    rng = np.random.default_rng(seed)
    f = f.copy()
    cur = _score(f)
    best = f.copy()
    best_sc = cur

    mults = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.5, 0.0]

    for outer in range(max_outer):
        if time.monotonic() > stop_time:
            break
        b = _conv(f)
        peak_lags = np.where(b >= b.max() * 0.999)[0]
        # Importance of each coordinate: how much it contributes to peak lags
        importance = np.zeros(n)
        for kl in peak_lags:
            lo_j = max(0, int(kl) - n + 1)
            hi_j = min(n - 1, int(kl))
            for j in range(lo_j, hi_j + 1):
                importance[j] += b[kl] * (f[j] + (f[kl - j] if 0 <= kl - j < n else 0))
        importance /= max(1e-15, importance.max())

        # Order by importance (descending) with some randomness
        order = np.argsort(-importance + rng.uniform(0, 0.1, n)).astype(int)

        improved_any = False
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            best_v = orig
            best_local = cur
            for m in mults:
                v = max(0.0, orig * m) if orig > 1e-12 else (0.0 if m < 1.0 else 0.01)
                f[i] = v
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_local:
                    best_local = sc
                    best_v = v
            f[i] = best_v
            if best_local < cur - 1e-15:
                cur = best_local
                improved_any = True
                if cur < best_sc:
                    best_sc = cur
                    best = f.copy()

        if not improved_any and outer >= 5:
            break

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(400, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(400, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    n_parent = len(parent)
    n = n_parent

    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(200)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_and_update(res, sc, trial_name=""):
        nonlocal best_seq, best_score
        if res is None or len(res) == 0 or not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0, 1000.0)
        if res.sum() < 0.01:
            return
        if sc < best_score - 1e-14:
            try:
                exact = float(evaluate_sequence([float(x) for x in res]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-14:
                best_seq = res.copy()
                best_score = exact

    # --- Generate initializations ---
    i_arr = np.arange(n, dtype=np.float64)
    ctr = (n - 1) / 2.0
    rng_main = np.random.default_rng(seed)

    init_list = [
        ("parent", parent.copy()),
        ("cos", 0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / n))),
        ("cos_shift", 0.5 * (1.0 + np.cos(2.0 * np.pi * (i_arr + 0.12 * n) / n))),
        ("parab", np.maximum(0.0, 1.0 - ((i_arr - ctr) / (0.42 * n)) ** 2)),
        ("triang", 1.0 - 2.0 * np.abs(i_arr / n - 0.5)),
        ("two_lobe", np.exp(-0.5 * ((i_arr - 0.25 * n) / (0.07 * n)) ** 2) +
                   np.exp(-0.5 * ((i_arr - 0.75 * n) / (0.07 * n)) ** 2)),
        ("superflat", 1.0 + 0.5 * np.cos(2 * np.pi * i_arr / n) + 0.3 * np.cos(4 * np.pi * i_arr / n)),
        ("exp_sym", np.exp(-((i_arr - ctr) / (n / 3.5)) ** 2)),
        ("noise_parent", np.abs(parent + rng_main.normal(0, 0.12 * max(1.0, np.mean(parent)), n)) + 1e-4),
        ("alt", np.where(np.arange(n) % 2 == 0, 1.0, 0.35) * (0.5 + 0.5 * np.cos(2 * np.pi * i_arr / n))),
        ("flat_ring", np.where((i_arr < 0.2 * n) | (i_arr > 0.8 * n), 0.3, 1.0)),
    ]

    # Score and sort initializations
    scored = []
    for name, ix in init_list:
        if ix.sum() < 1e-10:
            continue
        sc = _score(ix)
        scored.append((sc, name, ix))
    scored.sort(key=lambda x: x[0])

    # --- Phase 1: LP optimization (55% of budget) ---
    phase1_end = t0 + budget * 0.55
    time_per_init = max(12.0, (phase1_end - t0) / max(1, len(scored)))

    for idx, (sc_q, name, init) in enumerate(scored):
        if time.monotonic() > phase1_end - 10.0:
            break
        if best_score < 1.45:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        try:
            res, sc = _lp_loop(init, n, stop, seed + 1000 * (idx + 1))
        except Exception:
            continue
        _try_and_update(res, sc, name)

    # --- Phase 2: Warm restart on best (15% of budget) ---
    phase2_end = t0 + budget * 0.72
    if time.monotonic() < phase2_end - 15.0:
        stop_p2 = min(phase2_end - 5.0, time.monotonic() + (phase2_end - time.monotonic()) * 0.85)
        try:
            res, sc = _lp_loop(best_seq.copy(), len(best_seq), stop_p2, seed + 77777)
        except Exception:
            pass
        else:
            _try_and_update(res, sc, "warm")

    # --- Phase 3: Peak-focused CD (12% of budget) ---
    phase3_end = t0 + budget * 0.84
    if time.monotonic() < phase3_end - 5.0:
        try:
            f_c, sc_c = _peak_focused_cd(best_seq, phase3_end, max_outer=80, seed=seed * 3 + 1)
        except Exception:
            f_c, sc_c = best_seq.copy(), best_score
        _try_and_update(f_c, sc_c, "peak_cd")

    # --- Phase 4: Try nearby lengths (8% of budget) ---
    phase4_end = t0 + budget * 0.93
    if time.monotonic() < phase4_end - 12.0:
        for L in [n - 8, n + 8, n - 16, n + 16, n + 32, n - 32]:
            if L < 80 or time.monotonic() > phase4_end - 10.0:
                continue
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
            stop_l = min(phase4_end - 5.0, time.monotonic() + 18.0)
            try:
                res, sc = _lp_loop(init_l, L, stop_l, seed + 90000 + L)
            except Exception:
                continue
            _try_and_update(res, sc, f"len{L}")

    # --- Phase 5: Final aggressive CD (remaining time) ---
    final_end = deadline - 5.0
    if time.monotonic() < final_end - 8.0:
        try:
            f_c, sc_c = _cd_polish(best_seq, final_end, max_outer=200, seed=seed * 7 + 42)
        except Exception:
            f_c, sc_c = best_seq.copy(), best_score
        _try_and_update(f_c, sc_c, "final_cd")

        # One more peak-focused pass
        if time.monotonic() < final_end - 5.0 and best_score > 1.5027:
            try:
                f_c2, sc_c2 = _peak_focused_cd(best_seq, final_end, max_outer=50, seed=seed * 11 + 7)
            except Exception:
                f_c2, sc_c2 = best_seq.copy(), best_score
            _try_and_update(f_c2, sc_c2, "final_peak_cd")

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0, 1.0, 1.0]
        return [float(x) for x in fallback]

    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    if not result:
        result = [1.0]
    if any(math.isnan(x) or math.isinf(x) for x in result):
        return [float(x) for x in fallback]

    return result
