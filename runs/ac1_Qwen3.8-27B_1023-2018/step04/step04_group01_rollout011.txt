            f = np.clip(f + rng.normal(0, 0.005, n), 0, None)
            f = f / f.sum()
            S = 1.0
            conv = _conv_full(f)
            no_improve_sweeps = 0
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            v = f[i]
            S_rest = S - v
            fm = 0.0
            if i > 0:
                fm = max(fm, float(conv[:i].max()))
            if i + n <= 2 * n - 2:
                fm = max(fm, float(conv[i + n:].max()))
            end = i + n
            base_conv = conv[i:end]
            base_f = f[:n]
            pos_2i = i
            x_max = max(4.0 * v, 0.2 * float(f.max()), 0.01)
            x_max = min(x_max, 1.5)
            n_coarse = 60
            x_grid = np.linspace(0.0, x_max, n_coarse)
            best_x = v
            best_obj = best_score
            for x in x_grid:
                delta = x - v
                if abs(delta) < 1e-15:
                    m_new = float(conv.max())
                else:
                    affected = base_conv + 2.0 * delta * base_f
                    if 0 <= pos_2i < n:
                        affected = affected.copy()
                        affected[pos_2i] += delta * delta
                    m_new = max(fm, float(affected.max()))
                S_new = S_rest + x
                if S_new < 0.01:
                    continue
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-16:
                    best_obj = obj
                    best_x = x
            if best_x != v:
                dw = x_max / n_coarse
                x_lo = max(0.0, best_x - 2 * dw)
                x_hi = min(x_max, best_x + 2 * dw)
                x_fine = np.linspace(x_lo, x_hi, 30)
                for x in x_fine:
                    delta = x - v
                    if abs(delta) < 1e-15:
                        continue
                    affected = base_conv + 2.0 * delta * base_f
                    if 0 <= pos_2i < n:
                        affected = affected.copy()
                        affected[pos_2i] += delta * delta
                    m_new = max(fm, float(affected.max()))
                    S_new = S_rest + x
                    if S_new < 0.01:
                        continue
                    obj = 2.0 * n * m_new / (S_new * S_new)
                    if obj < best_obj - 1e-16:
                        best_obj = obj
                        best_x = x
            if best_x != v and best_obj < best_score - 1e-15:
                delta = best_x - v
                conv[i:end] += 2.0 * delta * f[:n]
                if 0 <= 2 * i <= 2 * n - 2:
                    conv[2 * i] += delta * delta
                f[i] = best_x
                S += delta
                best_score = best_obj
                improved = True
        if improved:
            no_improve_sweeps = 0
            cur_sc = 2.0 * n * conv.max() / (S * S)
            if cur_sc < best_score + 1e-14:
                best_f = f.copy()
                best_score = cur_sc
        else:
            no_improve_sweeps += 1
    f = np.clip(f, 0, 1000)
    return f, best_score


def _grad_phase(f0, stop_time, seed=123, max_iters=30000):
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / S
    best_f = f.copy()
    best_score = _score(f)
    rng = np.random.default_rng(seed)
    alpha = 0.03
    alpha_min = 1e-9
    tau = 60.0
    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break
        conv = _conv_full(f)
        S_val = float(f.sum())
        M = float(conv.max())
        if M < 1e-18:
            break
        score = 2.0 * n * M / (S_val * S_val)
        if score < best_score - 1e-14:
            best_score = score
            best_f = f.copy()
        b_shifted = conv - M
        tau_b = tau * b_shifted
        tau_b -= tau_b.max()
        p = np.exp(tau_b)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(2 * n - 1)
            p[np.argmax(conv)] = 1.0
            p_sum = 1.0
        p /= p_sum
        f_rev = f[::-1]
        conv_size = 2 * n - 1
        P = rfft(p, conv_size)
        Fr = rfft(f_rev, conv_size)
        conv_val = irfft(P * Fr, conv_size)[:conv_size]
        grad = 2.0 * conv_val[n - 1:2 * n - 1]
        score_grad = (2.0 * n / (S_val * S_val)) * (grad - 2.0 * M / S_val)
        f_raw = f - alpha * score_grad
        # Project to simplex
        f_proj = np.clip(f_raw, 0, None)
        fp_sum = f_proj.sum()
        if fp_sum < 1e-10:
            f_proj = np.ones(n) / n
        else:
            f_proj = f_proj / fp_sum
        sc_new = _score(f_proj)
        if sc_new < score - 1e-15:
            f = f_proj
            alpha = max(alpha_min, alpha * 1.01)
        else:
            alpha = max(alpha_min, alpha * 0.5)
            if alpha < 1e-7:
                f = np.clip(f + rng.normal(0, 0.003, n), 0, None)
                f = f / f.sum()
                alpha = 0.01
        if it % 5000 == 4999:
            f = np.clip(f + rng.normal(0, 0.008, n), 0, None)
            f = f / f.sum()
            alpha = 0.015
            tau = max(30.0, tau * 0.8)
    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 20.0
    parent = _get_parent()
    n_p = len(parent)
    # Fallback
    fallback = parent.copy()
    fb_sc = _score(fallback)
    if not math.isfinite(fb_sc):
        fallback = np.ones(32) / 32.0
        fb_sc = _score(fallback)
    best_seq = fallback.copy()
    best_score = fb_sc
    rng = np.random.default_rng(seed)

    def _update_best(seq, sc=None):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64).ravel()
        seq = np.clip(seq, 0, 1000)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        if sc is None:
            sc = _score(seq)
        if not math.isfinite(sc):
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-15:
            best_seq = seq.copy()
            best_score = exact

    # === Phase 1: LP optimization on parent ===
    t1_end = t0 + budget * 0.35
    try:
        res, sc = _lp_phase(parent, n_p, t1_end, rng, max_iters=100)
        if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
            res = res / res.sum()
            _update_best(res, sc)
    except Exception:
        pass

    # === Phase 2: Intensive coordinate descent ===
    t2_end = min(t0 + budget * 0.60, hard_deadline - 60)
    if time.monotonic() < t2_end - 5:
        try:
            res, sc = _coord_descent(best_seq, t2_end, max_sweeps=400, seed=seed + 1)
            _update_best(res, sc)
        except Exception:
            pass

    # === Phase 3: Gradient refinement ===
    t3_end = min(t0 + budget * 0.72, hard_deadline - 50)
    if time.monotonic() < t3_end - 5 and best_score > 1.50275:
        try:
            res, sc = _grad_phase(best_seq, t3_end, seed=seed + 2, max_iters=20000)
            _update_best(res, sc)
        except Exception:
            pass

    # === Phase 4: More coordinate descent ===
    t4_end = min(t0 + budget * 0.82, hard_deadline - 40)
    if time.monotonic() < t4_end - 5:
        try:
            res, sc = _coord_descent(best_seq, t4_end, max_sweeps=500, seed=seed + 3)
            _update_best(res, sc)
        except Exception:
            pass

    # === Phase 5: Try a second LP pass with perturbation ===
    t5_end = min(t0 + budget * 0.90, hard_deadline - 30)
    if time.monotonic() < t5_end - 10 and best_score > 1.50275:
        perturbed = np.clip(best_seq + rng.normal(0, 0.01, len(best_seq)), 0, None)
        perturbed = perturbed / perturbed.sum()
        try:
            res, sc = _lp_phase(perturbed, len(best_seq), t5_end - 5, rng, max_iters=50)
            if res is not None:
                res = res / res.sum()
                _update_best(res, sc)
        except Exception:
            pass

    # === Phase 6: Final coordinate polish ===
    t6_end = hard_deadline - 10
    if time.monotonic() < t6_end - 5:
        try:
            res, sc = _coord_descent(best_seq, t6_end, max_sweeps=300, seed=seed + 99)
            _update_best(res, sc)
        except Exception:
            pass

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf
    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]
    return [float(x) for x in best_seq]
