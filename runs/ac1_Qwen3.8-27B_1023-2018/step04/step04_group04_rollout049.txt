2. Solve the resulting convex problem (LP)
3. Update f

Let me think about the QP approach. The linearization of (f*f)_k at point f is:

def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _precompute_A(n):
    col = np.repeat(np.arange(n, dtype=np.int32), n)
    off = np.tile(np.arange(n, dtype=np.int32), n)
    row = (col + off).astype(np.int32)
    return row, col, off

def _lp_solve(f, n, M, struct, timeout=3.0):
    row, col, off = struct
    vals = f[off]
    A = csr_matrix((vals, (row, col)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g
