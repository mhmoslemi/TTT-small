        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)

    b = _conv_self(a)
    m = float(b.max())
    if not math.isfinite(m):

def _conv_self(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _cross_n(p, w):
    """Cross-correlation: (reverse(p) * w) centered, giving n-length output."""
    n = p.size
    L = 3 * n - 2
    size = next_fast_len(L)
    F = rfft(p[::-1], size)
    G = rfft(w, size)
    c = irfft(F * G, size)[:L]
    q = c[n-1:2*n-1]
    np.maximum(q, 0.0, out=q)
    return q


def _softmax(u):
    u = np.asarray(u, dtype=np.float64).ravel()
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0/u.size)
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.full(u.size, 1.0/u.size)
    return e / s

def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0/n)

        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m) + 0.5) / m
        xp_new = (np.arange(n) + 0.5) / n
