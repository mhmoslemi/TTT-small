```python
def project_simplex(v, s=1):
    """Project v onto {x >= 0, sum(x) = s}."""
    u = v.copy()
    # We want to find tau such that sum(max(u - tau, 0)) = s
    # Sort u in decreasing order
    u_sorted = np.sort(u)[::-1]
    # Find the largest k such that u_sorted[k] - (sum(u_sorted[:k]) - s) / k > 0
    cssv = np.cumsum(u_sorted)
    # Find where (cssv[k] - s) / (k+1) < u_sorted[k+1]
    # Or equivalently: cssv[k] - s < (k+1) * u_sorted[k+1]
    # This is the standard algorithm
    rho = np.nonzero(u_sorted * np.arange(1, len(u)+1) > (cssv - s))[0][-1]
    tau = (cssv[rho] - s) / (rho + 1)
    return np.maximum(u - tau, 0)
```

This is O(n log n) per projection, which is fast for n=384.

```python
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _project_simplex(v, s=1.0):
    n = len(v)
    u = np.asarray(v, dtype=np.float64).copy()
    if n == 0:
        return u
    u_sorted = np.sort(u)[::-1]
    cssv = np.cumsum(u_sorted)
    indices = np.arange(1, n + 1, dtype=np.float64)
    cond = u_sorted * indices > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n, dtype=np.float64)
    rho = int(np.nonzero(cond)[0][-1])
    tau = (cssv[rho] - s) / (rho + 1)
    return np.maximum(u - tau, 0.0)

    """Compute gradient of max(a*a) or log-sum-exp approximation."""
    n = len(a)
    b = _conv_full(a)
    if beta is None or beta == 0:
        # Subgradient at argmax
        idx = np.where(b == b.max())[0]
        w = np.zeros_like(b)
        w[idx] = 1.0 / len(idx)
    else:
        b_max = b.max()
        exp_b = np.exp(beta * (b - b_max))
        w = exp_b / exp_b.sum()
    
    # g[j] = sum_k w[k] * a[k-j]
    fsize = next_fast_len(2 * n + 1)
    W = rfft(w, fsize)
    A = rfft(a, fsize)
    g = irfft(W * np.conjugate(A), fsize)[:n]
    return g

    b = _conv_full(f)

    A = np.zeros((n_rows, n))

        j_end = min(n, k + 1)
        for j in range(j_start, j_end):

                     method='highs', options={'presolve': True, 'time_limit': time_limit})

        return None
    
    if res.status not in (0, 3) or res.x is None:
```

The line search finds the optimal t that actually decreases the max. This is what makes the LP approach work in practice.

```python
        if t_opt > 0 and m_opt < M - 1e-14:

            f = np.maximum(f, 0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            M = float(_conv_full(f).max())
            if M < best_M:
                best_M = M
                best_f = f.copy()
                top_k = max(100, top_k - 5)
            stall = 0
        else:
            stall += 1
            top_k = min(2 * n - 1, top_k + 10)
        iter_count += 1
    best_sc = _score(best_f)
    return best_f, best_sc


def _smoothed_grad_desc(a_init, stop_time, seed):
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = _project_simplex(a_init.copy(), 1.0)
    best_a = a.copy()
    best_sc = _score(best_a)
    fsize = next_fast_len(2 * n + 1)
    beta = 15.0
    lr = 0.08 / n
    no_improve = 0
    max_no_improve = 500
    while time.monotonic() < stop_time - 1.0 and no_improve < max_no_improve:
        b = _conv_full(a)
        b_max = b.max()
        scaled = beta * (b - b_max)
        scaled = np.clip(scaled, -50, 50)
        exp_b = np.exp(scaled)
        exp_sum = exp_b.sum()
        if exp_sum < 1e-300:
            w = np.zeros_like(b)
            mask = b == b_max
            w[mask] = 1.0 / mask.sum()
        else:
            w = exp_b / exp_sum
        W = rfft(w, fsize)
        A = rfft(a, fsize)
        g = irfft(W * np.conjugate(A), fsize)[:n]
        a_new = _project_simplex(a - lr * g, 1.0)
        sc = _score(a_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            beta = min(beta * 1.008, 500.0)
            lr = max(lr * 1.003, 1e-9)
            no_improve = 0
        else:
            if score_a := _score(a) > sc:
                a = a_new
            no_improve += 1
            if no_improve % 100 == 0:
                a = _project_simplex(best_a + rng.normal(0, 0.02, n), 1.0)
                lr = 0.08 / n
                beta = 15.0
                no_improve = 0
    return best_a, best_sc


def _coord_polish(a_init, stop_time, seed):
    rng = np.random.default_rng(seed)
    n = len(a_init)
    best = np.clip(np.asarray(a_init, dtype=np.float64).ravel().copy(), 0, None)
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    no_imp = 0
    max_no_imp = 10
    while time.monotonic() < stop_time - 1.0 and no_imp < max_no_imp:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            orig = best[i]
            if orig > 1e-10:
                mults = (0.0, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0)
            else:
                mults = (0.0, 0.005, 0.01, 0.05, 0.1, 0.5, 1.0)
            best_val = orig
            best_val_sc = best_sc
            for mlt in mults:
                nv = orig * mlt if orig > 1e-10 else mlt
                best[i] = nv
                sc = _score(best)
                if math.isfinite(sc) and sc < best_val_sc:
                    best_val = nv
                    best_val_sc = sc
            best[i] = best_val
            if best_val_sc < best_sc - 1e-15:
                best_sc = best_val_sc
                improved = True
        for _ in range(min(200, n * 2)):
            if time.monotonic() > stop_time - 1.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            total = oi + oj
            if total < 1e-12:
                continue
            for frac in (0.0, 0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9, 1.0):
                best[i] = frac * total
                best[j] = (1.0 - frac) * total
                sc = _score(best)
                if math.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    improved = True
        best[i], best[j] = oi, oj
        for _ in range(min(40, n // 4)):
            if time.monotonic() > stop_time - 1.0:
                break
            idx = rng.choice(n, size=3, replace=False)
            i, j, k = int(idx[0]), int(idx[1]), int(idx[2])
            oi, oj, ok = best[i], best[j], best[k]
            total = oi + oj + ok
            if total < 1e-12:
                continue
            for fi in (0.0, 0.2, 0.4, 0.5):
                for fj in (0.0, 0.2, 0.4, 0.5):
                    fk = 1.0 - fi - fj
                    if_fk = max(0.0, fk)
                    s = fi + fj + if_fk
                    if s < 1e-9:
                        continue
                    fi2, fj2, fk2 = fi / s, fj / s, if_fk / s
                    best[i], best[j], best[k] = fi2 * total, fj2 * total, fk2 * total
                    sc = _score(best)
                    if math.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        improved = True
        best[i], best[j], best[k] = oi, oj, ok
        if not improved:
            no_imp += 1
        else:
            no_imp = 0
    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0
    try:
        h = globals()["height_sequence_1"]
        parent = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(384, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(384, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    n_p = len(parent)
    n = n_p
    fallback = parent.copy()
    try:
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(n) / n
        fallback_sc = _score(fallback)
    best_seq = fallback.copy()
    best_score = fallback_sc
    rng_main = np.random.default_rng(seed)
    phase1_end = t0 + budget * 0.50
    phase2_end = t0 + budget * 0.72
    phase3_end = deadline - 5.0
    inits = []
    inits.append(("parent", parent.copy() / max(parent.sum(), 1e-10)))
    i_arr = np.arange(n, dtype=np.float64)
    taper = 0.5 + 0.5 * np.cos(np.pi * i_arr / (n - 1))
    inits.append(("taper", taper / taper.sum()))
    kern = np.exp(-0.5 * (np.arange(-8, 9) / 3.0) ** 2)
    kern /= kern.sum()
    smooth_noise = np.convolve(rng_main.normal(0, 1, n), kern, 'same')
    p_noisy = np.clip(parent + 0.12 * smooth_noise * max(parent), 0, None)
    if p_noisy.sum() > 1e-10:
        inits.append(("noisy", p_noisy / p_noisy.sum()))
    z = np.ones(n)
    z[::2] *= 1.4
    z[1::2] *= 0.65
    inits.append(("zipper", z / z.sum()))
    sup = np.ones(n) * 0.15
    for fr in (1, 2, 3, 5, 7, 11):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.25 * np.cos(2 * math.pi * fr * i_arr / n + ph)
    sup = np.abs(sup) + 0.001
    inits.append(("multifreq", sup / sup.sum()))
    time_per_init = max(20.0, (phase1_end - t0) / len(inits))
    for idx, (name, init) in enumerate(inits):
        if time.monotonic() > phase1_end - 15.0:
            break
        stop = min(phase1_end - 10.0, time.monotonic() + time_per_init)
        if time.monotonic() >= stop:
            break
        try:
            res, sc = _lp_optimize(init, n, stop, seed + 7919 * (idx + 1))
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_clipped = np.clip(res, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in res_clipped]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = res_clipped.copy()
                    best_score = exact
        except Exception:
            pass
    if time.monotonic() < phase2_end - 10.0:
        grad_stop = min(phase2_end - 5.0, time.monotonic() + (phase2_end - time.monotonic()) * 0.7)
        if time.monotonic() < grad_stop - 5.0:
            try:
                res, sc = _smoothed_grad_desc(best_seq, grad_stop, seed + 31337)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res_clipped = np.clip(res, 0, 1000)
                    exact = float(evaluate_sequence([float(x) for x in res_clipped]))
                    if math.isfinite(exact) and exact < best_score:
                        best_seq = res_clipped.copy()
                        best_score = exact
            except Exception:
                pass
        if time.monotonic() < phase2_end - 8.0:
            try:
                res, sc = _lp_optimize(best_seq, n, phase2_end - 5.0, seed + 99991)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res_clipped = np.clip(res, 0, 1000)
                    exact = float(evaluate_sequence([float(x) for x in res_clipped]))
                    if math.isfinite(exact) and exact < best_score:
                        best_seq = res_clipped.copy()
                        best_score = exact
            except Exception:
                pass
    if time.monotonic() < phase3_end - 8.0:
        polish_time_each = max(15.0, (phase3_end - time.monotonic()) / 3.0)
        for pidx in range(3):
            p_stop = min(time.monotonic() + polish_time_each, phase3_end - 3.0)
            if time.monotonic() >= p_stop - 2.0:
                break
            try:
                res, sc = _coord_polish(best_seq, p_stop, seed + 50000 + pidx * 997)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res_clipped = np.clip(res, 0, 1000)
                    exact = float(evaluate_sequence([float(x) for x in res_clipped]))
                    if math.isfinite(exact) and exact < best_score:
                        best_seq = res_clipped.copy()
                        best_score = exact
            except Exception:
                pass
    final_seq = np.clip(best_seq, 0, 1000)
    final_seq = np.where(np.isfinite(final_seq), final_seq, 0.0)
    if final_seq.sum() < 0.01:
        final_seq = fallback.copy()
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf
    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        final_seq = fallback
    return [float(x) for x in final_seq]
```
