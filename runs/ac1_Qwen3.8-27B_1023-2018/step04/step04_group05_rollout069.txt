    if time.monotonic() < stop - 8.0:
        p3_end = min(stop - 2.0, t0 + budget * 0.985)
        try:
            r, sc = _polish(best_f, p3_end, seed + 999999)
            asc = _exa(r)
            if math.isfinite(asc) and asc < best_sc - 1e-12:
                best_f = r.copy()
                best_sc = asc
        except Exception:
            pass

    # Final sanitise + authoritative validation
    best_f = np.clip(np.asarray(best_f, dtype=np.float64).ravel(), 0, 1000.0)
    best_f = np.where(np.isfinite(best_f), best_f, 0.0)
    if best_f.size == 0 or best_f.sum() < 0.01:
        best_f = np.ones(198)
    final = _exa(best_f)
    if not math.isfinite(final):
        fb = np.clip(p.copy(), 0, 1000.0)
        fb = np.where(np.isfinite(fb), fb, 0.0)
        fbv = _exa(fb)
        if not math.isfinite(fbv):
            return [1.0]
        return [float(x) for x in fb]
    if time.monotonic() > hard:
        pass
    return [float(x) for x in best_f]
