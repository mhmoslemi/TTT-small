```python
def _lp_direction_v2(f):
    """Find g >= 0, sum(g)=1, minimizing max_k (f*g)_k"""
    n = len(f)
    # Build the LP
    # Variables: g_0, ..., g_{n-1}, t (total n+1)
    # Objective: minimize t -> c = [0, ..., 0, 1]
    # Constraints: sum_i f_i * g_{k-i} - t <= 0 for k = 0, ..., 2n-2
    #              sum g_i = 1
    #              g_i >= 0
    
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Build constraint matrix for (f*g)_k <= t
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i  # index into g
            if 0 <= j < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
        # -t term
        rows.append(k)
        cols.append(n)  # t is the last variable
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    # Equality constraint: sum g = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # g_i >= 0, t free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = res.x[:n]
        g = np.clip(g, 0, None)
        if g.sum() > 1e-15:
            g = g / g.sum()
        return g
    return None
```
