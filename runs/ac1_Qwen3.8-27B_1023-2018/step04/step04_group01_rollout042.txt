    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(v - theta, 0)


def _grad_optimize(f0, n, stop_time, seed, max_iters=80000):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    S = float(f.sum())
    if S < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / S
    best_f = f.copy()
    best_score = _score(f)

    alpha = 0.5
    alpha_min = 1e-10
    tau = 30.0
    fsize = next_fast_len(2 * n - 1)
    conv_size = 2 * n - 1

    for it in range(max_iters):
        if time.monotonic() > stop_time - 5.0:
            break

        F = rfft(f, fsize)
        c = irfft(F * F, fsize)[:conv_size]
        np.maximum(c, 0.0, out=c)
        M = float(c.max())
        if M < 1e-18:
            break

        # Soft-max weights
        shifted = c - M
        scaled = tau * shifted
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-30 or not np.isfinite(p_sum):
            p = np.zeros(conv_size)
            p[np.argmax(c)] = 1.0
            p_sum = 1.0
        p /= p_sum

        # Gradient computation via FFT
        f_rev = f[::-1]
        Fr = rfft(f_rev, fsize)
        P = rfft(p, fsize)
        conv_val = irfft(P * Fr, fsize)
        grad = 2.0 * conv_val[n - 1:2 * n - 1]

        # Projected gradient step with backtracking
        step = alpha
        accepted = False
        curr_score = _score(f)
        for _bt in range(10):
            g_raw = f - step * grad
            f_new = _proj_simplex(g_raw)
            sc_new = _score(f_new)
            if np.isfinite(sc_new) and sc_new < curr_score - 1e-16:
                f = f_new
                if sc_new < best_score - 1e-14:
                    best_score = sc_new
                    best_f = f.copy()
                alpha = min(2.0, alpha * 1.1)
                accepted = True
                break
            step *= 0.5
            if step < alpha_min:
                break

        if not accepted:
            alpha = max(alpha_min, alpha * 0.7)
            if alpha < 1e-7:
                f = np.clip(f + rng.normal(0, 0.005, n), 0, None)
                f /= f.sum()
                alpha = 0.01

        # Adaptive temperature
        if it % 3000 == 2999:
            tau = min(tau * 1.4, 800.0)

        # Periodic perturbation
        if it % 8000 == 7999:
            f = np.clip(f + rng.normal(0, 0.008, n), 0, None)
            f /= f.sum()
            alpha = 0.05

    return best_f, best_score


def _coord_descent(f0, stop_time, max_sweeps=300, seed=42):
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        f = np.ones(n) / n
        S = 1.0
    else:
        f = f / S
        S = 1.0

    conv = _conv_full(f)
    best_score = 2.0 * n * float(conv.max())
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break

            v = f[i]
            S_rest = S - v
            start = i
            end = min(i + n, 2 * n - 1)
            affected_len = end - start

            # Precompute unaffected max
            if start > 0 or end < 2 * n - 1:
                if start == 0:
                    unaffected_max = float(conv[end:].max())
                elif end == 2 * n - 1:
                    unaffected_max = float(conv[:start].max())
                else:
                    unaffected_max = max(float(conv[:start].max()), float(conv[end:].max()))
            else:
                unaffected_max = 0.0

            curr_max = float(conv.max())

            # Determine search range
            if v > 1e-10:
                x_max = min(10.0 * v, 5.0 / n)
            else:
                x_max = 3.0 / n
            x_max = max(x_max, 1.0 / n)

            # Coarse grid
            n_pts = 48
            x_grid = np.linspace(0.0, x_max, n_pts)
            best_x = v
            best_obj = best_score

            f_seg = f[:affected_len]  # f[0], f[1], ..., f[affected_len-1] corresponds to f[k-i]

            for x in x_grid:
                S_new = S_rest + x
                if S_new < 0.01:
                    continue
                w = x - v
                if abs(w) < 1e-15:
                    m_new = curr_max
                else:
                    c_affected = conv[start:end] + 2.0 * w * f_seg
                    if 0 <= 2 * i < 2 * n - 1:
                        c_affected[2 * i - start] += w * w
                    m_affected = float(c_affected.max())
                    m_new = max(m_affected, unaffected_max)
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-16:
                    best_obj = obj
                    best_x = x

            # Fine refinement
            if best_x != v and n_pts > 2:
                dw = x_max / (n_pts - 1)
                x_lo = max(0.0, best_x - 3 * dw)
                x_hi = min(x_max, best_x + 3 * dw)
                if x_hi > x_lo + 1e-15:
                    x_fine = np.linspace(x_lo, x_hi, 24)
                    for x in x_fine:
                        S_new = S_rest + x
                        if S_new < 0.01:
                            continue
                        w = x - v
                        if abs(w) < 1e-15:
                            continue
                        c_affected = conv[start:end] + 2.0 * w * f_seg
                        if 0 <= 2 * i < 2 * n - 1:
                            c_affected[2 * i - start] += w * w
                        m_affected = float(c_affected.max())
                        m_new = max(m_affected, unaffected_max)
                        obj = 2.0 * n * m_new / (S_new * S_new)
                        if obj < best_obj - 1e-16:
                            best_obj = obj
                            best_x = x

            # Accept
            if best_x != v and best_obj < best_score - 1e-15:
                w = best_x - v
                conv[start:end] += 2.0 * w * f_seg
                if 0 <= 2 * i < 2 * n - 1:
                    conv[2 * i] += w * w
                f[i] = best_x
                S += w
                best_score = best_obj
                best_f = f.copy()
                improved = True

        if not improved and sweep > 8:
            # Try perturbation
            f = np.clip(f + rng.normal(0, 0.003, n), 0, None)
            f /= f.sum()
            S = 1.0
            conv = _conv_full(f)
            sc = 2.0 * n * float(conv.max())
            if sc < best_score - 1e-14:
                best_score = sc
                best_f = f.copy()

    return best_f, best_score


def _lp_refine(f0, n, stop_time, seed, max_iters=2000):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    S = float(f.sum())
    if S < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / S
    best_f = f.copy()
    best_score = _score(f)

    stall = 0
    max_stall = 50

    for it in range(max_iters):
        if time.monotonic() > stop_time - 5.0:
            break

        g_dir = _lp_direction(f, n, time_limit=4.0)
        if g_dir is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.01, n), 0, None)
            f /= f.sum()
            continue

        # Line search on direction f + t*(g - f)
        curr_score = _score(f)
        best_t = 0.0
        best_sc = curr_score
        lo, hi = 0.0, 1.0
        for _ in range(14):
            mid = (lo + hi) / 2.0
            f_new = np.clip(f + mid * (g_dir - f), 0, None)
            fs = f_new.sum()
            if fs < 0.01:
                hi = mid
                continue
            f_new = f_new / fs
            sc = _score(f_new)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_t = mid
                lo = mid
            else:
                hi = mid

        if best_t > 1e-5 and best_sc < best_score - 1e-14:
            f_new = np.clip(f + best_t * (g_dir - f), 0, None)
            f_new /= f_new.sum()
            f = f_new
            best_f = f.copy()
            best_score = best_sc
            stall = 0
        elif best_t > 1e-5 and best_sc < curr_score - 1e-14:
            f_new = np.clip(f + best_t * (g_dir - f), 0, None)
            f_new /= f_new.sum()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
                f /= f.sum()
                stall = 0

    return best_f, best_score


def _pair_polish(f0, stop_time, max_rounds=30, seed=777):
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / f.sum()
    best_score = _score(f)
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 5.0:
            break
        improved = False
        for _ in range(40):
            if time.monotonic() > stop_time - 5.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-12 and oj < 1e-12:
                continue
            multipliers = (0.3, 0.5, 0.7, 1.0, 1.4, 2.0, 0.0)
            best_vals = (oi, oj)
            best_sc = best_score
            for mi in multipliers:
                ni = oi * mi if oi > 1e-12 else (0.01 / n if 0.1 < mi < 10 else 0.0)
                for mj in multipliers:
                    nj = oj * mj if oj > 1e-12 else (0.01 / n if 0.1 < mj < 10 else 0.0)
                    f[i], f[j] = ni, nj
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        best_vals = (ni, nj)
            f[i], f[j] = best_vals
            if best_sc < best_score - 1e-14:
                best_score = best_sc
                best_f = f.copy()
                improved = True
        if not improved and rnd > 5:
            break

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard_deadline = t0 + budget - 10.0

    parent = _get_parent()
    n_p = len(parent)

    # Fallback
    fallback = parent.copy()
    try:
        fallback_sc = _score(fallback)
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(400) / 400.0
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update_best(seq, sc=None):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64).ravel()
        seq = np.clip(seq, 0, 1000)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        if sc is None:
            try:
                sc = _score(seq)
            except Exception:
                return
        if not math.isfinite(sc):
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-14:
            best_seq = seq.copy()
            best_score = exact

    # === Phase 1: LP refinement on parent ===
    phase1_end = t0 + budget * 0.25
    if time.monotonic() < phase1_end - 10:
        try:
            res, sc = _lp_refine(parent, n_p, phase1_end, seed)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res = res / res.sum()
                _update_best(res, sc)
        except Exception:
            pass

    # === Phase 2: Gradient descent ===
    phase2_end = t0 + budget * 0.45
    if time.monotonic() < phase2_end - 10 and best_score > 1.5028:
        try:
            res, sc = _grad_optimize(best_seq, len(best_seq), phase2_end, seed + 1)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update_best(res, sc)
        except Exception:
            pass

    # === Phase 3: Coordinate descent ===
    phase3_end = t0 + budget * 0.70
    if time.monotonic() < phase3_end - 10:
        try:
            res, sc = _coord_descent(best_seq, phase3_end, max_sweeps=200, seed=seed + 2)
            _update_best(res, sc)
        except Exception:
            pass

    # === Phase 4: Another LP round ===
    phase4_end = t0 + budget * 0.80
    if time.monotonic() < phase4_end - 10 and best_score > 1.5028:
        try:
            res, sc = _lp_refine(best_seq, len(best_seq), phase4_end, seed + 100)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update_best(res, sc)
        except Exception:
            pass

    # === Phase 5: Fine coordinate descent ===
    phase5_end = t0 + budget * 0.92
    if time.monotonic() < phase5_end - 10:
        try:
            res, sc = _coord_descent(best_seq, phase5_end, max_sweeps=150, seed=seed + 200)
            _update_best(res, sc)
        except Exception:
            pass

    # === Phase 6: Pair polish ===
    phase6_end = hard_deadline - 15.0
    if time.monotonic() < phase6_end - 5:
        try:
            res, sc = _pair_polish(best_seq, phase6_end, max_rounds=20, seed=seed + 300)
            _update_best(res, sc)
        except Exception:
            pass

    # === Phase 7: Try different lengths briefly ===
    if time.monotonic() < hard_deadline - 30 and best_score > 1.5028:
        for L in [n_p - 16, n_p + 16, n_p // 2]:
            if time.monotonic() > hard_deadline - 25:
                break
            if L < 32 or L > 1024:
                continue
            init_l = np.interp(
                np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq
            )
            init_l = np.clip(init_l, 0, None)
            if init_l.sum() < 0.01:
                init_l = np.ones(L) / L
            init_l = init_l / init_l.sum()
            t_alloc = min(hard_deadline - 20, time.monotonic() + 25.0)
            try:
                res, sc = _coord_descent(init_l, t_alloc, max_sweeps=30, seed=seed + L)
                _update_best(res, sc)
            except Exception:
                pass

    # === Final validation ===
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64).ravel(), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
