            obj[~valid] = math.inf

            idx_best = int(np.argmin(obj))
            if obj[idx_best] < best_obj - 1e-15:
                best_obj = obj[idx_best]
                best_x = x_grid[idx_best]

            # Fine grid around best
            if best_x != v:
                dw = (x_max - x_min) / n_pts
                x_lo = max(0.0, best_x - 4 * dw)
                x_hi = min(x_max, best_x + 4 * dw)
                x_fine = np.linspace(x_lo, x_hi, 60)
                d_fine = x_fine - v
                new_conv_fine = conv_affected[None, :] + 2.0 * (d_fine * f_slice)[None, :]
                new_conv_fine[:, 0] += d_fine**2
                max_new_fine = np.zeros(len(x_fine))
                for k in range(0, end - start):
                    max_new_fine = np.maximum(max_new_fine, new_conv_fine[:, k])
                if start > 0:
                    max_new_fine = np.maximum(max_new_fine, conv[:start].max())
                if end < 2 * n - 1:
                    max_new_fine = np.maximum(max_new_fine, conv[end:].max())
                new_S_fine = S_rest + x_fine
                valid_fine = new_S_fine > 0.01
                if valid_fine.any():
                    obj_fine = 2.0 * n * max_new_fine / (new_S_fine * new_S_fine)
                    obj_fine[~valid_fine] = math.inf
                    idx_fine = int(np.argmin(obj_fine))
                    if obj_fine[idx_fine] < best_obj - 1e-15:
                        best_obj = obj_fine[idx_fine]
                        best_x = x_fine[idx_fine]

            if abs(best_x - v) > 1e-12 and best_obj < best_score - 1e-14:
                w = best_x - v
                f[i] = best_x
                S += w
                conv[start:end] += 2.0 * w * f_slice
                conv[i] += w * w
                best_score = best_obj
                best_f = f.copy()
                improved = True

        if not improved and sweep > 5:
            break

    f = np.clip(best_f, 0, 1000)
    final_score = _score(f)
    return f, final_score


def _grad_optimize(f0, n, stop_time, seed, max_iters=30000):
    """Gradient-based with smoothed max, projected onto simplex."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum
    best_f = f.copy()
    best_score = _score(f)

    alpha = 0.08
    alpha_min = 1e-12
    tau = 100.0
    fsize = next_fast_len(2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() > stop_time - 3.0:
            break

        b = _conv_full(f)
        S = float(f.sum())
        M = float(b.max())
        if M < 1e-18:
            break
        score = 2.0 * n * M / (S * S)

        if score < best_score - 1e-14:
            best_score = score
            best_f = f.copy()

        b_shifted = b - M
        tau_b = tau * b_shifted
        tau_b -= tau_b.max()
        p = np.exp(tau_b)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(2 * n - 1)
            p[np.argmax(b)] = 1.0
            p_sum = 1.0
        p /= p_sum

        f_rev = f[::-1]
        conv_size = 2 * n - 1
        P = rfft(p, conv_size)
        Fr = rfft(f_rev, conv_size)
        conv_val = irfft(P * Fr, conv_size)[:conv_size]
        grad = 2.0 * conv_val[n - 1:2 * n - 1]

        score_grad = (2.0 * n / (S * S)) * (grad - 2.0 * M / S)

        g_raw = f - alpha * score_grad
        f_new = _project_simplex(g_raw)

        sc_new = _score(f_new)
        if sc_new < score - 1e-15:
            f = f_new
            alpha = max(alpha_min, alpha * 0.97)
            # Reduce tau periodically for sharper max approximation
            if it % 5000 == 4999:
                tau = max(20.0, tau * 0.8)
        else:
            alpha = max(alpha_min, alpha * 0.5)
            if alpha < alpha_min:
                f = np.clip(f + rng.normal(0, 0.003, n), 0, None)
                f /= f.sum()
                alpha = 0.01
                tau = 80.0

        if it % 3000 == 2999:
            f = np.clip(f + rng.normal(0, 0.01, n), 0, None)
            f /= f.sum()
            alpha = 0.02

    return best_f, best_score


def _pair_polish(f0, stop_time, max_rounds=80, seed=9999):
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    if f.sum() < 0.01:
        f = np.ones(n) / n
    f = f / f.sum()
    best_score = _score(f)
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 3.0:
            break
        improved = False
        n_trials = 80
        for _ in range(n_trials):
            if time.monotonic() > stop_time - 2.0:
                break
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-12 and oj < 1e-12:
                continue

            multipliers = (0.1, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 0.0)
            best_pair = (oi, oj)
            best_pair_sc = best_score

            for mi in multipliers:
                ni = oi * mi if oi > 1e-12 else (0.01 if mi > 0 else 0.0)
                for mj in multipliers:
                    nj = oj * mj if oj > 1e-12 else (0.01 if mj > 0 else 0.0)
                    f[i], f[j] = ni, nj
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_pair_sc - 1e-13:
                        best_pair_sc = sc
                        best_pair = (ni, nj)

            f[i], f[j] = best_pair
            if best_pair_sc < best_score - 1e-13:
                best_score = best_pair_sc
                best_f = f.copy()
                improved = True

        if not improved and rnd > 10:
            break

    return best_f, best_score


def _block_optimize(f0, stop_time, seed=555):
    """Optimize blocks of 4 coordinates using Nelder-Mead."""
    from scipy.optimize import minimize
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / f.sum()
    best_score = _score(f)
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    block_size = 4
    n_blocks = n // block_size

    for sweep in range(30):
        if time.monotonic() > stop_time - 3.0:
            break
        improved = False
        order = rng.permutation(n_blocks)
        for bi in order:
            if time.monotonic() > stop_time - 2.0:
                break
            start_idx = bi * block_size
            end_idx = start_idx + block_size
            x0 = f[start_idx:end_idx].copy()
            if x0.max() < 1e-10:
                continue

            def obj(x):
                x = np.maximum(x, 0.0)
                f_trial = f.copy()
                f_trial[start_idx:end_idx] = x
                s = f_trial.sum()
                if s < 0.01:
                    return 1e10
                f_trial = f_trial / s
                return _score(f_trial)

            try:
                res = minimize(obj, x0, method='Nelder-Mead',
                               options={'maxiter': 50, 'xatol': 1e-8, 'fatol': 1e-10})
                if res.fun < best_score - 1e-12:
                    f_new_vals = np.maximum(res.x, 0.0)
                    f_trial = f.copy()
                    f_trial[start_idx:end_idx] = f_new_vals
                    f_trial = f_trial / f_trial.sum()
                    sc = _score(f_trial)
                    if sc < best_score - 1e-12:
                        f = f_trial
                        best_score = sc
                        best_f = f.copy()
                        improved = True
            except Exception:
                pass

        if not improved:
            break

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    fallback = parent.copy()
    try:
        fallback_sc = _score(fallback)
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(32) / 32.0
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update_best(seq, sc=None):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64)
        seq = np.clip(seq, 0, 1000)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        if sc is None:
            sc = _score(seq)
        if not math.isfinite(sc):
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-15:
            best_seq = seq.copy()
            best_score = exact

    # === Phase 1: LP optimization on parent ===
    lp1_end = min(t0 + budget * 0.30, hard_deadline - 200)
    if time.monotonic() < lp1_end - 5:
        try:
            res, sc = _lp_optimize(parent, n_p, lp1_end, seed)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res = res / res.sum()
                _update_best(res, sc)
        except Exception:
            pass

    # === Phase 2: Coordinate descent ===
    cd1_end = min(t0 + budget * 0.50, hard_deadline - 150)
    if time.monotonic() < cd1_end - 5:
        try:
            res, sc = _coord_descent(best_seq, cd1_end, max_sweeps=150, seed=seed + 1)
            _update_best(res, sc)
        except Exception:
            pass

    # === Phase 3: More LP ===
    lp2_end = min(t0 + budget * 0.60, hard_deadline - 120)
    if time.monotonic() < lp2_end - 5:
        try:
            res, sc = _lp_optimize(best_seq, n_p, lp2_end, seed + 100)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res = res / res.sum()
                _update_best(res, sc)
        except Exception:
            pass

    # === Phase 4: Gradient optimization ===
    grad_end = min(t0 + budget * 0.70, hard_deadline - 100)
    if time.monotonic() < grad_end - 5:
        try:
            res, sc = _grad_optimize(best_seq, n_p, grad_end, seed + 200)
            _update_best(res, sc)
        except Exception:
            pass

    # === Phase 5: Intensive coordinate descent ===
    cd2_end = min(t0 + budget * 0.82, hard_deadline - 80)
    if time.monotonic() < cd2_end - 5:
        try:
            res, sc = _coord_descent(best_seq, cd2_end, max_sweeps=200, seed=seed + 300)
            _update_best(res, sc)
        except Exception:
            pass

    # === Phase 6: Block optimization ===
    blk_end = min(t0 + budget * 0.88, hard_deadline - 50)
    if time.monotonic() < blk_end - 5:
        try:
            res, sc = _block_optimize(best_seq, blk_end, seed=seed + 400)
            _update_best(res, sc)
        except Exception:
            pass

    # === Phase 7: Pair polish ===
    pair_end = min(t0 + budget * 0.93, hard_deadline - 30)
    if time.monotonic() < pair_end - 5:
        try:
            res, sc = _pair_polish(best_seq, pair_end, max_rounds=60, seed=seed + 500)
            _update_best(res, sc)
        except Exception:
            pass

    # === Phase 8: Final LP + coord ===
    final_lp_end = min(t0 + budget * 0.96, hard_deadline - 20)
    if time.monotonic() < final_lp_end - 5 and best_score > 1.5028:
        try:
            res, sc = _lp_optimize(best_seq, n_p, final_lp_end, seed + 600, max_iters=50)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res = res / res.sum()
                _update_best(res, sc)
        except Exception:
            pass

    # Final coordinate polish
    if time.monotonic() < hard_deadline - 10:
        try:
            res, sc = _coord_descent(best_seq, hard_deadline - 8, max_sweeps=50, seed=88888)
            _update_best(res, sc)
        except Exception:
            pass

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
