        s = float(a.sum())
        if s < 0.01:
            return math.inf
        return float(2.0 * n * c.max() / (s * s))

    # ===== Phase 1: LSE + L-BFGS (brief) =====
    def lse_func(u, T):
        um = u - u.max()
        e = np.exp(um)
        p_loc = e / e.sum()
        F = rfft(p_loc, conv_size)
        c = irfft(F * F, conv_size)[:2 * n - 1]
        np.maximum(c, 0.0, out=c)
        cmax = float(c.max())
        T_safe = max(T, 1e-30)
        scaled = np.minimum((c - cmax) / T_safe, 50.0)
        exps = np.exp(scaled)
        s_exp = float(exps.sum())
        if not math.isfinite(s_exp) or s_exp <= 0:
            return cmax, np.zeros(n)
        f = cmax + T_safe * math.log(s_exp / c.size)
        w = exps / s_exp
        Wf = rfft(w, cross_size)
        Pf = rfft(p_loc, cross_size)
        conv_wp = irfft(Wf * Pf, cross_size)
        cross_wp = irfft(np.conj(Wf) * Pf, cross_size)
        g = (conv_wp + cross_wp)[:n]
        tdp = float(np.dot(p_loc, g))
        grad_u = p_loc * (g - tdp)
        grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
        return f, grad_u

    u = np.log(np.maximum(p, 1e-30))
    u -= u.mean()
    c_init = conv_self(p)
    M_init = float(c_init.max())

    lse_end = min(work_deadline - 60.0, t0 + 50.0)
    if lse_end > t0 + 5.0:
        temps = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002]
        for tf in temps:
            if time.monotonic() > lse_end - 3.0:
                break
            p_cur = np.exp(u - u.max())
            p_cur /= p_cur.sum()
            cm = float(conv_self(p_cur).max())
            T = max(1e-30, tf * cm)
            try:
                res = minimize(lse_func, u, args=(T,), method='L-BFGS-B',
                              jac=True, options={'maxiter': 25, 'gtol': 1e-12,
                              'ftol': 1e-16, 'maxls': 30, 'maxcor': 20})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    u_new = res.x - res.x.mean()
                    p_new = np.exp(u_new - u_new.max())
                    p_new /= p_new.sum()
                    sc_new = score_fft(p_new)
                    if math.isfinite(sc_new):
                        consider(p_new)
                        u = u_new
                        p = p_new.copy()
            except Exception:
                pass

    # Use best if better
    if best_a.size == n:
        pb = best_a / best_a.sum()
        if score_fft(pb) < score_fft(p) - 1e-12:
            p = pb.copy()
    consider(p)

    # ===== Phase 2: Coordinate Descent =====
    cd_end = work_deadline - 20.0
    G = 48
    check_interval = 20  # call consider every N sweeps
    sweep = 0
    flat = 0
    rng = np.random.default_rng(seed)

    # Recompute c from current p
    p = p / p.sum()
    c = conv_self(p)
    S_tot = 1.0

    while time.monotonic() < cd_end - 1.0:
        improved = False
        order = rng.permutation(n)
        for ii in range(n):
            i = int(order[ii])
            if time.monotonic() > cd_end - 0.3:
                break

            if S_tot > 1.25 or S_tot < 0.75:
                inv = 1.0 / S_tot
                p *= inv
                c *= inv * inv
                S_tot = 1.0

            S = float(S_tot)
            v = float(p[i])

            k_lo = i
            k_hi = i + n - 1  # always <= 2n-2 since i <= n-1

            base = c[k_lo:k_hi + 1]  # length n

            max_const = 0.0
            if i > 0:
                v1 = float(c[:i].max())
                if v1 > max_const:
                    max_const = v1
            if k_hi + 1 < 2 * n - 1:
                v2 = float(c[k_hi + 1:].max())
                if v2 > max_const:
                    max_const = v2

            x_hi = max(3.0 * v, 3.0 * S / n, 0.01 * S, 1e-14)
            x_hi = min(x_hi, 2.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = S / n

            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 < v < x_hi:
                pts = np.unique(np.append(pts, v))

            dw = pts - v
            denom = S + dw
            # aff[j, m] = base[m] + dw[j] * 2*p[m]
            aff = base[np.newaxis, :] + dw[:, np.newaxis] * (2.0 * p)[np.newaxis, :]
            # Add dw^2 at position i
            aff[np.arange(pts.size), i] += dw * dw
            maff = np.maximum(np.max(aff, axis=1), max_const)
            valid = denom > 1e-15
            vals = np.full(pts.shape, math.inf)
            if np.any(valid):
                vals[valid] = 2.0 * n * maff[valid] / (denom[valid] ** 2)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            bx = float(pts[imin])
            bv = float(vals[imin])

            # Fine search
            dw0 = x_hi / G
            lo = max(0.0, bx - 2.0 * dw0)
            hi = min(x_hi, bx + 2.0 * dw0)
            if hi - lo < 1e-15:
                hi = lo + 1e-15
