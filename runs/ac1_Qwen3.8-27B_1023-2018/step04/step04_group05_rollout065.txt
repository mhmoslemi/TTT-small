def _fft_conv(a):
    n = len(a)
    if n <= 160:
        return np.convolve(a, a, 'full')
    m = next_fast_len(2 * n - 1)
    F = rfft(a, m)
    c = irfft(F * F, m)[:2 * n - 1]
    return np.maximum(c, 0.0)

    n = len(a)

        return math.inf
    b = _fft_conv(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)


def _toeplitz_A(f, n):

def _gradient_refine(f0, n, stop_t, seed):
    """Simple gradient-based refinement using subgradients of the max."""
    rng = np.random.default_rng(seed)

    while f.size < n:

    step_size = 0.02

    while time.monotonic() < stop_t - 5.0:
        b = _fft_conv(f)
        M = float(b.max())
        s = float(f.sum())
        # Find active lags (within 5% of max)
        thresh = M * 0.95
        active = np.where(b >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            for i in range(n):
                gi = 0.0
                if 0 <= k - i < n:
                    gi += f[k - i]
                if 0 <= i - k < n:
                    gi += f[i - k]
                grad[i] += gi
        grad /= len(active)

        # score = 2n * M / s^2
        # d(score)/d(a_i) = 2n * (grad_i * s - 2*M) / s^3
        sg = 2.0 * n * (grad * s - 2.0 * M) / (s ** 3)

        # Normalize
        sg_norm = float(np.max(np.abs(sg)))
        if sg_norm < 1e-18:
            break
        sg = sg / sg_norm

        # Gradient step with projection
        f_new = f - step_size * sg
        f_new = np.clip(f_new, 0, None)
        fns = float(f_new.sum())
        if fns < 0.01:
            f_new = f.copy()
            break
        f_new = f_new / fns * tgt

        sc = _score_fast(f_new)
        if sc < best_sc - 1e-15:
            f = f_new
            best = f.copy()
            best_sc = sc
            step_size = min(0.08, step_size * 1.1)
        else:
            step_size *= 0.7
            if step_size < 1e-5:
                break

    return best, best_sc
