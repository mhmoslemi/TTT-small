def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _score_norm(f):
    """Score assuming sum(f)=1. Returns 2*n*max(f*f)."""
    n = len(f)
    b = _fft_conv(f)
    m = float(np.max(b))
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m


def _build_A_sym(f, n):
    """Build n x n lower-triangular matrix: A[k,i] = f[k-i] for i<=k < n."""
    A = np.zeros((n, n), dtype=np.float64)
    for k in range(n):
        A[k, :k + 1] = f[:k + 1][::-1]
    return A


def _lp_solve(f, n, M, deadline):

        return None

        return None

    return None
