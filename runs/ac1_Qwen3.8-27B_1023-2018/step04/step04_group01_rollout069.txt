```python
from scipy.fft import rfft, irfft, next_fast_len

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(400, dtype=np.float64)
    if p.size == 0:
        p = np.ones(400, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(400, dtype=np.float64)
    return p

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)
```

```python
def _smoothed_grad(a, tau):
    n = len(a)
    N = next_fast_len(2 * n - 1)
    b = _conv_full(a)
    M = b.max()
    
    # Log-sum-exp weights
    b_shift = b - M  # <= 0
    p = np.exp(tau * b_shift)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.zeros(2*n-1)
        p[np.argmax(b)] = 1.0
        p_sum = 1.0
    p /= p_sum
    
    # Gradient via FFT
    # c_i = sum_m a[m] * p[m+i] = cross-correlation
    # d_i = sum_m a[m] * p[i-m] = convolution
    Fa = rfft(a, N)
    Fp = rfft(p, N)
    
    # c = ifft(conj(FFT(a)) * FFT(p))[:n]
    c = irfft(np.conj(Fa) * Fp, N)[:n]
    # d = ifft(FFT(p) * FFT(a))[:n]
    d = irfft(Fp * Fa, N)[:n]
    
    grad = c + d
    return grad, M, p
```

```python
    n = len(a)
    N = next_fast_len(2 * n - 1)
    F = rfft(a, N)
    c = irfft(F * F, N)[:2 * n - 1]

    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _smoothed_scores(a, tau):
    """Compute smoothed max and gradient using log-sum-exp."""
    n = len(a)
    N = next_fast_len(2 * n - 1)
    b = _conv_full(a)
    M = float(b.max())
    
    # Softmax weights
    b_shift = b - M
    p = np.exp(tau * b_shift)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.zeros(len(b))
        p[np.argmax(b)] = 1.0
        p_sum = 1.0
    p = p / p_sum
    
    # Gradient: grad[i] = 2 * sum_m a[m] * p[m+i]
    # = 2 * cross_correlation(a, p)[i]
    Fa = rfft(a, N)
    Fp = rfft(p, N)
    cross = irfft(np.conj(Fa) * Fp, N)[:n]
    grad = 2.0 * cross
    
    return grad, M, p, b

        if stall >= max_stall:

        S = a.sum()

            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # d_score/da_i = (2n/S^2) * (grad_i - 2M/S)
        sg = (2.0 * n / (S * S)) * (grad - 2.0 * M / S)

        S_new = a_new.sum()

            a_new = a.copy()
            lr *= 0.5
            continue

        if sc_new < score - 1e-15:
            a = a_new
            lr = min(lr * 1.05, 1.0)

            tau *= 1.002

                a /= a.sum()
                lr = 0.05
                tau = max(50.0 / a.sum(), tau * 0.9)
        
        it += 1
    
    return best_a, best_score

def _lp_direction(f, n, time_limit=5.0):
    """Find LP direction: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    b = _conv_full(f)
    M = float(b.max())
    if M < 1e-15:
        return None, M, b
    
    # Build constraint matrix A (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None, M, b
    
    if res.status != 0 or res.x is None:
        return None, M, b
    
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None, M, b
    
    return g / g_sum, M, b
```

```python
            continue
```

            a = a_new

    best_a = a.copy()
    best_score = _score(a)
    tau = 200.0 / S0
    lr = 0.05 * S0 / n
    stall = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() < stop_time - 3 and it < 30000:
        if stall > 80:
            break
        grad, M, p, b = _smoothed_grad(a, tau)
        S = float(a.sum())
        score = 2.0 * n * M / (S * S)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        sg = (2.0 * n / S**2) * (grad - 2.0 * M / S)
        a_new = np.clip(a - lr * sg, 0, None)
        S_new = a_new.sum()
        if S_new < 0.01:
            lr *= 0.5
            stall += 1
            it += 1
            continue

        sc_new = _score(a_new)
        if sc_new < score - 1e-14:
            a = a_new
            lr = min(lr * 1.03, 2.0 * S0 / n)
            tau *= 1.001
        else:
            lr *= 0.5
            if lr < 1e-12:
                a = np.clip(a + rng.normal(0, 0.01 * S0 / n, n), 0, None)
                lr = 0.02 * S0 / n
                tau = max(50.0 / a.sum(), tau * 0.95)
        it += 1
    return best_a, best_score

    n = len(parent)

    a0 = np.clip(parent.copy(), 0, None).astype(np.float64)
    s0 = a0.sum()
    if s0 < 0.01:
        a0 = np.ones(400) / 400.0

    # Initial score
    best_a = a0.copy()
    best_score = _score(a0)

    def _update(seq, sc=None):
        nonlocal best_a, best_score
        seq = np.clip(np.asarray(seq, dtype=np.float64), 0, 1000)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        if sc is None:
            sc = _score(seq)
        if not math.isfinite(sc):
            return
        try:
            ex = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            ex = math.inf
        if math.isfinite(ex) and ex < best_score - 1e-15:
            best_a = seq.copy()
            best_score = ex

    # Phase 1: Gradient descent (~40% of budget)
    p1_end = t0 + budget * 0.35
    if time.monotonic() < p1_end - 5:
        try:
            res, sc = _grad_descent(a0, p1_end, seed=seed)
            _update(res, sc)
        except Exception:
            pass

    # Phase 2: LP-based optimization (~30% of budget)
    p2_end = t0 + budget * 0.60
    if time.monotonic() < p2_end - 5:
        try:
            res, sc = _lp_optimize(best_a, p2_end, seed=seed + 1, max_iters=150)
            _update(res, sc)
        except Exception:
            pass

    p3_end = t0 + budget * 0.80

            _update(res, sc)
        except Exception:
            pass

    # Phase 4: Round 2 - gradient + LP + CD
    p4_end = min(deadline - 30, t0 + budget * 0.92)
    if time.monotonic() < p4_end - 20:
        t_mid = time.monotonic() + (p4_end - time.monotonic()) * 0.3
        if time.monotonic() < t_mid - 5:

                _update(res, sc)
            except Exception:

                _update(res, sc)
            except Exception:
                pass

            except Exception:
