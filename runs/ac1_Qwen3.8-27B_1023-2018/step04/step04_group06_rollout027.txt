def _build_A(n, a):
    """Convolution matrix A where (A @ g)[k] = (a * g)[k]"""
    # Using the fact that A is a Toeplitz-like matrix
    # A[k, j] = a[k-j] if 0 <= k-j < n else 0
    # This is a banded matrix with bandwidth n
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[k, j] = a[k - j]
    return A

def _build_A_fast(n, a):
    """Faster construction using scipy's toeplitz-like structure."""
    # The matrix A has the structure:
    # Row k has a[k], a[k-1], ..., a[0], 0, ... in the first k+1 positions
    #         and 0, ..., 0, a[n-1-k], ..., a[0] in the last n-1-k positions
    # Actually it's: A[k,j] = a[k-j] if 0 <= k-j < n
    # So column j has: 0 (first j entries), a[k-j] for k=j to j+n-1, 0 (rest)
    # This is a convolution matrix (Toeplitz with some zeros)
    
    # Use the fact that this is the convolution matrix
    # We can build it using scipy.signal or manually
    # For n up to 600, the O(n^2) construction is fine (600^2 = 360000)
    k = np.arange(2*n-1)[:, None]

    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2*n-1, n), dtype=np.float64)

    return A

def _lp_best_response(A, M, n, time_limit=5.0):
    """max 1^T g s.t. A @ g <= M, g >= 0"""
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    try:

def _golden_section_line_search(a, g, M_cur, n_evals=30):
    """Golden section search for best t in [0,1] minimizing max conv of (1-t)*a + t*g."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~ 0.618
    
    def f(t):
        cand = (1.0 - t) * a + t * g
        cs = cand.sum()
        if cs < 1e-15:
            return float('inf')
        cand = cand / cs
        return float(np.convolve(cand, cand).max())
    
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    
    for _ in range(n_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    
    t_best = (lo + hi) / 2
    val_best = f(t_best)
    return t_best, val_best

def _active_gradient_step(a, k_star, n, step_sizes):
    """Try gradient descent steps in the direction that reduces the active constraint."""
    # Gradient of (a*a)_{k_star} w.r.t. a:
    # grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n else 0
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    for i in range(lo, hi):
        idx = k_star - i
        if 0 <= idx < n:
            grad[i] = 2.0 * a[idx]

    # f = M / (sum_a)^2 where M = (a*a)_{k_star}
    # To decrease f, we can either decrease M or increase sum_a (or both)
    # Since we normalize, we should decrease M.
    # Gradient descent: a_new = a - alpha * grad / ||grad||
    # But we need to maintain non-negativity and normalization.

        return None
    
    for alpha in step_sizes:
        cand = a - alpha * grad / gn
        cand = np.maximum(cand, 0.0)
        cs = cand.sum()
        if cs < 1e-12:
            continue

        new_M = float(np.convolve(cand, cand).max())
        if new_M < float(np.convolve(a, a).max()) * (1 - 1e-10):
            return cand
    return None

def _smoothed_lbfgs(a0, beta, max_iter=200, time_limit=30.0):
    """Optimize using L-BFGS-B on the smoothed objective."""
    n = len(a0)
    t_start = time.monotonic()
    
    def obj_grad(x):
        if time.monotonic() - t_start > time_limit:
            return float('inf'), np.zeros_like(x)

        s = x.sum()
        if s < 1e-15:
            return float('inf'), np.zeros_like(x)

        bc = beta * c
        max_bc = bc.max()
        exp_bc = np.exp(bc - max_bc)
        sum_exp = exp_bc.sum()
        w = exp_bc / sum_exp
        lse = max_bc + np.log(sum_exp)
        obj = 2.0 * n * np.exp(lse) / (s * s)
        
        r = np.correlate(w, x, mode='valid')
        grad = (2.0 * n * np.exp(lse) / (s * s)) * (2.0 * beta * r - 2.0 / s)
        return obj, grad

                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})
    x_opt = np.maximum(res.x, 0.0)
    if x_opt.sum() > 1e-15:
        x_opt /= x_opt.sum()
    return x_opt

def _build_A_fast(n, a):
    """Convolution matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[d[mask]]
    return A


def _lp_solve(A, M, n, tl=4.0):
    """max 1^T g s.t. A @ g <= M, g >= 0."""
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tl,
                               'dual_feasibility_tolerance': 1e-14,
                               'primal_feasibility_tolerance': 1e-14})
    except Exception:

    if res.status != 0 or res.x is None:
        return None
    return np.clip(res.x, 0.0, None)


def _golden_section(a, g, n_evals=28):
    """Golden section search for best t in [0,1]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    def f(t):

        cs = cand.sum()
        if cs < 1e-15:

        cand = cand / cs
        return float(np.convolve(cand, cand).max())

    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc, fd = f(c), f(d)

            c = hi - gr * (hi - lo)

        else:

            d = lo + gr * (hi - lo)
            fd = f(d)

    bc = beta * c
    max_bc = bc.max()
    if max_bc > 700.0:
        # Handle overflow: shift more
        exp_bc = np.exp(bc - max_bc)
    else:
        exp_bc = np.exp(bc - max_bc)
    sum_exp = exp_bc.sum()
    if sum_exp < 1e-300:
        return float('inf'), np.zeros(n)
    w = exp_bc / sum_exp
    lse = max_bc + np.log(sum_exp)
    exp_lse = np.exp(lse)
    if not np.isfinite(exp_lse):
        return float('inf'), np.zeros(n)
    obj = 2.0 * n * exp_lse / (s * s)
    if not np.isfinite(obj):
        return float('inf'), np.zeros(n)
    r = np.correlate(w, x, mode='valid')
    grad = (2.0 * n * exp_lse / (s * s)) * (2.0 * beta * r - 2.0 / s)
    grad = np.clip(grad, -1e10, 1e10)
    return obj, grad


def _smoothed_lbfgs(a0, beta, t_limit, max_iter=300):
    """L-BFGS-B on smoothed objective with timeout callback."""
    n = len(a0)
    t_start = time.monotonic()
    state = {'count': 0}

    def objective(x):
        state['count'] += 1
        return _smoothed_obj_grad(x, beta, t_limit, t_start)[0]

    def callback(xk):
        return time.monotonic() - t_start > t_limit

    def full_obj_grad(x):
        state['count'] += 1
        return _smoothed_obj_grad(x, beta, t_limit, t_start)

    try:
        res = minimize(full_obj_grad, a0.copy(), method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': max_iter * 2,
                                'callback': callback})
        x_opt = np.maximum(res.x, 0.0)
        if x_opt.sum() > 1e-15:
            x_opt /= x_opt.sum()
        return x_opt
    except Exception:
        return a0.copy()


def _optimize_lp(a0, stop_time, seed=42, max_stuck=100):
    """LP-based best-response with golden section line search."""
    a0 = np.asarray(a0, dtype=np.float64)
    n = a0.size
    a = np.clip(a0, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a /= a.sum()

    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0
    iter_count = 0

    while time.monotonic() < stop_time:
        iter_count += 1
        c_full = np.convolve(a, a)
        M = float(c_full.max())
        k_star = int(c_full.argmax())

        if M < best_M * (1.0 - 5e-14):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > max_stuck:
            sigma = min(0.015 + 0.008 * (stuck // max_stuck), 0.35)
            xx = np.arange(n, dtype=np.float64)
            mode = stuck % 6
            if mode == 0:
                a *= np.exp(rng.normal(0, sigma, n))
            elif mode == 1:
                freq = rng.uniform(0.5, 5.0)
                phase = rng.uniform(0, 2 * np.pi)
                a *= (1.0 + sigma * np.sin(2 * np.pi * freq * xx / n + phase))
            elif mode == 2:
                blk = max(2, n // rng.integers(3, 8))
                start = rng.integers(0, max(1, n - blk))
                a[start:start + blk] *= (1.0 + sigma * rng.normal(0, 1, blk))
            elif mode == 3:
                # Two-frequency perturbation
                f1, f2 = rng.uniform(0.5, 4.0), rng.uniform(2.0, 6.0)
                a *= (1.0 + sigma * (0.5 * np.sin(2 * np.pi * f1 * xx / n) +
                                      0.5 * np.cos(2 * np.pi * f2 * xx / n)))
            elif mode == 4:
                # Mirror blend
                a_mix = 0.5 * a + 0.5 * a[::-1]
                a = a_mix * np.exp(rng.normal(0, sigma * 0.5, n))
            else:
                # Localized bump
                center = rng.integers(0, n)
                width = max(2, n // rng.integers(8, 20))
                bump = np.exp(-0.5 * ((xx - center) / width) ** 2)
                a *= (1.0 + sigma * (bump - bump.mean()))
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            stuck = 0
            continue

        made_move = False

        # LP best response with multiple M levels
        A = _build_A_fast(n, a)
        for M_factor in [1.0 + 1e-10, 1.0 + 1e-7, 1.0 + 1e-5]:
            if made_move:
                break
            if time.monotonic() > stop_time - 2:
                break
            M_lp = M * M_factor
            g = _lp_solve(A, M_lp, n, tl=3.0)
            if g is None:
                continue
            gs = g.sum()
            if gs < 1e-15:
                continue
            g = g / gs
            if float(np.max(np.abs(g - a))) < 1e-11:
                continue
            t_best, val_best = _golden_section(a, g, n_evals=22)
            if val_best < M * (1.0 - 1e-12):
                a = (1.0 - t_best) * a + t_best * g
                a = np.maximum(a, 0.0)
                a /= a.sum()
                made_move = True

        if not made_move:
            # Gradient step on active constraint
            grad = np.zeros(n)
            lo = max(0, k_star - n + 1)
            hi = min(n, k_star + 1)
            for i in range(lo, hi):
                idx = k_star - i
                if 0 <= idx < n:
                    grad[i] = 2.0 * a[idx]
            # Subtract mean for normalization invariance
            grad -= grad.mean()
            gn = float(np.linalg.norm(grad))
            if gn > 1e-14:
                for alpha in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002]:
                    cand = a - alpha * grad / gn
                    cand = np.maximum(cand, 0.0)
                    cs = cand.sum()
                    if cs < 1e-12:
                        continue
                    cand /= cs
                    if float(np.convolve(cand, cand).max()) < M * (1.0 - 1e-12):
                        a = cand
                        made_move = True
                        break

        if not made_move:
            a *= np.exp(rng.normal(0, 0.012, n))
            a = np.maximum(a, 1e-12)
            a /= a.sum()

    return best_a


def _optimize_smoothed(a0, stop_time, seed=42):
    """Optimize using smoothed L-BFGS-B with increasing beta schedule."""
    a0 = np.asarray(a0, dtype=np.float64)
    n = a0.size
    a = np.clip(a0, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a /= a.sum()

    best_a = a.copy()
    best_score = _eval_score(a)
    rng = np.random.default_rng(seed)

    # Schedule of beta values (smoothing parameter)
    beta_schedule = [5, 10, 20, 40, 80, 160, 320, 640, 1280, 2560]

    for beta in beta_schedule:
        if time.monotonic() > stop_time - 10:
            break
        # Multiple rounds at this beta
        for rd in range(4):
            if time.monotonic() > stop_time - 5:
                break
            alloc = min(20.0, stop_time - time.monotonic() - 3)
            if alloc < 5:
                break
            t_limit = alloc
            try:
                a_new = _smoothed_lbfgs(a, beta, t_limit, max_iter=200)
                sc = _eval_score(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new
            except Exception:
                break

        # Perturbation between beta levels
        if time.monotonic() > stop_time - 15:
            break
        a_pert = a * np.exp(rng.normal(0, 0.04, n))
        a_pert = np.maximum(a_pert, 1e-12)
        a_pert /= a_pert.sum()
        try:
            alloc = min(10.0, stop_time - time.monotonic() - 3)
            if alloc > 3:
                a_new = _smoothed_lbfgs(a_pert, beta, alloc, max_iter=100)
                sc = _eval_score(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new
        except Exception:
            pass

    return best_a


def _quick_optimize(a0, t_alloc, seed=0):
    """Quick combined optimization for length exploration."""
    best = a0.copy()
    best_sc = _eval_score(a0)

    if t_alloc > 15:
        try:
            r = _optimize_lp(a0, time.monotonic() + t_alloc * 0.6, seed=seed, max_stuck=40)
            sc = _eval_score(r)
            if np.isfinite(sc) and sc < best_sc:
                best, best_sc = r, sc
        except Exception:
            pass

    if t_alloc > 25:
        try:
            r = _optimize_smoothed(best, time.monotonic() + t_alloc * 0.3, seed=seed + 1)
            sc = _eval_score(r)
            if np.isfinite(sc) and sc < best_sc:
                best, best_sc = r, sc
        except Exception:
            pass

    return best


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 22.0

    parent = _get_parent()
    n_parent = parent.size

    best_seq = np.clip(parent.copy(), 0, 1000).astype(np.float64)
    best_score = _eval_score(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(32, dtype=np.float64)
        best_score = _eval_score(best_seq)

    def try_update(cand):
        nonlocal best_seq, best_score
        try:
            cand = np.asarray(cand, dtype=np.float64)
            cand = np.clip(cand, 0, 1000)
            sc = _eval_score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand.copy()
        except Exception:
            pass

    # ===== Phase 1: LP optimization on parent (250s) =====
    t1 = t0 + min(250.0, budget * 0.22)
    if time.monotonic() < t1 - 5:
        try:
            r = _optimize_lp(parent, t1, seed=1, max_stuck=80)
            try_update(r)
        except Exception:
            pass

    # ===== Phase 2: Smoothed optimization (250s) =====
    t2 = t0 + min(500.0, budget * 0.42)
    if time.monotonic() < t2 - 10:
        try:
            r = _optimize_smoothed(best_seq, t2, seed=2)
            try_update(r)
        except Exception:
            pass

    # ===== Phase 3: LP optimization on best (250s) =====
    t3 = t0 + min(750.0, budget * 0.62)
    if time.monotonic() < t3 - 10:
        try:
            r = _optimize_lp(best_seq, t3, seed=3, max_stuck=150)
            try_update(r)
        except Exception:
            pass

    # ===== Phase 4: Length exploration (200s) =====
    t4 = t0 + min(950.0, budget * 0.80)
    if time.monotonic() < t4 - 30:
        dn_list = [-30, -20, -15, -10, -7, -5, -3, -2, -1, 1, 2, 3, 5, 7, 10, 15, 20, 30]
        for dn in dn_list:
            if time.monotonic() > t4 - 20:
                break
            nt = n_parent + dn
            if nt < 40 or nt > 700:
                continue
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, best_seq)
            init_new = np.clip(init_new, 0, 1000)
            sn = init_new.sum()
            if sn < 1e-10:
                continue
            init_new /= sn
            alloc = min(25.0, t4 - time.monotonic() - 5)
            if alloc < 8:
                continue
            try:
                r = _quick_optimize(init_new, alloc, seed=2000 + nt)
                try_update(r)
            except Exception:
                pass

    # ===== Phase 5: Final smoothed polish (150s) =====
    if time.monotonic() < deadline - 60:
        t5 = min(deadline - 60, t0 + budget * 0.92)
        try:
            r = _optimize_smoothed(best_seq, t5, seed=777)
            try_update(r)
        except Exception:
            pass

    # ===== Phase 6: Final LP polish =====
    if time.monotonic() < deadline - 10:
        try:
            r = _optimize_lp(best_seq, deadline - 5, seed=9999, max_stuck=200)
            try_update(r)
        except Exception:
            pass

    # ===== Final validation =====
    try:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if not np.isfinite(fs):
            raise ValueError("invalid final")
    except Exception:
        return [1.0] * 32

    return [float(x) for x in best_seq]
