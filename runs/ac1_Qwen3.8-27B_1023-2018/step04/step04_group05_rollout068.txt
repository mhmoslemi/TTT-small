    cd_inits.append(np.clip(best_seq + rng_main.normal(0, 0.05, n), 0, None).copy())

    n_cd = len(cd_inits)
    time_per_cd = max(30.0, (phase2_end - time.monotonic()) / max(1, n_cd))

    for idx, init in enumerate(cd_inits):
        if time.monotonic() > phase2_end - 10.0:
            break
        stop = min(phase2_end - 5.0, time.monotonic() + time_per_cd)
        try:
            res, sc = _cd_optimize(init, stop, seed + 99999 * (idx + 1))
            if res is not None and len(res) > 0 and np.all(np.isfinite(res)):
                _update_best(res, sc)
        except Exception:
            pass

    # Phase 3: 2D polish (~15% of time)
    phase3_end = t0 + budget * 0.85
    if time.monotonic() < phase3_end - 10.0:
        try:
            res, sc = _polish_2d(best_seq, phase3_end - 5.0, seed + 31337)
            if res is not None:
                _update_best(res, sc)
        except Exception:
            pass

    # Phase 4: Final CD polish from best
    if time.monotonic() < deadline - 10.0:
        try:
            res, sc = _cd_optimize(best_seq.copy(), deadline - 5.0, seed + 777777)
            if res is not None and len(res) > 0 and np.all(np.isfinite(res)):
                _update_best(res, sc)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf
    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
