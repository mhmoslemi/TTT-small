def _lp_mindir(f, n, stop_time):
    """
    Find g that minimizes max_k (f*g)_k subject to sum(g)=1, g>=0.
    This is: minimize M' s.t. (f*g)_k <= M', sum(g)=1, g>=0
    """
    A_conv = _build_A(f, n)
    # Variables: g_0, ..., g_{n-1}, M'
    # Objective: minimize M' (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: (f*g)_k - M' <= 0 for k=0,...,2n-2
    # [A_conv, -ones(2n-1)] * [g; M'] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_conv
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(None, None)]  # g_j >= 0, M' free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 10.0,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
    except Exception:
        return None, None
    
    if res.status != 0 or res.x is None:
        return None, None
    
    g = np.clip(res.x[:n], 0, None)
    M_prime = res.x[n]
    
    if g.sum() < 1e-15:
        return None, None
    
    g = g / g.sum()  # Ensure sum = 1
    return g, M_prime

def _lp_maxdir(f, n, M, timeout=5.0):
    """Original LP: maximize sum(g) subject to (f*g)_k <= M, g >= 0."""
    A_conv = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g

import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _conv(a):
    n = len(a)
    if n <= 2048:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_normalized(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m


def _project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def _subgradient_step(a, b, active_k, step):
    """Compute subgradient and take projected step."""
    n = len(a)
    # Subgradient: average of gradients at active lags
    v = np.zeros(n)
    for k in active_k:
        # grad b[k] at a: v[i] = 2*a[k-i] if 0 <= k-i < n
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        v[mask] += 2.0 * a[idx[mask]]
    v /= len(active_k)
    
    # Projected step
    a_new = _project_simplex(a - step * v)
    return a_new

import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _conv(a):
    n = len(a)
    if n <= 2048:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_norm(a):
    """Score for a sequence with sum=1: 2*n*max(conv(a,a))."""
    n = len(a)
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m


def _project_simplex(v):
    """Euclidean projection onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        rho = np.array([0])
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def _build_conv_matrix(f, n):
    """Matrix A such that A @ g = f * g (full convolution)."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_mindir(f, n, timeout=4.0):
    """LP: minimize M' s.t. (f*g)_k <= M', sum(g)=1, g>=0. Returns g or None."""
    A_conv = _build_conv_matrix(f, n)
    nv = n + 1
    c = np.zeros(nv)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, nv))
    A_ub[:, :n] = A_conv
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(-1e4, 1e4)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.clip(np.asarray(res.x[:n], dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g / g.sum()


def _lp_maxdir(f, n, M, timeout=3.0):
    """LP: maximize sum(g) s.t. (f*g)_k <= M, g>=0. Returns normalized g or None."""
    A_conv = _build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_conv, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g / g.sum()


def _subgrad_v(a, b, active_k):
    """Compute subgradient of max_k b[k] at a, averaged over active_k."""
    n = len(a)
    v = np.zeros(n)
    for k in active_k:
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        v[mask] += 2.0 * a[idx[mask]]
    v /= len(active_k)
    return v
