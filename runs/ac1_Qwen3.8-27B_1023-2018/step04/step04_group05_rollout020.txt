                stall = 0
                max_stall += 60
                t = 0.3

    return best_f, best_score


def _polish_fast(f0, stop_time, seed=4242):
    n = len(f0)
    best = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0.0, None)
    s = float(best.sum())
    if s < 1e-10:
        return best, _score(best)
    best = best / s
    s = 1.0

    c = _conv_full(best)
    best_sc = 2.0 * n * float(c.max())
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 25
    round_num = 0

    while time.monotonic() < stop_time - 5.0 and no_imp < max_no_imp:
        round_num += 1
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 5.0:
                break
            orig = best[i]
            best_i_val = orig
            best_i_sc = best_sc
            best_i_s = s
            saved_c = c[i:i + n].copy()

            multipliers = [0.0, 0.1, 0.2, 0.35, 0.5, 0.65, 0.8, 1.2, 1.5, 2.0, 3.0, 5.0]
            for mlt in multipliers:
                new_val = orig * mlt if mlt > 0 else 0.0
                delta = new_val - orig
                if abs(delta) < 1e-15:
                    continue
                new_s = s + delta
                if new_s < 0.01:
                    continue
                c[i:i + n] += 2.0 * delta * best
                new_sc = 2.0 * n * float(c.max()) / (new_s * new_s)
                if new_sc < best_i_sc - 1e-15:
                    best_i_sc = new_sc
                    best_i_val = new_val
                    best_i_s = new_s
                    c[i:i + n] = saved_c + 2.0 * delta * best
                elif new_sc >= best_i_sc:
                    c[i:i + n] = saved_c
                    c[i:i + n] += 2.0 * delta * best
                    c[i:i + n] = saved_c

            # Restore c after trying all multipliers
            c[i:i + n] = saved_c

            if best_i_val != orig and best_i_sc < best_sc - 1e-15:
                delta_final = best_i_val - orig
                best[i] = best_i_val
                c[i:i + n] += 2.0 * delta_final * best
                s = best_i_s
                best_sc = best_i_sc
                improved = True
                # Refine around best
                for frac in [0.9, 1.05, 1.1, 0.8, 1.2, 0.95, 1.02]:
                    trial = best_i_val * frac
                    delta_r = trial - best_i_val
                    if abs(delta_r) < 1e-15:
                        continue
                    trial_s = s + delta_r
                    if trial_s < 0.01:
                        continue
                    c[i:i + n] += 2.0 * delta_r * best
                    trial_sc = 2.0 * n * float(c.max()) / (trial_s * trial_s)
                    if trial_sc < best_sc - 1e-15:
                        best[i] = trial
                        s = trial_s
                        best_sc = trial_sc
                    else:
                        c[i:i + n] -= 2.0 * delta_r * best

        # Two-coordinate moves
        n_pairs = 200 if round_num <= 5 else 100
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 5.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            si, sj = oi, oj
            for mi in [0.0, 0.3, 0.5, 0.7, 1.3, 1.7, 2.5]:
                for mj in [0.0, 0.3, 0.5, 0.7, 1.3, 1.7, 2.5]:
                    if mi == 1.0 and mj == 1.0:
                        continue
                    ni = oi * mi if (mi > 0 and oi > 1e-14) else 0.0
                    nj = oj * mj if (mj > 0 and oj > 1e-14) else 0.0
                    best[i], best[j] = ni, nj
                    new_s = s - oi - oj + ni + nj
                    if new_s < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    # Recompute c from scratch for accuracy (only 2 coords changed)
                    c_new = c.copy()
                    di = ni - oi
                    dj = nj - oj
                    c_new[i:i + n] += 2.0 * di * best
                    c_new[j:j + n] += 2.0 * dj * best
                    new_sc = 2.0 * n * float(c_new.max()) / (new_s * new_s)
                    if new_sc < best_sc - 1e-14:
                        c = c_new
                        s = new_s
                        best_sc = new_sc
                        improved = True
                    else:
                        best[i], best[j] = oi, oj
            best[i], best[j] = si, sj

        # Random perturbations if stuck
        if not improved and round_num > 3:
            for _ in range(50):
                if time.monotonic() > stop_time - 5.0:
                    break
                k = int(rng.integers(n))
                old = best[k]
                old_c = c[k:k + n].copy()
                sigma = 0.04 * max(1.0, best.mean())
                best[k] = max(0.0, old + rng.normal(0, sigma))
                delta = best[k] - old
                new_s = s + delta
                if new_s < 0.01:
                    best[k] = old
                    c[k:k + n] = old_c
                    continue
                c[k:k + n] += 2.0 * delta * best
                new_sc = 2.0 * n * float(c.max()) / (new_s * new_s)
                if new_sc < best_sc - 1e-14:
                    s = new_s
                    best_sc = new_sc
                    improved = True
                else:
                    best[k] = old
                    c[k:k + n] = old_c

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    best_sc_exact = _score(best)
    return best, best_sc_exact


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    n_p = len(parent)

    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(100)
        fallback_sc = _score(fallback)
    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_opt(n, init_arr, stop, sd):
        nonlocal best_seq, best_score
        if time.monotonic() > stop - 3.0:
            return
        try:
            res, sc = _lp_optimize(init_arr, n, stop, sd)
        except Exception:
            return
        if res is None or len(res) == 0:
            return
        res = np.clip(np.asarray(res, dtype=np.float64), 0.0, 1000.0)
        if res.sum() < 0.01 or not np.all(np.isfinite(res)):
            return
        if sc < best_score - 1e-13:
            try:
                exact = float(evaluate_sequence([float(x) for x in res]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-14:
                best_seq = res.copy()
                best_score = exact

    # Phase 1: LP on parent length with diverse inits (55% time)
    n = n_p
    phase1_end = t0 + budget * 0.55
    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    inits = [parent.copy()]
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n)))
    sup = np.ones(n) * 0.2
    for fr in (1, 2, 3, 5, 7, 11, 13):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.12 * np.cos(2.0 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.01)
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.02)
    sig = 0.2 * max(1.0, float(np.mean(parent)))
    inits.append(np.abs(parent + rng_main.normal(0, sig, n)) + 0.01)
    center = (n - 1) / 2.0
    inits.append(np.exp(-0.5 * ((i_arr - center) / (n / 5.0)) ** 2))
    inits.append(np.exp(-0.5 * ((i_arr - n * 0.3) / (n * 0.12)) ** 2) +
                 np.exp(-0.5 * ((i_arr - n * 0.7) / (n * 0.12)) ** 2))
    inits.append(np.ones(n))
    inits.append(1.0 + 0.7 * np.cos(math.pi * i_arr / max(1, n - 1)))
    x = np.clip(np.abs(i_arr - (n - 1) / 2.0) / (n / 4.0), 0, 1)
    inits.append((1 - x) ** 3 + 0.01)

    n_inits = len(inits)
    time_per_init = max(8.0, (phase1_end - t0) / n_inits)

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 8.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        _try_opt(n, init, stop, seed + 7777 * (idx + 1))

    # Phase 2: Alternative lengths (15% time)
    phase2_end = t0 + budget * 0.72
    other_lengths = [400, 384, 448, 320, 480, 288, 352, 512]
    for L in other_lengths:
        if time.monotonic() > phase2_end - 8.0:
            break
        if L == n:
            continue
        stop = min(phase2_end - 5.0, time.monotonic() + max(6.0, (phase2_end - phase1_end) / 10.0))
        init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
        _try_opt(L, init_l, stop, seed + 50000 + L)

    # Phase 3: Second LP pass on best (warm-start, 12% time)
    phase3_end = t0 + budget * 0.84
    if time.monotonic() < phase3_end - 20.0:
        stop_p3 = min(phase3_end - 5.0, time.monotonic() + (phase3_end - time.monotonic()))
        _try_opt(len(best_seq), best_seq.copy(), stop_p3, seed + 999999)

    # Phase 4: Polish (18% time)
    polish_end = deadline - 8.0
    if time.monotonic() < polish_end - 10.0:
        try:
            b_seq, b_sc = _polish_fast(best_seq, polish_end, seed=4242)
            if b_sc < best_score - 1e-14:
                best_seq = b_seq
                best_score = b_sc
        except Exception:
            pass

    # Phase 5: Second polish with different seed
    if time.monotonic() < deadline - 12.0:
        try:
            b_seq2, b_sc2 = _polish_fast(best_seq, deadline - 8.0, seed=88888)
            if b_sc2 < best_score - 1e-14:
                best_seq = b_seq2
                best_score = b_sc2
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0] * 100
        return [float(x) for x in fallback]

    if final_sc < best_score - 1e-12:
        best_score = final_sc

    out = [float(x) for x in best_seq]
    # Final sanity: ensure valid
    if len(out) == 0 or any(not math.isfinite(x) or x < 0 for x in out):
        return [1.0] * 100
    return out
