    b = _conv_full(a)

    return 2.0 * n * m / (s * s)

def _lp_direction(f, b, M, frac=0.05, timeout=10.0):
    """
    Find LP direction g: maximize sum(g) subject to (f*g)[k] <= M for active k, g >= 0.
    """
    n = len(f)
    threshold = M * (1.0 - frac)
    active = np.where(b >= threshold)[0]
    
    # Build sparse matrix A where A[r, i] = f[k - i] for k = active[r]
    rows, cols, vals = [], [], []
    for r, k in enumerate(active):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-14:
                rows.append(r)
                cols.append(i)
                vals.append(v)
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c_vec = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout})

        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _line_search(f, g, M, max_iter=20):
    """Find best t in (0, 1] such that max_b of (1-t)f + t*g <= M."""
    n = len(f)
    lo, hi = 0.0, 1.0
    
    # First check if t=1 works
    f_new = g  # (1-1)*f + 1*g
    b_new = _conv_full(f_new)
    if b_new.max() <= M + 1e-10:
        return 1.0

    M_mid = M + 1e-10

        mid = (lo + hi) / 2
        f_mid = (1 - mid) * f + mid * g
        b_mid = _conv_full(f_mid)
        if b_mid.max() <= M_mid:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-6:
            break
    
    return lo

def _optimize_lp_v2(f0, stop_time, seed, peak_frac=0.05):
    """LP-based optimization with line search."""
    n = len(f0)

    f_sum = f.sum()
    if f_sum < 0.01:
        return f, math.inf
    f = f / f_sum  # normalize to sum = 1
    best_f = f.copy()
    best_score = _score(f)

    max_stall = 50
    frac = peak_frac
    
    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        
        g = _lp_direction(f, b, M, frac=frac, timeout=8.0)
        if g is None:
            stall += 1
            if stall > max_stall:
                # Perturb and restart
                rng = np.random.default_rng(hash((seed, stall)) & 0xFFFFFFFF)
                noise = rng.normal(0, 0.05, n)
                f = np.clip(f + noise, 0, None)
                fs = f.sum()
                if fs > 0.01:
                    f = f / fs
                stall = 0
                max_stall = min(max_stall + 30, 200)
                frac = min(frac * 2, 0.5)
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            stall += 1
            continue
        
        # Scale g to have same sum as f (so the update is a convex combination)
        g = g * (1.0 / g_sum)  # now sum(g) = 1, same as sum(f)
        
        # Line search
        t = _line_search(f, g, M, max_iter=15)
        if t < 1e-8:
            stall += 1
            continue
        
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        f_new_sum = f_new.sum()
        if f_new_sum < 0.01:
            stall += 1
            continue
        f_new = f_new / f_new_sum
        
        sc = _score(f_new)
        if sc < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = sc
            stall = 0
            frac = max(frac * 0.9, 0.01)  # tighten over time
        else:
            stall += 1
            frac = min(frac * 1.5, 0.5)  # loosen
        
        f = f_new
    
    return best_f, best_score

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _lp_minmax(f):
    """Minimize max_k (f*g)[k] s.t. sum(g)=1, g>=0."""
    n = len(f)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-14:
                rows.append(k)
                cols.append(i)
                vals.append(v)
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n+1))
    c_obj = np.zeros(n+1); c_obj[n] = 1.0
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1)); A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)]*n + [(-np.inf, np.inf)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x[:n], 0, None)
    gs = g.sum()
    if gs < 1e-10:
        return None
    return g / gs

def _line_search(f, g, M, max_iter=20):
    """Find largest t in [0,1] s.t. max_b((1-t)f+tg) <= M."""
    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        f_mid = (1 - mid) * f + mid * g
        if f_mid.sum() < 1e-10:
            hi = mid
            continue
        if _conv_full(f_mid).max() <= M + 1e-12:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-8:
            break
    return lo

    n = len(parent)

    best_score = _score(best_f)
    
    # Phase 1: LP optimization

    no_imp = 0
    max_no_imp = 100
    
    while time.monotonic() < deadline - 400:
        g = _lp_minmax(f)
        if g is None:
            no_imp += 1
            break
        
        # Check if g is essentially f (stagnation)
        if np.allclose(g, f, atol=1e-6):

            break
        
        b = _conv_full(f)
        M = float(b.max())
        
        t = _line_search(f, g, M)
        if t < 1e-8:

            if no_imp > max_no_imp:
                break
            continue
        
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        fs = f_new.sum()
        if fs < 1e-10:
            no_imp += 1
            continue
        f_new = f_new / fs
        
        sc = _score(f_new)
        if sc < best_score - 1e-14:
            best_f = f_new.copy()
            best_score = sc
            f = f_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max_no_imp:
                break
    
    # Phase 2: Subgradient (if time)
    if time.monotonic() < deadline - 200:
        f = best_f.copy()

        for it in range(50000):

            # Subgradient: direction of increase of max

                        f[i] = orig
            if not improved:

    except:
        final_sc = math.inf
    
    if not math.isfinite(final_sc):
        return [float(x) for x in parent]
    
    return [float(x) for x in best_f]
