- For coordinate i, the objective as a function of f_i is:
  F(f_i) = 2n * max_k conv(f,f)_k / (S - f_i_old + f_i)^2

  For each such k, conv_k depends on f_i linearly (since conv is bilinear):
  conv_k(f_i) = conv_k(f_i=0) + f_i * (2*f_{k-i} - 2*f_i_old + f_i_old... )
  
  Actually, conv_k = sum_j f_j * f_{k-j}. The terms involving f_i are:
  - j=i: f_i * f_{k-i}
  - k-j=i, i.e., j=k-i: f_{k-i} * f_i
  So conv_k = (const) + 2*f_{k-i}*f_i (if i ≠ k-i) or (const) + f_i^2 (if i = k-i)
  
  And S_new = S_old - f_i_old + f_i.
  
  So F(f_i) = 2n * max_k (a_k + b_k * f_i) / (S_old - f_i_old + f_i)^2
  
  where a_k is conv_k without the f_i terms, and b_k is the coefficient (2*f_{k-i} or 2*f_i for the diagonal).
  
  The function to minimize is the max of linear functions divided by a linear function squared. This is a fractional function.
  
  For the 1D minimization, I can use golden-section search on f_i ∈ [0, f_i_max] where f_i_max is some reasonable upper bound.

def _golden1d(f, i, lo, hi, fn, iters=30):
    """1D golden-section search for f[i] in [lo, hi] minimizing fn."""
    ip = (math.sqrt(5) + 1) / 2
    a, b = lo, hi
    c = b - (b - a) / ip
    d = a + (b - a) / ip
    f[i] = c
    fc = fn()
    f[i] = d
    fd = fn()
    for _ in range(iters):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - (b - a) / ip
            f[i] = c
            fc = fn()
        else:
            a = c
            c = d
            fc = fd
            d = a + (b - a) / ip
            f[i] = d
            fd = fn()
    best = c if fc < fd else d
    f[i] = best
    return min(fc, fd)

                rows.append(ri)
                cols.append(i)
                vals.append(v)
    if not rows:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(pk), n))
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=bv,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': 3.0, 'presolve': True})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _lp_run(f0, t_end, seed, pfracs=None):
    if pfracs is None:
        pfracs = [0.10, 0.06, 0.04, 0.02, 0.01]
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0.0, None)
    best_f, best_sc = f.copy(), _sc(f)
    t, stall, mstall, pi = 0.3, 0, 120, 0

    while time.monotonic() < t_end - 3.0:
        b = _conv(f)
        M = float(b.max())
        if M < 1e-18:
            break

        pf = pfracs[min(pi, len(pfracs) - 1)]
        th = M * (1.0 - pf)
        pk = np.where(b >= th)[0]
        if len(pk) == 0:
            pk = np.array([int(np.argmax(b))])

        g = _lp_dir(f, n, pk, np.full(len(pk), M))
        if g is None:
            stall += 4
            if stall > mstall:
                pi = min(pi + 1, len(pfracs) - 1)
                f = np.clip(f + rng.normal(0, 0.06 * max(1.0, f.mean()), n), 0.0, None)
                s = f.sum()
                if s > 1e-10:
                    f *= best_f.sum() / s
                t, stall, mstall = 0.3, 0, min(mstall + 50, 500)
            continue

        g *= f.sum() / g.sum()
        rd = np.linalg.norm(g - f) / max(1.0, np.linalg.norm(f))
        if rd < 1e-10:
            stall += 3
            if stall > mstall:
                pi = min(pi + 1, len(pfracs) - 1)
                f = np.clip(f + rng.normal(0, 0.08 * max(1.0, f.mean()), n), 0.0, None)
                s = f.sum()
                if s > 1e-10:
                    f *= best_f.sum() / s
                t, stall, mstall = 0.3, 0, min(mstall + 50, 500)
            continue

        bt, bls = t, math.inf
        for tt in (t, t * .5, t * 1.5, .05, .1, .15, .2, .3, .4, .5, .6, .7, .8, .9):
            tt = max(.01, min(.99, tt))
            fn = np.clip((1.0 - tt) * f + tt * g, 0.0, None)
            if fn.sum() < .01:
                continue
            sc = _sc(fn)
            if sc < bls:
                bls, bt = sc, tt

        if bls < best_sc - 1e-14:
            f = np.clip((1.0 - bt) * f + bt * g, 0.0, None)
            best_f, best_sc, t, stall = f.copy(), bls, bt, 0
        else:
            stall += 1
            t = max(.02, t * .8)
            if stall > mstall:
                pi = min(pi + 1, len(pfracs) - 1)
                f = np.clip(f + rng.normal(0, .10 * max(1.0, f.mean()), n), 0.0, None)
                s = f.sum()
                if s > 1e-10:
                    f *= best_f.sum() / s
                t, stall, mstall = 0.3, 0, min(mstall + 60, 600)

    return best_f, best_sc


def _cd(f0, t_end, seed):
    """Coordinate descent with golden-section 1D search."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64), 0.0, None).copy()
    best_sc = _sc(f)
    best_f = f.copy()
    ni, mni = 0, 8
    ip = (math.sqrt(5.0) + 1.0) / 2.0
    inv_ip = 1.0 / ip

    while time.monotonic() < t_end - 3.0 and ni < mni:
        imp = False
        for i in rng.permutation(n):
            if time.monotonic() > t_end - 3.0:
                break
            o = f[i]
            if o < 1e-14:
                f[i] = 0.01 * max(1.0, f.mean())
                sc = _sc(f)
                if sc < best_sc - 1e-14:
                    best_sc, best_f = sc, f.copy()
                    imp = True
                else:
                    f[i] = 0.0
                continue

            a, b_ = 0.0, min(o * 5.0 + 0.5, 1000.0)
            if b_ - a < 1e-15:
                b_ = a + 0.1

            c = b_ - inv_ip * (b_ - a)
            d = a + inv_ip * (b_ - a)
            f[i] = c
            sc_c = _sc(f)
            f[i] = d
            sc_d = _sc(f)

            for _ in range(28):
                if b_ - a < 1e-10 * max(1.0, o):
                    break
                if sc_c < sc_d:
                    b_, d, sc_d = c, c, sc_c  # Wait, this is wrong
                    # Let me redo:
                    # c = b - (b-a)/ip, d = a + (b-a)/ip
                    # If sc_c < sc_d, optimum is in [a, d]
                    b_ = d
                    d = c
                    sc_d = sc_c
                    c = b_ - inv_ip * (b_ - a)
                    f[i] = c
                    sc_c = _sc(f)
                else:
                    a = c
                    c = d
                    sc_c = sc_d
                    d = a + inv_ip * (b_ - a)
                    f[i] = d
                    sc_d = _sc(f)

            bv = c if sc_c <= sc_d else d
            bsv = min(sc_c, sc_d)
            f[i] = bv
            if bsv < best_sc - 1e-14:
                best_sc, best_f = bsv, f.copy()
                imp = True
            else:
                f[i] = o

        ni = 0 if imp else ni + 1

    sc = _sc(best_f)
    if sc < best_sc:
        best_sc = sc
    return best_f, best_sc
