import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _lp_solve(f, n, M, time_limit=3.0):
    # Build A matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                     bounds=[(0, None)] * n,
                     method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs

def _golden_minconv(f, g, lo=0.0, hi=1.0, max_evals=14):
    gr = (1.0 + math.sqrt(5.0)) / 2.0
    a, b = lo, hi
    c_pt = b - (b - a) / gr
    d_pt = a + (b - a) / gr
    fc = float(_conv_full((1 - c_pt) * f + c_pt * g).max())
    fd = float(_conv_full((1 - d_pt) * f + d_pt * g).max())
    evals = 2
    best_t, best_v = (c_pt, fc) if fc < fd else (d_pt, fd)
    for _ in range(max_evals // 2):
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - (b - a) / gr
            fc = float(_conv_full((1 - c_pt) * f + c_pt * g).max())
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + (b - a) / gr
            fd = float(_conv_full((1 - d_pt) * f + d_pt * g).max())
        evals += 1

        v_mid = float(_conv_full((1 - t_mid) * f + t_mid * g).max())
        evals += 1
        if v_mid < best_v:
            best_v = v_mid
            best_t = t_mid
        if b - a < 1e-10:

    best_sc = _score(best_f)
    return best_f, best_sc

def _coord_descent(a, stop_time, seed, n_sweeps_max=200):
    """Cyclic coordinate descent with 1D minimization."""
    n = len(a)
    best = np.clip(a.astype(np.float64).copy(), 0, None)
    bs = float(best.sum())
    if bs < 1e-10:
        return best, math.inf
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp_sweeps = 0
    max_no_imp = 15

    for sweep in range(n_sweeps_max):
        if time.monotonic() > stop_time - 1.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            old = best[i]
            # Try multiplier range
            if old > 1e-12:
                mults = np.array([0.0, 0.1, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0, 5.0])
            else:
                mults = np.array([0.0, 0.01, 0.05, 0.1, 0.5, 1.0])
            
            best_nv = old
            best_sc_i = best_sc
            for m in mults:
                nv = old * m if old > 1e-12 else m
                best[i] = nv
                sc = _score(best)
                if sc < best_sc_i - 1e-15:
                    best_sc_i = sc
                    best_nv = nv
            best[i] = best_nv
            
            # Golden section around best
            if best_nv != old and time.monotonic() < stop_time - 2.0:
                lo = max(0, old * 0.3) if old > 0 else 0
                hi = old * 3.0 if old > 0 else 1.0
                # Narrow to around best_nv
                ratio = best_nv / old if old > 1e-12 else 1.0
                lo = max(0.0, best_nv * 0.5)
                hi = best_nv * 2.0 if best_nv > 0 else old * 2.0 + 0.01
                gr = (1 + math.sqrt(5)) / 2
                a_g, b_g = lo, hi
                c_g = b_g - (b_g - a_g) / gr
                d_g = a_g + (b_g - a_g) / gr
                fc = _score(best.copy().take([i]).squeeze() if False else np.where(np.arange(n)==i, c_g, best)[np.arange(n)])
                # Actually let me just do it simply
                best[i] = c_g
                fc = _score(best)
                best[i] = d_g
                fd = _score(best)
                for _ in range(6):
                    if fc < fd:
                        b_g = d_g; d_g = c_g; fd = fc
                        c_g = b_g - (b_g - a_g) / gr
                        best[i] = c_g; fc = _score(best)
                    else:
                        a_g = c_g; c_g = d_g; fc = fd
                        d_g = a_g + (b_g - a_g) / gr
                        best[i] = d_g; fd = _score(best)
                    if b_g - a_g < 1e-12:
                        break
                t_mid = (a_g + b_g) / 2
                best[i] = t_mid
                sc_mid = _score(best)
                if sc_mid < best_sc_i:
                    best_sc_i = sc_mid
                    best_nv = t_mid
                best[i] = best_nv
            
            if best_sc_i < best_sc - 1e-15:
                best_sc = best_sc_i
                improved = True

        # 2-coord moves
        n_pairs = min(500, n * 2)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 1.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            total = oi + oj
            if total < 1e-12:

            best_sc_p = best_sc

                best[i] = t * total
                best[j] = (1.0 - t) * total
                sc = _score(best)
                if sc < best_sc_p - 1e-15:
                    best_sc_p = sc
                    best_t = t
            # Golden section refinement
            lo = max(0.0, best_t - 0.15)
            hi = min(1.0, best_t + 0.15)
            gr = (1 + math.sqrt(5)) / 2
            a_g, b_g = lo, hi
            c_g = b_g - (b_g - a_g) / gr
            d_g = a_g + (b_g - a_g) / gr
            best[i] = c_g * total; best[j] = (1-c_g) * total
            fc = _score(best)
