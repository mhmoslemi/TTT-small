import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _cross_corr(p, w):
    """Compute the n-point cross-correlation (centered part of convolution)."""
    p = np.asarray(p, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = p.size
    L = 3 * n - 2
    size = next_fast_len(L)
    F = rfft(p[::-1], size)
    G = rfft(w, size)
    c = irfft(F * G, size)[:L]
    q = c[n - 1:2 * n - 1]
    q = np.asarray(q, dtype=np.float64)
    np.maximum(q, 0.0, out=q)
    return q


def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        e = np.full(u.size, 1.0 / u.size, dtype=np.float64)
        s = 1.0
    return e / s


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))


def _score_exact_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    return float(2.0 * a.size * m / (s * s))


def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == 1:
        return np.full(n, float(src[0]) / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / float(n), xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _lp_direction(f, max_time=60.0):
    """LP: max sum(g) s.t. (f_norm * g)_k <= M, g >= 0. Returns g normalized to sum=1."""
    n = f.size
    s = float(f.sum())
    if s < 1e-300:
        return None
    f_norm = f / s  # sum = 1
    c = _conv_self(f_norm)
    M = float(c.max())
    if M < 1e-300 or not math.isfinite(M):
        return None
    
    n_con = 2 * n - 1
    # Build sparse convolution matrix
    # A[k, j] = f_norm[k-j] for 0<=k-j<=n-1 (i.e., j<=k<=j+n-1)
    rows_list = []
    cols_list = []
    data_list = []
    for j in range(n):
        nnz = min(n, 2 * n - 1 - j)
        r = np.arange(j, j + nnz, dtype=np.int32)
        rows_list.append(r)
        cols_list.append(np.full(nnz, j, dtype=np.int32))
        data_list.append(f_norm[:nnz])
    
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    data = np.concatenate(data_list)
    del rows_list, cols_list, data_list
    
    A = csr_matrix((data, (rows, cols)), shape=(n_con, n))
    del rows, cols, data
    
    c_obj = -np.ones(n)
    b_ub = np.full(n_con, M)
    
    t_start = time.monotonic()
    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0.0, 1000.0)] * n,
            method='highs',
            options={'time_limit': max(max_time - 5.0, 5.0), 'output_flag': 0}
        )
    except Exception:
        return None
    
    del A
    
    if time.monotonic() - t_start > max_time:
        return None
    
    if res.status == 0 and res.x is not None:
        g = np.asarray(res.x, dtype=np.float64)
        g = np.maximum(g, 0.0)
        sg = float(g.sum())
        if sg > 1e-300 and math.isfinite(sg):
            g = g / sg
            return g
    return None


def _lse_optimize(p, n, stop_time, consider=None, fine_only=False):
    """LSE-based L-BFGS optimizer."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300), dtype=np.float64)
        return uu - float(np.mean(uu))

    u = p_to_u(p)

    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu, dtype=np.float64)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0.0:
            return 1e100, np.zeros_like(uu, dtype=np.float64)
        if T <= 0.0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    def run_chunk(uu, T, end_stage):
        try:
            f0, _ = lse_func(uu, T)
        except Exception:
            return uu
        if not math.isfinite(f0):
            return uu
        best_u = np.array(uu, dtype=np.float64, copy=True)
        best_f = float(f0)
        stale = 0
        while time.monotonic() < end_stage - 2.0:
            maxit = 30 if n > 2200 else 50
            opts = {
                "maxiter": maxit,
                "gtol": 1e-12 if n > 1200 else 1e-13,
                "ftol": 1e-16,
                "maxls": 30,
                "maxcor": 30,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(
                    lse_func,
                    best_u,
                    args=(T,),
                    method="L-BFGS-B",
                    jac=True,
                    options=opts,
                )
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == uu.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, "status", 1) == 0:
                break
        return best_u

    if fine_only:
        stages = ((0.002, 0.30), (0.0005, 0.35), (0.0001, 0.35))
    else:
        stages = (
            (0.20, 0.10),
            (0.06, 0.15),
            (0.018, 0.20),
            (0.005, 0.20),
            (0.0015, 0.25),
            (0.0004, 0.10),
        )

    start = time.monotonic()
    avail = float(stop_time - start - 12.0)
    if avail <= 5.0:
        return p

    best_p = p.copy()
    best_sc = _score_fft(p)
    total_w = float(sum(w for _, w in stages))
    if total_w <= 0.0:
        total_w = 1.0

    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 8.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 6.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        sc_cur = _score_fft(p_cur)
        if not math.isfinite(sc_cur):
            p = best_p.copy()
            u = p_to_u(p)
            continue
        if sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif sc_cur > best_sc + max(0.003, 3e-4 * best_sc):
            p = best_p.copy()
        else:
            p = p_cur.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, max_sweeps=50000, G=None, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p, math.inf
    p = p / s

    if G is None:
        if n <= 450:
            G = 56
        elif n <= 900:
            G = 44
        elif n <= 1800:
            G = 36
        elif n <= 4000:
            G = 28
        else:
            G = 20
    G = int(max(16, G))

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.6:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.25 or S_total < 0.75:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end]
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start > 0:
                ml = float(np.max(c[:start]))
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(np.max(c[end:]))
                if mr > max_const:
                    max_const = mr

            x_hi = max(20.0 * v, 60.0 * S / float(n), 0.05 * S, 1e-12)
            x_hi = min(x_hi, 10.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, S / float(n))

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            dots = float(v)
            if 0.0 <= dots <= x_hi:
                pts = np.concatenate((pts, [dots]))
            pts = np.unique(pts)

            diag = i

            def eval_xs(xs, base=base, infl=infl, max_const=max_const, diag=diag):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                dval = base[diag] + w * infl[diag] + w * w
                maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if G > 1:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 32, dtype=np.float64)
                fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-12 and 0.0 <= best_x <= 10.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        np.maximum(c, 0.0, out=c)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 10.0:
                    p = p * np.exp(rng.normal(0.0, 0.04, n, dtype=np.float64))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0.0:
                        p /= ss
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            consider(best_p)
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 2.0:
        hard = t0 + max(2.0, budget - 1.0)
    usable = max(1.0, hard - t0)
    final_reserve = max(10.0, 0.10 * usable) if usable > 80.0 else 0.0
    phase_limit = hard - final_reserve

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64)
        best_arr /= 64.0
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-13:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    def make_skew(src, n, sd, amp_range=(0.8, 2.5)):
        p = _resample_mid(src, n)
        rng = np.random.default_rng(sd)
        x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
        side = 1.0 if rng.random() < 0.5 else -1.0
        amp = amp_range[0] + (amp_range[1] - amp_range[0]) * float(rng.random())
        q = p * np.exp(side * amp * (2.0 * x - 1.0))
        q = np.maximum(q, 0.0)
        s = float(q.sum())
        if s > 0.0:
            q /= s
        else:
            q = p
        return q

    def run_lp_kick(p, stop_time, consider):
        """Try LP-based direction and apply it."""
        nonlocal best_arr, best_score
        rem = stop_time - time.monotonic()
        if rem < 20.0:
            return p
        n = p.size
        if n > 4096:
            return p  # LP too expensive for very large n
        max_lp_time = min(rem * 0.3, 120.0)
        g = _lp_direction(p, max_time=max_lp_time)
        if g is None:
            return p
        # Try mixing: f_new = (1-t)*f + t*g for several t
        best_p_local = p.copy()
        best_sc_local = _score_fft(p)
        for t in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7):
            mix = (1 - t) * p + t * g
            mix = np.maximum(mix, 0.0)
            sm = float(mix.sum())
            if sm < 1e-300:
                continue
            mix = mix / sm
            sc = _score_fft(mix)
            if math.isfinite(sc) and sc < best_sc_local:
                best_sc_local = sc
                best_p_local = mix.copy()
                consider(mix)
        return best_p_local

    # Main optimization on the parent length
    parent_len = int(best_arr.size) if best_arr.size > 0 else 3072
    
    # We'll focus on the parent length since it's already well-optimized
    # Try a couple of variants for the same length
    rng_top = np.random.default_ng(seed)  # BUG: should be np.random.default_rng
    rng_top = np.random.default_rng(seed)
    
    phase_limit = min(phase_limit, hard - 15.0)
    
    # Phase 1: LSE optimization on parent (70% of time)
    # Phase 2: LP kicks + LSE fine (15% of time)
    # Phase 3: Coordinate polish (15% of time)
    
    n_opt = parent_len
    p_work = best_arr.copy()
    p_work = p_work / float(p_work.sum())
    
    work_budget = phase_limit - t0
    t_lse_end = t0 + 0.55 * work_budget
