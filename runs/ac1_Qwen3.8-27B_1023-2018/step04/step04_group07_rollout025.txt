import numpy as np

from scipy.optimize import linprog

def _conv_full(a):
    """Full linear convolution via FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    """Compute the exact score."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_step(f, n, M, time_limit=2.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Return normalized g."""
    A = _build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                     method='highs', options={'presolve': True, 'time_limit': time_limit,
                                              'mip_gap': 0.0})

        return None

    g = np.asarray(res.x, dtype=np.float64)

    if g_sum < 1e-12:

            fs4 = f_try.sum()
            if fs4 > 1e-10:
                f_try = f_try / fs4
                M_try = float(_conv_full(f_try).max())
                if M_try < best_M:
                    f = f_try
                    M = M_try
                    best_M = M
                    best_f = f.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        
        iter_count += 1
    
    return best_f, _score(best_f)
