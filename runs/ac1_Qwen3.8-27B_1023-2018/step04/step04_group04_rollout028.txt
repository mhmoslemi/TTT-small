def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m):

    return 2.0 * n * m / (s * s)

def _lp_direction(f, n, M, timeout=3.0):

        return None
    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

        if gs < 1e-15:

            continue
        g = g / gs

        if dist < 1e-10:
            stall += 1
            if stall > max_stall:
                break
            continue

        best_sc_try = best_sc

            best_f = f.copy()
            best_sc = best_sc_try

                stall = 0

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=3.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _lp_optimize(f0, n, stop_time, seed, max_iter=50000):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    fs = f.sum()
    f = f / max(fs, 1e-10)

    best_f = f.copy()
    best_sc = _score(f)
    t_param = 0.3
    stall = 0
    max_stall = 400

    for it in range(max_iter):
        if time.monotonic() > stop_time:
            break

        b = _fft_conv(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, n, M, timeout=2.5)
        if g is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.015, n), 0, None)
            f = f / max(f.sum(), 1e-10)
            continue

        gs = g.sum()
        if gs < 1e-15:
            stall += 3
            continue
        g = g / gs

        dist = np.linalg.norm(g - f)
        if dist < 1e-11:
            stall += 2
            if stall > max_stall:
                break
            continue

        best_t = 0.0
        best_sc_try = best_sc
        # Adaptive step sizes centered around t_param
        candidates = sorted(set([
            t_param, t_param * 0.4, t_param * 0.7, t_param * 1.3,
            t_param * 1.8, 0.05, 0.15, 0.3, 0.5, 0.7, 0.85, 0.95
        ]))
        for tt in candidates:
            tt = min(0.99, max(0.005, tt))
            fn = (1 - tt) * f + tt * g
            fn = np.clip(fn, 0, None)
            if fn.sum() < 0.01:
                continue
            sc_try = _score(fn)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_t = tt

        if best_t > 0 and best_sc_try < best_sc - 1e-14:
            fn = (1 - best_t) * f + best_t * g
            fn = np.clip(fn, 0, None)
            f = fn
            best_f = f.copy()
            best_sc = best_sc_try
            t_param = min(0.9, best_t * 1.15)
            stall = 0
        else:
            t_param = max(0.01, t_param * 0.5)
            stall += 1
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                t_param = 0.3
                stall = 0
                max_stall = min(max_stall + 150, 1500)

    return best_f, best_sc


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smoothed_grad(a, tau):
    n = len(a)
    b = _fft_conv(a)
    bm = b.max()
    exp_b = np.exp(tau * (b - bm))
    w = exp_b / exp_b.sum()
    val = (bm + math.log(exp_b.sum()) / tau) / tau if tau > 0 else bm
    grad = np.zeros(n)
    for m in range(n):
        k_lo = m
        k_hi = min(m + n, 2 * n - 1)
        if k_lo < k_hi:
            grad[m] = 2.0 * np.dot(w[k_lo:k_hi], a[:k_hi - k_lo])
    return val, grad, w


def _fista_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = _project_simplex(np.asarray(f0, dtype=np.float64).copy())
    best_f = f.copy()
    best_sc = _score(f)

    tau = 5.0 * n
    step = 1.0 / (4.0 * n * tau)
    t_k = 1.0
    y = f.copy()

    for it in range(50000):
        if time.monotonic() > stop_time:
            break

        _, grad, w = _smoothed_grad(y, tau)
        # Backtracking line search
        s = step
        z = y - s * grad
        a_new = _project_simplex(z)
        # Check sufficient decrease
        val_y, _, _ = _smoothed_grad(y, tau)
        val_new, _, _ = _smoothed_grad(a_new, tau)
        if val_new > val_y - 0.5 * s * np.dot(grad, y - a_new):
            # Backtrack
            for _ in range(10):
                s *= 0.5
                z = y - s * grad
                a_new = _project_simplex(z)
                val_new, _, _ = _smoothed_grad(a_new, tau)
                if val_new <= val_y - 0.5 * s * np.dot(grad, y - a_new):
                    break

        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - f)
        y = np.clip(y, 0, None)
        ys = y.sum()
        if ys > 1e-10:
            y = y / ys
        f = a_new
        t_k = t_new
        step = s * 1.05  # slightly increase step for next iter

        sc = _score(f)
        if sc < best_sc - 1e-13:
            best_f = f.copy()
            best_sc = sc

        # Increase tau periodically
        if it % 200 == 0:
            tau *= 1.5
            step = 1.0 / (4.0 * n * tau)

        # Occasional restart
        if it % 500 == 0 and it > 0:
            if _score(f) > best_sc * 1.01:
                f = _project_simplex(best_f + rng.normal(0, 0.02, n))
                y = f.copy()
                t_k = 1.0
                step = 1.0 / (4.0 * n * tau)

    return best_f, best_sc


def _coord_descent(f0, stop_time, seed, max_epochs=200):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    s0 = f.sum()
    n = len(f)
    best_sc = _score(f)
    best_f = f.copy()

    for epoch in range(max_epochs):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            if orig < 1e-15:
                continue
            # Try different values for f[i], adjust f[(i+1)%n] to maintain sum
            j = (i + 1) % n
            orig_j = f[j]
            best_v = orig
            best_vj = orig_j
            best_v_sc = best_sc

            # Binary-ish search: try several multipliers
            trials = [0.0, 0.1, 0.25, 0.4, 0.6, 0.8, 1.1, 1.4, 1.8, 2.5]
            for mult in trials:
                new_i = orig * mult
                delta = new_i - orig
                new_j = orig_j - delta
                if new_j < -1e-10:
                    # Redistribute delta among all other coords proportionally
                    continue
                f[i] = max(0.0, new_i)
                f[j] = max(0.0, new_j)
                if f.sum() < 0.01:
                    f[i], f[j] = orig, orig_j
                    continue
                sc = _score(f)
                if sc < best_v_sc:
                    best_v_sc = sc
                    best_v = f[i]
                    best_vj = f[j]

            f[i] = best_v
            f[j] = best_vj
            if best_v_sc < best_sc - 1e-15:
                best_sc = best_v_sc
                best_f = f.copy()
                improved = True

        # Also try adding mass to zero coordinates
        zero_idx = np.where(f < 1e-12)[0]
        if len(zero_idx) > 0:
            for _ in range(min(50, len(zero_idx))):
                if time.monotonic() > stop_time:
                    break
                zi = int(zero_idx[rng.integers(len(zero_idx))])
                src = int(rng.integers(n))
                if src == zi or f[src] < 1e-10:
                    continue
                for mult in [0.1, 0.3, 0.5, 0.8]:
                    transfer = f[src] * mult
                    f[zi] = transfer
                    f[src] -= transfer
                    if f.sum() < 0.01:
                        f[zi] = 0
                        f[src] += transfer
                        continue
                    sc = _score(f)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        best_f = f.copy()
                        improved = True
                    else:
                        f[zi] = 0
                        f[src] += transfer
                        break

        if not improved:
            break

    return best_f, best_sc
