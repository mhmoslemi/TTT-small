- x <= min over linear constraints

def _get_parent():

        h = globals()["height_sequence_1"]

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _lp_direction(f, n, b, peak_frac=0.05, timeout=6.0):
    M = float(b.max())
    if M < 1e-18:
        return None
    threshold = M * (1.0 - peak_frac)
    peak_idx = np.where(b >= threshold)[0]
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(b)])

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(peak_idx):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-20:
                rows.append(idx)
                cols.append(i)
                vals.append(v)

    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    c_vec = -np.ones(n)
    b_ub = np.full(len(peak_idx), M)
    bounds = [(0.0, None)] * n

                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None

        return None

    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

            if stall >= max_stall:
                noise = rng.normal(0, 0.15 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                if f.sum() > 1e-10:
                    f = f / f.sum() * best_f.sum()
                t = 0.25
                stall = 0
                max_stall = min(max_stall + 30, 300)

    return best_f, best_score


def _coord_descent_exact(f0, stop_time, seed, max_rounds=100):
    """Exact coordinate descent: for each coordinate, find the optimal value."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    if f.sum() < 0.01:
        f = np.ones(n)
    best_f = f.copy()
    best_score = _score(f)
    b = _conv_full(f)
    unaffected_mask_cache = {}

    def _score_at_x(i, v, x, S_rest, b_cur, k_range, coeffs, quad_k, unaff_max):
        if S_rest + x < 0.01:
            return math.inf
        new_vals = b_cur[k_range] + 2.0 * (x - v) * coeffs
        qi = np.where(k_range == quad_k)[0]
        if len(qi) > 0:
            new_vals[qi[0]] = b_cur[quad_k] + x * x - v * v
        max_b = max(float(np.max(new_vals)) if len(new_vals) > 0 else 0.0, unaff_max)
        if max_b < 0:
            max_b = 0.0
        return 2.0 * n * max_b / ((S_rest + x) ** 2)

    round_num = 0
    no_imp_rounds = 0
    max_no_imp = 15

    while time.monotonic() < stop_time - 3.0 and round_num < max_rounds:
        round_num += 1
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break

            v = f[i]
            S_rest = float(f.sum()) - v
            if S_rest < 0.001:
                S_rest = 0.001

            k_range = np.arange(max(0, i), min(2 * n - 1, i + n), dtype=np.int64)
            diffs = k_range - i
            coeffs = 2.0 * f[diffs]
            quad_k = 2 * i if 2 * i < 2 * n - 1 else -1

            # Compute unaffected max
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[k_range] = False
            unaff_max = float(b[mask].max()) if mask.any() else 0.0

            # Search for optimal x
            x_hi = max(3.0 * abs(v), 0.5) if v > 0 else 1.0
            x_hi = min(x_hi, 50.0)

            # Coarse search
            xs = np.linspace(0, x_hi, 60)
            scores = np.array([_score_at_x(i, v, x, S_rest, b, k_range, coeffs, quad_k, unaff_max) for x in xs])
            valid = np.isfinite(scores)
            if not valid.any():
                continue
            best_idx = np.argmin(np.where(valid, scores, np.inf))
            best_x = xs[best_idx]
            best_local_sc = scores[best_idx]

            # Fine search
            step = x_hi / 60.0
            for _ in range(25):
                best_found = False
                for trial in (best_x - step, best_x + step):
                    if trial < 0 or trial > x_hi:
                        continue
                    sc = _score_at_x(i, v, trial, S_rest, b, k_range, coeffs, quad_k, unaff_max)
                    if sc < best_local_sc - 1e-15:
                        best_local_sc = sc
                        best_x = trial
                        best_found = True
                step *= 0.45
                if not best_found and step < 1e-12:
                    break

            if best_local_sc < best_score - 1e-14:
                f[i] = best_x
                best_f = f.copy()
                best_score = best_local_sc
                # Update b
                b = _conv_full(f)
                improved = True
            else:
                # Try small perturbation
                if v > 1e-10:
                    for mult in (0.5, 0.8, 1.2, 0.3):
                        trial_x = v * mult
                        sc = _score_at_x(i, v, trial_x, S_rest, b, k_range, coeffs, quad_k, unaff_max)
                        if sc < best_score - 1e-14:
                            f[i] = trial_x
                            best_f = f.copy()
                            best_score = sc
                            b = _conv_full(f)
                            improved = True
                            break

        if not improved:
            no_imp_rounds += 1
            if no_imp_rounds >= max_no_imp:
                break
        else:
            no_imp_rounds = 0

    return best_f, best_score


def _subgrad_optimize(f0, stop_time, seed):
    """Mirror descent / multiplicative weights for subgradient optimization."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-15, None)
    f = f / f.sum()
    best_f = f.copy()
    best_score = _score(f)

    lr = 0.3
    stall = 0
    max_stall = 60
    momentum = np.zeros(n)
    mu = 0.5  # Nesterov momentum

    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        peak_threshold = M * 0.99
        peak_idx = np.where(b >= peak_threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])

        grad = np.zeros(n)
        for k in peak_idx:
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                j = k - i
                if 0 <= j < n:
                    if i == j:
                        grad[i] += 2.0 * f[i]
                    else:
                        grad[i] += 2.0 * f[j]
        grad /= len(peak_idx)

        # Nesterov momentum
        momentum = mu * momentum + (1 - mu) * grad
        grad_scale = np.max(momentum)
        if grad_scale < 1e-15:
            break

        # Multiplicative weights update
        exponent = -lr * momentum / grad_scale
        exponent -= np.max(exponent)
        weights = np.exp(exponent)
        f_new = f * weights
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new /= s

        sc = _score(f_new)
        if sc < best_score - 1e-13:
            best_f = f_new.copy()
            best_score = sc
            f = f_new
            stall = 0
            lr *= 0.995
        else:
            # Try damped
            f_damp = 0.7 * f + 0.3 * f_new
            f_damp = np.clip(f_damp, 1e-15, None)
            f_damp /= f_damp.sum()
            sc_d = _score(f_damp)
            if sc_d < best_score - 1e-13:
                best_f = f_damp.copy()
                best_score = sc_d
                f = f_damp
                stall = 0
                lr *= 0.98
            else:
                f = f_new  # still move (explore)
                stall += 1
                lr *= 0.97
                if stall > max_stall:
                    f = np.clip(f + rng.normal(0, 0.02, n), 1e-15, None)
                    f /= f.sum()
                    stall = 0
                    lr = 0.2
                    max_stall = min(max_stall + 40, 300)

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = _score(best_seq)

    def _update_best(candidate):
        nonlocal best_seq, best_score
        if candidate is None or len(candidate) == 0:
            return
        candidate = np.clip(np.asarray(candidate, dtype=np.float64), 0, 1000)
        if not np.all(np.isfinite(candidate)):
            return
        if candidate.sum() < 0.01:
            return
        sc = _score(candidate)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq = candidate.copy()
            best_score = sc

    rng = np.random.default_rng(seed)
    i_arr = np.arange(n_p, dtype=np.float64)

    # Build diverse initializations
    inits = [parent.copy()]

    sig = 0.1 * max(1.0, float(parent.mean()))
    inits.append(np.clip(parent + rng.normal(0, sig, n_p), 0, None))
    inits.append(np.clip(parent * (1.0 + 0.2 * rng.normal(0, 1, n_p)), 0, None))

    # Cosine-based
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n_p)))

    # Multi-frequency
    sup = np.ones(n_p) * 0.3
    for fr in (1, 2, 3, 5, 7, 11, 13):
        ph = rng.uniform(0, 2 * math.pi)
        sup += 0.1 * np.cos(2.0 * math.pi * fr * i_arr / n_p + ph)
    inits.append(np.abs(sup) + 0.01)

    # Gaussian bumps
    center = (n_p - 1) / 2.0
    for width in (n_p * 0.15, n_p * 0.25, n_p * 0.35):
        inits.append(np.exp(-0.5 * ((i_arr - center) / width) ** 2))

    # Two bumps
    c1, c2 = n_p * 0.25, n_p * 0.75
    w = n_p * 0.1
    inits.append(np.exp(-0.5 * ((i_arr - c1) / w) ** 2) + np.exp(-0.5 * ((i_arr - c2) / w) ** 2))

    # Ramped
    inits.append(np.linspace(0.1, 1.0, n_p))
    inits.append(np.linspace(1.0, 0.1, n_p))

    # Uniform
    inits.append(np.ones(n_p))

    # Normalize all inits
    inits = [np.clip(ix, 0, None) for ix in inits]
    inits = [ix / ix.sum() * max(1.0, best_seq.sum()) if ix.sum() > 0 else ix for ix in inits]

    # Phase 1: LP optimization on multiple inits (40% time)
    phase1_end = t0 + budget * 0.40
    time_per_init = max(8.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 5.0:
            break
        stop = min(phase1_end - 3.0, time.monotonic() + time_per_init)
        if time.monotonic() > stop - 2.0:
            continue
        try:
            res, sc = _optimize_lp(init, stop, seed + 10000 * (idx + 1), peak_frac=0.05)
            _update_best(res)
        except Exception:
            pass

    # Phase 2: Coordinate descent on best (30% time)
    phase2_end = t0 + budget * 0.70
    if time.monotonic() < phase2_end - 10.0:
        stop = min(phase2_end - 3.0, time.monotonic() + (phase2_end - time.monotonic() - 5.0))
        try:
            res, sc = _coord_descent_exact(best_seq, stop, seed + 55555)
            _update_best(res)
        except Exception:
            pass

    # Phase 3: More LP on best (10% time)
    phase3_end = t0 + budget * 0.80
    if time.monotonic() < phase3_end - 10.0:
        stop = min(phase3_end - 3.0, time.monotonic() + (phase3_end - time.monotonic() - 5.0))
        try:
            res, sc = _optimize_lp(best_seq, stop, seed + 777777, peak_frac=0.03)
            _update_best(res)
        except Exception:
            pass

    # Phase 4: Subgradient on best (5% time)
    phase4_end = t0 + budget * 0.85
    if time.monotonic() < phase4_end - 10.0:
        stop = min(phase4_end - 3.0, time.monotonic() + (phase4_end - time.monotonic() - 5.0))
        try:
            res, sc = _subgrad_optimize(best_seq, stop, seed + 888888)
            _update_best(res)
        except Exception:
            pass

    # Phase 5: More coordinate descent (5% time)
    phase5_end = t0 + budget * 0.90
    if time.monotonic() < phase5_end - 10.0:
        stop = min(phase5_end - 3.0, time.monotonic() + (phase5_end - time.monotonic() - 5.0))
        try:
            res, sc = _coord_descent_exact(best_seq, stop, seed + 999999, max_rounds=50)
            _update_best(res)
        except Exception:
            pass

    # Phase 6: Try nearby lengths (5% time)
    phase6_end = deadline + 2.0
    if time.monotonic() < phase6_end - 15.0:
        lengths_to_try = sorted(set([n_p - 16, n_p + 16, n_p - 32, n_p + 32]),
                                key=lambda L: abs(L - n_p))
        for L in lengths_to_try:
            if time.monotonic() > phase6_end - 10.0:
                break
            if L < 100 or L > 700:
                continue
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, L)
            init_L = np.interp(x_new, x_old, best_seq)
            stop = min(phase6_end - 5.0, time.monotonic() + 20.0)
            try:
                res, sc = _coord_descent_exact(init_L, stop, seed + L * 7, max_rounds=30)
                _update_best(res)
            except Exception:
                pass

    # Final coordinate descent polish
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        try:
            res, sc = _coord_descent_exact(best_seq, deadline - 3.0, seed + 11111, max_rounds=40)
            _update_best(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    return [float(x) for x in best_seq]
