def _score_fast(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    b = np.convolve(a, a)
    M = float(b.max())
    if not math.isfinite(M) or M < 0:
        return math.inf
    return 2.0 * n * M / (S * S)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, tlimit=3.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': tlimit,
                                               'mip_rel_gap': 0.0})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g

def _grad_step(f, n, alpha=0.05, tau=60.0):
    b = np.convolve(f, f)
    S = float(f.sum())
    M = float(b.max())
    if M < 1e-18 or S < 1e-12:
        return f
    log_p = (b - M) / tau
    log_p -= log_p.max()
    p = np.exp(log_p)
    p /= p.sum()
    f_rev = f[::-1]
    cv_size = 2 * n - 1
    P = rfft(p, cv_size)
    Fr = rfft(f_rev, cv_size)
    grad_conv = irfft(P * Fr, cv_size)[:cv_size]
    grad = 2.0 * grad_conv[n - 1:2 * n - 1]
    score_grad = (2.0 * n / (S * S)) * (grad - 2.0 * M / S)
    f_new = f - alpha * score_grad
    f_new = np.clip(f_new, 0, None)
    sn = f_new.sum()
    if sn > 1e-12:
        f_new = f_new / sn
    else:
        f_new = f.copy()
    return f_new

            res, sc = _coord_descent(best_seq, hard_deadline - 6, max_sweeps=30, seed=seed + 9999)
            _try_update(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = parent.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        fb = parent.copy()
        try:
            fb_sc = float(evaluate_sequence([float(x) for x in fb]))
        except Exception:
            fb_sc = math.inf
        if math.isfinite(fb_sc):
            return [float(x) for x in fb]
        return [1.0 / n_p] * n_p

    return [float(x) for x in best_seq]
