    fsize = next_fast_len(2*n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n - 1]
    np.maximum(c, 0.0, out=c)

def _score(a):

    m = float(b.max())

            # c_new[d_k, m] = c[m] + 2*d_k * shifted_f[m]
            # shifted_f[m] = f[m-i] if 0<=m-i<n else 0
            shifted = np.zeros(2*n-1)
            if i + n <= 2*n - 1:
                shifted[i:i+n] = f
            elif i <= 2*n - 1:
                shifted[i:] = f[:2*n-1-i]
            
            # c_new shape: (n_pts, 2n-1)
            c_new = c[None, :] + 2.0 * d_grid[:, None] * shifted[None, :]
            # Correction at m = 2i
            if 0 <= 2*i <= 2*n-2:
                c_new[:, 2*i] += d_grid**2
            
            c_max = c_new.max(axis=1)
            S_new = 1.0 + d_grid
            valid = S_new > 0.01
            if not valid.any():
                continue
            obj = np.where(valid, 2.0 * n * c_max / S_new**2, np.inf)
            
            best_idx = int(np.argmin(obj))
            best_d = d_grid[best_idx]
            best_obj = obj[best_idx]
            
            if best_obj < best_score - 1e-14 and abs(best_d) > 1e-12:
                # Fine-tune: narrow the grid around best_d
                dw = d_max / n_pts
                d_lo = max(-v, best_d - 3*dw)
                d_hi = min(d_max, best_d + 3*dw)

                c_fine = c[None, :] + 2.0 * d_fine[:, None] * shifted[None, :]
                if 0 <= 2*i <= 2*n-2:
                    c_fine[:, 2*i] += d_fine**2
                c_max_f = c_fine.max(axis=1)
                S_fine = 1.0 + d_fine
                valid_f = S_fine > 0.01
                obj_f = np.where(valid_f, 2.0 * n * c_max_f / S_fine**2, np.inf)
                best_fi = int(np.argmin(obj_f))
                best_d = d_fine[best_fi]
                best_obj = obj_f[best_fi]
                
                if best_obj < best_score - 1e-14:
                    # Update f and c
                    f[i] = v + best_d
                    c += 2.0 * best_d * shifted
                    if 0 <= 2*i <= 2*n-2:
                        c[2*i] += best_d**2
                    S += best_d  # S was 1, now 1 + d
                    # Renormalize
                    f /= S
                    c /= S**2
                    S = 1.0
                    best_score = 2.0 * n * c.max()
                    best_f = f.copy()
                    improved = True
        
        if not improved and sweep > 2:
            break
    
    return best_f, 2.0 * len(best_f) * _conv_full(best_f).max() / (best_f.sum()**2)

Wait, I'm making an error with the normalization. If f has sum=1 and I change f[i] from v to v+d, the new sum is 1+d. Then I renormalize: f_new = f / (1+d). The new convolution is c_new / (1+d)².

Let me redo:
- f has sum = 1, convolution c
- Change f[i] from v to x = v+d
- New f' has sum = 1+d
- New convolution c' = c + 2d*shifted + d²*(at position 2i)
- Score of f' = 2n * max(c') / (1+d)²
- If we accept, renormalize: f_new = f'/(1+d), c_new = c'/(1+d)²
  Score of f_new = 2n * max(c_new) / 1² = 2n * max(c')/(1+d)² (same score)

So the score formula is correct. After updating:
- f[i] = (v+d)/(1+d)
- All other f[j] = f[j]/(1+d)
- c = c'/(1+d)²

This renormalization is O(n) which is fine.

Actually, I realize I should NOT renormalize after every coordinate update. Instead, I should just track the current sum S and use it in the score formula. This avoids the O(n) renormalization per update.

Let me revise:
- Track f and S = sum(f)
- c = conv(f) (of the actual f, not normalized)
- Score = 2n * max(c) / S²
- When changing f[i] from v to x:
  - d = x - v

  - S_new = S + d
