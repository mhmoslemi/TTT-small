```python
def _optimize(f0, stop_time, seed):
    n = len(f0)
    f = f0.copy()
    M = max(convolve(f, f))
    
    while time.monotonic() < stop_time:
        # Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
        # Build constraint matrix
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            for i in range(max(0, k-n+1), min(k+1, n)):
                A[k, i] = f[k-i]
        
        # Solve LP
        from scipy.optimize import linprog
        c = -np.ones(n)  # minimize -sum(g)
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                     bounds=[(0, None)]*n, method='highs')
        
        if res.status != 0:
            break
        
        g = res.x
        if g.sum() < 1e-15:
            break
        
        # Rescale g to match sum of f
        g = g * (f.sum() / g.sum())
        
        # Line search
        best_t = 0
        best_score = score(f)
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            f_new = (1-t)*f + t*g
            s = score(f_new)
            if s < best_score:
                best_score = s
                best_t = t
                best_f = f_new
        
        if best_t == 0:
            break
        
        f = best_f
        M = max(convolve(f, f))
    
    return f
```
