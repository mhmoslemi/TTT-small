The gradient of the softmax:
$\frac{\partial Q}{\partial a_i} = 2 \sum_{k=i}^{i+n-1} w_k \cdot a_{k-i}$ (with bounds)

Let me just think of it as: for each output index $i$ in $[0, n-1]$, we compute $\sum_{j=0}^{n-1} w_{i+j} \cdot a_j$.

from scipy.fft import rfft, irfft, next_fast_len


def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())

        return math.inf

    m = float(b.max())

def _build_A(f, n):
