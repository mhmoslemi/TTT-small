    conv = np.convolve(p, p)

    max_conv = conv.max()
    shifted = beta * (conv - max_conv)
    log_sum = max_conv * beta + np.log(np.sum(np.exp(shifted)))
    F = log_sum / beta  # = max_conv + (1/beta)*log(sum exp(beta*(conv-max_conv)))

= G[j] * p[j] - p[j] * sum_i G[i] * p[i]
= p[j] * (G[j] - sum_i G[i]*p[i])

Let S = sum_i G[i]*p[i] (a scalar). Then:
∂f/∂θ[j] = p[j] * (G[j] - S)

This is clean! And S can be computed as a dot product.

Let me verify: conv[p] = sum_k (sum_i p[i]*p[k-i]). F ≈ max of these.
∂F/∂p[i] should be the "influence" of p[i] on the max convolution value.

If k* is the argmax: ∂conv[k*]/∂p[i] = 2*p[k*-i] (for 0<=k*-i<n).
So G[i] ≈ 2*p[k*-i] (for 0<=k*-i<n), else 0. This makes sense: increasing p[i] increases conv[k*] by 2*p[k*-i].

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _get_parent():
    try:
        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(281, dtype=np.float64)
    p = np.clip(p, 0, 1000)
    if p.size == 0 or not np.all(np.isfinite(p)):
        p = np.ones(281, dtype=np.float64)
    return p


def _fast_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * len(a) * c.max()) / (s * s)


def _fft_len(n):
    m = 3 * n - 2
    return 1 << (m - 1).bit_length()


def _smooth_obj_grad(p, beta, fft_n):
    n = len(p)
    conv = np.convolve(p, p)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    exp_s = np.exp(shifted)
    Z = exp_s.sum()
    obj = max_c + np.log(Z) / beta
    w = exp_s / Z
    W = np.fft.rfft(w, fft_n)
    P = np.fft.rfft(p, fft_n)
    C = np.fft.irfft(W * np.conj(P), fft_n)[:n]
    g_p = 2.0 * C
    s = p.sum()
    sg = 2.0 * n * (g_p / (s * s) - 2.0 * obj / (s * s * s))
    return obj, sg


def _smooth_optimize(p0, time_budget, seed=42):
    p = np.clip(np.asarray(p0, dtype=np.float64), 0, None).copy()
    s = p.sum()
    if s < 1e-12:
        p = np.ones_like(p)
        s = 1.0
    p = p / s
    n = len(p)
    fft_n = _fft_len(n)
    best_p = p.copy()
    best_obj = _fast_score(best_p)
    rng = np.random.default_rng(seed)
    betas = [2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 400.0]
    per_beta = max(0.5, time_budget / len(betas))
    t_start = time.monotonic()
    for beta in betas:
        if time.monotonic() - t_start > time_budget - 1.0:
            break
        alloc = min(per_beta, time_budget - (time.monotonic() - t_start) - 0.5)
        if alloc < 0.3:
            break

        def func(p_curr):
            val, g = _smooth_obj_grad(p_curr, beta, fft_n)
            return val, g

        try:
            res = minimize(func, p, method='L-BFGS-B',
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': 150, 'ftol': 1e-12, 'gtol': 1e-10,
                                    'maxls': 40})
            p_new = np.clip(res.x, 0, None)
            ps = p_new.sum()
            if ps > 1e-10:
                p_new = p_new / ps
                sc = _fast_score(p_new)
                if sc < best_obj:
                    best_obj = sc
                    best_p = p_new.copy()
                p = p_new
        except Exception:
            pass
    return best_p


def _build_A(n, a):
    k = np.arange(2 * n - 1)[:, None].astype(np.int64)
    j = np.arange(n)[None, :].astype(np.int64)
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[d[mask]]
    return A


def _lp_solve(A, M, n, tl=5.0):
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tl,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    return np.clip(res.x, 0.0, None)


def _line_search(a, g, M_cur, n):
    best_t = 0.0
    best_val = M_cur
    cs = a.sum()
    if cs < 1e-12:
        return None, M_cur
    a_norm = a / cs
    g_norm = g / max(g.sum(), 1e-12)
    ts = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    for t in ts:
        cand = (1.0 - t) * a_norm + t * g_norm
        cand = np.maximum(cand, 0)
        csum = cand.sum()
        if csum < 1e-12:
            continue
        val = float(np.convolve(cand, cand).max()) / (csum * csum)
        if val < best_val:
            best_val = val
            best_t = t
    if best_t == 0.0:
        return None, M_cur
    lo = max(0.0, best_t - 0.08)
    hi = min(1.0, best_t + 0.08)
    for _ in range(30):
        if hi - lo < 1e-8:
            break
        mid = 0.5 * (lo + hi)
        cand = (1.0 - mid) * a_norm + mid * g_norm
        cand = np.maximum(cand, 0)
        csum = cand.sum()
        if csum < 1e-12:
            hi = mid
            continue
        val = float(np.convolve(cand, cand).max()) / (csum * csum)
        if val < best_val:
            best_val = val
            best_t = mid
            hi = mid
        else:
            lo = mid
    cand = (1.0 - best_t) * a_norm + best_t * g_norm
    cand = np.maximum(cand, 0)
    cand /= cand.sum()
    return cand, best_val


def _lp_polish(a0, time_budget, seed=42):
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, None).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a)
        s = 1.0
    a = a / s
    n = a.size
    best_a = a.copy()
    best_score = _fast_score(best_a)
    rng = np.random.default_rng(seed)
    stuck = 0
    t_start = time.monotonic()
    while time.monotonic() - t_start < time_budget - 2.0:
        conv = np.convolve(a, a)
        M = float(conv.max())
        k_star = int(conv.argmax())
        A = _build_A(n, a)
        made_move = False
        for M_f in [1.0 + 1e-8, 1.0 + 1e-6, 1.0 + 1e-4]:
            if time.monotonic() - t_start > time_budget - 3.0:
                break
            g = _lp_solve(A, M * M_f, n, tl=3.0)
            if g is None:
                continue
            gs = g.sum()
            if gs < 1e-12:
                continue
            g_norm = g / gs
            if float(np.max(np.abs(g_norm - a))) < 1e-10:
                continue
            cand, val = _line_search(a, g, M, n)
            if cand is not None and val < M * (1.0 - 1e-10):
                a = cand
                made_move = True
                break
        if not made_move:
            grad = np.zeros(n)
            lo = max(0, k_star - n + 1)
            hi = min(n, k_star + 1)
            for j in range(lo, hi):
                idx = k_star - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * a[idx]
            grad -= grad.mean()
            gn = float(np.linalg.norm(grad))
            if gn > 1e-14:
                for alpha in [0.03, 0.01, 0.005, 0.001]:
                    cand = a - alpha * grad / gn
                    cand = np.maximum(cand, 0)
                    cs = cand.sum()
                    if cs < 1e-12:
                        continue
                    cand /= cs
                    if float(np.convolve(cand, cand).max()) < M * (1.0 - 1e-10):
                        a = cand
                        made_move = True
                        break
        sc = _fast_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        if not made_move or stuck > 60:
            a *= np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            stuck = 0
    return best_a


def _generate_inits(n, parent, seed):
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    pn = parent / max(parent.sum(), 1e-15)
    inits = [pn.copy()]
    inits.append(np.ones(n) / n)
    for p in [0.3, 0.5, 0.8, 1.2, 2.0, 4.0]:
        v = np.maximum(1.0 - x / (n - 1), 1e-10) ** p
        inits.append(v / v.sum())
    for al in [0.5, 1.0, 2.0, 4.0, 8.0]:
        v = np.exp(-al * x / n)
        inits.append(v / v.sum())
        v2 = np.exp(-al * (n - 1 - x) / n)
        inits.append(v2 / v2.sum())
    for w in [0.05, 0.1, 0.15, 0.25, 0.4]:
        ctr = (n - 1) / 2.0
        v = np.exp(-0.5 * ((x - ctr) / (w * n / 2.0)) ** 2)
        inits.append(v / v.sum())
    for fr in [1, 2, 3]:
        v = np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * fr * x / (n - 1)), 0.01)
        inits.append(v / v.sum())
    for fr in [1, 2, 3, 4]:
        v = np.maximum(np.sin(np.pi * x / (n - 1)) + 0.1 * np.sin(2.0 * np.pi * fr * x / (n - 1)), 0.01)
        inits.append(v / v.sum())
    f1 = rng.uniform(1, 8)
    f2 = rng.uniform(1, 8)
    v = np.exp(0.2 * np.sin(2 * np.pi * f1 * x / n) + 0.15 * np.cos(2 * np.pi * f2 * x / n))
    inits.append(v / v.sum())
    for frac in [0.3, 0.5, 0.7]:
        for init in inits[1:4]:
            blended = frac * pn + (1.0 - frac) * init
            inits.append(blended / blended.sum())
    for _ in range(5):
        v = np.exp(rng.normal(0, 0.3, n))
        inits.append(v / v.sum())
    ends = np.zeros(n)
    ends[0] = 1.0
    ends[-1] = 1.0
    ends[n // 4] = 0.5
    ends[3 * n // 4] = 0.5
    ends += 0.01
    inits.append(ends / ends.sum())
    two_block = np.zeros(n)
    m = max(5, n // 8)
    two_block[:m] = 1.0
    two_block[-m:] = 1.0
    two_block += 0.01
    inits.append(two_block / two_block.sum())
    return inits
