    a = np.clip(a, 0, 1000)

    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)

    """Build (n x n_half) matrix for symmetric LP.

Let me be careful about the line search. Given a and g (both symmetric, sum=1), I want to find t in [0,1] that minimizes max_k (h*h)_k where h = (1-t)*a + t*g.

Precompute:
- A_k = (a*a)_k for k=0,...,n-1
- C_k = (g*g)_k for k=0,...,n-1
- D_k = (a*g)_k for k=0,...,n-1

Then (h*h)_k = (1-t)^2 * A_k + 2t(1-t) * D_k + t^2 * C_k

f(t) = max_k [(1-t)^2 * A_k + 2t(1-t) * D_k + t^2 * C_k]

This is the max of n quadratics, hence convex. I can minimize it with ternary search.

Actually, since f is convex, I can also find the minimum by checking where f'(t) changes sign. f'(t) = derivative of the active quadratic at t. But the active set changes, so it's not straightforward. Ternary search is simpler and reliable.

For ternary search on [0,1]:
- Evaluate f at 1/3 and 2/3
- Keep the half where f is smaller
- Repeat 40-50 times for precision ~ 2/3^50 ≈ 10^-9

Each evaluation is O(n). Total: O(50*n) ≈ O(14000) ops. Very fast.

            stuck += 1
            if stuck > max_stuck:
                sigma = min(0.03 + 0.02 * (pert_level % 5), 0.35)
                half = a[:n_half].copy()
                mode = pert_level % 3
                if mode == 0:
                    half *= np.exp(rng.normal(0, sigma, n_half))
                elif mode == 1:
                    xx = np.arange(n_half, dtype=np.float64)
                    freq = rng.uniform(0.5, 4.0)
                    half *= (1.0 + sigma * np.sin(2 * np.pi * freq * xx / n_half))
                    half = np.maximum(half, 1e-15)
                else:
                    half *= (1.0 + sigma * rng.uniform(-1, 1, n_half))
                    half = np.maximum(half, 1e-15)
                a = _expand_fast(half, n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                stuck = 0
                pert_level += 1
                if pert_level > 60:
                    break

    return best_a


def _optimize_nonsym(a_init, stop_time, seed=42, max_stuck=60):
    """Non-symmetric LP-based optimization (fallback/supplement)."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    best_a = a.copy()
    c = np.convolve(a, a)
    best_M = float(c.max())

    rng = np.random.default_rng(seed)
    stuck = 0

    while time.monotonic() < stop_time:
        c = np.convolve(a, a)
        M = float(c.max())

        if M < best_M * (1.0 - 1e-13):
            best_M = M
            best_a = a.copy()
            stuck = 0

        # Build full (2n-1) x n convolution matrix
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            lo_j = max(0, k - n + 1)
            hi_j = min(n, k + 1)
            for j in range(lo_j, hi_j):
                A[k, j] = a[k - j]

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * (1 + 1e-10))
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'time_limit': 4.0})
        except Exception:
            res = None

        made_move = False
        if res is not None and res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                g /= gs
                if np.max(np.abs(g - a)) > 1e-12:
                    # Line search (full)
                    aa_c = np.convolve(a, a)
                    gg_c = np.convolve(g, g)
                    ag_c = np.convolve(a, g)

                    def f(t):
                        h2 = (1 - t) ** 2 * aa_c + 2 * t * (1 - t) * ag_c + t ** 2 * gg_c
                        return float(h2.max())

                    lo, hi = 0.0, 1.0
                    for _ in range(40):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        if f(m1) < f(m2):
                            hi = m2
                        else:
                            lo = m1
                    t = (lo + hi) / 2.0
                    val = f(t)
                    if t > 1e-6 and val < M * (1 - 1e-13):
                        a = (1 - t) * a + t * g
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        made_move = True

        if not made_move:
            stuck += 1
            if stuck > max_stuck:
                sigma = min(0.04 * (1 + 0.3 * (stuck // max_stuck)), 0.4)
                a *= np.exp(rng.normal(0, sigma, n))
                a = np.maximum(a, 1e-15)
                a /= a.sum()
                stuck = 0
                if stuck > 200:
                    break

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 18.0

    # Get parent
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(281, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.size == 0 or not np.all(np.isfinite(parent)):
        parent = np.ones(281, dtype=np.float64)

    n = parent.size

    # Initial best: symmetrized parent
    a0 = (parent + parent[::-1]) / 2.0
    a0 = np.maximum(a0, 0.0)
    if a0.sum() < 1e-15:
        a0 = np.ones(n)
    a0 /= a0.sum()

    best_a = a0.copy()
    best_score = _score(a0)

    def try_update(cand):
        nonlocal best_a, best_score
        cand = np.asarray(cand, dtype=np.float64)
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 0.01:
            return
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_a = (cand / s).copy()

    # Phase 1: Symmetric LP optimization on symmetrized parent (250s)
    t1 = t0 + min(250.0, budget_s * 0.20)
    try:
        r = _optimize_sym(a0, t1, seed=1)
        try_update(r)
    except Exception:
        pass

    # Phase 2: Diverse symmetric initializations (250s)
    t2 = t0 + min(500.0, budget_s * 0.40)
    inits = []
    pn = best_a.copy()

    # Power taper
    x = np.arange(n, dtype=np.float64)
    xy = x / max(n - 1, 1)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        v = np.maximum(1.0 - xy, 1e-8) ** p
        v = (v + v[::-1]) / 2.0
        v = np.maximum(v, 1e-10)
        inits.append(v / v.sum())

    # Gaussian
    for w in [0.1, 0.15, 0.2, 0.3, 0.5]:
        v = np.exp(-0.5 * ((xy - 0.5) / w) ** 2)
        v = (v + v[::-1]) / 2.0
        v = np.maximum(v, 1e-10)
        inits.append(v / v.sum())

    # Cosine
    for fr in [1, 2, 3, 4, 5]:
        v = np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * fr * xy), 0.01)
        v = (v + v[::-1]) / 2.0
        v = np.maximum(v, 1e-10)
        inits.append(v / v.sum())

    # Blends with current best
    for frac in [0.3, 0.5, 0.7]:
        for init in inits[:5]:
            v = frac * pn + (1 - frac) * init
            v = np.maximum(v, 1e-10)
            v = (v + v[::-1]) / 2.0
            inits.append(v / v.sum())

    remaining = max(t2 - time.monotonic(), 0)
    per_init = max(4.0, remaining / max(1, len(inits)))

    for i, init in enumerate(inits):
        if time.monotonic() > t2 - 5.0:
            break
        alloc = min(time.monotonic() + per_init, t2 - 3.0)
        try:
            r = _optimize_sym(init, alloc, seed=500 + i)
            try_update(r)
        except Exception:
            pass

    # Phase 3: Deep polish on current best (symmetric) (200s)
    t3 = t0 + min(700.0, budget_s * 0.55)
    try:
        r = _optimize_sym(best_a, t3, seed=777, max_stuck=200)
        try_update(r)
    except Exception:
        pass

    # Phase 4: Try non-symmetric on best (150s)
    if time.monotonic() < deadline - 200:
        t4 = min(time.monotonic() + 150.0, deadline - 180)
        try:
            r = _optimize_nonsym(best_a, t4, seed=1234)
            try_update(r)
        except Exception:
            pass

    # Phase 5: Try nearby lengths (symmetric)
    if time.monotonic() < deadline - 100:
        for dn in [-20, -10, -5, 5, 10, 20, 30]:
            if time.monotonic() > deadline - 50:
                break
            nt = n + dn
            if nt < 20 or nt > 600:
                continue
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, best_a)
            init_new = (init_new + init_new[::-1]) / 2.0
            init_new = np.maximum(init_new, 1e-10)
            init_new /= init_new.sum()
            try:
                r = _optimize_sym(init_new, time.monotonic() + 25, seed=2000 + nt)
                try_update(r)
            except Exception:
                pass

    # Phase 6: Final deep polish (remaining time)
    if time.monotonic() < deadline - 5:
        try:
            r = _optimize_sym(best_a, deadline - 2, seed=9999, max_stuck=300)
            try_update(r)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        fs = float(evaluate_sequence([float(x) for x in best_a]))
        if not np.isfinite(fs):
            raise ValueError("invalid score")
    except Exception:
        # Fallback
        best_a = np.ones(32)
        try:
            evaluate_sequence([float(x) for x in best_a])
        except Exception:
            best_a = [1.0] * 32
            return [1.0] * 32

    result = [float(x) for x in best_a]
    return result
