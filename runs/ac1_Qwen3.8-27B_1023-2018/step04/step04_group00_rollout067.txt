    amp = 0.4 + 0.8 * float(rng.random())
    q = p * np.exp(side * amp * (2.0 * x - 1.0))

                p2 = _lse_lbfgs(best_arr, nbest, final_end - 3.0, consider=consider)
                if p2 is not None and p2.size == nbest and np.all(np.isfinite(p2)):
                    ss = float(p2.sum())
                    if ss > 0:
                        p2 /= ss
                        consider(p2)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.clip(np.where(np.isfinite(out), out, 0.0), 0.0, 1000.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        print(f"Final score: {final_sc:.10f} (n={len(ls)}, time={time.monotonic()-t0:.1f}s)", flush=True)
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
