def _subgrad_step(f, n, step, mu, momentum, rng):
    """One subgradient step."""
    b = _fft_conv(f)
    m = float(b.max())
    S = float(f.sum())
    
    # Subgradient of max(a*a) / S^2 w.r.t. a
    # d/da [max_k (a*a)_k] = 2 * (active components of a convolved with active set)
    eps = 1e-10 * max(m, 1e-20)
    active = np.flatnonzero(b >= m - eps)
    
    grad = np.zeros(n)
    for k in active:
        il = max(0, k - n + 1)
        ih = min(k, n - 1)
        if il <= ih:
            idx = k - np.arange(il, ih + 1)
            grad[il:ih+1] += 2.0 * a[idx]
    
    if len(active) > 0:
        grad /= len(active)
    
    # Gradient of the full objective: 2n * (S*grad_maxconv - 2*M) / S^3
    f_sub = 2.0 * n * (grad * S - 2.0 * m) / (S ** 3)
    f_sub -= np.mean(f_sub)  # Renormalize (since sum is fixed)
    
    norm_sg = np.linalg.norm(f_sub)
    if norm_sg < 1e-18:
        return f, 0.0
    
    momentum = mu * momentum - step * f_sub / norm_sg
    fn = np.clip(f + momentum, 0, None)
    s = fn.sum()
    if s < 0.01:
        fn = np.clip(f + 0.5 * momentum, 0, None)
        s = fn.sum()
    if s < 0.01:
        return f, step * 0.5
    fn = fn / s
    return fn, step

def _subgrad_optimize(f0, n, stop_time, seed):

    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)

def _subgrad_optimize(f0, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    n = len(f)
    f = f / max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = _score(f)
    step = 0.03
    mu = 0.8
    momentum = np.zeros(n)
    no_imp = 0
    max_no_imp = 500
    
    while time.monotonic() < stop_time:
        b = _fft_conv(f)
        M = float(b.max())
        S = float(f.sum())
        
        # Subgradient of max(a*a)
        eps = 1e-10 * max(M, 1e-20)
        active = np.flatnonzero(b >= M - eps)
        
        grad = np.zeros(n)
        for k in active:
            il = max(0, k - n + 1)
            ih = min(k, n - 1)
            if il <= ih:
                idx = k - np.arange(il, ih + 1)
                grad[il:ih+1] += 2.0 * f[idx]
        
        if len(active) > 0:
            grad /= len(active)
        
        # Subgradient of 2n * M / S^2
        f_sub = 2.0 * n * (grad * S - 2.0 * M) / (S ** 3)
        f_sub -= np.mean(f_sub)
        
        norm_sg = np.linalg.norm(f_sub)
        if norm_sg < 1e-18:
            break
        
        momentum = mu * momentum - step * f_sub / norm_sg
        fn = np.clip(f + momentum, 0, None)
        s = fn.sum()
        if s < 0.01:
            step *= 0.5
            momentum *= 0.5
            continue
        fn = fn / s
        sc = _score(fn)
        
        if sc < best_sc - 1e-14:
            f = fn
            best_f = f.copy()
            best_sc = sc
            step = min(0.2, step * 1.02)
            no_imp = 0
        else:
            step *= 0.93
            no_imp += 1
            if no_imp > max_no_imp:
                f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                momentum = np.zeros(n)
                step = 0.03
                no_imp = 0
                max_no_imp += 200
    
    return best_f, best_sc

def _coord_descent(f0, stop_time, seed, max_epochs=40):

    best_sc = _score(f)
    best_f = f.copy()
    
    for epoch in range(max_epochs):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            if orig < 1e-15:
                continue
            best_v = orig
            best_v_sc = best_sc
            for mult in [0.1, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.6, 2.0]:
                f[i] = max(0.0, orig * mult)

                if sc < best_v_sc - 1e-15:
                    best_v = f[i]

                if sc < best_v_sc - 1e-15:
                    best_v = 0.0

            f[i] = best_v
            if best_v_sc < best_sc - 1e-15:
                best_sc = best_v_sc
                best_f = f.copy()
                improved = True
        
        # Pair optimization
        n_pairs = min(200, n)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-15 and oj < 1e-15:
                continue

                for mj in [0.3, 0.6, 1.4]:

                    sc = _score(f)

                        best_f = f.copy()
                        improved = True
            f[i], f[j] = oi, oj
        
        if not improved:
            break
    
    return best_f, best_sc

            else:

                stall += 1
                if stall >= max_stall:
                    f = _normalize(f + rng.normal(0, 0.04, nn))
                    t_param = 0.3
                    stall = 0
                    max_stall += 100
        
        return best_f, best_sc
    
    def _subgrad_optimize(f0, stop_time, seed):
        rng = np.random.default_rng(seed)
        f = _normalize(f0.copy())
        nn = len(f)
        best_f = f.copy()
        best_sc = _score(f)
        step = 0.04
        mu = 0.75
        momentum = np.zeros(nn)
        no_imp = 0
        
        while time.monotonic() < stop_time:
            b = _fft_conv(f)
            M = float(b.max())
            S = float(f.sum())
            
            eps = 1e-10 * max(M, 1e-20)
            active = np.flatnonzero(b >= M - eps)
            
            grad = np.zeros(nn)
            for k in active:
                il = max(0, k - nn + 1)
                ih = min(k, nn - 1)
                if il <= ih:
                    idx = k - np.arange(il, ih + 1)
                    grad[il:ih+1] += 2.0 * f[idx]
            
            if len(active) > 0:
                grad /= len(active)
            
            f_sub = 2.0 * nn * (grad * S - 2.0 * M) / (S ** 3)
            f_sub -= np.mean(f_sub)
            
            norm_sg = np.linalg.norm(f_sub)
            if norm_sg < 1e-18:
                break
            
            momentum = mu * momentum - step * f_sub / norm_sg
