```python
import numpy as np
import time
import math
from scipy.optimize import linprog

def _conv(a):
    n = len(a)
    if n <= 600:
        return np.convolve(a, a, mode='full')
    from scipy.fft import rfft, irfft, next_fast_len
    L = next_fast_len(2*n-1)
    F = rfft(a, L)
    b = irfft(F*F, L)[:2*n-1]
    np.maximum(b, 0.0, out=b)
    return b

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    M = float(b.max())
    if not math.isfinite(M) or M < 0:
        return math.inf
    return 2.0 * n * M / (s*s)

def _lp_direction(f, M, timeout=5.0):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None."""
    n = len(f)
    # Build A matrix: A[k, j] = f[k-j] if 0 <= k-j < n
    # Shape: (2n-1, n)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    
    if res.status != 0:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0, None)
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs  # normalize to sum=1
```

((1-t)f + tg) * ((1-t)f + tg) at lag k
= (1-t)^2 (f*f)[k] + 2t(1-t) (f*g)[k] + t^2 (g*g)[k]

Wait, that's not right either. Let h = (1-t)f + tg. Then:
(h*h)[k] = sum_j h[j] h[k-j]
= sum_j [(1-t)f[j] + tg[j]] [(1-t)f[k-j] + tg[k-j]]
= (1-t)^2 sum_j f[j]f[k-j] + (1-t)t sum_j (f[j]g[k-j] + g[j]f[k-j]) + t^2 sum_j g[j]g[k-j]
= (1-t)^2 (f*f)[k] + t(1-t) [(f*g)[k] + (g*f)[k]] + t^2 (g*g)[k]

Now, (f*g)[k] = sum_j f[j] g[k-j] and (g*f)[k] = sum_j g[j] f[k-j].

In general, (f*g)[k] ≠ (g*f)[k]. They are reverses of each other: (f*g)[k] = (g*f)[2n-2-k].

So I need both f*g and g*f. Or I can compute (f*g)[k] + (g*f)[k] directly.

Actually, (g*f)[k] = sum_j g[j] f[k-j]. If I let m = k-j, then j = k-m:
(g*f)[k] = sum_m g[k-m] f[m] = sum_m f[m] g[k-m] = (f*g)[k].

Wait, that's the same! So (f*g) = (g*f). Convolution is commutative.

So (h*h)[k] = (1-t)^2 (f*f)[k] + 2t(1-t) (f*g)[k] + t^2 (g*g)[k]. ✓
