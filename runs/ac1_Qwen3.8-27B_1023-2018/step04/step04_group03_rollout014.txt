```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(416, dtype=np.float64)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    return p

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_conv_matrix(f, n):
    """Build the matrix A where (A @ g)[k] = (f * g)[k] for full convolution."""
    # A is (2n-1) x n
    # A[k, j] = f[k-j] if max(0,k-n+1) <= j <= min(n-1,k)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)  # exclusive
        for j in range(j_start, j_end):
            v = f[k - j]
            if v > 1e-20:
                rows.append(k)
                cols.append(j)
                vals.append(v)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def _solve_full_lp(f, M, timeout=5.0):
    n = len(f)
    A = _build_conv_matrix(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g
```
