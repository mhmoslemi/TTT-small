But wait - the LP is maximizing `Σg`, not minimizing the peak. The connection to the objective is through the Frank-Wolfe framework: we're minimizing a convex function, and at each step, we solve the linearization over the feasible set. The "feasible set" here is `{g ≥ 0 : (f*g)_k ≤ M for peak k}`, and the linearization of the objective at `f` in the direction of `g` is proportional to `Σg` (or rather, the decrease in the objective is related to the LP gap).

Actually, let me think about this more carefully. The Frank-Wolfe algorithm for minimizing `F(g) = max_k (g*g)_k` over the simplex `g ≥ 0, Σg = 1` would:
1. At point `f`, compute the linearization: `∇F(f) · g` where `∇F(f)` is a subgradient.
2. Solve `min_g ∇F(f) · g` over the feasible set.
3. Do a line search.

The subgradient of `F(g) = max_k (g*g)_k` at `f` is `(f*·)_k` for an active k (i.e., a k where `(f*f)_k = max`). So the subgradient is a vector whose i-th component is `Σ_k (f*f)'_k[i]` averaged over active k's, which is `(f*·)_k` for those k.

The FW step would solve: `min_g Σ_i grad_i * g_i` subject to `g ≥ 0, Σg = 1`. This is just putting all mass on the coordinate with the smallest `grad_i`. That's not very useful.

So the parent's LP approach is different from standard FW. import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(416, dtype=np.float64)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _solve_lp(f, n, b, peak_frac, timeout=5.0):
    """Solve LP to find direction g"""
    M = float(b.max())
    if M < 1e-18:
        return None
    threshold = M * (1.0 - peak_frac)
    peak_idx = np.where(b >= threshold)[0]
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(b)])
    
    # Build sparse matrix A where A[idx, i] = f[k - i] for constraint idx (lag k)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(peak_idx):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-20:
                rows.append(idx)
                cols.append(i)
                vals.append(v)
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    c_vec = -np.ones(n)
    b_ub = np.full(len(peak_idx), M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _line_search(f, g, n, fsize):
    """Exact line search using FFT linearity"""
    F = rfft(f, fsize)
    G = rfft(g, fsize)
    f_sum = f.sum()
    g_sum = g.sum()
    
    def eval_at_t(t):
        A = (1 - t) * F + t * G
        C = irfft(A * A, fsize)[:2 * n - 1]
        np.maximum(C, 0.0, out=C)
        s = (1 - t) * f_sum + t * g_sum
        if s < 0.01:
            return math.inf
        return 2.0 * n * float(C.max()) / (s * s)
    
    best_t = 0.0
    best_val = eval_at_t(0.0)
    
    # Coarse grid
    ts = np.linspace(0.02, 0.98, 60)
    for t in ts:
        v = eval_at_t(t)
        if v < best_val:
            best_val = v
            best_t = t
    
    # Refine
    if best_t > 0:
        # Find neighborhood
        grid_idx = int((best_t - 0.02) / (0.96 / 59))
        lo = ts[max(0, grid_idx - 3)]
        hi = ts[min(len(ts)-1, grid_idx + 3)]
        ts_fine = np.linspace(lo, hi, 30)
        for t in ts_fine:
            v = eval_at_t(t)
            if v < best_val:
                best_val = v
                best_t = t
    
    return best_t, best_val

    best_f = f.copy()

def _subgrad_optimize(f0, stop_time, seed):
    n = len(f0)
    fsize = next_fast_len(2 * n - 1)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-10, None)
    f = f / f.sum()
    
    best_f = f.copy()
    best_score = _score(f)
    
    lr = 0.3
    stall = 0
    max_stall = 150
    
    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        # Find active peaks
        threshold = M * 0.99
        peak_idx = np.where(b >= threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])
        
        # Subgradient: average of (f * ·)_k for active k
        grad = np.zeros(n)
        for k in peak_idx:
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                grad[i] += f[k - i]
        grad /= len(peak_idx)
        
        # Multiplicative weight update
        grad_max = grad.max()
        if grad_max < 1e-15:
            break
        exponent = -lr * grad / grad_max
        exponent -= exponent.max()
        weights = np.exp(exponent)
        f_new = (f + 1e-15) * weights
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        
        sc = _score(f_new)
        if sc < best_score - 1e-12:
            f = f_new
            best_f = f.copy()
            best_score = sc
            stall = 0
            lr = min(0.5, lr * 1.02)
        else:
            # Try damped
            for damp in (0.5, 0.3, 0.1):

                    f = f / f.sum()
                    stall = 0

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(416, dtype=np.float64)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _solve_lp(f, n, b, peak_frac, timeout=4.0):
    M = float(b.max())
    if M < 1e-18:
        return None
    threshold = M * (1.0 - peak_frac)
    peak_idx = np.where(b >= threshold)[0]
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(b)])

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(peak_idx):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-20:
                rows.append(idx)
                cols.append(i)
                vals.append(v)

    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    c_vec = -np.ones(n)
    b_ub = np.full(len(peak_idx), M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None

    if res.status not in (0, 3) or res.x is None:
        return None

    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _line_search_fast(f, g, n, fsize, f_sum):
    F = rfft(f, fsize)
    G = rfft(g, fsize)
    g_sum = g.sum()

    def eval_at_t(t):
        A = (1.0 - t) * F + t * G
        C = irfft(A * A, fsize)[:2 * n - 1]
        np.maximum(C, 0.0, out=C)
        s = (1.0 - t) * f_sum + t * g_sum
        if s < 0.01:
            return math.inf
        return 2.0 * n * float(C.max()) / (s * s)

    best_t = 0.0
    best_val = eval_at_t(0.0)

    n_coarse = 40
    ts = np.linspace(0.02, 0.98, n_coarse)
    vals = np.empty(n_coarse)
    for i, t in enumerate(ts):
        vals[i] = eval_at_t(t)
        if vals[i] < best_val:
            best_val = vals[i]
            best_t = t

    if best_t > 0.01:
        idx = int(np.argmin(vals))
        lo = ts[max(0, idx - 2)]
        hi = ts[min(n_coarse - 1, idx + 2)]
        ts_fine = np.linspace(lo, hi, 20)
        for t in ts_fine:
            v = eval_at_t(t)
            if v < best_val:
                best_val = v
                best_t = t

    return best_t, best_val


def _lp_optimize(f0, stop_time, seed, initial_peak_frac=0.08, min_peak_frac=0.015):
    n = len(f0)
    fsize = next_fast_len(2 * n - 1)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f_sum = f.sum()
    if f_sum < 0.01:
        return np.asarray(f0, dtype=np.float64), _score(np.asarray(f0, dtype=np.float64))

    best_f = f.copy()
    best_score = _score(f)

    stall = 0
    max_stall = 80
    peak_frac = initial_peak_frac

    while time.monotonic() < stop_time - 2.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _solve_lp(f, n, b, peak_frac, timeout=3.0)
        if g is None:
            stall += 5
            if stall > max_stall:
                noise = rng.normal(0, 0.06 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                fs = f.sum()
                if fs > 0.01:
                    f = f * (f_sum / fs)
                stall = 0
                max_stall = min(max_stall + 40, 350)
                peak_frac = min(0.15, peak_frac * 1.4)
            continue

        g = g * (f_sum / g.sum())
        dist = np.linalg.norm(g - f) / max(1.0, np.linalg.norm(f))
        if dist < 1e-10:
            stall += 2
            if stall > max_stall:
                break
            continue

        t_new, val_new = _line_search_fast(f, g, n, fsize, f_sum)

        accepted = False
        if t_new > 1e-6 and val_new < best_score - 1e-13:
            f = (1.0 - t_new) * f + t_new * g
            f = np.clip(f, 0, None)
            fs = f.sum()
            if fs < 0.01:
                f = best_f.copy()
            else:
                best_f = f.copy()
                best_score = val_new
                stall = 0
                peak_frac = max(min_peak_frac, peak_frac * 0.97)
                accepted = True

        if not accepted:
            stall += 1
            if stall > 30 and t_new > 0:
                t_small = max(0.05, t_new * 0.3)
                f_try = np.clip((1.0 - t_small) * f + t_small * g, 0, None)
                if f_try.sum() > 0.01:
                    sc = _score(f_try)
                    if sc < best_score - 1e-13:
                        f = f_try
                        best_f = f.copy()
                        best_score = sc
                        stall = 0
                        peak_frac = max(min_peak_frac, peak_frac * 0.96)
                        accepted = True

            if stall > max_stall:
                noise = rng.normal(0, 0.08 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                fs = f.sum()
                if fs > 0.01:
                    f = f * (f_sum / fs)
                stall = 0
                max_stall = min(max_stall + 40, 400)
                peak_frac = min(0.12, peak_frac * 1.3)

    return best_f, best_score


def _subgrad_optimize(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-10, None)
    f = f / f.sum()

    best_f = f.copy()
    best_score = _score(f)

    lr = 0.25
    stall = 0
    max_stall = 120

    while time.monotonic() < stop_time - 2.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        threshold = M * 0.99
        peak_idx = np.where(b >= threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])

        grad = np.zeros(n)
        for k in peak_idx:
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                grad[i] += f[k - i]
        grad /= len(peak_idx)

        grad_max = grad.max()
        if grad_max < 1e-15:
            break

        exponent = -lr * grad / grad_max
        exponent -= exponent.max()
        weights = np.exp(exponent)
        f_new = (f + 1e-15) * weights
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        sc = _score(f_new)
        if sc < best_score - 1e-12:
            f = f_new
            best_f = f.copy()
            best_score = sc
            stall = 0
            lr = min(0.4, lr * 1.03)
        else:
            improved = False
            for damp in (0.5, 0.25, 0.1):
                f_d = (1.0 - damp) * f + damp * f_new
                sc_d = _score(f_d)
                if sc_d < best_score - 1e-12:
                    f = f_d
                    best_f = f.copy()
                    best_score = sc_d
                    stall = 0
                    improved = True
                    break
            if not improved:
                stall += 1
                lr *= 0.96
                if stall > max_stall:
                    noise = rng.normal(0, 0.015, n)
                    f = np.clip(f + noise, 1e-10, None)
                    f = f / f.sum()
                    stall = 0
                    lr = 0.2
                    max_stall += 40
                    if max_stall > 400:
                        break

    return best_f, best_score


def _polish(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    best_sc = _score(f)
    no_imp = 0
    max_no_imp = 10

    mults_coarse = (0.3, 0.5, 0.7, 0.85, 1.15, 1.35, 1.7, 2.5, 0.0)
    mults_fine = (0.85, 0.92, 1.08, 1.15, 0.75, 1.25)

    while time.monotonic() < stop_time - 2.0 and no_imp < max_no_imp:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 2.0:
                return f.copy(), best_sc

            orig = f[i]
            if orig < 1e-14:
                f[i] = 0.05 * max(1.0, f.mean())
                sc = _score(f)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    improved = True
                else:
                    f[i] = 0.0
                continue

            best_local = best_sc
            best_val = orig

            for mlt in mults_coarse:
                f[i] = orig * mlt
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_val = f[i]

            if best_local < best_sc - 1e-14:
                f[i] = best_val
                for mlt in mults_fine:
                    f[i] = best_val * mlt
                    if f.sum() < 0.01:
                        f[i] = best_val
                        continue
                    sc = _score(f)
                    if sc < best_local - 1e-14:
                        best_local = sc
                        best_val = f[i]
                    else:
                        f[i] = best_val
                f[i] = best_val
                best_sc = best_local
                improved = True
            else:
                f[i] = orig

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return f.copy(), best_sc

    best_score = _score(best_seq)

    def _update_best(candidate):

        if candidate is None or len(candidate) == 0:

            return
        sc = _score(candidate)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq = candidate.copy()
            best_score = sc

    rng = np.random.default_rng(seed)
    i_arr = np.arange(n_p, dtype=np.float64)

    # Diverse initializations
    inits = [parent.copy()]

    sig = 0.08 * max(1.0, float(parent.mean()))
    inits.append(np.clip(parent + rng.normal(0, sig, n_p), 0, None))
    inits.append(np.clip(parent * (1.0 + 0.15 * rng.normal(0, 1, n_p)), 0, None))

    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n_p)))

    sup = np.ones(n_p) * 0.3
    for fr in (2, 3, 5, 7):
        ph = rng.uniform(0, 2 * math.pi)
        sup += 0.12 * np.cos(2.0 * math.pi * fr * i_arr / n_p + ph)
    inits.append(np.abs(sup) + 0.02)

    center = (n_p - 1) / 2.0
    for width in (n_p * 0.15, n_p * 0.25, n_p * 0.35):
        inits.append(np.exp(-0.5 * ((i_arr - center) / width) ** 2))

    for frac in (0.25, 0.35):
        c1 = n_p * frac
        c2 = n_p * (1.0 - frac)
        w = n_p * 0.1
        inits.append(np.exp(-0.5 * ((i_arr - c1) / w) ** 2) +
                     np.exp(-0.5 * ((i_arr - c2) / w) ** 2))

    center_dist = np.abs(i_arr - center) / center
    for pwr in (0.5, 1.5, 3.0):
        inits.append(np.maximum(0.0, 1.0 - center_dist) ** pwr + 0.01)

    # Phase 1: LP optimization on multiple inits (55% time)
    phase1_end = t0 + budget * 0.55
    time_per_init = max(8.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 5.0:
            break
        stop = min(phase1_end - 3.0, time.monotonic() + time_per_init)

            continue
        try:

                                    initial_peak_frac=0.10, min_peak_frac=0.02)
            _update_best(res)
        except Exception:
            pass

    # Second LP pass on best with tighter constraints
    if time.monotonic() < phase1_end - 8.0:
        stop = phase1_end - 1.0
        try:
            res, sc = _lp_optimize(best_seq, stop, seed + 777777,
                                    initial_peak_frac=0.05, min_peak_frac=0.01)
            _update_best(res)
        except Exception:

    # Phase 2: Subgradient (13% time)
    phase2_end = t0 + budget * 0.68
    if time.monotonic() < phase2_end - 6.0:
        stop = phase2_end - 1.0
        try:

            _update_best(res)
        except Exception:
            pass

    # Phase 3: Polish (17% time)
    phase3_end = t0 + budget * 0.85
    if time.monotonic() < phase3_end - 6.0:
        stop = phase3_end - 1.0
        try:
            res, sc = _polish(best_seq, stop, seed + 33333)
            _update_best(res)
        except Exception:
            pass

    # Phase 4: LP on polished (10% time)
    phase4_end = deadline
    if time.monotonic() < phase4_end - 8.0:
        stop = phase4_end - 1.0
        try:
            res, sc = _lp_optimize(best_seq, stop, seed + 444444,
                                    initial_peak_frac=0.04, min_peak_frac=0.01)
            _update_best(res)
        except Exception:
            pass

    # Final polish
    remaining = deadline - time.monotonic()
    if remaining > 8.0:
        try:
            res, sc = _polish(best_seq, deadline - 1.0, seed + 22222)
