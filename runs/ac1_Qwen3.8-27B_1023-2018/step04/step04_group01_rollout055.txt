    if n <= 2048:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _lp_direction(f, n, time_limit=10.0):
    """Find LP direction: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    b = _conv_full(f)
    M = float(b.max())
    if M < 1e-15:
        return None, M, b
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c_neg = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_neg, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": time_limit})
    except Exception:
        return None, M, b
    if res.status != 0 or res.x is None:
        return None, M, b
    g = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None, M, b
    return g / g_sum, M, b

def _lp_optimize(f0, n, stop_time, seed, max_iters=600):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum

    best_f = f.copy()
    best_score = _score(f)
    stall = 0
    max_stall = 60
    iteration = 0

    while time.monotonic() < stop_time - 5.0 and iteration < max_iters:
        g_dir, M, b = _lp_direction(f, n, time_limit=10.0)
        if g_dir is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f /= fs
            else:
                f = best_f.copy()
            iteration += 1
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-8:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f /= fs
            else:
                f = best_f.copy()
            iteration += 1
            continue

        # Line search
        f_curr_score = _score(f)
        best_t = 0.0
        best_sc = f_curr_score
        lo, hi = 0.0, 1.0
        for _ in range(22):
            mid = (lo + hi) / 2
            f_new = f + mid * (g_dir - f)
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 1e-10:
                hi = mid
                continue
            f_new = f_new / fs_new
            sc = _score(f_new)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_t = mid
                lo = mid
            else:
                hi = mid

        if best_t > 1e-5 and best_sc < best_score - 1e-14:
            f_new = np.clip(f + best_t * (g_dir - f), 0, None)
            f_new /= f_new.sum()
            f = f_new.copy()
            best_f = f_new.copy()
            best_score = best_sc
            stall = 0
        elif best_t > 1e-5 and best_sc < f_curr_score - 1e-14:
            f_new = np.clip(f + best_t * (g_dir - f), 0, None)
            f_new /= f_new.sum()
            f = f_new.copy()
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f /= fs
                else:
                    f = best_f.copy()
                stall = 0

        iteration += 1

    return best_f, best_score

def _grad_optimize(f0, n, stop_time, seed, max_iters=50000):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f_sum

    best_f = f.copy()
    best_score = _score(f)
    alpha = 0.05
    alpha_min = 1e-12
    tau = 100.0

    conv_len = 2 * n - 1

    for it in range(max_iters):
        if time.monotonic() > stop_time - 5.0:
            break

        b = _conv_full(f)
        S = float(f.sum())
        M = float(b.max())
        if M < 1e-18:
            break
        score = 2.0 * n * M / (S * S)

        if score < best_score - 1e-14:
            best_score = score
            best_f = f.copy()

        # Softmax for smoothed max
        b_shift = b - M
        tau_b = tau * b_shift
        tau_b -= tau_b.max()
        p = np.exp(tau_b)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(conv_len)
            p[np.argmax(b)] = 1.0
            p_sum = 1.0
        p /= p_sum

        # Gradient: grad[i] = 2 * sum_k p[k] * f[k-i] (where valid)
        # = 2 * (p * f_flipped)[i + n - 1]
        f_flipped = f[::-1]
        P = rfft(p, conv_len)
        Ff = rfft(f_flipped, conv_len)
        grad_conv = irfft(P * Ff, conv_len)[:conv_len]
        grad = 2.0 * grad_conv[n-1:2*n-1]

        # Score gradient
        score_grad = (2.0 * n / (S * S)) * (grad - 2.0 * M / S)

        # Update
        g_raw = f - alpha * score_grad
        # Project onto simplex {x >= 0, sum = 1}
        g_raw = np.maximum(g_raw, 0)
        g_sum = g_raw.sum()
        if g_sum < 1e-10:
            f = np.ones(n) / n
        else:
            f = g_raw / g_sum

        sc_new = _score(f)
        if sc_new < score - 1e-14:
            alpha = max(alpha_min, alpha * 0.99)
        else:
            alpha = max(alpha_min, alpha * 0.5)
            if alpha < alpha_min:
                f = np.clip(f + rng.normal(0, 0.005, n), 0, None)
                f /= f.sum()
                alpha = 0.01

        if it % 5000 == 4999:
            tau = max(20.0, tau * 0.8)
            f = np.clip(f + rng.normal(0, 0.01, n), 0, None)
            f /= f.sum()
            alpha = 0.02

    return best_f, best_score

def _coord_optimize(f0, stop_time, max_sweeps=200, seed=12345):
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, math.inf
    f = f / S
    S = 1.0
    conv = _conv_full(f)
    best_score = 2.0 * n * conv.max()
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 3.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break

            v = f[i]
            S_rest = S - v
            lo_k = i
            hi_k = i + n - 1

            # C0: max outside affected range
            C0 = -1e30
            if lo_k > 0:
                C0 = max(C0, float(conv[:lo_k].max()))
            if hi_k < 2 * n - 2:
                C0 = max(C0, float(conv[hi_k+1:].max()))

            conv_seg = conv[lo_k:hi_k+1]
            self_idx = i  # 2i - lo_k = 2i - i = i

            # Search range
            w_max = max(3.0 * v, 0.5) if v > 1e-6 else 2.0
            w_max = min(w_max, 50.0)

            # Coarse grid
            n_pts = 80
            w_grid = np.linspace(0.0, w_max, n_pts)

            S_new = np.maximum(S_rest + w_grid, 0.01)
            new_cv = conv_seg[None, :] + 2.0 * w_grid[:, None] * f[None, :]
            new_cv[:, self_idx] += w_grid**2
            new_max = np.maximum(C0, new_cv.max(axis=1))
            scores = 2.0 * n * new_max / (S_new * S_new)

            min_idx = int(np.argmin(scores))
            best_w = w_grid[min_idx]
            best_sc = float(scores[min_idx])

            # Fine grid
            if 0 < w_grid[min_idx] < w_max:
                dw = w_max / n_pts
                w_fine = np.linspace(max(0.0, best_w - 3*dw), min(w_max, best_w + 3*dw), 30)
                S_new_f = np.maximum(S_rest + w_fine, 0.01)
                new_cv_f = conv_seg[None, :] + 2.0 * w_fine[:, None] * f[None, :]
                new_cv_f[:, self_idx] += w_fine**2
                new_max_f = np.maximum(C0, new_cv_f.max(axis=1))
                scores_f = 2.0 * n * new_max_f / (S_new_f * S_new_f)
                min_idx_f = int(np.argmin(scores_f))
                if scores_f[min_idx_f] < best_sc:
                    best_w = w_fine[min_idx_f]
                    best_sc = float(scores_f[min_idx_f])

            # Accept
            dw_val = best_w - v
            if abs(dw_val) > 1e-14 and best_sc < best_score - 1e-15:
                conv[lo_k:hi_k+1] += 2.0 * dw_val * f
                conv[2*i] += dw_val * dw_val
                f[i] = best_w
                S += dw_val
                best_score = best_sc
                best_f = f.copy()
                improved = True

        if not improved and sweep > 8:
            f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
            S = f.sum()
            if S > 0.01:
                f /= S
                S = 1.0
                conv = _conv_full(f)
                best_score = 2.0 * n * conv.max()
                best_f = f.copy()

    return best_f, best_score
