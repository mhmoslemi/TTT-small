        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _fft_conv(a, fsize=None):
    n = len(a)
    if fsize is None:
        fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, M, timeout=3.0):
    n = len(f)
    A = _build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-14:
        return None
    return g


def _lp_optimize(f0, stop_time, seed, max_stall=800):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / fs
    n = len(f)
    fsize = next_fast_len(2 * n - 1)

    best_f = f.copy()
    best_sc = _score(f)
    t_param = 0.3
    stall = 0
    m_scale = 1.0

    for it in range(200000):
        if time.monotonic() > stop_time:
            break

        b = _fft_conv(f, fsize)
        M = float(b.max())
        if M < 1e-18:
            break

        # Try LP with current M
        g = _lp_solve(f, M * m_scale, timeout=2.0)

        if g is None:
            stall += 3
            if stall > 100:
                m_scale = max(0.995, m_scale - 0.002)
            if stall > max_stall:
                # Perturb
                f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                t_param = 0.3
                m_scale = 1.0
                stall = 0
            continue

        g = g / g.sum()

        # Check convergence
        dist = np.linalg.norm(g - f)
        if dist < 1e-11:
            stall += 5
            if stall > max_stall:
                break
            # Try slightly different M
            m_scale = 1.0 - rng.uniform(0, 0.005)
            continue

        # Line search with golden-ratio refinement
        best_t = t_param
        best_sc_try = best_sc

        # Coarse search
        candidates = [t_param, t_param * 0.5, t_param * 1.5,
                      0.05, 0.15, 0.3, 0.5, 0.7, 0.9, 1.0]
        candidates = sorted(set(max(0.01, min(1.0, c)) for c in candidates))
        sc_vals = {}
        for tt in candidates:
            fn = (1 - tt) * f + tt * g
            fn = np.clip(fn, 0, None)
            s = fn.sum()
            if s < 1e-10:
                continue
            fn = fn / s
            sc = _score(fn)
            sc_vals[tt] = sc
            if sc < best_sc_try:
                best_sc_try = sc
                best_t = tt

        # Fine search around best_t using golden section
        if best_sc_try < best_sc - 1e-14 and len(sc_vals) >= 3:
            lo = max(0.01, best_t - 0.1)
            hi = min(1.0, best_t + 0.1)
            gr = (math.sqrt(5) - 1) / 2
            for _ in range(8):
                m1 = hi - gr * (hi - lo)
                m2 = lo + gr * (hi - lo)
                fn1 = np.clip((1 - m1) * f + m1 * g, 0, None)
                fn2 = np.clip((1 - m2) * f + m2 * g, 0, None)
                s1, s2 = fn1.sum(), fn2.sum()
                if s1 < 1e-10 or s2 < 1e-10:
                    break
                fn1, fn2 = fn1 / s1, fn2 / s2
                sc1, sc2 = _score(fn1), _score(fn2)
                if sc1 < sc2:
                    hi = m2
                else:
                    lo = m1
                if sc1 < best_sc_try:
                    best_sc_try = sc1
                    best_t = m1
                if sc2 < best_sc_try:
                    best_sc_try = sc2
                    best_t = m2

        if best_sc_try < best_sc - 1e-14:
            fn = (1 - best_t) * f + best_t * g
            fn = np.clip(fn, 0, None)
            fn = fn / max(fn.sum(), 1e-10)
            f = fn
            best_f = f.copy()
            best_sc = best_sc_try
            t_param = min(0.95, max(0.05, best_t * 1.15))
            stall = 0
            if m_scale < 1.0:
                m_scale = min(1.0, m_scale + 0.001)
        else:
            t_param = max(0.02, t_param * 0.5)
            stall += 1
            m_scale = 1.0
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.12, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                t_param = 0.3
                stall = 0
                max_stall += 200

    return best_f, best_sc


def _subgrad_polish(f0, stop_time, seed, max_iters=50000):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / max(f.sum(), 1e-10)
    n = len(f)
    fsize = next_fast_len(2 * n - 1)

    best_f = f.copy()
    best_sc = _score(f)
    step = 0.02
    momentum = np.zeros(n)
    mu = 0.8
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break

        b = _fft_conv(f, fsize)
        M = float(b.max())
        eps = max(1e-14, 1e-10 * M)
        active = np.flatnonzero(b >= M - eps)

        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo <= i_hi:
                idx = k - np.arange(i_lo, i_hi + 1)
                grad[i_lo:i_hi + 1] += 2.0 * f[idx]
        if len(active) > 0:
            grad /= len(active)

        S = float(f.sum())
        # Subgradient of 2n*max_b/S^2 w.r.t. f (with S not fixed)
        # d/d f: 2n * (grad * S - 2*M * ones) / S^3
        obj_grad = 2.0 * n * (grad * S - 2.0 * M) / (S ** 3)
        obj_grad -= np.mean(obj_grad)  # remove mean for stability

        nrm = np.linalg.norm(obj_grad)
        if nrm < 1e-16:
            break

        momentum = mu * momentum - step * obj_grad / nrm
        fn = np.clip(f + momentum, 0, None)
        s = fn.sum()
        if s < 1e-10:
            step *= 0.5
            momentum *= 0.5
            continue
        fn = fn / s
        sc = _score(fn)

        if sc < best_sc - 1e-14:
            f = fn
            best_f = f.copy()
            best_sc = sc
            step = min(0.3, step * 1.04)
            no_imp = 0
        else:
            step *= 0.92
            no_imp += 1
            if no_imp > 300:
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                momentum = np.zeros(n)
                step = 0.02
                no_imp = 0

    return best_f, best_sc


def _coord_descent(f0, stop_time, seed, max_sweeps=100):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    n = len(f)
    fsize = next_fast_len(2 * n - 1)
    best_sc = _score(f)
    best_f = f.copy()

    mults = [0.0, 0.3, 0.5, 0.7, 1.3, 1.6, 2.0, 3.0]

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            if orig < 1e-15:
                continue
            best_v = orig
            best_v_sc = best_sc
            for m in mults:
                f[i] = orig * m
                s = f.sum()
                if s < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_v_sc - 1e-15:
                    best_v = f[i]
                    best_v_sc = sc
            f[i] = best_v
            if best_v_sc < best_sc - 1e-15:
                best_sc = best_v_sc
                best_f = f.copy()
                improved = True

        # Random pair updates
        n_pairs = min(300, 3 * n)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-15 and oj < 1e-15:
                continue
            for mi in [0.0, 0.5, 1.5, 3.0]:
                for mj in [0.0, 0.5, 1.5, 3.0]:
                    f[i] = oi * mi if oi > 1e-15 else 0.0
                    f[j] = oj * mj if oj > 1e-15 else 0.0
                    s = f.sum()
                    if s < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        best_f = f.copy()
                        improved = True
            f[i], f[j] = oi, oj

        if not improved:
            break

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    deadline = t0 + budget - 25.0

    parent = _get_parent()
    n = len(parent)

    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(64)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update(a, sc):
        nonlocal best_seq, best_score
        if sc < best_score - 1e-12:
            if np.all(np.isfinite(a)) and np.all(a >= -1e-10) and a.sum() >= 0.01:
                best_seq = np.clip(a, 0, 1000.0).copy()
                best_score = sc

    # Phase 1: Main LP optimization (45% of budget)
    phase1_end = t0 + budget * 0.45
    p_norm = parent / max(parent.sum(), 1e-10)
    try:
        res, sc = _lp_optimize(p_norm, phase1_end, seed)
        if math.isfinite(sc):
            _update(res, sc)
    except Exception:
        pass

    # Phase 2: LP from perturbed starts (15%)
    phase2_end = t0 + budget * 0.60
    rng_main = np.random.default_rng(seed + 500)

    # Perturbed parent
    if time.monotonic() < phase2_end:
        p2 = np.clip(p_norm + rng_main.normal(0, 0.1, n), 0, None)
        p2 = p2 / max(p2.sum(), 1e-10)
        stop2 = min(phase2_end, time.monotonic() + budget * 0.08)
        try:
            res2, sc2 = _lp_optimize(p2, stop2, seed + 600)
            if math.isfinite(sc2):
                _update(res2, sc2)
        except Exception:
            pass

    # Smoothed parent
    if time.monotonic() < phase2_end:
        from scipy.ndimage import gaussian_filter1d
        p3 = gaussian_filter1d(p_norm, sigma=3.0)
        p3 = np.clip(p3, 0, None)
        p3 = p3 / max(p3.sum(), 1e-10)
        stop3 = min(phase2_end, time.monotonic() + budget * 0.07)
        try:
            res3, sc3 = _lp_optimize(p3, stop3, seed + 700)
            if math.isfinite(sc3):
                _update(res3, sc3)
        except Exception:
            pass

    # Phase 3: Subgradient polish (12%)
    phase3_end = t0 + budget * 0.72
    if time.monotonic() < phase3_end:
        f_sg = best_seq / max(best_seq.sum(), 1e-10)
        try:
            res_sg, sc_sg = _subgrad_polish(f_sg, phase3_end, seed + 800)
            if math.isfinite(sc_sg):
                _update(res_sg, sc_sg)
        except Exception:
            pass

    # Phase 4: Coordinate descent (13%)
    phase4_end = t0 + budget * 0.85
    if time.monotonic() < phase4_end:
        try:
            res_cd, sc_cd = _coord_descent(best_seq / max(best_seq.sum(), 1e-10),
                                           phase4_end, seed + 900, max_sweeps=60)
            if math.isfinite(sc_cd):
                _update(res_cd, sc_cd)
        except Exception:
            pass

    # Phase 5: Length exploration (8%)
    phase5_end = deadline - 10
    if time.monotonic() < phase5_end:
        best_norm = best_seq / max(best_seq.sum(), 1e-10)
        alt_lengths = sorted(set([n - 8, n - 4, n - 2, n - 1, n + 1, n + 2, n + 4, n + 8]))
        alt_lengths = [L for L in alt_lengths if L >= 8 and L <= 2 * n]
        time_per_alt = min(30.0, (phase5_end - time.monotonic()) / max(1, len(alt_lengths)))

        for L in alt_lengths:
            if time.monotonic() > phase5_end - 5:
                break
            # Initialize by interpolation
            init_L = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n), best_norm)
            init_L = np.clip(init_L, 0, None)
            init_L = init_L / max(init_L.sum(), 1e-10)
            stop_L = min(phase5_end - 5, time.monotonic() + time_per_alt)
            try:
                rL, sL = _lp_optimize(init_L, stop_L, seed + 2000 + L)
                if math.isfinite(sL):
                    _update(rL, sL)
            except Exception:
                pass
            # Also try truncation from center
            if L < n:
                start = (n - L) // 2
                trunc = best_norm[start:start + L]
                trunc = trunc / max(trunc.sum(), 1e-10)
                stop_L2 = min(phase5_end - 5, time.monotonic() + time_per_alt * 0.5)
                try:
                    rT, sT = _lp_optimize(trunc, stop_L2, seed + 3000 + L)
                    if math.isfinite(sT):
                        _update(rT, sT)
                except Exception:
                    pass

    # Phase 6: Final coordinate descent polish
    if time.monotonic() < deadline - 5:
        try:
            res_fin, sc_fin = _coord_descent(best_seq / max(best_seq.sum(), 1e-10),
                                             deadline - 5, seed + 9000, max_sweeps=40)
            if math.isfinite(sc_fin):
                _update(res_fin, sc_fin)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
