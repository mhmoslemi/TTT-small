                     bounds=[(0, None)] * n, method='highs',
                     options={'time_limit': 3.0, 'presolve': True})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    g = np.clip(res.x, 0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _lp_optimize(a0, n, stop_time, seed, update_cb):
    """LP-based optimization."""
    rng = np.random.default_rng(seed)
    a = _project_simplex(a0.copy())
    best_a = a.copy()
    best_sc = _score(a)
    t_param = 0.1
    stall = 0
    max_stall = 300
    
    for it in range(10000):
        if time.monotonic() > stop_time:
            break
        
        g = _lp_direction(a, n, None)
        if g is None:
            stall += 10
            if stall > max_stall:
                break
            # Perturb
            a = _project_simplex(a + rng.normal(0, 0.02, n))
            continue
        
        g = g / g.sum()  # normalize to simplex
        dist = np.linalg.norm(g - a)
        if dist < 1e-12:
            stall += 5
            if stall > max_stall:
                break
            continue
        
        # Line search
        best_t = t_param
        best_s = best_sc
        for tt in [t_param, t_param*0.5, t_param*1.5, 0.02, 0.05, 0.1, 0.2, 0.4, 0.6, 0.8]:
            tt = min(1.0, max(0.005, tt))
            a_try = (1 - tt) * a + tt * g
            a_try = _project_simplex(a_try)
            s = _score(a_try)
            if s < best_s - 1e-15:
                best_t = tt
                best_s = s
        
        if best_s < best_sc - 1e-13:
            a = (1 - best_t) * a + best_t * g
            a = _project_simplex(a)
            best_a = a.copy()
            best_sc = _score(a)
            update_cb(a, best_sc)
            t_param = min(0.9, best_t * 1.1)
            stall = 0
        else:
            t_param = max(0.01, t_param * 0.6)
            stall += 1
            if stall >= max_stall:
                a = _project_simplex(a + rng.normal(0, 0.05, n))
                t_param = 0.1
                stall = 0
                max_stall += 100
    
    return best_a, best_sc

def _coord_descent(a0, stop_time, seed, epoch, update_cb):
    """Coordinate descent refinement."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for ep in range(epoch):
        if time.monotonic() > stop_time:
            break
        improved = False
        idx = np.arange(n)
        rng.shuffle(idx)
        for i in idx:
            if time.monotonic() > stop_time:
                break
            orig = a[i]
            best_v = orig
            best_s = best_sc
            for mult in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0, 1.1, 1.3, 1.5, 2.0, 3.0]:
                a[i] = orig * mult
                if a.sum() < 0.01:
                    a[i] = orig
                    continue
                s = _score(a)
                if s < best_s - 1e-15:
                    best_v = a[i]
                    best_s = s
            a[i] = best_v
            if best_s < best_sc - 1e-15:
                best_sc = best_s
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_sc
