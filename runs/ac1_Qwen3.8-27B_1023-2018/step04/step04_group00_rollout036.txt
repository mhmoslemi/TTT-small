                c *= inv * inv
                S = 1.0

    if coord_stop > time.monotonic() + 60.0:
        try:
            p2, _ = _coord_descent(best_arr, coord_stop, consider=consider, seed=int(seed + 7))
            if p2 is not None and p2.size == n_best and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p2 = p2 / float(p2.sum())
                consider(p2)
        except Exception:
            pass

    # Phase 3: Final LSE at low T + short coordinate
    rem = hard - 15.0 - time.monotonic()
    if rem > 40.0:
        lse2_stop = time.monotonic() + max(20.0, 0.4 * rem)
        try:
            p3 = _lse_optimize(best_arr, lse2_stop, consider=consider, fine_only=True)
            if p3 is not None and p3.size == n_best and np.all(np.isfinite(p3)) and p3.sum() > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)
        except Exception:
            pass

        coord2_stop = hard - 15.0
        if time.monotonic() < coord2_stop - 20.0:
            try:
                p4, _ = _coord_descent(best_arr, coord2_stop, consider=consider, seed=int(seed + 12345))
                if p4 is not None and p4.size == n_best and np.all(np.isfinite(p4)) and p4.sum() > 0:
                    p4 = p4 / float(p4.sum())
                    consider(p4)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
