    """Fast line search using precomputed convolutions.
    (1-t)f + tg has convolution = (1-t)^2 * ff + 2*t*(1-t) * fg + t^2 * gg
    """
    # Precompute: conv of (1-t)f + t*g = (1-t)^2*ff + 2t(1-t)*fg + t^2*gg
    # where ff = f*f, fg = f*g (which equals g*f), gg = g*g
    # Max over k of [(1-t)^2 * ff[k] + 2t(1-t)*fg[k] + t^2*gg[k]]
    
    best_t = 0.0
    best_val = float(conv_ff.max())
    
    # Try candidate t values
    candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    for t in candidates:
        if time.monotonic() > stop_time:
            break
        val = (1-t)**2 * conv_ff + 2*t*(1-t) * conv_fg + t**2 * conv_gg
        max_val = float(val.max())
        if max_val < best_val:
            best_val = max_val
            best_t = t
    
    # Golden section refinement
    if 0 < best_t < 1.0:
        lo = max(0.0, best_t - 0.1)
        hi = min(1.0, best_t + 0.1)
        gr = (1.0 + math.sqrt(5)) / 2.0
        for _ in range(12):
            if time.monotonic() > stop_time:
                break
            m1 = hi - (hi - lo) / gr
            m2 = lo + (hi - lo) / gr
            v1 = float(((1-m1)**2 * conv_ff + 2*m1*(1-m1) * conv_fg + m1**2 * conv_gg).max())
            v2 = float(((1-m2)**2 * conv_ff + 2*m2*(1-m2) * conv_fg + m2**2 * conv_gg).max())
            if v1 < v2:
                hi = m2
            else:
                lo = m1
            tm = (lo + hi) / 2.0
            vm = float(((1-tm)**2 * conv_ff + 2*tm*(1-tm) * conv_fg + tm**2 * conv_gg).max())
            if vm < best_val:
                best_val = vm
                best_t = tm
    
    return best_t, best_val

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _convolve(a, b):

    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy import sparse


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _convolve(a, b):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    G = rfft(b, fsize)
    c = irfft(F * G, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A_sparse(f, n, indices):
    rows = []
    cols = []
    vals = []
    for ri, k in enumerate(indices):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            f_idx = k - j
            if 0 <= f_idx < n:
                rows.append(ri)
                cols.append(j)
                vals.append(f[f_idx])
    return sparse.csr_matrix((vals, (rows, cols)), shape=(len(indices), n))


def _solve_lp(A, M, n, time_limit=5.0):
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": time_limit})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if g.sum() <= 1e-12:
        return None
    return g


def _fast_line_search(f, g, conv_ff, conv_fg, conv_gg, sum_f, sum_g, n, stop_time):
    best_t = 0.0
    best_val = 2.0 * n * float(conv_ff.max()) / (sum_f * sum_f)

    candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    for t in candidates:
        if time.monotonic() > stop_time:
            break
        s = (1 - t) * sum_f + t * sum_g
        if s < 0.01:
            continue
        val = (1 - t) ** 2 * conv_ff + 2 * t * (1 - t) * conv_fg + t ** 2 * conv_gg
        sc = 2.0 * n * float(val.max()) / (s * s)
        if sc < best_val:
            best_val = sc
            best_t = t

    if 0 < best_t < 1.0:
        lo = max(0.0, best_t - 0.15)
        hi = min(1.0, best_t + 0.15)
        gr = (1.0 + math.sqrt(5)) / 2.0
        for _ in range(14):
            if time.monotonic() > stop_time:
                break
            m1 = hi - (hi - lo) / gr
            m2 = lo + (hi - lo) / gr
            s1 = (1 - m1) * sum_f + m1 * sum_g
            v1 = (1 - m1) ** 2 * conv_ff + 2 * m1 * (1 - m1) * conv_fg + m1 ** 2 * conv_gg
            sc1 = 2.0 * n * float(v1.max()) / (s1 * s1) if s1 > 0.01 else math.inf
            s2 = (1 - m2) * sum_f + m2 * sum_g
            v2 = (1 - m2) ** 2 * conv_ff + 2 * m2 * (1 - m2) * conv_fg + m2 ** 2 * conv_gg
            sc2 = 2.0 * n * float(v2.max()) / (s2 * s2) if s2 > 0.01 else math.inf
            if sc1 < sc2:
                hi = m2
            else:
                lo = m1
            tm = (lo + hi) / 2.0
            sm = (1 - tm) * sum_f + tm * sum_g
            vm = (1 - tm) ** 2 * conv_ff + 2 * tm * (1 - tm) * conv_fg + tm ** 2 * conv_gg
            scm = 2.0 * n * float(vm.max()) / (sm * sm) if sm > 0.01 else math.inf
            if scm < best_val:
                best_val = scm
                best_t = tm

    return best_t, best_val


def _lp_optimize(f0, n, stop_time, seed):

    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n) / n
        f_sum = 1.0
    else:
        f = f / f_sum
        f_sum = 1.0

    best_f = f.copy()
    conv_f = _conv_full(f)
    best_M = float(conv_f.max())
    best_score = 2.0 * n * best_M / (f_sum * f_sum)
    M = best_M
    stall = 0

    iter_count = 0

            elif strategy == 1:

            elif strategy == 4:

            conv_f = _conv_full(f)

            stall = 0
            iter_count += 1
            continue

        conv_f = _conv_full(f)
        M = float(conv_f.max())

        # Select constraints near peak

        active_ks = np.where(conv_f >= M * frac)[0]

            sorted_idx = np.argsort(-conv_f)
            active_ks = sorted_idx[:80]
        elif len(active_ks) > 200:

        if g is None:
            stall += 4
            iter_count += 1
            continue

        dist = float(np.linalg.norm(g - f))
        if dist < 1e-8:
            stall += 3
            iter_count += 1
            continue

        # Precompute for fast line search
        sum_f = float(f.sum())
        sum_g = float(g.sum())
        conv_fg = _convolve(f, g)
        conv_gg = _conv_full(g)

        t_opt, sc_opt = _fast_line_search(f, g, conv_f, conv_fg, conv_gg, sum_f, sum_g, n, stop_time)

        if t_opt > 0 and sc_opt < best_score - 1e-13:
            f = (1.0 - t_opt) * f + t_opt * g
            f = np.clip(f, 0, None)
            fs = f.sum()
            if fs > 1e-10:

            conv_f = _conv_full(f)

                best_M = M

        else:
            # Try with less relaxed M
            if stall < 10:
                A2 = _build_A_sparse(f, n, active_ks.tolist())

                if g2 is not None:

                        conv_fg2 = _convolve(f, g2)
                        conv_gg2 = _conv_full(g2)

                            stall = 0

    return best_f, best_score


def _coordinate_polish(a, stop_time, seed=99):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel().copy(), 0, None)
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0

    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            best_val = orig
            best_sc_i = best_sc
            mults = (0.0, 0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0) if orig > 1e-10 else (0.0, 0.01, 0.05, 0.1, 0.5)
            for mlt in mults:
                best[i] = orig * mlt if orig > 1e-10 else mlt
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_sc_i - 1e-14:
                    best_sc_i = sc
                    best_val = best[i]
            best[i] = best_val
            if best_sc_i < best_sc - 1e-14:
                best_sc = best_sc_i
                improved = True

        if not improved:
            no_imp += 1
            if no_imp >= 3:
                break
        else:
            no_imp = 0

    return best, best_sc


def _pair_polish(a, stop_time, seed=123):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel().copy(), 0, None)
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0

    while time.monotonic() < stop_time - 2.0 and no_imp < 4:
        improved = False
        n_pairs = min(200, n * 2)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 2.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            total = oi + oj
            if total < 1e-12:
                continue
            for frac in (0.0, 0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9, 1.0):
                best[i] = frac * total
                best[j] = (1.0 - frac) * total
                sc = _score(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc
