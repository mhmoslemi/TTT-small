            f, sc = _lp_optimize(best_seq, n, p1 - 3, seed)
            if f is not None and np.all(np.isfinite(f)) and sc < best_score:
                best_seq = f
                best_score = sc
        except Exception:
            pass

    # Phase 2: Perturbed LP (14%)
    p2 = t0 + budget * 0.62
    if p2 - time.monotonic() > 15:
        rng2 = np.random.default_rng(seed + 31337)
        pert = best_seq + rng2.normal(0, 0.025 * max(1.0, best_seq.mean()), n)
        pert = np.clip(pert, 0, None)
        try:
            f, sc = _lp_optimize(pert, n, p2 - 3, seed + 4444)
            if f is not None and np.all(np.isfinite(f)) and sc < best_score:
                best_seq = f
                best_score = sc
        except Exception:
            pass

    # Phase 3: Polish (20%)
    p3 = t0 + budget * 0.82
    if p3 - time.monotonic() > 12:
        try:
            best_seq, best_score = _polish(best_seq, p3 - 3, seed + 555)
        except Exception:
            pass

    # Phase 4: Second LP warm-start (10%)
    p4 = t0 + budget * 0.93
    if p4 - time.monotonic() > 20:
        try:
            f, sc = _lp_optimize(best_seq, n, p4 - 3, seed + 99999)
            if f is not None and np.all(np.isfinite(f)) and sc < best_score:
                best_seq = f
                best_score = _score(f)
        except Exception:
            pass

    # Phase 5: Final polish
    p5 = deadline - 8.0
    if p5 - time.monotonic() > 8:
        try:
            best_seq, best_score = _polish(best_seq, p5 - 1, seed + 77777)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        return [1.0] * max(1, n)

    if final_sc < best_score:
        best_score = final_sc

    return [float(x) for x in best_seq]
