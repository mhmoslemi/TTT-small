import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.signal import correlate

def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c

def _score_fast(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return math.inf
    b = _fft_conv(a)
    M = b.max()
    return 2.0 * n * M / (S * S)

def _proj_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x) = target_sum, x >= 0}."""
    n = len(v)
    # First, project onto {x >= 0, sum(x) = target_sum}
    # Standard algorithm:
    if v.sum() <= 0:
        return np.full(n, target_sum / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _smooth_objective_grad(a, lam):
    """Compute smooth log-sum-exp objective and its gradient."""
    n = len(a)
    b = _fft_conv(a)
    # Numerically stable log-sum-exp
    b_max = b.max()
    exp_b = np.exp(lam * (b - b_max))
    log_sum = b_max * lam + np.log(exp_b.sum())
    obj = log_sum / lam
    w = exp_b / exp_b.sum()
    # Gradient
    v = correlate(w, a, mode='valid')
    grad = 2.0 * v
    return obj, grad, b

def _lp_direction(a, n, M, timeout=3.0):
    """Solve LP to find direction that maximizes sum without exceeding M."""
    # Build constraint matrix: (a * g)_k <= M
    # (a * g)_k = sum_j a_j g_{k-j} (with appropriate bounds)
    # This is a Toeplitz-like matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(lo + n - 1, 2 * n - 1)
        A[lo:hi, j] = a[:hi-lo]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    g = np.clip(res.x, 0, None)
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs

def _gradient_optimize(a0, stop_time, seed, lam_init=20, lam_max=200):
    """Optimize using projected gradient with smooth approximation."""
    rng = np.random.default_rng(seed)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 0, None)
    S = a.sum()
    if S < 1e-10:
        a = np.ones_like(a) / len(a)
        S = 1.0
    a = a / S  # Normalize to sum = 1
    n = len(a)
    
    best_a = a.copy()
    best_sc = _score_fast(a)
    eta = 0.01
    lam = lam_init
    no_imp = 0
    max_no_imp = 500
    
    while time.monotonic() < stop_time:
        obj, grad, b = _smooth_objective_grad(a, lam)
        M = b.max()
        
        # Projected gradient step
        a_try = a - eta * grad
        a_try = _proj_simplex(a_try, 1.0)
        
        sc = _score_fast(a_try)
        
        if sc < best_sc - 1e-12:
            a = a_try
            best_a = a.copy()
            best_sc = sc
            no_imp = 0
            eta = min(0.5, eta * 1.05)
            lam = min(lam_max, lam * 1.01)
        else:
            eta *= 0.7
            no_imp += 1
            if no_imp > max_no_imp:
                # Restart
                a = np.clip(a + rng.normal(0, 0.03, n), 0, None)
                a = _proj_simplex(a, 1.0)
                eta = 0.01
                lam = lam_init
                no_imp = 0
                max_no_imp = int(max_no_imp * 1.3)
    
    return best_a * n, best_sc  # Scale back (score is scale-invariant, but return consistent)

def _coord_descent(a0, stop_time, seed, max_sweeps=100):
    """Coordinate descent polish."""
    rng = np.random.default_rng(seed)
    a = np.clip(np.asarray(a0, dtype=np.float64).copy(), 0, None)
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a, math.inf
    best_sc = _score_fast(a)
    best_a = a.copy()
    
    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = a[i]
            best_v = orig
            best_v_sc = best_sc
            for mult in [0.1, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.6, 2.0, 0.0]:
                a[i] = orig * mult if orig > 0 else (0.01 if mult > 1 else 0.0)
                if a.sum() < 0.01:
                    a[i] = orig
                    continue
                sc = _score_fast(a)
                if sc < best_v_sc - 1e-15:
                    best_v = a[i]
                    best_v_sc = sc
            a[i] = best_v
            if best_v_sc < best_sc - 1e-14:
                best_sc = best_v_sc
                best_a = a.copy()
                improved = True
        if not improved:
            break
    
    return best_a, best_sc

import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.signal import correlate

def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    return irfft(F * F, fsize)[:2 * n - 1]

def _score_fast(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return math.inf
    b = _fft_conv(a)
    M = b.max()
    if M < 0:
        return math.inf
    return 2.0 * n * M / (S * S)

def _proj_simplex(v, target=1.0):
    n = len(v)
    if v.sum() < 1e-15:
        return np.full(n, target / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - target)
    if not cond.any():
        return np.full(n, target / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _build_A(a, n):
    """Build the constraint matrix for the LP. A[k, j] = a[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(lo + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A

def _lp_solve(a, n, M, timeout=2.0):
    A = _build_A(a, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs

    n = len(a)

def _proj_simplex(v, target=1.0):
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-15:
        return np.full(n, target / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - target)
    if not cond.any():
        return np.full(n, target / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _build_A(a, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(lo + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': timeout})

        return None

        return None
    g = np.clip(res.x, 0.0, None)

    return g / gs

    best_seq = parent.copy()
    best_score = parent_sc

    def _update(a, sc):
        nonlocal best_seq, best_score
        if math.isfinite(sc) and sc < best_score - 1e-13:
            a_c = np.clip(np.asarray(a, dtype=np.float64), 0, 1000.0)
            if a_c.size > 0 and np.all(np.isfinite(a_c)) and a_c.sum() >= 0.01:
                best_seq = a_c
                best_score = sc

    p_norm = parent / parent.sum()

    # Phase 1: LP optimization
    phase1_end = t0 + budget * 0.28
    if time.monotonic() < phase1_end:
        res, sc = _optimize_lp(p_norm, n, phase1_end, seed)
        _update(res, sc)

    # Phase 2: LP from perturbed init
    if time.monotonic() < t0 + budget * 0.38:
        rng_p = np.random.default_rng(seed + 1)
        p2 = np.clip(p_norm + rng_p.normal(0, 0.12, n), 0, None)
        p2 = p2 / max(p2.sum(), 1e-10)
        stop2 = min(t0 + budget * 0.38, phase1_end + 15)
        res2, sc2 = _optimize_lp(p2, n, stop2, seed + 100)
        _update(res2, sc2)

    # Phase 3: Adam gradient (simplex)
    phase3_end = t0 + budget * 0.52
    if time.monotonic() < phase3_end:
        a_cur = best_seq / max(best_seq.sum(), 1e-10)
        res3, sc3 = _optimize_grad(a_cur, n, phase3_end, seed + 200)
        _update(res3, sc3)

    # Phase 4: LP refinement from improved point
    phase4_end = t0 + budget * 0.68
    if time.monotonic() < phase4_end:
        a_cur = best_seq / max(best_seq.sum(), 1e-10)
        res4, sc4 = _optimize_lp(a_cur, n, phase4_end, seed + 300)
        _update(res4, sc4)

    # Phase 5: Adam gradient (softmax)
    phase5_end = t0 + budget * 0.78
    if time.monotonic() < phase5_end:
        a_cur = best_seq / max(best_seq.sum(), 1e-10)
        res5, sc5 = _optimize_grad_softmax(a_cur, n, phase5_end, seed + 400)
        _update(res5, sc5)

    # Phase 6: Coordinate descent polish
    phase6_end = deadline - 12
    if time.monotonic() < phase6_end:
        res6, sc6 = _coord_descent(best_seq, phase6_end, seed + 500, max_sweeps=400)
        _update(res6, sc6)

    # Phase 7: Brief LP from final CD result
    if time.monotonic() < deadline - 8:
        a_cur = best_seq / max(best_seq.sum(), 1e-10)
        res7, sc7 = _optimize_lp(a_cur, n, deadline - 8, seed + 600)
        _update(res7, sc7)

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            alt_sc = float(evaluate_sequence([float(x) for x in parent]))
            if math.isfinite(alt_sc):
                return [float(x) for x in parent]
        except Exception:
            pass
        return [1.0]

    return [float(x) for x in best_seq]
