```python
def _coord_descent(a, stop_time, max_rounds=3):
    n = len(a)
    target_sum = math.sqrt(2.0 * n)
    best = a.copy()
    best_sc = _score(best)
    
    for round_num in range(max_rounds):
        if time.monotonic() > stop_time - 5:
            break
        improved = False
        order = np.random.default_rng(round_num * 1000 + 42).permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time - 5:
                break
            
            # Compute C_k, p_k, q_k for coordinate i
            a_i = best[i]
            best[i] = 0.0
            b_base = _conv_full(best)  # C_k
            best[i] = a_i
            
            # p_k = 2*a[k-i] if 0<=k-i<n and k!=2i, else 0
            # q_k = 1 if k==2i and 0<=k-i<n, else 0
            p = np.zeros(2*n-1)
            q = np.zeros(2*n-1)
            if 0 <= 2*i < 2*n-1:
                q[2*i] = 1.0
            # For k where 0 <= k-i < n and k != 2i:
            # k-i = j, so k = i+j for j in [0, n-1], k != 2i means j != i
            for j in range(n):
                k = i + j
                if k < 2*n-1 and k != 2*i:
                    p[k] = 2.0 * best[j]  # Wait, this is a[k-i] = a[j]
            
            # Actually, a[k-i] where k-i = j means a[j]. And k = i+j.
            # So for j in [0, n-1]: k = i+j, and if k < 2n-1 and k != 2i: p[k] = 2*a[j]
            # But k != 2i means i+j != 2i means j != i.
            
            # Golden section search for x in [0, x_max]
            # f(x) = max_k (C_k + p_k*x + q_k*x^2)
            # We want to minimize f(x)
            
            def M_of_x(x):
                vals = b_base + p * x
                if q.any():
                    vals += q * x * x
                return vals.max()
            
            # Upper bound for x
            x_max = max(2.0 * a_i, 2.0 * best.mean()) if a_i > 0 else 2.0 * best.mean()
            
            # Golden section
            gr = (math.sqrt(5) - 1) / 2
            lo, hi = 0.0, x_max
            a_gs = hi - gr * (hi - lo)
            b_gs = lo + gr * (hi - lo)
            fa = M_of_x(a_gs)
            fb = M_of_x(b_gs)
            for _ in range(40):
                if fa < fb:
                    hi = b_gs
                    b_gs = a_gs
                    fb = fa
                    a_gs = hi - gr * (hi - lo)
                    fa = M_of_x(a_gs)
                else:
                    lo = a_gs
                    a_gs = b_gs
                    fa = fb
                    b_gs = lo + gr * (hi - lo)
                    fb = M_of_x(b_gs)
            
            x_opt = (lo + hi) / 2
            best[i] = x_opt
            
            # Renormalize
            s = best.sum()
            if s < 0.01:
                best[i] = a_i
                continue
            best = best / s * target_sum
            
            sc = _score(best)
            if sc < best_sc - 1e-14:
                best_sc = sc
                improved = True
        
        if not improved:
            break
    
    return best, best_sc
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_direction(f, n, M, timeout=5.0):
    if M <= 1e-18:
        return None
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-15)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"presolve": True, "time_limit": timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g

            if time.monotonic() > stop_time - 3:
                break
            
            a_i = best[i]
            k_lo = i
            k_hi = min(i + n - 1, n_lags - 1)
            
            # max_out: max of B outside [k_lo, k_hi]
            if k_lo > 0 or k_hi < n_lags - 1:
                max_out = 0.0
                if k_lo > 0:
                    max_out = max(max_out, float(B[:k_lo].max()))
                if k_hi < n_lags - 1:
                    max_out = max(max_out, float(B[k_hi+1:].max()))
            else:
                max_out = 0.0
            
            # Inside coefficients
            in_range = np.arange(k_lo, k_hi + 1)
            if 2 * i >= k_lo and 2 * i <= k_hi:
                mask = in_range != 2 * i
                in_indices = in_range[mask]
            else:
                in_indices = in_range
            
            if len(in_indices) > 0:
                base_in = B[in_indices] - 2.0 * a_i * best[in_indices - i]
                coeff_in = 2.0 * best[in_indices - i]
            else:
                base_in = np.array([])
                coeff_in = np.array([])
            
            has_quad = (2 * i <= n_lags - 1)
            if has_quad:
                quad_base = B[2 * i] - a_i * a_i
            
            # Golden section for x in [0, x_max]
            x_max = max(3.0 * a_i, 1.5 * best.mean()) if a_i > 0 else 3.0 * best.mean()
            
            def M_of_x(x):
                m = max_out
                if base_in.size > 0:
                    m = max(m, float(np.max(base_in + coeff_in * x)))
                if has_quad:
                    m = max(m, quad_base + x * x)
                return m
            
            lo, hi = 0.0, x_max
            a_gs = hi - gr * (hi - lo)
            b_gs = lo + gr * (hi - lo)
            fa = M_of_x(a_gs)
            fb = M_of_x(b_gs)
            for _ in range(40):
                if fa < fb:
                    hi = b_gs
                    b_gs = a_gs
                    fb = fa
                    a_gs = hi - gr * (hi - lo)
                    fa = M_of_x(a_gs)
                else:
                    lo = a_gs
                    a_gs = b_gs
                    fa = fb
                    b_gs = lo + gr * (hi - lo)
                    fb = M_of_x(b_gs)
            
            x_opt = (lo + hi) / 2.0
            
            # Update B incrementally
            if k_lo <= k_hi:
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        B[k] = quad_base + x_opt * x_opt
                    else:
                        B[k] += 2.0 * (x_opt - a_i) * best[k - i]
            best[i] = x_opt
            
            # Renormalize
            s_new = best.sum()
            if s_new < 0.01:
                best[i] = a_i
                # Recompute B
                B = _conv_full(best)
                continue
            
            scale = target_sum / s_new
            if abs(scale - 1.0) > 1e-12:
                best *= scale
                B *= scale * scale
            
            sc = 2.0 * n * B.max() / (target_sum * target_sum)
            if sc < best_sc - 1e-15:
                best_sc = sc
                improved = True
        
        if not improved:
            break
    
    return best, best_sc
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=4.0):
    if M <= 1e-18:
        return None
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-15)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"presolve": True, "time_limit": timeout})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g


def _coord_descent(a0, stop_time, max_rounds=15, seed=42):
    n = len(a0)
    target_sum = math.sqrt(2.0 * n)
    best = np.array(a0, dtype=np.float64).copy()
    s = best.sum()
    if s < 0.01:
        return a0, _score(a0)
    best = best / s * target_sum
    B = _conv_full(best)
    best_sc = 2.0 * n * B.max() / (target_sum * target_sum)

    rng = np.random.default_rng(seed)
    gr = (math.sqrt(5) - 1) / 2
    n_lags = 2 * n - 1

    for round_num in range(max_rounds):
        if time.monotonic() > stop_time - 5:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3:
                break

            a_i = best[i]
            k_lo = i
            k_hi = min(i + n - 1, n_lags - 1)

            max_out = 0.0
            if k_lo > 0:
                max_out = max(max_out, float(B[:k_lo].max()))
            if k_hi < n_lags - 1:
                max_out = max(max_out, float(B[k_hi + 1:].max()))

            in_range = np.arange(k_lo, k_hi + 1)
            if 2 * i >= k_lo and 2 * i <= k_hi:
                mask = in_range != 2 * i
                in_indices = in_range[mask]
            else:
                in_indices = in_range

            if len(in_indices) > 0:
                base_in = B[in_indices] - 2.0 * a_i * best[in_indices - i]
                coeff_in = 2.0 * best[in_indices - i]
            else:
                base_in = np.array([])
                coeff_in = np.array([])

            has_quad = (2 * i <= n_lags - 1)
            if has_quad:
                quad_base = B[2 * i] - a_i * a_i

            x_max = max(3.0 * a_i, 1.5 * best.mean()) if a_i > 0 else 3.0 * best.mean()

            def M_of_x(x):
                m = max_out
                if base_in.size > 0:
                    m = max(m, float(np.max(base_in + coeff_in * x)))
                if has_quad:
                    m = max(m, quad_base + x * x)
                return m

            lo, hi = 0.0, x_max
            a_gs = hi - gr * (hi - lo)
            b_gs = lo + gr * (hi - lo)
            fa = M_of_x(a_gs)
            fb = M_of_x(b_gs)
            for _ in range(45):
                if fa < fb:
                    hi = b_gs
                    b_gs = a_gs
                    fb = fa
                    a_gs = hi - gr * (hi - lo)
                    fa = M_of_x(a_gs)
                else:
                    lo = a_gs
                    a_gs = b_gs
                    fa = fb
                    b_gs = lo + gr * (hi - lo)
                    fb = M_of_x(b_gs)

            x_opt = (lo + hi) / 2.0

            if k_lo <= k_hi:
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        B[k] = quad_base + x_opt * x_opt
                    else:
                        B[k] += 2.0 * (x_opt - a_i) * best[k - i]
            best[i] = x_opt

            s_new = best.sum()
            if s_new < 0.01:
                best[i] = a_i
                B = _conv_full(best)
                continue

            scale = target_sum / s_new
            if abs(scale - 1.0) > 1e-12:
                best *= scale
                B *= scale * scale

            sc = 2.0 * n * B.max() / (target_sum * target_sum)
            if sc < best_sc - 1e-15:
                best_sc = sc
                improved = True

        if not improved:
            break

    return best, best_sc


def _lp_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n)
        f_sum = n

    target_sum = math.sqrt(2.0 * n)
    f = f / f_sum * target_sum

    best_f = f.copy()
    best_score = _score(f)

    t = 0.3
    t_min = 0.05
    t_max = 0.9
    stall = 0
    max_stall = 150
    conv_thresh = 1e-11
    lp_fail_count = 0

    while time.monotonic() < stop_time - 8.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, n, M)
        if g is None:
            lp_fail_count += 1
            stall += 5
            if stall > max_stall:
                f = best_f.copy()
                idx = rng.choice(n, size=min(20, n), replace=False)
                for k in idx:
                    f[k] = max(0.0, f[k] * rng.uniform(0.3, 2.5))
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                else:
                    f = best_f.copy()
                stall = 0
                max_stall += 80
                t = 0.3
            continue

        lp_fail_count = 0
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            stall += 2
            continue
        g = g * (target_sum / g_sum)

        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < conv_thresh:
            stall += 3
            if stall > max_stall:
                f = best_f.copy()
                idx = rng.choice(n, size=min(15, n), replace=False)
                for k in idx:
                    f[k] = max(0.0, f[k] * rng.uniform(0.4, 2.0))
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                stall = 0
                max_stall += 60
                t = 0.3
            continue

        improved = False
        candidates = sorted(
            {max(t_min, min(1.0, c)) for c in
             [t, t * 0.5, t * 0.75, t * 1.25, t * 1.5, 0.1, 0.25, 0.5, 0.7, 1.0]},
            reverse=True
        )
        for tc in candidates:
            f_new = (1.0 - tc) * f + tc * g
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 0.01:
                continue
            f_new = f_new / fs_new * target_sum
            sc = _score(f_new)
            if sc < best_score - 1e-13:
                f = f_new.copy()
                best_f = f.copy()
                best_score = sc
                improved = True
                t = min(t_max, tc * 1.2)
                stall = 0
                break

        if not improved:
            stall += 1
            t = max(t_min, t * 0.6)
            if stall >= max_stall:
                f = best_f.copy()
                strength = 0.06 + 0.02 * (stall // max_stall)
                f += rng.normal(0, strength * max(1.0, best_f.mean()), n)
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                else:
                    f = best_f.copy()
                stall = 0
                max_stall += 50
                t = 0.3

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(100)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_update(seq, sc):
        nonlocal best_seq, best_score
        if seq is None or len(seq) == 0:
            return
        if not np.all(np.isfinite(seq)):
            return
        seq = np.clip(seq, 0, 1000)
        if seq.sum() < 0.01:
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-14:
            best_seq = seq.copy()
            best_score = exact

    def _try_lp_and_cd(n, init_arr, stop, sd):
        nonlocal best_seq, best_score
        if time.monotonic() > stop - 5.0:
            return
        try:
            lp_res, lp_sc = _lp_optimize(init_arr, n, stop, sd)
        except Exception:
            lp_res = init_arr
            lp_sc = _score(init_arr)
        if lp_res is None or len(lp_res) == 0:
            return
        _try_update(lp_res, lp_sc)
        # Coordinate descent polish
        if time.monotonic() < stop - 8.0:
            try:
                cd_res, cd_sc = _coord_descent(lp_res, stop, max_rounds=12, seed=sd + 777)
            except Exception:
                cd_res = lp_res
                cd_sc = lp_sc
            if cd_res is not None and len(cd_res) == n:
                _try_update(cd_res, cd_sc)

    # Phase 1: LP+CD on parent length
    n = n_p
    phase1_end = t0 + budget * 0.45
    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    inits = []
    inits.append(parent.copy())

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n)))

    # Boundary-weighted
    inits.append(1.0 + 0.5 * np.cos(math.pi * i_arr / (n - 1)))

    # Multi-frequency
    sup = np.ones(n) * 0.3
    for fr in (1, 2, 3, 5, 7):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.12 * np.cos(2.0 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.01)

    # Parent + noise
    sig = 0.15 * max(1.0, float(np.mean(parent)))
    inits.append(np.abs(parent + rng_main.normal(0, sig, n)) + 0.01)

    # Gaussian bump
    center = (n - 1) / 2.0
    width = n / 4.0
    inits.append(np.exp(-0.5 * ((i_arr - center) / width) ** 2))

    # Two-lobe
    two = np.exp(-0.5 * ((i_arr - n * 0.3) / (n * 0.12)) ** 2) + \
          np.exp(-0.5 * ((i_arr - n * 0.7) / (n * 0.12)) ** 2)
    inits.append(two)

    n_inits = len(inits)
    time_per_init = max(15.0, (phase1_end - t0) / n_inits)

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        _try_lp_and_cd(n, init, stop, seed + 1001 * (idx + 1))

    # Phase 2: Try other lengths
    phase2_end = t0 + budget * 0.72
    other_lengths = [448, 400, 512, 384, 480, 320, 288, 576]
    time_per_len = max(10.0, (phase2_end - phase1_end) / max(1, len(other_lengths)))

    for L in other_lengths:
        if time.monotonic() > phase2_end - 10.0:
            break
        stop = min(phase2_end - 5.0, time.monotonic() + time_per_len)
        init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
        init_l = np.clip(init_l, 0, None)
        if init_l.sum() < 0.01:
            init_l = np.ones(L)
        _try_lp_and_cd(L, init_l, stop, seed + 50000 + L)

    # Phase 3: Second LP pass on best (warm-start)
    phase3_end = t0 + budget * 0.88
    if time.monotonic() < phase3_end - 30.0:
        n_best = len(best_seq)
        stop_p3 = min(phase3_end - 5.0, time.monotonic() + (phase3_end - max(t0, time.monotonic())))
        _try_lp_and_cd(n_best, best_seq.copy(), stop_p3, seed + 999999)

    # Phase 4: Aggressive coordinate descent polish
    polish_end = deadline - 12.0
    if time.monotonic() < polish_end - 15.0:
        try:
            b_seq, b_sc = _coord_descent(best_seq, polish_end, max_rounds=30, seed=4242)
            if b_seq is not None and len(b_seq) > 0:
                _try_update(b_seq, b_sc)
        except Exception:
            pass

    # Phase 5: Another CD with different seed
    if time.monotonic() < deadline - 15.0:
        try:
            b_seq2, b_sc2 = _coord_descent(best_seq, deadline - 10.0, max_rounds=20, seed=88888)
            if b_seq2 is not None and len(b_seq2) > 0:
                _try_update(b_seq2, b_sc2)
        except Exception:
            pass

    # Phase 6: Final short LP refinement
    if time.monotonic() < deadline - 25.0:
        n_best = len(best_seq)
        try:
            lp_res, lp_sc = _lp_optimize(best_seq, n_best, deadline - 15.0, seed + 777777)
            if lp_res is not None and len(lp_res) > 0:
                _try_update(lp_res, lp_sc)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
```
