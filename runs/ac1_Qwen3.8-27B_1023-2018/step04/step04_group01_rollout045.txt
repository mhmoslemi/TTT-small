    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)

    return c

    n = len(a)
    S = a.sum()

def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho_val = rho_idx[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(v - theta, 0)

def _lp_direction(f, n, time_limit=5.0):
    """Find LP direction: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    b = _conv(f)
    M = b.max()
    if M < 1e-15:
        return None, M, b
    
    # Build constraint matrix: A[k,j] = f[k-j] if 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        start = j
        end = min(j+n, 2*n-1)
        A[start:end, j] = f[:end-start]
    
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2*n-1, M)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)]*n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'mip_rel_gap': 0.1})
    except Exception:
        return None, M, b
    
    if res.status != 0 or res.x is None:
        return None, M, b
    
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None, M, b
    return g/g_sum, M, b
