                if len(new_lags) > 2 * n:
                    # Keep only top violations
                    new_arr = np.array(new_lags)
                    top_idx = np.argsort(fg[new_arr])[-n:]
                    new_lags = list(new_arr[top_idx])
                active_lags_list = new_lags
                b_ub = np.array([M * 1.0 for _ in active_lags_list])
                g = _solve_lp(f, active_lags_list, b_ub, n, timeout=4.0)
                if g is None:
                    break

        if g is None:
            stall += 4
            if stall > max_stall:
                noise = rng.normal(0, 0.05 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs
                t = 0.3
                stall = 0
                max_stall = min(max_stall + 15, 200)
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            stall += 2
            continue
        g = g / g_sum  # normalize to sum=1

        dist = np.linalg.norm(g - f)
        if dist < 1e-10:
            stall += 2
            if stall > max_stall:
                break
            continue

        # Line search
        improved = False
        t_candidates = [t, t * 0.5, t * 1.5, t * 0.25, t * 2.0,
                        0.05, 0.1, 0.15, 0.25, 0.4, 0.5, 0.65, 0.8]
        t_candidates = sorted(set(max(0.01, min(0.95, x)) for x in t_candidates),
                              key=lambda x: -abs(x - t))[:12]

        for t_try in t_candidates:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.clip(f_new, 0, None)
            fs_new = f_new.sum()
            if fs_new < 1e-10:
                continue
            f_new = f_new / fs_new
            sc = _score_fast(f_new)
            if sc < best_score - 1e-14:
                f = f_new
                best_f = f.copy()
                best_score = sc
                t = t_try
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            t = max(0.01, t * 0.75)
            if stall >= max_stall:
                # Adiabatic rescue
                rescued = False
                for ts in [0.01, 0.005, 0.002, 0.001]:
                    f_new = (1.0 - ts) * f + ts * g
                    f_new = np.clip(f_new, 0, None)
                    fs_new = f_new.sum()
                    if fs_new > 1e-10:
                        f_new = f_new / fs_new
                        sc = _score_fast(f_new)
                        if sc < best_score - 1e-14:
                            f = f_new
                            best_f = f.copy()
                            best_score = sc
                            t = ts
                            rescued = True
                            break
                if not rescued:
                    noise = rng.normal(0, 0.10 * max(1.0, f.mean()), n)
                    f = np.clip(f + noise, 0, None)
                    fs = f.sum()
                    if fs > 1e-10:
                        f = f / fs
                    t = 0.25
                    stall = 0
                    max_stall += 20
                    if max_stall > 400:
                        break

        total_iters += 1

    return best_f, best_score


def _optimize_subgrad_v2(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-12, None)
    f = f / f.sum()

    best_f = f.copy()
    best_score = _score_fast(f)

    lr = 0.3
    stall = 0
    max_stall = 60
    momentum = np.zeros(n)
    mu = 0.5

    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-20:
            break

        # Near-peak lags
        frac = 0.02
        thr = M * (1.0 - frac)
        peak_idx = np.where(b >= thr)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([int(np.argmax(b))])

        # Subgradient of max_k (f*f)(k) averaged over peaks
        grad = np.zeros(n)
        for k in peak_idx:
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                grad[i] += 2.0 * f[k - i]
        grad /= len(peak_idx)

        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        # Exponentiated gradient update with momentum
        exponent = -lr * grad / max(g_norm, 1e-8) + mu * momentum
        exponent -= exponent.max()
        weights = np.exp(exponent)
        f_new = f * weights
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        # Also try direct gradient step (projected onto simplex)
        g_step = lr * grad / max(g_norm, 1e-8)
        f_sg = f - g_step
        f_sg = np.clip(f_sg, 0, None)
        sg_sum = f_sg.sum()
        if sg_sum > 1e-10:
            f_sg = f_sg / sg_sum

        sc_exp = _score_fast(f_new)
        sc_sg = _score_fast(f_sg) if sg_sum > 1e-10 else math.inf

        if sc_exp < best_score - 1e-13:
            f = f_new
            best_f = f.copy()
            best_score = sc_exp
            momentum = mu * momentum + (1 - mu) * grad / max(g_norm, 1e-8)
            stall = 0
            lr *= 0.995
        elif sc_sg < best_score - 1e-13:
            f = f_sg
            best_f = f.copy()
            best_score = sc_sg
            momentum = mu * momentum + (1 - mu) * grad / max(g_norm, 1e-8)
            stall = 0
            lr *= 0.995
        else:
            # Try damped
            for damp in [0.3, 0.15, 0.05]:
                f_d = (1 - damp) * f + damp * f_new
                f_d = np.clip(f_d, 1e-15, None)
                f_d = f_d / f_d.sum()
                sc_d = _score_fast(f_d)
                if sc_d < best_score - 1e-13:
                    f = f_d
                    best_f = f.copy()
                    best_score = sc_d
                    stall = 0
                    break
            else:
                stall += 1
                lr *= 0.9
                if stall > max_stall:
                    f = best_f.copy()
                    noise = rng.normal(0, 0.005, n)
                    f = np.clip(f + noise, 1e-12, None)
                    f = f / f.sum()
                    stall = 0
                    lr = 0.15
                    max_stall += 20
                    if max_stall > 300:
                        break

    return best_f, best_score


def _polish_cd(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    best_sc = _score_fast(f)
    no_imp = 0
    max_no_imp = 15

    mults = (0.5, 0.7, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4, 1.7, 0.3, 0.0)

    while time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = f[i]
            if orig < 1e-14:
                continue

            best_local = best_sc
            best_val = orig

            for mlt in mults:
                f[i] = orig * mlt
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score_fast(f)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_val = f[i]

            if best_local < best_sc - 1e-15:
                f[i] = best_val
                # Fine search
                for mlt_fine in (0.95, 1.05, 0.98, 1.02, 0.92, 1.08):
                    trial = best_val * mlt_fine
                    f[i] = trial
                    if f.sum() < 0.01:
                        f[i] = best_val
                        continue
                    sc = _score_fast(f)
                    if sc < best_local - 1e-16:
                        best_local = sc
                        best_val = trial
                    else:
                        f[i] = best_val
                f[i] = best_val
                best_sc = best_local
                improved = True
            else:
                f[i] = best_val

        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

    return f.copy(), best_sc


def _try_pairwise(f, i, j):
    """Try to improve by adjusting pair (i, j) keeping sum(i,j) fixed."""
    s = f[i] + f[j]
    if s < 1e-14:
        return f.copy(), _score_fast(f)
    # f[i] + f[j] = s, try different splits
    base_sc = _score_fast(f)
    best_sc = base_sc
    best_ratio = f[i] / s if s > 0 else 0.5
    orig_i, orig_j = f[i], f[j]
    for ratio in np.linspace(0.01, 0.99, 19):
        f[i] = s * ratio
        f[j] = s * (1.0 - ratio)
        sc = _score_fast(f)
        if sc < best_sc:
            best_sc = sc
            best_ratio = ratio
    f[i] = s * best_ratio
    f[j] = s * (1.0 - best_ratio)
    return f.copy(), best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 25.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = _score_fast(best_seq)

    def _update_best(candidate):
        nonlocal best_seq, best_score
        if candidate is None or len(candidate) == 0:
            return
        candidate = np.clip(np.asarray(candidate, dtype=np.float64), 0, 1000)
        if not np.all(np.isfinite(candidate)):
            return
        if candidate.sum() < 0.01:
            return
        sc = _score_fast(candidate)
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_seq = candidate.copy()
            best_score = sc

    # Build diverse initializations
    rng = np.random.default_rng(seed)
    inits = [parent.copy()]

    # Reversed parent
    inits.append(parent[::-1].copy())

    # Symmetrized parent
    inits.append(0.5 * (parent + parent[::-1]).copy())

    # Noisy variants
    sig = 0.08 * max(1.0, float(parent.mean()))
    inits.append(np.clip(parent + rng.normal(0, sig, n_p), 0, None))
    inits.append(np.clip(parent * (1.0 + 0.12 * rng.normal(0, 1, n_p)), 0, None))

    # Fourier-based
    i_arr = np.arange(n_p, dtype=np.float64)
    fourier_init = np.ones(n_p)
    for fr in (1, 2, 3, 5, 7):
        ph = rng.uniform(0, 2 * math.pi)
        fourier_init += 0.15 * np.cos(2.0 * math.pi * fr * i_arr / n_p + ph)
    inits.append(np.abs(fourier_init) + 0.02)

    # Gaussian
    center = (n_p - 1) / 2.0
    inits.append(np.exp(-0.5 * ((i_arr - center) / (n_p * 0.25)) ** 2))
    inits.append(np.exp(-0.5 * ((i_arr - center) / (n_p * 0.4)) ** 2))

    # Two-bump
    two_c = n_p * 0.28
    two_w = n_p * 0.1
    inits.append(np.exp(-0.5 * ((i_arr - two_c) / two_w) ** 2) +
                 np.exp(-0.5 * ((i_arr - (n_p - two_c - 1)) / two_w) ** 2))

    # Linear ramp + noise
    ramp = np.linspace(0.3, 1.0, n_p) + 0.05 * rng.normal(0, 1, n_p)
    inits.append(np.clip(ramp, 0.01, None))

    # Sinusoid
    inits.append(0.5 * (1.0 + np.sin(2.0 * math.pi * i_arr / n_p + 0.3)))

    # Phase 1: LP optimization on multiple inits (55% time)
    phase1_end = t0 + budget * 0.55
    n_inits = len(inits)
    time_per_init = max(8.0, (phase1_end - t0) / n_inits)

    for idx, init in enumerate(inits):
        elapsed = time.monotonic() - t0
        if elapsed > phase1_end - 10.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        if time.monotonic() > stop - 2.0:
            continue
        try:
            res, sc = _optimize_lp_stage(init, stop, seed + 1000 * (idx + 1),
                                          pf_start=0.06, pf_end=0.02)
            _update_best(res)
        except Exception:
            pass

    # LP refinement pass on best (within phase 1)
    if time.monotonic() < phase1_end - 12.0:
        stop = phase1_end - 4.0
        try:
            res, sc = _optimize_lp_stage(best_seq, stop, seed + 777000,
                                          pf_start=0.04, pf_end=0.015)
            _update_best(res)
        except Exception:
            pass

    # Phase 2: Subgradient optimization (15% time)
    phase2_end = t0 + budget * 0.72
    if time.monotonic() < phase2_end - 10.0:
        stop = min(phase2_end - 4.0, time.monotonic() + (phase2_end - time.monotonic() - 4.0))
        try:
            res, sc = _optimize_subgrad_v2(best_seq, stop, seed + 888000)
            _update_best(res)
        except Exception:
            pass

    # Phase 3: Coordinate descent polishing (18% time)
    phase3_end = t0 + budget * 0.90
    if time.monotonic() < phase3_end - 10.0:
        stop = min(phase3_end - 4.0, time.monotonic() + (phase3_end - time.monotonic() - 4.0))
        try:
            res, sc = _polish_cd(best_seq, stop, seed + 999000)
            _update_best(res)
        except Exception:
            pass

    # Phase 4: Second LP pass after polish (5% time)
    phase4_end = deadline + 2.0
    if time.monotonic() < phase4_end - 10.0:
        stop = min(phase4_end - 3.0, time.monotonic() + 40.0)
        try:
            res, sc = _optimize_lp_stage(best_seq, stop, seed + 111000,
                                          pf_start=0.03, pf_end=0.01)
            _update_best(res)
        except Exception:
            pass

    # Phase 5: PD (coordinate) after second LP
    remaining = deadline - time.monotonic()
    if remaining > 15.0:
        try:
            res, sc = _polish_cd(best_seq, deadline - 3.0, seed + 222000)
            _update_best(res)
        except Exception:
            pass

    # Final: try another LP micro-pass
    remaining = deadline - time.monotonic()
    if remaining > 12.0:
        try:
            res, sc = _optimize_lp_stage(best_seq, deadline - 3.0, seed + 333000,
                                          pf_start=0.02, pf_end=0.01)
            _update_best(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    result_list = [float(x) for x in best_seq]
    try:
        final_sc = float(evaluate_sequence(result_list))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Fallback to parent
        try:
            parent_list = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(parent_list))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return parent_list

    return result_list
