import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p.astype(np.float64)

def _score(a):
    """Fast score computation using FFT."""
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    M = float(c.max())
    if not math.isfinite(M) or M < 0:
        # M should be nonnegative for nonnegative a, but handle edge case
        M = max(M, 0.0)
        if not math.isfinite(M):
            return math.inf
    return 2.0 * n * M / (S * S)

def _conv_self(a, fsize=None):
    """Self-convolution via FFT."""
    n = len(a)
    if fsize is None:
        fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _conv_cross(a, b, fsize=None):
    """Cross-convolution a*b via FFT."""
    n = len(a)
    if fsize is None:
        fsize = next_fast_len(2 * n - 1)
    Fa = rfft(a, fsize)
    Fb = rfft(b, fsize)
    c = irfft(Fa * Fb, fsize)[:2 * n - 1]
    return c

def _concise_score(a, fsize=None):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    c = _conv_self(a, fsize)
    M = float(c.max())
    if not math.isfinite(M):
        return math.inf
    return 2.0 * n * M / (S * S)

def _lp_direction(f, n, M, fsize=None):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3.0})
    except Exception:
        return None
    
    if res.status not in (0, 3):
        return None
    if res.x is None:
        return None
    
    g = np.maximum(res.x, 0.0)
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs  # Normalize to sum 1


def _lp_optimize(f0, n, stop_time, seed):
    """LP-based optimization."""
    rng = np.random.default_rng(seed)
    
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    S_orig = float(f.sum())
    if S_orig < 1e-10:
        f = np.ones(n, dtype=np.float64) / n
        S_orig = 1.0
    else:
        f = f / S_orig  # Normalize to sum 1
    
    fsize = next_fast_len(2 * n - 1)
    
    best_f = f.copy()
    best_sc = _concise_score(f, fsize)
    t_param = 0.3
    stall = 0
    max_stall = 100
    iter_count = 0
    
    for it in range(200000):
        if time.monotonic() > stop_time:
            break
        
        b = _conv_self(f, fsize)
        M = float(b.max())
        if M < 1e-20:
            break
        
        g = _lp_direction(f, n, M, fsize)
        if g is None:
            stall += 5
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                fs = f.sum()
                if fs < 1e-10:
                    f = np.ones(n) / n
                else:
                    f = f / fs
                stall = 0
                max_stall += 50
            continue
        
        dist = np.linalg.norm(g - f)
        if dist < 1e-12:
            stall += 1
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                stall = 0
                max_stall += 30
            continue
        
        # Line search: try multiple t values
        best_t = 0.0
        best_sc_try = best_sc
        
        t_list = [t_param, t_param * 0.5, t_param * 1.5,
                  0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9]
        
        for t in t_list:
            t = min(1.0, max(0.001, t))
            fn = (1.0 - t) * f + t * g
            fn = np.maximum(fn, 0.0)
            if fn.sum() < 0.01:
                continue
            sc = _concise_score(fn, fsize)
            if sc < best_sc_try:
                best_sc_try = sc
                best_t = t
        
        if best_sc_try < best_sc - 1e-14:
            f = (1.0 - best_t) * f + best_t * g
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-10:
                f = np.ones(n) / n
            else:
                f = f / fs
            best_f = f.copy()
            best_sc = _concise_score(f, fsize)
            t_param = min(0.5, max(0.05, best_t * 1.05))
            stall = 0
            iter_count += 1
        else:
            t_param = max(0.02, t_param * 0.6)
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                t_param = 0.3
                stall = 0
    
    return best_f * S_orig, best_sc


def _subgrad_optimize(f0, n, stop_time, seed):
    """Subgradient optimization with Nesterov acceleration."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    S_orig = float(f.sum())
    if S_orig < 1e-10:
        f = np.ones(n) / n
        S_orig = 1.0
    else:
        f = f / S_orig
    
    fsize = next_fast_len(2 * n - 1)
    best_f = f.copy()
    best_sc = _concise_score(f, fsize)
    
    step = 0.03
    y = f.copy()  # Nesterov momentum
    m = np.zeros(n)
    mu = 0.8
    no_imp = 0
    
    for it in range(100000):
        if time.monotonic() > stop_time:
            break
        
        # Compute subgradient of score
        S = float(f.sum())
        b = _conv_self(f, fsize)
        M = float(b.max())
        if M < 1e-20:
            break
        
        eps = max(1e-14, 1e-10 * M)
        active = np.flatnonzero(b >= M - eps)
        
        grad_M = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo <= i_hi:
                idx = k - np.arange(i_lo, i_hi + 1)
                grad_M[i_lo:i_hi + 1] += 2.0 * f[idx]
        if len(active) > 0:
            grad_M /= len(active)
        
        # Gradient of 2nM/S^2
        grad = 2.0 * n * grad_M / (S * S) - 4.0 * n * M / (S ** 3)
        
        # Normalize and apply
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Nesterov step
        yn = (1 + mu) * y - mu * f  # Look-ahead
        # Actually, let's use a simpler momentum approach
        m = mu * m - step * grad / gnorm
        fn = np.clip(f + m, 0, None)
        fn = fn / max(fn.sum(), 1e-10)
        
        sc = _concise_score(fn, fsize)
        
        if sc < best_sc - 1e-14:
            f = fn
            y = f.copy()
            best_f = f.copy()
            best_sc = sc
            step = min(0.2, step * 1.02)
            no_imp = 0
        else:
            step *= 0.95
            no_imp += 1
            if no_imp > 300:
                f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                m = np.zeros(n)
                step = 0.03
                no_imp = 0
                if no_imp > 5000:
                    break
    
    return best_f * S_orig, best_sc


def _coord_descent(f0, stop_time, seed, max_epochs=30):
    """Coordinate descent."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    n = len(f)
    fsize = next_fast_len(2 * n - 1)
    best_sc = _concise_score(f, fsize)
    best_f = f.copy()
    S_orig = float(f.sum())
    
    mults = [0.1, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.6, 2.0, 0.0]
    
    for epoch in range(max_epochs):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            if orig < 1e-15:
                continue
            
            best_v = orig
            best_v_sc = best_sc
            
            for mult in mults:
                f[i] = max(0.0, orig * mult)
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _concise_score(f, fsize)
                if sc < best_v_sc:
                    best_v = f[i]
                    best_v_sc = sc
            
            f[i] = best_v
            if best_v_sc < best_sc - 1e-15:
                best_sc = best_v_sc
                best_f = f.copy()

    return g / gs


def _lp_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    S_orig = float(f.sum())
    if S_orig < 1e-10:
        f = np.ones(n, dtype=np.float64) / n
        S_orig = 1.0
    else:
        f = f / S_orig
    fsize = next_fast_len(2 * n - 1)
    best_f = f.copy()
    best_sc = _score(f, fsize)
    t_param = 0.25
    stall = 0
    max_stall = 80
    A_buf = np.zeros((2 * n - 1, n), dtype=np.float64)

    for it in range(500000):
        if time.monotonic() > stop_time:
            break
        b = _conv_self(f, fsize)
        M = float(b.max())
        if M < 1e-20:
            break
        for j in range(n):
            A_buf[j:j + n, j] = f
        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c_vec, A_ub=A_buf, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 2.0})
        except Exception:
            stall += 3
            continue
        if res.status not in (0, 3) or res.x is None:
            stall += 3
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                stall = 0
                max_stall += 40
            continue
        g = np.maximum(res.x, 0.0)
        gs = float(g.sum())
        if gs < 1e-15:
            stall += 2
            continue
        g = g / gs
        dist = float(np.linalg.norm(g - f))
        if dist < 1e-12:
            stall += 1
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                stall = 0
                max_stall += 30
            continue
        gg = _conv_self(g, fsize)
        if float(gg.max()) >= M + 1e-15:
            stall += 1
            t_param = max(0.03, t_param * 0.7)
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                stall = 0
                max_stall += 20
            continue
        best_t = 0.0
        best_sc_try = best_sc
        t_list = [t_param, t_param * 0.4, t_param * 0.7, t_param * 1.3,
                  0.05, 0.1, 0.2, 0.35, 0.5, 0.65, 0.8, 0.95]
        for t in t_list:
            t = min(1.0, max(0.001, t))
            fn = (1.0 - t) * f + t * g
            fn = np.maximum(fn, 0.0)
            fs = float(fn.sum())
            if fs < 0.01:
                continue
            sc = _score(fn, fsize)
            if sc < best_sc_try:
                best_sc_try = sc
                best_t = t
        if best_sc_try < best_sc - 1e-14:
            f = (1.0 - best_t) * f + best_t * g
            f = np.maximum(f, 0.0)
            f = f / max(f.sum(), 1e-10)
            best_f = f.copy()
            best_sc = best_sc_try
            t_param = min(0.45, max(0.04, best_t * 1.08))
            stall = 0
        else:
            t_param = max(0.02, t_param * 0.55)
            stall += 1
            if stall >= max_stall:
                f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                t_param = 0.25
                stall = 0
                max_stall += 60
    return best_f * S_orig, best_sc


def _subgrad_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    S_orig = float(f.sum())
    if S_orig < 1e-10:
        f = np.ones(n) / n
        S_orig = 1.0
    else:
        f = f / S_orig
    fsize = next_fast_len(2 * n - 1)
    best_f = f.copy()
    best_sc = _score(f, fsize)
    step = 0.04
    m = np.zeros(n)
    mu = 0.7
    no_imp = 0
    t_check = time.monotonic()

    for it in range(200000):
        if time.monotonic() > stop_time:
            break
        S = float(f.sum())
        b = _conv_self(f, fsize)
        M = float(b.max())
        if M < 1e-20:
            break
        eps = max(1e-14, 1e-10 * M)
        active = np.flatnonzero(b >= M - eps)
        ga = _conv_self((2.0 / len(active)) * f, fsize)
        grp = 2.0 * n * ga / (S * S) - 4.0 * n * M / (S ** 3)
        gn = float(np.linalg.norm(grp))
        if gn < 1e-18:
            break
        m = mu * m - step * grp / gn
        fn = np.clip(f + m, 0, None)
        fs = float(fn.sum())
        if fs < 0.01:
            step *= 0.5
            m *= 0.5
            continue
        fn = fn / fs
        sc = _score(fn, fsize)
        if sc < best_sc - 1e-14:
            f = fn
            best_f = f.copy()
            best_sc = sc
            step = min(0.3, step * 1.03)
            no_imp = 0
        else:
            step *= 0.94
            no_imp += 1
            if no_imp > 200:
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                m = np.zeros(n)
                step = 0.04
                no_imp = 0
    return best_f * S_orig, best_sc


def _coord_descent(f0, stop_time, seed, max_epochs=40):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    n = len(f)
    fsize = next_fast_len(2 * n - 1)
    best_sc = _score(f, fsize)
    best_f = f.copy()
    mults = [0.05, 0.2, 0.4, 0.6, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0, 0.0]

    for epoch in range(max_epochs):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            if orig < 1e-15:
                continue
            best_v = orig
            best_v_sc = best_sc
            for mult in mults:
                f[i] = max(0.0, orig * mult)
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f, fsize)
                if sc < best_v_sc:
                    best_v = f[i]
                    best_v_sc = sc
            f[i] = best_v
            if best_v_sc < best_sc - 1e-15:
                best_sc = best_v_sc
                best_f = f.copy()
                improved = True
        if not improved:
            break
    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 22.0

    parent = _get_parent()
    n = len(parent)

    fallback = np.clip(parent, 0, 1000.0).copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(64.0)
        fallback_sc = _score(fallback)

    best_f = fallback.copy()
    best_sc = fallback_sc

    def _upd(a, sc):
        nonlocal best_f, best_sc
        if math.isfinite(sc) and sc < best_sc - 1e-13:
            try:
                ac = np.clip(np.asarray(a, dtype=np.float64), 0, 1000.0)
            except Exception:
                return
            if ac.size == 0:
                return
            if ac.sum() >= 0.01 and np.all(np.isfinite(ac)):
                best_f = ac.copy()
                best_sc = sc

    # Phase 1: Primary LP optimization
    t1 = t0 + (deadline - t0) * 0.50
    p_norm = parent / max(parent.sum(), 1e-10)
    try:
        r1, s1 = _lp_optimize(p_norm, n, t1, seed)
        _upd(r1, s1)
    except Exception:
        pass

    # Phase 1b: LP with perturbed initialization
    t2 = t0 + (deadline - t0) * 0.65
    if time.monotonic() < t2:
        rng_p = np.random.default_rng(seed + 7)
        p2 = np.clip(best_f + rng_p.normal(0, 0.03 * max(float(best_f.max()), 0.01),
                                            len(best_f)), 0, None)
        p2s = float(p2.sum())
        if p2s > 0.01:
            p2 = p2 / p2s
            try:
                r2, s2 = _lp_optimize(p2, len(best_f), t2, seed + 107)
                _upd(r2 * p2s, s2)
            except Exception:
                pass

    # Phase 1c: Second perturbed LP if time permits
    t2b = t0 + (deadline - t0) * 0.74
    if time.monotonic() < t2b:
        rng_p2 = np.random.default_rng(seed + 23)
        p3 = np.clip(best_f * (1.0 + rng_p2.uniform(-0.15, 0.15, len(best_f))), 0, None)
        p3s = float(p3.sum())
        if p3s > 0.01:
            p3 = p3 / p3s
            try:
                r3, s3 = _lp_optimize(p3, len(best_f), t2b, seed + 233)
                _upd(r3 * p3s, s3)
            except Exception:
                pass

    # Phase 2: Subgradient refinement
    t3 = t0 + (deadline - t0) * 0.82
    if time.monotonic() < t3:
        try:
            _upd(_subgrad_optimize(best_f / max(best_f.sum(), 1e-10),
                                   len(best_f), t3, seed + 500))
        except Exception:
            pass

    # Phase 3: Coordinate descent
    t4 = t0 + (deadline - t0) * 0.92
    if time.monotonic() < t4:
        try:
            rc, sc = _coord_descent(best_f, t4, seed + 777, max_epochs=50)
            _upd(rc, sc)
        except Exception:
            pass

    # Phase 4: Explore nearby lengths
    t5 = deadline - 18
    if time.monotonic() < t5:
        cn = len(best_f)
        alts = sorted(set([cn - 2, cn - 1, cn + 1, cn + 2, cn + 4, cn + 8]))
        alts = [L for L in alts if 4 <= L <= 3000]
        per_alt = min(12.0, max(5.0, (t5 - time.monotonic()) / max(1, len(alts))))
        for L in alts:
            if time.monotonic() > t5 - 5:
                break
            try:
                il = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, cn),
                               best_f.astype(np.float64))
                sl = min(t5, time.monotonic() + per_alt)
                rL, scL = _lp_optimize(il, L, sl, seed + 5000 + L)
                _upd(rL, scL)
            except Exception:
                pass

    # Phase 5: Final coordinate polish
    if time.monotonic() < deadline - 5:
        try:
            r5, s5 = _coord_descent(best_f, deadline - 5, seed + 9999, max_epochs=30)
            _upd(r5, s5)
        except Exception:
            pass

    # Final validation with exact evaluator
    best_seq = np.clip(best_f, 0, 1000.0)
    seq_list = [float(x) for x in best_seq]
    try:
        fsc = float(evaluate_sequence(seq_list))
    except Exception:
        fsc = math.inf

    if not math.isfinite(fsc) or fsc <= 0:
        fb_list = [float(x) for x in fallback]
        try:
            fsc2 = float(evaluate_sequence(fb_list))
        except Exception:
            fsc2 = math.inf
        if math.isfinite(fsc2) and fsc2 > 0:
            return fb_list
        return [1.0]

    return seq_list
