            if time.monotonic() > stop_time - 6.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            if oi < 1e-15 and oj < 1e-15:
                continue
            for mi in (0.0, 0.25, 0.5, 0.7, 1.3, 1.6):
                for mj in (0.0, 0.25, 0.5, 0.7, 1.3, 1.6):
                    if mi == 1.0 and mj == 1.0:
                        continue
                    ni = oi * mi if (oi > 1e-15 and mi > 0) else 0.0
                    nj = oj * mj if (oj > 1e-15 and mj > 0) else 0.0
                    best[i], best[j] = ni, nj
                    s_val = float(best.sum())
                    if s_val < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score_fast(best)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        # Random single-coordinate perturbations
        if not improved:
            for _ in range(150):
                if time.monotonic() > stop_time - 6.0:
                    break
                k = int(rng.integers(n))
                old = best[k]
                sigma = 0.04 * max(1.0, best.mean())
                best[k] = max(0.0, old + rng.normal(0, sigma))
                s_val = float(best.sum())
                if s_val < 0.01:
                    best[k] = old
                    continue
                sc = _score_fast(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                else:
                    best[k] = old

        # Block perturbation
        if not improved and round_num > 2:
            for _ in range(40):
                if time.monotonic() > stop_time - 6.0:
                    break
                k0 = int(rng.integers(n - 5))
                blen = int(rng.integers(3, 9))
                old_block = best[k0:k0 + blen].copy()
                sigma = 0.08 * max(1.0, best.mean())
                best[k0:k0 + blen] = np.maximum(0, old_block + rng.normal(0, sigma, blen))
                s_val = float(best.sum())
                if s_val < 0.01:
                    best[k0:k0 + blen] = old_block
                    continue
                sc = _score_fast(best)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    improved = True
                else:
                    best[k0:k0 + blen] = old_block

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    fallback = parent.copy()
    fallback_sc = _score_fast(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(100)
        fallback_sc = _score_fast(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_optimize(n, init_arr, stop, sd):
        nonlocal best_seq, best_score
        if time.monotonic() > stop - 4.0:
            return
        try:
            res, sc = _lp_optimize(init_arr, n, stop, sd)
        except Exception:
            return
        if res is None or len(res) == 0:
            return
        if not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0, 1000)
        if res.sum() < 0.01:
            return
        if sc < best_score - 1e-12:
            try:
                exact = float(evaluate_sequence([float(x) for x in res]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-13:
                best_seq = res.copy()
                best_score = exact

    # Phase 1: LP optimization on parent length (55% of budget)
    n = n_p
    phase1_end = t0 + budget * 0.55
    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    inits = []
    inits.append(parent.copy())

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n)))

    # |sin|
    inits.append(np.abs(np.sin(math.pi * i_arr / (n - 1))) + 0.01)

    # Multi-frequency superposition
    sup = np.ones(n) * 0.3
    for fr in (1, 2, 3, 5, 7, 11, 13, 17):
        ph = rng_main.uniform(0, 2 * math.pi)
        sup += 0.1 * np.cos(2.0 * math.pi * fr * i_arr / n + ph)
    inits.append(np.abs(sup) + 0.01)

    # |Normal|
    inits.append(np.abs(rng_main.normal(0, 1, n)) + 0.02)

    # Parent + noise
    sig = 0.2 * max(1.0, float(np.mean(parent)))
    inits.append(np.abs(parent + rng_main.normal(0, sig, n)) + 0.01)

    # Gaussian
    center = (n - 1) / 2.0
    width = n / 4.0
    inits.append(np.exp(-0.5 * ((i_arr - center) / width) ** 2))

    # Two-lobe
    inits.append(np.exp(-0.5 * ((i_arr - n * 0.25) / (n * 0.12)) ** 2) +
                  np.exp(-0.5 * ((i_arr - n * 0.75) / (n * 0.12)) ** 2))

    # Flat
    inits.append(np.ones(n))

    # Boundary-weighted (peaks at edges)
    inits.append(1.0 + 0.9 * np.cos(math.pi * i_arr / (n - 1)))

    # Tricube B-spline
    x = np.clip(np.abs(i_arr - (n - 1) / 2.0) / (n / 4.0), 0, 1)
    inits.append((1 - x) ** 3 + 0.01)

    # Sinc-squared
    inits.append(np.sinc(i_arr / (n / 3.0)) ** 2 + 0.005)

    # Power-law from center
    inits.append(1.0 / (1.0 + 0.08 * np.abs(i_arr - (n - 1) / 2.0)) ** 2)

    # Anti-symmetric emphasis: high left, low right
    inits.append(1.0 + 0.5 * np.cos(math.pi * i_arr / (n - 1)) +
                  0.3 * np.cos(2 * math.pi * i_arr / n))

    # Exponential decay pair
    inits.append(np.exp(-np.abs(i_arr - n * 0.3) / (n * 0.15)) +
                  np.exp(-np.abs(i_arr - n * 0.7) / (n * 0.15)))

    n_inits = len(inits)
    time_per_init = max(8.0, (phase1_end - t0 - 15.0) / n_inits)

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 12.0:
            break
        stop = min(phase1_end - 8.0, time.monotonic() + time_per_init)
        _try_optimize(n, init, stop, seed + 7777 * (idx + 1))

    # Phase 2: Alternative lengths (15% of budget)
    phase2_end = t0 + budget * 0.72
    other_lengths = [400, 448, 480, 384, 512, 320, 288, 544, 420, 368]
    time_per_len = max(5.0, (phase2_end - phase1_end - 5.0) / max(1, len(other_lengths)))

    for L in other_lengths:
        if time.monotonic() > phase2_end - 8.0:
            break
        stop = min(phase2_end - 5.0, time.monotonic() + time_per_len)
        init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
        _try_optimize(L, init_l, stop, seed + 50000 + L)

    # Phase 3: Warm-start LP on best (13% of budget)
    phase3_end = t0 + budget * 0.85
    if time.monotonic() < phase3_end - 30.0:
        stop_p3 = min(phase3_end - 8.0, time.monotonic() + (phase3_end - time.monotonic() - 10.0))
        _try_optimize(len(best_seq), best_seq.copy(), stop_p3, seed + 999999)

    # Phase 4: Polish (7% of budget)
    polish_end = deadline - 8.0
    if time.monotonic() < polish_end - 20.0:
        try:
            b_seq, b_sc = _polish(best_seq, polish_end, seed=4242)
            if b_sc < best_score - 1e-13:
                best_seq = b_seq
                best_score = b_sc
        except Exception:
            pass

    # Phase 5: Second polish with different seed
    if time.monotonic() < deadline - 12.0:
        try:
            b_seq2, b_sc2 = _polish(best_seq, deadline - 6.0, seed=88888)
            if b_sc2 < best_score - 1e-13:
                best_seq = b_seq2
                best_score = b_sc2
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    if final_sc < best_score:
        best_score = final_sc

    return [float(x) for x in best_seq]
