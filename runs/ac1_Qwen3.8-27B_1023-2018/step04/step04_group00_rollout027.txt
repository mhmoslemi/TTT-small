    a = np.asarray(a, dtype=np.float64)

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300), dtype=np.float64)
        return uu - float(np.mean(uu))

    def softmax_u(uu):
        m = float(np.max(uu))
        e = np.exp(uu - m, dtype=np.float64)
        s2 = float(e.sum())
        if not math.isfinite(s2) or s2 <= 0:
            return np.full(n, 1.0 / n, dtype=np.float64)
        return e / s2

    u = p_to_u(p)

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = softmax_u(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax):
            return 1e100, np.zeros(n)
        if T <= 1e-300:
            T = max(1e-300, 1e-5 * cmax)
        scaled = (c - cmax) / T
        # Clip to avoid overflow
        scaled = np.clip(scaled, -100.0, 100.0)
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            f_val = cmax
        else:
            f_val = cmax + T * math.log(sum_exp / conv_len)
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f_val), grad

    best_p = p.copy()
    best_sc = _score_norm(p)

    stages = [
        (0.05, 0.15),
        (0.02, 0.20),
        (0.006, 0.25),
        (0.002, 0.20),
        (0.0006, 0.20),
    ]

    start = time.monotonic()
    avail = max(0.0, stop_time - start - 10.0)
    if avail < 20.0:
        return p

    total_w = sum(w for _, w in stages)
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 8.0:
            break
        cum += wt
        end_stage = min(stop_time - 5.0, start + avail * (cum / total_w))
        if end_stage <= now + 5.0:
            continue

        c = cfast(p)
        cmax = float(c.max()) if c.size > 0 else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)

        maxit = 30 if n <= 2000 else 20
        opts = {
            "maxiter": maxit,
            "gtol": 1e-10,
            "ftol": 1e-16,
            "maxls": 15,
            "maxcor": 15,
            "maxdisp": 0,
            "disp": False,
        }

        stale = 0
        while time.monotonic() < end_stage - 2.0 and stale < 3:
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break

            if res.x is not None and np.all(np.isfinite(res.x)):
                u = res.x
                p_cur = softmax_u(u)
                p_cur = np.maximum(p_cur, 0.0)
                sc_cur = _score_norm(p_cur)
                if math.isfinite(sc_cur) and sc_cur < best_sc - 1e-12:
                    best_sc = sc_cur
                    best_p = p_cur.copy()
                    p = p_cur.copy()
                    u = p_to_u(p)
                    stale = 0
                    if consider:
                        consider(best_p)
                else:
                    stale += 1
            if getattr(res, "status", 1) == 0:
                break

    return best_p


def _coord_descent(p, stop_time, consider=None, seed=12345):
    """Coordinate descent with exact score evaluation."""
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s_total = float(p.sum())
    if s_total < 1e-300:
        return p, math.inf
    p = p / s_total
    s_total = 1.0

    Ltot = 2 * n - 1
    c = _conv_fft(p)
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()

    if consider:
        consider(best_p)

    rng = np.random.default_rng(seed)
    G = 24  # grid points per coordinate

    sweep = 0
    flat = 0
    max_sweeps = 50000

    while sweep < max_sweeps and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break

            # Renormalize if drifted
            if s_total > 1.25 or s_total < 0.75:
                inv = 1.0 / s_total
                p *= inv
                c *= inv * inv
                s_total = 1.0

            S = s_total
            v = float(p[i])

            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]

            # Max of unchanged parts
            max_const = 0.0
            if start > 0:
                ml = float(np.max(c[:start]))
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(np.max(c[end:]))
                if mr > max_const:
                    max_const = mr

            x_hi = max(20.0 * v, 60.0 / n, 0.05, 1e-12)
            x_hi = min(x_hi, 10.0 * S)

            pts = np.linspace(0.0, x_hi, G, dtype=np.float64)
            if 0.0 <= v <= x_hi:
                pts = np.concatenate((pts, [v]))
            pts = np.unique(pts)

            diag = i - start  # index within base where k=2i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64)
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = (denom > 1e-15) & (w > -v - 1e-15)
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                if 0 <= diag < Ls:
                    dval = base[diag] + w * infl[diag] + w * w
                    maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Fine search around best
            dw = x_hi / G
            lo = max(0.0, best_x - 2.0 * dw)
            hi = min(x_hi, best_x + 2.0 * dw)
            if hi <= lo:
                hi = lo + 1e-12
            fine = np.linspace(lo, hi, 20, dtype=np.float64)
            fine = np.unique(fine)
            vals2 = eval_xs(fine)
            if np.any(np.isfinite(vals2)):
                imin2 = int(np.argmin(vals2))
                if float(vals2[imin2]) < best_val:
                    best_x = float(fine[imin2])
                    best_val = float(vals2[imin2])

            if math.isfinite(best_val) and best_val < cur_score - 1e-14:
                w = best_x - v
                if abs(w) > 1e-16:
                    p[i] = best_x
                    s_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        # Renormalize at end of sweep
        if s_total > 1e-300:
            inv = 1.0 / s_total
            p *= inv
            c *= inv * inv
            s_total = 1.0
        else:
            break

        # Recompute c exactly to avoid drift
        c = _conv_fft(p)
        new_score = 2.0 * n * float(c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
            if consider:
                consider(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4 and time.monotonic() < stop_time - 20.0:
                # Perturbation to escape local minimum
                p = p * np.exp(rng.normal(0.0, 0.04, n, dtype=np.float64))
                p = np.maximum(p, 0.0)
                ss = float(p.sum())
                if ss > 0:
                    p /= ss
                    s_total = 1.0
                c = _conv_fft(p)
                cur_score = 2.0 * n * float(c.max())
                flat = 0

        sweep += 1

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 20.0

    parent = _get_parent()
    n = parent.size

    # Normalize
    p = parent.copy()
    s = float(p.sum())
    if s < 0.01:
        p = np.ones(n, dtype=np.float64) / n
        s = 1.0
    else:
        p = p / s

    best_arr = p.copy()
    best_score = _score_norm(p)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            ss = float(a.sum())
            if ss < 0.01:
                return
            a = a / ss
            sc = _score_norm(a)
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_score = sc
                best_arr = a.copy()
        except Exception:
            pass

    consider(p)
    print(f"Initial score: {best_score:.10f}")

    # Phase 1: LP-based optimization
    lp_stop = min(hard_deadline - 400.0, t0 + 250.0)
    if lp_stop > time.monotonic() + 60.0 and n <= 4096:
        try:
            p_lp, _ = _lp_optimize(p, n, lp_stop, consider=consider)
            if p_lp is not None and p_lp.size == n and np.all(np.isfinite(p_lp)):
                p = p_lp
                consider(p)
        except Exception as e:
            print(f"LP phase error: {e}")

    print(f"After LP: score = {best_score:.10f}, time = {time.monotonic()-t0:.0f}s")

    # Phase 2: L-BFGS fine-tuning
    lbfgs_stop = min(hard_deadline - 200.0, time.monotonic() + 200.0)
    if lbfgs_stop > time.monotonic() + 30.0:
        try:
            p_lb = _lbfgs_optimize(best_arr, n, lbfgs_stop, consider=consider)
            if p_lb is not None and p_lb.size == n:
                consider(p_lb)
        except Exception as e:
            print(f"L-BFGS phase error: {e}")

    print(f"After L-BFGS: score = {best_score:.10f}, time = {time.monotonic()-t0:.0f}s")

    # Phase 3: Coordinate descent (main polishing)
    coord_stop = hard_deadline - 15.0
    if coord_stop > time.monotonic() + 20.0:
        try:
            p_cd, _ = _coord_descent(best_arr, coord_stop, consider=consider,
                                     seed=int(seed + 777))
            if p_cd is not None and p_cd.size == n:
                consider(p_cd)
        except Exception as e:
            print(f"Coord descent error: {e}")

    print(f"After coord: score = {best_score:.10f}, time = {time.monotonic()-t0:.0f}s")

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = evaluate_sequence(ls)
            if not math.isfinite(final_sc):
                return [1.0]
        print(f"Final score: {final_sc:.10f}")
        return ls
    except Exception:
        return [float(x) for x in parent]
