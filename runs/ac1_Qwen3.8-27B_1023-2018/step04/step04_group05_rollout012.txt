$\|f_{new} * f_{new}\|_\infty = \|(1-t)^2 f*f + 2t(1-t) f*g + t^2 g*g\|_\infty$
$\le (1-t)^2 M + 2t(1-t) M + t^2 \|g*g\|_\infty$
$= (1-t)M(2-t) + t^2 \|g*g\|_\infty$

For small $t$, this is approximately $M + t(2M - 2M) + ... = M$ to first order. Hmm, that doesn't show improvement.

Let me be more careful. $(f_{new} * f_{new})_k = (1-t)^2 (f*f)_k + 2t(1-t)(f*g)_k + t^2 (g*g)_k$.

For the peak constraint: $(f*g)_k \le M$ for all $k$. But $(g*g)_k$ could be larger than $M$.

The maximum of $(f_{new} * f_{new})_k$ over $k$:
At the peak of $f*f$ (say $k^*$):
$(1-t)^2 M + 2t(1-t)(f*g)_{k^*} + t^2 (g*g)_{k^*}$
$\le (1-t)^2 M + 2t(1-t) M + t^2 (g*g)_{k^*}$
$= M(1-t^2) + t^2 (g*g)_{k^*}$

For this to be less than $M$: $(g*g)_{k^*} < M$. This might not hold.

At a non-peak $k$: $(1-t)^2 (f*f)_k + 2t(1-t)(f*g)_k + t^2 (g*g)_k$. Since $(f*f)_k < M$ and $(f*g)_k \le M$, for small $t$ this is $< M$.

So the improvement comes from the non-peak lags, not the peak lag. The peak lag might stay the same or increase slightly, but the overall max decreases because the non-peak lags were close to $M$ and they decrease.

Actually, I think the key insight is that the LP direction $g$ is chosen to maximize $\|g\|_1$ while keeping $f*g$ bounded. This means $g$ "spreads out" the mass in a way that reduces the convolution peaks when $f$ is replaced by $g$.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(200, dtype=np.float64)
    if p.size == 0:
        p = np.ones(200, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(200, dtype=np.float64)
    return p

def _conv_full(a):
    n = len(a)
    if n <= 800:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """A[k,j] = f[k-j] if 0 <= k-j < n, else 0. Shape (2n-1, n)."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_solve(f, n, M, timeout=4.0):
    """LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    if M <= 1e-18:
        return None
    A = _build_A(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs",
                      options={"time_limit": timeout, "presolve": True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g

    phase1_end = max(t0 + 80.0, min(t0 + budget * 0.52, deadline - 500.0))
    time_per_init = max(12.0, (phase1_end - t0) / n_inits)

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 8.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        _try(n, init, stop, seed + 7777 * (idx + 1))

    # Phase 2: Try other lengths
    phase2_end = max(t0 + 40.0, min(t0 + budget * 0.72, deadline - 300.0))
    if phase2_end > time.monotonic() + 25.0:
        other_lengths = [320, 384, 400, 448, 480, 368, 416]
        remaining = max(0, phase2_end - time.monotonic())
        time_per_len = max(6.0, remaining / max(1, len(other_lengths)))
        for L in other_lengths:
            if time.monotonic() > phase2_end - 6.0:
                break
            stop = min(phase2_end - 4.0, time.monotonic() + time_per_len)
            if L != len(best_seq):
                init_l = np.interp(np.linspace(0, 1, L),
                                   np.linspace(0, 1, len(best_seq)), best_seq)
            else:
                init_l = best_seq.copy()
            _try(L, init_l, stop, seed + 50000 + L)

    # Phase 3: Second LP pass on best (warm-start for extra refinement)
    phase3_end = max(t0 + 30.0, min(t0 + budget * 0.87, deadline - 150.0))
    if time.monotonic() < phase3_end - 20.0:
        stop_p3 = min(phase3_end - 5.0, deadline - 150.0)
        _try(len(best_seq), best_seq.copy(), stop_p3, seed + 999999)

    # Phase 4: Coordinate-descent polishing
    polish_end = min(deadline - 20.0, t0 + budget - 25.0)
    if time.monotonic() < polish_end - 10.0:
        try:
            b_seq, b_sc = _polish(best_seq, polish_end, seed=4242)
            if math.isfinite(b_sc) and b_sc < best_score - 1e-14:
                best_seq = b_seq
                best_score = b_sc
        except Exception:
            pass

    # Phase 5: Second polish with different seed
    if time.monotonic() < deadline - 25.0:
        try:
            b_seq2, b_sc2 = _polish(best_seq, deadline - 20.0, seed=88888)
            if math.isfinite(b_sc2) and b_sc2 < best_score - 1e-14:
                best_seq = b_seq2
                best_score = b_sc2
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
