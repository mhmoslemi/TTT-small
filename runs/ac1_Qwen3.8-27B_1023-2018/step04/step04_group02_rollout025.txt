def _conv_fast(a):
    n = len(a)
    if n <= 2048:
        return np.convolve(a, a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _lp_direction(f, n, M, timeout=3.0):
    """LP to find best direction g s.t. f*g <= M, g >= 0, maximize sum(g)."""
    from scipy.optimize import linprog
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g

            f_final, sc_final = _cd_optimize(best_seq, deadline - 5.0, max_outer=40, seed=seed + 99)
        except Exception:
            f_final, sc_final = best_seq.copy(), best_score
        if sc_final < best_score - 1e-12:
            best_score = sc_final
            best_seq = f_final.copy()

    # Final validation with exact evaluator
    if best_seq is None:
        best_seq = np.ones(n_parent)

    final_sc = _eval(best_seq)
    if not math.isfinite(final_sc):
        best_seq = np.ones(n_parent)
        final_sc = _eval(best_seq)

    # Ensure valid output
    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    if not result or any(math.isnan(x) or math.isinf(x) for x in result):
        result = [1.0] * n_parent

    return result
