        stop_p3 = min(phase3_end - 5.0, time.monotonic() + (phase3_end - time.monotonic()))
        try:
            res, sc = _lp_iterate(best_ref[0].copy(), n_best, stop_p3, seed + 999999)
            _try_update(best_ref, best_sc_ref, res, stop_p3)
        except Exception:
            pass

    # Phase 4: Polish (12%)
    phase4_end = deadline - 12.0
    if time.monotonic() < phase4_end - 10.0:
        try:
            b_seq, b_sc = _polish(best_ref[0], phase4_end, seed=77777)
            _try_update(best_ref, best_sc_ref, b_seq, phase4_end)
        except Exception:
            pass

    # Phase 5: Second polish with different seed
    if time.monotonic() < deadline - 14.0:
        try:
            b_seq2, b_sc2 = _polish(best_ref[0], deadline - 10.0, seed=123456)
            _try_update(best_ref, best_sc_ref, b_seq2, deadline)
        except Exception:
            pass

    # Final: try one more LP round if time permits
    if time.monotonic() < deadline - 18.0:
        n_best = len(best_ref[0])
        stop_final = deadline - 14.0
        try:
            res, sc = _lp_iterate(best_ref[0].copy(), n_best, stop_final, seed + 777777)
            _try_update(best_ref, best_sc_ref, res, stop_final)
        except Exception:
            pass

    # Final validation with exact evaluator
    final_candidate = best_ref[0]
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_candidate]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Fallback to parent
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in parent]

    if final_sc < best_sc_ref[0]:
        best_sc_ref[0] = final_sc

    return [float(x) for x in final_candidate]
