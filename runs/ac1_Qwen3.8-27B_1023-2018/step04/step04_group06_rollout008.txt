        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = cand.copy()

    # Phase 1: Optimize parent directly
    t1 = t0 + min(200.0, budget_s * 0.18)
    r = optimize_lp(best_a, t1, seed=1, max_stuck=100)
    try_update(r / r.sum())

    # Phase 2: Coarse-to-fine on parent
    if time.monotonic() < deadline - 60:
        t2 = min(t0 + 400.0, deadline - 60)
        r = optimize_coarse_to_fine(best_a / best_a.sum(), t2, seed=200)
        try_update(r)

    # Phase 3: Diverse initializations at parent length
    inits = []
    pn = best_a / best_a.sum()
    x = np.arange(n, dtype=np.float64)
    # Power-law decay
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        v = np.maximum(1.0 - x / max(n - 1, 1), 1e-10) ** p
        inits.append(v / v.sum())
    # Exponential decay
    for al in [0.5, 1.0, 1.5, 2.5, 4.0]:
        v = np.exp(-al * x / n)
        inits.append(v / v.sum())
    # Gaussian center
    for w in [0.08, 0.12, 0.2, 0.3, 0.5]:
        ctr = (n - 1) / 2.0
        v = np.exp(-0.5 * ((x - ctr) / max(w * n / 2, 1)) ** 2)
        inits.append(v / v.sum())
    # Cosine modulated
    for fr in [1, 2, 3]:
        for amp in [0.2, 0.4, 0.6]:
            v = np.maximum(1.0 + amp * np.cos(2 * np.pi * fr * x / max(n - 1, 1)), 0.01)
            inits.append(v / v.sum())
    # Blends with parent
    for frac in [0.2, 0.4, 0.6, 0.8]:
        for init in inits[:6]:
            b = frac * pn + (1 - frac) * init
            inits.append(b / b.sum())

    t3 = min(t0 + 700.0, deadline - 40)
    per_init = max(2.0, (t3 - time.monotonic()) / max(1, len(inits)))
    for i, init in enumerate(inits):
        if time.monotonic() > t3 - 5.0:
            break
        alloc = min(time.monotonic() + per_init, t3 - 2.0)
        r = optimize_lp(init, alloc, seed=500 + i, max_stuck=40)
        try_update(r)

    # Phase 4: Explore nearby lengths
    if time.monotonic() < deadline - 80:
        for dn in [-20, -10, -5, -2, -1, 1, 2, 5, 10, 20, 30]:
            if time.monotonic() > deadline - 30:
                break
            nt = n + dn
            if nt < 10 or nt > 500:
                continue
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, best_a / best_a.sum())
            init_new = np.maximum(init_new, 1e-12)
            init_new /= init_new.sum()
            r = optimize_lp(init_new, time.monotonic() + 15.0, seed=1000 + nt, max_stuck=30)
            try_update(r)

    # Phase 5: Final deep polish on best
    if time.monotonic() < deadline - 10:
        af = best_a / best_a.sum()
        r = optimize_lp(af, deadline, seed=9999, max_stuck=300)
        try_update(r)

    # Final validation with exact evaluator
    try:
        fs = float(evaluate_sequence([float(x) for x in best_a]))
        if not (np.isfinite(fs) and fs > 0):
            raise ValueError("invalid")
    except Exception:
        fallback = [1.0] * 32
        return fallback

    return [float(x) for x in best_a]
