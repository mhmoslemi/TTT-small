        if time.monotonic() > stop - 3.0:
            return
        try:
            res, sc = _lp_optimize(init_arr, n, stop, sd)
        except Exception:
            return
        if res is None or len(res) == 0:
            return
        if not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0, 1000)
        if res.sum() < 0.01:
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in res]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-14:
            best_seq = res.copy()
            best_score = exact

    # Phase 1: LP on parent length with diverse inits
    n = n_p
    phase1_end = t0 + budget * 0.48
    rng_main = np.random.default_rng(seed)
    ia = np.arange(n, dtype=np.float64)

    inits = [parent.copy(),
             1.0 + 0.9 * ia / (n - 1),
             1.0 + 0.9 * (1.0 - ia / (n - 1)),
             0.3 + 0.7 * np.sin(np.pi * ia / (n - 1)),
             np.exp(-4.0 * ia / n) + 0.04,
             np.exp(-4.0 * (n - 1 - ia) / n) + 0.04,
             np.abs(rng_main.normal(0, 1, n)) + 0.03,
             1.0 + rng_main.uniform(-0.35, 0.35, n),
             np.maximum(0.0, 1.6 - 3.5 * np.abs(ia - 0.25 * n) / n) + 0.02,
             np.maximum(0.0, 1.6 - 3.5 * np.abs(ia - 0.75 * n) / n) + 0.02,
             1.0 + 0.6 * np.cos(2.0 * np.pi * ia / n) + 0.4 * np.cos(4.0 * np.pi * ia / n)]

    n_inits = len(inits)
    time_per_init = max(12.0, (phase1_end - t0) / n_inits)

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 6.0:
            break
        stop = min(phase1_end - 3.0, time.monotonic() + time_per_init)
        _try(n, init, stop, seed + 7777 * (idx + 1))

    # Phase 2: Alternative lengths
    phase2_end = t0 + budget * 0.70
    other_lengths = [400, 448, 480, 384, 512, 320, 288, 600, 700]
    time_per_len = max(8.0, (phase2_end - phase1_end) / max(1, len(other_lengths)))

    for L in other_lengths:
        if time.monotonic() > phase2_end - 5.0:
            break
        stop = min(phase2_end - 3.0, time.monotonic() + time_per_len)
        if L == len(best_seq):
            init_l = best_seq.copy()
        else:
            init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
        _try(L, init_l, stop, seed + 50000 + L)

    # Phase 3: Third LP pass (warm-start from best)
    phase3_end = t0 + budget * 0.86
    if time.monotonic() < phase3_end - 20.0:
        stop_p3 = min(phase3_end - 5.0, time.monotonic() + (phase3_end - time.monotonic()) - 5.0)
        _try(len(best_seq), best_seq.copy(), stop_p3, seed + 999999)

    # Phase 4: Coordinate-descent polish
    polish_end = deadline - 10.0
    if time.monotonic() < polish_end - 10.0:
        try:
            b_seq, b_sc = _polish(best_seq, polish_end, seed=4242)
            if b_sc < best_score - 1e-14:
                best_seq = b_seq
                best_score = b_sc
        except Exception:
            pass

    # Phase 5: Second polish with different seed
    if time.monotonic() < deadline - 14.0:
        try:
            b_seq2, b_sc2 = _polish(best_seq, deadline - 8.0, seed=88888)
            if b_sc2 < best_score - 1e-14:
                best_seq = b_seq2
                best_score = b_sc2
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    if final_sc >= best_score - 1e-12:
        best_score = final_sc

    return [float(x) for x in best_seq]
