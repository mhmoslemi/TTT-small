    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)

For coordinate descent: fix sum(a) = S, all a_j for j ≠ i fixed. Optimize a_i.
The score is `2n * max_k c_k(a_i) / S^2` where S = (S - a_i_old) + a_i = S (fixed).

Each c_k is:

- `c_k = base_k + a_i^2` (quadratic) for k = 2i

So max_k c_k(a_i) is the max of linear and quadratic functions, which is a unimodal (well, convex piecewise) function of a_i. We can find the minimum by:

1. For linear terms, the "kink" points are where one linear function crosses another.
2. For the quadratic term, the minimum is at a_i = 0 (since the quadratic has positive coefficient).

Actually, the max of convex functions is convex, so `f(a_i) = max_k c_k(a_i)` is convex in a_i (it's the max of affine functions, plus possibly a_i^2). The minimum is where the derivative goes from negative to positive.

We can find it by:
- Compute c_k(0) and c_k(1000)
- The minimum is in [0, 1000]
- Use golden section search or binary search on the derivative

from scipy.optimize import linprog


def _get_parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(200, dtype=np.float64)
    if p.size == 0:

    p = np.clip(p, 0.0, 1000.0)

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    c = _conv(a)
    m = float(c.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _make_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    return A


def _lp_dir(f, n, M, tlimit=2.0):
    try:
        A = _make_A(f, n)
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M*(1+1e-10)+1e-10),
                      bounds=[(0,None)]*n, method='highs',
                      options={'presolve':True, 'time_limit':tlimit})
        if res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0, 1e4)
            if g.sum() > 1e-14:
                return g
    except Exception:
        pass
    return None

                cm = max(cn.max(), c_rest_max)
