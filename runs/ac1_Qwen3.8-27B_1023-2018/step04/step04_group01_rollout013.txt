def _golden_section(max_func, lo, hi, iters=50):
    gr = (math.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = max_func(c)
    fd = max_func(d)
    for _ in range(iters):

            hi = d; d = c; fd = fc
            c = hi - gr * (hi - lo)
            fc = max_func(c)
        else:
            lo = c; c = d; fc = fd
            d = lo + gr * (hi - lo)
            fd = max_func(d)

    def max_at(t):
        return float(np.max((1.0 - t) ** 2 * bf + 2.0 * t * (1.0 - t) * mix + t ** 2 * bg))

    return _golden_section(max_at, 0.0, 1.0, iters=48)


def _lp_optimize(f0, n, stop_time, seed, max_iters=500):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs

    best_f = f.copy()
    best_score = _score(f)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 5:
            break

        b = _conv(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, M, time_limit=4.0)
        if g is None:
            stall += 3
            if stall > 25:
                break
            f = np.clip(f + rng.normal(0, 0.015, n), 0, None)
            f /= f.sum()
            continue

        gs = float(g.sum())
        if gs < 1e-15:
            stall += 3
            if stall > 25:
                break
            f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
            f /= f.sum()
            continue
        g = g / gs

        dist = float(np.linalg.norm(g - f))
        if dist < 1e-8:
            stall += 2
            if stall > 25:
                break
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            f /= f.sum()
            continue

        best_t = _line_search_conv(f, g, n)

        candidates = sorted(set([max(0.0, best_t - 0.05), best_t, min(1.0, best_t + 0.05), 0.5, 1.0]))
        best_sc = _score(f)
        best_new = f.copy()
        for t_try in candidates:
            f_new = f + t_try * (g - f)
            f_new = np.clip(f_new, 0, None)
            s_new = f_new.sum()
            if s_new < 0.01:
                continue
            f_new = f_new / s_new
            sc = _score(f_new)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_new = f_new

        if best_sc < best_score - 1e-14:
            f = best_new
            best_f = f.copy()
            best_score = best_sc
            stall = 0
        elif best_sc < _score(f) - 1e-14:
            f = best_new
            stall = 0
        else:
            stall += 1
            if stall >= 20:
                f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
                f /= f.sum()
                stall = 0
                if time.monotonic() > stop_time - 30:
                    break

    return best_f, best_score


def _coord_descent(f0, stop_time, max_sweeps=80, seed=0):
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        return f, math.inf
    f = f / S
    S = 1.0
    conv = _conv(f)
    best_score = 2.0 * n * float(conv.max())
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 5:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 5:
                break
            v = f[i]
            lo_idx = i
            hi_idx = min(i + n, 2 * n - 1)
            seg_len = hi_idx - lo_idx

            max_out = 0.0
            if lo_idx > 0:
                max_out = max(max_out, float(conv[:lo_idx].max()))
            if hi_idx < 2 * n - 1:
                max_out = max(max_out, float(conv[hi_idx:].max()))

            base = conv[lo_idx:hi_idx]
            w = f[:seg_len]
            j_quad = 2 * i - lo_idx

            x_max = max(3.0 * v + 0.02, 0.3)
            x_max = min(x_max, 5.0)
            G = 80
            xs = np.linspace(0.0, x_max, G)

            delta = 2.0 * (xs[:, None] - v) * w[None, :]
            new_vals = base[None, :] + delta
            if 0 <= j_quad < seg_len:
                new_vals[:, j_quad] += xs ** 2 - v ** 2

            max_inside = new_vals.max(axis=1)
            max_total = np.maximum(max_inside, max_out)
            S_new = S - v + xs
            valid = S_new > 0.01
            if not valid.any():
                continue
            scores = np.full(G, np.inf)
            scores[valid] = 2.0 * n * max_total[valid] / (S_new[valid] ** 2)
            idx = int(np.argmin(scores))

            if scores[idx] < best_score - 1e-15:
                best_x = xs[idx]
                best_sc = scores[idx]
                dw = x_max / G
                x_lo = max(0.0, best_x - 4 * dw)
                x_hi = min(x_max, best_x + 4 * dw)
                xs_f = np.linspace(x_lo, x_hi, 50)
                delta_f = 2.0 * (xs_f[:, None] - v) * w[None, :]
                nv_f = base[None, :] + delta_f
                if 0 <= j_quad < seg_len:
                    nv_f[:, j_quad] += xs_f ** 2 - v ** 2
                mi_f = nv_f.max(axis=1)
                mt_f = np.maximum(mi_f, max_out)
                Sn_f = S - v + xs_f
                va_f = Sn_f > 0.01
                if va_f.any():
                    sf = np.full(len(xs_f), np.inf)
                    sf[va_f] = 2.0 * n * mt_f[va_f] / (Sn_f[va_f] ** 2)
                    idx_f = int(np.argmin(sf))
                    if sf[idx_f] < best_sc:
                        best_x = xs_f[idx_f]
                        best_sc = sf[idx_f]

                if best_sc < best_score - 1e-15:
                    w_diff = best_x - v
                    conv[lo_idx:hi_idx] += w_diff * 2.0 * w
                    if 0 <= j_quad < seg_len:
                        conv[2 * i] += best_x ** 2 - v ** 2
                    f[i] = best_x
                    S += best_x - v
                    best_score = best_sc
                    best_f = f.copy()
                    improved = True

        if not improved and sweep > 2:
            break

    final_score = _score(best_f)
    return best_f, final_score


def _grad_optimize(f0, n, stop_time, seed, max_iters=8000):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f = f / f.sum()
    best_f = f.copy()
    best_score = _score(f)
    alpha = 0.04
    alpha_min = 1e-10
    tau = 60.0
    fsize = next_fast_len(2 * n - 1)
    idx_i = np.arange(n)

    for it in range(max_iters):
        if time.monotonic() > stop_time - 5:
            break

        b = _conv(f)
        S = float(f.sum())
        M = float(b.max())
        if M < 1e-18:
            break
        score = 2.0 * n * M / (S * S)
        if score < best_score - 1e-14:
            best_score = score
            best_f = f.copy()

        b_shift = b - M + 1e-15
        logits = tau * b_shift
        logits -= logits.max()
        p = np.exp(logits)
        ps = p.sum()
        if ps < 1e-30:
            p = np.zeros(2 * n - 1)
            p[int(np.argmax(b))] = 1.0
            ps = 1.0
        p /= ps

        indices = idx_i[:, None] + idx_i[None, :]
        M_mat = p[indices]
        grad = 2.0 * (M_mat @ f)
        idx2 = 2 * idx_i
        valid_2 = idx2 < (2 * n - 1)
        grad[valid_2] += p[idx2[valid_2]] * f[valid_2]

        sg = (2.0 * n / (S * S)) * (grad - 2.0 * M / S)
        f_new = f - alpha * sg
        f_new = np.maximum(f_new, 0.0)
        if f_new.sum() < 0.01:
            f_new = np.maximum(f + rng.normal(0, 0.01, n), 0)
        f_new = f_new / f_new.sum()

        sc_new = _score(f_new)
        if sc_new < score - 1e-15:
            f = f_new
            alpha = min(0.15, alpha * 1.02)
        else:
            alpha = max(alpha_min, alpha * 0.65)
            if alpha < 1e-8:
                f = np.maximum(f + rng.normal(0, 0.012, n), 0)
                f /= f.sum()
                alpha = 0.03

        if it % 1500 == 1499:
            f = np.maximum(f + rng.normal(0, 0.02, n), 0)
            f /= f.sum()
            alpha = 0.04
            tau = max(20.0, tau * 0.85)

    return best_f, best_score


def _pair_polish(f0, stop_time, max_rounds=25, seed=0):
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / f.sum()
    best_score = _score(f)
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 5:
            break
        improved = False
        for _ in range(35):
            if time.monotonic() > stop_time - 5:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-12 and oj < 1e-12:
                continue
            mults = (0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0, 0.0)
            best_pair = (oi, oj)
            best_sc = best_score
            for mi in mults:
                ni = oi * mi if oi > 1e-12 else (0.005 if mi > 0 else 0.0)
                for mj in mults:
                    nj = oj * mj if oj > 1e-12 else (0.005 if mj > 0 else 0.0)
                    f[i], f[j] = ni, nj
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _score(f)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        best_pair = (ni, nj)
            f[i], f[j] = best_pair
            if best_sc < best_score - 1e-14:
                best_score = best_sc
                best_f = f.copy()
                improved = True
        if not improved and rnd > 2:
            break
    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 15.0

    parent = _get_parent()
    n = len(parent)

    fallback = parent.copy()
    try:
        fallback_sc = _score(fallback)
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(n) / n
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update_best(seq, sc=None):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64).ravel()
        seq = np.clip(seq, 0, 1000)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        if sc is None:
            sc = _score(seq)
        if not math.isfinite(sc):
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-14:
            best_seq = seq.copy()
            best_score = exact

    # Phase 1: LP optimization from parent
    p1_end = min(t0 + budget * 0.30, hard_deadline - 300)
    if time.monotonic() < p1_end - 10:
        try:
            res, sc = _lp_optimize(parent, n, p1_end, seed, max_iters=500)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res = res / res.sum()
                _update_best(res, sc)
        except Exception:
            pass

    # Phase 2: Coordinate descent polish
    p2_end = min(t0 + budget * 0.40, hard_deadline - 250)
    if time.monotonic() < p2_end - 10:
        try:
            res, sc = _coord_descent(best_seq, p2_end, max_sweeps=100, seed=seed + 1)
            _update_best(res, sc)
        except Exception:
            pass

    # Phase 3: More LP from perturbed best
    p3_end = min(t0 + budget * 0.55, hard_deadline - 200)
    if time.monotonic() < p3_end - 10 and best_score > 1.503:
        rng3 = np.random.default_rng(seed + 100)
        pert = np.clip(best_seq + rng3.normal(0, 0.02, n), 0, None)
        pert /= pert.sum()
        try:
            res, sc = _lp_optimize(pert, n, p3_end, seed + 200, max_iters=400)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res = res / res.sum()
                _update_best(res, sc)
        except Exception:
            pass

    # Phase 4: Gradient descent
    p4_end = min(t0 + budget * 0.70, hard_deadline - 150)
    if time.monotonic() < p4_end - 10:
        try:
            res, sc = _grad_optimize(best_seq, n, p4_end, seed + 300, max_iters=8000)
            _update_best(res, sc)
        except Exception:
            pass

    # Phase 5: Coordinate descent again
    p5_end = min(t0 + budget * 0.80, hard_deadline - 100)
    if time.monotonic() < p5_end - 10:
        try:
            res, sc = _coord_descent(best_seq, p5_end, max_sweeps=80, seed=seed + 400)
            _update_best(res, sc)
        except Exception:
            pass

    # Phase 6: Pair polish
    p6_end = min(t0 + budget * 0.87, hard_deadline - 60)
    if time.monotonic() < p6_end - 10:
        try:
            res, sc = _pair_polish(best_seq, p6_end, max_rounds=25, seed=seed + 500)
            _update_best(res, sc)
        except Exception:
            pass

    # Phase 7: Final coordinate descent
    p7_end = hard_deadline - 20.0
    if time.monotonic() < p7_end - 10:
        try:
            res, sc = _coord_descent(best_seq, p7_end, max_sweeps=50, seed=seed + 600)
            _update_best(res, sc)
        except Exception:
            pass

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf
    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
