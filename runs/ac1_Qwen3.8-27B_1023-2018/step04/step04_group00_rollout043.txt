                found = True
                no_improve_count = 0
                # Adapt alpha
                if alpha_try > 0.1:
                    alpha = min(1.0, alpha_try * 1.1)
                else:

    def coord_polish(arr, deadline, max_sweeps=20000, seed=42):
        n_local = arr.size
        p = np.maximum(arr, 0.0).astype(np.float64).copy()
        ps = p.sum()
        if ps < 1e-300:
            return p, math.inf
        p /= ps
        
        c = np.convolve(p, p)
        np.maximum(c, 0.0, out=c)
        cur_score = float(2.0 * n_local * c.max())
        best_local_score = cur_score
        best_local_p = p.copy()
        
        Ltot = 2 * n_local - 1
        G = 24  # grid points per coordinate
        
        rng = np.random.default_rng(seed)
        sweep = 0
        flat = 0
        
        while sweep < max_sweeps and time.monotonic() < deadline:
            improved = False
            order = rng.permutation(n_local)
            
            for ii in order:
                i = int(ii)
                if time.monotonic() > deadline - 0.5:
                    break
                
                v = float(p[i])
                start = i
                end = min(i + n_local, Ltot)
                Ls = end - start
                if Ls <= 0:
                    continue
                
                # Constant max from unaffected region
                max_const = 0.0
                if start > 0:
                    mc = float(np.max(c[:start]))
                    if mc > max_const:
                        max_const = mc
                if end < Ltot:
                    mc = float(np.max(c[end:]))
                    if mc > max_const:
                        max_const = mc
                
                # Search range
                x_hi = max(5.0 * v, 5.0 / n_local, 1e-8)
                x_hi = min(x_hi, 3.0)
                if not math.isfinite(x_hi) or x_hi <= 0:
                    x_hi = 1.0 / n_local
                
                # Grid
                pts = np.linspace(0.0, x_hi, G, dtype=np.float64)
                # Include current value
                if 0.0 <= v <= x_hi:
                    pts = np.unique(np.concatenate([pts, [v]]))
                
                # Vectorized evaluation
                base = c[start:end]
                infl = 2.0 * p[:Ls]
                diag_pos = 2 * i - start  # = i
                
                ws = pts - v  # shape (len(pts),)
                aff = base[None, :] + ws[:, None] * infl[None, :]  # shape (len(pts), Ls)
                if 0 <= diag_pos < Ls:
                    aff[:, diag_pos] += ws * ws
                max_aff = aff.max(axis=1)
                scores = 2.0 * n_local * np.maximum(max_aff, max_const)
                
                if not np.all(np.isfinite(scores)):
                    scores = np.where(np.isfinite(scores), scores, math.inf)
                
                imin = int(np.argmin(scores))
                best_x = float(pts[imin])
                best_val = float(scores[imin])
                
                # Fine search around best
                if best_val < cur_score - 1e-14:
                    dw = x_hi / G
                    lo = max(0.0, best_x - 2*dw)
                    hi = min(x_hi, best_x + 2*dw)
                    if hi > lo:
                        fine_pts = np.linspace(lo, hi, 16, dtype=np.float64)
                        fine_ws = fine_pts - v
                        fine_aff = base[None, :] + fine_ws[:, None] * infl[None, :]
                        if 0 <= diag_pos < Ls:
                            fine_aff[:, diag_pos] += fine_ws * fine_ws
                        fine_max = fine_aff.max(axis=1)
                        fine_scores = 2.0 * n_local * np.maximum(fine_max, max_const)
                        fi = int(np.argmin(fine_scores))
                        if float(fine_scores[fi]) < best_val:
                            best_x = float(fine_pts[fi])
                            best_val = float(fine_scores[fi])
                
                # Apply update
                if best_val < cur_score - 1e-14 and 0 <= best_x <= 10.0:
                    w = best_x - v
                    if abs(w) > 1e-15:
                        p[i] = best_x
                        c[start:end] += w * infl
                        if 0 <= diag_pos < Ls:
                            c[start + diag_pos] += w * w
                        # Renormalize
                        new_s = float(p.sum())
                        if new_s > 1e-300:
                            p /= new_s
                            c /= (new_s * new_s)
                        cur_score = best_val
                        improved = True
                        if cur_score < best_local_score:
                            best_local_score = cur_score
                            best_local_p = p.copy()
            
            if improved:
                flat = 0
            else:
                flat += 1
                if flat >= 4:
                    if time.monotonic() < deadline - 3.0:
                        p = p * np.exp(rng.normal(0, 0.025, n_local))
                        p = np.maximum(p, 0)
                        ps2 = p.sum()
                        if ps2 > 0:
                            p /= ps2
                        c = np.convolve(p, p)
                        np.maximum(c, 0, out=c)
                        cur_score = float(2.0 * n_local * c.max())
                        if cur_score < best_local_score:
                            best_local_score = cur_score
                            best_local_p = p.copy()
                        flat = 0
                    else:
                        break
            sweep += 1
        
        return best_local_p, best_local_score
