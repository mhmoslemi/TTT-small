def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, float_a=fsize) if hasattr(rfft, 'float_a') else rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _lp_direction(f, b, n, num_peaks, timeout=3.0):
    M = float(b.max())
    if M < 1e-20:
        return None
    # Select peak indices
    sorted_idx = np.argsort(b)[::-1][:num_peaks]
    rows, cols, vals = [], [], []
    for row_idx, k in enumerate(sorted_idx):
        k_i = int(k)
        i_start = max(0, k_i - n + 1)
        i_end = min(n, k_i + 1)
        for i in range(i_start, i_end):
            v = f[k_i - i]
            if v > 1e-20:
                rows.append(row_idx)
                cols.append(i)
                vals.append(v)
    if not rows:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(sorted_idx), n))
    c_vec = -np.ones(n)
    b_ub = np.full(len(sorted_idx), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _line_search(f, g, n, max_evals=20):
    """Golden-section search for optimal t mixing f and g."""
    a, b = 0.01, 0.99
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    t1 = a + (1.0 - gr) * (b - a)
    t2 = a + gr * (b - a)
    f_new1 = (1.0 - t1) * f + t1 * g
    f_new1 = np.clip(f_new1, 0, None)
    f_new2 = (1.0 - t2) * f + t2 * g
    f_new2 = np.clip(f_new2, 0, None)
    sc1 = _score(f_new1)
    sc2 = _score(f_new2)
    evals = 2
    best_t = 0.0
    best_sc = min(sc1, sc2)
    for _ in range(max(1, max_evals - 2)):
        if (b - a) < 5e-4:
            break
        if sc1 < sc2:
            b = t2
            t2 = t1
            sc2 = sc1
            t1 = a + (1.0 - gr) * (b - a)
            fnt = np.clip((1.0 - t1) * f + t1 * g, 0, None)
            sc1 = _score(fnt)
        else:
            a = t1
            t1 = t2
            sc1 = sc2
            t2 = a + gr * (b - a)
            fnt = np.clip((1.0 - t2) * f + t2 * g, 0, None)
            sc2 = _score(fnt)
        evals += 1
        cur_best = min(sc1, sc2)
        if cur_best < best_sc:
            best_sc = cur_best
            best_t = t1 if sc1 < sc2 else t2
    return best_t, best_sc

def _subgrad_optimize(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-8, None)
    f = f / f.sum()
    best_f = f.copy()

    lr = 0.3
    stall = 0
    max_stall = 60

    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break
        # Select top peaks
        peak_count = max(3, int(0.03 * (2 * n - 1)))
        peak_idx = np.argsort(b)[::-1][:peak_count]
        grad = np.zeros(n)
        for k in peak_idx:
            k_i = int(k)
            for i in range(max(0, k_i - n + 1), min(n, k_i + 1)):
                grad[i] += 2.0 * f[k_i - i]
        grad /= peak_count
        gscale = grad.max()
        if gscale < 1e-15:
            break
        lr_eff = lr / gscale
        exponent = -lr_eff * grad
        exponent -= exponent.max()
        weights = np.exp(exponent)
        f_new = (f + 1e-12) * weights
        s = f_new.sum()
        if s < 1e-12:
            break
        f_new = f_new / s
        sc = _score(f_new)
        if sc < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = sc
            f = f_new
            stall = 0
            lr *= 0.995
        else:
            half = 0.5 * f + 0.5 * f_new
            sc_h = _score(half)
            if sc_h < best_score - 1e-12:
                best_f = half.copy()
                best_score = sc_h
                f = half
                stall = 0

                    f = f / f.sum()

def _coord_polish(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0.0, None)
    best_sc = _score(f)
    no_imp = 0
    max_no_imp = 8
    mults_c = (0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 0.0)
    mults_f = (0.9, 1.1, 0.85, 1.15, 0.95, 1.05)

    while time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = f[i]
            best_local = best_sc
            best_val = orig
            if orig < 1e-12:
                for mv in (0.01, 0.05, 0.1, 0.5 * max(1.0, f.mean())):
                    f[i] = mv
                    sc = _score(f)
                    if sc < best_local - 1e-14:
                        best_local = sc
                        best_val = mv
                f[i] = best_val
                if best_local < best_sc - 1e-14:
                    best_sc = best_local
                    improved = True
                continue
            for mlt in mults_c:
                f[i] = orig * mlt
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_val = f[i]
            if best_local < best_sc - 1e-14:
                f[i] = best_val
                for mlt in mults_f:
                    f[i] = orig * mlt
                    if f.sum() < 0.01:
                        f[i] = orig
                        continue
                    sc = _score(f)
                    if sc < best_local - 1e-14:
                        best_local = sc
                        best_val = f[i]
            f[i] = best_val
            if best_local < best_sc - 1e-14:
                best_sc = best_local
                improved = True
        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0
    return f.copy(), best_sc

    phase3_end = t0 + budget * 0.82
    if time.monotonic() < phase3_end - 8.0:
        stop = min(phase3_end - 4.0, time.monotonic() + (phase3_end - time.monotonic() - 4.0))
        try:
            res, sc = _subgrad_optimize(best_seq, stop, seed + 77777)
            _update_best(res)
        except Exception:
            pass

    # Phase 4: Coordinate descent polish (10% time)
    phase4_end = t0 + budget * 0.92
    if time.monotonic() < phase4_end - 6.0:
        stop = min(phase4_end - 4.0, time.monotonic() + (phase4_end - time.monotonic() - 4.0))
        try:
            res, sc = _coord_polish(best_seq, stop, seed + 55555)
            _update_best(res)
        except Exception:
            pass

    # Phase 5: Final LP + polish on best (remaining)
    final_end = deadline
    if time.monotonic() < final_end - 20.0:
        remaining = final_end - time.monotonic() - 5.0
        if remaining > 30:
            half = remaining / 2.0
            stop1 = time.monotonic() + half - 3.0
            try:
                res, sc = _optimize_lp(best_seq, stop1, seed + 999999, init_peaks=10)
                _update_best(res)
            except Exception:
                pass
            if time.monotonic() < final_end - 8.0:
                stop2 = final_end - 5.0
                try:
                    res, sc = _coord_polish(best_seq, stop2, seed + 888888)
                    _update_best(res)
                except Exception:
                    pass

    # Try one alternative length if time permits
    if time.monotonic() < final_end - 25.0:
        alt_n = n_p + 16
        if alt_n >= 100 and alt_n <= 600:
            x_old = np.linspace(0, 1, n_p)
            x_new = np.linspace(0, 1, alt_n)
            init_alt = np.interp(x_new, x_old, best_seq)
            stop_alt = final_end - 10.0
            if time.monotonic() < stop_alt - 10.0:
                try:
                    res, sc = _optimize_lp(init_alt, stop_alt, seed + alt_n * 7, init_peaks=8)
                    if sc < best_score - 1e-6:
                        # Quick polish
                        stop_pol = final_end - 5.0
                        if time.monotonic() < stop_pol - 8.0:
                            res2, sc2 = _coord_polish(res, stop_pol, seed + alt_n * 13)
                            _update_best(res2)
                        else:
                            _update_best(res)
                except Exception:
                    pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0] * 4
        return [float(x) for x in parent]

    return [float(x) for x in best_seq]
