def _lse_optimize(p, n, stop_time, consider=None):

    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    
    conv_size = next_fast_len(2*n - 1)
    cross_size = next_fast_len(3*n - 2)
    conv_len = 2*n - 1
    cross_len = 3*n - 2
    
    def conv_fast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return c
    
    def softmax(u):
        m = float(np.max(u))
        e = np.exp(u - m)
        s = e.sum()
        if s < 1e-300:
            return np.full(n, 1.0/n)
        return e / s
    
    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - uu.mean()
    
    u = p_to_u(p)
    
    def obj(uu, T):
        pp = softmax(uu)
        c = conv_fast(pp)
        cmax = float(c.max())
        if T < 1e-10:
            # Nearly exact max
            return cmax, 2.0 * conv_fast(pp) * 0  # placeholder
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if sum_exp < 1e-300:
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * np.log(sum_exp / conv_len)
            w = exps / sum_exp
        # Gradient: df/da_j = grad_j (from weight w)
        ar = pp[::-1]
        F_ar = rfft(ar, cross_size)
        F_w = rfft(w, cross_size)

        gda = 2.0 * cross[n-1:2*n-1]
        # Gradient w.r.t. u: da_j/du_k = a_j*(delta_jk - a_k)
        gu = pp * (gda - float(pp @ gda))
        gu = np.where(np.isfinite(gu), gu, 0.0)
        return float(f), gu
    
    # Multi-stage temperature reduction
    best_p = p.copy()
    best_f = float(_score_fft(p))
    
    stages = [
        (0.20, 0.10),
        (0.06, 0.15),
        (0.018, 0.20),
        (0.005, 0.20),
        (0.0015, 0.25),
        (0.0003, 0.10),
    ]
    
    start = time.monotonic()
    avail = max(0.0, stop_time - start - 5.0)
    if avail < 5.0:
        return p
    
    total_w = sum(w for _, w in stages)
    cum = 0.0
    for frac, wt in stages:

        if now > stop_time - 3.0:
            break
        cum += wt
        end_stage = min(stop_time - 2.0, start + avail * cum / total_w)
        if end_stage <= now + 2.0:
            continue
        
        c = conv_fast(p)
        cmax = float(c.max())
        T = max(1e-15, frac * cmax)
        
        # Run L-BFGS for this temperature
        best_u = p_to_u(p)
        max_iter_inner = 30
        stale = 0
        f0, _ = obj(best_u, T)
        if not math.isfinite(f0):
            continue
        best_local_f = f0
        
        while time.monotonic() < end_stage - 1.0:
            opts = {
                "maxiter": 25,
                "gtol": 1e-12,
                "ftol": 1e-16,
                "maxls": 30,
                "maxcor": 30,
                "maxdisp": 0,
            }
            try:

            except Exception:
                break

            best_f = sc
            best_p = p_cur.copy()
            p = p_cur.copy()
            if consider:
