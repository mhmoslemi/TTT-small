def _subgrad_direction(a):
    """Compute subgradient of f(a) = 2n*max(a*a)/(sum a)^2 w.r.t. a, projected to sum-preserving."""
    n = len(a)
    b = _fft_conv(a)
    M = float(b.max())
    S = float(a.sum())
    
    eps = max(1e-14, 1e-10 * M)
    active = np.flatnonzero(b >= M - eps)
    
    # Subgradient of max_b w.r.t. a_i: for each active k, grad[i] += 2*a[k-i] (valid indices)
    grad_max = np.zeros(n)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            idx = k - np.arange(i_lo, i_hi + 1)
            grad_max[i_lo:i_hi+1] += 2.0 * a[idx]
    grad_max /= max(1, len(active))
    
    # d/da_i [2n * M / S^2] = 2n * (grad_max_i * S - 2*M) / S^3
    grad = 2.0 * n * (grad_max * S - 2.0 * M) / (S ** 3)
    
    # Project to sum-preserving (remove mean)
    grad -= np.mean(grad)
    return grad

    if prev_grad is None or prev_step is None:
        return 0.1
    dg = np.dot(grad - prev_grad, grad)
    if abs(dg) < 1e-20:
        return prev_step * 0.8
    s_bb = dg / max(np.dot(grad - prev_grad, grad - prev_grad), 1e-20)
    return np.clip(s_bb, 1e-6, 1.0)

import time

from scipy.optimize import linprog, minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)

def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, n, M, timeout=3.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _line_search(f0, f1, sc0, stop_time=None):
