def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _gradient_step(f, fsize, top_frac=0.3, lr=0.05):
    """One subgradient step using top-m lags."""
    n = len(f)
    F = rfft(f, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    
    M = float(c.max())
    S = float(f.sum())
    
    # Top m lags
    m = max(3, int((2 * n - 1) * top_frac))
    # Use a threshold instead of exact top-m for stability
    thresh = M - 0.01 * M  # 1% below max
    active = np.where(c >= thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    if len(active) > m:
        # Take the actual top m
        top_indices = np.argsort(c)[-m:]
        active = top_indices
    
    # Compute gradient: grad[i] = (2/len(active)) * sum_{k in active} f[k-i] (valid)
    # This is 2/|active| * cross_correlation(f, indicator_active)[i]
    
    indicator = np.zeros(2 * n - 1)
    indicator[active] = 1.0
    
    ind_pad = np.zeros(fsize)
    ind_pad[:2 * n - 1] = indicator
    F_ind = rfft(ind_pad, fsize)
    F_f = rfft(f, fsize)
    
    # cross[i] = sum_j f[j] * indicator[j+i]
    cross = irfft(np.conj(F_f) * F_ind, fsize)[:n]
    
    # Gradient of M (subgradient)
    grad_M = (2.0 / len(active)) * cross
    
    # Gradient of score = 2n*M/S^2
    # d(score)/df = (2n/S^2) * grad_M - (4n*M/S^3)
    grad = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M / (S ** 3))
    
    # Step
    f_new = f - lr * grad
    f_new = np.clip(f_new, 0, None)
    
    # Normalize
    S_new = float(f_new.sum())
    if S_new > 0.01:
        f_new = f_new * (S / S_new)
    else:
        f_new = f.copy()
    
    return f_new, M, S, float(c.min())

def _lp_direction(f, n, M, timeout=5.0):

        return None

    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
            options={"presolve": True, "time_limit": timeout}
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g
