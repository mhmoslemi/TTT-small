    """Find t in [t_lo, t_hi] minimizing max_k (b_orig[k] + b_fh[k]*t + b_hh[k]*t^2)"""
    # Binary search on V
    def feasible(V):
        # Find t in [t_lo, t_hi] such that for all k: b_hh[k]*t^2 + b_fh[k]*t + (b_orig[k]-V) <= 0
        lo = t_lo
        hi = t_hi
        for k in range(len(b_orig)):
            a = b_hh[k]
            b = b_fh[k]
            c = b_orig[k] - V
            if a < 1e-15:
                # Linear: b*t + c <= 0
                if abs(b) < 1e-15:
                    if c > 0:
                        return None
                else:
                    # t <= -c/b (if b>0) or t >= -c/b (if b<0)
                    root = -c / b
                    if b > 0:
                        hi = min(hi, root)
                    else:
                        lo = max(lo, root)
            else:
                # Quadratic: a*t^2 + b*t + c <= 0
                disc = b*b - 4*a*c
                if disc < 0:
                    if a > 0:
                        # Always positive (or zero at one point)
                        if c > 0:
                            return None
                    # If a < 0, always negative outside roots... but a >= 0 since it's (h*h)[k] >= 0
                    # Actually h*h has non-negative values, so a >= 0
                if disc < 0:
                    return None  # a>0, no real roots, always positive
                # Actually: if disc < 0 and a > 0, quadratic is always positive -> infeasible
                # if disc >= 0: roots are (-b ± sqrt(disc)) / (2a)
                # quadratic <= 0 between the roots
                sqrt_d = math.sqrt(max(0, disc))
                r1 = (-b - sqrt_d) / (2*a)
                r2 = (-b + sqrt_d) / (2*a)
                lo = max(lo, r1)
                hi = min(hi, r2)
            if lo > hi + 1e-12:
                return None
        return (lo + hi) / 2
    
    # Binary search on V
    # Lower bound: max of b_orig (at t=0)
    # Upper bound: max of b_orig + b_fh + b_hh (at t=1)
    V_lo = b_orig.max()
    V_hi = (b_orig + b_fh + b_hh).max()
    
    for _ in range(60):
        V_mid = (V_lo + V_hi) / 2
        t = feasible(V_mid)
        if t is not None:
            V_hi = V_mid
            best_t = t
        else:
            V_lo = V_mid
    
    return V_hi, best_t
