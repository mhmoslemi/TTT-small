import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _build_conv_matrix(f, n):
    """A where (A @ g)[k] = (f * g)[k] for full convolution."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(f, n, M, tl=30.0):
    """Solve max sum(g) s.t. (f*g) <= M, g >= 0; return g normalized to sum 1."""
    A = _build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": tl})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-14:
        return None
    return g / gs

                res_c = np.clip(res, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in res_c]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = res_c.copy()
                    best_score = exact
        except Exception:
            pass

    # Phase 3: Polish (28% of time)
    p3_start = max(time.monotonic(), p2_end)
    p3_end = deadline - 8.0
    p3_duration = max(10.0, p3_end - p3_start)

    # 3a: Coordinate descent
    polish_time1 = p3_duration * 0.40
    if time.monotonic() < p3_end - 5.0:
        try:
            best_seq, _ = _coord_polish(best_seq, time.monotonic() + polish_time1, seed=11111)
        except Exception:
            pass

    # 3b: Pair moves
    polish_time2 = p3_duration * 0.30
    if time.monotonic() < p3_end - 5.0:
        try:
            best_seq, _ = _pair_polish(best_seq, time.monotonic() + polish_time2, seed=22222)
        except Exception:
            pass

    # 3c: Another coordinate pass
    polish_time3 = p3_duration * 0.30
    if time.monotonic() < p3_end - 3.0:
        try:
            best_seq, _ = _coord_polish(best_seq, min(time.monotonic() + polish_time3, p3_end), seed=33333)
        except Exception:
            pass

    # Final validation
    final_seq = np.clip(best_seq, 0, 1000)
    final_seq = np.where(np.isfinite(final_seq), final_seq, 0.0)
    if len(final_seq) < 1:
        final_seq = np.array([1.0])

    try:
        exact_final = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        exact_final = math.inf

    if not math.isfinite(exact_final):
        try:
            exact_fb = float(evaluate_sequence([float(x) for x in fb]))
        except Exception:
            exact_fb = math.inf
        if math.isfinite(exact_fb):
            return [float(x) for x in fb]
        return [1.0]

    if exact_final < best_score:
        best_score = exact_final

    return [float(x) for x in final_seq]
