import time

from scipy.optimize import linprog


def _get_parent():

def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_conv_matrix(f, n):
    """Build A such that A @ g = convolve(f, g)"""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f
    return A


def _lp_direction(f, n, M, time_limit=3.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0"""
    A = _build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0.0, None)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def _subgrad_polish(f0, n, stop_time, seed):
    """Subgradient method for polishing"""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = 2.0 * n * float(_fft_conv(f).max())
    
    step = 0.1
    momentum = np.zeros(n)
    mu = 0.8
    no_imp = 0
    max_no_imp = 3000

    for it in range(100000):
        if time.monotonic() > stop_time:
            break
        
        b = _fft_conv(f)
        M = float(b.max())
        
        # Subgradient of max(conv(f,f))
        eps = max(1e-14, 1e-10 * M)
        active = np.flatnonzero(b >= M - eps)
        
        # Gradient of conv(f,f) at each active point
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            idx = np.arange(i_lo, i_hi + 1)
            other = k - idx
            grad[idx] += 2.0 * f[other]
        
        if len(active) > 0:
            grad /= len(active)
        
        # Projected subgradient step on simplex
        # We want to minimize max(conv(f,f)) subject to f >= 0, sum(f) = 1
        # The constraint sum(f) = 1 means we should project onto the simplex
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-16:
            break
        
        momentum = mu * momentum - step * grad / g_norm
        fn = f + momentum
        # Project onto simplex {f >= 0, sum(f) = 1}
        fn = _project_simplex(fn)
        sc = 2.0 * n * float(_fft_conv(fn).max())
        
        if sc < best_sc - 1e-13:
            f = fn
            best_f = f.copy()
            best_sc = sc
            step = min(1.0, step * 1.02)
            no_imp = 0
        else:
            step *= 0.97
            no_imp += 1
            if no_imp > max_no_imp:
                f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
                f = _project_simplex(f)
                momentum = np.zeros(n)
                step = 0.1
                no_imp = 0
                max_no_imp = int(max_no_imp * 1.3)
    
    return best_f, best_sc


def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def _dca_step(f_t, n):
    """One DCA iteration. Returns new f and new t."""
    f = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [f >= 0, cp.sum(f) == 1]
    
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        indices = list(range(i_lo, i_hi + 1))

        g_k = 0.0
        for i in indices:
            g_k += 0.25 * cp.square(f[i] + f[k-i])
        
        # h_k(f) = 1/4 * sum_i (f[i] - f[k-i])^2
        # ∇h_k at f_t:
        # d_j = sum_i [contributions from terms involving f[j]]
        d_k = np.zeros(n)
        for i in indices:
            j = k - i
            # d/d f[j] of 1/4 * (f[i] - f[j])^2 = -1/2 * (f_t[i] - f_t[j])
            # d/d f[i] of 1/4 * (f[i] - f[j])^2 = 1/2 * (f_t[i] - f_t[j])

            d_k[j] -= 0.5 * diff

        constraints.append(g_k - d_k @ f <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    if f.value is None:

    return np.clip(f.value, 0, None) / max(np.sum(np.clip(f.value, 0, None)), 1e-10), t.value

def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

                        best_vi = frac
            else:
                for mult in [0.0, 0.1, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.6, 2.0, 3.0]:
                    f[i] = oi * mult
                    s = f.sum()
                    if s < 0.01:
                        f[i] = oi
                        continue
                    sc = 2.0 * n * float(_fft_conv(f / s).max())
                    if sc < best_local:
                        best_local = sc
                        best_vi = oi * mult
            f[i] = best_vi

        if best_local < best_sc - 1e-14:
            best_sc = best_local
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 8000:
                break

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 30.0

    parent = _get_parent()
    n = len(parent)

    best_seq = parent.copy()
    best_score = _score(parent)
    if not math.isfinite(best_score):
        best_seq = np.ones(n)
        best_score = _score(best_seq)

    def _update(a, sc):
        nonlocal best_seq, best_score
        if not math.isfinite(sc):
            return
        s = a.sum()
        if s < 0.01:
            return
        a_clean = np.clip(np.asarray(a, dtype=np.float64), 0, 1000.0)
        if not np.all(np.isfinite(a_clean)):
            return
        actual = _score(a_clean)
        if actual < best_score - 1e-12:
            best_seq = a_clean.copy()
            best_score = actual

    # Phase 1: Main LP optimization from parent (most time)
    phase1_end = t0 + budget * 0.45
    f_norm = parent / max(parent.sum(), 1e-10)
    try:
        res, sc = _lp_main(f_norm, n, phase1_end, seed)
        _update(res * parent.sum(), sc)
    except Exception:
        pass

    # Phase 2: LP from perturbed initializations (multiple starts)
    phase2_end = t0 + budget * 0.60
    if time.monotonic() < phase2_end:
        n_restarts = 3
        time_per = (phase2_end - time.monotonic()) / n_restarts
        for r in range(n_restarts):
            if time.monotonic() > phase2_end - 5:
                break
            rng_r = np.random.default_rng(seed + 1000 + r * 100)
            sigma = 0.03 + 0.02 * r
            f_r = np.clip(best_seq / max(best_seq.sum(), 1e-10) + rng_r.normal(0, sigma, n), 0, None)
            f_r = f_r / max(f_r.sum(), 1e-10)
            stop_r = min(phase2_end - 5, time.monotonic() + max(20.0, time_per))
            try:
                rr, sr = _lp_main(f_r, n, stop_r, seed + 2000 + r * 100)
                _update(rr * best_seq.sum(), sr)
            except Exception:
                pass

    # Phase 3: Subgradient polish
    phase3_end = t0 + budget * 0.72
    if time.monotonic() < phase3_end:
        f_sg = best_seq / max(best_seq.sum(), 1e-10)
        try:
            res_sg, sc_sg = _subgrad_polish(f_sg, n, phase3_end, seed + 3000)
            _update(res_sg * best_seq.sum(), sc_sg)
        except Exception:
            pass

    # Phase 4: Coordinate descent polish
    phase4_end = t0 + budget * 0.82
    if time.monotonic() < phase4_end:
        try:
            res_cd, sc_cd = _coord_polish(best_seq, phase4_end, seed + 4000)
            _update(res_cd, sc_cd)
        except Exception:
            pass

    # Phase 5: Try alternate lengths
    if time.monotonic() < deadline - 80:
        alt_lengths = [n - 2, n - 1, n + 1, n + 2, n + 4, n - 4]
        alt_lengths = [L for L in alt_lengths if 4 <= L <= 2000]
        time_remaining = deadline - 80 - time.monotonic()
        time_per_alt = max(15.0, time_remaining / max(1, len(alt_lengths)))
        for L in alt_lengths[:4]:
            if time.monotonic() > deadline - 40:
                break
            if time.monotonic() - t0 > budget * 0.88:
                break
            stop_L = min(deadline - 40, time.monotonic() + time_per_alt)
            if stop_L - time.monotonic() < 10:
                break
            # Interpolate best_seq to new length
            init_L = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n), best_seq)
            init_L = init_L / max(init_L.sum(), 1e-10)
            try:
                rL, sL = _lp_main(init_L, L, stop_L, seed + 5000 + L)
                _update(rL * best_seq.sum(), sL)
            except Exception:
                pass

    # Phase 6: Final coordinate polish on best
    if time.monotonic() < deadline - 10:
        try:
            res_final, sc_final = _coord_polish(best_seq, deadline - 10, seed + 6000)
            _update(res_final, sc_final)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        return [1.0]

    return [float(x) for x in best_seq]
