    c1 = n_p * 0.25
    c2 = n_p * 0.75
    w = n_p * 0.1
    inits.append(np.exp(-0.5 * ((i_arr - c1) / w) ** 2) +
                 np.exp(-0.5 * ((i_arr - c2) / w) ** 2))

    # Random smoothed
    raw = rng.uniform(0.5, 1.5, n_p)
    kernel = np.exp(-0.5 * (np.arange(-10, 11) / 3.0) ** 2)
    kernel /= kernel.sum()
    smoothed = np.convolve(raw, kernel, mode='same')
    inits.append(np.clip(smoothed, 0.01, None))

    # Phase 1: LP optimization on diverse inits (55% of time)
    phase1_end = t0 + budget * 0.55
    time_per_init = max(15.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        if time.monotonic() > stop - 3.0:
            continue
        try:
            res, sc = _optimize_lp(init, stop, seed + 1000 * (idx + 1), peak_frac=0.05)
            _update_best(res)
        except Exception:
            pass

    # Second LP pass on best with tighter peak fraction
    if time.monotonic() < phase1_end - 20.0:
        stop = phase1_end - 5.0
        try:
            res, sc = _optimize_lp(best_seq, stop, seed + 777777, peak_frac=0.02)
            _update_best(res)
        except Exception:
            pass

    # Phase 2: Subgradient refinement (12% time)
    phase2_end = t0 + budget * 0.67
    if time.monotonic() < phase2_end - 10.0:
        stop = min(phase2_end - 5.0, time.monotonic() + (phase2_end - time.monotonic() - 5.0))
        try:
            res, sc = _optimize_subgrad(best_seq, stop, seed + 55555)
            _update_best(res)
        except Exception:
            pass

    # Phase 3: Coordinate descent polish (18% time)
    phase3_end = t0 + budget * 0.85
    if time.monotonic() < phase3_end - 10.0:
        stop = min(phase3_end - 5.0, time.monotonic() + (phase3_end - time.monotonic() - 5.0))
        try:
            res, sc = _polish(best_seq, stop, seed + 44444)
            _update_best(res)
        except Exception:
            pass

    # Phase 4: Second LP round on polished best (5% time)
    phase4_end = t0 + budget * 0.90
    if time.monotonic() < phase4_end - 10.0:
        stop = min(phase4_end - 5.0, time.monotonic() + (phase4_end - time.monotonic() - 5.0))
        try:
            res, sc = _optimize_lp(best_seq, stop, seed + 333333, peak_frac=0.01)
            _update_best(res)
        except Exception:
            pass

    # Phase 5: Coordinate polish again (3% time)
    phase5_end = t0 + budget * 0.93
    if time.monotonic() < phase5_end - 10.0:
        stop = min(phase5_end - 5.0, time.monotonic() + (phase5_end - time.monotonic() - 5.0))
        try:
            res, sc = _polish(best_seq, stop, seed + 22222)
            _update_best(res)
        except Exception:
            pass

    # Phase 6: Try nearby lengths (6% time)
    phase6_end = deadline + 5.0
    if time.monotonic() < phase6_end - 25.0:
        lengths_to_try = []
        for delta in (-32, -16, 16, 32, -48, 48, -64, 64):
            L = n_p + delta
            if 100 <= L <= 800 and L not in lengths_to_try:
                lengths_to_try.append(L)
        lengths_to_try.sort(key=lambda L: abs(L - n_p))

        for L in lengths_to_try[:4]:
            if time.monotonic() > phase6_end - 10.0:
                break
            x_old = np.linspace(0, 1, n_p)
            x_new = np.linspace(0, 1, L)
            init_L = np.interp(x_new, x_old, parent)
            stop = min(phase6_end - 5.0, time.monotonic() + 20.0)
            try:
                res, sc = _optimize_lp(init_L, stop, seed + L * 100, peak_frac=0.05)
                if time.monotonic() < stop - 3.0:
                    res2, sc2 = _polish(res, stop, seed + L * 200)
                    res = res2
                _update_best(res)
            except Exception:
                pass

    # Final polish with remaining time
    remaining = deadline - time.monotonic()
    if remaining > 20.0:
        try:
            res, sc = _polish(best_seq, deadline - 5.0, seed + 11111)
            _update_best(res)
        except Exception:
            pass
        # Quick LP on final best
        if deadline - time.monotonic() > 15.0:
            try:
                res, sc = _optimize_lp(best_seq, deadline - 5.0, seed + 99999, peak_frac=0.02)
                _update_best(res)
            except Exception:
                pass
            if deadline - time.monotonic() > 10.0:
                try:
                    res, sc = _polish(best_seq, deadline - 3.0, seed + 88888)
                    _update_best(res)
                except Exception:
                    pass

    # Final validation with exact evaluator
    final_seq = [float(x) for x in best_seq]
    try:
        final_sc = float(evaluate_sequence(final_seq))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc) or final_sc > 2.0:
        # Fallback to parent
        parent_list = [float(x) for x in parent]
        try:
            psc = float(evaluate_sequence(parent_list))
            if math.isfinite(psc):
                return parent_list
        except Exception:
            pass
        return [1.0] * 10

    return final_seq
