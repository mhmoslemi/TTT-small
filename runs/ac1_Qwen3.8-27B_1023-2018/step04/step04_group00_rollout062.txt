```python
    conv_len = 2 * n - 1
```

```python
        np.maximum(c, 0.0, out=c)
```

```python
            w[int(np.argmax(c))] = 1.0
            f = cm
        else:
            w = ex / se
            f = cm + T * math.log(se / clen)
        q = xf(pp, w)
        gd = 2.0 * q
        t = float(pp @ gd)
        gr = pp * (gd - t)
        return float(f), np.where(np.isfinite(gr), gr, 0.0)
    
    c0 = cf(p)
    cm0 = float(c0.max())
    if not np.isfinite(cm0) or cm0 <= 0:
        return p
    
    if fine:
        fracs = [0.001, 0.0003, 0.0001, 0.00003]
    else:
        fracs = [0.10, 0.04, 0.012, 0.004, 0.001, 0.0003, 0.0001]
    
    # Time allocation
    ns = len(fracs)
    if fine:
        alloc = np.array([0.25, 0.30, 0.25, 0.20])
    else:
        alloc = np.array([0.06, 0.10, 0.14, 0.16, 0.16, 0.16, 0.22])
    alloc /= alloc.sum()
    
    t_start = time.monotonic()
    t_avail = max(5.0, stop_time - t_start - 8.0)
    
    for i in range(ns):
        t_now = time.monotonic()
        t_end = t_start + float(alloc[:i+1].sum()) * t_avail
        t_end = min(t_end, stop_time - 5.0)
        if t_end < t_now + 3.0:
            continue
        T = max(fracs[i] * cm0, 1e-30)
        
        while time.monotonic() < t_end - 2.0:
            maxit = 25 if n > 2500 else 40
            try:
                res = minimize(lf, u, args=(T,), method='L-BFGS-B',
                              jac=True, options={'maxiter': maxit, 'gtol': 1e-12,
                                               'ftol': 1e-15, 'maxls': 30,
                                               'maxcor': 25, 'disp': False})
            except Exception:
                break
            if res.x is None:
                break
            xx = np.asarray(res.x, dtype=np.float64)
            if xx.size != n or not np.all(np.isfinite(xx)):
                break
            u = xx
            p_new = _softmax(u)
            sn = float(p_new.sum())
            if sn > 1e-300:
                p_new = p_new / sn
                p = p_new
                if consider:
                    try: consider(p_new)
                    except: pass
            if getattr(res, 'status', 1) == 0:
                break
    
    p = np.maximum(p, 0)
    p /= max(p.sum(), 1e-300)
    return p
```
