    except Exception:

def _conv_full(a):
    n = len(a)
    N = next_fast_len(2*n - 1)
    F = rfft(a, N)
    c = irfft(F * F, N)[:2*n-1]
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_conv_matrix(a, n):
    """Build matrix A where (A @ b)[k] = (a*b)[k] for k=0..2n-2."""
    A = np.zeros((2*n-1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2*n - 1)
        A[lo:hi, j] = a[:hi-lo]
    return A

def _lp_step(a, n, c_vec=None):
    """Given a, find b minimizing max(a*b) s.t. b>=0, sum(b)=1.
    Returns (b, max_val)."""
    A = _build_conv_matrix(a, n)
    # Minimize t
    # Subject to: A @ b - t <= 0, sum(b) = 1, b >= 0
    # Variables: [b_0..b_{n-1}, t]

    c[-1] = 1.0  # minimize t

    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # t is free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.status != 0:
        return None, None
    b = res.x[:n]
    t = res.x[-1]
    return b, t

            #   Δconv[2i] = (x-v)(x+v) for 0≤2i≤2n-2
            #   (Note: Δconv[2i] does NOT include the 2(x-v)*f[i] term, because the j=i

            # For k=2i, 2(x-v)*f[k-i] = 2(x-v)*f[i]. These are different!

                if S_new < 0.01:
                    return math.inf

                if abs(w) < 1e-15:
                    m = conv.max()
                else:

                    # For speed, don't copy the whole array
                    old_vals = {}
                    base_idx = list(range(i, min(i+n, 2*n-1)))
                    if 2*i < 2*n-1 and 2*i >= 0:
                        old_vals[2*i] = conv[2*i]
                    for k in base_idx:
                        conv[k] += 2.0 * w * f[k-i]
                    if 0 <= 2*i < 2*n-1:
                        conv[2*i] += w * (w + 2.0 * f[i])  # total: 2w*f[i] + w(w+2f[i]) = w² + 4w*f[i]? No...
                        # Wait: we already added 2w*f[i] to conv[2i] from the loop above.
                        # We need the total change to be (x-v)(x+v) = w(w+2v) = w²+2wv.
                        # We already added 2w*f[i] = 2wv.
                        # So we need to add w² more. But wait, that's not right either.
                        # Δconv[2i] should be w(w+2v).
                        # We added 2wv from the loop.
                        # We need to add w².
                        # But w² = (x-v)².

                    m = conv.max()

                        conv[k] -= 2.0 * w * f[k-i]
                    if 0 <= 2*i < 2*n-1:
                        conv[2*i] = old_vals[2*i]
                return 2.0 * n * m / (S_new * S_new)
            
            # Golden section search
            gr = (math.sqrt(5) - 1) / 2
            a_lo, a_hi = 0.0, max(5.0 * v, 0.5)
            a_hi = min(a_hi, 20.0)
            c_ = a_hi - gr * (a_hi - a_lo)
            d_ = a_lo + gr * (a_hi - a_lo)
            fc, fd = obj_at(c_), obj_at(d_)
            for _ in range(25):
                if fc < fd:

                    c_ = a_hi - gr * (a_hi - a_lo)
                    fc = obj_at(c_)

                    a_lo, c_, fc = c_, d_, fd
                    d_ = a_lo + gr * (a_hi - a_lo)
                    fd = obj_at(d_)
            
            x_opt = (a_lo + a_hi) / 2
            sc_opt = obj_at(x_opt)
            
            if sc_opt < best_score - 1e-13:
                w = x_opt - v
                f[i] = x_opt
                S += w
                # Update conv permanently
                base_idx = list(range(i, min(i+n, 2*n-1)))
                for k in base_idx:
                    conv[k] += 2.0 * w * f_old[k-i]  # Need old f!
                if 0 <= 2*i < 2*n-1:
                    conv[2*i] += w * w
                best_score = sc_opt
                best_f = f.copy()
                improved = True
        
        if not improved and sweep > 3:
            break
    
    # Final check
    sc = _score(f)
    if sc < best_score:
        best_f, best_score = f.copy(), sc
    return best_f, best_score

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(400, dtype=np.float64)
    if p.size == 0:
        p = np.ones(400, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    return p

def _conv_full(a):
    n = len(a)
    if n <= 256:
        return np.convolve(a, a, 'full')
    N = next_fast_len(2*n-1)
    F = rfft(a, N)
    return irfft(F*F, N)[:2*n-1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m <= 0:
        return math.inf
    return 2.0 * n * m / (s*s)

def _lp_dir(f, n, M, top_k=None):
    """LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    conv_len = 2*n - 1
    if top_k is None or top_k >= conv_len:
        # Build full A
        A = np.zeros((conv_len, n))
        for j in range(n):
            A[j:j+n, j] = f
        b_ub = np.full(conv_len, M)
    else:
        # Only top K constraints
        # Approximate: use the lags where f shifted is largest
        # f*g at lag k is sum_j f[j]*g[k-j]
        # The "width" of the constraint is n (f has n entries)
        # Include the center and nearby lags
        center = n - 1
        idx = [center + d if 0 <= center + d < conv_len else center - d
               for d in range(top_k) if 0 <= center + d < conv_len]
        idx = list(dict.fromkeys(idx))[:top_k]
        A = np.zeros((len(idx), n))
        for ii, k in enumerate(idx):
            lo = max(0, k-n+1)
            hi = min(n, k+1)
            A[ii, lo:hi] = f[k-hi+1:k-lo+1][::-1]
        b_ub = np.full(len(idx), M)
    
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)]*n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 3.0})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    gs = g.sum()
    if gs < 1e-12:
        return None
    return g / gs

def _lp_optimize(f0, n, stop_time, seed, max_iters=20000):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f /= fs
    
    best_f = f.copy()
    best_score = _score(f)
    stall = 0
    t = 0.5  # initial step
    
    for it in range(max_iters):
        if time.monotonic() > stop_time - 3:
            break
        
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        g = _lp_dir(f, n, M)
        if g is None:
            stall += 3
            if stall > 80:
                break
            f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
            f /= f.sum()
            continue
        
        diff = g - f
        dnorm = float(np.linalg.norm(diff))
        if dnorm < 1e-10:
            stall += 5
            if stall > 80:
                break
            f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
            f /= f.sum()
            continue
        
        # Line search: bisection on t in [0, 1]
        sc0 = _score(f)
        best_t = 0.0
        best_sc = sc0
        lo, hi = 0.0, 2.0
        prev_good = sc0
        for _ in range(16):
            mid = (lo + hi) / 2
            fn = np.clip(f + mid * diff, 0, None)
            sn = fn.sum()
            if sn < 1e-10:
                hi = mid
                continue
            fn /= sn
            sc = _score(fn)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_t = mid
                lo = mid
            else:
                hi = mid
        
        if best_t > 1e-4:
            fn = np.clip(f + best_t * diff, 0, None)
            fn /= fn.sum()
            f = fn
            if best_sc < best_score - 1e-12:
                best_score = best_sc
                best_f = f.copy()
            stall = 0
        else:
            stall += 1
            if stall > 40:
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
                f /= f.sum()
                stall = 0
    
    return best_f, best_score

def _grad_optimize(f0, n, stop_time, seed):
    """Gradient descent on log-softmax surrogate."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f /= f.sum()
    best_f = f.copy()
    best_score = _score(f)
    
    alpha = 0.02
    tau = 40.0
    N = next_fast_len(2*n-1)
    L = 2*n-1
    
    for it in range(100000):
        if time.monotonic() > stop_time - 3:
            break
        
        b = _conv_full(f)
        S = float(f.sum())
        M = float(b.max())
        if M < 1e-18:
            break
        score = 2.0 * n * M / (S*S)
        
        if score < best_score - 1e-13:
            best_score = score
            best_f = f.copy()
        
        # Softmax weight
        bt = tau * (b - M)
        bt -= bt.max()
        p = np.exp(bt)
        ps = p.sum()
        if ps < 1e-30:
            p = np.zeros(L); p[np.argmax(b)] = 1.0; ps = 1.0
        p /= ps
        
        # grad of smoothed max: (p * f_rev)[i+n-1] * 2
        f_rev = f[::-1]
        P = rfft(p, N)
        Fr = rfft(f_rev, N)
        cv = irfft(P*Fr, N)[:L]
        grad_max = 2.0 * cv[n-1:2*n-1]
        
        sg = (2.0*n/(S*S)) * (grad_max - 2.0*M/S)
        
        raw = f - alpha * sg
        raw = np.maximum(raw, 0)
        raw /= raw.sum()
        
        sc_new = _score(raw)
        if sc_new < score - 1e-14:
            f = raw
            alpha = min(0.05, alpha * 1.01)
            tau = max(3.0, tau * 0.9995)
        else:
            alpha = max(1e-7, alpha * 0.5)
            if alpha < 1e-6:
                f = np.clip(f + rng.normal(0, 0.003, n), 0, None)
                f /= f.sum()
                alpha = 0.01
                tau = 40.0
    
    return best_f, best_score
