import numpy as np
import time
from scipy.optimize import minimize, linprog


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * len(a) * c.max() / (s * s))


def _softmax(b):
    b = b - b.max()
    e = np.exp(b)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(b) / len(b)
    return e / s


def _smooth_max_opt(a0, tau_list, max_iter_each=300, t_deadline=None):
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1e-300)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    b = np.log(a)

    for tau in tau_list:
        if t_deadline is not None and time.monotonic() > t_deadline:
            break

        def obj(b_in):
            ai = _softmax(b_in)
            cv = np.convolve(ai, ai)
            m = cv.max()
            sh = tau * cv - tau * m
            return (tau * m + np.log(np.sum(np.exp(sh)))) / tau

        def grad(b_in):
            ai = _softmax(b_in)
            cv = np.convolve(ai, ai)
            m = cv.max()
            sh = tau * cv - tau * m
            p = np.exp(sh)
            ps = p.sum()
            if ps < 1e-300:
                return np.zeros(n)
            p = p / ps
            t1 = np.correlate(p, ai, mode='valid')
            cp = np.concatenate([[0.0], np.cumsum(p)])
            t2 = ai * (cp[n:2 * n] - cp[0:n])
            ga = t1 + t2
            return ai * (ga - np.dot(ai, ga))

        try:
            res = minimize(obj, b, jac=grad, method='L-BFGS-B',
                           options={'maxiter': max_iter_each, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxls': 50})
            if np.isfinite(res.fun):
                b = np.asarray(res.x, dtype=np.float64)

        except Exception:
            pass

    a = np.maximum(a, 0.0)
    s2 = a.sum()
    if s2 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s2
    return a


def _build_conv_matrix(a):
    n = len(a)
    k = np.arange(2 * n - 1)[:, None].astype(np.int64)
    j = np.arange(n)[None, :].astype(np.int64)
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[d[mask]]
    return A


def _lp_solve(a, M, time_limit=8.0):
    n = len(a)
    A = _build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-14,
                               'primal_feasibility_tolerance': 1e-14})
        if res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0.0, None)
            return g
    except Exception:
        pass
    return None

def _gradient_step(a, k_star, lr_list):
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    for j in range(lo, hi):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] += a[idx]
            grad[j] += a[j]
    # Remove mean to project onto sum-preserving subspace
    grad = grad - grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-16:
        return None
    grad = grad / gn
    for lr in lr_list:
        cand = a - lr * grad
        cand = np.maximum(cand, 0.0)
        cs = cand.sum()
        if cs < 1e-15:
            continue
        cand = cand / cs
        return cand
    return None
