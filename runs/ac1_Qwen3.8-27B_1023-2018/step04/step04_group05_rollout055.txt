    n = len(a)
    if n <= 600:
        c = np.convolve(a, a, mode="full")
    else:
        fsize = next_fast_len(2 * n - 1)
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2 * n - 1]
        np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_direction(f, n, M, timeout=10.0):

        return None
    A = _build_A(f, n)
    c = -np.ones(n)

    if res.status != 0 or res.x is None:

def _subgrad_polish(a, stop_time, seed=0):
    n = len(a)
    tS = math.sqrt(2.0*n)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    S = best.sum()
    if S < 1e-10: return best, math.inf
    best = best/S*tS
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    step = 0.005 * tS / n
    no_imp = 0
    
    while time.monotonic() < stop_time - 3.0 and no_imp < 300:
        b = _conv_full(best)
        M = float(b.max())
        k* = int(b.argmax())
        S_cur = float(best.sum())
        
        # Negative subgradient direction
        d = np.full(n, M/S_cur)  # default: M/S
        # For i where k*-i is in [0, n): d[i] = M/S - best[k*-i]
        i_min = max(0, k* - (n-1))
        i_max = min(n-1, k*)
        for i in range(i_min, i_max+1):
            d[i] = M/S_cur - best[k* - i]
        
        # Normalize
        d = d - d.mean()
        dn = np.linalg.norm(d)
        if dn < 1e-15:
            no_imp += 1; step *= 0.9
            continue
        d /= dn
        
        # Try step with multiple scales
        improved = False
        for scale in [step, step*0.5, step*2, step*0.1, step*5]:
            if scale > 0.5*tS: scale = 0.5*tS
            bn = best + scale * d
            bn = np.clip(bn, 0, None)
            fs = bn.sum()
            if fs < 0.01: continue
            bn = bn/fs*tS
            sc = _score_fast(bn)
            if sc < best_sc - 1e-14:
                best = bn; best_sc = sc
                improved = True; step = min(step*1.05, 0.1*tS/n)
                no_imp = 0
                break
        if not improved:
            no_imp += 1; step *= 0.85
    
    return best, best_sc

def _coord_descent(a, stop_time, seed=0):

                if nv == orig: continue

                    best_vs = sc; best_v = nv
            best[i] = best_v
            if best_vs < best_sc - 1e-15:

        # Double coordinate
        for _ in range(200):

            if i == j: continue

                for mj in [0, 0.5, 1.5]:
                    if mi == 1 and mj == 1: continue

                    nj = oj*mj if (oj > 1e-15 and mj > 0) else 0
                    best[i], best[j] = ni, nj

                continue
            best[i], best[j] = oi, oj  # reset for next pair

                sc = _score_fast(best)

                    best[k] = old
    
    return best, best_sc

Hmm, the coordinate descent is getting messy. Let me simplify.

Actually, I realize I should think about this more carefully. The score is:
score = 2n * max(f*f) / S^2

This is a ratio. The coordinate descent should work on the *shape* of f, not its scale. Since the score is scale-invariant, changing f_i by a factor changes both the numerator and denominator.

For a single coordinate change: f_i → f_i + δ
- S → S + δ
- The max coefficient might change
- score → 2n * max(f'f') / (S+δ)^2

The effect depends on which coefficient is the max and how it changes.

For the coordinate descent, I should:
1. For each i, try changing f_i by various amounts
2. Evaluate the score
3. Keep the best

This is straightforward but expensive (O(n * n_evals * conv_cost) per round).

For n=352, conv_cost is about 700 multiplications (for np.convolve of length 352). So one score evaluation is ~700 operations. With 352 coordinates and 10 evaluations each, one round is ~3.5M operations, which takes maybe 10-50ms. That's fine.

Let me simplify the coordinate descent:
