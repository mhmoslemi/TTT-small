    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _fw_step(f, n):
    """One Frank-Wolfe step. Returns new f, improvement."""
    b = _conv_full(f)
    k_star = int(np.argmax(b))
    M = float(b[k_star])
    # Gradient direction: 2 * f shifted by k_star
    # grad[j] = 2*f[k_star - j] if 0 <= k_star - j < n
    grad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] = 2.0 * f[idx]
    # Also consider nearby k values for a better subgradient
    # Take top 5 active constraints
    top_k = np.argsort(-b)[:5]
    for k2 in top_k[1:]:
        for j in range(n):
            idx = k2 - j
            if 0 <= idx < n:
                grad[j] += f[idx]  # add f[k2-j] (without the factor of 2 since we're averaging)
    # FW: move toward the vertex with smallest gradient
    j_star = int(np.argmin(grad))
    v = np.zeros(n)
    v[j_star] = 1.0
    return v, M, k_star

def _polish_cd(a, stop_time, seed=42, n_max_passes=100):
    """Coordinate descent polishing."""
    n = len(a)
    best = np.clip(a.copy(), 0, None)
    s = best.sum()
    if s < 1e-10:
        return best, math.inf
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    
    for passes in range(n_max_passes):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = best[i]
            # Try multipliers
            mults = [0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]
            if orig < 1e-10:
                mults = [0.0, 0.01, 0.05, 0.1, 0.5]
            best_i_val = orig
            best_i_sc = best_sc
            for m in mults:
                best[i] = orig * m + (1e-6 if m > 0 and orig < 1e-10 else 0.0)
                if best[i] > 1000:
                    best[i] = 1000
                sc = _score_fast(best)
                if sc < best_i_sc - 1e-15:
                    best_i_val = best[i]
                    best_i_sc = sc
            best[i] = best_i_val
            if best_i_sc < best_sc - 1e-15:
                best_sc = best_i_sc
                improved = True
        if not improved and passes > 3:
            break
    return best, best_sc

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 30.0  # 30s headroom
    
    # Get parent
    try:
        h = globals()["height_sequence_1"]
        parent = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(384, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(384, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    if parent.size == 0:
        parent = np.ones(384, dtype=np.float64)
    
    n = len(parent)
    rng = np.random.default_rng(seed)
    
    # Fallback
    fallback = parent.copy()
    fallback_sc = _score_fast(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(n) / n
        fallback_sc = _score_fast(fallback)
    
    best_seq = fallback.copy()
    best_score = fallback_sc
    
    # Normalize parent to sum=1 for optimization (score is scale-invariant)
    psum = parent.sum()
    if psum > 1e-10:
        f = parent / psum
    else:
        f = np.ones(n) / n
    
    # Phase allocation
    phase1_end = t0 + budget * 0.15  # FW + initial LP
    phase2_end = t0 + budget * 0.55  # Main LP optimization
    phase3_end = deadline - 30.0     # Polish
    
    # === Phase 1: Frank-Wolfe warm start ===
    while time.monotonic() < phase1_end:
        b = _conv_full(f)
        M = float(b.max())
        k_star = int(np.argmax(b))
        
        # Compute gradient from multiple active constraints
        top_k_idx = np.argsort(-b)[:10]
        grad = np.zeros(n)
        for k2 in top_k_idx:
            ks = int(k2)
            for j in range(n):
                idx = ks - j
                if 0 <= idx < n:
                    grad[j] += f[idx]
        # Normalize grad (it's a subgradient of the max)
        # FW: move toward argmin grad
        j_star = int(np.argmin(grad))
        v = np.zeros(n)
        v[j_star] = 1.0
        
        # Line search
        best_t = 0.0
        best_M = M
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_new = (1-t)*f + t*v
            M_new = float(_conv_full(f_new).max())
            if M_new < best_M:
                best_M = M_new
                best_t = t
        
        if best_t > 0:
            f = (1-best_t)*f + best_t*v
            M = float(_conv_full(f)
