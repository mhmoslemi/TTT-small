from scipy.optimize import linprog

def _conv_full(a):
    n = len(a)
    if n <= 512:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _cross_conv(a, b):
    na, nb = len(a), len(b)
    fsize = next_fast_len(na + nb - 1)
    A = rfft(a, fsize)
    B = rfft(b, fsize)
    return irfft(A * B, fsize)[:na + nb - 1]


def _build_tmat(f, n):
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _solve_lp(f, M, n, time_limit=8.0):
    if M < 1e-18:
        return None
    A = _build_tmat(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": time_limit})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-15 or not np.all(np.isfinite(g)):
        return None
    return g / gs


def _score_local(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    conv = _conv_full(a)
    m = float(conv.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (S * S)


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    ws = w.sum()
    if ws < 1e-30:
        return np.ones(n) / n
    return w / ws

    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    f_init = np.clip(parent, 0, None).astype(np.float64)
    f_sum = f_init.sum()
    if f_sum < 0.01:
        f_init = np.ones(400) / 400.0
        f_sum = 1.0
    f_init = f_init / f_sum

    best_seq = f_init.copy()
    try:
        best_score = float(evaluate_sequence([float(x) for x in f_init]))
    except Exception:
        best_score = math.inf

    def _try_update(seq, sc_local):
        nonlocal best_seq, best_score
        if seq is None:
            return
        seq = np.asarray(seq, dtype=np.float64)
        seq = np.clip(seq, 0, 1000)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        if not math.isfinite(sc_local):
            try:
                sc_local = _score_local(seq)
            except Exception:
                return
        if math.isfinite(sc_local) and sc_local < best_score + 1e-12:
            try:
                exact = float(evaluate_sequence([float(x) for x in seq]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-14:
                best_seq = seq.copy()
                best_score = exact

    f_work = f_init.copy()

    lp1_end = t0 + min(650.0, budget * 0.50)
    if time.monotonic() < lp1_end - 10:
        try:
            res, sc = _lp_phase(f_work, n_p, lp1_end, seed)
            _try_update(res, sc)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                f_work = res / res.sum()
        except Exception:
            pass

    lp2_end = min(t0 + budget * 0.65, hard_deadline - 250.0)
    if time.monotonic() < lp2_end - 10:
        rng2 = np.random.default_rng(seed + 1000)
        for trial in range(3):
            if time.monotonic() > lp2_end - 10:
                break
            if best_score < 1.5028:
                break
            f_pert = np.clip(f_work + rng2.normal(0, 0.02, n_p), 0, None)
            fps = f_pert.sum()
            if fps < 0.01:
                f_pert = f_work.copy()
            else:
                f_pert = f_pert / fps
            sub_end = min(lp2_end - 5, time.monotonic() + 80.0)
            try:
                res, sc = _lp_phase(f_pert, n_p, sub_end, seed + 2000 + trial)
                _try_update(res, sc)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res_norm = res / res.sum()
                    sc_r = _score_local(res_norm)
                    sc_w = _score_local(f_work)
                    if sc_r < sc_w:
                        f_work = res_norm
            except Exception:
                pass

    grad_end = min(t0 + budget * 0.80, hard_deadline - 150.0)
    if time.monotonic() < grad_end - 10:
        try:
            res, sc = _grad_phase(best_seq, grad_end, seed + 3000)
            _try_update(res, sc)
        except Exception:
            pass

    cd_end = min(t0 + budget * 0.92, hard_deadline - 50.0)
    if time.monotonic() < cd_end - 5:
        try:
            res, sc = _coord_polish(best_seq, cd_end, seed + 4000, max_sweeps=200)
            _try_update(res, sc)
        except Exception:
            pass

    g2_end = min(hard_deadline - 25.0, cd_end + 60.0)
    if time.monotonic() < g2_end - 10 and best_score > 1.503:
        try:
            res, sc = _grad_phase(best_seq, g2_end, seed + 5000)
            _try_update(res, sc)
        except Exception:
            pass

    cd2_end = hard_deadline - 12.0
    if time.monotonic() < cd2_end - 3:
        try:
            res, sc = _coord_polish(best_seq, cd2_end, seed + 6000, max_sweeps=50)
            _try_update(res, sc)
        except Exception:
            pass

    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        fb = f_init.copy()
        try:
            fc = float(evaluate_sequence([float(x) for x in fb]))
            if math.isfinite(fc):
                return [float(x) for x in fb]
        except Exception:
            pass
        return [1.0]

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        fb = f_init.copy()
        try:
            fc = float(evaluate_sequence([float(x) for x in fb]))
            if
