One approach: use Nesterov's accelerated gradient or heavy-ball method on the smoothed objective.

Actually, let me try a different angle. The problem is equivalent to:
min_a ||a * a||_∞ / (Σa)² × 2n

If I normalize Σa = 1, it's just 2n * ||a*a||_∞.

An alternative formulation: Let b_k = (a*a)[k]. We want min max_k b_k subject to b = a*a, a >= 0, Σa = 1.

The constraint b = a*a means b_k = Σ_j a_j * a_{k-j}. This is a set of quadratic equations.

I think the most practical approach is:
1. Use the LP best-response with a good line search
2. Add a smoothed gradient-based polish at the end
3. Try a few different initializations and lengths
4. Be careful about the time budget

Let me now write the actual code. I'll focus on:
1. Efficient LP solving (pre-build the matrix structure, only update the values)
2. Adaptive active set (use only near-max constraints)
3. Good line search
4. Smooth gradient polish
5. Multiple restarts

Actually, for the active set approach: if I only use constraints where (a*a)[k] >= M * (1 - δ) for some small δ, the LP is smaller and might give a better direction (focused on the binding constraints).

Let me also think about the "tall" vs "wide" aspect. The convolution has 2n-1 values. The peak is typically at k = n-1 (the center). The constraints at the center are the most binding. But off-center constraints can also be important.

import numpy as np
import time
from scipy.optimize import linprog

def _eval_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)

def _get_parent():
    try:
        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(201, dtype=np.float64)
    p = np.clip(p, 0, 1000)
    if p.size == 0 or not np.all(np.isfinite(p)):
        p = np.ones(201, dtype=np.float64)
    return p

def _build_conv_mat(n, a):
    """Build (2n-1) x n convolution matrix."""
    M = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        M[j:j+n, j] = a
    return M

def _subgradient(a, k_star):
    """Subgradient of (a*a)[k_star] w.r.t. a."""
    n = a.size
    grad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] = 2.0 * a[idx]
    return grad

def _smooth_grad(a, beta=200.0):
    """Gradient of log-sum-exp approximation."""
    n = a.size
    c = np.convolve(a, a)
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    exp_sum = exp_c.sum()
    if exp_sum < 1e-300:
        return np.zeros(n), c_max
    w = exp_c / exp_sum
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n, k + 1)
        for j in range(j_min, j_max):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += 2.0 * w[k] * a[idx]
    return grad, c_max

def _optimize(a0, stop_time, seed=42):
    """Main optimization loop."""
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / s
    
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0
    
    while time.monotonic() < stop_time:
        c = np.convolve(a, a)
        M = float(c.max())
        k_star = int(c.argmax())
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 80:
            # Escape
            sigma = min(0.02 + 0.005 * (stuck // 80), 0.25)
            mode = stuck % 3
            if mode == 0:
                a *= np.exp(rng.normal(0, sigma, n))
            elif mode == 1:
                x = np.arange(n, dtype=np.float64)
                f = rng.uniform(1, 5)
                a *= (1 + sigma * np.sin(2*np.pi*f*x/n + rng.uniform(0, 6.28)))
            else:
                # Random coordinate perturbation
                idx = rng.choice(n, size=max(n//5, 5), replace=False)
                a[idx] *= np.exp(rng.normal(0, sigma*2, len(idx)))
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            stuck = 0
            continue
        
        # Build LP matrix
        M_mat = _build_conv_mat(n, a)
        
        # Active set: use constraints near the max
        c_vals = c  # already computed
        threshold = M * 0.95
        active = np.where(c_vals >= threshold)[0]
        if len(active) < 3:
            active = np.where(c_vals >= M * 0.8)[0]
        if len(active) < 2:
            active = np.arange(2*n-1)
        
        # Solve LP on active constraints
        A_act = M_mat[active]
        c_obj = -np.ones(n)
        b_act = np.full(len(active), M * (1 + 1e-10))
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c_obj, A_ub=A_act, b_ub=b_act, bounds=bounds,
                         method='highs', options={'time_limit': 15.0})
        except Exception:
            res = None
        
        made_move = False
        if res is not None and res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0, None)
            gs = g.sum()
            if gs > 1e-12:
                g = g / gs
                if np.max(np.abs(g - a)) > 1e-10:
                    # Line search
                    best_t = 0.0
                    best_val = M
                    for t in np.linspace(0.05, 1.0, 20):
                        cand = (1-t)*a + t*g
                        cand = np.maximum(cand, 0)
                        cs = cand.sum()
                        if cs < 1e-15: continue
                        cand /= cs
                        val = float(np.convolve(cand, cand).max())
                        if val < best_val:
                            best_val = val
                            best_t = t
                    if best_t > 0:
                        lo, hi = max(0, best_t-0.08), min(1, best_t+0.08)
                        for _ in range(40):
                            mid = (lo+hi)/2
                            cand = (1-mid)*a + mid*g
                            cand = np.maximum(cand, 0)
                            cs = cand.sum()
                            if cs < 1e-15:
                                hi = mid; continue
                            cand /= cs
                            val = float(np.convolve(cand, cand).max())
                            if val < best_val:
                                best_val = val; best_t = mid; hi = mid
                            else:
                                lo = mid
                        a = (1-best_t)*a + best_t*g
                        a = np.maximum(a, 0)
                        a /= a.sum()
                        made_move = True
        
        if not made_move:
            # Smooth gradient step (mirror descent)
            grad, c_max = _smooth_grad(a, beta=300)
            gn = np.linalg.norm(grad)
            if gn > 1e-14:
                for alpha in [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]:
                    log_a = np.log(np.maximum(a, 1e-30)) - alpha * grad / gn
                    a_new = np.exp(log_a - log_a.max())
                    a_new /= a_new.sum()
                    val = float(np.convolve(a_new, a_new).max())
                    if val < M * (1 - 1e-10):
                        a = a_new
                        made_move = True
                        break
        
        if not made_move:
            a *= np.exp(rng.normal(0, 0.015, n))
            a = np.maximum(a, 1e-15)
            a /= a.sum()
    
    return best_a, 2.0 * n * best_M

    for j in range(n):
        M[j:j+n, j] = a
    return M

def _smooth_objective_and_grad(w, a_w, beta):
    """Compute log-sum-exp smoothed max and its gradient w.r.t. w (log-parameterization)."""
    n = a_w.size
    c = np.convolve(a_w, a_w)
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    exp_sum = exp_c.sum()
    if exp_sum < 1e-300:
        return c_max, np.zeros(n)
    w_k = exp_c / exp_sum
    obj = c_max + np.log(exp_sum) / beta

    # As derived: grad[j] = 2 * a[j] * (S1[j] - a[j] * S2)
    # where S1[j] = sum_m w_k[m] * a[m-j] and S2 = sum_m w_k[m] * c[m]

    for j in range(n):

    S2 = np.dot(w_k, c)
    grad = 2.0 * a_w * (S1 - a_w * S2)
    return obj, grad

def _polish_with_minimize(a0, stop_time):
    """Polish using scipy minimize on the smoothed objective."""
    n = a0.size
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    w0 = np.log(a)
    
    beta_schedule = [100, 200, 500, 1000]
    
    for beta in beta_schedule:
        if time.monotonic() > stop_time - 10:
            break
        
        def obj(w):
            aw = np.exp(w - w.max())
            aw = aw / aw.sum()
            v, _ = _smooth_objective_and_grad(w, aw, beta)
            return v
        
        def grad(w):
            aw = np.exp(w - w.max())
            aw = aw / aw.sum()
            _, g = _smooth_objective_and_grad(w, aw, beta)
            return g
        
        try:
            result = minimize(obj, w0, jac=grad, method='L-BFGS-B',
                            options={'maxiter': 200, 'gtol': 1e-12,
                                     'maxfun': 5000})
            if result.success or result.fun < obj(w0):
                w0 = result.x
        except Exception:
            pass
    
    a_final = np.exp(w0 - w0.max())
    a_final = a_final / a_final.sum()
    return a_final

    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)

def _conv_max(a):

def _build_A(n, a):
    """Build (2n-1) x n convolution matrix: A[k,j] = a[k-j] if 0<=k-j<n."""
    M = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):

def _smooth_obj_grad(a_w, beta):
    """Compute log-sum-exp smoothed max(conv) and gradient w.r.t. log-parameter."""
    n = a_w.size
    c = np.convolve(a_w, a_w)
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    exp_sum = exp_c.sum()
    if exp_sum < 1e-300:

            aw = np.exp(w_in - w_in.max())
            aw = aw / aw.sum()
            v, _ = _smooth_obj_grad(aw, beta)
