def _lp_dir(f, b, peak_frac, timeout=4.0):

    M = b.max()

        return None

    for r, k in enumerate(pk):

        seg = f[k - i_hi + 1 : k - i_lo + 1]  # f[k-i] for i in [i_lo, i_hi)

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = _score(best_seq)

    def _update(cand):
        nonlocal best_seq, best_score
        if cand is None or len(cand) == 0:
            return
        cand = np.clip(np.asarray(cand, dtype=np.float64).ravel(), 0, 1000)
        if not np.all(np.isfinite(cand)) or cand.sum() < 0.01:
            return
        sc = _score(cand)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq = cand.copy()
            best_score = sc

    rng = np.random.default_rng(seed)
    i_arr = np.arange(n_p, dtype=np.float64)

    # Build diverse initializations
    inits = [parent.copy()]
    sig = 0.07 * max(1.0, float(parent.mean()))
    inits.append(np.clip(parent + rng.normal(0, sig, n_p), 0, None))
    inits.append(np.clip(parent * (1.0 + 0.2 * rng.normal(0, 1, n_p)), 0, None))
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / n_p)))
    sup = np.ones(n_p) * 0.35
    for fr in (1, 2, 3, 5, 7, 11, 13, 17):
        ph = rng.uniform(0, 2 * np.pi)
        sup += 0.09 * np.cos(2.0 * np.pi * fr * i_arr / n_p + ph)
    inits.append(np.abs(sup) + 0.02)
    for w in (n_p * 0.15, n_p * 0.25, n_p * 0.4):
        inits.append(np.exp(-0.5 * ((i_arr - (n_p - 1) / 2.0) / w) ** 2))
    c1, c2 = n_p * 0.25, n_p * 0.75
    ww = n_p * 0.1
    inits.append(np.exp(-0.5 * ((i_arr - c1) / ww) ** 2) + np.exp(-0.5 * ((i_arr - c2) / ww) ** 2))

    # Phase 1: LP optimization from multiple inits (48%)
    phase1_end = t0 + budget * 0.48
    tp_init = max(12.0, (phase1_end - t0) / len(inits))
    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10:
            break
        stop = min(phase1_end - 4, time.monotonic() + tp_init)
        if time.monotonic() > stop - 3:
            continue
        try:
            res, sc = _lp_optimize(init, stop, seed + 1000 * (idx + 1), peak_frac=0.06)
            _update(res)
        except Exception:
            pass

    # Phase 2: Best LP refinement (10%)
    phase2_end = t0 + budget * 0.58
    if time.monotonic() < phase2_end - 10:
        try:
            res, sc = _lp_optimize(best_seq, phase2_end - 4, seed + 12345, peak_frac=0.04)
            _update(res)
        except Exception:
            pass

    # Phase 3: Frequency equalization (8%)
    phase3_end = t0 + budget * 0.66
    if time.monotonic() < phase3_end - 8:
        try:
            res, sc = _freq_equalize(best_seq, phase3_end - 4, seed + 23456)
            _update(res)
        except Exception:
            pass

    # Phase 4: Subgradient (12%)
    phase4_end = t0 + budget * 0.78
    if time.monotonic() < phase4_end - 8:
        try:
            res, sc = _subgrad_optimize(best_seq, phase4_end - 4, seed + 34567)
            _update(res)
        except Exception:
            pass

    # Phase 5: Coordinate polish (12%)
    phase5_end = t0 + budget * 0.90
    if time.monotonic() < phase5_end - 8:
        try:
            res, sc = _polish(best_seq, phase5_end - 4, seed + 45678)
            _update(res)
        except Exception:
            pass

    # Phase 6: Pairwise polish (5%)
    phase6_end = t0 + budget * 0.95
    if time.monotonic() < phase6_end - 8:
        try:
            res, sc = _pairwise_polish(best_seq, phase6_end - 4, seed + 56789)
            _update(res)
        except Exception:
            pass

    # Phase 7: Final LP + polish
    if time.monotonic() < deadline - 15:
        try:
            res, sc = _lp_optimize(best_seq, deadline - 12, seed + 67890, peak_frac=0.02)
            _update(res)
        except Exception:
            pass
    if time.monotonic() < deadline - 8:
        try:
            res, sc = _polish(best_seq, deadline - 5, seed + 78901)
            _update(res)
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
        if not math.isfinite(final_sc):
            raise ValueError("invalid")
    except Exception:
        try:
            fallback_sc = float(evaluate_sequence([float(x) for x in parent]))
            if math.isfinite(fallback_sc):
                return [float(x) for x in parent]
        except Exception:
            pass
        return [1.0] * 8

    return [float(x) for x in best_seq]
