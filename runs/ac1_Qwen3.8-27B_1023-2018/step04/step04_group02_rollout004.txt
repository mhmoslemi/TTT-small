        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)

1. Fix a, solve for g that maximizes sum(g) s.t. a*g <= t, g >= 0 (this is the LP)
2. Update a = (1-t_step)*a + t_step*g

This is like a proximal gradient method with the "gradient" being the LP direction.

def _conv(a):

    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c
