        if M < best_M * (1.0 - 1e-13):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 150:
            n_escapes += 1
            if n_escapes > max_escapes:
                break
            mode = n_escapes % 7
            xx = np.arange(n, dtype=np.float64)
            if mode == 0:
                sigma = 0.03 + 0.02 * (n_escapes // 7)
                a *= np.exp(rng.normal(0, min(sigma, 0.3), n))
            elif mode == 1:
                f = rng.uniform(0.3, 8.0)
                a *= (1.0 + 0.07 * np.sin(2 * np.pi * f * xx / n + rng.uniform(0, 6.28)))
            elif mode == 2:
                a = np.roll(a, int(rng.integers(1, n)))
            elif mode == 3:
                a = 0.85 * a + 0.15 * a[::-1]
            elif mode == 4:
                f1, f2 = rng.uniform(0.5, 5, 2)
                a *= (1.0 + 0.05 * (np.sin(2 * np.pi * f1 * xx / n) + np.cos(2 * np.pi * f2 * xx / n)))
            elif mode == 5:
                a *= (1.0 + 0.04 * rng.standard_normal(n))
            else:
                a *= np.exp(0.12 * rng.standard_normal(n))
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            stuck = 0
            continue

        A_mat = _build_toeplitz(n, a)
        b_vec = np.full(2 * n - 1, M * (1.0 + 1e-11))

        try:
            res = linprog(c_obj, A_ub=A_mat, b_ub=b_vec, bounds=bounds_list,
                          method='highs', options={'time_limit': 3})
        except Exception:
            stuck += 20
            continue

        if res.status != 0 or res.x is None:
            stuck += 20
            continue

        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            stuck += 20
            continue
        g /= gs

        if float(np.max(np.abs(g - a))) < 1e-12:
            stuck += 10
            continue

        c_ag = np.convolve(a, g)
        c_gg = np.convolve(g, g)

        Aq = c_aa - 2.0 * c_ag + c_gg
        Bq = 2.0 * (c_ag - c_aa)
        Cq = c_aa

        def obj_t(t):
            return float((Cq + t * Bq + t * t * Aq).max())

        best_t, best_val = 0.0, M
        for tt in np.linspace(0.03, 1.0, 20):
            v = obj_t(tt)
            if v < best_val:
                best_val = v
                best_t = tt

        if best_t < 0.02:
            stuck += 10
            continue

        lo = max(0.0, best_t - 0.06)
        hi = min(1.0, best_t + 0.06)
        for _ in range(18):
            mid = 0.5 * (lo + hi)
            if obj_t(mid) < best_val:
                best_val = obj_t(mid)
                best_t = mid
                hi = mid
            else:
                lo = mid

        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a /= a.sum()

    return best_a

def _lp_optimize_raised(a0, t_end, seed=42, max_escapes=40):
    """LP optimization with progressive M relaxation for escape."""
    n = a0.size
    a = np.clip(a0.copy(), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0
    n_escapes = 0
    relax = 1.0

    c_obj = -np.ones(n)
    bounds_list = [(0.0, None)] * n

    while time.monotonic() < t_end:
        c_aa = np.convolve(a, a)
        M = float(c_aa.max())

        if M < best_M * (1.0 - 1e-13):
            best_M = M
            best_a = a.copy()
            stuck = 0
            relax = max(0.95, relax * 0.99)
        else:
            stuck += 1

        if stuck > 120:
            n_escapes += 1
            if n_escapes > max_escapes:
                break
            relax = min(1.001, relax * 1.003)
            mode = n_escapes % 5
            xx = np.arange(n, dtype=np.float64)
            if mode == 0:
                a *= np.exp(rng.normal(0, 0.04, n))
            elif mode == 1:
                f = rng.uniform(0.3, 6.0)
                a *= (1.0 + 0.06 * np.sin(2 * np.pi * f * xx / n + rng.uniform(0, 6.28)))
            elif mode == 2:
                a = np.roll(a, int(rng.integers(1, n)))
            elif mode == 3:
                a = 0.8 * a + 0.2 * a[::-1]
            else:
                a *= (1.0 + 0.05 * rng.standard_normal(n))
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            stuck = 0
            continue

        A_mat = _build_toeplitz(n, a)
        M_lp = M * relax
        b_vec = np.full(2 * n - 1, M_lp)

        try:
            res = linprog(c_obj, A_ub=A_mat, b_ub=b_vec, bounds=bounds_list,
                          method='highs', options={'time_limit': 3})
        except Exception:
            stuck += 20
            continue

        if res.status != 0 or res.x is None:
            stuck += 20
            continue

        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            stuck += 20
            continue
        g /= gs

        if float(np.max(np.abs(g - a))) < 1e-12:
            stuck += 10
            continue

        c_ag = np.convolve(a, g)
        c_gg = np.convolve(g, g)
        Aq = c_aa - 2.0 * c_ag + c_gg
        Bq = 2.0 * (c_ag - c_aa)
        Cq = c_aa

        def obj_t(t):
            return float((Cq + t * Bq + t * t * Aq).max())

        best_t, best_val = 0.0, M
        for tt in np.linspace(0.03, 1.0, 16):
            v = obj_t(tt)
            if v < best_val:
                best_val = v
                best_t = tt

        if best_t < 0.02:
            stuck += 10
            continue

        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a /= a.sum()
        relax = max(0.95, relax * 0.999)

    return best_a

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    DEADLINE = t0 + budget - 15.0

    parent = _get_parent()
    n_parent = parent.size

    pn = np.clip(parent, 0, 1000)
    sn = pn.sum()
    if sn < 1e-15:
        pn = np.ones(n_parent) / n_parent
    else:
        pn /= sn

    best_a = pn.copy()
    best_score = _score_fast(pn)

    def try_update(cand):
        nonlocal best_a, best_score
        try:
            cand = np.asarray(cand, dtype=np.float64)
            sc = _score_fast(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = np.clip(cand, 0, 1000).copy()
        except Exception:
            pass

    # Phase 1: Main LP optimization on parent (55% budget)
    t1 = t0 + budget * 0.55
    try:
        r = _lp_optimize(pn, t1, seed=seed, max_escapes=60)
        try_update(r)
    except Exception:
        pass

    # Try palindromic symmetrization
    try_update(0.5 * best_a + 0.5 * best_a[::-1])

    # Phase 2: Multi-start with diverse initializations (25% budget)
    t2 = t0 + budget * 0.80
    xx = np.arange(n_parent, dtype=np.float64)
    inits = []

    for p in [0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0]:
        v = np.maximum(1.0 - xx / max(n_parent - 1, 1), 0.0) ** p
        if v.sum() > 1e-10:
            inits.append(v / v.sum())

    ctr = (n_parent - 1) / 2.0
    for w in [0.15, 0.25, 0.35, 0.5, 0.7]:
        v = np.exp(-0.5 * ((xx - ctr) / (w * n_parent)) ** 2)
        inits.append(v / v.sum())

    for alpha in [0.1, 0.2, 0.4, 0.6]:
        v = np.maximum(0.5 + 0.5 * np.cos(np.pi * xx / max(n_parent - 1, 1) * (1 + alpha)), 0.01)
        inits.append(v / v.sum())

    inits.append(np.ones(n_parent) / n_parent)

    # Harmonic perturbations of parent
    for fr in [1, 2, 3, 4, 5]:
        for amp in [0.05, 0.1, 0.15]:
            v = pn * (1.0 + amp * np.cos(2 * np.pi * fr * xx / max(n_parent - 1, 1)))
            v = np.maximum(v, 1e-8)
            inits.append(v / v.sum())

    # Blends
    for frac in [0.3, 0.5, 0.7]:
        for init in inits[:8:2]:
            v = frac * pn + (1 - frac) * init
            inits.append(v / v.sum())

    rng_main = np.random.default_rng(seed + 99999)
    idx = rng_main.permutation(len(inits))
    inits = [inits[i] for i in idx]

    remaining = max(t2 - time.monotonic(), 10.0)
    per_init = max(4.0, remaining / max(1, len(inits)))

    for i, init in enumerate(inits):
        if time.monotonic() > t2 - 5.0:
            break
        alloc = min(time.monotonic() + per_init, t2 - 3.0)
        try:
            r = _lp_optimize_raised(init, alloc, seed=int(rng_main.integers(0, 10**7)), max_escapes=25)
            try_update(r)
        except Exception:
            pass

    # Phase 3: Final deep polish on best (remaining)
    if time.monotonic() < DEADLINE - 10:
        ab = np.clip(best_a, 0, 1000).copy()
        ab /= ab.sum()
        try:
            r = _lp_optimize(ab, DEADLINE, seed=7777, max_escapes=80)
            try_update(r)
        except Exception:
            pass

        # One more pass with raised version
        if time.monotonic() < DEADLINE - 10:
            ab2 = np.clip(best_a, 0, 1000).copy()
            ab2 /= ab2.sum()
            try:
                r = _lp_optimize_raised(ab2, DEADLINE, seed=4444, max_escapes=40)
                try_update(r)
            except Exception:
                pass

    # Final validation with exact evaluator
    try:
        final_list = [float(x) for x in best_a]
        fs = float(evaluate_sequence(final_list))
        if not np.isfinite(fs):
            raise ValueError("invalid")
    except Exception:
        return [1.0] * 32

    return [float(x) for x in best_a]
