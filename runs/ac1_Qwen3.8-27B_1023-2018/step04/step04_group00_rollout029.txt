        c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        cur_score = new_score
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 2:
                if time.monotonic() < stop_time - 8.0:
                    # Randomized restart
                    sigmas = [0.04, 0.025, 0.015, 0.08]
                    for sigma in sigmas:
                        p_try = best_p * np.exp(rng.normal(0.0, sigma, n, dtype=np.float64))
                        p_try = np.maximum(p_try, 0.0)
                        ss = float(p_try.sum())
                        if ss > 0.0:
                            p_try /= ss
                        sc_try = _score_norm(p_try)
                        if math.isfinite(sc_try) and sc_try < best_score - 1e-14:
                            p = p_try.copy()
                            c = _conv_self(p)
                            S_total = 1.0
                            cur_score = sc_try
                            flat = 0
                            improved = True
                            break
                    if not improved:
                        p = best_p.copy()
                        c = _conv_self(p)
                        S_total = 1.0
                        cur_score = best_score
                else:
                    break
        sweep += 1

    # Final: use best found
    p = np.maximum(best_p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    else:
        p = best_p.copy()
        ss = float(p.sum())
        if ss > 1e-300:
            p = p / ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _subgrad_step(p, n, nsteps=20, lr_base=0.001):
    """Subgradient descent on the max of the autocorrelation."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    Ltot = 2 * n - 1

    for step in range(nsteps):
        c = _conv_self(p)
        cmax = float(c.max())
        # Find active set (lags where c is within 1% of max)
        threshold = cmax * 0.999
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        # Compute subgradient: average gradient over active lags
        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            # grad c_k / d a[j] = 2*a[k-j] if 0 <= k-j <= n-1
            lo_j = max(0, k - n + 1)
            hi_j = min(k, n - 1)
            if hi_j >= lo_j:
                j_range = np.arange(lo_j, hi_j + 1)
                k_minus_j = k - j_range
                grad[j_range] += 2.0 * p[k_minus_j]
        grad /= float(len(active))
        # Remove mean component (to maintain sum=1)
        grad -= float(np.mean(grad))
        # Take step
        lr = lr_base / (1.0 + 0.1 * step)
        p_new = p - lr * grad
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if ss > 1e-300:
            p_new = p_new / ss
        else:
            p_new = p
        # Check if improved
        sc_old = float(2.0 * n * _conv_self(p).max())
        sc_new = float(2.0 * n * _conv_self(p_new).max())
        if sc_new < sc_old:
            p = p_new
        else:
            lr_base *= 0.7
            if lr_base < 1e-6:
                break
    return p


def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / float(n), xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 28.0
    if hard <= t0 + 2.0:
        hard = t0 + max(2.0, budget - 1.0)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64)
        best_arr /= 64.0
        try:
            best_score = float(_score_exact_arr(best_arr))
        except Exception:
            best_score = math.inf

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    # Main phase: focus on the parent length, with some time for other lengths
    n_main = int(best_arr.size)
    if n_main == 0:
        n_main = 384

    # Normalize parent to sum=1 for optimization
    p_main = best_arr / float(best_arr.sum())
    score_main = _score_norm(p_main)

    # Time allocation
    total_time = hard - t0
    # Phase 1: L-BFGS (20%)
    # Phase 2: Coordinate polish (55%)
    # Phase 3: Subgradient + fine polish (20%)
    # Phase 4: Other lengths (5%)
    # Reserve for finalization (5%)

    phase1_end = t0 + 0.18 * total_time
    phase2_end = t0 + 0.70 * total_time
    phase3_end = t0 + 0.90 * total_time
    phase4_end = hard - 10.0

    # Phase 1: L-BFGS
    if time.monotonic() < phase1_end - 5.0:
        try:
            p1 = _lbfgs_phase(p_main, n_main, phase1_end, consider=consider)
            if p1 is not None and p1.size == n_main and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                sc1 = _score_norm(p1)
                if math.isfinite(sc1) and sc1 < score_main:
                    p_main = p1
                    score_main = sc1
                    consider(p_main)
        except Exception:
            pass

    # Phase 2: Intensive coordinate polish with multiple rounds
    if time.monotonic() < phase2_end - 10.0:
        # First round with moderate G
        G1 = 50 if n_main <= 2048 else 45
        try:
            p2, sc2 = _coord_polish(p_main, min(phase2_end - 20.0, time.monotonic() + 0.55 * (phase2_end - time.monotonic())),
                                     consider=consider, seed=int(seed + 777), G=G1)
            if p2 is not None and p2.size == n_main and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p2 = p2 / float(p2.sum())
                sc2 = _score_norm(p2)
                if math.isfinite(sc2) and sc2 < score_main:
                    p_main = p2
                    score_main = sc2
                    consider(p_main)
        except Exception:
            pass

        # Second round with finer G
        if time.monotonic() < phase2_end - 10.0:
            G2 = 70 if n_main <= 2048 else 60
            try:
                p2b, sc2b = _coord_polish(p_main, min(phase2_end - 5.0, time.monotonic() + 0.4 * (phase2_end - time.monotonic())),
                                           consider=consider, seed=int(seed + 999), G=G2)
                if p2b is not None and p2b.size == n_main and np.all(np.isfinite(p2b)) and p2b.sum() > 0:
                    p2b = p2b / float(p2b.sum())
                    sc2b = _score_norm(p2b)
                    if math.isfinite(sc2b) and sc2b < score_main:
                        p_main = p2b
                        score_main = sc2b
                        consider(p_main)
            except Exception:
                pass

    # Phase 3: Subgradient + very fine polish
    if time.monotonic() < phase3_end - 5.0:
        # Subgradient steps
        try:
            p3 = _subgrad_step(p_main, n_main, nsteps=30, lr_base=0.0008)
            sc3 = _score_norm(p3)
            if math.isfinite(sc3) and sc3 < score_main:
                p_main = p3
                score_main = sc3
                consider(p_main)
        except Exception:
            pass

        # Very fine coordinate polish
        if time.monotonic() < phase3_end - 3.0:
            G3 = 90 if n_main <= 2048 else 75
            try:
                p3b, sc3b = _coord_polish(p_main, min(phase3_end - 2.0, time.monotonic() + 0.8 * (phase3_end - time.monotonic())),
                                           consider=consider, seed=int(seed + 31337), G=G3)
                if p3b is not None and p3b.size == n_main and np.all(np.isfinite(p3b)) and p3b.sum() > 0:
                    p3b = p3b / float(p3b.sum())
                    sc3b = _score_norm(p3b)
                    if math.isfinite(sc3b) and sc3b < score_main:
                        p_main = p3b
                        score_main = sc3b
                        consider(p3b)
            except Exception:
                pass

    # Phase 4: Try other lengths (brief)
    if time.monotonic() < phase4_end - 10.0 and n_main > 200:
        other_lengths = []
        if n_main >= 1024:
            other_lengths.append(n_main // 2)
        if n_main <= 4096:
            other_lengths.append(n_main + n_main // 4)  # e.g., 3072 -> 3840
        for ol in other_lengths:
            if time.monotonic() > phase4_end - 15.0:
                break
            if ol < 64 or ol > 8192:
                continue
            try:
                p_ol = _resample_mid(best_arr, ol)
                # Quick coordinate polish
                G_ol = 40
                stop_ol = min(phase4_end - 12.0, time.monotonic() + max(10.0, 0.15 * (phase4_end - time.monotonic())))
                p_ol2, sc_ol = _coord_polish(p_ol, stop_ol, consider=consider, seed=int(seed + ol), G=G_ol)
            except Exception:
                pass

    # Final: squeeze the best
    if time.monotonic() < hard - 15.0 and best_arr.size > 0:
        p_final = best_arr / float(best_arr.sum())
        n_final = int(p_final.size)
        if n_final <= 4096:
            G_final = 80 if n_final <= 2048 else 65
            stop_final = hard - 12.0
            try:
                p_fin, sc_fin = _coord_polish(p_final, stop_final, consider=consider,
                                               seed=int(seed + 271828), G=G_final)
                if p_fin is not None and p_fin.size == n_final and np.all(np.isfinite(p_fin)) and p_fin.sum() > 0:
                    p_fin = p_fin / float(p_fin.sum())
                    consider(p_fin)
            except Exception:
                pass

    # Final validation and return
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        try:
            ls = [float(x) for x in parent]
            if len(ls) > 0 and all(math.isfinite(x) and x >= 0 for x in ls):
                return ls
        except Exception:
            pass
        return [1.0]
