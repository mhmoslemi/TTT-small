- The second factor when $k-j = i$, i.e., $j = k-i$ (if $0 \le k-i < n$): term is $a_{k-i} a_i$

    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0 or p.sum() < 0.01:
        p = np.ones(414, dtype=np.float64)
    return p

def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    n = len(a)
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _lp_minpeak(f, n, timeout=4.0):
    c = np.zeros(n + 1)
    c[n] = 1.0
    rows, cols, vals = [], [], []
    for j in range(n):
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        vals.extend(f)
    rows.extend(range(2*n-1))
    cols.extend([n] * (2*n-1))
    vals.extend([-1.0] * (2*n-1))
    A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    if res.status not in (0, 3):
        return None
    h = np.clip(res.x[:n], 0, None)
    hs = h.sum()
    if hs < 1e-15:
        return None
    return h / hs

def _lp_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(f0, 0, None).copy()
    fs = f.sum()
    f = f / max(fs, 1e-10)
    best_f = f.copy()
    best_sc = _score(f)
    no_imp = 0
    max_no_imp = 60
    t_param = 0.3
    
    for it in range(200000):
        if time.monotonic() > stop_time:
            break
        h = _lp_minpeak(f, n, timeout=3.0)
        if h is None:
            no_imp += 3
            if no_imp > max_no_imp:
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                f = f / f.sum()
                no_imp = 0
            continue
        
        best_t = t_param
        best_sc_try = best_sc
        for tt in [0.05, 0.1, 0.2, t_param, 0.4, 0.6, 0.8, 1.0]:
            tt = min(max(tt, 0.01), 1.0)
            fn = np.clip((1-tt)*f + tt*h, 0, None)
            s2 = fn.sum()
            if s2 < 1e-10:
                continue
            fn = fn / s2
            sc = _score(fn)
            if sc < best_sc_try:
                best_sc_try = sc
                best_t = tt
        
        if best_sc_try < best_sc - 1e-12:
            fn = np.clip((1-best_t)*f + best_t*h, 0, None)
            fn = fn / fn.sum()
            f = fn
            best_f = f.copy()
            best_sc = best_sc_try
            t_param = min(0.9, best_t * 1.2)
            no_imp = 0
        else:
            no_imp += 1
            t_param = max(0.02, t_param * 0.6)
            if no_imp > max_no_imp:
                scale = min(1.0, (no_imp - max_no_imp) * 0.01)
                f = np.clip(f + rng.normal(0, 0.03 + 0.1*scale, n), 0, None)
                f = f / f.sum()
                no_imp = max(0, no_imp - max_no_imp // 2)
    
    return best_f, best_sc

def _softmax_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(f0, 0, None).copy()
    f = f / f.sum()
    best_f = f.copy()
    best_sc = _score(f)
    
    eps0 = max(1e-4, 0.01 * np.max(_fft_conv(f)))
    eps = eps0
    step = 0.5
    no_imp = 0
    
    for it in range(50000):
        if time.monotonic() > stop_time:
            break
        
        b = _fft_conv(f)
        b_max = b.max()
        shifted = b - b_max
        p = np.exp(shifted / eps)
        p_sum = p.sum()
        p = p / p_sum
        
        # Gradient via FFT
        q = p.copy()
        q_flipped = q[::-1]
        fsize = next_fast_len(3*n)
        F = rfft(f, fsize)
        Q = rfft(q_flipped, fsize)
        conv = irfft(F * Q, fsize)
        idx = 2*n-2 - np.arange(n)
        grad = 2.0 * conv[idx]
        
        # Project: remove mean, clip, renormalize
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        grad = grad / gnorm
        
        fn = np.clip(f - step * grad, 0, None)
        s2 = fn.sum()
        if s2 < 1e-10:
            step *= 0.5
            continue
        fn = fn / s2
        sc = _score(fn)
        
        if sc < best_sc - 1e-13:
            f = fn
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            eps = max(1e-6, eps * 0.995)
            step = min(2.0, step * 1.01)
        else:
            step *= 0.9
            no_imp += 1
            if no_imp > 200:
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                f = f / f.sum()
                step = 0.3
                no_imp = 0
                eps = max(eps * 2, 0.01 * eps0)
    
    return best_f, best_sc

def _coord_descent(f0, stop_time, seed, max_rounds=200):
    rng = np.random.default_rng(seed)
    f = np.clip(f0, 0, None).copy()
    n = len(f)
    best_sc = _score(f)
    best_f = f.copy()
    
    for rnd in range(max_rounds):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            if orig < 1e-14:
                # Try turning on
                for mult in [0.01, 0.05, 0.1, 0.5]:
                    f[i] = mult * f.max()
                    if f.sum() < 0.01:
                        f[i] = 0
                        continue
                    sc = _score(f)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        best_f = f.copy()
                        improved = True
                        break
                f[i] = 0
                continue
            
            best_v = orig
            best_v_sc = best_sc
            for mult in [0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.6, 2.0, 0.0]:
                f[i] = orig * mult
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_v_sc - 1e-14:
                    best_v = f[i]
                    best_v_sc = sc
            f[i] = best_v
            if best_v_sc < best_sc - 1e-14:
                best_sc = best_v_sc
                best_f = f.copy()
                improved = True
        
        if not improved:
            break
    
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 30.0
    
    parent = _get_parent()
    n = len(parent)
    p_sum = float(parent.sum())
    
    best_seq = parent.copy()
    best_score = _score(parent)
    
    def _update(arr, sc):
        nonlocal best_seq, best_score
        if sc < best_score - 1e-11:
            a = np.clip(np.asarray(arr, dtype=np.float64), 0, 1000.0)
            if a.sum() >= 0.01 and np.all(np.isfinite(a)):
                best_seq = a.copy()
                best_score = sc
    
    f_init = parent / p_sum
    
    # Phase 1: LP optimization (primary)
    phase1_end = t0 + budget * 0.50
    res, sc = _lp_optimize(f_init, n, phase1_end, seed)
    if math.isfinite(sc):
        _update(res * p_sum, sc)
    
    # Phase 2: Softmax gradient
    phase2_end = t0 + budget * 0.68
    if time.monotonic() < phase2_end:
        f_sg = best_seq / max(best_seq.sum(), 1e-10)
        res_sg, sc_sg = _softmax_optimize(f_sg, n, phase2_end, seed+100)
        if math.isfinite(sc_sg):
            _update(res_sg * p_sum, sc_sg)
    
    # Phase 3: LP again from best
    phase3_end = t0 + budget * 0.80
    if time.monotonic() < phase3_end:
        f_lp2 = best_seq / max(best_seq.sum(), 1e-10)
        res_lp2, sc_lp2 = _lp_optimize(f_lp2, n, phase3_end, seed+200)
        if math.isfinite(sc_lp2):
            _update(res_lp2 * p_sum, sc_lp2)
    
    # Phase 4: Coordinate descent
    phase4_end = deadline - 10
    res_cd, sc_cd = _coord_descent(best_seq, phase4_end, seed+300, max_rounds=500)
    if math.isfinite(sc_cd):
        _update(res_cd, sc_cd)
    
    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf
    
    if not math.isfinite(final_sc):
        return [1.0]
    
    return [float(x) for x in best_seq]
