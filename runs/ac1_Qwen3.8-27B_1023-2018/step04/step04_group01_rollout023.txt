                    # But end-start = min(i+n, n_c) - i = min(n, n_c-i) = n (since n_c=2n-1 > i for i<n)
                    # Actually end = min(i+n, 2n-1), start = i
                    # f_part length = end - start
                    # f[k-i] for k = i, i+1, ..., end-1 is f[0], f[1], ..., f[end-i-1]
                    f_seg = f[0 : end - start]
                    tmp[start:end] += 2.0 * dw * f_seg
                    if 0 <= 2 * i < n_c:
                        tmp[2 * i] += dw * dw
                    m = float(tmp.max())
                return 2.0 * n * m / (S_new * S_new)

            # Golden section / grid search
            best_w = v
            best_obj = best_score  # current best (which equals _obj_1d(v))
            w_grid = np.linspace(0.0, w_max, n_pts)
            for w_val in w_grid:
                obj = _obj_1d(w_val)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_w = w_val

            if best_w != v:
                # Refine
                idx_best = int(round(best_w / w_max * (n_pts - 1)))
                w_lo = max(0.0, w_grid[max(0, idx_best - 2)])
                w_hi = min(w_max, w_grid[min(n_pts - 1, idx_best + 2)])
                w_fine = np.linspace(w_lo, w_hi, 20)
                for w_val in w_fine:
                    obj = _obj_1d(w_val)
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_w = w_val

                if best_obj < best_score - 1e-13:
                    dw = best_w - v
                    f[i] = best_w
                    S += dw
                    b[start:end] += 2.0 * dw * f[0 : end - start]
                    if 0 <= 2 * i < n_c:
                        b[2 * i] += dw * dw
                    # Recompute max
                    bm = float(b.max())
                    best_score = 2.0 * n * bm
                    best_f = f.copy()
                    improved = True

        if not improved and sweep > 5:
            break

    f = np.clip(f, 0, 1000)
    return best_f, _score_internal(best_f)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(60.0, float(budget_s))
    hard_deadline = t0 + budget - 20.0

    parent = _get_parent()
    n = len(parent)

    # Initialize
    f_init = np.clip(parent.astype(np.float64), 0.0, 1000.0)
    f_init = f_init / max(f_init.sum(), 1e-10)

    best_seq = f_init.copy()
    best_score = math.inf
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        best_score = _score_internal(best_seq)
    if not math.isfinite(best_score):
        best_seq = np.ones(n) / n
        best_score = 2.0

    def _update(seq, sc=None):
        nonlocal best_seq, best_score
        seq = np.clip(np.asarray(seq, dtype=np.float64), 0.0, 1000.0)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        if sc is None:
            sc = _score_internal(seq)
        if not math.isfinite(sc):
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = sc
        if math.isfinite(exact) and exact < best_score - 1e-15:
            best_score = exact
            best_seq = seq.copy()

    # === Phase 1: LP-based optimization (50% time) ===
    t1_end = t0 + budget * 0.50
    if time.monotonic() < t1_end - 10:
        try:
            res, sc = _lp_optimize(f_init, n, t1_end, seed + 100)
            _update(res, sc)
        except Exception:
            pass

    # === Phase 2: More LP from best with different seed/perturbations ===
    t2_end = t0 + budget * 0.65
    if time.monotonic() < t2_end - 10 and best_score > 1.503:
        # Perturb best and re-run LP
        rng2 = np.random.default_rng(seed + 200)
        f_pert = np.clip(best_seq + rng2.normal(0, 0.02, n), 0.0, None)
        f_pert /= f_pert.sum()
        try:
            res2, sc2 = _lp_optimize(f_pert, n, t2_end, seed + 300)
            _update(res2, sc2)
        except Exception:
            pass

    # === Phase 3: Gradient-based refinement (15% time) ===
    t3_end = t0 + budget * 0.80
    if time.monotonic() < t3_end - 10:
        try:
            res3, sc3 = _grad_optimize(best_seq, n, t3_end, seed + 400)
            _update(res3, sc3)
        except Exception:
            pass

    # === Phase 4: Coordinate descent polish (12% time) ===
    t4_end = t0 + budget * 0.92
    if time.monotonic() < t4_end - 10:
        try:
            res4, sc4 = _coord_polish(best_seq, t4_end, max_sweeps=80, seed=seed + 500)
            _update(res4, sc4)
        except Exception:
            pass

    # === Phase 5: Final fine polish (8% time) ===
    t5_end = hard_deadline - 10.0
    if time.monotonic() < t5_end - 5:
        try:
            res5, sc5 = _coord_polish(best_seq, t5_end, max_sweeps=40, seed=seed + 600)
            _update(res5, sc5)
        except Exception:
            pass

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = np.ones(n) / n

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        for fb in [f_init, np.ones(n) / n, [1.0]]:
            try:
                s = float(evaluate_sequence([float(x) for x in fb]))
                if math.isfinite(s):
                    return [float(x) for x in fb]
            except Exception:
                continue
        return [1.0]

    return [float(x) for x in best_seq]
