import time
from scipy.optimize import linprog

    a = np.asarray(a, dtype=np.float64)

def _project_simplex(v):
    """Project onto probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u * rho) > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:

    rho_star = idx[-1]
    theta = (cssv[rho_star] - 1.0) / (rho_star + 1.0)
    return np.maximum(v - theta, 0.0)

def _conv(a):
    """Full autocorrelation."""
    return np.convolve(a, a)

def _grad_logsumexp(a, tau):
    """Gradient of log-sum-exp smoothed max of autocorrelation."""
    n = a.size
    c = _conv(a)
    c_max = c.max()
    exponents = (c - c_max) / tau
    # Clip for numerical stability
    exponents = np.clip(exponents, -50, 50)
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        w /= w_sum
    
    # grad_j = 2 * sum_k w_k * a_{k-j} / tau
    corr = np.correlate(w, a, mode='full')
    grad = 2.0 * corr[n - 1: 2 * n - 1] / tau
    return grad, c

def _grad_subgrad(a, top_k=10):
    """Subgradient using weighted top-k active constraints."""
    n = a.size
    c = _conv(a)
    c_max = c.max()
    
    # Get top-k lags
    order = np.argsort(c)[::-1][:top_k]
    # Exponential weights based on proximity to max
    gaps = c_max - c[order]
    w_active = np.exp(-gaps / max(c_max * 0.01, 1e-10))
    w_active /= w_active.sum()
    
    # Build weight vector
    w = np.zeros(2 * n - 1)
    w[order] = w_active
    
    corr = np.correlate(w, a, mode='full')
    grad = 2.0 * corr[n - 1: 2 * n - 1]
    return grad, c

    """
    FISTA with log-sum-exp smoothing for min max(a*a) s.t. sum(a)=1, a>=0.
    """

    a = _project_simplex(a)
    n = a.size
    
    # Initial Lipschitz estimate
    if L_init is None:
        # Rough estimate: ||Hessian|| <= 2 * ||a||_inf * (2n-1) / tau

def _lp_best_response(a, M, n, time_limit=3.0):
    """Solve LP: max sum(g) s.t. A(a) g <= M, g >= 0."""
    # Build convolution matrix
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = a[d[mask]]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    return np.clip(res.x, 0, None)
