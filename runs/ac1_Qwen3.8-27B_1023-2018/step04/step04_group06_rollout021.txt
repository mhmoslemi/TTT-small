    a = np.clip(a, 0, 1000)

        M = c.max()

            stuck += 1
        
        if stuck > 100:

            a /= a.sum()

            a *= np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-10)

        if np.max(np.abs(g - a)) < 1e-10:
            # At LP fixed point
            a *= np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 1e-10)
            a /= a.sum()
            continue
        
        # Line search
        best_t = 0
        best_val = M
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1-t)*a + t*g

This is an important insight. The LP best-response converges to a fixed point where:

- g is the "primal variables"

So the LP approach finds stationary points of the non-convex objective, which may be local minima, saddles, or local maxima. The line search helps by taking the best step in the LP direction, which should move towards local minima.

    ag = np.convolve(a, g)

        return float(np.max((1.0 - t) * (1.0 - t) * aa + 2.0 * t * (1.0 - t) * ag + t * t * gg))

def _lp_solve(A, M, n, tl=3.0):
    """max 1^T g s.t. A g <= M, g >= 0."""
    try:
        b_ub = M * np.ones(A.shape[0])
        res = linprog(-np.ones(n), A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': tl,
                               'primal_feasibility_tolerance': 1e-12,
                               'dual_feasibility_tolerance': 1e-12})
        if res.status == 0 and res.x is not None:
            return np.clip(res.x, 0.0, None)
    except Exception:
        pass
    return None
