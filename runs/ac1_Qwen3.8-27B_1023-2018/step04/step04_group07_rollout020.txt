    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)

    def conv_fft(a):

        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2 * n - 1]
        np.maximum(c, 0.0, out=c)
        return c

    def score_from_conv(c, a):
        s = float(a.sum())
        if s < 0.01:
            return float('inf')
        m = float(c.max())
        return 2.0 * n * m / (s * s)

        return score_from_conv(conv_fft(a), a)

    def conv_inplace(c, a, i, delta):
        """Update c = a*a in place after a[i] += delta."""
        old = a[i]
        # k ranges from max(0, i) to min(2n-2, i+n-1)

        # And k in [0, 2n-2]
        k_lo = max(0, i)

                c[k] += 2.0 * old * delta + delta * delta
            else:
                c[k] += 2.0 * delta * a[k - i]
