For coordinate descent, given all other coordinates fixed, we want to find f_i that minimizes:
2n * max_k (conv_k + 2*f_i*f_{k-i} + f_i^2 * delta_{k,2i}) / (S - old_f_i + f_i)^2

This is a 1D optimization problem. The function is piecewise (changes when a different k achieves the max), but we can use grid search or ternary search.

Let me think about the structure: for a given i, the function of x = f_i is:
obj(x) = 2n * max_k (c_k + 2x * f_{k-i} + x^2 * δ_{k,2i}) / (S' + x)^2

where c_k = conv_k - 2*v*f_{k-i} - v^2*δ_{k,2i} (the part not involving f_i), S' = S - v.

The max over k of (c_k + 2x*f_{k-i} + x^2*δ_{k,2i}) is a piecewise quadratic function of x. The denominator (S'+x)^2 is a square. So obj(x) is a ratio of piecewise quadratic to quadratic, which is a piecewise rational function.

def _convolve(a):

        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_inner(a, n):

    s = float(a.sum())

        return math.inf
    b = _convolve(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):

def _solve_lp(f, n, M, time_limit=4.0):

    c = -np.ones(n)

                      method="highs",

def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]

    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]

        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

        return f, _score_inner(f, n)
    f = f / S
    S = 1.0

    conv = _convolve(f)
    best_score = 2.0 * n * float(conv.max())
    best_f = f.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 3.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:

                break

            v = f[i]
            S_rest = S - v

            si_valid = (0 <= si <= 2 * n - 2)

            x_max = max(3.0 * v + 0.1, 0.5)
            if v > 1e-6:
                x_max = max(x_max, 5.0 * v)
            x_max = min(x_max, 50.0)
            
            n_pts = 80
            x_grid = np.linspace(0.0, x_max, n_pts)
            
            best_x = v
            best_obj = best_score

            for x in x_grid:
                w = x - v
                if abs(w) < 1e-15:
                    m_new = float(conv.max())
                else:
                    tmp = conv.copy()
                    tmp[start:end] += w * infl
                    if si_valid:
                        tmp[si] += w * w
                    m_new = float(tmp.max())
                S_new = S_rest + x
                if S_new < 0.01:
                    continue
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-16:
                    best_obj = obj
                    best_x = x

            # Fine search
            if best_x != v:
                dw = x_max / n_pts
                x_lo = max(0.0, best_x - 4 * dw)
                x_hi = min(x_max, best_x + 4 * dw)
                x_fine = np.linspace(x_lo, x_hi, 40)
                for x in x_fine:
                    w = x - v
                    if abs(w) < 1e-15:
                        continue
                    tmp = conv.copy()
                    tmp[start:end] += w * infl
                    if si_valid:
                        tmp[si] += w * w
                    m_new = float(tmp.max())
                    S_new = S_rest + x
                    if S_new < 0.01:
                        continue
                    obj = 2.0 * n * m_new / (S_new * S_new)
                    if obj < best_obj - 1e-16:
                        best_obj = obj
                        best_x = x

                w = best_x - v
                f[i] = best_x
                S += w
                conv[start:end] += w * infl
                if si_valid:
                    conv[si] += w * w
                best_score = best_obj
                best_f = f.copy()
                improved = True

        if not improved and sweep > 5:
            break

    return best_f, best_score

Given f (sum=1), M = max(f*f).

LP: max sum(g) s.t. (f*g)_k <= M for all k in [0, 2n-2], g >= 0.

The solution g has sum = some value S_g. We normalize: g = g / S_g, so sum(g) = 1.

Then we step: f_new = (1-t)*f + t*g, renormalize to sum=1.

The key question: does this step decrease the score? The score is 2n * max(f*f) / sum(f)^2. With sum=1, it's 2n * max(f*f).

When we move from f to (1-t)f + tg:
new_f*f = (1-t)^2 f*f + 2t(1-t) f*g + t^2 g*g

max(new_f*f) <= (1-t)^2 M + 2t(1-t) * max(f*g) + t^2 * max(g*g)
<= (1-t)^2 M + 2t(1-t) M + t^2 * max(g*g)
= M(1-t)^2 + 2M*t*(1-t) + t^2 * max(g*g)
= M(1 - 2t + t^2 + 2t - 2t^2) + t^2 * max(g*g)
= M(1 - t^2) + t^2 * max(g*g)
= M + t^2(max(g*g) - M)

So if max(g*g) < M, the score decreases for any t > 0. If max(g*g) >= M, it might increase.

- w = x - v
- conv[k] changes by 2w * f[k-i] for k in [i, i+n-1] (with appropriate bounds)
- Plus w^2 at k = 2i

new_conv = conv + w * infl + w^2 * e_{2i}

where infl[k] = 2f[k-i] for k in [i, i+n-1] and 0 elsewhere.

The score is: 2n * max(new_conv) / (S - v + x)^2

For each candidate x, I need:
1. Compute the affected part of new_conv: conv[i:i+n] + w * 2*f[:n]
2. Add w^2 at position 2i (relative)
3. Find the max of new_conv = max(max(affected), max(unaffected))
4. Compute the score

The "unaffected" max can be precomputed: it's the max of conv outside [i, i+n-1].

import numpy as np
import time
import math
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(400, dtype=np.float64)
    if p.size == 0:
        p = np.ones(400, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(400, dtype=np.float64)
    return p


def _convolve(a):
    n = len(a)
    if n <= 4096:
        return np.convolve(a, a, mode="full")
    from scipy.fft import rfft, irfft, next_fast_len
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_inner(a, n):
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _convolve(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp(f, n, M, time_limit=5.0):
    A = _build_A(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": time_limit, "disp": False}
        )
        if res.status != 0:
            res = linprog(
                c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                method="highs",
                options={"presolve": False, "time_limit": time_limit, "disp": False}
            )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0.0)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs

def _conv(a):
    n = len(a)
    if n <= 4096:
        return np.convolve(a, a, mode="full")
    from scipy.fft import rfft, irfft, next_fast_len
    fz = next_fast_len(2 * n - 1)
    F = rfft(a, fz)
    c = irfft(F * F, fz)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _sc(a, n):
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)


def _lpsolve(f, n, M, tl=5.0):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n)
    bu = np.full(2 * n - 1, M)
    bd = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=bu, bounds=bd, method="highs",
                    options={"presolve": True, "time_limit": tl, "disp": False})
        if r.status != 0:
            r = linprog(c, A_ub=A, b_ub=bu, bounds=bd, method="highs",
                        options={"presolve": False, "time_limit": tl, "disp": False})
    except Exception:
        return None
    if r.status != 0 or r.x is None:
        return None
    g = np.maximum(np.asarray(r.x, dtype=np.float64), 0.0)
    gs = float(g.sum())
    return g / gs if gs > 1e-12 else None
