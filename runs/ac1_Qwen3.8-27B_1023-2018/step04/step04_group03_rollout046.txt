                t = 0.3
                stall = 0
                max_stall = min(max_stall + 30, 200)
            continue

        g_sum = g.sum()
        f_sum = f.sum()
        if g_sum < 1e-15:
            stall += 3
            continue
        g = g * (f_sum / g_sum)

        dist = np.linalg.norm(g - f) / max(1.0, np.linalg.norm(f))
        if dist < 1e-10:
            stall += 2
            if stall > max_stall:
                break
            continue

        improved = False
        t_trials = [t, t * 0.5, t * 1.5, t * 0.25, t * 2.0,
                   0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]
        t_trials = sorted(set(max(0.01, min(0.95, x)) for x in t_trials))

        best_t = t
        best_f_new = None
        best_sc_new = best_score

        for t_try in t_trials:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.clip(f_new, 0, None)
            if f_new.sum() < 0.01:
                continue
            sc = _score(f_new)
            if sc < best_sc_new - 1e-14:
                best_sc_new = sc
                best_f_new = f_new
                best_t = t_try

        if best_f_new is not None:
            f = best_f_new
            t = best_t
            improved = True
            stall = 0
        else:
            stall += 1
            t = max(0.02, t * 0.75)
            if stall >= max_stall:
                if stall > max_stall * 2:
                    break
                noise = rng.normal(0, 0.08 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * best_f.sum()
                t = 0.3
                stall = 0
                max_stall = min(max_stall + 30, 300)

    return best_f, best_score


def _subgrad_optimize(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-8, None)
    f_sum = f.sum()
    best_f = f.copy()
    best_score = _score(f)

    lr = 0.5
    stall = 0
    max_stall = 50

    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        sc = _score(f)
        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f.copy()
            stall = 0
            lr = min(lr * 1.01, 2.0)
        else:
            stall += 1
            lr *= 0.97

        f_new = _subgrad_step(f, b, M, lr)
        sc_new = _score(f_new)
        if sc_new < _score(f) + 1e-10:
            f = f_new
        else:
            f = 0.5 * f + 0.5 * f_new

        if stall > max_stall:
            noise = rng.normal(0, 0.02 * max(1.0, f.mean()), n)
            f = np.clip(f + noise, 1e-8, None)
            stall = 0
            lr = 0.3
            max_stall += 20
            if max_stall > 200:
                break

    return best_f, best_score


def _polish(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    best_sc = _score(f)
    best_f = f.copy()
    no_imp = 0
    max_no_imp = 15

    mults = (0.3, 0.5, 0.7, 0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.3, 1.5, 2.0, 3.0, 0.0)

    while time.monotonic() < stop_time - 3.0 and no_imp < max_no_imp:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = f[i]
            best_local = best_sc
            best_val = orig

            for mlt in mults:
                f[i] = orig * mlt
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_val = f[i]

            if best_local < best_sc - 1e-15:
                f[i] = best_val
                best_sc = best_local
                best_f = f.copy()
                improved = True
            else:
                f[i] = best_val

        if improved:
            no_imp = 0
        else:
            no_imp += 1

    return best_f, best_sc


def _try_different_lengths(f_best, f_score, stop_time, seed, n_parent):
    best_f = f_best
    best_sc = f_score
    rng = np.random.default_rng(seed)

    for L in [n_parent - 32, n_parent - 16, n_parent + 16, n_parent + 32,
              n_parent + 64, n_parent - 64]:
        if time.monotonic() > stop_time - 15.0:
            break
        if L < 50 or L > 1000:
            continue
        x_old = np.linspace(0, 1, n_parent)
        x_new = np.linspace(0, 1, L)
        init_L = np.interp(x_new, x_old, f_best)
        init_L = np.clip(init_L, 0, 1000)

        stop = min(stop_time - 10.0, time.monotonic() + 30.0)
        try:
            res, sc = _optimize_lp(init_L, stop, seed + L, peak_frac=0.05)
            if sc < best_sc - 1e-12:
                best_f = res
                best_sc = sc
        except Exception:
            pass

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 25.0

    parent = _get_parent()
    n_p = len(parent)

    best_seq = parent.copy()
    best_score = _score(best_seq)

    def _update_best(candidate):
        nonlocal best_seq, best_score
        if candidate is None or len(candidate) == 0:
            return
        candidate = np.clip(np.asarray(candidate, dtype=np.float64).ravel(), 0, 1000)
        if not np.all(np.isfinite(candidate)):
            return
        if candidate.sum() < 0.01:
            return
        sc = _score(candidate)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq = candidate.copy()
            best_score = sc

    rng = np.random.default_rng(seed)
    i_arr = np.arange(n_p, dtype=np.float64)

    # Build diverse initializations
    inits = [parent.copy()]

    # Uniform
    inits.append(np.ones(n_p))

    # Cosine window
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / (n_p - 1))))

    # Gaussian bump
    center = (n_p - 1) / 2.0
    inits.append(np.exp(-0.5 * ((i_arr - center) / (n_p * 0.25)) ** 2))

    # Two bumps
    c1, c2 = n_p * 0.25, n_p * 0.75
    inits.append(np.exp(-0.5 * ((i_arr - c1) / (n_p * 0.1))) +
                 np.exp(-0.5 * ((i_arr - c2) / (n_p * 0.1))))

    # Multi-freq
    sup = np.ones(n_p) * 0.5
    for fr in (1, 2, 3, 5, 7, 11, 13):
        ph = rng.uniform(0, 2 * np.pi)
        sup += 0.08 * np.cos(2.0 * np.pi * fr * i_arr / n_p + ph)
    inits.append(np.abs(sup) + 0.02)

    # Perturbed parent
    inits.append(np.clip(parent * (1.0 + 0.2 * rng.normal(0, 1, n_p)), 0, 1000))
    inits.append(np.clip(parent + 0.1 * max(1.0, parent.mean()) * rng.normal(0, 1, n_p), 0, 1000))

    # Sawtooth-like
    inits.append(np.abs(2 * (i_arr / n_p - 0.5)))
    # Triangle
    inits.append(1.0 - 2 * np.abs(i_arr / n_p - 0.5))

    # Evaluate all inits
    for init in inits:
        sc = _score(init)
        if sc < best_score - 1e-14:
            best_seq = init.copy()
            best_score = sc

    # Phase 1: LP optimization on top initializations (50% of time)
    phase1_end = t0 + budget * 0.50
    top_inits = []
    for init in inits:
        sc = _score(init)
        top_inits.append((sc, init))
    top_inits.sort()
    top_inits = top_inits[:5]

    time_per = max(10.0, (phase1_end - t0 - 10.0) / len(top_inits))
    for idx, (sc_init, init) in enumerate(top_inits):
        if time.monotonic() > phase1_end - 8.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per)
        if time.monotonic() > stop - 3.0:
            continue
        try:
            res, sc = _optimize_lp(init, stop, seed + 10000 * (idx + 1), peak_frac=0.05)
            _update_best(res)
        except Exception:
            pass

    # Phase 2: Continue LP on best (15% time)
    phase2_end = t0 + budget * 0.65
    if time.monotonic() < phase2_end - 15.0:
        try:
            res, sc = _optimize_lp(best_seq, phase2_end - 5.0, seed + 77777, peak_frac=0.03)
            _update_best(res)
        except Exception:
            pass

    # Phase 3: Subgradient (10% time)
    phase3_end = t0 + budget * 0.75
    if time.monotonic() < phase3_end - 15.0:
        try:
            res, sc = _subgrad_optimize(best_seq, phase3_end - 5.0, seed + 55555)
            _update_best(res)
        except Exception:
            pass

    # Phase 4: Polish (12% time)
    phase4_end = t0 + budget * 0.87
    if time.monotonic() < phase4_end - 15.0:
        try:
            res, sc = _polish(best_seq, phase4_end - 5.0, seed + 33333)
            _update_best(res)
        except Exception:
            pass

    # Phase 5: LP again on polished (5% time)
    phase5_end = t0 + budget * 0.92
    if time.monotonic() < phase5_end - 15.0:
        try:
            res, sc = _optimize_lp(best_seq, phase5_end - 5.0, seed + 88888, peak_frac=0.02)
            _update_best(res)
        except Exception:
            pass

    # Phase 6: Try different lengths (3% time)
    phase6_end = t0 + budget * 0.95
    if time.monotonic() < phase6_end - 20.0:
        try:
            bf, bs = _try_different_lengths(best_seq, best_score, phase6_end - 5.0, seed + 99999, n_p)
            _update_best(bf)
        except Exception:
            pass

    # Final polish
    remaining = deadline - time.monotonic()
    if remaining > 20.0:
        try:
            res, sc = _polish(best_seq, deadline - 5.0, seed + 11111)
            _update_best(res)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        final_val = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_sc = float(evaluate_sequence(final_val))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_val = [float(max(0.0, min(1000.0, x))) for x in parent]
            final_sc = float(evaluate_sequence(final_val))
        except Exception:
            return [1.0]

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
