```python
def project_simplex(v, s=1.0):
    # Project v onto {x >= 0, sum(x) = s}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u * np.arange(1, n+1) > cssv)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0)
```

```python
def _lp_direction(f, n, M, timeout=5.0):
    """Find LP direction: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    # Build constraint matrix A: (f*g)_k <= M for k=0,...,2n-2
    # (f*g)_k = sum_j f_{k-j} * g_j for valid j
    # A[k,j] = f[k-j] if 0<=k-j<n else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        lo = max(0, j - n + 1)  # k-j >= 0 => k >= j, but k starts from 0
        # Actually: (f*g)_k = sum_j g_j * f[k-j] for valid k-j
        # For fixed j, the constraint (f*g)_k <= M gives coefficient f[k-j] in row k
        # But we need A[k,j] = f[k-j] if 0<=k-j<n
        # For row k: A[k,j] = f[k-j] if 0<=k-j<n
        # For column j: A[k,j] nonzero when 0<=k-j<n, i.e., j<=k<=j+n-1
        # So A[j:j+n, j] = f (shifted)
        A[j:j+n, j] = f
    
    c = -np.ones(n)  # maximize sum(g) = minimize -sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.status != 0:
            return None
        g = np.clip(res.x, 0, None)
        if g.sum() < 1e-15:
            return None
        return g
    except Exception:
        return None
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _autocorr(f):
    n = len(f)
    size = next_fast_len(2 * n - 1)
    F = rfft(f, size)
    return irfft(F * F, size)[:2 * n - 1]


def _max_ac(f):
    b = _autocorr(f)
    m = float(b.max())
    return max(0.0, m)


def _subgrad(f, b, topk=8):
    """Subgradient of max_k b_k using top-k active lags."""
    n = len(f)
    nl = len(b)
    if topk >= nl:
        idx = np.arange(nl)
    else:
        idx = np.argpartition(-b, topk)[:topk]
    w = b[idx]
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for ki, wi in zip(idx, w):
        k = int(ki)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        seg = f[k - hi : k - lo + 1][::-1].copy()
        grad[lo:hi + 1] += 2.0 * wi * seg
    return grad


def _proj_simplex(v, s=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    cond = u * np.arange(1, n + 1) > cssv
    if not cond.any():
        return np.full(n, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

        f = np.ones(n) / n
    else:
        f /= s

    best_f = f.copy()
    best_val = _max_ac(f)

    momentum = np.zeros(n)
    beta = 0.82
    alpha = 0.004
    alpha_min = 1e-9
    alpha_max = 0.25
    topk = 4
    stall = 0
    max_stall = 250
    iterations = 0

    while time.monotonic() < stop_time - 3.0:
        iterations += 1
        b = _autocorr(f)
        M = float(b.max())

        if M < best_val - 1e-14:
            best_val = M
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

            f = best_f.copy()
            scale = 0.003 + 0.002 * min(5, stall // max_stall)
            f += rng.normal(0, scale / math.sqrt(n), n)
            f = np.clip(f, 0, None)
            fs = f.sum()
            if fs > 1e-10:

            else:

            momentum *= 0.3
            stall = 0
            max_stall = int(max_stall * 1.4)
            alpha = max(alpha_min, alpha * 0.4)

        grad = _subgrad(f, b, topk)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:

        f_try = _proj_simplex(f - momentum, 1.0)
        b_try = _autocorr(f_try)
        M_try = float(b_try.max())

    if bs > 1e-10:

def _lp_direction(f, n, M, timeout=4.0):
    """LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    if M <= 1e-18:
        return None
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-14)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"time_limit": timeout, "presolve": True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-14:
        return None
    return g

    rng = np.random.default_rng(seed)
    f = np.clip(f.copy(), 0, None)

    if fs < 1e-10:
        return f, _max_ac(f)
    f /= fs
    best_f = f.copy()
    best_val = _max_ac(f)
    t = 0.3
    stall = 0

    while time.monotonic() < stop_time - 3.0:
        b = _autocorr(f)

        if M < best_val - 1e-14:
            best_val = M
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        if stall > 80:
            break

        g = _lp_direction(f, n, M, timeout=3.0)
        if g is None:
            stall += 5
            continue

        gs = g.sum()
        if gs < 1e-14:
            continue
        g = g / gs

        improved = False
        for tc in [t, t * 0.5, t * 1.5, 0.1, 0.5, 0.8]:

    # Final validation
    final_sc = _exact_score(best_f)
    if not math.isfinite(final_sc):
        return [1.0]
    return [float(x) for x in best_f]
```
