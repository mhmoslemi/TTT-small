        # Projected subgradient step: descend M while keeping sum = target_sum, a >= 0
        # f_new = f - step * grad
        # Then project: f_new = max(0, f_new), then rescale to sum = target_sum
        
        f_new = f - step * grad / max(1.0, np.max(np.abs(grad)))
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 0.01:
            f_new = f.copy()
        else:
            f_new = f_new / s_new * target_sum
        
        sc = _score(f_new)
        if sc < best_sc - 1e-14:
            f = f_new
            best_f = f.copy()
            best_sc = sc
            no_imp = 0
            step = min(step * 1.1, 0.2 * max(1.0, f.mean()))
        else:
            no_imp += 1
            step *= 0.7
            if no_imp % 10 == 0:
                # Small random kick
                idx = rng.choice(n, max(3, n//50), replace=False)
                f = best_f.copy()
                f[idx] *= rng.uniform(0.7, 1.3, len(idx))
                s = f.sum()
                if s > 0.01:
                    f = f/s*target_sum
            if step < 1e-10:
                break
    
    return best_f, best_sc

                s = f.sum()

                    f = f/s*target_sum
                best_sc = best_v_sc
                best_f = f.copy()

            no_imp = 0

            # Random perturbation

                f = best_f.copy()
                f[k] = max(0.0, f[k]*rng.uniform(0.3, 2.5))

                if s > 0.01:
                    f = f/s*target_sum
    
    return best_f, best_sc

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p


def _conv(a):
    n = len(a)
    m = 2 * n - 1
    fsize = next_fast_len(m)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, n, M, timeout=0.5):
    if M <= 1e-14:
        return None
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10,
                               'mip_rel_gap': 1e-8})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, None)
    if g.sum() < 1e-12:
        return None
    return g


def _lp_optimize(f0, n, stop_time, seed, target_sum=None):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    if f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n)
        fs = n
    if target_sum is None:
        target_sum = math.sqrt(2.0 * n)
    f = f / fs * target_sum

    best_f = f.copy()
    best_sc = _score(f)

    t = 0.3
    stall = 0
    max_stall = 60
    conv_thresh = 5e-10

    while time.monotonic() < stop_time - 2.0:
        b = _conv(f)
        M = float(b.max())
        if M < 1e-14:
            break

        g = _lp_solve(f, n, M, timeout=0.4)
        if g is None:
            stall += 2
            if stall > max_stall:
                f = best_f.copy()
                n_pert = max(3, n // 15)
                idx = rng.choice(n, n_pert, replace=False)
                f[idx] *= rng.uniform(0.2, 3.0, n_pert)
                f = np.clip(f, 0, None)
                s = f.sum()
                f = f / s * target_sum if s > 0.01 else best_f.copy()
                stall = 0
                max_stall += 30
                t = 0.3
            continue

        stall = 0
        g = g * (target_sum / g.sum())

        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < conv_thresh:
            stall += 2
            if stall > max_stall:
                f = best_f.copy()
                idx = rng.choice(n, max(5, n // 10), replace=False)
                f[idx] *= rng.uniform(0.4, 2.0, len(idx))
                f = np.clip(f, 0, None)
                s = f.sum()
                f = f / s * target_sum if s > 0.01 else best_f.copy()
                stall = 0
                max_stall += 25
                t = 0.3
            continue

        improved = False
        candidates = sorted(
            {max(0.02, min(1.0, c)) for c in
             [t, t * 0.6, t * 0.9, t * 1.2, t * 1.5, 0.15, 0.4, 0.7, 1.0]},
            reverse=True
        )
        for tc in candidates:
            fn = (1.0 - tc) * f + tc * g
            fn = np.clip(fn, 0, None)
            s = fn.sum()
            if s < 0.01:
                continue
            fn = fn / s * target_sum
            sc = _score(fn)
            if sc < best_sc - 1e-12:
                f = fn.copy()
                best_f = f.copy()
                best_sc = sc
                improved = True
                t = min(1.0, tc * 1.3)
                break

        if not improved:
            stall += 1
            t = max(0.05, t * 0.6)
            if stall >= max_stall:
                f = best_f.copy()
                strength = 0.05 + 0.03 * min(stall // max_stall, 5)
                f += rng.normal(0, strength * max(1.0, best_f.mean()), n)
                f = np.clip(f, 0, None)
                s = f.sum()
                f = f / s * target_sum if s > 0.01 else best_f.copy()
                stall = 0
                max_stall += 20
                t = 0.3

    return best_f, best_sc
