        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            best_v = orig
            best_v_sc = best_sc
            if orig > 1e-14:
                vals = [0.0, orig * 0.15, orig * 0.4, orig * 0.7, orig * 0.9,
                        orig * 1.1, orig * 1.4, orig * 1.8, orig * 2.5, orig * 4.0]
            else:
                vals = [0.0, mean_val * 0.05, mean_val * 0.1, mean_val * 0.2, mean_val * 0.4]
            for v in vals:
                f[i] = v
                sc = _score_fast(f)
                if sc < best_v_sc - 1e-15:
                    best_v_sc = sc
                    best_v = v
            f[i] = best_v
            if best_v_sc < best_sc - 1e-15:
                best_sc = best_v_sc
                best_f = f.copy()
                improved = True
                mean_val = f.sum() / n

        for _ in range(min(250, n)):
            if time.monotonic() > stop_time:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            bv, bj = oi, oj
            bsc = best_sc
            for mi in [0.0, 0.5, 1.5, 2.5]:
                for mj in [0.0, 0.5, 1.5, 2.5]:
                    if mi == 1.0 and mj == 1.0:
                        continue
                    f[i] = oi * mi if oi > 1e-14 else 0.0
                    f[j] = oj * mj if oj > 1e-14 else 0.0
                    sc = _score_fast(f)
                    if sc < bsc:
                        bsc = sc
                        bv, bj = f[i], f[j]
            f[i], f[j] = bv, bj
            if bsc < best_sc:
                best_sc = bsc
                best_f = f.copy()
                improved = True

        if not improved:
            no_imp_sweeps += 1
            if no_imp_sweeps >= 6:
                break
        else:
            no_imp_sweeps = 0

    return best_f, best_sc
