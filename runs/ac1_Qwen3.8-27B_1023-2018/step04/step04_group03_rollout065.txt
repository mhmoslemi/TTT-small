def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _pf(b, s, n):
    if s < 0.01:
        return math.inf
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    return _pf(b, s, n)


def _lp_direction(f, n, b, peak_frac, timeout=5.0, row_cap=220):
    M = float(b.max())
    if M < 1e-18:
        return None
    threshold = M * (1.0 - peak_frac)
    peak_idx = np.where(b >= threshold)[0]
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(b)])
    if len(peak_idx) > row_cap:
        order = np.argsort(-b[peak_idx])
        peak_idx = peak_idx[order[:row_cap]]

    rows, cols, vals = [], [], []
    for idx, k in enumerate(peak_idx):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-20:
                rows.append(idx)
                cols.append(i)
                vals.append(v)
    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    c_vec = -np.ones(n)
    b_ub = np.full(len(peak_idx), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

            except Exception:

    # final coordinate polish with remaining time
    if time.monotonic() < deadline - 12.0:
        try:
            bf, bb, bs = _opt_coords(best_seq, _conv_full(best_seq), deadline - 8.0, seed + 999999)
            _relax(bf)
        except Exception:
            pass

    # exact validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf
    if math.isfinite(final_sc):
        best_score = min(best_score, final_sc)

    seq_out = np.clip(best_seq, 0, 1000)
    if not np.all(np.isfinite(seq_out)) or seq_out.size == 0 or seq_out.sum() < 0.01:
        seq_out = np.clip(parent, 0, 1000).copy()
    return [float(x) for x in seq_out]
