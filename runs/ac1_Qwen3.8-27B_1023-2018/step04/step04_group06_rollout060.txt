import numpy as np

from scipy.optimize import linprog

def _score_fast(a):
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)

def _conv_max(a):
    return float(np.convolve(a, a).max())

def _build_A(a, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        A[k, js:je] = a[k - je + 1 : k - js + 1][::-1]
    return A

def _lp_br(a, M, n, tl=4.0):
    A = _build_A(a, n)
    c_vec = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tl,
                               'dual_feasibility_tolerance': 1e-13,
                               'primal_feasibility_tolerance': 1e-13})
        if res.status == 0 and res.x is not None:
            return np.clip(res.x, 0.0, None)
    except Exception:
        pass
    return None
