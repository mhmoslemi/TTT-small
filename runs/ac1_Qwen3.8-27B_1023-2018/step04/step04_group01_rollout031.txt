    """Project v onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]

def _coord_descent(f0, n, deadline, seed, max_sweeps=100):
    """Coordinate descent with grid search per coordinate."""
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0, None)
    f = f / f.sum()
    
    best_f = f.copy()
    best_score = _score(f)
    
    S = 1.0  # since normalized
    b = _conv_full(f)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline - 3:
            break
        
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline - 3:
                break
            
            v = f[i]
            S_rest = S - v
            
            # Effect on convolution:
            # For k in [i, i+n-1] (clipped to [0, 2n-2]):
            #   Δb[k] = (x-v) * 2*f[k-i] (if k != 2i) or (x^2-v^2) (if k == 2i)
            # Wait, more precisely:
            # Δb[k] = (x-v)*f[k-i] + (x-v)*f[k-i] = 2*(x-v)*f[k-i] if k-i != i
            # Δb[2i] = x^2 - v^2 (additional to the linear term)
            # Actually: Δb[k] = 2*(x-v)*f[k-i] for all k in range (including k=2i)
            #             + (x^2 - v^2 - (x-v)*f[i] - (x-v)*f[i]) for k=2i
            # = 2*(x-v)*f[k-i] + (x^2 - v^2 - 2*(x-v)*f[i]) for k=2i
            # = 2*(x-v)*f[k-i] + ((x-v)*(x+v) - 2*(x-v)*v) for k=2i
            # = 2*(x-v)*f[k-i] + (x-v)*(x-v) for k=2i
            # = 2*(x-v)*f[k-i] + (x-v)^2 for k=2i
            
            # So: Δb[k] = 2*(x-v)*f[k-i] + (x-v)^2 * δ(k, 2i)
            
            lo_k = i
            hi_k = min(i + n - 1, 2*n - 2)
            
            # Current max at unaffected positions
            # The affected range is [lo_k, hi_k]
            # Unaffected max = max(b[0:lo_k], b[hi_k+1:])
            
            # For the 1D optimization, we need to find x that minimizes:
            # score(x) = 2*n*max_k (b[k] + Δb[k]) / (S + (x-v))^2
            
            # The max is the max of:
            # 1. Unaffected positions: max of b[k] for k outside [lo_k, hi_k]
            # 2. Affected positions: b[k] + 2*(x-v)*f[k-i] + (x-v)^2*δ(k,2i)
            
            # For (2), each affected position k gives a quadratic in x (if k=2i) or linear in x (if k≠2i).
            
            # The objective is max of these, divided by a quadratic in x.
            # This is a non-smooth 1D problem. We'll use grid search + refinement.
            
            if v < 1e-12:
                # Coordinate is essentially zero; try small values
                x_max = 0.1 / S
            else:
                x_max = max(3.0 * v, 0.5 / S)
            
            # Grid search
            n_pts = 80
            x_grid = np.linspace(0.0, min(x_max, 10.0), n_pts)
            
            best_x = v
            best_obj = best_score
            
            # Precompute unaffected max
            if lo_k == 0 and hi_k == 2*n-2:
                unaff_max = 0.0
            elif lo_k == 0:
                unaff_max = b[hi_k+1:].max() if hi_k+1 < 2*n-1 else 0.0
            elif hi_k == 2*n-2:
                unaff_max = b[:lo_k].max() if lo_k > 0 else 0.0
            else:
                unaff_max = max(b[:lo_k].max(), b[hi_k+1:].max())
            
            f_slice = f[hi_k - lo_k + 1 : hi_k - lo_k + 1]  # f[0 : hi_k-i+1] = f[0:n] but clipped
            # Actually f[k-i] for k in [lo_k, hi_k] means f[0 : hi_k-i+1] = f[0 : n] (since hi_k <= i+n-1)
            f_aff = f[0 : hi_k - lo_k + 1]  # = f[0:n] clipped
            
            has_2i = (2*i >= lo_k and 2*i <= hi_k)
            idx_2i = 2*i - lo_k  # index of 2i in the affected range
            
            for x in x_grid:
                w = x - v
                if abs(w) < 1e-15:
                    m_new = b.max()
                else:
                    aff_vals = b[lo_k:hi_k+1] + 2.0*w*f_aff
                    if has_2i:
                        aff_vals[idx_2i] += w*w
                    m_aff = aff_vals.max()
                    m_new = max(m_aff, unaff_max)
                
                S_new = S_rest + x
                if S_new < 0.01:
                    continue
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_x = x
            
            # Refine around best_x
            if best_x != v:
                dw = x_grid[1] - x_grid[0]
                x_lo = max(0.0, best_x - 3*dw)
                x_hi = min(x_max, best_x + 3*dw)
                x_fine = np.linspace(x_lo, x_hi, 40)
                for x in x_fine:
                    w = x - v
                    S_new = S_rest + x
                    if S_new < 0.01:
                        continue
                    aff_vals = b
