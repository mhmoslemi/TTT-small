The LP: maximize sum(g) subject to:
- For k in active_set: conv(f, g)[k] <= M
- g >= 0

Where M is the current peak of conv(f, f).

The constraint conv(f, g)[k] = sum_i f[i]*g[k-i] <= M for each active k.

This is a linear program with |active_set| constraints and n variables. For n=416 and |active_set| up to ~200, this is very manageable.

The key is choosing the active set well. If we include all positions where conv(f,f) is within a small epsilon of M, the LP gives a direction that reduces the peak. If we include too many, the LP might be infeasible (no nonnegative g with conv(f,g) <= M at those positions while having positive sum).

def _cross_conv(f, g):
    """Compute conv(f, g) efficiently."""
    n = len(f)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(f, fsize)
    G = rfft(g, fsize)
    c = irfft(F * G, fsize)[:2 * n - 1]
    return c

def _lp_direction(f, n, b, active_frac, timeout=5.0):
    """Find LP direction to reduce peak."""
    M = float(b.max())
    if M < 1e-18:
        return None
    
    threshold = M * (1.0 - active_frac)
    peak_idx = np.where(b >= threshold)[0]
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(b)])
    
    # Build constraint matrix: for each active k, conv(f,g)[k] = sum_i f[i]*g[k-i] <= M
    rows = []
    cols = []
    vals = []
    for constraint_row, k in enumerate(peak_idx):
        # conv(f,g)[k] = sum_{i} f[i] * g[k-i] where 0<=k-i<n
        # = sum_{j: 0<=j<n, 0<=k-j<n} f[k-j] * g[j]
        # So for variable g[j], the coefficient is f[k-j] (when valid)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            coeff = f[k - j]
            if coeff > 1e-15:
                rows.append(constraint_row)
                cols.append(j)
                vals.append(coeff)
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    c_vec = -np.ones(n)  # minimize -sum = maximize sum
    b_ub = np.full(len(peak_idx), M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    
    if res.status not in (0, 3):
        return None
    if res.x is None:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _line_search(f, g, current_score, n):
    """Find best step size t for f_new = (1-t)*f + t*g."""
    f_sum = f.sum()
    g = g * (f_sum / g.sum())  # normalize g to same sum
    
    # Try a range of t values
    best_t = 0.0
    best_sc = current_score
    
    # Coarse search
    for t in [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9]:
        f_new = (1.0 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        if f_new.sum() < 0.01:
            continue
        sc = _score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    
    if best_t == 0.0:
        return None, best_sc
    
    # Fine search around best_t
    t_lo = max(0.01, best_t - 0.05)
    t_hi = min(0.95, best_t + 0.05)
    for _ in range(8):
        t_mid = (t_lo + t_hi) / 2
        f_new = (1.0 - t_mid) * f + t_mid * g
        f_new = np.clip(f_new, 0, None)
        if f_new.sum() < 0.01:
            break
        sc = _score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_t = t_mid
            t_lo = t_mid
        else:
            t_hi = t_mid
    
    f_new = (1.0 - best_t) * f + best_t * g
    f_new = np.clip(f_new, 0, None)
    if f_new.sum() < 0.01:
        return None, best_sc
    return f_new, best_sc

def _main_optimize(f0, stop_time, seed, start_frac=0.15):
    """Main LP-guided optimization loop."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    best_f = f.copy()
    best_score = _score(f)
    
    frac = start_frac  # active constraint fraction (how "wide" the peak set is)
    stall = 0
    max_stall = 50
    frac_schedule = [0.15, 0.10, 0.08, 0.06, 0.05, 0.04, 0.03, 0.02, 0.015, 0.01]
    frac_idx = 0
    
    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        g = _lp_direction(f, n, b, frac, timeout=4.0)
        if g is None:
            # Widen the constraint set
            if frac_idx < len(frac_schedule) - 1:
                frac_idx += 1
                frac = frac_schedule[frac_idx]
            stall += 3
            if stall > max_stall:
                break
            continue
        
        f_new, sc = _line_search(f, g, best_score, n)
        
        if f_new is not None and sc < best_score - 1e-14:
            f = f_new
            best_f = f.copy()
            best_score = sc
            stall = 0
            # Tighten constraints as we improve
            if frac_idx < len(frac_schedule) - 1 and stall == 0:
                frac_idx = min(frac_idx + 1, len(frac_schedule) - 1)
                frac = frac_schedule[frac_idx]
        else:
            stall += 1
            if stall > max_stall:
                # Perturb and restart
                noise = rng.normal(0, 0.05 * max(1.0, f.mean()), n)
                f = np.clip(f + noise, 0, None)
                fs = f.sum()
                if fs < 0.01:
                    f = best_f.copy()
                else:
                    f = f / fs * best_f.sum()
                stall = 0
                frac_idx = max(0, frac_idx - 2)
                frac = frac_schedule[frac_idx]
                max_stall = min(max_stall + 20, 200)
    
    return best_f, best_score

def _subgrad_optimize(f0, stop_time, seed):
    """Nesterov-accelerated subgradient optimization."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 1e-10, None)
    f = f / f.sum()
    best_f = f.copy()
    best_score = _score(f)
    
    # Nesterov momentum
    y = f.copy()  # momentum point
    beta = 0.9
    lr = 0.3
    stall = 0
    
    iter_count = 0
    while time.monotonic() < stop_time - 3.0:
        b = _conv_full(y)
        M = float(b.max())
        if M < 1e-18:
            break
        
        # Find peak positions
        threshold = M * 0.99
        peak_idx = np.where(b >= threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])
        
        # Compute subgradient at y: for each i, sum over peak k of y[k-i]
        grad = np.zeros(n)
        for k in peak_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):
                grad[j] += y[k - j]
        grad /= len(peak_idx)
        
        # Nesterov update
        beta = min(0.999, 2.0 / (iter_count + 2.0))
        
        # Gradient step
        lr_eff = lr / max(1e-10, np.linalg.norm(grad, ord=np.inf))
        v = y - lr_eff * grad
        v = np.clip(v, 0, None)
        s = v.sum()
        if s < 1e-15:
            break
        v = v / s
        
        # Nesterov combination
        f_new = (1.0 - beta) * y + beta * v
        f_new = np.clip(f_new, 1e-10, None)
        f_new = f_new / f_new.sum()
        
        y = f_new
        iter_count += 1
        
        sc = _score(y)
        if sc < best_score - 1e-13:
            best_f = y.copy()
            best_score = sc
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Perturb
                noise = rng.normal(0, 0.005, n)
                y = np.clip(y + noise, 1e-10, None)
                y = y / y.sum()
                stall = 0
                lr *= 0.7
                if lr < 0.01:
                    break
        
        # Reduce learning rate slowly
        lr *= 0.9995
    
    return best_f, best_score

def _coord_descent(f0, stop_time, seed):
    """Coordinate descent polish."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    best_sc = _score(f)
    no_imp = 0
    max_no_imp = 15
    
    # Multipliers for 1D search
    mults = (0.1, 0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5, 2.0, 3.0, 0.0)
    
    while time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            
            orig = f[i]
            best_local_sc = best_sc
            best_val = orig
            
            for m in mults:
                f[i] = orig * m if m > 0 else 0.0
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_local_sc - 1e-15:
                    best_local_sc = sc
                    best_val = f[i]
            
            if best_local_sc < best_sc - 1e-15:
                f[i] = best_val
                best_sc = best_local_sc
                improved = True
            else:
                f[i] = best_val
        
        if not improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0
    
    return f.copy(), best_sc
