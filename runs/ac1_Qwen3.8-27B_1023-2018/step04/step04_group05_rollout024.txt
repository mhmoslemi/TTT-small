        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=8.0):
    if M <= 1e-18:
        return None
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-12)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
            options={"presolve": True, "time_limit": timeout,
                     "dual_feasibility_tolerance": 1e-12,
                     "primal_feasibility_tolerance": 1e-12}
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g


def _lp_optimize(f0, n, stop_time, seed, max_iters=500):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n)
        f_sum = n

    target_sum = math.sqrt(2.0 * n)
    f = f / f_sum * target_sum

    best_f = f.copy()
    best_score = _score_fast(f)

    t = 0.3
    t_min = 0.01
    t_max = 1.0
    stall = 0
    max_stall = 150
    conv_thresh = 5e-12
    iteration = 0

    while time.monotonic() < stop_time - 3.0 and iteration < max_iters:
        iteration += 1
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, n, M, timeout=6.0)
        if g is None:
            stall += 5
            if stall > max_stall:
                f = best_f + rng.normal(0, 0.05 * max(1.0, best_f.mean()), n)
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                else:
                    f = best_f.copy()
                stall = 0
                max_stall += 100
                t = 0.3
            continue

        stall = 0
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            continue
        g = g * (target_sum / g_sum)

        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < conv_thresh:
            stall += 5
            if stall > max_stall:
                f = best_f.copy()
                idx = rng.choice(n, size=min(15, n), replace=False)
                for k in idx:
                    f[k] = max(0.0, f[k] * rng.uniform(0.2, 3.0))
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                stall = 0
                max_stall += 80
                t = 0.3
            continue

        improved = False
        step_candidates = sorted(
            {max(t_min, min(1.0, c)) for c in
             [t, t * 0.6, t * 0.8, t * 1.2, t * 1.5, 0.1, 0.3, 0.5, 0.8, 1.0]},
            reverse=True
        )
        for tc in step_candidates:
            f_new = (1.0 - tc) * f + tc * g
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 0.01:
                continue
            f_new = f_new / fs_new * target_sum
            sc = _score_fast(f_new)
            if sc < best_score - 1e-13:
                f = f_new.copy()
                best_f = f.copy()
                best_score = sc
                improved = True
                t = min(1.0, tc * 1.2)
                break

        if not improved:
            t = max(t_min, t * 0.6)
            stall += 1
            if stall >= max_stall:
                f = best_f.copy()
                strength = 0.06 + 0.02 * min(5, stall // max_stall)
                f += rng.normal(0, strength * max(1.0, best_f.mean()), n)
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                else:
                    f = best_f.copy()
                stall = 0
                max_stall += 60
                t = 0.3

    return best_f, best_score


def _polish(a, stop_time, seed=4242):
    n = len(a)
    target_sum = math.sqrt(2.0 * n)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0, None).copy()
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs * target_sum
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 15
    round_num = 0

    while time.monotonic() < stop_time - 4.0 and no_imp < max_no_imp:
        round_num += 1
        improved = False

        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 4.0:
                break
            orig = best[i]
            best_val = orig
            best_val_sc = best_sc

            multipliers = [0.0, 0.1, 0.3, 0.5, 0.7, 0.85, 0.9, 1.1, 1.2, 1.4, 1.8, 2.5]
            for mlt in multipliers:
                nv = orig * mlt if mlt > 0 else 0.0
                best[i] = nv
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score_fast(best)
                if sc < best_val_sc - 1e-14:
                    best_val = nv
                    best_val_sc = sc

            # Refine
            if best_val > 0 and best_val < 500:
                for frac in (0.92, 0.96, 1.04, 1.08, 0.85, 1.15):
                    nv = best_val * frac
                    best[i] = nv
                    if best.sum() < 0.01:
                        best[i] = best_val
                        continue
                    sc = _score_fast(best)
                    if sc < best_val_sc - 1e-14:
                        best_val = nv
                        best_val_sc = sc

            best[i] = best_val
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True

        # Pairs
        n_pairs = 200 if round_num < 4 else 100
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 4.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            if oi < 1e-14 and oj < 1e-14:
                continue
            for mi in (0.0, 0.3, 0.6, 0.8, 1.3, 1.7):
                for mj in (0.0, 0.3, 0.6, 0.8, 1.3, 1.7):
                    if mi == 1.0 and mj == 1.0:
                        continue
                    if mi == 0.0 and mj == 0.0:
                        continue
                    ni = oi * mi if (oi > 1e-14 and mi > 0) else 0.0
                    nj = oj * mj if (oj > 1e-14 and mj > 0) else 0.0
                    best[i], best[j] = ni, nj
                    if best.sum() < 0.01:
                        best[i], best[j] = oi, oj
                        continue
                    sc = _score_fast(best)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        improved = True
            best[i], best[j] = oi, oj

        # Random perturbations
        if not improved:
            for _ in range(60):
                if time.monotonic() > stop_time - 4.0:
                    break
                k = int(rng.integers(n))
                old = best[k]
                sigma = 0.02 * max(1.0, best.mean())
                best[k] = max(0.0, old + rng.normal(0, sigma))
                if best.sum() < 0.01:
                    best[k] = old
                    continue
                sc = _score_fast(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                else:
                    best[k] = old

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc


def _make_init(n, idx, rng):
    i_arr = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    if idx == 0:
        return np.ones(n)
    elif idx == 1:
        return 0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n))
    elif idx == 2:
        sup = np.ones(n) * 0.3
        for fr in (1, 2, 3, 5, 7):
            ph = rng.uniform(0, 2 * math.pi)
            sup += 0.12 * np.cos(2.0 * math.pi * fr * i_arr / n + ph)
        return np.abs(sup) + 0.01
    elif idx == 3:
        return np.abs(rng.normal(0, 1, n)) + 0.02
    elif idx == 4:
        width = n / 4.0
        return np.exp(-0.5 * ((i_arr - center) / width) ** 2)
    elif idx == 5:
        two = np.exp(-0.5 * ((i_arr - n * 0.25) / (n * 0.12)) ** 2) + \
              np.exp(-0.5 * ((i_arr - n * 0.75) / (n * 0.12)) ** 2)
        return two + 0.01
    elif idx == 6:
        # Edge-weighted (arcsine-like)
        d = np.abs(i_arr - center) / (n / 2.0)
        return 1.0 / np.sqrt(np.maximum(d, 0.05))
    elif idx == 7:
        # Slight depression in center
        return 1.0 + 0.3 * np.cos(2.0 * math.pi * i_arr / n)
    elif idx == 8:
        # Quadratic
        x = i_arr / (n - 1)
        return 1.0 + 0.4 * (1 - 4 * (x - 0.5) ** 2)
    elif idx == 9:
        # Central binomial inspired (decay from edges)
        d = np.minimum(i_arr, n - 1 - i_arr).astype(float)
        return 1.0 / np.sqrt(d + 1.0)
    elif idx == 10:
        # Multi-scale
        return (1.0 + 0.2 * np.cos(2 * math.pi * i_arr / n) +
                0.1 * np.cos(4 * math.pi * i_arr / n) +
                0.05 * np.cos(6 * math.pi * i_arr / n))
    else:
        return np.ones(n)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    parent = _get_parent()
    n_p = len(parent)

    fallback = parent.copy()
    fallback_sc = _score_fast(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(100)
        fallback_sc = _score_fast(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_and_update(n, init_arr, stop, sd):
        nonlocal best_seq, best_score
        if time.monotonic() > stop - 2.0:
            return
        try:
            res, sc = _lp_optimize(init_arr, n, stop, sd)
        except Exception:
            return
        if res is None or len(res) == 0:
            return
        if not np.all(np.isfinite(res)):
            return
        res = np.clip(res, 0, 1000)
        if res.sum() < 0.01:
            return
        if sc < best_score - 1e-12:
            try:
                exact = float(evaluate_sequence([float(x) for x in res]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-13:
                best_seq = res.copy()
                best_score = exact

    rng_main = np.random.default_rng(seed)

    # Phase 1: LP on parent length with diverse inits
    n = n_p
    phase1_end = t0 + budget * 0.40
    n_inits = 10
    time_per_init = max(8.0, (phase1_end - t0) / n_inits)

    # First: the parent itself gets the most time
    stop_p = min(phase1_end - 3.0, t0 + budget * 0.15)
    _try_and_update(n, parent.copy(), stop_p, seed + 1)

    for idx in range(n_inits):
        if time.monotonic() > phase1_end - 5.0:
            break
        init = _make_init(n, idx, rng_main)
        # Mix with parent for warm start
        init = 0.5 * init / max(init.sum(), 1e-10) * n + 0.5 * parent
        init = np.clip(init, 0, None)
        stop = min(phase1_end - 3.0, time.monotonic() + time_per_init)
        _try_and_update(n, init, stop, seed + 1000 * (idx + 1))

    # Phase 2: Try different lengths
    phase2_end = t0 + budget * 0.62
    other_lengths = [n + 32, n - 32, n + 64, n + 128, n - 64, n + 192, n + 96]
    other_lengths = [L for L in other_lengths if 64 <= L <= 800]
    # Remove duplicates and sort by closeness to n
    other_lengths = sorted(set(other_lengths), key=lambda x: abs(x - n))
    other_lengths = other_lengths[:6]

    time_per_len = max(6.0, (phase2_end - phase1_end) / max(1, len(other_lengths)))
    for L in other_lengths:
        if time.monotonic() > phase2_end - 5.0:
            break
        init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
        stop = min(phase2_end - 3.0, time.monotonic() + time_per_len)
        _try_and_update(L, init_l, stop, seed + 50000 + L)

    # Phase 3: Warm restart on best
    phase3_end = t0 + budget * 0.80
    if time.monotonic() < phase3_end - 20.0:
        stop_p3 = min(phase3_end - 3.0, time.monotonic() + (phase3_end - time.monotonic() - 5.0))
        _try_and_update(len(best_seq), best_seq.copy(), stop_p3, seed + 999999)

    # Phase 4: Polish
    polish_end = deadline - 12.0
    if time.monotonic() < polish_end - 10.0:
        try:
            b_seq, b_sc = _polish(best_seq, polish_end, seed=4242)
            if b_sc < best_score - 1e-13:
                exact = float(evaluate_sequence([float(x) for x in b_seq]))
                if math.isfinite(exact) and exact < best_score - 1e-12:
                    best_seq = b_seq
                    best_score = exact
        except Exception:
            pass

    # Phase 5: Second polish
    if time.monotonic() < deadline - 18.0:
        stop_p5 = min(deadline - 12.0, time.monotonic() + 60.0)
        try:
            b_seq2, b_sc2 = _polish(best_seq, stop_p5, seed=77777)
            if b_sc2 < best_score - 1e-13:
                exact = float(evaluate_sequence([float(x) for x in b_seq2]))
                if math.isfinite(exact) and exact < best_score - 1e-12:
                    best_seq = b_seq2
                    best_score = exact
        except Exception:
            pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    if final_sc < best_score - 1e-12:
        best_score = final_sc

    return [float(x) for x in best_seq]
