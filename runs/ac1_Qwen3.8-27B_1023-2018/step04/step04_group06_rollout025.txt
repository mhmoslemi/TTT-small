import numpy as np
import time
from scipy.optimize import linprog, minimize

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)

def _get_parent():
    try:
        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(200, dtype=np.float64)
    p = np.clip(p, 0, 1000)
    if p.size == 0 or not np.all(np.isfinite(p)):
        p = np.ones(200, dtype=np.float64)
    return p

def _build_conv_mat(n, a):
    """Build (2n-1) x n matrix where A[k,j] = a[k-j] if valid."""
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            A[k, lo:hi] = a[k - lo: k - lo + (hi - lo)]
    return A

def _lp_solve(A, M, n, tl=2.0):
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n,
                      method='highs', options={'time_limit': tl})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    return np.clip(res.x, 0.0, None)

def _line_search(a, d, M_cur):
    """Find best blend minimizing max conv. a and d are normalized (sum=1)."""
    best_t = 0.0
    best_val = M_cur
    # Coarse grid
    for t in np.linspace(0.05, 1.0, 16):
        cand = (1-t)*a + t*d
        cand = np.maximum(cand, 0)
        s = cand.sum()
        if s < 1e-15:
            continue
        cn = cand / s
        v = float(np.convolve(cn, cn).max())
        if v < best_val:
            best_val = v
            best_t = t
    if best_t == 0.0:
        return None, M_cur
    # Bisection refinement
    lo = max(0.0, best_t - 0.07)
    hi = min(1.0, best_t + 0.07)
    for _ in range(35):
        if hi - lo < 1e-9:
            break
        mid = 0.5*(lo+hi)
        cand = (1-mid)*a + mid*d
        cand = np.maximum(cand, 0)
        s = cand.sum()
        if s < 1e-15:
            lo = mid
            continue
        cn = cand / s
        v = float(np.convolve(cn, cn).max())
        if v < best_val:
            best_val = v
            best_t = mid
            hi = mid
        else:
            lo = mid
    cand = (1-best_t)*a + best_t*d
    cand = np.maximum(cand, 0)
    s = cand.sum()
    if s < 1e-15:
        return None, M_cur
    return cand/s, best_val

def _logsumexp_obj(a, T):
    """Smoothed objective: 2n/(sum a)^2 * (1/T)*log(sum exp(T*(a*a)_k))"""
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return 1e10
    cn = a / s
    conv = np.convolve(cn, cn)
    # log-sum-exp for numerical stability
    max_c = conv.max()
    lse = max_c + np.log(np.sum(np.exp(T*(conv - max_c)))) / T
    return 2.0 * n * lse

def _logsumexp_grad(a, T):
    """Gradient of smoothed objective w.r.t. a (unnormalized)."""
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.zeros(n)
    cn = a / s
    conv = np.convolve(cn, cn)
    max_c = conv.max()
    exp_vals = np.exp(T*(conv - max_c))
    weights = exp_vals / exp_vals.sum()  # softmax weights
    # d/d(cn)_j of sum_k weights_k * conv_k
    # conv_k = sum_i cn_i * cn_{k-i}
    # d(conv_k)/d(cn_j) = cn_{k-j} (if k-j valid, and k!=2j) or 2*cn_j (if k=2j)
    grad_cn = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k]
        if w < 1e-15:
            continue
        lo = max(0, k-n+1)
        hi = min(n, k+1)
        for j in range(lo, hi):
            idx = k - j
            if idx == j:
                grad_cn[j] += 2.0 * w * cn[idx]
            else:
                grad_cn[j] += w * cn[idx]
    # Chain rule: a = s*cn, so d/d a = (1/s) * d/d(cn)
    # Also need to account for normalization... 
    # Actually let's just use cn as the variable and scale back
    # f(cn) = 2n * lse(conv(cn)) where cn/sum(cn) is implicit... 
    # Let's just work with cn directly (sum=1 constraint handled by L-BFGS bounds trick)
    # Simpler: just return grad_cn / s for the original parameterization
    # But for L-BFGS it's easier to optimize over cn with sum=1 constraint
    # Let's just return the gradient of f(a) = 2n*(1/T)*log(sum exp(T*(a/s * a/s)_k))
    # This is getting complicated. Let's use a simpler approach.
    return grad_cn

def _gradient_polish(a0, T_start=5.0, T_end=50.0, n_iters=50, deadline=None):
    """Polish using gradient descent on smoothed objective."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s  # work with sum=1
    
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    
    lr = 0.05
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        T = T_start * (T_end/T_start)**(it/max(n_iters-1,1))
        conv = np.convolve(a, a)
        max_c = conv.max()
        exp_vals = np.exp(T*(conv - max_c))
        weights = exp_vals / exp_vals.sum()
        
        # Gradient of sum_k weights_k * conv_k w.r.t. a_j
        grad = np.zeros(n)
        for k in range(2*n-1):
            w = weights[k]
            if w < 1e-14:
                continue
            lo = max(0, k-n+1)
            hi = min(n, k+1)
            for j in range(lo, hi):
                idx = k - j
                if idx == j:
                    grad[j] += 2.0 * w * a[idx]
                else:
                    grad[j] += w * a[idx]
        
        # Project to zero-sum (since we're working with sum=1)
        grad -= grad.mean()
        
        # Adaptive step size
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        step = lr / gn
        for _ in range(10):
            cand = a - step * grad
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                step *= 0.5
                continue
            cand = cand / cs
            M_new = float(np.convolve(cand, cand).max())
            if M_new < best_M:
                best_M = M_new
                best_a = cand.copy()
                break
            step *= 0.5
        lr *= 0.95
    
    return best_a

def _lp_optimize(a0, stop_time, seed=42, max_stuck=100):
    """LP-based best-response optimization."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a / s
    
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0
    pert_count = 0
    
    while time.monotonic() < stop_time:
        conv = np.convolve(a, a)
        M = float(conv.max())
        k_star = int(conv.argmax())
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > max_stuck:
            # Escalating perturbation
            sigma = min(0.02 + 0.01*stuck/max_stuck, 0.3)
            xx = np.arange(n, dtype=np.float64)
            mode = pert_count % 5
            if mode == 0:
                a *= np.exp(rng.normal(0, sigma, n))
            elif mode == 1:
                f = rng.uniform(0.5, 8.0)
                a *= (1 + sigma*np.sin(2*np.pi*f*xx/n + rng.uniform(0, 6.28)))
            elif mode == 2:
                f1, f2 = rng.uniform(0.5, 5, 2)
                a *= np.exp(sigma*(np.sin(2*np.pi*f1*xx/n) + 0.5*np.cos(2*np.pi*f2*xx/n)))
            elif mode == 3:
                # Targeted: reduce mass at k_star neighborhood
                center = k_star / 2.0
                a *= np.exp(-sigma * np.exp(-((xx-center)/5.0)**2))
            else:
                # Random block changes
                bl = rng.integers(5, max(6, n//5))
                st = rng.integers(0, n-bl)
                a[st:st+bl] *= (1 + sigma*rng.uniform(-1, 1))
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            stuck = 0
            pert_count += 1
            if pert_count > 60:
                break
            continue
        
        # Build LP matrix
        A = _build_conv_mat(n, a)
        made_move = False
        
        # Try LP with current M (and slight relaxations)
        for M_factor in [1.0 + 1e-10, 1.0 + 1e-8, 1.0 + 1e-6]:
            g = _lp_solve(A, M * M_factor, n, tl=2.5)
            if g is None:
                continue
            gs = g.sum()
            if gs < 1e-15:
                continue
            g = g / gs
            if np.max(np.abs(g - a)) < 1e-10:
                continue
            
            cand, val = _line_search(a, g, M)
            if cand is not None and val < M * (1 - 1e-10):
                a = cand
                made_move = True
                break
            # Also try reversed/modified directions
            for frac in [0.5, 0.7]:
                d2 = frac * g + (1-frac) * g[::-1]
                d2 = np.maximum(d2, 0)
                if d2.sum() < 1e-15:
                    continue
                d2 = d2 / d2.sum()
                cand, val = _line_search(a, d2, M)
                if cand is not None and val < M * (1 - 1e-10):
                    a = cand
                    made_move = True
                    break
            if made_move:
                break
        
        if not made_move:
            # Gradient step on active constraint
            grad = np.zeros(n)
            lo = max(0, k_star - n + 1)
            hi = min(n, k_star + 1)
            for j in range(lo, hi):
                idx = k_star - j
                if idx == j:
                    grad[j] += 2.0 * a[idx]
                else:
                    grad[j] += a[idx]
            grad -= grad.mean()
            gn = np.linalg.norm(grad)
            if gn > 1e-14:
                for alpha in [0.03, 0.01, 0.005, 0.002, 0.001]:
                    cand = a - alpha * grad / gn
                    cand = np.maximum(cand, 0)
                    cs = cand.sum()
                    if cs < 1e-12:
                        continue
                    cand /= cs
                    if float(np.convolve(cand, cand).max()) < M * (1 - 1e-10):
                        a = cand
                        made_move = True
                        break
        
        if not made_move:
            a *= np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 1e-12)
            a /= a.sum()
    
    return best_a
