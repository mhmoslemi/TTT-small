def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0)
    if n <= 512:
        c = np.convolve(a, a, mode="full")
    else:
        L = 2 * n - 1
        size = next_fast_len(L)
        F = rfft(a, size)
        c = irfft(F * F, size)[:L]
    return np.maximum(c, 0.0)


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):

def run_lse_lbgfs(u, T, end_time, lse_func, n):

    best_f, _ = lse_func(u, T)

        curv = 0.5 + 2.0 * float(rng.random())
        q = p * np.exp(side * amp * (2.0 * x - 1.0) + curv * (2.0 * x - 1.0) ** 2)
        q = np.maximum(q, 0.0)
        s = float(q.sum())
        if s > 0:
            q /= s
        else:
            q = p
        return q

    def make_geometric_init(n, sd):
        rng = np.random.default_rng(sd)
        r = 0.92 + 0.06 * float(rng.random())
        x = np.arange(n, dtype=np.float64)
        if rng.random() < 0.5:
            a = r ** x
        else:
            a = r ** (n - 1 - x)
        s = float(a.sum())
        if s > 0:
            a /= s
        return a

    def run_full_opt(n, init_arr, end_time, sd, do_lse=True, do_coord=True):
        nonlocal best_arr, best_score
        p = _resample_mid(init_arr, n)
        if p.size != n or float(p.sum()) <= 0:
            p = np.full(n, 1.0 / n)
        p = p / float(p.sum())
        consider(p)

        if time.monotonic() > end_time - 10.0:
            return
        phase_len = max(0.0, float(end_time) - time.monotonic())

        if do_lse and phase_len > 30.0:
            lse_stop = min(float(end_time) - max(20.0, 0.3 * phase_len),
                           time.monotonic() + 0.65 * phase_len)
            if lse_stop > time.monotonic() + 10.0:
                try:
                    p2 = _lse_optimize(p, lse_stop, consider=consider, seed=sd)
                    if p2 is not None and p2.size == n and float(p2.sum()) > 0:
                        p = p2 / float(p2.sum())
                        consider(p)
                except Exception:
                    pass

        rem = float(end_time) - time.monotonic()
        if do_coord and rem > 15.0:
            cstop = min(float(end_time) - 1.5, time.monotonic() + rem - 2.0)
            if cstop > time.monotonic() + 10.0:
                try:
                    p3, _ = _coord_polish(p, cstop, seed=sd, consider=consider)
                    if p3 is not None and p3.size == n and float(p3.sum()) > 0:
                        p = p3 / float(p3.sum())
                        consider(p)
                except Exception:
                    pass
        return p

    n_parent = int(best_arr.size)

    # Phase 1: Optimize at parent length (main effort)
    p1_end = min(phase_limit, t0 + 0.55 * (phase_limit - t0))
    p_opt = run_full_opt(n_parent, best_arr, p1_end, int(seed + 100), do_lse=True, do_coord=True)

    # Phase 2: Perturb and re-optimize at parent length
    if time.monotonic() < phase_limit - 40.0:
        rng2 = np.random.default_rng(seed + 777)
        n_perturb_rounds = 2
        for pr in range(n_perturb_rounds):
            if time.monotonic() > phase_limit - 25.0:
                break
            pert = best_arr * np.exp(rng2.normal(0, 0.04, best_arr.size))
            pert = np.maximum(pert, 0.0)
            ps = float(pert.sum())
            if ps > 0:
                pert /= ps
            sub_end = min(phase_limit - 20.0, time.monotonic() + 0.4 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 20.0:
                run_full_opt(n_parent, pert, sub_end, int(seed + 2000 + pr * 100),
                             do_lse=True, do_coord=True)

    # Phase 3: Try different initializations at parent length
    if time.monotonic() < phase_limit - 50.0:
        for init_seed in range(3):
            if time.monotonic() > phase_limit - 35.0:
                break
            init = make_asym_init(best_arr, n_parent, int(seed + 5000 + init_seed * 777))
            sub_end = min(phase_limit - 30.0, time.monotonic() + 0.25 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 20.0:
                run_full_opt(n_parent, init, sub_end, int(seed + 6000 + init_seed * 333),
                             do_lse=True, do_coord=True)

    # Phase 4: Try a different length
    if time.monotonic() < phase_limit - 80.0:
        n_alt = min(n_parent * 2, 6144) if n_parent <= 4096 else max(n_parent // 2, 512)
        if n_alt != n_parent:
            sub_end = min(phase_limit - 20.0, time.monotonic() + 0.3 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 30.0:
                run_full_opt(n_alt, best_arr, sub_end, int(seed + 8000),
                             do_lse=True, do_coord=True)

    # Phase 5: Final squeeze on the best
    final_end = min(hard - 12.0, t0 + budget - 8.0)
    if final_end > time.monotonic() + 30.0 and best_arr.size > 0:
        nbest = int(best_arr.size)
        rem = final_end - time.monotonic()
        if rem > 60.0 and nbest <= 5000:
            lstop = min(final_end - 35.0, time.monotonic() + 0.4 * rem)
            if lstop > time.monotonic() + 15.0:
                try:
                    p2 = _lse_optimize(best_arr, lstop, consider=consider, seed=int(seed + 99999))
                    if p2 is not None and p2.size == nbest and float(p2.sum()) > 0:
                        p2 = p2 / float(p2.sum())
                        consider(p2)
                except Exception:
                    pass

        cstop = min(final_end - 5.0, time.monotonic() + max(20.0, 0.5 * (final_end - time.monotonic())))
        if cstop > time.monotonic() + 10.0:
            try:
                p3, _ = _coord_polish(best_arr, cstop, seed=int(seed + 123456789), consider=consider)
                if p3 is not None and p3.size == nbest and float(p3.sum()) > 0:
                    p3 = p3 / float(p3.sum())
                    consider(p3)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
