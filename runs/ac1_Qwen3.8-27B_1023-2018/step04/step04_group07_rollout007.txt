            elif strat == 4:
                # Shift
                sh = rng.integers(1, max(2, n // 8) + 1)
                f = np.roll(f, sh * rng.choice([-1, 1]))
                f = np.clip(f, 0, None)
            else:
                f = np.clip(f + rng.normal(0, 0.12, n), 0, None)
            fs2 = float(f.sum())
            if fs2 > 1e-10:
                f /= fs2
            else:
                f = best_f.copy()
            M = float(np.max(_conv_full(f)))
            stall = 0
            iters += 1
            continue

        # Solve LP
        if use_focused > 0 and iters % 3 == 1:
            g_dir = _lp_direction_focused(f, M, n, top_k=150, time_limit=3.0)
            if g_dir is None:
                g_dir = _lp_direction(f, M, n, time_limit=4.0)
        else:
            g_dir = _lp_direction(f, M, n, time_limit=4.0)

        if g_dir is None:
            stall += 4
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            fs2 = f.sum()
            if fs2 > 1e-10:
                f /= fs2
            M = float(np.max(_conv_full(f)))
            iters += 1
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-8:
            stall += 3
            f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
            fs2 = f.sum()
            if fs2 > 1e-10:
                f /= fs2
            M = float(np.max(_conv_full(f)))
            iters += 1
            continue

        t_opt, m_opt = _line_search(f, g_dir, stop_time)

        if t_opt > 0 and m_opt < M - 1e-13:
            f_new = (1.0 - t_opt) * f + t_opt * g_dir
            np.maximum(f_new, 0.0, out=f_new)
            fs3 = float(f_new.sum())
            if fs3 > 1e-10:
                f_new /= fs3
            M_new = float(np.max(_conv_full(f_new)))
            f = f_new
            M = M_new
            if M < best_M - 1e-14:
                best_M = M
                best_f = f.copy()
            stall = 0
        else:
            stall += 1

        iters += 1

    return best_f, best_M


def _coord_descent(a, stop_time, seed=42):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel().copy(), 0, None)
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 6

    while time.monotonic() < stop_time - 2.0 and no_imp < max_no_imp:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            if orig < 1e-12:
                candidates = [0.001, 0.005, 0.01, 0.05, 0.1]
            else:
                candidates = [0.0, orig * 0.3, orig * 0.5, orig * 0.7,
                              orig * 0.9, orig, orig * 1.1, orig * 1.3,
                              orig * 1.6, orig * 2.5]
            best_mult = orig
            best_sc_local = best_sc
            for nv in candidates:
                nv = max(0.0, nv)
                if nv == orig:
                    continue
                best[i] = nv
                s = best.sum()
                if s < 0.01:
                    best[i] = orig
                    continue
                sc = _score_fast(best)
                if sc < best_sc_local - 1e-16:
                    best_sc_local = sc
                    best_mult = nv
            best[i] = best_mult
            if best_sc_local < best_sc - 1e-16:
                best_sc = best_sc_local
                improved = True

        # Pair moves
        n_pairs = min(200, n * 2)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 2.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            tot = oi + oj
            if tot < 1e-12:
                continue
            for frac in (0.0, 0.2, 0.35, 0.5, 0.65, 0.8, 1.0):
                best[i] = frac * tot
                best[j] = (1.0 - frac) * tot
                sc = _score_fast(best)
                if sc < best_sc - 1e-16:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc


def _block_polish(a, stop_time, seed=77):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel().copy(), 0, None)
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    block_sizes = [2, 3, 4, 6, 8, 12]

    while time.monotonic() < stop_time - 2.0 and no_imp < 5:
        improved = False
        for bs in block_sizes:
            step = max(1, bs // 3)
            for start in range(0, n - bs + 1, step):
                if time.monotonic() > stop_time - 2.0:
                    break
                block = best[start:start + bs].copy()
                tot = block.sum()
                if tot < 1e-12:
                    continue
                for _ in range(2):
                    w = rng.dirichlet(np.ones(bs) * 1.5)
                    best[start:start + bs] = w * tot
                    sc = _score_fast(best)
                    if sc < best_sc - 1e-16:
                        best_sc = sc
                        improved = True
                best[start:start + bs] = block
        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc


def _multi_start_lp(f_init, n, t_stop, seed, n_starts=4):
    rng = np.random.default_rng(seed)
    best_f = np.clip(np.asarray(f_init, dtype=np.float64).ravel()[:n].copy(), 0, None)
    fs = best_f.sum()
    if fs > 1e-10:
        best_f /= fs
    else:
        best_f = np.ones(n) / n
    best_M = float(np.max(_conv_full(best_f)))

    inits = [best_f.copy()]
    # Add diverse starts
    i_arr = np.arange(n, dtype=np.float64)
    # Cosine taper
    inits.append(0.5 + 0.5 * np.cos(np.pi * i_arr / (n - 1)))
    # Sine bump
    inits.append(np.maximum(0.3, np.sin(np.pi * i_arr / (n - 1))))
    # Two-hump
    inits.append(0.4 + 0.6 * np.abs(np.sin(2 * np.pi * i_arr / (n - 1))))
    # Random smooth
    kern = np.exp(-0.5 * (np.arange(-8, 9) / 3.0) ** 2)
    kern /= kern.sum()
    rn = np.convolve(rng.normal(0, 1, n), kern, 'same')
    inits.append(np.abs(rn) + 0.05)

    time_per = max(15.0, (t_stop - time.monotonic()) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > t_stop - 10.0:
            break
        stop_i = min(t_stop - 5.0, time.monotonic() + time_per)
        if time.monotonic() >= stop_i:
            break
        try:
            res_f, res_M = _lp_optimize(init, n, stop_i, seed + 9973 * (idx + 1),
                                        use_focused=(2 if idx > 0 else 0))
            if res_f is not None and np.all(np.isfinite(res_f)) and res_f.sum() > 0.01:
                res_M = float(np.max(_conv_full(res_f)))
                if res_M < best_M:
                    best_M = res_M
                    best_f = res_f.copy()
        except Exception:
            pass

    return best_f, best_M


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 25.0

    # Get parent sequence
    try:
        parent = np.asarray(
            globals().get("height_sequence_1", None), dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(384, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(384, dtype=np.float64)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    parent = np.clip(parent, 0.0, 1000.0)
    n = len(parent)

    # Fallback
    fallback = parent.copy()
    fb_sum = float(fallback.sum())
    if fb_sum < 0.01:
        fallback = np.ones(n)
    fallback_sc = _score_fast(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    # Phase 1: LP optimization (65% of time)
    phase1_end = t0 + (budget - 25.0) * 0.65

    try:
        res_f, res_M = _multi_start_lp(parent, n, phase1_end, seed, n_starts=5)
        if res_f is not None and np.all(np.isfinite(res_f)) and res_f.sum() > 0.01:
            res_clipped = np.clip(res_f, 0, 1000)
            exact = float(evaluate_sequence([float(x) for x in res_clipped]))
            if math.isfinite(exact) and exact < best_score:
                best_score = exact
                best_seq = res_clipped.copy()
    except Exception:
        pass

    # Phase 2: Coordinate descent + block polish (25% of time)
    phase2_end = t0 + (budget - 25.0) * 0.90
    if time.monotonic() < phase2_end - 10.0:
        time_per_polish = max(15.0, (phase2_end - time.monotonic()) / 6.0)
        for pidx in range(6):
            p_stop = time.monotonic() + time_per_polish
            if p_stop > phase2_end:
                p_stop = phase2_end
            if time.monotonic() >= p_stop - 3.0:
                break
            try:
                if pidx % 3 == 0:
                    best_seq, best_score = _coord_descent(best_seq, p_stop,
                                                          seed=1000 + pidx * 137)
                elif pidx % 3 == 1:
                    best_seq, best_score = _block_polish(best_seq, p_stop,
                                                         seed=2000 + pidx * 211)
                else:
                    best_seq, best_score = _coord_descent(best_seq, p_stop,
                                                          seed=3000 + pidx * 151)
            except Exception:
                pass

    # Phase 3: Final short LP + polish (remaining time)
    final_stop = t0 + budget - 10.0
    if time.monotonic() < final_stop - 5.0:
        try:
            res_f, res_M = _lp_optimize(best_seq, n, final_stop - 3.0,
                                        seed + 77777, use_focused=2)
            if res_f is not None and np.all(np.isfinite(res_f)) and res_f.sum() > 0.01:
                res_clipped = np.clip(res_f, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in res_clipped]))
                if math.isfinite(exact) and exact < best_score:
                    best_score = exact
                    best_seq = res_clipped.copy()
        except Exception:
            pass

    # Quick final coordinate descent
    if time.monotonic() < final_stop - 2.0:
        try:
            best_seq, best_score = _coord_descent(best_seq, final_stop,
                                                  seed=99999)
        except Exception:
            pass

    # Final validation
    final_seq = np.clip(best_seq, 0, 1000)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    if final_sc < best_score:
        best_score = final_sc
        best_seq = final_seq

    # Ensure we return a valid list
    result = [float(x) for x in np.clip(best_seq, 0, 1000)]
    if not result or any(not math.isfinite(x) for x in result):
        result = [float(x) for x in fallback]

    return result
