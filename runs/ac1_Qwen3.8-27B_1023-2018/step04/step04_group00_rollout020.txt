import numpy as np
import time

def _get_parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n <= 1024:
        c = np.convolve(a, a, mode="full")
    else:
        L = 2 * n - 1
        size = next_fast_len(L)
        F = rfft(a, size)
        c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    if c.size == 0:
        return math.inf
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _score_exact_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.where(np.isfinite(a), a, 0.0)

        return math.inf
    b = np.convolve(a, a)
    if b.size == 0:
        return math.inf
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * a.size * m / (s * s))


def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    return e / s


def _lbfgs_optimize(p, stop_time, consider=None):
    """L-BFGS on log-sum-exp smoothed objective with schedule."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300), dtype=np.float64)
        return uu - float(np.mean(uu))

    u = p_to_u(p)

    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size * 2)
        G = rfft(ww, cross_size * 2)
        c = irfft(F * G, cross_size * 2)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0.0:
            cmax = 1e100
            return cmax, np.zeros_like(uu)
        if T <= 0.0:
            T = max(1e-300, 1e-5 * cmax)
        scaled = (c - cmax) / T
        max_scaled = float(scaled.max())
        exps = np.exp(scaled - max_scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + max_scaled * T + T * float(np.log(sum_exp / float(c.size)))
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    best_p = p.copy()
    best_sc = _score_fft(p)

    stages = (
        (0.05, 0.15),
        (0.02, 0.20),
        (0.008, 0.25),
        (0.003, 0.25),
        (0.001, 0.15),
    )

    start = time.monotonic()
    avail = max(0.0, float(stop_time) - start - 8.0)
    if avail <= 5.0:
        return p

    total_w = float(sum(w for _, w in stages))
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 6.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 5.0, start + avail * (cum / total_w))
        if end_stage <= now + 5.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)

        stale = 0
        while time.monotonic() < end_stage - 1.0 and stale < 4:
            maxit = 30 if n <= 2000 else 20
            opts = {
                "maxiter": maxit,
                "gtol": 1e-10,
                "ftol": 1e-15,
                "maxls": 20,
                "maxcor": 20,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    fval = float(res.fun)
                    if math.isfinite(fval):
                        u = x
                        p_cur = _softmax(u)
                        p_cur = np.maximum(p_cur, 0.0)
                        ss = float(p_cur.sum())
                        if ss > 1e-300:
                            p_cur /= ss
                        sc_cur = _score_fft(p_cur)
                        if math.isfinite(sc_cur) and sc_cur < best_sc:
                            best_sc = sc_cur
                            best_p = p_cur.copy()
                            stale = 0
                        else:
                            stale += 1
                    else:
                        stale += 1
            else:
                stale += 1

        p = _softmax(u)
        p = np.maximum(p, 0.0)
        ss = float(p.sum())
        if ss > 1e-300:
            p /= ss
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, G=40, seed=12345, consider=None):

    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf

    if s < 1e-300 or not math.isfinite(s):
        return p, math.inf
    p = p / s

    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    flat = 0
    sweep = 0
    max_sweeps = 200000

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]

            # Precompute max outside affected range
            max_const = 0.0
            if start > 0:
                mc1 = float(c[:start].max())
                if mc1 > max_const:
                    max_const = mc1
            if end < Ltot:
                mc2 = float(c[end:].max())
                if mc2 > max_const:
                    max_const = mc2

            diag = i  # index of diagonal term: c[2i] is at position 2i - i = i in base

            # Determine search range
            x_hi = max(20.0 * v, 80.0 / n, 0.02, 1e-15)
            x_hi = min(x_hi, 20.0 / n)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = 1.0 / n

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 <= v <= x_hi:
                pts = np.unique(np.concatenate((pts, [v])))

            w_arr = pts - v
            aff = base[None, :] + w_arr[:, None] * infl[None, :]
            maff = np.max(aff, axis=1)
            dval = base[diag] + w_arr * infl[diag] + w_arr * w_arr
            maff = np.maximum(maff, dval)
            maff = np.maximum(maff, max_const)
            denom = 1.0 + w_arr
            valid = denom > 1e-15
            vals = np.full(pts.shape, math.inf, dtype=np.float64)
            if np.any(valid):
                vals[valid] = (2.0 * n * maff[valid]) / (denom[valid] ** 2)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Fine search around best
            dw = x_hi / float(G)
            lo = max(0.0, best_x - 3.0 * dw)
            hi = min(x_hi, best_x + 3.0 * dw)
            if hi > lo:
                fine = np.linspace(lo, hi, 40, dtype=np.float64)
                if 0.0 <= best_x <= x_hi:
                    fine = np.unique(np.concatenate((fine, [best_x])))
                w_f = fine - v
                aff_f = base[None, :] + w_f[:, None] * infl[None, :]
                maff_f = np.max(aff_f, axis=1)
                dval_f = base[diag] + w_f * infl[diag] + w_f * w_f
                maff_f = np.maximum(maff_f, dval_f)
                maff_f = np.maximum(maff_f, max_const)
                den_f = 1.0 + w_f
                valid_f = den_f > 1e-15

                if np.any(valid_f):

                    di = 2 * i

        else:

                        p = np.maximum(p, 0.0)

                        if ss > 0:

                        c = _conv_self(p)

                            if consider is not None:

                        break

                    pass

            if sc_check < best_score:

    return best_p, best_score

            if p4 is not None and p4.size == n_cd and np.all(np.isfinite(p4)) and p4.sum() > 0:
                p4 /= float(p4.sum())
                consider(p4)
                if math.isfinite(_score_fft(p4)) and _score_fft(p4) < _score_fft(p) + 1e-8:
                    p = p4
        except Exception:
            pass

    # Phase 4: Final coordinate sweep
    if time.monotonic() < hard - 8.0:
        G_final = 50 if n_cd <= 2500 else 40
        try:
            p5, _ = _coord_polish(p, hard - 5.0, G=G_final, seed=int(seed + 99999), consider=consider)
            if p5 is not None and p5.size == n_cd and np.all(np.isfinite(p5)) and p5.sum() > 0:
                p5 /= float(p5.sum())
                consider(p5)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0, 1.0, 1.0, 1.0]
        return ls
    except Exception:
        return [1.0, 1.0, 1.0, 1.0]
