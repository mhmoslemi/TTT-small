    """Solve max 1^T g s.t. (a*g)_k <= M, g >= 0."""
    # Build constraint matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = np.arange(max(0, k-n+1), min(n, k+1))
        A[k, js] = a[k - js]

                     method='highs', options={'time_limit': time_limit})

    except Exception:
        pass
    return None

    """Find best t in [0,1] to blend a and g to minimize max_conv."""
    s_g = g.sum()
    if s_g < 1e-12:
        return None, M_current
    g = g / s_g  # normalize
    
    best_t = 0.0
    best_val = M_current

    ts = np.linspace(0, 1, n_steps)
    for t in ts:
        cand = (1-t)*a + t*g
        v = float(np.convolve(cand, cand).max())
        if v < best_val:
            best_val = v
            best_t = t
    
    if best_t == 0:
        return None, M_current

    lo = max(0, best_t - 0.1/n_steps * 2)
    hi = min(1, best_t + 0.1/n_steps * 2)
    gr = (np.sqrt(5)-1)/2
    c1 = hi - gr*(hi-lo)
    c2 = lo + gr*(hi-lo)
    f1 = float(np.convolve((1-c1)*a + c1*g, (1-c1)*a+c1*g).max())
    f2 = float(np.convolve((1-c2)*a + c2*g, (1-c2)*a+c2*g).max())
    for _ in range(30):
        if hi - lo < 1e-7:
            break
        if f1 < f2:
            hi = c2; c2 = c1; f2 = f1
            c1 = hi - gr*(hi-lo)
            f1 = float(np.convolve((1-c1)*a + c1*g, (1-c1)*a+c1*g).max())
        else:

    cand = np.maximum(cand, 0)
    cs = cand.sum()

    cand /= cs
    return cand, best_val

def _subgradient_step(a, k_star, alpha, n):
    """One subgradient step to reduce (a*a)_{k_star}."""
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    for j in range(lo, hi):

        if 0 <= idx < n:
            grad[j] += a[idx]

    grad -= grad.mean()
    cand = a - alpha * grad
    cand = np.maximum(cand, 0)
    cs = cand.sum()
    if cs < 1e-12:
        return None
    cand /= cs
    return cand

    perturb_count = 0

        c = np.convolve(a, a)
        M = float(c.max())

            best_a = a.copy()
            stuck = 0

    return np.clip(res.x, 0.0, None)

                if cand is not None:
                    v = float(np.convolve(cand, cand).max())
                    if v < Mk * (1.0 - 1e-9):
                        a, moved = cand, True
                        break

        if not moved:
            a *= np.exp(rng.normal(0, 0.015, n))
            a = np.maximum(a, 1e-12)
            a /= a.sum()

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 20.0

    parent = _get_parent()
    n = parent.size

    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(32)
        best_score = _score(best_seq)

    def try_update(cand):
        nonlocal best_seq, best_score
        try:
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.clip(np.asarray(cand, dtype=np.float64), 0, 1000).copy()
        except Exception:
            pass

    # Phase 1: Primary LP optimization on parent
    t1 = t0 + min(500.0, budget_s * 0.38)
    try:
        r = _optimize_lp(parent, t1, seed=1, max_perturb=50)
        try_update(r)
    except Exception:
        pass

    # Phase 2: Alternative initializations at same length
    xx = np.arange(n, dtype=np.float64)
    pn = np.clip(parent, 0, 1000).copy()
    ps = pn.sum()
    if ps > 0:
        pn /= ps
    else:
        pn = np.ones(n) / n

    inits = []
    # Triangle
    v = np.maximum(1.0 - np.abs(2.0 * xx / max(n - 1, 1) - 1.0), 0.01)
    inits.append(v / v.sum())
    # Gaussian center
    v = np.exp(-0.5 * ((xx - n / 2) / (n / 5)) ** 2)
    inits.append(v / v.sum())
    # Two bumps
    v = np.exp(-((xx - n / 4) / (n / 6)) ** 2) + np.exp(-((xx - 3 * n / 4) / (n / 6)) ** 2)
    inits.append(v / v.sum())
    # Blend parent with uniform
    inits.append(0.5 * pn + 0.5 * np.ones(n) / n)
    # Cosine window
    v = 0.5 * (1 + np.cos(2 * np.pi * xx / n))
    inits.append(np.maximum(v, 0.01) / np.maximum(v, 0.01).sum())
    # Blended parent with Gaussian
    v = 0.3 * pn + 0.7 * np.exp(-0.5 * ((xx - n / 2) / (n / 4)) ** 2)
    inits.append(v / v.sum())

    t2 = t0 + min(820.0, budget_s * 0.65)
    per_init = max(30.0, max(t2 - time.monotonic(), 0) / max(len(inits), 1))
    for i, init in enumerate(inits):
        if time.monotonic() > t2 - 20.0:
            break
        alloc = min(time.monotonic() + per_init, t2 - 10.0)
        try:
            r = _optimize_lp(init, alloc, seed=200 + i, max_perturb=40)
            try_update(r)
        except Exception:
            pass

    # Phase 3: Quick length variations (try shorter to reduce n factor)
    if time.monotonic() < deadline - 120:
        for dn in [-30, -20, -10, 10, 20]:
            if time.monotonic() > deadline - 30:
                break
            nt = n + dn
            if nt < 16:
                continue
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, parent)
            init_new = np.clip(init_new, 0, 1000)
            sn = init_new.sum()
            if sn < 1e-10:
                continue
            init_new /= sn
            try:
                r = _optimize_lp(init_new, time.monotonic() + 25.0, seed=500 + nt, max_perturb=20)
                try_update(r)
            except Exception:
                pass

    # Phase 4: Final deep polish on best found
    if time.monotonic() < deadline - 15:
        af = np.clip(best_seq, 0, 1000).astype(np.float64)
        afs = af.sum()
        if afs > 1e-10:
            af /= afs
        else:
            af = np.ones(n) / n
        try:
            r = _optimize_lp(af, deadline, seed=7777, max_perturb=80)
            try_update(r)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        fs = evaluate_sequence([float(x) for x in best_seq])
        if not np.isfinite(fs) or fs >= 1e10:
            return [1.0] * 32
    except Exception:
        return [1.0] * 32

    return [float(x) for x in best_seq]
