def _score(a):

        return math.inf

import numpy as np

from scipy.fft import rfft, irfft, next_fast_len


def _conv(a):
    n = len(a)
    if n <= 512:
        return np.convolve(a, a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=3.0):
    A = _build_A(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-8,
                               'primal_feasibility_tolerance': 1e-8})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _subgradient_step(f, n, M, k_star, lr):
    """Take a subgradient step to reduce the peak at k_star."""
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        j1 = k_star - i
        if 0 <= j1 < n:
            g += f[j1]
        j2 = i - k_star
        if 0 <= j2 < n:
            g += f[j2]
        if k_star == 2 * i and 0 <= i < n:
            g = 2.0 * f[i]
        grad[i] = g
    return grad
