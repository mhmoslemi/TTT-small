    """Compute subgradient of max(f*f) at f."""
    n = len(f)
    h = np.convolve(f, f)
    K = max(1, int(top_frac * (2 * n - 1)))
    top_idx = np.argpartition(h, -K)[-K:]

    # where f[m] = 0 for m outside [0, n-1]

        # f[k-j] for j in [0,n-1], valid when 0 <= k-j < n, i.e., k-(n-1) <= j <= k
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            # f[k-j] for j in [j_lo, j_hi] is f[k-j_hi]...f[k-j_lo] = f[j_lo]...f[k-j_lo]... 
            # Actually: for j = j_lo, k-j = k-j_lo; for j = j_hi, k-j = k-j_hi
            # So f[k-j] ranges from f[k-j_hi] to f[k-j_lo]
            # That's f[k-j_hi], f[k-j_hi+1], ..., f[k-j_lo]
            # Which is f[0], f[1], ..., f[k-j_lo] if k-j_hi = 0
            # In general: indices in f are k-j_hi, k-j_hi+1, ..., k-j_lo
            # And these correspond to j = j_hi, j_hi-1, ..., j_lo (reversed)
            # So: grad[j_hi-ii] += f[k-j_hi+ii] for ii = 0..(j_hi-j_lo)

        if k % 2 == 0:

            if 0 <= j_mid < n:

    return grad, h.max()

                no_improve = 0

import time
import math
from scipy.optimize import linprog
from scipy.fft import next_fast_len, rfft, irfft

def _convolve(a):
    """Self-convolution, using FFT for efficiency."""
    n = len(a)
    if n <= 256:
        return np.convolve(a, a)
    size = next_fast_len(2 * n - 1)
    F = rfft(a, size)
    result = irfft(F * F, size)[:2*n-1]
    np.maximum(result, 0, out=result)  # numerical issues
    return result

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    h = _convolve(a)
    return 2.0 * n * float(h.max()) / (s * s)

    """Build the (2n-1) x n matrix A where (A @ g)[k] = (f*g)[k]."""
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_direction(f, n, M, timeout=3.0):
    """LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    A = _build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M + 1e-12)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except:
        return None
    if res.status != 0:
        return None
    g = np.clip(res.x, 0, None)
    if g.sum() < 1e-15:
        return None
    return g
