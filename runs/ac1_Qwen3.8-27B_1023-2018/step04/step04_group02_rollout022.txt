import time
import math
from scipy.optimize import linprog

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)


def _lp_direction(f, M, timeout=3.0):
    """LP: max sum(g) s.t. f*g <= M, g >= 0"""
    n = len(f)
    # f*g has 2n-1 bins. Build the system.
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j, k-j
    # We need: for each k, sum_j f[max(0,k-n+1):min(k+1,n)] * g[...] <= M
    # This is a row of the constraint matrix.
    # Row k: [0, ..., f[0]*g[k], f[1]*g[k-1], ..., f[k]*g[0], 0, ...] (for k < n)
    # More precisely: coefficient of g[j] in row k is f[k-j] if 0<=k-j<n, else 0
    
    # Build A_ub sparsely... but for n=400, dense is 799*400 = 320K, fine.
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # g[j] appears in conv[k] with coefficient f[k-j] if 0<=k-j<n
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[k, lo:hi+1] = f[k - np.arange(lo, hi+1)]
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _subgrid_step(a, eta=0.01, n_iter=200):
    """Subgradient descent on normalized a."""
    n = len(a)
    a = a.copy().astype(float)
    S = a.sum()
    a = a / S  # normalize to sum=1
    best = a.copy()
    best_sc = _score(a)
    
    for it in range(n_iter):
        b = _conv(a)
        M = float(b.max())
        S = a.sum()
        
        # Active set
        tol = 1e-6 * max(1.0, M)
        active = np.where(b >= M - tol)[0]
        
        # Subgradient of max conv
        grad_M = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                grad_M[lo:hi+1] += 2.0 * a[k - np.arange(lo, hi+1)]
        grad_M /= len(active)
        
        # Step
        a_new = np.maximum(0.0, a - eta * grad_M)
        s_new = a_new.sum()
        if s_new < 0.01:
            eta *= 0.5
            continue
        a_new /= s_new
        
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best = a_new.copy()
            a = a_new
            eta = min(eta * 1.05, 0.1)
        else:
            eta *= 0.7
            a = a + 0.5 * (a_new - a)  # partial step? No, just keep a
            a = np.maximum(0, a)
        
        if eta < 1e-6:
            break
    
    return best, best_sc

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

    """Build the constraint matrix for f*g <= M."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[k, lo:hi+1] = f[k - np.arange(lo, hi+1)]
    return A


def _lp_solve(f, M, timeout=2.0):

                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
    except Exception:
        return None

        return None

    parent = np.where(np.isfinite(parent), parent, 0.0)
    n_par = len(parent)

    # Fallback
    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(200)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_update(seq, sc):
        nonlocal best_seq, best_score
        if seq is None or len(seq) == 0:
            return
        seq = np.clip(np.asarray(seq, dtype=float), 0, 1000)
        if not np.all(np.isfinite(seq)) or seq.sum() < 0.01:
            return
        if sc < best_score - 1e-14:
            try:
                exact = float(evaluate_sequence([float(x) for x in seq]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-14:
                best_seq = seq.copy()
                best_score = exact

    # Generate initializations
    n = n_par
    rng0 = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    inits = []
    inits.append(parent.copy())
    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / n)))
    # Parabolic
    ctr = (n - 1) / 2.0
    inits.append(np.maximum(0.0, 1.0 - ((i_arr - ctr) / (0.48 * n)) ** 2))
    # Two-lobe Gaussian
    inits.append(np.exp(-0.5 * ((i_arr - 0.25 * n) / (0.08 * n)) ** 2) +
                 np.exp(-0.5 * ((i_arr - 0.75 * n) / (0.08 * n)) ** 2))
    # Slightly perturbed parent
    inits.append(np.abs(parent + rng0.normal(0, 0.05 * max(1.0, parent.mean()), n)) + 1e-6)
    # Alternating weights
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.4)
    inits.append(alt * (0.5 + 0.5 * np.cos(2.0 * np.pi * i_arr / n)))
    # Uniform
    inits.append(np.ones(n) * 0.01)

    # Deduplicate & normalize
    inits_clean = []
    seen_sums = set()
    for ix in inits:
        ix = np.clip(ix, 0, 1000)
        if ix.sum() < 1e-10:
            continue
        ix = ix / ix.sum()
        key = ix[:10].tobytes()
        if key not in seen_sums:
            seen_sums.add(key)
            inits_clean.append(ix)

    # Phase 1: LP optimization on multiple inits (~55% time)
    phase1_end = t0 + budget * 0.55
    time_per_init = max(10.0, (phase1_end - t0) / max(1, len(inits_clean)))

    for idx, init in enumerate(inits_clean):
        if time.monotonic() > phase1_end - 10.0:
            break
        stop = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        try:
            res, sc = _lp_loop(init.copy(), stop, seed=seed + 1000 * (idx + 1))
            _try_update(res, sc)
        except Exception:
            pass

    # Phase 2: Warm restart LP on best (~15% time)
    phase2_end = t0 + budget * 0.72
    if time.monotonic() < phase2_end - 20.0:
        stop_p2 = min(phase2_end - 5.0, time.monotonic() + (phase2_end - time.monotonic()) * 0.85)
        try:
            res, sc = _lp_loop(best_seq.copy(), stop_p2, seed=seed + 77777)
            _try_update(res, sc)
        except Exception:
            pass

    # Phase 3: CD polish on best (~12% time)
    phase3_end = t0 + budget * 0.84
    if time.monotonic() < phase3_end - 5.0:
        try:
            f_cd, sc_cd = _cd_polish(best_seq.copy(), phase3_end, max_outer=60, seed=seed * 3 + 1)
            _try_update(f_cd, sc_cd)
        except Exception:
            pass

    # Phase 4: Try other lengths (~8% time)
    phase4_end = t0 + budget * 0.92
    if time.monotonic() < phase4_end - 15.0:
        for L in [n - 20, n + 20, n - 50, n + 50, n - 80]:
            if L < 100 or time.monotonic() > phase4_end - 10.0:
                continue
            try:
                init_l = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, len(best_seq)), best_seq)
                init_l = np.maximum(init_l, 0)
                if init_l.sum() < 1e-10:
                    continue
                init_l = init_l / init_l.sum()
                stop_l = min(phase4_end - 5.0, time.monotonic() + max(8.0, (phase4_end - time.monotonic()) * 0.35))
                res, sc = _lp_loop(init_l, stop_l, seed=seed + 90000 + L)
                _try_update(res, sc)
            except Exception:
                continue

    # Phase 5: Final CD + subgradient polish
    final_end = deadline - 5.0
    if time.monotonic() < final_end - 8.0:
        try:
            f_cd, sc_cd = _cd_polish(best_seq.copy(), final_end, max_outer=80, seed=seed * 7 + 42)
            _try_update(f_cd, sc_cd)
        except Exception:
            pass

    # Final subgradient sharpening
    if time.monotonic() < deadline - 4.0:
        try:
            f = best_seq.copy() / best_seq.sum()
            for _ in range(200):
                if time.monotonic() > deadline - 3.0:
                    break
                b = _conv(f)
                M = float(b.max())
                grad_M = _subgrad_dir(f, b, M)
                for eta in [0.005, 0.002, 0.001, 0.0005]:
                    f_try = np.maximum(0, f - eta * grad_M)
                    s_try = f_try.sum()
                    if s_try > 0.01:
                        f_try = f_try / s_try
                        sc_try = _score(f_try)
                        if sc_try < best_score - 1e-14:
                            f = f_try
                            break
        except Exception:
            pass
        _try_update(f, _score(f))

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0, 1.0, 1.0]
        return [float(x) for x in fallback]

    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    if not result:
        result = [1.0]
    if any(math.isnan(x) or math.isinf(x) for x in result):
        result = [float(x) for x in fallback]

    return result
