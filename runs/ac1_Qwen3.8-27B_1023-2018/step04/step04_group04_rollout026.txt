            if math.isfinite(sc2):

        except Exception:
            pass

    # Phase 2: Subgradient refinement
    p2_end = t0 + budget * 0.65
    if time.monotonic() < p2_end:
        f_sg = best_seq / max(best_seq.sum(), 1e-10)
        try:
            res_sg, sc_sg = _subgrad_loop(f_sg, n, p2_end, seed + 400)
            if math.isfinite(sc_sg):
                _update(res_sg * best_seq.sum(), sc_sg)
        except Exception:
            pass

    # Phase 3: Coordinate descent
    p3_end = t0 + budget * 0.82
    if time.monotonic() < p3_end:
        try:
            res_cd, sc_cd = _coord_descent(best_seq, p3_end, seed + 600, max_epochs=60)
            if math.isfinite(sc_cd):
                _update(res_cd, sc_cd)
        except Exception:
            pass

    # Phase 4: Try a few nearby lengths
    p4_end = deadline - 10
    if time.monotonic() < p4_end:
        remaining = p4_end - time.monotonic()
        for L in [n, n + 2, n - 2, n + 6, n - 6]:
            if time.monotonic() > p4_end - 5:
                break
            if L < 4:
                continue
            time_per = min(60.0, remaining / 4.0)
            init_L = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n), parent)
            init_L = init_L / max(init_L.sum(), 1e-10)
            stop_L = min(p4_end, time.monotonic() + time_per)
            if L == n:
                init_L = best_seq / max(best_seq.sum(), 1e-10)
            try:
                rL, sL = _lp_loop(init_L, L, stop_L, seed + 1000 + L)
                if math.isfinite(sL):
                    _update(rL * parent.sum(), sL)
            except Exception:
                pass

    # Phase 5: Final coordinate polish
    if time.monotonic() < deadline - 5:
        try:
            res_fin, sc_fin = _coord_descent(best_seq, deadline - 5, seed + 8000, max_epochs=40)
            if math.isfinite(sc_fin):
                _update(res_fin, sc_fin)
        except Exception:
            pass

    # Final validation
    try:
        fin_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        fin_sc = math.inf

    if not math.isfinite(fin_sc):
        try:
            fb_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            fb_sc = math.inf
        if math.isfinite(fb_sc):
            return [float(x) for x in fallback]
        return [1.0]

    return [float(x) for x in best_seq]
