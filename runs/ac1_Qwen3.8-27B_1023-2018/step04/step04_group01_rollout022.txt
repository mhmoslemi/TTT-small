def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        A[m : m + n, m] = f
    return A


def _lp_direction(f, n, M, time_limit=5.0):
    A = _build_A(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c_obj, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None
    return g / g_sum


def _line_search(f, g, n, conv_f, M, f_sum, rng):
    """Find best t in (0,1] to minimize score(f + t*(g-f))."""
    d = g - f
    lo, hi = 0.0, 1.0
    best_t = 0.0
    best_sc = _score(f)
    
    # Golden section search (faster than binary for unimodal-ish)
    gr = (1 + 5**0.5) / 2
    c1 = 1 - 1.0 / gr
    c2 = 1 / gr
    x1 = c1 * (hi - lo) + lo
    x2 = c2 * (hi - lo) + lo
    
    def _eval_t(t):
        if t <= 0:
            return _score(f)
        f_new = f + t * d
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 0.01:
            return math.inf
        b_new = _conv_full(f_new)
        return 2.0 * n * b_new.max() / (s_new * s_new)
    
    f1 = _eval_t(x1)
    f2 = _eval_t(x2)
    for _ in range(30):
        hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = c1 * (hi - lo) + lo
            f1 = _eval_t(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = c2 * (hi - lo) + lo
            f2 = _eval_t(x2)
    
    best_t = (lo + hi) / 2
    best_sc = min(f1, f2, _eval_t(best_t))
    return best_t, best_sc

    for it in range(max_iters):
        if time.monotonic() > stop_time - 10:

            continue

        d = g - f
        best_t = 0.0

def _coord_descent(f0, stop_time, max_sweeps=500, seed=0):
    n = len(f0)
    a = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    a_sum = float(a.sum())
    if a_sum < 0.01:
        a = np.ones(n) / n
        a_sum = 1.0
    else:
        a = a / a_sum
        a_sum = 1.0
    
    conv = _conv_full(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 8:
            break
        
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time - 8:
                break
            
            v = a[i]
            # Search range for delta
            if v < 1e-12:
                deltas = np.array([0.0, 0.001, 0.005, 0.01, 0.05])
            else:
                deltas = np.unique(np.concatenate([
                    np.linspace(-v, 0, 12),
                    np.linspace(0, v, 8)[1:],
                    np.array([2*v, 4*v, 8*v, 0.5, 1.0])
                ]))
                deltas = deltas[deltas <= 50.0]
            
            best_d = 0.0
            best_sc = best_score
            
            for d in deltas:
                new_v = v + d
                if new_v < -1e-12 or new_v > 1000:
                    continue
                new_v = max(0.0, new_v)
                d_adj = new_v - v
                if abs(d_adj) < 1e-14:
                    sc = best_score
                else:
                    S_new = a_sum + d_adj
                    if S_new < 0.01:
                        continue
                    # Update conv
                    # conv[k] += 2*d_adj*a[k-i] for k in [i, i+n-1]
                    # conv[2i] += d_adj^2
                    # Outside [i, i+n-1]: unchanged
                    seg_lo = i
                    seg_hi = min(i + n, 2 * n - 1)
                    # Quick max computation without full copy:
                    # Inside the segment: conv[k] + 2*d_adj*a[k-i] = conv[k] + 2*d_adj*a[0:n] (shifted)
                    # a[k-i] for k in [i, i+n-1] is a[0], a[1], ..., a[n-1]
                    inside = conv[seg_lo:seg_hi] + 2.0 * d_adj * a
                    outside_max = conv.max()  # This includes the segment, but that's OK for upper bound
                    # Actually for correctness:
                    # outside = conv with segment zeroed
                    # But that's expensive. Instead:
                    # new_max = max(max(inside + quad), max(conv outside segment))
                    inside_max = inside.max()
                    if 2 * i >= seg_lo and 2 * i < seg_hi:
                        inside_max = max(inside_max, conv[2*i] + 2*d_adj*a[i] + d_adj*d_adj)
                    # For outside:
                    # The max of conv outside [seg_lo, seg_hi]
                    # We can precompute this but for now just use overall max as bound
                    # More precisely:
                    if seg_lo > 0:
                        out_left = conv[:seg_lo].max()
                    else:
                        out_left = 0.0
                    if seg_hi < 2 * n - 1:
                        out_right = conv[seg_hi:].max()
                    else:
                        out_right = 0.0
                    out_max = max(out_left, out_right)
                    new_max = max(inside_max, out_max)
                    sc = 2.0 * n * new_max / (S_new * S_new)
                
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_d = d_adj
            
            # Fine search around best_d
            if best_d != 0:
                span = max(abs(best_d) * 0.5, v * 0.1, 1e-4)
                lo_f = max(-v, best_d - span)
                hi_f = min(1.0, best_d + span)
                fine_deltas = np.linspace(lo_f, hi_f, 40)
                for d in fine_deltas:
                    new_v = v + d
                    if new_v < -1e-12 or new_v > 1000:
                        continue
                    new_v = max(0.0, new_v)
                    d_adj = new_v - v
                    if abs(d_adj) < 1e-14:
                        continue
                    S_new = a_sum + d_adj
                    if S_new < 0.01:
                        continue
                    seg_lo = i
                    seg_hi = min(i + n, 2 * n - 1)
                    inside = conv[seg_lo:seg_hi] + 2.0 * d_adj * a
                    inside_max = inside.max()
                    if 2 * i >= seg_lo and 2 * i < seg_hi:
                        inside_max = max(inside_max, conv[2*i] + 2*d_adj*a[i] + d_adj*d_adj)
                    if seg_lo > 0:
                        out_left = conv[:seg_lo].max()
                    else:
                        out_left = 0.0
                    if seg_hi < 2 * n - 1:
                        out_right = conv[seg_hi:].max()
                    else:
                        out_right = 0.0
                    out_max = max(out_left, out_right)
                    new_max = max(inside_max, out_max)
                    sc = 2.0 * n * new_max / (S_new * S_new)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        best_d = d_adj
            
            # Apply update
            if best_d != 0 and best_sc < best_score - 1e-14:
                # Update conv
                seg_lo = i
                seg_hi = min(i + n, 2 * n - 1)
                conv[seg_lo:seg_hi] += 2.0 * best_d * a
                if 2 * i < 2 * n - 1:
                    conv[2 * i] += best_d * best_d
                a[i] = v + best_d
                a_sum += best_d
                best_score = best_sc
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 5:
                # Perturbation
                a = np.clip(a + rng.normal(0, 0.005, n), 0, None)
                a_sum = a.sum()
                if a_sum > 1e-10:
                    a /= a_sum
                    a_sum = 1.0
                else:
                    a = best_a.copy()
                    a_sum = 1.0
                conv = _conv_full(a)
                best_score = 2.0 * n * conv.max()
                no_improve = 0
            elif no_improve >= 15:
                break
        else:
            no_improve = 0
    
    # Final verification
    final_score = _score(best_a)
    if math.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    return best_a, best_score


def _pair_descent(f0, stop_time, max_rounds=100, seed=12345):
    n = len(f0)
    a = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    a_sum = float(a.sum())
    if a_sum < 0.01:
        a = np.ones(n) / n
        a_sum = 1.0
    else:
        a = a / a_sum
        a_sum = 1.0
    
    conv = _conv_full(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    
    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 8:
            break
        improved = False
        n_trials = 80
        for _ in range(n_trials):
            if time.monotonic() > stop_time - 8:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            vi, vj = a[i], a[j]
            if vi < 1e-13 and vj < 1e-13:
                continue
            
            # Try scaling both
            best_pair_sc = best_score
            best_mul = (1.0, 1.0)
            
            for mi in (0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0):
                for mj in (0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0):
                    ni = vi * mi
                    nj = vj * mj
                    if ni + nj > 50:
                        continue
                    a[i], a[j] = ni, nj
                    S = a_sum - vi - vj + ni + nj
                    if S < 0.01:
                        a[i], a[j] = vi, vj
                        continue
                    b_new = _conv_full(a)
                    sc = 2.0 * n * b_new.max() / (S * S)
                    if sc < best_pair_sc - 1e-14:
                        best_pair_sc = sc
                        best_mul = (ni, nj)
            
            a[i], a[j] = best_mul
            if best_pair_sc < best_score - 1e-14:
                conv = _conv_full(a)
                a_sum = a.sum()
                best_score = best_pair_sc
                best_a = a.copy()
                improved = True
        
        if not improved and rnd > 5:
            break
    
    final_score = _score(best_a)
    if math.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    return best_a, best_score

                # Quick coord descent
                if time.monotonic() < t_alloc + budget * 0.02:
                    res2, sc2 = _coord_descent(best_seq, t_alloc + budget * 0.02, max_sweeps=100, seed=seed+2000+idx)
                    _update_best(res2, sc2)
            except Exception:
                pass
    
    # === Phase 5: Try different lengths ===
    phase5_end = min(t0 + budget * 0.87, hard_deadline - 60)
    if time.monotonic() < phase5_end - 30 and best_score > 1.5028:
        other_lengths = []
        for L in [n-20, n-10, n+10, n+20, n+40, n-40, 2*n, n//2]:
            if 20 <= L <= 2048 and L not in other_lengths and L != n:
                other_lengths.append(L)
        other_lengths.sort(key=lambda L: abs(L - n))
        
        time_per = max(20.0, (phase5_end - time.monotonic()) / max(1, len(other_lengths)))
        for L in other_lengths[:5]:
            if time.monotonic() > phase5_end - 15:
                break
            if best_score < 1.5028:
                break
            # Interpolate
            if L != len(best_seq):
                init_l = np.interp(
                    np.linspace(0, 1, L),
                    np.linspace(0, 1, len(best_seq)),
                    best_seq
                )
                init_l = np.clip(init_l, 0, None)
            else:
                init_l = best_seq.copy()
            if init_l.sum() < 0.01:
                init_l = np.ones(L) / L
            init_l = init_l / init_l.sum()
            
            t_stop = min(phase5_end - 5, time.monotonic() + time_per)
            try:
                res, sc = _lp_optimize(init_l, L, t_stop, seed=seed + 50000 + L, max_iters=50)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res = res / res.sum()
                    _update_best(res, sc)
                if time.monotonic() < t_stop + 15:
                    res2, sc2 = _coord_descent(best_seq, t_stop + 15, max_sweeps=80, seed=seed + L)
                    _update_best(res2, sc2)
            except Exception:
                pass
    
    # === Phase 6: Intensive coordinate descent ===
    phase6_end = min(t0 + budget * 0.94, hard_deadline - 20)
    if time.monotonic() < phase6_end - 10:
        try:
            res, sc = _coord_descent(best_seq, phase6_end, max_sweeps=600, seed=99999)
            _update_best(res, sc)
        except Exception:
            pass
    
    # === Phase 7: Pair descent ===
    if time.monotonic() < hard_deadline - 15:
        try:
            res, sc = _pair_descent(best_seq, hard_deadline - 10, max_rounds=50, seed=77777)
            _update_best(res, sc)
        except Exception:
            pass
    
    # === Final coordinate polish ===
    if time.monotonic() < hard_deadline - 10:
        try:
            res, sc = _coord_descent(best_seq, hard_deadline - 5, max_sweeps=100, seed=31337)
            _update_best(res, sc)
        except Exception:
            pass
    
    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = fallback.copy()
    
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf
    
    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]
    
    return [float(x) for x in best_seq]
