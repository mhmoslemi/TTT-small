            # infl[j] = 2*p[j] for j=0..Ls-1 (since k-i ranges from 0 to Ls-1)
            infl = 2.0*p[:Ls]
            
            diag = 2*i - st
            hd = 0 <= diag < Ls
            
            # unchanged max
            mc = 0.0
            if st > 0: mc = max(mc, c[:st].max())
            if en < L: mc = max(mc, c[en:].max())
            
            # Grid
            xhi = max(100.0*v, 200.0/n, 1e-15)
            xhi = min(xhi, 2.0)
            
            pts = np.linspace(0.0, xhi, G+1)
            w = pts - v
            aff = base[None,:] + w[:,None]*infl[None,:]
            maff = aff.max(axis=1)
            if hd:
                maff = np.maximum(maff, base[diag]+w*infl[diag]+w*w)
            maff = np.maximum(maff, mc)
            
            denom = (1.0+w)**2  # S=1
            valid = denom > 1e-30
            sc = np.full(pts.shape, np.inf)
            sc[valid] = 2.0*n*maff[valid]/denom[valid]
            
            im = np.argmin(sc)
            bx = pts[im]; bv = sc[im]
            
            if bv < cur - 1e-16 and 0 <= bx:
                w = bx - v
                if abs(w) > 1e-18:
                    p[i] = bx
                    S_new = 1.0 + w
                    c[st:en] += w*infl
                    if hd: c[2*i] += w*w
                    # Renormalize
                    inv = 1.0/S_new
                    p *= inv
                    c *= inv*inv
                    cur *= inv*inv
                    improved = True
        
        # Recompute c to avoid drift
        c = _conv(p)
        psum = p.sum()
        if psum < 1e-300: break
        p /= psum
        cur = 2.0*n*c.max()
        
        if cur < best - 1e-15:
            best = cur
            best_p = p.copy()
        
        if consider: consider(p)
        
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 8:
                if time.monotonic() < end_t - 10:
                    p *= np.exp(rng.normal(0, perturb_sigma, n))
                    p = np.maximum(p, 0)
                    ps = p.sum()
                    if ps > 0: p /= ps
                    c = _conv(p)
                    cur = 2.0*n*c.max()
                    flat = 0
                else:
                    break
        if time.monotonic() > end_t - 1: break
    
    return best_p, best
