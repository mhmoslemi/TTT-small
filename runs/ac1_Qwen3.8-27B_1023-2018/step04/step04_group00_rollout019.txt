def _lp_dual_step(f, M, stop_time):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)(k)<=M for all k. Return g or None."""
    n = f.size
    if n > 4096 or time.monotonic() > stop_time - 30.0:
        return None
    # Build sparse constraint matrix A: A[k,j] = f[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        js = np.arange(j_lo, j_hi + 1)
        ks = k - js  # f indices
        rows.extend(k for _ in range(len(js)))
        cols.extend(js.tolist())
        vals.extend(f[ks].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-12)
    bounds = [(0, None)] * n
    from scipy.optimize import linprog
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"time_limit": max(1.0, stop_time - time.monotonic() - 10.0)})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None
