def _lp_direction(f, M):
    """Find g that maximizes sum(g) subject to (f*g)_k <= M, g >= 0."""
    n = len(f)
    # Build constraint matrix A: A[k, i] = f[k-i] if valid
    # There are 2n-1 constraints, n variables
    
    n_cons = 2 * n - 1
    nnz = 0
    for k in range(n_cons):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        nnz += (i_end - i_start)
    
    rows = np.zeros(nnz, dtype=int)
    cols = np.zeros(nnz, dtype=int)
    vals = np.zeros(nnz, dtype=float)
    
    idx = 0
    for k in range(n_cons):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-30:
                rows[idx] = k
                cols[idx] = i
                vals[idx] = v
                idx += 1
    
    if idx < nnz:
        rows = rows[:idx]
        cols = cols[:idx]
        vals = vals[:idx]
    
    A = csr_matrix((vals, (rows, cols)), shape=(n_cons, n))
    c_vec = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status not in (0, 3):
        return None
    g = np.clip(res.x, 0, None)
    if g.sum() < 1e-15:
        return None
    return g
