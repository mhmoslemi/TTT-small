    """Subgradient descent on the score."""
    n = len(f)
    f = f.copy()
    S = f.sum()
    best_f = f.copy()
    best_score = _score_fast(f)
    lr = lr_init
    stall = 0
    max_stall = 50

        if time.monotonic() > stop_time:
            break
        b = _conv_full(f)
        M = float(b.max())
        # Find all lags within 1% of max
        active = np.where(b > 0.99 * M)[0]
        S = float(f.sum())

            best_score = score
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

            break
        
        # Compute subgradient: average gradient of active b_k's
        # db_k/df_i = f[k-i] (if valid) * multiplicity
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    if 2 * i == k:
                        grad[i] += f[j]  # self-term
                    else:
                        grad[i] += 2.0 * f[j]

        # But for subgradient, we just need direction: minimize score => minimize M/S^2
        # Direction: -grad(M) * S + 2*M*grad(S) = -A*f + 2*M*f... 
        # Simpler: just use the gradient of M/S^2
        # d(M/S^2)/df_i = (dM/df_i * S^2 - M * 2*S * f_i) / S^4 = (dM/df_i * S - 2*M*f_i/S) / S^3
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        # Score gradient direction
        sg = grad * S - 2.0 * M * f / S
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg = -sg / sg_norm  # negative direction (minimize)
        
        # Take step
        step = lr * np.linalg.norm(f) / max(1.0, n)
        f_new = f + step * sg
        f_new = np.clip(f_new, 0, None)
        fs = f_new.sum()
        if fs < 0.01:
            f = best_f.copy()
            continue
        # Re-normalize to keep S constant (optional)
        f = f_new
        # Decay learning rate
        lr *= 0.98
        
    return best_f, best_score
