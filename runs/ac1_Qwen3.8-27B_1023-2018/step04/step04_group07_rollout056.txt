import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)  # clip tiny negative values from FFT
    return c

def _score(a):
    """Compute the AC1 score."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """Build the (2n-1) x n matrix where A @ g = conv(f, g)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_solve(f, n, M, time_limit=10.0):
    """Solve the LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': time_limit},
        )
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum

    iter_count = 0
    prev_M = M

            stall = max(0, stall - 3)
        else:
            stall += 1
        prev_M = M

        if stall > max_stall:

            if strategy == 0:
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)

            elif strategy == 2:

                f = 0.7 * f + 0.3 * mono
            else:

            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:

def _polish(a, stop_time, seed=99):
    """Coordinate descent polish."""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel().copy(), 0, None)
    fs = best.sum()
    if fs < 1e-10:
        return best, math.inf
    best = best / fs
    best_sc = _score(best)
    rng = np.random.default_rng(seed)
    no_imp = 0
    max_no_imp = 6

    while time.monotonic() < stop_time - 2.0 and no_imp < max_no_imp:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            if orig > 1e-10:
                mults = (0.0, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0)
            else:
                mults = (0.001, 0.01, 0.05, 0.1, 0.5, orig + 0.01)
            best_val_sc = best_sc
            best_nv = orig
            for mlt in mults:
                nv = orig * mlt if orig > 1e-10 else mlt
                best[i] = nv
                if best.sum() < 0.01:
                    best[i] = orig
                    continue
                sc = _score(best)
                if sc < best_val_sc - 1e-15:
                    best_val_sc = sc
                    best_nv = nv
            best[i] = best_nv
            if best_val_sc < best_sc - 1e-14:
                best_sc = best_val_sc
                improved = True

        # Pair moves
        for _ in range(min(200, n * 2)):
            if time.monotonic() > stop_time - 2.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            total = oi + oj
            if total < 1e-12:
                continue
            for frac in (0.0, 0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9, 1.0):
                best[i] = frac * total
                best[j] = (1.0 - frac) * total
                sc = _score(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj

        if not improved:
            no_imp += 1
        else:
            no_imp = 0

    return best, best_sc
