def _solve_lp(f, M):
    """Solve the LP: max sum(b) s.t. b >= 0, (f*b)(k) <= M for all k."""
    n = f.size
    # Constraint matrix: (2n-1) x n
    # A[k,j] = f[j] if 0 <= k-j <= n-1 else 0
    
    # For efficiency, build sparse matrix
    # Actually for n=3072, the dense matrix is 6143 x 3072 ≈ 19M entries ≈ 152MB
    # Sparse: n^2 ≈ 9.4M nonzeros
    
    # Let me use cvxpy
    b = cp.Variable(n, nonneg=True)
    obj = cp.Maximize(cp.sum(b))
    
    # Build constraints
    # (f*b)(k) = sum_j f[j] * b[k-j] for valid j
    # This is a matrix-vector product where A[k,j] = f[j] if 0 <= k-j <= n-1
    
    # For efficiency, let's build the constraint as A @ b <= M * ones
    # where A is (2n-1) x n
    
    # Actually, for large n, building the full matrix might be memory-intensive.
    # Let me think of an alternative.
    
    # (f*b) is the full convolution of f and b.
    # We can express this as: for each k, sum_{j=max(0,k-n+1)}^{min(k,n-1)} f[j]*b[k-j]
    
    # This is the same as the diagonal bands of an outer product.
    # Let's build A as a sparse matrix.
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [A @ b <= M]
    
    prob = cp.Problem(obj, constraints)
    
    # Try different solvers
    try:
        prob.solve(solver=cp.CBC, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.GLPK, verbose=False)
            except Exception:
                b.value = f.copy()
    
    if b.value is None:
        return f.copy()
    
    g = np.asarray(b.value, dtype=np.float64).ravel()
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s < 1e-12:
        return f.copy()
    return g / s

def _solve_lp_direction(f, M):
    """Solve LP: max sum(b) s.t. b >= 0, (f*b)(k) <= M for all k."""
    n = f.size
    if n == 0:
        return f.copy()
    
    # Build sparse constraint matrix A (2n-1) x n
    # A[k,j] = f[j] if max(0, k-n+1) <= j <= min(k, n-1)
    
    row_parts = []
    col_parts = []
    val_parts = []
    
    for j in range(n):
        r0 = j
        r1 = min(j + n, 2*n - 1)
        cnt = r1 - r0
        if cnt > 0:
            col_parts.append(np.full(cnt, j, dtype=np.int32))
            row_parts.append(np.arange(r0, r1, dtype=np.int32))
            val_parts.append(np.full(cnt, f[j], dtype=np.float64))
    
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # linprog: minimize c @ x s.t. A_ub @ x <= b_ub, bounds
    # We want to maximize sum(b), so minimize -sum(b)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            g = np.maximum(g, 0.0)
            s = float(g.sum())
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    
    # Fallback to cvxpy
    try:
        b = cp.Variable(n, nonneg=True)
        obj = cp.Maximize(cp.sum(b))
        constraints = [A @ b <= M]
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            prob.solve(solver=cp.GLOP, verbose=False)
        if b.value is not None:
            g = np.asarray(b.value, dtype=np.float64)
            g = np.maximum(g, 0.0)
            s = float(g.sum())
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    
    return f.copy()
