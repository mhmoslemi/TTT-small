def _conv_full(a):
    n = len(a)
    if n == 1:
        return np.array([a[0]*a[0]])
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    c = irfft(F*F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _conv_cross(a, b):
    n, m = len(a), len(b)
    fsize = next_fast_len(n+m-1)
    return irfft(rfft(a, fsize)*rfft(b, fsize), fsize)[:n+m-1]

def _score(a):

    conv = _conv_full(best)
    best_sc = float(_score(best))
    best_M = float(conv.max())
    rng = np.random.default_rng(seed)
    no_imp = 0
    while time.monotonic() < stop - 1.0 and no_imp < 12:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop - 1.0:
                break
            ai = best[i]
            # Compute factors: for each m, 2*a[m-i] if valid, and delta^2 if m=2i
            # new_conv[m] = conv[m] + 2*delta*a[m-i] + delta^2*(m==2i)
            lin = np.zeros(2*n-1)
            quad = np.zeros(2*n-1)
            k = np.arange(2*n-1) - i
            valid = (k >= 0) & (k < n)
            lin[valid] = 2.0 * best[k[valid]]
            if 2*i < 2*n-1:
                quad[2*i] = 1.0
            # Try candidate deltas
            if ai > 1e-12:
                deltas = ai * np.array([-0.8, -0.5, -0.3, -0.15, -0.05, 0.05, 0.15, 0.3, 0.5, 0.8, 1.2])
            else:
                deltas = np.array([0.001, 0.005, 0.01, 0.05, 0.1])
            best_d, best_mmax = 0.0, best_M
            for d in deltas:
                nv = ai + d
                if nv < -1e-15:
                    continue
                nv = max(0.0, nv)
                d2 = nv - ai
                new_max = float((conv + d2*lin + d2*d2*quad).max())
                if new_max < best_mmax - 1e-15:
                    best_mmax = new_max
                    best_d = d2
            if abs(best_d) > 1e-14:
                best[i] += best_d
                conv = np.maximum(conv + best_d*lin + best_d*best_d*quad, 0.0)
                cur_M = float(conv.max())
                if cur_M < best_M - 1e-14:
                    best_M = cur_M
                    cur_sc = float(_score(best))
                    if cur_sc < best_sc - 1e-14:
                        best_sc = cur_sc
                    improved = True
            # Also try pair moves
        # Pair moves
        for _ in range(min(n, 300)):
            if time.monotonic() > stop - 1.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            tot = oi + oj
            if tot < 1e-12:
                continue
            for frac in [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]:
                best[i] = frac*tot
                best[j] = (1-frac)*tot
                conv2 = _conv_full(best)
                m2 = float(conv2.max())
                if m2 < best_M - 1e-14:
                    best_M = m2
                    conv = conv2
                    cur_sc = float(_score(best))
                    if cur_sc < best_sc - 1e-14:
                        best_sc = cur_sc
                    improved = True
        if not improved:
            no_imp += 1
        else:
            no_imp = 0
    return best, best_sc

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 30.0

    # Get parent
    try:
        h = globals()["height_sequence_1"]
        parent = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(384)
    parent = np.clip(np.where(np.isfinite(parent), parent, 0.0), 0.0, 1000.0)
    if parent.size == 0:
        parent = np.ones(384)
    n_p = len(parent)
    n = n_p

    # Fallback
    fallback = parent.copy()
    try:
        fb_sc = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fb_sc = math.inf
    if not math.isfinite(fb_sc):
        fallback = np.ones(n)/n
        fb_sc = float(evaluate_sequence([1.0]))
    best_seq = fallback.copy()
    best_score = fb_sc

    # Phase 1: LP optimization (60% of budget)
    p1_end = t0 + budget * 0.60
    rng_main = np.random.default_rng(seed)

    inits = [("parent", parent.copy())]
    # Taper
    ia = np.arange(n, dtype=np.float64)
    inits.append(("taper", 0.5 + 0.5*np.cos(np.pi*ia/(n-1))))
    # Noisy parent
    kern = np.exp(-0.5*(np.arange(-5,6)/2.0)**2)
    kern /= kern.sum()
    sn = np.convolve(rng_main.normal(0,1,n), kern, 'same')
    inits.append(("noisy", np.abs(parent + 0.1*sn)))
    # Multi-freq
    sup = np.ones(n)*0.1
    for fr in [1,2,3,5,7]:
        ph = rng_main.uniform(0, 2*math.pi)
        sup += 0.3*np.cos(2*math.pi*fr*ia/n + ph)
    inits.append(("mf", np.abs(sup)+0.001))

    tp = max(20.0, (p1_end - t0)/len(inits))
    for idx, (nm, init) in enumerate(inits):
        if time.monotonic() > p1_end - 15.0:
            break
        stop_p = min(p1_end - 5.0, time.monotonic() + tp)
        if time.monotonic() >= stop_p:
            break
        try:
            res, _ = _lp_opt(init, n, stop_p, seed+10000*(idx+1))
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                rc = np.clip(res, 0, 1000)
                es = float(evaluate_sequence([float(x) for x in rc]))
                if math.isfinite(es) and es < best_score:
                    best_score = es
                    best_seq = rc.copy()
        except Exception:
            pass

    # Phase 2: More LP from best (15% of budget)
    p2_end = min(t0 + budget*0.75, deadline - 60.0)
    if time.monotonic() < p2_end - 10.0:
        try:
            res, _ = _lp_opt(best_seq, n, p2_end, seed+99999)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                rc = np.clip(res, 0, 1000)
                es = float(evaluate_sequence([float(x) for x in rc]))
                if math.isfinite(es) and es < best_score:
                    best_score = es
                    best_seq = rc.copy()
        except Exception:
            pass

    # Phase 3: Coordinate polish (remaining time minus margin)
    polish_stop = deadline - 15.0
    if time.monotonic() < polish_stop - 20.0:
        try:
            best_seq, best_score = _coord_polish(best_seq, polish_stop, seed=4242)
        except Exception:
            pass

    # Final validation
    final = np.clip(best_seq, 0, 1000)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final]))
    except Exception:
        final_sc = math.inf
    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]
    if final_sc < best_score:
        best_score = final_sc
    return [float(x) for x in final]
