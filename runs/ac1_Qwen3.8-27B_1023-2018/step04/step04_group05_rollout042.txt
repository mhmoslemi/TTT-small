def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

    """A[k,j] = f[k-j] if 0<=k-j<n, else 0. Shape (2n-1, n)."""
    # Efficient construction using the convolution structure
    # Row k of A is: [f[k], f[k-1], ..., f[0], 0, ..., 0, 0, ..., f[k-n+1]]
    # More precisely: A[k,j] = f[k-j] if 0 <= k-j < n else 0
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        # A[k,j] = f[k-j] for k = j, j+1, ..., j+n-1
        A[j:j+n, j] = f
    return A

def _lp_optimize(f0, n, stop_time, seed):
    """Main LP-based iterative optimization."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n)
        f_sum = n

    target_sum = math.sqrt(2.0 * n)
    f = f / f_sum * target_sum

    best_f = f.copy()
    best_score = _score(f)

    t = 0.3
    t_min = 0.05
    t_max = 0.9
    stall = 0
    max_stall = 100
    iteration = 0

    while time.monotonic() < stop_time - 10.0:
        iteration += 1
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, n, M, timeout=6.0)
        if g is None:
            stall += 5
            if stall > max_stall:
                # Reset with perturbation
                f = best_f.copy()
                idx = rng.choice(n, size=min(50, n), replace=False)
                f[idx] *= rng.uniform(0.3, 3.0, len(idx))
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                stall = 0
                max_stall += 50
                t = 0.3
            continue

        g_sum = float(g.sum())
        if g_sum < 1e-15:
            stall += 2
            continue
        g = g * (target_sum / g_sum)

        dist = float(np.linalg.norm(g - f)) / max(1.0, float(np.linalg.norm(f)))
        if dist < 1e-10:
            stall += 3
            if stall > max_stall:
                f = best_f.copy()
                idx = rng.choice(n, size=min(30, n), replace=False)
                f[idx] *= rng.uniform(0.2, 4.0, len(idx))
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                stall = 0
                max_stall += 50
                t = 0.3
            continue

        # Line search
        improved = False
        candidates = sorted(
            {max(t_min, min(t_max, c)) for c in
             [t, t * 0.7, t * 0.5, t * 1.3, t * 1.5, 0.1, 0.3, 0.5, 0.7, 1.0]},
            reverse=True
        )
        for tc in candidates:
            f_new = (1.0 - tc) * f + tc * g
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 0.01:
                continue
            f_new = f_new / fs_new * target_sum
            sc = _score(f_new)
            if sc < best_score - 1e-13:
                f = f_new.copy()
                best_f = f.copy()
                best_score = sc
                improved = True
                t = min(t_max, tc * 1.2)
                stall = 0
                break

        if not improved:
            stall += 1
            t = max(t_min, t * 0.6)
            if stall >= max_stall:
                f = best_f.copy()
                strength = 0.1
                f += rng.normal(0, strength * max(1.0, best_f.mean()), n)
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                stall = 0
                max_stall += 40
                t = 0.3

    return best_f, best_score
