from scipy.fft import rfft, irfft, next_fast_len

def _conv(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    if n <= 512:
        return np.convolve(a, a, mode='full')

The LP-based method is essentially a projection method: project onto the set {g : conv(f,g) <= M, g >= 0}. The issue is that this set depends on the current f, so it's a "moving target." The method converges to a fixed point where f = g (the direction equals the current point), which means conv(f,f) <= M where M = max(conv(f,f)), i.e., all convolution coefficients are equal to M. This is the "uniform convolution" condition.

If we could achieve perfectly uniform convolution (all c[k] = M), the score would be 2n*M/S0^2. With sum p = 1, M = 1/(2n-1) (since sum of all c[k] = 1), and score = 2n/(2n-1) ≈ 1.

But in practice, we can't achieve perfectly uniform convolution with a nonnegative sequence on a finite support. The best we can do is get close to uniform, and the deviation from uniform determines how much above the lower bound we are.

The LP method should converge to a local optimum where the convolution is as flat as possible. The question is whether we're getting to the best local optimum, or getting stuck.

    threshold = M * 0.999

            if f_new.sum() < 1e-10:
                continue
            f_new = f_new / f_new.sum()
            sc = _score(f_new)
            if sc < best_score - 1e-14:
                f = f_new
                best_f = f.copy()
                best_score = sc
                cur_score = sc
                t = min(0.9, ti * 1.2)
                improved = True
                stall = 0
                break
            elif sc < cur_score - 1e-14:
                f = f_new
                cur_score = sc
                t = min(0.9, ti)
                improved = True
                stall = 0
                break
            else:
                t = max(0.01, t * 0.5)

        if not improved:
            stall += 1
            if stall >= max_stall:
                # Reset with perturbation
                f = np.clip(best_f + rng.normal(0, 0.05 * max(1.0, best_f.mean()), n), 0, None)
                if f.sum() < 1e-10:
                    f = best_f.copy()
                else:
                    f = f / f.sum()
                cur_score = _score(f)
                t = 0.3
                stall = 0
                max_stall = min(max_stall + 100, 600)

        it += 1
        # Periodic CD polish
        if it % 20 == 0 and stall < max_stall // 3 and time.monotonic() < stop - 10.0:
            f_c, sc_c = _cd_polish(f, min(stop - 5.0, time.monotonic() + 15.0),
                                   max_outer=3, seed=seed + it)
            if sc_c < best_score - 1e-14:
                best_score = sc_c
                best_f = f_c.copy()
                f = f_c.copy()
                cur_score = sc_c

    # Final CD
    if time.monotonic() < stop - 5.0:
        f_c, sc_c = _cd_polish(best_f, min(stop - 1.0, time.monotonic() + 20.0),
                               max_outer=5, seed=seed + 9999)
        if sc_c <= best_score + 1e-14:
            best_f = f_c
            best_score = sc_c

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 15.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(400, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(400, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    n = len(parent)

    # Fallback
    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(100)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_candidate(cand):
        nonlocal best_seq, best_score
        if cand is None or len(cand) == 0:
            return
        cand = np.clip(np.asarray(cand, dtype=np.float64), 0, 1000)
        if cand.sum() < 0.01 or not np.all(np.isfinite(cand)):
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in cand]))
        except Exception:
            sc = math.inf
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_seq = cand.copy()

    # Generate initializations
    rng = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)
    inits = [parent.copy()]

    # Uniform
    inits.append(np.ones(n) / n)
    # Alternating (anti-symmetric about center)
    for a_val in [0.3, 0.5, 0.7, 0.85]:
        alt = 1.0 + a_val * np.cos(np.pi * i_arr)
        alt = np.maximum(alt, 0.0)
        if alt.sum() > 1e-6:
            inits.append(alt / alt.sum())
    # Cosine modes
    for mode in [1, 2, 3, 5, 7, 10, 16, 25]:
        c = 1.0 + 0.6 * np.cos(2.0 * np.pi * mode * i_arr / n)
        c = np.maximum(c, 0.0)
        if c.sum() > 1e-6:
            inits.append(c / c.sum())
    # Gaussian (centered)
    for sigma in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((i_arr - n / 2) / (sigma * n)) ** 2)
        inits.append(g / g.sum())
    # Edge-concentrated (anti-Gaussian)
    edge = np.exp(-((i_arr - n / 2) / (0.2 * n)) ** 2)
    edge = 1.0 / (edge + 0.1)
    inits.append(edge / edge.sum())
    # Two-lobe
    for split in [0.3, 0.5, 0.7]:
        tl = (np.exp(-((i_arr - split * n * 0.3) / (0.1 * n)) ** 2) +
              np.exp(-((i_arr - (1 - split) * n * 0.7) / (0.1 * n)) ** 2))
        inits.append(tl / tl.sum())
    # Random Dirichlet
    for alpha in [0.3, 1.0, 3.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))
    # Perturbed parent
    for sig in [0.1, 0.3]:
        pp = np.abs(parent + rng.normal(0, sig * max(1.0, np.mean(parent)), n))
        if pp.sum() > 1e-6:
            inits.append(pp / pp.sum())

    # Filter valid inits
    inits = [x for x in inits if x is not None and x.sum() > 0.01 and np.all(np.isfinite(x))]

    # Phase 1: LP optimization with multiple starts (~55% of time)
    phase1_end = t0 + budget * 0.55
    time_per_init = max(15.0, (phase1_end - t0 - 10.0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10.0:
            break
        if best_score < 1.49:
            break
        stop_p1 = min(phase1_end - 5.0, time.monotonic() + time_per_init)
        try:
            res, sc = _lp_loop(init, n, stop_p1, seed + 1000 * (idx + 1))
            _try_candidate(res)
        except Exception:
            pass

    # Phase 2: Gradient refinement on best (~15% of time)
    phase2_end = t0 + budget * 0.70
    if time.monotonic() < phase2_end - 10.0:
        try:
            f_g, sc_g = _grad_descent(best_seq, phase2_end - 5.0, max_iter=5000)
            _try_candidate(f_g)
        except Exception:
            pass
        # CD after gradient
        if time.monotonic() < phase2_end - 5.0:
            try:
                f_c, sc_c = _cd_polish(best_seq, phase2_end - 2.0, max_outer=10, seed=seed + 55555)
                _try_candidate(f_c)
            except Exception:
                pass

    # Phase 3: Re-run LP on best with more time (~15% of time)
    phase3_end = t0 + budget * 0.85
    if time.monotonic() < phase3_end - 15.0:
        stop_p3 = min(phase3_end - 5.0, time.monotonic() + (phase3_end - time.monotonic()) * 0.8)
        try:
            res, sc = _lp_loop(best_seq.copy(), len(best_seq), stop_p3, seed + 77777)
            _try_candidate(res)
        except Exception:
            pass

    # Phase 4: Try slightly different lengths (~8% of time)
    phase4_end = t0 + budget * 0.93
    if time.monotonic() < phase4_end - 12.0 and best_score > 1.50:
        for L in [n - 16, n + 16]:
            if L < 64 or L > 2000:
                continue
            if time.monotonic() > phase4_end - 10.0:
                break
            init_l = np.interp(np.linspace(0, 1, L),
                               np.linspace(0, 1, len(best_seq)), best_seq)
            stop_l = min(phase4_end - 5.0, time.monotonic() + 25.0)
            try:
                res, sc = _lp_loop(init_l, L, stop_l, seed + 90000 + L)
                _try_candidate(res)
            except Exception:
                pass

    # Phase 5: Final CD polish (~7% of time)
    final_end = hard_deadline - 8.0
    if time.monotonic() < final_end - 5.0:
        try:
            f_c, sc_c = _cd_polish(best_seq, final_end, max_outer=80, seed=seed * 3 + 42)
            _try_candidate(f_c)
        except Exception:
            pass

    # Final validation
    _try_candidate(best_seq)

    # Ensure valid return
    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    if not result or any(math.isnan(x) or math.isinf(x) for x in result):
        result = [float(max(0.0, min(1000.0, x))) for x in fallback]
    if not result:
        result = [1.0]
    return result
