def _fft_conv(a):
    """Full self-convolution via FFT."""
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return c


def _score_fft(a):
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _fft_conv(a)
    m = float(c.max())
    return 2.0 * n * m / (s * s)
