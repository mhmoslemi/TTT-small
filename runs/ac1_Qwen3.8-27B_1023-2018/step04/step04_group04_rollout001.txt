    return p

def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

    conv_k = sum_i a[i] * a[k-i]

    Actually, for a vector a of length n:
    conv_k = sum_{i: max(0,k-n+1) <= i <= min(k,n-1)} a_i * a_{k-i}
    
    For the gradient:
    d(conv_k)/da_j = [k-j is valid] * a_{k-j} + [k-j is valid] * a_{k-j}

    d(conv_k)/da_j = 2 * a[(k-j) mod n] if we were doing circular convolution.

    d(conv_2)/db = 2b. k-j = 2-1 = 1. a[1] = b. So 2*a[k-j] = 2b. ✓
    d(conv_2)/dc = 2a. k-j = 2-2 = 0. a[0] = a. So 2*a[k-j] = 2a. ✓
    d(conv_2)/dd = 0. k-j = 2-3 = -1. Not valid. ✓

    d(conv_k)/da_j = 2 * a[k-j] if 0 <= k-j < n, else 0.

    Actually more precisely: grad[j] = 2 * a[k-j] if 0 <= k-j < n, else 0.
    This means: grad = 2 * reversed_a shifted, which can be computed as:
    grad[j] = 2 * a[k-j] for j in [max(0,k-n+1), min(k, n-1)]
    = 2 * a[k - np.arange(max(0,k-n+1), min(k,n-1)+1)]
    
    For the subgradient of max_k conv_k, we take the average over active k's:
    subgrad[j] = mean over active k of 2*a[k-j] (with validity check)
    """
    n = len(a)
    b = _fft_conv(a)
    m = float(np.max(b))
    eps = max(1e-14, 1e-10 * m)
    active = np.flatnonzero(b >= m - eps)
    grad = np.zeros(n)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += 2.0 * a[k - idx]
    if len(active) > 0:
        grad /= len(active)
    return grad, m, active

def _logsumexp_conv(a, tau):
    """Smooth approximation of max(conv) using log-sum-exp."""
    b = _fft_conv(a)
    # For numerical stability
    bm = np.max(b)
    if bm < 1e-30:
        return 0.0, np.zeros_like(a)
    logsum = bm + tau * np.log(np.sum(np.exp((b - bm) / tau)))
    # Gradient: (1/tau) * sum(exp((b_k - max)/tau) * db_k/da_j) / sum(exp((b_k-max)/tau))
    # = weighted average of db_k/da_j weighted by softmax((b-k_max)/tau)
    w = np.exp((b - bm) / tau)
    w /= np.sum(w)
    grad = np.zeros(len(a))
    n = len(a)
    for k in range(2*n-1):
        if w[k] < 1e-15:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += w[k] * 2.0 * a[k - idx]
    return logsum, grad

def _lp_direction(f, n, M, timeout=8.0):
    """Find descent direction via LP: max sum(g) s.t. ||f*g||_inf <= M, g >= 0."""
    # Build constraint matrix A such that (A @ g)[k] = (f * g)[k] for k = 0..2n-2
    # (f*g)[k] = sum_j f[k-j] * g[j] where k-j is valid
    # For fixed j: contributes f[k-j] * g[j] to (f*g)[k]
    # So A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        # A[k, j] = f[k-j] for k where 0 <= k-j < n
        # i.e., k in [j, j+n-1]
        A[j:j+n, j] = f
    
    c = -np.ones(n)  # minimize -sum(g) = max sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout, 'dual_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _lp_optimize(f0, n, stop_time, seed, max_iters=5000):
    """LP-based iterative optimization."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs
    
    best_f = f.copy()
    best_sc = _score(f)
    t_param = 0.3
    stall = 0
    max_stall = 400
    it = 0
    
    while time.monotonic() < stop_time:
        b = _fft_conv(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        # Try to find LP direction
        g = _lp_direction(f, n, M, timeout=6.0)
        
        if g is None:
            # Fallback: use subgradient direction
            sg, M, _ = _grad_maxconv(f)
            sg -= np.mean(sg)  # project to tangent plane of simplex
            if np.linalg.norm(sg) > 1e-15:
                g = np.clip(-sg, 0, None)
                gs = g.sum()
                if gs < 1e-15:
                    g = np.ones(n) / n
                else:
                    g = g / gs
            else:
                stall += 10
                if stall > max_stall:
                    break
                f = np.clip(f + rng.normal(0, 0.03, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                it += 1
                continue
        else:
            gs = g.sum()
            if gs < 1e-15:
                stall += 5
                it += 1
                continue
            g = g / gs
        
        # Line search
        dist = np.linalg.norm(g - f)
        if dist < 1e-12:
            stall += 3
            it += 1
            if stall > max_stall:
                break
            continue
        
        best_t = 0
        best_sc_t = best_sc
        for tt in [t_param, t_param*0.6, t_param*1.5, 0.1, 0.2, 0.4, 0.6, 0.8, 0.95]:
            tt = min(1.0, max(0.001, tt))
            fn = np.clip((1 - tt) * f + tt * g, 0, None)
            fn_s = fn.sum()
            if fn_s < 0.01:
                continue
            sc_t = _score(fn)
            if sc_t < best_sc_t - 1e-14:
                best_t = tt
                best_sc_t = sc_t
        
        if best_t > 0 and best_sc_t < best_sc - 1e-13:
            fn = np.clip((1 - best_t) * f + best_t * g, 0, None)
            f = fn / fn.sum()
            sc = _score(f)
            if sc < best_sc - 1e-14:
                best_f = f.copy()
                best_sc = sc
            t_param = min(0.9, best_t * 1.1)
            stall = 0
        else:
            t_param = max(0.02, t_param * 0.5)
            stall += 1
            if stall >= max_stall:
                # Perturb
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
                f = f / max(f.sum(), 1e-10)
                t_param = 0.25
                stall = 0
                max_stall += 100
        
        it += 1
    
    return best_f, best_sc

def _smooth_optimize(f0, n, stop_time, seed):
    """Optimize using log-sum-exp smooth approximation."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / max(f.sum(), 1e-10)
    
    best_f = f.copy()
    best_sc = _score(f)
    
    # Start with relatively large tau, then decrease
    tau = max(1e-4, _score(f) * 1e-4)
    
    while time.monotonic() < stop_time:
        # Use scipy minimize with L-BFGS-B
        def obj_grad(x):
            x = np.maximum(x, 0)
            ls, g = _logsumexp_conv(x, tau)
            return ls, g
        
        try:
            res = minimize(obj_grad, f, method='L-BFGS-B', bounds=[(0, None)]*n,
                          options={'maxiter': 50, 'gtol': 1e-8, 'maxls': 30})
            fn = np.clip(res.x, 0, None)
            fn_s = fn.sum()
            if fn_s > 0.01:
                fn = fn / fn_s
                sc = _score(fn)
                if sc < best_sc - 1e-12:
                    best_f = fn.copy()
                    best_sc = sc
                    f = fn
                    tau *= 0.7  # decrease tau for sharper approximation
        except Exception:
            pass
        
        if tau < 1e-8:
            break
        
        # Occasionally perturb
        if rng.random() < 0.1:
            f = np.clip(f + rng.normal(0, 0.02, n), 0, None)
            f = f / max(f.sum(), 1e-10)
    
    return best_f, best_sc
