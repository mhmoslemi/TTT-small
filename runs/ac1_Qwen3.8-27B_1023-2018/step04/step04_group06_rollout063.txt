    """max sum(g) s.t. A@g <= M, g >= 0."""
    c = -np.ones(n)
    b = np.full(A.shape[0], M)

            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > max_stuck:

            sigma = min(0.03 + 0.015 * (stuck // max_stuck), 0.35)
            xx = np.arange(n, dtype=np.float64)

            else:

def _optimize_gradient(a0, time_limit, seed=42):
    """Subgradient + subgradient polishing without LP."""
    a0 = np.asarray(a0, dtype=np.float64)
    n = a0.size
    a = np.clip(a0, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a /= s

    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0

    while time.monotonic() < time_limit:
        c_full = np.convolve(a, a)
        M = float(c_full.max())

        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 50:
            sigma = min(0.03 + 0.005 * (stuck // 50), 0.25)
            a *= np.exp(sigma * rng.normal(0, 1, n))
            a = np.maximum(a, 1e-14)
            a /= a.sum()
            stuck = 0
            if stuck > 40:
                break
            continue

        made_move = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
            cand = _subgrad_step(a, None, step)
            vc = float(np.convolve(cand, cand).max())
            if vc < M * (1.0 - 1e-11):
                a = cand
                made_move = True
                break

        if not made_move:
            a *= np.exp(rng.normal(0, 0.015, n))
            a = np.maximum(a, 1e-14)
            a /= a.sum()

    return best_a


def _make_bowl(n, alpha):
    """Bowl-shaped initialization: high at edges, low in center."""
    xx = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * xx / max(n - 1, 1))
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def _make_trapezoid(n, width_frac):
    """Trapezoidal initialization."""
    xx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 - width_frac * np.abs(xx - center) / max(center, 1)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def _make_dual_bowl(n, alpha):
    """Dual bowl: low at edges, high in center."""
    xx = np.arange(n, dtype=np.float64)
    a = 1.0 - alpha * np.cos(np.pi * xx / max(n - 1, 1))
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 20.0  # safety margin

    parent = _get_parent()
    n_parent = parent.size

    # Normalize parent
    p = np.clip(parent, 0, 1000).astype(np.float64)
    ps = p.sum()
    if ps < 1e-15:
        p = np.ones(201, dtype=np.float64)
    else:
        p = p / ps

    best_seq = p.copy()
    best_score = _eval_score(best_seq)

    def try_update(cand):
        nonlocal best_seq, best_score
        try:
            cand = np.clip(np.asarray(cand, dtype=np.float64), 0, 1000)
            sc = _eval_score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand.copy()
        except Exception:
            pass

    # ===== Phase 1: Optimize parent with LP (first 25% of budget) =====
    t1 = t0 + 0.25 * budget
    try:
        r = _optimize_lp(p, t1, seed=100)
        try_update(r)
    except Exception:
        pass

    # ===== Phase 2: Structured initializations (next 25%) =====
    t2 = t0 + 0.50 * budget
    n = n_parent

    inits = []
    # Bowl shapes
    for alpha in [0.3, 0.5, 0.7, 0.9, 1.2, 1.5, 2.0]:
        inits.append(_make_bowl(n, alpha))
    # Trapezoids
    for wf in [0.3, 0.5, 0.7, 0.9]:
        inits.append(_make_trapezoid(n, wf))
    # Dual bowls
    for alpha in [0.3, 0.5, 0.7]:
        inits.append(_make_dual_bowl(n, alpha))
    # Cosine harmonics
    xx = np.arange(n, dtype=np.float64)
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7]:
            v = 1.0 + amp * np.cos(2 * np.pi * freq * xx / n)
            v = np.maximum(v, 1e-10)
            inits.append(v / v.sum())
    # Blends with parent
    for frac in [0.3, 0.5, 0.7]:
        for init in inits[:4]:
            b = frac * p + (1 - frac) * init
            b = np.maximum(b, 1e-12)
            inits.append(b / b.sum())

    # Allocate time per initialization
    time_per_init = max(2.0, (t2 - time.monotonic()) / max(1, len(inits)))
    for i, init in enumerate(inits):
        if time.monotonic() > t2 - 5.0:
            break
        alloc = min(time.monotonic() + time_per_init, t2 - 3.0)
        if time.monotonic() >= alloc:
            continue
        try:
            r = _optimize_lp(init, alloc, seed=5000 + i)
            try_update(r)
        except Exception:
            pass

    # ===== Phase 3: Deep polish on best (next 25%) =====
    t3 = t0 + 0.75 * budget
    ab = np.clip(best_seq, 0, 1000).astype(np.float64)
    ab_s = ab.sum()
    if ab_s < 1e-15:
        ab = np.ones(n_parent)
    else:
        ab = ab / ab_s

    try:
        r = _optimize_lp(ab, t3, seed=9999, max_stuck=200)
        try_update(r)
    except Exception:
        pass

    # ===== Phase 4: Try different lengths (remaining time before deadline) =====
    if time.monotonic() < deadline - 60:
        # Try nearby lengths
        for dn in [-20, -10, 5, 10, 20]:
            if time.monotonic() > deadline - 30:
                break
            nt = n_parent + dn
            if nt < 10 or nt > 600:
                continue
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, best_seq)
            init_new = np.clip(init_new, 1e-10, 1000)
            init_new /= init_new.sum()
            tt = min(time.monotonic() + 25, deadline - 10)
            try:
                r = _optimize_lp(init_new, tt, seed=2000 + nt)
                try_update(r)
            except Exception:
                pass

    # ===== Phase 5: Final gradient polish =====
    if time.monotonic() < deadline - 8:
        af = np.clip(best_seq, 0, 1000).astype(np.float64)
        af_s = af.sum()
        if af_s < 1e-15:
            af = np.ones(int(len(af)))
        else:
            af = af / af_s
        try:
            r = _optimize_gradient(af, deadline - 5, seed=12345)
            try_update(r)
        except Exception:
            pass

    # ===== Final validation =====
    try:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if not np.isfinite(fs):
            raise ValueError("invalid")
    except Exception:
        # Fallback: simple bowl
        fb = _make_bowl(32, 0.5)
        return [float(x) for x in fb]

    return [float(x) for x in best_seq]
