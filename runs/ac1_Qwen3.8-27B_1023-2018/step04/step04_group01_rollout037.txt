import time
import math

    def _conv(a):
        """Full convolution via FFT."""
        n_a = len(a)
        size = next_fast_len(2 * n_a - 1)
        fa = rfft(a, size)
        c = irfft(fa * fa, size)[:2 * n_a - 1]
        return c
    
    def _score(a):
        """Compute the AC1 score."""
        s = float(a.sum())
        if s < 0.01:
            return math.inf
        b = _conv(a)
        m = float(b.max())
        return 2.0 * len(a) * m / (s * s)
    
    def _update_best(seq):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return

            sc = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    
    def _subgrad_direction(a, b):
        """
        Compute a subgradient of max_k (a*a)_k.
        Returns the subgradient vector (not yet combined with the score gradient).
        """
        M = float(b.max())
        # Find all argmax (within tolerance)
        tol = max(1e-10, 1e-8 * M)
        active = np.where(b >= M - tol)[0]
        K = len(active)
        if K == 0:
            return None
        
        n_a = len(a)
        # Subgradient of f(a) = max_k (a*a)_k
        # grad_i = 2 * sum_{k in active} lambda_k * a_{k-i} (if 0 <= k-i <= n_a-1)
        # Using uniform lambda = 1/K
        grad = np.zeros(n_a)
        for k in active:
            # a_{k-i} is valid when 0 <= k-i <= n_a-1, i.e., k-(n_a-1) <= i <= k
            i_lo = max(0, k - (n_a - 1))
            i_hi = min(n_a - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi+1] += a[:i_hi - i_lo + 1]
        grad *= 2.0 / K
        return grad
    
    def _lp_direction(f, n_f, time_limit=5.0):
        """
        LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
        Returns normalized direction g/sum(g) or None.
        """
        b = _conv(f)
        M = float(b.max())
        if M < 1e-15:
            return None
        
        # Build constraint matrix A (2n-1 x n)
        # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
        A = np.zeros((2 * n_f - 1, n_f))
        for j in range(n_f):
            A[j:j+n_f, j] = f
        
        c = -np.ones(n_f)
        b_ub = np.full(2 * n_f - 1, M)
        
        try:
            res = linprog(
                c, A_ub=A, b_ub=b_ub,
                bounds=[(0, None)] * n_f,
                method="highs",
                options={"presolve": True, "time_limit": time_limit, "dual_feasibility_tolerance": 1e-7}
            )
        except Exception:
            return None
        
        if res.status != 0 or res.x is None:
            return None
        
        g = np.asarray(res.x, dtype=np.float64)
        gs = float(g.sum())
        if gs < 1e-15:
            return None
        return g / gs
    
    def _project_nonneg(v):
        return np.maximum(v, 0)

        """
        Coordinate descent with exact 1D optimization.
        f is normalized (sum = 1).

                S_rest = S - v

                # plus w^2 at k=2i
                start = i
                end = min(i + n_c, 2 * n_c - 1)

                # Unaffected max

                else:

                else:

                        best_x = x

                    x_flo = max(0.0, best_x - 3*dw)
                    x_fhi = min(x_hi, best_x + 3*dw)
                    x_fine = np.linspace(x_flo, x_fhi, 30)
                    for x in x_fine:
                        w = x - v
                        S_new = S_rest + x
                        if S_new < 0.01:
                            continue

                        m_new = max(float(new_vals.max()), C)
                        obj = 2.0 * n_c * m_new / (S_new * S_new)
                        if obj < best_obj:
                            best_obj = obj
                            best_x = x
                
                if best_x != v and best_obj < best_sc - 1e-15:
                    w = best_x - v
                    f[i] = best_x
                    S += w
                    conv[start:end] += w * infl
                    if 2*i >= start and 2*i < end:
                        conv[2*i] += w*w
                    best_sc = best_obj
                    best_local = f.copy()
                    improved = True
            
            if not improved and sweep > 5:
                break
        
        return best_local, best_sc

    conv = _conv(f)

    while time.monotonic() < phase1_end - 5:

        g_dir = _lp_direction(f, n, time_limit=8.0)
        
        if g_dir is not None:

            best_t = 0.0
            best_sc = _score(f)
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:

                if f_try.sum() < 0.01:
                    continue
                f_try = _normalize(f_try)
                sc = _score(f_try)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t
            
            if best_t > 0:
                f = _project_nonneg(f + best_t * (g_dir - f))
                f = _normalize(f)
                conv = _conv(f)
        else:
            # Subgradient step
            pass
        
        # Subgradient steps between LP calls
        for sg_iter in range(20):
            if time.monotonic() > phase1_end - 5:
                break
            
            M = float(conv.max())
            S = float(f.sum())
            grad_f = _subgrad_direction(f, conv)
            if grad_f is None:
                break
            
            # Subgradient of score
            score_grad = (2.0 * n / (S * S)) * grad_f - (4.0 * n * M / (S ** 3)) * np.ones(n)

                conv = _conv(f)
