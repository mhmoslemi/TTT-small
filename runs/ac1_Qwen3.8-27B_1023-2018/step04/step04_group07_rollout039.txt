The constraint is: $(f * g)_k \leq M$ for $k$ in the active set.

$(f * g)_k = \sum_j f_j g_{k-j}$ where $g_i = 0$ for $i \notin [0, n-1]$.

After mixing: $a_{new} = (1-t)f + tg/\sum g$ (with $g$ normalized to sum 1).

import numpy as np

from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a, fsize):
    n = len(a)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_from_b(n, b, s):
    if s < 1e-10:
        return math.inf
    m = float(b.max())
    return 2.0 * n * m / (s*s)

def _build_A(f, n, k_list):
    m = len(k_list)
    A = np.zeros((m, n))
    for idx in range(m):
        k = int(k_list[idx])
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end+1)
            A[idx, i_range] = f[k - i_range]
    return A

def _solve_lp(f, n, M, k_list, time_limit=2.5):
    A = _build_A(f, n, k_list)
    c = -np.ones(n)
    b_ub = np.full(len(k_list), M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                     bounds=[(0, None)]*n,
                     method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None, None
    if res.status not in (0, 3) or res.x is None:
        return None, None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None, None
    g_norm = g / g_sum
    # Get dual variables if available
    dual = getattr(res, 'ineqlin', None)
    return g_norm, dual
