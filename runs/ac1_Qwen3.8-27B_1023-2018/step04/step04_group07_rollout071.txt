import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

_FSIZE = {}
_SEG = {}


def _fsize(n):
    if n not in _FSIZE:
        _FSIZE[n] = next_fast_len(2 * n - 1)
    return _FSIZE[n]


def _segments(n):
    if n not in _SEG:
        _SEG[n] = [(max(0, k - n + 1), min(n, k + 1)) for k in range(2 * n - 1)]
    return _SEG[n]


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s < 1e-12:
        return np.ones(a.size, dtype=np.float64) / a.size
    a *= 1.0 / s
    return a


def _conv_same(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    fs = _fsize(n)
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _conv_diff(a, b):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    fs = _fsize(n)
    c = irfft(rfft(a, fs) * rfft(b, fs), fs)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    c = _conv_same(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m


def _score_general(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s < 0.01:
        return math.inf
    c = _conv_same(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * a.size * m / (s * s)


def _lse_and_p(c, beta):
    c = np.asarray(c, dtype=np.float64)
    cmax = float(c.max())
    x = beta * (c - cmax)
    x = np.clip(x, -700.0, 0.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0.0 or not math.isfinite(s):
        idx = np.flatnonzero(c == cmax)
        p = np.zeros(c.size, dtype=np.float64)
        p[idx] = 1.0 / idx.size
        return cmax, p
    p = ex / s
    val = cmax + math.log(s) / beta if beta > 1e-12 else cmax
    return val, p


def _grad_from_p(a, p):
    n = a.size
    conv = np.convolve(a, p[::-1], "full")
    h = conv[2 * n - 2 : n - 2 : -1]
    return 2.0 * h


def _choose_indices(c, n, top_k):
    m = 2 * n - 1
    if top_k is None or top_k >= m:
        return np.arange(m, dtype=int)
    top_k = int(max(1, min(top_k, m)))
    order = np.argsort(-c).astype(int)
    keep = set(order[:top_k].tolist())
    for k in (0, n // 2, n - 1, m - 1):
        if 0 <= k < m:
            keep.add(int(k))
    arr = np.fromiter(keep, dtype=int, count=len(keep))
    return arr


def _lp_linearized(a, n, c=None, top_k=None, time_limit=3.0):
    if c is None:
        c = _conv_same(a)
    K = _choose_indices(c, n, top_k)
    seg = _segments(n)
    m = K.size
    Aub = np.zeros((m, n + 1), dtype=np.float64)

    for r, k in enumerate(K):

        if hi > lo:

        Aub[r, n] = -1.0

    Aeq[0, :n] = 1.0

            method="highs",
            options={"presolve": True, "time_limit": float(time_limit)},

        return None

    if res is None or getattr(res, "x", None) is None:

def _lp_cross(a, n, M, c=None, top_k=None, time_limit=2.0, target_factor=1.0):
    if c is None:
        c = _conv_same(a)
    K = _choose_indices(c, n, top_k)
    seg = _segments(n)
    A = np.zeros((K.size, n), dtype=np.float64)
    b_ub = np.full(K.size, float(M) * float(target_factor), dtype=np.float64)

    for r, k in enumerate(K):
        lo, hi = seg[k]
        if hi > lo:
            A[r, lo:hi] = a[k - hi + 1:k - lo + 1][::-1]

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": float(time_limit)},
        )
    except Exception:
        return None

    if res is None or getattr(res, "x", None) is None:
        return None
    if res.status not in (0, 1, 3):
        return None

    b = np.asarray(res.x, dtype=np.float64)

        return None
    b = np.clip(b, 0.0, 1000.0)
    s = float(b.sum())

    return b / s


def _line_search(f, g, M, stop, cff=None):
    cfg = _conv_diff(f, g)
    cgg = _conv_same(g)
    if cff is None:
        cff = _conv_same(f)

    def value(t):
        ut = 1.0 - t
        c = cff * (ut * ut) + cfg * (2.0 * t * ut) + cgg * (t * t)
        m = float(c.max())
        if not math.isfinite(m):
            return math.inf
        return m if m > 0.0 else 0.0

    best_t = 0.0
    best_val = float(M)

    for t in (0.0, 0.005, 0.01, 0.02, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.7, 1.0):
        if time.monotonic() > stop:
            break
        v = value(t)
        if v < best_val - 1e-15:
            best_val = v
            best_t = t

        lo = max(0.0, best_t - 0.18)
        hi = min(1.0, best_t + 0.18)
        if hi - lo > 1e-7:

            m1 = hi - (hi - lo) / gr
            m2 = lo + (hi - lo) / gr
            v1 = value(m1)
            v2 = value(m2)
            if v1 < v2:
                hi = m2
            else:
                lo = m1
            for _ in range(18):
                if time.monotonic() > stop:
                    break
                m1 = hi - (hi - lo) / gr
                m2 = lo + (hi - lo) / gr
                v1 = value(m1)
                v2 = value(m2)
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
                tm = 0.5 * (lo + hi)
                vm = value(tm)
                if vm < best_val - 1e-15:
                    best_val = vm
                    best_t = tm
    return best_t, best_val


def _perturb(a, rng, best=None, strength=0.1):
    n = a.size
    if best is None:
        best = a
    choice = int(rng.integers(8))
    mean = max(float(a.mean()), 1e-15)

    if choice == 0:
        out = a + rng.normal(0.0, strength * mean, n)
    elif choice == 1:
        out = a[::-1].copy()
    elif choice == 2:
        out = 0.5 * (a + a[::-1])
        out = out + rng.normal(0.0, 0.05 * mean, n)
    elif choice == 3:
        out = 0.65 * a + 0.35 * np.abs(rng.normal(1.0, 0.5, n)) * mean
    elif choice == 4:
        blk = max(1, n // 6)
        out = a.copy()
        out[:blk], out[n - blk:] = a[n - blk:].copy(), a[:blk].copy()
    elif choice == 5:
        k = min(n, max(5, n // 10))
        idx = rng.choice(n, size=k, replace=False)
        out = a.copy()
        local = max(float(a[idx].mean()), mean)
        out[idx] = rng.uniform(0.0, 2.5 * local, size=k)
    elif choice == 6:
        kern = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
        kern /= kern.sum()
        out = np.convolve(a, kern, "same")
        out = out + rng.normal(0.0, strength * mean, n)
    else:
        out = 0.7 * best + 0.3 * np.abs(rng.normal(1.0, 0.35, n)) * max(float(best.mean()), mean)

    if rng.random() < 0.35:
        out = 0.5 * out + 0.5 * best

    out = np.nan_to_num(out, nan=0.0, posinf=1000.0, neginf=0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s < 1e-10:
        return best.copy()
    return out / s


def _smooth_update(a, beta, target=0.02):
    c = _conv_same(a)
    val, p = _lse_and_p(c, beta)
    g = _grad_from_p(a, p)
    gt = g - g.mean()
    nrm = float(np.linalg.norm(gt, ord=1))
    if nrm < 1e-15:
        return a, c, float(c.max())

    step = target / nrm
    best_a = a
    best_c = c
    best_m = float(c.max())
    best_lse = val

    for frac in (1.0, 0.5, 0.25):
        an = a - frac * step * gt
        an = np.clip(an, 0.0, None)
        an = np.nan_to_num(an, nan=0.0, posinf=1000.0, neginf=0.0)
        s = float(an.sum())
        if s < 1e-12:
            continue
        an /= s
        cn = _conv_same(an)
        mn = float(cn.max())
        if mn < best_m - 1e-15:
            best_a, best_c, best_m, best_lse = an, cn, mn, val
            break
        if mn < best_m + 1e-12:
            lsen, _ = _lse_and_p(cn, beta)
            if lsen < best_lse - 1e-12:
                best_a, best_c, best_m, best_lse = an, cn, mn, lsen
                break
    return best_a, best_c, best_m


def _smooth_phase(a_in, stop, seed, beta_start=20.0, beta_end=6000.0):
    n = a_in.size
    a = _normalize(a_in)
    c = _conv_same(a)
    best = a.copy()
    best_c = c.copy()
    best_M = float(c.max())
    rng = np.random.default_rng(seed)
    beta = beta_start
    stall = 0
    target = 0.05
    it = 0

    while time.monotonic() < stop - 2.0:
        c = _conv_same(a)
        M = float(c.max())
        if M < best_M - 1e-15:
            best_M = M
            best = a.copy()
            best_c = c.copy()
            stall = 0
        else:
            stall += 1

        if M > best_M * 1.25 and stall > 60:
            a = best.copy()
            c = best_c.copy()
            M = best_M
            stall = 0
            continue

        if stall > 3500:
            a = _perturb(a, rng, best, strength=0.15)
            c = _conv_same(a)
            M = float(c.max())
            if M < best_M:
                best_M = M
                best = a.copy()
                best_c = c.copy()
            stall = 0
            continue

        if it % 35 == 0:
            beta = min(beta_end, beta * 1.28)
        target = max(0.006, 0.055 * math.exp(-beta / 4500.0))

        val, p = _lse_and_p(c, beta)
        g = _grad_from_p(a, p)
        gt = g - g.mean()
        nrm = float(np.linalg.norm(gt, ord=1))
        if nrm < 1e-15:
            break

        step = target / nrm
        accepted = False
        for frac in (1.0, 0.5, 0.25, 0.125):
            if time.monotonic() > stop - 2.0:
                break
            an = a - frac * step * gt
            an = np.clip(an, 0.0, None)
            an = np.nan_to_num(an, nan=0.0, posinf=1000.0, neginf=0.0)
            s = float(an.sum())
            if s < 1e-12:
                continue
            an /= s
            cn = _conv_same(an)
            Mn = float(cn.max())
            if Mn < M - 1e-15:
                a, c, M, accepted = an, cn, Mn, True
                target = min(0.10, target * 1.06)
                break
            if Mn < M + 1e-12:
                lsen, _ = _lse_and_p(cn, beta)
                if lsen < val - 1e-12:
                    a, c, M, accepted = an, cn, Mn, True
                    target = min(0.08, target * 1.03)
                    break

        if not accepted:
            target = max(0.002, target * 0.55)
            stall += 2
            if stall % 120 == 0:
                a = _perturb(a, rng, best, strength=0.08)
                c = _conv_same(a)
                M = float(c.max())
                if M < best_M:
                    best_M = M
                    best = a.copy()
                    best_c = c.copy()
        it += 1

    return best, 2.0 * n * best_M


def _lp_phase(a_in, stop, seed):
    n = a_in.size
    a = _normalize(a_in)
    c = _conv_same(a)
    M = float(c.max())
    best = a.copy()
    best_c = c.copy()
    best_M = M
    rng = np.random.default_rng(seed)
    stall = 0
    it = 0

    while time.monotonic() < stop - 2.0:
        if M < best_M - 1e-15:
            best_M = M
            best = a.copy()
            best_c = c.copy()
            stall = 0
        else:
            stall += 1

        if M > best_M * 1.35 and stall > 25:
            a = best.copy()
            c = best_c.copy()
            M = best_M
            stall = 0
            continue

        if stall > 90:
            a = _perturb(a, rng, best, strength=0.22)
            c = _conv_same(a)
            M = float(c.max())
            if M < best_M:
                best_M = M
                best = a.copy()
                best_c = c.copy()
            stall = 0
            continue

        g = None
        if it % 18 == 0:
            g = _lp_linearized(a, n, c, top_k=None, time_limit=4.5)
        elif it % 4 == 3:
            if stall < 12:
                g = _lp_cross(a, n, M, c, top_k=90, time_limit=1.8)
            elif stall < 30:
                g = _lp_cross(a, n, M, c, top_k=160, time_limit=2.5, target_factor=0.998)
            else:
                g = _lp_cross(a, n, M, c, top_k=260, time_limit=3.5, target_factor=0.995)
        else:
            if stall < 12:
                g = _lp_linearized(a, n, c, top_k=80, time_limit=2.0)
            elif stall < 30:
                g = _lp_linearized(a, n, c, top_k=160, time_limit=3.0)
            else:
                g = _lp_linearized(a, n, c, top_k=300, time_limit=4.0)

        if g is None and stall >= 15:
            g = _lp_cross(a, n, M, c, top_k=220, time_limit=3.0, target_factor=0.996)

        if g is None:
            a, c, M = _smooth_update(a, beta=1200.0, target=0.012)
            if M < best_M - 1e-15:
                best_M = M
                best = a.copy()
                best_c = c.copy()
                stall = 0
            else:
                stall += 2
            it += 1
            continue

        g = np.asarray(g, dtype=np.float64)
        s = float(g.sum())
        if not np.all(np.isfinite(g)) or s < 1e-12:
            stall += 1
            it += 1
            continue
        g /= s

        if float(np.linalg.norm(g - a)) < 1e-10:
            stall += 2
            it += 1
            continue

        candidates = [g]
        gs = 0.5 * (g + g[::-1])
        s = float(gs.sum())
        if s > 1e-12 and float(np.linalg.norm(gs / s - g)) > 1e-5:
            candidates.append(gs / s)
        gr = g[::-1].copy()
        s = float(gr.sum())
        if s > 1e-12 and float(np.linalg.norm(gr / s - g)) > 1e-5:
            candidates.append(gr)

        improved = False
        for cand in candidates:
            t, val = _line_search(a, cand, M, stop - 2.0, cff=c)
            if t > 0.0 and val < M - 1e-14:
                a = (1.0 - t) * a + t * cand
                a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
                a = np.clip(a, 0.0, 1000.0)
                s = float(a.sum())
                if s <= 0.0:
                    a = best.copy()
                    c = best_c.copy()
                    M = best_M
                    break
                a /= s
                c = _conv_same(a)
                M = float(c.max())
                if M < best_M - 1e-15:
                    best_M = M
                    best = a.copy()
                    best_c = c.copy()
                    stall = 0
                improved = True
                break

        if not improved:
            a, c, M = _smooth_update(a, beta=1500.0, target=0.008)
            if M < best_M - 1e-15:
                best_M = M
                best = a.copy()
                best_c = c.copy()
                stall = 0
            else:
                stall += 3
        it += 1

    return best, 2.0 * n * best_M


def _coord_polish(a_in, stop, seed=31337):
    n = a_in.size
    a = _normalize(a_in)
    c = _conv_same(a)
    S = 1.0
    best = a.copy()
    best_score = float(_score_normalized(best))
    rng = np.random.default_rng(seed)
    no_round = 0
    rnd = 0

    while time.monotonic() < stop - 2.0 and no_round < 8:
        rnd += 1
        improved = False
        order = rng.permutation(n)
        mean = S / n

        for i in order:
            if time.monotonic() > stop - 2.0:
                break
            v = float(a[i])
            idx = np.arange(i, i + n)
            shift = a[idx - i]
            base = c[idx]
            old = 2.0 * v * shift
            old[i] = v * v

            cands = np.array([0.0], dtype=np.float64)
            if v > 0.0:
                mults = np.array([0.2, 0.5, 0.75, 1.25, 1.6, 2.2, 3.5], dtype=np.float64)
                cands = np.concatenate((cands, v * mults))
            factors = np.array([0.02, 0.08, 0.2, 0.5], dtype=np.float64)
            cands = np.concatenate((cands, mean * factors))
            cands = np.concatenate((cands, np.array([1e-5, 1e-4, 5e-4, 2e-3, 5e-3, 1e-2, 2e-2], dtype=np.float64)))
            cands = cands[(cands >= 0.0) & (cands <= 1000.0)]
            if cands.size:
                cands.sort()
                filt = [float(cands[0])]
                for x in cands[1:]:
                    x = float(x)
                    if x - filt[-1] > 1e-11 * max(1.0, abs(x)):
                        filt.append(x)
                cands = np.array(filt, dtype=np.float64)
                cands = cands[np.abs(cands - v) > 1e-12 * max(1.0, abs(v))]

            best_x = v
            best_c = c
            best_sc = float(2.0 * n * c.max() / (S * S))

            def eval_x(x, cur_best_sc, cur_best_c, cur_best_x):
                S_try = S + x - v
                if S_try <= 1e-12:
                    return cur_best_sc, cur_best_c, cur_best_x
                new = 2.0 * x * shift
                new[i] = x * x
                c_try = c.copy()
                c_try[idx] = base - old + new
                m = float(c_try.max())
                if not math.isfinite(m):
                    return cur_best_sc, cur_best_c, cur_best_x
                if m < 0.0:
                    m = 0.0
                sc = float(2.0 * n * m / (S_try * S_try))
                if sc < cur_best_sc - 1e-15:
                    return sc, c_try, x
                return cur_best_sc, cur_best_c, cur_best_x

            for x in cands:
                if time.monotonic() > stop - 2.0:
                    break
                best_sc, best_c, best_x = eval_x(float(x), best_sc, best_c, best_x)

            extra = []
            if best_x > 0.0:
                extra = [best_x * 0.72, best_x * 0.85, best_x * 1.15, best_x * 1.32]
            else:
                extra = [1e-5, 1e-4, 5e-4, 2e-3]
            for x in extra:
                if time.monotonic() > stop - 2.0:
                    break
                if x < 0.0 or x > 1000.0:
                    continue
                if abs(x - v) <= 1e-12 * max(1.0, abs(v)):
                    continue
                best_sc, best_c, best_x = eval_x(float(x), best_sc, best_c, best_x)

            if best_x != v:
                a[i] = best_x
                c = best_c
                S = S + best_x - v
                if S <= 0.0 or abs(S - 1.0) > 1e-3:
                    if S <= 0.0:
                        break
                    scale = 1.0 / S
                    a *= scale
                    c *= scale * scale
                    S = 1.0
                cur_sc = float(2.0 * n * c.max() / (S * S))
                if cur_sc < best_score - 1e-15:
                    best_score = cur_sc
                    best = a.copy()
                improved = True

        if rnd % 10 == 0:
            a = _normalize(a)
            c = _conv_same(a)
            S = 1.0
            cur_sc = float(_score_normalized(a))
            if cur_sc < best_score - 1e-15:
                best_score = cur_sc
                best = a.copy()

        if improved:
            no_round = 0
        else:
            no_round += 1

    a = _normalize(a)
    sc = float(_score_normalized(a))
    if sc < best_score:
        best_score = sc
        best = a
    return best, best_score


def _block_polish(a_in, stop, seed=90001):
    n = a_in.size
    a = _normalize(a_in)
    best = a.copy()
    best_score = float(_score_normalized(best))
    rng = np.random.default_rng(seed)
    no_imp = 0
    sizes = (3, 5, 8)

    while time.monotonic() < stop - 2.0 and no_imp < 6:
        improved = False
        for bs in sizes:
            if bs > n:
                continue
            step = max(1, bs // 2)
            for start in range(0, n - bs + 1, step):
                if time.monotonic() > stop - 2.0:
                    break
                block = a[start:start + bs].copy()
                total = float(block.sum())
                if total <= 1e-14:
                    continue
                current = float(_score_normalized(a))
                best_block = block
                best_local = current
                for _ in range(3):
                    w = rng.dirichlet(np.ones(bs) * 1.8)
                    a[start:start + bs] = w * total
                    sc = float(_score_normalized(a))
                    if sc < best_local - 1e-14:
                        best_local = sc
                        best_block = a[start:start + bs].copy()
                if best_local < current - 1e-14:
                    a[start:start + bs] = best_block
                    improved = True
                    if best_local < best_score - 1e-14:
                        best_score = best_local
                        best = a.copy()
                else:
                    a[start:start + bs] = block

        if improved:
            no_imp = 0
        else:
            no_imp += 1

    a = _normalize(a)
    sc = float(_score_normalized(a))
    if sc < best_score:
        best_score = sc
        best = a
    return best, best_score
