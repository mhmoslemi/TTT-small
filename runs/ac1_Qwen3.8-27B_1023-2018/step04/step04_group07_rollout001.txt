import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _build_A(f, n):
    """A has shape (2n-1, n). A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_step(f, n, M, top_k=None, time_limit=10.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    A = _build_A(f, n)
    b_ub = np.full(2*n-1, M)
    if top_k is not None and top_k < 2*n-1:
        # Use only top-k constraints (those with largest f*f values)
        ff = _conv(f)
        idx = np.argsort(-ff)[:top_k]
        A_sub = A[idx]
        b_sub = np.full(len(idx), M)
    else:
        A_sub = A
        b_sub = b_ub
    
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_sub, b_ub=b_sub,
                     bounds=[(0, None)]*n, method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = res.x.copy()
    s = g.sum()
    if s < 1e-12:
        return None
    return g / s

def _line_search(f, g, n, M_curr, stop_t):
    """Find best t for (1-t)f + t*g."""
    best_t = 0.0
    best_m = M_curr
    # Try discrete values
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_step(f, n, M, top_k=None, time_limit=8.0):
    A = _build_A(f, n)
    c_obj = -np.ones(n)
    if top_k is not None and top_k < 2 * n - 1:
        ff = _conv(f)
        idx = np.argsort(-ff)[:top_k]
        A_sub = A[idx]
        b_sub = np.full(len(idx), M)
    else:
        A_sub = A
        b_sub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A_sub, b_ub=b_sub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-7})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64).copy()
    s = g.sum()
    if s < 1e-12:
        return None
    return g / s


def _line_search(f, g, n, M_curr, stop_t):
    best_t = 0.0
    best_m = M_curr
    for t in [0.1, 0.2, 0.35, 0.5, 0.7, 0.85, 1.0]:
        if time.monotonic() > stop_t:
            break
        f_new = (1.0 - t) * f + t * g
        m = float(_conv(f_new).max())
        if m < best_m:
            best_m = m
            best_t = t
    if 0.0 < best_t < 1.0:
        lo = max(0.0, best_t - 0.12)
        hi = min(1.0, best_t + 0.12)
        gr = (1.0 + math.sqrt(5.0)) / 2.0
        for _ in range(12):
            if time.monotonic() > stop_t:
                break
            m1 = hi - (hi - lo) / gr
            m2 = lo + (hi - lo) / gr
            v1 = float(_conv((1.0 - m1) * f + m1 * g).max())
            v2 = float(_conv((1.0 - m2) * f + m2 * g).max())
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        t_mid = (lo + hi) / 2.0
        v_mid = float(_conv((1.0 - t_mid) * f + t_mid * g).max())
        if v_mid < best_m:
            best_m = v_mid
            best_t = t_mid
    return best_t, best_m


def _lp_optimize(f0, n, stop_time, seed=42):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n], 0.0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n].copy()
    fs = f.sum()
    f = f / fs if fs > 1e-10 else np.ones(n) / n

    best_f = f.copy()
    best_M = float(_conv(f).max())
    M = best_M
    stall = 0
    iter_count = 0

    while time.monotonic() < stop_time - 3.0:
        if stall > 60:
            strat = stall % 6
            if strat == 0:
                f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
            elif strat == 1:
                blk = max(1, n // 4)
                f2 = f.copy()
                f2[:blk], f2[n - blk:] = f[n - blk:].copy(), f[:blk].copy()
                f = f2
            elif strat == 2:
                kern = np.exp(-0.5 * (np.arange(-8, 9) / 3.0) ** 2)
                kern /= kern.sum()
                noise = np.convolve(rng.normal(0, 0.15, n), kern, 'same')
                f = np.clip(f * (1.0 + noise), 0, None)
            elif strat == 3:
                f = 0.6 * best_f + 0.4 * np.abs(rng.normal(0, 1, n))
            elif strat == 4:
                f = np.clip(f + 0.1 * rng.standard_normal(n), 0, None)
            else:
                # Targeted: boost low elements, reduce high
                med = np.median(f)
                f = np.clip(f + 0.05 * (med - f), 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            M = float(_conv(f).max())
            stall = 0
            continue

        # Adaptive M: slightly above current to allow flexibility
        M_lp = M * 1.005

        # Choose top_k: use all constraints early, subset later
        if iter_count < 20:
            top_k = None
        elif iter_count < 80:
            top_k = max(n, (2 * n - 1) * 3 // 4)
        else:
            top_k = max(n, (2 * n - 1) * 2 // 3)

        g_dir = _lp_step(f, n, M_lp, top_k=top_k, time_limit=7.0)

        if g_dir is None:
            stall += 4
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            fs = f.sum()
            f = f / fs if fs > 1e-10 else best_f.copy()
            M = float(_conv(f).max())
            continue

        dist = float(np.linalg.norm(g_dir - f))
        if dist < 1e-8:
            stall += 2
            continue

        t_opt, m_opt = _line_search(f, g_dir, n, M, stop_time)
        if t_opt > 0.01 and m_opt < M - 1e-14:
            f = (1.0 - t_opt) * f + t_opt * g_dir
            f = np.clip(f, 0, None)
            fs = f.sum()
            f = f / fs if fs > 1e-10 else best_f.copy()
            M = float(_conv(f).max())
            if M < best_M - 1e-14:
                best_M = M
                best_f = f.copy()
            stall = 0
        else:
            stall += 1

        iter_count += 1

    return best_f, _score_fast(best_f)


def _polish(a, stop_time, seed=99):
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel().copy(), 0, None)
    fs = best.sum()
    best = best / fs if fs > 1e-10 else np.ones(n) / n
    best_sc = _score_fast(best)
    rng = np.random.default_rng(seed)
    gr = (1.0 + math.sqrt(5.0)) / 2.0
    no_imp = 0

    while time.monotonic() < stop_time - 2.0:
        improved = False

        # 1D coordinate descent with golden section
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            orig = best[i]
            hi = max(3.0 * orig + 0.02, 0.05)
            lo = 0.0
            # Quick scan
            for mlt in (0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5):
                nv = orig * mlt if orig > 1e-10 else mlt * 0.01
                if nv < 0:
                    nv = 0.0
                best[i] = nv
                s = best.sum()
                if s < 0.01:
                    best[i] = orig
                    continue
                sc = 2.0 * n * float(_conv(best).max()) / (s * s)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    orig = nv
                    improved = True
            # Golden section refinement around current best
            if orig > 0:
                hi2 = orig * 1.3 + 0.01
                lo2 = orig * 0.7
                for _ in range(6):
                    if time.monotonic() > stop_time - 2.0:
                        break
                    m1 = hi2 - (hi2 - lo2) / gr
                    m2 = lo2 + (hi2 - lo2) / gr
                    best[i] = m1
                    s1 = best.sum()
                    v1 = 2.0 * n * float(_conv(best).max()) / (s1 * s1) if s1 > 0.01 else math.inf
                    best[i] = m2
                    s2 = best.sum()
                    v2 = 2.0 * n * float(_conv(best).max()) / (s2 * s2) if s2 > 0.01 else math.inf
                    if v1 < v2:
                        hi2 = m2
                    else:
                        lo2 = m1
                best[i] = (lo2 + hi2) / 2
                sc = _score_fast(best)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
                else:
                    # Keep the best from golden section
                    pass
            # If no improvement found, restore
            if not improved:
                best[i] = orig

        # Pair moves
        n_pairs = min(200, n * 2)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 2.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = best[i], best[j]
            total = oi + oj
            if total < 1e-12:
                continue
            for frac in (0.0, 0.15, 0.3, 0.5, 0.7, 0.85, 1.0):
                best[i] = frac * total
                best[j] = (1.0 - frac) * total
                s = best.sum()
                if s < 0.01:
                    break
                sc = 2.0 * n * float(_conv(best).max()) / (s * s)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
            best[i], best[j] = oi, oj

        # Block moves (3-5 consecutive)
        n_blocks = min(60, n // 2)
        for _ in range(n_blocks):
            if time.monotonic() > stop_time - 2.0:
                break
            bs = int(rng.integers(3, 6))
            start = int(rng.integers(max(1, n - bs + 1)))
            block = best[start:start + bs].copy()
            total = block.sum()
            if total < 1e-12:
                continue
            for _ in range(4):
                w = rng.dirichlet(np.ones(bs) * 1.5)
                best[start:start + bs] = w * total
                s = best.sum()
                if s < 0.01:
                    continue
                sc = 2.0 * n * float(_conv(best).max()) / (s * s)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
            best[start:start + bs] = block

        if not improved:
            no_imp += 1
            if no_imp >= 6:
                break
        else:
            no_imp = 0

    return best, best_sc
