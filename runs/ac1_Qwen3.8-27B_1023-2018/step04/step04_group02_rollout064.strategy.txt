    Ensures the sequence has a sum >= target, clipping values to bounds [0,1000].
    Includes repair for NaN/Inf and zero-sum cases.
    """
    seq = np.array(seq, dtype=float)
    if len(seq)<1e5:
        # Run repair: ensure sum
        if np.isnan(seq).any() or np.isinf(seq).any():
            seq = np.clip(seq, 0, 1000)
            return list(seq) + [1000]*(400-len(seq))
        if np.min(seq) < 0 or np.max(seq) > 1000:
            seq = np.clip(seq, 0, 1000)
        total = np.sum(seq)
        if total < 0.01:
            # Worth scaling?
            if total == 0:
                # Only then, at least one value >0
                seq += 1000/len(seq)  # rudimentary enforcement
            else:
                seq = seq / total * (0.01) * 1000  # Linear scale
                # Reclip to avoid spikes
                seq = np.clip(seq, 0, 1000)
    return list(seq[:400])
