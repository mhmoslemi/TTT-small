def _score(a):

def _conv_max(a):
    c = np.convolve(a, a)

        return float("inf"), -1
    return float(c.max()), int(c.argmax())


def _build_A(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size

def _line_search(a, g, m0, tol=1e-14, grid=21, refine=25):
    a = np.asarray(a, dtype=np.float64).ravel()
    g = np.asarray(g, dtype=np.float64).ravel()
    if a.size != g.size:
        return None
    d = g - a

    def val(t):
        if t <= 0.0:
            return m0, a
        if t >= 1.0:
            c = g
        else:
            c = a + t * d
        c = np.where(np.isfinite(c), c, 0.0)
        if c.min() < 0.0:
            c = np.maximum(c, 0.0)
        s = float(c.sum())
        if not np.isfinite(s) or s <= 1e-300:
            return float("inf"), None
        if abs(s - 1.0) > 1e-9:
            c = c / s
        m, _ = _conv_max(c)
        return m, c

    best_m = m0
    best_c = None
    best_t = 0.0

    for t in np.linspace(0.0, 1.0, grid):
        m, c = val(float(t))
        if c is not None and np.isfinite(m) and m < best_m - tol:
            best_m = m
            best_c = c
            best_t = float(t)

    if best_c is None:
        return None

    for span in (0.30, 0.12, 0.045):
        lo = max(0.0, best_t - span)
        hi = min(1.0, best_t + span)
        if hi - lo < 2e-6:

        for t in np.linspace(lo, hi, refine + 1):
            m, c = val(float(t))
            if c is not None and np.isfinite(m) and m < best_m - tol:
                best_m = m
                best_c = c
                best_t = float(t)

        for alpha in alphas:
            cand = _norm(a - float(alpha) * u)
            m, _ = _conv_max(cand)
            if m < base - 1e-14:
                return cand
    return None


def _perturb(a, rng, sigma, mode=0):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return _norm(np.ones(32))
    mode = int(mode) % 5
    if mode == 0:
        b = a * np.exp(rng.normal(0.0, float(sigma), n))
    elif mode == 1:
        xx = np.arange(n, dtype=np.float64) / float(n)
        f = rng.uniform(0.8, 6.0)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        b = a * (1.0 + float(sigma) * np.sin(2.0 * np.pi * f * xx + ph))
    elif mode == 2:
        b = a + float(sigma) * float(a.mean()) * rng.normal(0.0, 1.0, n)
    elif mode == 3:
        b = np.power(np.maximum(a, 1e-300), float(rng.uniform(0.65, 1.35)))
    else:
        b = _rev_blend_dir(a, float(rng.uniform(0.15, 0.50)))
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return a.copy()
    return b / s


def _optimize(a0, t_end, seed=1, max_stuck=70, use_lp2=True, use_lp1=True):
    a = _norm(a0)
    n = a.size
    c0 = np.convolve(a, a)
    best_m = float(c0.max())
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    stuck = 0
    pert = 0

    def replace(cand):
        nonlocal a, best_a, best_m
        cand = _norm(cand)
        m, _ = _conv_max(cand)
        a = cand
        if m < best_m:
            best_m = m
            best_a = a.copy()

    while time.monotonic() < t_end:
        if time.monotonic() > t_end - 0.25:
            break
        c = np.convolve(a, a)
        m = float(c.max())
        if m < best_m - 1e-14:
            best_m = m
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck >= max_stuck:
            sigma = min(0.02 + 0.012 * pert, 0.35)
            mode = pert % 5
            if mode == 4:
                a = _norm(0.5 * best_a + 0.5 * a)
            else:
                a = _perturb(a, rng, sigma, mode)
            c = np.convolve(a, a)
            m = float(c.max())
            if m < best_m:
                best_m = m
                best_a = a.copy()
            stuck = 0
            pert += 1
            if pert > 45:
                break
            continue

        made = False

        if use_lp2 and time.monotonic() < t_end - 1.5:
            g, _ = _lp_min_peak(a, tl=1.3)
            if g is not None:
                cand = _try_direction(a, g, m)
                if cand is not None:
                    replace(cand)
                    c = np.convolve(a, a)
                    m = float(c.max())
                    made = True
                else:
                    for beta in (0.25, 0.45):
                        cand = _try_direction(a, _rev_blend_dir(g, beta), m)
                        if cand is not None:
                            replace(cand)
                            c = np.convolve(a, c.size if False else a * 0 + c, c) if False else np.convolve(a, a)
                            m = float(c.max())
                            made = True
                            break
                    if not made:
                        for tf in (_smooth_dir, lambda x: _power_dir(x, 0.85), lambda x: _power_dir(x, 1.15)):
                            cand = _try_direction(a, tf(g), m)
                            if cand is not None:
                                replace(cand)
                                c = np.convolve(a, a)
                                m = float(c.max())
                                made = True
                                break

        if (not made) and use_lp1 and stuck >= 3 and time.monotonic() < t_end - 1.5:
            for fac in (1.0 + 1e-9, 1.0 + 1e-7, 1.0 - 1e-9):
                if time.monotonic() > t_end - 1.0:
                    break
                Mtry = m * fac
                if Mtry <= 1e-300:
                    continue
                g = _lp_maxvol(a, Mtry, tl=1.1)
                if g is None:
                    continue
                cand = _try_direction(a, g, m)
                if cand is not None:
                    replace(cand)
                    c = np.convolve(a, a)
                    m = float(c.max())
                    made = True
                    break

        if (not made) and stuck >= 5:
            for tf in (_smooth_dir, lambda x: _power_dir(x, 0.82), lambda x: _power_dir(x, 1.18), lambda x: _rev_blend_dir(x, 0.22)):
                cand = _try_direction(a, tf(a), m)
                if cand is not None:
                    replace(cand)
                    c = np.convolve(a, a)
                    m = float(c.max())
                    made = True
                    break

        if not made:
            cand = _flattening_step(a, c)
            if cand is not None:
                replace(cand)
                c = np.convolve(a, a)
                m = float(c.max())
                made = True

        if not made:
            cand = _active_step(a, c)
            if cand is not None:
                replace(cand)
                c = np.convolve(a, a)
                m = float(c.max())
                made = True

        if not made:
            cand = _subgrad_step(a, c)
            if cand is not None:
                replace(cand)
                c = np.convolve(a, a)
                m = float(c.max())
                made = True

        if not made:
            a = _perturb(a, rng, 0.018, 0)

    return best_a


def _resample(seq, new_n):
    seq = np.asarray(seq, dtype=np.float64).ravel()
    old_n = seq.size
    if old_n == int(new_n):
        return _norm(seq)
    if old_n == 0:
        return _norm(np.ones(int(new_n)))
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / float(old_n)
    x_new = (np.arange(int(new_n), dtype=np.float64) + 0.5) / float(int(new_n))
    return _norm(np.interp(x_new, x_old, seq))


def _fourier_seq(p, x, K=4):
    p = np.asarray(p, dtype=np.float64).ravel()
    need = 2 + 2 * K
    if p.size < need:
        p = np.pad(p, (0, need - p.size), mode="constant")
    v = p[0] * (x - 0.5) ** 2 + p[1] * (x - 0.5)
    idx = 2
    for m in range(1, K + 1):
        v = v + p[idx] * np.sin(np.pi * float(m) * x) + p[idx + 1] * np.cos(np.pi * float(m) * x)
        idx += 2
    v = np.where(np.isfinite(v), v, 0.0)
    vmax = float(v.max())
    if not np.isfinite(vmax):
        return None
    v = v - vmax
    v = np.clip(v, -30.0, 0.0)
    f = np.exp(v)
    s = float(f.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return f / s


def _lowdim_candidates(pool, n, t_end, seed):
    if minimize is None:
        return
    n = int(n)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    K = 4
    dim = 2 + 2 * K
    starts = [
        np.array([-1.2, 1.0, 0.35, 0.0, 0.0, 0.0, 0.7, 0.15, 0.25, 0.0], dtype=np.float64),
        np.array([1.2, -1.0, 0.0, 0.5, -0.3, 0.2, 0.4, -0.2, 0.1, 0.3], dtype=np.float64),
        np.array([-0.8, 0.4, 0.0, -0.6, 0.5, 0.0, 0.2, 0.4, -0.2, 0.1], dtype=np.float64),
        np.array([0.6, -0.3, -0.5, 0.4, 0.0, 0.3, 0.3, -0.4, 0.0, -0.2], dtype=np.float64),
    ]
    rng = np.random.default_rng(seed + 101)
    starts.extend(rng.normal(0.0, 0.8, size=(2, dim)).tolist())

    for s in range(len(starts)):
        if time.monotonic() > t_end - 4.0:
            break
        x0 = np.asarray(starts[s], dtype=np.float64).reshape(dim)
        local = {"score": float("inf"), "seq": None}

        def obj(pp, _x=x, _K=K, _local=local):
            seq = _fourier_seq(pp, _x, _K)
            if seq is None:
                return 1e9
            sc = _score(seq)
            return sc if np.isfinite(sc) else 1e9

        def cb(xx, *args, **kwargs):
            seq = _fourier_seq(xx, x, K)
            if seq is None:
                return
            sc = _score(seq)
            if np.isfinite(sc) and sc < local["score"]:
                local["score"] = sc
                local["seq"] = seq.copy()

        try:
            minimize(
                obj,
                x0,
                method="Powell",
                options={"maxiter": 90, "ftol": 1e-10, "xtol": 1e-10},
                callback=cb,
            )
        except Exception:
            pass
        try:
            seq = _fourier_seq(x0, x, K)
            if seq is not None:
                sc = _score(seq)
                if np.isfinite(sc) and sc < local["score"]:
                    local["score"] = sc
                    local["seq"] = seq.copy()
        except Exception:
            pass
        if local["seq"] is not None and np.isfinite(local["score"]):
            pool.append((local["score"], local["seq"]))


def _populate_pool(pool, parent, rng):
    n = parent.size
    p = _norm(parent)

    def add(seq):
        try:
            seq = _norm(seq)
            sc = _score(seq)
            if np.isfinite(sc):
                pool.append((sc, seq))
        except Exception:
            pass

    add(p)
    for sig in (0.02, 0.05, 0.09):
        add(p * np.exp(rng.normal(0.0, sig, n)))
    if n >= 5:
        add(np.convolve(p, np.array([1.0, 2.0, 1.0], dtype=np.float64) / 4.0, mode="same"))
    for gamma in (0.70, 0.85, 1.15, 1.35):
        add(_power_dir(p, gamma))
    for beta in (0.12, 0.30, 0.55):
        add(_rev_blend_dir(p, beta))

    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    for lam in (0.5, 1.0, 1.7, 2.6):
        add(np.exp(lam * x))
        add(np.exp(-lam * x))

    xc = np.linspace(0.5 / n, 1.0 - 0.5 / n, n, dtype=np.float64)
    for aa in (0.25, 0.55, 1.0, 1.8):
        for bb in (0.25, 0.55, 1.0, 1.8):
            add(np.power(xc, aa) * np.power(1.0 - xc, bb))

    for pp in (1.5, 2.0):
        a1 = np.maximum(np.power(np.cos(0.5 * np.pi * x), pp) + 0.25 * np.sin(2.0 * np.pi * x), 1e-8)
        a2 = np.maximum(np.power(np.cos(0.5 * np.pi * (1.0 - x)), pp) + 0.25 * np.sin(2.0 * np.pi * (1.0 - x)), 1e-8)
        add(a1)
        add(a2)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + max(60.0, budget - 10.0)

    parent = _height_parent()
    n0 = int(parent.size)

    best_seq = _norm(parent)
    best_score = _score(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(64, dtype=np.float64)
        best_score = _score(best_seq)
        n0 = int(best_seq.size)

    def consider(arr):
        nonlocal best_seq, best_score
        try:
            sc = _score(arr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = _norm(arr)
        except Exception:
            pass

    consider(best_seq)

    rng = np.random.default_rng(seed)
    pool = [(best_score, best_seq.copy())]

    t_pool_end = t0 + min(95.0, max(35.0, budget * 0.075))
    if time.monotonic() < t_pool_end:
        _populate_pool(pool, best_seq, rng)
        if time.monotonic() < min(t_pool_end, deadline - 15.0):
            _lowdim_candidates(pool, n0, min(t_pool_end, deadline - 15.0), seed)

    for sc, seq in pool:
        consider(seq)
    consider(best_seq)

    pool.sort(key=lambda z: z[0])
    chosen = []
    for sc, seq in pool:
        seq = np.asarray(seq, dtype=np.float64).ravel()
        if not seq.size or not np.all(np.isfinite(seq)):
            continue
        close = False
        for s2 in chosen:
            if np.max(np.abs(seq - s2)) < 1e-6:
                close = True
                break
        if not close:
            chosen.append(seq)
        if len(chosen) >= 6:
            break
    if not chosen:
        chosen = [best_seq.copy()]

    t_now = time.monotonic()
    main_alloc = min(570.0, max(130.0, (deadline - t_now) * 0.45))
    t_main_end = min(deadline - 130.0, t_now + main_alloc)
    if t_main_end > t_now + 5.0:
        try:
            cand = _optimize(chosen[0], t_main_end, seed=seed + 1, max_stuck=75)
            consider(cand)
        except Exception:
            pass

    for idx, cand0 in enumerate(chosen[1:5]):
        t_now = time.monotonic()
        time_left = deadline - t_now - 85.0
        if time_left < 55.0:
            break
        alloc = min(85.0, time_left)
        try:
            cand = _optimize(cand0, t_now + alloc, seed=seed + 17 + idx, max_stuck=55)
            consider(cand)
        except Exception:
            pass

    t_now = time.monotonic()
    if t_now < deadline - 110.0:
        current = np.asarray(best_seq, dtype=np.float64).ravel()
        nc = int(current.size)
        nvars = []
        for dn in (-20, -40, 20, 40, -80, 80):
            nt = nc + dn
            if 64 <= nt <= 512 and nt != nc and nt not in nvars:
                nvars.append(nt)
        for idx, nt in enumerate(nvars[:3]):
            t_now = time.monotonic()
            time_left = deadline - t_now - 70.0
            if time_left < 45.0:
                break
            alloc = min(50.0, time_left)
            try:
                init = _resample(current, nt)
                consider(init)
                cand = _optimize(init, t_now + alloc, seed=seed + 211 + idx, max_stuck=45)
                consider(cand)
            except Exception:
                pass

    t_now = time.monotonic()
    if t_now < deadline - 12.0:
        try:
            cand = _optimize(best_seq, deadline - 3.0, seed=seed + 777, max_stuck=350)
            consider(cand)
        except Exception:
            pass

    best_seq = _norm(best_seq)
    consider(best_seq)

    out = [float(x) for x in best_seq]
    try:
        ev = globals().get("evaluate_sequence", None)
        if ev is not None:
            fs = ev(out)
            if not np.isfinite(fs):
                raise ValueError("invalid final sequence")
    except Exception:
        out = [1.0] * max(1, min(64, int(best_seq.size)))
        try:
            ev = globals().get("evaluate_sequence", None)
            if ev is not None:
                fs = ev(out)
                if not np.isfinite(fs):
                    out = [1.0] * 32
        except Exception:
            out = [1.0] * 32

    return out
