def _conv(a):

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _lp_direction(f, n, M, top_k=150, timeout=3.0):
    """Find LP direction: max sum(g) s.t. (f*g)_j <= M for top-k constraints, g >= 0."""
    b_f = _conv(f)
    # Get top-k constraints
    k = min(top_k, 2*n-1)
    top_idx = np.argsort(b_f)[-k:]
    
    # Build constraint matrix for these constraints
    # (f*g)_j = sum_i f[j-i] * g[i] for valid j-i
    A = np.zeros((k, n))
    for row, j in enumerate(top_idx):
        for i in range(n):
            ji = j - i
            if 0 <= ji < n:
                A[row, i] = f[ji]
    
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(k, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    if g.sum() < 1e-15:
        return None
    return g

Let me think about this differently. The score as a function of a[i] (with all other a's fixed):

Let s = sum(a) - a[i] (sum excluding coordinate i).
Then sum(a) with a[i]=v is s + v.
b_j(v) = b_j + 2(v-a[i]) * a[j-i] for j≠2i (where valid)
b_{2i}(v) = b_{2i} + 2(v-a[i]) * a[i] + (v-a[i])²

score(v) = 2n * max_j b_j(v) / (s+v)²

The max makes this piecewise, but the pieces are of the form:
score(v) = 2n * (c_j + d_j*(v-a[i])) / (s+v)² for some j (the active one)

or for j=2i: score(v) = 2n * (c_{2i} + d_{2i}*(v-a[i]) + (v-a[i])²) / (s+v)²

Each piece has at most 2 critical points (can find by derivative = 0). So I can:
1. Find which j gives max b_j(v) for the current v = a[i]
2. Compute the optimal v for that piece
3. Check if that v is still in the region where that j is the max
4. If not, check the next j
5. Repeat (usually converges in 1-2 iterations)

This is O(n) per coordinate (to find the argmax and check conditions). Total: O(n²) per sweep. Very fast.

But this is getting complicated to implement correctly. Let me just go with the simple approach: for each coordinate, try a few values (0, current, current*0.5, current*2, mean*0.5, mean) and recompute the full score. Recomputing the score is O(n²) due to the convolution.

For 400 coordinates * 6 values, each O(n²) = O(160K), total per sweep: 400*6*160K = 384M operations. In Python with numpy, the convolution part is in C, so maybe 400*6*0.05ms = 120ms per sweep. 200 sweeps = 24 seconds. That's acceptable.

import numpy as np
import time
import math
from scipy.optimize import linprog

def _conv(a):
    return np.convolve(a, a, mode='full')

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    return 2.0 * n * m / (s * s)

def _lp_direction(f, n, M, top_k=200, timeout=2.0):
    b_f = _conv(f)
    k = min(top_k, 2*n-1)
    top_idx = np.argsort(b_f)[-k:]
    
    rows = top_idx[:, None].astype(np.intp)
    cols = np.arange(n)[None, :].astype(np.intp)
    indices = rows - cols
    mask = (indices >= 0) & (indices < n)
    
    A = np.zeros((k, n), dtype=np.float64)
    A[mask] = f[indices[mask]]
    
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(k, M),
                  bounds=[(0, None)]*n, method='highs',
                  options={'time_limit': timeout, 'mip_rel_gap': 0})
    
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x), 0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _lp_phase(f0, n, stop_time, seed, top_k=200):
    rng = np.random.default_rng(seed)
    f = np.clip(f0, 0, None).astype(np.float64).copy()
    if f.sum() < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f.sum()
    
    best_f = f.copy()
    best_sc = _score(f)
    t = 0.3
    stall = 0
    max_stall = 100
    
    while time.monotonic() < stop_time - 1.0:
        b = _conv(f)
        M = float(b.max())
        
        g = _lp_direction(f, n, M, top_k=top_k)
        if g is None:
            stall += 5
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.01 * f.std(), n), 0, None)
            f /= f.sum()
            continue
        
        g_sum = g.sum()
        g = g * (f.sum() / g_sum)
        
        dist = float(np.linalg.norm(g - f))
        if dist < 1e-12 * max(1.0, float(np.linalg.norm(f))):
            stall += 3
            if stall > max_stall:
                break
            continue
        
        improved = False
        for _ in range(10):
            f_new = (1-t)*f + t*g
            f_new = np.maximum(f_new, 0)
            if f_new.sum() < 1e-10:
                t *= 0.5
                continue
            sc = _score(f_new)
            if sc < best_sc - 1e-13:
                f = f_new
                best_f = f.copy()
                best_sc = sc
                t = min(0.9, t * 1.15)
                improved = True
                stall = 0
                break
            elif sc < _score(f) - 1e-13:

                stall = 0
                break

                f /= f.sum()
                t = 0.3
                stall = 0
                max_stall = min(max_stall + 50, 300)
    
    return best_f, best_sc

def _grad_phase(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    a = np.clip(f0, 0, None).copy()
    a = a / a.sum()
    best_a = a.copy()

    T = 30.0

    no_improve = 0

        b = _conv(a)
        Tb = T * b
        Tb = Tb - Tb.max()
        p = np.exp(Tb)
        p_sum = p.sum()
        if p_sum < 1e-300:
            break
        p = p / p_sum
        
        # Gradient: grad[i] = 2 * sum_k a[k] * p[i+k]
        grad = 2.0 * np.correlate(p, a, mode='valid')
        
        # Project onto simplex tangent
        grad_tan = grad - grad.mean()
        
        improved = False
        lr_try = lr
        for _ in range(15):
            a_new = a - lr_try * grad_tan
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                lr_try *= 0.5
                continue
            a_new = a_new / s
            sc = _score(a_new)
            if sc < best_sc - 1e-13:
                a = a_new
                best_a = a.copy()
                best_sc = sc
                lr = min(1.0, lr_try * 1.1)
                improved = True
                no_improve = 0
                break
            elif sc < _score(a) - 1e-13:
                a = a_new
                improved = True
                no_improve = 0
                break
            else:
                lr_try *= 0.5
        
        if not improved:
            no_improve += 1
            lr *= 0.9

                    break
            if no_improve > 300:
                # Perturb

                no_improve = 0
    
    return best_a, best_sc
