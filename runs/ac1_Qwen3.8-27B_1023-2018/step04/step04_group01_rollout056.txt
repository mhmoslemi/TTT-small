import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    n = len(a)
    if n <= 2048:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _lp_direction(f, n, M, time_limit=10.0):
    """LP: max sum(g) s.t. conv(f,g)_k <= M, g >= 0"""
    # Build constraint matrix A: A[k,j] = f[k-j] if 0<=k-j<n else 0
    # Shape: (2n-1) x n

    b_ub = np.full(2 * n - 1, M)
    
    try:

                      bounds=[(0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": time_limit})

        return None
    
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None
    return g

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    if n <= 2048:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: 2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _quick_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _lp_direction(f, n, M, time_limit=8.0):
    """LP: max sum(g) s.t. conv(f,g)_k <= M, g >= 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j : j + n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g

            f /= max(f.sum(), 1e-10)
            continue

        dist = float(np.linalg.norm(g - f))
        if dist < 1e-7:
            stall += 3
            if stall > 40:
                break
            f = np.clip(f + rng.normal(0, 0.04, n), 0, None)
            f /= max(f.sum(), 1e-10)
            continue

        # Line search
        cur_sc = _quick_score(f)
        best_t = 0.0
        best_sc = cur_sc
        lo, hi = 0.0, 1.0
        for _ in range(22):
            mid = (lo + hi) * 0.5
            f_new = np.clip(f + mid * (g - f), 0, None)
            snew = f_new.sum()
            if snew < 1e-10:
                hi = mid
                continue
            f_new /= snew
            sc = _quick_score(f_new)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_t = mid
                lo = mid
            else:
                hi = mid

        if best_t > 1e-4 and best_sc < best_score - 1e-14:
            f_new = np.clip(f + best_t * (g - f), 0, None)
            f_new /= max(f_new.sum(), 1e-10)
            f = f_new
            best_f = f.copy()
            best_score = best_sc
            stall = 0
            t = min(0.9, t * 1.05)
        elif best_t > 1e-4:
            f_new = np.clip(f + best_t * (g - f), 0, None)
            f_new /= max(f_new.sum(), 1e-10)
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall >= 25:
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
                f /= max(f.sum(), 1e-10)
                stall = 0
                t = max(0.1, t * 0.5)

        it += 1

    return best_f, best_score


def _coord_descent(f0, stop_time, max_sweeps=3000, seed=777):
    """Vectorized coordinate descent."""
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    S = float(f.sum())
    if S < 0.01:
        f = np.ones(n) / n
        S = 1.0
    conv = _conv_full(f)
    best_score = 2.0 * n * conv.max() / (S * S)
    best_f = f.copy()
    rng = np.random.default_rng(seed)
    stall_sweeps = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 3.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break

            v = f[i]
            S_rest = S - v
            if S_rest < 1e-15:
                continue

            # Influence: a_shift[k] = f[k-i] if 0<=k-i<n else 0
            a_shift = np.zeros(2 * n - 1)
            st = i
            en = min(i + n, 2 * n - 1)
            a_shift[st:en] = f[: en - st]

            # Delta at position 2i
            have_2i = 2 * i <= 2 * n - 2
            pos_2i = 2 * i

            # Coarse grid
            x_max = min(max(3.0 * max(v, 0.005), 0.05), 50.0)
            nc = 96
            x_grid = np.linspace(0.0, x_max, nc)
            xm = x_grid - v

            # d_conv[m,k] = 2*xm[m]*a_shift[k] + xm[m]^2 * delta[k]
            d_conv = 2.0 * xm[:, None] * a_shift[None, :]
            if have_2i:
                d_conv[:, pos_2i] += xm * xm
            conv_new = conv[None, :] + d_conv
            conv_new = np.maximum(conv_new, 0.0)
            max_c = conv_new.max(axis=1)
            S_new = S_rest + x_grid
            valid = S_new > 0.01
            sc_grid = np.full(nc, math.inf)
            sc_grid[valid] = 2.0 * n * max_c[valid] / (S_new[valid] ** 2)

            bi = int(np.argmin(sc_grid))
            best_x = x_grid[bi]
            best_obj = sc_grid[bi]

            # Fine grid
            if best_x != v and math.isfinite(best_obj):
                dw = x_max / nc
                x_lo = max(0.0, best_x - 3 * dw)
                x_hi = min(x_max, best_x + 3 * dw)
                nf = 64
                x_fine = np.linspace(x_lo, x_hi, nf)
                xf = x_fine - v
                d_fine = 2.0 * xf[:, None] * a_shift[None, :]
                if have_2i:
                    d_fine[:, pos_2i] += xf * xf
                conv_fine = conv[None, :] + d_fine
                conv_fine = np.maximum(conv_fine, 0.0)
                max_f = conv_fine.max(axis=1)
                S_f = S_rest + x_fine
                vf = S_f > 0.01
                sc_fine = np.full(nf, math.inf)
                sc_fine[vf] = 2.0 * n * max_f[vf] / (S_f[vf] ** 2)
                fi = int(np.argmin(sc_fine))
                if sc_fine[fi] < best_obj:
                    best_x = x_fine[fi]
                    best_obj = sc_fine[fi]

            # Accept
            if best_x != v and best_obj < best_score - 1e-14:
                w = best_x - v
                f[i] = best_x
                S += w
                conv += 2.0 * w * a_shift
                if have_2i:
                    conv[pos_2i] += w * w
                np.maximum(conv, 0.0, out=conv)
                best_score = best_obj
                improved = True

        if improved:
            stall_sweeps = 0
        else:
            stall_sweeps += 1
            if stall_sweeps > 30:
                # Perturb
                f = np.clip(f + rng.normal(0, 0.008, n), 0, None)
                S = f.sum()
                if S < 0.01:
                    f = best_f.copy()
                    S = f.sum()
                conv = _conv_full(f)
                stall_sweeps = 0
                if sweep > 200 and stall_sweeps > 0:
                    pass

    # Final
    S = f.sum()
    if S < 0.01:
        return best_f, _quick_score(best_f)
    f = f / S
    fs = _quick_score(f)
    if fs < best_score:
        best_score = fs
        best_f = f.copy()
    else:
        best_score = _quick_score(best_f)
    return best_f, best_score


def _grad_phase(f0, n, stop_time, seed, max_iters=50000):
    """Gradient descent with softmax-smoothed max."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel().copy(), 0, None)
    f /= max(f.sum(), 1e-10)
    best_f = f.copy()
    best_score = _quick_score(f)

    alpha = 0.02
    alpha_min = 1e-12
    tau = 60.0

    it = 0
    while it < max_iters and time.monotonic() < stop_time - 3.0:
        if it % 2000 == 0:
            if time.monotonic() > stop_time - 3.0:
                break

        b = _conv_full(f)
        S = float(f.sum())
        M = float(b.max())
        if M < 1e-20:
            break
        score = 2.0 * n * M / (S * S)
        if score < best_score - 1e-14:
            best_score = score
            best_f = f.copy()

        # Softmax weights
        bs = b - M
        tau_b = tau * bs
        tau_b -= tau_b.max()
        p = np.exp(tau_b)
        ps = p.sum()
        if ps < 1e-30:
            p = np.zeros_like(b)
            p[np.argmax(b)] = 1.0
        else:
            p /= ps

        # Gradient: grad[i] = 2 * sum_j p[i+j] * f[j]
        # = 2 * cross-correlation of p and f
        fsize = next_fast_len(4 * n)
        P = rfft(p, fsize)
        F = rfft(f, fsize)
        # cross-correlation: (p * f_rev)
        fr = f[::-1].copy()
        Fr = rfft(fr, fsize)
        cc = irfft(P * np.conj(Fr), fsize)
        # (p * f_rev)[i + n - 1] gives sum_j p[i+j] * f[j]
        grad = 2.0 * cc[n - 1 : 2 * n - 1]

        # Score gradient: (2n/S^2) * (grad - 2*M/S)
        sg = (2.0 * n / (S * S)) * (grad - 2.0 * M / S)

        # Projected gradient step
        f_raw = f - alpha * sg
        # Project onto {x >= 0, sum(x) = 1}
        f_new = np.maximum(f_raw, 0)
        sn = f_new.sum()
        if sn < 1e-15:
            f_new = np.ones(n) / n
            sn = 1.0
        f_new /= sn

        sc_new = _quick_score(f_new)
        if sc_new < score - 1e-15:
            f = f_new
            alpha = max(alpha_min, alpha * 0.99)
        else:
            alpha = max(alpha_min, alpha * 0.5)
            if alpha < alpha_min * 10:
                f = np.clip(f + rng.normal(0, 0.003, n), 0, None)
                f /= max(f.sum(), 1e-10)
                alpha = 0.01

        it += 1

    return best_f, best_score


def _pair_polish(f0, stop_time, max_rounds=100, seed=55555):
    """Random pair coordinate moves."""
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f /= max(f.sum(), 1e-10)
    best_score = _quick_score(f)
    best_f = f.copy()
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 3.0:
            break
        n_trials = 30
        trial_improved = False
        for _ in range(n_trials):
            if time.monotonic() > stop_time - 2.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = f[i], f[j]
            if oi < 1e-12 and oj < 1e-12:
                continue

            best_pair_sc = best_score
            best_pi, best_pj = oi, oj
            for mi in (0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 0.0):
                ni = oi * mi if oi > 1e-12 else (0.02 if mi > 0 else 0.0)
                for mj in (0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 0.0):
                    nj = oj * mj if oj > 1e-12 else (0.02 if mj > 0 else 0.0)
                    f[i], f[j] = ni, nj
                    if f.sum() < 0.01:
                        f[i], f[j] = oi, oj
                        continue
                    sc = _quick_score(f)
                    if sc < best_pair_sc - 1e-14:
                        best_pair_sc = sc
                        best_pi, best_pj = ni, nj
            f[i], f[j] = best_pi, best_pj
            if best_pair_sc < best_score - 1e-14:
                best_score = best_pair_sc
                best_f = f.copy()
                trial_improved = True

        if not trial_improved and rnd > 10:
            break

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 20.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except (NameError, Exception):
        parent = np.ones(400, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(400, dtype=np.float64)
    parent = np.clip(np.nan_to_num(parent, nan=0.0, posinf=1000.0, neginf=0.0), 0.0, 1000.0)
    n = len(parent)

    # Fallback
    f_init = parent.copy()
    if f_init.sum() < 0.01:
        f_init = np.ones(n) / n

    best_seq = f_init.copy()
    best_score = math.inf
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        best_score = _quick_score(best_seq)

    def _update(seq):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64)
        seq = np.clip(np.nan_to_num(seq, nan=0.0), 0.0, 1000.0)
        if seq.size == 0 or seq.sum() < 0.01:
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            sc = math.inf
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_seq = seq.copy()

    # Phase 1: LP optimization (~35% of budget)
    lp_end = t0 + budget * 0.35
    if time.monotonic() < lp_end - 10:
        try:
            res, sc = _lp_phase(f_init, n, lp_end, seed)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update(res)
        except Exception:
            pass

    # Phase 2: Coordinate descent (~30%)
    cd_end = min(t0 + budget * 0.65, hard_deadline - 100)
    if time.monotonic() < cd_end - 10:
        try:
            res, sc = _coord_descent(best_seq, cd_end, max_sweeps=3000, seed=seed + 100)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update(res)
        except Exception:
            pass

    # Phase 3: Gradient polishing (~20%)
    grad_end = min(t0 + budget * 0.85, hard_deadline - 80)
    if time.monotonic() < grad_end - 10 and best_score > 1.502:
        try:
            res, sc = _grad_phase(best_seq, len(best_seq), grad_end, seed + 200)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update(res)
        except Exception:
            pass

    # Phase 4: Another CD pass on improved solution
    cd2_end = min(t0 + budget * 0.93, hard_deadline - 50)
    if time.monotonic() < cd2_end - 10:
        try:
            res, sc = _coord_descent(best_seq, cd2_end, max_sweeps=2000, seed=seed + 300)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update(res)
        except Exception:
            pass

    # Phase 5: Pair polish
    pair_end = min(t0 + budget * 0.97, hard_deadline - 30)
    if time.monotonic() < pair_end - 10:
        try:
            res, sc = _pair_polish(best_seq, pair_end, max_rounds=80, seed=seed + 400)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update(res)
        except Exception:
            pass

    # Final CD micro-polish
    if time.monotonic() < hard_deadline - 10:
        try:
            res, sc = _coord_descent(best_seq, hard_deadline - 8, max_sweeps=500, seed=seed + 500)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update(res)
        except Exception:
            pass

    # Final validation and return
    best_seq = np.clip(np.nan_to_num(np.asarray(best_seq, dtype=np.float64), nan=0.0), 0.0, 1000.0)
    if best_seq.size == 0 or best_seq.sum() < 0.01:
        best_seq = f_init.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in f_init]))
        except Exception:
            return [1.0]
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in f_init]

    return [float(x) for x in best_seq]
