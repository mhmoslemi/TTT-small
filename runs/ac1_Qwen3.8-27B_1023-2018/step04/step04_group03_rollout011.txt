from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import random

def _conv_full(a):

    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)

    if not math.isfinite(m) or m < 0:

    return 2.0 * n * m / (s * s)

However, if we keep ε small enough (say ε = 0.01 * mean(f)), the quadratic term (δ*δ)_k is at most sum_i ε^2 = n*ε^2, which is small compared to M. So the linearization is good.

                if stall > max_stall:
                    # Try max_sum approach
                    b = _conv_full(f)
                    M = float(b.max())
                    g = _lp_max_sum(f, n, M, timeout=6.0)
                    if g is not None:
                        g_sum = g.sum()
                        f_sum = f.sum()
                        g = g * (f_sum / g_sum)
                        f_try = (1-0.1) * f + 0.1 * g
                        f_try = np.clip(f_try, 0, None)
                        sc_try = _score(f_try)
                        if sc_try < sc - 1e-12:
                            f = f_try
                            sc = sc_try
                            n_steps += 1
                        stall = 0
                        continue
                    else:
                        break
                continue
        else:
            b = _conv_full(f)
            M = float(b.max())
            g = _lp_max_sum(f, n, M, timeout=6.0)
            if g is None:
                stall += 3
                if stall > max_stall:
                    use_min_peak = True
                    stall = 0
                continue
            g_sum = g.sum()
            f_sum = f.sum()
            if g_sum < 1e-15:
                stall += 2
                continue
            g = g * (f_sum / g_sum)
            f_new = (1 - 0.15) * f + 0.15 * g
            f_new = np.clip(f_new, 0, None)
        
        sc_new = _score(f_new)
        if sc_new < sc - 1e-12:
            f = f_new
            sc = sc_new
            n_steps += 1
            stall = 0
            epsilon *= 0.95  # slightly decrease step
            # Occasionally switch approach
            if n_steps % 20 == 0:
                use_min_peak = not use_min_peak
        else:
            # Try damped step
            f_try = 0.7 * f + 0.3 * f_new
            f_try = np.clip(f_try, 0, None)
            sc_try = _score(f_try)
            if sc_try < sc - 1e-12:
                f = f_try
                sc = sc_try
                n_steps += 1
                stall = 0
                epsilon *= 0.7

                epsilon *= 0.8
                if stall > max_stall:
                    # Random perturbation to escape
                    noise = rng.normal(0, 0.02 * max(1.0, f.mean()), n)
                    f = np.clip(f + noise, 0, None)
                    sc = _score(f)
                    epsilon = 0.03 * max(1.0, f.mean())
                    stall = 0
                    max_stall = min(max_stall + 20, 150)
                    use_min_peak = not use_min_peak
    
    return f, sc

def _coord_polish(f0, stop_time, seed):
    """Coordinate descent for fine-tuning."""
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 0, None)
    sc = _score(f)
    
    mults = np.array([0.0, 0.05, 0.1, 0.25, 0.5, 0.7, 0.8, 0.9, 1.0,
                      1.1, 1.2, 1.5, 2.0, 3.0, 0.0])
    
    stall_rounds = 0
    max_stall = 12
    
    while time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = f[i]
            best_local = sc
            best_val = orig
            
            for m in mults:
                f[i] = orig * m
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                s = _score(f)
                if s < best_local - 1e-14:
                    best_local = s
                    best_val = f[i]
            
            f[i] = best_val
            if best_local < sc - 1e-13:
                sc = best_local
                improved = True
        
        if not improved:
            stall_rounds += 1
            if stall_rounds >= max_stall:
                break
        else:
            stall_rounds = 0
    
    return f.copy(), sc
