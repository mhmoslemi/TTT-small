import numpy as np

from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_solve(f, n, M, time_limit=5.0):
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    return g / g_sum

    # subgradient of c[k_star] w.r.t. a[i] = 2*a[k_star - i] if 0 <= k_star - i < n
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for i in range(lo, hi + 1):
        grad[i] = 2.0 * a[k_star - i]
    
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a, c
    a_new = a_new / s
    c_new = _conv_full(a_new)
    return a_new, c_new

def _nesterov_optimize(a0, n, stop_time, seed=99):
    """Nesterov accelerated gradient descent on the smooth surrogate."""

    a = np.clip(np.asarray(a0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_c = _conv_full(a)
    best_sc = 2.0 * n * float(best_c.max())
    
    y = a.copy()
    momentum = 0.9
    lr = 0.01
    
    for it in range(5000):
        if time.monotonic() > stop_time - 2.0:
            break
        c_y = _conv_full(y)
        k_star = int(np.argmax(c_y))
        
        # Subgradient of c[k_star]
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            grad[i] = 2.0 * y[k_star - i]
        
        # Projected gradient step
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        # Nesterov update
        t_k = (1.0 + math.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        y = a_new + ((momentum - 1.0) / t_k) * (a_new - a)
        y = np.maximum(y, 0)
        a = a_new
        
        c_a = _conv_full(a)
        sc = 2.0 * n * float(c_a.max())
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
        
        # Adaptive learning rate
        if it % 100 == 99:
            lr *= 0.95

    return best_a, best_sc

            if 0 <= k_star - i < n:
                contributing.append((i, a * best_a[k_star - i]))
        
        # Focus optimization on top contributors
        contributing.sort(key=lambda x: -x[1])
        top_idx = [x[0] for x in contributing[:min(32, len(contributing))]]
        
        # Try random redistributions among top contributors
        for trial in range(20):
            if time.monotonic() > stop_time - 2.0:
                break
            trial_a = best_a.copy()
            # Randomly perturb top contributors
            for idx in top_idx:
                factor = rng.uniform(0.3, 2.0)
                trial_a[idx] *= factor
            s_t = trial_a.sum()
            if s_t < 1e-10:
                continue
            trial_a = trial_a / s_t
            c_t = _conv_full(trial_a)
            sc_t = 2.0 * n * float(c_t.max())
            if sc_t < best_sc - 1e-13:
                best_sc = sc_t
                best_a = trial_a.copy()
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    # Get parent
    try:
        h = globals()["height_sequence_1"]
        parent = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(384, dtype=np.float64)
    if parent.size == 0:
        parent = np.ones(384, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    if parent.size == 0:
        parent = np.ones(384, dtype=np.float64)

    n_p = len(parent)
    n = n_p

    # Fallback
    fallback = parent.copy()
    fs = fallback.sum()
    if fs < 0.01:
        fallback = np.ones(n_p)
    try:
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(n_p) / n_p
        fallback_sc = float(evaluate_sequence([float(x) for x in fallback]))

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _try_update(seq):
        nonlocal best_seq, best_score
        if seq is None or len(seq) == 0:
            return
        seq = np.clip(np.asarray(seq, dtype=np.float64), 0, 1000)
        try:
            sc = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # Phase 1: LP optimization (40% of time)
    phase1_end = t0 + budget * 0.40
    rng_main = np.random.default_rng(seed)

    # Initializations
    inits = [
        ("parent", parent.copy()),
    ]
    # Sine-based
    i_arr = np.arange(n, dtype=np.float64)
    inits.append(("sine", 0.5 + 0.5 * np.sin(np.pi * i_arr / (n - 1))))
    # Cosine bell
    inits.append(("cos_bell", 0.5 + 0.5 * np.cos(np.pi * (i_arr - (n-1)/2) / ((n-1)/2 + 1e-9))))
    # Parent smoothed
    from scipy.ndimage import gaussian_filter1d
    inits.append(("parent_smooth", gaussian_filter1d(parent, sigma=3.0)))
    # Zipper
    z = np.ones(n)
    z[::2] *= 1.4
    z[1::2] *= 0.55
    inits.append(("zipper", z))

    time_per_init = max(10.0, (phase1_end - t0) / len(inits))

    for idx, (name, init) in enumerate(inits):
        if time.monotonic() > phase1_end - 5.0:
            break
        stop = min(phase1_end - 3.0, time.monotonic() + time_per_init)
        if time.monotonic() >= stop:
            break
        try:
            res, sc = _lp_optimize(init, n, stop, seed + 1009 * (idx + 1), max_iters=150)
            if res is not None and np.all(np.isfinite(res)):
                _try_update(res)
        except Exception:
            pass

    # Phase 2: Fast coordinate descent (35% of time)
    phase2_end = t0 + budget * 0.75
    cd_start = best_seq.copy()
    try:
        res, sc = _fast_coord_desc(cd_start, min(phase2_end, time.monotonic() + (phase2_end - phase1_end) * 0.5),
                                   seed=5000 + seed)
        if res is not None:
            _try_update(res)
    except Exception:
        pass

    # Aggressive CD
    if time.monotonic() < phase2_end - 10.0:
        try:
            res, sc = _aggressive_cd(best_seq, phase2_end - 5.0, seed=7000 + seed)
            if res is not None:
                _try_update(res)
        except Exception:
            pass

    # Phase 3: Nesterov + multi-scale (15% of time)
    phase3_end = t0 + budget * 0.90
    if time.monotonic() < phase3_end - 10.0:
        try:
            res, sc = _nesterov_optimize(best_seq, n, min(phase3_end - 5.0, time.monotonic() + 60),
                                         seed=11000 + seed)
            if res is not None:
                _try_update(res)
        except Exception:
            pass

    if time.monotonic() < phase3_end - 5.0:
        try:
            res, sc = _multi_scale_optimize(best_seq, n, min(phase3_end - 3.0, time.monotonic() + 30),
                                            seed=13000 + seed)
            if res is not None:
                _try_update(res)
        except Exception:
            pass

    # Phase 4: Final fine polishing (remaining time)
    final_stop = deadline - 3.0
    if time.monotonic() < final_stop - 5.0:
        for pidx in range(3):
            if time.monotonic() >= final_stop - 3.0:
                break
            p_stop = min(final_stop - 2.0, time.monotonic() + 30)
            try:
                if pidx % 2 == 0:
                    res, sc = _aggressive_cd(best_seq, p_stop, seed=20000 + pidx * 7)
                else:
                    res, sc = _fast_coord_desc(best_seq, p_stop, seed=21000 + pidx * 13)
                if res is not None:
                    _try_update(res)
            except Exception:
                pass

    # Final LP burst if time permits
    if time.monotonic() < final_stop - 15.0:
        try:
            res, sc = _lp_optimize(best_seq, n, final_stop - 5.0, seed=30000 + seed, max_iters=50)
            if res is not None:
                _try_update(res)
        except Exception:
            pass

    # One last coordinate descent
    if time.monotonic() < final_stop - 2.0:
        try:
            res, sc = _fast_coord_desc(best_seq, final_stop, seed=40000)
            if res is not None:
                _try_update(res)
        except Exception:
            pass

    # Final validation
    final_seq = np.clip(best_seq, 0, 1000)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in final_seq]
