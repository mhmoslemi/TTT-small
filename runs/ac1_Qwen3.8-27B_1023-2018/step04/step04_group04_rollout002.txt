def _subgrad_step(f, n):
    b = _fft_conv(f)
    m = float(b.max())
    eps = max(1e-14, 1e-10 * m)
    active = np.flatnonzero(b >= m - eps)
    grad = np.zeros(n)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            idx = k - np.arange(i_lo, i_hi + 1)
            grad[i_lo:i_hi + 1] += 2.0 * f[idx]
    if len(active) > 0:
        grad /= len(active)
    S = f.sum()
    f_sub = 2.0 * n * (grad * S - 2.0 * m) / (S ** 3)
    f_sub -= np.mean(f_sub)
    return f_sub

                    for mj in [0.3, 0.6, 1.4, 0.0]:
                        f[i] = oi * mi if oi > 1e-15 else 0.0
                        f[j] = oj * mj if oj > 1e-15 else 0.0
                        if f.sum() < 0.01:
                            f[i], f[j] = oi, oj
                            continue
                        sc = _score(f)
                        if sc < best_sc - 1e-15:
                            best_sc = sc
                            best_f = f.copy()
                            improved = True
                            found = True
                            break
                    if found:
                        break
                if not found:
                    f[i], f[j] = oi, oj

        if not improved:
            stall_epochs += 1
            if stall_epochs >= 4:
                break
        else:
            stall_epochs = 0

    return best_f, best_sc


def _annealing_refine(f0, stop_time, seed, max_iter=2000):
    """Simple simulated-annealing-like perturbation for local escape."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    n = len(f)
    f = f / max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = _score(f)
    cur_sc = best_sc
    T = 0.005
    T_min = 0.0001
    cool = 0.998

    for it in range(max_iter):
        if time.monotonic() > stop_time:
            break
        if it % 200 == 0:
            T = max(T_min, T * 0.95)

        pert = f + rng.normal(0, T, n)
        np.clip(pert, 0, None, out=pert)
        ps = pert.sum()
        if ps < 0.01:
            continue
        pert = pert / ps
        sc_new = _score(pert)

        if sc_new < cur_sc or rng.uniform() < math.exp(-(sc_new - cur_sc) / max(T, 1e-10)):
            f = pert
            cur_sc = sc_new
            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_f = f.copy()

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_parent = len(parent)

    # Fallback
    fallback = parent.copy()
    fallback[~np.isfinite(fallback)] = 0.0
    np.clip(fallback, 0, 1000.0, out=fallback)
    if fallback.sum() < 0.01:
        fallback = np.ones(320)
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(320) / 320.0
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update(a, sc):
        nonlocal best_seq, best_score
        if math.isfinite(sc) and sc < best_score - 1e-13:
            a_clip = np.clip(a, 0, 1000.0)
            if a_clip.sum() >= 0.01 and np.all(np.isfinite(a_clip)):
                best_seq = a_clip.copy()
                best_score = sc

    # Phase 1: LP optimization on parent length (majority of time)
    p_norm = parent / max(parent.sum(), 1e-10)
    phase1_end = t0 + budget * 0.45
    res, sc = _lp_optimize(p_norm, n_parent, phase1_end, seed)
    if math.isfinite(sc):
        _update(res, sc)

    # Phase 1b: Second LP run with perturbed init
    if time.monotonic() < t0 + budget * 0.52:
        rng_p = np.random.default_rng(seed + 7)
        p2 = np.clip(best_seq / max(best_seq.sum(), 1e-10) + rng_p.normal(0, 0.06, n_parent), 0, None)
        p2 = p2 / max(p2.sum(), 1e-10)
        stop1b = min(t0 + budget * 0.52, phase1_end + 20)
        res2, sc2 = _lp_optimize(p2, n_parent, stop1b, seed + 100)
        if math.isfinite(sc2):
            _update(res2, sc2)

    # Phase 1c: Third LP run from a different angle
    if time.monotonic() < t0 + budget * 0.57:
        rng_p2 = np.random.default_rng(seed + 13)
        p3 = np.clip(best_seq / max(best_seq.sum(), 1e-10) + rng_p2.choice([0, 1], n_parent).astype(float) * 0.1, 0, None)
        p3 = p3 / max(p3.sum(), 1e-10)
        stop1c = min(t0 + budget * 0.57, t0 + budget * 0.52 + 25)
        res3, sc3 = _lp_optimize(p3, n_parent, stop1c, seed + 200)
        if math.isfinite(sc3):
            _update(res3, sc3)

    # Phase 2: Subgradient refinement
    phase2_end = t0 + budget * 0.65
    if time.monotonic() < phase2_end:
        f_sg = best_seq / max(best_seq.sum(), 1e-10)
        res_sg, sc_sg = _subgrad_optimize(f_sg, n_parent, phase2_end, seed + 300)
        if math.isfinite(sc_sg):
            _update(res_sg, sc_sg)

    # Phase 3: Coordinate descent (important for small improvements)
    phase3_end = t0 + budget * 0.78
    if time.monotonic() < phase3_end:
        res_cd, sc_cd = _coord_descent(best_seq, phase3_end, seed + 400, max_epochs=50)
        if math.isfinite(sc_cd):
            _update(res_cd, sc_cd)

    # Phase 4: Try nearby lengths
    phase4_end = deadline - 30
    if time.monotonic() < phase4_end:
        alt_lengths = sorted(set([n_parent - 2, n_parent - 1, n_parent + 1, n_parent + 2,
                                   n_parent + 4, n_parent + 8, n_parent - 4]))
        alt_lengths = [L for L in alt_lengths if L >= 16]
        # Deduplicate with parent
        alt_lengths = [L for L in alt_lengths if L != n_parent]
        time_per_alt = min(25.0, (phase4_end - time.monotonic()) / max(1, len(alt_lengths)))
        for L in alt_lengths:
            if time.monotonic() > phase4_end - 5:
                break
            init_L = np.interp(np.linspace(0, 1, L), np.linspace(0, 1, n_parent),
                               best_seq / max(best_seq.sum(), 1e-10))
            init_L = init_L / max(init_L.sum(), 1e-10)
            stop_L = min(phase4_end, time.monotonic() + time_per_alt)
            try:
                rL, sL = _lp_optimize(init_L, L, stop_L, seed + 1000 + L)
                if math.isfinite(sL):
                    _update(rL, sL)
            except Exception:
                pass

    # Phase 5: APH (annealing) for escape
    if time.monotonic() < deadline - 15:
        f_an = best_seq / max(best_seq.sum(), 1e-10)
        res_an, sc_an = _annealing_refine(f_an, deadline - 15, seed + 2000)
        if math.isfinite(sc_an):
            _update(res_an, sc_an)

    # Phase 6: Final coordinate descent polish
    if time.monotonic() < deadline - 5:
        res_final, sc_final = _coord_descent(best_seq, deadline - 5, seed + 5000, max_epochs=30)
        if math.isfinite(sc_final):
            _update(res_final, sc_final)

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
