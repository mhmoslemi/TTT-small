```python
import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len
import scipy.sparse as sp

def _rfft_conv(a):
    """Fast convolution using FFT."""
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _rfft_conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A_sparse(f, n):
    """Build sparse cross-correlation matrix A where (Ag)(k) = (f*g)(k)."""
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            idx = k - j
            if 0 <= idx < n:
                rows_list.append(k)
                cols_list.append(j)
                vals_list.append(f[idx])
    A = sp.csr_matrix((vals_list, (rows_list, cols_list)), shape=(2*n-1, n))
    return A

def _lp_direction(f, n, M, timeout=3.0):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M, g >= 0."""
    A = _build_A_sparse(f, n)
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g
```
