where C_k(a[i]) is linear in a[i] for each k (since C_k = sum_j a[j]*a[k-j], and a[i] appears in at most 2 terms for each k... wait, actually a[i] appears in C_k as a[i]*a[k-i] and a[k-i]*a[i], so C_k is linear in a[i] for each k).

So score(a[i]) = 2n * max_k (C_k_base + a[i] * w_k) / (S_base + a[i])^2

where C_k_base = C_k(a) - a[i]*w_k and w_k = a[k-i] (with the appropriate convention).

This is a ratio of a piecewise-linear function to a quadratic, which can be minimized in 1D relatively easily (find the breakpoints where the max switches, and minimize on each interval).

But this is getting complex. For simplicity, I'll use a grid + refinement approach in the coordinate descent.

        F = rfft(a, _fsize)
        return irfft(F * F, _fsize)[:2 * n - 1]

    def _score(a):
        S = float(a.sum())
        if S < 0.01:
            return math.inf
        C = _convolver(a)
        max_C = float(np.max(C))
        if max_C < 0:
            max_C = 0.0
        return 2.0 * n * max_C / (S * S)

    def _score_grad(a, beta=20.0):
        n = len(a)
        S = float(a.sum())
        if S < 1e-10:
            return 1e10, np.zeros(n)
        C = _convolver(a)
        C = np.clip(C, 0, None)
        max_C = float(C.max())
        score = 2.0 * n * max_C / (S * S)
        c_range = max_C - float(C.min())
        if c_range < 1e-14:
            p = np.full(2 * n - 1, 1.0 / (2 * n - 1))
            phi = max_C
        else:
            scaled = beta * (C - max_C) / c_range
            es = np.exp(scaled)
            p = es / es.sum()
            phi = max_C + (c_range / beta) * math.log(float(es.sum()))
        # h[i] = sum_k p[k]*a[k-i] = (a_rev * p)[n-1+i]
        a_rev = a[::-1]
        cs = next_fast_len(3 * n + 1)
        A_r = rfft(a_rev, cs)
        P = rfft(p, cs)
        conv = irfft(A_r * P, cs)[:3 * n - 1]
        h = conv[n - 1:2 * n - 1]
        grad = (2.0 * n / (S * S)) * (h + a) - (4.0 * n * phi / (S * S * S))
        return score, grad

    def _lbfgs_run(a0, stop_time, beta=15.0, max_iter=150):

            sc, g = _score_grad(x, beta=beta)
            if not np.isfinite(sc):
                return 1e8, np.zeros_like(x)
            if not np.all(np.isfinite(g)):
                g = np.zeros_like(x)
            return sc, g

    def _lp_direction(f, timeout=7.0):
        M = float(np.max(_convolver(f)))

        except Exception:

        return None, M

    def _lp_optimize(f0, stop_time):

                    stall = 0
