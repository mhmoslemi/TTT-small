def _lp_best_response(a, M, n, time_limit=4.0):
    """LP: max sum(g) s.t. (g*a)_k <= M, g >= 0. Returns normalized g or None."""
    # Build A: A[k,j] = a[k-j] if 0<=k-j<n else 0
    # A is (2n-1) x n
    k = np.arange(2 * n - 1)[:, None].astype(np.int64)
    j = np.arange(n)[None, :].astype(np.int64)
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[d[mask]]

    try:

                      bounds=[(0.0, None)] * n,

                               'primal_feasibility_tolerance': 1e-12})
    except Exception:
        return None

        return None
    g = np.clip(res.x, 0.0, None)
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs

        inits.append(v / v.sum())
    
    # Exponential
    for al in [0.5, 1.0, 2.0, 3.0, 5.0]:
        v = np.exp(-al * x / n_parent)
        inits.append(v / v.sum())
    
    # Gaussian (various widths)
    for w in [0.05, 0.1, 0.2, 0.3, 0.5]:
        v = np.exp(-0.5 * ((x - ctr) / (w * n_parent / 2.0)) ** 2)
        inits.append(v / v.sum())
    
    # Cosine
    for fr in [1, 2, 3]:
        v = np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * fr * x / max(n_parent - 1, 1)), 0.01)
        inits.append(v / v.sum())
    
    # Blends with parent
    for frac in [0.3, 0.5, 0.7]:
        for init in inits[:4]:
            v = frac * pn + (1.0 - frac) * init
            inits.append(v / v.sum())
    
    # Random smooth
    rng_init = np.random.default_rng(12345)
    for _ in range(5):
        v = np.exp(rng_init.normal(0, 0.5, n_parent))
        inits.append(v / v.sum())
    
    t3 = min(t0_global + 900.0, deadline - 120.0)
    per_init = max(4.0, (t3 - time.monotonic()) / max(1, len(inits)))
    
    for i, init in enumerate(inits):
        if time.monotonic() > t3 - 5.0:
            break
        alloc = min(time.monotonic() + per_init, t3 - 3.0)
        try:
            r = _optimize_softmax(init, alloc, seed=5000 + i)
            update(r)
        except Exception:
            pass
    
    # ===== Phase 4: LP polish again on best =====
    t4 = min(t0_global + 1100.0, deadline - 60.0)
    ab = best_seq / max(best_seq.sum(), 1e-15)
    try:
        r = _optimize_lp(ab, t4, seed=777, max_stuck=200)
        update(r)
    except Exception:
        pass
    
    # ===== Phase 5: Explore different lengths =====
    if time.monotonic() < deadline - 40:
        dn_list = [-10, -5, -2, -1, 1, 2, 5, 10, 20]
        for dn in dn_list:
            if time.monotonic() > deadline - 15:
                break
            nt = n_parent + dn
            if nt < 10 or nt > 500:
                continue
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, nt)
            a_old = best_seq / max(best_seq.sum(), 1e-15)
            init_new = np.interp(x_new, x_old, a_old)
            init_new = np.clip(init_new, 0, 1000)
            sn = init_new.sum()
            if sn < 1e-10:
                continue
            init_new /= sn
            try:
                r = _optimize_softmax(init_new, time.monotonic() + 30, seed=10000 + nt)
                update(r)
            except Exception:
                pass
    
    # ===== Phase 6: Final polish =====
    if time.monotonic() < deadline - 10:
        af = best_seq / max(best_seq.sum(), 1e-15)
        try:
            r = _optimize_softmax(af, deadline - 5, seed=99999, eps_start=0.5 / af.size)
            update(r)
        except Exception:
            pass
    
    # ===== Final validation with exact evaluator =====
    final_list = [float(x) for x in best_seq]
    try:
        fs = float(evaluate_sequence(final_list))
        if not np.isfinite(fs):
            raise ValueError("invalid final")
    except Exception:
        # Fallback: uniform sequence
        return [1.0] * 32
    
    return final_list
