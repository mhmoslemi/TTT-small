            w[int(np.argmax(c))] = 1.0

        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return f, grad
    
    stages = [

        p_cur = np.maximum(p_cur, 0)

            best_p = p_cur.copy()
            p = p_cur.copy()
            u = np.log(np.maximum(p, 1e-30))
            u = u - u.mean()
        else:
            p = best_p.copy()
            u = np.log(np.maximum(p, 1e-30))
            u = u - u.mean()
        if consider:
            try: consider(best_p)
            except: pass
    
    return best_p

def _coord_descent(p, stop_time, consider=None, seed=12345):
    n = int(p.size)
    if n == 0:
        return p, float('inf')
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p, float('inf')
    p = p / s
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    Ltot = 2 * n - 1
    G = 48
    flat = 0
    sweep = 0
    GR = (math.sqrt(5.0) - 1.0) / 2.0

    def eval_1d_scalar(w, base, infl, diag, max_const, S, n):
        s_new = S + w
        if s_new <= 1e-15:
            return float('inf')
        aff_max = float(np.max(base + w * infl))
        if 0 <= diag < len(base):
            dval = base[diag] + w * infl[diag] + w * w
            if dval > aff_max:
                aff_max = dval
        m = max(aff_max, max_const)
        return 2.0 * n * m / (s_new * s_new)

    max_sweeps = 50000

        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 1.0:
                break
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end]
            infl = 2.0 * p[:Ls]
            diag = i  # 2i - start = 2i - i = i

            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < Ltot:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

            S = 1.0  # normalized sum
            x_hi = max(30.0 * v, 20.0 / n, 0.01)
            x_hi = min(x_hi, 0.5)
            x_hi = max(x_hi, 1e-8)

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                ins = np.searchsorted(pts, v)
                pts = np.insert(pts, ins, v)

            ws = pts - v
            s_new = S + ws
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag < Ls:
                aff[:, diag] += ws * ws
            maff = np.max(aff, axis=1)

            if np.any(valid):
                scores[valid] = 2.0 * n * maff[valid] / (s_new[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            # Golden section refinement
            if 0 < imin < len(pts) - 1:
                lo = float(pts[max(0, imin - 2)])
                hi = float(pts[min(len(pts) - 1, imin + 2)])
            else:
                lo = 0.0
                hi = x_hi
            if hi - lo < 1e-14:
                lo = max(0.0, best_x - 1e-6)
                hi = best_x + 1e-6

            a_gs = lo
            b_gs = hi
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)
            f_c = eval_1d_scalar(c_gs - v, base, infl, diag, max_const, S, n)
            f_d = eval_1d_scalar(d_gs - v, base, infl, diag, max_const, S, n)
            for _ in range(22):
                if b_gs - a_gs < 1e-12 * max(1.0, abs(best_x)):
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = eval_1d_scalar(c_gs - v, base, infl, diag, max_const, S, n)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d

                    f_d = eval_1d_scalar(d_gs - v, base, infl, diag, max_const, S, n)
            x_mid = (a_gs + b_gs) / 2.0

                        aff_seg[diag] += w * w

                    except Exception:
                        pass

        else:

                    # Kick
                    kick_size = 0.02 + 0.04 * min(1.0, flat / 20.0)
                    p = p * np.exp(rng.normal(0.0, kick_size, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    # Final check
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s

        # Try a smooth "tent" initialization
        x = (np.arange(n_work, dtype=np.float64) + 0.5) / n_work
        tent = np.maximum(0.0, 1.0 - 2.0 * np.abs(x - 0.5))
        s_t = float(tent.sum())
        if s_t > 0:
            tent /= s_t
        # Blend with current best
        p_blend = 0.5 * tent + 0.5 * _resample(best_arr, n_work)
        p_blend = np.maximum(p_blend, 0.0)
        p_blend /= float(p_blend.sum())
        consider(p_blend)

        blend_end = min(hard - 5.0, time.monotonic() + rem * 0.6)
        if blend_end > time.monotonic() + 20.0:
            try:
                p_b, _ = _coord_descent(p_blend, blend_end, consider=consider, seed=int(seed + 8888))
                if p_b is not None and p_b.size > 0 and np.all(np.isfinite(p_b)) and p_b.sum() > 0:
                    consider(p_b)
            except Exception:
                pass

    # Phase 4: Final coordinate descent squeeze on the overall best
    rem = hard - 8.0 - time.monotonic()
    if rem > 30.0:
        p_final = _resample(best_arr, n_work)
        try:
            p_f, _ = _coord_descent(p_final, hard - 8.0, consider=consider, seed=int(seed + 9999))
            if p_f is not None and p_f.size > 0 and np.all(np.isfinite(p_f)) and p_f.sum() > 0:
                consider(p_f)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
