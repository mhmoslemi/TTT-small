= conv[2i] + 2vx - 2v^2 + x^2 - 2vx + v^2 = conv[2i] + x^2 - v^2. ✓

```python
w = x - v
conv[i:i+n] += 2.0 * w * a[:n]  # a[i] is now x
conv[2*i] -= w * w  # correction
```

Both have a w^2 correction at position 2i, but the sign differs because the base value of a[i] differs (v in evaluation, x in update).

Let me double-check the update formula one more time:
- After a[i] = x, a[:n] at position i is x.
- conv[i:i+n] += 2*w*a[:n] adds 2*w*x at position i (k=2i).
- We need the total delta at k=2i to be x^2-v^2 = w*(x+v).

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(400, dtype=np.float64)
    if p.size == 0:
        p = np.ones(400, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    return p


def _conv(a):
    n = len(a)
    if n <= 512:
        return np.convolve(a, a, mode='full')
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _eval(a):
    try:
        sc = float(evaluate_sequence([float(x) for x in a]))
        return sc
    except Exception:
        return math.inf


def _lp_direction(a, n, M, time_limit=6.0):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g


def _cd_routine(a0, stop_time, seed, max_sweeps=50000):
    """Fast coordinate descent with exact 1D search."""
    rng = np.random.default_rng(seed)
    a = np.clip(a0.copy(), 0, None)
    S = float(a.sum())
    n = len(a)
    if S < 0.01:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a / S
        S = 1.0

    conv = _conv(a)
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()

    # Precompute prefix/suffix max of conv for O(1) unaffected max query
    def _make_prefix_suffix(conv_arr):
        sz = len(conv_arr)
        pmx = np.empty(sz)
        smx = np.empty(sz)
        pmx[0] = conv_arr[0]
        for i in range(1, sz):
            pmx[i] = max(pmx[i - 1], conv_arr[i])
        smx[sz - 1] = conv_arr[sz - 1]
        for i in range(sz - 2, -1, -1):
            smx[i] = max(smx[i + 1], conv_arr[i])
        return pmx, smx

    no_improve = 0
    max_no_improve = 80
    sweep = 0

    while sweep < max_sweeps:
        if time.monotonic() > stop_time - 3:
            break

        improved = False
        order = rng.permutation(n)
        pmx, smx = _make_prefix_suffix(conv)

        for i in order:
            if time.monotonic() > stop_time - 3:
                break

            v = a[i]
            x_max = max(4.0 * v, 0.03) if v > 1e-8 else 0.1
            x_max = min(x_max, 50.0)

            # Determine search range more tightly
            # The max of conv is at some index. If we increase x, conv increases
            # in [i, i+n-1]. We want to find x that minimizes 2n*max/(S-v+x)^2
            n_coarse = 24
            x_grid = np.linspace(0.0, x_max, n_coarse)

            best_x = v
            best_obj = 2.0 * n * float(conv.max())  # current (S=1)

            for x in x_grid:
                S_new = S - v + x
                if S_new < 0.01:
                    continue
                w = x - v
                if abs(w) < 1e-15:
                    m_new = float(conv.max())
                else:
                    # Evaluate new max without modifying conv
                    # affected = conv[i:i+n] + 2*w*a[:n] + w^2 at position i
                    aff = conv[i:i + n] + 2.0 * w * a[:n]
                    aff[i] += w * w
                    m_aff = float(aff.max())
                    # Unaffected max
                    m_unaff = 0.0
                    if i > 0:
                        m_unaff = max(m_unaff, float(pmx[i - 1]))
                    if i + n < 2 * n - 1:
                        m_unaff = max(m_unaff, float(smx[i + n]))
                    m_new = max(m_aff, m_unaff)
                obj = 2.0 * n * m_new / (S_new * S_new)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_x = x

            # Fine search around best
            if best_x != v:
                dw = x_max / n_coarse
                x_lo = max(0.0, best_x - 2 * dw)
                x_hi = min(x_max, best_x + 2 * dw)
                x_fine = np.linspace(x_lo, x_hi, 20)
                for x in x_fine:
                    S_new = S - v + x
                    if S_new < 0.01:
                        continue
                    w = x - v
                    if abs(w) < 1e-15:
                        m_new = float(conv.max())
                    else:
                        aff = conv[i:i + n] + 2.0 * w * a[:n]
                        aff[i] += w * w
                        m_aff = float(aff.max())
                        m_unaff = 0.0
                        if i > 0:
                            m_unaff = max(m_unaff, float(pmx[i - 1]))
                        if i + n < 2 * n - 1:
                            m_unaff = max(m_unaff, float(smx[i + n]))
                        m_new = max(m_aff, m_unaff)
                    obj = 2.0 * n * m_new / (S_new * S_new)
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_x = x

            if best_x != v and best_obj < best_score - 1e-13:
                w = best_x - v
                # Commit update
                a[i] = best_x
                S += w
                conv[i:i + n] += 2.0 * w * a[:n]
                if 2 * i < 2 * n - 1:
                    conv[2 * i] -= w * w
                best_score = best_obj
                best_a = a.copy()
                improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                # Perturb a random subset
                n_pert = max(10, n // 8)
                idx = rng.choice(n, size=n_pert, replace=False)
                a[idx] = np.clip(a[idx] * (1.0 + rng.normal(0, 0.15, n_pert)), 0, None)
                a /= a.sum()
                S = 1.0
                conv = _conv(a)
                best_score = 2.0 * n * float(conv.max())
                no_improve = 0

        sweep += 1

    a = np.clip(a, 0, 1000)
    final_sc = _eval(a)
    if math.isfinite(final_sc) and final_sc < _score(best_a):
        return a, final_sc
    return best_a, best_score


def _lp_phase(a0, stop_time, seed):
    """LP-based direction finding with line search."""
    rng = np.random.default_rng(seed)
    a = np.clip(a0.copy(), 0, None)
    S = float(a.sum())
    n = len(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_score = _score(a)
    stall = 0
    max_stall = 20
    it = 0

    while time.monotonic() < stop_time - 8 and stall < max_stall:
        conv = _conv(a)
        M = float(conv.max())

        g = _lp_direction(a, n, M, time_limit=6.0)
        if g is None:
            stall += 3
            a = np.clip(a + rng.normal(0, 0.02, n), 0, None)
            a /= a.sum()
            continue

        g_norm = g / g.sum()
        dist = float(np.linalg.norm(g_norm - a))
        if dist < 1e-7:
            stall += 2
            a = np.clip(a + rng.normal(0, 0.015, n), 0, None)
            a /= a.sum()
            continue

        # Line search
        best_t = 0.0
        best_sc = _score(a)
        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
        for t in t_candidates:
            a_try = np.clip(a + t * (g_norm - a), 0, None)
            s_try = a_try.sum()
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            sc = _score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t > 0 and best_sc < best_score - 1e-12:
            a_new = np.clip(a + best_t * (g_norm - a), 0, None)
            a_new /= a_new.sum()
            a = a_new
            best_a = a.copy()
            best_score = _score(a)
            stall = 0
        else:
            stall += 1
            if stall >= 8:
                a = np.clip(a + rng.normal(0, 0.04, n), 0, None)
                a /= a.sum()
                stall = 0

        it += 1

    return best_a, best_score
```
