def _clean_seq(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    return np.clip(a, 0.0, 1000.0)


def _normalize(a, target=None):
    a = _clean_seq(a)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    if target is None:
        target = math.sqrt(2.0 * float(a.size))
    target = float(max(1e-8, target))
    s = float(a.sum())
    if s <= 1e-15:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    a = a * (target / s)
    return np.clip(a, 0.0, 1000.0)


def _conv_full(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_raw(a):
    a = _clean_seq(a)
    n = int(a.size)
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    if m < 0.0:
        m = 0.0
    return float(2.0 * n * m / (s * s))


def _active_indices(b, frac, max_active=None):
    M = float(b.max())
    if M <= 1e-18:
        return np.array([0], dtype=int)
    frac = float(min(max(frac, 0.0), 0.95))
    thresh = M * (1.0 - frac)
    idx = np.where(b >= thresh)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(b))], dtype=int)
    if max_active is not None and idx.size > max_active:
        max_active = int(max_active)
        if max_active > 0:
            sv = b[idx]
            order = np.argsort(sv)[::-1][:max_active]
            idx = idx[order]
    return idx


def _lp_direct_direction(f, b, active, timeout=1.2):
    n = int(f.size)
    if n == 0 or active.size == 0:
        return None, 0.0
    M = float(b.max())
    S = float(f.sum())
    if S <= 1e-15 or not math.isfinite(M):
        return None, 0.0

    m = int(active.size)
    A = np.zeros((m, n + 1), dtype=np.float64)
    rhs = np.zeros(m, dtype=np.float64)

    for r, k in enumerate(active):
        js = max(0, int(k) - n + 1)
        je = min(n, int(k) + 1)
        if js < je:
            idx = np.arange(js, je, dtype=np.int64)
            A[r, js:je] = 2.0 * f[k - idx]
        rhs[r] = max(0.0, float(M - b[k]))

    A[:, n] = 1.0
    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = -1.0

    ub = max(1.0, S)
    delta_ub = max(1.0, 2.0 * M + 1.0)
    bounds = [(max(0.0, -float(f[i])), ub) for i in range(n)]
    bounds.append((0.0, delta_ub))

    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, :n] = 1.0
    beq = np.zeros(1, dtype=np.float64)

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=rhs,
            A_eq=Aeq,
            b_eq=beq,
            bounds=bounds,
            method="highs",
            options={"time_limit": float(timeout), "presolve": True},
        )
    except Exception:
        return None, 0.0

    if res is None or getattr(res, "x", None) is None:
        return None, 0.0
    if int(res.status) not in (0, 3):
        return None, 0.0

    d = np.asarray(res.x[:n], dtype=np.float64).copy()
    delta = float(res.x[n])
    if not math.isfinite(delta):
        return None, 0.0
    if not np.all(np.isfinite(d)):
        d = np.where(np.isfinite(d), d, 0.0)

    tol = 1e-12 * max(1.0, M) + 1e-15
    if delta <= tol:
        return None, 0.0

    d -= float(d.mean())
    d = np.maximum(d, -f)
    d -= float(d.mean())
    d = np.maximum(d, -f)

    # A tiny sum imbalance is harmless because the line search re-normalizes.
    norm = float(np.max(np.abs(d))) if d.size else 0.0
    if not math.isfinite(norm) or norm <= 1e-14:
        return None, 0.0

    desired = 0.5 * max(1.0, S)
    if norm > desired:
        d *= desired / norm

    return d, max(0.0, delta)


def _line_search(f, d, base_score):
    n = int(f.size)
    if n == 0 or d is None:
        return None, math.inf
    S = float(f.sum())
    if S <= 1e-12:
        return None, math.inf
    d = np.asarray(d, dtype=np.float64)
    if not np.all(np.isfinite(d)):
        return None, math.inf

    best_sc = float(base_score)
    best_t = 0.0
    best_cand = None

    def _try_t(t):
        nonlocal best_sc, best_t, best_cand
        cand = f + float(t) * d
        cand = np.maximum(cand, 0.0)
        cs = float(cand.sum())
        if cs <= 1e-12:
            return
        if abs(cs - S) > 1e-8 * max(1.0, S):
            cand = cand * (S / cs)
        sc = _score_raw(cand)
        if math.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_t = float(t)
            best_cand = cand.copy()

    ts = (0.0, 0.02, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.65, 0.8, 0.9, 1.0)
    for t in ts:
        _try_t(t)

    if best_cand is None or best_t <= 0.0:
        return None, math.inf

    low = max(0.0, best_t - 0.15)
    high = min(1.0, best_t + 0.15)
    for t in np.linspace(low, high, 17):
        _try_t(float(t))

    if best_cand is None or best_sc >= float(base_score) - 1e-15:
        return None, math.inf
    return best_cand, best_sc


def _direct_lp_loop(f0, stop_time, seed, initial_frac=0.035):
    f0 = _clean_seq(f0)
    if f0.size == 0:
        return np.ones(1, dtype=np.float64), _score_raw(np.ones(1, dtype=np.float64))
    n = int(f0.size)
    S_target = math.sqrt(2.0 * n)
    f = _normalize(f0, S_target)
    b = _conv_full(f)
    M = float(b.max())
    S = float(f.sum())
    if S <= 1e-12 or not math.isfinite(M):
        best = f.copy()
        return best, _score_raw(best)

    best_sc = float(2.0 * n * M / (S * S))
    best = f.copy()
    rng = np.random.default_rng(seed)

    frac = float(max(0.005, min(0.5, initial_frac)))
    stall = 0
    max_stall = 60
    improvements = 0

    while time.monotonic() < stop_time - 1.0:
        b = _conv_full(f)
        M = float(b.max())
        S = float(f.sum())
        if S <= 1e-12 or not math.isfinite(M) or M <= 1e-18:
            break
        base_sc = float(2.0 * n * M / (S * S))
        if not math.isfinite(base_sc):
            break

        max_active = min(
            400,
            max(
                8,
                int((2 * n - 1) * min(0.35, max(0.02, 1.8 * frac))),
            ),
        )
        active = _active_indices(b, frac, max_active)
        d, delta = _lp_direct_direction(f, b, active, timeout=1.2)

        accepted = False
        if d is not None:
            cand, sc = _line_search(f, d, base_sc)
            if cand is not None and sc < base_sc - 1e-14:
                cand = _clean_seq(cand)
                cs = float(cand.sum())
                if cs > 1e-12:
                    cand *= S / cs
                cand = np.clip(cand, 0.0, 1000.0)
                f = cand
                if math.isfinite(sc) and sc < best_sc - 1e-14:
                    best = f.copy()
                    best_sc = sc
                stall = 0
                improvements += 1
                if improvements % 5 == 0 and frac > 0.01:
                    frac = max(0.008, frac * 0.85)
                elif frac < 0.12:
                    frac = min(0.18, frac * 1.08 + 0.002)
                accepted = True

        if not accepted:
            stall += 1
            if stall == 5 or stall == 15:
                frac = min(0.45, frac * 1.5 + 0.004)
            if stall >= max_stall:
                noise_scale = 0.025 + 0.015 * min(1.0, float(stall) / float(max_stall))
                noise = rng.normal(0.0, noise_scale * max(1.0, float(f.mean())), n)
                f = _normalize(np.clip(f + noise, 0.0, None), S)
                stall = 0
                frac = 0.03
                max_stall = min(max_stall + 30, 260)

    scf = _score_raw(f)
    if math.isfinite(scf) and scf < best_sc:
        best = f.copy()
        best_sc = scf
    return best, best_sc


def _subgrad_loop(f0, stop_time, seed):
    f0 = _clean_seq(f0)
    if f0.size == 0:
        return np.ones(1, dtype=np.float64), _score_raw(np.ones(1, dtype=np.float64))
    n = int(f0.size)
    S = math.sqrt(2.0 * n)
    f = _normalize(f0, S)

    b = _conv_full(f)
    sc0 = _score_raw(f)
    best = f.copy()
    best_sc = sc0
    rng = np.random.default_rng(seed)

    lr = 0.18
    frac = 0.015
    stall = 0
    max_stall = 120

    while time.monotonic() < stop_time - 1.0:
        b = _conv_full(f)
        M = float(b.max())
        if not math.isfinite(M) or M <= 1e-18:
            break

        active = _active_indices(b, frac, max_active=110)
        g = np.zeros(n, dtype=np.float64)
        for k in active:
            js = max(0, int(k) - n + 1)
            je = min(n, int(k) + 1)
            if js < je:
                idx = np.arange(js, je, dtype=np.int64)
                g[js:je] += 2.0 * f[k - idx]
        if active.size > 0:
            g /= float(active.size)

        gmax = float(g.max())
        if not math.isfinite(gmax) or gmax <= 1e-14:
            break

        cand_best = None
        cand_best_sc = best_sc
        for lr_try in (lr, 0.5 * lr, 1.6 * lr, 0.04, 0.09, 0.22, 0.38, 0.6):
            if lr_try <= 0.0:
                continue
            e = -lr_try * g / max(1e-15, gmax)
            e -= float(e.max())
            cand = f * np.exp(e)
            cand = _normalize(cand, S)
            sc = _score_raw(cand)
            if math.isfinite(sc) and sc < cand_best_sc - 1e-14:
                cand_best_sc = sc
                cand_best = cand

        if cand_best is not None:
            f = cand_best
            if cand_best_sc < best_sc - 1e-14:
                best = f.copy()
                best_sc = cand_best_sc
            stall = 0
            lr = max(0.03, lr * 0.99)
        else:
            stall += 1
            lr = max(0.025, lr * 0.88)
            if stall == 40:
                frac = 0.035
            if stall >= max_stall:
                noise = rng.normal(0.0, 0.012 * max(1.0, float(f.mean())), n)
                f = _normalize(np.clip(f + noise, 0.0, None), S)
                stall = 0
                lr = 0.15
                frac = 0.02
                max_stall = min(max_stall + 50, 450)

    scf = _score_raw(f)
    if math.isfinite(scf) and scf < best_sc:
        best = f.copy()
        best_sc = scf
    return best, best_sc


def _coordinate_descent(f0, stop_time, seed):
    f0 = _clean_seq(f0)
    if f0.size == 0:
        return np.ones(1, dtype=np.float64), _score_raw(np.ones(1, dtype=np.float64))
    n = int(f0.size)
    S = math.sqrt(2.0 * n)
    f = _normalize(f0, S)
    b = _conv_full(f)
    sum_a = float(f.sum())
    cur_score = _score_raw(f)

    best = f.copy()
    best_exact = cur_score
    rng = np.random.default_rng(seed)

    aff = []
    for i in range(n):
        start = i
        stop = min(2 * n - 1, i + n)
        ks = np.arange(start, stop, dtype=np.int64)
        partners = ks - i
        diag_mask = partners == i
        diag_pos = int(np.argmax(diag_mask)) if np.any(diag_mask) else -1
        aff.append((ks, partners, diag_pos))

    no_imp = 0
    max_no_imp = 3

    while time.monotonic() < stop_time - 1.0:
        b = _conv_full(f)
        sum_a = float(f.sum())
        if sum_a <= 1e-12:
            break
        mx = float(b.max())
        cur_score = float(2.0 * n * mx / (sum_a * sum_a))
        if not math.isfinite(cur_score):
            break

        pass_improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break

            ks, partners, diag_pos = aff[i]
            orig = float(f[i])
            base = sum_a / float(n)

            xs = []
            if orig > 1e-14:
                for mlt in _COORD_MULTS:
                    xs.append(orig * mlt)
            else:
                xs.extend([0.0, base * 0.01, base * 0.05, base * 0.2, base, 2.0 * base])

            xs.extend([0.05 * S, 0.2 * S, 0.5 * S, S])

            best_x = orig
            best_local = cur_score

            best_before = best_local
            for x in xs:
                if not math.isfinite(x) or x < 0.0:
                    continue
                if x > 1000.0:
                    x = 1000.0
                if abs(x - orig) < 1e-15:
                    continue
                d = float(x - orig)
                new_sum = float(sum_a + d)
                if new_sum < 0.01:
                    continue

                deltas = (2.0 * d) * f[partners]
                if diag_pos >= 0:
                    deltas[diag_pos] = x * x - orig * orig

                np.add(b, deltas, out=b)
                mxv = float(b.max())
                np.subtract(b, deltas, out=b)

                if not math.isfinite(mxv):
                    continue
                sc = float(2.0 * n * max(0.0, mxv) / (new_sum * new_sum))
                if math.isfinite(sc) and sc < best_local - 1e-15:
                    best_local = sc
                    best_x = float(x)

            if best_x != orig and best_local < best_before - 1e-15:
                d = float(best_x - orig)
                new_sum = float(sum_a + d)
                deltas = (2.0 * d) * f[partners]
                if diag_pos >= 0:
                    deltas[diag_pos] = best_x * best_x - orig * orig
                np.add(b, deltas, out=b)
                f[i] = best_x
                sum_a = new_sum
                cur_score = best_local
                pass_improved = True

        if not pass_improved:
            no_imp += 1
            if no_imp >= max_no_imp:
                break
        else:
            no_imp = 0

        # Renormalize after each pass; score is scale invariant.
        if sum_a > 1e-12:
            factor = S / sum_a
            if abs(factor - 1.0) > 1e-5:
                f *= factor
                b *= factor * factor
                sum_a = S
                cur_score = float(2.0 * n * float(b.max()) / (sum_a * sum_a))

        if pass_improved:
            exact_cur = _score_raw(f)
            if math.isfinite(exact_cur) and exact_cur < best_exact - 1e-12:
                best = f.copy()
                best_exact = exact_cur

    exact_best = _score_raw(best)
    if math.isfinite(exact_best):
        return best, exact_best
    return best, best_exact


def _try_length(seq, length, stop_time, seed):
    seq = _clean_seq(seq)
    n0 = int(seq.size)
    L = int(length)
    if n0 <= 1 or L < max(10, n0 // 4) or L > 1000:
        return seq
    if time.monotonic() >= stop_time - 8.0:
        return seq

    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, L)
    init = np.interp(x1, x0, seq)
    init = np.clip(init, 0.0, None)
    if float(init.sum()) <= 1e-12:
        init = np.ones(L, dtype=np.float64)

    best = init.copy()

    if time.monotonic() < stop_time - 18.0:
        lp_stop = min(stop_time - 18.0, time.monotonic() + 12.0)
        try:
            f1, _ = _direct_lp_loop(init, lp_stop, seed + 1000 + L, initial_frac=0.04)
            best = f1
        except Exception:
            pass

    if time.monotonic() < stop_time - 5.0:
        cd_stop = min(stop_time - 4.0, time.monotonic() + 9.0)
        try:
            f2, _ = _coordinate_descent(best, cd_stop, seed + 9000 + L)
            best = f2
        except Exception:
            pass

    return _clean_seq(best)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    reserve = max(5.0, 0.02 * budget)
    deadline = t0 + budget - reserve

    parent = _get_parent()
    best_seq = _clean_seq(parent)
    if best_seq.size == 0 or float(best_seq.sum()) < 0.01:
        best_seq = np.ones(416, dtype=np.float64)

    best_score = _score_raw(best_seq)
    if not math.isfinite(best_score):
        best_seq = np.ones(parent.size if parent.size > 0 else 416, dtype=np.float64)
        best_score = _score_raw(best_seq)

    def _update_best(candidate):
        nonlocal best_seq, best_score
        if candidate is None:
            return
        cand = _clean_seq(candidate)
        if cand.size == 0 or float(cand.sum()) < 0.01:
            return
        sc = _score_raw(cand)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_seq = cand.copy()
            best_score = sc

    np.random.seed(seed)

    # Main LP-guided descent from the parent.
    stop1 = min(deadline - 300.0, t0 + 0.38 * budget)
    if time.monotonic() < stop1 - 5.0:
        try:
            f1, _ = _direct_lp_loop(best_seq, stop1, seed, initial_frac=0.035)
            _update_best(f1)
        except Exception:
            pass

    # Continue on the best sequence with a tighter active set.
    stop2 = min(deadline - 250.0, t0 + 0.49 * budget)
    if time.monotonic() < stop2 - 10.0:
        try:
            f2, _ = _direct_lp_loop(best_seq, stop2, seed + 1234567, initial_frac=0.02)
            _update_best(f2)
        except Exception:
            pass

    # A short perturbed LP run to escape a stick point.
    if time.monotonic() < stop2 - 3.0:
        try:
            base = _clean_seq(best_seq)
            if base.size > 0:
                noise = np.random.default_rng(seed + 9).normal(
                    0.0, 0.012 * max(1.0, float(base.mean())), base.size
                )
                pert = _normalize(np.clip(base + noise, 0.0, None), math.sqrt(2.0 * base.size))
                f3, _ = _direct_lp_loop(pert, stop2, seed + 7654321, initial_frac=0.04)
                _update_best(f3)
        except Exception:
            pass

    # Subgradient / multiplicative smoothing.
    stop3 = min(deadline - 180.0, t0 + 0.62 * budget)
    if time.monotonic() < stop3 - 10.0:
        try:
            f4, _ = _subgrad_loop(best_seq, stop3, seed + 55555)
            _update_best(f4)
        except Exception:
            pass

    # Coordinate descent polish.
    stop4 = min(deadline - 120.0, t0 + 0.85 * budget)
    if time.monotonic() < stop4 - 10.0:
        try:
            f5, _ = _coordinate_descent(best_seq, stop4, seed + 33333)
            _update_best(f5)
        except Exception:
            pass

    # One more short LP pass after polishing.
    stop5 = min(deadline - 80.0, t0 + 0.91 * budget)
    if time.monotonic() < stop5 - 10.0:
        try:
            f6, _ = _direct_lp_loop(best_seq, stop5, seed + 444444, initial_frac=0.015)
            _update_best(f6)
        except Exception:
            pass

    # Nearby lengths can occasionally help.
    n_base = int(best_seq.size)
    lens = []
    for L in (n_base + 8, n_base - 8, n_base + 16, n_base - 16,
              n_base + 32, n_base - 32, n_base + 48, n_base - 48):
        if 100 <= L <= 700 and int(L) not in lens:
            lens.append(int(L))

    for L in lens[:4]:
        if time.monotonic() > deadline - 35.0:
            break
        try:
            cand = _try_length(
                best_seq,
                L,
                min(deadline - 25.0, time.monotonic() + 22.0),
                seed + 100000 + L,
            )
            _update_best(cand)
        except Exception:
            pass

    # Final coordinate polish with whatever time remains.
    if time.monotonic() < deadline - 5.0:
        try:
            f7, _ = _coordinate_descent(best_seq, deadline - 3.0, seed + 987654)
            _update_best(f7)
        except Exception:
            pass

    # Final exact validation against the authoritative evaluator.
    final = _clean_seq(best_seq)
    try:
        exact = float(evaluate_sequence([float(x) for x in final]))
    except Exception:
        exact = _score_raw(final)

    if not math.isfinite(exact) or exact >= best_score - 1e-12:
        fallback = _clean_seq(parent)
        if fallback.size > 0 and float(fallback.sum()) >= 0.01:
            final = fallback
        else:
            final = np.ones(416, dtype=np.float64)

    final = _clean_seq(final)
    if final.size == 0 or float(final.sum()) < 0.01:
        final = np.ones(416, dtype=np.float64)

    return [float(x) for x in final]
