def _lp_direction_full(f, n, M, timeout=10.0):

    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):

                rows.append(k)
                cols.append(i)
                vals.append(v)

def _mirror_descent(f0, stop_time, seed):
    n = len(f0)
    rng = np.random.default_rng(seed)
    f = np.clip(f0.copy(), 1e-10, None)
    f = f / f.sum()
    best_f = f.copy()
    best_score = _score(f)
    
    lr = 1.0
    for it in range(100000):
        if time.monotonic() > stop_time:
            break
        
        b = _conv_full(f)
        M = b.max()
        # Find active peaks
        peak_idx = np.where(b >= 0.999 * M)[0]
        
        # Gradient of max(b) ≈ average gradient over near-peak lags
        grad = np.zeros(n)
        for k in peak_idx:
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                grad[i] += f[k - i]
        grad /= len(peak_idx)
        
        # Multiplicative update
        # f_new[i] = f[i] * exp(-lr * grad[i]) / sum(...)
        exponent = -lr * grad
        exponent -= exponent.max()
        weights = np.exp(exponent)
        f_new = f * weights
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new /= s
        
        sc = _score(f_new)
        if sc < best_score - 1e-14:
            best_f = f_new.copy()
            best_score = sc
            lr *= 0.999
        else:
            # Damped step
            for alpha in [0.5, 0.3, 0.1]:
                f_damp = (1 - alpha) * f + alpha * f_new
                sc_d = _score(f_damp)
                if sc_d < best_score - 1e-14:
                    f_new = f_damp
                    best_f = f_new.copy()
                    best_score = sc_d
                    break
            lr *= 0.99
        
        f = f_new
    
    return best_f, best_score

    return 2.0 * n * m / (s * s)


def _build_lp_matrix(f, n, use_all=True, peak_frac=1.0):
    """Build the LP constraint matrix A where (A@g)[k] = (f*g)[k]"""
    b = _conv_full(f)
    M = float(b.max())
    
    if use_all or peak_frac >= 1.0:
        peak_idx = np.arange(2 * n - 1)
    else:
        threshold = M * (1.0 - peak_frac)
        peak_idx = np.where(b >= threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(peak_idx):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if abs(v) > 1e-20:
                rows.append(idx)
                cols.append(i)
                vals.append(v)
    
    if not rows:
        return None, None, len(peak_idx)
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    return A, peak_idx, len(peak_idx)


def _lp_direction(f, n, b, use_all=True, timeout=8.0):
    """LP: maximize sum(g) subject to f*g <= M (selected positions), g >= 0"""
    M = float(b.max())
    if M < 1e-18:
        return None
    
    A, _, ncons = _build_lp_matrix(f, n, use_all=use_all)
    if A is None:
        return None

    b_ub = np.full(ncons, M)
