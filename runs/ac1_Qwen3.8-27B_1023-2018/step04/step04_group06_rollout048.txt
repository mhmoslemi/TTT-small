from scipy.optimize import linprog


def _score_fast(a):
    """Fast score computation matching the exact evaluator."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')

    return 2.0 * len(a) * float(c.max()) / (s * s)


def _get_parent():

def _build_A(n, a):
    """Build (2n-1) x n matrix: (A @ g)_k = (a * g)_k."""
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    d = k_idx - j_idx
    mask = (d >= 0) & (d < n)

    A[mask] = a[d[mask]]
    return A


def _solve_lp_fast(A, M, n, tl=8.0):
    """max 1^T g s.t. A g <= M, g >= 0 via HiGHS."""
    c_obj = -np.ones(n)
    b_ub = np.full(A.shape[0], M, dtype=np.float64)
    try:
        res = linprog(
            c_obj, A_ub=A, b_ub=b_ub,

    return np.clip(res.x, 0.0, None)

    a_n = a / a.sum()
    g_n = g / g.sum()

        return float(np.convolve(c, c).max())

    best_t = 0.0

def _make_initializations(n, pn, rng):

    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 4.0]:

    # Exponential decay
    for al in [0.5, 1.0, 2.0, 3.0, 5.0, 8.0]:

        inits.append((1 - frac) * pn + frac * base)
