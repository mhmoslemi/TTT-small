import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(198, dtype=np.float64)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(198, dtype=np.float64)
    return p

def _conv_full(a):
    n = len(a)
    if n <= 600:
        return np.convolve(a, a, mode="full")
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_direction(f, n, M, timeout=8.0):
    if M <= 1e-18:
        return None
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
            options={"presolve": True, "time_limit": timeout,
                     "dual_feasibility_tolerance": 1e-12,
                     "primal_feasibility_tolerance": 1e-12}
        )
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g

def _lp_optimize(f0, n, stop_time, seed, verbose=False):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    f = f[:n]
    f_sum = float(f.sum())
    if f_sum < 1e-10:
        f = np.ones(n)
        f_sum = n

    target_sum = math.sqrt(2.0 * n)
    f = f / f_sum * target_sum

    best_f = f.copy()
    best_score = _score(f)

    t = 0.3
    t_min = 0.01
    t_max = 1.0
    stall = 0
    max_stall = 100
    iter_count = 0

    while time.monotonic() < stop_time - 5.0:
        iter_count += 1
        b = _conv_full(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, n, M)
        if g is None:
            stall += 3
            if stall > max_stall:
                # Restart with perturbation
                f = best_f.copy()
                idx = rng.choice(n, size=min(20, n), replace=False)
                f[idx] = np.maximum(0, f[idx] + rng.normal(0, 0.1 * (f.mean() + 1), len(idx)))
                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                stall = 0
                max_stall = int(max_stall * 1.3)
                t = 0.3
            continue

        stall = 0
        max_stall = max(50, int(0.9 * max_stall))
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            continue
        g = g * (target_sum / g_sum)

        # Line search
        improved = False
        for tc in [t, t*0.5, t*0.75, t*1.5, 0.1, 0.3, 0.5, 1.0]:
            tc = max(t_min, min(t_max, tc))
            f_new = (1.0 - tc) * f + tc * g
            f_new = np.clip(f_new, 0, None)
            fs_new = float(f_new.sum())
            if fs_new < 0.01:
                continue
            f_new = f_new / fs_new * target_sum
            sc = _score(f_new)
            if sc < best_score - 1e-14:
                f = f_new
                best_f = f.copy()
                best_score = sc
                improved = True
                t = min(t_max, tc * 1.2)
                break
            elif sc < _score(f) + 1e-10:
                # Accept even if not globally better (to escape)
                f = f_new
                t = max(t_min, tc * 0.8)
        
        if not improved:
            t = max(t_min, t * 0.6)
            # Check convergence
            dist = np.linalg.norm(g - f) / max(1.0, np.linalg.norm(f))
            if dist < 1e-9:

                fs = f.sum()
                if fs > 1e-10:
                    f = f / fs * target_sum
                t = 0.3
                max_stall = 50

    return best_f, best_score

Actually wait, I want to double-check the LP formulation. The constraint is `(f*g)_k <= M` for all k. Here `(f*g)_k = sum_j f[k-j]*g[j]` (the convolution of f and g, not f with itself). This is a LINEAR constraint in g (for fixed f). The objective is to maximize `sum(g)`, also linear. So it IS a valid LP.

The idea: we want to find a nonnegative `g` with large total mass whose convolution with `f` doesn't exceed the current peak `M`. When we interpolate `f_new = (1-t)f + t*g`, the new peak should be:
`max(f_new * f_new) <= (1-t)^2 * M + 2t(1-t) * M + t^2 * max(g*g)`
`= ((1-t)^2 + 2t(1-t)) * M + t^2 * max(g*g)`
`= (1-t^2) * M + t^2 * max(g*g)`

For this to be less than M, we need `t^2 * max(g*g) < t^2 * M`, i.e., `max(g*g) < M`. But we don't control max(g*g) directly!

Hmm, so the LP constraint ensures `f*g <= M` but not `g*g <= M`. For small t, the `t^2 * max(g*g)` term is small, so the peak decreases. But for large t, it might increase.

This is why the line search is important: try different values of t and accept only if the score actually decreases.
