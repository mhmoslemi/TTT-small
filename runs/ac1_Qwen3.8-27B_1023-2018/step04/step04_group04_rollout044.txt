    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def _lp_direction(f, n, M, timeout=10.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    # conv(f, g)_k = sum_j f_j * g_{k-j} (with appropriate bounds)
    # This is a linear function of g.
    # Build the constraint matrix A where (Ag)_k = conv(f, g)_k
    # For k = 0, ..., 2n-2:
    # (Ag)_k = sum_{j: 0<=j<n, 0<=k-j<n} f_j * g_{k-j}
    #        = sum_{j: max(0,k-n+1) <= j <= min(k,n-1)} f_j * g_{k-j}
    # Let i = k-j, so j = k-i:
    #        = sum_{i: max(0,n-1-k) <= i <= min(k,n-1)} f_{k-i} * g_i
    
    # Actually, let's just build it as: for each variable g_i (i=0..n-1),
    # it appears in constraints k = i, i+1, ..., i+n-1 (since conv(f,g)_k has g_i when j = k-i)
    # Specifically, conv(f,g)_k = sum_j f_j * g_{k-j}
    # g_i appears when k-j = i, i.e., k = j+i, with coefficient f_j.
    # So for variable g_i, it appears in constraints k = i, i+1, ..., i+n-1 with coefficients f_0, f_1, ..., f_{n-1}.
    
    # A[k, i] = f_{k-i} if 0 <= k-i < n, else 0
    # But we need A[k, i] such that (A @ g)[k] = conv(f, g)[k]
    # conv(f, g)[k] = sum_j f[j] * g[k-j] where 0<=j<n and 0<=k-j<n
    # = sum_i g[i] * f[k-i] where 0<=i<n and 0<=k-i<n
    # So A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    
    # For n=414, A is 827 x 414. Dense but manageable (~340K floats = 2.7 MB)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for i in range(n):
        # g[i] appears in constraints k = i to i+n-1
        # A[k, i] = f[k-i]
        k_lo = i
        k_hi = min(i + n - 1, 2*n - 2)
        A[k_lo:k_hi+1, i] = f[0:k_hi-k_lo+1]
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout, 'mip_rel_gap': 0})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _lp_optimize(f0, n, stop_time, seed, verbose=False):
    """LP-based optimization with improved line search."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs  # normalize to sum = 1
    
    best_f = f.copy()
    best_sc = _score(f)
    
    t_param = 1.0  # start with full step
    stall = 0
    max_stall = 200
    
    for it in range(100000):
        if time.monotonic() > stop_time:
            break
        
        b = _fft_conv(f)
        M = float(b.max())
        if M < 1e-18:
            break
        
        g = _lp_direction(f, n, M, timeout=8.0)
        if g is None:
            stall += 5
            if stall > max_stall:
                break
            # Perturb
            f = _proj_simplex(f + rng.normal(0, 0.01, n))
            continue
        
        gs = g.sum()
        if gs < 1e-15:
            stall += 3
            continue
        g = g / gs  # normalize to sum = 1
        
        # Check distance
        dist = np.linalg.norm(g - f)
        if dist < 1e-12:
            stall += 1
            if stall > max_stall:
                break
            continue
        
        # Line search: try various t values
        # Key: try t=1 first (jump to g), then binary search
        candidates_t = [1.0, t_param, 0.5, 0.75, 0.25, 0.9, 0.6, 0.8, 0.4, 0.3]
        candidates_t = sorted(set(min(1.0, max(0.0, t)) for t in candidates_t))
        
        best_t = None
        best_sc_try = best_sc
        
        for tt in candidates_t:
            fn = (1 - tt) * f + tt * g
            fn = np.clip(fn, 0, None)
            fn_sum = fn.sum()
            if fn_sum < 0.01:
                continue
            fn = fn / fn_sum
            sc_try = _score(fn)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_t = tt
        
        if best_t is not None and best_sc_try < best_sc - 1e-14:
            fn = (1 - best_t) * f + best_t * g
            fn = np.clip(fn, 0, None)
            f = fn / fn.sum()
            best_f = f.copy()
            best_sc = _score(f)
            t_param = best_t  # remember good step
            stall = 0
        else:
            # Try smaller steps
            found = False
            for tt in [0.1, 0.05, 0.02, 0.01, 0.005]:
                fn = (1 - tt) * f + tt * g
                fn = np.clip(fn, 0, None)
                if fn.sum() < 0.01:
                    continue
                fn = fn / fn.sum()
                sc_try = _score(fn)
                if sc_try < best_sc - 1e-14:
                    f = fn
                    best_f = f.copy()
                    best_sc = sc_try
                    t_param = tt
                    stall = 0
                    found = True
                    break
            if not found:
                stall += 1
                t_param = max(0.005, t_param * 0.5)
                if stall >= max_stall:
                    # Restart with perturbation
                    f = _proj_simplex(f + rng.normal(0, 0.02, n))
                    t_param = 0.5
                    stall = 0
                    max_stall = min(500, max_stall + 100)
    
    return best_f, best_sc

def _subgrad_optimize(f0, n, stop_time, seed):
    """Projected subgradient on simplex."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / max(f.sum(), 1e-10)
    
    best_f = f.copy()
    best_sc = _score(f)
    
    # Lower bound: 2n/(2n-1) (from average of convolution)
    f_lower = 2.0 * n / (2.0 * n - 1.0)
    
    step = 0.1
    no_imp = 0
    best_sc_window = best_sc
    window = []
    
    velocity = np.zeros(n)
    mu = 0.5
    
    for it in range(200000):
        if time.monotonic() > stop_time:
            break
        
        sg, M, b = _subgrad(f)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-18:
            break
        
        # Polyak-like step: (f(a) - f*) / ||sg||^2
        # f(a) = 2n * M (since sum=1)
        f_current = 2.0 * n * M
        step = (f_current - f_lower) / (sg_norm * sg_norm + 1e-15)
        step = min(step, 0.5)
        
        velocity = mu * velocity - step * sg
        f_trial = _proj_simplex(f + velocity)
        
        sc = _score(f_trial)
        window.append(sc)
        if len(window) > 100:
            window.pop(0)
        
        if sc < best_sc - 1e-13:
            f = f_trial
            best_f = f.copy()
            best_sc = sc
            best_sc_window = min(best_sc_window, sc)
            no_imp = 0
        else:
            f = f_trial  # still move (for convex problems, this is OK)
            no_imp += 1
            if no_imp > 2000:
                # Restart
                f = _proj_simplex(f + rng.normal(0, 0.01, n))
                velocity = np.zeros(n)
                no_imp = 0
                if best_sc > 1.6:  # not good enough, more aggressive restart
                    f = _proj_simplex(best_f + rng.normal(0, 0.03, n))
    
    return best_f, best_sc

def _active_set_method(f0, n, stop_time, seed):
    """
    Active set SQP-like method for min max_k (a^T M_k a) s.t. sum(a)=1, a>=0.
    At each step, identify active constraints, solve QP subproblem.
    """
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / max(f.sum(), 1e-10)
    
    best_f = f.copy()
    best_sc = _score(f)
    M = 2.0 * n * f.sum()**-2  # will compute properly
    
    for it in range(5000):
        if time.monotonic() > stop_time:
            break
        
        b = _fft_conv(f)
        M_val = float(b.max())
        obj_val = 2.0 * n * M_val / (f.sum()**2)
        
        if obj_val < best_sc - 1e-14:
            best_sc = obj_val
            best_f = f.copy()
        
        # Identify active set
        eps = 1e-10 * M_val
        active = np.flatnonzero(b >= M_val - eps)
        
        if len(active) == 0:
            break
        if len(active) > n * 2:
            # Too many active constraints, fall back to subgradient
            active = np.flatnonzero(b >= M_val - eps * 10)
        
        # Compute gradient of active objective
        # grad = (1/|A|) sum_{k in A} 2 * shift_k(a)
        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            if j_lo <= j_hi:
                idx = k - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi+1] += 2.0 * f[idx]
        grad /= len(active)
        
        # Hessian approximation: use identity * lambda (diagonal)
        # Or use the actual Hessian of the active constraints
        # For simplicity, use steepest descent with line search
        # Normalize gradient to be orthogonal to the constraint sum(a)=1
        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Line search
        best_t = None
        best_sc_ls = best_sc
        for tt in [0.05, 0.1, 0.2, 0.3, 0.5, 0.8]:
            fn = f - tt * grad / grad_norm
            fn = _proj_simplex(fn)
            sc = _score(fn)
            if sc < best_sc_ls:
                best_sc_ls = sc
                best_t = tt
        
        if best_t is not None:
            fn = f - best_t * grad / grad_norm
            f = _proj_simplex(fn)
        else:
            # Small step
            fn = f - 0.01 * grad / grad_norm
            f = _proj_simplex(fn)
            if _score(f) > best_sc + 1e-10:
                f = best_f.copy()
                break
    
    return best_f, best_sc
