Let me also think about the LSE optimization more carefully.

The LSE (log-sum-exp) function is:
$$\text{LSE}(c, T) = T \log \sum_k \exp(c_k / T)$$

which approximates $\max_k c_k$ as $T \to 0$. The gradient with respect to $a$ is:
$$\nabla_a \text{LSE}(c, T) = \sum_k \frac{\exp(c_k / T)}{\sum_j \exp(c_j / T)} \nabla_a c_k$$

Since $c_k = (a*a)_k = \sum_i a_i a_{k-i}$, we have $\nabla_a c_k = 2 a_{\cdot - k}$ (the $k$-th cross-correlation of $a$ with itself).

So the gradient is $\nabla_a \text{LSE} = 2 \sum_k w_k a_{\cdot - k}$ where $w_k = \exp(c_k/T) / \sum_j \exp(c_j/T)$.

This is exactly what the current code computes as `q = xfast(pp, w)` and `gdp = 2 * q`.

import numpy as np
import time
import math

from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n <= 512:
        c = np.convolve(a, a, mode="full")
    else:
        L = 2 * n - 1
        size = next_fast_len(L)
        F = rfft(a, size)
        c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    return float(2.0 * a.size * m / (s * s))


def _lse_optimize(p0, stop_time, consider=None):
    """LSE-smoothed optimization."""
    p = np.maximum(np.asarray(p0, dtype=np.float64).ravel(), 0.0).copy()
    n = p.size
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c
    
    def xfast(pp, ww):
        F = rfft(pp[::-1], conv_size)
        G = rfft(ww, conv_size)
        c = irfft(F * G, conv_size)[:conv_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q
    
    u = np.log(np.maximum(p, 1e-300))
    u -= float(np.mean(u))
    
    best_p = p.copy()
    best_score = _score_fft(p)
    
    stages = [(0.12, 0.10), (0.04, 0.15), (0.01, 0.20), (0.003, 0.25), (0.0008, 0.30)]
    total_w = sum(w for _, w in stages)
    start = time.monotonic()
    avail = max(0, stop_time - start - 10.0)
    if avail < 20:
        return best_p
    
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 8:
            break
        cum += wt
        end_stage = min(stop_time - 5, start + avail * (cum / total_w))
        if end_stage <= now + 5:
            continue
        
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0

            break

            pp = _softmax(uu)
            c = cfast(pp)
            if c.size == 0:
                return 1e100, np.zeros_like(uu)
            cm = float(c.max())
            scaled = (c - cm) / T
            exps = np.exp(scaled)

                w = np.zeros(c.size)
                w[int(np.argmax(c))] = 1.0
                f = cm
            else:
                f = cm + T * math.log(sum_exp / c.size)
                w = exps / sum_exp
            q = xfast(pp, w)
            gdp = 2.0 * q
            tdp = float(np.dot(pp, gdp))
            grad = pp * (gdp - tdp)
            grad = np.where(np.isfinite(grad), grad, 0.0)
            return float(f), grad
        
        maxit = max(15, min(50, 300000 // n))

        stale = 0

            f0, _ = lse_func(u)
        except:
            f0 = math.inf
        
        while time.monotonic() < end_stage - 2:
            try:
                res = minimize(lse_func, u, method="L-BFGS-B", jac=True, options=opts)
            except:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fval = float(res.fun)

                    stale = 0

            p = p_cur.copy()
            if consider:

        else:
            p = best_p.copy()
