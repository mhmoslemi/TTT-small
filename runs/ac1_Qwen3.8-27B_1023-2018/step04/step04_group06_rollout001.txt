The objective is to minimize max(a*a) subject to sum(a) = 1, a >= 0.

The subgradient of max(a*a) w.r.t. a at the point where k* is the argmax is:
∂(max)/∂a[j] = 2*a[k*-j] (if 0 <= k*-j < n), else 0

To minimize this subject to sum(a) = 1, we project the gradient to the zero-sum subspace:
grad_proj[j] = 2*a[k*-j] (if valid) - mean(2*a[k*-j] (if valid))

Then take a step: a_new = a - α * grad_proj, project to a >= 0, renormalize.

But this only works if we're trying to decrease the specific convolution value at k*. The issue is that this might increase the value at another index. So we need to be careful.

Actually, a better approach: minimize the sum of top-K convolution values. Or use the smooth approximation:
f_ε(a) = (1/K) * sum_{k in top-K} (a*a)[k]

where K is the number of indices within, say, 99% of the max.

The gradient of f_ε is:
∂f_ε/∂a[j] = (2/K) * sum_{k in top-K} a[k-j] (if valid)

This is a smoother objective and the gradient descent should work better.

import numpy as np
import time
from scipy.optimize import linprog


def _sc(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * a.size * float(np.convolve(a, a).max()) / (s * s)


def _par():
    try:
        p = np.asarray(globals()['height_sequence_1'], np.float64).ravel()
        p = np.clip(p, 0, 1000)
        if p.size and np.all(np.isfinite(p)) and p.sum() > 0.01:
            return p
    except Exception:
        pass
    return np.ones(201)


def _n1(a):
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    return a / s if s > 1e-15 else a


def _Afull(n, a):
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    m = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n))
    A[m] = a[d[m]]
    return A


def _lpcg(a, M, n, ks=None, tl=2.5, mrnd=18):
    co = -np.ones(n)
    if ks is None:
        ks = int(np.convolve(a, a).argmax())
    act = [ks]
    for _ in range(mrnd):
        A = np.zeros((len(act), n))
        for i, kk in enumerate(act):
            js, je = max(0, kk - n + 1), min(n, kk + 1)
            if je > js:
                A[i, js:je] = a[kk - np.arange(js, je)]
        try:
            r = linprog(co, A_ub=A, b_ub=np.full(len(act), M),
                        bounds=[(0, None)] * n, method='highs',
                        options={'time_limit': tl})
        except Exception:
            return None
        if r.status != 0 or r.x is None:
            return None
        g = np.clip(r.x, 0, None)
        cv = np.convolve(a, g)
        v = np.where(cv > M * (1 + 1e-8))[0]
        if len(v) == 0:
            return g
        act = sorted(set(act) | set(v.tolist()))
        if len(act) > 80:
            try:
                Af = _Afull(n, a)
                r = linprog(co, A_ub=Af, b_ub=np.full(2 * n - 1, M),
                            bounds=[(0, None)] * n, method='highs',
                            options={'time_limit': tl + 2.0})
            except Exception:
                return None
            if r.status != 0:
                return None
            return np.clip(r.x, 0, None)
    return None


def _lpsolve(a, M, n, ks, tl=3.0):
    Af = _Afull(n, a)
    co = -np.ones(n)
    try:
        r = linprog(co, A_ub=Af, b_ub=np.full(2 * n - 1, M),
                    bounds=[(0, None)] * n, method='highs',
                    options={'time_limit': tl})
    except Exception:
        return None
    if r.status != 0 or r.x is None:
        return None
    return np.clip(r.x, 0, None)


def _ls(a, g, M, n):
    bt, bv = 0.0, M
    for t in np.linspace(0.04, 1.0, 16):
        c = np.maximum((1 - t) * a + t * g, 0)
        cs = c.sum()
        if cs < 1e-15:
            continue
        c /= cs
        v = float(np.convolve(c, c).max())
        if v < bv:
            bv, bt = v, t
    if bt == 0:
        return None, M
    lo, hi = max(0.0, bt - 0.04), min(1.0, bt + 0.04)
    gr = (1.0 + 5.0 ** 0.5) / 2.0
    for _ in range(28):
        if hi - lo < 1e-9:
            break
        c1 = hi - (hi - lo) / gr
        c2 = lo + (hi - lo) / gr
        x1 = np.maximum((1 - c1) * a + c1 * g, 0)
        x1 /= max(x1.sum(), 1e-15)
        x2 = np.maximum((1 - c2) * a + c2 * g, 0)
        x2 /= max(x2.sum(), 1e-15)
        f1 = float(np.convolve(x1, x1).max())
        f2 = float(np.convolve(x2, x2).max())
        if f1 < f2:
            hi = c2
        else:
            lo = c1
        if f1 < bv:
            bv, bt = f1, c1
        if f2 < bv:
            bv, bt = f2, c2
    c = np.maximum((1 - bt) * a + bt * g, 0)
    c /= max(c.sum(), 1e-15)
    return c, bv


def _grad(a, ks, n):
    gr = np.zeros(n)
    js, je = max(0, ks - n + 1), min(n, ks + 1)
    for j in range(js, je):
        idx = ks - j
        if 0 <= idx < n:
            gr[j] = 2.0 * a[idx]
    gr -= gr.mean()
    return gr

            a = np.maximum(a, 1e-12)
            a /= a.sum()
            st = 0
            continue
        g = None
        if use_cg:
            g = _lpcg(a, M * (1 + 1e-10), n, ks, tl=2.5)
        else:

        mm = False
        if g is not None and g.sum() > 1e-15:

            if np.max(np.abs(g - a)) > 1e-10:
