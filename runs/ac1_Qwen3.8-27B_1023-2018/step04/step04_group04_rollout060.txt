def _get_parent():

def _fft_conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(np.max(b))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _score_n1(a):
    """Score for sum(a)==1: returns 2n*max_conv"""
    n = len(a)
    b = _fft_conv(a)
    return 2.0 * n * float(np.max(b))


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_dir(a, n, timeout=1.5):
    """LP: max sum(g) s.t. (a*g)_k <= 1, g >= 0. Returns normalized g."""
    A = _build_A(a, n)
    c = -np.ones(n)
    b_ub = np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})

        return None

        return None
    g = np.clip(res.x, 0.0, None)
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs

                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or f[i] < 1e-14 and f[j] < 1e-14:
                    continue
                oi, oj = f[i], f[j]
                for mi in [0.4, 0.7, 1.3, 0.0]:
                    for mj in [0.4, 0.7, 1.3, 0.0]:
                        f[i] = oi * mi if oi > 1e-14 else 0.0
                        f[j] = oj * mj if oj > 1e-14 else 0.0
                        s = f.sum()
                        if s < 0.01:
                            f[i], f[j] = oi, oj
                            continue
                        sc = _score_n1(f / s)
                        if sc < best_sc - 1e-14:
                            best_sc = sc
                            best_f = f.copy()
                            two_imp = True
                f[i], f[j] = oi, oj
            if not two_imp:
                break

    return best_f, best_sc


def _peak_coord(f0, stop_time, seed, max_epochs=25):
    """Coordinate descent targeting peak-contributing elements."""
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = np.clip(np.asarray(f0, dtype=np.float64).copy(), 0, None)
    f = f / max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = _score_n1(f)

    for epoch in range(max_epochs):
        if time.monotonic() > stop_time:
            break
        b = _fft_conv(f)
        M = float(b.max())
        active = np.flatnonzero(b >= 0.92 * M)

        # Identify top-contributing indices for active peaks
        contrib = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            for i in range(i_lo, i_hi + 1):
                kj = k - i
                if 0 <= kj < n:
                    contrib[i] += f[i] * f[kj]

        # Prioritize high-contribution indices
        order = np.argsort(-contrib)

        improved = False
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            if orig < 1e-14:
                continue
            best_v = orig
            best_s = best_sc
            for mult in [0.3, 0.5, 0.7, 0.9, 1.1, 1.4, 0.0]:
                f[i] = orig * mult
                s = f.sum()
                if s < 0.01:
                    f[i] = orig
                    continue
                sc = _score_n1(f / s)
                if sc < best_s:
                    best_s = sc
                    best_v = f[i]
            f[i] = best_v
            if best_s < best_sc - 1e-14:
                best_sc = best_s
                best_f = f.copy()
                improved = True
        if not improved:
            break

    return best_f, best_sc


def _make_init(n, parent, rng):
    """Create diverse initializations for length n."""
    inits = []
    # Interpolated parent
    if len(parent) > 1:
        t = np.linspace(0, 1, n)
        s = np.linspace(0, 1, len(parent))
        ip = np.interp(t, s, parent)
        ip = np.clip(ip, 0, None)
        if ip.sum() > 0.01:
            inits.append(ip / ip.sum())
    # Uniform
    inits.append(np.ones(n) / n)
    # Random
    r = np.abs(rng.normal(1.0, 0.5, n))
    r = np.clip(r, 0, None)
    if r.sum() > 0.01:
        inits.append(r / r.sum())
    # Comb pattern
    comb = np.zeros(n)
    comb[::2] = 1.0
    if comb.sum() > 0.01:
        inits.append(comb / comb.sum())
    # Triangle
    tri = np.ones(n)
    m = n // 2
    tri[:m] = np.linspace(1, 0.3, m)
    tri[m:] = np.linspace(0.3, 1, n - m)
    tri = np.clip(tri, 0, None)
    inits.append(tri / tri.sum())
    return inits


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    deadline = t0 + budget - 18.0

    parent = _get_parent()
    n = len(parent)
    psum = float(parent.sum())

    fallback = parent.copy()
    fallback_sc = _score(fallback)
    if not math.isfinite(fallback_sc):
        fallback = np.ones(200)
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update(a, sc):
        nonlocal best_seq, best_score
        if sc < best_score - 1e-13:
            if not np.all(np.isfinite(a)) or a.sum() < 0.01:
                return
            best_seq = np.clip(a, 0.0, 1000.0).copy()
            best_score = sc

    # --- Phase 1: LP optimization on parent length (main workhorse) ---
    p_norm = parent / max(psum, 1e-10)
    stop1 = t0 + budget * 0.38
    res1, sc1 = _lp_loop(p_norm, n, stop1, seed)
    if math.isfinite(sc1):
        _update(res1 * psum, sc1)

    # --- Phase 2: LP with perturbed start ---
    if time.monotonic() < t0 + budget * 0.48:
        rng2 = np.random.default_rng(seed + 17)
        p2 = np.clip(p_norm + rng2.normal(0, 0.12, n), 0, None)
        p2 = p2 / max(p2.sum(), 1e-10)
        stop2 = min(t0 + budget * 0.48, stop1 + 60)
        res2, sc2 = _lp_loop(p2, n, stop2, seed + 200)
        if math.isfinite(sc2):
            _update(res2 * psum, sc2)

    # --- Phase 3: Peak-directed coordinate descent ---
    if time.monotonic() < t0 + budget * 0.58:
        f_cur = best_seq / max(best_seq.sum(), 1e-10)
        stop3 = t0 + budget * 0.58
        res3, sc3 = _peak_coord(f_cur, stop3, seed + 400, max_epochs=30)
        if math.isfinite(sc3):
            _update(res3 * psum, sc3)

    # --- Phase 4: General coordinate descent ---
    if time.monotonic() < t0 + budget * 0.68:
        f_cur = best_seq / max(best_seq.sum(), 1e-10)
        stop4 = t0 + budget * 0.68
        res4, sc4 = _coord_descent(f_cur, stop4, seed + 600, max_epochs=40)
        if math.isfinite(sc4):
            _update(res4 * psum, sc4)

    # --- Phase 5: Explore nearby lengths ---
    stop5 = deadline - 40
    if time.monotonic() < stop5:
        alt_lens = sorted(set([n - 2, n - 1, n + 1, n + 2, n + 4, n + 8, n - 4, n - 8, n + 16]))
        alt_lens = [L for L in alt_lens if 16 <= L <= 800]
        rng5 = np.random.default_rng(seed + 800)
        time_per = min(35.0, max(10.0, (stop5 - time.monotonic()) / max(1, len(alt_lens))))
        for L in alt_lens:
            if time.monotonic() > stop5 - 5:
                break
            stop_L = min(stop5, time.monotonic() + time_per)
            inits = _make_init(L, parent, rng5)
            for ix, init in enumerate(inits[:2]):
                if time.monotonic() > stop_L:
                    break
                try:
                    rL, sL = _lp_loop(init, L, stop_L, seed + 2000 + L * 10 + ix, max_iters=3000)
                    if math.isfinite(sL):
                        _update(rL * psum, sL)
                except Exception:
                    pass

    # --- Phase 6: Final LP polish on best ---
    if time.monotonic() < deadline - 20:
        f_cur = best_seq / max(best_seq.sum(), 1e-10)
        nc = len(f_cur)
        stop6 = deadline - 20
        res6, sc6 = _lp_loop(f_cur, nc, stop6, seed + 9000, max_iters=3000)
        if math.isfinite(sc6):
            _update(res6 * psum, sc6)

    # --- Phase 7: Final coordinate descent ---
    if time.monotonic() < deadline - 8:
        f_cur = best_seq / max(best_seq.sum(), 1e-10)
        stop7 = deadline - 8
        res7, sc7 = _coord_descent(f_cur, stop7, seed + 12000, max_epochs=35)
        if math.isfinite(sc7):
            _update(res7 * psum, sc7)

    # Final validation with exact evaluator
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            fs = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            fs = math.inf
        if not math.isfinite(fs):
            return [1.0]
        return [float(x) for x in fallback]

    # Ensure non-negative and finite
    out = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    if any(not math.isfinite(x) for x in out) or len(out) == 0:
        return [float(x) for x in fallback]
    return out
