import numpy as np
import time
import math
from scipy.optimize import linprog


def _conv(a):
    """Full autocorrelation of a."""
    n = len(a)
    if n <= 4096:
        return np.convolve(a, a, mode="full")
    from scipy.fft import rfft, irfft, next_fast_len
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    """Fast score computation (same as evaluator but clipped)."""
    n = len(a)
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    """Build constraint matrix A where (f*b)_j = (A*b)_j."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    idx = np.arange(n)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, timeout=2.0):
    """Solve LP: max sum(b) s.t. (f*b)_j <= M, b >= 0."""
    A = _build_A(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout,
                                                'dual_feasibility_tolerance': 1e-9,
                                                'primal_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-14:
        return None
    return g


def _smooth_step(f, n, temp, lr):
    """One step of projected gradient with softmax-smoothed max."""
    b = _conv(f)
    b_shift = b - b.max()
    exp_b = np.exp(b_shift / temp)
    p = exp_b / exp_b.sum()
    # grad[i] = 2 * sum_k p[i+k] * f[k] for valid indices
    # Use correlate: grad = 2 * cross_correlate(p, f)
    # cross_correlate(p, f)[i] = sum_k p[i+k] * f[k]
    # Pad f to length 2n-1 for the convolution
    f_pad = np.zeros(2 * n - 1)
    f_pad[:n] = f
    # We need grad[i] = 2 * sum_{k=0}^{n-1} p[i+k] * f[k] for i=0..n-1
    # This is the cross-correlation of p and f evaluated at lags 0..n-1
    # np.correlate(p, f_pad, mode='valid') won't work directly
    # Let's use: grad[i] = 2 * dot(p[i:i+n], f)
    grad = np.zeros(n)
    for i in range(n):
        lo = i
        hi = i + n
        if hi <= 2 * n - 1:
            grad[i] = 2.0 * np.dot(p[lo:hi], f)
    f_new = f - lr * grad
    f_new = np.maximum(f_new, 0)
    if f_new.sum() > 1e-12:
        f_new = f_new / f_new.sum()
    else:
        f_new = f.copy()
    return f_new


def _lp_optimize(f0, n, stop_time, seed, lp_timeout=1.5):
    """Main LP-based optimization loop."""
    rng = np.random.default_rng(seed)
    f = np.maximum(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0)
    if f.size < n:
        f = np.concatenate([f, np.zeros(n - f.size)])
    if f.sum() < 1e-10:
        f = np.ones(n, dtype=np.float64)
    f = f / f.sum()

    best_f = f.copy()
    best_sc = _score_fast(f)
    t = 0.3
    stall = 0
    max_stall = 80

    # Temperature for smooth step fallback
    temp = 0.05

    while time.monotonic() < stop_time - 2.0:
        b = _conv(f)
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(f, n, M, timeout=lp_timeout)

        if g is not None:
            g = g / g.sum()
            improved = False
            # Line search with multiple candidates
            candidates = [t, 0.5 * t, 0.25 * t, 1.5 * t, 0.1 * t, 0.7 * t]
            base_sc = _score_fast(f)
            for ti in candidates:
                if ti <= 0 or ti >= 1:
                    continue
                f_new = np.maximum((1 - ti) * f + ti * g, 0)
                if f_new.sum() < 1e-12:
                    continue
                f_new = f_new / f_new.sum()
                sc = _score_fast(f_new)
                if sc < best_sc - 1e-15:
                    f = f_new
                    best_f = f_new.copy()
                    best_sc = sc
                    improved = True
                    t = min(0.7, ti * 1.25)
                    stall = 0
                    break
                elif sc < base_sc - 1e-15:
                    f = f_new
                    improved = True
                    t = min(0.7, ti)
                    stall = 0
                    break
            if not improved:
                t = max(0.02, t * 0.5)
                stall += 1
        else:
            # Fallback: smooth gradient step
            f_new = _smooth_step(f, n, temp, lr=0.3)
            sc = _score_fast(f_new)
            if sc < best_sc - 1e-15:
                f = f_new
                best_f = f_new.copy()
                best_sc = sc
                stall = 0
                temp = max(0.01, temp * 0.9)
            else:
                # Add small noise
                f = np.maximum(f + rng.normal(0, 0.01, n), 0)
                if f.sum() < 1e-10:
                    f = best_f.copy()
                else:
                    f = f / f.sum()
                stall += 2
                temp = min(0.5, temp * 1.2)

        if stall >= max_stall:
            # Random restart from best with perturbation
            f = np.maximum(best_f + rng.normal(0, 0.05, n), 0)
            if f.sum() < 1e-10:
                f = best_f.copy()
            else:
                f = f / f.sum()
            t = 0.3
            stall = 0
            max_stall = min(max_stall + 40, 200)

    return best_f, best_sc


def _cd_polish(f0, stop_time, max_outer=30, seed=0):
    """Coordinate descent with adaptive multipliers and line search."""
    n = len(f0)

    if f.sum() < 1e-10:
        f = np.ones(n)
    cur = _score_fast(f)
    best = f.copy()
    best_sc = cur

    orig_scores = []

    for outer in range(max_outer):
        if time.monotonic() > stop_time - 3.0:
            break
        improved_any = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            orig = f[i]
            # Generate candidate values
            cands = set()
            cands.add(0.0)
            if orig > 1e-12:
                for m in [0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0, 0.05]:
                    cands.add(max(0.0, orig * m))
            else:
                base = max(1e-4, 0.3 * np.median(f[f > 0])) if np.any(f > 0) else 1e-4
                for m in [0.5, 1.0, 2.0, 4.0, 8.0]:
                    cands.add(base * m)

            best_v = orig
            best_local = cur
            for v in cands:
                f[i] = v
                if f.sum() < 0.01:
                    f[i] = orig
                    continue
                sc = _score_fast(f)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_v = v
            f[i] = best_v
            if best_local < cur - 1e-15:
                cur = best_local
                improved_any = True
                if cur < best_sc:
                    best_sc = cur
                    best = f.copy()

        # 2-coordinate moves if stuck
        if not improved_any:
            for _ in range(100):
                if time.monotonic() > stop_time - 3.0:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                oi, oj = f[i], f[j]
                bi, bj, bsc = oi, oj, cur
                for mi in [0.0, 0.3, 0.6, 1.5, 2.5, 4.0]:
                    for mj in [0.0, 0.3, 0.6, 1.5, 2.5, 4.0]:
                        if mi == 1.0 and mj == 1.0:
                            continue
                        f[i] = oi * mi if oi > 1e-15 else (0.0 if mi == 0 else 0.3 * np.median(f[f>0]) if mi > 0 else 0)
                        f[j] = oj * mj if oj > 1e-15 else (0.0 if mj == 0 else 0.3 * np.median(f[f>0]) if mj > 0 else 0)
                        if f.sum() < 0.01:
                            f[i], f[j] = oi, oj
                            continue
                        sc = _score_fast(f)
                        if sc < bsc - 1e-15:
                            bsc = sc
                            bi, bj = f[i], f[j]
                f[i], f[j] = bi, bj
                if bsc < cur - 1e-15:
                    cur = bsc
                    improved_any = True
                    if cur < best_sc:
                        best_sc = cur
                        best = f.copy()

        if not improved_any and outer >= 3:
            break

    return best, best_sc
