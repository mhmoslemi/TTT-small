import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def propose_candidate(seed=42, budget_s=1300, **kwargs):

    hard = t0 + budget - 22.0

    def exact_score(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        nv = a.size
        s = float(a.sum())
        if s < 0.01:
            return math.inf
        b = np.convolve(a, a)
        m = float(b.max())
        if not math.isfinite(m) or m < 0:
            return math.inf
        return float(2.0 * nv * m / (s * s))

    best_score = exact_score(p)

        nonlocal best_p, best_score

            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)

                return

            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_p = a.copy()
                best_score = sc
        except Exception:
            pass

    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 1)
    cross_len = 3 * n - 2

        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xcorr(p_arr, w_arr):
        F = rfft(p_arr[::-1], cross_size)
        G = rfft(w_arr, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    def u_to_p(uu):
        m = float(np.max(uu))
        e = np.exp(uu - m)
        s = float(e.sum())
        if s < 1e-300 or not math.isfinite(s):
            return np.full(n, 1.0 / n)
        return e / s

    # ============ LSE Objective ============
    def lse_obj(uu, T):
        pp = u_to_p(uu)
        c = conv_self(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros(n)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        s_exp = float(exps.sum())
        if not math.isfinite(s_exp) or s_exp <= 0:
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * math.log(s_exp / conv_len)
            w = exps / s_exp
        q = xcorr(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    # ============ Coordinate Descent ============
    def coord_descent(p_in, stop_time, max_sweeps, rng, G=32, Gf=24, consider_fn=consider):

                        vx2 = float(fine[imin2])
                        vv2 = float(vals2[imin2])
                        if vv2 < best_val:
                            best_x = vx2
                            best_val = vv2

                if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 10.0:
                    w_delta = best_x - v
                    if abs(w_delta) > 1e-14:
                        p[i] = best_x
                        c[start:end] += w_delta * infl
                        di = 2 * i
                        if 0 <= di < conv_len:
                            c[di] += w_delta * w_delta
                        cur_score = best_val
                        improved = True

            # End of sweep
            s = float(p.sum())
            if s > 1e-300 and math.isfinite(s):
                inv = 1.0 / s
                p *= inv
                c *= inv * inv
            else:

                        p = p * np.exp(rng.normal(0.0, sigma, n))

                        ss = float(p.sum())
                        if ss > 0:
                            p /= ss
                        c = conv_self(p)

                        flat = 0
                    else:

            sweep += 1

        p = np.maximum(p, 0.0)
        ss = float(p.sum())
        if ss > 1e-300:
            p /= ss
        consider_fn(p)
        return p

    c0 = conv_self(p)
    cmax0 = float(c0.max())

                break

            if getattr(res, "status", 1) == 0:
                break
        return u_cur

    stages = [
        (0.08, 0.10),
        (0.03, 0.12),
        (0.01, 0.15),

    for frac, wt in stages:

        if now > lse_progress_end - 5.0:

        end_stage = min(lse_progress_end, t0 + budget * 0.45 * (cum / total_w))
        end_stage = min(end_stage, t0 + budget * 0.45)
        if end_stage <= now + 4.0:

        T = max(1e-300, frac * cmax0)

        pp = u_to_p(u)

        p = pp.copy()
        c0 = conv_self(p)

    # ============ Coordinate Descent Phase 1 ============
