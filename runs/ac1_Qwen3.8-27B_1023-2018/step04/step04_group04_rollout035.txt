        b = _fft_conv(f)
        M = float(b.max())
        g, t_opt = _solve_minpeak_lp(f, n, time_limit=6.0)
        if g is None:
            stall += 4
            if stall > max_stall:
                break
            f = np.clip(f + rng.normal(0, 0.008, n), 0.0, None)
            f = f / max(f.sum(), 1e-10)
            continue
        if t_opt >= M * (1.0 - 1e-10):
            stall += 2
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.03, n), 0.0, None)
                f = f / max(f.sum(), 1e-10)
                stall = 0
                max_stall = min(250, max_stall + 40)
            continue
        best_alpha, best_sc_try = _line_search(f, g, n)
        if best_sc_try < best_sc - 1e-14:
            a_new = (1.0 - best_alpha) * f + best_alpha * g
            a_new = np.clip(a_new, 0.0, None)
            f = a_new / max(a_new.sum(), 1e-10)
            best_f = f.copy()
            best_sc = best_sc_try
            stall = 0
            max_stall = max(80, int(max_stall * 0.9))
        else:
            stall += 1
            if stall > max_stall:
                f = np.clip(f + rng.normal(0, 0.04, n), 0.0, None)
                f = f / max(f.sum(), 1e-10)
                stall = 0
                max_stall = min(300, max_stall + 50)
    return best_f, best_sc


def _subgrad_optimize(f0, n, stop_time, rng):
    f = np.clip(np.asarray(f0, dtype=np.float64), 0.0, None).copy()
    f = f / max(f.sum(), 1e-10)
    best_f = f.copy()
    best_sc = _score(f)
    step = 0.04
    momentum = np.zeros(n)
    mu = 0.6
    no_imp = 0
    for _ in range(100000):
        if time.monotonic() > stop_time:
            break
        b = _fft_conv(f)
        M = float(b.max())
        if M < 1e-18:
            break
        S = float(f.sum())
        eps = max(1e-14, 1e-10 * M)
        active = np.flatnonzero(b >= M - eps)
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo <= i_hi:
                idx = k - np.arange(i_lo, i_hi + 1)
                grad[i_lo:i_hi + 1] += 2.0 * f[idx]
        if len(active) > 0:
            grad /= len(active)
        f_sub = 2.0 * n * (grad * S - 2.0 * M) / (S ** 3)
        f_sub -= np.mean(f_sub)
        ng = np.linalg.norm(f_sub)
        if ng < 1e-16:
            break
        momentum = mu * momentum - step * f_sub / ng
        fn = np.clip(f + momentum, 0.0, None)
        s = fn.sum()
        if s < 0.01:
            step *= 0.5
            momentum *= 0.5
            continue
        fn = fn / s
        sc = _score(fn)
        if sc < best_sc - 1e-14:
            f = fn
            best_f = f.copy()
            best_sc = sc
            step = min(0.4, step * 1.02)
            no_imp = 0
        else:
            step *= 0.92
            no_imp += 1
            if no_imp > 400:
                f = np.clip(f + rng.normal(0, 0.03, n), 0.0, None)
                f = f / max(f.sum(), 1e-10)
                momentum = np.zeros(n)
                step = 0.04
                no_imp = 0
    return best_f, best_sc


def _coord_descent(f0, stop_time, rng, max_rounds=25):
    f = np.clip(np.asarray(f0, dtype=np.float64), 0.0, None).copy()
    n = len(f)
    best_sc = _score(f)
    best_f = f.copy()
    for _ in range(max_rounds):
        if time.monotonic() > stop_time:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time:
                break
            orig = f[i]
            if orig < 1e-15:
                continue
            best_v = orig
            best_v_sc = best_sc
            for mult in [0.0, 0.2, 0.4, 0.6, 0.8, 1.1, 1.3, 1.6, 2.0]:
                f[i] = orig * mult
                s = f.sum()
                if s < 0.01:
                    f[i] = orig
                    continue
                sc = _score(f)
                if sc < best_v_sc:
                    best_v_sc = sc
                    best_v = f[i]
            f[i] = best_v
            if best_v_sc < best_sc - 1e-15:
                best_sc = best_v_sc
                best_f = f.copy()
                improved = True
        if not improved:
            break
    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 30.0

    parent = _get_parent()
    n = len(parent)
    rng = np.random.default_rng(seed)

    p_norm = parent / max(parent.sum(), 1e-10)
    best_seq = p_norm.copy()
    best_score = _score(p_norm)

    def _update(a, sc):
        nonlocal best_seq, best_score
        if sc < best_score - 1e-12:
            if np.all(np.isfinite(a)) and a.sum() > 0.01:
                best_seq = np.clip(a, 0.0, 1000.0).copy()
                best_score = sc

    # Phase 1: Primary LP optimization
    phase1_end = t0 + budget * 0.30
    res, sc = _lp_optimize(p_norm, n, phase1_end, rng)
    if math.isfinite(sc):
        _update(res, sc)

    # Phase 2: LP from perturbed best
    phase2_end = t0 + budget * 0.45
    if time.monotonic() < phase2_end:
        p2 = np.clip(best_seq / max(best_seq.sum(), 1e-10) + rng.normal(0, 0.04, n), 0.0, None)
        p2 = p2 / max(p2.sum(), 1e-10)
        res2, sc2 = _lp_optimize(p2, n, phase2_end, rng)
        if math.isfinite(sc2):
            _update(res2, sc2)

    # Phase 3: LP from random init
    phase3_end = t0 + budget * 0.58
    if time.monotonic() < phase3_end:
        f_rand = rng.random(n)
        # Add some structure: half-plate
        f_rand[:n // 2] *= 2.0
        f_rand = f_rand / max(f_rand.sum(), 1e-10)
        res3, sc3 = _lp_optimize(f_rand, n, phase3_end, rng)
        if math.isfinite(sc3):
            _update(res3, sc3)

    # Phase 4: Subgradient refinement
    phase4_end = t0 + budget * 0.68
    if time.monotonic() < phase4_end:
        f_sg = best_seq / max(best_seq.sum(), 1e-10)
        res4, sc4 = _subgrad_optimize(f_sg, n, phase4_end, rng)
        if math.isfinite(sc4):
            _update(res4, sc4)

    # Phase 5: LP again from best
    phase5_end = t0 + budget * 0.78
    if time.monotonic() < phase5_end:
        res5, sc5 = _lp_optimize(best_seq / max(best_seq.sum(), 1e-10), n, phase5_end, rng)
        if math.isfinite(sc5):
            _update(res5, sc5)

    # Phase 6: Coordinate descent polish
    phase6_end = deadline - 15
    if time.monotonic() < phase6_end:
        res_cd, sc_cd = _coord_descent(best_seq, phase6_end, rng, max_rounds=30)
        if math.isfinite(sc_cd):
            _update(res_cd, sc_cd)

    # Phase 7: Final LP + CD
    if time.monotonic() < deadline - 5:
        res_f, sc_f = _lp_optimize(best_seq / max(best_seq.sum(), 1e-10), n, deadline - 8, rng)
        if math.isfinite(sc_f):
            _update(res_f, sc_f)
        if time.monotonic() < deadline - 3:
            res_cd2, sc_cd2 = _coord_descent(best_seq, deadline - 3, rng, max_rounds=10)
            if math.isfinite(sc_cd2):
                _update(res_cd2, sc_cd2)

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        fallback = [1.0] * n
        try:
            fb_sc = float(evaluate_sequence(fallback))
        except Exception:
            fb_sc = math.inf
        if math.isfinite(fb_sc):
            return fallback
        return [1.0]

    return [float(x) for x in best_seq]
