import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _conv_cross(a, b):
    n = len(a)
    m = len(b)
    fsize = next_fast_len(n + m - 1)
    A = rfft(a, fsize)
    B = rfft(b, fsize)
    c = irfft(A * B, fsize)[:n + m - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_local(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_conv_A(f, n):
    """A[k,j] = f[k-j] if 0<=k-j<n else 0; shape (2n-1, n)"""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n, M, top_k, time_limit):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g>=0. Use top_k tightest constraints."""
    b = _conv_full(f)
    if top_k is None or top_k >= 2 * n - 1:
        rows = np.arange(2 * n - 1)
    else:
        rows = np.argsort(-b)[:top_k]
    
    A = _build_conv_A(f, n)
    A_sub = A[rows]
    c = -np.ones(n)
    b_ub = np.full(len(rows), M)
    
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-12:
        return None
    return g / gs


def _line_search_quad(f, g, stop_time):
    """Fast line search using precomputed convolutions. Minimize max conv along (1-t)f+t*g."""
    n = len(f)
    ff = _conv_full(f)
    fg = _conv_cross(f, g)
    gg = _conv_full(g)
    
    best_t = 0.0
    best_M = float(ff.max())
    
    ts = np.linspace(0.0, 1.0, 41)
    for t in ts:
        at2 = (1.0 - t) ** 2 * ff + 2.0 * t * (1.0 - t) * fg + t ** 2 * gg
        Mt = float(at2.max())
        if Mt < best_M:
            best_M = Mt
            best_t = t
    
    if 0.0 < best_t < 1.0:
        lo = max(0.0, best_t - 0.06)
        hi = min(1.0, best_t + 0.06)
        gr = (1.0 + math.sqrt(5.0)) / 2.0
        for _ in range(18):
            if time.monotonic() > stop_time:
                break
            m1 = hi - (hi - lo) / gr
            m2 = lo + (hi - lo) / gr
            a1 = (1.0 - m1) ** 2 * ff + 2.0 * m1 * (1.0 - m1) * fg + m1 ** 2 * gg
            a2 = (1.0 - m2) ** 2 * ff + 2.0 * m2 * (1.0 - m2) * fg + m2 ** 2 * gg
            v1 = float(a1.max())
            v2 = float(a2.max())
            if v1 < v2:
                hi = m2
            else:
                lo = m1
            tm = (lo + hi) / 2.0
            am = (1.0 - tm) ** 2 * ff + 2.0 * tm * (1.0 - tm) * fg + tm ** 2 * gg
            vm = float(am.max())
            if vm < best_M:
                best_M = vm
                best_t = tm
    elif best_t == 0.0:
        for t in [0.02, 0.05, 0.1]:
            at2 = (1.0 - t) ** 2 * ff + 2.0 * t * (1.0 - t) * fg + t ** 2 * gg
            Mt = float(at2.max())
            if Mt < best_M:
                best_M = Mt
                best_t = t
    
    return best_t, best_M


def _lp_optimize(f0, n, stop_time, seed):

    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    while f.size < n:

    f = f[:n]
    fs = float(f.sum())
    if fs < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / fs
    
    best_f = f.copy()
    best_M = float(_conv_full(f).max())
    M = best_M
    stall = 0
    prev_M = M
    
    while time.monotonic() < stop_time - 3.0:
        if M < prev_M - 1e-14:
            stall = max(0, stall - 3)
        else:
            stall += 1
        prev_M = M
        
        if stall > 40:
            # Perturbation strategies
            strategy = stall % 5
            if strategy == 0:
                f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            elif strategy == 1:
                blk = max(1, n // 4)
                f2 = f.copy()
                f2[:blk], f2[n - blk:] = f[n - blk:].copy(), f[:blk].copy()
                f = f2
            elif strategy == 2:
                kern = np.exp(-0.5 * (np.arange(-8, 9) / 3.0) ** 2)
                kern /= kern.sum()
                noise = np.convolve(rng.normal(0, 0.15, n), kern, 'same')
                f = np.clip(f * (1.0 + noise), 0, None)
            elif strategy == 3:

            else:

                f2 = f * np.cos(2 * math.pi * phase_shift) + rng.normal(0, 0.03, n)
                f = np.clip(f2, 0, None)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            M = float(_conv_full(f).max())
            stall = 0
            continue

        else:

def _coord_descent(a, stop_time, seed=0, n_passes=None):
    """Fast coordinate descent using incremental convolution updates."""
    n = len(a)
    best = np.clip(np.asarray(a, dtype=np.float64).ravel().copy(), 0, None)
    fs = best.sum()
    if fs < 1e-10:
        best = np.ones(n) / n
    else:
        best = best / fs
    best_M = float(_conv_full(best).max())
    rng = np.random.default_rng(seed)
    pass_count = 0
    no_imp_count = 0
    max_no_imp = 6
    
    while time.monotonic() < stop_time - 3.0:
        if n_passes is not None and pass_count >= n_passes:
            break
        improved = False
        b = _conv_full(best)
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > stop_time - 3.0:
                break
            oi = best[i]
            # Try several multiplier values
            if oi > 1e-8:
                mults = np.array([0.0, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0])
            else:
                mults = np.array([0.0, 0.01, 0.05, 0.1, 0.3, 1.0])
            
            best_ai = oi
            best_local_M = float(b.max())
            
            for mlt in mults:
                nv = oi * mlt if oi > 1e-8 else float(mlt)
                if nv < 0:
                    nv = 0.0
                delta = nv - oi
                if abs(delta) < 1e-15:
                    continue
                # Compute new max conv incrementally
                # b_k changes: for k in [i, i+n-1] (valid range), b_k += 2*delta*a_{k-i}
                # Plus quadratic correction at k=2i: + delta^2
                new_b_vals = np.empty(0)
                k_lo = i
                k_hi = min(i + n, 2 * n - 1)
                if k_hi > k_lo:
                    indices = np.arange(k_lo, k_hi)
                    # a_{k-i} for k in [i, i+n-1] -> a[0], a[1], ..., a[n-1]
                    affected = best[0:n]
                    new_vals = b[k_lo:k_hi] + 2.0 * delta * affected
                    # Quadratic correction at k=2i if in range
                    if 2 * i >= k_lo and 2 * i < k_hi:
                        new_vals[2 * i - k_lo] += delta * delta
                    # Unaffected regions
                    unaffected_max_left = b[:k_lo].max() if k_lo > 0 else 0.0
                    unaffected_max_right = b[k_hi:].max() if k_hi < 2 * n - 1 else 0.0
                    local_max = max(float(new_vals.max()), float(unaffected_max_left), float(unaffected_max_right))
                else:
                    local_max = float(b.max())
                
                if local_max < best_local_M - 1e-15:
                    best_local_M = local_max
                    best_ai = nv
            
            if best_ai != oi:
                best[i] = best_ai
                if best_local_M < best_M - 1e-14:
                    best_M = best_local_M
                    improved = True
        
        # Renormalize
        fs = best.sum()
        if fs > 1e-10:
            best = best / fs
            best_M = float(_conv_full(best).max())
        
        pass_count += 1
        if not improved:
            no_imp_count += 1
            if no_imp_count >= max_no_imp:
                break
        else:
            no_imp_count = 0
    
    return best, 2.0 * n * best_M


def _pair_descent(a, stop_time, seed=1):

    no_imp = 0

        n_pairs = min(200, n * 2)

            j = int(rng.integers(n))

        for ri in range(n_rounds):
            s = time.monotonic() + t_each
            if s > phase3_end:
                s = phase3_end
            if time.monotonic() >= s - 3.0:
                break
            try:
                if ri % 3 == 0:
                    best_seq, sc = _coord_descent(best_seq, s, seed=10000 + ri * 7919)
                elif ri % 3 == 1:
                    best_seq, sc = _pair_descent(best_seq, s, seed=20000 + ri * 7919)
                else:
                    best_seq, sc = _coord_descent(best_seq, s, seed=30000 + ri * 7919, n_passes=5)
                
                if np.all(np.isfinite(best_seq)) and best_seq.sum() > 0.01:
                    best_seq = np.clip(best_seq, 0, 1000)
                    exact = float(evaluate_sequence([float(x) for x in best_seq]))
                    if math.isfinite(exact):
                        if exact < best_score:
                            best_score = exact
                else:
                    break
            except Exception:
                break
    
    # Final short polish
    final_stop = t0 + budget - 10.0
    if time.monotonic() < final_stop - 5.0:
        try:
            best_seq, _ = _coord_descent(best_seq, final_stop, seed=99999, n_passes=3)
            best_seq = np.clip(best_seq, 0, 1000)
            if np.all(np.isfinite(best_seq)) and best_seq.sum() > 0.01:
                exact = float(evaluate_sequence([float(x) for x in best_seq]))
                if math.isfinite(exact) and exact < best_score:
                    best_score = exact
        except Exception:
            pass
    
    # Final validation
    final_seq = np.clip(best_seq, 0, 1000)
    final_seq = np.where(np.isfinite(final_seq), final_seq, 0.0)
    try:
        final_sc = float(evaluate_sequence([float(x) for x in final_seq]))
    except Exception:
        final_sc = math.inf
    
    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            return [1.0]
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]
    
    if final_sc < best_score:
        best_score = final_sc
        final_seq = best_seq.copy()
    
    return [float(x) for x in final_seq]
