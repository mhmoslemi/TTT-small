        return np.zeros(0, dtype=np.float64)

    size = next_fast_len(L)

    if s > 1e-300:

def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    sz = next_fast_len(L)
    F = rfft(a, sz)
    return irfft(F * F, sz)[:L]


def _score_norm(a):
    c = _conv_self(a)
    if c.size == 0:
        return math.inf
    return 2.0 * a.size * float(c.max())


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
        out = np.interp((np.arange(n, dtype=np.float64) + 0.5) / float(n), xp, src)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s > 1e-300:
        out /= s
    else:
        out = np.full(n, 1.0 / n, dtype=np.float64)
    return out


def _build_lp_indices(n):
    rows_l = []
    cols_l = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows_l.append(k)
            cols_l.append(j)
    rows = np.array(rows_l, dtype=np.int32)
    cols = np.array(cols_l, dtype=np.int32)
    return rows, cols
