```python
def _line_search_quadratic(f, g, bf, bm, bg, M_current, stop_time):
    # Precomputed: bf=conv(f,f), bm=conv(f,g), bg=conv(g,g)
    # Find t in [0,1] minimizing max_k [(1-t)^2*bf[k] + 2t(1-t)*bm[k] + t^2*bg[k]]
    
    # Coarse grid
    ts = np.linspace(0, 1, 21)
    best_t, best_val = 0.0, M_current
    for t in ts:
        ct, st = 1-t, t
        vals = ct*ct*bf + 2*st*ct*bm + st*st*bg
        m = vals.max()
        if m < best_val:
            best_val = m
            best_t = t
    
    # Refine with golden section
    lo, hi = max(0, best_t - 0.05), min(1, best_t + 0.05)
    gr = (1 + math.sqrt(5)) / 2
    for _ in range(30):
        if time.monotonic() > stop_time:
            break
        m1 = hi - (hi-lo)/gr
        m2 = lo + (hi-lo)/gr
        ct1, st1 = 1-m1, m1
        v1 = (ct1*ct1*bf + 2*st1*ct1*bm + st1*st1*bg).max()
        ct2, st2 = 1-m2, m2
        v2 = (ct2*ct2*bf + 2*st2*ct2*bm + st2*st2*bg).max()
        if v1 < v2:
            hi = m2
        else:
            lo = m1
        t_mid = (lo+hi)/2
        ct, st = 1-t_mid, t_mid
        vm = (ct*ct*bf + 2*st*ct*bm + st*st*bg).max()
        if vm < best_val:
            best_val = vm
            best_t = t_mid
    
    return best_t, best_val
```

    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

    Fg = rfft(g, fsize)
    c = irfft(Ff * Fg, fsize)[:2 * n - 1]

```python
def _subgradient_step(a, n, step_size, stop_time):
    b = _conv_full(a)
    k_star = int(b.argmax())
    # Subgradient of b[k_star] w.r.t. a[j] = 2*a[k_star - j] for valid j
    grad = np.zeros(n)
    j_lo = max(0, k_star - (n-1))
    j_hi = min(n-1, k_star)
    for j in range(j_lo, j_hi+1):
        grad[j] = 2 * a[k_star - j]
    # We want to DECREASE b[k_star], so move a in -grad direction
    a_new = a - step_size * grad
    a_new = np.clip(a_new, 0, None)
    a_new = a_new / a_new.sum()
    return a_new
```

Actually, for non-negative sequences, the subgradient direction might push some elements negative. The projection to [0, ∞) handles this, but it might not be the most effective direction.

A better approach: instead of subgradient, use "pursuit" - identify the peak lag k*, and move mass from positions that contribute most to b[k*] to positions that don't.

For b[k*] = sum_j a[j]*a[k*-j], the "contributors" are pairs (j, k*-j). To decrease b[k*], we want to decrease these products. One way: reduce a[j] for j that pairs with a large a[k*-j], and increase a at positions that don't contribute to b[k*].

This is essentially what the LP does (find the best direction to update), so I'll rely on the LP for the main optimization and use coordinate descent for polish.

Let me finalize the code. Key decisions:
- n = 384 (same as parent)
- Phase 1: LP optimization with multiple inits (55% of time)
- Phase 2: Try different lengths (15% of time)  
- Phase 3: Coordinate descent + pair polish (30% of time)

For the LP, I'll solve it with HiGHS, using all 767 constraints. The matrix A is rebuilt each iteration (O(n^2) = 147k ops, fast).

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _conv_cross(f, g, n):
    fsize = next_fast_len(2 * n - 1)
    Ff = rfft(f, fsize)
    Fg = rfft(g, fsize)
    c = irfft(Ff * Fg, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _line_search_quad(bf, bm, bg, M_cur):
    best_t, best_val = 0.0, M_cur
    for t in np.linspace(0, 1, 25):
        ct, st = 1 - t, t
        m = (ct * ct * bf + 2 * st * ct * bm + st * st * bg).max()
        if m < best_val:
            best_val, best_t = m, t
    lo, hi = max(0.0, best_t - 0.06), min(1.0, best_t + 0.06)
    gr = (1.0 + math.sqrt(5.0)) / 2.0
    for _ in range(50):
        m1 = hi - (hi - lo) / gr
        m2 = lo + (hi - lo) / gr
        ct1, st1 = 1 - m1, m1
        v1 = (ct1 * ct1 * bf + 2 * st1 * ct1 * bm + st1 * st1 * bg).max()
        ct2, st2 = 1 - m2, m2
        v2 = (ct2 * ct2 * bf + 2 * st2 * ct2 * bm + st2 * st2 * bg).max()
        if v1 < v2:
            hi = m2
        else:
            lo = m1
        tm = (lo + hi) / 2.0
        ct, st = 1 - tm, tm
        vm = (ct * ct * bf + 2 * st * ct * bm + st * st * bg).max()
        if vm < best_val:
            best_val, best_t = vm, tm
    return best_t, best_val


def _lp_optimize(f0, n, stop_time, seed):
    rng = np.random.default_rng(seed)
    f = np.clip(np.asarray(f0, dtype=np.float64).ravel()[:n].copy(), 0, None)
    fs = f.sum()
    if fs < 1e-10:
        f = np.ones(n)
    else:
        f = f / fs
    bf = _conv_full(f)
    M = float(bf.max())
    best_f = f.copy()
    best_M = M
    stall = 0
    iter_ct = 0

    while time.monotonic() < stop_time - 3.0:
        A = _build_A(f, n)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0.0, 20.0)] * n,
                          method='highs',
                          options={'time_limit': 12, 'presolve': True})
        except Exception:
            stall += 5
            f = np.clip(f + rng.normal(0, 0.06, n), 0, None)
            f = f / f.sum()
            continue

        if res.status not in (0, 3) or res.x is None or res.x.sum() < 1e-12:
            stall += 3
            continue

        g = np.asarray(res.x, dtype=np.float64).copy()
        g = g / g.sum()

        if np.linalg.norm(g - f) < 1e-10:
            stall += 3
            f = np.clip(f + rng.normal(0, 0.05, n), 0, None)
            f = f / f.sum()
            continue

        bm = _conv_cross(f, g, n)
        bg = _conv_full(g)
        t_opt, M_opt = _line_search_quad(bf, bm, bg, M)

        if t_opt > 1e-6 and M_opt < M - 1e-13:
            f = (1.0 - t_opt) * f + t_opt * g
            f = np.clip(f, 0, None)
            f = f / f.sum()
            bf = _conv_full(f)
            M = float(bf.max())
            if M < best_M:
                best_M = M
                best_f = f.copy()
            stall = 0
        else:
            stall += 1

        if stall > 40:
            strategy = int(rng.integers(6))
            if strategy == 0:
                f = np.clip(f + rng.normal(0, 0.08, n), 0, None)
            elif strategy == 1:
                blk = int(rng.integers(1, n // 3))
                st = int(rng.integers(0, n - blk * 2 + 1))
                f[st:st+blk], f[n-st-blk:n-st] = f[n-st-blk:n-st].copy(), f[st:st+blk].copy()
            elif strategy == 2:
                f = f[::-1].copy()
            elif strategy == 3:
                f = 0.4 * best_f + 0.6 * np.abs(rng.normal(0.5, 0.4, n))
            elif strategy == 4:
                kern = np.exp(-0.5 * (np.arange(-8, 9) / 3.0) ** 2)
                kern /= kern.sum()
                noise = np.convolve(rng.normal(0, 0.3, n), kern, 'same')
                f = np.clip(f * (1 + noise), 0, None)
            else:
                f = np.clip(f + 0.1 * rng.standard_cauchy(n), 0, None)
            f = np.clip(f, 0, None)
            if f.sum() < 1e-10:
                f = best_f.copy()
            else:
                f = f / f.sum()
            bf = _conv_full(f)
            M = float(bf.max())
            stall = 0

        iter_ct += 1

    return best_f, best_M


def _coord_descent(a, n, stop_time, max_sweeps=30, seed=0):
    rng = np.random.default_rng(seed)
    a = np.clip(a.copy(), 0, None)
    a = a / a.sum()
    b = _conv_full(a)
    M = float(b.max())
    no_imp = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 2.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            old = a[i]
            b = _conv_full(a)
            M_cur = float(b.max())
            best_v, best_m = old, M_cur
            lo_k = i
            hi_k = min(2 * n - 1, i + n - 1)
            a_slice_len = hi_k - lo_k + 1

            if old < 1e-8:
                cands = np.array([0.0, 0.001, 0.003, 0.01, 0.03, 0.1, 0.3])
            else:
                cands = np.linspace(0.0, 2.2 * old, 35)

            for v in cands:
                v = float(max(0.0, v))
                delta = v - old
                if abs(delta) < 1e-15:
                    continue
                b_new = b.copy()
                b_new[lo_k:hi_k + 1] += 2.0 * delta * a[0:a_slice_len]
                k2 = 2 * i
                if lo_k <= k2 <= hi_k:
                    b_new[k2] = b[k2] + v * v - old * old
                m = float(b_new.max())
                if m < best_m - 1e-15:
                    best_m, best_v = m, v

            if best_v != old:
                a[i] = best_v
                b = _conv_full(a)
                M = float(b.max())
                improved = True

        if not improved:
            no_imp += 1
            if no_imp >= 3:
                break
        else:
            no_imp = 0

    b = _conv_full(a)
    return a, b, float(b.max())


def _pair_polish(a, n, stop_time, seed=0):
    rng = np.random.default_rng(seed)
    a = np.clip(a.copy(), 0, None)
    a = a / a.sum()
    b = _conv_full(a)
    M = float(b.max())
    no_imp = 0
    max_rounds = 30

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 2.0:
            break
        improved = False
        n_pairs = min(200, n * 2)
        for _ in range(n_pairs):
            if time.monotonic() > stop_time - 2.0:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            oi, oj = a[i], a[j]
            total = oi + oj
            if total < 1e-12:
                continue
            b = _conv_full(a)
            M_cur = float(b.max())
            best_frac, best_m = (oi / total if total > 0 else 0.5), M_cur
            fracs = np.linspace(0.0, 1.0, 25)
            for fr in fracs:
                a[i], a[j] = fr * total, (1.0 - fr) * total
                bn = _conv_full(a)
                m = float(bn.max())
                if m < best_m - 1e-15:
                    best_m, best_frac = m, fr
            a[i], a[j] = best_frac * total, (1.0 - best_frac) * total
            if best_frac != (oi / total if total > 0 else 0.5):
                M = best_m
                improved = True

        if not improved:
            no_imp += 1
            if no_imp >= 4:
                break
        else:
            no_imp = 0

    b = _conv_full(a)
    return a, float(b.max())


def propose_candidate(seed=42, budget_s=1300, **kwargs):

    i_arr = np.arange(n, dtype=np.float64)
    taper = 0.5 + 0.5 * np.cos(np.pi * i_arr / (n - 1))
    inits.append(("taper", taper))

    kern = np.exp(-0.5 * (np.arange(-8, 9) / 3.0) ** 2)

    smooth_noise = np.convolve(rng_main.normal(0, 1, n), kern, 'same')

        sup += 0.25 * np.cos(2 * math.pi * fr * i_arr / n + ph)
    inits.append(("multifreq", np.abs(sup) + 0.001))

    # Zipper
    z = np.ones(n)
    z[::2] *= 1.4
    z[1::2] *= 0.6
    inits.append(("zipper", z))

    # Gaussian bumps
    bumps = np.zeros(n)
    for _ in range(5):
        ctr = rng_main.integers(n // 4, 3 * n // 4)

    inits.append(("bumps", bumps))

    n_inits = len(inits)

            break
        if best_score < 1.5028:

            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res_c = np.clip(res, 0, 1000)
                exact = float(evaluate_sequence([float(x) for x in res_c]))
                if math.isfinite(exact) and exact < best_score:
                    best_seq = res_c.copy()
                    best_score = exact
        except Exception:
            pass

            break
