            # Actually for 0<=k-i<n: both j=i and j=k-i are valid indices.
            #   If k-i != i: two distinct terms, each contributing f[k-i] to the derivative. Total: 2*f[k-i].
            #   If k-i == i (k=2i): one term f[i]*f[i], derivative is 2*f[i].
            # So: d(f*f)[k]/d f[i] = 2*f[k-i] for all valid k (where 0<=k-i<n).
            # The formula is simply: 2*f[k-i] when 0<=k-i<n.
            
            grad[i] = 2.0 * s1  # since d(f*f)[k]/d f[i] = 2*f[k-i] for all valid k
        
        # score gradient: d/d f[i] of 2n*M_smooth/S^2
        # = (2n/S^2) * grad[i] - (4n*M_smooth/S^3) * 1
        score_grad = (2.0 * n / (S * S)) * (grad - (2.0 * M) / S)
        
        g_raw = f - alpha * score_grad
        f_new = np.maximum(g_raw, 0.0)
        fn = f_new.sum()
        if fn < 1e-10:
            f_new = f
        else:
            f_new = f_new / fn
        
        sc_new = 2.0 * n * float(_conv_full(f_new).max()) / (float(f_new.sum()) ** 2)
        if sc_new < score_val - 1e-15:
            f = f_new
            alpha = max(alpha_min, alpha * 0.97)
        else:
            alpha = max(alpha_min, alpha * 0.5)
            if alpha < alpha_min:
                f = np.clip(f + rng.normal(0, 0.003, n), 0, None)
                f /= f.sum()
                alpha = 0.01

        if it % 1000 == 999:
            f = np.clip(f + rng.normal(0, 0.008, n), 0, None)
            f /= f.sum()
            alpha = 0.02

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_p = len(parent)

    # Fallback
    fallback = parent.copy()
    try:
        fallback_sc = _score(fallback)
    except Exception:
        fallback_sc = math.inf
    if not math.isfinite(fallback_sc):
        fallback = np.ones(32) / 32.0
        fallback_sc = _score(fallback)

    best_seq = fallback.copy()
    best_score = fallback_sc

    def _update_best(seq, sc=None):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=np.float64).ravel()
        seq = np.clip(seq, 0, 1000)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        if seq.size == 0 or float(seq.sum()) < 0.01:
            return
        if sc is None:
            sc = _score(seq)
        if not math.isfinite(sc):
            return
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-14:
            best_seq = seq.copy()
            best_score = exact

    def _try_pipeline(n, init_arr, stop_time, phase_label=""):
        nonlocal best_seq, best_score
        if time.monotonic() > stop_time - 15:
            return
        init_arr = np.clip(np.asarray(init_arr, dtype=np.float64).ravel(), 0, None)
        while init_arr.size < n:
            init_arr = np.concatenate([init_arr, np.zeros(n - init_arr.size)])
        init_arr = init_arr[:n]
        if init_arr.sum() < 1e-10:
            init_arr = np.ones(n) / n

        # LP phase
        t_lp_end = min(stop_time - 30, t0 + (stop_time - t0) * 0.45)
        if time.monotonic() < t_lp_end - 5:
            try:
                res, sc = _lp_optimize(init_arr, n, t_lp_end, seed)
            except Exception:
                res, sc = init_arr, _score(init_arr)
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                res = res / res.sum()
                _update_best(res, sc)

        # Gradient phase
        t_grad_end = min(stop_time - 20, t0 + (stop_time - t0) * 0.6)
        if time.monotonic() < t_grad_end - 5 and best_score < 1.51:
            try:
                res, sc = _grad_optimize(best_seq, len(best_seq), t_grad_end, seed + 1)
            except Exception:
                res, sc = best_seq, best_score
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update_best(res, sc)

        # Coord descent
        t_cd_end = min(stop_time - 5, t0 + (stop_time - t0) * 0.85)
        if time.monotonic() < t_cd_end - 5:
            try:
                res, sc = _coord_polish(best_seq, t_cd_end, max_sweeps=60, seed=seed + 2)
            except Exception:
                res, sc = best_seq, best_score
            if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                _update_best(res, sc)

    # === Phase 1: Primary optimization on parent length ===
    phase1_end = t0 + budget * 0.40
    _try_pipeline(n_p, parent, phase1_end)

    # === Phase 2: Try alternative initializations on same length ===
    if time.monotonic() < phase1_end + budget * 0.12 and best_score > 1.503:
        i_arr = np.arange(n_p, dtype=np.float64)
        alt_inits = [
            np.clip(0.2 + 0.8 * np.cos(np.pi * i_arr / n_p) ** 2, 0, None),
            np.clip(0.5 + 0.5 * np.cos(2 * np.pi * i_arr / n_p), 0.01, None),
            np.clip(1.0 - 0.8 * (i_arr / n_p - 0.5) ** 2, 0.01, None),
        ]
        alt_inits = [x / x.sum() for x in alt_inits if x.sum() > 0.01]
        alt_stop = min(t0 + budget * 0.55, hard_deadline - 60)
        for idx, init in enumerate(alt_inits):
            if time.monotonic() > alt_stop:
                break
            if best_score < 1.503:
                break
            t_alloc = min(alt_stop - 10, time.monotonic() + budget * 0.08)
            try:
                res, sc = _lp_optimize(init, n_p, t_alloc, seed + 200 + idx)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res = res / res.sum()
                    _update_best(res, sc)
                # Quick coord polish
                if time.monotonic() < t_alloc + 30:
                    res2, sc2 = _coord_polish(res, t_alloc + 20, max_sweeps=20, seed=seed + 300 + idx)
                    _update_best(res2, sc2)
            except Exception:
                pass

    # === Phase 3: Intensive coordinate descent on best ===
    t_cd_end = min(t0 + budget * 0.72, hard_deadline - 50)
    if time.monotonic() < t_cd_end - 10:
        try:
            best_seq, best_score = _coord_polish(best_seq, t_cd_end, max_sweeps=100, seed=424242)
        except Exception:
            pass
        _update_best(best_seq, best_score)

    # === Phase 4: Try nearby lengths ===
    t_len_end = min(t0 + budget * 0.87, hard_deadline - 50)
    if time.monotonic() < t_len_end - 30 and best_score > 1.5028:
        other_lengths = []
        for L in [n_p - 16, n_p - 32, n_p + 16, n_p + 32, n_p * 2, n_p // 2]:
            if L >= 32 and L <= 1024 and L not in other_lengths:
                other_lengths.append(L)
        other_lengths.sort(key=lambda L: abs(L - n_p))

        time_per_len = max(15.0, (t_len_end - time.monotonic()) / max(1, len(other_lengths)))
        for L in other_lengths[:4]:
            if time.monotonic() > t_len_end - 10:
                break
            if best_score < 1.5028:
                break
            # Interpolate best to new length
            init_l = np.interp(
                np.linspace(0, 1, L),
                np.linspace(0, 1, len(best_seq)),
                best_seq
            )
            init_l = np.clip(init_l, 0, None)
            if init_l.sum() < 0.01:
                init_l = np.ones(L) / L
            init_l = init_l / init_l.sum()

            stop_lp = min(t_len_end - 5, time.monotonic() + time_per_len * 0.55)
            stop_pol = min(t_len_end - 2, stop_lp + time_per_len * 0.45)
            try:
                res, sc = _lp_optimize(init_l, L, stop_lp, seed + 50000 + L)
                if res is not None and np.all(np.isfinite(res)) and res.sum() > 0.01:
                    res = res / res.sum()
                    _update_best(res, sc)
                if time.monotonic() < stop_pol - 5:
                    res2, sc2 = _coord_polish(res, stop_pol, max_sweeps=20, seed=seed + L)
                    _update_best(res2, sc2)
            except Exception:
                pass

    # === Phase 5: Final intense polish ===
    final_end = hard_deadline - 15.0
    if time.monotonic() < final_end - 10:
        try:
            best_seq, best_score = _coord_polish(best_seq, final_end - 10, max_sweeps=80, seed=31337)
        except Exception:
            pass
        _update_best(best_seq, best_score)

    # Final gradient polish
    if time.monotonic() < hard_deadline - 10:
        try:
            best_seq, best_score = _grad_optimize(best_seq, len(best_seq), hard_deadline - 8, seed=55555, max_iters=5000)
        except Exception:
            pass
        _update_best(best_seq, best_score)

    # Final coordinate polish
    if time.monotonic() < hard_deadline - 8:
        try:
            best_seq, best_score = _coord_polish(best_seq, hard_deadline - 6, max_sweeps=30, seed=88888)
        except Exception:
            pass
        _update_best(best_seq, best_score)

    # Final validation
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0, 1000)
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    if best_seq.size == 0 or float(best_seq.sum()) < 0.01:
        best_seq = fallback.copy()

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
