def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(len(u), 1.0 / len(u))
    return e / s

    cross_size = next_fast_len(cross_L)

def _coord_descent(p, stop_time, consider_callback, seed=12345, verbose=False):

        return p, math.inf

    S = 1.0

                break
            if S <= 1e-300:
                break
            # Periodic renormalization to prevent drift
            if S > 1.3 or S < 0.7:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0

            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            infl = 2.0 * p[:Ls].copy()  # influences c[start:end]

                    m_val = max_const

                    m_val = 0.0

    p = np.maximum(p, 0.0)

                    #   a_old_k = p[k] if k != i else v
                    #   c[j] += 2*w*a_old_k + w^2*(k==i)
                    #   = 2*w*p_old[k] + w^2*delta_{k,i}
                    
                    # Vectorized (but need to handle the diagonal):
                    k_idx = np.arange(Ls)  # 0, 1, ..., Ls-1
                    a_old = p[:Ls].copy()
                    if i < Ls:
                        a_old[i] = v  # restore old value at position i
                    c[start:end] += 2.0 * w * a_old
                    if 0 <= diag_offset < Ls:
                        c[start + diag_offset] += w * w
                    cur_score = best_val
                    improved = True

        # Renormalize
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        if sweep % 2 == 0:
            c = _conv_self(p)
            np.maximum(c, 0.0, out=c)

        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
            consider_callback(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 8:
                break

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    return best_p, best_score


# Global reference for time in helper functions
t0_global = 0.0


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    global t0_global
    t0_global = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0_global + budget - 25.0
    if hard <= t0_global + 5.0:
        hard = t0_global + max(3.0, budget * 0.9)

    # Get parent
    try:
        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(512, dtype=np.float64)
    if p.size == 0:
        p = np.ones(512, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(512, dtype=np.float64)
    n = len(p)
    p = p / float(p.sum())

    # Track best
    best_p = p.copy()
    best_score = _fft_score(p)

    def consider(arr):
        nonlocal best_p, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.clip(a, 0.0, 1000.0)
            a = np.where(np.isfinite(a), a, 0.0)
            if a.size == 0 or float(a.sum()) < 0.01:
                return
            sc = _fft_score(a)
            if math.isfinite(sc) and sc < best_score - 1e-13:
                a = a / float(a.sum())
                best_p = a.copy()
                best_score = sc
        except:
            pass

    # Verify initial score with exact method
    try:
        exact_sc = _exact_score(best_p)
        if math.isfinite(exact_sc):
            best_score = exact_sc
    except:
        pass

    # Phase 1: L-BFGS warm-up (~120s)
    lbfgs_stop = min(hard - 1050.0, t0_global + 120.0)
    if lbfgs_stop > t0_global + 30.0:
        try:
            p2 = _lbfgs_phase(p, n, lbfgs_stop, consider)
            if p2 is not None and len(p2) == n and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p = p2 / float(p2.sum())
                consider(p)
        except Exception:
            pass
        # Restore if L-BFGS made things worse
        if _fft_score(p) > _fft_score(best_p) + 0.001:
            p = best_p.copy()

    # Phase 2: Main coordinate descent (~1000s)
    coord_stop = min(hard - 120.0, t0_global + 1150.0)
    if coord_stop > time.monotonic() + 30.0:
        try:
            p3, sc3 = _coord_descent(p, coord_stop, consider, seed=seed + 777)
            if p3 is not None and len(p3) == n and np.all(np.isfinite(p3)) and p3.sum() > 0:
                p = p3 / float(p3.sum())
                consider(p)
        except Exception:
            pass
        p = best_p.copy() if _fft_score(best_p) < _fft_score(p) - 1e-6 else p

    # Phase 3: Focus polish on top coordinates (~100s)
    polish_stop = min(hard - 30.0, t0_global + 1250.0)
    if polish_stop > time.monotonic() + 20.0:
        try:
            p4, sc4 = _focus_polish(best_p, polish_stop, consider, top_k=300)
            if p4 is not None and len(p4) == n and np.all(np.isfinite(p4)) and p4.sum() > 0:
                p = p4 / float(p4.sum())
                consider(p)
        except Exception:
            pass

    # Phase 4: Another round of coord descent if time remains
    rem_stop = hard - 30.0
    if rem_stop > time.monotonic() + 30.0:
        try:
            p5, sc5 = _coord_descent(best_p, rem_stop, consider, seed=seed + 54321)
            if p5 is not None and len(p5) == n and np.all(np.isfinite(p5)) and p5.sum() > 0:
                consider(p5)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        out = np.asarray(best_p, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(128, 1.0 / 128.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            # Fallback to parent
            try:
                pp = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
                pp = np.clip(pp, 0.0, 1000.0)
                ls = [float(x) for x in pp]
                final_sc = float(evaluate_sequence(ls))
            except:
                return [1.0]
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
