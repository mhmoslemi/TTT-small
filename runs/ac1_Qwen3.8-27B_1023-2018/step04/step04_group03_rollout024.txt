def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    cv_sum = np.cumsum(v_sorted)
    # Find rho
    idx = np.arange(1, n + 1)
    cond = v_sorted * idx > (cv_sum - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.max(np.where(cond))
    tau = (cv_sum[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - tau, 0.0)


def _subgrad_conv(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    for i in range(n):

        if 0 <= j < n:
            if 2 * i == k:
                g[i] += a[j]  # = a[i]
            else:
                g[i] += 2.0 * a[j]
    return g


def _subgrad_conv_fast(a, k):
    """Faster gradient of (a*a)[k] using vectorization."""
    n = len(a)
    # valid i range: 0 <= i < n and 0 <= k-i < n => max(0, k-n+1) <= i <= min(n-1, k)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)

    i_range = np.arange(i_start, i_end)
    j_range = k - i_range

    if k % 2 == 0:

            g[i_mid] = a[i_mid]
    return g

        b = _conv_full(a)
