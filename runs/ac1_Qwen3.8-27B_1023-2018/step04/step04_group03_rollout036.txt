import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(416, dtype=np.float64)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(416, dtype=np.float64)
    return p


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _lp_direction(f, n, b, num_constraints, timeout=5.0):
    """Find LP direction: max sum(g) s.t. (f*g)_k <= M for top peaks, g >= 0."""
    M = float(b.max())
    if M < 1e-18:
        return None
    
    # Select top constraints
    sorted_idx = np.argsort(b)[::-1]
    peak_idx = sorted_idx[:num_constraints]
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(peak_idx):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            v = f[k - i]
            if v > 1e-15:
                rows.append(idx)
                cols.append(i)
                vals.append(v)
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
    c_vec = -np.ones(n)
    b_ub = np.full(len(peak_idx), M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    
    if res.status not in (0, 3) or res.x is None:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g


def _golden_section(f, g, f_sum, stop_time, max_iter=30):
    """Find best step size t in [0, 1] for f_new = (1-t)*f + t*g."""
    a, b = 0.0, 1.0
    gr = (math.sqrt(5) - 1) / 2
    
    # First evaluate at two interior points
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    f_sum_d = (1-d)*f_sum + d*g.sum()
    f_new_d = (1-d)*f + d*g
    f_new_d = np.clip(f_new_d, 0, None)
    sc_d = _score(f_new_d) if f_new_d.sum() > 0.01 else math.inf
    
    f_new_c = (1-c)*f + c*g
    f_new_c = np.clip(f_new_c, 0, None)
    f_sum_c = (1-c)*f_sum + c*g.sum()
    sc_c = _score(f_new_c) if f_new_c.sum() > 0.01 else math.inf
    
    best_t = 0.0
    best_sc = _score(f)
    
    if sc_d < best_sc:
        best_t = d
        best_sc = sc_d
    if sc_c < best_sc:
        best_t = c
        best_sc = sc_c
    
    for _ in range(max_iter):
        if time.monotonic() > stop_time:
            break
        if sc_c < sc_d:
            b = d
            d = c
            sc_d = sc_c
            c = b - gr * (b - a)
            f_new_c = np.clip((1-c)*f + c*g, 0, None)
            sc_c = _score(f_new_c) if f_new_c.sum() > 0.01 else math.inf
            if sc_c < best_sc:
                best_t = c
                best_sc = sc_c
        else:
            a = c
            c = d
            sc_c = sc_d
            d = a + gr * (b - a)
            f_new_d = np.clip((1-d)*f + d*g, 0, None)
            sc_d = _score(f_new_d) if f_new_d.sum() > 0.01 else math.inf
            if sc_d < best_sc:
                best_t = d
                best_sc = sc_d
        if abs(b - a) < 1e-6:
            break
    
    return best_t, best_sc

    rows = []
    cols = []
    vals = []

        i_end = min(n, k + 1)
        for i in range(i_start, i_end):

                rows.append(k)
                cols.append(i)
                vals.append(v)

    rows = []
    cols = []

        if f_new.sum() < 0.01:
            return math.inf
        return _score(f_new)

    c = b_ls - gr * (b_ls - a)
    d = a + gr * (b_ls - a)
    sc_c = eval_t(c)
    sc_d = eval_t(d)

    best_sc = _score(f)
    if sc_c < best_sc:
        best_t, best_sc = c, sc_c

            break

            break

            b_ls = d
            d, sc_d = c, sc_c
            c = b_ls - gr * (b_ls - a)

        f_new = np.clip(f_new, 0, None)

Actually, I just realized something important. The LP direction finding:
$$\max \sum g_i \quad \text{s.t.} \quad (f*g)_k \leq M \; \forall k, \; g \geq 0$$

This finds a $g$ that maximizes total mass while keeping its convolution with $f$ bounded. The intuition is:
- If $f$ has a concentration at position $p$, then to keep $(f*g)_k$ bounded, $g$ must avoid position $k-p$ (for the peak $k$).
- So the LP "pushes" $g$ away from positions that would create peaks when convolved with $f$.
- Moving from $f$ toward $g$ should reduce the peaks of $f*f$.

This is exactly the Frank-Wolfe algorithm for the convex problem:
$$\min_{s \geq 0, \sum s_i = S} \max_k (s*s)_k$$

And it's known to converge (sublinearly) for convex problems.

The convergence rate is $O(1/k)$ where $k$ is the number of iterations. To get from 1.509 to 1.503 (a relative improvement of 0.4%), we might need a few thousand iterations. Each iteration takes ~15ms (LP + line search), so ~15-30 seconds for the Frank-Wolfe phase. That's well within budget.
