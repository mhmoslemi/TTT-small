import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _cd_update(b, f, i, old_val):
    """O(n) in-place update of convolution b when f[i] changes from old_val to f[i]."""
    n = len(f)
    dv = f[i] - old_val
    if abs(dv) < 1e-18:
        return
    lo = i
    hi = min(2 * n - 1, i + n - 1)
    if lo > hi:
        return
    # Vectorized update
    ki = np.arange(0, hi - lo + 1)  # f indices
    b[lo:hi + 1] += 2.0 * dv * f[ki]
    # Fix diagonal: at k=2i, the term f[i]^2 was updated as 2*dv*f[i]_new
    # but should be (old+dv)^2 - old^2 = 2*old*dv + dv^2
    # We added 2*dv*(old+dv) = 2*old*dv + 2*dv^2, excess is dv^2
    dk = 2 * i
    if lo <= dk <= hi:
        b[dk] -= dv * dv

def _lp_dir(f, n, M, timeout=2.0):
    A = np.empty((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g

def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)

        return math.inf
    b = _conv(a)
    m = float(b.max())

def _cd_update(b, f, i, old_val):
    n = len(f)
    dv = f[i] - old_val
    if abs(dv) < 1e-18:

    lo = i
    hi = min(2 * n - 1, i + n - 1)
    if lo > hi:

    ki = np.arange(0, hi - lo + 1)
    b[lo:hi + 1] += 2.0 * dv * f[ki]
    dk = 2 * i
    if lo <= dk <= hi:
        b[dk] -= dv * dv

def _lp_dir(f, n, M, timeout=2.0):
    A = np.empty((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
    if g.sum() < 1e-15:
        return None
    return g

        try:
            exact = float(evaluate_sequence([float(x) for x in res]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-15:
            best_seq = res.copy()
            best_score = exact

    n = n_parent
    i_arr = np.arange(n, dtype=np.float64)
    rng_main = np.random.default_rng(seed)

    inits = [parent.copy()]
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * i_arr / n)))
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * (i_arr + 0.1 * n) / n)))
    ctr = (n - 1) / 2.0
    inits.append(np.maximum(0.0, 1.0 - ((i_arr - ctr) / (0.4 * n)) ** 2))
    inits.append(np.exp(-0.5 * ((i_arr - 0.25 * n) / (0.12 * n)) ** 2)
                 + np.exp(-0.5 * ((i_arr - 0.75 * n) / (0.12 * n)) ** 2))
    sig = 0.12 * max(1.0, np.mean(parent))
    inits.append(np.abs(parent + rng_main.normal(0, sig, n)) + 1e-4)

    seen = {inits[0].tobytes()}
    inits_dedup = [inits[0]]
    for ix in inits[1:]:
        if ix.sum() < 1e-10:
            continue
        key = ix.tobytes()
        if key not in seen:
            seen.add(key)
            inits_dedup.append(ix)

    phase1_end = t0 + budget * 0.50
    tpi = max(10.0, (phase1_end - t0) / len(inits_dedup))
    for idx, init in enumerate(inits_dedup):
        if time.monotonic() > phase1_end - 8.0:
            break
        if best_score < 1.43:
            break
        stop = min(phase1_end - 4.0, time.monotonic() + tpi)
        _try_run(init, n, stop, seed + 1000 * (idx + 1))

    phase2_end = t0 + budget * 0.62
    if time.monotonic() < phase2_end - 15.0:
        _try_run(best_seq.copy(), len(best_seq),
                 min(phase2_end - 3.0, time.monotonic() + (phase2_end - time.monotonic()) * 0.85),
                 seed + 77777)

    phase3_end = t0 + budget * 0.80
    if time.monotonic() < phase3_end - 5.0:
        try:
            f_c, sc_c = _cd_polish(best_seq, phase3_end, max_sweeps=400, seed=seed * 3 + 1)
        except Exception:
            f_c, sc_c = best_seq.copy(), best_score
        if sc_c < best_score - 1e-15:
            try:
                exact = float(evaluate_sequence([float(x) for x in f_c]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-15:
                best_seq = f_c.copy()
                best_score = exact

    phase4_end = t0 + budget * 0.88
    if time.monotonic() < phase4_end - 12.0:
        for L in (n - 20, n + 20, n - 40, n + 40):
            if L < 64 or time.monotonic() > phase4_end - 10.0:
                continue
            init_l = np.interp(np.linspace(0, 1, L),
                               np.linspace(0, 1, len(best_seq)), best_seq)
            stop_l = min(phase4_end - 3.0,
                         time.monotonic() + max(8.0, (phase4_end - time.monotonic()) * 0.35))
            _try_run(init_l, L, stop_l, seed + 90000 + L)

    final_end = deadline - 5.0
    if time.monotonic() < final_end - 8.0:
        try:
            f_c, sc_c = _cd_polish(best_seq, final_end, max_sweeps=600, seed=seed * 7 + 42)
        except Exception:
            f_c, sc_c = best_seq.copy(), best_score
        if sc_c < best_score - 1e-15:
            try:
                exact = float(evaluate_sequence([float(x) for x in f_c]))
            except Exception:
                exact = math.inf
            if math.isfinite(exact) and exact < best_score - 1e-15:
                best_seq = f_c.copy()
                best_score = exact

    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf
    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0, 1.0, 1.0]
        return [float(x) for x in fallback]

    best_score = min(best_score, final_sc)
    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    if not result:
        result = [1.0]
    if any(not math.isfinite(x) for x in result):
        return [float(x) for x in fallback]
    return result
