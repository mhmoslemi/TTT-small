```python
        F = rfft(f, N)
        mag = np.abs(F)
        # Blend current magnitude with a flatter target
        # Target: geometric mean of magnitudes (preserves "energy" in freq domain)
        log_mag = np.log(mag + 1e-20)
        target_log = np.full_like(log_mag, np.mean(log_mag))
        # Blend

        # Inverse FFT

        if s < 1e-10:
            break
        f_new = f_new / s
        f = f_new
    return f
```

```python
def _line_search(f, g, f_best, sc_best, max_iter=50):
    """
    Minimize J((1-t)*f + t*g) over t in [0, 1]
    where f and g are both unit-sum nonnegative vectors.
    Uses golden section search since J is convex in t.
    """
    phi = (1 + math.sqrt(5)) / 2
    a, b = 0.0, 1.0
    c = b - (b - a) / phi
    d = a + (b - a) / phi
    
    fc = _score_unit((1-c)*f + c*g)
    fd = _score_unit((1-d)*f + d*g)
    
    for _ in range(max_iter):
        if b - a < 1e-10:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - (b - a) / phi

            a = c
            c = d
            fc = fd
            d = a + (b - a) / phi

    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0"""
    # Build constraint matrix: A[k, j] = f[k-j] if valid, 0 otherwise
    # (f*g)_k = sum_j g_j * f[k-j]
    A = np.zeros((2 * n - 1, n), dtype=np.float64)

            idx = k - j
            if 0 <= idx < n:

    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    g = np.clip(res.x, 0.0, None)
    if g.sum() < 1e-15:
        return None
    return g
```

```python
def _gs_optimize(f0, stop_time, alpha=0.5, max_iter=50000):
    """Gerchberg-Saxton style optimization"""
    f = f0.copy()
    n = len(f)
    N = next_fast_len(2 * n - 1)
    best_f = f.copy()
    best_sc = _score(f)
    
    for it in range(max_iter):
        if time.monotonic() > stop_time:
            break
        
        # Frequency domain step
        F = rfft(f, N)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target: flat magnitude
        # Preserve total "energy" in frequency domain
        # sum(|F|^2) is preserved by the FFT (Parseval)
        flat_mag = np.full_like(mag, np.sqrt(np.mean(mag**2)))
        
        # Blend
        new_mag = (1 - alpha) * mag + alpha * flat_mag
        # Avoid zero magnitude (phase undefined)
        new_mag = np.maximum(new_mag, 1e-15)
        
        F_new = new_mag * np.exp(1j * phase)
        f_new = irfft(F_new, N)[:n]
        
        # Time domain projection
        f_new = np.clip(f_new, 0.0, None)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s
        
        sc = _score(f_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_f = f_new.copy()
        
        f = f_new
    
    return best_f, best_sc
```

The question "does there exist `f >= 0, sum(f) = 1` with `max_k (f*f)_k <= M`" can be answered by an LP:

But the constraints are quadratic, so this is not an LP. However, we can use the "linearization" trick:
At a given `f_0`, linearize `(f*f)_k <= M` around `f_0`:
`(f*f)_k ≈ (f_0*f_0)_k + 2 * (f_0 * (f - f_0))_k` (first-order Taylor)

This gives: `2 * (f_0 * f)_k <= M - (f_0*f_0)_k + 2*(f_0*f_0)_k = M + (f_0*f_0)_k`

`(f*f)_k ≈ (f_0*f_0)_k + sum_j [f_j f_{0,k-j} + f_{0,j} f_{k-j}]` (first order in `f - f_0`)
`= (f_0*f_0)_k + 2 * (f_0 * (f - f_0))_k`

So the linearized constraint is:
`2 * (f_0 * f)_k - 2 * (f_0*f_0)_k + (f_0*f_0)_k <= M`
`2 * (f_0 * f)_k <= M + (f_0*f_0)_k`

```
minimize    (f - f_0)^2  (or just check feasibility)
```

```python
def _subgrad_fista(f0, stop_time, seed=42, max_iter=100000):

    step = 1.0 / L
    no_improve = 0

    def _project_simplex(v):
        d = len(v)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, d + 1) > (cssv - 1))[0][-1]
        theta = (cssv[rho] - 1) / (rho + 1.0)
        return np.maximum(v - theta, 0)

    for it in range(max_iter):
        if time.monotonic() > stop_time:
            break

        b = _fft_conv(y)

            no_improve = 0

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 30.0

    parent = _get_parent()
    n = len(parent)
    parent_sum = float(parent.sum())

    f_init = parent / max(parent_sum, 1e-10)

    best_seq = parent.copy()
    best_score = _score(best_seq)

    def _try_update(a_unit):
        nonlocal best_seq, best_score
        a_unit = np.clip(a_unit, 0, None)
        s = a_unit.sum()
        if s < 1e-10 or not np.all(np.isfinite(a_unit)):
            return
        a_unit = a_unit / s
        # Scale back to original sum
        a_scaled = a_unit * parent_sum
        a_scaled = np.clip(a_scaled, 0, 1000.0)
        sc = _score(a_scaled)
        if sc < best_score - 1e-13:
            best_score = sc
            best_seq = a_scaled.copy()

    # Phase 1: GS optimization (fast exploration)
    phase1_end = min(t0 + budget * 0.22, deadline - 60)
    try:
        f_gs, sc_gs = _gs_optimize(f_init, phase1_end, seed=seed)
        _try_update(f_gs)
    except Exception:
        pass

    # Phase 2: LP optimization (precise refinement)
    phase2_end = min(t0 + budget * 0.50, deadline - 60)
    if time.monotonic() < phase2_end:
        try:
            f_lp, sc_lp = _lp_optimize(best_seq / max(best_seq.sum(), 1e-10), phase2_end, seed=seed + 1)
            _try_update(f_lp)
        except Exception:
            pass

    # Phase 3: FISTA / subgradient
    phase3_end = min(t0 + budget * 0.68, deadline - 40)
    if time.monotonic() < phase3_end:
        try:
            f_sg, sc_sg = _subgrad_fista(best_seq / max(best_seq.sum(), 1e-10), phase3_end, seed=seed + 2)
            _try_update(f_sg)
        except Exception:
            pass

    # Phase 4: Alternating GS + LP
    phase4_end = min(t0 + budget * 0.82, deadline - 30)
    if time.monotonic() < phase4_end:
        try:
            f_alt, sc_alt = _alternating_optimize(best_seq / max(best_seq.sum(), 1e-10), phase4_end, seed=seed + 3)
            _try_update(f_alt)
        except Exception:
            pass

    # Phase 5: Coordinate descent polish
    phase5_end = min(t0 + budget * 0.93, deadline - 15)
    if time.monotonic() < phase5_end:
        try:
            f_cd, sc_cd = _coord_descent(best_seq / max(best_seq.sum(), 1e-10), phase5_end, seed=seed + 4)
            _try_update(f_cd)
        except Exception:
            pass

    # Phase 6: Final GS polish
    if time.monotonic() < deadline - 8:
        try:
            f_final_gs, _ = _gs_optimize(best_seq / max(best_seq.sum(), 1e-10), deadline - 8, seed=seed + 5)
            _try_update(f_final_gs)
        except Exception:
            pass

    # Validate with exact evaluator
    try:
        final_list = [float(x) for x in best_seq]
        final_sc = float(evaluate_sequence(final_list))
        if math.isfinite(final_sc):
            return final_list
    except Exception:
        pass

    # Fallback
    try:
        fb_list = [float(x) for x in parent]
        if math.isfinite(float(evaluate_sequence(fb_list))):
            return fb_list
    except Exception:
        pass

    return [1.0]
```
