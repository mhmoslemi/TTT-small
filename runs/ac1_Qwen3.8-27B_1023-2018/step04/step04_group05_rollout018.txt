```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """Convolution constraint matrix: A[k,j] = f[k-j] if 0<=k-j<n else 0"""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _lp_direction(f, n, M, timeout=4.0):
    """LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    if M <= 1e-18:
        return None
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-14)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
                     options={"presolve": True, "time_limit": timeout,
                              "dual_feasibility_tolerance": 1e-12,
                              "primal_feasibility_tolerance": 1e-12,
                              "dual_edge_tap": 1e-8})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0, 1e6)
    if g.sum() < 1e-15:
        return None
    return g
```

    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]

        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf

                lo = x1

                break
        best_x = (lo + hi) / 2
    
    f[i] = best_x
    return best_sc, best_x

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _conv_cross(a, b):
    n1, n2 = len(a), len(b)
    fsize = next_fast_len(n1 + n2 - 1)
    F = rfft(a, fsize)
    G = rfft(b, fsize)
    c = irfft(F * G, fsize)[:n1 + n2 - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp(f, n, M, timeout=4.0):
    if M <= 1e-18:
        return None
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-14)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"presolve": True, "time_limit": timeout,
                               "dual_feasibility_tolerance": 1e-12,
                               "primal_feasibility_tolerance": 1e-12,
                               "dual_edge_tap": 1e-8})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0, 1e6)
    return g if g.sum() > 1e-15 else None


def _best_line_search(f, g, n, S):
    """Find best t in [0,1] to minimize score of (1-t)f + t*g."""
    sf, sg = float(f.sum()), float(g.sum())
    if sg < 1e-15:
        return 0.0, math.inf
    g_norm = g / sg * S
    f_norm = f / sf * S
    ff = _conv_full(f_norm)
    gg = _conv_full(g_norm)
    fg = _conv_cross(f_norm, g_norm)

    A_q = ff - 2.0 * fg + gg
    B_q = 2.0 * (fg - ff)
    C_q = ff
    ds = S - S  # both normalized to same sum, so sum is constant = S

    # After normalization both have sum S, so sum(c(t)) = S for all t
    # score(t) = 2n * max_k(A_k t^2 + B_k t + C_k) / S^2
    # Just minimize max_k(A_k t^2 + B_k t + C_k)

    best_t = 0.0
    best_val = float(ff.max())

    ts = np.linspace(0.0, 1.0, 200)
    for t in ts:
        Q = A_q * t * t + B_q * t + C_q
        np.maximum(Q, 0.0, out=Q)
        val = float(Q.max())
        if val < best_val - 1e-15:
            best_val = val
            best_t = t

    if best_t == 0.0 or best_t == 1.0:
        return best_t, best_val

    # Refine with golden section around best_t
    lo = max(0.0, best_t - 5.0 / 200.0)

        np.maximum(Q1, 0.0, out=Q1)
        v1 = float(Q1.max())
        Q2 = A_q * x2**2 + B_q * x2 + C_q
        np.maximum(Q2, 0.0, out=Q2)
        v2 = float(Q2.max())
        if v1 < v2:
            hi = x2
        else:
            lo = x1
        if hi - lo < 1e-10:

    best_t = (lo + hi) / 2.0
    Qb = A_q * best_t**2 + B_q * best_t + C_q
    np.maximum(Qb, 0.0, out=Qb)
    best_val = float(Qb.max())
    return best_t, best_val

                best_f = f.copy()

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_f]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in best_f]))
        except Exception:
            return [1.0]
        if not math.isfinite(final_sc):
            return [1.0]

    return [float(x) for x in best_f]
