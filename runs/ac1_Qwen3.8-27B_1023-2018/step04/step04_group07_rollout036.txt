import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    # Don't clip to 0 - the true convolution of non-negative sequences is non-negative
    # but numerical rounding might give tiny negatives
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _lse_grad(a, tau):
    """Smooth gradient of (1/tau)*log(sum exp(tau*(a*a)_k))"""
    n = len(a)
    c = _conv_full(a)  # length 2n-1
    tc = tau * c
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    weights = exp_tc / exp_tc.sum()
    
    # grad[j] = 2 * sum_{m=0}^{n-1} weights[j+m] * a[m]
    # Compute via FFT (cross-correlation)
    fsize = next_fast_len(3 * n)
    W = rfft(weights[::-1], fsize)
    A = rfft(a, fsize)
    corr = irfft(W * A, fsize)
    # corr[k] = (flip(weights) * a)[k] = sum_m weights[2n-2-m] * a[k-m]
    # We want (weights ⊛ a)[j] = sum_m weights[j+m] * a[m]
    # = (flip(weights) * a)[2n-2-j]
    grad = 2.0 * corr[2*n-2 - np.arange(n)]
    return grad

def _mirror_step(a, grad, eta, floor=1e-30):
    """Mirror descent step on the simplex"""
    n = len(a)
    # a_new = a * exp(-eta * grad) / Z
    log_a = np.where(a > floor, np.log(a / a.sum()), -50.0)  # log of normalized a

    a_new = np.exp(log_new)
    a_new = np.maximum(a_new, floor)
    a_new /= a_new.sum()
    return a_new

def _conv_full(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    return irfft(F * F, fsize)[:2 * n - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _build_A(f, n):
    """A[k, j] = f[k-j] if 0 <= k-j < n else 0, for k=0..2n-2, j=0..n-1"""
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _lp_solve(f, n, M, time_limit=3.0, focus_k=0):
    """Solve LP: max sum(g) s.t. A@f * g <= M, g >= 0
    If focus_k > 0, only use top focus_k constraints."""
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    if focus_k > 0 and focus_k < 2*n-1:
        # Use only constraints where (f*f) is largest (most likely to be binding)
        ff = _conv_full(f)

        A = A[top_idx]
        b_ub = b_ub[top_idx]
    
    try:
