def _convolve_max(a):

    n = len(a)
    size = 2 * n - 1
    # Pad to next power of 2 for FFT efficiency
    fsize = 1
    while fsize < size:

    A = fft(a, fsize)

    # But we need linear conv, and fsize >= 2n-1 ensures no wrap-around

def _eval_score(a):

from scipy.optimize import linprog


def _eval_fast(a):
    """Fast score computation (matches evaluator logic)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)


def _conv_and_max(a):
    """Return (max_conv, argmax_k, full_conv)."""
    c = np.convolve(a, a)
    k = int(np.argmax(c))
    return float(c[k]), k, c


def _build_A(n, a):
    """Convolution matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    diff = np.arange(2 * n - 1)[:, None] - np.arange(n)[None, :]
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[diff[mask]]
    return A


def _lp_best_response(A, M, n, time_limit=4.0):
    """Solve max 1^T g s.t. A@G <= M, g >= 0."""
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    try:
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit,
                               'ipm_gap_tolerance': 1e-9,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    return np.clip(res.x, 0.0, None)


def _line_search(a, g, M_cur, n_steps_coarse=12, n_steps_fine=30):
    """Find best blend t minimizing max conv of (1-t)a + t*g."""
    best_t = 0.0
    best_val = M_cur
    ts = np.linspace(0.0, 1.0, n_steps_coarse)
    for t in ts:

        cs = cand.sum()

            continue
        cand_n = cand / cs
        val = float(np.convolve(cand_n, cand_n).max())
        if val < best_val:
            best_val = val
            best_t = t
    if best_t == 0.0:
        return None, M_cur
    lo = max(0.0, best_t - 0.08)

        mid = 0.5 * (lo + hi)
        cand = (1.0 - mid) * a + mid * g
        cs = cand.sum()

        cand_n = cand / cs
        val = float(np.convolve(cand_n, cand_n).max())

            best_val = val
            best_t = mid
            hi = mid
        else:
            lo = mid
    cand = (1.0 - best_t) * a + best_t * g
    cand = np.maximum(cand, 0.0)
    cs = cand.sum()
    if cs < 1e-12:
        return None, M_cur
    cand /= cs

        for i, init in enumerate(inits):
            if time.monotonic() > t_phase3 - 3.0:
                break
            alloc = min(time.monotonic() + per_init, t_phase3 - 2.0)
            try:
                r = _optimize_lp(init, alloc, seed=5000 + i, max_stuck=60)
                try_update(r)
            except Exception:
                pass

    # ===== Phase 4: Try different lengths =====
    if time.monotonic() < deadline - 60:
        dn_list = [-40, -20, -10, -5, -2, -1, 1, 2, 5, 10, 20, 40]
        for dn in dn_list:
            if time.monotonic() > deadline - 30:
                break
            nt = n_parent + dn
            if nt < 10 or nt > 500:
                continue
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, parent)
            init_new = np.clip(init_new, 0, 1000)
            sn = init_new.sum()
            if sn < 1e-10:
                continue
            init_new /= sn
            try:
                r = _optimize_lp(init_new, time.monotonic() + 15,
                                 seed=10000 + nt, max_stuck=40)
                try_update(r)
            except Exception:
                pass

    # ===== Phase 5: Final deep polish =====
    if time.monotonic() < deadline - 15:
        af = np.clip(best_seq, 0, 1000).astype(np.float64)
        sf = af.sum()
        if sf < 1e-15:
            af = parent / parent.sum()
        else:
            af /= sf
        try:
            r = _optimize_lp(af, deadline, seed=77777, max_stuck=200)
            try_update(r)
        except Exception:
            pass
        # One more round with multi
        if time.monotonic() < deadline - 10:
            try:
                r = _optimize_multi(af, deadline, seed=12345, max_stuck=150)
                try_update(r)
            except Exception:
                pass

    # ===== Final validation with exact evaluator =====
    try:
        candidate_list = [float(x) for x in best_seq]
        fs = float(evaluate_sequence(candidate_list))
        if not np.isfinite(fs) or fs >= 2*n_parent:
            # Fallback to a safe sequence
            fallback = [1.0] * 32
            return fallback
        return candidate_list
    except Exception:
        return [1.0] * 32
