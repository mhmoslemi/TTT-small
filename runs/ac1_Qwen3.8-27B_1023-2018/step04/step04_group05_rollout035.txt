- The max is the max of these linear functions
- S = S_base + a_i
- Score = 2n * max_k(c_k_base + a_i * h_k) / (S_base + a_i)^2

This is a piecewise function where each piece is of the form (A + B*a_i) / (C + a_i)^2. The minimum of each piece can be found analytically, and the overall minimum is at the minimum of all pieces (restricted to the region where that piece is the max).

            return
        try:
            exact = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            exact = math.inf
        if math.isfinite(exact) and exact < best_score - 1e-14:
            best_seq = seq.copy()
            best_score = exact

    # Phase 1: L-BFGS-B from parent and diverse inits (40% of time)
    n = n_p
    phase1_end = t0 + budget * 0.40
    target_sum = math.sqrt(2.0 * n)

    rng_main = np.random.default_rng(seed)
    i_arr = np.arange(n, dtype=np.float64)

    inits = []
    inits.append(parent.copy() / parent.sum() * target_sum)

    # Noisy parent
    sig = 0.15 * max(1.0, float(np.mean(parent)))
    noisy = np.abs(parent + rng_main.normal(0, sig, n)) + 0.01
    inits.append(noisy / noisy.sum() * target_sum)

    # Raised cosine
    rc = 0.5 * (1.0 + np.cos(2.0 * math.pi * i_arr / n))
    inits.append(rc / rc.sum() * target_sum)

    # Gaussian
    gauss = np.exp(-0.5 * ((i_arr - n / 2) / (n / 4)) ** 2)
    inits.append(gauss / gauss.sum() * target_sum)

    # Two-lobe
    two_lobe = (np.exp(-0.5 * ((i_arr - n * 0.25) / (n * 0.1)) ** 2) +
                np.exp(-0.5 * ((i_arr - n * 0.75) / (n * 0.1)) ** 2))
    inits.append(two_lobe / two_lobe.sum() * target_sum)

    n_inits = len(inits)
    time_per_init = max(8.0, (phase1_end - t0) / n_inits)

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 10:
            break
        stop = min(phase1_end - 5, time.monotonic() + time_per_init)
        try:
            res, sc = _lbfgs_optimize(init, n, stop)
            if sc < best_score - 1e-14:
                _try_update(res, sc)
        except Exception:
            pass

    # Phase 2: LP iteration on best (15% of time)
    phase2_end = t0 + budget * 0.55
    if time.monotonic() < phase2_end - 15:
        stop_p2 = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()))
        try:
            res, sc = _lp_iterate(best_seq, len(best_seq), stop_p2, max_iters=30)
            if sc < best_score - 1e-14:
                _try_update(res, sc)
        except Exception:
            pass

    # Phase 3: Coordinate descent polish (30% of time)
    phase3_end = t0 + budget * 0.85
    if time.monotonic() < phase3_end - 15:
        stop_p3 = min(phase3_end - 5, deadline - 20)
        try:
            res, sc = _coordinate_descent(best_seq, len(best_seq), stop_p3, max_rounds=300)
            if sc < best_score - 1e-14:
                _try_update(res, sc)
        except Exception:
            pass

    # Phase 4: Refined coordinate descent (10% of time)
    phase4_end = deadline - 10
    if time.monotonic() < phase4_end - 10:
        try:
            res, sc = _refined_coordinate_descent(best_seq, len(best_seq), phase4_end)
            if sc < best_score - 1e-14:
                _try_update(res, sc)
        except Exception:
            pass

    # Phase 5: Try nearby lengths
    if time.monotonic() < deadline - 30:
        other_lengths = [n - 2, n + 2, n - 4, n + 4, n - 1, n + 1]
        time_left = deadline - 30 - time.monotonic()
        time_per_len = max(5.0, time_left / max(1, len(other_lengths)))

        for L in other_lengths:
            if time.monotonic() > deadline - 20:
                break
            if L < 50 or L > 600:
                continue
            stop = min(deadline - 15, time.monotonic() + time_per_len)
            try:
                # Interpolate best_seq to new length
                x_old = np.linspace(0, 1, len(best_seq))
                x_new = np.linspace(0, 1, L)
                init_l = np.interp(x_new, x_old, best_seq)
                tl = math.sqrt(2.0 * L)
                init_l = init_l / init_l.sum() * tl

                res, sc = _lbfgs_optimize(init_l, L, stop)
                if sc < best_score - 1e-14:
                    _try_update(res, sc)
            except Exception:
                pass

    # Final validation
    try:
        final_sc = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        try:
            final_sc = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_sc = math.inf
        if not math.isfinite(final_sc):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
