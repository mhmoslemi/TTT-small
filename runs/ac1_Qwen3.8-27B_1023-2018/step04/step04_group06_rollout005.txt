def _fast_convolve_max(a):
    """Compute max of convolve(a,a) and its index."""
    c = np.convolve(a, a, mode='full')
    k = int(c.argmax())
    return float(c[k]), k

def _build_conv_matrix(n, a):
    """Build (2n-1) x n matrix where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    # Efficient construction
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # A[k, j] = a[k-j] for k = j, j+1, ..., j+n-1
        A[j:j+n, j] = a
    return A

def _score_norm(a):
    """Score for normalized (sum=1) sequence: 2*n*max_conv."""
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * float(c.max())


def _build_A(n, a):
    """(2n-1) x n convolution matrix: A[k,j]=a[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _grad_active(a, k_star, n):
    """Subgradient of (a*a)_{k_star} projected to sum=0."""
    grad = np.zeros(n)

        if 0 <= idx < n:
            grad[j] = 2.0 * a[idx]
    grad -= grad.mean()
    return grad
