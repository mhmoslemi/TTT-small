    n = len(a)
    F_len = next_fast_len(2*n-1)
    F = rfft(a, F_len)
    c = irfft(F*F, F_len)[:2*n-1]
    return np.maximum(c, 0.0)

    n = len(a)
    S = a.sum()
    if S < 0.01:
        return math.inf
    b = _fft_conv(a)
    m = float(b.max())

        return math.inf

def _conv_matrix(f):
    """Build matrix A where (A @ x) = convolve(f, x)[:2n-1]"""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

The score is $F(a) = \frac{2n \cdot M}{S^2}$ where $M = \max_k (a*a)_k$ and $S = \sum a_i$.

$\nabla_a F = \frac{2n}{S^2} \nabla_a M - \frac{2n M \cdot 2S}{S^4} \nabla_a S$

$= \frac{2n}{S^2} \nabla_a M - \frac{4nM}{S^3} \mathbf{1}$

$= \frac{2n}{S^3} \left( S \nabla_a M - 2M \mathbf{1} \right)$

The subgradient of $M = \max_k (a*a)_k$ at a is:
$\nabla_a M \in \frac{1}{|\mathcal{A}|} \sum_{k \in \mathcal{A}} \nabla_a (a*a)_k$

where $\mathcal{A} = \{k : (a*a)_k = M\}$.

And $\nabla_a (a*a)_k$ has component $i$:
$[\nabla_a (a*a)_k]_i = 2 a_{k-i}$ if $0 \leq k-i < n$, else 0.
