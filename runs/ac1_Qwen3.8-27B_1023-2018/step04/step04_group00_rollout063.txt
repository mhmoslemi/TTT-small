def _get_parent():

def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):

def _lse_optimize(p, stop_time, consider=None):
    """Optimize using log-sum-exp smoothed objective with L-BFGS-B."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300), dtype=np.float64)
        return uu - float(np.mean(uu))

    def softmax(uu):
        m = float(np.max(uu))
        e = np.exp(uu - m)
        ss = float(e.sum())
        if not math.isfinite(ss) or ss <= 0:
            return np.full(n, 1.0 / n)
        return e / ss

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = softmax(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        T_eff = max(1e-300, T)
        scaled = (c - cmax) / T_eff
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T_eff * float(np.log(sum_exp / c.size))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    u = p_to_u(p)
    c = cfast(p)
    cmax = float(c.max()) if c.size else 1.0
    if not math.isfinite(cmax) or cmax <= 0:
        cmax = 1.0

    best_u = u.copy()
    best_f = float('inf')

    stages = (
        (0.15, 0.12),
        (0.05, 0.15),
        (0.015, 0.20),
        (0.004, 0.20),
        (0.001, 0.22),
        (0.0003, 0.11),
    )

    avail = max(0.0, float(stop_time) - time.monotonic() - 5.0)
    if avail <= 5:
        return p

    total_w = sum(w for _, w in stages)
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4:
            break
        cum += wt
        end_stage = min(float(stop_time) - 3.0, time.monotonic() + avail * (cum / total_w))
        if end_stage <= now + 4:
            continue
        T = max(1e-300, frac * cmax)
        maxit = 40 if n <= 2000 else 25
        opts = {
            "maxiter": maxit, "gtol": 1e-12, "ftol": 1e-15,
            "maxls": 30, "maxcor": 25, "maxdisp": 0, "disp": False,
        }
        stale = 0
        while time.monotonic() < end_stage - 2:
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    try:
                        fv = float(res.fun)
                    except Exception:
                        fv = math.inf
                    if math.isfinite(fv) and fv < best_f:
                        best_f = fv
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, "status", 1) == 0:
                break
        p_cur = softmax(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc = _score_fast(p_cur)
        if math.isfinite(sc):
            p = p_cur.copy()
            u = p_to_u(p)
            c = cfast(p)
            cmax = float(c.max()) if c.size else cmax
        if consider is not None:

            except Exception:
                pass

def _tilt_uniform(n, tilt, seed=0):
    """Create a slightly tilted uniform sequence."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    p = 1.0 + tilt * (2.0 * x - 1.0)
    # Add small random perturbation
    p += 0.01 * rng.standard_normal(n)
    p = np.maximum(p, 0.01)
    p /= p.sum()
    return p

            p_lse2 = p_lse2 / p_lse2.sum()
            consider(p_lse2)

    # Phase 4: Final coordinate descent polish (~150s)
    cd2_end = min(hard - 10.0, t0 + budget - 30.0)
    if cd2_end > time.monotonic() + 30:
        p_cd2, _ = _coord_descend(best_arr, cd2_end, consider=consider, G_base=40, seed=seed + 99999)
        if p_cd2 is not None and p_cd2.size == best_arr.size and np.all(np.isfinite(p_cd2)) and p_cd2.sum() > 0:
            p_cd2 = p_cd2 / p_cd2.sum()
            consider(p_cd2)

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.ones(64, dtype=np.float64) / 64.0
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
