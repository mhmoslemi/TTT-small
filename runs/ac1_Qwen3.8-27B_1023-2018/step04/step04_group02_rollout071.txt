def _as_norm(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        a = np.zeros(n, dtype=float)
    elif a.size != n:
        if a.size < n:
            a = np.pad(a, (0, n - a.size))
        else:
            a = a[:n]
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.ones(n, dtype=float) / float(n)
    return a / s


def _score_seq(a):
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        return math.inf
    n = len(a)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _exact_score(a):
    a = np.asarray(a, dtype=float).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    try:
        return float(evaluate_sequence([float(x) for x in a]))
    except NameError:
        return _score_seq(a)
    except Exception:
        return math.inf


def _lp_direction(f, M, timeout=0.6):
    n = len(f)
    if M <= 1e-300:
        return f.copy()
    A = np.zeros((2 * n - 1, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n, dtype=float)
    b_ub = np.full(2 * n - 1, float(M), dtype=float)
    bounds = [(0.0, None)] * n

    res = None
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": timeout},
        )
    except Exception:
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
        except Exception:
            res = None

    if res is None or res.x is None:
        return None
    if res.status not in (0, 1):
        return None

    g = np.asarray(res.x, dtype=float).ravel()
    if not np.all(np.isfinite(g)):
        return None
    g = np.clip(g, 0.0, None)
    gs = float(g.sum())
    if gs <= 1e-15:
        return None
    return g / gs


def _fast_cd(f, stop, max_sweeps=1000, seed=0):
    n = len(f)
    f = _as_norm(f, n).copy()
    b = np.convolve(f, f)
    cur = 2.0 * n * float(b.max())
    best = f.copy()
    best_sc = cur

    if time.monotonic() >= stop - 1.0:
        return best, best_sc

    tmp = np.empty(n, dtype=float)
    rng = np.random.default_rng(seed)

    rel_muls = (0.25, 0.5, 0.75, 1.25, 1.6, 2.5, 4.0)
    abs_muls = (0.02, 0.05, 0.1, 0.3, 1.0)
    fixed_vals = (0.5, 1.0, 2.0, 5.0)
    fine_r = (0.8, 0.9, 0.95, 1.05, 1.1, 1.2)
    upper = 50.0
    L = 2 * n - 1

    def eval_v(v, i, orig, out_base):
        if v < 0.0 or v > upper:
            return math.inf
        delta = v - orig
        if delta == 0.0 or abs(delta) <= 1e-11 * (1.0 + abs(orig)):
            return math.inf
        S1 = 1.0 - orig + v
        if S1 <= 1e-12:
            return math.inf
        np.copyto(tmp, b[i:i + n])
        tmp += (2.0 * delta) * f
        tmp[i] += delta * delta
        tm = float(tmp.max())
        mtemp = out_base if out_base > tm else tm
        if not math.isfinite(mtemp):
            return math.inf
        if mtemp < 0.0:
            mtemp = 0.0
        return (2.0 * n * mtemp) / (S1 * S1)

    for _ in range(max_sweeps):
        if time.monotonic() >= stop - 1.5:
            break

        s = float(f.sum())
        if s < 1e-12:
            break
        f /= s
        b = np.convolve(f, f)
        cur = 2.0 * n * float(b.max())
        if cur < best_sc:
            best_sc = cur
            best = f.copy()

        mean = float(f.mean())
        improved = False
        clean = 0
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() >= stop - 1.5:
                break

            if clean >= 50:
                s = float(f.sum())
                if s < 1e-12:
                    break
                f /= s
                b = np.convolve(f, f)
                cur = 2.0 * n * float(b.max())
                mean = float(f.mean())
                if cur < best_sc:
                    best_sc = cur
                    best = f.copy()
                clean = 0
            clean += 1

            orig = float(f[i])
            left = float(b[:i].max()) if i > 0 else 0.0
            right = float(b[i + n:].max()) if i + n < L else 0.0
            out_base = left if left > right else right

            coord_best = cur
            coord_v = orig

            if orig > 1e-14:
                sc = eval_v(0.0, i, orig, out_base)
                if sc < coord_best:
                    coord_best = sc
                    coord_v = 0.0
                for m in rel_muls:
                    v = orig * m
                    sc = eval_v(v, i, orig, out_base)
                    if sc < coord_best:
                        coord_best = sc
                        coord_v = v

            for m in abs_muls:
                v = mean * m
                sc = eval_v(v, i, orig, out_base)
                if sc < coord_best:
                    coord_best = sc
                    coord_v = v

            for v in fixed_vals:
                sc = eval_v(v, i, orig, out_base)
                if sc < coord_best:
                    coord_best = sc
                    coord_v = v

            center = coord_v
            if center > 1e-8 and center < 0.9 * upper:
                for r in fine_r:
                    sc = eval_v(center * r, i, orig, out_base)
                    if sc < coord_best:
                        coord_best = sc
                        coord_v = center * r

            if coord_v != orig:
                delta = coord_v - orig
                S1 = 1.0 - orig + coord_v
                b[i:i + n] += (2.0 * delta) * f
                b[2 * i] += delta * delta
                f[i] = coord_v
                inv = 1.0 / S1
                f *= inv
                b *= inv * inv
                cur = coord_best
                improved = True
                if cur < best_sc:
                    best_sc = cur
                    best = f.copy()

        if not improved:
            break

    s = float(f.sum())
    if s > 1e-12:
        f = f / s
        b = np.convolve(f, f)
        sc = 2.0 * n * float(b.max())
        if sc < best_sc:
            best = f.copy()
            best_sc = sc

    return best, best_sc


def _run_lp(f0, stop, seed, max_stall=180, lp_timeout=0.6):

        if it - best_it > 4000:

        s = float(f.sum())

            else:

            stall += 4

            noise_scale = 0.02
            f = _as_norm(best + rng.normal(0.0, noise_scale, n), n)
            cur = _score_seq(f)
            t = 0.3
            stall = 0
            it += 1
            continue

        steps = [t, 0.5 * t, 0.25 * t, min(0.9, 1.5 * t), 0.1]

        for sx in steps:

        for sx in uniq:

            fn = np.clip(fn, 0.0, None)
            ssum = float(fn.sum())

            fn /= ssum
            bb = np.convolve(fn, fn)

            if not math.isfinite(mm):
                continue
            sc = 2.0 * n * mm
            if sc < best_step_sc - 1e-12:

                best_step_f = fn

        if best_step_f is not None:

            cur = best_step_sc
            if best_step_s is not None:

                    t = max(0.05, 0.8 * t)

                else:
                    t = min(0.8, max(0.05, 1.05 * t))
            stall = 0
            if cur < best_sc - 1e-14:
                best_sc = cur
                best = f.copy()
                best_it = it
        else:
            t = max(0.05, 0.7 * t)
            stall += 1
            if stall > max_stall_cur:
                cd_stop = min(stop - 10.0, time.monotonic() + 30.0)
                if cd_stop > time.monotonic() + 2.0:
                    cf, csc = _fast_cd(best, cd_stop, max_sweeps=20, seed=seed + it)
                    if csc < best_sc:
                        best = cf.copy()
                        best_sc = csc

                noise_scale = 0.02 if stall < 2 * max_stall else 0.04

                    if csc < cur:
                        f = cf.copy()
                        cur = csc
                    if csc < best_sc:
                        best_sc = csc
                        best = cf.copy()
                        best_it = it

        it += 1

        if it % 40 == 0 and time.monotonic() < stop - 25.0:
            cd_stop = min(stop - 20.0, time.monotonic() + 15.0)
            if cd_stop > time.monotonic():
