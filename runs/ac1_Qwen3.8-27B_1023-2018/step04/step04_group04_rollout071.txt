    if not math.isfinite(m):
        return math.inf
    return float(2.0 * a.size * m / (s * s))


def _eval_w(ws, base, a, self_idx, base_i, out_max, s0, n):
    ws = np.asarray(ws, dtype=np.float64)
    lam = (1.0 - ws) / s0
    lam2 = lam * lam
    inside = base[None, :] * lam2[:, None]
    inside += a[None, :] * (2.0 * lam * ws)[:, None]
    inside[:, self_idx] = lam2 * base_i + ws * ws
    imax = inside.max(axis=1)
    if out_max > 0.0:
        imax = np.maximum(imax, lam2 * out_max)
    imax = np.maximum(imax, 0.0)
    return 2.0 * n * imax


def _coord_step(a, b, i, cur_score):
    n = a.size
    old = float(np.clip(a[i], 0.0, 1.0))
    s0 = 1.0 - old
    if s0 <= 1e-12:
        return old

    m = 2 * n - 1
    d = 2 * i

    base = b[i:i + n].copy()
    if not np.all(np.isfinite(base)):
        return old
    base -= 2.0 * old * a
    base_i = float(b[d]) - old * old
    base[i] = base_i
    if not math.isfinite(base_i) or not np.all(np.isfinite(base)):
        return old

    out_max = 0.0
    if i > 0:
        out_max = max(out_max, float(np.max(b[:i])))
    if i + n < m:
        out_max = max(out_max, float(np.max(b[i + n:])))
    if not math.isfinite(out_max):
        return old
    out_max = max(0.0, out_max)

    seg_a = a[i:i + n]
    if not np.all(np.isfinite(seg_a)):
        return old

    def eval_one(w):
        w = float(np.clip(w, 0.0, 1.0))
        lam = (1.0 - w) / s0
        lam2 = lam * lam
        tmp = base + (2.0 * lam * w) * a
        tmp[self_idx := i] = lam2 * base_i + w * w
        mx = float(np.max(tmp))
        ox = lam2 * out_max
        if ox > mx:
            mx = ox
        if mx < 0.0:
            mx = 0.0
        return 2.0 * n * mx

    sc_old = eval_one(old)
    if not math.isfinite(sc_old):
        return old

    n1 = 40
    vs1 = np.linspace(0.0, 1.0, n1)
    sc1 = _eval_w(vs1, base, a, i, base_i, out_max, s0, n)
    if not np.all(np.isfinite(sc1)):
        return old
    k1 = int(np.argmin(sc1))
    lo1 = float(vs1[max(0, k1 - 1)])
    hi1 = float(vs1[min(n1 - 1, k1 + 1)])

    n2 = 112
    vs2 = np.linspace(lo1, hi1, n2)
    sc2 = _eval_w(vs2, base, a, i, base_i, out_max, s0, n)
    if not np.all(np.isfinite(sc2)):
        return old
    k2 = int(np.argmin(sc2))
    lo2 = float(vs2[max(0, k2 - 1)])
    hi2 = float(vs2[min(len(vs2) - 1, k2 + 1)])

    if hi2 - lo2 <= 1e-13:
        vbest = 0.5 * (lo2 + hi2)
    else:
        gl = lo2
        gh = hi2
        aphi = (math.sqrt(5.0) - 1.0) / 2.0
        c = gl + (1.0 - aphi) * (gh - gl)
        dd = gl + aphi * (gh - gl)
        fc = eval_one(c)
        fd = eval_one(dd)
        for _ in range(28):
            if fc <= fd:
                gh = dd
                dd = c
                fd = fc
                c = gl + (1.0 - aphi) * (gh - gl)
                fc = eval_one(c)
            else:
                gl = c
                c = dd
                fc = fd
                dd = gl + aphi * (gh - gl)
                fd = eval_one(dd)
        vbest = 0.5 * (gl + gh)

    cands = np.asarray([vbest, old, 0.0, 1.0, float(vs2[k2])], dtype=np.float64)
    cands = np.clip(cands, 0.0, 1.0)
    cands = np.unique(cands)
    scs = np.asarray([eval_one(float(w)) for w in cands], dtype=np.float64)
    if not np.all(np.isfinite(scs)):
        return old

    bi = int(np.argmin(scs))
    vbest = float(cands[bi])
    sbest = float(scs[bi])

    if sbest < sc_old - 1e-13 and abs(vbest - old) > 1e-12:
        return vbest
    return old


def _apply_coord(a, b, i, w):
    n = a.size
    old = float(a[i])
    s0 = 1.0 - old
    if s0 <= 1e-14:
        return None
    w = float(np.clip(w, 0.0, 1.0))
    d = 2 * i
    lam = (1.0 - w) / s0
    lam2 = lam * lam

    base_full = b.copy()
    base_full[i:i + n] -= 2.0 * old * a
    base_full[d] = float(b[d]) - old * old
    if not np.all(np.isfinite(base_full)):
        return None

    new_a = lam * a
    new_a[i] = w
    na = float(new_a.sum())
    if na <= 1e-14:
        return None
    new_a /= na

    new_b = base_full * lam2
    new_b[i:i + n] += (2.0 * lam * w) * a
    new_b[d] = float(base_full[d]) * lam2 + w * w
    new_b = np.where(np.isfinite(new_b), new_b, 0.0)
    np.maximum(new_b, 0.0, out=new_b)
    sb = float(new_b.sum())
    if sb > 0.0 and math.isfinite(sb):
        new_b /= sb

    if not np.all(np.isfinite(new_a)) or not np.all(np.isfinite(new_b)):
        return None
    return new_a, new_b


def _polish(a, stop_time, seed):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64), nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if n == 0 or s < 1e-12:
        return None, math.inf
    a = a / s

    best_a = a.copy()
    best_s = _score_arr(a)
    if not math.isfinite(best_s):
        return None, math.inf

    cur_a = a.copy()
    b = _fft_conv(cur_a)
    np.maximum(b, 0.0, out=b)
    sb = float(b.sum())
    if sb > 1e-300:
        b /= sb
    cur_s = 2.0 * n * float(np.max(b))
    if not math.isfinite(cur_s):
        return best_a, best_s

    rng = np.random.default_rng(seed)
    sweeps = 0
    stagn = 0
    perturb = 0
    max_perturb = 10

    while time.monotonic() < stop_time:
        sweeps += 1

        b = _fft_conv(cur_a)
        np.maximum(b, 0.0, out=b)
        sb = float(b.sum())
        if sb > 1e-300:
            b /= sb
        cur_s = 2.0 * n * float(np.max(b))
        if math.isfinite(cur_s) and cur_s < best_s - 1e-12:
            best_s = cur_s
            best_a = cur_a.copy()

        improved = 0
        order = rng.permutation(n)

        for it, ii in enumerate(order):
            if time.monotonic() > stop_time:
                break
            w = _coord_step(cur_a, b, int(ii), cur_s)
            if w is None or abs(w - cur_a[int(ii)]) < 1e-12:
                continue

            updated = _apply_coord(cur_a, b, int(ii), w)
            if updated is None:
                continue
            cur_a, b = updated
            np.maximum(b, 0.0, out=b)
            sb = float(b.sum())
            if sb > 1e-300:
                b /= sb
            cur_s = 2.0 * n * float(np.max(b))
            if math.isfinite(cur_s) and cur_s < best_s - 1e-12:
                best_s = cur_s
                best_a = cur_a.copy()
            improved += 1

        if improved == 0:
            stagn += 1
            if stagn >= 3:
                if perturb < max_perturb and time.monotonic() < stop_time - 10.0:
                    sigma = 0.05 + 0.02 * perturb
                    p = best_a * np.exp(rng.normal(0.0, sigma, n))
                    p = np.where(np.isfinite(p), p, 0.0)
                    p = np.maximum(p, 0.0)
                    ps = float(p.sum())
                    if ps > 1e-12:
                        cur_a = p / ps
                        b = _fft_conv(cur_a)
                        np.maximum(b, 0.0, out=b)
                        sb = float(b.sum())
                        if sb > 1e-300:
                            b /= sb
                        cur_s = 2.0 * n * float(np.max(b))
                        perturb += 1
                        stagn = 0
                    else:
                        break
                else:
                    break
        else:
            stagn = 0

    final_s = _score_arr(best_a)
    if not math.isfinite(final_s):
        return None, math.inf
    return best_a, final_s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    margin = max(3.0, min(30.0, 0.03 * budget))
    deadline = t0 + budget - margin

    parent = _get_parent()
    n = parent.size

    inits = []
    best_seq = None
    best_score = math.inf

    def consider(a):
        nonlocal best_seq, best_score
        a = np.nan_to_num(np.asarray(a, dtype=np.float64), nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01 or not np.all(np.isfinite(a)):
            return
        a = a / s
        sc = _score_arr(a)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()

    def add_init(a):
        a = np.nan_to_num(np.asarray(a, dtype=np.float64), nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size != n:
            return
        s = float(a.sum())
        if s < 1e-12:
            return
        a = a / s
        for _, aa in inits:
            if np.linalg.norm(a - aa) < 1e-5:
                return
        sc = _score_arr(a)
        if math.isfinite(sc):
            inits.append((sc, a.copy()))
        consider(a)

    rng0 = np.random.default_rng(int(seed) if seed is not None else 42)

    add_init(parent)

    x = np.linspace(0.0, 1.0, n)
    for alpha in (0.5, 0.75, 1.0, 1.25, 1.5, 1.75, 2.0, 2.5, 3.0, 4.0):
        add_init((1.0 + x) ** alpha)
    for beta in (0.5, 1.0, 1.5, 2.0, 3.0):
        add_init(np.exp(beta * x))
    for beta in (2.0, 4.0, 6.0):
        add_init(np.exp(-beta * (x - 0.78) ** 2))
    for sig in (0.015, 0.04, 0.09, 0.16):
        p = parent / max(float(parent.sum()), 1e-12)
        add_init(p * np.exp(rng0.normal(0.0, sig, n)))

    inits.sort(key=lambda z: z[0])
    inits = inits[:8]

    if not inits:
        fallback = np.ones(n) / float(n)
        consider(fallback)

    if best_seq is None:
        fallback = parent / max(float(parent.sum()), 1e-12)
        if not np.all(np.isfinite(fallback)) or float(fallback.sum()) < 0.01:
            fallback = np.ones(max(16, n)) / float(max(16, n))
        consider(fallback)
        if best_seq is None:
            return [1.0]

    stop_points = [
        t0 + budget * 0.34,
        t0 + budget * 0.49,
        t0 + budget * 0.61,
        t0 + budget * 0.71,
        t0 + budget * 0.79,
        t0 + budget * 0.84,
        t0 + budget * 0.865,
        t0 + budget * 0.88,
    ]

    for idx, (_, init_arr) in enumerate(inits):
        stop_i = min(stop_points[min(idx, len(stop_points) - 1)], deadline - 12.0)
        if time.monotonic() >= stop_i or time.monotonic() >= deadline - 18.0:
            break
        try:
            polished, sc = _polish(init_arr.copy(), stop_i, int(seed) + 1000 * (idx + 1))
        except Exception:
            polished = None
        if polished is not None and np.all(np.isfinite(polished)):
            consider(polished)

    # Final polish on the globally best candidate.
    try:
        polished, sc = _polish(best_seq.copy(), deadline - 4.0, int(seed) + 987654)
    except Exception:
        polished = None
    if polished is not None and np.all(np.isfinite(polished)):
        consider(polished)

    seq = [float(x) for x in best_seq]
    try:
        if evaluate_sequence(seq) < math.inf:
            return seq
    except Exception:
        pass

    try:
        if evaluate_sequence([1.0]) < math.inf:
            return [1.0]
    except Exception:
        pass

    return seq
