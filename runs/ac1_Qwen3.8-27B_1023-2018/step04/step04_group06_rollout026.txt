def _build_A(n, a):

def _lp_best_response(A, M, n, time_limit=5.0):
    """max 1^T g s.t. A @ g <= M, g >= 0."""
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10,
                               'mip_rel_gap': 1e-10})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    return np.clip(res.x, 0.0, None)


def _lp_min_crosscorr(A, n, time_limit=5.0):
    """min t s.t. A @ g <= t, sum(g) = 1, g >= 0."""
    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: minimize t -> c = [0,...,0,1]
    # Inequality: A @ g - t <= 0 for each row -> [A, -1] @ [g;t] <= 0
    # Equality: sum(g) = 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ineq = np.hstack([A, -np.ones((A.shape[0], 1))])
    b_ineq = np.zeros(A.shape[0])
    A_eq = np.hstack([np.ones(n), [[0.0]]])
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,

def _line_search(a, g, M_cur, n_evals=30):
    """Find best blend t for (1-t)*a + t*g minimizing max conv. Returns (cand, val) or (None, M_cur)."""
    n = a.size
    best_t = 0.0
    best_val = M_cur

    # Coarse search
    ts_coarse = np.linspace(0.0, 1.0, 15)
    for t in ts_coarse:
        cand = (1.0 - t) * a + t * g
        cs = cand.sum()
        if cs < 1e-12:
            continue
        cand_n = cand / cs
        val = float(np.convolve(cand_n, cand_n).max())
        if val < best_val:
            best_val = val
            best_t = t

    if best_t == 0.0:
        return None, M_cur

    # Fine search (golden section around best)
    lo = max(0.0, best_t - 0.1)
    hi = min(1.0, best_t + 0.1)
    for _ in range(25):
        if hi - lo < 1e-8:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        c1 = (1.0 - m1) * a + m1 * g
        c1 /= c1.sum()
        v1 = float(np.convolve(c1, c1).max())
        c2 = (1.0 - m2) * a + m2 * g
        c2 /= c2.sum()
        v2 = float(np.convolve(c2, c2).max())
        if v1 < v2:
            hi = m2
        else:
            lo = m1
        m = 0.5 * (lo + hi)
        cm = (1.0 - m) * a + m * g
        cm /= cm.sum()
        vm = float(np.convolve(cm, cm).max())
        if vm < best_val:
            best_val = vm
            best_t = m

    cand = (1.0 - best_t) * a + best_t * g
    cand = np.maximum(cand, 0.0)
    cand_s = cand.sum()
    if cand_s < 1e-12:
        return None, M_cur
    cand /= cand_s
    return cand, best_val


def _subgradient_step(a, k_star, eta):
    """One projected subgradient step for minimizing max(conv)."""
    n = a.size
    # Gradient of conv(a,a)[k_star] wrt a
    grad = np.zeros(n)
    j_lo = max(0, k_star - n + 1)
    j_hi = min(n, k_star + 1)
    for j in range(j_lo, j_hi):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] += 2.0 * a[idx]
    # Project onto simplex: a - eta * grad, then project
    v = a - eta * grad
    # Euclidean projection onto simplex {x: sum(x)=a.sum(), x>=0}
    target = a.sum()
    # Projection: find lambda such that (v_i - lambda)+ sums to target
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (css[-1] - target))[0][-1]
    lambda_val = (css[rho] - target) / (rho + 1)
    proj = np.maximum(v - lambda_val, 0.0)
    proj_s = proj.sum()
    if proj_s < 1e-15:
        return a.copy()
    proj /= proj_s * proj_s  # No, just normalize to same sum
    proj = proj / proj.sum() * target
    return np.maximum(proj, 0.0)


def _subgradient_step_simplex(a, k_star, eta):
    """Projected subgradient step onto the probability simplex."""
    n = a.size
    grad = np.zeros(n)
    j_lo = max(0, k_star - n + 1)
    j_hi = min(n, k_star + 1)
    for j in range(j_lo, j_hi):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] += 2.0 * a[idx]
    # a is on simplex (sum=1, >=0)
    v = a - eta * grad
    # Project onto simplex
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    inds = np.arange(1, n + 1)
    cond = v_sorted * inds > (css[-1] - 1.0)
    if not np.any(cond):
        return a.copy()
    rho = np.nonzero(cond)[0][-1]
    theta = (css[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    ps = proj.sum()
    if ps < 1e-15:
        return a.copy()
    return proj / ps


def _coordinate_descent(a, max_iters=200, tol=1e-10):
    """Cycle through coordinates, optimizing each one greedily."""
    n = a.size
    a = a.copy()
    a_s = a.sum()
    a = a / a_s  # normalize to sum=1

    for it in range(max_iters):
        improved = False
        for i in range(n):
            old_val = a[i]
            # conv(a,a) depends on a[i] quadratically
            # For each lag k, conv[k] = c_k + b_k * a[i] + a[i]^2 * q_k
            # where q_k = 1 if 0 <= k-i < n (the self-term), b_k = a[k-i] + (a[i] if symmetric)
            # Actually: conv[k] = sum_j a[j]*a[k-j]
            # Terms involving a[i]: a[i]*a[k-i] (j=i) and a[k-i]*a[i] (j=k-i) if k-i != i
            # So conv[k] = C_k + a[i] * (a[k-i] + (a[k-i] if i != k-i else 0)) + (a[i]^2 if i == k-i)
            # = C_k + 2*a[k-i]*a[i] + a[i]^2*(1 if i==k-i else 0)

            # For each k: conv[k] = base_k + alpha_k * a[i] + beta_k * a[i]^2
            # where alpha_k = 2*a[k-i] (if 0<=k-i<n) else 0
            #       beta_k = 1 if k-i == i (i.e., k = 2i) and 0<=k<n, else 0
            #       base_k = conv[k] - alpha_k*old_val - beta_k*old_val^2

            k_range_lo = max(0, i)  # k-i >= 0 -> k >= i
            k_range_hi = min(2 * n - 1, i + n - 1)  # k-i < n -> k < i+n

            # Compute base, alpha, beta for affected lags
            ks = np.arange(k_range_lo, k_range_hi + 1)
            idxs = ks - i  # 0 to n-1
            alpha = 2.0 * np.array([a[j] for j in idxs])
            beta = np.where(ks == 2 * i, 1.0, 0.0)

            # Current conv values at these lags
            conv_cur = np.convolve(a, a)
            base = conv_cur[ks] - alpha * old_val - beta * old_val * old_val

            # We want to minimize max_k (base_k + alpha_k * x + beta_k * x^2)
            # subject to x >= 0
            # The max is at most M_cur (current max)
            # Try to find x that reduces the peak
            M_cur = float(conv_cur.max())

            # Binary search for best x
            # The function h(x) = max_k (base_k + alpha_k*x + beta_k*x^2) is convex in x
            # (each term is convex since alpha_k >= 0, beta_k >= 0 for nonneg a)
            # So h(x) is convex, minimum can be found by ternary search or gradient

            # Simple approach: try a few values
            best_x = old_val
            best_h = M_cur

            # Candidate values: old_val, 0, and some fractions
            candidates = [old_val, 0.0, old_val * 0.5, old_val * 1.5]
            # Also try to reduce the active constraint
            k_act = int(conv_cur[ks].argmax()) + k_range_lo
            active_base = base[k_act - k_range_lo]
            active_alpha = alpha[k_act - k_range_lo]
            active_beta = beta[k_act - k_range_lo]
            if active_alpha + 2 * active_beta * old_val > 0:
                # Optimal x for this constraint: (base + alpha*x + beta*x^2) = M_cur
                # May not have a simple solution, just add some candidates
                candidates.append(old_val * 0.7)
                candidates.append(old_val * 0.8)
                candidates.append(old_val * 1.1)
                candidates.append(old_val * 1.3)

            # Evaluate
            for x in candidates:
                if x < 0 or x > 1000:
                    continue
                # New max at affected lags
                vals = base + alpha * x + beta * x * x
                affected_max = float(vals.max())
                # Unaffected lags stay the same
                # The new overall max is max(affected_max, max of unaffected)
                # For speed, just check affected_max < M_cur AND old lag was the global max
                if affected_max < best_h:
                    # Need to verify globally
                    a[i] = x
                    new_conv = np.convolve(a, a)
                    new_max = float(new_conv.max())
                    a[i] = old_val
                    if new_max < best_h:
                        best_h = new_max
                        best_x = x

            if best_x != old_val:
                a[i] = best_x
                improved = True

        if not improved:
            break

    a_s = a.sum()
    if a_s > 1e-15:
        a /= a_s
    return a


def _optimize_lp(a0, stop_time, seed=42, use_cd=False):
    """Main LP-based optimization."""
    a0 = np.asarray(a0, dtype=np.float64)
    n = a0.size
    a = np.clip(a0, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a /= s

    best_a = a.copy()
    best_M, best_k = _convolve_max(a)
    best_score = 2.0 * n * best_M

    rng = np.random.default_rng(seed)
    A = _build_A(n, a)
    stuck = 0
    total_iters = 0

    while time.monotonic() < stop_time:
        total_iters += 1
        conv = np.convolve(a, a)
        M = float(conv.max())
        k_star = int(conv.argmax())

        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 150:
            # Perturbation
            sigma = 0.05 + 0.02 * min(stuck // 50, 5)
            mode = int(rng.integers(0, 4))
            xx = np.arange(n, dtype=np.float64)
            if mode == 0:
                a = a * np.exp(rng.normal(0, sigma, n))
            elif mode == 1:
                freq = rng.uniform(1.0, 8.0)
                a = a * (1.0 + sigma * np.sin(2.0 * np.pi * freq * xx / n + rng.uniform(0, 6.28)))
            elif mode == 2:
                # Targeted perturbation: increase values at low-convolution positions
                conv_flat = conv / max(conv.max(), 1e-15)
                a = a * (1.0 + sigma * (1.0 - conv_flat[:n]))
            else:
                # Swap perturbation
                i1, i2 = rng.integers(0, n, 2)
                delta = 0.1 * a.sum()
                a[i1] = max(0, a[i1] - delta)
                a[i2] += delta
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            A = _build_A(n, a)
            stuck = 0
            continue

        # LP best response
        made_move = False
        for M_factor in [1.0, 1.0 + 1e-9, 1.0 + 1e-7]:
            g = _lp_best_response(A, M * M_factor, n, time_limit=4.0)
            if g is None:
                continue
            gs = g.sum()
            if gs < 1e-10:
                continue
            g_n = g / gs
            if float(np.max(np.abs(g_n - a))) < 1e-12:
                continue
            cand, val = _line_search(a, g_n, M)
            if cand is not None and val < M * (1.0 - 1e-11):
                a = cand
                A = _build_A(n, a)
                made_move = True
                break

        if not made_move:
            # Try min-crosscorr LP
            g2 = _lp_min_crosscorr(A, n, time_limit=4.0)
            if g2 is not None:
                g2s = g2.sum()
                if g2s > 1e-10:
                    g2n = g2 / g2s
                    if float(np.max(np.abs(g2n - a))) > 1e-10:
                        cand, val = _line_search(a, g2n, M)
                        if cand is not None and val < M * (1.0 - 1e-11):
                            a = cand
                            A = _build_A(n, a)
                            made_move = True

        if not made_move:
            # Subgradient step
            eta = 1.0 / (2.0 * M * n + 1.0)
            a_new = _subgradient_step_simplex(a, k_star, eta)
            conv_new = np.convolve(a_new, a_new)
            M_new = float(conv_new.max())
            if M_new < M * (1.0 - 1e-12):
                a = a_new
                A = _build_A(n, a)
                made_move = True

        if not made_move:
            # Small kick
            a = a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            A = _build_A(n, a)

    return best_a


def _optimize_cd(a0, stop_time):
    """Coordinate descent optimization."""
    a = np.asarray(a0, dtype=np.float64)
    n = a.size
    a = np.clip(a, 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s

    best_a = a.copy()
    best_M, _ = _convolve_max(a)
    best_score = 2.0 * n * best_M

    while time.monotonic() < stop_time:
        a = _coordinate_descent(a, max_iters=50)
        M, k = _convolve_max(a)
        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()
        else:
            # Try small perturbation + CD
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            a_pert = a * np.exp(rng.normal(0, 0.03, n))
            a_pert = np.maximum(a_pert, 1e-12)
            a_pert /= a_pert.sum()
            a = _coordinate_descent(a_pert, max_iters=30)
            M, k = _convolve_max(a)
            if M < best_M * (1.0 - 1e-12):
                best_M = M
                best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 20.0  # safety margin

    parent = _get_parent()
    n_parent = parent.size

    best_seq = np.clip(parent.copy(), 0, 1000)
    best_score = _eval_score(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(32, dtype=np.float64)
        best_score = _eval_score(best_seq)

    def try_update(cand):
        nonlocal best_seq, best_score
        try:
            cand = np.asarray(cand, dtype=np.float64)
            cand = np.clip(cand, 0, 1000)
            if cand.size == 0 or not np.all(np.isfinite(cand)) or cand.sum() < 0.01:
                return
            sc = _eval_score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand.copy()
        except Exception:
            pass

    # ===== Phase 1: LP optimization on parent =====
    t1 = t0 + min(400.0, budget * 0.35)
    try:
        r = _optimize_lp(parent, t1, seed=1)
        try_update(r)
    except Exception:
        pass

    # ===== Phase 2: Coordinate descent refinement =====
    t2 = t0 + min(550.0, budget * 0.45)
    if time.monotonic() < t2 - 30:
        try:
            r = _optimize_cd(best_seq, t2, )
            try_update(r)
        except Exception:
            pass

    # ===== Phase 3: More LP with different seeds =====
    t3 = t0 + min(800.0, budget * 0.65)
    if time.monotonic() < t3 - 100:
        try:
            r = _optimize_lp(best_seq, t3, seed=123)
            try_update(r)
        except Exception:
            pass

    # ===== Phase 4: Try different lengths =====
    if time.monotonic() < deadline - 200:
        x_old = np.linspace(0, 1, n_parent)
        lengths_to_try = [n_parent, n_parent - 20, n_parent + 20,
                          n_parent - 50, n_parent + 50,
                          int(n_parent * 0.8), int(n_parent * 1.2),
                          int(n_parent * 0.6), int(n_parent * 1.5)]
        lengths_to_try = sorted(set(l for l in lengths_to_try if 50 <= l <= 800))

        for nt in lengths_to_try:
            if time.monotonic() > deadline - 60:
                break
            if nt == n_parent:
                continue
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, parent)
            init_new = np.clip(init_new, 0, 1000)
            sn = init_new.sum()
            if sn < 1e-10:
                continue
            init_new /= sn
            alloc = min(60.0, deadline - time.monotonic() - 30)
            if alloc < 10:
                break
            try:
                r = _optimize_lp(init_new, time.monotonic() + alloc, seed=2000 + nt)
                try_update(r)
            except Exception:
                pass

    # ===== Phase 5: Final deep LP polish =====
    if time.monotonic() < deadline - 50:
        ab = np.clip(best_seq, 0, 1000).astype(np.float64)
        ab_s = ab.sum()
        if ab_s > 1e-15:
            ab /= ab_s
        try:
            r = _optimize_lp(ab, deadline - 10, seed=7777)
            try_update(r)
        except Exception:
            pass

    # ===== Phase 6: Final CD polish =====
    if time.monotonic() < deadline - 20:
        try:
            r = _optimize_cd(best_seq, deadline - 5)
            try_update(r)
        except Exception:
            pass

    # ===== Final validation =====
    try:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if not np.isfinite(fs):
            raise ValueError("invalid")
    except Exception:
        fallback = np.ones(32, dtype=np.float64)
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
