```python
import numpy as np
import time
from scipy.optimize import linprog

def _convolve_full(a, b):
    """Full convolution using FFT for efficiency."""
    n = len(a) + len(b) - 1
    fa = np.fft.rfft(a, 2*n)
    fb = np.fft.rfft(b, 2*n)
    return np.fft.irfft(fa * fb, 2*n)[:n]

def _build_conv_sparse(a, n):
    """Build sparse convolution matrix C where C @ b = np.convolve(a, b)."""
    # C[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    # C is (2n-1) x n
    from scipy.sparse import lil_matrix
    C = lil_matrix((2*n-1, n), dtype=np.float64)
    for j in range(n):
        # C[j + i, j] = a[i] for i = 0..n-1
        C[j:j+n, j] = a
    return C.tocsr()

def _solve_lp(a, M, n):
    """Solve the LP: max 1^T b s.t. C(a) b <= M, b >= 0."""
    C = _build_conv_sparse(a, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.status != 0 or not np.isfinite(res.x).all():
        return None
    return res.x

def _score(a):
    """Compute the AC1 score."""
    n = len(a)
    if a.sum() < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (a.sum()**2)
```

This means $a$ is an eigenfunction of the "dual operator" $\sum_{k \in K^*} \lambda_k S_k$ (where $S_k$ is the shift operator), with eigenvalue $\mu$.

If $K^* = \{k_0\}$ (only one active constraint), then $a_{k_0 - j} = \mu / \lambda_{k_0}$ for all $j$, which means $a$ is constant. This gives the uniform solution with score 2.

For multiple active constraints, $a$ can be non-constant. The question is: can we find a non-constant $a$ that satisfies the KKT conditions?

    # Diagonal 1: a[1] at positions (1,0), (2,1), ..., (n, n-1)
    # ...
