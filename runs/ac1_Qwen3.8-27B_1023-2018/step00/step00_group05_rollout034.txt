```python
def project_to_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    # t_i = (cumsum[i] - 1) / (i+1)
    # Find largest i such that sorted_v[i] > t_i
    t_vals = (cumsum - 1.0) / np.arange(1, n+1)
    mask = sorted_v > t_vals
    m = np.sum(mask)
    if m == 0:
        t = 0.0
    else:
        t = (cumsum[m-1] - 1.0) / m
    result = np.maximum(v - t, 0.0)
    # Normalize to ensure sum = 1 (numerical precision)
    s = result.sum()
    if s > 0:
        result /= s
    return result
```
