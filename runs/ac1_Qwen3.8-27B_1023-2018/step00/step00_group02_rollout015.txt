        candidates.append(a0 / a0.sum())
    
    # Perturbed parent
    pert = a0 / a0.sum() + 0.01 * rng.standard_normal(n)
    candidates.append(np.clip(pert, 0, 1000) / (np.clip(pert, 0, 1000).sum() + 1e-30))
    
    # Parabolic (high in middle)
    j = np.arange(n)
    parab = 1.0 - (2 * (j - (n-1)/2) / (n-1))**2
    parab = np.clip(parab, 0.01, None)
    candidates.append(parab / parab.sum())
    
    # Inverted parabolic (high at edges)
    inv_parab = ((2 * (j - (n-1)/2) / (n-1))**2 + 0.1)
    candidates.append(inv_parab / inv_parab.sum())
    
    # Linear ramp
    ramp = (j + 1) / n
    candidates.append(ramp / ramp.sum())
    
    best_a = None
    best_score = np.inf
    
    # Evaluate all candidates
    for cand in candidates:
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        s = fast_score(cand)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = cand.copy()
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_score = 2.0
    
    # Verify with exact evaluator
    exact_s = float(evaluate_sequence(best_a.tolist()))
    best_score_exact = exact_s
    best_a_exact = best_a.copy()
    
    # Phase 1: Gradient descent
    a = best_a.copy()
    lr = 0.005
    prev_fast = fast_score(a)
    stall_count = 0
    
    for it in range(200000):
        if time.monotonic() > deadline - 10:
            break
        if (it + 1) % 1000 == 0:
            s = float(evaluate_sequence(a.tolist()))
            if np.isfinite(s) and s < best_score_exact:
                best_score_exact = s
                best_a_exact = a.copy()
            # Decay learning rate
            lr = max(1e-6, lr * 0.9)
            if lr <= 1e-6:
                break
        
        fs, g = compute_grad(a)
        if not np.isfinite(fs) or not np.isfinite(g).all():
            break
        
        # Gradient step with projection
        # Remove mean component (scale invariance)
        g_mean = g.mean()
        g_centered = g - g_mean  # This doesn't change the score (scale invariance)
        
        new_a = np.clip(a - lr * g_centered, 0.0, 1000.0)
        if new_a.sum() < 1e-10:
            new_a = a
            continue
        
        new_fs = fast_score(new_a)
        
        if np.isfinite(new_fs) and new_fs < prev_fast:
            a = new_a
            prev_fast = new_fs
            stall_count = 0
        else:
            stall_count += 1
            lr = max(1e-7, lr * 0.7)
            if stall_count > 500:
                # Kicks: random perturbation
                a = np.clip(a * np.exp(0.01 * rng.standard_normal(n)), 0, 1000)
                if a.sum() < 1e-10:
                    break
                stall_count = 0
                lr = 0.005
    
    # Check final state
    s = float(evaluate_sequence(a.tolist()))
