- This is a smooth approximation of max(conv(a,a))
- Minimize F(a) subject to sum(a) = 1, a >= 0
- Use projected gradient descent

The gradient of F(a) with respect to a[j]:
dF/da[j] = (1/tau) * sum_k softmax(conv(a,a)/tau)[k] * d(conv(a,a)[k])/da[j]
= (1/tau) * sum_k softmax(conv(a,a)/tau)[k] * 2*a[k-j] (with a[m]=0 outside [0,n-1])

This can be computed in O(n^2) time (or O(n log n) with FFT if we're clever).

For n = 3809, O(n^2) = 14.5M operations per gradient evaluation. That's fast.

```python
def smooth_max(values, tau):
    """Compute log-sum-exp as a smooth max."""
    m = values.max()
    return m + tau * np.log(np.sum(np.exp((values - m) / tau)))

def ac1_smooth(a, tau):
    """Smooth version of AC1."""
    n = len(a)
    S = a.sum()
    autoc = np.convolve(a, a)
    log_sum_exp = autoc.max() + tau * np.log(np.sum(np.exp((autoc - autoc.max()) / tau)))
    return 2 * n * log_sum_exp / (S ** 2)

def ac1_smooth_grad(a, tau):
    """Gradient of the smooth AC1."""
    n = len(a)
    S = a.sum()
    autoc = np.convolve(a, a)
    
    # Softmax weights
    shifted = (autoc - autoc.max()) / tau
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Gradient of log-sum-exp with respect to a[j]:
    # = (1/S^2) * d(log_sum_exp)/da[j] * 2n - (2n * log_sum_exp / S^3) * d(S^2)/da[j] * ... 
    # Let me be more careful.
    # AC1 = 2n * LSE / S^2
    # d(AC1)/da[j] = 2n * (dLSE/da[j] * S^2 - LSE * 2*S * 1) / S^4
    # = 2n * (dLSE/da[j] * S - 2 * LSE) / S^3
    
    # dLSE/da[j] = sum_k weights[k] * d(autoc[k])/da[j]
    # = sum_k weights[k] * 2*a[k-j] (with a[m]=0 outside [0,n-1])
    # = 2 * sum_k weights[k] * a[k-j]
    # This is 2 * (weights conv a)[j] (with appropriate indexing)
    
    # (weights conv a)[j] = sum_k weights[k] * a[k-j]
    # = sum_i weights[j+i] * a[i] (letting i = k-j, k = j+i)
    # = (a * weights)[j] in the "cross-correlation" sense
    # Actually, sum_k weights[k] * a[k-j] is the cross-correlation of weights and a at lag j
    # = (a * weights)[j] where * denotes cross-correlation
    # In numpy: np.correlate(a, weights, mode='full')[n-1+j] ... hmm
    # Or: np.convolve(a, weights[::-1], mode='full')[n-1+j]
    
    # Let me just compute it directly:
    # grad_lse[j] = 2 * sum_{k: 0<=k-j<n} weights[k] * a[k-j]
    # = 2 * sum_{i=0}^{n-1} weights[j+i] * a[i] (where j+i < 2n-1)
    
    # This is a convolution-like operation
    # grad_lse[j] = 2 * (a * weights)[j] where (a * weights)[j] = sum_i a[i] * weights[j+i]
    
    # Using np.convolve:
    # np.convolve(a, weights, mode='full') gives h[j] = sum_i a[i] * weights[j-i]
    # We want sum_i a[i] * weights[j+i] = sum_i a[i] * weights[-(-j-i)]
    # Let w_rev = weights[::-1], then sum_i a[i] * weights[j+i] = sum_i a[i] * w_rev[2n-2-j-i]
    # = (a * w_rev)[2n-2-j] ... this is getting confusing.
    
    # Let me just use a direct O(n^2) computation:
    grad_lse = np.zeros(n)
    for j in range(n):
        # grad_lse[j] = 2 * sum_{k=max(0,j)}^{min(2n-2, j+n-1)} weights[k] * a[k-j]
        k_start = max(0, j)  # k-j >= 0 => k >= j
        k_end = min(2*n-2, j+n-1)  # k-j <= n-1 => k <= j+n-1
        if k_end >= k_start:
            grad_lse[j] = 2.0 * np.dot(weights[k_start:k_end+1], a[0:k_end-k_start+1])
    
    # This is O(n^2) total, which is fine for n=3809.
    
    dAC1 = 2*n * (grad_lse * S - 2 * (autoc.max() + tau * np.log(np.sum(np.exp(shifted)))) ) / S**3
    
    # Projected gradient (project onto sum(a) = 1, a >= 0)
    # For the constraint sum(a) = 1: subtract the mean of dAC1
    # For a >= 0: clip
    # Simple projection: a_new = max(a - eta * dAC1 + eta * mean(dAC1), 0), then normalize
    
    return dAC1
```

1. Compute the autocorrelation
2. Find the peak positions
3. The "subgradient" of the peak at position k* is proportional to 2*a[k*-j] for each j
4. To reduce the peak, we want to decrease a[j] in proportion to a[k*-j]
5. But we also want to increase a[j] where it doesn't contribute to the peak
6. This gives a "direction" that reduces the peak

More precisely, the "water-filling" approach:
- Let T be the target level (slightly below current peak)
- For positions k where autoc[k] > T, we need to reduce them
- The reduction at position k is achieved by reducing a[j] for j that contribute to autoc[k]
- The amount of reduction at a[j] should be proportional to the "excess" at the positions it contributes to

This is essentially solving:
min sum_k max(0, autoc[k] - T)² subject to sum(a) = 1, a >= 0

Which is a non-convex optimization problem. But we can use gradient descent.

```python
def refine(a, n, budget):
    """Iteratively refine the sequence to reduce AC1."""
    deadline = time.monotonic() + budget
    a = a.copy()
    a = a / a.sum()
    best = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    
    iter_count = 0
    while time.monotonic() < deadline:
        iter_count += 1
        autoc = np.convolve(a, a)
        M = autoc.max()
        
        # Find all positions where autoc is close to the peak
        threshold = M * 0.95
        active = np.where(autoc >= threshold)[0]
        
        # Compute the "pressure" at each position j
        # pressure[j] = sum_{k in active, 0<=k-j<n} autoc[k] / M * a[k-j]
        # This is like a weighted sum of the a values that contribute to the peak
        pressure = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            pressure[j_start:j_end+1] += autoc[k] * a[j_start:j_end+1] / M
        
        # Normalize pressure
        if pressure.max() > 0:
            pressure = pressure / pressure.max()
        
        # Update: reduce a where pressure is high, increase where pressure is low
        # a_new = a * (1 - eta * pressure) + eta * a_min
        # Or: a_new = exp(-eta * pressure) * a, then normalize
        
        eta = 0.1  # Step size
        a_new = a * np.exp(-eta * pressure)
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            a_new = a
        a_new = a_new / a_new.sum()
        
        score = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a_new.copy()
            best_score = score
            a = a_new.copy()
        else:
            # Try a smaller step
            a_new = (1 - eta*0.1) * a + eta*0.1 * a_new
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            score = float(evaluate_sequence(a_new.tolist()))
            if np.isfinite(score) and score < best_score:
                best = a_new.copy()
                best_score = score
                a = a_new.copy()
    
    return best, best_score
```
