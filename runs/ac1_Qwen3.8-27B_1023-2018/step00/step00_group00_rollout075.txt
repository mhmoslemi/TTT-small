def _fast_score(a, n):
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * float(np.max(c)) / (s * s))


def _make_ramp(n, p, lo=1.0, hi=1000.0, reverse=False):
