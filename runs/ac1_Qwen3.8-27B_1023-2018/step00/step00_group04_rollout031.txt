import numpy as np
import time
import math

def normalize_simplex(a):
    """Project onto simplex {a >= 0, sum(a) = 1}."""
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    # Solve for tau: sum(max(a - tau, 0)) = 1
    # Equivalent: find tau such that sum_{a_i > tau} (a_i - tau) = 1
    # Let's use the standard algorithm
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv[-1] - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(a - theta, 0)

def compute_score_fast(a):
    """Fast computation of 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    corr = np.convolve(a, a)
    max_corr = np.max(corr)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * max_corr / (s * s)

def subgradient_corr(a, top_k=20, eps=0.0):
    """Compute subgradient of max(conv(a,a)) using top-k lags.
    
    Returns (subgradient, max_corr, corr)
    """
    n = len(a)
    corr = np.convolve(a, a)
    max_corr = np.max(corr)
    
    # Get top-k lags
    if eps > 0:
        active = np.where(corr >= max_corr - eps)[0]
    else:
        idx = np.argsort(corr)[::-1][:top_k]
        active = idx
    
    grad = np.zeros(n)
    for k in active:
        j_min = max(0, int(k) - n + 1)
        j_max = min(n - 1, int(k))
        if j_max >= j_min:
            # grad[j] += a[k-j] for j in [j_min, j_max]
            segment = a[j_min:j_max+1]
            grad[j_min:j_max+1] += np.flip(segment)  # a[k-j] for j in range
    
    grad /= len(active)
    return grad, max_corr, corr

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv[-1] - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0)

def optimize_subgrad(a, max_iters, lr_init, deadline):
    """Projected subgradient descent to minimize max(conv(a,a)) subject to sum(a)=1, a>=0."""
    n = len(a)
    a = project_simplex(a.copy())
    
    best_a = a.copy()
    best_score = compute_score_fast(a)
    lr = lr_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, max_corr, corr = subgradient_corr(a, top_k=50)
        
        # Project gradient to tangent space (sum = 0)
        grad = grad - np.mean(grad)
        
        # Gradient step
        a_new = a - lr * grad
        
        # Project to simplex
        a_new = project_simplex(a_new)
        
        # Evaluate
        score = compute_score_fast(a_new)
        
        if score < best_score:
            best_a = a_new.copy()
            best_score = score
            # Increase lr slightly
            lr *= 1.01
        else:
            # Decrease lr
            lr *= 0.95
        
        a = a_new
    
    return best_a, best_score

import numpy as np
import time
import math

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv[-1] - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0)

def fast_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2 without calling evaluate_sequence."""
    n = len(a)
    corr = np.convolve(a, a)
    max_corr = np.max(corr)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_corr / (s * s))

def smooth_max_grad(a, n_smooth=50):
    """Compute smooth approximation gradient of max(conv(a,a)).
    
    Returns (gradient, max_corr, full_corr)
    """
    n = len(a)
    corr = np.convolve(a, a)
    max_corr = float(np.max(corr))
    
    # Compute softmax weights, focusing on top n_smooth
    sorted_idx = np.argsort(corr)[::-1]
    top = corr[sorted_idx][:n_smooth]
    
    if top[0] <= 0:
        # Fallback: point to max
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        T = max((top[0] - top[-1]) / max(np.log(max(n_smooth, 2)), 1), max_corr * 1e-10)
        shifted = (corr - max_corr) / T
        shifted = np.maximum(shifted, -50)  # prevent underflow
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(corr))
            w[np.argmax(corr)] = 1.0
            w_sum = 1.0
        w /= w_sum
    
    # Compute gradient: sum_k w_k * 2 * a[k-j] for valid j
    grad = np.zeros(n)
    # Only process lags with significant weight
    active_mask = w > 1e-12
    active_idx = np.where(active_mask)[0]
    
    for k in active_idx:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        if je > js:
            seg = a[k - js : k - je : -1]  # a[k-j] for j in [js, je)
            grad[js:je] += 2.0 * w[k] * seg
    
    return grad, max_corr, corr

def optimize_sequence(a0, iters, deadline, n_smooth=50, verbose=False):
    """Optimize max(conv(a,a)) subject to sum(a)=1, a>=0 using projected subgradient."""
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = fast_score(a)
    
    step = 1.0 / n
    no_improve_count = 0
    step_history = []
    
    for it in range(iters):
        if time.monotonic() > deadline:
            break
        
        grad, max_corr, corr = smooth_max_grad(a, n_smooth)
        
        # Project gradient to tangent space of simplex
        grad = grad - np.mean(grad)
        
        # Gradient step
        a_new = a - step * grad
        
        # Project to simplex
        a_new = project_simplex(a_new)
        
        # Evaluate
        score = fast_score(a_new)
        
        if score < best_score - 1e-12:
            best_a = a_new.copy()
            best_score = score
            no_improve_count = 0
            step = min(step * 1.05, 5.0)
        else:
            no_improve_count += 1
            step *= 0.9
        
        a = a_new
        
        if verbose and it % 100 == 0:
            print(f"Iter {it}: score={best_score:.6f}, step={step:.6f}")
    
    return best_a, best_score

Let me also consider a **spectral approach**. The condition that conv(a,a) is flat (equal to S^2/(2n-1) for all k) means:
- The DFT of conv(a,a) is zero for all non-DC frequencies
- The DFT of conv(a,a) is |DFT(a)|^2 (for circular convolution)
- So |DFT(a)|^2 should be concentrated at DC, i.e., DFT(a) should be a delta at frequency 0

But that means a should be a delta, which contradicts the flat convolution. The issue is that conv(a,a) is linear convolution, not circular. For linear convolution, the DFT relationship is different.

For linear convolution: DFT[conv(a,a)] = DFT[a] * DFT[a] (with zero-padding to length >= 2n-1).

If we zero-pad a to length 2n-1 (or more), then:
DFT[conv_padded] = DFT[a_padded]^2

If conv_padded is flat (= S^2/(2n-1)), then DFT[conv_padded] = S^2 at DC and 0 elsewhere.
So DFT[a_padded]^2 = S^2 at DC and 0 elsewhere.
So DFT[a_padded] = S at DC and 0 elsewhere.
So a_padded is a constant (= S/(2n-1) for length 2n-1).
But a_padded = [a, 0, ..., 0] (padded with zeros).
So a[i] = S/(2n-1) for all i, and 0 = S/(2n-1) for the padding. Contradiction unless S = 0.

So a perfectly flat linear convolution is impossible (for non-trivial a). The best we can do is approximately flat.

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv[-1] - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0)

def _fast_score(a):

    corr = np.convolve(a, a)
    mc = corr.max()
    s = a.sum()
    if s < 0.01:

    return 2.0 * n * mc / (s * s)

def _softmax_grad(a, n_top=100):
    """Softmax-weighted subgradient of max(conv(a,a))."""
    n = len(a)
    corr = np.convolve(a, a)
    max_c = corr.max()
    
    # Temperature from top n_top
    idx = np.argsort(corr)[::-1][:n_top]
    top = corr[idx]

    if ws < 1e-30:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    
    # Gradient
    grad = np.zeros(n)
    active = np.where(w > 1e-14)[0]

        js = max(0, ki - n + 1)
        je = min(n, ki + 1)
        if je > js:
            grad[js:je] += 2.0 * w[ki] * a[ki - js:ki - je:-1]
    
    return grad, max_c, corr
