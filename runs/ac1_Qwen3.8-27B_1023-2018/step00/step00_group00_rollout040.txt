    # Get parent
    try:
        parent = np.array(height_sequence_1, dtype=np.float64).copy()
    except NameError:
        parent = np.ones(4806, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.all(np.isfinite(parent)):
        parent = np.ones(4806, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.size

    # Normalize parent
    S0 = parent.sum()
    if S0 < 0.01:
        parent = np.ones(n, dtype=np.float64)
        S0 = float(n)

    # Try several parametric initializations
    candidates = [parent / S0]
    x = np.arange(n, dtype=np.float64)

    par = np.sin(np.pi * (x + 1) / (n + 1)) ** 2
    if par.sum() > 1e-10:
        candidates.append(par / par.sum())

    for lam in [1.0, 2.0, 3.0, 5.0]:
        es = np.exp(-lam * x / n)
        if es.sum() > 1e-10:
            candidates.append(es / es.sum())

    cos_w = 0.5 * (1.0 + np.cos(np.pi * x / (n - 1)))
    if cos_w.sum() > 1e-10:
        candidates.append(cos_w / cos_w.sum())

    ramp = np.arange(1, n + 1, dtype=np.float64)
    candidates.append(ramp / ramp.sum())

    inv_ramp = ramp[::-1].copy()
    candidates.append(inv_ramp / inv_ramp.sum())

    # Gaussian centered
    gc = np.exp(-0.5 * ((x - 0.4 * n) / (0.2 * n)) ** 2)
    if gc.sum() > 1e-10:
        candidates.append(gc / gc.sum())

    gc2 = np.exp(-0.5 * ((x - 0.6 * n) / (0.2 * n)) ** 2)
    if gc2.sum() > 1e-10:
        candidates.append(gc2 / gc2.sum())

    # Evaluate all
    best = candidates[0].copy()
    best_score = float('inf')
    for c in candidates:
        sc = float(evaluate_sequence(c.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = c.copy()

    a = best.copy()

    if not np.isfinite(best_score):
        return [float(v) for v in parent]

    # Precompute CSR structure for LP convolution matrix
    n_rows = 2 * n - 1
    nnz_per_row = np.empty(n_rows, dtype=np.int64)
    for k in range(n_rows):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        nnz_per_row[k] = hi - lo + 1
    indptr = np.zeros(n_rows + 1, dtype=np.int64)
    indptr[1:] = np.cumsum(nnz_per_row)
    total_nnz = int(indptr[-1])
    col_indices = np.empty(total_nnz, dtype=np.int32)
    for k in range(n_rows):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        s = int(indptr[k])
        col_indices[s:s + nnz_per_row[k]] = np.arange(lo, hi + 1, dtype=np.int32)

    def build_data(a_vec):
        data = np.empty(total_nnz, dtype=np.float64)
        for k in range(n_rows):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            s = int(indptr[k])
            data[s:s + nnz_per_row[k]] = a_vec[lo:hi + 1]
        return data

    # Main optimization loop
    alpha = 0.05
    best_alpha = alpha
    no_improve = 0
    max_stall = 200
    lp_done = 0
    max_lps = 45
    next_lp = 80
    grad_steps = 0

    while tl() > 60:
        use_lp = (grad_steps >= next_lp and lp_done < max_lps and tl() > 120)

        if use_lp:
            try:
                b = _fft_conv(a)
                b = np.maximum(b, 0)
                M = b.max()
                data = build_data(a)

                import scipy.sparse as sp
                A_mat = sp.csr_matrix((data, col_indices, indptr), shape=(n_rows, n))

                import cvxpy as cp
                g_var = cp.Variable(n, nonneg=True)
                cons = [A_mat @ g_var <= M]
                prob = cp.Problem(cp.Maximize(cp.sum(g_var)), cons)

                solve_tl = max(5, min(120, tl() - 120))
                success = False
                for solver_name in [cp.GLOP, cp.CBC, cp.SCIP]:
                    try:
                        prob.solve(solver=solver_name,
                                   time_limit=solve_tl)
                        if prob.status in ['optimal', 'optimal_inaccurate']:
                            success = True
                            break
                    except Exception:
                        continue

                if success and g_var.value is not None:
                    g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0)
                    g_sum = g_val.sum()
                    if g_sum > 1e-10:
                        g_norm = g_val / g_sum
                        for t_blend in [0.5, 0.3, 0.7, 0.2, 0.1, 0.9, 0.4, 0.6]:
                            if tl() < 65:
                                break
                            a_mix = (1.0 - t_blend) * a + t_blend * g_norm
                            am = a_mix.sum()
                            if am < 1e-10:
                                continue
                            a_mix /= am
                            sc = float(evaluate_sequence(a_mix.tolist()))
                            if np.isfinite(sc) and sc < best_score:
                                best = a_mix.copy()
                                best_score = sc
                                a = a_mix
                                no_improve = 0
                                break

                lp_done += 1
                next_lp = grad_steps + 80 + 25 * lp_done
            except Exception:
                lp_done += 1
                next_lp = grad_steps + 80
        else:
            # Gradient step
            b = _fft_conv(a)
            b = np.maximum(b, 0)
            M = b.max()

            tol = 1e-7 * M + 1e-15
            active = np.where(b >= M - tol)[0]

            grad = np.zeros(n)
            for k in active:
                ki = int(k)
                lo = max(0, ki - n + 1)
                hi = min(n - 1, ki)
                if lo <= hi:
                    is_ = ki - hi
                    ie = ki - lo + 1
                    grad[lo:hi + 1] += a[is_:ie][::-1]
            grad /= len(active)

            a_new = a - alpha * grad
            a_new = np.maximum(a_new, 0.0)
            ns = a_new.sum()
            if ns < 1e-10:
                alpha = max(alpha * 0.5, 1e-8)
                no_improve += 1
                grad_steps += 1
                continue
            a_new /= ns

            sc = float(evaluate_sequence(a_new.tolist()))
            grad_steps += 1

            if np.isfinite(sc) and sc < best_score:
                best = a_new.copy()
                best_score = sc
                a = a_new
                no_improve = 0
                if alpha < 0.12:
                    alpha = min(alpha * 1.04, 0.12)
                    if alpha > best_alpha:
                        best_alpha = alpha
            else:
                no_improve += 1
                alpha = max(alpha * 0.93, 1e-8)

                if no_improve >= max_stall:
                    a = best * np.exp(rng.normal(0, 0.06, n))
                    a = np.maximum(a, 0.0)
                    as_ = a.sum()
                    if as_ < 1e-10:
                        a = best.copy()
                    else:
                        a /= as_
                    alpha = max(0.02, best_alpha * 0.5)
                    no_improve = 0

    # Final fine-tuning phase
    while tl() > 10:
        a = best.copy()
        b = _fft_conv(a)
        b = np.maximum(b, 0)
        M = b.max()
        tol = 1e-8 * M + 1e-15
        active = np.where(b >= M - tol)[0]
        grad = np.zeros(n)
        for k in active:
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            if lo <= hi:
                is_ = ki - hi
                ie = ki - lo + 1
                grad[lo:hi + 1] += a[is_:ie][::-1]
        grad /= len(active)

        improved = False
        for at in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            if tl() < 12:
                break
            a_new = a - at * grad
            a_new = np.maximum(a_new, 0.0)
            ns = a_new.sum()
            if ns < 1e-10:
                continue
            a_new /= ns
            sc = float(evaluate_sequence(a_new.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = a_new
                best_score = sc
                a = a_new
                improved = True
                break

        # Random perturbations
        for _ in range(30):
            if tl() < 12:
                break
            sigma = rng.choice([0.003, 0.005, 0.01, 0.02, 0.04])
            a_p = best * np.exp(rng.normal(0, sigma, n))
            a_p = np.maximum(a_p, 0.0)
            a_p /= a_p.sum()
            sc = float(evaluate_sequence(a_p.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = a_p
                best_score = sc
                improved = True

        if not improved and tl() < 30:
            break

    # Validate and return
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        fb = np.clip(parent, 0.0, 1000.0)
        fb_sc = float(evaluate_sequence(fb.tolist()))
        if np.isfinite(fb_sc):
            return [float(v) for v in fb]
        return [1.0]
    return [float(v) for v in best]
