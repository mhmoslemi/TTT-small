def conv_self(f):
    N1=1; while N1 < 2*n-1: N1*=2
    F=np.fft.rfft(f,N1)
    h=np.fft.irfft(F*F,N1)[:2*n-1]
    np.maximum(h,0,h) # avoid neg
    return h

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _clean_sequence(f, n):
    if f is None:
        return None
    f = np.asarray(f, dtype=np.float64).reshape(-1)
    if f.size == 0:
        return None
    if f.size != n:
        return None
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.clip(f, 0.0, 1000.0)
    s = float(f.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    f = f / s
    f = np.maximum(f, 1e-300)
    f /= float(f.sum())
    if not np.isfinite(f).all():
        return None
    return f


def _fft_conv_self(f):
    n = f.size
    L = 2 * n - 1
    N = _next_pow2(L)
    F = np.fft.rfft(f, N)
    h = np.fft.irfft(F * F, N)[:L]
    np.maximum(h, 0.0, out=h)
    return h


def _max_self(f):
    h = _fft_conv_self(f)
    return float(h.max()) if h.size else 0.0


def _fast_score(f):
    s = float(f.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return float("inf")
    return float(2.0 * f.size * _max_self(f) / (s * s))


def _fft_corr_pf(p, f, F2=None, N2=None):
    n = f.size
    L = p.size
    if N2 is None:
        N2 = _next_pow2(L + n - 1)
    if F2 is None:
        tmp = np.zeros(N2, dtype=np.float64)
        tmp[:n] = f
        F2 = np.fft.rfft(tmp, N2)
    pr = p[::-1]
    P = np.fft.rfft(pr, N2)
    conv = np.fft.irfft(P * F2, N2)
    idx = L - 1 - np.arange(n)
    return conv[idx]


def _softmax_weights(y, beta):
    m = float(y.max()) if y.size else 0.0
    z = float(beta) * (y - m)
    np.clip(z, -30.0, 30.0, out=z)
    p = np.exp(z)
    ps = float(p.sum())
    if not np.isfinite(ps) or ps <= 0.0:
        tol = max(1e-12, 1e-7 * max(0.0, m))
        p = np.where(y >= m - tol, 1.0, 0.0)
        ps = float(p.sum())
        if ps <= 0.0:
            p = np.full(y.size, 1.0 / y.size, dtype=np.float64) if y.size else np.ones(1)
            ps = 1.0
    p /= ps
    return p, m


def make_half_cauchy(n):
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    if n > 1:
        k = np.arange(n - 1, dtype=np.float64)
        ratios = (2.0 * k + 1.0) / (2.0 * k + 2.0)
        a[1:] = np.cumprod(ratios)
    s = float(a.sum())
    if s > 0 and np.isfinite(s):
        a /= s
    return a


class SearchState:
    def __init__(self, n):
        self.n = n
        self.limit = 30
        self.pool = []
        self.scores = []

    def consider(self, f):
        f = _clean_sequence(f, self.n)
        if f is None:
            return None
        sc = _fast_score(f)
        if not np.isfinite(sc):
            return None
        if (
            len(self.pool) < self.limit
            or sc < max(self.scores) - 1e-14
        ):
            self.pool.append(f.copy())
            self.scores.append(float(sc))
            if len(self.pool) > self.limit:
                worst = int(np.argmax(self.scores))
                self.pool.pop(worst)
                self.scores.pop(worst)
        return f, sc

    def best(self):
        if not self.pool:
            return None, float("inf")
        i = int(np.argmin(self.scores))
        return self.pool[i].copy(), self.scores[i]


def _run_self_optimize(f0, state, deadline, rng):
    n = state.n
    L = 2 * n - 1
    N1 = _next_pow2(L)

    f = f0.copy()

        F1 = np.fft.rfft(f, N1)
        h = np.fft.irfft(F1 * F1, N1)[:L]

        m = float(y.max()) if y.size else 0.0

        tmp[:n] = f
        F2 = np.fft.rfft(tmp, N2)
        corr = _fft_corr_pf(p, f, F2=F2, N2=N2)

        cum = np.empty(L + 1, dtype=np.float64)
        cum[0] = 0.0
        np.cumsum(p, out=cum[1:])
        Pwin = cum[n:] - cum[:n]

            fn /= s

            sc = _fast_score(fn)

        if candidates:
            candidates.sort(key=lambda x: x[0])

            sc, fn = candidates[0]
            if np.isfinite(sc) and sc < old_sc - 1e-12:
                f = fn

                eta = min(0.25, eta * 1.15)

            elif len(candidates) > 1:
                sc, fn = candidates[1]

                    f = fn
                    best_state, best_sc = state.best()
                    if best_sc is None or sc < best_sc + 1e-10:
                        state.consider(f)
                eta *= 0.65
            else:
                eta *= 0.5

        if improved:
            if beta < 2000.0:
                beta = beta * 1.15
        elif time.monotonic() - no_info_since > 45.0:

def _max_cross_with_aff(f, v, F1, N1, L):
    V = np.fft.rfft(v, N1)
    r = np.fft.irfft(F1 * V, N1)[:L]
    return float(np.max(r)) if r.size else 0.0


def _solve_cross_lp(f, v_init, deadline, rng):

    L = 2 * n - 1
    N1 = _next_pow2(L)
    N2 = _next_pow2(L + n - 1)

    v = v_init.copy()

def _run_cross_iter(f0, state, deadline, rng):
    n = state.n
    f = f0.copy()
    v = f.copy()
    last_improve = time.monotonic()
    base_score, _ = state.best()
    if base_score is None:
        base_score = _fast_score(f)

    for outer in range(200):
        if time.monotonic() > deadline - 20.0:
            break
        sub_deadline = min(deadline - 20.0, time.monotonic() + 35.0)
        v, _ = _solve_cross_lp(f, v, sub_deadline, rng)

        cur_sc = _fast_score(f)
        cands = []
        for t in (1.0, 0.85, 0.65, 0.45, 0.25, 0.12, 0.05):
            u = (1.0 - t) * f + t * v
            u = _clean_sequence(u, n)
            if u is not None:
                u_sc = _fast_score(u)
                if np.isfinite(u_sc):
                    cands.append((u_sc, u))
                    state.consider(u)

        new_f = f
        new_sc = cur_sc
        if cands:
            cands.sort(key=lambda x: x[0])
            if cands[0][0] < new_sc - 1e-12:
                new_sc, new_f = cands[0]

        _, _ = state.consider(new_f)

        cur_best, cur_score = state.best()
        if cur_score < base_score - 1e-11:
            base_score = cur_score
            last_improve = time.monotonic()
            f = new_f.copy()
            v = v.copy()
        elif outer > 0 and time.monotonic() - last_improve > 130.0:
            break
        else:
            f = new_f

    state.consider(f)


def _random_smooth_perturbations(base, state, deadline, rng, trials=1200):

    if n < 8:
        return
    nfft = _next_pow2(n)
    freqs = np.fft.rfftfreq(nfft, d=1.0) * 2.0 * np.pi
    kernel = 1.0 / (1.0 + (freqs / 35.0) ** 2)

    for _ in range(trials):
        if time.monotonic() > deadline - 10.0:
            break
        noise = rng.normal(0.0, 1.0, size=nfft)
        low = np.fft.irfft(np.fft.rfft(noise) * kernel, nfft)[:n]
        sig = float(np.sqrt(max(1e-12, 1.0 / max(1, n))))
        amp = rng.uniform(0.05, 0.35)
        u = base.copy()
        u = u * np.exp(amp * low * sig)
        u = _clean_sequence(u, n)
        if u is not None:
            state.consider(u)
            if rng.random() < 0.25:
                v = 0.5 * u + 0.5 * make_half_cauchy(n)
                state.consider(v)
