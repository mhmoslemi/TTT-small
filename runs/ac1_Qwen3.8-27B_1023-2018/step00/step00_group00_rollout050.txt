from scipy.fft import fft, ifft
import time

def compute_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1 << (size - 1).bit_length()
    fa = fft(a, fft_size)
    conv = np.real(ifft(fa * fa, fft_size))[:size]
    return np.maximum(conv, 0)  # numerical errors might give tiny negatives

def _project(a, target_sum=1.0):
    """Project onto {a >= 0, sum(a) = target_sum}."""
    # Water-filling: a_proj[i] = max(0, a[i] - lambda)
    # Find lambda so sum = target_sum
    a_sorted = np.sort(a)[::-1]
    cumsum = np.cumsum(a_sorted)
    # For j active (largest j entries are positive):
    # lambda = (sum of top j - target_sum) / j
    # a_proj[i] = a_sorted[i] - lambda > 0 requires a_sorted[i] > lambda
    n = len(a)
    # Binary search for lambda
    lo = a.min() - 1
    hi = (a.sum() - target_sum) / n + 1  # rough upper bound
    # Actually, simpler: bisection
    lo = -1.0
    hi = a.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(a - mid, 0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.maximum(a - lam, 0)
