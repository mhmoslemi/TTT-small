import numpy as np
import time


def _next_pow2(x):
    return 1 << (int(x) - 1).bit_length()


def _convolve_full(a):
    n = a.size
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    k = np.arange(1, n + 1)
    cond = u - (css - 1.0) / k > 0
    if not cond.any():
        return np.full(n, 1.0 / n)
    kmax = int(np.max(np.nonzero(cond)[0]))
    theta = (css[kmax] - 1.0) / (kmax + 1.0)
    return np.clip(v - theta, 0.0, None)


def _soft_gradient(a, c, gamma):
    """Gradient of smooth-max of convolution lags w.r.t. a, via FFT cross-correlation."""
    n = a.size
    cmax = np.max(c)
    z = gamma * (c - cmax)
    w = np.exp(z - np.max(z))
    w /= np.sum(w)
    N = _next_pow2(3 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * corr[:n]
