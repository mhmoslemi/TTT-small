    n = init.size
    if n == 0:
        n = 6121
        init = np.ones(n)
    
    target_sum = np.sqrt(2.0 * n)
    C = 2.0 * n
    
    # FFT convolution helper
    def nextpow2(m):
        p = 1
        while p < m: p <<= 1
        return p
    # precompute sizes
    size_aa = nextpow2(2*n - 1)
    size_wra = nextpow2(3*n - 2)
    
    def conv_aa(a):
        A = np.fft.rfft(a, size_aa)
        return np.fft.irfft(A*A, size_aa)[:2*n-1]
    
    def conv_wra(wr, a):
        W = np.fft.rfft(wr, size_wra)
        A = np.fft.rfft(a, size_wra)
        return np.fft.irfft(W*A, size_wra)[:3*n-2]
    
    def score_of(a):
        # a already >= 0
        s = a.sum()
        if s <= 0: return np.inf
        conv = conv_aa(a)
        return C * conv.max() / (s*s)
    
    def smooth_obj_grad(a, beta):
        conv = conv_aa(a)
        cmax = conv.max()
        z = beta*(conv - cmax)
        ez = np.exp(z)
        E = ez.sum()
        weights = ez / E
        Fval = cmax + np.log(E)/beta
        wr = weights[::-1]
        conv_wra = conv_wra(wr, a)
        cross = conv_wra[n-1 : 2*n-1][::-1]
        Fgrad = 2.0 * cross
        s = a.sum()
        g = C*(Fgrad - 2.0*Fval/s) / (s*s)

        return deadline - time.monotonic()

    # ----------------------------------------------------------------------------

        n = 6121
        base = np.ones(n)
    n = int(base.size)

    target_sum = float(np.sqrt(2.0 * n))
    C = 2.0 * n

    def norm(a):
        a = np.asarray(a, dtype=np.float64)
        s = a.sum()
        if s <= 0:
            a = np.ones(n); s = n
        return a * (target_sum / s)

    # Fast convolutions (linear) via FFT
    # ----------------------------------------------------------------------------
    def _nextpow2(m):
        p = 1
        while p < m:
            p <<= 1
        return p

    size_aa = _nextpow2(2 * n - 1)
    size_wra = _nextpow2(3 * n - 2)

    def conv_aa(a):
        A = np.fft.rfft(a, size_aa)
        return np.fft.irfft(A * A, size_aa)[:2 * n - 1]

    def conv_wra(wr, a):
        W = np.fft.rfft(wr, size_wra)
        A = np.fft.rfft(a, size_wra)
        return np.fft.irfft(W * A, size_wra)[:3 * n - 2]

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0:
            return np.inf
        return float(C * conv_aa(a).max() / (s * s))

        a = np.maximum(a, 0.0)
        conv = conv_aa(a)
        cmax = conv.max()
        ez = np.exp(beta * (conv - cmax))
        E = ez.sum()
        weights = ez / E

        wr = weights[::-1]
        cw = conv_wra(wr, a)

        Fgrad = 2.0 * cross
        s = a.sum()
        g = C * (Fgrad - 2.0 * Fval / s) / (s * s)

            return 1e12 if not np.isfinite(sc) else sc

            return g
