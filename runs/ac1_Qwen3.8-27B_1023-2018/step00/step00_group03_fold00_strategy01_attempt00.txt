class FourierProjector:
    """Project current solution onto Fourier basis with energy preservation"""
    def __init__(self, length):
        self.length = int(length)
        self.omega_k = 2 * np.pi * np.arange(self.length) / self.length
        
    def project(self, coefficients):
        """Fourier basis projection with normalization"""
        N = self.length
        return np.real(np.fft.ifft(np.fft.fft(coefficients) * np.conj(np.fft.fft(np.ones(N)))))
