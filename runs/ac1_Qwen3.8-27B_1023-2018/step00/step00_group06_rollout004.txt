def _normalize(a, floor=1e-12):

    a = np.where(np.isfinite(a), a, 0.0)

```python
import numpy as np
import time


def _next_pow2(m):
    m = max(1, int(m))
    p = 1
    while p < m:
        p <<= 1
    return p


def _normalize(a, floor=1e-12):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        if a.size == 0:
            return np.ones(1)
        return np.full(a.size, 1.0 / float(a.size))
    a = a / s
    if floor > 0.0:
        a = np.maximum(a, floor)
        s2 = float(a.sum())
        if np.isfinite(s2) and s2 > 0.0:
            a = a / s2
    return a


def _conv_fft(a, nfft=None):
    n = int(a.size)
    if n <= 0:
        return np.zeros(0)
    if nfft is None:
        nfft = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, nfft)
    c = np.fft.irfft(fa * fa, nfft)
    need = 2 * n - 1
    if c.size >= need:
        c = c[:need]
    else:
        c = np.resize(c, need)
    c = np.maximum(c, 0.0)
    return c


def _score_from_c(c, n):
    """Score for a normalized sequence: sum(a) = 1."""
    if c.size == 0:
        return float("inf")
    cm = float(np.max(c))
    if not np.isfinite(cm):
        return float("inf")
    return float(2.0 * float(n) * cm)


def _score_fft(a, nfft=None, c=None):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return float("inf")
    if c is None:
        c = _conv_fft(a, nfft)
    cm = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cm):
        return float("inf")
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return float("inf")
    return float(2.0 * n * cm / (s * s))


def _weights_from_c(c, p, beta):
    if c.size == 0:
        return None
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return None
    u = c / cmax
    if p <= 1.0:
        w = u.copy()
    else:
        w = np.power(u, p)
    ws = float(np.sum(w))
    if not np.isfinite(ws) or ws <= 0.0:
        m = min(32, c.size)
        idx = np.argsort(c)[-m:]
        w = np.zeros_like(c)
        w[idx] = 1.0
        ws = float(w.sum())
        if ws <= 0.0:
            return None
    w /= ws
    if beta > 0.0:
        w = (1.0 - beta) * w + beta * (1.0 / float(c.size))
    return w


def _gradient_from_weights(a, w, nfft_d=None):
    n = int(a.size)
    if n == 0 or w.size != 2 * n - 1:
        return None
    if nfft_d is None:
        nfft_d = _next_pow2(3 * n - 2)
    ar = a[::-1]
    fw = np.fft.rfft(w, nfft_d)
    fa = np.fft.rfft(ar, nfft_d)
    conv = np.fft.irfft(fw * fa, nfft_d)
    if conv.size < n + n - 1:
        return None
    return conv[n - 1:n + n - 1].copy()


def _apply_update_z(a, z, alpha):
    z = np.clip(z, -4.0, 4.0)
    b = a * np.exp(-alpha * z)
    b = np.where(np.isfinite(b) & (b > 0.0), b, 1e-12)
    s = float(b.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    return b / s


def _resample(arr, N):
    arr = np.asarray(arr, dtype=np.float64)
    N = int(N)
    if arr.size == N:
        return _normalize(arr)
    if arr.size <= 1:
        return np.full(N, float(arr.mean()) if arr.size else 1.0)
    xs = np.linspace(0.0, 1.0, arr.size, endpoint=True)
    xt = np.linspace(0.0, 1.0, N, endpoint=True)
    out = np.interp(xt, xs, arr)
    return _normalize(out)


def _structured_candidates(N, seed=42, end_time=None, nfft=None):
    N = int(N)
    if N <= 0:
        return []
    if end_time is None:
        end_time = time.monotonic() + 0.0
    out = []

    def add(a):
        if end_time is not None and time.monotonic() > end_time:
            return
        an = _normalize(a)
        c = _conv_fft(an, nfft)
        s = _score_from_c(c, N)
        if np.isfinite(s):
            out.append((s, an))

    add(np.ones(N))
    if N <= 1:
        out.sort(key=lambda t: t[0])
        return [a for _, a in out]

    x = np.linspace(0.0, 1.0, N, endpoint=True)
    cos1 = np.cos(np.pi * x)

    Avals = (-1.0, -0.9, -0.8, -0.6, -0.4, -0.2,
             0.2, 0.4, 0.6, 0.8, 0.9, 1.0)
    pvals = (0.75, 1.0, 1.3, 1.7, 2.0, 2.5, 3.0, 4.0)

    for A in Avals:
        if end_time is not None and time.monotonic() > end_time:
            break
        base = 1.0 + A * cos1
        if float(np.min(base)) < -1e-7:
            base = np.clip(base, 0.0, None)
        base = np.maximum(base, 0.0)
        for p in pvals:
            v = np.power(base, p)
            if float(np.max(v)) > 0.0:
                add(v)

    M = 6
    basis = [np.cos(np.pi * m * x) for m in range(1, M + 1)]
    rng = np.random.default_rng(seed)
    for _ in range(24):
        if end_time is not None and time.monotonic() > end_time:
            break
        v = np.ones(N)
        for b in basis:
            v = v + rng.normal(0.0, 0.45) * b
        add(v)

    out.sort(key=lambda t: t[0])
    return [a for _, a in out]


def _lp_improve(a, end_time, max_iters=25):
    try:
        from scipy.optimize import linprog
    except Exception:
        return a

    a = _normalize(a)
    n = int(a.size)
    if n < 6:
        return a

    a = _normalize(np.maximum(a, 1e-8))
    nfft = _next_pow2(2 * n - 1)
    c = _conv_fft(a, nfft)
    score = _score_from_c(c, n)
    best = a.copy()
    best_score = score

    for _ in range(int(max_iters)):
        if time.monotonic() > end_time:
            break
        c = _conv_fft(a, nfft)
        M = float(np.max(c))
        if not np.isfinite(M) or M <= 0.0:
            break
        M = M * (1.0 + 1e-9)

        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = a

        res = None
        try:
            res = linprog(
                -np.ones(n),
                A_ub=A,
                b_ub=np.full(2 * n - 1, M),
                bounds=(0, None),
                method="highs",
            )
        except Exception:
            try:
                res = linprog(
                    -np.ones(n),
                    A_ub=A,
                    b_ub=np.full(2 * n - 1, M),
                    bounds=(0, None),
                    method="interior-point",
                )
            except Exception:
                res = None

        if res is None or not bool(getattr(res, "success", False)):
            break
        b = getattr(res, "x", None)
        if b is None:
            break
        b = np.asarray(b, dtype=np.float64)
        b = np.where(np.isfinite(b), b, 0.0)
        b = np.maximum(b, 0.0)
        sb = float(b.sum())
        if not np.isfinite(sb) or sb <= 1.0 + 1e-7:
            break

        g = b / sb
        g = _normalize(g, floor=1e-10)

        best_t = None
        best_cand = None
        best_cand_score = score
        for t in (0.25, 0.5, 0.1, 0.75, 0.9, 1.0):
            cand = (1.0 - t) * a + t * g
            cand = _normalize(cand)
            cc = _conv_fft(cand, nfft)
            sc = _score_from_c(cc, n)
            if np.isfinite(sc) and sc < best_cand_score - 1e-12:
                best_cand_score = sc
                best_t = t
                best_cand = cand

        if best_cand is not None:
            a = best_cand
            score = best_cand_score
            if score < best_score:
                best_score = score
                best = a.copy()
        else:
            break

    return best


def _improve(a, end_time, rng=None, nfft_c=None, nfft_d=None,
             p_start=1.6, p_max=24.0, alpha_start=0.35,
             do_exact=False, best_global=None, exact_every=350):
    if rng is None:
        rng = np.random.default_rng(20250518)
    a = _normalize(a)
    n = int(a.size)
    if n == 0:
        return a, float("inf")
    if nfft_c is None:
        nfft_c = _next_pow2(2 * n - 1)
    if nfft_d is None:
        nfft_d = _next_pow2(3 * n - 2)

    c = _conv_fft(a, nfft_c)
    score = _score_from_c(c, n)
    local_best_a = a.copy()
    local_best_score = score

    p = float(p_start)
    alpha = float(alpha_start)
    beta = 0.035
    fails = 0
    succ = 0

    while time.monotonic() < end_time:
        if not np.isfinite(score):
            a = local_best_a
            c = _conv_fft(a, nfft_c)
            score = local_best_score

        w = _weights_from_c(c, p, beta)
        if w is None:
            break

        d = _gradient_from_weights(a, w, nfft_d)
        if d is None:
            break

        d0 = d - float(np.mean(d))
        scale = float(np.std(d0))
        if not np.isfinite(scale) or scale <= 1e-15:
            fails += 1
            p = max(1.0, p * 0.8)
            beta = min(0.20, beta + 0.02)
            if fails > 60:
                base = local_best_a
                if best_global is not None:
                    gb = best_global.get("a")
                    gs = best_global.get("score", float("inf"))
                    if gb is not None and np.isfinite(gs) and gs < local_best_score - 1e-6:
                        base = np.asarray(gb, dtype=np.float64)
                b = base * np.exp(rng.normal(0.0, 0.12, size=n))
                b = _normalize(b)
                cb = _conv_fft(b, nfft_c)
                bs = _score_from_c(cb, n)
                if np.isfinite(bs):
                    a, c, score = b, cb, bs
                fails = 0
                p = max(1.2, min(p_max, p * 0.9))
                beta = 0.05
                alpha = 0.35
            continue

        z = d0 / scale

        accepted = False
        improved_local = False
        at_list = (alpha, max(0.02, alpha * 0.5), max(0.03, alpha * 1.6))
        for at in at_list:
            b = _apply_update_z(a, z, at)
            if b is None:
                continue
            cb = _conv_fft(b, nfft_c)
            bs = _score_from_c(cb, n)
            if np.isfinite(bs) and bs < score - 1e-13:
                a = b
                c = cb
                score = bs
                alpha = min(1.6, max(0.03, at * 1.08))
                accepted = True
                if score < local_best_score:
                    local_best_score = score
                    local_best_a = a.copy()
                    improved_local = True
                break

        if accepted:
            fails = 0
            succ += 1
            beta = max(0.02, beta * 0.96)
            if succ % 120 == 0:
                p = min(p_max, p * 1.08)
            if do_exact and best_global is not None:
                gb_score = float(best_global.get("score", float("inf")))
                if improved_local and np.isfinite(gb_score) and score < gb_score - 1e-4:
                    try:
                        es = float(evaluate_sequence(a.tolist()))
                        if np.isfinite(es) and es < best_global.get("score", float("inf")):
                            best_global["score"] = es
                            best_global["a"] = a.copy()
                    except Exception:
                        pass
                elif succ % exact_every == 0:
                    try:
                        es = float(evaluate_sequence(a.tolist()))
                        if np.isfinite(es) and es < best_global.get("score", float("inf")):
                            best_global["score"] = es
                            best_global["a"] = a.copy()
                    except Exception:
                        pass
        else:
            fails += 1
            alpha = max(0.02, alpha * 0.78)
            if fails == 12:
                p = min(p_max, p * 1.25)
            elif fails == 40:
                p = max(1.1, p * 0.85)
                beta = 0.08
                alpha = 0.32
            elif fails > 160:
                base = local_best_a
                if best_global is not None:
                    gb = best_global.get("a")
                    gs = best_global.get("score", float("inf"))
                    if gb is not None and np.isfinite(gs) and gs < local_best_score - 1e-6:
                        base = np.asarray(gb, dtype=np.float64)
                b = base * np.exp(rng.normal(0.0, 0.18, size=n))
                b = _normalize(b)
                cb = _conv_fft(b, nfft_c)
                bs = _score_from_c(cb, n)
                if np.isfinite(bs):
                    a, c, score = b, cb, bs
                else:
                    a = local_best_a
                    c = _conv_fft(a, nfft_c)
                    score = local_best_score
                fails = 0
                alpha = 0.36
                p = max(1.3, min(p_max, p * 0.75))
                beta = 0.05

    return local_best_a, local_best_score


def _optimize_small(N, large_cands, seed, small_end):
    if small_end - time.monotonic() < 15.0:
        return None
    if N <= 1:
        return None
    n = 256
    nfft_s = _next_pow2(2 * n - 1)
    nfft_ds = _next_pow2(3 * n - 2)

    starts = []

    gen_end = min(small_end, time.monotonic() + 10.0)
    for arr in _structured_candidates(n, seed=seed + 17, end_time=gen_end, nfft=nfft_s):
        starts.append(arr)

    for arr in large_cands[:6]:
        small = _resample(arr, n)
        starts.append(small)

    scored = []
    for arr in starts:
        s = _normalize(arr)
        c = _conv_fft(s, nfft_s)
        sc = _score_from_c(c, n)
        if np.isfinite(sc):
            scored.append((sc, s))
    if not scored:
        scored = [(float("inf"), np.ones(n))]
    scored.sort(key=lambda t: t[0])
    starts = [a for _, a in scored[:3]]

    best_small = None
    best_small_score = float("inf")

    for idx, start in enumerate(starts[:2]):
        rem = small_end - time.monotonic()
        if rem < 20.0:
            break
        s = _normalize(start)
        if idx == 0 and rem > 80.0:
            lp_end = time.monotonic() + min(35.0, rem - 50.0)
            s = _lp_improve(s, lp_end, max_iters=18)

        p_end = min(small_end, time.monotonic() + max(5.0, rem - 10.0))
        s2, sc = _improve(
            s,
            end_time=p_end,
            rng=np.random.default_rng(seed + 101 + idx),
            nfft_c=nfft_s,
            nfft_d=nfft_ds,
            p_start=1.25,
            p_max=14.0,
            alpha_start=0.32,
            do_exact=False,
            best_global=None,
            exact_every=100000,
        )
        if s2 is not None and np.isfinite(sc) and sc < best_small_score:
            best_small = s2
            best_small_score = sc

    return best_small


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    B = max(1.0, float(budget_s))
    end_all = t0 + max(1.0, B - 1.0)
    work_end = t0 + max(10.0, B - 45.0)
    work_end = min(work_end, end_all - 5.0)
    if work_end < t0 + 2.0:
        work_end = end_all - 0.5

    rng = np.random.default_rng(seed)

    parent = None
    N = 5582
    try:
        pa = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if pa.size > 0:
            pa = np.where(np.isfinite(pa), pa, 0.0)
            pa = np.clip(pa, 0.0, 1000.0)
            if pa.sum() > 0.0:
                parent = _normalize(pa)
                N = int(pa.size)
    except Exception:
        parent = None

    if N < 64:
        N = 512
        parent = None
    if N > 16384:
        if parent is not None:
            parent = _resample(parent, 12000)
        N = 12000

    uniform = np.full(N, 1.0 / float(N))
    nfft_c = _next_pow2(2 * N - 1)
    nfft_d = _next_pow2(3 * N - 2)

    cands = []
    if parent is not None:
        cands.append(parent)
    cands.append(uniform)

    if B > 60.0:
        gen_end = min(work_end, t0 + min(30.0, max(2.0, 0.03 * B)))
        try:
            cands.extend(_structured_candidates(N, seed=seed + 7, end_time=gen_end, nfft=nfft_c))
        except Exception:
            pass

    small_best = None
    if B > 260.0 and N > 256:
        small_end = min(work_end, time.monotonic() + min(170.0, 0.13 * B))
        if small_end > time.monotonic() + 20.0:
            try:
                small_best = _optimize_small(N, cands, seed=seed + 91, small_end=small_end)
            except Exception:
                small_best = None

    if small_best is not None:
        try:
            sl = _resample(small_best, N)
            cands.append(sl)
            cands.append(sl[::-1])
            for i in range(3):
                noisy = sl * np.exp(rng.normal(0.0, 0.06, size=N))
                cands.append(_normalize(noisy))
        except Exception:
            pass

    sur_scores = []
    for a in cands:
        try:
            a = _normalize(a)
            if a.size != N:
                a = _resample(a, N)
            c = _conv_fft(a, nfft_c)
            s = _score_from_c(c, N)
            if np.isfinite(s):
                sur_scores.append((s, a))
        except Exception:
            pass
    sur_scores.sort(key=lambda t: t[0])

    def exact_score_arr(a):
        return float(evaluate_sequence(a.tolist()))

    best_global = {"score": float("inf"), "a": None}
    for _, a in sur_scores[:18]:
        try:
            es = exact_score_arr(a)
            if np.isfinite(es) and es < best_global["score"]:
                best_global["score"] = es
                best_global["a"] = a.copy()
        except Exception:
            pass

    if best_global["a"] is None:
        try:
            es = exact_score_arr(uniform)
            if np.isfinite(es):
                best_global["score"] = es
                best_global["a"] = uniform.copy()
            else:
                best_global["score"] = 2.0
                best_global["a"] = uniform.copy()
        except Exception:
            best_global["score"] = 2.0
            best_global["a"] = uniform.copy()

    if best_global["score"] < 2.0 - 1e-8:
        start = np.asarray(best_global["a"], dtype=np.float64).copy()
    elif sur_scores:
        start = np.asarray(sur_scores[0][1], dtype=np.float64).copy()
    else:
        start = uniform.copy()

    if work_end - time.monotonic() > 20.0:
        try:
            local_best, local_score = _improve(
                start,
                end_time=work_end,
                rng=rng,
                nfft_c=nfft_c,
                nfft_d=nfft_d,
                p_start=1.5,
                p_max=22.0,
                alpha_start=0.34,
                do_exact=True,
                best_global=best_global,
                exact_every=(900 if N > 7000 else 380),
            )
        except Exception:
            local_best = local_score = None

    final_pool = []

    def add_final(a):
        if a is None:
            return
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
            if a.size != N:
                a = _resample(a, N)
            a = _normalize(a)
            if np.isfinite(a).all() and a.size > 0:
                final_pool.append(a)
        except Exception:
            pass

    add_final(best_global.get("a"))
    try:
        add_final(local_best)
    except Exception:
        pass
    add_final(start)
    for i in range(min(3, len(sur_scores))):
        add_final(sur_scores[i][1])
    add_final(parent)
    add_final(uniform)

    final_best = best_global.get("a") if best_global.get("a") is not None else uniform
    final_score = float(best_global.get("score", float("inf")))

    max_checks = 6 if N > 9000 else 12
    for a in final_pool[:max_checks]:
        if time.monotonic() > end_all - 1.0:
            break
        try:
            es = exact_score_arr(a)
            if np.isfinite(es) and es < final_score:
                final_score = es
                final_best = a.copy()
        except Exception:
            pass

    if final_best is None or not np.isfinite(final_score):
        final_best = uniform.copy()
        final_score = 2.0

    try:
        return [float(x) for x in final_best.tolist()]
    except Exception:
        return [float(x) for x in uniform.tolist()]
```
