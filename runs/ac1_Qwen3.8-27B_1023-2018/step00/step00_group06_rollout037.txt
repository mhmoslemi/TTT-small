import numpy as np

def compute_convolution(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    # Use FFT for efficiency
    size = 2 * n - 1
    # Round up to next power of 2 for FFT efficiency
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:size]
    return conv

def score_and_grad(a):
    """Compute score and gradient of the score w.r.t. a."""
    n = len(a)
    conv = compute_convolution(a)
    S = a.sum()
    M = conv.max()
    m_star = int(conv.argmax())
    score = 2.0 * n * M / (S * S)

    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    
    grad = np.full(n, -M, dtype=np.float64)  # default: -M (outside window)

        mirrored = m_star - indices  # these are in [0, n-1]
        grad[lo:hi+1] = S * a[mirrored] - M
    
    grad *= 4.0 * n / (S ** 3)
    return score, grad, m_star, M, S

    best_score = 2.0 * n * np.convolve(a, a).max() / a.sum()**2
