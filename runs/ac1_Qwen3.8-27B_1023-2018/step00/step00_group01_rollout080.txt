def solve_lp(a, M, lags, tlimit):
   if linprog is None: return None
   K=len(lags)
   try:
      A = np.zeros((K,n), dtype=np.float64)
      b = np.empty(K,dtype=np.float64)
      a_use = a.copy()
      if np.any(a_use == 0.0):
         a_use += 1e-12
      for r,k in enumerate(lags):
          lo = k - n + 1
          if lo <0: lo=0
          hi = k
          if hi >= n: hi=n-1
          if lo <= hi:
             A[r, lo:hi+1] = a_use[k-hi:k-lo+1][::-1]
          sc = A[r].max()
          if sc > 0:
             A[r] /= sc
             b[r] = M / sc
          else:
             b[r] = 1.0
      c_obj = -np.ones(n,dtype=np.float64)
      bounds = np.column_stack((np.zeros(n), np.full(n,100.0)))
      options = {'time_limit': float(tlimit)}
      res=linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs', options=options)
      x = getattr(res,'x',None)
      if x is not None and np.all(np.isfinite(x)):
          return x
   except Exception:
      pass
   return None

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def _conv_self(a, fft_size):
    n = int(a.size)
    if n <= 0:
        return np.array([], dtype=np.float64)
    if fft_size is None:
        fft_size = _next_pow2(2 * n - 1)
    fa = rfft(a, fft_size)
    c = irfft(fa * fa, fft_size)
    return c[:2 * n - 1]


def _score_normalized(a, fft_size):

    if c.size == 0:

    m = float(c.max())
    if not math.isfinite(m) or m < 0.0:
        m = 0.0
    return 2.0 * float(a.size) * m


def _hard_gradient(a, c):
    n = int(a.size)
    k = int(c.argmax())

    zs -= zs.max()
    e = np.exp(zs)
    s = float(e.sum())
    if s <= 1e-300:
        return np.ones_like(e, dtype=np.float64) / float(e.size)
    return e / s


def _next_prime(x):
    y = int(max(2, x))
    while True:
        r = int(math.isqrt(y))
        ok = True
        for d in range(2, r + 1):
            if y % d == 0:
                ok = False
                break
        if ok:
            return y
        y += 1


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not math.isfinite(budget) or budget <= 10.0:
        budget = 60.0
    deadline = start + max(1.0, budget - 35.0)
    rng = np.random.default_rng(seed)

    # The supplied parent, if it exists, is used directly.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        base = np.ones(6121, dtype=np.float64)

    if base.ndim != 1 or base.size == 0 or not np.all(np.isfinite(base)):
        base = np.ones(6121, dtype=np.float64)
    base = np.clip(base, 0.0, 1000.0)
    n = int(base.size)
    if base.sum() < 0.01:
        base = np.ones(n, dtype=np.float64)

    fft_size = _next_pow2(2 * n - 1)
    grad_fft_size = _next_pow2(3 * n - 2)
    k_lag_count = 2 * n - 1
    grad_idx = (k_lag_count - 1) - np.arange(n, dtype=np.int64)

    base_norm = _normalize(base)

    # Exact evaluator wrappers.
    def exact_score(a):
        if a is None or a.size == 0:
            return float("inf")
        try:
            return float(evaluate_sequence([float(x) for x in np.clip(a, 0.0, 1000.0)]))
        except Exception:
            return float("inf")

    best_seq = base_norm.copy()
    best_sur = float(_score_normalized(best_seq, fft_size))
    best_exact_seq = best_seq.copy()
    best_exact = exact_score(best_seq)
    last_exact = start

    def consider(seq, force_exact_interval_s=45.0):
        nonlocal best_seq, best_sur, best_exact_seq, best_exact, last_exact
        if seq is None:
            return
        seq = _normalize(seq)
        sc = float(_score_normalized(seq, fft_size))
        if not math.isfinite(sc):
            return
        if sc < best_sur - 1e-12:
            best_seq = seq.copy()
            best_sur = sc
        now = time.monotonic()
        if now - last_exact > force_exact_interval_s or not math.isfinite(best_exact):
            e = exact_score(seq)
            last_exact = now
            if math.isfinite(e) and (not math.isfinite(best_exact) or e < best_exact - 1e-12):
                best_exact = e
                best_exact_seq = seq.copy()

    # ---------- Candidate generation ----------
    cands = []

    def add(a):
        try:
            a = _normalize(a)
            if a.size == n and np.all(np.isfinite(a)) and a.sum() > 0:
                cands.append(a)
        except Exception:
            pass

    add(base_norm)
    add(base_norm + 1e-4 / n)
    add(np.ones(n, dtype=np.float64) / n)

    # Broken-symmetry parent variants.
    idx = np.linspace(0.0, 1.0, n)
    add(base_norm * (0.4 + 1.2 * idx))
    add(base_norm * (1.6 - 1.2 * idx))
    add(base_norm * (0.6 + 0.8 * np.sin(2.0 * math.pi * idx + 0.6)))

    # Smooth random positive variants.
    for _ in range(4):
        a = np.abs(rng.normal(0.0, 1.0, n)) + 0.05
        klen = int(min(21, max(3, (n // 150) | 1)))
        if klen % 2 == 0:
            klen += 1
        kernel = np.ones(klen, dtype=np.float64)
        kernel /= kernel.sum()
        try:
            a = np.convolve(a, kernel, mode="same")
        except Exception:
            pass
        add(np.maximum(a, 0.0))

    # Blocks.
    nb = max(16, min(256, n // 16))
    try:
        bw = rng.gamma(2.0, 1.0, nb)
        a = np.repeat(bw, n // nb + 1)[:n]
        a = np.convolve(a, np.ones(5, dtype=np.float64) / 5.0, mode="same")
        add(np.maximum(a, 0.0))

        bw = (rng.random(nb) > 0.5).astype(np.float64)
        a = np.repeat(bw, n // nb + 1)[:n]
        a = np.convolve(a, np.ones(7, dtype=np.float64) / 7.0, mode="same")
        add(np.maximum(a, 1e-3))
    except Exception:
        pass

    # Cosine-like variants.
    ii = np.arange(n, dtype=np.float64)
    for f in (2, 3, 5, 8, 13):
        ph = rng.uniform(0.0, 2.0 * math.pi)
        add(np.maximum(1.0 + 0.6 * np.cos(2.0 * math.pi * f * ii / n + ph), 0.02))

    # Legendre / quadratic-residue inspired binary candidates, lightly randomized.
    try:
        p = _next_prime(n)
        qr = np.zeros(n, dtype=np.float64)
        for x in range(1, p):
            if x >= n:
                break
            if pow(x, (p - 1) // 2, p) == 1:
                qr[x] = 1.0
        # Try both residue classes and a few shifts.
        for arr in (qr, 1.0 - qr):
            for s in (0, 1, 3, 7, int(n // 4), int(n // 2)):
                b = np.roll(arr, int(s % n), axis=0)
                add(np.maximum(b, 0.0))
        # Add zero sometimes.
        qr0 = qr.copy()
        qr0[0] = 1.0
        qn0 = 1.0 - qr0
        add(qr0)
        add(qn0)
    except Exception:
        pass

    # Score and keep best few.
    scored = []
    for a in cands:
        sc = _score_normalized(a, fft_size)
        if math.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda x: x[0])

    if not scored:
        scored = [(best_sur, best_seq)]

    for sc, a in scored[:3]:
        consider(a, force_exact_interval_s=1e9)

    # ---------- Lightweight local antipeak moves ----------
    def quick_subgrad(a, limit_s=3.0):
        a = _normalize(a)
        start_t = time.monotonic()
        cur = _score_normalized(a, fft_size)
        step_scale = 1.0
        no_improve = 0
        for it in range(600):
            if time.monotonic() - start_t > limit_s:
                break
            if time.monotonic() > deadline:
                break
            c = _conv_self(a, fft_size)
            if c.size == 0:
                break
            m = float(c.max())
            if m <= 1e-300:
                break
            if it % 4 == 0:
                q = min(5, c.size)
                part = np.argpartition(c, -q)[-q:]
                w = c[part]
                w = np.maximum(w, 1e-300)
                k = int(rng.choice(part, p=w / w.sum()))
            else:
                k = int(c.argmax())

            g = _hard_gradient(a, c)
            gm = float(g.max())
            if gm <= 1e-300:
                break
            r = g / gm
            lr = 0.08 * step_scale
            cand = a * np.exp(-lr * r)
            cand = np.maximum(cand, 0.0)
            s = float(cand.sum())
            if s <= 1e-300:
                step_scale *= 0.5
                continue
            cand /= s
            sc = _score_normalized(cand, fft_size)
            if sc < cur - 1e-12:
                a = cand
                cur = sc
                step_scale = min(1.0, 1.05 * step_scale)
                no_improve = 0
            else:
                step_scale *= 0.75
                no_improve += 1
                if no_improve > 40:
                    break
        return a

    top = [a for _, a in scored[:3]]
    for a in top:
        try:
            ra = quick_subgrad(a, limit_s=2.5)
            consider(ra, force_exact_interval_s=1e9)
        except Exception:
            pass

    if math.isfinite(best_exact) and best_exact < 1.45:
        return [float(x) for x in _normalize(best_exact_seq)]

    # ---------- Top-lag LP refinement ----------
    lp_cutoff = min(deadline, start + max(120.0, min(520.0, 0.40 * budget)))

    def solve_lp(a, M, lags, tlimit):
        if linprog is None:
            return None
        K = int(len(lags))
        if K <= 0:
            return None
        try:
            A = np.zeros((K, n), dtype=np.float64)
            b = np.empty(K, dtype=np.float64)
            a_use = a.copy()
            if np.any(a_use == 0.0):
                a_use += 1e-12
            for r, k in enumerate(lags):
                lo = k - n + 1
                if lo < 0:
                    lo = 0
                hi = k
                if hi >= n:
                    hi = n - 1
                if lo <= hi:
                    A[r, lo:hi + 1] = a_use[k - hi:k - lo + 1][::-1]
                sc = float(A[r].max())
                if sc > 0.0:
                    A[r] /= sc
                    b[r] = M / sc
                else:
                    b[r] = 1.0
            c_obj = -np.ones(n, dtype=np.float64)
            bounds = np.column_stack((np.zeros(n, dtype=np.float64), np.full(n, 50.0, dtype=np.float64)))
            options = {"time_limit": float(tlimit)}
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs", options=options)
            x = getattr(res, "x", None)
            if x is None:
                return None
            x = np.asarray(x, dtype=np.float64)
            if not np.all(np.isfinite(x)):
                return None
            return np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
        except Exception:
            return None

    def run_lp(a0, cutoff, max_iter=400):
        a = _normalize(a0)
        cur = _score_normalized(a, fft_size)
        best_local = a.copy()
        best_local_score = cur
        no_imp = 0
        K = min(240, k_lag_count)
        maxK = min(k_lag_count, 900)

        for it in range(max_iter):
            if time.monotonic() > cutoff:
                break
            rem = cutoff - time.monotonic()
            if rem < 12.0:
                break
            c = _conv_self(a, fft_size)
            M = float(c.max())
            if not math.isfinite(M) or M <= 1e-15:
                break

            if K >= c.size:
                lags = np.arange(c.size)
            else:
                lags = np.argpartition(c, -K)[-K:]
                if it % 5 == 4:
                    K2 = int(min(c.size, max(K, 2 * K)))
                    wide = np.argpartition(c, -K2)[-K2:]
                    if wide.size > 0:
                        rnd = rng.choice(wide, size=max(1, K // 4), replace=False)
                        lags = np.unique(np.concatenate([lags, rnd]))[:K]

            tlimit = max(1.0, min(7.0, rem / 25.0))
            g = solve_lp(a, M, lags, tlimit)
            if g is None:
                K = max(80, K // 2)
                continue
            g = np.maximum(np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0), 0.0)
            gs = float(g.sum())
            if gs <= 1e-12:
                continue
            g /= gs

            best_cand = None
            best_cscore = cur
            for alpha in (0.15, 0.30, 0.50, 0.10, 0.70, 1.0):
                cand = (1.0 - alpha) * a + alpha * g
                cand = np.maximum(cand, 0.0)
                s = float(cand.sum())
                if s <= 1e-12:
                    continue
                cand /= s
                sc = _score_normalized(cand, fft_size)
                if math.isfinite(sc) and sc < best_cscore:
                    best_cscore = sc
                    best_cand = cand

            if best_cand is not None and best_cscore < cur - 1e-10:
                a = best_cand
                cur = best_cscore
                if cur < best_local_score - 1e-12:
                    best_local = a.copy()
                    best_local_score = cur
                no_imp = 0
                K = int(min(maxK, K + 70))
            else:
                no_imp += 1
                if no_imp >= 5:
                    step = 0.05
                    a = _normalize(a * np.exp(rng.normal(0.0, step, n)))
                    cur = _score_normalized(a, fft_size)
                    if cur < best_local_score - 1e-12:
                        best_local = a.copy()
                        best_local_score = cur
                    no_imp = 0
                    K = int(min(maxK, max(120, K // 2 + 80)))

            if math.isfinite(best_exact) and best_exact < 1.45:
                break
        return best_local, best_local_score

    try:
        a_lp, sc_lp = run_lp(best_seq, lp_cutoff, max_iter=220)
        if a_lp is not None:
            consider(a_lp, force_exact_interval_s=1e9)
    except Exception:
        pass

    if math.isfinite(best_exact) and best_exact < 1.45:
        return [float(x) for x in _normalize(best_exact_seq)]

    # ---------- Smoothed L-BFGS-B refinement ----------
    lbfgs_cutoff = min(deadline, start + max(240.0, min(1000.0, 0.78 * budget)))

    def run_lbfgs(a0, cutoff, max_evals=3500):
        if minimize is None:
            return _normalize(a0), _score_normalized(a0, fft_size)
        a = _normalize(a0)
        cur = _score_normalized(a, fft_size)
        best_local = a.copy()
        best_local_score = cur
        total_evals = 0

        for factor in (60.0, 120.0, 240.0):
            if time.monotonic() > cutoff or total_evals > max_evals:
                break
            c0 = _conv_self(a, fft_size)
            cmax = float(c0.max())
            if not math.isfinite(cmax) or cmax <= 1e-300:
                break
            tau = max(1e-15, cmax / factor)
            logK = float(math.log(float(k_lag_count)))

            def fun(z, tau=tau, logK=logK):
                try:
                    zs = np.clip(z, -60.0, 60.0)
                    zs = zs - zs.max()
                    ez = np.exp(zs)
                    ss = float(ez.sum())
                    if ss <= 1e-300 or not math.isfinite(ss):
                        return 1e100, np.zeros_like(zs, dtype=np.float64)
                    aa = ez / ss

                    cc = _conv_self(aa, fft_size)
                    cc = np.maximum(cc, 0.0)
                    m = float(cc.max())
                    if not math.isfinite(m) or m <= 0.0:
                        return 0.0, np.zeros_like(aa, dtype=np.float64)

                    w = np.exp((cc - m) / tau)
                    wsum = float(w.sum())
                    if not math.isfinite(wsum) or wsum <= 1e-300:
                        p = np.zeros_like(cc, dtype=np.float64)
                        p[int(cc.argmax())] = 1.0
                    else:
                        p = w / wsum

                    q = p[::-1]
                    fa = rfft(aa, grad_fft_size)
                    fq = rfft(q, grad_fft_size)
                    cq = irfft(fa * fq, grad_fft_size)
                    C = cq[grad_idx]
                    g = 2.0 * C
                    dot = float(np.dot(g, aa))
                    grad_z = aa * (g - dot)
                    grad_z *= 2.0 * float(n)
                    grad_z = np.nan_to_num(grad_z, nan=0.0, posinf=0.0, neginf=0.0)

                    ell = float(np.log(wsum)) if wsum > 0.0 else -math.log(10.0)
                    fval = 2.0 * float(n) * (m + tau * (ell - logK))
                    if not math.isfinite(fval):
                        return 1e100, np.zeros_like(zs, dtype=np.float64)
                    return float(fval), grad_z
                except Exception:
                    return 1e100, np.zeros_like(np.asarray(z, dtype=np.float64))

            z = _to_z(a)
            stagnant = 0
            last_best = best_local_score

            # Run in bounded chunks.
            for _ in range(40):
                rem = cutoff - time.monotonic()
                if rem < 4.0 or total_evals > max_evals:
                    break
                maxfun = int(min(90, max(25, rem / 2.0)))
                maxfun = max(10, min(maxfun, max_evals - total_evals))
                if maxfun < 10:
                    break
                try:
                    res = minimize(
                        fun,
                        z,
                        jac=True,
                        method="L-BFGS-B",
                        options={
                            "maxiter": min(60, maxfun),
                            "maxfun": maxfun,
                            "maxls": 15,
                            "ftol": 1e-13,
                            "gtol": 1e-8,
                            "disp": False,
                            "maxcor": 10,
                        },
                    )
                except Exception:
                    break

                nfev = int(getattr(res, "nfev", 0) or 0)
                total_evals += max(0, nfev)
                x = getattr(res, "x", None)
                if x is None or not np.all(np.isfinite(x)):
                    if getattr(res, "status", 0) in (1, 2):
                        break
                    continue

                z = x
                a1 = _from_z(x)
                s1 = _score_normalized(a1, fft_size)
                if math.isfinite(s1) and s1 < best_local_score - 1e-12:
                    best_local = a1.copy()
                    best_local_score = s1
                    stale = best_local_score < last_best - 1e-10
                    stagnant = 0 if stale else stagnant
                    last_best = best_local_score

                if not stale:
                    stagnant += 1

                if getattr(res, "status", 0) in (1, 2) or stagnant >= 3:
                    break

                if math.isfinite(best_exact) and best_exact < 1.45:
                    return best_local, best_local_score

            # Optional small restart if still improving.
            if time.monotonic() > cutoff:
                break
            z = _to_z(best_local)
            z = z + rng.normal(0.0, 0.05, n)

        return best_local, best_local_score

    try:
        a_lb, sc_lb = run_lbfgs(best_seq, lbfgs_cutoff, max_evals=3500)
        if a_lb is not None:
            consider(a_lb, force_exact_interval_s=1e9)
    except Exception:
        pass

    # Second LP round if useful.
    if math.isfinite(best_exact) and best_exact >= 1.45 and time.monotonic() < deadline - 60.0:
        try:
            lp_cutoff2 = min(deadline, time.monotonic() + max(60.0, min(220.0, (deadline - time.monotonic()) * 0.55)))
            a_lp2, _ = run_lp(best_seq, lp_cutoff2, max_iter=120)
            if a_lp2 is not None:
                consider(a_lp2, force_exact_interval_s=1e9)
        except Exception:
            pass

    # ---------- Final bounded stochastic polish ----------
    cur_seq = best_seq.copy()
    cur_score = _score_normalized(cur_seq, fft_size)
    no_improve = 0
    while time.monotonic() < deadline - 8.0:
        # Random block perturbation or small subgradient step.
        mode = rng.random()
        if mode < 0.45:
            width = max(1, n // 80)
            s0 = int(rng.integers(0, max(1, n - width + 1)))
            cand = cur_seq.copy()
            cand[s0:s0 + width] *= float(np.exp(rng.normal(0.0, 0.07)))
            if rng.random() < 0.15:
                cand = 0.5 * (cand + cand[::-1])
        elif mode < 0.75:
            c = _conv_self(cur_seq, fft_size)
            g = _hard_gradient(cur_seq, c)
            gm = float(g.max())
            if gm <= 1e-300:
                continue
            lr = 0.04 * (0.5 + rng.random())
            cand = cur_seq * np.exp(-lr * (g / gm))
        else:
            cand = cur_seq + rng.normal(0.0, 0.01 / max(1, math.sqrt(n)), n)

        cand = np.maximum(cand, 0.0)
        s = float(cand.sum())
        if s <= 1e-12:
            continue
        cand /= s
        sc = _score_normalized(cand, fft_size)
        if math.isfinite(sc) and sc < cur_score - 1e-11:
            cur_seq = cand
            cur_score = sc
            consider(cand, force_exact_interval_s=20.0)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 120:
                cur_seq = best_seq.copy()
                cur_score = _score_normalized(cur_seq, fft_size)
                no_improve = 0

        if math.isfinite(best_exact) and best_exact < 1.45:
            break

    # ---------- Final validation and return ----------
    final = best_exact_seq
    if final is None or final.size == 0 or not np.all(np.isfinite(final)):
        final = best_seq
    if final is None or final.size == 0 or not np.all(np.isfinite(final)):
        final = np.ones(n, dtype=np.float64) / float(max(1, n))

    final = _normalize(final)

    best_final_arr = None
    best_final_score = float("inf")
    for scale in (500.0, 100.0, 10.0, 1.0, 1000.0):
        try:
            arr = np.clip(final * scale, 0.0, 1000.0)
            if not np.all(np.isfinite(arr)) or arr.sum() < 0.01:
                continue
            sc = exact_score(arr)
            if math.isfinite(sc) and sc < best_final_score:
                best_final_score = sc
                best_final_arr = arr
        except Exception:
            pass

    if best_final_arr is None:
        best_final_arr = np.ones(n, dtype=np.float64)
        if best_final_arr.size == 0:
            best_final_arr = np.ones(1, dtype=np.float64)

    return [float(x) for x in best_final_arr]
