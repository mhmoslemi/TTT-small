```python
import numpy as np
import time
from scipy.fft import fft, ifft

def autocorrelation(a):
    """Compute full autocorrelation (self-convolution) of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for efficient FFT
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    F = fft(a, n=fft_size)
    ac = np.real(ifft(F * F.conjugate(), n=fft_size))[:size]
    return ac

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```
