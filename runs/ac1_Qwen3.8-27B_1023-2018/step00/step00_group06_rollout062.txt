```python
def project_simplex(v, s=1.0):
    """Project v onto {x: sum(x)=s, x>=0}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
import numpy as np
import time
import cvxpy as cp


def _lp_direction(f, M, n, k_lo, k_hi):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in [k_lo,k_hi], g>=0."""
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in range(k_lo, k_hi + 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        row = np.zeros(n)
        idx = np.arange(j_lo, j_hi + 1)
        row[idx] = f[k - idx]
        constraints.append(row @ g <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    for solver in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.GLPK]:
        try:
            kw = {}
            if solver in (cp.SCS, cp.CLARABEL):
                kw = dict(max_iters=200000, verbose=False)
            elif solver == cp.ECOS:
                kw = dict(max_iters=50000)
            prob.solve(solver=solver, **kw)
            if g.value is not None:
                return np.maximum(g.value, 0)
        except Exception:
            continue
    return None


def _logsumexp(x, scale=50.0):
    shifted = x * scale
    m = shifted.max()
    return (m / scale) + np.log(np.sum(np.exp(shifted - m))) / scale
```
