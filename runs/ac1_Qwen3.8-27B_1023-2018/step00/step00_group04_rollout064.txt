import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 8.0)

    # --- Initialization ---

        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except (NameError, Exception):
        parent = np.ones(3809, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0).astype(np.float64)
    n = parent.size
    if n == 0:
        return [1.0]

    total = parent.sum()
    if total < 1e-10:
        parent[:] = 1.0 / n
    else:
        parent /= total

    def smooth_perturb(base, scale, rng_local):
        freqs = rng_local.integers(1, 6, size=10)
        phases = rng_local.uniform(0, 2 * np.pi, size=10)
        amps = rng_local.uniform(0, scale, size=10)
        idx = np.arange(n, dtype=np.float64)
        pert = np.zeros(n, dtype=np.float64)
        for k in range(10):
            pert += amps[k] * np.sin(2 * np.pi * freqs[k] * idx / n + phases[k])
        result = base + pert
        result = np.clip(result, 0, None)
        s = result.sum()
        if s < 1e-15:
            result = base.copy()
        else:
            result /= s
        return result

    current = smooth_perturb(parent, 0.05, rng)
    best = current.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    if not np.isfinite(best_score):
        current = parent.copy()
        best = parent.copy()
        best_score = float(evaluate_sequence(best.tolist()))

    no_improve = 0
    iteration = 0

    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        iteration += 1

        conv_result = np.convolve(current, current)
        M = float(conv_result.max())
        k_star = int(conv_result.argmax())

        direction = None
        used_lp = False

        # LP-based direction (anteater method)
        if remaining > 50 and (iteration % 3 != 0 or no_improve >= 10):
            try:
                j_idx = np.repeat(np.arange(n, dtype=np.int32), n)
                i_idx = np.tile(np.arange(n, dtype=np.int32), n)
                rows = j_idx + i_idx
                cols = j_idx
                vals = np.repeat(current, n)

                A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
                del rows, cols, vals, j_idx, i_idx

                c = -np.ones(n, dtype=np.float64)
                b_ub = np.full(2 * n - 1, M, dtype=np.float64)
                bounds = [(0.0, None)] * n

                lp_limit = min(remaining - 25, 30.0)
                res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                              method='highs',
                              options={'time_limit': lp_limit, 'presolve': True})
                del A

                if res.status == 0 and res.x is not None:
                    g = np.asarray(res.x, dtype=np.float64)
                    g_sum = g.sum()
                    if g_sum > 1e-10:
                        g /= g_sum
                        direction = g
                        used_lp = True
                del res
            except Exception:
                pass

        # Subgradient + noise fallback
        if direction is None:
            subgrad = np.zeros(n, dtype=np.float64)
            lo = max(0, k_star - (n - 1))
            hi = min(n - 1, k_star)
            for j in range(lo, hi + 1):
                subgrad[j] = 2.0 * current[k_star - j]

            noise_level = 0.05 + 0.3 * min(no_improve / 40.0, 1.0)
            direction = -subgrad + noise_level * M * rng.random(n)
            direction = np.clip(direction, 0, None)
            s = direction.sum()
            if s < 1e-10:
                direction = rng.random(n)
                direction /= (direction.sum() + 1e-15)
            else:
                direction /= s

        # Line search
        best_t = 0.0

                best_trial_M = trial_M
                best_t = t

            new_current = (1.0 - best_t) * current + best_t * direction
            new_M = best_trial_M

            # Accept if improved or via simulated annealing

                current = new_current
                score = float(evaluate_sequence(current.tolist()))

                    best_score = score
                    no_improve = 0

                    no_improve += 1

                no_improve += 1

                    no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 10:
                current = smooth_perturb(best, 0.03, rng)
                no_improve = 0

        # Periodic re-evaluation of best

    # Final validation

    final_score = float(evaluate_sequence(final_list))
    if not np.isfinite(final_score) or final_score >= 2.0:
        fallback = np.ones(n, dtype=np.float64)
        final_list = [float(x) for x in fallback]
        final_score = float(evaluate_sequence(final_list))

    return final_list
