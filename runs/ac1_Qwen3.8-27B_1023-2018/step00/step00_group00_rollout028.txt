def _compute_score(a):
    """Fast score computation using FFT convolution."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    b = fftconvolve(a, a)
    return 2.0 * n * b.max() / (s * s)


def _gradient_step(a, k_star, alpha):
    """One projected gradient step on the simplex {a>=0, sum(a)=1}."""
    n = len(a)
    # Subgradient of max_k (a*a)[k] at k_star: sg[m] = 2*a[k_star-m] if valid
    sg = np.zeros(n)
    j_max = min(n - 1, k_star)
    j_min = max(0, k_star - (n - 1))
    for m in range(j_min, j_max + 1):
        sg[m] = 2.0 * a[k_star - m]
    # Center to preserve sum
    sg -= sg.mean()
    # Step
    a_new = a - alpha * sg
    a_new = np.maximum(a_new, 0.0)
    a_new = np.minimum(a_new, 1000.0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(n) / n
    else:
        a_new /= s

def _smoothed_gradient_step(a, b, tau, alpha):
