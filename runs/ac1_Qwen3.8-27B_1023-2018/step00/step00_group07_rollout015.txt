```python
def convolve_max(x, n):
    """Compute max of x*x using FFT. Returns (max_val, max_idx)."""
    size = 1
    while size < 2*n - 1:
        size *= 2
    X = np.fft.rfft(x, size)
    Y = np.fft.irfft(X * X, size)
    # Y has length size, first 2*n-1 entries are the convolution
    Y = Y[:2*n-1]
    return np.max(Y), np.argmax(Y)
```
