If |S| is small (say < 100), this is a small LP that's easy to solve!

    else:

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fftconvolve(a, a)
        M = c.max()
        
        # Find active set (peaks within 1% of max)
        threshold = M * 0.99
        S = np.where(c >= threshold)[0]
        
        # Build linearized LP
        # min t
        # s.t. 2*(P_k(a))^T d - t <= -q_k(a) for k in S
        #      1^T d = 0
        #      d_i >= -a_i
        
        m = len(S)
        # A matrix: m rows for quadratic constraints, 1 row for sum constraint
        # Variables: d (n) and t (1), total n+1
        
        # For constraint k: 2*sum_i (P_k(a))_i * d_i - t <= -q_k(a)
        # (P_k(a))_i = a_{k-i} if 0 <= k-i < n else 0
        
        A_rows = []
        b_rhs = []
        for k in S:
            row = np.zeros(n + 1)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    row[i] = 2.0 * a[j]
            row[n] = -1.0  # coefficient of t
            A_rows.append(row)
            b_rhs.append(-c[k])  # q_k(a) = c[k]

        sum_row = np.zeros(n + 1)

        nonneg_A = np.zeros((n, n + 1))

        A_ub_full = np.vstack([A_ub, nonneg_A])
        b_ub_full = np.concatenate([b_rhs, a])

        b_eq = np.array([0.0])

                     A_eq=A_eq, b_eq=b_eq, method='highs')
