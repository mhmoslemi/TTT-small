import time
from scipy.fft import fft, ifft, next_fast_len

def ac1_score(a):
    """Fast AC1 computation using FFT."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    # Full convolution via FFT
    size = 2 * n
    Af = fft(a, size)
    bc = ifft(Af * Af, size).real[:2*n-1]

    # Using size = 2*n-1 or next_fast_len(2*n-1)
    M = bc.max()
    return 2.0 * n * M / (s * s)
