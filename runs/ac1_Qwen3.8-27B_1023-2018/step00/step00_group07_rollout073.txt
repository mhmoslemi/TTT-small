from scipy.signal import fftconvolve


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _smooth_grad_step(f, t, N):
    """One normalized smooth-gradient step for min  max_k (f*f)[k].

    Returns (new_f, c, surrogate) where surrogate = 2n*max(c) for sum(f)=1 f.
    f is assumed nonnegative, sums to 1. N is precomputed FFT length.
    Returns None on failure.
    """
    n = f.size
    c = fftconvolve(f, f, mode="full")
    c = np.maximum(c, 0.0)
    cmax = float(c.max())
    surrogate = 2.0 * n * cmax

    # softmax weights (numerically stable)
    cs = t * (c - cmax)
    w = np.exp(cs)
    ws = float(w.sum())
    if not np.isfinite(ws) or ws <= 0.0:
        return (f, c, surrogate)
    w = w / ws

    # cross[i] = sum_k w[k] f[k-i]  ==  (w * f_rev)[i + n - 1]
    f_rev = f[::-1]
    Fw = np.fft.rfft(w, N)
    Fr = np.fft.rfft(f_rev, N)
    conv = np.fft.irfft(Fw * Fr, N)
    cross = conv[(n - 1):(2 * n - 1)]

    # gradient of  (1/t) log sum exp(t c[k])
    grad = f + cross

    return (f, c, surrogate, grad, cross)
