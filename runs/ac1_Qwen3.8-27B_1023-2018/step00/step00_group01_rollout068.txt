import numpy as np
import time

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    import scipy.sparse as sp
    from scipy.optimize import linprog
    
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    
    # Choose sequence length
    n = 1000
    
    # Initialize with geometric decay (asymmetric)
    a = np.exp(-0.5 * np.arange(n) / n)
    a = a / a.sum()
    
    # Get initial score
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    # LP-based optimization
    for iteration in range(500):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution and peak
        conv = np.convolve(a, a)
        M = float(conv.max())
        
        # Build constraint matrix C: (2n-1) x n
        # C[k, j] = a[k-j] if 0 <= k-j < n, else 0
        idx = np.arange(2*n-1)[:, None] - np.arange(n)[None, :]
        mask = (idx >= 0) & (idx < n)
        rows_i, cols_i = np.where(mask)
        C = sp.csr_matrix(
            (a[idx[rows_i, cols_i]], (rows_i, cols_i)),
            shape=(2*n-1, n)
        )
        
        # LP: maximize sum(b) s.t. C @ b <= M, b >= 0
        # scipy form: minimize -1^T b s.t. C @ b <= M, b >= 0
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=C, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)]*n, method='highs')
        
        if not res.success:
            # Try with a relaxed constraint
            res = linprog(c_obj, A_ub=C, b_ub=np.full(2*n-1, M*1.5),
                          bounds=[(0, None)]*n, method='highs')
            if not res.success:
                break
        
        b = res.x
        b_sum = b.sum()
        if b_sum < 1e-12:
            break
        
        # Rescale
        b = b * (a.sum() / b_sum)
        
        # Adaptive step size
        t = 0.2
        a_new = np.clip((1-t)*a + t*b, 0, None)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum() * a.sum()
        
        score = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a_new.copy()
            best_score = score
        
        a = a_new
    
    # Final validation
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # Fallback: uniform sequence
        best = np.ones(10)
