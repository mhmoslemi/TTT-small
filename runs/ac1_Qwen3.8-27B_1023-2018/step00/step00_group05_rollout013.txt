```python
def solve_lp_active(a, M, active_set, tol=1e-8, max_iter=100):
    # Solve: max sum(g) s.t. (a*g)[k] <= M for k in active_set, g >= 0
    # Use scipy.linprog or cvxpy
    
    n = len(a)
    # Constraint matrix for active constraints:
    # Row k: A[k, j] = a[k-j] if 0 <= k-j < n
    import cvxpy as cp
    
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_set:
        # (a*g)[k] = sum_j a[j] * g[k-j] for valid indices
        # = sum_j g[j] * a[k-j] for valid k-j
        # Row: for j, coeff is a[k-j] if 0 <= k-j < n
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        row[lo:hi+1] = a[k - lo:k - lo + 1][::-1]  # a[k-(lo)], a[k-(lo+1)], ...
        # Wait: for column j, the coefficient is a[k-j].
        # So row[j] = a[k-j] if 0 <= k-j < n.
        for j in range(lo, hi+1):
            row[j] = a[k-j]
        constraints.append(row @ g <= M)
    
    prod = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prod.solve(solver=cp.ECOS)  # or SCS, or CBC
    
    if g.value is None:
        return None, active_set
    
    g_val = np.maximum(0, g.value)
    
    # Check for violated constraints
    b = np.convolve(a, g_val)
    violations = b - M
    if violations.max() < tol:
        return g_val, active_set
    
    # Add most violated constraint
    k_new = np.argmax(b)
    if k_new not in active_set:
        active_set.add(k_new)
    return None, active_set
```
