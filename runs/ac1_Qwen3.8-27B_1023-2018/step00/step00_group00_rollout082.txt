def _build_rows(f, rows, n):
    rows=np.asarray(rows, dtype=np.int64)
    if rows.size==0: return csr_matrix((0,n))
    i0 = np.maximum(rows-(n-1), 0)
    i1 = np.minimum(rows, n-1)
    cnt = i1 - i0 + 1
    total = int(cnt.sum())
    data=np.empty(total, dtype=np.float64)
    rarr=np.empty(total, dtype=np.int64)
    carr=np.empty(total, dtype=np.int64)
    pos=0
    for r,i0r,i1r in zip(rows, i0, i1):
       length=i1r-i0r+1
       if length <=0: continue
       segment = f[r - np.arange(i0r, i1r+1, dtype=np.int64)]
       data[pos:pos+length]=segment
       rarr[pos:pos+length]=r
       carr[pos:pos+length]=np.arange(i0r, i1r+1, dtype=np.int64)
       pos += length
    if pos < total:
       data=data[:pos]; rarr=rarr[:pos]; carr=carr[:pos]
    return coo_matrix((data,(rarr,carr)), shape=(rows.size,n)).tocsr()

def _downsample(parent, n):
    if parent is None: return None
    m=parent.size
    if m==0: return None
    if m==n: v=parent.copy()
    elif m>n:
       edges=np.linspace(0,m,n+1,dtype=int)
       v=np.empty(n)
       for i in range(n):
          seg=parent[edges[i]:edges[i+1]]
          v[i]=seg.mean() if seg.size else 0.0
    else:
       v=np.interp(np.linspace(0,1,n), np.linspace(0,1,m), parent)
    return _normalize(v)
