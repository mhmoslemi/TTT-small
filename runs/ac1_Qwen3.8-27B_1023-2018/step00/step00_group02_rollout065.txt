def _fft_convolve_self(a, fft_n):
    """Compute full linear convolution a*a via FFT."""
    A = np.fft.rfft(a, fft_n)
    B = np.fft.irfft(A * A, fft_n)
    return B[:2 * len(a) - 1]


def _compute_gradient_info(a, B, w, grad_fft):
    """Compute c_j = sum_m w[j+m] * a_m via FFT cross-correlation."""
    n = len(a)
