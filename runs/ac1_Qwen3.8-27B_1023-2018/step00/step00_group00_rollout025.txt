            c = _conv(a, N_fft)
            score = 2 * n * c.max()
            
            if score < local_best:
                local_best = score
                no_improve = 0
            else:
                no_improve += 1
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            # Periodic exact evaluation
            eval_counter += 1
            if eval_counter % 500 == 0:
                es = evaluate_sequence(a.tolist())
                if np.isfinite(es) and es < best_score:
                    best_score = es
                    best_a = a.copy()
            
            # Gradient
            if top_k >= 2*n-1:
                top_idx = np.arange(2*n-1)
                w = np.full(2*n-1, 1.0/(2*n-1))
            else:
                top_idx = np.argpartition(c, -top_k)[-top_k:]
                w = np.exp(c[top_idx] / smoothing)
                w /= w.sum()
            
            grad = _grad_from_lags(c, a, top_idx, w)
            grad -= grad.mean()  # sum constraint
            
            # ADAM
            t_adam += 1
            m_hat = beta1 * m_hat + (1 - beta1) * grad
            v_hat = beta2 * v_hat + (1 - beta2) * (grad * grad)
            m_ub = m_hat / (1 - beta1 ** t_adam)
            v_ub = v_hat / (1 - beta2 ** t_adam)
            
            step = lr / np.sqrt(1 + 0.001 * t_adam)
            a = a - step * m_ub / (np.sqrt(v_ub) + eps)
            a = np.maximum(a, 0)
            
            if (t_adam - 1) % sym_every == 0:
                a = 0.5 * (a + a[::-1])
            
            s = a.sum()
            a = a / s if s > 1e-15 else np.ones(n) / n
            
            # Restart
            if no_improve > 500:
                a = a + 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = 0.5 * (a + a[::-1])
                s = a.sum()
                a = a / s if s > 0 else np.ones(n)/n
                m_hat[:] = 0
                v_hat[:] = 0
                t_adam = 0
                no_improve = 0
        
        return a
    
    # Create initializations
    x = np.linspace(0, 1, n)
    inits = [
        parent.copy(),                                    # 0: parent
        1.0 + 2.0*(2*x-1)**2,                            # 1: mild U
        1.0 + 5.0*(2*x-1)**2,                            # 2: strong U
        1.0/np.sqrt(1.0+3.0*(2*x-1)**2),                 # 3: inverse bell
        rng.random(n),                                    # 4: random
        np.exp(-((x-0.25)/0.2)**2) + np.exp(-((x-0.75)/0.2)**2) + 0.1,  # 5: two peaks
        1.0/np.sqrt(np.maximum(1e-10, x*(1-x))+0.01),    # 6: arcsine-like
        0.5*(1+np.cos(2*np.pi*x)),                        # 7: cosine bell
    ]
    
    ni = len(inits)
    per_init_time = min(120.0, (deadline - time.monotonic()) / ni)
    
    for i, a0 in enumerate(inits):
        rem = deadline - time.monotonic()
        if rem < 10:
            break
        t_end = time.monotonic() + min(per_init_time, rem - 5)
        optimize(a0, t_end)
    
    # Final polishing on best
    rem = deadline - time.monotonic()
    if rem > 10:
        optimize(best_a.copy(), deadline - 5)
    
    best_a = np.clip(best_a, 0, 1000)
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    if not np.isfinite(fs):
        best_a = parent.copy()
        fs = float(evaluate_sequence(best_a.tolist()))
    
    return [float(v) for v in best_a]
