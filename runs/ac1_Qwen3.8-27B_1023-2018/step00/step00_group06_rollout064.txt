import time


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    # Get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(2048, dtype=np.float64)

    n = parent.size
    conv_len = 2 * n - 1

    # FFT size: need >= 3n-2 for cross-correlation of w(len 2n-1) and u(len n)
    fft_size = 1
    while fft_size < 3 * n + 16:
        fft_size *= 2

    def exact_eval(a):
        return float(evaluate_sequence(a.tolist()))

    # Initial best
    best = parent.copy()
    best_score = exact_eval(best)
    if not np.isfinite(best_score):
        best = np.ones(n)
        best_score = exact_eval(best)

    def optimize_softmax(u0, max_iters, T_sched, alpha_sched):
        """Gradient descent on softmax parameterization to minimize smooth max of conv(u,u)."""
        u = np.maximum(u0.copy(), 1e-15)
        s = u.sum()
        if s < 1e-15:
            u = np.ones(n) / n
        else:
            u = u / s
        v = np.log(u) - np.log(u).max()  # center for stability

        best_u = u.copy()
        best_L = float('inf')
        stall = 0

        for it in range(max_iters):
            if time.monotonic() > deadline - 8.0:
                break

            frac = min(1.0, it / max(1, max_iters - 1))
            if len(T_sched) == 1:
                T = T_sched[0]
            else:
                T = T_sched[0] + (T_sched[-1] - T_sched[0]) * frac
            if len(alpha_sched) == 1:
                alpha = alpha_sched[0]
            else:
                alpha = alpha_sched[0] + (alpha_sched[-1] - alpha_sched[0]) * frac

            # Compute full convolution via FFT
            U = np.fft.rfft(u, fft_size)
            b = np.fft.irfft(U * U, fft_size)[:conv_len]

            # Log-sum-exp smooth max
            b_max = b.max()
            shifted = T * (b - b_max)
            shifted = np.clip(shifted, -400, 400)
            exp_s = np.exp(shifted)
            sum_exp = exp_s.sum()
            if sum_exp < 1e-300:
                w = np.zeros(conv_len)
                w[int(np.argmax(b))] = 1.0
                L = b_max
            else:
                w = exp_s / sum_exp
                L = b_max + np.log(sum_exp) / T

            if L < best_L - 1e-12:
                best_L = L
                best_u = u.copy()
                stall = 0
            else:
                stall += 1

            # Gradient w.r.t. u: 2 * cross_corr(w, u)
            W = np.fft.rfft(w, fft_size)
            C = np.fft.irfft(W * np.conj(U), fft_size)[:n]
            grad_u = 2.0 * C

            # Convert to gradient w.r.t. v via softmax: grad_v = u * (grad_u - dot(grad_u, u))
            dual = (grad_u * u).sum()
            grad_v = u * (grad_u - dual)

            # Update v
            v -= alpha * grad_v
            v -= v.max()
            u = np.exp(v)
            us = u.sum()
            if us < 1e-300:
                break
            u = u / us

            # Adaptive: if stalled, try a random jump
            if stall > 3000 and rng.random() < 0.01:
                v += rng.normal(0, 0.1, n)
                v -= v.max()
                u = np.exp(v)
                u /= u.sum()
                stall = 0

        return best_u

    # Build initializations
    inits = []

    # 1. Uniform
    inits.append(np.ones(n) / n)

    # 2. Parent normalized

    if sp > 1e-10:

    # 3. Raised cosine window
    x = np.arange(n, dtype=np.float64)

    if cos_w.sum() > 1e-10:

    # 4. Asymmetric (ramp down at one end)

    inits.append(ramp / ramp.sum())

    # 5. Two-block

    two[:n // 3] *= 0.3
    two[n // 3:] *= 1.0
    inits.append(two / two.sum())

    # 6. Random
    inits.append(np.abs(rng.standard_normal(n)))
    inits.append(np.abs(rng.standard_normal(n)) ** 2)

    # Phase 1: Broad optimization with moderate T
    T_broad = [5.0, 80.0]
    a_broad = [0.3, 0.02]
    iters_broad = 6000

    for idx in range(len(inits)):
        if time.monotonic() > deadline - 60.0:
            break
        u0 = inits[idx]
        result = optimize_softmax(u0, iters_broad, T_broad, a_broad)
        es = exact_eval(result)
        if np.isfinite(es) and es < best_score:
            best_score = es
            best = result.copy()

    # Phase 2: Refinement from best with higher T
    if time.monotonic() < deadline - 40.0 and best_score > 1.005:
        sb = best.sum()
        u_ref = best / sb if sb > 1e-10 else best.copy()
        T_ref = [50.0, 500.0]
        a_ref = [0.05, 0.001]
        result = optimize_softmax(u_ref, 15000, T_ref, a_ref)
        es = exact_eval(result)
        if np.isfinite(es) and es < best_score:
            best_score = es
            best = result.copy()

    # Phase 3: Very fine refinement
    if time.monotonic() < deadline - 30.0 and best_score > 1.001:
        sb = best.sum()
        u_ref = best / sb if sb > 1e-10 else best.copy()
        T_fine = [200.0, 2000.0]
        a_fine = [0.01, 0.0001]
        result = optimize_softmax(u_ref, 10000, T_fine, a_fine)
        es = exact_eval(result)
        if np.isfinite(es) and es < best_score:
            best_score = es
            best = result.copy()

    # Phase 4: Random perturbations around best
    if time.monotonic() < deadline - 10.0:
        perturb_count = 0
        while time.monotonic() < deadline - 5.0 and perturb_count < 50:
            u_try = best.copy()
            s_try = u_try.sum()
            if s_try > 1e-10:
                u_try = u_try / s_try
            # Random multiplicative block perturbation
            nblocks = rng.integers(1, 5)
            for _ in range(nblocks):
                width = max(1, rng.integers(n // 200, n // 10))
                start = rng.integers(0, max(1, n - width))
                stop = min(n, start + width)
                scale = np.exp(rng.normal(0, 0.05 + 0.1 * (1.0 / (1.0 + perturb_count * 0.05))))
                u_try[start:stop] *= scale
            u_try = np.maximum(u_try, 0.0)
            s_try2 = u_try.sum()
            if s_try2 > 1e-10:
                u_try = u_try / s_try2
            es = exact_eval(u_try)
            if np.isfinite(es) and es < best_score:
                best_score = es
                best = u_try.copy()
            perturb_count += 1

    # Final validation
    best = np.clip(best, 0.0, 1000.0)
    if best.size == 0 or best.sum() < 0.01:
        best = np.ones(n)

    final_score = exact_eval(best)
    if not np.isfinite(final_score):
        best = np.ones(n)

    return [float(x) for x in best]
