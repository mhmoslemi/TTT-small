import numpy as np
from scipy.fft import fft, ifft
import time

def _fft_convolve(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    fa = fft(a, size)
    c = ifft(fa * fa).real[:2*n-1]
    return np.maximum(c, 0)  # clip small negatives from FFT error

def _compute_score_fft(a):
    """Compute the score using FFT (fast, for internal use)."""
    n = len(a)
    C = _fft_convolve(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2 * n * C.max() / (s * s)

def _gradient_descent(a, n, rng, max_iters, time_limit):
    """Run projected gradient descent to minimize max autocorrelation."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s  # normalize to sum=1
    
    alpha = 0.5 / n  # initial step size (scaled)
    best_local = a.copy()
    best_local_score = _compute_score_fft(a)
    
    K = max(3, n // 100)  # number of top lags to use
    
    for it in range(max_iters):
        if time.monotonic() > time_limit:
            break
        
        C = _fft_convolve(a)
        current_max = C.max()
        
        # Get top-K lags
        if K < len(C):
            top_ks = np.argpartition(C, -K)[-K:]
            top_ks = top_ks[C[top_ks] > 0.9 * current_max]  # only near-max
        else:
            top_ks = np.arange(len(C))
        
        if len(top_ks) == 0:
            break
            
        # Average gradient over top-K lags
        grad = np.zeros(n)
        for k in top_ks:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                grad[idx] += a[k - idx]
        grad /= len(top_ks)
        grad *= 2  # factor of 2 from the derivative
        
        # Project (subtract mean for normalization constraint)
        grad -= grad.mean()

        for _ in range(10):
            a_new = a - alpha * grad
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            
            C_new = _fft_convolve(a_new)
            if C_new.max() < current_max:
                a = a_new
                current_max = C_new.max()
                alpha = min(alpha * 1.1, 5.0/n)
                break
            else:
                alpha *= 0.5
        
        # Track best
        score = _compute_score_fft(a)
        if score < best_local_score:
            best_local_score = score
            best_local = a.copy()
    
    return best_local

import numpy as np
from scipy.fft import fft, ifft
import time


def _fft_convolve(a):
    """Compute full autocorrelation (convolution of a with itself) via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    fa = fft(a, size)
    c = ifft(fa * fa).real[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_fast(a):
    """Fast score computation (assumes a is already non-negative)."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    C = _fft_convolve(a)
    return 2.0 * n * C.max() / (s * s)

def _make_step_init(n, splits, heights, rng):
    """Create a step-function initialization."""
    a = np.zeros(n)
    boundaries = [0] + [int(n * s) for s in splits] + [n]
    for i in range(len(heights)):
        a[boundaries[i]:boundaries[i + 1]] = heights[i]
    # Random small perturbation
    a += rng.normal(0, 0.005, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    return a
