import numpy as np
from scipy import sparse
from scipy.optimize import linprog
import time

def build_conv_matrix(f):
    """Build the sparse convolution matrix A where (A@g)[p] = (f*g)[p]."""
    n = len(f)
    # A has shape (2n-1, n)
    # A[p, j] = f[p-j] if 0 <= p-j < n, else 0
    rows = []
    cols = []
    vals = []
    for p in range(2*n - 1):
        j_start = max(0, p - n + 1)
        j_end = min(n - 1, p)
        for j in range(j_start, j_end + 1):
            rows.append(p)
            cols.append(j)
            vals.append(f[p - j])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'time_limit': 30})
    if res.success:
        return res.x
    return None

    sum_f = f.sum()

        # Compute current M

        M = conv.max()

        if g is None or not np.all(np.isfinite(g)):
