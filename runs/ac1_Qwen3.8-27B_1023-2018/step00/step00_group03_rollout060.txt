```python
import numpy as np
import time

def proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if v.sum() == 0:
        return np.ones(n) / n * s
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0
    
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    n = parent.size
    s = parent.sum()
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a = parent / s
    
    # Track best
    best = a.copy()
    best_score = 2 * n * np.max(np.convolve(a, a))
    
    # Projected gradient descent
    alpha = 0.1
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        k_star = np.argmax(c)
        max_b = c[k_star]
        
        # Gradient
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            g[i] = 2 * a[k_star - i]
        
        # Step
        a_new = a - alpha * g
        a_new = proj_simplex(a_new, 1.0)
        
        # Check
        c_new = np.convolve(a_new, a_new)
        max_b_new = np.max(c_new)
        
        if max_b_new < max_b:
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            alpha *= 0.9
            if alpha < 1e-8:
                alpha = 0.1  # reset
        
        # Update best
        score = 2 * n * max_b_new if max_b_new < max_b else 2 * n * max_b
        if score < best_score:
            best = a.copy()
            best_score = score
        
        # Periodic exact evaluation
        if no_improve % 100 == 0:
            exact = evaluate_sequence(a.tolist())
            if exact < evaluate_sequence(best.tolist()):
                best = a.copy()
                best_score = exact
    
    return best.tolist()
```

```python
def proj_simplex(v, s=1.0):

    n = len(v)
    if n == 0:

    theta = (cssv[rho] - s) / (rho + 1.0)
```

```python
import numpy as np
```

```python
def _proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s} in O(n log n)."""
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not cond.any():
        return np.full(n, s / n, dtype=np.float64)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```
