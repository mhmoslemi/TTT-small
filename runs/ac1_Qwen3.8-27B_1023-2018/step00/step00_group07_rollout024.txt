def _autocorr_fft(a):
    """Fast autocorrelation via FFT. O(n log n)."""
    n = len(a)

    while size < 2*n:
        size <<= 1
    f = np.fft.rfft(a, size)
    ac = np.fft.irfft(f * f.conjugate(), size)[:2*n-1]
    return ac
