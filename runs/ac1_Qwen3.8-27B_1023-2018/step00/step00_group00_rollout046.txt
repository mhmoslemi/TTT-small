def propose_candidate(seed=42, budget_s=1300, **kwargs):
    import numpy as np
    import time
    
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 5.0
    
    parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
    n = parent.size
    parent = np.clip(parent, 0.0, 1000.0)
    s = parent.sum()
    a = parent / s if s > 0 else np.ones(n) / n
    
    # FFT size for linear convolution
    fft_size = 1
    while fft_size < 2*n - 1:
        fft_size *= 2
    
    def convolve_sq(a):
        fa = np.fft.rfft(a, fft_size)
        return np.fft.irfft(fa * fa, fft_size)[:2*n-1]
    
    def get_score(a):
        s = a.sum()
        if s < 0.01:
            return np.inf
        b = convolve_sq(a)
        return 2.0 * n * b.max() / (s*s)
    
    best = a.copy()
    best_score = get_score(best)
    
    # Gradient descent
    lr = 0.005
    for iteration in range(100000):
        if time.monotonic() > deadline:
            break
        
        b = convolve_sq(a)  # length 2n-1
        M = b.max()
        
        # Softmax weight on lags for smooth gradient
        # Focus on top lags
        top_lags = np.argsort(b)[-20:]  # top 20 lags
        # Weight by how close to max
        weights = np.exp((b[top_lags] - M) / 0.01)
        weights /= weights.sum()
        
        # Gradient of weighted sum of b_k w.r.t. a
        grad = np.zeros(n)
        for w, k in zip(weights, top_lags):
            k = int(k)
            # b_k = sum_j a_j * a_{k-j}, ∂/∂a_i = 2*a_{k-i} for valid k-i
            lo = max(0, k - (n-1))
            hi = min(n-1, k)
            # grad[lo:hi+1] += 2 * w * a[k - (lo:hi+1)]
            indices = np.arange(lo, hi+1)
            grad[indices] += 2.0 * w * a[k - indices]
        
        # Add small regularization to encourage flatness
        # Subtract mean to be variance-free
        grad -= grad.mean()
        
        # Gradient descent (minimize weighted peak)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        new_s = a_new.sum()
        if new_s < 1e-10:
            a_new[:] = 0
            new_s = 1
        a_new /= new_s  # renormalize to sum=1
        
        new_score = get_score(a_new)
        if new_score < best_score:
            best = a_new.copy()
            best_score = new_score
        
        # Adaptive learning rate
        if new_score < get_score(a):
            lr = min(lr * 1.05, 0.05)
        else:
            lr = max(lr * 0.9, 0.0001)
        
        a = a_new
    
    # Optionally scale back and evaluate
    # The evaluator clips to [0, 1000], so we need to make sure values are reasonable
    # Scale so max is well below 1000
    a_out = best.copy()
    if a_out.max() > 999:
        a_out = a_out * 999 / a_out.max()
    
    # Final evaluation
    final_score = get_score(a_out)
    if final_score >= best_score:
        a_out = best
    
    return [float(x) for x in a_out]
