```python
    # Actually, let me just use the cumulative sum approach:
    # For row k, the entries are a[k-j] for j in [max(0,k-n+1), min(k,n-1)]
    # Let's build it as: for each j, the column j has a[k-j] for k in [j, j+n-1]
    
    # Column j: positions k = j, j+1, ..., j+n-1 have values a[n-1], a[n-2], ..., a[0]
    # Wait: C[k,j] = a[k-j], so for fixed j, as k goes from j to j+n-1:
    # C[j,j] = a[0], C[j+1,j] = a[1], ..., C[j+n-1,j] = a[n-1]
    
    # So column j is: zeros for k < j, then a[0], a[1], ..., a[n-1] for k = j, ..., j+n-1, then zeros.
    
    # This means C can be built as a sum of shifted versions of [a[0], a[1], ..., a[n-1]]
    # Column j has a shifted by j positions.
    
    # In sparse format:
    rows = []
    cols = []
    data = []
    # For column j (j = 0, ..., n-1):
    # Nonzero rows: k = j, j+1, ..., j+n-1 (but capped at 2n-2)
    # Values: a[0], a[1], ..., a[n-1] (but only up to row 2n-2)
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2*n - 2)
        n_entries = k_end - k_start + 1
        rows.extend(range(k_start, k_end + 1))
        cols.extend([j] * n_entries)
        data.extend(a[:n_entries])
    
    C = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return C.tocsr()
```

This is still O(n^2) Python-level. For n=5582, the inner loop runs 31M times total. In Python, that's about 31M * ~0.1μs = 3.1 seconds (optimistic) to 31M * 1μs = 31 seconds (realistic).

```python
def build_C_fast(a):
    n = len(a)
    # For column j, nonzero rows are j to min(j+n-1, 2n-2)
    # Number of entries in column j: min(n, 2n-1-j)
    
    # Total entries: sum_{j=0}^{n-1} min(n, 2n-1-j) = sum_{j=0}^{n-1} (2n-1-j) for j <= n-1
    # = sum_{j=0}^{n-1} (2n-1-j) = n(2n-1) - n(n-1)/2 = 2n^2-n - n^2/2 + n/2 = 3n^2/2 - n/2
    # Wait, that's not right. For j = 0: rows 0 to n-1, so n entries.
    # For j = n-1: rows n-1 to 2n-2, so n entries.
    # So every column has exactly n entries!
    # Total entries: n * n = n^2. ✓
    
    # Vectorized:
    # Column j has entries a[0], a[1], ..., a[n-1] at rows j, j+1, ..., j+n-1
    # But row j+n-1 can be at most 2n-2 (which is fine since j <= n-1, so j+n-1 <= 2n-2)
    
    # So every column j (0 <= j < n) has exactly n entries: a[0] at row j, a[1] at row j+1, ..., a[n-1] at row j+n-1
    
    # This is beautiful! C is a "block" matrix where each column is a shifted copy of a.
    
    # In COO:
    # col j: row = j + arange(n), value = a
    j = np.arange(n)
    offset = np.arange(n)
    # For all columns simultaneously:
    # rows = j[:, None] + offset[None, :]  (n x n matrix)
    # cols = j[:, None] * np.ones((1, n))  (n x n matrix)  
    # data_vals = a[None, :] * np.ones((n, 1))  (n x n matrix)
    
    rows_2d = j[:, None] + offset[None, :]  # (n, n)
    cols_2d = j[:, None] + np.zeros((1, n), dtype=int)  # (n, n)
    data_2d = a[None, :]  # (1, n) broadcast to (n, n)
    
    rows = rows_2d.ravel()
    cols = cols_2d.ravel()
    data = np.tile(a, n)  # a repeated n times
    
    C = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
    return C
```

```python
def build_C_csr(a):
    n = len(a)
    num_rows = 2*n - 1
    
    # indptr
    indptr = np.zeros(num_rows + 1, dtype=np.int64)
    for k in range(num_rows):
        nnz_in_row = min(k+1, n, 2*n-1-k)
        indptr[k+1] = indptr[k] + nnz_in_row
    
    total_nnz = indptr[-1]  # = n^2
    indices = np.empty(total_nnz, dtype=np.int64)
    data = np.empty(total_nnz, dtype=np.float64)
    
    # Fill in row by row
    pos = 0
    for k in range(num_rows):
        j_start = max(0, k-n+1)
        j_end = min(k, n-1)
        count = j_end - j_start + 1
        indices[pos:pos+count] = np.arange(j_start, j_end+1)
        data[pos:pos+count] = a[k-j_end : k-j_start+1][::-1]  # a[k-j_start], ..., a[k-j_end]... 
        
        # Wait, C[k,j] = a[k-j]. So for j = j_start, ..., j_end:
        # C[k, j_start] = a[k - j_start], C[j_start+1] = a[k-j_start-1], ..., C[j_end] = a[k-j_end]
        # So the values are a[k-j_start], a[k-j_start-1], ..., a[k-j_end]
        # That's a[k-j_start : k-j_end-1 : -1] or equivalently a[k-j_end : k-j_start+1][::-1]
        
        pos += count
    
    return sp.csr_matrix((data, indices, indptr), shape=(num_rows, n))
```
