def _project_simplex(z):
    """Project z onto {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 1:
        return np.array([max(0, z[0])])
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = (u - (cssv - 1.0) / np.arange(1, n + 1) > 0).argmax()
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

    # Find the main contributors to b[k_star]

    pairs = [(i, idx[i]) for i in np.where(valid)[0] if i <= idx[i]]
