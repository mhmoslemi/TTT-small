from scipy.fft import fft, ifft
import time

def full_convolution(f):
    """Compute full linear convolution f*f using FFT."""
    n = len(f)
    N = 1 << (2*n - 1 - 1).bit_length()
    F = fft(f, N)
    conv = np.real(ifft(F * F))[:2*n-1]
    return conv

def soft_max_gradient(f, conv, beta):
    """Compute gradient of soft-max of conv w.r.t. f."""
    n = len(f)
    # Soft-max weights
    logits = beta * conv
    logits -= np.max(logits)
    weights = np.exp(logits)
    weights /= np.sum(weights)
