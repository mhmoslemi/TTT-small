def _binomial_init(n):
    """Generate a[k] = C(2k,k)/4^k for k=0..n-1 and normalize to sum=1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k) / 2.0
    s = a.sum()
    a /= s
    return a

def _score_np(a):
    """Compute 2*n*max(conv)/(sum)^2 using FFT."""
    n = a.size
    S = a.sum()
    if S < 1e-12:
        return float('inf'), 0

    c = irfft(fa * fa, N)[:2 * n - 1]
    mc = c.max()
    return 2.0 * n * mc / (S * S), mc

    s_hb = a_hb.sum()

        cands.append(a_hb / s_hb)

    for a in cands:
        if time.monotonic() > deadline:
            break
        update_best(a.tolist())

    a_work = best_list is not None and a_bin[0] > 0.4 * a_work[0] if False else a_bin.copy()
    # Reset working to the best-scored candidate
    # Re-derive: pick the candidate with best evaluate_sequence score
    a_work = None
    for a in cands:
        sc, _ = _score_np(a)
        if np.isfinite(sc) and (a_work is None or sc < _score_np(cands[0])[0]):
            pass
    # Just use binomial as primary work array
    a_work = a_bin.copy()

    # ---- Phase 2: Projected subgradient on full convolution ----
    eta = 0.3
    n_iters = 0
    stuck = 0
    prev_base = _score_np(a_work)[1]

    while time.monotonic() < deadline:
        sc, mc = _score_np(a_work)
        if not np.isfinite(sc):
            break
        update_best(a_work.tolist())
        if sc >= best_score - 1e-12:
            stuck += 1
        else:
            stuck = 0
        if stuck > 5000:
            break

        # Find argmax lag
        Nf = next_fast_len(2 * n - 1)
        fa = rfft(a_work, Nf)
        conv = irfft(fa * fa, Nf)[:2 * n - 1]
        kstar = int(np.argmax(conv))

        # Compute subgradient and step
        g = np.zeros(n)
        lo = max(0, kstar - (n - 1))
        hi = min(kstar, n - 1)
        # Only indices where both i and kstar-i are in [0, n-1]
        i_arr = np.arange(lo, hi + 1)
        j_arr = kstar - i_arr
        # g[i] gets contribution from a[j], g[j] gets contribution from a[i]
        g[i_arr] += a_work[j_arr]
        mask_neq = i_arr != j_arr
        g[j_arr[mask_neq]] += a_work[i_arr[mask_neq]]
        # Fix diagonal: for i == j, should be 2*a[i] instead of a[i]
        diag = i_arr[i_arr == j_arr]
        if len(diag) > 0:
            g[diag] = 2.0 * a_work[diag]

        # Adaptive learning rate
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        base = 2.0 * n * mc / (a_work.sum() ** 2)
        step_size = eta * base / gnorm
        a_work = a_work - step_size * g
        a_work = np.maximum(a_work, 0.0)
        s = a_work.sum()
        if s < 1e-15:
            a_work = a_bin.copy()
        else:
            a_work /= s
        eta *= 0.9998
        n_iters += 1

    # ---- Phase 3: Sort-based peak fix ----
    if time.monotonic() < deadline - 3.0:
        sc, mc = _score_np(a_work)
        Nf = next_fast_len(2 * n - 1)
        fa = rfft(a_work, Nf)
        conv = irfft(fa * fa, Nf)[:2 * n - 1]
        flat = conv.ravel()
        for _ in range(300):
            if time.monotonic() > deadline - 3.0:
                break
            kmax = int(np.argmax(flat))
            lo = max(0, kmax - (n - 1))
            hi = min(kmax, n - 1)
            i_idx = np.arange(lo, hi + 1)
            j_idx = kmax - i_idx
            vals = a_work[i_idx] * a_work[j_idx]
            # If kmax is even and i==j exists, fix that term
            even_mask = (i_idx == j_idx)
            if even_mask.any():
                vals[even_mask] = a_work[i_idx[even_mask]] ** 2
            if vals.size < 2:
                break
            mi = np.argmin(vals)
            ma = np.argmax(vals)
            if ma == mi:
                break
            if vals[ma] <= vals[mi]:
                break
            av = a_work[i_idx[mi]]
            bv = a_work[j_idx[mi]]
            ac = a_work[i_idx[ma]]
            bc = a_work[j_idx[ma]]
            # Swap to increase min and decrease max (symmetrize)
            a_work[i_idx[mi]] = a_work[i_idx[ma]]
            a_work[j_idx[mi]] = a_work[j_idx[ma]]
            a_work[i_idx[ma]] = av
            a_work[j_idx[ma]] = bv
            a_work = np.maximum(a_work, 0.0)
            s = a_work.sum()
            if s < 1e-15:
                break
            a_work /= s
            # Recompute only the peaks
            fa2 = rfft(a_work, Nf)
            flat = irfft(fa2 * fa2, Nf)[:2 * n - 1].ravel()
            update_best(a_work.tolist())

    # ---- Phase 4: Random local search ----
    if time.monotonic() < deadline - 2.0:
        for it in range(500000):
            if time.monotonic() > deadline - 2.0:
                break
            p = rng.integers(n)
            a_trial = a_work.copy()
            a_trial[p] *= float(np.exp(rng.normal(0, 0.05)))
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial /= s
            sc, _ = _score_np(a_trial)
            if np.isfinite(sc) and sc < best_score - 1e-10:
                a_work = a_trial
                update_best(a_trial.tolist())

    # ---- Final validation ----
    if best_list is not None:
        try:
            fs = evaluate_sequence(best_list)
            if np.isfinite(fs):
                return best_list
        except Exception:
            pass
    # Fallback: binomial
    fb = a_bin.tolist()
    try:
        if np.isfinite(evaluate_sequence(fb)):
            return fb
    except Exception:
        pass
    return [1.0 / n] * n
