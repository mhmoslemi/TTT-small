```python
def compute_pressure(w, x):
    """g[i] = sum_j x[j] * w[i+j], for w of length Mw and x of length n."""
    n = len(x)
    Mw = len(w)
    N = 1
    while N < n + Mw - 1:
        N *= 2
    fx = np.fft.rfft(x, N)
    fw = np.fft.rfft(w, N)
    g = np.fft.irfft(fw * np.conj(fx), N)[:n]
    return g
```
