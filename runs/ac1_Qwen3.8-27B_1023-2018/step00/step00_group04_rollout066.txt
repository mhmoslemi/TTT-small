    if S0 < 0.01:

    else:

    best_score = evaluate_sequence(best.tolist())
    if not np.isfinite(best_score):

        best_score = evaluate_sequence(best.tolist())

    def run_optim(f_init, end_time, lam, lr, mom):
        """Projected gradient descent with momentum and softmax-smoothed max."""
        f = f_init.copy().astype(np.float64)
        n = len(f)
        sf = f.sum()
        if sf < 1e-12:
            f = np.ones(n) / n
        else:
            f = f / sf

        vel = np.zeros(n)
        best_local = f.copy()
        best_local_sc = evaluate_sequence(f.tolist())
        it = 0
        no_improve = 0

        while time.monotonic() < end_time and (end_time - time.monotonic()) > 8.0:
            it += 1

            # Full convolution
            b = np.convolve(f, f)  # length 2*n-1

            # Softmax weights (stable)
            b_max = b.max()
            shifted = b - b_max
            shifted = np.maximum(shifted, -50.0 / lam)
            eb = np.exp(lam * shifted)
            ws = eb.sum()
            if ws < 1e-300:
                break
            w = eb / ws

            # Gradient: grad[j] = sum_{i=0}^{n-1} f[i] * w[j+i]
            # Computed as: fv = conv(f, w_reversed); grad[j] = fv[2n-2-j]
            w_rev = w[::-1]
            fv = np.convolve(f, w_rev)  # length 3n-2
            idx = 2 * n - 2 - np.arange(n)

            gm = grad.mean()
            grad -= gm

            # Momentum
            vel = mom * vel + grad
            vn = np.sqrt(np.dot(vel, vel))
            if vn < 1e-18:
                break

            # Normalized step
            step_magnitude = lr * f.mean() * n / vn
            f = f - step_magnitude * vel

            np.maximum(f, 0.0, out=f)

            if sn < 1e-12:
                f = np.ones(n) / n

                    best_local = f.copy()
