```python
import time
import cvxpy as cp

def solve_lp(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b) <= M*ones"""
    n = len(a)
    b = cp.Variable(n)
    # conv(a,b) has 2n-1 entries
    # conv(a,b)[k] = sum_j a[j]*b[k-j] for valid j
    # Build constraint: for each k, sum_j A[k,j]*b[j] <= M
    # A[k,j] = a[j] if max(0,k-n+1) <= j <= min(k,n-1) else 0
    
    constraints = [b >= 0]
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        # conv(a,b)[k] = sum_{j=lo}^{hi} a[j] * b[k-j]
        # Let i = k-j, so j = k-i, i ranges from k-hi to k-lo
        # b index: j, so we need b[j] for j in [lo, hi]
        # coefficient of b[j] is a[k-j]
        row = 0
        for j in range(lo, hi+1):
            row += a[k-j] * b[j]
        constraints.append(row <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.CBC, max_time=30)
    if b.value is not None and np.isfinite(b.value).all():
        return np.maximum(b.value, 0)
    return None
```
