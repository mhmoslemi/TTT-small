import numpy as np
import time
from scipy.optimize import linprog


def _unit_sum(u):
    u = np.asarray(u, dtype=np.float64)
    s = float(u.sum())
    if not np.isfinite(s) or s <= 0.0:
        u = np.ones_like(u)
        s = float(u.size)
    return u / s


def _score_seq(seq):
    try:
        v = evaluate_sequence(seq)
        return v if np.isfinite(v) else np.inf
    except Exception:
        return np.inf


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s)) - 5.0
    rng = np.random.default_rng(seed)

    parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    if parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(min(512, max(2, int(np.sqrt(2 * 400)))))
    parent = np.clip(parent, 0.0, 1000.0)
    n = int(parent.size)

    arr = np.arange(n, dtype=np.float64)

    # ---- initial candidates (homogeneous, so work at unit sum) ----
    cands = [_unit_sum(parent)]
    for freq in (2, 3, 5, 7, 11):
        w = parent * (1.0 + 0.55 * np.sin(2.0 * np.pi * freq * arr / n))
        w = np.clip(w, 1e-4, None)
        cands.append(_unit_sum(w))
    cands.append(_unit_sum(parent * (1.0 + 0.5 * rng.standard_normal(n))))
    cands.append(_unit_sum(rng.uniform(0.3, 1.0, size=n)))

    best = None
    best_score = np.inf
    for c in cands:
        s = _score_seq(c.tolist())
        if s < best_score:
            best_score = s
            best = c.copy()
    f = best.copy()

    # convolution-matrix buffer (A[j,k] = f[j-k] on the valid band)
    Abuf = np.zeros((2 * n - 1, n), dtype=np.float64)
    ones_ub = np.ones(2 * n - 1, dtype=np.float64)
    neg_one = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    tstep = 0.6
    kick_after = 0
    iter_kick = 0

    def build_A():
        Abuf[:] = 0.0
        for k in range(n):
            Abuf[k:k + n, k] = f

    while True:
        remaining = deadline - time.monotonic()
        if remaining < 8.0:
            break

        # current peak
        ff = np.convolve(f, f)
        M = float(ff.max())
        if not np.isfinite(M) or M <= 0.0:
            break

        # assemble LP constraints  (f * b)_j <= M ,  b >= 0
        build_A()
        b_ub = M * ones_ub
        tlim = max(0.5, remaining - 6.0)
        try:
            res = linprog(neg_one, A_ub=Abuf, b_ub=b_ub,
                          bounds=bounds, method="highs",

        except Exception:
            break
        if res is None or getattr(res, "x", None) is None or not np.all(np.isfinite(res.x)):
            # solver trouble: break symmetry and keep searching
            f = np.clip(f * (1.0 + 0.12 * rng.standard_normal(n)), 1e-6, None)
            f = _unit_sum(f)

        b = np.clip(res.x, 0.0, None)
        sb = float(b.sum())
        if sb <= 1e-12:
            f = np.clip(f * (1.0 + 0.12 * rng.standard_normal(n)), 1e-6, None)
