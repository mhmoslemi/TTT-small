    nc = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(nc):
        j0 = max(0, k - n + 1)
        j1 = min(k, n - 1)
        for j in range(j0, j1 + 1):
            rows.append(k)
            cols.append(j)

def _lp_blend(f, n, deadline, rng):
    """Run LP-based refinement on f. Returns best (f, score) found."""
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    for it in range(200):
        if time.monotonic() > deadline:
            break
        
        s = f.sum()
        if s < 1e-10:
            break
        f_norm = f / s
        
        ff = np.convolve(f_norm, f_norm)
        M = ff.max()
        if M < 1e-15:
            break
        
        A = _build_conv_matrix(f_norm, n)
        c = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        bounds = [(0, None)] * n

            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'max_iter': 50000, 'dual_feasibility_tolerances': 1e-8})
        except Exception:

            new_f = np.clip(new_f, 0.0, 1000.0)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _conv_matrix(f, n):
    """(2n-1) x n sparse matrix: row k has f[k-j] at col j for valid j."""
    nc = 2 * n - 1
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for k in range(nc):
        j0 = max(0, k - n + 1)
        j1 = min(k, n - 1)
        cnt = j1 - j0 + 1
        rows[idx:idx + cnt] = k
        cols[idx:idx + cnt] = np.arange(j0, j1 + 1)
        vals[idx:idx + cnt] = f[k - np.arange(j0, j1 + 1)]
        idx += cnt
    return csr_matrix((vals, (rows, cols)), shape=(nc, n))


def _refine(f0, n, deadline, rng, max_iters=100):
    """LP-based refinement. Returns (best_f, best_score)."""
    best = f0.copy()
    bs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(bs):
        return best, bs
    
    f = f0.copy()
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        s = f.sum()
        if s < 1e-12:
            break
        fn = f / s
        ff = np.convolve(fn, fn)
        M = float(ff.max())
        if M < 1e-15:
            break
        
        A = _conv_matrix(fn, n)
        c = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        bounds = [(0.0, None)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'max_iter': 100000})
        except Exception:
            break
        if not r.success or r.x is None:
            break
        
        g = np.maximum(r.x, 0.0)
        gs = g.sum()
        if gs < 1e-12:
            break
        
        improved = False
        for a in (0.05, 0.1, 0.2, 0.3, 0.5):
            nf = (1.0 - a) * f + a * g
            nf = np.clip(nf, 0.0, 1000.0)
            sc = float(evaluate_sequence(nf.tolist()))
            if np.isfinite(sc) and sc < bs:
                best = nf
                bs = sc
                f = nf.copy()
                improved = True
                break
        if not improved and it > 5:
            # Perturb to escape
            f = best + rng.normal(0, 0.02 * max(best.mean(), 1e-6), n)
            f = np.clip(f, 0, 1000)
    
    return best, bs
