def _fft_conv(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def _compute_conv_and_grad(a, s):
    """Compute convolution and gradient of smooth-max objective. Returns (conv, grad, smooth_max_val)."""
    n = len(a)
    size = 1
    while size < 4 * n:
        size *= 2
    
    conv = _fft_conv(a, a)
    S = a.sum()
    g = conv / (S * S)

    sg = s * g
    sg -= sg.max()
    w = np.exp(sg)
    w_sum = w.sum()
    w /= w_sum
    smooth_max = g.max() + np.log(w_sum) / s  # approx
    
    # Gradient via FFT
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    
    # term1[j] = sum_i a[i]*w[i+j] (cross-correlation)
    corr1 = np.fft.irfft(np.conj(A) * W, size)[:n]
    # term2[j] = sum_i a[i]*w[j-i] (convolution)
    conv_aw = np.fft.irfft(A * W, size)[:n]
    
    grad_conv = corr1 + conv_aw  # sum_k w_k * dc_k/da_j
    
    # Full gradient: d/d(a_j) [sum_k w_k * c_k / S^2]
    dot_wc = np.dot(w, conv)
    const_term = 2.0 * dot_wc / (S ** 3)
    grad = grad_conv / (S * S) - const_term
    
    return conv, grad, smooth_max, w
