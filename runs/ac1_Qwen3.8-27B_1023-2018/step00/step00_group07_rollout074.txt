def _grad_logsumexp(a, C, beta):
    n = a.size
    mc = C.max()
    w = np.exp(beta*(C-mc))
    wsum = w.sum()
    if wsum ==0: return np.zeros(n)
    p = w / wsum
    # v_i = sum_j a_j p_{i+j}
    # conv a with p_rev
    L = 1 << (3*n - 2 - 1).bit_length()  # >=3n-2
    fa = np.fft.rfft(a, L)
    fp = np.fft.rfft(p[::-1], L)
    conv = np.fft.irfft(fa*fp, L)
    # v_i = conv[2n-2 - i] for i=0..n-1
    v = conv[n-1:2*n-1][::-1]
    return 2.0 * v
```

def _perturb(a, rng, scale=None):
    n=a.size
    if scale is None: scale = 0.2
    b = a.copy()
    kind = int(rng.integers(0, 6))
    if kind==0:
       width = int(rng.integers(max(2,n//500), max(3,n//5)))
       start = int(rng.integers(0,n-width))
       b[start:start+width] *= float(np.exp(rng.normal(0, scale)))
    elif kind==1:
       b *= np.exp(rng.normal(0, scale/2, n))
    elif kind==2:
       # random block mass transfer
       w1 = int(rng.integers(max(5,n//100), n//4))
       w2 = int(rng.integers(max(5,n//100), n//4))
       s1 = int(rng.integers(0,n-w1)); s2 = int(rng.integers(0,n-w2))
       total = b[s1:s1+w1].sum()
       frac = float(rng.uniform(0.02,0.2))
       delta = frac*total
       b[s1:s1+w1] -= delta/(w1)
       # distribute to block weighted by current to avoid huge spikes? uniform or proportional
       target = b[s2:s2+w2].copy()
       if target.sum() > 1e-18:
           b[s2:s2+w2] += delta * target/target.sum()
       else:
           b[s2:s2+w2] += delta/w2
    elif kind==3:
       # additive smooth bump at random location
       width = int(rng.integers(n//20, n//3))
       start = int(rng.integers(0,n-width))
       x = np.linspace(0,1,width)
       shape = int(rng.integers(0,3))
       if shape==0: bump=np.ones(width)
       elif shape==1: bump=np.cos(np.pi*x/2)**2
       else: bump=np.sin(np.pi*x/2)**2
       amp = float(rng.uniform(0,0.1)) * (b.sum()/width)
       b[start:start+width] += amp*bump
    elif kind==4:
       f = int(rng.integers(1,4))
       phase = float(rng.uniform(0,2*np.pi))
       mod = 1.0 + scale*float(rng.uniform(-1,1))*np.cos(2*np.pi*f*np.linspace(0,1,n,endpoint=False)+phase)
       b *= mod
    else:
       # tail bump specifically
       width = int(rng.integers(n//4, n))
       x = np.linspace(0,1,width)
       shape = int(rng.integers(0,2))
       bump = np.cos(np.pi*x/2)**2 if shape==0 else np.ones(width)
       amp = float(rng.uniform(0,0.15)) * b.sum()/width
       b[n-width:n] += amp*bump
    b = np.clip(b,0,None)
    s=b.sum()
    if s<1e-18: return a.copy()

import numpy as np
import time
import math


def _pow2_at_least(m):
    if m <= 1:
        return 1
    return 1 << max(0, (m - 1).bit_length())


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    L = _pow2_at_least(m)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)[:m]
    # Guard harmless roundoff; the true convolution of a nonnegative sequence is nonnegative.
    np.clip(c, 0.0, None, out=c)
    return c


def _fft_conv_max(a):
    c = _fft_conv(a)
    if c.size == 0:
        return 0.0
    return float(np.max(c))


def _normalize_valid(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return None
    a = a / s
    # Keep leave room for evaluator caps; a sum-one nonnegative sequence is already safe.
    return a


def _fast_score_norm(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    maxc = _fft_conv_max(a)
    return float(2.0 * n * maxc)


def _grad_topk(a, c, rng, k=32, rel=0.04):
    n = a.size
    maxc = float(np.max(c)) if c.size else 0.0
    if maxc <= 0.0:
        return np.zeros(n, dtype=np.float64)
    active = np.flatnonzero(c >= maxc - rel * maxc - 1e-15)
    if active.size == 0:
        active = np.array([int(np.argmax(c))], dtype=np.int64)
    if active.size > k:
        idx = rng.integers(0, active.size, size=k)
        lags = active[idx]
    else:
        lags = active

    g = np.zeros(n, dtype=np.float64)
    for L in lags:
        L = int(L)
        lo = max(0, L - (n - 1))
        hi = min(n - 1, L)
        if lo > hi:
            continue
        start_idx = L - lo
        end_idx = L - hi
        if start_idx == end_idx:
            g[lo] += a[start_idx]
        else:
            if end_idx == 0:
                rev = a[start_idx::-1]
            else:
                rev = a[start_idx:end_idx - 1:-1]
            g[lo:hi + 1] += rev
    g *= (2.0 / float(lags.size))
    return g


def _grad_full_logsumexp(a, c, beta):
    n = a.size
    maxc = float(np.max(c)) if c.size else 0.0
    if maxc <= 0.0 or beta <= 0.0:
        return np.zeros(n, dtype=np.float64)
    w = np.exp(beta * (c - maxc))
    sw = float(w.sum())
    if not np.isfinite(sw) or sw <= 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        sw = 1.0
    p = w / sw

    # v_i = sum_j a_j p_{i+j}.  Compute v as a slice of conv(a, reverse(p)).
    need = 3 * n - 2
    L = _pow2_at_least(need)
    fa = np.fft.rfft(a, L)
    fp = np.fft.rfft(p[::-1], L)
    conv = np.fft.irfft(fa * fp, L)
    v = conv[n - 1:2 * n - 1][::-1]
    return 2.0 * v


def _perturb(a, rng, scale=0.20):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    b = a.copy()
    if n <= 2:
        b = np.clip(b + scale * rng.normal(0.0, 1.0, n), 0.0, None)
    else:
        kind = int(rng.integers(0, 7))
        if kind == 0:
            width = int(rng.integers(max(2, n // 700), max(3, n // 8)))
            start = int(rng.integers(0, n - width))
            factor = float(np.exp(rng.normal(0.0, max(1e-3, scale))))
            b[start:start + width] *= factor
        elif kind == 1:
            b *= np.exp(rng.normal(0.0, max(1e-3, 0.5 * scale), n))
        elif kind == 2:
            w1 = int(rng.integers(max(5, n // 250), max(6, n // 4)))
            w2 = int(rng.integers(max(5, n // 250), max(6, n // 4)))
            s1 = int(rng.integers(0, n - w1))
            s2 = int(rng.integers(0, n - w2))
            block = b[s1:s1 + w1]
            total = float(block.sum())
            if total > 1e-18:
                frac = float(rng.uniform(0.02, max(0.03, 0.5 * scale + 0.05)))
                delta = frac * total
                b[s1:s1 + w1] -= delta / w1
                target = b[s2:s2 + w2].copy()
                ts = float(target.sum())
                if ts > 1e-18:
                    b[s2:s2 + w2] += delta * (target / ts)
                else:
                    b[s2:s2 + w2] += delta / w2
        elif kind == 3:
            width = int(rng.integers(max(4, n // 40), max(5, n // 2)))
            start = int(rng.integers(0, n - width))
            x = np.linspace(0.0, 1.0, width)
            typ = int(rng.integers(0, 3))
            if typ == 0:
                bump = np.ones(width)
            elif typ == 1:
                bump = np.cos(0.5 * np.pi * x) ** 2
            else:
                bump = np.sin(0.5 * np.pi * x) ** 2
            amp = float(rng.uniform(0.0, max(0.02, scale))) * (1.0 / width)
            b[start:start + width] += amp * bump
        elif kind == 4:
            f = int(rng.integers(1, 5))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            amp = float(rng.uniform(-1.0, 1.0)) * max(1e-3, scale)
            mod = 1.0 + amp * np.cos(2.0 * np.pi * f * np.arange(n) / float(n) + phase)
            b *= mod
        elif kind == 5:
            # Explicitly try moving some mass into a smooth tail bump.
            width = int(rng.integers(n // 6, n))
            x = np.linspace(0.0, 1.0, width)
            typ = int(rng.integers(0, 3))
            if typ == 0:
                bump = np.ones(width)
            elif typ == 1:
                bump = np.cos(0.5 * np.pi * x) ** 2
            else:
                bump = np.sin(0.5 * np.pi * x) ** 2
            amp = float(rng.uniform(0.0, max(0.05, scale))) * (1.0 / width)
            b[n - width:n] += amp * bump
        else:
            # Random low-rank multiplicative modulation.
            f1 = int(rng.integers(1, 4))
            f2 = int(rng.integers(1, 6))
            ph1 = float(rng.uniform(0.0, 2.0 * np.pi))
            ph2 = float(rng.uniform(0.0, 2.0 * np.pi))
            t = np.arange(n) / float(max(1, n - 1))
            amp = float(rng.uniform(-1.0, 1.0)) * max(1e-3, scale)
            mod = 1.0 + amp * (0.6 * np.cos(2.0 * np.pi * f1 * t + ph1)
                               + 0.4 * np.cos(2.0 * np.pi * f2 * t + ph2))
            b *= mod

    b = np.where(np.isfinite(b), b, 0.0)
    b = np.clip(b, 0.0, None)
    s = float(b.sum())
    if s <= 1e-15:
        return a.copy()
    return b / s


def _local_improve(a, rng, deadline, consider):
    a = _normalize_valid(a)
    if a is None:
        return
    n = a.size
    cur = _fast_score_norm(a)
    consider(a)
    best_local = a.copy()
    best_local_score = cur

    base_amp = 0.035
    stall = 0
    it = 0
    beta_log = 60.0

    while time.monotonic() < deadline:
        it += 1
        c = _fft_conv(a)
        maxc = float(np.max(c)) if c.size else 0.0
        cur = float(2.0 * n * maxc)

        if cur < best_local_score - 1e-13:
            best_local_score = cur
            best_local = a.copy()
            consider(a)

        # Choose a gradient.  Full softmax gradient every so often for smoother descent.
        if it % 12 == 0 and maxc > 0.0:
            beta = beta_log / maxc
            g = _grad_full_logsumexp(a, c, beta)
        else:
            g = _grad_topk(a, c, rng, k=48, rel=0.035)

        g -= g.mean()
        gn = float(np.linalg.norm(g))
        an = float(np.linalg.norm(a))

        accepted = False
        if gn > 1e-18 and an > 1e-18:
            direction = g / gn
            for amp in (base_amp, 0.5 * base_amp, 0.25 * base_amp, 0.10 * base_amp, 0.04 * base_amp):
                if time.monotonic() >= deadline:
                    break
                trial = a - amp * an * direction
                trial = np.clip(trial, 0.0, None)
                s = float(trial.sum())
                if s <= 1e-15:
                    continue
                trial = trial / s
                tcur = _fast_score_norm(trial)
                if tcur < cur - 1e-15:
                    a = trial
                    cur = tcur
                    accepted = True
                    base_amp = min(0.12, base_amp * 1.07)
                    break

        if not accepted:
            # Try a few structured random moves, both local and around the best known in this basin.
            improving = False
            for restart_best in (False, True):
                src = best_local if restart_best else a
                trial = _perturb(src, rng, scale=0.25 if restart_best else 0.18)
                tcur = _fast_score_norm(trial)
                if tcur < cur - 1e-15:
                    a = trial
                    cur = tcur
                    accepted = True
                    improving = True
                    base_amp = 0.035
                    break
            if not improving:
                stall += 1
                beta_log *= 1.15
                beta_log = min(beta_log, 500.0)
                base_amp *= 0.85
                if base_amp < 1e-4:
                    a = _perturb(best_local, rng, scale=0.45)
                    cur = _fast_score_norm(a)
                    base_amp = 0.03
                    beta_log = 60.0
                    stall = 0
        else:
            if accepted:
                stall = 0
            else:
                stall += 1

        # Occasional exploratory jump even when accepted, to avoid being trapped on a smooth plateau.
        if it % 35 == 0 and time.monotonic() < deadline:
            trial = _perturb(a, rng, scale=0.30)
            tcur = _fast_score_norm(trial)
            if tcur < cur - 1e-15:
                a = trial
                cur = tcur
