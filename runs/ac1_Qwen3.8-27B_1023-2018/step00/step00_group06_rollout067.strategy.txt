```python
    n = len(signal)
    if length is None:
        length = n
    padded = np.fft.rfft(np.pad(signal, int(np.ceil(length/2))-n, mode='symmetric'), length=int(np.ceil(length)))
    padded[np.logical_or(padded < center-width/2, padded > center+width/2)] = 0
    return np.fft.irfft(padded)[:n]
```

```python
    full = np.zeros(n*2)

    time_coeffs = np.fft.irfft(full, n=n)
```

```python
    x = np.clip(x, 0.0, 1000.0)
    total_energy = sum(x)
    if total_energy < 0.01:
```

```python
def calculate_gradient(seq):

    dx = 1e-2
    n = len(seq)

    def perturb(i):

        return calculate_objective(x_pert)
    
    # Compute non-normalized gradient via perturbation
    for i in range(n):
```
