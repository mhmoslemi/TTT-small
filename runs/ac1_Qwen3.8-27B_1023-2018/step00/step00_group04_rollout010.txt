            t_step = max(0.01, t_step * 0.7)
            no_improve += 1
            continue
        
        g = res.x
        gs = g.sum()
        if gs < 1e-10:
            t_step = max(0.01, t_step * 0.7)
            no_improve += 1
            continue
        
        g = g / gs  # normalize to sum=1
        
        # Blend
        a_new = (1.0 - t_step) * a + t_step * g
        a_new = np.clip(a_new, 0.0, None)
        sn = a_new.sum()
        if sn < 1e-10:
            a = a  # skip
            no_improve += 1
            continue
        a_new = a_new / sn
        
        s_new = fast_score(a_new)

            best = a_new.copy()

            no_improve = 0
        else:
            no_improve += 1

        t_step = max(0.01, t_step * 0.997)

            scale = 0.05
            noise = rng.normal(0, scale, n)
            a = np.clip(a + noise, 0, None)
            a = a / a.sum()

            if s_pert < best_score:
                best = a.copy()
                best_score = s_pert
            no_improve = 0
            t_step = min(0.5, t_step * 2)

    if time.monotonic() < deadline - 20:

        for _ in range(20000):

            k_max = int(np.argmax(b))

            ilo = max(0, k_max - n + 1)
            ihi = min(k_max, n - 1)
            if ilo <= ihi:

            a = a - lr * grad
            a = np.clip(a, 0, None)
            s_a = a.sum()

                a = a / s_a
            s_cur = fast_score(a)
            if s_cur < best_score:

                best_score = s_cur
            lr *= 0.9995
    
    # Final: verify with exact evaluator
    final_list = [float(x) for x in best]
    fs = float(evaluate_sequence(final_list))
    
    if not np.isfinite(fs) or fs > 2.0:
        # Fallback to parent
        parent_list = [float(x) for x in parent]
        ps = float(evaluate_sequence(parent_list))
        if np.isfinite(ps):
            return parent_list
        # Last resort: uniform
        return [1.0/n] * n
    
    return final_list

Wait, I need to double-check the gradient computation for the subgradient step.

$b_k = \sum_j a_j a_{k-j}$

$\frac{\partial b_k}{\partial a_i}$:
- Term from $j=i$ in the sum: $\frac{\partial}{\partial a_i}(a_i a_{k-i}) = a_{k-i}$
- Term from $k-j=i$ (i.e., $j=k-i$): $\frac{\partial}{\partial a_i}(a_{k-i} a_i) = a_{k-i}$

Wait, if $j=i$ and $j=k-i$ are the same (i.e., $2i=k$), then we only have one term: $\frac{\partial}{\partial a_i}(a_i^2) = 2a_i$. This is consistent with the two contributions: $a_i + a_i = 2a_i$.

So in general: $\frac{\partial b_k}{\partial a_i} = 2a_{k-i}$ (when $0 \le k-i < n$).

In code: for the gradient vector, entry $i$ is $2a[k_{max} - i]$ if $0 \le k_{max} - i < n$, else 0.
