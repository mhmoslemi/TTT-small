def _prep(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)

    n = a.size

    m = float(c.max())
    s = float(a.sum())
    if s <= 0:
        return np.inf
    return 2.0 * n * m / (s * s), c


def _grad_mhat(a, w):

        if j0 <= j1:
            aidx2 = np.arange(j0 - k, j1 - k + 1)
            grad[j0:j1 + 1] += wk * a[aidx2]
    return grad
