        budget = float(budget_s)
        if np.isfinite(budget) and budget > 30.0:
            deadline = time.monotonic() + max(0.0, budget - 15.0)
            if time.monotonic() < deadline:
                score = float(evaluate_sequence(candidate))
                if not np.isfinite(score) or score > 2.0 + 1e-11:
                    # A shorter constant sequence has the same exact optimum and
                    # is cheap to evaluate; use it only if the long one misbehaves.
                    candidate = [1.0] * 256
    except Exception:
        # The constant candidate is already valid; do not risk a failure.
        pass

    return candidate
