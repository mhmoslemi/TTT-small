def subgradient_step(a, alpha):
    n = len(a)
    b = fftconvolve(a, a, mode='full')
    k_star = np.argmax(b)
    # Compute gradient of b_{k_star} w.r.t. a
    # b_k = sum_i a_i * a_{k-i}
    # db_k/da_j = 2*a_{k-j} if k-j in [0,n-1] and j != k-j
    #           = a_{k-j} if j == k-j (i.e., 2j == k)
    #           = 0 if k-j not in [0,n-1]
    
    k = k_star
    grad = np.zeros(n)
    for j in range(n):
        idx = k - j
        if 0 <= idx < n:
            if 2*j == k:
                grad[j] = a[idx]
            else:
                grad[j] = 2*a[idx]
    
    # Center the gradient to maintain sum
    grad_centered = grad - np.mean(grad)
    
    a_new = a - alpha * grad_centered
    a_new = np.clip(a_new, 0, 1000)
    
    # Renormalize
    if np.sum(a_new) > 0:
        a_new = a_new * (np.sum(a) / np.sum(a_new))
        a_new = np.clip(a_new, 0, 1000)
    
    return a_new
