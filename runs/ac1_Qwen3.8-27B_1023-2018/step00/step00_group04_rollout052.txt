    deadline = time.monotonic() + budget_s - 5

    a = height_sequence_1.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    S0 = a.sum()

    a = a / S0 * a.size
    n = a.size
    
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    
    # Optimization
    while time.monotonic() < deadline:
        b = np.convolve(a, a, mode='full')
        # Find top peaks
        top_k = np.argsort(b)[-5:]  # top 5 elements
        max_b = b.max()
        
        # Gradient: average of gradients at top peaks
        grad = np.zeros(n)
        for k in top_k:
            # d b(k) / d a[m] = 2 * a[k-m] if 0 <= k-m < n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for m in range(lo, hi + 1):
                grad[m] += 2 * a[k - m]
        grad /= len(top_k)
        
        # Project gradient onto tangent space of simplex
        grad -= grad.mean()
        
        # Line search
        alpha = 0.1
        best_a_local = a.copy()
        best_f_local = max_b
        
        for _ in range(20):
            a_trial = a - alpha * grad
            a_trial = np.maximum(a_trial, 0)
            if a_trial.sum() > 1e-10:
                a_trial = a_trial / a_trial.sum() * n
            b_trial = np.convolve(a_trial, a_trial, mode='full')
            f_trial = b_trial.max()
            if f_trial < best_f_local:
                best_f_local = f_trial
                best_a_local = a_trial.copy()
                alpha *= 1.5  # grow step
            else:
                alpha *= 0.5
            if alpha < 1e-6:
                break
        
        a = best_a_local
        
        # Occasionally perturb
        if rng.random() < 0.1:
            a += 0.01 * n * rng.normal(size=n)
            a = np.maximum(a, 0)
            a = a / a.sum() * n
        
        # Check score
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best = a.copy()
