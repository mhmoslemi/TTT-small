import numpy as np
import time


def _proj_simplex(v, s=1.0):
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u * rho) > (cssv - s)
    if cond.any():
        rho_idx = np.nonzero(cond)[0][-1]
        theta = (cssv[rho_idx] - s) / (rho_idx + 1.0)
    else:
        theta = 0.0
    out = np.maximum(v - theta, 0.0)
    # enforce exact normalization for numerical safety
    ss = out.sum()
    if ss > 0:
        out *= (s / ss)
    return out
