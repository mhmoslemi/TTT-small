def _simplex_project(v, target):
    """Project v onto {x>=0, sum(x)=target} using Duchi et al. O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cumulative = np.cumsum(u)
    # Find rho: max j such that (cumulative[j-1] - target)/j < u[j-1]
    # i.e., cumulative[j-1] - target < j * u[j-1]
    # i.e., cumulative[j-1] < j * u[j-1] + target
    Swe = np.arange(1, n+1, dtype=np.float64)

    rho = int(np.sum(cond))

        result = np.zeros(n)
        result[np.argmax(v)] = target  # shouldn't happen if target > 0
        return result
    theta = (cumulative[rho-1] - target) / rho
    return np.maximum(v - theta, 0.0)
