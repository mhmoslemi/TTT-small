                return None
            zfull = zfull - float(np.mean(zfull))
            zfull = zfull - float(zfull[-1])
            mx = float(np.max(zfull))
            if not np.isfinite(mx):
                return None
            ez = np.exp(zfull - mx)
            s = float(ez.sum())
            if s <= 1e-300 or not np.isfinite(s):
                return None
            return ez / s

        def func(z):
            if time.monotonic() > stop_time:
                raise _Timeout()
            f = make_f(z)
            if f is None:
                return 1e12, np.zeros_like(z, dtype=np.float64)

            c = ctx.conv(f)
            if c.size == 0:
                return 1e12, np.zeros_like(z, dtype=np.float64)
            cmax = float(np.max(c))
            if not np.isfinite(cmax) or cmax < 0.0:
                cmax = float(np.abs(cmax)) if np.isfinite(cmax) else 1e12
            if cmax < state["C"]:
                state["C"] = cmax
                state["f"] = f.copy()

            if K <= 0 or K >= m:
                active = np.arange(m, dtype=int)
            else:
                active = np.argpartition(c, -K)[-K:].astype(int)

            mu = max(1e-12, mu_factor * cmax)
            vals = (c[active] - cmax) / mu
            # vals are <= 0 up to roundoff; no overflow.
            w = np.exp(np.minimum(vals, 0.0))
            wsum = float(w.sum())
            if wsum <= 1e-300 or not np.isfinite(wsum):
                return 1e12, np.zeros_like(z, dtype=np.float64)

            obj = cmax + mu * float(np.log(wsum))
            p = w / wsum

            wpad = np.zeros(L, dtype=np.float64)
            wpad[active] = p

            Ff = np.fft.fft(f, L)
            Fw = np.fft.fft(wpad, L)
            v = np.fft.ifft(Fw * np.conj(Ff)).real[:n]
            g = 2.0 * v

            sfg = float(np.dot(f, g))
            if not np.isfinite(sfg):
                sfg = 0.0
            grad = f * (g - sfg)
            grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
            if not np.isfinite(obj):
                obj = 1e12
            return float(obj), grad[:-1]

        z = np.log(f_init + 1e-300)
        z = z - z[-1]
        z = z - float(np.mean(z))
        z = z - z[-1]
        z0 = z[:-1].astype(np.float64, copy=True)

        try:
            minimize(
                func,
                z0,
                jac=True,
                method="L-BFGS-B",
                options={
                    "maxiter": int(maxiter),
                    "maxls": 10,
                    "ftol": 1e-16,
                    "gtol": 1e-10,
                },
            )
        except _Timeout:
            pass
        except Exception:
            pass

        if not np.isfinite(state["C"]):
            state["f"] = f_init
        return clean_normalize(state["f"], n)

    def mirror_refine(f_init, limit_time):
        f = clean_normalize(f_init, N)
        local_best = fin.score(f)
        bestf = f.copy()
        stall = 0
        it = 0

        while time.monotonic() < limit_time:
            c = fin.conv(f)
            if c.size == 0:
                break
            cmax = float(np.max(c))
            if not np.isfinite(cmax) or cmax <= 1e-15:
                break
            current = float(2.0 * N * cmax)
            if current < local_best * (1.0 - 1e-11):
                local_best = current
                bestf = f.copy()

            K = min(2048, fin.M)
            if K <= 1:
                active = np.arange(fin.M, dtype=int)
            else:
                active = np.argpartition(c, -K)[-K:].astype(int)

            mu = max(1e-12, 0.05 * cmax)
            vals = (c[active] - cmax) / mu
            w = np.exp(np.minimum(vals, 0.0))
            wsum = float(w.sum())
            if wsum <= 1e-300:
                break
            p = w / wsum

            wpad = np.zeros(fin.L, dtype=np.float64)
            wpad[active] = p
            Ff = np.fft.fft(f, fin.L)
            Fw = np.fft.fft(wpad, fin.L)
            v = np.fft.ifft(Fw * np.conj(Ff)).real[:N]
            g = 2.0 * v
            hmax = float(np.max(np.abs(g)))
            if hmax <= 1e-300:
                break

            improved = False
            for eta in (2.0, 1.3, 0.9, 0.6, 0.4, 0.25, 0.15, 0.08):
                fnew = f * np.exp(-eta * g / hmax)
                snew = float(fnew.sum())
                if snew <= 1e-300 or not np.isfinite(snew):
                    continue
                fnew = fnew / snew
                cnew = fin.conv(fnew)
                scnew = float(2.0 * N * np.max(cnew))
                if np.isfinite(scnew) and scnew < current * (1.0 - 1e-10):
                    f = fnew
                    current = scnew
                    improved = True
                    break

            if not improved:
                # Small random smooth perturb to escape plateaus.
                scale = max(0.01, 0.1 / float(N))
                noise = rng.normal(0.0, 0.03, size=N)
                fnew = f * np.exp(noise)
                snew = float(fnew.sum())
                if snew > 1e-300 and np.isfinite(snew):
                    fnew = fnew / snew
                    scnew = fin.score(fnew)
                    if np.isfinite(scnew) and scnew < current * (1.0 + 1e-9):
                        f = fnew
                        current = scnew
                        improved = True

                if not improved:
                    stall += 1
                    if stall > 30:
                        f = bestf.copy()
                        stall = 0
            else:
                stall = 0

            if it % 40 == 0:
                update_best(bestf)
            it += 1

        if time.monotonic() < limit_time + 2.0:
            scfinal = fin.score(f)
            if np.isfinite(scfinal) and scfinal < local_best:
                local_best = scfinal
                bestf = f.copy()
        update_best(bestf)
        return clean_normalize(bestf, N)

    # Coarse-scale search: quickly find a good smooth continuous shape.
    remaining = safe_deadline - time.monotonic()
    coarse_f = None
    if remaining > 150.0:
        Nc = min(256, N)
        ctxc = Ctx(Nc)
        inits_c = make_inits(Nc, rng, nrand=8)
        scored = []
        for ff in inits_c:
            sc = ctxc.score(ff)
            if np.isfinite(sc):
                scored.append((sc, ff))
        scored.sort(key=lambda z: z[0])
        coarse_iters = min(6, len(scored))
        time_per = min(8.0, max(2.0, (safe_deadline - time.monotonic() - 120.0) * 0.05))
        for _, ff in scored[:coarse_iters]:
            if time.monotonic() >= safe_deadline - 130.0:
                break
            fopt = run_scipy(ff, ctxc, time_per, 500, min(80, 2 * Nc - 1), 0.05)
            if coarse_f is None:
                coarse_f = fopt
            else:
                if ctxc.score(fopt) < ctxc.score(coarse_f):
                    coarse_f = fopt

    # Build final-scale starting points.
    starts = []
    if coarse_f is not None and coarse_f.size > 1:
        xold = np.linspace(0.0, 1.0, coarse_f.size)
        xnew = np.linspace(0.0, 1.0, N)
        starts.append(clean_normalize(np.interp(xnew, xold, coarse_f), N))

    if parent is not None:
        starts.append(clean_normalize(parent, N))

    # A modest number of final-grid deterministic starts.
    inits_f = make_inits(N, rng, nrand=6)
    scored_f = []
    for ff in inits_f:
        sc = fin.score(ff)
        if np.isfinite(sc):
            scored_f.append((sc, ff))
    scored_f.sort(key=lambda z: z[0])
    starts.extend(ff for sc, ff in scored_f[:2])

    def add_unique(starts_list, vec, tol=0.02):
        v = clean_normalize(vec, N)
        for u in starts_list:
            d = float(np.linalg.norm(np.asarray(v) - np.asarray(u)))
            if d < tol:
                return
        starts_list.append(v)

    unique_starts = []
    for s in starts:
        add_unique(unique_starts, s)
    if len(unique_starts) == 0:
        unique_starts.append(np.ones(N, dtype=np.float64) / float(N))

    # Final-scale L-BFGS refinement on a few best starts.
    remaining = safe_deadline - time.monotonic()
    reserve = 260.0
    if remaining > reserve + 80.0:
        final_budget = max(60.0, min(780.0, remaining - reserve))
    else:
        final_budget = max(0.0, remaining - 30.0)

    max_runs = min(3, len(unique_starts))
    if final_budget > 30.0 and max_runs > 0:
        for run_i in range(max_runs):
            local = min(260.0, final_budget / float(max(1, max_runs - run_i)))
            if local < 8.0 or time.monotonic() >= safe_deadline - reserve:
                break
            f0 = unique_starts[run_i]
            fopt = run_scipy(f0, fin, local, 600, min(2048, fin.M), 0.05)
            if fopt is not None:
                update_best(fopt)
                # Use the best exact candidate for later runs if available.
                if best_out is not None and run_i + 1 < max_runs:
                    fnext = clean_normalize(best_out, N)
                    unique_starts[run_i + 1] = fnext

    # Subgradient/mirror refinement.
    remaining = safe_deadline - time.monotonic()
    if remaining > 80.0:
        start_norm = clean_normalize(best_out, N) if best_out is not None else None
        if start_norm is None:
            start_norm = np.ones(N, dtype=np.float64) / float(N)
        mirror_limit = safe_deadline - max(8.0, 0.04 * budget)
        f_refined = mirror_refine(start_norm, mirror_limit)
        update_best(f_refined)

    # Final validation and safe fallback.
    if best_out is None:
        best_out = np.ones(N, dtype=np.float64)
    result = [float(max(0.0, min(1000.0, x))) for x in best_out]
    try:
        fs = evaluate_sequence(result)
    except Exception:
        fs = np.inf
    if (not np.isfinite(fs)) or len(result) == 0:
        result = [1.0] * max(1, N)
        try:
            fs = evaluate_sequence(result)
        except Exception:
            fs = 2.0
        if not np.isfinite(fs):
            result = [1.0]

    return result
