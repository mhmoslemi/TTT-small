def fast_score(a):
    n = a.size
    if n <= 1:
        s = a.sum()
        return 2.0*n*a[0]*a[0]/(s*s) if s>0 else np.inf
    m = 2*n - 1
    size = 1 << (m - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)[:m]
    mb = float(conv.max())
    s = float(a.sum())
    if s <= 0: return np.inf
    return 2.0*n*mb/(s*s)

import numpy as np
import time
import math


def _fast_score(a):
    """Fast surrogate for the exact evaluator using a zero-padded FFT.
    Matches 2*n*max(convolve(a,a))/(sum a)**2 up to floating point rounding."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 0:
        return np.inf
    s = float(a.sum())
    if s <= 0.0:
        return np.inf
    m = 2 * n - 1
    if m == 1:
        return 2.0 * float(a[0] * a[0]) / (s * s)
    size = 1
    while size < m:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:m]
    mb = float(conv.max())
    return 2.0 * n * mb / (s * s)


def _exact(a):
    """Authoritative score via the pre-imported evaluator."""
    try:
        v = evaluate_sequence([float(x) for x in a])
        v = float(v)
    except Exception:
        return np.inf
    if not np.isfinite(v):
        return np.inf
    return v


def _perturb(base, rng):
    """O(n) nonnegative multiplicative block perturbation."""
    a = base.copy()
    n = a.size
    if n < 2:
        return a
    nblk = int(rng.integers(1, 5))
    for _ in range(nblk):
        lo = max(1, n // 2000)
        hi = max(2, n // 40)
        w = int(rng.integers(lo, hi + 1))
        if w > n:
            w = n
        st = int(rng.integers(0, max(1, n - w)))
        a[st:st + w] *= float(rng.normal(1.0, 0.10))
    np.clip(a, 0.0, 1000.0, out=a)
    if not np.isfinite(a).all():
        a = np.nan_to_num(a, nan=1.0, posinf=1000.0, neginf=0.0)
        np.clip(a, 0.0, 1000.0, out=a)
    return a


def _amplitude_map(shape):
    """Map a unit-range shape to coefficients in [1, 1000]; score is
    scale invariant so the absolute scale does not change the objective."""
    s = np.asarray(shape, dtype=np.float64)
    s = np.nan_to_num(s, nan=0.0, posinf=1.0, neginf=0.0)
    s = np.clip(s, 0.0, 1.0)
    return 1.0 + 999.0 * s
