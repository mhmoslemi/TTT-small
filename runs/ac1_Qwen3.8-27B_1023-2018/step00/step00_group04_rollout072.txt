import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog


def _ac1(a):
    """Fast AC1 score for a 1-D float array (same formula as evaluator)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = a.sum()
    if s < 0.01:
        return float("inf")
    c = np.convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _build_conv_matrix(f):
    """Return sparse csc matrix A with A[k,l] = f[k-l] if 0<=k-l<=n-1."""
    n = f.size
    rows = np.repeat(np.arange(n, dtype=np.int32), n) + np.tile(np.arange(n, dtype=np.int32), n)
    cols = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    return sparse.csc_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_direction(f, M, max_iters=400):
    """Solve max sum(g) s.t. (f*g)<=M, g>=0.  Return g normalized to sum 1, or None."""
    n = f.size
    top = float(f.max())
    if top <= 1e-15:
        return None
    ub = 2.0 * M / top + 1.0
    try:
        A = _build_conv_matrix(f)
    except Exception:
        return None
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0.0, ub)] * n,
            method="highs",
            options={"time_limit": float(max(1.0, (3.0))), "presolve": True},
        )
    except TypeError:
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, ub)] * n, method="highs",
                          options={"presolve": True})
        except Exception:
            return None
    if getattr(res, "status", None) not in (0,) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    s = g.sum()
    if s <= 1e-12:
        return None
    return g / s

    return np.clip(c, 0.0, 1000.0)
