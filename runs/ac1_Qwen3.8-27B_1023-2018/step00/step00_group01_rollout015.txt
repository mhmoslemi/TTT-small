import numpy as np
import time


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0

    # Get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(1024, dtype=np.float64)

    n = parent.size

    # FFT size for linear convolution
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2

    def compute_C(a):
        F = np.fft.rfft(a, fft_size)
        C_full = np.fft.irfft(F * F, fft_size)
        C_full = np.maximum(C_full, 0.0)
        return C_full[:2 * n - 1]

    def fast_score(a):
        total = a.sum()
        if total < 1e-12:
            return np.inf
        C = compute_C(a)
        return 2.0 * n * C.max() / (total * total)

    # --- Phase 1: Try diverse asymmetric initializations ---
    best = None
    best_score = np.inf

    # Parent
    if parent.sum() > 0:
        c = parent / parent.sum()
        s = float(evaluate_sequence(c.tolist()))
        if np.isfinite(s) and s < best_score:
            best = c.copy()
            best_score = s

    x = np.linspace(0.0, 1.0, n)
    shapes = [
        (1 + 0.3 * (2 * x - 1)) / n,
        (1 + 0.7 * (2 * x - 1)) / n,
        (1 + 1.0 * (2 * x - 1)) / n,
        x / n,
        (1 - x) / n,
        x * x / n,
        (1 - x) * (1 - x) / n,
        np.exp(0.5 * x) / np.exp(0.5 * x).sum(),
        np.exp(-0.5 * x) / np.exp(-0.5 * x).sum(),
        np.exp(1.5 * x) / np.exp(1.5 * x).sum(),
        np.exp(-1.5 * x) / np.exp(-1.5 * x).sum(),
        (1 + 0.2 * np.cos(2 * np.pi * x)) / n,
        (1 + 0.2 * np.sin(2 * np.pi * x)) / n,
        (1 + 0.4 * np.cos(4 * np.pi * x)) / n,
        (x * (1 - x)) / (x * (1 - x)).sum(),
    ]

    for sh in shapes:
        c = np.maximum(sh, 0.0)
        if c.sum() < 1e-10:
            continue
        c = c / c.sum()
        c = np.clip(c, 0.0, 1000.0)
        s = float(evaluate_sequence(c.tolist()))
        if np.isfinite(s) and s < best_score:
            best = c.copy()
            best_score = s
            # Try tilted versions of this shape
            for beta in [0.2, 0.5, 0.8]:
                tilt = 1 + beta * np.linspace(-1, 1, n)
                c2 = np.maximum(c * tilt, 0.0)
                if c2.sum() < 1e-10:
                    continue
                c2 = c2 / c2.sum()
                s2 = float(evaluate_sequence(c2.tolist()))
                if np.isfinite(s2) and s2 < best_score:
                    best = c2.copy()
                    best_score = s2

    if best is None:
        fallback = parent / parent.sum()
        return [float(v) for v in fallback]

    # --- Phase 2: Iterative peak-reduction optimization ---
    a = best.copy()
    K = min(50, 2 * n - 1)

    for it in range(500000):
        if time.monotonic() > deadline - 5.0:
            break

        C = compute_C(a)
        max_C = C.max()

        if max_C < 1e-15:
            break

        # Identify top-K peaks
        if K < 2 * n - 1:
            top_idx = np.argpartition(C, -K)[-K:]
        else:
            top_idx = np.arange(2 * n - 1)

        # Compute blame: total sensitivity of peak C[k] to each a[j]
        blame = np.zeros(n)
        for k in top_idx:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            js = np.arange(j_min, j_max + 1)
            other = k - js
            blame[js] += a[other] + a[js]

        bmax = blame.max()
        if bmax < 1e-15:
            break

        blame_n = blame / bmax

        # Step size with annealing
        eta = 0.04 / (1.0 + it / 2000.0)

        # Multiplicative update: reduce a where blame high, increase where low
        shift = blame_n - blame_n.mean()
        a_new = a * np.exp(-eta * shift / max(shift.std(), 1e-10))
        a_new = np.maximum(a_new, 1e-15)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

            best_score = s_new
            a = a_new

            a = a / a.sum()

        # Periodic: re-seed from best with larger perturbation
        if it > 0 and it % 3000 == 2999:

            a = a / a.sum()

        # Occasional full restart from different shape
        if it > 0 and it % 10000 == 9999:
            x_r = np.linspace(0.0, 1.0, n)
            phase = rng.uniform(0, 2 * np.pi)
            shape_new = 1.0 + 0.3 * np.sin(2 * np.pi * x_r + phase) + 0.2 * rng.uniform(-1, 1) * (2 * x_r - 1)
            shape_new = np.maximum(shape_new, 0.01)
            s_test = fast_score(shape_new / shape_new.sum())
            if np.isfinite(s_test) and s_test < best_score * 1.05:
                a = shape_new / shape_new.sum()

    # --- Phase 3: Final local refinement ---
    a = best.copy()
    K2 = min(100, 2 * n - 1)
    eta_fine = 0.015

    for it2 in range(200000):
        if time.monotonic() > deadline - 3.0:
            break

        C = compute_C(a)
        max_C = C.max()
        if max_C < 1e-15:
            break

        if K2 < 2 * n - 1:
            top_idx = np.argpartition(C, -K2)[-K2:]
        else:
            top_idx = np.arange(2 * n - 1)

        blame = np.zeros(n)
        for k in top_idx:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            js = np.arange(j_min, j_max + 1)
            other = k - js
            blame[js] += a[other] + a[js]

        bmax = blame.max()
        if bmax < 1e-15:
            break

        blame_n = blame / bmax
        shift = blame_n - blame_n.mean()
        denom = max(shift.std(), 1e-10)
        a_new = a * np.exp(-eta_fine * shift / denom)
        a_new = np.maximum(a_new, 1e-15)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            continue
        a_new = a_new / a_new_sum

        s_new = fast_score(a_new)
        if np.isfinite(s_new) and s_new < best_score:
            best = a_new.copy()
            best_score = s_new
            a = a_new
        else:
            a = a_new  # still move (exploratory)

    # --- Final validation with exact evaluator ---
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # Fallback to parent
        return [float(v) for v in (parent / parent.sum())]

    return [float(v) for v in best]
