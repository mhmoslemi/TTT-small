import numpy as np
import time


def _conv_fft(a, fsize):
    """Linear convolution a*a via FFT, returning 2n-1 entries."""
    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)[:2 * len(a) - 1]


def _corr_affine(a, w, corr_fsize):
    """Return corr[j] = sum_m a[m]*w[j+m]  and  wsum[j] = sum_{k=j}^{j+n-1} w[k]."""
    n = len(a)
    # wsum via cumsum
    tc = np.empty(len(w) + 1)
    tc[0] = 0.0
    np.cumsum(w, out=tc[1:])
    wsum = tc[n:] - tc[:n]
    # cross-correlation via FFT
    A = np.fft.rfft(a, corr_fsize)
    W = np.fft.rfft(w, corr_fsize)
    corr = np.fft.irfft(np.conj(A) * W, corr_fsize)[:n]
    return corr, wsum


def make_block_perturbation(base, rng):
    """O(n) nonnegative perturbation."""
    c = base.copy()
    n = c.size
    width = int(rng.integers(max(1, n // 300), max(2, n // 10)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    c[start:stop] *= float(np.exp(rng.normal(0.0, 0.08)))
    if rng.random() < 0.3:
        c = 0.5 * (c + c[::-1])
    return np.clip(c, 0.0, 1000.0)
