    Jold=1e18

        bc=np.fft.ifft(A*A).real[:2*n-1]

        m=b.max()

def _run_once(n, u0, deadline):
    u = np.clip(np.asarray(u0, dtype=np.float64), 0.0, 1000.0).copy()
    s = float(np.sum(u))
    if (not np.isfinite(s)) or s < 1e-9:
        u = np.ones(n, dtype=np.float64)
        s = float(n)
    u = u / s  # normalize to sum = 1 (score is scale invariant)

    N = 2 * n
    L = 2 * n - 1
    beta = 10.0
    beta_max = 250.0
    eta = 1.0
    eta_floor = 1e-6
    uniform = 1.0 / n

    best = u.copy()
    best_exact = float(evaluate_sequence(best.tolist()))
    J = 1e18
    stag = 0
    it = 0

    while time.monotonic() < deadline:
        try:
            A = np.fft.fft(u, N)
            b = np.fft.ifft(A * A).real[:L]
            b = np.maximum(b, 0.0)
            m = float(b.max())
            eb = np.exp(beta * (b - m))
            ss = float(eb.sum())
            if ss <= 0.0 or not np.isfinite(ss):
                break
            p = eb / ss
            J = m + np.log(ss) / beta

            P = np.fft.fft(p, N)
            g = 2.0 * (np.fft.ifft(P * np.conj(A)).real[:n])
            if not np.all(np.isfinite(g)):
                break
            gnorm = float(np.max(np.abs(g)))
            if gnorm < 1e-14:
                # converged enough at this beta: sharpen
                beta = min(beta_max, beta * 2.0)
                if beta >= beta_max:
                    break
                continue

            # adaptive / backtracking step
            eta = min(eta, 0.9 / gnorm)
            accepted = False
            for _ in range(10):
                un = u - eta * g
                un = np.maximum(un, 0.0)
                su = float(un.sum())
                if su < 1e-12:
                    eta = max(eta_floor, 0.5 * eta)
                    continue
                un = un / su
                A2 = np.fft.fft(un, N)
                b2 = np.fft.ifft(A2 * A2).real[:L]
                b2 = np.maximum(b2, 0.0)
                m2 = float(b2.max())
                eb2 = np.exp(beta * (b2 - m2))

                    continue

                    u = un
                    J = J2
                    accepted = True
                    break
                eta = max(eta_floor, 0.4 * eta)

                # tiny exploration to escape numerically stuck states

                su = float(u.sum())
                if su < 1e-12:
                    break
                u = u / su

                    best = u.copy()

            # anneal smoothing towards the true maximum

            if J < Jold - 1e-8:
                stag = 0

    if np.isfinite(ex):
        best_exact = ex
