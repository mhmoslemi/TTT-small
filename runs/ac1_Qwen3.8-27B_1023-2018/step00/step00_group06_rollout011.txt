```python
def solve_best_response(f, M):
    n = len(f)
    # Build constraint matrix A (2n-1 x n)
    # A[k,i] = f[i] if max(0,k-n+1) <= i <= min(n-1,k)
    n_rows = 2*n - 1
    # For each k, the valid i range
    # Total nonzeros = n^2
    # Build as sparse
    rows = []
    cols = []
    vals = []
    for k in range(n_rows):
        i_start = max(0, k - n + 1)
        i_end = min(n-1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(n_rows, n))
    # LP: max sum(g) s.t. A @ g <= M, g >= 0
    # Convert to min: min -sum(g) s.t. A @ g <= M, g >= 0
    c = -np.ones(n)
    b = np.full(n_rows, M)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    return res.x
```

```python
def build_conv_matrix(f):
    n = len(f)
    n_rows = 2*n - 1
    # For each k, valid i range: [max(0,k-n+1), min(n-1,k)]
    k_vals = np.arange(n_rows)
```

```python
    block_starts = np.repeat(np.cumsum(lengths) - lengths, lengths)
    offsets = np.arange(total_nz) - block_starts
    cols = np.repeat(i_starts, lengths) + offsets
    return rows, cols, n_rows


def _make_triangular(n):
    """Create a normalized triangular sequence [1,2,...,n]/sum."""
    a = np.arange(1, n + 1, dtype=np.float64)
    a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t_start = time.monotonic()
    t_deadline = t_start + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    # Determine sequence length from parent if available
    try:
        n = len(height_sequence_1)
    except (NameError, TypeError):
        n = 5582

    # Triangular initialization: score ~ 4/3 for large n (already below target 1.5027)
    a = _make_triangular(n)
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())

    # Also consider parent as a possible starting point
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size == n and np.all(np.isfinite(parent)):
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.sum() >= 0.01:
                pn = parent / parent.sum()
                ps = evaluate_sequence(pn.tolist())
                if np.isfinite(ps) and ps < best_score:
                    best = pn.copy()
                    best_score = ps
                    a = pn.copy()
    except Exception:
        pass

    # Precompute LP matrix sparsity pattern (depends only on n)
    rows, cols, n_rows = _precompute_pattern(n)

    # LP best-response iteration
    t_step = 0.5
    stall_count = 0

    for it in range(2000):
        if time.monotonic() >= t_deadline:
            break

        # Compute current convolution peak
        conv = np.convolve(a, a)
        M = float(conv.max())
        if M < 1e-15:
            break

        # Build constraint matrix with current a (ensure no zeros for LP boundedness)
        a_safe = np.maximum(a, 1e-12)
        a_safe /= a_safe.sum()
        vals = a_safe[cols]
        A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(n_rows, n))

        # Solve LP: maximize sum(g) subject to (a*g)_k <= M for all k, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(n_rows, M)
        remaining = t_deadline - time.monotonic() - 5.0
        if remaining < 2.0:
            break

        try:
            res = scipy.optimize.linprog(
                c_obj, A_ub=A, b_ub=b_ub,
                bounds=[(0.0, None)] * n,
                method='highs',
                options={'time_limit': min(90.0, remaining), 'presolve': True}
            )
        except Exception:
            break

        if res.x is None or not np.all(np.isfinite(res.x)):
            break

        g = np.maximum(res.x, 0.0)
        S = g.sum()
        if S < 1e-12:
            break
        g /= S  # normalize to unit sum

        # Blend: new_a = (1-t)*a + t*g
        a_new = (1.0 - t_step) * a + t_step * g
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-12:
            break
        a_new /= a_new_sum
        a = a_new

        # Evaluate with the exact evaluator
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            stall_count = 0
        else:
            stall_count += 1
            # Reduce step size when stalling
            t_step = max(0.05, t_step * 0.8)

        # Occasionally try a slightly different step for exploration
        if stall_count > 15:
            t_step = 0.3
            # Add tiny perturbation to escape potential plateau
            noise = rng.multivariate_normal(np.zeros(n), np.diag(np.maximum(a, 1e-6)))
            a = np.maximum(a + 0.01 * noise, 0.0)
            a /= a.sum()
            stall_count = 0

    # If LP didn't help much, try a few quick perturbative improvements
    if time.monotonic() < t_deadline - 10:
        for _ in range(200):
            if time.monotonic() >= t_deadline - 5:
                break
            # Block perturbation on best
            cand = best.copy()
            width = max(1, n // 50)
            start = int(rng.integers(0, n - width))
            cand[start:start + width] *= np.exp(rng.normal(0, 0.1))
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 1e-10:
                continue
            cand /= cand.sum()
            sc = evaluate_sequence(cand.tolist())
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc

    # Final validation: ensure we return a valid sequence
    try:
        final_score = evaluate_sequence(best.tolist())
        if not np.isfinite(final_score) or best_score >= 2.0:
            raise ValueError("Fallback to triangular")
    except Exception:
        best = _make_triangular(n)

    # Ensure all values are finite and in valid range
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = _make_triangular(n)

    return [float(x) for x in best]
```
