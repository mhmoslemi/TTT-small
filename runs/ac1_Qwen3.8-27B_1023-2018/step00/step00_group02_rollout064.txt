def _fft_len(need):
    L = 1
    while L < need:
        L <<= 1
    return L


def _fast_score(a, n):
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _cross_corr(w, a, L):
    # (w * a_corr)[j] = sum_m w[j+m] * a[m], j = 0..len(a)-1
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    return np.fft.irfft(W * np.conj(A), L)


def _softmax(x, beta):
    x = beta * (x - x.max())
    e = np.exp(np.clip(x, -50.0, 50.0))
    z = e.sum()
    if z <= 0:
        return np.full_like(x, 1.0 / x.size)
    return e / z


def _anti_sym_init(n, rng, amp):
    t = np.arange(n)
    u = 2.0 * t / max(1, n - 1) - 1.0
    env = np.sin(np.pi * (t + 0.5) / n)
    base = 1.0 + amp * (u * env)
    base += rng.normal(0.0, 0.02, n)
    return np.clip(base, 1e-3, 1000.0)

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)

            rows.append(r)
            cols.append(j)
            data.append(a[j])
    A = sp.csc_matrix((data, (rows, cols)), shape=(K, n))

    return b / sb * n
