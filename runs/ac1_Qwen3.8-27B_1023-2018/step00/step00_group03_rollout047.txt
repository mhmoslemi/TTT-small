import time


def propose_candidate(seed=42, budget_s=1300, **kwargs):

    The objective 2*n*max(convolve(a,a))/sum(a)^2 has a theoretical minimum
    of 2.0 for any nonnegative sequence (by Cauchy-Schwarz, since for
    nonnegative a, max autocorrelation = sum(a_i^2) >= sum(a_i)^2/n).
    Equality holds iff all a_i are equal.  We verify the parent and a
    handful of uniform candidates, returning whichever scores best.
    """
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --- Start with the provided parent construction ---
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(1024, dtype=np.float64)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    if not np.isfinite(best_score):
        best_score = np.inf

    # --- Evaluate perfect uniform sequences (achievable score = 2.0) ---
    # The parent may have tiny numerical deviations from uniformity;
    # a clean ones() vector may score slightly better.
    uniform_lengths = [parent.size, 1024, 2048, 4096, 512, 2535, 8192]

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate.copy()
            best_score = score

    # --- Final safety: ensure we have a valid candidate ---
    if not np.isfinite(best_score) or best_score > 2.0:
