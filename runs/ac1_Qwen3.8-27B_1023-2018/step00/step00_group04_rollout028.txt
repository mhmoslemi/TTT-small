import numpy as np
import time
import scipy.signal

def _project_to_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if rho.size == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fast_conv(a):
    """Compute auto-correlation using FFT for speed."""
    n = a.size
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fa2 = fa * fa
    return np.fft.irfft(fa2, size)[:2 * n - 1]


def _subgradient_at_lags(a, lags, n):
    """Compute sum of subgradients at given lags. grad[m] = 2*a[k-m] for 0<=k-m<n."""
    grad = np.zeros(n)
    for k in lags:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start >= j_end:
            continue
        j = np.arange(j_start, j_end)
        m = k - j
        grad[m] += 2.0 * a[j]
    return grad
