def build_basis(n, n_dct=512, n_rand=128, seed):
  Ks = list(range(1, min(n, n_dct+1)))
  i=np.arange(n)[:,None]
  # DCT-II normalized columns: for k=1..K
  kcols=np.arange(1,K+1)
  Vdct = np.sqrt(2.0/n)*np.cos(np.pi * np.outer(i+0.5, kcols)/n)
  V=Vdct
  if n_rand>0:
    Vr = rng.standard_normal((n,n_rand))
    norms=np.linalg.norm(Vr,axis=0)
    Vr[norms>0] /= norms[norms>0]
    V=np.hstack([Vdct,Vr])
  # maybe normalize columns to unit norm again
  col_norms=np.linalg.norm(V,axis=0)
  V = V/col_norms
  return V
```

import numpy as np
import time

def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s

def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0: return z
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0/z.size)
    a = e/s
    # due to numerical sum maybe 1
    return a

def _score_array(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0: return np.inf
    s = float(a.sum())
    if s < 1e-12: return np.inf
    c = np.convolve(a, a)
    if c.size == 0: return np.inf
    return float(2.0 * a.size * np.max(c) / (s*s))

def _cross(a, q):
    n = a.size
    m = q.size
    if m < n:
       q = np.concatenate([q, np.zeros(n-m)])
    size = n + m - 1
    L = 1 << max(1, (size - 1).bit_length())
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(q[::-1], L)
    conv = np.fft.irfft(A * B, L)
    idx = (m - 1) - np.arange(n, dtype=np.int64)
    return conv[idx]

def _gradient_v(a, c, beta, e_weight):
    cmax = np.max(c)
    # if beta*range too small? p uniform
    zc = beta * (c - cmax)
    # clip lower? no.
    p = np.exp(zc)
    ps = p.sum()
    if not np.isfinite(ps) or ps <= 0.0:
       p = np.zeros_like(c); p[int(np.argmax(c))] = 1.0
    else:
       p /= ps
    r_c = _cross(a, c)
    r_p = _cross(a, p)
    v = 2.0 * r_p
    if e_weight != 0.0:
       v = v + (4.0 * e_weight) * r_c
    dot = float(np.dot(v, a))
    return a * (v - dot)

def _logit(a):
    a = np.asarray(a, dtype=np.float64)
    z = np.log(np.maximum(a, 1e-12))
    return z - np.mean(z)

def _adam(params, grad, state, lr, cap=1.0, beta1=0.9, beta2=0.999, eps=1e-8):
    m = state['m']; v = state['v']; t = state['t']
    t += 1.0
    m = beta1*m + (1.0-beta1)*grad
    v = beta2*v + (1.0-beta2)*(grad*grad)
    bc1 = 1 - beta1**t
    bc2 = 1 - beta2**t
    step = lr * (m/bc1) / (np.sqrt(v/bc2) + eps)
    if not np.all(np.isfinite(step)):
        step = np.zeros_like(params)
    if cap is not None:
        step = np.clip(step, -cap, cap)
    params = params - step
    state['m']=m; state['v']=v; state['t']=t
    return params

def _build_basis(m, rng, k_dct=None, k_rand=None):
    if k_dct is None:
       k_dct = int(min(m-1, 256 if m>1024 else 128))
    if k_rand is None:
       k_rand = int(min(m-1, 96 if m>1024 else 48))
    i = np.arange(m, dtype=np.float64)[:, None]
    parts=[]
    if k_dct>0:
      cols=np.arange(1, k_dct+1, dtype=np.float64)
      Vd = np.sqrt(2.0/m)*np.cos(np.pi * np.outer(i[0], cols) / m)
      # reshape: i shape (m,1), cols (k,), outer okay: np.cos(np.pi * (np.arange(m)[:,None]+0.5) * cols[None,:] / m)
      Vd = np.sqrt(2.0/m)*np.cos(np.pi * (np.arange(m, dtype=np.float64)[:,None] + 0.5) * cols[None,:] / m)
      parts.append(Vd)
    if k_rand>0:
      Vr = rng.standard_normal((m,k_rand))
      nr = np.linalg.norm(Vr, axis=0)
      Vr[nr>1e-12] /= np.where(nr>1e-12,nr,1)[None,:]
      parts.append(Vr)
    V=np.hstack(parts)
    nr=np.linalg.norm(V,axis=0)
    nr[nr<1e-12]=1
    V=V/nr[None,:]
    return V

import numpy as np
import time


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0 / float(z.size), dtype=np.float64)
    a = e / s
    return a


def _upsample(a, m):
    a = np.asarray(a, dtype=np.float64)
    if a.size == m:
        out = a.copy()
    else:
        if a.size == 0:
            out = np.ones(m, dtype=np.float64) / float(m)
        else:
            x1 = np.linspace(0.0, 1.0, m)
            x0 = np.linspace(0.0, 1.0, a.size)
            out = np.interp(x1, x0, a)
    return _normalize(out)


def _score_array(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    c = np.convolve(a, a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * a.size * mx / (s * s))


def _cross(a, q):
    n = int(a.size)
    m = int(q.size)
    if m < n:
        q = np.concatenate([q, np.zeros(n - m, dtype=np.float64)])
        m = n
    size = n + m - 1
    if size <= 1:
        L = 1
    else:
        L = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(q[::-1], L)
    conv = np.fft.irfft(A * B, L)
    idx = (m - 1) - np.arange(n, dtype=np.int64)
    return conv[idx]


def _gradient_v(a, c, beta, e_weight):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return np.zeros_like(a)
    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        zc = float(beta) * (c - cmax)
        p = np.exp(np.where(np.isfinite(zc), zc, -745.0))
        ps = float(p.sum())
        if not np.isfinite(ps) or ps <= 0.0:
            p = np.zeros_like(c)
            p[int(np.argmax(c))] = 1.0
        else:
            p = p / ps

    rc = _cross(a, c)
    rp = _cross(a, p)
    v = 2.0 * rp
    if e_weight != 0.0:
        v = v + (4.0 * float(e_weight)) * rc

    if not np.all(np.isfinite(v)):
        v = np.zeros_like(a)
    dot = float(np.dot(v, a))
    grad = a * (v - dot)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad


def _logit(a):
    a = np.asarray(a, dtype=np.float64)
    z = np.log(np.maximum(a, 1e-12))
    z = np.where(np.isfinite(z), z, 0.0)
    return z - float(np.mean(z))


def _adam_state(dim):
    return {
        "m": np.zeros(dim, dtype=np.float64),
        "v": np.zeros(dim, dtype=np.float64),
        "b1": 0.1,
        "b2": 0.001,
        "t": 0.0,
    }


def _adam(params, grad, state, lr, cap=0.8):
    dim = int(params.size)
    g = np.asarray(grad, dtype=np.float64)
    if not np.all(np.isfinite(g)):
        g = np.zeros(dim, dtype=np.float64)

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    state["t"] += 1.0
    state["b1"] += 1.0 - b1
    state["b2"] += 1.0 - b2

    state["m"] = b1 * state["m"] + (1.0 - b1) * g
    state["v"] = b2 * state["v"] + (1.0 - b2) * (g * g)

    mhat = state["m"] / state["b1"]
    vhat = state["v"] / state["b2"]
    step = float(lr) * mhat / (np.sqrt(vhat) + eps)

    if not np.all(np.isfinite(step)):
        step = np.zeros(dim, dtype=np.float64)
    if cap is not None:
        step = np.clip(step, -float(cap), float(cap))

    params = params - step
    params = np.where(np.isfinite(params), params, 0.0)
    return params


def _build_basis(m, rng):
    if m <= 1:
        return np.zeros((m, 0), dtype=np.float64)

    if m > 1024:
        k_dct = min(m - 1, 256)
        k_rand = min(m - 1, 96)
    elif m > 128:
        k_dct = min(m - 1, 160)
        k_rand = min(m - 1, 64)
    else:
        k_dct = min(m - 1, 72)
        k_rand = min(m - 1, 32)

    parts = []
    if k_dct > 0:
        cols = np.arange(1, k_dct + 1, dtype=np.float64)[None, :]
        rows = (np.arange(m, dtype=np.float64)[:, None] + 0.5) / float(m)
        Vd = np.sqrt(2.0 / float(m)) * np.cos(np.pi * rows * cols)
        parts.append(Vd)

    if k_rand > 0:
        Vr = rng.standard_normal((m, k_rand))
        nrm = np.linalg.norm(Vr, axis=0)
        Vr = Vr / np.where(nrm > 1e-12, nrm, 1.0)[None, :]
        parts.append(Vr)

    if not parts:
        return np.zeros((m, 0), dtype=np.float64)

    V = np.hstack(parts).astype(np.float64, copy=False)
    nrm = np.linalg.norm(V, axis=0)
    nrm = np.where(nrm > 1e-12, nrm, 1.0)
    V = V / nrm[None, :]
    return V


def _make_inits(m, rng, parent=None):
    inits = []

    def add(x):
        try:
            a = _normalize(x)
            if a.size == m and np.all(np.isfinite(a)):
                inits.append(a)
        except Exception:
            pass

    if m == 1:
        add(np.ones(1))
        return inits

    if m == 3:
        add(np.array([0.5, 0.1, 0.4], dtype=np.float64))
        add(np.array([0.4, 0.1, 0.5], dtype=np.float64))

    if parent is not None and parent.size > 0:
        p = _upsample(parent, m)
        add(p)
        add(p * (1.0 + 0.03 * rng.standard_normal(m)))
        add(p * (1.0 + 0.10 * rng.standard_normal(m)))

    add(np.ones(m, dtype=np.float64) / float(m))
    add(np.ones(m, dtype=np.float64) + 0.20 * rng.standard_normal(m))
    add(_softmax(rng.normal(size=m, scale=0.4)))
    add(_softmax(rng.normal(size=m, scale=1.0)))

    if m > 2:
        x = np.linspace(0.0, 1.0, m)
        for pw in (0.5, 1.0, 1.8, -0.6, -1.0):
            if pw < 0.0:
                vals = 1.0 + 0.6 / (x + 0.08) ** (-pw)
            else:
                vals = 1.0 + 0.6 * (x ** pw)
            add(vals)

        for dec in (-0.7, 0.7):
            add(np.exp(-dec * x) + 1e-3)

        for f in (2, 5):
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            vals = 1.0 + 0.35 * np.sin(2.0 * np.pi * f * x + ph)
            vals += 0.12 * np.sin(4.0 * np.pi * f * x + 2.0 * ph)
            add(np.maximum(vals, 0.03))

    if len(inits) > 16:
        inits = inits[:16]
    if not inits:
        inits.append(np.ones(m, dtype=np.float64) / float(m))
    return inits


def _random_polish(a, rng, trials=2):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a, np.inf
    best = a.copy()
    best_score = _score_array(best)
    if not np.isfinite(best_score):
        best = np.ones(a.size, dtype=np.float64) / float(a.size)
        best_score = _score_array(best)

    for _ in range(int(max(1, trials))):
        cand = best.copy()
        nb = int(rng.integers(1, 4))
        for _i in range(nb):
            lo = max(2, a.size // 300)
            hi = max(lo + 1, min(a.size, max(4, a.size // 8)))
            width = int(rng.integers(lo, hi + 1))
            width = min(max(1, width), a.size)
            if a.size > 1:
                start = int(rng.integers(0, a.size - width + 1))
            else:
                start = 0
            factor = float(np.exp(rng.normal(0.0, 0.18)))
            cand[start:start + width] *= factor

        if a.size > 6 and rng.random() < 0.35:
            off = int(rng.integers(-a.size // 20, a.size // 20 + 1))
            cand = 0.5 * (cand + np.roll(cand, off))

        cand = _normalize(cand)
        sc = _score_array(cand)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best = cand

    return best, best_score


def _optimize_length(m, inits, start_time, end_time, rng, basis_frac=0.40):
    m = int(m)
    if m <= 0:
        return np.ones(1, dtype=np.float64)

    best_arr = np.ones(m, dtype=np.float64) / float(m)
    best_score = _score_array(best_arr)

    for a in inits:
        a = _normalize(a)
        if a.size != m:
            continue
        sc = _score_array(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_arr = a.copy()

    if end_time <= start_time + 0.25:
        return best_arr

    V = _build_basis(m, rng)
    K = int(V.shape[1])

    basis_start = time.monotonic()
    basis_end = start_time + float(basis_frac) * max(1e-3, end_time - start_time)

    if K > 0 and basis_end > time.monotonic() + 0.6:
        z0 = _logit(best_arr)
        theta = (V.T @ z0) if K > 0 else np.zeros(0)
        theta = np.where(np.isfinite(theta), theta, 0.0)
        theta += 0.25 * rng.standard_normal(K)

        state = _adam_state(K)
        lr = 0.55
        cap = 0.8
        no_improve = 0
        it = 0
        thr = 650 if m > 256 else 260
        span = max(1e-3, basis_end - basis_start)

        while time.monotonic() < basis_end:
            it += 1
            z = V @ theta
            z = np.where(np.isfinite(z), z, 0.0)
            z -= float(np.mean(z))
            a = _softmax(z)
            c = np.convolve(a, a)
            s = float(a.sum())
            if s <= 1e-12 or not np.isfinite(s):
                a = np.ones(m, dtype=np.float64) / float(m)
                c = np.convolve(a, a)
                s = float(a.sum())
            sc = float(2.0 * m * np.max(c) / (s * s))

            if np.isfinite(sc) and sc < best_score - max(1e-12, 1e-7 * best_score):
                best_score = sc
                best_arr = a.copy()
                no_improve = 0
                lr = min(1.1, lr * 1.01)
            else:
                no_improve += 1

            frac = min(1.0, max(0.0, (time.monotonic() - basis_start) / span))
            beta = 20000.0 + 80000.0 * frac
            e_weight = max(0.10, 1.0 - 1.5 * frac)
            grad_x = _gradient_v(a, c, beta, e_weight)
            g = V.T @ grad_x
            if not np.all(np.isfinite(g)):
                g = rng.standard_normal(K) * 1e-3

            theta = _adam(theta, g, state, lr, cap)

            if it % 160 == 0:
                polished, pscore = _random_polish(best_arr, rng, trials=1)
                if np.isfinite(pscore) and pscore < best_score - 1e-10:
                    best_score = pscore
                    best_arr = polished
                    zr = _logit(best_arr)
                    theta = (V.T @ zr) + 0.08 * rng.standard_normal(K)
                    state = _adam_state(K)
                    no_improve = 0

            if no_improve > thr:
                zr = _logit(best_arr)
                if rng.random() < 0.3:
                    zr = zr[::-1]
                theta = (V.T @ zr) + 0.45 * rng.standard_normal(K)
                state = _adam_state(K)
                lr = max(0.12, lr * 0.62)
                no_improve = 0
                thr = max(220, int(thr * 1.25))

    z = _logit(best_arr)
    z = np.where(np.isfinite(z), z, 0.0)
    z += 0.05 * rng.standard_normal(m)

    state = _adam_state(m)
    lr = 0.55
    cap = 0.8
    no_improve = 0
    it = 0
    thr = 900 if m > 256 else 320
    full_start = time.monotonic()
    full_end = end_time - 0.6

    while time.monotonic() < full_end:
        it += 1
        a = _softmax(z)
        c = np.convolve(a, a)
        s = float(a.sum())
        if s <= 1e-12 or not np.isfinite(s):
            a = _normalize(best_arr)
            c = np.convolve(a, a)
            s = float(a.sum())
            z = _logit(a)
        sc = float(2.0 * m * np.max(c) / (s * s))

        if np.isfinite(sc) and sc < best_score - max(1e-12, 1e-7 * best_score):
            best_score = sc
            best_arr = a.copy()
            no_improve = 0
            lr = min(1.1, lr * 1.01)
        else:
            no_improve += 1

        span = max(1e-3, full_end - full_start)
        frac = min(1.0, max(0.0, (time.monotonic() - full_start) / span))
        beta = 30000.0 + 150000.0 * frac
        e_weight = max(0.0, 0.55 - 0.9 * frac)
        grad_x = _gradient_v(a, c, beta, e_weight)

        z = _adam(z, grad_x, state, lr, cap)
        if not np.all(np.isfinite(z)):
            z = _logit(best_arr) + 0.05 * rng.standard_normal(m)
            state = _adam_state(m)

        if it % 220 == 0:
            polished, pscore = _random_polish(best_arr, rng, trials=2)
            if np.isfinite(pscore) and pscore < best_score - 1e-10:
                best_score = pscore
                best_arr = polished
                z = _logit(best_arr) + 0.04 * rng.standard_normal(m)
                state = _adam_state(m)
                no_improve = 0

        if no_improve > thr:
            zr = _logit(best_arr)
            if rng.random() < 0.3:
                zr = zr[::-1]
            z = zr + 0.35 * rng.standard_normal(m)
            state = _adam_state(m)
            lr = max(0.15, lr * 0.62)
            no_improve = 0
            thr = max(300, int(thr * 1.25))

    return _normalize(best_arr)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    safety = 8.0
    deadline = start + max(1.0, float(budget_s) - safety)

    n = 2535
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if float(parent.sum()) > 1e-12:
                parent = parent / float(parent.sum())
                n = int(parent.size)
            else:
                parent = None
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(n, dtype=np.float64) / float(n)

    short3 = np.array([0.5, 0.1, 0.4], dtype=np.float64)
    best_any_arr = short3.copy()
    best_any_score = _score_array(short3)
    full_best = None
    full_score = np.inf

    try:
        if deadline - start < 25.0:
            pass
        else:
            base_lengths = (3, 7, 15, 31, 63, 127, 255, 511, 1023, 1536)
            lengths = sorted({L for L in base_lengths if L < n})
            if n == 4 and 3 not in lengths:
                lengths = [3]
            elif n == 5:
                lengths = sorted({3, 4})
            elif n == 6:
                lengths = sorted({3, 5})

            reserve_final = max(180.0, 0.55 * (deadline - start))
            current = parent

            for L in lengths:
                if time.monotonic() >= deadline - reserve_final:
                    break
                stage_time = max(2.0, 1.5 * np.sqrt(float(L)))
                stage_end = time.monotonic() + stage_time
                if stage_end > deadline - reserve_final:
                    stage_end = time.monotonic() + max(1.0, deadline - reserve_final - time.monotonic())
                if stage_end <= time.monotonic() + 0.5:
                    continue

                inits = _make_inits(L, rng, parent=current)
                best_L = _optimize_length(
                    L,
                    inits,
                    time.monotonic(),
                    stage_end,
                    rng,
                    basis_frac=0.45 if L < 64 else 0.35,
                )
                best_L = _normalize(best_L)
                sc = _score_array(best_L)
                if L != n and np.isfinite(sc) and sc < best_any_score:
                    best_any_score = sc
                    best_any_arr = best_L.copy()
                current = best_L

            final_inits = []
            if parent is not None and parent.size == n:
                final_inits.append(parent.copy())
            if current is not None:
                final_inits.append(_upsample(current, n))
            if best_any_arr is not None and best_any_arr.size != n:
                final_inits.append(_upsample(best_any_arr, n))
            final_inits.extend(_make_inits(n, rng, parent=parent))

            full_start = time.monotonic()
            full_end = min(deadline - 2.0, full_start + max(30.0, deadline - full_start))
            if full_end > full_start + 1.0:
                full_best = _optimize_length(
                    n,
                    final_inits,
                    full_start,
                    full_end,
                    rng,
                    basis_frac=0.32,
                )
                full_best = _normalize(full_best)
                full_score = _score_array(full_best)
                if np.isfinite(full_score) and full_score < best_any_score:
                    best_any_score = full_score
                    best_any_arr = full_best.copy()
    except Exception:
        pass

    candidates = []
    if full_best is not None:
        candidates.append(full_best)
    if best_any_arr is not None:
        candidates.append(best_any_arr)
    candidates.append(short3)

    final_list = None
    final_score = np.inf
    for cand in candidates:
        try:
            lst = [float(x) for x in np.asarray(cand, dtype=np.float64)]
            if not lst:
                continue
            if any(not np.isfinite(x) for x in lst):
                continue
            sc = float(evaluate_sequence(lst))
            if np.isfinite(sc) and sc < final_score:
                final_score = sc
                final_list = lst
        except Exception:
            pass

    if final_list is None:
        final_list = [0.5, 0.1, 0.4]

    return final_list
