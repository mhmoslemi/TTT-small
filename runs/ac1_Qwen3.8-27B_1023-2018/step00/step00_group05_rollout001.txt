    """Call the authoritative evaluator, converting all failures to +inf."""
    try:
        value = float(evaluate_sequence(sequence))
    except Exception:
        return float("inf")
    try:
        return float(value) if np.isfinite(value) else float("inf")
    except Exception:
        return float("inf")


def _valid_sequence(sequence, limit_len=_EVAL_LEN_LIMIT):
    """Return a clipped, finite, plain-float list, or [] if unusable."""
    try:
        m = len(sequence)
    except Exception:
        return []
    if m <= 0 or m > limit_len:
        return []

    out = []
    for x in sequence:
        try:
            y = float(x)
        except Exception:
            return []
        if not np.isfinite(y):
            return []
        if y < 0.0:
            y = 0.0
        elif y > 1000.0:
            y = 1000.0
        out.append(float(y))
    return out


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    start = time.monotonic()
    try:
        total = float(budget_s)
    except Exception:
        total = 1300.0
    if not np.isfinite(total) or total < 0.0:
        total = 1300.0

    # Leave headroom for final cleanup/return.
    deadline = start + max(0.0, total - 2.0)

    # A trivial exact optimum constant, always valid.
    best = [1.0]
    best_score = _safe_score(best)

    def consider(candidate):
        nonlocal best, best_score
        if time.monotonic() > deadline:
            return

        cand = _valid_sequence(candidate)
        if not cand:
            return

        try:
            if sum(cand) < 0.01:
                return
        except Exception:
            pass

        score = _safe_score(cand)
        if score < best_score:
            best_score = score
            best = cand

    # Use the available parent construction directly when it is cheap to score.
    try:
        parent = height_sequence_1
    except NameError:
        parent = None

    if parent is not None:
        try:
            parr = np.asarray(parent, dtype=np.float64).ravel()
        except Exception:
            parr = None

        if parr is not None and parr.ndim == 1 and parr.size > 0:
            if parr.size <= _EVAL_LEN_LIMIT:
                if bool(np.all(np.isfinite(parr))):
                    clipped = np.clip(parr, 0.0, 1000.0)
                    try:
                        parent_sum = float(np.sum(clipped))
                    except Exception:
                        parent_sum = float("nan")
                    if np.isfinite(parent_sum) and parent_sum >= 0.01:
                        consider(clipped.tolist())

                # A constant of the same affordable length also attains the bound.
                consider(np.ones(int(parr.size), dtype=np.float64).tolist())
            else:
                # If the parent is too large to cheaply exact-score, a short
                # constant remains an optimal valid return.
                consider(np.ones(1024, dtype=np.float64).tolist())

    # Cheap exact-evaluator scan of constant sequences.  Mathematically all
    # attain 2; exact rounding can in principle select a marginally better float.
    try:
        if time.monotonic() < deadline:
            for n in (1, 2, 3, 4, 5, 7, 8, 16, 32, 64, 128, 256, 512, 1024, 2048):
                if time.monotonic() > deadline:
                    break
                consider(np.ones(n, dtype=np.float64).tolist())

        if time.monotonic() + 0.5 < deadline:
            heights = (
                0.1, 0.125, 0.2, 0.25, 0.3, 0.5, 0.6, 0.7, 0.75, 0.9,
                1.0, 1.1, 1.25, 1.5, 2.0, 4.0, 10.0, 16.0, 100.0, 1000.0
            )
            for n in (2, 3, 4, 5, 6, 7, 8, 12, 16, 24, 32, 48):
                if time.monotonic() > deadline:
                    break
                for h in heights:
                    if time.monotonic() > deadline:
                        break
                    consider([float(h)] * n)
    except Exception:
        pass

    # Ensure the returned object is a plain non-empty list of finite floats.
    try:
        best = [float(x) for x in best]
        if not best or any(not np.isfinite(x) for x in best):
            best = [1.0]
    except Exception:
        best = [1.0]

    # Final exact verification if there is safe time to do so.
    try:
        if len(best) <= _EVAL_LEN_LIMIT and time.monotonic() < start + max(
            0.0, total - 1.0
        ):
            if not np.isfinite(_safe_score(best)):
                best = [1.0]
    except Exception:
        best = [1.0]

    return best
