import cvxpy as cp
import numpy as np

def lp_iteration(a, M):
    """Solve: max sum(g) s.t. conv(a, g)[k] <= M for all k, g >= 0."""
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    
    constraints = [cp.sum(g) >= 0]  # trivial
    
    # conv(a, g) has length 2n-1
    # conv(a, g)[k] = sum_j a[j]*g[k-j] for valid j
    # We need this <= M for all k in [0, 2n-2]
    
    # The convolution constraint can be written as a matrix inequality:
    # T @ g <= M * ones(2n-1)
    # where T is (2n-1) x n, T[k, j] = a[j] if 0 <= k-j < n else 0
    
    # For efficiency, we can use the structure
    expr = cp.convolve(a, g)  # cvxpy supports this?
    # Actually, I'm not sure cvxpy has convolve. Let me construct the matrix.
    
    # Alternative: construct the constraint matrix explicitly
    # T is (2n-1) x n
    # T[k, j] = a[j] if max(0, k-n+1) <= j <= min(n-1, k) else 0
    
    # For n=4498, this matrix is 8995 x 4498, which is about 40M entries.
    # Most are zero, so sparse representation is essential.
    
    # Actually, each row has at most n nonzeros, and there are 2n-1 rows.
    # Total nonzeros: sum_k (min(n, k+1) - max(0, k-n+1) + 1) 
    # = sum_k (number of j with 0<=j<n and 0<=k-j<n)
    # = n*n (each pair (j, k-j) with 0<=j<n and 0<=k-j<n contributes to exactly one k)
    # Wait, that's not right. The total number of nonzeros in T is:
    # For each (k, j) pair where the entry is nonzero: 0<=j<n and 0<=k-j<n
    # = for each j in [0,n-1] and k in [j, j+n-1]: that's n*(2n-1) - ... 
    # Actually it's simpler: for each j, k ranges from j to j+n-1 (n values), so total = n*n.
    
    # So the sparse matrix has n^2 nonzeros. For n=4498, that's about 20M. That's a lot.
    # Memory: 20M * 8 bytes = 160MB for the values, plus indices. Might be tight.
    
    # Alternative: use the fact that T is banded (each row is a shifted version of a).
    # We can express the constraint as: for each k, sum_{j in range} a[j]*g[k-j] <= M
    # This is like a "moving dot product."
    
    # For the LP solver, we need to pass the constraint matrix. Let's use scipy's sparse.
    
    from scipy.sparse import csr_matrix
    import scipy.sparse as sp
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    T = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [T @ g <= M * np.ones(2*n-1)]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GLPK, verbose=False)
    
    if g.value is not None:
        return np.maximum(0, g.value)
    return None
```

import numpy as np
import time

def compute_conv_and_peak(a):
    """Compute convolution and find peak."""
    b = np.convolve(a, a)
    k_star = np.argmax(b)
    return b, k_star, b[k_star]

def grad_at_peak(a, k_star):
    """Compute subgradient of b[k_star] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(n - 1, k_star)
    for i in range(j_start, j_end + 1):
        grad[i] = 2.0 * a[k_star - i]
    return grad

def projected_grad_step(a, grad, t):
    """Take a projected gradient step."""
    a_new = a - t * grad
    a_new = np.maximum(0, a_new)
    s = a_new.sum()
    if s < 1e-30:
        return a  # fallback
    a_new /= s
    return a_new

    a = np.maximum(0, a)
    a /= a.sum()

    # But a[k_star - i] for i in [j_start, j_end] is a[max(0,k_star-n+1) ... min(n-1,k_star)] reversed
    # Let's just compute it directly
    for i in range(j_start, j_end + 1):
        a_new[i] -= t * 2.0 * a[k_star - i]
    a_new = np.maximum(0.0, a_new)
    s = a_new.sum()
    if s < 1e-30:
        return a
    a_new /= s
    return a_new
```

def _grad_step_vec(a, k_star, t):
    n = a.size
    # Index: for i in [j_start, j_end], k_star - i ranges from k_star-j_end to k_star-j_start
    j_start = max(0, k_star - n + 1)
    j_end = min(n - 1, k_star)
    
    a_new = a.copy()
    # The values a[k_star - i] for i in [j_start, j_end] are:
    # a[k_star - j_start], a[k_star - j_start - 1], ..., a[k_star - j_end]
    # = a[k_star - j_start : k_star - j_end - 1 : -1]
    # which is a reversed slice of a
    
    idx = np.arange(k_star - j_start, k_star - j_end - 1, -1)
    a_new[j_start:j_end+1] -= t * 2.0 * a[idx]
    
    a_new = np.maximum(0.0, a_new)
    s = a_new.sum()
    if s < 1e-30:
        return a
    a_new /= s
    return a_new

This is O(n) and vectorized.

Now, for the full optimization loop, I want to avoid recomputing the convolution in the line search. Instead, I'll use a "barzilai-borwein" style adaptive step size:
- After each step, compute the ratio of objective change to step size
- Adjust t accordingly

Or more simply: use a fixed t that works well empirically, and use patience-based restarts.

Let me think about what a "good" t is. At the start (uniform init), the peak is at the center (k* = n-1) and the gradient is roughly uniform (grad[i] ≈ 2/n for all i). The step a_new = a - t*2*a (approximately) = a*(1-2t). After normalization, a_new ≈ a (unchanged!). So the step does nothing for a uniform initial condition!

This is a problem. The gradient direction at the uniform point is proportional to a itself (since conv(uniform, uniform) is symmetric and the gradient at the center is proportional to a). So the projected gradient step at the uniform point is a no-op (after normalization).

I need a different initialization or a different direction. Options:
1. Start from a perturbed uniform (a + noise)
2. Use the LP direction (which gives a different direction than the gradient)
3. Use a "random direction" in the tangent space of the constraint set

Actually, the issue is that at the uniform point, ALL directions in the tangent space give the same first-order change in the peak (because the Hessian is what matters). So gradient descent is stuck at the uniform point to first order. I need second-order information or a different method.

This is a well-known issue: gradient descent on a non-convex function can be slow at "plateaus" where the gradient is small or proportional to the current point.

Solution: start from a non-uniform initialization. For example:
- a = (1 + 0.5 * sin(2πj/n)) / sum(...), which is nonnegative and non-uniform
- Or a random nonnegative vector

With a non-uniform init, the gradient will point in a useful direction.

Let me verify: if a[j] = 1 + cos(2πj/n), then the peak of conv(a,a) is NOT at the center (it's also large at lag n and lag 2n, but those are out of range for linear convolution... actually for linear convolution of length n, the peak should be near the center but modulated by the cosine structure).

Actually, for a "wavy" sequence, the autocorrelation will have a peak at lag 0 (center) and secondary peaks at lags related to the wavelength. The gradient at the center peak will point in a direction that reduces the center peak, which is useful.

OK, let me just start from a random initialization and use gradient descent. The random init will break the symmetry.

Let me also add a "momentum" term to help with convergence:
- v = β*v - t*grad
- a = a + v
- Project: a = max(0, a); a /= sum(a)

This is basically SGD with momentum.

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix


def _convolve_peak(a):
    """Compute convolution and return (peak_value, argmax_index)."""
    b = np.convolve(a, a)
    k = int(np.argmax(b))
    return b[k], k


def _grad_step(a, k_star, t, v=None, beta=0.9):
    """One momentum projected-gradient step. Returns (a_new, v_new)."""
    n = a.size
    j_start = max(0, k_star - n + 1)
    j_end = min(n - 1, k_star)
    
    # Compute gradient: grad[i] = 2*a[k_star - i] for i in [j_start, j_end]
    idx = np.arange(k_star - j_start, k_star - j_end - 1, -1)
    grad = np.zeros(n)
    grad[j_start:j_end + 1] = 2.0 * a[idx]
    
    if v is None:
        v_new = -t * grad
    else:
        v_new = beta * v - t * grad
    
    a_new = a + v_new
    a_new = np.maximum(0.0, a_new)
    s = a_new.sum()
    if s < 1e-30:
        return a, v
    a_new /= s
    return a_new, v_new


def _optimize_gradient(a0, deadline, rng, verbose=False):
    """Run momentum projected-gradient descent to minimize the peak."""
    n = a0.size
    a = np.maximum(0.0, a0.astype(np.float64))
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_a = a.copy()
    peak, _ = _convolve_peak(a)
    best_score = 2.0 * n * peak
    
    v = None
    t = 0.5
    patience = 0
    max_patience = 2000
    
    it = 0
    while time.monotonic() < deadline:
        peak, k_star = _convolve_peak(a)
        score = 2.0 * n * peak
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_patience:
            # Reset with perturbation from best
            a = best_a.copy()
            noise_scale = rng.choice([0.005, 0.01, 0.02, 0.05])
            a = np.maximum(0.0, a + noise_scale * rng.standard_normal(n))
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a /= s
            v = None
            t = 0.5
            patience = 0
            continue
        
        a, v = _grad_step(a, k_star, t, v)
        
        # Adapt step size: slowly increase, decrease if stuck
        it += 1
        if patience < 10:
            t = min(t * 1.01, 2.0)
        elif patience > 50:
            t = max(t * 0.95, 0.01)
        
        # Periodic: try a larger random perturbation
        if it % 5000 == 0 and patience > 500:
            a = best_a.copy()
            mask = rng.random(n) < 0.3
            a[mask] *= np.exp(rng.normal(0, 0.3, mask.sum()))
            a = np.maximum(0.0, a)
            s = a.sum()
            if s > 1e-15:
                a /= s
            v = None
            patience = 0
    
    return best_a, best_score


def _optimize_lp_refine(a0, deadline, rng):
    """LP-based refinement for smaller n. Uses the active-set style inner loop."""
    n = a0.size
    if n > 500:
        return _optimize_gradient(a0, deadline, rng)
    
    a = np.maximum(0.0, a0.astype(np.float64))
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_a = a.copy()
    peak, _ = _convolve_peak(a)
    best_score = 2.0 * n * peak
    
    # Build the convolution constraint matrix (sparse)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    # We'll rebuild this each iteration since a changes
    outer_iters = 0
    max_outer = 200
    
    while time.monotonic() < deadline and outer_iters < max_outer:
        outer_iters += 1
        M = np.convolve(a, a).max()
        
        # Build sparse matrix for current a
        rows = np.zeros(n * n, dtype=np.int32)
        cols = np.zeros(n * n, dtype=np.int32)
        vals_arr = np.zeros(n * n, dtype=np.float64)
        ptr = 0
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                rows[ptr] = k
                cols[ptr] = j
                vals_arr[ptr] = a[j]
                ptr += 1
        rows = rows[:ptr]
        cols = cols[:ptr]
        vals_arr = vals_arr[:ptr]
        T = csr_matrix((vals_arr, (rows, cols)), shape=(2 * n - 1, n))
        
        # Solve LP: max sum(g) s.t. T@g <= M, g >= 0
        g = cp.Variable(n, nonneg=True)
        try:
            prob = cp.Problem(cp.Maximize(cp.sum(g)), [T @ g <= M])
            for solver in [cp.GLPK, cp.CBC, cp.SCIP]:
                try:
                    prob.solve(solver=solver, verbose=False, time_limit=30)
                    if g.value is not None:
                        break
                except Exception:
                    continue
            
            if g.value is None:
                break
            
            g_val = np.maximum(0, g.value)
            s_g = g_val.sum()
            if s_g < 1e-10:
                break
            
            g_val /= s_g
            # Blend
            blah = 0.1
            a_new = (1 - blah) * a + blah * g_val
            a_new = np.maximum(0, a_new)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new /= s_new
            
            peak_new = np.convolve(a_new, a_new).max()
            score_new = 2.0 * n * peak_new
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            a = a_new
        except Exception:
            break
    
    # Continue with gradient from LP result
    remaining = deadline - time.monotonic()
    if remaining > 5:
        a, score = _optimize_gradient(best_a, deadline, rng)
        if score < best_score:
            best_a = a
            best_score = score
    
    return best_a, best_score


def _random_init(n, rng, style=0):
    """Generate random nonnegative initializations."""
    if style == 0:  # uniform + noise
        a = 1.0 + 0.5 * rng.standard_normal(n)
    elif style == 1:  # cosine wave
        a = 1.0 + 0.8 * np.cos(2 * np.pi * rng.integers(1, max(2, n // 2)) * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
    elif style == 2:  # exponential decay from random center
        center = rng.integers(0, n)
        d = np.abs(np.arange(n) - center)
        a = np.exp(-d / (n * 0.1)) + 0.1
    elif style == 3:  # block structure
        nblocks = rng.integers(2, min(10, n // 5) + 1)
        a = np.ones(n)
        for _ in range(nblocks):
            start = rng.integers(0, n)
            width = rng.integers(1, max(2, n // 10))
            a[start:start + width] = rng.uniform(0.3, 2.0)
    else:  # pure random
        a = rng.random(n)
    a = np.maximum(0.01, a)
    a /= a.sum()
    return a
