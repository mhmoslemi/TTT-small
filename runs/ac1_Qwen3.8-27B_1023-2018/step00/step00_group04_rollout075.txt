import numpy as np
import time


def _fft_convolve_square(a, N):
    """Compute b = convolve(a, a) via FFT, using pre-allocated size N."""
    n = len(a)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    b = np.fft.irfft(A * A, N)
    return b[:2 * n - 1]


def _fft_grad(a, b, tau, N):
    """Compute gradient of log-sum-exp(b/tau) w.r.t. a via FFT cross-correlation."""
    n = len(a)
    x = b / tau
    x -= x.max()
    ex = np.exp(x)
    w = ex / ex.sum()
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    w_pad = np.zeros(N, dtype=np.float64)
    w_pad[:2 * n - 1] = w
    A = np.fft.rfft(a_pad)
    W = np.fft.rfft(w_pad)
    c = np.fft.irfft(A * np.conj(W), N)[:n]
    return 2.0 * c


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s
