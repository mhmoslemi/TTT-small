def fast_conv(x):
    """Compute linear convolution using FFT."""
    n = len(x)
    size = 2*n - 1
    # Use next power of 2 for FFT
    fft_size = 1
    while fft_size < size:

    F = np.fft.rfft(x, fft_size)
    conv = np.fft.irfft(F * F, fft_size)[:size]
