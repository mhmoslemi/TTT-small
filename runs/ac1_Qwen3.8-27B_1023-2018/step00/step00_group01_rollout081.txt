def build_basis(n,d,rng):
   # DCT/DST deterministic + random lowpass
   idx=np.arange(n)
   cols=[]
   half=d//2
   for k in range(1, half+1):
      cols.append(np.cos(2*np.pi*k*idx/n))
      cols.append(np.sin(2*np.pi*k*idx/n))
   # trim/pad to d maybe random smoothed
   B=np.column_stack(cols[:d]) if len>=d else ...
   # random lowpass components to fill
   for j in range(max(0,d-len)):
      rr=rng.standard_normal(n)
      F=np.fft.rfft(rr)
      mask= len(F)
      keep = min(mask, 4*d)
      F[keep:]=0
      rr=np.fft.irfft(F,n)
      rr=rr-rr.mean()
      nr=np.linalg.norm(rr)
      if nr>0: cols.append(rr/nr)
   B=np.column_stack(cols)
   # orthonormalize
   Q,_=np.linalg.qr(B)  # shape n x min
   return Q[:, :d] if Q.shape[1]>=d else Q
