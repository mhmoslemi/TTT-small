    cols = []
    vals = []

```python
import numpy as np
import time
import math

def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    # Find λ such that sum max(0, v[i] - λ) = 1
    # Use bisection
    v_sorted = np.sort(v)[::-1]  # descending
    # The solution has v[i] - λ > 0 iff v[i] > λ
    # Let k be the number of positive components
    # Then sum_{i=1}^{k} (v[i] - λ) = 1, so λ = (sum_{i=1}^{k} v[i] - 1) / k
    
    # Find the right k
    cumsum = np.cumsum(v_sorted)
    # v_sorted[i] > (cumsum[i+1] - 1) / (i+1) for i = 0, ..., k-1
    # i.e., (i+1) * v_sorted[i] > cumsum[i+1] - 1
    # i.e., v_sorted[i] + 1/(i+1) > cumsum[i+1] / (i+1)
    
    rho = np.nonzero(v_sorted - (cumsum - 1.0) / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    k = rho[-1] + 1
    theta = (cumsum[k-1] - 1.0) / k
    return np.maximum(v - theta, 0.0)
```

Wait, the standard simplex projection algorithm:
Given v, find the closest point on {x >= 0, sum x = 1} in Euclidean distance.

The solution is x[i] = max(0, v[i] - τ) where τ is chosen so sum x = 1.

To find τ: sort v in descending order: v_(1) >= v_(2) >= ... >= v_(n).
The correct τ satisfies: for some k, v_(k) > τ >= v_(k+1), and:
sum_{i=1}^{k} (v_(i) - τ) = 1
⟹ k*v_(1) + ... + v_(k) - k*τ = 1
⟹ τ = (sum_{i=1}^{k} v_(i) - 1) / k

And k is the largest integer such that v_(k) > (sum_{i=1}^{k} v_(i) - 1) / k.

This can be found in O(n log n) time (for the sort) + O(n) time (for the scan).

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u > (cssv - 1.0) / np.arange(1, n + 1)
    k = np.nonzero(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    k = k[-1] + 1
    theta = (cssv[k-1] - 1.0) / k
    return np.maximum(v - theta, 0.0)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)

def optimize_gradient(a, n, eta0=0.1, max_iters=10000, deadline=None):
    """Gradient descent on log-sum-exp of convolution, with simplex projection."""
    a = a.copy()
    best = a.copy()
    best_obj = np.inf
    eta = eta0
    tau = 50.0  # smoothing parameter
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        b = np.convolve(a, a)
        # Log-sum-exp for numerical stability
        b_max = b.max()
        log_sum_exp = b_max + np.log(np.sum(np.exp(tau * (b - b_max))))
        obj = log_sum_exp / tau
        
        if obj < best_obj:
            best_obj = obj
            best = a.copy()
        
        # Gradient
        # Weights
        w = np.exp(tau * (b - b_max))
        w /= w.sum()
        
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] > 1e-15:
                j_min = max(0, k - n + 1)
                j_max = min(n - 1, k)
                js = np.arange(j_min, j_max + 1)
                g[js] += 2.0 * w[k] * a[k - js]
        
        # Step
        a_new = a - eta * g
        a_new = project_simplex(a_new)

        b_new = np.convolve(a_new, a_new)
        b_max_new = b_new.max()
        log_sum_exp_new = b_max_new + np.log(np.sum(np.exp(tau * (b_new - b_max_new))))

import numpy as np
import time

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    r = np.arange(1, n + 1)
    cond = u > (cv - 1.0) / r
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    k = k[-1] + 1
    theta = (cv[k - 1] - 1.0) / k
    return np.maximum(v - theta, 0.0)
