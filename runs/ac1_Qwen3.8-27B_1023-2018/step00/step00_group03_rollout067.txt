def _score_and_grad(a):
    """Return (score, grad) for a nonneg sequence a using the exact formula.
    grad is for the objective 2*n*max_k(b_k)/S^2 at the currently active max lag."""
    n = a.size
    b = np.convolve(a, a)
    k_star = int(np.argmax(b))
    max_b = float(b[k_star])
    S = float(a.sum())
    if S < 0.01:
        return np.inf, None
    # gradient of b_{k*} w.r.t a: grad_b[m] = 2*a[k*-m] when 0 <= k*-m < n
    grad_b = np.zeros(n)
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    if lo <= hi:
        idx = k_star - np.arange(lo, hi + 1)
        grad_b[lo:hi + 1] = 2.0 * a[idx]
    score = 2.0 * n * max_b / (S * S)
    grad = (2.0 * n / (S * S)) * grad_b - (4.0 * n * max_b / (S ** 3))
    return float(score), grad
