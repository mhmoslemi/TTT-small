def evaluate_sequence(sequence: list) -> float:
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)

    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

                cand[:boundary+1] *= np.exp(rng.normal(0, 0.2))

                cand = current.copy()
                cand[boundary+1:] *= np.exp(rng.normal(0, 0.2))

            f = rng.uniform(0.1, 0.4)
            m = max(1, int(n * f))
            c = np.exp(rng.normal(np.log(2), 0.3))
            for side in [0, 1]:

                if side == 0:
                    cand[:m] = c
                else:

    if np.isfinite(final_score) and final_score < 1000:
        result = [float(x) for x in best]
        assert len(result) > 0
        assert all(np.isfinite(x) for x in result)
