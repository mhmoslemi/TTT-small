            m = parent_len
        else:
            remaining = max(0.0, float(budget_s) - 10.0)
            # Dense direct convolution of an m-vector is O(m^2); use a
            # conservative operation estimate and keep memory/time bounded.
            m = int(math.sqrt(max(1.0, remaining * 2.0e7)))
            if m > parent_len:
                m = parent_len
            if m > 20000:
                m = 20000
            if m < 1:
                m = 1
    if m is None:
        m = 1

    # Evaluate the constant candidate only if there is realistic headroom.
    # The known 5582-length case is very small; this estimate is conservative.
    estimated_eval_s = 0.02 * (float(m) / 5582.0) ** 2
    if m >= 1 and time.monotonic() < deadline - max(1.0, estimated_eval_s):
        try:
            cand = [1.0] * m
            c_score = float(evaluate_sequence(cand))

            if math.isfinite(c_score):
                if c_score < best_score - 1e-14:
                    best = cand
                    best_score = c_score
                else:
                    # On a numerical tie near the exact lower bound, prefer a
                    # constant vector matching the supplied parent length.
                    if (
                        abs(c_score - best_score) <= 1e-14
                        and best_score >= 2.0 - 1e-14
                        and m == parent_len
                        and m >= len(best)
                    ):
                        best = cand
                        best_score = c_score
        except Exception:
            pass

    # Final authoritative validation.  If anything invalid happens, strictly
    # return a known valid finite nonnegative singleton.
    try:
        final_score = float(evaluate_sequence(best))
        if math.isfinite(final_score):
            best_score = final_score
        else:
            best = [1.0]
    except Exception:
        best = [1.0]

    # Ensure the returned object is exactly a plain nonempty list of finite
    # real coefficients, clipped to the evaluator's allowed range.
    out = []
    for x in best:
        try:
            y = float(x)
        except Exception:
            y = 1.0
        if not math.isfinite(y):
            y = 1.0
        if y < 0.0:
            y = 0.0
        if y > 1000.0:
            y = 1000.0
        out.append(y)

    if not out:
        out = [1.0]

    return out
