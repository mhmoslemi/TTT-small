import numpy as np
import time

def compute_convolution(f):
    """Compute full convolution of f with itself using FFT."""
    n = len(f)
    N = 1 << (2*n - 1).bit_length()  # next power of 2 >= 2n-1
    F = np.fft.rfft(f, N)
    conv = np.fft.irfft(F * F, N)
    return conv[:2*n-1]  # take only valid part

def ac1_score(f):
    """Compute the AC1 score of sequence f (already nonnegative)."""
    n = len(f)
    b = compute_convolution(f)
    max_b = max(b)
    sum_a = np.sum(f)
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def gradient_of_max_conv(f):
    """Compute subgradient of max(b) with respect to f."""
    n = len(f)
    b = compute_convolution(f)
    max_b = b.max()
    # Find all locations where b[k] is close to max
    tol = 1e-10 * max_b + 1e-15
    K_star = np.where(b >= max_b - tol)[0]
    # Subgradient: average of 2f[k-i] over k in K_star
    grad = np.zeros(n)
    for k in K_star:
        # d(conv_k)/df_i = 2*f[k-i] if 0 <= k-i < n
        valid_i = np.arange(max(0, k-n+1), min(n, k+1))
        valid_kminus_i = k - valid_i
        grad[valid_i] += 2.0 * f[valid_kminus_i]
    grad /= len(K_star)
    return max_b, grad, K_star

        if np.isfinite(score) and score < best_score:

            best_score = score
            patience = 0
            # Decrease learning rate slightly on improvement
            lr *= 0.999

            patience += 1
            lr *= 0.95
