def lp_refine(a, M, K=200, max_rounds=10, deadline=None):
    n = len(a)
    # Start with top K constraints
    c = np.convolve(a, a)
    top_k = np.argsort(c)[-K:]
    
    b = np.ones(n)  # initial guess
    for round in range(max_rounds):
        if deadline and time.monotonic() > deadline:
            break
        # Build constraint matrix for selected k's
        # T[k, j] = a[k-j] if 0 <= k-j < n else 0
        A_selected = np.array([
            [a[k-j] if 0 <= k-j < n else 0 for j in range(n)]
            for k in top_k
        ])
        # This is K x n
        
        # Solve: max 1^T b s.t. A_selected b <= M*1, b >= 0
        # Equivalent: min -1^T b s.t. A_selected b <= M*1, -b <= 0
        from scipy.optimize import linprog
        c_obj = -np.ones(n)
        A_ub = np.vstack([A_selected, -np.eye(n)])
        b_ub = np.concatenate([M*np.ones(K), np.zeros(n)])
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, method='highs')
        if not res.success:
            break
        b = res.x
        
        # Check if all constraints are satisfied
        c_new = np.convolve(a, b)
        violations = np.where(c_new > M + 1e-8)[0]
        if len(violations) == 0:
            break
        # Add violated constraints
        top_k = np.concatenate([top_k, violations])
        top_k = np.unique(top_k)
        if len(top_k) > n * 2:  # cap on constraints
            top_k = np.argsort(c_new)[-n*2:]
    
    return b

    else:
        parent = np.ones(256)

        return [float(x) for x in parent]
    
    # Normalize parent
    parent_norm = parent / parent.sum()
    best = parent_norm.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    N = 1
    while N < 3 * n:
        N <<= 1
    
    def compute_score(a):
        """Compute the score efficiently."""
        c = np.fft.ifft(np.fft.fft(a, N) ** 2).real[:2*n-1]
        sum_a = a.sum()
        if sum_a < 1e-15:
            return np.inf
        return float(2 * n * c.max() / (sum_a ** 2))
    
    def smooth_gradient_descent(a, deadline, beta_start=100.0, max_iters=50000):
        """Smoothed gradient descent on the AC1 objective."""
        a = a / a.sum()  # normalize
        n_local = a.size
        N_local = 1
        while N_local < 3 * n_local:
            N_local <<= 1
        
        beta = beta_start
        lr = 1.0 / (beta * n_local)
        v = np.zeros(n_local)
        momentum = 0.85

        best_sc = compute_score(a)

            p = exp_bc / exp_bc.sum()
