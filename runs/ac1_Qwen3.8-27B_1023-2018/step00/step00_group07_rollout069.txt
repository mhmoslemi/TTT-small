```
def project_simplex(v):
    v = np.asarray(v, float)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum x == 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if rho.size == 0:
        return (1.0 / n) * np.ones(n, dtype=np.float64)
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _autocorr(x):
    n = x.size
    L = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(x, L)
    ac = np.fft.irfft(F * F.conj(), L)[: 2 * n - 1]
    return ac


def _score_fast(x):
    s = x.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * x.size * float(ac_max(x)) / (s * s)


def ac_max(x):
    return float(np.max(_autocorr(x)))


def _central_binomial(n, S=1.0):
    y = np.empty(n, dtype=np.float64)
    y[0] = 1.0
    for k in range(1, n):
        y[k] = y[k - 1] * (2 * k - 1) / (2 * k)
    y *= S / y.sum()
    return y


def _grad_lag(x, r):
    """d (x*x)_r / d x  = 2 * x_{r-j} where 0<=r-j<n else 0."""
    n = x.size
    g = np.zeros(n, dtype=np.float64)
    lo = max(0, r - (n - 1))
    hi = min(n - 1, r)
    idx = r - np.arange(lo, hi + 1)
    g[lo:hi + 1] = 2.0 * x[idx]
    return g


def _polish_mirror(f, max_iters, deadline, eta0=0.035, log_every=200):
    """Mirror (exponentiated) descent on the simplex reducing max autocorr."""
    n = f.size
    x = np.asarray(f, dtype=np.float64).copy()
    x = _project_simplex(x)
    best = x.copy()
    best_sc = _score_fast(x)
    u = np.log(x) - np.log(x).max()
    eta = eta0
    stall = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        ac = _autocorr(x)
        # pick among the top few hot lags to escape plateaus
        k = min(5, ac.size)
        top = np.argpartition(ac, -k)[-k:]
        r = int(top[np.argmax(ac[top])])
        g = _grad_lag(x, r)
        u -= eta * g
        # stabilise log-weights
        if not np.isfinite(u).all():
            u = np.log(x) - np.log(x).max()
        nx = np.exp(u - u.max())
        ns = nx.sum()
        if not np.isfinite(ns) or ns <= 0:
            u = np.log(x) - np.log(x).max()
            nx = np.exp(u - u.max()); ns = nx.sum()
        nx = _project_simplex(nx / ns)
        sc = _score_fast(nx)
        if np.isfinite(sc) and sc < best_sc - 1e-7:
            best = nx.copy()
            best_sc = sc
            stall = 0
        else:
            stall += 1
        x = nx
        if stall > 40:
            eta *= 0.7
            stall = 0
            u = np.log(x) - np.log(x).max()
        eta = max(eta, 1e-4)
    return best, best_sc


def _sota_lp(f, M, max_iters, deadline):
    """Short peak of the active-set LP_iters SoTA scheme (reduced n only)."""
    import cvxpy as cp
    from scipy import sparse
    n = f.size
    f = np.asarray(f, dtype=np.float64).copy()
    f = f / f.sum()
    M = max(M, ac_max(f))
    solvers = ["ECOS", "CLARABEL", "SCIP", "GLOP", "GLPK", "CBC"]
    chosen = None
    for s in solvers:
        try:
            cp.settings.default_solver = cp.get_solver(s)
            chosen = s
            break
        except Exception:
            continue
    if chosen is None:
        return f
    c = np.ones(n)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        # build A (2n-1, n)  A[r,j] = f[r-j] if 0<=r-j<n
        nlen = 2 * n - 1
        rows = np.empty(nlen * n, dtype=np.int32)
        cols = np.empty(nlen * n, dtype=np.int32)
        vals = np.empty(nlen * n, dtype=np.float64)
        p = 0
        for r in range(nlen):
            lo = max(0, r - (n - 1))
            hi = min(n - 1, r)
            idx = np.arange(lo, hi + 1)
            cnt = idx.size
            rows[p:p + cnt] = r
            cols[p:p + cnt] = idx
            vals[p:p + cnt] = f[r - idx]
            p += cnt
        A = sparse.csc_matrix((vals[:p], (rows[:p], cols[:p])), shape=(nlen, n))
        g = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(c @ g), [A @ g <= M, g >= 0])
        try:
            prob.solve(solver=chosen)
        except Exception:
            break
        gv = g.value
        if gv is None or not np.isfinite(gv).all():
            break
        sg = gv.sum()
        if sg < 1e-8:
            break
        gn = gv / sg
        improved = False
        for t in (0.5, 0.3, 0.15, 0.05):
            fn = (1.0 - t) * f + t * gn
            fn = np.maximum(fn, 0.0)
            if fn.sum() < 0.01:

            Mn = ac_max(fn)

                f = fn.copy()
                M = Mn
                improved = True
                break
        if not improved:
            # shrink M slowly to keep the LP informative
            M *= 0.99
    return f
```
