                a_s = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
                a_s = np.maximum(a_s, 0.0)
                if a_s.sum() > 0:
                    a_s /= a_s.sum()
                    inits.append(a_s)
        # V-shape (high edges, low center)
        a_v = np.abs(x - n / 2.0) / (n / 2.0)
        a_v = 0.3 + 0.7 * a_v
        a_v /= a_v.sum()
        inits.append(a_v)
        # Periodic blocks
        for block in [4, 8, 16]:
            a_b = np.ones(n)
            a_b[::block] = 0.0
            if a_b.sum() > 0:
                a_b /= a_b.sum()
                inits.append(a_b)

        time_per_init = min(120.0, remaining_budget / max(1, len(inits)))
        for a_init in inits:
            if time.monotonic() > deadline or (deadline - time.monotonic()) < 30:
                break
            a_t, s_t = optimize(a_init, tau_start=10.0,
                                max_steps=80000,
                                t_limit=min(deadline, time.monotonic() + time_per_init))
            if s_t < best_score:
                best_a, best_score = a_t, s_t

    # --- Phase 3: Local polish from best ---
    remaining_budget = deadline - time.monotonic()
    if remaining_budget > 60:
        n_polish = min(2000, int(remaining_budget / 2))
        for _ in range(n_polish):
            if time.monotonic() > deadline:
                break
            a_t = best_a * (1.0 + 0.02 * rng.standard_normal(n))
            a_t = np.maximum(a_t, 0.0)
            st = a_t.sum()
            if st < 1e-10:
                continue
            a_t /= st
            s_t = fast_score(a_t)
            if s_t < best_score:
                a_t2, s_t2 = optimize(a_t, tau_start=50.0,
                                      max_steps=20000,
                                      t_limit=min(deadline, time.monotonic() + 30))
                if s_t2 < best_score:
                    best_a, best_score = a_t2, s_t2

    # --- Final: ensure validity and return ---
    best_a = np.maximum(best_a, 0.0)
    best_a = np.minimum(best_a, 1000.0)
    if best_a.sum() < 0.01:
        best_a = np.ones(n) / n

    final_list = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    if not np.isfinite(final_score):
        # Fallback to parent
        final_list = [float(x) for x in parent]
        final_score = float(evaluate_sequence(final_list))

    return final_list
