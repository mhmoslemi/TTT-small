import numpy as np
import time

def fast_conv_max(a, N):
    """Compute max of a*a using FFT. N is padded FFT length."""
    fa = np.fft.rfft(a, N)
    ff = fa * fa  # |FA|^2 for real a
    conv = np.fft.irfft(ff, N)
    return conv[:2*len(a)-1].max()

def fast_score(a, N):

    S = a.sum()
    if S < 0.01:
        return float('inf')
    M = fast_conv_max(a, N)
    return 2*n*M/S**2
