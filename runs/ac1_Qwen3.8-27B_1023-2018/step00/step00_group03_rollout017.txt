def _softmax_obj(params, n, tau):
    # params: unbounded, a = softmax(params)
    logits = params - np.max(params)  # for numerical stability
    exps = np.exp(logits)
    a = exps / exps.sum()
    c = np.convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    weights = np.exp(shifted)
    weights_sum = weights.sum()
    smooth_max = c_max + tau * np.log(weights_sum)
    obj = 2.0 * n * smooth_max  # since sum a = 1
    return obj

def _conv(a):
    return np.convolve(a, a)

def _smoothed_max(c, tau):
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    log_sum = np.log(np.sum(np.exp(shifted)))
    return c_max + tau * log_sum

def _obj_and_grad(a, n, tau):
    """Compute smoothed objective and gradient w.r.t. a (nonneg space)."""
    c = _conv(a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exps = np.exp(shifted)
    wsum = exps.sum()
    w = exps / wsum  # weights for each lag
    smooth_max = c_max + tau * np.log(wsum)
    S = np.sum(a)
    obj = 2.0 * n * smooth_max / (S * S)
    # Gradient of smooth_max w.r.t. a_j = 2 * sum_k w_k * a_{k-j}
    grad_sm = np.zeros(n)
    for j in range(n):
        k_lo = max(0, j)
        k_hi = min(2 * n - 2, j + n - 1)
        # a_{k-j} for k in [k_lo, k_hi]
        idx = k_hi - j  # length of range
        grad_sm[j] = 2.0 * np.sum(w[k_lo:k_hi + 1] * a[:idx + 1])
    # d obj / d a_j = 2n * (grad_sm_j / S^2 - 2*smooth_max / S^3)
    grad_obj = 2.0 * n * (grad_sm / (S * S) - 2.0 * smooth_max / (S ** 3))
    return obj, grad_obj

def _optimize_streaming(n, p0, tau, max_iter=500, tol=1e-12):
    """Optimize a (nonneg, normalized) of length n using projected gradient in log-space."""
    # Work in log-space to maintain nonnegativity
    p = np.log(np.maximum(p0, 1e-15))
    best_p = p.copy()
    best_val = 2.0 * n * np.max(_conv(np.exp(p) / np.sum(np.exp(p))))
    
    for it in range(max_iter):
        a = np.exp(p - np.max(p))
        a = a / a.sum()
        c = _conv(a)
        c_max = np.max(c)
        shifted = (c - c_max) / tau
        exps = np.exp(shifted)
        wsum = exps.sum()
        w = exps / wsum
        smooth_max = c_max + tau * np.log(wsum)
        
        # grad of smooth_max w.r.t. a
        grad_sm = np.zeros(n)
        for j in range(n):
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            idx = k_hi - j
            grad_sm[j] = 2.0 * np.dot(w[k_lo:k_hi + 1], a[:idx + 1])
        
        # Objective (normalized): 2n * smooth_max / sum(a)^2, but sum(a)=1 here
        val = 2.0 * n * smooth_max
        # grad w.r.t. a: 2n * grad_sm (since S=1)
        grad_a = 2.0 * n * grad_sm
        # Remove mean to preserve normalization (tangent plane)
        grad_a = grad_a - np.sum(grad_a) / n  # project to simplex tangent
        # grad w.r.t. p: grad_a * a
        grad_p = grad_a * a
        
        # Adaptive step
        if it < 5:
            lr = 1.0
        else:
            lr = 2.0 / (np.linalg.norm(grad_p) + 1e-8)
            lr = min(lr, 1.0)
        
        p_new = p - lr * grad_p
        # Check if improved
        a_new = np.exp(p_new - np.max(p_new))
        a_new = a_new / a_new.sum()
        val_new = 2.0 * n * np.max(_conv(a_new))
        
        if val_new < val:
            p = p_new
            if val_new < best_val:
                best_val = val_new
                best_p = p_new.copy()
            if val_new < val * (1 - tol):
                pass  # good step
        else:
            # Halve step
            p_new = p - 0.5 * lr * grad_p
            a_new = np.exp(p_new - np.max(p_new))
            a_new = a_new / a_new.sum()
            val_new = 2.0 * n * np.max(_conv(a_new))
            if val_new < val:
                p = p_new
                if val_new < best_val:
                    best_val = val_new
                    best_p = p_new.copy()
            else:
                break  # stuck

            break  # degenerate
    
    a_best = np.exp(best_p - np.max(best_p))
    a_best = a_best / a_best.sum()
    return a_best, 2.0 * n * np.max(_conv(a_best))

def _sota_lp_step(f, M, max_time=30.0):
    """One LP step from SOTA: max sum(g) s.t. g>=0, f*g<=M componentwise."""
    n = len(f)
    t0 = time.monotonic()
    c = np.convolve(f, f)
    # M = max(c) or we pass it in
    # Build constraint matrix: (f*g)_k <= M for k=0..2n-2
    # (f*g)_k = sum_j f_j * g_{k-j} where 0<=k-j<n
    # As a matrix: A[k,j] = f_{k-j} if 0<=k-j<=n-1 else 0
    # This is a (2n-1) x n matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        A[k_lo:k_hi + 1, j] = f[n - 1 - (k_hi - j):n - 1 - (k_lo - j) - 1:-1] if False else \
                              f[max(0, n - 1 - (k_hi - j)):max(1, n - (k_lo - j))][::-1]
    # Actually let me just build it directly
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        for i in range(n):
            k = i + j  # (f*g)_k gets contribution f_j * g_i? No...
            # (f*g)_k = sum_i f_i * g_{k-i}
            # So for variable g_j, it appears in (f*g)_{j+i} with coefficient f_i, for i=0..n-1
            # (f*g)_{j+i} += f_i * g_j, for i = 0..n-1, if j+i <= 2n-2
            pass
    # Redo: (f*g)_k = sum_{i=0}^{n-1} f_i * g_{k-i} (where g_i=0 for i outside [0,n-1])
    # Variable g_j appears in (f*g)_k when k-j is in [0,n-1], i.e., k in [j, j+n-1]
    # Coefficient is f_{k-j}
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        for i in range(n):
            k = j + i
            A[k, j] = f[i]
    
    # LP: max -1^T g  (i.e. minimize -sum(g)) s.t. A g <= M * 1, g >= 0
    c_obj = -np.ones(n)
    c_max = np.max(c)
    M_eff = c_max  # use current max as bound
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M_eff),
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': max_time})
    if res.status == 0:
        g = res.x
        if np.sum(g) > 0:
            g = g / np.sum(g) * np.sum(f)  # normalize to match sum
            return g
    return None
