import time

from scipy import optimize, sparse

def _conv_full(a, b=None):
    """Linear convolution a*b (b defaults to a) via FFT, O(n log n)."""
    n = a.size
    if b is None:
        b = a
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    fa = np.fft.fft(a, N)
    if b is a:
        fb = fa
    else:
        fb = np.fft.fft(b, N)
    c = np.fft.ifft(fa * fb).real[:m]
    # remove tiny negative noise from round-off
    c[c < 0.0] = 0.0
    return c


def _resample_to(arr, n):
    """Mass/shape-preserving resample of a nonnegative vector to length n."""
    arr = np.asarray(arr, dtype=np.float64).ravel()
    arr = arr[np.isfinite(arr)]
    arr = np.clip(arr, 0.0, None)
    if arr.size == 0 or arr.sum() <= 0.0:
        return np.ones(n)
    arr = arr / arr.sum()
    if arr.size == n:
        return arr
    if arr.size < n:
        arr = np.resize(arr, n)
        return arr / arr.sum()
    # equal-mass inverse-CDF resampling
    cum = np.cumsum(arr) / arr.sum()
    tgt = (np.arange(n) + 0.5) / n
    out = np.interp(tgt, cum, arr)
    out = np.clip(out, EPS, None)
    return out / out.sum()

    """Sparse A (2n-1 x n) with (A @ g == np.convolve(f, g))."""

        f = np.clip(0.75 + 0.5 * np.cos(np.pi * i), EPS, None)
        f = f / f.sum()

    f = np.clip(f, EPS, None)
    f = f / f.sum()

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    t_candidates = (1.0, 0.66, 0.4, 0.22, 0.11)

    while left() > 2.0:
        conv = _conv_full(f, f)
        M = float(conv.max())
        if not np.isfinite(M) or M <= 0.0:
            break

        g = _dc_lp_step(f, M)
        if g is None:
            break

        accepted = False
        for t in t_candidates:
            cand = (1.0 - t) * f + t * g
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score - 1e-10:
                f = np.clip(cand, 0.0, None)
                f = f / f.sum()
                best = f.copy()
                best_score = sc
                accepted = True
                break
        if not accepted:
            t = 0.03
            cand = (1.0 - t) * f + t * g
            cand = np.clip(cand, 0.0, None)
            f = cand / cand.sum()
            sc = float(evaluate_sequence(f.tolist()))
            if sc < best_score:
                best = f.copy()
                best_score = sc

    if left() > 25.0:
        best, bs = _pool_refine(f, left)
        if np.isfinite(bs) and bs < best_score:
            best_score = bs

    out = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    out = out.tolist()
    vs = float(evaluate_sequence(out))
    if not np.isfinite(vs):
        out = [1.0 / n] * n
    return out
