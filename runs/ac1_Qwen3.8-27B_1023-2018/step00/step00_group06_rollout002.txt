import time
import numpy as np


def _safe_exact_score(sequence):
    """Return a finite exact evaluator score, or inf on failure/invalid."""
    try:
        value = float(evaluate_sequence(sequence))
    except Exception:
        return float("inf")
    return value if np.isfinite(value) else float("inf")


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    """
    Return a globally optimal sequence for the authoritative evaluator.

    For any finite list a accepted by the evaluator:
      - the evaluator clips it to nonnegative values,
      - n = len(a),
      - S = sum(a) > 0,
      - max_b >= (a*a)[0] = sum(a_i^2).

    By Cauchy-Schwarz,
      sum(a_i^2) >= S^2 / n.

    Therefore
      2*n*max_b / S^2 >= 2*n*(S^2/n)/S^2 = 2.

    Equality is attained by any positive constant vector.  Hence the exact
    minimum possible evaluator score is 2, and no value below 2 can be reached
    by any valid nonnegative sequence.  The program uses a compact exact
    minimizer and only optionally checks another exact minimizer of the
    supplied length if doing so is computationally cheap.
    """
    budget = max(0.0, float(budget_s))
    deadline = time.monotonic() + max(0.0, budget - 0.25)

    # Compact exact minimizer: length 1, value 1.
    best = [1.0]
    best_score = float("inf")
    if time.monotonic() < deadline:
        best_score = _safe_exact_score(best)

    # If the supplied parent construction exists and has a practical length,
    # test a uniform vector of that length.  This uses the supplied length
    # directly without copying its entries into the source program.
    parent_len = None
    try:
        parent_len = int(len(height_sequence_1))
    except Exception:
        parent_len = None

    if (
        parent_len is not None
        and 1 <= parent_len <= 8000
        and time.monotonic() < deadline
    ):
        candidate = np.ones(parent_len, dtype=np.float64).tolist()
        candidate_score = _safe_exact_score(candidate)
        if candidate_score < best_score - 1e-12:
            best = candidate
            best_score = candidate_score

    # Final validation.  If anything unexpected happens, fall back to a valid
    # exact minimizer.
    if time.monotonic() < deadline:
        final_score = _safe_exact_score(best)
        if not np.isfinite(final_score):
            best = [1.0]

    return [float(x) for x in best]
