def _smooth_max_and_grad(a, beta):
    """Compute log-sum-exp approximation to max(b) and its gradient (a, beta given)."""
    n = a.size
    b = np.convolve(a, a, mode='full')
    max_b = b.max()
    shifted = beta * (b - max_b)
    exp_b = np.exp(shifted)

        a7[j] = 1.0 + 0.5 * np.cos(2.0 * np.pi * (j % p_prime) / p_prime)
    a7 = np.maximum(a7, 0.01)
    inits.append(_normalize(a7))

    global_best = parent.copy()
    global_best_score = parent_sc

    # Optimize each initialization with time budget
    n_inits = len(inits)
    time_per_init = (hard_deadline - time.monotonic()) / (n_inits + 2)

    for idx in range(n_inits):
        if time.monotonic() > hard_deadline - 10.0:
            break
        init_budget = min(hard_deadline - 10.0, time.monotonic() + time_per_init * 1.5)
        try:
            best_a, best_sc = _optimize_one_init(inits[idx], n, init_budget, rng, idx)
            if np.isfinite(best_sc) and best_sc < global_best_score:
                global_best = best_a
                global_best_score = best_sc
        except Exception:
            pass

    # Final: do another round of optimization from global_best if time remains
    if time.monotonic() < hard_deadline - 20.0:
        try:
            best_a, best_sc = _optimize_one_init(global_best, n, hard_deadline - 5.0, rng, 99)
            if np.isfinite(best_sc) and best_sc < global_best_score:
                global_best = best_a
                global_best_score = best_sc
        except Exception:
            pass

    # Final validation with the exact evaluator
    final_check = float(evaluate_sequence(global_best.tolist()))
    if np.isfinite(final_check):
        global_best_score = final_check

    # Ensure valid output
    result = np.clip(global_best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(n, dtype=np.float64)

    return [float(x) for x in result]
