def _as_list_candidate(seq):

import time
import numpy as np


def _clean_candidate(seq, max_len=None):
    """Return a plain list of finite float coefficients in [0, 1000]."""
    try:
        arr = np.asarray(seq, dtype=np.float64)
        arr = np.ravel(arr)
    except Exception:
        return None

    if max_len is not None and arr.size > max_len:
        return None
    if arr.size == 0 or not np.all(np.isfinite(arr)):
        return None

    arr = np.clip(arr, 0.0, 1000.0)

    try:
        if float(np.sum(arr)) < 0.01:
            return None
    except Exception:
        return None

    return [float(x) for x in arr]


def _safe_score(seq):
    """Exact evaluator score, with exceptions mapped to infinity."""
    try:
        s = float(evaluate_sequence(seq))
    except Exception:
        return np.inf
    return s if np.isfinite(s) else np.inf


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    """
    Return a globally optimal sequence for the supplied evaluator.

    For the authoritative evaluator, after clipping all coefficients are
    nonnegative and max_b includes the zero lag:
        max_b >= sum_i a_i^2 >= (sum_i a_i)^2 / len(a)
    hence
        2 * n * max_b / (sum a)^2 >= 2.
    Equality is attained by any constant nonzero sequence.  Therefore the
    read-only objective cannot go below 2; the program validates the supplied
    parent when affordably possible and otherwise returns a constant optimum.
    """
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0

    # Leave one strict headroom so we are not killed after final work.
    hard_deadline = t0 + max(0.0, budget) - 2.0

    best = None
    best_score = np.inf

    def consider(seq, max_len=None):
        nonlocal best, best_score
        cand = _clean_candidate(seq, max_len)
        if cand is None:
            return
        score = _safe_score(cand)
        if score < best_score:
            best_score = score
            best = cand

    # Dense exact evaluation is O(L^2).  The announced parent has length 4806,
    # so this is affordable; guard against unexpectedly large parents.
