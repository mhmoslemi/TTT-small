import numpy as np
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (int(n) - 1).bit_length()


def _clean_sequence(a, n_expected=None):
    try:
        a = np.asarray(a, dtype=np.float64)
    except Exception:
        a = np.ones(1 if n_expected is None else n_expected, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        a = np.ones(1 if n_expected is None else n_expected, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    if not np.all(np.isfinite(a)):
        a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
        a = np.clip(a, 0.0, 1000.0)
    if a.sum() <= 0.01:
        a = np.ones(a.size, dtype=np.float64)
    return a


def _fft_score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    sm = float(a.sum())
    if sm <= 0.01:
        return float("inf")
    n = a.size
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)[:L]
    b = np.where(b > 0.0, b, 0.0)
    return float(2.0 * n * b.max() / (sm * sm))


def _sequence_intensity(a, alpha):
    """Create a smooth positive sequence by normalizing exp(alpha * shape)."""
    a = np.asarray(a, dtype=np.float64)
    a = a - a.mean()
    z = alpha * a
    z -= z.max()
    w = np.exp(z)
    if w.sum() <= 0:
        w = np.ones_like(a)
    return w / w.sum()


def _candidate_pool(n, rng, parent):
    cands = []

    p = _clean_sequence(parent, n).copy()
    if p.size != n:
        p = np.resize(p, n)
    cands.append(p)
    cands.append(np.ones(n, dtype=np.float64))

    x = np.linspace(0.0, 1.0, n)

    shapes = []
    shapes.append(1.0 + 1.35 * (x - 0.5))
    shapes.append(1.0 + 1.35 * (0.5 - x))
    t = 2.0 * x - 1.0
    shapes.append(1.0 + 2.3 * t * t - 0.7)
    shapes.append(1.0 + 2.3 * np.abs(t) - 0.85)
    shapes.append(1.0 + 1.7 * np.sin(2.0 * np.pi * x))
    shapes.append(1.0 + 1.7 * np.cos(2.0 * np.pi * x))
    shapes.append(1.0 + 1.5 * np.sin(4.0 * np.pi * x) + 0.7 * np.cos(2.0 * np.pi * x))
    shapes.append(1.0 + 0.9 * (x - x ** 2))
    shapes.append(1.0 + 1.1 * (x * x - x))
    shapes.append(1.0 + 1.4 * np.sin(6.28318 * x + 0.65) + 0.6 * np.cos(12.56636 * x + 1.31))

    for s in shapes:
        s = np.clip(s, 1e-6, None)
        s = s / s.sum()
        cands.append(s / s.mean())

    for q in (16, 32, 64, 128):
        d = 1.0 + 0.7 * rng.standard_normal(q)
        d = np.clip(d, 0.05, None)
        d = d / d.mean()
        idx = np.minimum((np.arange(n) * q) // max(1, n), q - 1)
        cands.append(d[idx] / d.mean())

        d = 1.0 + 1.25 * rng.standard_normal(q)
        d = np.clip(d, 0.03, None)
        d = d / d.mean()
        idx = np.minimum((np.arange(n) * q) // max(1, n), q - 1)
        cands.append(d[idx] / d.mean())

    for amp in (0.8, 1.4, 2.1):
        u = np.zeros(n, dtype=np.float64)
        max_mode = min(n // 2, 18)
        for m in range(1, max_mode + 1):
            a1 = rng.standard_normal() / (m ** 0.35)
            a2 = rng.standard_normal() / (m ** 0.35)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            u += amp * (a1 * np.cos(2.0 * np.pi * m * x / n + ph) +
                        a2 * np.sin(2.0 * np.pi * m * x / n + ph))
        w = _sequence_intensity(x, 1.0 / max(1.0, np.max(np.abs(u)) + 1e-12))
        cands.append(w)

    # A few coarse random block signs, kept positive and normalized.
    for q in (4, 8, 13, 21):
        d = 1.0 + 0.85 * rng.choice([-1.0, 1.0], size=q)
        d = np.clip(d, 0.05, None)
        d = d / d.mean()
        idx = np.minimum((np.arange(n) * q) // max(1, n), q - 1)
        cands.append(d[idx] / d.mean())

    # Deterministic three-scale pattern sometimes helps finite length.
    for c in (0.55, 0.85, 1.25):
        base = np.ones(n, dtype=np.float64)
        base[n // 4: 3 * n // 4] *= c
        base[0:n // 8] *= 1.0 + 0.25 * (1.0 - c)
        base[-n // 8:] *= 1.0 + 0.25 * (1.0 - c)
        base = np.clip(base, 1e-5, None)
        cands.append(base / base.mean())

    # Remove exact duplicates roughly by hashing rounded values.
    out = []
    seen = set()
    for c in cands:
        c = np.clip(np.asarray(c, dtype=np.float64), 1e-8, None)
        if c.sum() <= 0:
            continue
        c /= c.sum()
        key = tuple(np.round(c[::max(1, c.size // 16)], 9))
        if key in seen:
            continue
        seen.add(key)
        out.append(c)
    if not out:
        out.append(np.full(n, 1.0 / n, dtype=np.float64))
    return out


def _optimize_sequence(a, deadline, rng):
    a = _clean_sequence(a)
    n = a.size

    # Use the scale matching the evaluator's homogeneous objective.
    S = float(np.sqrt(2.0 * n))
    a = S * a / a.sum()

    L = 2 * n - 1
    N = _next_pow2(L)
    rfft = np.fft.rfft
    irfft = np.fft.irfft

    def from_u(u):
        u = np.asarray(u, dtype=np.float64)
        u = u - np.max(u)
        e = np.exp(np.clip(u, -50.0, 0.0))
        z = e.sum()
        if z <= 0 or not np.isfinite(z):
            return np.full(n, S / n, dtype=np.float64)
        return S * e / z

    u = np.log(np.maximum(a, 1e-300))
    u = np.clip(u, -45.0, 0.0)
    u = u - u.mean()

    best = a.copy()
    best_score = _fft_score(a)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    lr = 0.30
    b1, b2, eps = 0.90, 0.999, 1e-8
    best_t = 0

    # Number of active lags for softmax.  Broad at first, sharper later.
    while time.monotonic() < deadline:
        a = from_u(u)
        fa = rfft(a, N)
        b = irfft(fa * fa, N)[:L]
        b = np.where(b > 0.0, b, 0.0)
        peak = float(b.max())
        if not np.isfinite(peak) or peak <= 0.0:
            break

        score = 2.0 * n * peak / (S * S)
        if not np.isfinite(score):
            break

        if score < best_score - 1e-13:
            best = a.copy()
            best_score = score
            best_t = t
            no_imp = 0
        else:
            no_imp = t - best_t

        if no_imp > 3500:
            lr *= 0.55
            if lr < 1e-5:
                break
            best_t = t

        progress = min(1.0, 0.05 * t / 3000.0) if t > 0 else 0.0
        width = max(1, min(L, int(L * (0.45 - 0.30 * progress))))
        idx = np.argpartition(-b, width - 1)[:width]
        vals = b[idx]
        vmax = float(vals.max())
        scale = max(1e-8, vmax * (0.45 - 0.30 * progress))
        tw = np.exp(np.clip((vals - vmax) / scale, -50.0, 0.0))
        z = tw.sum()
        if z <= 0 or not np.isfinite(z):
            tw = np.ones_like(vals)
            z = twidth if False else float(vals.size)
        w = np.zeros(L, dtype=np.float64)
        w[idx] = tw / z

        fw = rfft(w, N)
        g = 2.0 * irfft(np.conj(fa) * fw, N)[:n]
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)

        dot = float(np.dot(a, g))
        gu = (a / S) * (g - dot / S)

        t += 1
        b1t = b1 ** t
        m = b1 * m + (1.0 - b1) * gu

        if t % 250 == 0:
            b2t = b2 ** t
            v = b2t * v + (1.0 - b2t) * (gu * gu)
        else:
            b2t = b2 ** t
            v = b2 * v + (1.0 - b2) * (gu * gu)

        mh = m / (1.0 - b1t)
        vh = v / (1.0 - b2t)
        step = lr * mh / (np.sqrt(vh) + eps)
        step = np.nan_to_num(step, nan=0.0, posinf=0.0, neginf=0.0)
        u = u - np.clip(step, -0.22, 0.22)
        u = u - np.mean(u)
        u = np.clip(u, -48.0, 48.0)

        if not np.all(np.isfinite(u)):
            break

    best = from_u(u)
    best = np.clip(best, 0.0, 1000.0)
    return best


def _refine_short(a, deadline, rng):
    """Strict time-bounded final pass to make sure returned object is exact-bestsofar."""
    try:
        a = _optimize_sequence(a, deadline, rng)
    except Exception:
        a = _clean_sequence(a)
    return a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = max(1.0, float(budget_s))
    t0 = time.monotonic()
    hard_deadline = t0 + max(10.0, budget_s - 9.0)
    rng = np.random.default_rng(seed)

    n = 4806
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            p = _clean_sequence(p)
            if p.sum() > 0.01:
                parent = p
                n = p.size
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(n, dtype=np.float64)
    parent = _clean_sequence(parent, n)

    # Initial authoritative score.  This is the guaranteed fallback.
    best_seq = parent.copy()
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_score = float("inf")
    if not np.isfinite(best_score):
        best_seq = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    def maybe_update(a):
        nonlocal best_seq, best_score
        try:
            a = _clean_sequence(a, n)
            if a.size != n:
                return
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_seq = a.copy()
                best_score = sc
        except Exception:
            pass

    # Build a pool of plausible initial shapes.  The pool is cheap and is
    # intended only to seed the iterative optimizer.
    try:
        pool = _candidate_pool(n, rng, parent)
    except Exception:
        pool = [parent.copy(), np.ones(n, dtype=np.float64)]

    # Allocate budget adaptively: parent gets the largest share, then others.
    candidates = [parent.copy()] + pool
    # Put current best parent first, then several diverse starts.
    if len(candidates) > 1:
        candidates = candidates[:max(2, min(12, len(candidates)))]

    # Time shares (fractions of the safe budget).  They need not sum to 1.
    shares = [0.32, 0.12, 0.10, 0.09, 0.08, 0.07, 0.06, 0.05, 0.04, 0.04, 0.03, 0.03]
    safe_budget = max(5.0, float(budget_s) - 20.0)

    for i, cand in enumerate(candidates):
        if time.monotonic() > hard_deadline - 12.0:
            break
        share = safe_budget * min(0.35, shares[i] if i < len(shares) else 0.03)
        # For very large n, reduce time per candidate automatically.
        if n > 8000:
            share = min(share, 180.0)
        lim = min(hard_deadline - 12.0, time.monotonic() + max(1.0, share))
        if lim <= time.monotonic():
            continue
        try:
            seq = _optimize_sequence(cand, lim, rng)
        except Exception:
            seq = cand
        maybe_update(seq)

    # Final polishing/re-validation on the globally best sequence.
    if time.monotonic() < hard_deadline - 15.0:
        rem = max(10.0, (hard_deadline - 15.0) - time.monotonic())
        # Polish the best, then a fresh random perturbation of it if time remains.
        try:
            seq = _optimize_sequence(best_seq, min(hard_deadline - 15.0, time.monotonic() + rem * 0.55), rng)
        except Exception:
            seq = best_seq
        maybe_update(seq)

        if time.monotonic() < hard_deadline - 15.0:
            try:
                local_rng = np.random.default_rng(seed + 101)
                perturb = best_seq * np.exp(0.10 * local_rng.standard_normal(best_seq.size))
                perturb = _clean_sequence(perturb, n)
                seq = _optimize_sequence(perturb, min(hard_deadline - 15.0, time.monotonic() + max(5.0, rem * 0.30)), local_rng)
            except Exception:
                seq = best_seq
            maybe_update(seq)

    # A couple of exact low-cost local perturbations as a safety net.
    if time.monotonic() < hard_deadline - 15.0:
        local_rng = np.random.default_rng(seed + 777)
        for _ in range(60):
            if time.monotonic() >= hard_deadline - 15.0:
                break
            try:
                c = best_seq.copy()
                w = int(max(2, min(n, local_rng.integers(1, max(2, n // 80)))))
                s = int(local_rng.integers(0, max(1, n - w + 1)))
                e = min(n, s + w)
                factor = float(np.exp(0.20 * local_rng.standard_normal(e - s)))
                c[s:e] *= factor
                c = _clean_sequence(c, n)
                maybe_update(c)
            except Exception:
                break
        # Symmetrization experiment: sometimes makes finite-length best better.
        try:
            sym = 0.5 * (best_seq + best_seq[::-1])
            maybe_update(sym)
        except Exception:
            pass

    # Verify and normalize for final return.
    final = _clean_sequence(best_seq, n)
    if final.size == 0 or not np.all(np.isfinite(final)):
        final = np.ones(n, dtype=np.float64)

    # One last exact check; if it somehow fails, fall back to constant.
    try:
        sc = float(evaluate_sequence(final.tolist()))
        if not np.isfinite(sc):
            final = np.ones(n, dtype=np.float64)
            sc = float(evaluate_sequence(final.tolist()))
    except Exception:
        final = np.ones(n, dtype=np.float64)
        try:
            sc = float(evaluate_sequence(final.tolist()))
        except Exception:
            sc = float("inf")

    if not np.isfinite(sc):
        final = np.ones(n, dtype=np.float64)

    return [float(x) for x in final]
