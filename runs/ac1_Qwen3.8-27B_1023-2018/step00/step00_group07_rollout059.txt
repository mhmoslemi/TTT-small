def _fft_conv(a, fft_size, cutoff):
    """Full linear self-convolution via FFT."""
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:cutoff]


def _grad_at_lag(a, k, n):
    """Subgradient of conv[k] w.r.t. a. O(n) memory and time."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    m1 = np.arange(lo, hi + 1)
    if m1.size:

    lo2 = max(0, k - hi)
    hi2 = min(n - 1, k - lo)
    if lo2 <= hi2:
        m2 = np.arange(lo2, hi2 + 1)
        grad[m2] += a[k - m2]
    return grad

        a /= a.sum()
        sc_cur = quick_score(a)
        local_best = a.copy()
        local_best_sc = sc_cur
        lr = 0.3
        mom = np.zeros(n)
        beta_m = 0.8
        stagnant = 0
        topK = 5
        it = 0
        t_it0 = time.monotonic()

        while it < max_iter:
            if time.monotonic() > deadline:
                break
            # every few hundred iters check wall-time budget for this sub-init
            if it % 400 == 400 and (time.monotonic() - t_it0) > budget_s * iter_budget_frac:
                break

            conv = _fft_conv(a, fft_size, cutoff)
            mb = conv.max()
            if mb <= 0 or not np.isfinite(mb):
                break
            sc_cur = 2.0 * n * mb / (a.sum() ** 2)

            # top-K lags
            if topK < cutoff:
                top_idx = np.argpartition(conv, -topK)[-topK:]
            else:
                top_idx = np.arange(cutoff)
            top_vals = conv[top_idx]
            order = np.argsort(top_vals)[::-1]
            top_idx = top_idx[order]
            top_vals = top_vals[order]

            w = top_vals / top_vals.sum()
            grad = np.zeros(n)
            for i in range(min(topK, len(top_idx))):
                grad += w[i] * _grad_at_lag(a, int(top_idx[i]), n)

            mom = beta_m * mom - lr * grad
            a_new = a + mom
            a_new = np.maximum(0.0, a_new)
            sn = a_new.sum()
            if sn < 1e-10:
                a_new = a.copy()
            else:
                a_new /= sn

            sc_new = quick_score(a_new)
            if np.isfinite(sc_new) and sc_new < sc_cur:
                a = a_new
                sc_cur = sc_new
                stagnant = 0
                if sc_new < local_best_sc:
                    local_best = a.copy()
                    local_best_sc = sc_new
            else:
                stagnant += 1
                if stagnant > 40:
                    lr *= 0.75
                    if lr < 0.01:
                        # anneal: add noise proportional to current scale
                        noise_scale = max(0.01, 0.05 * np.mean(a))
                        a = a + noise_scale * rng.standard_normal(n)
                        a = np.maximum(0.0, a)
                        sn2 = a.sum()
                        if sn2 > 0:
                            a /= sn2
                        mom *= 0.0
                        lr = 0.3
                        stagnant = 0
            it += 1

        return local_best, local_best_sc

    # ---------- run optimization from each init ----------
    n_inits = len(inits)
    per_init_time_frac = 0.5 / max(1, n_inits)  # leave 50% for polish

    for idx, init in enumerate(inits):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 1.0:
            break
        max_iter = 15000
        a_opt, sc_opt = optimize(init, max_iter, per_init_time_frac)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            # verify with exact evaluator
            exact = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best = a_opt.copy()
                best_score = exact

    # ---------- final polish: targeted local search ----------
    remaining = deadline - time.monotonic()
    if remaining > 5.0 and best is not None:
        a_pol = best.copy()
        if a_pol.sum() > 1e-10:
            a_pol /= a_pol.sum()
        polish_sc = quick_score(a_pol)
        lr_p = 0.005
        width = max(1, n // 300)
        for _ in range(100000):
            if time.monotonic() > deadline - 3.0:
                break
            # random block perturbation
            w = rng.integers(max(1, width // 4), width + 1)
            s0 = rng.integers(0, max(1, n - w))
            a_try = a_pol.copy()
            a_try[s0:s0 + w] *= np.exp(
                rng.normal(0.0, 0.08 * (1.0 + lr_p))
            )
            # occasional global nudge
            if rng.random() < 0.1:
                a_try *= np.exp(rng.normal(0.0, 0.02, n))
            a_try = np.maximum(0.0, a_try)
            st = a_try.sum()
            if st < 1e-10:
                continue
            a_try /= st
            sc_try = quick_score(a_try)
            if np.isfinite(sc_try) and sc_try < polish_sc:
                a_pol = a_try
                polish_sc = sc_try
                lr_p = min(lr_p * 1.05, 0.2)
                if polish_sc < best_score:
                    best = a_pol.copy()
                    best_score = polish_sc
            else:
                lr_p = max(lr_p * 0.97, 0.002)

    # ---------- exact final validation ----------
    if best is not None:
        fin = float(evaluate_sequence(best.tolist()))
        if np.isfinite(fin):
            best_score = fin
        else:
            best = parent.copy()
            best_score = float(evaluate_sequence(best.tolist()))

    if best is None or not np.isfinite(best_score) or best.size == 0:
        best = np.ones(n)
        best_score = float(evaluate_sequence(best.tolist()))

    # ensure output is clean
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
