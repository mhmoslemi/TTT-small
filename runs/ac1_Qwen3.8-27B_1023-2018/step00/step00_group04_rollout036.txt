```python
def _solve_lp(a, M, n):
    """Solve the LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    # Build constraint matrix A where A[k,j] = a[j] if 0 <= k-j <= n-1
    # Row k, column j: nonzero when k-j is in [0, n-1], i.e., j in [k-n+1, k]
    # Value: a[j]
    
    # For efficiency, build in CSR or LIL format
    # Column j has values a[j] in rows j through j+n-1
    # Total nonzeros: sum_j min(n, 2n-1-j) ≈ n^2/2 + n(n-1)/2 ... 
    
    # Let's build using the column structure
    indptr = np.zeros(n+1, dtype=np.int32)
    for j in range(n):
        row_start = j
        row_end = min(j + n, 2*n - 1)
        indptr[j+1] = indptr[j] + (row_end - row_start)
    
    # This is actually not quite right for CSR. Let me use COO.
    row_indices = []
    col_indices = []
    values = []

        for k in range(j, min(j + n, 2*n - 1)):
            row_indices.append(k)
            col_indices.append(j)
            values.append(a[j])

    return None
```

```python
        if j_start < j_end:

            cols.extend(js.tolist())
            vals.extend(a[j_start:j_end].tolist())
    
    return csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
```

```python
import numpy as np
import time
from scipy.sparse import csc_matrix
from scipy.optimize import linprog


def _build_conv_matrix_structure(n):
    """Precompute COO index structure for the convolution constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
    return (np.array(rows, dtype=np.int32),
            np.array(cols, dtype=np.int32))


def _solve_lp(a, M, n, row_idx, col_idx):
    """Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    vals = a[col_idx]
    A = csc_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, method='highs',
                  bounds=[(0, None)] * n)
    if res.success:
        return np.maximum(res.x, 0.0)
    return None
```
