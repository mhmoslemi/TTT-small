        A[k, j_start:j_end] = a[k - j_end + 1:k - j_start + 1]
    
    # LP: max sum(g) s.t. A @ g <= M*1, g >= 0
    # In linprog form: min -sum(g) s.t. A @ g <= M*1, g >= 0
    c = -np.ones(n)
    b_lp = np.full(conv_len, M)
    bounds = [(0, None)] * n
    
    try:
        from scipy.optimize import linprog as lp
        result = lp(c, A_ub=A, b_ub=b_lp, bounds=bounds, method='highs')
        if result.success:
            g = result.x
            g_sum = np.sum(g)
            if g_sum > 1e-10:
                g = g / g_sum
                # Blend
                t = 0.5
                a_new = (1 - t) * a + t * g
                a_new = np.clip(a_new, 0, None)
                a_new = a_new / np.sum(a_new)
                candidate = a_new * S0
                score = float(evaluate_sequence(candidate.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = candidate.copy()
                    best_score = score
                    return (a_new, best, best_score)
    except Exception:
        pass
    
    return None

def _fft_convolve_sq(a, fft_n):
    """Compute linear convolution a*a using FFT."""
    a_fft = np.fft.rfft(a, n=fft_n)
    b = np.fft.irfft(a_fft * a_fft, n=fft_n)
    return b, a_fft


def _subgradient_update(a, a_fft, fft_n, n, eta, rng):
    """One multiplicative subgradient step. Returns updated a and new a_fft."""
    conv_len = 2 * n - 1
    b = np.fft.irfft(a_fft * a_fft, n=fft_n)[:conv_len]
    M = float(np.max(b))
    if M <= 1e-15:
        return a, a_fft, 0.0
    
    # Find near-argmax positions (top 20% by value, threshold at 70% of max)
    thresh = 0.7 * M
    active = np.where(b >= thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    # Subgradient: s_i = (1/|K|) * sum_{k in K} a_{k-i} (clamped to valid range)
    # This is equivalent to convolution of indicator with a
    indicator = np.zeros(conv_len)
    indicator[active] = 1.0 / len(active)
    ind_fft = np.fft.rfft(indicator, n=fft_n)
    s = np.fft.irfft(ind_fft * a_fft, n=fft_n)[:n]

    a_sum = float(np.sum(a_new))

    a_new = a_new / a_sum
    a_new_fft = np.fft.rfft(a_new, n=fft_n)
