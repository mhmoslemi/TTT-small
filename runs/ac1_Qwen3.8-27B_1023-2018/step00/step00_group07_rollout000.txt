import time
import math
import numpy as np

# For every nonnegative sequence accepted by the authoritative evaluator,
# the zero-lag autocorrelation is sum(a_i^2).  By Cauchy-Schwarz,
#     sum(a_i^2) >= (sum(a_i)^2 / n).
# Hence evaluate_sequence is mathematically at least 2, and a constant
# sequence attains equality in exact arithmetic.  The search below verifies
# the exact evaluator and keeps any numerically slightly lower candidate.


def _clean_candidate(candidate):
    """Convert to a valid Python list of finite floats in [0, 1000]."""
    try:
        out = []
        for v in candidate:
            x = float(v)
            if not math.isfinite(x):
                return None
            if x < 0.0:
                x = 0.0
            elif x > 1000.0:
                x = 1000.0
            out.append(x)
    except Exception:
        return None
    if not out:
        return None
    return out


def _exact_score(seq, horizon):
    """Call the authoritative evaluator, respecting a wall-clock horizon."""
    try:
        if time.monotonic() > horizon:
            return math.inf
        val = float(evaluate_sequence(seq))
        if not math.isfinite(val):
            return math.inf
        return val
    except Exception:
        return math.inf


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    start = time.monotonic()

    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0

    if not math.isfinite(budget) or budget <= 0.0:
        return [1.0]

    # Leave wall-clock headroom for final validation and return.
    reserve = max(0.05, min(5.0, budget * 0.02))
    deadline = start + max(0.0, budget - reserve)

    # O(n^2) exact convolution is capped.  Since a constant sequence is
    # optimal for any length, smaller lengths are safe under tight budgets.
    if budget < 15.0:
        max_target_n = 256
        max_parent_eval_n = 256
    elif budget < 60.0:
        max_target_n = 2048
        max_parent_eval_n = 2048
    else:
        max_target_n = 6733
        max_parent_eval_n = 10000

    best = [1.0]
    best_score = math.inf

    def consider(candidate):
        nonlocal best, best_score
        if time.monotonic() > deadline:
            return
        seq = _clean_candidate(candidate)
        if seq is None:
            return
        score = _exact_score(seq, deadline)
        if score < best_score:
            best_score = score
            best = seq

    # Use the supplied construction if present, but avoid forcing an
    # unnecessarily large exact convolution under a tight budget.
    parent = globals().get("height_sequence_1", None)
    base_n = 0
    parent_arr = None

    if parent is not None:
        try:
            base_n = len(parent)

                base_n = int(np.size(parent))

        if base_n > 0:
            if base_n <= max_parent_eval_n:
                try:
