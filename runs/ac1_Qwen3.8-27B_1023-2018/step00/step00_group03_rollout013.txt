    from scipy.linalg import toeplitz
    # First column: A_conv[0,0]=a[0], A_conv[1,0]=a[1], ..., A_conv[n-1,0]=a[n-1], A_conv[n,0]=0, ...

    # Actually it's an anti-Toeplitz or convolution matrix
    # Let me just build it directly for clarity, or use scipy's toeplitz

    # Row k has a[i] at position i where 0 <= k-i < n, i.e., k-n+1 <= i <= k
    # So row 0: a[0] at col 0
    # Row 1: a[0] at col 1, a[1] at col 0
    # Row 2: a[0] at col 2, a[1] at col 1, a[2] at col 0
    # ...
    # Row n-1: a[0] at col n-1, a[1] at col n-2, ..., a[n-1] at col 0
    # Row n: a[1] at col n-1, a[2] at col n-2, ..., a[n-1] at col 1
    # ...
    # Row 2n-2: a[n-1] at col n-1
    
    # This is the standard convolution matrix. Let me build it.
    A_conv = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        A_conv[k, i_min:i_max+1] = a[i_min:i_max+1][::-1]

    #             b >= 0
    
    c = -np.ones(n)  # minimize -sum(b)
    A_ub = A_conv  # (2n-1) x n
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

import time


def propose_candidate(seed=42, budget_s=1300, **kwargs):

    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 5.0)

    def _remain():
        return max(0.0, deadline - time.monotonic())

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad parent")
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    S0 = float(parent.sum())
    if S0 < 1e-10:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = parent / S0
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / a.sum()

    # --- helpers ---
    def quick_obj(v):
        c = np.convolve(v, v)
        return 2.0 * n * float(c.max())

    def proj_simplex(v):
        v = np.array(v, dtype=np.float64, copy=True)
        nv = len(v)
        if nv == 1:
            v[0] = 1.0
            return v
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        # find rho
        cond = u * np.arange(1, nv + 1) > (cssv - 1.0)
        idx = np.where(cond)[0]
        if idx.size == 0:
            return np.ones(nv, dtype=np.float64) / nv
        rho = idx[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        out = np.maximum(v - theta, 0.0)
        s = out.sum()
        if s > 0:
            out /= s
        else:
            out = np.ones(nv, dtype=np.float64) / nv
        return out

    def subgrad_step(v, eta, top_k=1):
        conv = np.convolve(v, v)
        if top_k <= 1:
            ki = int(conv.argmax())
            lags = np.array([ki - (n - 1)], dtype=int)
            wts = np.array([1.0])
        else:
            top_idx = np.argpartition(conv, -top_k)[-top_k:][::-1]
            mx = float(conv[top_idx[0]])
            if mx < 1e-18:
                mx = 1.0
            wts = np.exp(conv[top_idx] / (0.1 * mx))
            wts /= wts.sum()
            lags = top_idx - (n - 1)
        g = np.zeros(n, dtype=np.float64)
        for lag, wk in zip(lags, wts):
            lag = int(lag)
            if lag == 0:
                g += 2.0 * wk * v
            elif lag > 0:
                g[:n - lag] += 2.0 * wk * v[lag:]
                g[lag:] += 2.0 * wk * v[:n - lag]
            else:
                ll = -lag
                g[ll:] += 2.0 * wk * v[:n - ll]
                g[:n - ll] += 2.0 * wk * v[ll:]
        v_new = v - eta * g
        return proj_simplex(v_new)

    def solve_lp(a_cur, M):
        """LP: max 1'b s.t. A_conv b <= M, b>=0. Returns normalised b or None."""
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = a_cur
        from scipy.optimize import linprog
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0.0, None)] * n,
                          method='highs',
                          options={'time_limit': 4})
            if res.success and res.x is not None:
                b = np.maximum(res.x, 0.0)
                bs = b.sum()
                if bs > 1e-14:
                    return b / bs
        except Exception:
            pass
        return None

    p2_t = 0.50 * float(budget_s)
    p2_start = time.monotonic()
    a = best_a.copy()
    t_blend = 0.14
    lp_calls = 0
    while (time.monotonic() - p2_start) < p2_t and _remain() > 15.0:
        conv = np.convolve(a, a)
        M = float(conv.max())

            break
        b_star = solve_lp(a, M)

        a_new = (1.0 - t_blend) * a + t_blend * b_star
