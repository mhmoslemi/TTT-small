    def normalize(v, m):
        v = np.asarray(v, dtype=np.float64)
        v = np.where(np.isfinite(v), v, 0.0)
        v = np.maximum(v, 0.0)
        s = float(v.sum())
        if not np.isfinite(s) or s < 1.0e-15:
            v = np.ones(m, dtype=np.float64)
            s = float(m)
        return v / s

    def to_list(v):
        v = normalize(v, int(np.size(v)))
        v = np.minimum(v, 1000.0)
        return [float(x) for x in v]

    def safe_perturb(v, m):
        noise = np.exp(rng.normal(0.0, 0.12, m))
        w = v * noise
        if m > 2:
            ramp = 0.05 * (np.arange(m, dtype=np.float64) / float(m - 1)) * float(v.mean())
            w = w + ramp
        w = np.where(np.isfinite(w), w, 0.0)
        w = np.maximum(w, 0.0)
        s = float(w.sum())
        if not np.isfinite(s) or s < 1.0e-15:
            return normalize(v, m)
        return w / s

    def optimize_local(x0, m, max_iters, deadline, l2_frac=0.55):
        x = normalize(x0, m)
        L = 2 * m - 1
        Nc = _pow2_at_least(L)
        Ns = _pow2_at_least(3 * m - 2)

        def conv_local(v):
            V = rfft(v, Nc)
            c = irfft(V * V, Nc).real
            return c[:L]

        def cross_local(w, v):
            W = rfft(w, Ns)
            Y = rfft(v[::-1], Ns)
            out = irfft(W * Y, Ns).real
            return out[m - 1:2 * m - 1]

        c = conv_local(x)
        cmax = float(c.max())
        if cmax <= 0.0 or not np.isfinite(cmax):
            x = normalize(np.linspace(0.1, 2.0, m, dtype=np.float64), m)
            c = conv_local(x)
            cmax = float(c.max())

        score = 2.0 * m * cmax / (float(x.sum()) ** 2)
        G = float(np.sum(c * c))
        best = x.copy()
        best_score = score
        best_G = G

        eta = 0.7
        max_eta = 2.5
        rejects = 0
        min_iters = max(100, int(max_iters * 0.05))
        l2_iters = max(min_iters, int(max_iters * l2_frac))
        gamma_frac = 0.02
        it = 0

        while it < max_iters:
            if time.monotonic() > deadline:
                break

            if it < l2_iters:
                w = c
            else:
                cmax2 = float(c.max())
                if (it - l2_iters) % 2000 == 0:
                    gamma_frac = max(0.003, gamma_frac * 0.9)
                top_k = int(0.02 * L)
                top_k = min(L, max(8, top_k))
                if top_k == L:
                    idx = np.arange(L)
                else:
                    idx = np.argpartition(c, -top_k)[-top_k:]
                vals = c[idx]
                spread = max(cmax2 * gamma_frac, float(vals.min()) * 0.25 + 1.0e-30, 1.0e-30)
                base = np.full(L, 1.0e-12, dtype=np.float64)
                base[idx] = np.exp((vals - vals.max()) / spread)
                sw = float(base.sum())
                if sw <= 0.0 or not np.isfinite(sw):
                    base = np.zeros(L, dtype=np.float64)
                    base[int(np.argmax(c))] = 1.0
                    sw = 1.0
                w = base / sw

            d = cross_local(w, x)
            if not np.all(np.isfinite(d)):
                break

            dd = d - float(d.mean())
            scale = float(np.abs(dd).max())
            ok = scale > 1.0e-30
            cand = None
            if ok:
                z = -eta * dd / scale
                z = np.clip(z, -max_eta, max_eta)
                z = z - float(z.mean())
                cand = x * np.exp(z)
                cand = np.where(np.isfinite(cand), cand, 0.0)
                cand = np.maximum(cand, 0.0)
                cs = float(cand.sum())
                if not np.isfinite(cs) or cs < 1.0e-15:
                    ok = False
                    cand = None
                else:
                    cand = cand / cs

            if not ok:
                rejects += 1
                eta = max(0.02, eta * 0.7)
                if rejects > 200:
                    x = safe_perturb(x, m)
                    c = conv_local(x)
                    score = 2.0 * m * float(c.max())
                    G = float(np.sum(c * c))
                    rejects = 0
                    eta = 0.7
                it += 1
                continue

            c2 = conv_local(cand)
            cmax3 = float(c2.max())
            if cmax3 <= 0.0 or not np.isfinite(cmax3):
                rejects += 1
                eta = max(0.02, eta * 0.7)
                it += 1
                continue

            score2 = 2.0 * m * cmax3
            G2 = float(np.sum(c2 * c2))

            improve = False
            if it < l2_iters:
                if G2 < G * (1.0 - 1.0e-11):
                    improve = True
                elif score2 < score * (1.0 - 1.0e-9):
                    improve = True
            else:
                if score2 < score * (1.0 - 1.0e-9):
                    improve = True
                elif G2 < G * (1.0 - 1.0e-11):
                    improve = True

            if improve:
                x, c, score, G = cand, c2, score2, G2
                if score < best_score:
                    best = x.copy()
                    best_score = score
                    best_G = G
                rejects = 0
                eta = min(max_eta, eta * 1.02)
            else:
                rejects += 1
                eta = max(0.02, eta * 0.85)
                if rejects > 300:
                    x = safe_perturb(x, m)
                    c = conv_local(x)
                    score = 2.0 * m * float(c.max())
                    G = float(np.sum(c * c))
                    rejects = 0
                    eta = 0.7

            it += 1

        if best is None or best.size != m or not np.all(np.isfinite(best)):
            best = normalize(x, m)
        else:
            best = normalize(best, m)
        return best, best_score

    def expand_block(y, m_b):
        y = normalize(y, m_b)
        x = np.empty(n, dtype=np.float64)
        for r in range(m_b):
            st = int(r * n / m_b)
            en = int((r + 1) * n / m_b)
            cnt = en - st
            if cnt > 0:
                x[st:en] = y[r] / float(cnt)
        return normalize(x, n)

    def make_fourier_seed(m):
        base = np.ones(m, dtype=np.float64)
        jj = np.arange(m, dtype=np.float64)
        total_amp = 0.75
        for k in range(1, 6):
            freq = int(rng.integers(1, max(2, m // 20)))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            amp = total_amp / float(k + 1) * float(rng.uniform(0.4, 1.0))
            base += amp * np.sin(2.0 * np.pi * freq * jj / float(m) + phase)
        base = np.where(base > 0.0, base, 1.0e-6)
        return normalize(base, m)

    def make_antimirror_seed(m, a=0.55):
        vals = np.ones(m, dtype=np.float64)
        half = m // 2
        if half > 0:
            left = np.arange(half)
            right = m - 1 - left
            flip = rng.random(half) < 0.5
            vals[left] = np.where(flip, 1.0 - a, 1.0 + a)
            vals[right] = np.where(flip, 1.0 + a, 1.0 - a)
        if m % 2 == 1:
            vals[half] = 1.0
        return normalize(vals, m)

    def make_spectral_seed(m, iterations=300):
        N = _pow2_at_least(2 * m - 1)
        L = 2 * m - 1
        q = np.ones(L, dtype=np.float64) / float(L)
        Q = np.fft.fft(q, N)
        A = np.sqrt(np.abs(Q))
        amax = float(A.max())
        if amax > 0.0:
            A = A + 1.0e-4 * amax

        k = np.arange(1, N // 2)

        def sym(P):
            P[0] = P[0].real
            N2 = N // 2
            P[N2] = P[N2].real
            if N2 > 1:
                pk = 0.5 * (P[k] + np.conj(P[N - k]))
                P[k] = pk
                P[N - k] = np.conj(pk)
            return P

        P = A * np.exp(0.5j * np.angle(Q))
        sym(P)
        p = np.fft.ifft(P).real[:m]
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if s <= 0.0:
            p = np.ones(m, dtype=np.float64)
        else:
            p = p / s
        prev = p.copy()

        for _ in range(int(iterations)):
            P = np.fft.fft(p, N)
            mag = np.abs(P)
            safe = np.where(mag > 1.0e-300, mag, 1.0e-300)
            P = P * (A / safe)
            sym(P)
            p2 = np.fft.ifft(P).real[:m]
            p2 = np.maximum(p2, 0.0)
            s2 = float(p2.sum())
            if s2 <= 0.0 or not np.isfinite(s2):
                break
            p2 = p2 / s2
            p = 0.5 * p + 0.5 * p2
            p[~np.isfinite(p)] = 0.0
            p = np.maximum(p, 0.0)
            s = float(p.sum())
            if s > 0.0:
                p = p / s
            else:
                p = prev
        return normalize(p, m)

    # Baseline candidate from the supplied parent.
    best_list = to_list(normalize(parent_arr, n))
    best_exact = np.inf
    if time.monotonic() < hard - 2.0:
        best_exact = evaluate_sequence(best_list)
    if not np.isfinite(best_exact):
        best_list = to_list(np.ones(n, dtype=np.float64))
        if time.monotonic() < hard - 1.0:
            best_exact = evaluate_sequence(best_list)

    full_cands = []

    # Optional spectral seed, often already close to a flat convolution.
    if hard - t_start > 90.0:
        try:
            spec = make_spectral_seed(n, iterations=280)
            if spec.size == n and np.all(np.isfinite(spec)):
                full_cands.append(spec)
        except Exception:
            pass

    # Optional coarse block optimization, expanded to full length.
    if hard - t_start > 220.0:
        m_b = 512 if n >= 1024 else max(64, min(256, n))
        if m_b < 4:
            m_b = 4
        jj = np.arange(m_b, dtype=np.float64)
        blk_seeds = []
        blk_seeds.append(np.ones(m_b, dtype=np.float64))
        blk_seeds.append(jj + 1.0)
        blk_seeds.append(m_b - jj + 1.0)
        blk_seeds.append(np.clip(1.0 + 0.6 * np.sin(2.0 * np.pi * jj / m_b), 1.0e-3, None))
        blk_seeds.append(np.clip(1.0 + 0.6 * np.sin(2.0 * np.pi * jj / m_b + np.pi / 2.0), 1.0e-3, None))
        blk_seeds.append(np.exp(rng.normal(0.0, 0.45, m_b)))
        blk_seeds.append(make_antimirror_seed(m_b, a=0.50))

        small_time = min(45.0, max(10.0, (hard - t_start - 80.0) * 0.20))
        small_start = time.monotonic()
        per_small = small_time / max(1, len(blk_seeds))
        small_results = []
        for idx, ys in enumerate(blk_seeds):
            end = min(small_start + small_time, small_start + per_small * (idx + 1))
            if end <= time.monotonic():
                break
            try:
                yb, _ = optimize_local(ys, m_b, 1000000, end, l2_frac=0.60)
                cb = np.convolve(yb, yb)
                sc = 2.0 * m_b * float(np.max(cb))
                if np.isfinite(sc):
                    small_results.append((sc, yb))
            except Exception:
                pass
        small_results.sort(key=lambda z: z[0])
        for _, yb in small_results[:2]:
            try:
                xb = expand_block(yb, m_b)
                if xb.size == n and np.all(np.isfinite(xb)):
                    full_cands.append(xb)
            except Exception:
                pass

    # Structured full-length seeds.
    if parent_arr is not None:
        full_cands.append(normalize(parent_arr, n))
    full_cands.append(make_fourier_seed(n))
    full_cands.append(make_antimirror_seed(n, a=0.55))
    iarr = np.arange(n, dtype=np.float64)
    full_cands.append(normalize(iarr + 1.0, n))
    full_cands.append(normalize(n - iarr + 1.0, n))

    # Remove exact duplicates by a cheap signature hash of rounded values.
    seen = set()
    uniq_full = []
    for c in full_cands:
        if c.size != n or not np.all(np.isfinite(c)):
            continue
        ccl = normalize(c, n)
        key = (round(float(np.mean(ccl)), 12),
               round(float(np.median(ccl)), 12),
               round(float(np.sum(ccl ** 4)), 9),
               round(float(ccl[0]), 12),
               round(float(ccl[-1]), 12))
        if key not in seen:
            seen.add(key)
            uniq_full.append(ccl)
    full_cands = uniq_full

    if not full_cands:
        full_cands = [normalize(np.ones(n, dtype=np.float64), n)]

    search_deadline = hard - 10.0
    remaining = max(0.0, search_deadline - time.monotonic())
    if remaining < 80.0:
        full_cands = full_cands[:3]
    elif remaining < 240.0:
        full_cands = full_cands[:4]

    search_start = time.monotonic()
    full_total = max(0.0, search_deadline - search_start)
    per_candidate = full_total / max(1, len(full_cands))

    eval_count = 1
    max_evals = 10

    for idx, c0 in enumerate(full_cands):
        if time.monotonic() >= search_deadline:
            break
        end = min(search_deadline, search_start + per_candidate * (idx + 1))
        if end <= time.monotonic():
            break
        try:
            xb, _ = optimize_local(c0, n, 1000000, end, l2_frac=0.55)
            cand_list = to_list(xb)
            if eval_count < max_evals and time.monotonic() < hard - 3.0:
                sc = evaluate_sequence(cand_list)
                eval_count += 1
                if np.isfinite(sc) and sc < best_exact:
                    best_exact = sc
                    best_list = cand_list
        except Exception:
            pass

    # Final polish on the current incumbent.
    if time.monotonic() < hard - 18.0 and n <= 20000:
        l2_frac = 0.50 if not np.isfinite(best_exact) or best_exact > 1.8 else 0.12
        try:
            xb, _ = optimize_local(np.asarray(best_list, dtype=np.float64), n, 1000000,
                                   hard - 7.0, l2_frac=l2_frac)
            cand_list = to_list(xb)
            sc = evaluate_sequence(cand_list)
            if np.isfinite(sc) and sc < best_exact:
                best_exact = sc
                best_list = cand_list
        except Exception:
            pass

    # Final safety evaluation if possible.
    if not np.isfinite(best_exact) and time.monotonic() < hard - 1.0:
        try:
            sc = evaluate_sequence(best_list)
            if np.isfinite(sc):
                best_exact = sc
        except Exception:
            pass

    # Ensure the returned object is a valid non-empty list of finite reals.
    try:
        out = np.asarray(best_list, dtype=np.float64)
        if out.ndim != 1 or out.size == 0:
            out = np.ones(n, dtype=np.float64)
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.maximum(out, 0.0)
        s = float(out.sum())
        if s < 0.01:
            out = np.ones(n, dtype=np.float64)
            s = float(n)
        else:
            out = out / s
        out = np.minimum(out, 1000.0)
        return [float(x) for x in out]
    except Exception:
        return [1.0 / float(n) for _ in range(n)]
