```python
def optimize(a, n, rng, deadline, max_iter=500):
    # Fix sum for scale invariance
    S = np.sqrt(2 * n)
    a = a / a.sum() * S  # normalize
    
    step = 1.0 / n  # initial step size
    
    for it in range(max_iter):
        if time.monotonic() > deadline - 10:
            break
        
        # Compute convolution via FFT
        conv = fft_convolve(a, a)
        kstar = np.argmax(conv)
        
        # Compute gradient of conv[kstar]
        # d/dl conv[kstar] = 2*a[kstar-l] (if kstar-l valid) + (correction for diagonal)
        grad = np.zeros(n)
        # For l in [0, n-1]: if 0 <= kstar-l < n, grad[l] += 2*a[kstar-l]
        # Handle the diagonal: if 2l = kstar, grad[l] should be a[l] not 2*a[l]
        for l in range(n):
            j = kstar - l
            if 0 <= j < n:
                grad[l] = 2.0 * a[j]
            # If 2l == kstar, we double-counted; fix:
            # Actually no: conv[k] = sum_i a[i]*a[k-i]
            # d/dl = a[k-l] (from a[l] in first factor) + a[k-l] (from a[l] in second factor, i.e., k-i=l → i=k-l)
            # If k-l = l (i.e., 2l=k), it's just a[l] (one term, not two)
            if 2*l == kstar:
                grad[l] = a[l]
        
        # Project gradient onto the hyperplane sum=0 (since we constrain sum=S)
        grad -= np.mean(grad)
        
        # Update
        a = a - step * grad
        
        # Project onto {sum=S, 0<=a<=1000}
        a = project(a, S, 1000)
        
        # Decrease step
        step *= 0.99
        
    return a
```

For the projection onto {sum=S, 0<=a<=1000}:
This is a "clipped" projection. Use bisection on a threshold:

```python
def project(a, S, cap=1000.0):
    # Find mu such that sum(clip(a - mu, 0, cap)) = S
    lo, hi = a.min() - cap - 1, a.max() + 1
    for _ in range(100):
        mu = (lo + hi) / 2
        val = np.clip(a - mu, 0, cap)
        s = val.sum()
        if s > S:
            lo = mu
        else:
            hi = mu
    return np.clip(a - mu, 0, cap)
```

```python
def fft_convolve(a, b):
    n = len(a)
    size = 2 * n - 1
    # Use FFT with zero-padding
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B)[:size]
```
