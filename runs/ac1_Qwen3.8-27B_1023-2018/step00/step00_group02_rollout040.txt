    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.full(n, 1.0/n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2*n-1]

def _score(a):
    """Compute AC1 score (scale-invariant, so sum a doesn't matter)."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _conv(a)
    return 2.0 * n * c.max() / (s * s)
