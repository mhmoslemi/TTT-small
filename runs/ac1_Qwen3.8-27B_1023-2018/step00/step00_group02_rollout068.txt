import numpy as np
import time
from scipy.optimize import minimize


def _obj_grad(a, K):
    """Scale-invariant surrogate: mean of top-K of (a*a) over (sum a)^2.

    Returns (value, gradient) with db_k/da_j = 2*a[k-j] for 0<=k-j<n.
    """
    n = a.size
    S = a.sum()
    if S <= 0:
        return 1e300, np.zeros(n)
    b = np.convolve(a, a)
    K = min(K, b.size)
    idx = np.argpartition(b, -K)[-K:]
    m = b[idx].mean()
    gsum = np.zeros(n)
    for k in idx:
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js <= je:
            gsum[js:je + 1] += 2.0 * a[k - js:k - je - 1:-1]
    gsum /= K
    val = m / (S * S)
    grad = gsum / (S * S) - (2.0 * m / (S * S * S)) * np.ones(n)
    return val, grad


def _exact_subgrad_refine(a, deadline, rng):
    """Projected subgradient on the EXACT max(b)/S^2; keeps best by exact eval."""
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = a.size
    step = max(1.0, a.mean() / n)
    while time.monotonic() < deadline - 5.0:
        b = np.convolve(a, a)
        k = int(np.argmax(b))
        S = a.sum()
        bmax = b[k]
        # subgradient of bmax w.r.t. a
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        g = np.zeros(n)
        if js <= je:
            g[js:je + 1] = 2.0 * a[k - js:k - je - 1:-1]
        grad = g / (S * S) - (2.0 * bmax / (S * S * S)) * np.ones(n)
        # backtracking line search
        best_x = None
        best_v = None
        s0 = step
