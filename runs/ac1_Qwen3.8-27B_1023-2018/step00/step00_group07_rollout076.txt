def _flat_conv_init(N):
    """Shape whose low lags of x*x are flat (=1 with x[0]=1); good nonneg init."""
    x = np.zeros(N, dtype=np.float64)
    x[0] = 1.0
    for k in range(1, N):
        acc = 0.0
        upper = (k - 1) // 2
        for j in range(1, upper + 1):
            acc += 2.0 * x[j] * x[k - j]
        if k % 2 == 0:
            acc += x[k // 2] * x[k // 2]
        x[k] = (1.0 - acc) / 2.0
    return x


def _fft_conv(x):
    """Full linear autocorrelation x*x via FFT (length 2n-1)."""
    n = x.size

    need = 2 * n - 1
    while L < need:
        L <<= 1
    X = np.fft.rfft(x, L)
    c = np.fft.irfft(X * X, L)
    return c[:need]
