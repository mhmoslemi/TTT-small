import numpy as np
import time


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    from scipy.fft import rfft, irfft, next_fast_len

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s))
    deadline = t0 + max(0.0, budget - 12.0)

    try:
        scorer = evaluate_sequence
    except NameError:
        def scorer(seq):
            if not isinstance(seq, list) or not seq:
                return float("inf")
            for x in seq:
                if isinstance(x, bool) or not isinstance(x, (int, float)):
                    return float("inf")
            seq = [float(x) for x in seq]
            if any(np.isnan(x) or np.isinf(x) for x in seq):
                return float("inf")
            seq = [max(0.0, x) for x in seq]
            seq = [min(1000.0, x) for x in seq]
            n = len(seq)
            b = np.convolve(seq, seq)
            s = float(np.sum(seq))
            if s < 0.01:
                return float("inf")
            return float(2 * n * float(np.max(b)) / (s * s))

    def normalize_positive(v):
        v = np.asarray(v, dtype=np.float64).ravel()
        if v.size == 0:
            return None
        v = np.where(np.isfinite(v), v, 0.0)
        v = np.clip(v, 0.0, None)
        s = float(v.sum())
        if not np.isfinite(s) or s <= 1e-300:
            return None
        v /= s
        v = np.maximum(v, 1e-16)
        s = float(v.sum())
        if s > 0.0:
            v /= s
        return v

    # Choose length from the supplied parent when sensible.
    parent = None
    n = 4498
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64, copy=False).ravel()
        if arr.size >= 32 and np.all(np.isfinite(arr)):
            if 32 <= arr.size <= 12000:
                n = int(arr.size)
                parent = np.clip(arr, 0.0, 1000.0).copy()
    except Exception:
        parent = None

    if not (32 <= n <= 12000):
        n = 4498
        parent = None

    L = 2 * n - 1
    m = next_fast_len(L)

    def conv_b(x):
        F = rfft(x, m)
        b = irfft(F * F, m)
        b = b[:L]
        np.maximum(b, 0.0, out=b)
        return b

    def conv_b_F(x):
        F = rfft(x, m)
        b = irfft(F * F, m)
        b = b[:L]
        np.maximum(b, 0.0, out=b)
        return b, F

    # Initializations.
    inits = []

    if parent is not None and parent.size == n:
        pv = normalize_positive(parent)
        if pv is not None:
            inits.append(pv)

    # Coefficients of (1-z)^(-1/2), truncated.  A strong non-symmetric start.
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * float(2 * k - 1) / float(2 * k)
    binom = normalize_positive(a)
    if binom is not None:
        inits.append(binom)

    ix = np.arange(1, n + 1, dtype=np.float64)
    for alpha in (0.45, 0.5, 0.55):
        pv = normalize_positive(ix ** (-alpha))
        if pv is not None:
            inits.append(pv)

    # Smooth / block perturbations of the binomial seed.
    if binom is not None:
        for sig in (0.08, 0.18, 0.4):
            z = np.cumsum(rng.normal(0.0, 1.0, n))
            z -= z.mean()
            z *= sig / max(1.0, float(np.sqrt(n)))
            pv = normalize_positive(binom * np.exp(z))
            if pv is not None:
                inits.append(pv)

        for _ in range(10):
            v = binom.copy()
            for _ in range(4):
                w = int(rng.integers(1, max(2, n // 40)))
                s = int(rng.integers(0, max(1, n - w)))
                v[s:s + w] *= float(np.exp(rng.normal(0.0, 0.25)))
            pv = normalize_positive(v)
            if pv is not None:
                inits.append(pv)

    # Pick the best fast-internal initialization.
    best_x = None
    best_mx = float("inf")
    for v in inits:
        if v is None:
            continue
        vv = np.maximum(v, 0.0)
        ss = float(vv.sum())
        if ss <= 1e-300:
            continue
        vv /= ss
        b = conv_b(vv)
        mx = float(b.max())
        if np.isfinite(mx) and mx < best_mx:
            best_mx = mx
            best_x = vv.copy()

    if best_x is None:
        best_x = binom if binom is not None else normalize_positive(np.ones(n))
        if best_x is None:
            best_x = normalize_positive(np.ones(n, dtype=np.float64))

    current = best_x.copy()
    best_internal_x = best_x.copy()
    best_internal_mx = best_mx

    best_exact_x = best_x.copy()
    best_exact_score = np.inf
    last_exact_time = -10.0

    def update_exact(x, force=False):
        nonlocal best_exact_x, best_exact_score, last_exact_time
        now = time.monotonic()
        if not force and (now - last_exact_time) < 3.0:
            return
        last_exact_time = now
        if x is None:
            return
        try:
            sc = float(scorer([float(y) for y in x]))
        except Exception:
            sc = float("inf")
        if np.isfinite(sc) and sc < best_exact_score:
            best_exact_score = sc
            best_exact_x = x.copy()

    update_exact(best_x, force=True)

    base_eta = 0.5
    min_eta = 1e-4
    stale = 0
    iter_count = 0

    while time.monotonic() < deadline:
        b, Fx = conv_b_F(current)
        max_b = float(b.max())

        if np.isfinite(max_b) and max_b < best_internal_mx * (1.0 - 1e-12):
            best_internal_mx = max_b
            best_internal_x = current.copy()
            stale = 0
            update_exact(best_internal_x)

        if max_b <= 1e-300:
            break

        beta = max(20.0, 8.0 / max(max_b, 1e-15))
        beta = min(beta, 1e7)

        e_shift = np.clip(beta * (b - max_b), -50.0, 0.0)
        e = np.exp(e_shift)
        e_sum = float(e.sum())
        if not np.isfinite(e_sum) or e_sum <= 1e-300:
            w = np.zeros(L, dtype=np.float64)
            mask = b >= max_b * (1.0 - 1e-7)
            if not np.any(mask):
                mask[int(np.argmax(b))] = True
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        else:
            w = e / e_sum

        Fw = rfft(w, m)
        h = irfft(np.conj(Fw) * Fx, m)
        h = h[:n]
        np.maximum(h, 0.0, out=h)

        scale = float(h.max())
        if not np.isfinite(scale) or scale <= 1e-300:
            # No useful gradient: perturb or fall back.
            sigma = 0.1
            noise = rng.normal(0.0, sigma, n)
            x_try = current * np.exp(noise)
            x_try = np.maximum(x_try, 1e-16)
            s = float(x_try.sum())
            if s > 0.0:
                x_try /= s
                b_try = conv_b(x_try)
                if float(b_try.max()) < max_b * 2.0:
                    current = x_try
                else:
                    current = best_internal_x.copy()
            stale += 1
            base_eta = max(min_eta, base_eta * 0.5)
            if stale > 200:
                current = best_internal_x.copy()
                stale = 0
                base_eta = 0.3
            iter_count += 1
            continue

        wm = float(np.dot(current, h))
        q = (h - wm) / scale
        q = np.clip(q, -4.0, 4.0)

        etas = (base_eta, 0.5 * base_eta, 0.25 * base_eta)
        best_trial = None
        best_trial_mx = max_b
        accepted_eta = None
        allow_neutral = (stale > 60) or (base_eta < min_eta / 10.0)
        neutral_limit = max_b * (1.0 + 1e-6) if allow_neutral else -1.0

        for eta in etas:
            if eta <= 1e-6:
                continue
            try:
                factor = np.exp(-eta * q)
                x_try = current * factor
                x_try = np.maximum(x_try, 1e-16)
                s = float(x_try.sum())
                if not np.isfinite(s) or s <= 1e-300:
                    continue
                x_try /= s
                b_try = conv_b(x_try)
                mx_try = float(b_try.max())
                if (mx_try < best_trial_mx - 1e-15) or (
                    mx_try <= neutral_limit and mx_try < best_trial_mx + 1e-18
                ):
                    best_trial = x_try
                    best_trial_mx = mx_try
                    accepted_eta = eta
            except Exception:
                continue

        if best_trial is not None and (best_trial_mx < max_b or best_trial_mx <= neutral_limit):
            current = best_trial
            if best_trial_mx < best_internal_mx * (1.0 - 1e-12):
                best_internal_mx = best_trial_mx
                best_internal_x = current.copy()
                stale = 0
                update_exact(best_internal_x)
                if accepted_eta is not None and accepted_eta == etas[0]:
                    base_eta = min(2.0, base_eta * 1.08)
                else:
                    base_eta = max(min_eta, base_eta)
            else:
                stale = min(stale + 1, 300)
                base_eta = max(min_eta, base_eta * 0.97)
        else:
            stale += 1
            base_eta = max(min_eta, base_eta * 0.55)

            if stale > 120 or base_eta <= min_eta * 1.5:
                sigma = 0.08 + 0.04 * float(min(8, stale // 30))
                noise = rng.normal(0.0, sigma, n)
                x_try = current * np.exp(noise)
                x_try = np.maximum(x_try, 1e-16)
                s = float(x_try.sum())
                if s > 0.0:
                    x_try /= s
                    b_try = conv_b(x_try)
                    if float(b_try.max()) < max_b * 1.75:
                        current = x_try
                    else:
                        current = best_internal_x.copy()
                stale = 0
                base_eta = 0.35

        iter_count += 1

    # Final validation and selection among strong candidates.
    finalists = []

    def add_candidate(cand):
        if cand is None:
            return
        c = np.asarray(cand, dtype=np.float64).ravel()
        if c.size != n or not np.all(np.isfinite(c)):
            return
        c = np.maximum(c, 0.0)
        s = float(c.sum())
        if s <= 1e-300:
            return
        c /= s
        for old in finalists:
            if old.shape == c.shape and np.array_equal(old, c):
                return
        finalists.append(c)

    add_candidate(best_internal_x)
    add_candidate(best_exact_x)
    add_candidate(current)
    add_candidate(binom)
    if parent is not None and parent.size == n:
        add_candidate(parent)

    final_best = None
    final_score = float("inf")

    for cand in finalists:
        if time.monotonic() > deadline - 4.0:
            break
        try:
            sc = float(scorer([float(v) for v in cand]))
        except Exception:
            sc = float("inf")
        if np.isfinite(sc) and sc < final_score:
            final_score = sc
            final_best = cand

    if final_best is None:
        final_best = best_internal_x if best_internal_x is not None else current

    out = [float(v) for v in np.clip(final_best, 0.0, 1000.0)]
    if not out:
        out = [1.0]
    return out
