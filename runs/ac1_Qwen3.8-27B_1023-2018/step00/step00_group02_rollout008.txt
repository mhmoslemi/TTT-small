```python
def solve_lp_direction(a, M, n, max_constraints=200, time_limit=5):
    """Solve the direction-finding LP with a small number of constraints."""
    import cvxpy as cp
    
    b = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(b))
    
    # Start with the most violated constraints (top peaks)
    b_conv = np.convolve(a, a)
    top_k = np.argsort(b_conv)[-max_constraints:]
    
    constraints = []
    for k in top_k:
        # conv(a, b)[k] = sum_i a[k-i]*b[i] for valid i
        # = a_reversed_convolved_with_b at position k
        # As a linear expression: sum_i a[k-i]*b[i]
        # where a[k-i] is the coefficient for b[i]
        
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        coeffs = a[k - i_max : k - i_min + 1][::-1]  # a[k-i_max], ..., a[k-i_min]
        constraints.append(coeffs @ b[i_min:i_max+1] <= M)
    
    try:
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, max_iter=2000)
        if b.value is not None:
            return np.array(b.value).flatten()
    except:
        pass
    return None
```
