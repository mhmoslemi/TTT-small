For the projection onto the simplex, I'll use the standard algorithm:
- Given x, find the projection p onto {a >= 0, sum(a) = 1}
- p_i = max(x_i - tau, 0) where tau is chosen so sum(p) = 1
- tau is found by sorting x and trying O(n) candidate values

```python
import numpy as np
import time
import math

def simplex_project(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    if n == 0:
        return x
    # Algorithm from Duchi et al. 2008
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)

def logsumexp(x, t):
    """Compute t * log(sum(exp(x/t))) stably."""
    x_max = np.max(x)
    return x_max + t * np.log(np.sum(np.exp((x - x_max) / t)))

def compute_smoothed_objective_and_grad(a, t):
    """Compute the smoothed objective and its gradient."""
    n = len(a)
    g = np.convolve(a, a)  # length 2n-1
    # Log-sum-exp
    x = g / t
    x_max = np.max(x)
    log_sum = x_max + np.log(np.sum(np.exp(x - x_max)))
    S = t * log_sum  # This is the smoothed approximation to max(g)
    # Actually S = t * (x_max + log(sum(exp(x - x_max)))) = max(g) + t*log(sum(exp((g-max)/t)))
    # = max(g) + t*log(sum_i exp((g_i - max(g))/t))
    # The gradient:
    w = np.exp(x - x_max) / np.sum(np.exp(x - x_max))  # softmax(g/t)
    # grad[j] = sum_k w[k] * a[k-j] = correlate(w, a, 'full')[n-1 : n-1+n]
    corr = np.correlate(w, a, mode='full')
    grad = corr[n-1:n-1+n]
    return S, grad

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)
    
    # Initialize
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except NameError:
        a = np.ones(3809, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    a_sum = a.sum()
    if a_sum < 1e-10:
        a = np.ones(3809) / 3809
    else:
        a = a / a_sum
    
    n = len(a)
    
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    # Phase 1: Smoothed gradient descent
    # Parameters
    t_start = 0.05  # Initial temperature for log-sum-exp
    t_end = 0.001   # Final temperature
    lr = 0.005     # Learning rate for Adam
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    iteration = 0
    no_improve_count = 0
    eval_interval = 50  # Check exact score every 50 iterations
    
    # Time per iteration estimate
    while time.monotonic() < deadline * 0.6:
        # Adaptive temperature
        t = t_start * (t_end / t_start) ** (iteration / max(1, 2000))
        
        # Compute gradient
        S, grad = compute_smoothed_objective_and_grad(a, t)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration+1))
        v_hat = v / (1 - beta2**(iteration+1))
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex
        a = simplex_project(a)
        
        iteration += 1
        
        # Periodic exact evaluation
        if iteration % eval_interval == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best = a.copy()
                best_score = score
                no_improve_count = 0
            else:
                no_improve_count += 1
            
            # Reset if stuck
            if no_improve_count > 20:
                # Add random perturbation
                a = a + 0.01 * rng.standard_normal(n)
                a = simplex_project(a)
                no_improve_count = 0
    
    # Phase 2: Fine-tuning with random perturbations
    while time.monotonic() < deadline:
        # Small random perturbation
        perturbation_scale = 0.005 * max(1, (deadline - time.monotonic()) / 60)
        a_trial = a + perturbation_scale * rng.standard_normal(n)
        a_trial = simplex_project(a_trial)
        
        score = float(evaluate_sequence(a_trial.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a_trial.copy()
            best_score = score
            a = a_trial.copy()
    
    # Final check
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # Fallback to uniform
        return [1.0/n] * n
```

```python
import numpy as np
import time
import math

def _simplex_project(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)


def _smoothed_grad(a, t):
    """Compute gradient of t*log(sum(exp(conv(a,a)/t))) w.r.t. a."""
    n = len(a)
    g = np.convolve(a, a)
    x = g / t
    x_max = np.max(x)
    e = np.exp(x - x_max)
    w = e / np.sum(e)
    # grad[j] = sum_k w[k] * a[k-j] (a zero-padded)
    corr = np.correlate(w, a, mode='full')
    grad = corr[n - 1:n - 1 + n]
    return grad


def _lp_direction(f, M, max_time=15.0):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0. Return normalized b or None."""

        import cvxpy as cp

        return None

    rows = []
    cols = []
    vals = []
    for j in range(n):
        # b[j] appears in (f*b)[k] for k from j to j+n-1 with coefficient f[k-j]
        for k in range(j, j + n):
            if k < 2 * n - 1:
                rows.append(k)
                cols.append(j)
                vals.append(f[k - j])
    
    from scipy.sparse import csc_matrix
    A = csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    prob = cp.Problem(objective, constraints)
```
