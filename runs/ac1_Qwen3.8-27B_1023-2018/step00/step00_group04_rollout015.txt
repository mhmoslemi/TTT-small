        F = np.fft.rfft(p, L1)
        c = np.fft.irfft(F * F, L1)[:2 * n - 1]
        # Remove harmless round-off noise.
        c = np.where(c > 0.0, c, 0.0)
        c = np.asarray(c, dtype=np.float64)
        return c

        if tau <= 0.0:
            return float(c.max())

            return float("inf")

        e = np.exp(z)
        s = float(e.sum())
        if not math.isfinite(s) or s <= 0.0:
            return m
        return m + tau * math.log(s)

        Fw = np.fft.rfft(W, L2)

        g = 2.0 * corr[n - 1:n - 1 + n]
        if not np.isfinite(g).all():
            g = np.zeros(n, dtype=np.float64)
        return g

            cache["f"] = f
            cache["g"] = g

            f, _ = get_eval(x)
            return f

            _, g = get_eval(x)
            return g

    # ------------------------------------------------------------------

        if s <= 1e-12 or not math.isfinite(s):
            arr = uniform.copy()
        else:
            arr = arr / s
        return [float(x) for x in arr.tolist()]

    cand = _safe_list(out)
    try:
        sc = evaluate_sequence(cand)
        if not math.isfinite(sc):
            cand = _safe_list(uniform)
            sc = evaluate_sequence(cand)
    except Exception:
        cand = _safe_list(uniform)

    return cand
