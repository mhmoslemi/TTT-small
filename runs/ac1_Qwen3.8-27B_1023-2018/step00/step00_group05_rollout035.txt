def _project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n+1)
    # Find largest k such that u[k-1] > (cssv[k-1] - s) / k
    tau_candidates = (cssv - s) / rho_candidates
    valid = u > tau_candidates
    if not valid.any():
        tau = 0.0
    else:
        rho = np.max(np.where(valid)[0])
        tau = tau_candidates[rho]
    return np.maximum(v - tau, 0.0)
