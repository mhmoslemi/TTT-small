        if not chunks_r:
            return sp.csc_matrix((2 * nn - 1, nn), dtype=np.float64)
        rr = np.concatenate(chunks_r)
        cc = np.concatenate(chunks_c)
        vv = np.concatenate(chunks_v)
        T = sp.csc_matrix((vv, (rr, cc)), shape=(2 * nn - 1, nn))
        return T

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    step = 0.5
    no_improve = 0
    prev_score = best_score

    for it in range(2000):
        t_start = time.monotonic()
        remaining = deadline - t_start
        if remaining < 20.0:
            break

        # Current max convolution
        conv_a = np.convolve(a, a)
        M = float(np.max(conv_a))
        if M <= 1e-15:
            break

        # Build T (recomputed because a changes each iteration)
        T = build_T(a)
        rhs = np.full(2 * n - 1, M, dtype=np.float64)

        # Solve LP: max sum(b)  s.t.  T@b <= M,  b >= 0
        lp_limit = min(60.0, remaining - 25.0)
        if lp_limit < 2.0:
            break
        try:
            res = opt.linprog(
                c_obj, A_ub=T, b_ub=rhs, bounds=bounds,
                method='highs', options={'time_limit': lp_limit}
            )
        except Exception:
            break

        if res.x is None:
            step = max(0.05, step * 0.8)
            continue

        b_dir = np.maximum(res.x, 0.0)
        S_b = b_dir.sum()
        if S_b < 1e-12:
            # Fallback: use reversed a
            b_dir = a[::-1].copy()
            S_b = b_dir.sum()
            if S_b < 1e-12:
                break
            b_dir /= S_b
            a = (1 - step) * a + step * b_dir
            a = np.maximum(a, 0)
            a /= a.sum()
            prev_score = score_norm(a)
            continue

        b_dir /= S_b  # normalise to sum 1

        # Step toward b_dir
        a_trial = (1.0 - step) * a + step * b_dir
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-12:
            step *= 0.5
            continue
        a_trial /= s_t

        new_score = score_norm(a_trial)
        if not np.isfinite(new_score):
            step = max(0.05, step * 0.5)
            continue

        if new_score < best_score:
            best = a_trial.copy()
            best_score = new_score
            no_improve = 0
            step = min(1.0, step * 1.08)
        else:
            no_improve += 1
            if new_score < prev_score:
                no_improve = max(0, no_improve - 1)
            step = max(0.05, step * 0.75)

        a = a_trial
        prev_score = new_score

        # Perturb to escape plateaus
        if no_improve >= 12:
            noise_scale = 0.003 * (1 + 0.005 * no_improve)
            a = a + rng.normal(0, noise_scale, n)
            a = np.maximum(a, 0.0)
            sa = a.sum()
            if sa > 1e-12:
                a /= sa
            # Also try a reversed blend sometimes
            if rng.random() < 0.3:
                a = 0.7 * a + 0.3 * a[::-1]
                a = np.maximum(a, 0)
                a /= a.sum()
            no_improve = 0
            step = 0.35

        # Check deadline
        if time.monotonic() > deadline - 12.0:
            break

    # --- Finalise: scale to a safe range and validate ---
    s_best = best.sum()
    scale = 100.0 / s_best if s_best > 1e-12 else 1.0
    result = best * scale
    result = np.clip(result, 0.0, 1000.0)

    # Clamp: ensure sum >= 0.01 for the evaluator
    if result.sum() < 0.01:
        result = parent.copy()

    final_score = evaluate_sequence(result.tolist())
    if not np.isfinite(final_score):
        result = parent.copy()
        final_score = evaluate_sequence(result.tolist())
    if not np.isfinite(final_score):
        result = np.ones(max(n, 1)) * 1.0
        final_score = evaluate_sequence(result.tolist())

    return [float(x) for x in result]
