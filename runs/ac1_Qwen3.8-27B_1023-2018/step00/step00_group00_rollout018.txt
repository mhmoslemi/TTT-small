    def objective_and_grad(x):
        # x is the unnormalized vector; we normalize internally
        a = np.maximum(x, 0.0)
        S = a.sum()
        if S < 1e-10:
            return 1e10, np.zeros_like(x)
        a = a / S  # normalize
        c = np.convolve(a, a)

        c_max = c.max()
        exp_tau_c = np.exp(tau * (c - c_max))
        sum_exp = exp_tau_c.sum()
        M_smooth = (c_max + np.log(sum_exp)) / tau
        p = exp_tau_c / sum_exp
        d = np.correlate(p, a, mode='valid')
        # Objective: 2n * M_smooth (since S=1)
        val = 2.0 * n * M_smooth
        # Gradient: (2n / S^3) * (2S*d - M_smooth) with S=1: 2n*(2*d - M_smooth)
        grad = 2.0 * n * (2.0 * d - M_smooth)
        # But we need to account for the normalization... 
        # Actually, since we normalize inside, the gradient w.r.t. x (unnormalized)
        # is more complex. Let's just use the projected gradient approach instead.
        return val, grad
    
    # Let's use a manual gradient descent with the normalized parameterization
    a = a_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    lr = 0.5  # learning rate for Adam
    
    for phase in range(50):
        if time.monotonic() > deadline:
            break
        
        # Adjust tau: start moderate, increase over time
        tau_current = 10.0 + phase * 2.0
        tau_current = min(tau_current, 500.0)

            if time.monotonic() > deadline:

            c_max = c.max()

            sum_exp = exp_tc.sum()
            M_s = (c_max + np.log(sum_exp)) / tau_current
            p = exp_tc / sum_exp

            m_hat = m / (1 - 0.9 ** t_adam)
            v_hat = v / (1 - 0.999 ** t_adam)
            
            # Step

            a_new = a - step
            
            # Project and renormalize
            a_new = np.maximum(a_new, 0.0)

            if S_new < 1e-10:

            else:
                a_new = a_new / S_new

        if np.isfinite(score) and score < best_score:

            best_a = a.copy()
            # Slightly increase lr for exploration

            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return [float(x) for x in best_a]
