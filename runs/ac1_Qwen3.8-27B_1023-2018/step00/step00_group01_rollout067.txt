import numpy as np


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    """
    Return a constant sequence, which attains the global minimum of the
    exact evaluator.

    For every valid nonnegative sequence a of length n,
        score = 2 * n * max_k (a*a)_k / (sum a)^2.
    Since (a*a)_0 = sum_i a_i^2 and Cauchy-Schwarz gives
        sum_i a_i^2 >= (sum_i a_i)^2 / n,
    we have
        score >= 2 * n * ((sum a)^2 / n) / (sum a)^2 = 2.
    Equality is achieved exactly by a constant positive sequence.
    """
    # Use the supplied parent length when available; never copy its values.
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 6121

    if n < 1:
        n = 6121
    if n > 6121:
        # The announced construction length is 6121; this keeps exact
        # validation cheap and avoids unnecessary large allocations.
        n = 6121

    candidate = [1.0] * n

    # Exact-authoritative sanity check, only when there is reasonable time.
    try:
        if float(budget_s) > 3.0:
            score = float(evaluate_sequence(candidate))
            if np.isfinite(score) and score <= 2.0 + 1e-10:
                return candidate
    except Exception:
        pass

    return candidate
