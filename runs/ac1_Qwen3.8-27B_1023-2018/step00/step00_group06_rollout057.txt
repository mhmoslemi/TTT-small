```python
def lp_step(f, M, n):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    # Constraint matrix: A, where A[k, i] = f[i] if 0 <= k-i < n
    # Number of constraints: 2n-1
    num_constraints = 2*n - 1
    
    # Build the constraint matrix (sparse)
    rows = []
    cols = []
    vals = []
    for k in range(num_constraints):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    
    # Objective: minimize -sum(g)
    c = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    # Solve
    result = linprog(c, A_ub=A, b_ub=np.full(num_constraints, M), bounds=bounds, method='highs')
    
    return result.x
```
