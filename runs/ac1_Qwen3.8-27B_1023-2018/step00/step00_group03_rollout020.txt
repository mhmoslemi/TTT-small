```python
def solve_lp(f, M):
    """Solve: max sum(g) s.t. B@f <= M, g >= 0"""
    n = len(f)
    # B is the convolution matrix: (B*g)[k] = (f*g)[k]
    # B[k,i] = f[k-i] if 0 <= k-i < n
    
    # Build B as sparse
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k-i])
    
    B = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: max c^T g s.t. B g <= M*ones, g >= 0
    c = np.ones(n)  # maximize sum(g)
    A_ub = B  # B@g <= M
    b_ub = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    
    result = optimize.linprog(-c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        return result.x
    return None
```

```python
def _make_conv_matrix(f):
    n = len(f)
    m = 2 * n - 1
    # For each output index k, the input indices i such that 0 <= k-i < n
    # i ranges from max(0, k-n+1) to min(n-1, k)
    row_idx = []
    col_idx = []
    data = []
    for k in range(m):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        n_elts = i_end - i_start + 1

        col_idx.extend(range(i_start, i_end + 1))
        data.extend(f[k - np.arange(i_start, i_end + 1)].tolist())
    A = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(m, n))
    return A
```

```python
def _subgradient_step(f, M, step):
    c = np.convolve(f, f)
    # Active set: lags within 1% of max
    active = np.where(c >= M * 0.99)[0]
    sg = np.zeros_like(f)
    n = len(f)
    for k in active:
        for j in range(n):
            if 0 <= k - j < n:
                sg[j] += f[k - j]
            if 0 <= j - k < n:
                sg[j] += f[j - k]
    sg /= len(active)
    f_new = f - step * sg
    f_new = np.clip(f_new, 0, None)
    f_new /= f_new.sum()
    return f_new
```
