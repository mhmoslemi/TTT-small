from scipy.linalg import toeplitz

def _conv_matrix(a, n):
    """Build (2n-1) x n Toeplitz convolution matrix."""
    c = np.zeros(2*n - 1)
    c[:n] = a
    r = np.zeros(n)
    r[0] = a[0]
    return toeplitz(c, r)

    res = opt.linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
