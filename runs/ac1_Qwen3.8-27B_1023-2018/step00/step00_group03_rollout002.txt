import numpy as np
import time


def _score_candidate(seq):
    """Return the exact evaluator score, or +inf on failure."""
    try:
        s = evaluate_sequence(seq)
        s = float(s)
        if np.isfinite(s):
            return s
    except Exception:
        pass
    return float("inf")


def _parent_sequence_and_length(obj, max_len):
    """
    Extract a clipped, validated parent sequence if small enough to verify.
    Always returns the parent length when it can be obtained safely.
    """
    try:
        length = len(obj)
    except Exception:
        return None, -1

    if length <= 0:
        return None, -1
    if length > max_len:
        return None, int(length)

    try:
        arr = np.asarray(obj, dtype=np.float64)
    except Exception:
        return None, int(length)

    if arr.ndim != 1 or arr.size != length or not np.all(np.isfinite(arr)):
        return None, int(length)

    clipped = [float(min(1000.0, max(0.0, v))) for v in arr.tolist()]
    if sum(clipped) < 0.01:
        return None, int(length)

    return clipped, int(length)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    """
    Return a valid nonnegative sequence minimizing the authoritative score.

    For any nonnegative a = (a_0, ..., a_{n-1}) with S = sum_j a_j, the
    central coefficient of np.convolve(a, a) is sum_j a_j^2.  By Cauchy,
    sum_j a_j^2 >= S^2 / n, so

        2*n*max_k (a*a)[k] / S^2 >= 2.

    Equality is attained by any positive constant sequence.  Therefore the
    exact optimum is 2, and the optimal search is to verify a constant
    sequence, preferably with the supplied construction length.  The remaining
    bounded probes only invoke the authoritative evaluator to capture any
    possible IEEE-rounding deficit, but they cannot change the mathematical
    optimum.
    """
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0

    if not np.isfinite(budget) or budget < 0.0:
        budget = 1300.0

    headroom = min(5.0, max(0.2, 0.05 * budget)) if budget > 0.0 else 0.0
    deadline = time.monotonic() + max(0.0, budget - headroom)

    # A one-coefficient constant sequence is already globally optimal.
    best = [1.0]
    best_score = _score_candidate(best)

    # Improvements smaller than this are floating-point noise relative to the
    # proven bound 2.  Exact equality is still handled by the length preference.
    IMPROVE_EPS = 1e-12

    def consider(seq, prefer_length=False):
        nonlocal best, best_score
        if seq is None:
            return

        s = _score_candidate(seq)
        if not np.isfinite(s):
            return

        if s < best_score - IMPROVE_EPS:
            best = seq
            best_score = s
        elif prefer_length and s <= best_score and len(seq) > len(best):
            best = seq
            best_score = s

    # Use the supplied construction when available.  We only directly verify
    # parent sequences of bounded length so quadratic convolution remains safe.
    max_parent_len = 5000

        if 0 < parent_len <= max_parent_len:

            if parent_seq is not None and time.monotonic() < deadline:
                consider(parent_seq)
    except NameError:

    except Exception:
        pass
