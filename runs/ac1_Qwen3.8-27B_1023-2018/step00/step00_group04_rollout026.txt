import numpy as np
import time

def compute_score_and_grad(a):
    """Compute AC1 score and its gradient."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf, np.zeros(n)
    c = np.convolve(a, a)
    M = np.max(c)
    score = 2.0 * n * M / (S * S)
    
    # Find all indices achieving the max
    max_indices = np.where(c == M)[0]
    
    # Gradient of score w.r.t. a[i]
    # score = 2n*M/S^2
    # d(score)/da[i] = 2n * (dM/da[i]) / S^2 - 2n*M * 2 / S^3
    # = 2n/S^2 * dM/da[i] - 4nM/S^3
    
    # dM/da[i] = average of dc_k/da[i] over active k (subgradient)
    dM = np.zeros(n)
    for k in max_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # dc_k/da[i] = 2*a[k-i] for i in [lo, hi]
            idx = np.arange(lo, hi + 1)
            dM[idx] += 2.0 * a[k - idx]
    dM /= len(max_indices)
    
    grad = (2.0 * n / (S * S)) * dM - (4.0 * n * M) / (S ** 3)
    return score, grad

        while Nfft < N:
            Nfft *= 2
        fa = np.fft.rfft(a, Nfft)
