            break

        alloc = _allocation_for(L)
        remaining = max(0.0, deadline - time.monotonic() - 20.0)
        alloc = min(alloc, remaining)
        if alloc < 3.0:
            continue

        try:
            candidate, fast_score = _optimize_softmax(L, int(seed) + 100000 + 7 * L, time.monotonic() + alloc, parent)
        except Exception:
            candidate = None

        if candidate is None:
            continue
        try:
            candidate = np.asarray(candidate, dtype=np.float64).ravel()
        except Exception:
            continue
        if candidate.size != L or not np.isfinite(candidate).all():
            continue
        candidate = np.clip(candidate, 0.0, 1000.0)

        if not np.isfinite(fast_score):
            fast_score = _fft_score(candidate)

        # Validate only if the internal surrogate suggests a plausible improvement.
        if np.isfinite(fast_score) and fast_score < best_score + 0.05:
            try:
                exact_score = float(evaluate_sequence([float(x) for x in candidate]))
            except Exception:
                exact_score = math.inf
            if np.isfinite(exact_score) and exact_score < best_score - 1e-12:
                best_seq = candidate.copy()
                best_score = exact_score

    if time.monotonic() < deadline - 12.0:
        try:
            best_seq, best_score = _polish(
                best_seq,
                best_score,
                max(time.monotonic(), min(deadline, time.monotonic() + max(0.0, deadline - time.monotonic() - 8.0)) - 3.0)
            )
        except Exception:
            pass

    # Final exact validation of the sequence we intend to return.
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_score = math.inf

    if not np.isfinite(final_score):
        try:
            final_score = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            final_score = math.inf
        if not np.isfinite(final_score):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
