import numpy as np
from scipy.sparse import csc_matrix
from scipy.optimize import linprog
import time

def compute_autocorr(f):
    """Compute full autocorrelation using FFT."""
    n = len(f)
    size = 2*n - 1
    # Use next power of 2 for FFT
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    F = np.fft.rfft(f, fft_size)
    conv = np.fft.irfft(F * F, fft_size)[:size]
    return conv

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10  # 10s buffer
    
    # Initialize
    f = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    n = len(f)
    S = f.sum()
    f = f / S  # normalize to sum = 1
    
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    
    t = 0.1  # blending parameter
    
    while time.monotonic() < deadline:
        # Compute peak
        r = compute_autocorr(f)
        M = r.max()
        k_star = r.argmax()
        
        # Build LP constraint matrix (sparse)
        # A[k, i] = f[k-i] if 0 <= k-i < n
        rows = []
        cols = []
        vals = []
        for i in range(n):
            for k in range(i, i+n):
                rows.append(k)
                cols.append(i)
                vals.append(f[k-i])
        
        A = csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        
        # LP: max 1^T g s.t. A @ g <= M, g >= 0
        # Minimize -1^T g s.t. A @ g <= M, g >= 0
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), 
                      bounds=[(0, None)]*n, method='highs')
        
        if res.success:
            g = res.x
            g_sum = g.sum()

                new_f = (1-t)*f + t*g
