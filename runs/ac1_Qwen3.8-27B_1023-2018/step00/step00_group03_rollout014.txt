        a = a0.copy()
        n_local = len(a)
        if a.sum() < 0.01:
            return a0, score_norm(a0)

        local_best_score = score_norm(a)

            fft_size *= 2

            A = np.fft.rfft(a, fft_size)
            b = np.fft.irfft(A * A, fft_size)[:need]
            b_max = b.max()
            t = max(t1, t0 * (t1/t0)**(it/5000))  # anneal temperature
            
            # Softmax weights
            b_shifted = (b - b_max) / t
            w = np.exp(b_shifted)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(need)
                w[np.argmax(b)] = 1
            else:
                w /= w_sum

            Ar = np.fft.rfft(a[::-1], fft_size)
            temp = np.fft.irfft(W * Ar, fft_size)
            grad_raw = temp[n_local-1 : 2*n_local-1]

            S = a.sum()
            M_smooth = b_max + t * np.log(w_sum)
            grad = 4 * n_local / S**3 * (S * grad_raw - M_smooth)
            
            # Step
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-10:
                lr *= 0.5
                continue
            a_new = a_new / S_new  # renormalize
            
            score_new = score_norm(a_new)
            
            if score_new < local_best_score:
                local_best = a_new.copy()
                local_best_score = score_new
                no_improve = 0
                lr = min(lr * 1.02, 0.1)
            else:
                no_improve += 1
                if no_improve > 200:
                    lr *= 0.95
                elif no_improve > 50:
                    lr *= 0.98
            
            a = a_new
        
        return local_best, local_best_score

    tri = np.concatenate([tri, tri[::-1][1:]])[:n]
    tri /= tri.sum()
    init_list.append(tri)

    x = np.linspace(0, 1, n)
    cos_seq = 0.5 * (1 + np.cos(2 * np.pi * x))
    cos_seq = np.maximum(cos_seq, 1e-6)
    cos_seq /= cos_seq.sum()
    init_list.append(cos_seq)
    
    # Power law (edge-concentrated)
    for alpha in [0.5, 1.0, 2.0]:
        pw = np.power(np.arange(1, n+1), alpha)
        pw /= pw.sum()
        init_list.append(pw)
    
    # Random
    for _ in range(3):
        r = rng.random(n)
        if r.sum() > 0:
            init_list.append(r / r.sum())

    for init_a in init_list:
        if time.monotonic() >= deadline - 10:
            break
        remaining = deadline - time.monotonic() - 10
        if remaining < 5:
            break
        # Give more time to promising starts

        opt_a, opt_sc = optimize(init_a, alloc)
        sc_check = evaluate_sequence(opt_a.tolist())
        if np.isfinite(sc_check) and sc_check < best_score:
            best = opt_a
            best_score = sc_check

    if time.monotonic() < deadline - 10:

        for _ in range(5):
            if time.monotonic() >= deadline - 10:

            # Small random perturbation

            pert = best + sigma * rng.standard_normal(n)
            pert = np.maximum(pert, 0)

            pert /= pert.sum()

            opt_a, opt_sc = optimize(pert, alloc, t0=0.05, t1=0.001)
            sc_check = evaluate_sequence(opt_a.tolist())
            if np.isfinite(sc_check) and sc_check < best_score:

                best_score = sc_check
            remaining = deadline - time.monotonic() - 10

    final_score = evaluate_sequence(final)
    if not np.isfinite(final_score):
        final = [1.0/n] * n
        final_score = evaluate_sequence(final)
