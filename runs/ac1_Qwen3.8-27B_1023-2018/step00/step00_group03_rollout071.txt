    for iteration in range(20000):
        if time.monotonic() > deadline:
            break

        conv = np.convolve(a, a)
        M = conv.max()
        if M < 1e-18:
            break

        # Build sparse A with current values
        dat = np.repeat(a, n)
        A = csr_matrix((dat, (row_idx, col_idx)), shape=(2 * n - 1, n))

        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b = np.maximum(res.x, 0.0)
        S_b = b.sum()
        if S_b < 1e-12:
            break
        b = b / S_b

        # Adaptive step size
        step = max(0.05, 0.5 * (0.998 ** iteration))
        a_new = (1.0 - step) * a + step * b
        a_new = np.clip(a_new, 0.0, 1000.0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            break
        a_new = a_new / s_new

        # Check score
        conv_new = np.convolve(a_new, a_new)
        sc = 2.0 * n * conv_new.max()  # sum=1 so score = 2n*max_b
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new

        # If no improvement in a while, nudge with small asymmetry perturbation
        if iteration % 200 == 199:
            rng_k = np.random.default_rng(seed + iteration)
            pert = np.exp(rng_k.normal(0, 0.02, n))
            a_p = a * pert
            a_p = np.clip(a_p, 0.0, 1000.0)
            sp = a_p.sum()
            if sp > 1e-12:
                a_p /= sp
                cp = np.convolve(a_p, a_p)
                scp = 2.0 * n * cp.max()
                if scp < best_score:
                    best_score = scp
                    best_a = a_p.copy()
                a = a_p  # use perturbed version for next LP

    # Local search refinement (block perturbations)
    rng = np.random.default_rng(seed)
    for _ in range(2000):
        if time.monotonic() > deadline:
            break
        a_try = best_a.copy()
        width = max(1, int(rng.integers(5, n // 5 + 1)))
        start = int(rng.integers(0, n - width + 1))
        a_try[start:start + width] *= float(np.exp(rng.normal(0, 0.04)))
        a_try = np.clip(a_try, 0.0, 1000.0)
        st = a_try.sum()
        if st < 0.01:
            continue
        conv_t = np.convolve(a_try, a_try)
        sc_t = 2.0 * n * conv_t.max() / (st * st)
        if sc_t < best_score:
            best_score = sc_t
            best_a = a_try.copy()

    # Final validation with exact evaluator
    final = np.clip(best_a, 0.0, 1000.0)
    final_score = float(evaluate_sequence(final.tolist()))
    if not np.isfinite(final_score):
        # Fallback: use a simple quadratic ramp
        final = (np.arange(n, dtype=np.float64) / (n - 1)) ** 2
        final_score = float(evaluate_sequence(final.tolist()))

    return [float(x) for x in final]
