```python
import numpy as np
import time
import math

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _finite_norm(a):
    """Return a finite, nonnegative, normalized copy, or None."""
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return None
    if a.size == 0 or not np.all(np.isfinite(a)):
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a, dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def _score_norm(a):
    """Exact scale-free score for arrays intended to have sum 1."""
    a = _finite_norm(a)
    if a is None:
        return float("inf")
    b = np.convolve(a, a)
    if b.size == 0 or not np.all(np.isfinite(b)):
        return float("inf")
    return float(2.0 * a.size * float(np.max(b)))


def _upsample(v, target):
    """Block-repeat a normalized vector to a target length, preserving mass."""
    v = _finite_norm(v)
    if v is None:
        return _finite_norm(np.ones(max(1, int(target))))
    target = int(target)
    if target <= 0:
        target = 1
    if v.size == target:
        return v.copy()
    idx = (np.arange(target, dtype=np.int64) * v.size) // v.size
    counts = np.bincount(idx, minlength=v.size)
    counts = np.where(counts > 0, counts, 1)
    out = v[idx] / counts[idx]
    return _finite_norm(out)


def _random_a(m, rng):
    m = int(m)
    style = int(rng.integers(0, 6))
    if style == 0:
        x = rng.exponential(scale=1.0, size=m)
    elif style == 1:
        x = rng.uniform(0.1, 1.0, size=m)
    elif style == 2:
        x = np.exp(-8.0 * np.abs(rng.standard_normal(m)))
    elif style == 3:
        t = np.linspace(0.0, 1.0, m, endpoint=False)
        x = (1.0 + np.cos(2.0 * math.pi * t)) / 2.0 + 0.1
    elif style == 4:
        x = rng.pareto(1.7, size=m) + 1e-3
    else:
        x = rng.exponential(scale=1.0, size=m)
        k = int(rng.integers(max(1, m // 5), max(2, m // 2)))
        x[rng.choice(m, size=k, replace=False)] *= 0.0
    out = _finite_norm(x)
    if out is None:
        out = np.ones(m, dtype=np.float64) / m
    return out


def _harmonic(m, amp, phase, kind=0):
    m = int(m)
    t = np.linspace(0.0, 1.0, m, endpoint=False)
    if kind == 0:
        f = 1.0 - amp * np.cos(math.pi * t + phase)
    elif kind == 1:
        f = 1.0 - amp * np.sin(math.pi * t + phase)
    else:
        f = 1.0 + amp * np.cos(2.0 * math.pi * t + phase)
    f = np.clip(f, 1e-5, None)
    return _finite_norm(f)


def _active_subgradient(a):
    """A subgradient of max_k (a*a)_k, averaged over active lags."""
    a = _finite_norm(a)
    if a is None:
        return None, None
    b = np.convolve(a, a)
    M = float(np.max(b))
    tol = max(1e-12, 1e-9 * abs(M))
    act = np.nonzero(b >= M - tol)[0]
    if act.size == 0:
        grad = np.zeros_like(a)
        return b, grad
    if act.size > 300:
        step = max(1, act.size // 300)
        act = act[::step]
    grad = np.zeros_like(a)
    n = a.size
    for k in act:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        cols = np.arange(lo, hi + 1)
        idx = k - cols
        grad[lo:hi + 1] += 2.0 * a[idx]
    grad /= max(1, act.size)
    return b, grad


def _lp_direction(f, M):
    """
    Solve the Frank-Wolfe-style LP direction:
        max sum(g)   s.t. g >= 0, (f*g)_k <= M for all k.
    The direction is returned normalized to sum 1.
    """
    if linprog is None:
        return None
    f = _finite_norm(f)
    if f is None:
        return None
    m = f.size
    if not np.isfinite(M) or M <= 0.0:
        return None

    fmax = float(np.max(f))
    if not np.isfinite(fmax) or fmax <= 0.0:
        return None

    scale = 1.0 / fmax
    f_use = f * scale
    M_use = M * scale
    if not np.isfinite(M_use) or M_use <= 0.0:
        return None

    try:
        A = np.empty((2 * m - 1, m), dtype=np.float64)
        A.fill(0.0)
        for k in range(2 * m - 1):
            lo = 0 if k < m else k - m + 1
            hi = k if k < m else m - 1
            A[k, lo:hi + 1] = f_use[k - lo : k - hi - 1 : -1]

        c = -np.ones(m, dtype=np.float64)
        rhs = np.full(2 * m - 1, float(M_use), dtype=np.float64)
        bounds = [(0.0, None)] * m

        res = linprog(c, A_ub=A, b_ub=rhs, bounds=bounds, method="highs")
    except Exception:
        return None

    if res is None:
        return None
    if not getattr(res, "success", False):
        status = int(getattr(res, "status", 1))
        if status not in (0, 2):
            return None

    g = np.asarray(res.x, dtype=np.float64).ravel()
    if g.size != m or not np.all(np.isfinite(g)):
        return None
    g = np.clip(g, 0.0, None)
    s = float(np.sum(g, dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-15:
        return None
    return g / s


def _mix_line_search(a, g, current_score):
    a = _finite_norm(a)
    g = _finite_norm(g)
    if a is None or g is None:
        return None, float("inf")
    best_a = a
    best_s = float(current_score)
    for t in (1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02):
        c = (1.0 - t) * a + t * g
        c = np.clip(c, 0.0, None)
        s = float(np.sum(c, dtype=np.float64))
        if not np.isfinite(s) or s <= 1e-300:
            continue
        c = c / s
        sc = _score_norm(c)
        if np.isfinite(sc) and sc < best_s:
            best_s = sc
            best_a = c
    return best_a, best_s


def _local_search(a, max_iter, rng, deadline, lp_cap):
    a = _finite_norm(a)
    if a is None:
        return None, float("inf")
    m = a.size
    cur = _score_norm(a)
    if not np.isfinite(cur):
        return None, float("inf")

    stagnation = 0
    alphas = (1e-3, 3e-3, 1e-2, 3e-2, 1e-1, 3e-1, 1.0, 3.0)

    for _ in range(max(int(max_iter), 0)):
        if time.monotonic() >= deadline:
            break

        improved = False

        # 1) LP-based Frank-Wolfe-style step.
        if m <= int(lp_cap) and linprog is not None:
            b = np.convolve(a, a)
            M = float(np.max(b))
            g = _lp_direction(a, M)
            if g is not None:
                cand, cand_s = _mix_line_search(a, g, cur)
                if cand is not None and np.isfinite(cand_s) and cand_s < cur - 1e-10:
                    a, cur = cand, cand_s
                    improved = True
                    stagnation = 0

        # 2) Exact active-set subgradient step.
        if (not improved) or stagnation > 3:
            _, grad = _active_subgradient(a)
            if grad is None or not np.all(np.isfinite(grad)):
                grad = np.zeros_like(a)
            if np.any(grad != 0.0):
                for alpha in alphas:
                    cand = a - alpha * grad
                    cand = np.clip(cand, 0.0, None)
                    s = float(np.sum(cand, dtype=np.float64))
                    if not np.isfinite(s) or s <= 1e-300:
                        continue
                    cand = cand / s
                    cand_s = _score_norm(cand)
                    if np.isfinite(cand_s) and cand_s < cur - 1e-10:
                        a, cur = cand, cand_s
                        improved = True
                        break

        if improved:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 6:
                break

            # 3) Random restart within the local neighborhood.
            rel = 0.05 + 0.30 * rng.random()
            cand = a * (1.0 + rel * rng.standard_normal(m))
            cand = np.clip(cand, 1e-12, None)
            s = float(np.sum(cand, dtype=np.float64))
            if np.isfinite(s) and s > 1e-300:
                cand = cand / s
                cand_s = _score_norm(cand)
                if np.isfinite(cand_s) and cand_s < cur:
                    a, cur = cand, cand_s
                    stagnation = 0
            else:
                cand = _random_a(m, rng)
                cand_s = _score_norm(cand)
                if np.isfinite(cand_s) and cand_s < cur:
                    a, cur = cand, cand_s
                    stagnation = 0

    return a, cur


def _iterations_for_m(m):
    m = int(m)
    if m <= 6:
        return 120
    if m <= 12:
        return 90
    if m <= 24:
        return 70
    if m <= 48:
        return 55
    if m <= 96:
        return 45
    if m <= 256:
        return 35
    if m <= 512:
        return 25
    return 15


def _optimize_m(m, rng, deadline, best_arr, best_score, lp_cap):
    m = int(m)
    if m < 1:
        return best_arr, best_score

    best_arr = _finite_norm(best_arr)
    best_score = float(best_score)

    inits = []
    if best_arr is not None:
        inits.append(_upsample(best_arr, m))
    inits.append(_finite_norm(np.ones(m, dtype=np.float64)))

    nrand = 10 if m <= 24 else 7 if m <= 96 else 5
    for _ in range(nrand):
        inits.append(_random_a(m, rng))

    for amp in (0.2, 0.35, 0.5, 0.7):
        inits.append(_harmonic(m, amp, 0.0, 0))
        inits.append(_harmonic(m, amp, math.pi, 0))
    inits.append(_harmonic(m, 0.35, 0.0, 1))
    inits.append(_harmonic(m, 0.35, math.pi, 1))

    scored = []
    for init in inits:
        a = _finite_norm(init)
        if a is None:
            continue
        sc = _score_norm(a)
        if np.isfinite(sc):
            scored.append((sc, a))

    if not scored:
        if best_arr is not None:
            return best_arr, float(best_score)
        a = np.ones(m, dtype=np.float64) / m
        return a, _score_norm(a)

    scored.sort(key=lambda z: z[0])
    max_inits = 10 if m <= 32 else 8 if m <= 128 else 6
    top = scored[:max_inits]

    local_best_arr, local_best_score = scored[0]
    if best_arr is not None and np.isfinite(best_score):
        return_arr, return_score = best_arr, best_score
    else:
        return_arr, return_score = local_best_arr, float(local_best_score)

    for sc, a in top:
        if not np.isfinite(return_score):
            pass
        elif sc > return_score + 0.4:
            continue
        if time.monotonic() >= deadline - 10:
            break

        iters = _iterations_for_m(m)
        new_a, new_s = _local_search(a, iters, rng, deadline, lp_cap)
        if new_a is not None and np.isfinite(new_s) and new_s < return_score - 1e-9:
            return_arr, return_score = new_a, float(new_s)

    return return_arr, return_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    start = time.monotonic()
    try:
        hard_deadline = start + max(1.0, float(budget_s) - 10.0)
    except Exception:
        hard_deadline = start + max(1.0, 1300.0 - 10.0)

    rng = np.random.default_rng(seed)

    # Parent construction, if provided, used only as an initialization candidate.
    parent = None
    parent_len = 4806
    try:
        p_series = height_sequence_1
        if hasattr(p_series, "size"):
            parent_len = int(p_series.size)
        p = _finite_norm(p_series)
        if p is not None and p.size > 0:
            parent = p
    except Exception:
        parent = None
        parent_len = 4806

    # Guaranteed non-uniform fallback for length 4.
    q = (3.0 - math.sqrt(3.0)) / 12.0
    p4 = 0.5 - q
    fallback = np.array([0.25, p4, q, 0.25], dtype=np.float64)
    fallback = _finite_norm(fallback)
    if fallback is None:
        fallback = np.ones(1, dtype=np.float64)

    best_arr = fallback
    best_score = _score_norm(fallback)

    if parent is not None:
        parent_score = _score_norm(parent)
        if np.isfinite(parent_score) and parent_score < best_score:
            best_arr, best_score = parent, parent_score

    lp_cap = 512
    m_list = [4, 5, 6, 7, 8, 10, 12, 16, 24, 32, 48, 64, 96, 128, 192, 256, 384, 512]

    for m in m_list:
        if time.monotonic() >= hard_deadline - 45.0:
            break
        best_arr, best_score = _optimize_m(
            m, rng, hard_deadline, best_arr, best_score, lp_cap
        )
        if not np.isfinite(best_score):
            best_arr = fallback
            best_score = _score_norm(fallback)

    # If time remains, deepen the best current representation.
    if time.monotonic() < hard_deadline - 90.0 and best_score > 1.503:
        for m in (256, 512):
            if time.monotonic() >= hard_deadline - 30.0:
                break
            if m < lp_cap or linprog is None:
                continue
            a0 = _upsample(best_arr, m)
            new_a, new_s = _local_search(a0, 80, rng, hard_deadline, lp_cap)
            if new_a is not None and np.isfinite(new_s) and new_s < best_score:
                best_arr, best_score = new_a, new_s

    # Assemble final candidates; exact evaluator is authoritative.
    candidates = [best_arr]
    if parent is not None:
        candidates.append(parent)
    if parent_len > 0 and best_arr is not None and best_arr.size != parent_len:
        candidates.append(_upsample(best_arr, parent_len))

    final_arr = fallback
    final_score = float("inf")

    for cand in candidates:
        if time.monotonic() >= hard_deadline - 1.0:
            break
        a = _finite_norm(cand)
        if a is None:
            continue
        lst = [float(x) for x in a]
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = float("inf")
        if np.isfinite(sc) and sc < final_score:
            final_score = float(sc)
            final_arr = a

    final_arr = _finite_norm(final_arr)
    if final_arr is None:
        final_arr = fallback

    return [float(x) for x in final_arr]
```
