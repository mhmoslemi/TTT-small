    # Steep ramp (quadratic increasing)
    quad = np.arange(1, n + 1, dtype=np.float64) ** 2
    inits.append(quad / quad.sum())
    # Exponential increasing
    r = 1 - 2.0 / n
    expo = r ** np.arange(n, dtype=np.float64)
    expo = expo / expo.sum()
    inits.append(expo)

    # Score all initializations
    best_a = inits[0].copy()
    best_score = min(exact_score(ip) for ip in inits)
    for ip in inits:
        sc = exact_score(ip)
        if sc < best_score:
            best_score = sc
            best_a = ip.copy()

    # Phase 1: L-BFGS-B with increasing soft-max sharpness
    betas = [15, 40, 100, 250, 600, 1500]
    current = best_a.copy()
    bounds = [(0.0, 1000.0)] * n

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        func = make_func(beta)
        try:
            res = minimize(func, current, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 400, 'ftol': 1e-14, 'gtol': 1e-7,
                                    'maxls': 30, 'maxfun': 8000, 'disp': False})
        except Exception:
            break
        if time.monotonic() >= deadline:
            break
        if res.fun < 1e9:
            current = np.clip(res.x, 0.0, 1000.0)
            s = current.sum()
            if s > 1e-10:
                current = current / s
            sc = exact_score(current)
            if np.isfinite(sc) and sc < best_score:
                best_a = current.copy()
                best_score = sc

    # Phase 2: Local search with FFT-based scoring
    if time.monotonic() < deadline:
        step_sizes = [0.15, 0.10, 0.05, 0.03, 0.015, 0.008]
        for ss in step_sizes:
            if time.monotonic() >= deadline:
                break
            iters = 0
            while time.monotonic() < deadline and iters < 8000:
                iters += 1
                pert = rng.random()
                if pert < 0.5:
                    # Multiplicative noise
                    cand = best_a * np.exp(rng.normal(0, ss, n))
                elif pert < 0.8:
                    # Block multiplicative
                    cand = best_a.copy()
                    wid = max(2, int(n * ss * 0.1))
                    st = int(rng.integers(0, n - wid + 1))
                    cand[st:st + wid] *= float(np.exp(rng.normal(0, ss * 2)))
                else:
                    # Symmetrize + noise
                    cand = 0.5 * (best_a + best_a[::-1])
                    cand *= np.exp(rng.normal(0, ss * 0.5, n))

                cand = np.clip(cand, 0.0, 1000.0)
                sc = exact_score(cand)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = cand.copy()

    # Phase 3: Fine-tune with very small steps
    if time.monotonic() < deadline - 10:
        ss = 0.003
        iters = 0
        while time.monotonic() < deadline - 10 and iters < 20000:
            iters += 1
            cand = best_a * np.exp(rng.normal(0, ss, n))
            cand = np.clip(cand, 0.0, 1000.0)
            sc = exact_score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = cand.copy()

    # Final validation with exact evaluator
    final_list = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    if not np.isfinite(final_score):
        # Fallback to uniform
        final_list = [1.0 / n] * n
        final_score = float(evaluate_sequence(final_list))

    return final_list
