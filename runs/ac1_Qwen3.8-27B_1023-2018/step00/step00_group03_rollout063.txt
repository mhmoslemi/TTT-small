def propose_candidate(seed=42, budget_s=1300, **kwargs):
    import numpy as np
    import time
    
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 10.0
    
    # Parent
    try:
        a_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    except:
        a_init = np.ones(2535, dtype=np.float64)
    
    n = a_init.size
    a = a_init.copy()
    
    # Normalize to sum = n
    s = a.sum()
    if s < 0.01:
        a = np.ones(n)
    else:
        a *= n / s
    
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    
    # FFT setup
    N = 1
    while N < 2*n - 1:
        N *= 2
    
    def autocorr(x):
        fa = np.fft.rfft(x, N)
        pw = fa * fa.conj()
        return np.fft.irfft(pw, N)[:2*n-1]
    
    lr = 0.1  # learning rate
    top_k = 3  # number of active peaks to use
    
    it = 0
    patience = 0
    
    while time.monotonic() < deadline:
        it += 1
        
        # Compute autocorrelation
        conv = autocorr(a)
        s = a.sum()
        score = 2.0 * n * conv.max() / (s * s)
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        # Find top-k lags
        # Use partial sort for efficiency
        top_indices = np.argpartition(conv, -top_k)[-top_k:]
        top_indices = top_indices[np.argsort(conv[top_indices])[::-1]]  # sorted by value desc
        
        # Compute subgradient (average over top-k active lags if desired, 
        # or just use the top one)
        g = np.zeros(n)
        for k in top_indices:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            cols = np.arange(lo, hi + 1)
            rows = k - cols
            valid = (rows >= 0) & (rows < n)
            cols = cols[valid]
            rows = rows[valid]
            # Subgradient contribution: a[rows] at positions cols
            # (Each a[m] appears with coefficient a[k-m], plus symmetry)
            np.add.at(g, cols, a[rows])
        
        g /= top_k  # average
        
        # Projected gradient step
        a = a - lr * g
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum < 0.01:
            a = best_a.copy()
            break
        a *= n / a_sum  # renormalize
        
        # Adaptive step size
        if it % 50 == 0:
            lr *= 0.995  # slow decay
        
        # Every 200 iterations, try a random perturbation
        if it % 200 == 0 and patience > 100:
            # Restart from best with perturbation
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 0)
            a *= n / a.sum()
            lr = 0.1
            patience = 0
    
    # Final validation
    final_score = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(final_score):
        # Fallback
        best_a = np.ones(n)
        final_score = float(evaluate_sequence(best_a.tolist()))
    
    return [float(x) for x in best_a]
