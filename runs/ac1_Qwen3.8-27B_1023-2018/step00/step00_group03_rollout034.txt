def full_conv_max_and_grad(a):
    """Compute max of full convolution and its gradient w.r.t. a."""
    n = len(a)
    b = np.convolve(a, a)  # length 2n-1
    k_star = np.argmax(b)
    max_b = b[k_star]
    S = np.sum(a)

    # (grad)_{m} = 2 * a_{k_star - m} if 0 <= k_star - m <= n-1, else 0
