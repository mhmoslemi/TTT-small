import time


def propose_candidate(seed=42, budget_s=1300, **kwargs):

    budget = max(1.0, float(budget_s))
    # Leave headroom for final exact evaluation and safe return.

    rng = np.random.default_rng(int(seed))

    # its entries into this source.

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        parent = None

    if parent is None or parent.size == 0:
        parent = np.ones(6962, dtype=np.float64)

    # become quadratic-ish and unsafe.  Fall back to a moderate synthetic size.
    if parent.size > 9000 or parent.size < 64:
        parent = np.ones(6962, dtype=np.float64)

    n = int(parent.size)
    L = 2 * n - 1
    need = 3 * n - 2
    N = 1
    while N < need:
        N <<= 1

    rfft = np.fft.rfft
    irfft = np.fft.irfft

    def _as_normalized_nonnegative(raw):
        x = np.asarray(raw, dtype=np.float64)
        if x.ndim != 1 or x.size == 0:
            x = np.ones(n, dtype=np.float64)
        x = x.astype(np.float64, copy=True)
        x[~np.isfinite(x)] = 0.0
        x[x < 0.0] = 0.0
        x[x > 1e300] = 1e300
        s = float(np.sum(x))
        if not np.isfinite(s) or s <= 0.0:
            x = np.ones(n, dtype=np.float64)
            s = float(n)
        x *= (1.0 / s)
        return x

    def _simplex_projection(v, target=1.0):
        v = np.asarray(v, dtype=np.float64)
        m = v.size
        if m == 0:
            return v
        # Fast path: already in the nonnegative simplex up to tiny error.
        if np.all(np.isfinite(v)) and v.min() >= -1e-12:
            w = np.maximum(v, 0.0)
            s = float(np.sum(w))
            if s > 0.0 and abs(s - target) <= 1e-10:
                return w * (target / s)
        if not np.all(np.isfinite(v)):
            return np.full(m, target / m, dtype=np.float64)

        u = np.sort(v)[::-1]
        rho = np.arange(1, m + 1, dtype=np.float64)
        css = np.cumsum(u)
        cond = u - (css - target) / rho > 0.0
        idx = np.nonzero(cond)[0]
        if idx.size == 0:
            return np.full(m, target / m, dtype=np.float64)
        k = int(idx[-1]) + 1
        theta = (css[k - 1] - target) / float(k)
        w = v - theta
        w[w < 0.0] = 0.0
        s = float(np.sum(w))
        if not np.isfinite(s) or s <= 0.0:
            return np.full(m, target / m, dtype=np.float64)
        return w * (target / s)

    def _convolution_and_cmax(x):
        # x is assumed to have sum exactly 1.
        X = rfft(x, N)
        c = irfft(X * X, N)[:L]
        c = np.maximum(c, 0.0)
        cmax = float(np.max(c)) if c.size else 0.0
        if not np.isfinite(cmax) or cmax < 0.0:
            cmax = 0.0
        return c, cmax

    def _fft_score_normalized(x):
        _, cmax = _convolution_and_cmax(x)
        return 2.0 * float(n) * cmax

    def _gradient_smooth_max(x, alpha_target=30.0):
        """
        Returns (cmax, grad), where grad is the gradient of the smooth max
        log-sum-exp(alpha * c) with p = softmax(alpha*(c-cmax)) normalized to
        sum 1.  For x with sum=1 this is the useful direction.
        """
        X = rfft(x, N)
        c = irfft(X * X, N)[:L]
        c = np.maximum(c, 0.0)
        cmax = float(np.max(c)) if c.size else 0.0
        if not np.isfinite(cmax) or cmax <= 0.0:
            return 0.0, np.zeros(n, dtype=np.float64)

        alpha = float(alpha_target) / max(cmax, 1e-300)
        y = alpha * (c - cmax)
        # y is <= 0.  Underflow is harmless and focuses on active peaks.
        p = np.exp(y)
        psum = float(np.sum(p))
        if not np.isfinite(psum) or psum <= 0.0:
            K = min(32, L)
            p = np.zeros(L, dtype=np.float64)
            if K <= 0:
                p[0] = 1.0
                psum = 1.0
            else:
                top = np.argpartition(c, -K)[-K:]
                p[top] = 1.0
                psum = float(K)
        else:
            p *= (1.0 / psum)

        # g_i = 2 * sum_t x[t] * p[i+t].  This is a cross-correlation; compute
        # it by convolving x with reversed p and taking a reversed slice.
        q = p[::-1]
        Pq = rfft(q, N)
        yfull = irfft(X * Pq, N)
        if n == 1:
            g = 2.0 * np.array([float(yfull[0])] if yfull.size else [0.0])
        else:
            g = 2.0 * yfull[2 * n - 2 : n - 2 : -1]
        if g.size != n:
            # Very defensive fallback; should not happen for the supported sizes.
            g = np.zeros(n, dtype=np.float64)
        g = np.asarray(g, dtype=np.float64)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n, dtype=np.float64)
        return cmax, g

    # only at final validation.
    store = []

    def _store_candidate(score, x):

            sc = float(score)
        except Exception:
            return
        if not np.isfinite(sc):
            return

            return

        store.append((sc, np.array(x, dtype=np.float64, copy=True)))
        if len(store) > 60:
            store.sort(key=lambda it: it[0])
            del store[-1:]

    candidates = []

    def _consider(raw):
        if time.monotonic() > hard_deadline - 30.0:
            return
        x = _as_normalized_nonnegative(raw)
        if x.size != n:
            return
        sc = _fft_score_normalized(x)
        if np.isfinite(sc):
            candidates.append((sc, x))

    # Parent and uniform reference.
    parent_norm = _as_normalized_nonnegative(parent)
    _store_candidate(_fft_score_normalized(parent_norm), parent_norm)
    _consider(parent_norm)
    _consider(np.ones(n, dtype=np.float64))

    idx = np.arange(n, dtype=np.float64)

    # One-sided power laws, including square-root tails.  These already provide
    # strong non-uniform starts and are cheap to evaluate with FFTs.

    if param_limit < start + 1.0:

            _consider(raw[::-1])

            if time.monotonic() > param_limit:
                break
            sq = np.sqrt(idx + 0.5)
            for A in (0.0, 0.25, 0.75, 2.0, 5.0, 12.0, 30.0, 80.0, 250.0):
                raw = (sq + A) ** (-p)
                _consider(raw)
                _consider(raw[::-1])

    if time.monotonic() < param_limit:

            P = max(1, n // m)
            frac = (idx % float(P)) / float(P)
            for A in (0.05, 0.2, 1.0):

                _consider(raw)
                _consider(((1.0 - frac) + A) ** (-0.5))

        for a in (-0.9, -0.7, -0.5, -0.35, -0.2, -0.1, 0.0, 0.1, 0.3, 0.6, 0.9):

            raw2 = 1.0 - a * t
            _consider(raw1)
            _consider(raw2)

        for m in (1, 2, 3, 5, 8, 13):
            for r in (0.2, 0.45, 0.7, 0.9):
                if time.monotonic() > param_limit:
                    break

                for ph in (0.0, 0.6, 1.2, 1.8):
                    raw = 1.0 + r * np.cos(th + ph)
                    _consider(raw)

    # A few smooth random restarts (cumulative random walk, then square).
    if time.monotonic() < param_limit:
        for _ in range(6):
            z = np.cumsum(rng.standard_normal(n))
            z -= np.min(z)
            z += 1e-12
            z = z / max(1e-12, np.max(z))
            _consider(np.square(z))
            _consider(np.square(z[::-1]))

    candidates.sort(key=lambda it: it[0])
    # Keep the best parametric starts, and make sure the very best are stored.
    candidates = candidates[:80]
    for sc, x in candidates[:12]:
        _store_candidate(sc, x)

    # --- Local smoothing-max optimizer ---------------------------------------
    def _optimize_one(x0, end_time):
        x = _as_normalized_nonnegative(x0)
        xd = x.copy()

        m_adv = np.zeros(n, dtype=np.float64)
        v_adv = np.zeros(n, dtype=np.float64)
        t = 0
        lr = 0.02
        min_lr = 0.0018
        beta1 = 0.90
        beta2 = 0.999
        eps = 1e-9

        c0, cmax0 = _convolution_and_cmax(x)
        local_best_score = float(2.0 * n * cmax0)
        local_best = x.copy()
        _store_candidate(local_best_score, local_best)

        no_improve = 0

        while time.monotonic() < end_time:
            alpha_target = min(200.0, 18.0 + 0.08 * t)
            _, cmax, g = _gradient_smooth_max(x, alpha_target=alpha_target)
            score = float(2.0 * n * cmax)

            if score < local_best_score - 1e-11:
                local_best_score = score
                local_best = x.copy()
                no_improve = 0
                _store_candidate(local_best_score, local_best)
            else:
                no_improve += 1

            # Escape from stagnant basins by restarting from the current local
            # best with a controlled smoothing/noise injection.
            if no_improve > 350:
                frac = min(0.35, 0.08 + 0.0005 * no_improve)
                noise = rng.standard_normal(n)
                noise = np.maximum(noise, -3.0)
                cand = (1.0 - frac) * local_best + frac * np.maximum(0.05 + noise, 0.0)
                x = _simplex_projection(cand)
                m_adv.fill(0.0)
                v_adv.fill(0.0)
                t = 0
                no_improve = 0
                lr = 0.02
            elif no_improve > 120:
                cand = (1.0 - 0.05) * x + 0.05 * np.maximum(0.01 + rng.standard_normal(n) * 0.05, 0.0)
                x = _simplex_projection(cand)
                no_improve = 0

            # Tangential gradient (sum zero), so the simplex projection mainly
            # handles nonnegativity.
            gm = g - float(np.mean(g))
            t += 1
            b1d = 1.0 - beta1 ** t
            b2d = 1.0 - beta2 ** t
            m_adv = beta1 * m_adv + (1.0 - beta1) * gm
            v_adv = beta2 * v_adv + (1.0 - beta2) * (gm * gm)
            mh = m_adv / b1d
            vh = v_adv / b2d
            step = lr * (mh / (np.sqrt(vh) + eps))

            # Preserve diversity occasionally.

                    x = _simplex_projection(0.80 * x + 0.20 / float(n))

            lr = max(min_lr, lr * 0.99985)

        return local_best_score, local_best, x

    # Select the most promising starts.
    top_inits = []
    for sc, x in candidates:
        if sc < 2.001 and len(top_inits) < 4:
            top_inits.append(x)
    if not top_inits:
        top_inits.append(candidates[0][1] if candidates else parent_norm)

    remaining = max(0.0, hard_deadline - time.monotonic() - 45.0)
    if remaining > 30.0:
        if len(top_inits) == 1:
            weights = (1.0,)
        else:
            weights = (0.55, 0.25, 0.15, 0.05)[:len(top_inits)]
            sw = float(sum(weights))
            weights = tuple(w / sw for w in weights)

        for init, w in zip(top_inits, weights):
            if time.monotonic() >= hard_deadline - 45.0:
                break
            allow = min(remaining * w, max(20.0, remaining * 0.8))
            end_t = min(time.monotonic() + allow, hard_deadline - 45.0)
            if end_t <= time.monotonic() + 5.0:
                continue
            # Restart from any newly found best occasionally.
            if store:
                store.sort(key=lambda it: it[0])
                if store[0][0] <= _fft_score_normalized(_as_normalized_nonnegative(init)):

        if store and time.monotonic() < hard_deadline - 45.0:
            store.sort(key=lambda it: it[0])
            polish_end = min(time.monotonic() + (hard_deadline - time.monotonic() - 30.0) * 0.75, hard_deadline - 30.0)
            if polish_end > time.monotonic() + 10.0:
                _optimize_one(store[0][1], polish_end)

    # Include parent, optimizer store, and the top parametric starts.
    final_candidates = []
    final_candidates.append(parent_norm)
    if store:
        store.sort(key=lambda it: it[0])
        final_candidates.extend([x for _, x in store[:30]])
    final_candidates.extend([x for _, x in candidates[:10]])

    # De-duplicate roughly by first differing value isn't needed; just evaluate
    # in order of FFT score.
    scored = []
    for x in final_candidates:
        x = _as_normalized_nonnegative(x)
        sc = _fft_score_normalized(x)
        scored.append((sc, x))
    scored.sort(key=lambda it: it[0])
    scored = scored[:35]

    # Scale candidates to a safe, numerically comfortable sum before exact
    # testing.  The objective is scale-invariant until the [0, 1000] cap, and
    # these sums are far below the cap.
    best_out = None
    best_exact = np.inf

    def _exact_score_scaled(x_norm):
        x = np.asarray(x_norm, dtype=np.float64).copy()

            return None
        mx = float(np.max(x)) if x.size else 0.0

        if mx > 0.0:

        x *= scale
        x = np.clip(x, 0.0, 1000.0)
        if not np.all(np.isfinite(x)) or float(np.sum(x)) < 0.01:
            return None
        try:
            sc = float(evaluate_sequence([float(v) for v in x]))
        except Exception:

        if np.isfinite(sc):
            return x, sc
        return None

    for _, x in scored:

            break
        res = _exact_score_scaled(x)
        if res is None:
            continue
        x_out, sc = res
        if sc < best_exact:
            best_exact = sc
            best_out = x_out

    # If all exact evaluations failed (should not happen), return the parent
    # scaled safely.

        best_out = np.asarray(parent, dtype=np.float64).copy()

        else:

    s = float(np.sum(best_out))

    # Final guard so the harness receives finite floats.
    return [float(v) for v in best_out]
