```python
def solve_ac1(n, max_iter=100, tol=1e-6):
    """Solve min max_k (a*a)_k subject to sum(a)=1, a>=0 using SLP."""
    # Initialize
    a = np.ones(n) / n
    M = 0.0
    
    for it in range(max_iter):
        # Compute current autocorrelation
        bb = fftconvolve(a, a)
        M_current = bb.max()
        
        # Solve LP:
        # min M
        # s.t. 2*(a * a_new)_k <= M + bb_k for all k
        #      sum(a_new) = 1
        #      a_new >= 0
        #      M >= 0
        #
        # In standard form:
        # min [0, 0, ..., 0, 1] · [a_new, M]
        # s.t. 2*A_a * a_new - M <= bb  (where A_a is the conv matrix of a)
        #      sum(a_new) = 1
        #      a_new >= 0, M >= 0
        
        # Build conv matrix: (A_a @ a_new)_k = (a * a_new)_k
        # A_a is (2n-1) x n, A_a[k,j] = a[k-j] if 0<=k-j<n else 0
        
        # LP: min M
        # Variables: x = [a_new (n), M (1)], total n+1 variables
        # Objective: [0,...,0, 1]
        
        # Inequality constraints: 2*(A_a @ a_new) - M <= bb
        # In matrix form: [2*A_a | -I] @ x <= bb
        
        # Equality: sum(a_new) + 0*M = 1
        # [1,...,1 | 0] @ x = 1
        
        # Nonnegativity: x >= 0
        
        A = np.zeros((2*n-1, n))
        for j in range(n):
            A[j:j+n, j] = a
        
        # Full constraint matrix for linprog
        # A_ub @ x <= b_ub
        A_ub = np.hstack([2*A, -np.ones((2*n-1, 1))])
        b_ub = bb
        # Add nonnegativity for M (a_new >= 0 handled by bounds)
        # Actually linprog handles bounds separately
        
        # A_eq @ x = b_eq
        A_eq = np.hstack([np.ones((1, n)), [[0]]])
        b_eq = np.array([1.0])
        
        # Bounds: a_new >= 0 (lower=0), M >= 0 (lower=0)
        c_obj = np.zeros(n+1)
        c_obj[-1] = 1.0
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0, None)]*(n+1), method='highs')
        
        if not res.success:
            break
        
        a_new = res.x[:n]
        M_new = res.x[-1]
        
        # Check convergence
        bb_new = fftconvolve(a_new, a_new)
        M_actual = bb_new.max()
        
        if abs(M_actual - M_new) < tol * max(1, M_new):
            a = a_new
            break
        
        # Update
        a = a_new
    
    return a, M_actual
```

So the objective is a mix of convex and non-convex terms. The non-convexity arises from the cross-terms.
