    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 >= size for efficient FFT
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = rfft(a, fft_size)
    c = irfft(A * A, fft_size)[:size]
    return c

    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (s * s)

from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fft_convolve_self(a):
    """Full linear convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    c = irfft(A * A, fft_size)[:size]
    # Clean up numerical noise
    c[np.abs(c) < 1e-12] = 0.0
    return c
