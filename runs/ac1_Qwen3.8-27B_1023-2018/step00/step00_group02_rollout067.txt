def _full_linear_conv(a, N):
    """Linear (full) autocorrelation a*a of nonneg vector a, via FFT. N >= len(a)+len(a)-1."""
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[: 2 * len(a) - 1]


def _subgrad_from_top(a, c, weights, top_idx):
    """grad_i = 2 * sum_k w_k * a[k-i]  (zeros out-of-range), restricted to top lags."""
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    for k in top_idx:
        wv = weights[k]
        wv = wv * 2.0
        if wv <= 0.0:

        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        a_slice = a[k - hi : k - lo + 1][::-1]
        g[lo : hi + 1] += wv * a_slice
    return g
