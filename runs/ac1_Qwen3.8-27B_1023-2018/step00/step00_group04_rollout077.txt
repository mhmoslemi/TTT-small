import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _build_conv_matrix(x):
    """Build sparse (2n-1) x n matrix A where A[k,j] = x[k-j] if 0<=k-j<n."""
    nn = len(x)
    cols = np.repeat(np.arange(nn), nn)
    rows = np.tile(np.arange(nn), nn) + cols
    vals = np.tile(x, nn)
    return csr_matrix((vals, (rows, cols)), shape=(2 * nn - 1, nn))


def _subgradient_step(x, eta):
    """One multiplicative subgradient step to reduce max auto-convolution."""
    n = len(x)
    b = np.convolve(x, x)
    max_b = b.max()
    if max_b < 1e-20:
        return x.copy()
    K_star = np.where(b >= max_b * (1.0 - 1e-9))[0]
    s = np.zeros(n)
    for k in K_star:
        if k < n:
            s[0:k + 1] += x[k::-1]
        else:
            s[k - n + 1:n] += x[n - 1:k - n:-1]
    s /= len(K_star)
    # Multiplicative update
    scaled = eta * s / (max_b + 1e-30)
    x_new = x * np.exp(-scaled)
    x_new = np.maximum(x_new, 1e-30)
    x_new /= x_new.sum()
    return x_new
