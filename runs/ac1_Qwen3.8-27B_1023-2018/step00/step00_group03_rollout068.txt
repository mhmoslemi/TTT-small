            for k in range(2 * n - 1):
                jlo = max(0, k - n + 1)
                jhi = min(k, n - 1)
                for j in range(jlo, jhi + 1):
                    v = a[k - j]
                    if v > 1e-14:
                        rows.append(k)
                        cols.append(j)
                        vals.append(v)
            F = coo_matrix((np.array(vals, dtype=np.float64),
                            (np.array(rows, dtype=np.int32),
                             np.array(cols, dtype=np.int32))),
                           shape=(2 * n - 1, n)).tocsr()

            from scipy.optimize import linprog
            rem = max(2.0, deadline - time.monotonic() - 60.0)
            res = linprog(
                c=-np.ones(n),
                A_ub=F,
                b_ub=np.full(2 * n - 1, M),
                bounds=[(0.0, None)] * n,
                method='highs',
                options={'presolve': True, 'time_limit': rem}
            )

            if time.monotonic() > deadline:
                break

            if res.status == 0 and res.x is not None:
                g = np.maximum(res.x, 0.0)
                gs = g.sum()
                if gs > 1e-10:
                    g = g * (n / gs)
                    cur_sc = score_fast(a)
                    for t_mix in [1.0, 0.6, 0.4, 0.2]:
                        a_new = (1.0 - t_mix) * a + t_mix * g
                        a_new = np.maximum(a_new, 0.0)
                        ns = a_new.sum()
                        if ns > 1e-10:
                            a_new = a_new * (n / ns)
                            if score_fast(a_new) < cur_sc:
                                a = a_new
                                update_best(a)
                                break
        except Exception:
            break
        lp_count += 1

    # Phase 2: Gradient descent with log-sum-exp smoothing
    alpha_ls = 25.0
    lr = 1.0
    stall = 0
    grad_iter = 0
    while in_time():
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break
        if grad_iter % 200 == 0 and remaining < 60.0:
            break

        b = fftconvolve(a, a)
        bmax = b.max()
        shifted = np.minimum(alpha_ls * (b - bmax), 50.0)
        w = np.exp(shifted)
        wsum = w.sum()
        if wsum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(b))] = 1.0
        else:
            w /= wsum

        conv_grad = fftconvolve(w, a[::-1])
        grad = 2.0 * conv_grad[n - 1: 2 * n - 1]
        grad = grad - grad.mean()

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        ns = a_new.sum()
        if ns < 1e-10:
            lr *= 0.5
            stall += 1
            grad_iter += 1
            continue
        a_new = a_new * (n / ns)

        sc_new = score_fast(a_new)
        sc_cur = score_fast(a)

        if sc_new < sc_cur * (1.0 - 1e-9):
            a = a_new
            update_best(a)
            lr = min(lr * 1.02, 8.0)
            stall = 0
        else:
            lr *= 0.8
            stall += 1
            if stall > 150:
                alpha_ls = min(alpha_ls * 1.8, 80.0)
                if alpha_ls > 60.0:
                    alpha_ls = 20.0
                    a = a + 0.03 * rng.standard_normal(n)
                    a = np.maximum(a, 0.0)
                    ns = a.sum()
                    if ns > 0:
                        a = a * (n / ns)
                    lr = 1.0
                stall = 0
        grad_iter += 1

    # Phase 3: Coordinate descent fine-tuning
    cd_deadline = deadline - 10.0
    for sweep in range(200):
        if time.monotonic() > cd_deadline:
            break
        cur_sc = score_fast(a)
        any_change = False
        for i in range(n):
            if time.monotonic() > cd_deadline:
                break
            old = a[i]
            if old < 1e-10:
                # Try setting to small positive
                a[i] = 0.5
                ns = a.sum()
                if ns > 0:
                    a[i] = a[i] * (n / ns)
                    if score_fast(a) < cur_sc - 1e-10:
                        cur_sc = score_fast(a)
                        any_change = True
                        update_best(a)
                a[i] = 0.0
            else:
                for factor in [0.5, 0.8, 1.2, 1.5]:
                    a[i] = old * factor
                    a[i] = max(0.0, min(1000.0, a[i]))
                    ns = a.sum()
                    if ns < 0.01:
                        a[i] = old
                        continue
                    sc_try = score_fast(a)
                    if sc_try < cur_sc - 1e-10:
                        cur_sc = sc_try
                        any_change = True
                        update_best(a)
                        break
                    a[i] = old
                # Try zeroing
                a[i] = 0.0
                ns = a.sum()
                if ns >= 0.01 and score_fast(a) < cur_sc - 1e-10:
                    cur_sc = score_fast(a)
                    any_change = True
                    update_best(a)
            a[i] = np.minimum(max(a[i], 0.0), 1000.0)
        if not any_change and sweep > 5:
            break

    # Final validation
    final = np.clip(best, 0.0, 1000.0)
    result = [float(x) for x in final]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [float(x) for x in np.clip(np.asarray(height_sequence_1), 0, 1000)]
        fs = evaluate_sequence(result)
        if not np.isfinite(fs):
            result = [1.0]

    return result
