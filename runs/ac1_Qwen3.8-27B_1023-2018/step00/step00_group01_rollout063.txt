import time

def propose_candidate(seed=42, budget_s=1300, **kwargs):

    deadline = time.monotonic() + max(0.0, float(budget_s)) - 10.0

    n = len(height_sequence_1)
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).copy()

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # FFT size for linear convolution/correlation of lengths n and 2n-1
    N_fft = 1
    while N_fft < 3 * n - 2:
        N_fft *= 2

    k_idx = np.arange(n)

    def eval_arr(a):
        s = a.sum()
        if s < 0.01:
            return np.inf
        b = np.convolve(a, a, mode='full')
        return float(2.0 * n * b.max() / (s * s))

    def optimize_shape(a_init, iters, beta, eta0):
        nonlocal best, best_score
        a = a_init.copy()
        s = a.sum()
        if s < 0.01:
            return a
        a *= n / s  # normalize mean to 1
        a = np.clip(a, 0.0, 1000.0)

        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score

        eta = eta0
        prev_peak = None

        for it in range(iters):
            if time.monotonic() > deadline:
                break

            # Autocorrelation via FFT
            F = np.fft.rfft(a, N_fft)
            B_full = np.fft.irfft(F * np.conj(F), N_fft)
            B = B_full[:2 * n - 1]

            if B.max() <= 0:
                break

            # Softmax weights emphasizing peaks
            bmax = B.max()
            w = np.exp(beta * (B - bmax))
            wsum = w.sum()
            if wsum < 1e-300:
                w = np.zeros(2 * n - 1)
                w[np.argmax(B)] = 1.0
            else:
                w /= wsum

            # Gradient: grad_j = sum_m a(m) * w(m+j)
            a_pad = np.zeros(N_fft)
            a_pad[:n] = a
            w_pad = np.zeros(N_fft)
            w_pad[:2 * n - 1] = w

            FA = np.fft.rfft(a_pad)
            FW = np.fft.rfft(w_pad)

            if prev_peak is not None and new_peak > prev_peak * 1.02:

                if eta < 1e-5:
                    eta = 0.1
                    beta *= 0.8
            else:
                if prev_peak is not None and new_peak < prev_peak * 0.99:
                    eta = min(eta * 1.1, 5.0)
                a = a_new
                prev_peak = new_peak

            # Periodic exact evaluation
            if it % 200 == 199 or it == iters - 1:

                if np.isfinite(score) and score < best_score:
                    best = a.copy()

    for _ in range(200):

        start = int(rng.integers(0, max(1, n - width + 1)))
