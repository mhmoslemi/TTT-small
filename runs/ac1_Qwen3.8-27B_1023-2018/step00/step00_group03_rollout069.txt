import numpy as np

def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum x = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    r = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (cv - 1.0) / r
    idx = np.nonzero(cond)[0]
    if idx.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = idx[-1] + 1
    theta = (cv[k - 1] - 1.0) / k
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.full(n, 1.0 / n, dtype=np.float64)
    return out


def _surrogate(a, b, tau):
    bm = b.max()
