    from scipy.optimize import linprog
except Exception:
    linprog = None


def _eval_exact(a):
    try:
        return float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        return float("inf")


def _conv(a):
    return np.convolve(a, a)


def _bump(n, mu, sig, power, skew, base, T):
    t = np.arange(n, dtype=np.float64)
    u = t / max(1, n - 1)
    d = (u - mu) / max(1e-9, sig)
    a = np.maximum(0.0, (1.0 - d * d) ** power)
    a = a * (1.0 + skew * (u - 0.5))
    a = np.clip(a, 0.0, None) + base
    a = a / a.sum() * T
    return a


def _lp_direction(f, n, T, M):
    """Return g (sum=T) maximizing sum s.t. (f*g)_k <= M, g>=0, via LP."""
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for k in range(L):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi < lo:
            continue
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    b = np.full(L, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                  method="highs",
                  options={"presolve": True, "dual_feasibility_tolerance": 1e-7})
    if res is None or not res.success or res.x is None:
        return None
    g = np.maximum(res.x, 0.0)
    s = g.sum()
    if not np.isfinite(s) or s <= 1e-12:
        return None
    return g / s * T


def _subgrad_step(f, n, T):
