```python
import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve(a, b):
    """Full convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 2 * max(n, m)  # next power of 2 or just 2*max for safety
    # Actually need at least n+m-1
    size = 1
    while size < n + m - 1:
        size *= 2
    A = fft(a, size)
    B = fft(b, size)
    return np.real(ifft(A * B))[:n+m-1]

def compute_score(a):
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    M = np.max(c)
    return 2.0 * n * M / (S * S)
```
