def _project_simplex(v, S=1.0):
    """Project v onto the simplex {x >= 0, sum x = S}."""
    n = len(v)
    u = v / S  # work with sum = 1
    # Sort in decreasing order
    idx = np.argsort(u)[::-1]
    u_sorted = u[idx]
    # Find rho
    cssv = np.cumsum(u_sorted)
    cond = u_sorted * np.arange(1, n+1) > (cssv - 1)

        if no_improve > 30 and rng.random() < 0.1:

            a_pert = np.clip(a_pert, 0, None)
            sp = np.sum(a_pert)
            if sp > 1e-10:
                a_pert /= sp
                sc_p = fast_score(a_pert)
                if sc_p < best_score:
                    best = a_pert.copy()
                    best_score = sc_p
                    a = a_pert

    # Final validation with exact evaluator
    final_candidate = best.tolist()
    final_score = float(evaluate_sequence(final_candidate))
    if not np.isfinite(final_score) or final_score > 2.0:
        # Fallback: return parent (score=2)
        parent_list = [float(x) for x in a_parent]
        ps = float(evaluate_sequence(parent_list))
        if np.isfinite(ps):
            return parent_list
        # Last resort: constant sequence
        return [1.0] * 256

    return final_candidate
