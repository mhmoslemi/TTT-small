import numpy as np
import time


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(2.0, float(budget_s) - 6.0)

    # Determine length from parent
    try:
        n = int(np.asarray(height_sequence_1).shape[0])
    except Exception:
        n = 5582
    if n < 1:
        n = 5582

    # --- Key mathematical insight ---
    # For non-negative a (which the evaluator enforces via max(0,x)):
    #   (a*a)_k = sum_i a_i * a_{i+k}  for lag k
    # By Cauchy-Schwarz:  (a*a)_k <= sqrt(sum a^2) * sqrt(sum a^2) = sum a^2 = (a*a)_0
    # So max_k (a*a)_k = (a*a)_0 = sum_i a_i^2 always.
    # Score = 2n * sum(a^2) / sum(a)^2 >= 2n * (sum(a)^2/n) / sum(a)^2 = 2
    # Equality holds IFF a is uniform.  The uniform sequence is the unique global optimum.
    # -------------------------------------------------------------------

    # Start with the unique optimum: uniform sequence
    best = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # Check parent for completeness (should score >= 2)
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size == n and np.isfinite(parent).all():
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps) and ps < best_score:
                best, best_score = parent.copy(), ps
    except Exception:
        pass

    # --- Gradient-based refinement (serves as verification) ---
    # Analytical gradient of f(a) = 2n*sum(a^2)/S^2 where S=sum(a):
    #   df/da_j = (4n / S^3) * (a_j*S - sum(a^2))
    # This is zero iff a_j = S/n for all j (uniform).  For non-negative a,
    # the only minimum is at uniformity.  We use this to nudge if not exactly 2.
    rng = np.random.default_rng(seed)
    it = 0
    while time.monotonic() < deadline and it < 500:
        it += 1
        if best_score <= 2.0 + 1e-10:
            break  # At theoretical minimum

        S = float(np.sum(best))
        if S < 1e-12:
            break
        a2 = float(np.sum(best * best))
        # Normalized gradient direction (points toward increasing score if a_j > S/n)
        grad = best * S - a2  # = S*(a_j - S/n)
        # Step toward uniformity
        lr = 0.05 / max(1.0, it * 0.01)
        cand = np.maximum(0.0, best - lr * grad / S)
        cand = np.minimum(1000.0, cand)

        cs = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cs) and cs < best_score - 1e-14:
            best, best_score = cand, cs
        else:
            # Try random perturbation as exploration
            if it % 10 == 0:
                pert = best * np.exp(rng.normal(0.0, 0.005, n))
                pert = np.clip(pert, 0.0, 1000.0)
                ps2 = float(evaluate_sequence(pert.tolist()))
                if np.isfinite(ps2) and ps2 < best_score:
                    best, best_score = pert, ps2
                else:
                    break  # No further improvement found

    # --- Final validation with exact evaluator ---
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        best = np.ones(n, dtype=np.float64)
        final_score = float(evaluate_sequence(best.tolist()))

    if final_score > 2.0 + 1e-6:

        us = float(evaluate_sequence(uni.tolist()))
        if np.isfinite(us) and us < final_score:
            best, final_score = uni, us

    return [float(x) for x in best]
