import time

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0

        parent = np.ones(n)

    best = parent.copy().astype(np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    
    def inline_score(a):
        """Fast score computation (same as evaluator but no type checking)."""
        S = a.sum()
        if S < 0.01:
            return np.inf
        b = np.convolve(a, a, mode='full')
        M = b[0:].max()  # full array max
        return 2.0 * n * M / (S * S)
    
    def grad_step(a, lr):
        """One gradient descent step. Returns updated a."""
        S = a.sum()
        if S < 1e-15:
            return a
        
        b = np.convolve(a, a, mode='full')
        M = b.max()
        m_star = int(np.argmax(b))
        
        # Gradient: d(2nM/S^2)/da_k
        # = (2n/S^2) * dM/da_k - (4nM/S^3)
        # dM/da_k = 2*a[m_star-k] if 0<=m_star-k<n else 0
        c1 = 4.0 * n / (S * S)   # 2n/S^2 * 2
        c2 = 4.0 * n * M / (S * S * S)  # 4nM/S^3
        
        g = np.full(n, -c2)
        lo = max(0, m_star - n + 1)
        hi = min(n - 1, m_star)
        if lo <= hi:
            g[lo:hi+1] += c1 * a[m_star - hi : m_star - lo + 1][::-1]
        
        a = a - lr * g
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        return a
    
    def run_gd(a, lr_init, n_steps, decay):
        """Run gradient descent from initialization a."""
        local_best = a.copy()
        local_best_score = inline_score(a)
        lr = lr_init

                    local_best = a.copy()
            
            lr *= decay
        
        return local_best, local_best_score
