    fft_size = 1 << (2 * n - 1).bit_length()
    full_len = 2 * n - 1

    def autocorr(a):
        F = np.fft.rfft(a, fft_size)
        power = F.real * F.real + F.imag * F.imag
        c = np.fft.irfft(power, fft_size)
        return c[:full_len]

    def fast_score_kstar(a):
        c = autocorr(a)
        S = a.sum()
        if S < 0.01:
            return np.inf, 0, 0.0, c
        k_star = int(np.argmax(c))
        max_c = c[k_star]
        score = 2.0 * n * max_c / (S * S)
        return score, k_star, max_c, c

    def grad_ac1(a, k_star, max_c, S):
        # dAC1/da_j = 4n*(S*a_{k*-j} - max_c)/S^3 where a_m=0 for invalid m
        # a_{k*-j} valid when 0 <= k*-j < n
        s3 = S * S * S
        neg_term = -4.0 * n * max_c / s3
        grad = np.full(n, neg_term, dtype=np.float64)
        j = np.arange(n)
        idx = k_star - j
        valid = (idx >= 0) & (idx < n)
        if np.any(valid):
            grad[valid] += 4.0 * n * S * a[idx[valid]] / s3
        return grad

    def grad_multi(a, c, S, n_top=5):
        # Smooth gradient using top-n_top lags
        s3 = S * S * S
        # Find indices of top n_top values
        if full_len > n_top:
            top_idx = np.argpartition(c, -n_top)[-n_top:]
        else:
            top_idx = np.arange(full_len)
        grad = np.zeros(n, dtype=np.float64)
        for k in top_idx:
            ck = c[k]
            neg = -4.0 * n * ck / s3
            grad += neg
            j = np.arange(n)
            idx = k - j
            valid = (idx >= 0) & (idx < n)
            if np.any(valid):
                grad[valid] += 4.0 * n * S * a[idx[valid]] / s3
        grad /= len(top_idx)
        return grad

    # Optimization: try multiple perturbation strategies
    for attempt in range(30):
        if time.monotonic() > deadline:
            break

        a = best.copy().astype(np.float64)
        mean_val = a.mean()
        if mean_val < 1e-6:
            continue

        # Different perturbation patterns
        if attempt == 0:
            f = np.zeros(n)
            f[:n // 2] = 1.0
            f[n // 2:] = -1.0
            a += 0.05 * mean_val * f
        elif attempt == 1:
            t = np.arange(n, dtype=np.float64) / n
            a += 0.05 * mean_val * np.cos(2 * np.pi * t)
        elif attempt == 2:
            t = np.arange(n, dtype=np.float64) / n
            a += 0.05 * mean_val * np.cos(4 * np.pi * t)
        elif attempt == 3:
            t = np.arange(n, dtype=np.float64) / n
            a += 0.05 * mean_val * np.sin(2 * np.pi * t)
        elif attempt == 4:
            t = np.arange(n, dtype=np.float64) / n
            a += 0.05 * mean_val * np.cos(6 * np.pi * t)
        elif attempt == 5:
            f = np.zeros(n)
            f[:n // 3] = 1.0
            f[n // 3:2 * n // 3] = -1.0
            f[2 * n // 3:] = 0.0
            a += 0.08 * mean_val * f
        elif attempt == 6:
            noise = rng.standard_normal(n)
            # Smooth
            k_half = max(1, n // 100)
            kernel = np.ones(2 * k_half + 1) / (2 * k_half + 1)
            noise = np.convolve(noise, kernel, mode='same')
            noise -= noise.mean()
            a += 0.05 * mean_val * noise
        elif attempt == 7:
            t = np.arange(n, dtype=np.float64) / n
            a += 0.08 * mean_val * np.cos(2 * np.pi * 1.5 * t)
        else:
            freq = rng.uniform(0.5, 8.0)
            phase = rng.uniform(0.0, 2.0 * np.pi)
            t = np.arange(n, dtype=np.float64) / n
            a += 0.05 * mean_val * np.cos(2 * np.pi * freq * t + phase)

        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 0.01:
            continue

        # Check if perturbation already improved
        init_score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(init_score) and init_score < best_score:
            best_score = init_score
            best = a.copy()

        # Gradient descent with multi-lag smoothing
        alpha = 0.01
        momentum = np.zeros(n, dtype=np.float64)
        beta_m = 0.3
        no_improve = 0
        max_no_improve = 300
        cur_score = init_score if np.isfinite(init_score) else fast_score_kstar(a)[0]

        for it in range(50000):
            if time.monotonic() > deadline:
                break
            if no_improve > max_no_improve:
                break

            score, k_star, max_c, c = fast_score_kstar(a)
            if not np.isfinite(score):
                break
            S = a.sum()
            if S < 0.01:
                break

            # Use multi-lag gradient for smoothness
            grad = grad_multi(a, c, S, n_top=5)

            # Momentum update
            momentum = beta_m * momentum - alpha * grad
            trial_step = momentum

            # Line search
            improved = False
            for ls in range(15):
                a_trial = np.clip(a + trial_step, 0.0, 1000.0)
                S_trial = a_trial.sum()
                if S_trial < 0.01:
                    trial_step *= 0.5
                    continue
                c_trial = autocorr(a_trial)
                score_trial = 2.0 * n * c_trial.max() / (S_trial * S_trial)
                if np.isfinite(score_trial) and score_trial < cur_score - 1e-13:
                    a = a_trial
                    cur_score = score_trial
                    improved = True
                    alpha = min(alpha * 1.02, 0.5)
                    break
                trial_step *= 0.5

            if improved:
                no_improve = 0
            else:
                no_improve += 1
                alpha = max(alpha * 0.5, 1e-12)

            # Periodic exact evaluation
            if it % 500 == 0 and it > 0:
                exact = float(evaluate_sequence(a.tolist()))
                if np.isfinite(exact) and exact < best_score:
                    best_score = exact
                    best = a.copy()
                    no_improve = 0

            # Time check (frequent for long sequences)
            if it % 1000 == 0:
                remaining = deadline - time.monotonic()
                if remaining < 10.0:
                    break

        # Final check for this attempt
        exact = float(evaluate_sequence(a.tolist()))
        if np.isfinite(exact) and exact < best_score:
            best_score = exact
            best = a.copy()

    # Final validation
    final = [float(x) for x in best]
    fs = float(evaluate_sequence(final))
    if not np.isfinite(fs):
        final = [1.0] * int(n)

    return final
