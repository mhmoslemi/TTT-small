    if not np.isfinite(final_score):
        # Fallback to constant sequence
        best_list = [1.0] * n
        try:
            final_score = float(evaluate_sequence(best_list))
        except Exception:
            pass

    if not np.isfinite(final_score):
        # Absolute last resort
        best_list = [1.0, 1.0, 1.0]
        final_score = float(evaluate_sequence(best_list))

    return best_list
