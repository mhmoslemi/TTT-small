```python
def compute_gradient(a, n):
    """Compute approximate gradient of score w.r.t. a."""
    conv = np.convolve(a, a)
    M = conv.max()
    S = a.sum()
    k_star = np.argmax(conv)
    
    # Gradient of conv[k_star] w.r.t. a
    # d(conv[k])/d(a_j) = a[k-j] (if 0<=k-j<n) + a[j+k] (if 0<=j+k<n)
    grad_conv = np.zeros(n)
    # First term: a[k*-j] for j where 0 <= k*-j < n, i.e., k*-n+1 <= j <= k*
    # Second term: a[j+k*] for j where 0 <= j+k* < n, i.e., 0 <= j <= n-1-k*
    
    for j in range(n):
        g = 0.0
        if 0 <= k_star - j < n:
            g += a[k_star - j]
        if 0 <= j + k_star < n:
            g += a[j + k_star]
        grad_conv[j] = g
    
    # Gradient of score
    grad = (2*n / S**2) * grad_conv - (4*n*M / S**3)
    
    return grad, M, k_star
```

```python
    # sum_j a[k-j] * b[j] <= M (where a[k-j] is defined as 0 if k-j out of range)
    # This gives row k of the matrix.
    #
    # Row k has nonzeros at columns j where 0 <= k-j < n, i.e., max(0, k-(n-1)) <= j <= min(n-1, k)
    # The values are a[k-j] for j in that range.

        j_end = min(n-1, k)
```

```python
    # Gradient of max is (average of gradients at active lags)
    grad = np.zeros(n)
    for k in active:
        # d(conv[k])/d(a_j)
        # = a[k-j] (if 0<=k-j<n) + a[j+k] (if 0<=j+k<n)
        j1_start = max(0, k - n + 1)
        j1_end = min(n-1, k)
        for j in range(j1_start, j1_end+1):
            grad[j] += a[k-j]
        j2_end = min(n-1, n-1-k)
        for j in range(0, j2_end+1):
            grad[j] += a[j+k]
    
    # Gradient of score = 2n*max/S^2
    grad *= 2 * n / S**2
    grad -= 4 * n * M / S**3  # from the -2nM/S^2 * 1/S derivative... 
    
    # Actually: d(2nM/S^2)/d(a_j) = 2n/S^2 * dM/da_j - 4nM/S^3
    # Wait: d(f(g)/h)/d = (g' h - g h')/h^2
    # f = 2nM, g = M, h = S^2
    # d(2nM/S^2) = 2n(dM/S^2 - M*2dS/S^3) = 2n(dM/S^2) - 4nM(dS/S^3)
    # dS = 1 (for each component), so:
    # d(score)/d(a_j) = 2n*dM/d(a_j)/S^2 - 4nM/S^3
    
    new_a = a - lr * grad
    new_a = np.clip(new_a, 0, 1000)
    return new_a
```

The gradient computation is O(n * |active|) which could be slow if many lags are active. For a symmetric sequence, the peak is at lags n-1, and nearby lags might also be close to the peak. Let's say |active| is small (say < 20).

Actually, let me vectorize the gradient computation:

```python
def compute_gradient(a):
    n = len(a)
    conv = np.convolve(a, a)
    M = conv.max()
    S = a.sum()
    
    # Use only the single max lag for simplicity
    k = int(np.argmax(conv))
    
    grad_conv = np.zeros(n)
    # Term 1: a[k-j] for j where 0 <= k-j < n
    # j ranges from max(0, k-n+1) to min(n-1, k)
    j_start = max(0, k - n + 1)
    j_end = min(n-1, k)
    if j_start <= j_end:
        indices = np.arange(k - j_start, k - j_end - 1, -1)  # = k-j for j in [j_start, j_end]
        grad_conv[j_start:j_end+1] += a[indices]
    
    # Term 2: a[j+k] for j where 0 <= j+k < n
    # j ranges from 0 to min(n-1-k, n-1)
    j_end2 = min(n-1, n-1-k)
    if j_end2 >= 0:
        grad_conv[0:j_end2+1] += a[k:k+j_end2+1]
    
    # score = 2n*M/S^2
    grad = (2.0 * n / S**2) * grad_conv - (4.0 * n * M / S**3) * np.ones(n)
    
    return grad, M, k
```

This is O(n) per call.
