        if time.monotonic() > deadline - 30.0:
            break
        
        # Current peak
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        
        if M < 1e-15:
            break
        
        # Build sparse convolution matrix A (lower-triangular Toeplitz)
        # A[k,j] = f[k-j] for 0 <= k-j < n
        # Diagonal m (offset m) has value f[m], length n
        # n diagonals, offsets 0 to n-1
        diag_data = [np.full(n, f[m]) for m in range(n)]
        offsets = np.arange(n)
        A = diags(diag_data, offsets, shape=(2*n-1, n), format='csr')
        
        # LP: max sum(g) s.t. A@g <= M, g >= 0
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        
        t_start = time.monotonic()
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, 
                         method='highs', options={'time_limit': deadline - time.monotonic() - 15})
        except Exception:
            break
        
        if res.status != 0 or res.x is None:
            # Fallback: try a smaller version or break
            break
        
        g = np.asarray(res.x, dtype=np.float64)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        # Rescale g to sum 1
        g = g / g_sum
        g = np.clip(g, 0, 1000)
        g_sum2 = g.sum()
        if g_sum2 < 1e-10:
            break
        g = g / g_sum2
        
        # Line search for best blending
        best_t = 0
        best_t_score = np.inf
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            new_f = (1-t)*f + t*g
            new_f = np.clip(new_f, 0, 1000)
            sf = new_f.sum()
            if sf < 1e-10:
                continue
            new_f = new_f / sf  # renormalize
            score = float(evaluate_sequence(new_f.tolist()))
            if np.isfinite(score) and score < best_t_score:
                best_t_score = score
                best_t = t
        
        if best_t > 0 and best_t_score < best_score:
            f = (1-best_t)*f + best_t*g
            f = np.clip(f, 0, 1000)
            sf = f.sum()
            f = f / sf
            best = f.copy()
            best_score = best_t_score
        else:
            # Try very small steps
            for t in [0.01, 0.02]:
                new_f = (1-t)*f + t*g
                new_f = np.clip(new_f, 0, 1000)
                sf = new_f.sum()
                if sf < 1e-10:
                    continue
                new_f = new_f / sf
                score = float(evaluate_sequence(new_f.tolist()))
                if np.isfinite(score) and score < best_score:
                    f = new_f
                    best = f.copy()
                    best_score = score
                    break
    
    # Final validation
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # Fallback
        fallback = np.ones(n, dtype=np.float64) / n
        return [float(x) for x in fallback]
    
    return [float(x) for x in best]
