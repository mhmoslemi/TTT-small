        s = _fast_score(trial)
        if np.isfinite(s) and s < best_score - 1e-9:
            # Verify with exact evaluator
            vs = float(evaluate_sequence(trial.tolist()))
            if np.isfinite(vs) and vs < best_score:
                best = trial.copy()
                best_score = vs

    # Local / simulated-annealing search
    temp = 0.03
    no_improve = 0
    max_no_improve = 3000
    eval_count = 0

    while time.monotonic() < deadline:
        # Adaptive strength: larger early, smaller late
        frac = (time.monotonic() - t0) / max(1.0, float(budget_s))
        strength = 0.12 * (1.0 - 0.7 * min(1.0, frac))

        candidate = _make_perturbation(best, rng, n, strength)
        s = _fast_score(candidate)

        if np.isfinite(s):
            if s < best_score - 1e-12:
                # Verify improvement with exact evaluator
                vs = float(evaluate_sequence(candidate.tolist()))
                if np.isfinite(vs) and vs < best_score:
                    best = candidate.copy()
                    best_score = vs
                    no_improve = 0
                    temp = max(0.002, temp * 0.97)
                    eval_count += 1
            else:
                # Metropolis: accept worse with probability
                if no_improve < max_no_improve:
                    delta = s - best_score
                    p = np.exp(-delta / max(1e-10, temp))
                    if rng.random() < p:
                        best = candidate
                        no_improve += 1
                else:
                    no_improve += 1

        eval_count += 1
        # Gradual cooling
        if eval_count % 500 == 0:
            temp = max(0.001, temp * 0.95)

        # Periodic verification with exact evaluator to guard against drift
        if eval_count % 5000 == 0:
            cur_score = float(evaluate_sequence(best.tolist()))
            if not np.isfinite(cur_score):
                best = parent.copy()
                best_score = float(evaluate_sequence(best.tolist()))
            elif cur_score > best_score + 1e-9:
                best_score = cur_score

    # Final exact validation
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        best = parent
        final_score = float(evaluate_sequence(best.tolist()))

    # Ensure output is a valid Python list of finite floats
    out = [float(x) for x in best]
    # Safety: verify all finite and in range
    for i, v in enumerate(out):
        if not np.isfinite(v):
            out[i] = 0.0
    return out
