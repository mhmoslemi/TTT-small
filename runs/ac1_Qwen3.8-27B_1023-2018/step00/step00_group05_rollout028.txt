            #   A[k, j] = a[j] for j in [max(0,k-n+1), min(n-1,k)]
            
            A_rows = np.repeat(np.arange(len(active_rows)), [min(n-1,k)-max(0,k-n+1)+1 for k in active_rows])
            A_cols = np.concatenate([np.arange(max(0,k-n+1), min(n-1,k)+1) for k in active_rows])
            A_vals = a[A_cols]
            A = sparse.csr_matrix((A_vals, (A_rows, A_cols)), shape=(len(active_rows), n))
            
            # LP: max sum(g) s.t. A @ g <= P, g >= 0
            g_var = cp.Variable(n, nonneg=True)
            constraints = [A @ g_var <= P]
            objective = cp.Maximize(cp.sum(g_var))
            prob = cp.Problem(objective, constraints)
            
            solved = False
            for solver in [cp.GLOP, cp.CBC, cp.GLPK, cp.SCIP]:
                try:
                    prob.solve(solver=solver, **{cp.GLOP: {'time_limit': 30}, cp.CBC: {'timeLimit': 30}, cp.GLPK: {'timeLimit': 30}, cp.SCIP: {'time_limit': 30}}.get(solver, {'time_limit': 30}))
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        solved = True
                        break
                except:
                    continue
            
            if not solved:
                break
            
            g_opt = g_var.value
            if g_opt is None:
                break
            
            # Check if all constraints are satisfied
            c_check = np.convolve(a, g_opt)
            max_violation = c_check.max() - P
            if max_violation > P * 1e-6:
                # Find violated constraints and add them
                violated = np.where(c_check > P + P * 1e-6)[0]
                # Add the most violated few
                most_violated = violated[np.argsort(-c_check[violated])[:min(20, len(violated))]]
                new_rows = list(set(active_rows + [int(k) for k in most_violated]))
                new_rows.sort()
                active_rows = new_rows
                continue
            
            # LP is feasible and optimal with all constraints
            sum_g = g_opt.sum()
            if sum_g <= target_sum * (1 + 1e-6):
                break  # No improvement possible
            
            # Rescale and update
            g_rescaled = g_opt * target_sum / sum_g
            
            # Line search
            best_t = 0.0
            best_new_peak = P
            for t in [1.0, 0.8, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
                a_try = (1 - t) * a + t * g_rescaled
                a_try = np.maximum(a_try, 0)
                c_try = np.convolve(a_try, a_try)
                peak_try = float(c_try.max())
                if peak_try < best_new_peak:
                    best_new_peak = peak_try
                    best_t = t
            
            if best_t == 0.0:
                break  # Can't improve
            
            a = (1 - best_t) * a + best_t * g_rescaled
            a = np.maximum(a, 0)
            a = a * target_sum / a.sum()
            break  # B
