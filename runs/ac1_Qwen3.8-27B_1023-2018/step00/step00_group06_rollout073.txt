                    cur_step = float(step_base)
                    evals_since_best = 0

        score = float(evaluate_sequence(a.tolist()))
        if math.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score

    return best, best_score


def _alternate_polish(a0, t_start, t_end, rng, n, L1, L2):
    a = _normalize(a0, 1e-15)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not math.isfinite(best_score):
        best_score = float("inf")

    # (beta, step, inner_iters, fraction)
    configs = (
        (10.0, 0.10, 30, 0.28),
        (30.0, 0.06, 40, 0.36),
        (100.0, 0.035, 30, 0.36),
    )

    total = max(1e-6, float(t_end) - float(t_start))
    cum = 0.0

    for beta, eta, inner, frac in configs:
        stage_end = min(float(t_end), float(t_start) + (cum + frac) * total)
        cum += frac
        if stage_end - time.monotonic() < 3.0:
            break

        outer = 0
        while time.monotonic() < stage_end - 2.0 and outer < 6:
            outer += 1
            b = a.copy()
            fa1 = np.fft.rfft(a, L1)
            fa2 = np.fft.rfft(a, L2)

            for _ in range(int(inner)):
                if time.monotonic() > stage_end - 0.5:
                    break
                fb1 = np.fft.rfft(b, L1)
                c = np.fft.irfft(fa1 * fb1, L1).real[:2 * n - 1]
                c = np.maximum(c, 0.0)
                V = c * float(n)
                vmax = float(V.max())
                if not math.isfinite(vmax) or vmax <= 0.0:
                    break
                diff = np.minimum(0.0, V - vmax)
                cut = max(1e-6, 700.0 / beta)
                diff = np.maximum(diff, -cut)
                w = np.exp(float(beta) * diff)
                sw = float(w.sum())
                if not math.isfinite(sw) or sw <= 0.0:
                    break
                w *= 1.0 / sw

                G = float(n) * _crosscorr_w_a(w, a, L2, fa2)
                G = _standardize(G)
                if float(np.abs(G).sum()) == 0.0:
                    break

                decay = np.exp(-float(eta) * G)
                b = _normalize(b * decay, 1e-15)

            # Accept the best nearby candidate, whether or not it beats the
            # global best yet.  This keeps the outer sequence moving along the
            # alternating direction.
            local_a = a.copy()
            local_score = float(evaluate_sequence(local_a.tolist()))
            candidates = [b]
            for lam in (0.75, 0.5, 0.25):
                candidates.append((1.0 - float(lam)) * a + float(lam) * b)

            for cand in candidates:
                cand = _normalize(cand, 1e-15)
                sc = float(evaluate_sequence(cand.tolist()))
                if math.isfinite(sc) and sc < local_score:
                    local_a = cand
                    local_score = sc

            a = local_a
            if math.isfinite(local_score) and local_score < best_score:
                best = a.copy()
                best_score = local_score

    score = float(evaluate_sequence(a.tolist()))
    if math.isfinite(score) and score < best_score:
        best = a.copy()
        best_score = score
    return best, best_score


def _local_jitter(a0, t_start, t_end, rng, n, L1):
    best = _normalize(a0, 1e-15).copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not math.isfinite(best_score):
        best_score = float("inf")

    sigma = 0.02
    stuck = 0
    while time.monotonic() < t_end - 1.0:
        cand = best * np.exp(rng.normal(0.0, sigma, n))
        cand = _normalize(cand, 1e-15)

        surv = _surrogate_score(cand, L1)
        if math.isfinite(surv) and surv < best_score * (1.0 - 1e-12):
            sc = float(evaluate_sequence(cand.tolist()))
            if math.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc
                stuck = 0
                sigma = max(0.005, sigma * 0.92)
            else:
                stuck += 1
                sigma = max(0.001, sigma * 0.98)
        else:
            stuck += 1
            sigma = max(0.001, sigma * 0.985)

        if sigma <= 0.0012 and stuck >= 100:
            break

        # Occasional structured kick to avoid a tiny local minimum.
        if stuck % 120 == 0:
            cand = 0.8 * best + 0.2 * np.ones(n, dtype=np.float64) / n
            cand = _normalize(cand, 1e-15)
            surv = _surrogate_score(cand, L1)
            if math.isfinite(surv) and surv < best_score * (1.0 - 1e-12):
                sc = float(evaluate_sequence(cand.tolist()))
                if math.isfinite(sc) and sc < best_score:
                    best = cand
                    best_score = sc
                    stuck = 0
                    sigma = 0.02

    return best, best_score


def _make_seeds(n, parent, rng):
    seeds = []

    def add(raw):
        try:
            raw = np.asarray(raw, dtype=np.float64).ravel()
        except Exception:
            return
        if raw.size != n:
            return
        if not np.all(np.isfinite(raw)):
            return
        raw = np.maximum(raw, 1e-12)
        s = float(raw.sum())
        if not math.isfinite(s) or s <= 0.0:
            return
        raw *= 1.0 / s
        seeds.append(raw)

    if parent is not None:
        add(_normalize(parent, 1e-15))
    add(np.ones(n, dtype=np.float64) / n)

    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)

    # Skewed / smooth shapes.
    for g in (-2.0, -1.2, -0.6, 0.6, 1.2, 2.0):
        add(np.exp(float(g) * (2.0 * x - 1.0)))

    for alpha in (0.2, 0.4, 0.6, 0.8):
        for m in (1, 2, 3):
            add(1.0 + float(alpha) * np.sin(2.0 * np.pi * float(m) * x))

    combos = (
        (1, 0.6, 2, 0.3),
        (1, 0.5, 3, 0.5),
        (1, 0.4, 4, 0.4),
        (1, 0.7, 5, 0.2),
        (2, 0.5, 3, 0.5),
    )
    for m1, a1, m2, a2 in combos:
        add(1.0 + float(a1) * np.sin(2.0 * np.pi * float(m1) * x) +
             float(a2) * np.sin(2.0 * np.pi * float(m2) * x))

    for alpha in (0.3, 0.6):
        add(1.0 + float(alpha) * np.cos(2.0 * np.pi * x))

    # Random band-limited / anti-palindromic seeds.
    M = 24
    freqs = np.arange(1, M + 1, dtype=np.float64)
    for _ in range(10):
        phase = rng.uniform(0.0, 2.0 * np.pi, M)
        amp = rng.uniform(0.25, 1.0, M) / (freqs ** 0.75)
        y = np.zeros(n, dtype=np.float64)
        for i in range(M):
            y += float(amp[i]) * np.sin(2.0 * np.pi * float(freqs[i]) * x + float(phase[i]))
        scale = 0.85 / max(1.0, float(np.max(np.abs(y))))
        add(1.0 + (scale * y))

    for _ in range(6):
        y = np.zeros(n, dtype=np.float64)
        for m in (1, 2, 3, 4, 6, 8):
            y += rng.normal(0.0, 1.0 / float(m)) * np.sin(2.0 * np.pi * float(m) * x)
        scale = 0.75 / max(1.0, float(np.max(np.abs(y))))
        add(1.0 + (scale * y))

    # Smooth random block scalings.
    nb = int(max(8, n // 100))
    for _ in range(6):
        a = np.ones(n, dtype=np.float64)
        width = max(2, n // nb)
        for _b in range(nb):
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + int(rng.integers(1, width + 1)))
            a[start:stop] *= float(rng.lognormal(0.0, 0.35))
        add(a)

    # Random anti-symmetric modulation around the uniform level.
    for _ in range(5):
        y = np.zeros(n, dtype=np.float64)
        for m in (1, 2, 3, 5, 7):
            y += rng.normal(0.0, 1.0 / float(m)) * np.sin(2.0 * np.pi * float(m) * x)
        u = y - y[::-1]
        sc = 0.7 / max(1.0, float(np.max(np.abs(u))))
        add(1.0 + (sc * u))

    # Deduplicate by a cheap fingerprint.
    uniq = []
    seen = set()
    for s in seeds:
        key = (tuple(np.round(np.maximum(s, 0.0) * 1024.0, 0)), n)
        if key in seen:
            continue
        seen.add(key)
        uniq.append(s)
    return uniq


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    rng = np.random.default_rng(seed)

    parent = _parent_array()
    if parent is None or parent.size < 64:
        n = 512
        parent = np.ones(n, dtype=np.float64) / n
    else:
        n = int(parent.size)

    # Safety cap for memory/time if an unexpected very long parent is given.
    if n > 12000:
        target = 12000
        p = np.asarray(parent, dtype=np.float64).ravel()
        starts = np.arange(target) * (p.size // target)
        ends = starts + (p.size // target)
        ends[-1] = p.size
        if p.size % target != 0:
            # Simple average pooling to exactly target length.
            idx = np.linspace(0, p.size, target + 1).astype(int)
            chunks = [p[idx[i]:idx[i + 1]] for i in range(target) if idx[i + 1] > idx[i]]
            p = np.concatenate(chunks) if chunks else np.ones(target)
        else:
            p = np.array([p[starts[i]:ends[i]].mean() for i in range(target)], dtype=np.float64)
        parent = _normalize(p, 1e-15)
        n = int(parent.size)

    L1 = _next_pow2(2 * n - 1)
    L2 = _next_pow2(3 * n - 2)

    # Choose safe internal deadlines, leaving validation headroom in the end.
    end_main = t0 + max(30.0, budget * 0.72)
    end_alt = t0 + max(60.0, budget * 0.91)
    end_local = t0 + max(100.0, budget * 0.98)
    end_local = min(end_local, t0 + budget - 3.0)
    end_alt = min(end_alt, end_local - 5.0)
    end_main = min(end_main, end_alt - 5.0)

    seeds = _make_seeds(n, parent, rng)
    if not seeds:
        seeds = [np.ones(n, dtype=np.float64) / n]

    # Sort seeds by exact-equivalent surrogate score.
    scored = []
    for s in seeds:
        try:
            sv = _surrogate_score(s, L1)
        except Exception:
            sv = float("inf")
        scored.append((sv, s))
    scored.sort(key=lambda z: z[0])
    scored = [z for z in scored if math.isfinite(z[0])][:6]
    if not scored:
        scored = [(float("inf"), np.ones(n, dtype=np.float64) / n)]

    global_best = _normalize(seeds[0], 1e-15).copy()
    global_score = float(evaluate_sequence(global_best.tolist()))
    if not math.isfinite(global_score):
        global_score = float("inf")

    for _, s in scored:
        sc = _surrogate_score(_normalize(s, 1e-15), L1)
        if math.isfinite(sc) and sc < global_score:
            global_best = _normalize(s, 1e-15).copy()
            global_score = float(evaluate_sequence(global_best.tolist()))
            if not math.isfinite(global_score):
                global_score = float("inf")

    # Main seed optimization.
    fracs = np.array([0.34, 0.20, 0.17, 0.14, 0.10, 0.05], dtype=np.float64)
    m = min(len(scored), len(fracs))
    if m == 0:
        m = 1
    fracs = fracs[:m]
    fracs = fracs / fracs.sum() if float(fracs.sum()) > 0 else np.ones(m)

    total_main = max(1.0, end_main - t0)
    cum = 0.0
    for i in range(m):
        a0, _ = scored[i]
        a0 = _normalize(a0, 1e-15)
        seg_start = t0 + cum * total_main
        seg_end = min(end_main, seg_start + float(fracs[i]) * total_main)
        cum += float(fracs[i])

        if time.monotonic() >= seg_start and seg_end - time.monotonic() > 2.0:
            try:
                b, sc = _optimize_direct(a0, seg_start, seg_end, rng, n, L1, L2)
                if math.isfinite(sc) and sc < global_score:
                    global_best = b
                    global_score = sc
            except Exception:
                pass

    # Alternating polish on the best found so far.
    if end_alt - time.monotonic() > 4.0:
        try:
            b, sc = _alternate_polish(global_best, max(t0, time.monotonic()), end_alt, rng, n, L1, L2)
            if math.isfinite(sc) and sc < global_score:
                global_best = b
                global_score = sc
        except Exception:
            pass

    # Small exact-score-guided jitter.
    if end_local - time.monotonic() > 3.0:
        try:
            b, sc = _local_jitter(global_best, max(t0, time.monotonic()), end_local, rng, n, L1)
            if math.isfinite(sc) and sc < global_score:
                global_best = b
                global_score = sc
        except Exception:
            pass

    final = np.asarray(global_best, dtype=np.float64).ravel()
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if not math.isfinite(s) or s < 0.01:
        final = np.ones(n, dtype=np.float64) / n
    else:
        final = final / s

    final_score = float(evaluate_sequence([float(x) for x in final]))
    if not math.isfinite(final_score):
        final = _normalize(np.ones(n, dtype=np.float64) / n, 1e-15)
        final_score = float(evaluate_sequence([float(x) for x in final]))
        if not math.isfinite(final_score):
            final = np.full(n, 1.0, dtype=np.float64) / n
            final_score = float(evaluate_sequence([float(x) for x in final]))

    return [float(x) for x in final]
