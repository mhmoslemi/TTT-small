from scipy.signal import fftconvolve
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def _compute_conv(x):
    return fftconvolve(x, x, mode='full')

def _subgradient(x, k_active, S, M, n):
    """Compute subgradient of 2n*max_c/S^2 at x, with peak at k_active."""
    s = np.zeros(n)
    for k in k_active:
        # s_j += x[k-j] if 0 <= k-j < n
        js = np.arange(max(0, k-n+1), min(n, k+1))
        idx = k - js  # indices into x
        s[js] += x[idx]
        # s_j += x[j-k] if 0 <= j-k < n
        js2 = np.arange(k, min(n, k+n))
        if js2.size > 0:
            idx2 = js2 - k
            s[js2] += x[idx2]
    
    # Full subgradient of score
    grad = (2.0 * n / S**2) * (s - 2.0 * M * x / S)
    return grad
