class _FFTConv:
    def __init__(self, n):
        self.n = int(n)
        self.m = 2 * self.n - 1
        self.Nc = 1 << (self.m - 1).bit_length()
        self.Ng = 1 << (3 * self.n - 2).bit_length()

    def conv(self, x):
        x = np.asarray(x, dtype=np.float64)
        fx = np.fft.rfft(x, self.Nc)
        return np.fft.irfft(fx * fx, self.Nc)[:self.m]

    def score(self, x):
        x = np.asarray(x, dtype=np.float64)
        s = float(x.sum())
        if s <= 0.0:
            return np.inf
        return float(2.0 * self.n * float(np.max(self.conv(x))) / (s * s))

    def conv_pair(self, x, y):
        x = np.asarray(x, dtype=np.float64)
        y = np.asarray(y, dtype=np.float64)
        fx = np.fft.rfft(x, self.Nc)
        fy = np.fft.rfft(y, self.Nc)
        xy = np.fft.irfft(fx * fy, self.Nc)[:self.m]
        yy = np.fft.irfft(fy * fy, self.M[:0] if False else fy * fy, self.Nc)[:self.m] if False else np.fft.irfft(fy * fy, self.Nc)[:self.m]
        return xy, yy

    def grad_weighted(self, x, w):
        x = np.asarray(x, dtype=np.float64)
        w = np.asarray(w, dtype=np.float64)
        if w.size != self.m:
            w = np.resize(w, self.m)
        fx = np.fft.rfft(x, self.Ng)
        fw = np.fft.rfft(w[::-1], self.Ng)
        full = np.fft.irfft(fx * fw, self.Ng)
        return full[2 * self.n - 2:self.n - 2:-1]


def _binomial_init(n):
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64)
    b = np.empty(n, dtype=np.float64)
    b[0] = 1.0
    i = np.arange(1, n, dtype=np.float64)
    b[1:] = np.cumprod((2.0 * i - 1.0) / (2.0 * i))
    return b


def _clean_nonneg(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    return a


def _normalize(a):
    a = _clean_nonneg(a)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-14:
        return np.ones(len(a), dtype=np.float64) / max(1, len(a))
    return a / s


def _softmax_weights(c, tau):
    c = np.asarray(c, dtype=np.float64)
    C = float(np.max(c))
    if tau <= 0.0 or not np.isfinite(tau):
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
        return p
    p = np.exp((c - C) / tau)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
        return p
    return p / s


def _active_indices(c, m, tol=None):
    c = np.asarray(c, dtype=np.float64)
    C = float(np.max(c))
    if tol is None:
        tol = max(1e-11 * C, 1e-14)
    idx = np.where(c >= C - tol)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))])
    if idx.size <= m:
        return idx
    pos = np.linspace(0, idx.size - 1, m).astype(np.int64)
    return idx[pos]


def _direction_lse(F, a, c, tau_rel):
    C = float(np.max(c))
    tau = max(1e-18, C * float(tau_rel))
    p = _softmax_weights(c, tau)
    g = F.grad_weighted(a, p)
    h = -(g - float(g.mean()))
    mh = float(np.max(np.abs(h)))
    if mh <= 0.0:
        return None
    return h / mh


def _direction_top(F, a, c, m=128):
    idx = _active_indices(c, m)
    p = np.zeros(F.m, dtype=np.float64)
    p[idx] = 1.0 / max(1, idx.size)
    g = F.grad_weighted(a, p)
    h = -(g - float(g.mean()))
    mh = float(np.max(np.abs(h)))
    if mh <= 0.0:
        return None
    return h / mh


def _direction_single(F, a, c, rng):
    idx = _active_indices(c, 1, tol=max(1e-10 * float(np.max(c)), 1e-14))
    k = int(rng.integers(idx.size))
    k = int(idx[k])
    g = np.zeros(a.size, dtype=np.float64)
    if k < a.size:
        g[:k + 1] = a[k::-1]
    else:
        start = k - a.size + 1
        idxs = k - np.arange(start, a.size, dtype=np.int64)
        g[start:] = a[idxs]
    h = -(g - float(g.mean()))
    mh = float(np.max(np.abs(h)))
    if mh <= 0.0:
        return None
    return h / mh


def _direction_random(F, a, rng):
    n = a.size
    for _ in range(20):
        r = rng.normal(size=n)
        rf = np.fft.rfft(r)
        freq = np.fft.rfftfreq(n)
        rf[(freq > 0.12) & (freq < 0.5)] = 0.0
        r = np.fft.irfft(rf, n)
        r -= float(r.mean())
        mh = float(np.max(np.abs(r)))
        if mh > 0:
            h = r / mh
            h -= float(h.mean())
            mh = float(np.max(np.abs(h)))
            if mh > 0:
                return h / mh
    r = rng.normal(size=n)
    r -= float(r.mean())
    mh = float(np.max(np.abs(r)))
    if mh <= 0:
        return None
    h = r / mh
    h -= float(h.mean())
    mh = float(np.max(np.abs(h)))
    if mh <= 0:
        return None
    return h / mh


def _choose_alpha(c, lin, q, a, h):
    hneg = h < -1e-16
    hpos = h > 1e-16
    pos_lim = np.inf
    if np.any(hneg):
        pos_lim = float(np.min(a[hneg] / (-h[hneg])))
    neg_lim = 0.0
    if np.any(hpos):
        neg_lim = -float(np.min(a[hpos] / h[hpos]))
    if not np.isfinite(pos_lim):
        pos_lim = np.inf
    if not np.isfinite(neg_lim):
        neg_lim = 0.0
    pos_lim *= 0.999
    neg_lim *= 0.999

    amax = float(np.max(a))
    if amax <= 0.0:
        return None, np.inf
    cap = min(0.10, max(1e-6, 0.45 * amax))
    if pos_lim > cap:
        pos_lim = cap
    if -neg_lim > cap:
        neg_lim = -cap

    alphas = []

    def add_side(mag, sgn):
        if mag > 1e-10:
            start = max(1e-12 * mag, 1e-15)
            linear = np.linspace(start, mag, 72)
            alphas.extend((linear * sgn).tolist())
            if mag / start > 8.0:
                log = np.geomspace(start, mag, 24)
                alphas.extend((log * sgn).tolist())

    add_side(pos_lim, 1.0)
    if neg_lim < 0.0:
        add_side(-neg_lim, -1.0)

    if not alphas:
        return None, np.inf

    alphas = np.asarray(alphas, dtype=np.float64)
    alphas = alphas[(alphas <= pos_lim + 1e-14) & (alphas >= neg_lim - 1e-14)]
    if alphas.size == 0:
        return None, np.inf

    best_val = np.inf
    best_alpha = 0.0
    chunk = 40
    for i in range(0, alphas.size, chunk):
        aa = alphas[i:i + chunk]
        vals = c[None, :] + aa[:, None] * lin[None, :] + (aa * aa)[:, None] * q[None, :]
        mo = vals.max(axis=1)
        mo = np.where(np.isfinite(mo), mo, np.inf)
        j = int(np.argmin(mo))
        if mo[j] < best_val:
            best_val = float(mo[j])
            best_alpha = float(aa[j])

    if best_alpha == 0.0:
        return None, best_val

    width = max(abs(best_alpha) * 0.12, 1e-9 * cap)
    lo = max(neg_lim, best_alpha - width)
    hi = min(pos_lim, best_alpha + width)
    if hi > lo:
        aa = np.linspace(lo, hi, 48)
        vals = c[None, :] + aa[:, None] * lin[None, :] + (aa * aa)[:, None] * q[None, :]
        mo = vals.max(axis=1)
        mo = np.where(np.isfinite(mo), mo, np.inf)
        j = int(np.argmin(mo))
        if mo[j] < best_val:
            best_val = float(mo[j])
            best_alpha = float(aa[j])

    return best_alpha, best_val


def _try_direction(F, a, c, h, improve_tol):
    if h is None:
        return False, None, None
    h = np.asarray(h, dtype=np.float64)
    h -= h.mean()
    mh = float(np.max(np.abs(h)))
    if mh <= 0.0:
        return False, None, None
    h /= mh
    lin = 2.0 * F.conv(a, h)
    q = F.conv(h, h)
    alpha, val = _choose_alpha(c, lin, q, a, h)
    if alpha is None or not np.isfinite(val):
        return False, h, lin
    if val < c.max() - improve_tol:
        a_new = a + alpha * h
        if a_new.min() < -1e-12:
            a_new = np.maximum(a_new, 0.0)
            a_new /= float(a_new.sum())
            c_new = F.conv(a_new)
        else:
            s = float(a_new.sum())
            c_new = c + alpha * lin + (alpha * alpha) * q
            if abs(s - 1.0) > 1e-12:
                a_new = a_new / s
                c_new = c_new * (s * s)
            if c_new.min() < 0.0:
                if c_new.min() < -1e-9:
                    a_new = np.maximum(a_new, 0.0)
                    a_new /= float(a_new.sum())
                    c_new = F.conv(a_new)
                else:
                    c_new = np.maximum(c_new, 0.0)
        return True, a_new, c_new
    return False, h, lin


def _optimize_length(n, seed, deadline, parent_a=None):
    rng = np.random.default_rng(seed)
    n = int(max(8, n))
    F = _FFTConv(n)

    base = _binomial_init(n)
    xlin = np.linspace(0.0, 1.0, n)
    candidates = [base.copy()]

    if parent_a is not None and len(parent_a) == n:
        candidates.append(_normalize(parent_a))

    for g in (-0.35, 0.35, -0.15, 0.15):
        candidates.append(base * np.exp(g * xlin))
    for g in (-0.35, 0.35):
        candidates.append(base * (1.0 + g * np.cos(np.pi * xlin)))
        candidates.append(base * (1.0 + g * np.sin(np.pi * xlin)))
    for g in (-0.25, 0.25):
        candidates.append(base * (1.0 + g * (2.0 * xlin - 1.0) ** 2))

    best_a = None
    best_score = np.inf
    for arr in candidates:
        arr = _normalize(arr)
        sc = F.score(arr)
        if sc < best_score:
            best_score = sc
            best_a = arr.copy()

    if best_a is None:
        best_a = _normalize(base)
        best_score = F.score(best_a)

    a = best_a.copy()
    c = F.conv(a)
    cur_score = 2.0 * n * float(c.max())
    start_best = best_a.copy()
    start_best_score = best_score
    start_c = F.conv(start_best)

    tau_list = (0.05, 0.02, 0.10, 0.20, 0.50, 0.01, 0.005, 1.00, 2.00)
    tau_idx = 0
    cur_tau = tau_list[0]
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        improve_tol = max(1e-14, 1e-10 * float(c.max()))
        variants = [("lse", cur_tau)]
        if stall > 8:
            variants.append(("top", 0.0))
        if stall > 20:
            variants.append(("single", 0.0))
        if stall > 50:
            variants.append(("random", 0.0))
        if stall > 100:
            variants.append(("lse_small", max(0.001, cur_tau * 0.12)))

        accepted = False
        for name, tau in variants:
            if name == "lse":
                h = _direction_lse(F, a, c, tau)
            elif name == "top":
                h = _direction_top(F, a, c, m=160)
            elif name == "single":
                h = _direction_single(F, a, c, rng)
            elif name == "random":
                h = _direction_random(F, a, rng)
            else:
                h = _direction_lse(F, a, c, tau)

            ok, a_new, c_new = _try_direction(F, a, c, h, improve_tol)
            if not ok and name == "lse":
                # One Newton-like reweighting using the failed direction's linear part.
                if ok is False and len(a) > 0:
                    # recover h/lin if available: easiest is to re-evaluate cheaply only when stalled
                    pass
            if ok:
                a = np.asarray(a_new, dtype=np.float64)
                c = np.asarray(c_new, dtype=np.float64)
                cur_score = 2.0 * n * float(c.max())
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                stall = 0
                it += 1
                accepted = True
                cur_tau = max(0.002, cur_tau * 0.992)
                break

        if not accepted:
            stall += 1
            tau_idx = (tau_idx + 1) % len(tau_list)
            cur_tau = tau_list[tau_idx]

            if stall == 250 or stall == 700 or (stall % 1200 == 0):
                # Explore from the globally best point with a small random push.
                h = _direction_random(F, start_best, rng)
                if h is not None:
                    h -= h.mean()
                    mh = float(np.max(np.abs(h)))
                    if mh > 0:
                        h /= mh
                        alpha = 0.02
                        for _ in range(8):
                            trial = start_best + alpha * h
                            if trial.min() >= 0:
                                break
                            alpha *= 0.5
                        trial = _normalize(trial)
                        trial_c = F.conv(trial)
                        trial_score = 2.0 * n * float(trial_c.max())
                        if trial_score < start_best_score * 1.25 + 1e-12:
                            a = trial
                            c = trial_c
                            cur_score = trial_score
                            stall = max(1, stall // 3)

            if stall > 4000:
                break
        else:
            it += 1

    # Periodic-like stiffening near deadline is unnecessary; the quadratic line search is exact.
    if time.monotonic() < max(deadline, time.monotonic()):
        final_score = 2.0 * n * float(F.conv(best_a).max())
        if final_score < best_score:
            best_score = final_score
    return best_a, float(best_score)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget_s = float(max(10.0, float(budget_s)))
    final_deadline = t0 + budget_s - 2.0
    opt_deadline = t0 + max(1.0, budget_s - 45.0)

    rng = np.random.default_rng(seed)

    parent = None
    hs = globals().get("height_sequence_1", None)
    if hs is not None:
        try:
            p = np.asarray(hs, dtype=np.float64).ravel()
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 8 and p.sum() > 0:
                parent = p
        except Exception:
            parent = None

    parent_n = int(parent.size) if parent is not None else 0

    if budget_s > 900:
        primary_n = parent_n if 256 <= parent_n <= 20000 else 4096
    elif budget_s > 250:
        primary_n = min(2048, parent_n if parent_n >= 128 else 1024)
    else:
        primary_n = min(512, parent_n if parent_n >= 64 else 256)

    primary_n = int(max(64, int(primary_n)))

    primary_parent = None
    if parent is not None and parent_n == primary_n:
        primary_parent = parent
    elif parent is not None:
        # Do not copy a different length; use a truncated/expanded parent only as a reservoir.
        if parent_n > primary_n:
            primary_parent = _normalize(parent[:primary_n])
        else:
            primary_parent = _normalize(np.tile(parent, (primary_n // max(1, parent_n) + 1))[:primary_n])

    results = []

    # Primary run.
    try:
        a, sc = _optimize_length(primary_n, seed, opt_deadline, primary_parent)
        results.append((a, sc, primary_n))
    except Exception:
        a = _normalize(_binomial_init(primary_n))
        results.append((a, 2.0, primary_n))

    # Secondary smaller runs if useful and time permits.
    tried_lengths = {results[0][2] if results else -1}
    candidate_lengths = [min(primary_n // 2, 2048), 1024, 512, 256]
    for n2 in candidate_lengths:
        n2 = int(max(64, n2))
        if n2 in tried_lengths:
            continue
        if time.monotonic() > opt_deadline - 70:
            break
        # If primary is already near the target, avoid spending too much on likely-worse lengths.
        primary_best = min((sc for _, sc, _ in results), default=np.inf)
        if primary_best < 1.505 and n2 < 1024:
            if time.monotonic() > opt_deadline - 90:
                break
        tlocal = min(opt_deadline - 5.0, time.monotonic() + 75.0)
        if tlocal <= time.monotonic() + 5.0:
            break
        p2 = None
        if parent is not None and parent_n == n2:
            p2 = parent
        elif parent is not None:
            if parent_n > n2:
                p2 = _normalize(parent[:n2])
            else:
                p2 = _normalize(np.tile(parent, (n2 // max(1, parent_n) + 1))[:n2])
        try:
            a2, sc2 = _optimize_length(n2, seed + 1000 + n2, tlocal, p2)
            results.append((a2, sc2, n2))
        except Exception:
            pass
        tried_lengths.add(n2)

    # Select by exact evaluator.
    best_list = None
    best_exact = np.inf

    def exact_list(arr):
        arr = _clean_nonneg(arr)
        if arr.size == 0:
            return None
        lst = [float(x) for x in arr]
        if lst and all(np.isfinite(v) and v >= 0.0 for v in lst) and float(np.sum(lst)) >= 0.01:
            return lst
        return None

    # Evaluate larger/closer-to-target candidates first, but stop if time is short.
    eval_order = sorted(range(len(results)), key=lambda i: (results[i][1], -results[i][2]))
    for idx in eval_order:
        if time.monotonic() > final_deadline - 8.0:
            break
        arr, _, _ = results[idx]
        lst = exact_list(arr)
        if lst is None:
            continue
        try:
            sc = float(evaluate_sequence(lst))
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < best_exact:
            best_exact = sc
            best_list = lst

    # If exact evaluation did not produce anything, fall back to a guaranteed finite candidate.
    if best_list is None:
        for _, _, _ in results:
            pass
        arr = results[0][0] if results else _normalize(_binomial_init(128))
        best_list = exact_list(arr)
        if best_list is None:
            best_list = [1.0]

    # Verify once more just before returning, with headroom.
    if time.monotonic() < final_deadline - 0.5:
        try:
            sc = float(evaluate_sequence(best_list))
            if np.isfinite(sc):
                best_exact = min(best_exact, sc)
        except Exception:
            pass

    return [float(max(0.0, min(1000.0, x))) for x in best_list]
