- Process constraints in order. For constraint j, if the current allocated y over [j, j+n-1] is less than d_j, add the deficit to the rightmost unallocated position.

One more consideration: for the LP, I should make sure the problem is well-scaled. The constraint matrix entries are a[j] (values of the sequence), and the RHS is M (the peak of the self-convolution). For a roughly uniform sequence with values ~1/n, the entries are ~1/n and M ~ 1/n. So the scaling is OK.

But after a few LP iterations, the sequence might become more structured, with some values much larger or smaller. In that case, the LP might be poorly scaled. I can normalize a before building the matrix (divide by max(a) or by mean(a)).

Actually, the LP is:
max sum(b) s.t. A @ b <= M, b >= 0

where A[k,j] = a[j] for valid (k,j). If I scale a by c (set a' = a/c), then A' = A/c, M' = M/c^2 (since M = max conv(conve(a,a)) scales as c^2). The LP becomes:

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + max(1.0, total_budget - 12.0)

    # --- Initialization ---
    parent = np.asarray(height_sequence_1, dtype=np.float64)
    a = np.clip(parent, 0.0, 1000.0).copy()
    n = a.size

    if n < 4 or not np.isfinite(a).all() or a.sum() < 0.01:
        a = np.ones(256, dtype=np.float64)
        n = 256

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        return [float(x) for x in best]

    # --- Precompute convolution matrix indices (structural, reused each LP call) ---
    # Column j holds a[j] at rows j, j+1, ..., j+n-1  =>  total nonzeros = n*n
    row_idx = (np.repeat(np.arange(n, dtype=np.int64), n) +
               np.tile(np.arange(n, dtype=np.int64), n))
    col_idx = np.tile(np.arange(n, dtype=np.int64), n)
    nc = 2 * n - 1

    # --- LP helper ---
    def lp_direction(a, t_limit=90.0):
        """Solve max 1'b s.t. conv(a,b) <= M, b >= 0.  Return scaled direction or None."""
        S = float(a.sum())
        if S < 1e-10:
            return None
        conv_aa = np.convolve(a, a)
        M = float(conv_aa.max())
        if M < 1e-15:
            return None

        vals = np.repeat(a, n)
        try:
            A_ub = csr_matrix((vals, (row_idx, col_idx)), shape=(nc, n))
        except Exception:
            return None

        c_obj = -np.ones(n)
        b_ub = np.full(nc, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'time_limit': t_limit,
                                   'presolve': True,
                                   'mip_rel_gap': 0.0})
        except Exception:
            return None

        if res.status != 0 or res.x is None:
            return None
        g0 = np.maximum(res.x, 0.0)
        gs = float(g0.sum())
        if gs < 1e-10:
            return None
        return g0 * (S / gs)

    # --- Reduced LP (top-K active constraints) as a faster fallback ---
    def lp_direction_reduced(a, K=80, t_limit=30.0):
        """LP using only the K largest self-convolution lags."""
        S = float(a.sum())
        if S < 1e-10:
            return None
        conv_aa = np.convolve(a, a)
        M = float(conv_aa.max())
        if M < 1e-15:
            return None

        # Pick top-K lags
        k_idx = np.argsort(conv_aa)[::-1][:K]
        # Build reduced matrix: K x n
        # For each selected lag k, column j has a[j] if j <= k <= j+n-1
        rows_r, cols_r, vals_r = [], [], []
        for ii, k in enumerate(k_idx):
            j_lo = max(0, k - (n - 1))
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                rows_r.append(ii)
                cols_r.append(j)
                vals_r.append(a[j])
        if not rows_r:
            return None
        A_red = csr_matrix((vals_r, (rows_r, cols_r)), shape=(K, n))

        c_obj = -np.ones(n)
        b_ub = np.full(K, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c_obj, A_ub=A_red, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': t_limit})
        except Exception:
            return None
        if res.status != 0 or res.x is None:
            return None
        g0 = np.maximum(res.x, 0.0)
        gs = float(g0.sum())
        if gs < 1e-10:
            return None
        return g0 * (S / gs)

    # --- Phase 1: LP iterations ---
    stall = 0
    lp_count = 0
    use_full_lp = True

    while time.monotonic() < deadline - 150:
        remaining = deadline - time.monotonic() - 30
        if remaining < 10:
            break

        if use_full_lp:
            t_lim = min(120.0, max(5.0, remaining * 0.35))
            g = lp_direction(a, t_limit=t_lim)
            if g is None:
                use_full_lp = False  # fall back to reduced
                continue
        else:
            g = lp_direction_reduced(a, K=80, t_limit=30.0)
            if g is None:
                stall += 1
                if stall >= 4:
                    break
                a = np.clip(best + rng.normal(0, 0.02 * best.mean(), n), 0, 1000)
                continue

        lp_count += 1
        improved = False
        for t in [0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.02]:
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0.0, 1000.0)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score - 1e-14:
                best = cand.copy()
                best_score = sc
                a = cand.copy()
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 5:
                break
            a = np.clip(best + rng.normal(0, 0.01 * best.mean(), n), 0, 1000)

    # --- Phase 2: Gradient-guided local search ---
    while time.monotonic() < deadline - 10:
        a_cur = best.copy()
        S = float(a_cur.sum())
        if S < 1e-10:
            break
        conv = np.convolve(a_cur, a_cur)
        k_star = int(np.argmax(conv))

        # Subgradient direction: increase a[i] where it contributes least to peak
        # Contribution of a[i] to conv[k_star]: a[k_star - i] (if valid) + a[i] (if 2i == k_star)
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a_cur[j]
            if k_star == 2 * i:
                contrib[i] += a_cur[i]
        # We want to move mass from high-contrib to low-contrib
        t = 0.05
        scale = 0.1 * S / n
        a_cur -= t * (contrib - contrib.mean()) * scale / max(contrib.max(), 1e-15)
        a_cur = np.clip(a_cur, 0.0, 1000.0)
        sc = float(evaluate_sequence(a_cur.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = a_cur.copy()
            best_score = sc

    # --- Phase 3: Block-perturbation local search ---
    ls_deadline = deadline
    a_ls = best.copy()
    cur_s = best_score
    n_ls_iter = 0
    while time.monotonic() < ls_deadline:
        w = int(max(1, n // rng.integers(200, 300)))
        s = int(rng.integers(0, n - w + 1))
        cand = a_ls.copy()
        factor = np.exp(rng.normal(0.0, 0.06))
        cand[s:s + w] *= factor
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])
        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        n_ls_iter += 1
        if np.isfinite(sc) and sc < cur_s:
            cur_s = sc
            a_ls = cand
            if sc < best_score:
                best = a_ls.copy()
                best_score = cur_s

    # --- Final validation ---
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        # Extreme fallback
        return [1.0] * 64

    return [float(x) for x in best]
