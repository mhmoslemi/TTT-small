from scipy.fft import fft, ifft

def _fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = n + m - 1
    # Next power of 2
    fft_size = 1 << (size - 1).bit_length()
    F = fft(a, fft_size) * fft(b, fft_size)
    return np.real(ifft(F))[:size]

    """Compute the objective for a normalized sequence (sum=1)."""
    n = len(f)
    conv = _fft_convolve(f, f)

def _gradient_at_lags(f, lags):

    n = len(f)

import numpy as np
import time
from scipy.fft import fft, ifft


def _fft_conv(a, b):
    """Full linear convolution via FFT, returns array of length len(a)+len(b)-1."""
    n, m = len(a), len(b)
    size = n + m - 1
    fs = 1 << (size - 1).bit_length()
    return np.real(ifft(fft(a, fs) * fft(b, fs)))[:size]


def _score(f):
    """Objective for normalized sequence (sum=1): 2*n*max(conv(f,f))."""
    n = len(f)
    return 2.0 * n * float(np.max(_fft_conv(f, f)))


def _grad_lags(f, lags):
    """Average gradient of (f*f)[k] w.r.t. f, averaged over given lags."""
    n = len(f)
    grad = np.zeros(n)
    for k in lags:
        k = int(k)
        # Term 1: f[k-j] valid when 0<=k-j<n => j in [max(0,k-n+1), min(k,n-1)]
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js <= je:
            idx = np.arange(js, je + 1)
            grad[js:je + 1] += f[k - idx]
        # Term 2: f[j-k] valid when 0<=j-k<n => j in [max(k,0), min(k+n-1,n-1)]
        js2 = max(k, 0)
        je2 = min(k + n - 1, n - 1)
        if js2 <= je2:
            idx = np.arange(js2, je2 + 1)
            grad[js2:je2 + 1] += f[idx - k]
    return grad / len(lags)

    n = 6962
