    # A[x, j] = f[x-j] if 0 <= x-j < n, else 0

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix


def _quick_score(a):
    """Fast score for internal comparison (no safety checks)."""
    c = np.convolve(a, a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    return 2.0 * len(a) * c.max() / (s * s)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 20.0)

    # Initialize from parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(3809, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    n = len(parent)
    total = parent.sum()
    if total < 0.01:
        f = np.ones(n, dtype=np.float64) / n
    else:
        f = (parent / total).astype(np.float64)

    best_f = f.copy()
    best_score = float(evaluate_sequence(f.tolist()))
    if not np.isfinite(best_score):
        f = np.ones(n, dtype=np.float64) / n
        best_f = f.copy()
        best_score = float(evaluate_sequence(f.tolist()))

    if n <= 1:
        return [float(x) for x in best_f]

    # Precompute sparsity pattern for constraint matrix A
    # A is (2n-1) x n; A[x,j] = f[x-j] if 0 <= x-j < n
    nnz = n * n
    row_arr = np.empty(nnz, dtype=np.int32)
    col_arr = np.empty(nnz, dtype=np.int32)
    pos_arr = np.empty(nnz, dtype=np.int32)
    idx = 0
    for x in range(2 * n - 1):
        js = max(0, x - n + 1)
        je = min(n - 1, x)
        cnt = je - js + 1
        row_arr[idx:idx + cnt] = x
        col_arr[idx:idx + cnt] = range(js, je + 1)
        pos_arr[idx:idx + cnt] = range(je, js - 1, -1)
        idx += cnt

    # Determine available solver
    solver = None
    for s in [cp.GUROBI, cp.GLOP, cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            v = cp.Variable(3, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(v)), [cp.sum(v) == 1])
            p.solve(solver=s)
            if v.value is not None:
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.SCS

    def solve_lp(current_f):
        """Solve the LP: min t s.t. A@g <= t, g>=0, sum(g)=1."""
        vals = current_f[pos_arr]
        A = csr_matrix((vals, (row_arr, col_arr)), shape=(2 * n - 1, n))
        g_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable(lower=0)
        constraints = [
            A @ g_var <= t_var,
            cp.sum(g_var) == 1
        ]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            try:
                prob.solve()
            except Exception:
                return None
        val = g_var.value
        if val is None:
            return None
        return np.asarray(val, dtype=np.float64).flatten()

    restart_count = 0

            gs = g_val.sum()
            if gs < 1e-10:

            g_val /= gs

            for alpha in (1.0, 0.7, 0.5, 0.3):

                candidate = np.clip(candidate, 0.0, None)

                sc = _quick_score(candidate)

                    best_candidate = candidate

            if best_candidate is not None:

            else:

        else:

            c = np.convolve(f, f)
            max_c = c.max()
            top_k = np.where(c >= max_c * (1.0 - 1e-4))[0]
            if len(top_k) == 0:

            for k in top_k:

                hi = min(n - 1, k)
                if hi >= lo:
                    grad[lo:hi + 1] += 2.0 * f[k - hi:k - lo + 1][::-1]
            grad /= float(len(top_k))
            grad -= grad.mean()

            step = 0.1 / (1.0 + 0.02 * stagnation)
            f_new = f - step * grad
            f_new = np.clip(f_new, 0.0, None)
            fs = f_new.sum()
            if fs < 1e-10:
                break
            f = f_new / fs

        # Update best
        sc = _quick_score(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Periodic restart to escape local minima
        if stagnation >= 80:
            rng = np.random.default_rng((seed + 7919 * (restart_count + 1)) & 0xFFFFFFFF)
            f = f * np.exp(rng.normal(0.0, 0.3, n))

            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()

            stagnation = 0

    final_score = float(evaluate_sequence(best_f.tolist()))
    if np.isfinite(final_score):
        return [float(x) for x in best_f]

    # Ultimate fallback
    return [1.0 / n] * n
