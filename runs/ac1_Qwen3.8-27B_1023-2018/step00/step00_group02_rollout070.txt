import math
import time

import numpy as np


def _clip_parent_to_array(raw):
    if raw is None:
        return None
    arr = np.asarray(raw, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return None
    arr = arr.copy()
    if not np.all(np.isfinite(arr)):
        return None
    arr = np.clip(arr, 0.0, 1000.0)
    return arr


def _fft_full_convolve(a, b, n_out, n_fft):
    fa = np.fft.rfft(a, n=n_fft)
    fb = np.fft.rfft(b, n=n_fft)
    out = np.fft.irfft(fa * fb, n=n_fft)
    return out[:n_out]


def _fft_autoconvolve(p, L, N):
    fp = np.fft.rfft(p, n=N)
    out = np.fft.irfft(fp * fp, n=N)
    return out[:L]


def _project_simplex(v):
    x = np.asarray(v, dtype=np.float64).copy()
    n = x.size
    if n == 0:
        return x
    if not np.all(np.isfinite(x)):
        x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
    s = x.sum()
    if s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    x = x / s
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    good = np.where(u * rho > (cssv - 1.0))[0]
    if good.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    idx = int(good[-1])
    theta = (cssv[idx] - 1.0) / float(idx + 1)
    y = np.maximum(x - theta, 0.0)
    ysum = y.sum()
    if ysum <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return y / ysum


def _logit_map(w):
    w = np.asarray(w, dtype=np.float64)
    if w.size == 0:
        return w
    if not np.all(np.isfinite(w)):
        w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    w = w - np.max(w)
    p = np.exp(w)
    s = p.sum()
    if s <= 1e-300 or not math.isfinite(float(s)):
        return np.full(w.size, 1.0 / w.size, dtype=np.float64)
    return p / s


def _soft_autocorrelation_max_and_gradient(p, L, Nq, Nc, beta):
    q = _fft_autoconvolve(p, L, Nq)
    if not np.all(np.isfinite(q)):
        q = np.nan_to_num(q, nan=0.0, posinf=0.0, neginf=0.0)

    qmax = float(np.max(q))
    if not math.isfinite(qmax):
        qmax = 0.0

    scaled = beta * (q - qmax)
    exp_q = np.exp(scaled)
    z = float(np.sum(exp_q))
    if z <= 1e-300 or not math.isfinite(z):
        s = np.zeros(L, dtype=np.float64)
        s[L // 2] = 1.0
    else:
        s = exp_q / z

    # r[j] = sum_m p[m] * s[m+j]
    # Use convolution with reversed s:
    # conv = p * s[::-1], r[j] = conv[L-1-j].
    srev = s[::-1]
    conv = _fft_full_convolve(p, srev, n_out=p.size + L - 1, n_fft=Nc)
    if p.size == 1:
        r = np.array([float(2.0 * p[0] * s[0])], dtype=np.float64)
    else:
        r = conv[L - 1 : L - p.size - 1 : -1]
        if r.size != p.size:
            r = np.zeros(p.size, dtype=np.float64)
            fill = min(r.size, L - 1)
            r[:fill] = 2.0 * p[:fill] * s[:fill]
    grad_p = 2.0 * r
    if not np.all(np.isfinite(grad_p)):
        grad_p = np.nan_to_num(grad_p, nan=0.0, posinf=0.0, neginf=0.0)

    return qmax, grad_p


def _fast_ac1(p, n, L, Nq):
    s = float(np.sum(p))
    if s < 0.01:
        return math.inf
    q = _fft_autoconvolve(p, L, Nq)
    m = float(np.max(q))
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))

        fast = float(2.0 * n * qmax)
        if not math.isfinite(fast):
            break

        if fast < best_fast - 1e-14:
            best_fast = fast
            best_p = p.copy()
            stagnant = 0
        else:
            stagnant += 1

        dg = grad_p - float(np.dot(p, grad_p))
        gw = p * dg
        if not np.all(np.isfinite(gw)):
            gw = np.nan_to_num(gw, nan=0.0, posinf=0.0, neginf=0.0)
        gw = np.clip(gw, -1e6, 1e6)
        if not np.all(np.isfinite(gw)):
            break

def _simple_structured_starts(n, rng, seed, parent):
    starts = []

    if parent is not None and parent.sum() > 0.01:
        p = np.clip(parent, 0.0, 1000.0)
        s = p.sum()
        if s > 0.01:
            starts.append((0.18, "parent", p / s))

    starts.append((0.12, "uniform", np.full(n, 1.0 / n, dtype=np.float64)))

    p = rng.dirichlet(np.ones(n, dtype=np.float64) * 1.0)
    starts.append((0.10, "dirichlet1", p))

    p = rng.dirichlet(np.ones(n, dtype=np.float64) * 0.25)
    starts.append((0.12, "dirichlet025", p))

    p = np.exp(rng.normal(0.0, 0.9, n))
    starts.append((0.10, "expnormal", p / p.sum()))

    p = np.ones(n, dtype=np.float64)
    half = max(1, n // 2)
    p[:half] = 1.25
    p[half:] = 0.75
    starts.append((0.08, "two_block", p / p.sum()))

    idx = np.arange(n, dtype=np.float64)
    p = 1.0 + 0.35 * np.cos(2.0 * math.pi * idx / float(n))
    p = np.maximum(p, 0.0)
    p = p / p.sum()
    starts.append((0.08, "cosine", p))

    p = 1.0 + 0.35 * np.sin(2.0 * math.pi * idx / float(n))
    p = np.maximum(p, 0.0)
    p = p / p.sum()
    starts.append((0.08, "sine", p))

    # A low-frequency random-walk-ish positive profile.
    x = np.cumsum(rng.normal(0.0, 1.0, n))
    x = x - 0.5 * x[0] - 0.5 * x[-1]
    x = x / max(1e-12, np.ptp(x))
    p = np.exp(0.7 * x)
    starts.append((0.12, "smooth_random", p / p.sum()))

    # Rescale weights conservatively to respect intended total budget.
    total = sum(w for w, _, _ in starts)
    return [(max(0.02, w / total), name, p) for w, name, p in starts if p is not None]
