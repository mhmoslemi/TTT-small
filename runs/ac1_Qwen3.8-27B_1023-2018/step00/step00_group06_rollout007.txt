def _compute_conv(f):
    """Compute linear convolution f*f efficiently."""
    n = len(f)
    # Use FFT for efficiency
    size = 2 * n - 1
    # Next power of 2 for efficient FFT
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    F = fft.rfft(f, fft_size)
    conv = fft.irfft(F * F, fft_size)[:size]

            # = 2 * sum_{m=0}^{n-1} f_m * w_{j+m} [j+m < 2n-1]
            # = 2 * (reverse-convolve)(f, w)[j]  ... 
            # = 2 * (w_rev * f)[2n-2-j] where w_rev is w reversed... 
            # 
            # Actually the simplest way: pad w to length 2n-1 (it's already that length)
            # g_j = 2 * sum_{m=0}^{n-1} f_m * w_{j+m} = 2 * cross-correlation(f, w)[j]
            # = 2 * (f * w_rev)[n-1-j] where w_rev[m] = w[2n-2-m]
            # Hmm, let me just compute it with FFT.
            
            # g_j = sum_k w_k * f_{k-j} [0<=k-j<n, 0<=k<2n-1]
            #      = sum_{m=0}^{n-1} f_m * w_{j+m} [j+m < 2n-1]
            # Let's define: h_j = sum_{m=0}^{n-1} f_m * w_{j+m}
            # This is like a "shifted sum" or "Sliding dot product".
            # h_j = <f, shift_j(w)> where shift_j(w)_m = w_{j+m}
            # 
            # In terms of FFT: h = ifft(fft(f, 2n) * conj(fft(w, 2n))) ... no that's circular cross-correlation.
            # 
            # Actually, h_j = sum_m f_m w_{j+m} is the cross-correlation of f and w at lag j
            # = (f ⛓ w)[j] = ifft(fft(f) * conj(fft(w)))[j] (circular, with zero padding)
            # 
            # With zero padding to 2n-1 (or next power of 2), this gives the linear cross-correlation.
            
            # So: grad_j ≈ 2 * (f ⛓ w)[j] = 2 * ifft(fft(f, N) * conj(fft(w, N)))[j] for j=0..n-1
            # where N >= 2n-1
            # 
            # Wait, cross-correlation (f ⛓ w)[j] = sum_m f[m] conj(w[m+j]) for real: sum_m f[m] w[m+j]
            # Yes! That's exactly h_j.
            
            N_fft = 1
            while N_fft < 2*n - 1:
                N_fft *= 2
            
            F_f = fft.rfft(f, N_fft)
            F_w = fft.rfft(weights, N_fft)
            cross_corr = fft.irfft(F_f * np.conj(F_w), N_fft)[:2*n-1]
            # cross_corr[j] = sum_m f[m] * weights[j+m] for m=0..min(n-1, 2n-2-j)
            # But we want sum_m f[m] * weights[j+m] for m=0..n-1 (only if j+m < 2n-1)
            # For j < n: j+m < j+n <= n-1+n = 2n-1, so all terms are valid
            # Actually j+m goes from j to j+n-1, and we need j+m < 2n-1, i.e., j < n
            # For j in 0..n-1, j+n-1 <= 2n-2 < 2n-1, so all valid.
            
            grad = 2.0 * cross_corr[:n]
            
            # Add regularization for sum constraint and non-negativity
            # Projected gradient: f_new = max(0, f - lr * grad) then rescale to sum=1
            f_new = f - lr * grad
            f_new = np.maximum(f_new, 0)
            if f_new.sum() > 1e-10:
                f_new = f_new / f_new.sum()
            
            score = _score(f_new)
            if np.isfinite(score):
                # Rescale back
                f_new_s = f_new * S_norm
                score_s = _score(f_new_s)
                if score_s < best_score:
                    best = f_new_s.copy()
                    best_score = score_s
                    f = f_new
                elif score_s < _score(f * S_norm):
                    f = f_new
            
            # Adaptive learning rate
            if it % 50 == 0 and it > 0:
                lr *= 0.9
    
    # Final check
    final_score = float(evaluate_sequence(best.tolist()))
    if final_score < best_score:
        best_score = final_score
    
    return [float(x) for x in best]
