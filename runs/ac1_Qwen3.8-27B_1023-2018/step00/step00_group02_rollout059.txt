        # f_rev[k] = f[2n-2-k]
        # (g * f_rev)[m] = sum_j g[j] f_rev[m-j] = sum_j g[j] f[2n-2-m+j]
        # Setting 2n-2-m+j = i+j: m = 2n-2-i
        # (g * f_rev)[2n-2-i] = sum_j g[j] f[i+j] = h[i]
        
        # So h[i] = (g * f_rev)[2n-2-i]
        # We can compute g * f_rev via FFT with padding to length >= n + (2n-1) - 1 = 3n-2.
        # But fft_len is only >= 2n-1. Need a larger FFT.
        
        # Let me use a separate, larger FFT for the gradient:
        # grad_fft_len = 1 << (3*n-1).bit_length()
        
        # For n=6962: 3n-2 = 20884, next power of 2 = 32768.
        # This doubles the FFT cost but is still fast.
        
        it += 1
        
        # Use direct correlation for now (simpler, and numpy is fast)
        h = np.correlate(phi, a, mode='full')[n-1:n-1+n]
        grad = 2.0 * p * h  # gradient of sum c^p
        
        # Projected gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        S_new = a_new.sum()
        if S_new > 1e-10:
            a_new /= S_new
        
        s_new = fast_score(a_new)
        
        if s_new < best_score:
            best = a_new.copy()
            best_score = s_new
            stuck_count = 0
        else:
            stuck_count += 1
        
        a = a_new
        
        # Adaptive lr
        if stuck_count > 50:
            lr *= 0.7
            stuck_count = 0
            # Occasionally add a perturbation
            if it % 200 == 0:
                a = a * (1 + 0.05 * rng.standard_normal(n))
