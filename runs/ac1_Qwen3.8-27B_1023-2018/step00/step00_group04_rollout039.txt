import numpy as np
from scipy import sparse
import cvxpy as cp
import time

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Build the constraint matrix A such that A @ g gives f*g
    # (f*g)[k] = sum_j f[j] * g[k-j] for k = 0, ..., 2n-2
    # Row k of A has entries A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                vals.append(f[j])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraint = A @ g <= M
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, [constraint])
    prob.solve()
    
    return g.value
```

def build_conv_matrix_fast(a, n):
    L = 2 * n - 1
    # Precompute row/column ranges
    i_starts = np.maximum(0, np.arange(L) - n + 1)
    i_ends = np.minimum(n, np.arange(L) + 1)
    counts = i_ends - i_starts
    
    total_nz = counts.sum()
    data = np.empty(total_nz, dtype=np.float64)
    indices = np.empty(total_nz, dtype=np.int32)
    row_ptr = np.zeros(L + 1, dtype=np.int64)
    row_ptr[1:] = np.cumsum(counts)
    
    pos = 0
    for k in range(L):
        cnt = counts[k]
        if cnt > 0:
            idx = np.arange(i_starts[k], i_ends[k])
            indices[pos:pos+cnt] = idx
            data[pos:pos+cnt] = a[k - idx]
            pos += cnt
    
    return sparse.csr_matrix((data, indices, row_ptr), shape=(L, n))

The EDGE autocorrelation entries (small k or large k) have FEWER terms in the sum, so their mean is smaller (proportional to the number of terms).

For example, (a*a)[0] = a[0]². E[a[0]²] = E[X[0]²/S²] ≈ (1/n²) * E[X²] = (1/n²) * 2. So the edge peak is about 2/n², which is much smaller than the center peak of 1/n.

So for a random sequence, the peak is at the center (value ≈ 1/n, same as uniform) with some variability. The peak-to-mean ratio is ≈ 2 (same as uniform) with some randomness that could go either way.

This confirms that a simple random initialization won't do better than uniform. We need a more clever construction.
