class _FFTConv:
    def __init__(self,n):
       self.n=n
       L=1
       need=3*n-2
       while L<need: L<<=1
       self.L=L
    def autocorr(self,p):
       f=np.fft.rfft(p,self.L)
       c=np.fft.irfft(f*f,self.L)[:2*self.n-1]
       return np.maximum(c,0)
    def conv_pp(self,a,b): # a len 2n-1, b len n
       need=self.n+(2*self.n-1)-1
       fa=np.fft.rfft(a,self.L)
       fb=np.fft.rfft(b,self.L)
       return np.fft.irfft(fa*fb,self.L)

import numpy as np

def _as_p(p):
    if p is None:
        return None
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if not np.isfinite(s) or s < 1e-12:
        return None
    return p / s


def _softmax(y):
    y = np.asarray(y, dtype=np.float64).ravel()
    if y.size == 0:
        return np.ones(0, dtype=np.float64)
    y = np.where(np.isfinite(y), y, -80.0)
    y = y - float(y.max())
    y = np.clip(y, -80.0, 80.0)
    e = np.exp(y)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(y.size, dtype=np.float64) / float(y.size)
    return e / s


class _FFTConv:
    def __init__(self, n):
        self.n = int(n)
        need = 3 * self.n - 2
        if need < 4:
            need = 4
        L = 1
        while L < need:
            L <<= 1
        self.L = L
        self.q_end = 2 * self.n - 1
        self.g0 = self.n - 1
        self.g1 = 2 * self.n - 1

    def autocorr(self, p):
        f = np.fft.rfft(p, self.L)
        c = np.fft.irfft(f * f, self.L)[:self.q_end]
        return np.maximum(c, 0.0)

    def conv_segment(self, a, b):
        fa = np.fft.rfft(a, self.L)
        fb = np.fft.rfft(b, self.L)
        c = np.fft.irfft(fa * fb, self.L)
        return c[self.g0:self.g1]


def _fft_score(p, fc):
    q = fc.autocorr(p)
    return float(2 * p.size * q.max())


def _weighted_gradient(fc, p, q, alpha):
    M = float(q.max())
    if not np.isfinite(M) or M <= 0.0:
        return None
    mn = float(q.min())
    span = M - mn
    if span < 1e-14:
        T = max(M / float(p.size), 1e-15)
    else:
        T = max(float(alpha) * span, 1e-15)
    z = (q - M) / T
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    w += 1e-12
    ws = float(w.sum())
    if not np.isfinite(ws) or ws <= 0.0:
        w = np.zeros(q.size, dtype=np.float64)
        w[int(np.argmax(q))] = 1.0
    else:
        w /= ws
    g = fc.conv_segment(w, p[::-1])
    return 2.0 * g

def _run_length(n, allot, rng, parent, target, state, global_deadline):
    if allot < 4.0:
        return None, np.inf
    if n is None or int(n) <= 0:
        return None, np.inf
    n = int(n)

    start = time.monotonic()
    deadline = min(start + float(allot), global_deadline)
    if deadline - start < 2.0:
        return None, np.inf

    try:
        keep = 4 if n >= 4096 else 5
        starts = _generate_starts(n, rng, parent, keep=keep)
    except Exception:
        starts = [np.ones(n, dtype=np.float64) / n]

    if not starts:
        starts = [np.ones(n, dtype=np.float64) / n]

    local_best_p = None
    local_best_int = np.inf

    rem = deadline - time.monotonic()
    if rem > 2.0:
        starts = starts[:max(1, int(rem // 20) + 2)]

    for i, p0 in enumerate(starts):
        rem = deadline - time.monotonic()
        if rem < 2.0:
            break
        lim = max(1.0, rem / float(len(starts) - i))
        try:
            p_int, int_sc = _optimize(n, p0, lim, rng, target)
        except Exception:
            p_int, int_sc = None, np.inf
        if p_int is not None and np.isfinite(int_sc):
            if int_sc < local_best_int:
                local_best_int = int_sc
                local_best_p = p_int
            _exact_validate(p_int, int_sc, state, target)
            if state["best_exact"] < target - 1e-7:
                return local_best_p, local_best_int

    if local_best_p is not None and np.isfinite(local_best_int):
        _exact_validate(local_best_p, local_best_int, state, target)

    return local_best_p, local_best_int

        budget = max(1.0, float(budget_s))

            if arr.size > 0 and np.any(np.isfinite(arr)):

    total = max(20.0, budget - 6.0)

    def try_run(n, allot):

        rem = global_deadline - time.monotonic()
        if rem < 5.0:
            return
        allot = min(float(allot), rem - 1.0)

        try_run.return = _run_length(int(n), allot, rng, parent, target, state, global_deadline)
