def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _convolve(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    out_len = a.size + b.size - 1
    if a.size == 0 or b.size == 0 or out_len <= 0:
        return np.zeros(0, dtype=np.float64)
    if a.size * b.size <= _CONV_DIRECT_LIMIT:
        return np.convolve(a, b)
    nff = _next_pow2(out_len)
    fa = np.fft.rfft(a, n=nff)
    fb = np.fft.rfft(b, n=nff)
    return np.fft.irfft(fa * fb, n=nff)[:out_len]

                try:
                    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0.0, 1.0))
                except Exception:
                    break
            if res is None or not getattr(res, "success", False) or getattr(res, "x", None) is None:
                break
            g = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, 1000.0)
            sg = float(g.sum())
            if not math.isfinite(sg) or sg <= 1.0 + 1e-8:
                break
            g /= sg
            fnew = (1.0 - t) * f + t * g
            fnew = np.clip(fnew, 0.0, 1000.0)
            s = float(fnew.sum())
            if s <= 0.0:
                break
            fnew /= s
            sc = _score_norm(fnew)
            sc0 = _score_norm(f)
            if math.isfinite(sc) and sc < sc0 - 1e-12:
                f = fnew
                _consider(f)
            else:
                t *= 0.5
                if t < 0.02:
                    break

    # ---- initialization ----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = None

    if parent is None or parent.size == 0 or not np.all(np.isfinite(parent)):
        parent = np.ones(2535, dtype=np.float64)
    else:
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() <= 0.0:
            parent = np.ones(parent.size, dtype=np.float64) if parent.size else np.ones(2535, dtype=np.float64)

    full_n = int(parent.size) if 8 <= parent.size <= 8000 else 2535

    _consider(parent)
    _consider(np.ones(full_n, dtype=np.float64))
    for n in (4, 5, 6, 7, 8, 10, 12, 16, 32):
        for cand in _simple_candidates(n, rng):
            _consider(cand)

    # ---- small / medium exploratory search ----
    small_ns = [
        (4, 10), (5, 12), (6, 14), (7, 14), (8, 18),
        (9, 18), (10, 22), (12, 25), (14, 30),
        (16, 35), (20, 45), (24, 55), (32, 70),
    ]
    for n, base in small_ns:
        rem = deadline - time.monotonic()
        if rem < 8.0:
            break
        scale = 0.45 if best["score"] < 1.55 else 1.0
        tlim = min(float(base) * scale, rem * 0.60)
        if tlim < 4.0:
            break
        loc = _de_search(n, tlim)
        if loc is not None:
            _consider(loc)
            rem = deadline - time.monotonic()
            if rem > 14.0:
                _optimize_lbfgs(loc, min(14.0, rem * 0.55), eps_factors=(10.0, 20.0))

    # ---- polish current best if medium-sized ----
    rem = deadline - time.monotonic()
    if best["seq"] is not None and best["seq"].size <= 512 and rem > 20.0:
        _optimize_lbfgs(best["seq"], min(45.0, rem * 0.30), eps_factors=(10.0, 20.0, 30.0))

    # ---- LP-inspired directions on medium-sized instances ----
    lp_targets = []
    if best["seq"] is not None:
        bn = best["seq"].size
        if 6 <= bn <= 1200:
            lp_targets.append(best["seq"])
        elif bn > 1200:
            lp_targets.append(_resample_to(best["seq"], 700))
            lp_targets.append(_resample_to(best["seq"], 1100))
    if parent is not None:
        if 6 <= parent.size <= 1200:
            lp_targets.append(parent)
        else:
            lp_targets.append(_resample_to(parent, 550))
            lp_targets.append(_resample_to(parent, 1000))

    seen_shapes = set()
    for arr in lp_targets:
        rem = deadline - time.monotonic()
        if rem < 45.0:
            break
        a = np.asarray(arr, dtype=np.float64).ravel()
        if a.size in seen_shapes:
            continue
        seen_shapes.add(a.size)
        tlim = min(55.0, rem * 0.40)
        _lp_heuristic(a, tlim)

    # ---- full-size softmax optimization around parent and promising seeds ----
    if deadline - time.monotonic() > 40.0:
        full_starts = []
        if parent is not None and parent.size == full_n:
            full_starts.append(parent.copy())
            full_starts.append(parent[::-1].copy())
            full_starts.append(np.roll(parent, max(1, full_n // 7)).copy())
            full_starts.append(np.roll(parent, max(1, full_n // 3)).copy())

        y = rng.normal(0.0, 0.7, full_n)
        full_starts.append(np.exp(y).astype(np.float64))
        y = rng.normal(0.0, 1.4, full_n)
        full_starts.append(np.exp(y).astype(np.float64))

        base = np.ones(full_n, dtype=np.float64)
        for period in (3, 4, 5, 7):
            z = int(rng.integers(0, period))
            v = (np.arange(full_n) % period != z).astype(np.float64)
            if v.sum() > 0.0:
                full_starts.append(v)

        if best["seq"] is not None and best["score"] < 2.2:
            full_starts.append(_resample_to(best["seq"], full_n))

        full_starts = [x for x in full_starts if x is not None and np.all(np.isfinite(x)) and x.sum() > 0.0]
        max_starts = 5
        for idx, arr in enumerate(full_starts[:max_starts]):
            rem = deadline - time.monotonic()
            if rem < 70.0:
                break
            if idx < 2:
                tlim = min(150.0, rem * 0.35)
            else:
                tlim = min(90.0, rem * 0.30)
            if tlim < 20.0:
                break
            eps_factors = (5.0, 10.0, 20.0) if tlim > 90 else (5.0, 10.0)
            _optimize_lbfgs(arr, tlim, eps_factors=eps_factors)

    # ---- final local polish ----
    rem = deadline - time.monotonic()
    if best["seq"] is not None and rem > 35.0:
        _peak_repair(best["seq"], min(90.0, rem * 0.55))
    rem = deadline - time.monotonic()
    if best["seq"] is not None and rem > 15.0:
        _block_local(best["seq"], min(60.0, rem * 0.80))

    # ---- final safe return ----
    seq = best["seq"]
    if seq is None or seq.size == 0 or not np.all(np.isfinite(seq)):
        seq = parent if (parent is not None and parent.size > 0) else np.ones(64, dtype=np.float64)
    seq = np.asarray(seq, dtype=np.float64).copy()
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(seq.sum())
    if s <= 0.0:
        seq = np.ones(64, dtype=np.float64)
    else:
        seq /= s

    out = [float(x) for x in seq]
    try:
        fs = float(evaluate_sequence(out))
        if not math.isfinite(fs):
            out = [1.0] * 64
    except Exception:
        out = [1.0] * 64

    return out
