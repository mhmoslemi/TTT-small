    if s < 1e-10:

    else:
        f = f / s
    
    best_f = f.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    
    # Main optimization loop
    tau = 50.0 / n
    lr = 1.0 / (n * tau) * 0.5
    momentum = np.zeros(n)
    beta_m = 0.85
    count = 0
    last_improve = 0
    
    while time.monotonic() < hard_deadline:
        # Compute autocorrelation
        c = np.convolve(f, f)
        c_max = c.max()
        # Smoothed max via log-sum-exp
        exp_c = np.exp(tau * (c - c_max))
        exp_sum = exp_c.sum()
        if exp_sum < 1e-300:
            w = np.zeros_like(c)
            w[c.argmax()] = 1.0
        else:
            w = exp_c / exp_sum
        
        # Gradient of smooth max
        grad = 2.0 * np.correlate(w, f, mode='full')[:n]
        
        # Momentum step
        momentum = beta_m * momentum + grad
        f_new = f - lr * momentum
        f_new = np.maximum(f_new, 0.0)
        s2 = f_new.sum()
        if s2 < 1e-15:
            f_new = f
        else:
            f_new = f_new / s2
        f = f_new
        count += 1
        
        # Every 200 iterations: evaluate and adapt
        if count % 200 == 0:
            score = float(evaluate_sequence(f.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_f = f.copy()
                last_improve = count
            
            # Adapt parameters
            if count - last_improve > 2000:
                # Stuck - add perturbation
                perturb = rng.normal(0, 0.01, n)
                f = f + perturb * f
                f = np.maximum(f, 0)
                s3 = f.sum()
                if s3 < 1e-15:
                    f = best_f.copy()
                else:
                    f = f / s3
                momentum *= 0.5
                last_improve = count
            elif count - last_improve > 500:
                # No improvement for a while - reduce tau slightly, reduce lr
                lr *= 0.95
        
        # Gradually increase tau (make max sharper)
        if count % 5000 == 0:
            tau = min(tau * 1.02, 500.0 / n)
            lr = 0.5 / (n * tau) * 0.1  # reduce step as tau increases
    
    # Final evaluation
    final_score = float(evaluate_sequence(best_f.tolist()))
    if not np.isfinite(final_score) or final_score > best_score:
        best_f = f.copy()
        best_score = float(evaluate_sequence(best_f.tolist()))
    
    if not np.isfinite(best_score):
        return [1.0/n] * n
    
    return [float(x) for x in best_f]
