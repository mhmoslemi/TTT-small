        best, best_score = a.copy(), sc

    p1_end = time.monotonic() + 0.28 * T
    lr, stall = 0.4, 0
    while time.monotonic() < p1_end:
        grad = _grad_multi_lag(a, K=10, eta=20.0)
        a_new = _proj_simplex(a - lr * grad)
        sc = _score_norm(a_new)
        if sc < best_score - 1e-13:
            best, best_score, a = a_new.copy(), sc, a_new
            stall = 0
            lr = min(lr * 1.15, 3.0)
        else:
            a, stall, lr = a_new, stall + 1, lr * 0.88
            if stall > 80:
                a = _proj_simplex(best * np.exp(rng.normal(0, 0.06, n)))
                stall, lr = 0, 0.4
        if time.monotonic() > deadline - 0.25 * T:
            break

    # ── Phase 2: LP-based direction ──
    p2_end = max(p1_end, time.monotonic()) + 0.28 * T
    try:
        import cvxpy as cp
        for _ in range(100):
            if time.monotonic() > p2_end:
                break
            conv_b = np.convolve(best, best)
            M = float(conv_b.max())
            # Build C: (2n-1) x n, C[k,j] = best[k-j] for valid k,j
            C = np.zeros((2 * n - 1, n))
            for j in range(n):
                C[j:j + n, j] = best
            bv = cp.Variable(n, nonneg=True)
            pr = cp.Problem(cp.Maximize(cp.sum(bv)), [C @ bv <= M])
            bval = None
            for sl in [cp.ECOS, cp.CBC, cp.GLOP, cp.PDLP, cp.GLPK]:
                try:
                    pr.solve(solver=sl, time_limit=15)
                    if bv.value is not None:
                        bval = np.asarray(bv.value, dtype=np.float64)
                        break
                except Exception:
                    pass
            if bval is None:
                if time.monotonic() > p2_end - 30:
                    break
                continue
            bval = np.maximum(bval, 0)
            bs = bval.sum()
            if bs > 1.0 + 1e-4:
                bn = bval / bs
                for t in (0.25, 0.45, 0.65, 0.85):
                    at = _proj_simplex((1 - t) * best + t * bn)
                    sc = _score_norm(at)
                    if sc < best_score:
                        best, best_score = at.copy(), sc
                        break
    except Exception:
        pass

    # ── Phase 3: diverse perturbations ──
    while time.monotonic() < deadline:
        at = best.copy()
        m = rng.integers(0, 6)
        if m == 0:
            w = int(rng.integers(max(1, n // 500), max(2, n // 5)))
            s = int(rng.integers(0, n - w + 1))
            at[s:s + w] *= np.exp(rng.normal(0, 0.10))
        elif m == 1:
            at = 0.7 * at + 0.3 * at[::-1]
            at *= np.exp(rng.normal(0, 0.04, n))
        elif m == 2:
            ix = rng.integers(0, n, size=min(200, n))
            at[ix] *= np.exp(rng.normal(0, 0.20, size=len(ix)))
        elif m == 3:
            c = np.cos(np.pi * np.arange(n) / max(1, n - 1))
            at *= 1.0 + 0.15 * (c - c.mean())
        elif m == 4:
            c = np.exp(-0.5 * ((np.arange(n) / n - rng.uniform()) ** 2) / (0.05 ** 2))
            at *= 1.0 + 0.1 * (c - c.mean())
        else:
            # Random smooth ridge replacement
            center = rng.uniform(0.1, 0.9) * n
            width = rng.uniform(0.05, 0.3) * n
            bump = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
            at = 0.8 * at + 0.2 * bump
        at = np.maximum(at, 0)
        if at.sum() < 0.01:
            continue
        at /= at.sum()
        sc = _score_norm(at)
        if sc < best_score:
            best, best_score = at, sc

    # ── Final: local gradient polish ──
    if time.monotonic() < deadline - 2:
        lr = 0.05
        for _ in range(200):
            if time.monotonic() > deadline - 2:
                break
            grad = _grad_multi_lag(best, K=15, eta=30.0)
            a_new = _proj_simplex(best - lr * grad)
            sc = _score_norm(a_new)
            if sc < best_score - 1e-13:
                best, best_score = a_new.copy(), sc
                lr = min(lr * 1.1, 0.5)
            else:
                lr *= 0.7
                if lr < 1e-5:
                    break

    # ── Return ──
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.ones(n) / n
    return [float(x) for x in best]
