import time
import numpy as np

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None
try:
    from scipy.sparse import csr_matrix
except Exception:
    csr_matrix = None


def _next_pow2(m):
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()


def _parent_array():
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.ndim == 1 and h.size > 0 and np.isfinite(h).all() and h.sum() > 0:
            return np.clip(h, 0.0, 1000.0)
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    horizon = max(0.0, float(budget_s))
    deadline = time.monotonic() + (horizon - 4.0)

    parent = _parent_array()
    if parent is None:
        # smooth bump fallback
        m = 2048
        x = np.linspace(0.0, 1.0, m)
        parent = np.exp(-8.0 * (x - 0.5) ** 2)
        parent = parent + 1e-3
    n = int(parent.size)
    f = parent.astype(np.float64).copy()
    s = f.sum()
    if s <= 0:
        f[:] = 1.0
        s = float(n)
    f = f / s  # work with sum == 1 (objective is homogeneous)

    L2 = _next_pow2(2 * n - 1)
    L3 = _next_pow2((2 * n - 1) + n)

    def autocorr(x):
        a = np.fft.rfft(x, L2)
        return np.fft.irfft(a * a, L2)[:2 * n - 1]

        return float(autocorr(x).max())

            return float(evaluate_sequence([float(v) for v in x]))

            return float("inf")

    best = f.copy()

        best = u.copy()

    K = 60
    n_iters = 0
    no_improve = 0

        b = autocorr(f)
        M = float(b.max())
        if M <= 0 or not np.isfinite(M):
            break

        # active lags: top K by convolution value

        S = np.sort(order)

            rows = []

        best_score = sc

        best = u
