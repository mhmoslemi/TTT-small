import numpy as np
import time


def _project_simplex(v):
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho = np.nonzero(u * idx > (cssv - 1))[0]
    if rho.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    out = v - theta
    np.maximum(out, 0.0, out)
    return out


def _conv_full(a):
    """Full linear convolution a * a via FFT, length 2n-1."""
    n = a.size
    N = 1
    need = 2 * n - 1
    while N < need:
        N <<= 1
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)
    return r[:need]


def _ac1(a):
    """Faithful (FFT-computed) AC1 objective: 2n*max(conv)/(sum a)^2."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    c = _conv_full(a)
    return float(2.0 * n * c.max() / (s * s))


def _smoothed_max_grad(a, mu):
    """
    Returns gradient of f = mu*log(sum_k exp(c[k]/mu)) where c=conv(a,a),
    i.e. g[j] = 2*sum_k softmax[c/mu][k] * a[k-j].
    Computed via one cross-correlation (FFT).
    """
    n = a.size
    c = _conv_full(a)
    cm = c / mu
    cm = cm - cm.max()
    e = np.exp(cm)
    s = e / e.sum()                 # softmax weights over lags
    af = a[::-1]
    N = 1
    need = (2 * n - 1) + (n - 1)
    while N < need:
        N <<= 1
    rs = np.fft.rfft(s, N)
    ra = np.fft.rfft(af, N)
    r = np.fft.irfft(rs * ra, N)
    g = 2.0 * r[n - 1:2 * n - 1]    # g[j] for j = 0..n-1
    return g, c
