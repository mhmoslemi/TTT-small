def _project_onto_simplex(v, target=1.0):
    """Project v onto {a >= 0, sum a = target}."""
    if v.sum() <= 0:
        return np.full_like(v, target / len(v))
    # Find lambda such that sum max(0, v_i - lambda) = target
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    rho = np.nonzero(sorted_v * np.arange(1, len(v)+1) > (cumsum[-1] - target))[0]
    if len(rho) == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = (cumsum[rho] - target) / (rho + 1)
    return np.maximum(v - theta, 0)

            # Reduce top-k smoothing for sharper descent
            top_k_smooth = max(1, top_k_smooth - 2)
            # Add noise
            sigma = 0.005 * (1 + patience / max_patience)
            current = current + sigma * rng.normal(0, 1, n)
            current = np.maximum(current, 0)
            if current.sum() > 0:
                current = current / current.sum()
            patience = 0
            lr = min(max_lr, lr * 2.0)

        # Major restart: try new parametrized shape
        if patience >= max_patience * 4 and total_iters > 500:
            alpha = rng.uniform(0.3, 3.0)
            new_shape = np.power(np.arange(n, 0, -1, dtype=np.float64), alpha)
            # Add asymmetry perturbation
            frac = rng.uniform(0.1, 0.5)
            new_shape[:int(n * frac)] *= rng.uniform(1.0, 2.0)
            new_shape *= np.exp(0.03 * rng.normal(0, 1, n))
            new_shape = np.maximum(new_shape, 0)
            if new_shape.sum() > 0:
                new_shape = new_shape / new_shape.sum()
                sc = _fast_score(new_shape)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best = new_shape.copy()
            current = new_shape if new_shape.sum() > 0 else best.copy()
            lr = 5e-5
            patience = 0
            top_k_smooth = 5

        # Fine-tune: smaller steps near convergence
        if best_score < 1.52 and patience < 20:
            lr = max(min_lr, lr * 0.99)

        it += 1
        # Check deadline more frequently
        if it % 50 == 0 and time.monotonic() > deadline - 30:
            # Enter polish mode: very small steps
            lr = min(lr, 1e-7)

    # Final polish: small deterministic steps
    if time.monotonic() < deadline - 5:
        lr_polish = 1e-7
        for _ in range(200):
            if time.monotonic() > deadline - 5:
                break
            conv = np.convolve(current, current)
            k_star = int(np.argmax(conv))
            i_lo = max(0, k_star - n + 1)
            i_hi = min(n - 1, k_star)
            idx = np.arange(i_lo, i_hi + 1)
            grad = np.zeros(n)
            grad[idx] = current[k_star - idx] + current[idx]
            v = current - lr_polish * grad
            current = _project_simplex(v, 1.0)
            sc = _fast_score(current)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = current.copy()

    # Final validation with official evaluator
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.ones(n, dtype=np.float64)
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # Fallback to uniform
        best = np.ones(n, dtype=np.float64)
        final_score = float(evaluate_sequence(best.tolist()))

    return [float(x) for x in best]
