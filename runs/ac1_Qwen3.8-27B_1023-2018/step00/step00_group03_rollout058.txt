def convolve_fast(p):
    """Compute full linear convolution of p with itself using FFT."""
    n = len(p)
    size = 2*n - 1
    # Pad to next power of 2 for efficiency
    fsize = 1
    while fsize < size:
        fsize *= 2
    P = fft(p, fsize)
    C = ifft(P * P).real[:size]
    return C
