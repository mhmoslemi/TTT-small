def _proj_simplex(v, target=1.0):
  v=np.asarray(v,float)
  n=v.size
  if n==0: return v
  if not np.all(np.isfinite(v)):
    v=np.where(np.isfinite(v), v, 0.0)
  u=np.sort(v)[::-1]
  K=np.arange(1,n+1,dtype=float)
  cum=u.cumsum()
  tmp = u - (cum - target)/K
  idx = np.flatnonzero(tmp > 0.0)
  if idx.size ==0:
     return np.full(n, target/n)
  rho=int(idx[-1])
  theta=(cum[rho]-target)/(rho+1)
  out=np.maximum(v-theta,0.0)
  s=out.sum()
  if s<=0 or not finite: return np.full(n,target/n)
  if abs(s-target)>1e-12: out *= target/s
  return out

def _convolve_auto(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)

        return np.empty(0, dtype=np.float64)

        if n <= 4096:
            for sig in (0.2, 0.6):
                add(np.maximum(p * np.exp(rng.normal(0.0, sig, size=n)), 0.0))

    num_rand = 5 if n <= 128 else 3 if n <= 512 else 2
    for conc in (0.15, 0.4, 1.0, 3.0):
        try:
            add(rng.dirichlet(np.full(n, conc, dtype=np.float64)))
        except Exception:
            pass

    for _ in range(num_rand):
        add(np.exp(rng.normal(0.0, 0.5, size=n)))

    return inits


def _optimize_start(c0, n, rng, start_deadline):
    n = int(n)
    c = np.asarray(c0, dtype=np.float64).copy()
    if c.size != n:
        c = np.resize(c, n)
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    s = c.sum()
    if not np.isfinite(s) or s <= 1e-12:
        c = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        c = c / s

    score_best = _score(c)
    if not np.isfinite(score_best):
        return c, np.inf

    x = _c_to_x(c)
    base_tau = 2.0 / (float(n) * float(n))
    tau_factors = (10.0, 2.0, 0.5, 0.12)
    obj_scale = max(1.0, (float(n) * float(n)) / 256.0)

    if n <= 32:
        maxfun_base = 260
    elif n <= 64:
        maxfun_base = 200
    elif n <= 128:
        maxfun_base = 150
    elif n <= 256:
        maxfun_base = 110
    elif n <= 512:
        maxfun_base = 80
    elif n <= 1024:
        maxfun_base = 60
    elif n <= 2048:
        maxfun_base = 45
    else:
        maxfun_base = 35

    for ti, fac in enumerate(tau_factors):
        now = time.monotonic()
        if now >= start_deadline - 3.0:
            break

        remaining = max(0.5, start_deadline - now)
        per_run = remaining / max(1, len(tau_factors) - ti)
        run_deadline = min(start_deadline - 0.5, time.monotonic() + per_run)

        tau = float(fac) * base_tau
        if not np.isfinite(tau) or tau <= 0.0:
            tau = 0.12 * base_tau

        maxfun = int(min(maxfun_base, max(30, per_run / 0.02)))

        last_x = np.array(x, dtype=np.float64, copy=True)

        def f_and_g(xk):
            nonlocal last_x
            if time.monotonic() > run_deadline:
                raise _Timeout
            xk = np.asarray(xk, dtype=np.float64)
            if not np.all(np.isfinite(xk)):
                raise _Timeout
            last_x = np.array(xk, dtype=np.float64, copy=True)

            cloc = _softmax(xk)
            if cloc.size != n or not np.all(np.isfinite(cloc)):
                raise _Timeout

            b = _convolve_auto(cloc, cloc)
            if b.size == 0 or not np.all(np.isfinite(b)):
                raise _Timeout

            bm = float(b.max())
            z = np.exp((b - bm) / tau)
            zsum = float(z.sum())

            if not np.isfinite(zsum) or zsum <= 0.0:
                H = bm
                p = np.zeros_like(b)
                amax = np.where(b >= bm - 1e-12 * max(1.0, bm))[0]
                if amax.size:
                    p[amax] = 1.0 / amax.size
            else:
                H = bm + tau * math.log(zsum)
                p = z / zsum

            score = 2.0 * float(n) * H
            grad = 2.0 * _corr_pc(p, cloc)
            grad = np.where(np.isfinite(grad), grad, 0.0)

            mean = float(np.dot(cloc, grad))
            gxc = cloc * (grad - mean)
            gxc = np.where(np.isfinite(gxc), gxc, 0.0)

            return float(score * obj_scale), gxc * obj_scale

        if minimize is not None:
            try:
                res = minimize(
                    f_and_g,
                    x,
                    method="L-BFGS-B",
                    jac=True,
                    options={
                        "maxfun": maxfun,
                        "maxiter": maxfun,
                        "ftol": 1e-12,
                        "gtol": 1e-9,
                        "maxls": 20,
                        "maxcor": 10,
                    },
                )
                if res is not None and getattr(res, "x", None) is not None:
                    x_new = np.asarray(res.x, dtype=np.float64)
                    if np.all(np.isfinite(x_new)):
                        c_new = _softmax(x_new)
                        c_new = np.maximum(c_new, 0.0)
                        sn = c_new.sum()
                        if np.isfinite(sn) and sn > 0.0:
                            c_new /= sn
                            sc_new = _score(c_new)
                            if np.isfinite(sc_new) and sc_new < score_best:
                                score_best = sc_new
                                c = c_new
                        x = x_new
            except _Timeout:
                if last_x is not None and np.all(np.isfinite(last_x)):
                    c_trial = _softmax(last_x)
                    c_trial = np.maximum(c_trial, 0.0)
                    st = c_trial.sum()
                    if np.isfinite(st) and st > 0.0:
                        c_trial /= st
                        sc_trial = _score(c_trial)
                        if np.isfinite(sc_trial) and sc_trial < score_best:
                            score_best = sc_trial
                            c = c_trial
                            x = last_x
            except Exception:
                pass

    if start_deadline - time.monotonic() > 1.5 and n <= 4096:
        polish_time = min(3.0 if n <= 1024 else 1.5, (start_deadline - time.monotonic()) * 0.5)
        pol_deadline = min(start_deadline, time.monotonic() + polish_time)
        try:
            c_pol, sc_pol = _hard_polish(c, n, rng, pol_deadline)
            if np.isfinite(sc_pol) and sc_pol < score_best:
                c = c_pol
                score_best = sc_pol
        except Exception:
            pass

    c = np.asarray(c, dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    s = c.sum()
    if np.isfinite(s) and s > 0.0:
        c /= s
    else:
        c = np.full(n, 1.0 / n, dtype=np.float64)

    return c, _score(c)


def _optimize_n(n, time_budget, rng, parent):
    n = int(n)
    if time_budget <= 5.0 or n < 2:
        return []

    deadline = time.monotonic() + float(time_budget)
    inits = _make_inits(n, rng, parent)

    scored = []
    for c in inits:
        sc = _score(c)
        if np.isfinite(sc):
            scored.append((sc, c))

    if not scored:
        return []

    scored.sort(key=lambda z: z[0])

    if n <= 32:
        max_starts = 10
    elif n <= 64:
        max_starts = 9
    elif n <= 128:
        max_starts = 8
    elif n <= 256:
        max_starts = 6
    elif n <= 512:
        max_starts = 5
    elif n <= 1024:
        max_starts = 4
    elif n <= 2048:
        max_starts = 3
    else:
        max_starts = 2

    candidates = []

    for idx, (_, c0) in enumerate(scored):
        now = time.monotonic()
        if now >= deadline - 6.0 or idx >= max_starts:
            break

        remaining = deadline - now
        per_start = min(120.0, max(6.0, remaining / max(1, max_starts - idx)))
        start_deadline = now + per_start

        try:
            c_opt, sc_opt = _optimize_start(c0, n, rng, start_deadline)
        except Exception:
            c_opt, sc_opt = c0, _score(c0)

        if np.isfinite(sc_opt):
            candidates.append((sc_opt, c_opt))

    if candidates and n <= 128 and linprog is not None:
        now = time.monotonic()
        if now < deadline - 12.0:
            lp_deadline = min(deadline - 2.0, now + min(18.0, max(5.0, deadline - now - 10.0)))
            try:
                c_lp, sc_lp = _lp_polish(candidates[0][1], lp_deadline, rng)
                if np.isfinite(sc_lp):
                    candidates.append((sc_lp, c_lp))
            except Exception:
                pass

    candidates.sort(key=lambda z: z[0])

    top = []
    for sc, c in candidates:
        if len(top) >= 3:
            break
        dup = False
        for _, pc in top:
            if c.size == pc.size and np.allclose(c, pc, rtol=0.0, atol=1e-7):
                dup = True
                break
        if not dup:
            top.append((float(sc), np.asarray(c, dtype=np.float64)))

    return top


def _de_search(n, budget_s, rng):
    if differential_evolution is None:
        return None, np.inf

    n = int(n)
    if n < 4:
        return None, np.inf

    deadline = time.monotonic() + max(5.0, float(budget_s) - 5.0)

    def obj(x):
        if time.monotonic() > deadline - 2.0:
            raise _Timeout
        c = np.asarray(x, dtype=np.float64)
        c = np.maximum(c, 0.0)
        s = c.sum()
        if not np.isfinite(s) or s <= 1e-12:
            return 1e6
        c = c / s
        return _score(c)

    bounds = [(0.0, 1.0) for _ in range(n)]
    maxiter = 70 if n <= 16 else 45

    try:
        seed = int(rng.integers(1, 2**31 - 1))
        res = differential_evolution(
            obj,
            bounds,
            seed=seed,
            maxiter=maxiter,
            popsize=15,
            tol=1e-6,
            polish=False,
        )
    except _Timeout:
        return None, np.inf
    except Exception:
        return None, np.inf

    if res is None or getattr(res, "x", None) is None:
        return None, np.inf

    c = _normalize(res.x)
    if c is None:
        return None, np.inf

    return c, _score(c)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(1.0, float(budget_s))
    if budget <= 10.0:
        deadline = t0 + max(1.0, budget - 1.0)
    else:
        deadline = t0 + max(1.0, budget - 10.0)

    rng = np.random.default_rng(seed)

    parent = None
    try:
        parent = _normalize(height_sequence_1)
    except Exception:
        parent = None

    best_seq = None
    best_score = np.inf

    if parent is not None:
        cand = [float(x) for x in parent]
        sc = _exact_score(cand)
        if np.isfinite(sc):
            best_seq = cand
            best_score = sc

    if best_seq is None:
        fb = np.full(128, 1.0 / 128.0, dtype=np.float64)
        cand = [float(x) for x in fb]
        sc = _exact_score(cand)
        if not np.isfinite(sc):
            sc = _score(fb)
        best_seq = cand
        best_score = sc if np.isfinite(sc) else 2.0

    rem = deadline - time.monotonic()
    if rem < 20.0:
        final = [float(x) for x in best_seq]
        return [float(x) for x in final if np.isfinite(x)] or [1.0]

    sizes = []
    if rem > 40.0:
        sizes.append(16)
    if rem > 80.0:
        sizes.extend([32, 64])
    if rem > 160.0:
        sizes.extend([128, 256])
    if rem > 400.0:
        sizes.append(512)
    if rem > 800.0:
        sizes.append(1024)
    if rem > 1100.0:
        sizes.append(2048)

    if parent is not None:
        pn = int(parent.size)
        if 32 <= pn <= 8192 and rem > 300.0:
            sizes.append(pn)

    uniq = []
    seen = set()
    for n in sizes:
        n = int(n)
        if n > 1 and n not in seen:
            seen.add(n)
            uniq.append(n)
    sizes = uniq

    for idx, n in enumerate(sizes):
        now = time.monotonic()
        remaining = deadline - now
        n_left = max(1, len(sizes) - idx)
        reserve = 5.0 * n_left + 5.0
        if remaining <= reserve + 15.0:
            break

        per_size = (remaining - reserve) / float(n_left)
        per_size = min(240.0, per_size)
        per_size = max(15.0, per_size)

        if best_score < 1.35:
            per_size = min(per_size, 60.0)
        elif best_score < 1.50:
            per_size = min(per_size, 120.0)

        try:
            cand_list = _optimize_n(n, per_size, rng, parent)
        except Exception:
            cand_list = []

        for fast_sc, c in cand_list:
            if not np.isfinite(fast_sc) or fast_sc > best_score + 0.1:
                continue
            try:
                sc_exact = _exact_score([float(x) for x in c])
            except Exception:
                sc_exact = np.inf
            if np.isfinite(sc_exact) and sc_exact < best_score:
                best_score = sc_exact
                best_seq = [float(x) for x in c]

    if best_score > 1.55:
        de_sizes = []
        rem_after = deadline - time.monotonic()
        if rem_after > 160.0:
            de_sizes = [16, 24]
        elif rem_after > 80.0:
            de_sizes = [16]

        for n in de_sizes:
            now = time.monotonic()
            if now > deadline - 25.0:
                break
            de_budget = min(100.0, max(20.0, deadline - now - 20.0))
            try:
                c, fast_sc = _de_search(n, de_budget, rng)
            except Exception:
                continue
            if c is None:
                continue
            if np.isfinite(fast_sc) and fast_sc <= best_score + 0.1:
                sc_exact = _exact_score([float(x) for x in c])
                if np.isfinite(sc_exact) and sc_exact < best_score:
                    best_score = sc_exact
                    best_seq = [float(x) for x in c]

    if best_seq is None:
        best_seq = [1.0 / 128.0] * 128

    final = []
    for x in best_seq:
        try:
            xf = float(x)
        except Exception:
            xf = 0.0
        if not np.isfinite(xf):
            xf = 0.0
        if xf < 0.0:
            xf = 0.0
        if xf > 1000.0:
            xf = 1000.0
        final.append(xf)

    if not final:
        final = [1.0]

    s = sum(final)
    if not np.isfinite(s) or s <= 0.01:
        final = [1.0 / 128.0] * 128
        s = 1.0

    final = [x / s for x in final]

    try:
        sc_final = _exact_score(final)
    except Exception:
        sc_final = np.inf

    if not np.isfinite(sc_final):
        fb = [1.0 / 128.0] * 128
        try:
            if np.isfinite(_exact_score(fb)):
                final = fb
        except Exception:
            pass

    return final
