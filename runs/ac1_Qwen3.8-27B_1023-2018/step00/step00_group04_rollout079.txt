import time
import math
import numpy as np

try:
    from scipy.optimize import linprog as _linprog
except Exception:
    _linprog = None


def _fallback_candidate():
    x = (3.0 + math.sqrt(2.0)) / 14.0
    y = 0.5 - x
    return np.array([x, y, x, y], dtype=np.float64)


def _as_valid_normalized(v, m=None):
    if v is None:
        return None
    a = np.asarray(v, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if m is not None and a.size != m:
        if a.size == 1:
            a = np.full(int(m), float(a[0]))
        else:
            xs = np.linspace(0.0, 1.0, int(a.size))
            xn = np.linspace(0.0, 1.0, int(m))
            a = np.interp(xn, xs, a)
    a = np.asarray(a, dtype=np.float64)
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-14:
        return None
    a = a / s
    return a


def _score_np(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float("inf")
    if not np.all(np.isfinite(a)):
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    b = np.convolve(a, a)
    mb = float(np.max(b)) if b.size else 0.0
    if not math.isfinite(mb) or mb <= 0.0:
        if mb < 0.0:
            mb = 0.0
        else:
            return float("inf")
    return float(2.0 * n * mb / (s * s))


def _fft_size_for(n):
    need = 2 * n - 1
    m = 1
    while m < need:
        m <<= 1
    return m


def _fast_score(a, n, fft_n=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    if n <= 0 or a.size != n:
        return float("inf")
    s = float(a.sum())
    if s <= 1e-300:
        return float("inf")
    if fft_n is None:
        fft_n = _fft_size_for(n)
    af = np.fft.rfft(a, n=fft_n)
    b = np.fft.irfft(af * af, n=fft_n)[: 2 * n - 1]
    mb = float(np.max(b))
    if not math.isfinite(mb):
        return float("inf")
    if mb < 0.0 and mb > -1e-14:
        mb = 0.0
    return float(2.0 * n * mb / (s * s))


def _build_convolution_matrix(a, n):
    conv_len = 2 * n - 1
    A = np.empty((conv_len, n), dtype=np.float64)
    if n <= 0:
        return A
    for k in range(n):
        A[k, : k + 1] = a[k::-1]
    rev = a[::-1]
    for k in range(n, conv_len):
        length = conv_len - k
        if length > 0:
            A[k, n - length :] = rev[:length]
    return A


def _lp_direction(a, n, time_limit):
    if _linprog is None:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 0 or time_limit <= 0.0:
        return None
    conv_len = 2 * n - 1
    M = _fast_score(a, n) / (2.0 * n)  # a is normalized, score = 2n*M
    if not math.isfinite(M) or M <= 1e-300:
        return None

    try:
        A = _build_convolution_matrix(a, n)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(conv_len, float(M), dtype=np.float64)
        bounds = [(0.0, None)] * n
        try:
            res = _linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"time_limit": max(0.05, float(time_limit)), "presolve": True},
            )
        except TypeError:
            res = _linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
            )
        finally:
            del A, b_ub

        x = getattr(res, "x", None)
        if x is None:
            return None
        g = np.asarray(x, dtype=np.float64)
        if g.size != n or not np.all(np.isfinite(g)):
            return None
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if sg <= 1e-14:
            return None
        return g / sg
    except Exception:
        return None


def _blend(a, b, t):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    q = (1.0 - t) * a + t * b
    q = np.clip(q, 0.0, None)
    s = float(q.sum())
    if not math.isfinite(s) or s <= 1e-14:
        return None
    return q / s


def _try_update(state, a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0 or not np.all(np.isfinite(a)):
        return
    a = _as_valid_normalized(a)
    if a is None:
        return
    sc = _score_np(a)
    if math.isfinite(sc) and sc < state["score"] - 1e-12:
        state["score"] = float(sc)
        state["p"] = a.copy()


def _fast_descent(a, n, budget, seed=0):
    a = _as_valid_normalized(a, n)
    if a is None or n <= 0 or budget <= 0.0:
        return np.full(n, 1.0 / n) if n else a, float("inf")

    p = a.copy()
    best = p.copy()
    fft_n = _fft_size_for(n)
    best_score = _fast_score(p, n, fft_n)
    deadline = time.monotonic() + float(budget)
    rng = np.random.default_rng(seed)
    eta = 0.7
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        p_fft = np.fft.rfft(p, n=fft_n)
        b = np.fft.irfft(p_fft * p_fft, n=fft_n)[: 2 * n - 1]
        M = float(np.max(b))
        if not math.isfinite(M) or M <= 1e-300:
            break

        beta = 8.0 / M
        w = np.exp(beta * (b - M))
        ws = float(w.sum())
        if not math.isfinite(ws) or ws <= 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(b))] = 1.0
        else:
            w /= ws

        w_fft = np.fft.rfft(w, n=fft_n)
        sens = np.fft.irfft(w_fft * p_fft, n=fft_n)[:n]
        sc = sens - float(np.mean(sens))
        step = np.clip(-eta * sc, -50.0, 50.0)

        p = p * np.exp(step)
        p = np.clip(p, 0.0, None)
        ps = float(p.sum())
        if not math.isfinite(ps) or ps <= 1e-300:
            p = best.copy()
            break
        p = p / ps

        score = _fast_score(p, n, fft_n)
        if math.isfinite(score) and score < best_score:
            best = p.copy()
            best_score = score
            no_improve = 0
            eta = min(eta * 1.02, 3.0)
        else:
            no_improve += 1
            if no_improve % 120 == 0:
                eta = max(eta * 0.88, 0.02)
            if no_improve > 1800:
                p = p * np.exp(0.12 * rng.normal(size=n))
                p = np.clip(p, 0.0, None)
                ps = float(p.sum())
                if ps > 1e-300:
                    p = p / ps
                eta = 0.7
                no_improve = 0

    return best, best_score


def _generate_inits(n, hs):
    if n <= 0:
        return []
    t = np.linspace(0.0, 1.0, n)
    raw = []

    raw.append(np.ones(n, dtype=np.float64))

    for A in (0.15, 0.30, 0.42, 0.55, 0.70):
        raw.append(np.clip(A + (1.0 - 2.0 * A) * t, 0.0, None))

    for lam in (0.8, 1.5, 2.5, 4.0):
        raw.append(np.exp(lam * t))
        raw.append(np.exp(lam * (t - 1.0)))

    for c3 in (-0.4, -0.1, 0.2, 0.5):
        raw.append(np.clip(0.5 + 0.35 * t + c3 * t * (1.0 - t), 0.0, None))

    for sd in (101, 211):
        rng = np.random.default_rng(sd)
        v = np.ones(n, dtype=np.float64)
        for k in range(1, 5):
            amp = rng.uniform(0.12, 0.55)
            phase = 2.0 * np.pi * rng.random()
            v += amp * np.cos(2.0 * np.pi * k * t + phase)
        raw.append(np.clip(v, 0.0, None))

    for sd in (301, 401):
        rng = np.random.default_rng(sd)
        raw.append(np.clip(np.abs(rng.normal(size=n)) + 0.05, 0.0, None))

    if hs is not None:
        raw.append(hs)
        try:
            raw.append(hs[::-1])
        except Exception:
            pass

    out = []
    seen = set()
    for v in raw:
        a = _as_valid_normalized(v, n)
        if a is None:
            continue
        key = tuple(np.round(10000.0 * a, 1).tolist())
        if key in seen:
            continue
        seen.add(key)
        out.append(a)
    return out


def _search_n(n, allowance, state, hs):
    if n <= 0 or allowance <= 0.0:
        return
    deadline = time.monotonic() + float(allowance)
    if deadline <= time.monotonic():
        return

    p = _as_valid_normalized(state.get("p"), n)
    if p is None:
        p = np.full(n, 1.0 / n, dtype=np.float64)

    inits = _generate_inits(n, hs)
    if not inits:
        inits = [np.full(n, 1.0 / n, dtype=np.float64)]

    local_best = p.copy()
    local_score = _fast_score(p, n)
    _try_update(state, p)

    per_init = max(0.3, min(4.5, (deadline - time.monotonic()) / (8.0 * max(1, len(inits)))))
    for i0, init in enumerate(inits):
        if time.monotonic() >= deadline - 1.0:
            break
        p_try = init.copy()
        _try_update(state, p_try)
        fb = min(max(0.2, per_init), deadline - time.monotonic())
        if fb > 0.25:
            p2, s2 = _fast_descent(p_try, n, fb, seed=1000 + i0)
            _try_update(state, p2)
            if math.isfinite(s2) and s2 < local_score:
                local_score = s2
                local_best = p2.copy()

    p = local_best.copy()
    p_fast = _fast_score(p, n)
    _try_update(state, p)

    lp_count = 0
    no_local = 0

    while time.monotonic() < deadline - 1.5:
        rem = deadline - time.monotonic()
        if rem <= 0.5:
            break

        per_lp = None
        if n <= 64:
            if lp_count < 800:
                per_lp = min(2.0, max(0.1, rem * 0.08))
        elif n <= 256:
            if lp_count < 500:
                per_lp = min(4.0, max(0.25, rem * 0.10))
        elif n <= 1024:
            if lp_count < 220:
                per_lp = min(7.0, max(0.5, rem * 0.12))
        elif n <= 2048:
            if lp_count < 70:
                per_lp = min(12.0, max(1.0, rem * 0.15))

        if per_lp is not None and rem > per_lp + 1.0:
            g = _lp_direction(p, n, per_lp)
            lp_count += 1
            if g is not None:
                cur_fast = _fast_score(p, n)
                best_q = None
                best_q_fast = float("inf")
                for t in (0.5, 0.25, 0.1, 0.75, 0.9, 0.05):
                    q = _blend(p, g, t)
                    if q is None:
                        continue
                    _try_update(state, q)
                    sf = _fast_score(q, n)
                    if math.isfinite(sf) and sf < best_q_fast:
                        best_q = q
                        best_q_fast = sf
                if best_q is not None:
                    if best_q_fast < cur_fast - 1e-12:
                        p = best_q.copy()
                        p_fast = best_q_fast
                        no_local = 0
                        if p_fast < local_score:
                            local_score = p_fast
                            local_best = p.copy()
                    else:
                        no_local += 1
                else:
                    no_local += 1
            else:
                no_local += 1

        fast_budget = min(5.0, max(0.3, rem * 0.20))
        if fast_budget > 0.3:
            p2, s2 = _fast_descent(p, n, fast_budget, seed=9000 + lp_count)
            _try_update(state, p2)
            if math.isfinite(s2) and s2 < p_fast - 1e-12:
                p = p2.copy()
                p_fast = s2
                no_local = 0
                if p_fast < local_score:
                    local_score = p_fast
                    local_best = p.copy()
            else:
                no_local += 1

        if no_local > 7:
            rng = np.random.default_rng(int(time.monotonic() * 1000) ^ (lp_count * 101 + 17))
            base = p * np.exp(0.25 * rng.normal(size=n))
            base = np.clip(base, 0.0, None)
            bs = float(base.sum())
            if bs > 1e-14:
                base = base / bs
                p2, s2 = _fast_descent(base, n, min(3.0, rem * 0.08), seed=int(rng.integers(5_000_000)))
                if math.isfinite(s2) and s2 < p_fast:
                    p = p2.copy()
                    p_fast = s2
                    _try_update(state, p)
            no_local = 0

    _try_update(state, local_best)
    _try_update(state, p)
