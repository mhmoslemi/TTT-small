import numpy as np
import time
import scipy.sparse as sp
import scipy.optimize as opt


def _pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def _norm1(a):
    """Normalize to a valid nonnegative finite vector, sum 1."""
    try:
        a = np.asarray(a, dtype=np.float64)
    except Exception:
        return None
    if a.ndim != 1 or a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    if not np.all(np.isfinite(a)):
        return None
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    return a / s


def _score_norm(a):
    """Fast score for a normalized/nonnegative array; evaluator is final."""
    a = _norm1(a)
    if a is None:
        return np.inf
    n = int(a.size)
    c = _conv(a)
    if c.size == 0:
        return np.inf
    S = float(a.sum())
    if S <= 1e-12:
        return np.inf
    M = float(c.max())
    if not np.isfinite(M) or M <= 0:
        return np.inf
    return float(2.0 * n * M / (S * S))


def _conv(a):
    """Full linear convolution a*a, with FFT for larger sizes."""
    a = _norm1(a)
    if a is None:
        return np.array([0.0])
    n = int(a.size)
    m = 2 * n - 1
    if n <= 128:
        return np.convolve(a, a)
    L = _pow2(m)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:m]
    return np.maximum(c, 0.0)


def _corr_w_a(w, a):
    """h_l = sum_i a_i w_{l+i}, for l = 0,...,n-1."""
    a = _norm1(a)
    if a is None:
        return None
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = 2 * n - 1
    if w.size != m:
        return None
    if n <= 128:
        h = np.empty(n, dtype=np.float64)
        for l in range(n):
            h[l] = np.dot(w[l:l + n], a)
        return np.maximum(h, 0.0)

    L = _pow2(n + m - 1)
    Fa = np.fft.rfft(a, L)
    Frw = np.fft.rfft(w[::-1], L)
    cv = np.fft.irfft(Fa * Frw, L)
    h = cv[m - 1:m - n - 1:-1]
    return np.maximum(h, 0.0)


def _weights_from_conv(c, tau):
    c = np.asarray(c, dtype=np.float64)
    M = float(c.max()) if c.size else 0.0
    if not np.isfinite(M) or M <= 0:
        wt = np.ones_like(c)
        return wt / max(1, wt.size), 1.0
    if tau is None or tau <= 1e-15:
        tol = max(1e-12, 1e-10 * M)
        active = (c <= M + tol) & (c >= M - tol)
        if not active.any():
            active[int(np.argmax(c))] = True
        wt = active.astype(np.float64)
        wt /= max(1.0, wt.sum())
        return wt, M
    z = (c - M) / float(tau)
    z -= np.max(z)
    wt = np.exp(z)
    s = wt.sum()
    if not np.isfinite(s) or s <= 0:
        active = (c <= M + max(1e-12, 1e-10 * M)) & (c >= M - max(1e-12, 1e-10 * M))
        if not active.any():
            active[int(np.argmax(c))] = True
        wt = active.astype(np.float64)
        wt /= max(1.0, wt.sum())
    else:
        wt /= s
    return wt, M


def _piecewise(N, lens, vals):
    N = int(N)
    lens = np.clip(np.asarray(lens, dtype=np.float64), 0.0, None)
    if lens.size == 0 or lens.sum() <= 0:
        return np.ones(N)
    lens = lens / lens.sum()
    vals = np.clip(np.asarray(vals, dtype=np.float64), 0.0, None)
    K = min(lens.size, vals.size)
    if K <= 0 or N <= 0:
        return None
    if K > N:
        K = N
    counts = np.maximum(0, np.round(lens[:K] * N).astype(np.int64))
    if counts.sum() == 0:
        counts = np.ones(K, dtype=np.int64)
        if N % K:
            counts[:N % K] += 1
    else:
        while counts.sum() < N:
            counts[int(np.argmax(lens[:K]))] += 1
        while counts.sum() > N:
            idx = int(np.argmax(counts))
            if counts[idx] <= 0:
                break
            counts[idx] -= 1
    a = np.repeat(vals[:K], counts)
    if a.size != N:
        if a.size > N:
            a = a[:N]
        else:
            a = np.concatenate([a, np.zeros(N - a.size)])
    return _norm1(a)


def _smooth_perturb(a, rng, scale=0.25):
    a = _norm1(a)
    if a is None:
        return None
    n = int(a.size)
    x = np.linspace(0.0, 1.0, n)
    z = np.zeros(n, dtype=np.float64)
    for f in (1, 2, 3, 5, 8):
        z += float(rng.normal(0.0, 0.5 * max(scale, 0.01))) * np.sin(
            2.0 * np.pi * f * x + float(rng.uniform(0.0, 2.0 * np.pi))
        )
    z = np.clip(z, -2.5, 2.5)
    a_pos = np.maximum(a, 1e-6)
    a_new = a_pos * np.exp(0.5 * z)
    return _norm1(a_new)

def _basis_search(N, budget_s, rng, t_start, deadline):
    budget_s = max(0.0, float(budget_s))
    deadline = min(deadline, t_start + budget_s + 3.0)
    N = int(N)
    if N <= 0:
        return

    x = (np.arange(N, dtype=np.float64) + 0.5) / float(N)
    B = np.column_stack((
        x,
        x * x,
        x * x * x,
        np.sin(2.0 * np.pi * x),
        np.cos(2.0 * np.pi * x),
        np.sin(4.0 * np.pi * x),
    ))
    d = B.shape[1]

    def decode(theta):
        try:
            z = np.asarray(theta, dtype=np.float64) @ B.T
        except Exception:
            return None
        z = z - float(np.mean(z))
        z = np.clip(z, -8.0, 8.0)
        a = np.exp(z)
        s = float(a.sum())
        if not np.isfinite(s) or s <= 1e-12:
            return None
        return a / s

    def fun(theta):
        a = decode(theta)
        if a is None:
            return 1e6
        sc = _score_norm(a)
        return float(sc) if np.isfinite(sc) else 1e6

    inits = []

    def project(a):
        a = _norm1(a)
        if a is None:
            return rng.normal(0.0, 0.5, size=d)
        z = np.log(np.clip(a, 1e-12, None))
        z = z - float(np.mean(z))
        try:
            theta = np.linalg.lstsq(B, z, rcond=None)[0]
        except Exception:
            theta = rng.normal(0.0, 0.5, size=d)
        return np.clip(theta, -5.0, 5.0)

    inits.append(np.zeros(d))
    inits.append(project(_piecewise(N, [0.5, 0.5], [2.0, 1.0])) or rng.normal(0, 0.5, d))
    inits.append(project(_piecewise(N, [0.3, 0.4, 0.3], [2.0, 1.0, 0.4])) or rng.normal(0, 0.5, d))
    inits.append(project(np.exp(-1.5 * x)) or rng.normal(0, 0.5, d))
    inits.append(project(np.exp(1.5 * x)) or rng.normal(0, 0.5, d))

    best_rand = None
    best_rand_sc = np.inf
    n_random = max(10, min(300, int(budget_s * 4)))
    for _ in range(n_random):
        if time.monotonic() >= deadline - 2.0:
            break
        theta = rng.normal(0.0, 1.0, size=d)
        a = decode(theta)
        if a is None:
            continue
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_rand_sc - 1e-9:
            best_rand_sc = sc
            best_rand = np.clip(theta, -6.0, 6.0)
    if best_rand is not None:
        inits.append(best_rand)
    if len(inits) < 5:
        inits.append(rng.normal(0.0, 0.8, size=d))

    inits = [theta for theta in inits if theta is not None and np.all(np.isfinite(theta))]
    if not inits:
        return

    for theta0 in inits:
        if time.monotonic() >= deadline - 3.0:
            break
        remaining = max(0.0, deadline - time.monotonic() - 3.0)
        if remaining < 5.0:
            break
        maxfev = int(max(120, min(650, remaining * 80)))
        try:
            res = opt.minimize(
                fun,
                np.clip(theta0, -6.0, 6.0),
                method="Powell",
                options={"maxiter": 35, "maxfev": maxfev, "disp": False},
            )
            if res is not None and getattr(res, "x", None) is not None:
                a = decode(res.x)
                if a is not None:
                    return _norm1(a)  # caller records
        except Exception:
            pass
    return None


def _lp_iteration(a0, budget_s, rng, t_start, deadline, n_limit=512):
    a0 = _norm1(a0)
    if a0 is None:
        return None
    n = int(a0.size)
    if n < 2 or n > n_limit:
        return a0

    budget_s = max(0.0, float(budget_s))
    deadline = min(deadline, t_start + budget_s + 3.0)

    best = a0.copy()
    best_score = _score_norm(best)

    i = np.repeat(np.arange(n), n)
    j = np.tile(np.arange(n), n)
    rows_flat = i + j
    cols_flat = j
    bounds = [(0.0, 1000.0)] * n
    c_obj = -np.ones(n)

    rng_state = rng
    for it in range(40):
        if time.monotonic() >= deadline - 3.0:
            break
        c = _conv(best)
        M = float(c.max()) if c.size else 1.0
        if M <= 0 or not np.isfinite(M):
            break
        M = M * (1.0 + 1e-10) + 1e-14

        data = np.repeat(best, n)
        try:
            A = sp.coo_matrix((data, (rows_flat, cols_flat)), shape=(2 * n - 1, n)).tocsr()
        except Exception:
            break

        b_ub = np.full(2 * n - 1, M)
        res = None
        remaining = max(0.0, deadline - time.monotonic() - 3.0)
        max_solve_time = max(0.5, min(45.0, remaining * 0.45))
        for opts in (
            {"presolve": True, "time_limit": max_solve_time},
            {"presolve": True},
        ):
            if time.monotonic() >= deadline - 3.0:
                break
            try:
                res = opt.linprog(
                    c_obj,
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=bounds,
                    method="highs",
                    options=opts,
                )
                if res is not None and getattr(res, "success", False):
                    break
            except Exception:
                res = None

        if res is None or not getattr(res, "success", False) or res.x is None:
            break
        b = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, None)
        sb = float(b.sum())
        if sb <= 1e-12:
            break
        g = b / sb

        accepted = False
        for t in (0.25, 0.10, 0.50, 0.05, 0.20, 1.00, 0.02):
            cand = best * (1.0 - float(t)) + g * float(t)
            cand = _norm1(cand)
            if cand is None:
                continue
            sc = _score_norm(cand)
            if np.isfinite(sc) and sc < best_score - 1e-10:
                best = cand
                best_score = sc
                accepted = True
                break
        if not accepted:
            break
    return best

            a = _piecewise(N, L, V)
            if a is not None:
                record(a)
        # Random smooth profiles.
        for _ in range(4):
            z = np.zeros(N, dtype=np.float64)
            for f in (1, 2, 3, 5):
                z += float(rng.normal(0.0, 0.7)) * np.sin(
                    2.0 * np.pi * f * x + float(rng.uniform(0.0, 2.0 * np.pi))
                )
            z += float(rng.normal(0.0, 0.7)) * x
            record(np.exp(np.clip(z, -3.0, 3.0)))
        # If time is modest, skip the very long expensive records in the next loop.

    for N in basic_lengths:
        if time.monotonic() >= deadline - 8.0:
            break
        add_simple(N)

    if best_score_box[1] is None:
        best_score_box[1] = np.ones(N_main, dtype=np.float64)
        best_score_box[0] = _score_norm(best_score_box[1])

    def remaining():
        return max(0.0, deadline - time.monotonic())

    # Low-dimensional profile search on the main and one moderate length.
    search_Ns = []
    if N_main > 0:
        search_Ns.append(N_main)
    if 512 not in best_len:
        search_Ns.append(512)
    if budget_s > 800:
        search_Ns.append(2048)
    search_Ns = sorted(set([int(n) for n in search_Ns if 16 <= int(n) <= 8192]), reverse=True)

    for N in search_Ns:
        if remaining() < 60.0:
            break
        alloc = min(180.0, remaining() * 0.22)
        if alloc <= 5.0:
            break
        t_start = time.monotonic()
        a = _basis_search(N, alloc, rng, t_start, deadline)
        if a is not None:
            record(a)

    # LP / Frank-Wolfe polish on small, solver-size instances.
    for N in (256, 128, 64):
        if remaining() < 30.0:
            break
        if N not in best_len:
            add_simple(N)
        if N not in best_len:
            continue
        base = best_len[N][1]
        t_start = time.monotonic()
        a2 = _lp_iteration(base, min(90.0, remaining() * 0.20), rng, t_start, deadline, n_limit=512)
        if a2 is not None:
            record(a2)

    # High-dimensional projected-gradient polishing.
    pg_targets = []
    if N_main in best_len:
        pg_targets.append((N_main, best_len[N_main][1]))
    else:
        pg_targets.append((N_main, np.ones(N_main, dtype=np.float64)))

    # Add the current best other lengths, but avoid huge small spends too many times.
    for n, (sc, arr) in sorted(best_len.items(), key=lambda kv: kv[1][0]):
        if len(pg_targets) >= 4:
            break
        if n == N_main:
            continue
        if sc < 1.70 or n <= 1024:
            pg_targets.append((n, arr))

    for idx, (N, arr) in enumerate(pg_targets):
        if remaining() < 25.0:
            break
        if N == N_main:
            alloc = min(650.0, remaining() * 0.50)
        else:
            alloc = min(75.0, remaining() * 0.12)
        if alloc < 10.0:
            continue
        t_start = time.monotonic()
        out = _projected_gradient(arr, alloc, rng, t_start, deadline)
        if out is not None:
            record(out)

    # If a small/medium solution is looking good, interpolate it to a longer length
    # and polish: often a cheap way to test whether continuity helps.
    if remaining() > 80.0 and best_score_box[1] is not None:
        small = best_score_box[1]
        if small.size < N_main:
            m = int(small.size)
            x_old = np.linspace(0.0, 1.0, m)
            x_new = np.linspace(0.0, 1.0, N_main)
            interp = np.interp(x_new, x_old, small)
            a_interp = _norm1(interp)
            if a_interp is not None:
                record(a_interp)
                t_start = time.monotonic()
                out = _projected_gradient(a_interp, min(180.0, remaining() * 0.30), rng, t_start, deadline)
                if out is not None:
                    record(out)

    # Final modest round of basis + gradient from the global best, if time remains.
    if remaining() > 50.0 and best_score_box[1] is not None:
        N_final = int(best_score_box[1].size)
        t_start = time.monotonic()
        a_basis = _basis_search(N_final, min(60.0, remaining() * 0.25), rng, t_start, deadline)
        if a_basis is not None:
            record(a_basis)

    if remaining() > 25.0 and best_score_box[1] is not None:
        t_start = time.monotonic()
        out = _projected_gradient(
            best_score_box[1],
            min(60.0, remaining() * 0.35),
            rng,
            t_start,
            deadline,
        )
        if out is not None:
            record(out)

    cand = best_score_box[1]
    if cand is None:
        cand = np.ones(N_main if N_main > 0 else 1, dtype=np.float64)

    # Exact final validation with the authoritative evaluator.
    try:
        lst = [float(v) for v in cand]
        exact = evaluate_sequence(lst)
        if not np.isfinite(exact) or exact > 1e4:
            lst = [1.0] * max(1, int(cand.size) if cand.size else 1)
            exact2 = evaluate_sequence(lst)
            if np.isfinite(exact2):
                return lst
    except Exception:
        lst = [float(v) for v in cand]

    # Defensive fallback: a valid score-2 sequence.
    if not lst or any((not np.isfinite(v)) or v < 0 for v in lst):
        lst = [1.0] * max(1, int(N_main))
    return lst
