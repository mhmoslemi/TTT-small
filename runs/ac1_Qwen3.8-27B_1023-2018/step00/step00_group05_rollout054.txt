import numpy as np
import time


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 8.0)

    n = len(height_sequence_1)
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()

    def fast_self_conv(a):
        fa = np.fft.rfft(a, fft_size)
        return np.fft.irfft(fa * fa, fft_size)[:conv_len]

    def fast_score(a):
        sa = a.sum()
        if sa < 1e-15:
            return np.inf
        c = fast_self_conv(a)
        return 2.0 * n * c.max() / (sa * sa)

    # Best tracking with exact evaluator
    best = parent.copy()
    best_score = float(evaluate_sequence([float(x) for x in best]))

    # Generate diverse initializations
    j = np.arange(n, dtype=np.float64)
    x = j / (n - 1)
    t = 2.0 * x - 1.0

    inits = [parent.copy()]
    inits.append((0.5 + 0.5 * np.cos(2.0 * np.pi * x)) ** 2)
    inits.append(np.maximum(0.0, 1.0 - t * t) ** 2 + 1e-12)
    inits.append(np.exp(-((j - 0.42 * n) / (0.28 * n)) ** 2))
    inits.append(np.maximum(0.0, np.sinc(0.4 * t)) ** 2 + 1e-12)
    inits.append((1.0 + 0.3 * t) ** 2 * np.maximum(0.0, 1.0 - t * t ** 2))
    inits.append(np.exp(-((j - 0.55 * n) / (0.22 * n)) ** 2) * (1.0 + 0.2 * np.sin(j * 0.05)))
    inits.append(np.maximum(0.0, 1.0 - 2.0 * x) ** 2 + 1e-12)

    # Select best initialization
    best_init = inits[0]
    best_init_sc = fast_score(best_init)
    for ini in inits[1:]:
        sc = fast_score(ini)
        if sc < best_init_sc:
            best_init_sc = sc
            best_init = ini

    a = best_init.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n, dtype=np.float64)
    a /= a.sum()

    # Subgradient descent
    step = 2e-4
    mu = 0.75
    vel = np.zeros(n)
    no_improve = 0

    while time.monotonic() < deadline:
        sa = a.sum()
        c = fast_self_conv(a)
        k_star = int(np.argmax(c))
        max_c = c[k_star]

        # Subgradient: d c[k*] / d a[j] = 2*a[k*-j] if 0<=k*-j<n
        grad_c = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo <= hi:
            grad_c[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]

        # Gradient of F = 2n*c[k*]/sa^2
        grad_F = (2.0 * n / (sa ** 3)) * (grad_c * sa - 2.0 * max_c)

        # Momentum update
        vel = mu * vel - step * grad_F
        a_new = a + vel
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new /= s_new

        # Evaluate new score
        s_new_val = 2.0 * n * fast_self_conv(a_new).max() / a_new.sum() ** 2
        s_old_val = 2.0 * n * max_c / (sa * sa)

        if s_new_val < s_old_val:
            a = a_new
            no_improve = 0
            step = min(step * 1.015, 3e-3)
        else:
            no_improve += 1
            step *= 0.85
            vel *= 0.4
            if no_improve > 400:
                # Escape local minimum with structured perturbation
                width = max(1, n // 30)
                s0 = int(rng.integers(0, n - width))
                a[s0:s0 + width] *= np.exp(rng.normal(0, 0.15, width))
                a = np.maximum(a, 0.0)
                a /= a.sum()
                step = 2e-4
                vel = np.zeros(n)
                no_improve = 0
            if no_improve > 2000:

                a /= a.sum()
                step = 1e-4
                vel = np.zeros(n)

        polish_step = 0.002

                break

            cand = polish_a.copy()
            cand[s0:s0 + w] *= np.exp(rng.normal(0, polish_step, w))
            cand = np.maximum(cand, 0.0)

                polish_a = cand
                polish_step = min(polish_step * 1.05, 0.05)
            else:
                polish_step *= 0.9
        eval_a = polish_a * np.sqrt(2.0 * n)
        sc = float(evaluate_sequence([float(v) for v in eval_a]))
        if np.isfinite(sc) and sc < best_score:
            best = eval_a.copy()
            best_score = sc

    result = [float(x) for x in best]
    fin_sc = float(evaluate_sequence(result))
    if np.isfinite(fin_sc) and fin_sc > 0:
        return result

    # Fallback: parent
    return [float(x) for x in parent]
