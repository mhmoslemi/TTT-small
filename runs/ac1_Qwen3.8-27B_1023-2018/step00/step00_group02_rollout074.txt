import numpy as np
import time


def _fft_conv(a, b):
    """FFT-based linear convolution, returns full result of length len(a)+len(b)-1."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def _score_fast(x, n):
    """Fast score: 2n * max(conv) / (sum)^2."""
    conv = _fft_conv(x, x)
    max_b = conv.max()
    s = x.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (s * s)
