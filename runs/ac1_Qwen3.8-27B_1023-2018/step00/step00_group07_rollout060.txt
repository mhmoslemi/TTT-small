    # = correlation of weights and p (with specific convention)
    # Pad both to size >= conv_len + n - 1
    size = 1
    while size < conv_len + n - 1:
        size *= 2
    
    # weights[k] * p[k-m]: this is (weights * p_flip)[m + ...]
    # Actually: sum_k weights[k] * p[k-m]
    # Let q = p reversed: q[j] = p[n-1-j]
    # p[k-m] = q[n-1-(k-m)] = q[n-1-k+m]
    # sum_k weights[k] * q[n-1-k+m]
    # = sum_k weights[k] * q[(m + n-1) - k]
    # = (weights * q)[m + n - 1]  (standard convolution)
    
    q = p[::-1].copy()
    size = 1
    while size < conv_len + n - 1:
        size *= 2
    
    W = np.fft.rfft(weights, size)
    Q = np.fft.rfft(q, size)
    conv_wq = np.fft.irfft(W * Q, size)
    
    # (weights * q)[m + n - 1] for m = 0, ..., n-1
    grad = conv_wq[n-1 : n-1+n] * 2.0
    
    return grad

def _fft_full_conv(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def _grad_via_fft(p, weights):
    """Compute grad[m] = 2*sum_k weights[k]*p[k-m] using FFT."""
    n = len(p)
    seg_len = len(weights) + n - 1
    size = 1 << (seg_len - 1).bit_length()
    q = p[::-1].copy()
    W = np.fft.rfft(weights, size)
    Q = np.fft.rfft(q, size)
    full = np.fft.irfft(W * Q, size)
    return 2.0 * full[n - 1: n - 1 + n]
