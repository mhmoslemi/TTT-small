import numpy as np
import time

# ---------------------------------------------------------------------------
# Helpers

def _proj_simplex(v):
    """Project v onto the probability simplex (sum 1, v>=0), O(n) expected."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rng = np.arange(1, n + 1)
    rho = np.nonzero(u * rng + 1.0 - css > 0)[0]
    if rho.size == 0:
        return np.full(n, 1.0 / n)
    r = rho[-1]
    theta = (css[r] - 1.0) / (r + 1.0)
    out = v - theta
    out[out < 0] = 0.0
    s = out.sum()
    if s <= 0:
        return np.full(n, 1.0 / n)
    out /= s
    return out


def _lse_value_and_weight(p, tau):
    """Return (logsumexp(tau*p*p)/tau, softmax weights over lags, autocorr)."""
    c = np.convolve(p, p)
    m = tau * c
    m -= m.max()
    e = np.exp(m)
    s = e.sum()
    w = e / s
    val = (m.max() + np.log(s)) / tau
    return val, w, c


def _grad_lse(p, w):
    """Gradient of the logsumexp surrogate at p (uses autocorr weights w)."""
    n = p.size
    L = 2 * n - 1
    rv = w[::-1]
    full = np.convolve(p, rv)
    h = full[L - n:L][::-1].copy()
    return 2.0 * h, c


def _score(a):
    """Exact objective 2*n*max(conv)/(sum)^2, robust."""
    n = len(a)
    if n == 0:
        return np.inf
    c = np.convolve(a, a)
    s = a.sum()
    if s <= 0:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def _run_lse(p0, n, budget_end, rng):
    """Projected gradient on the simplex for the LSE surrogate.

    Returns the best probability vector found (by exact objective).
    """
    p = _proj_simplex(p0.copy())
    a_scale = 64.0
    best = p.copy()
    best_score = _score(best * a_scale)
    if not np.isfinite(best_score):
        best = np.full(n, 1.0 / n)
        best_score = 2.0

    tau = 0.25 * n
    eta = 1.0 / (tau * p.max() + 1e-12)
    tau_goal = n

    start = time.monotonic()
    itr = 0
    stall = 0
    last_best_i = -1000

    while time.monotonic() < budget_end - 2.0:
        itr += 1

        val0, w0, c0 = _lse_value_and_weight(p, tau)
        grad0, _ = _grad_lse(p, w0)

        # adaptive line search on the surrogate
        moved = False
        for _bt in range(8):
            p2 = _proj_simplex(p - eta * grad0)
            val1, _, _ = _lse_value_and_weight(p2, tau)
            if val1 < val0 - 1e-14:
                p = p2
                moved = True
                break
            eta *= 0.5
        if not moved:
            eta *= 2.0
            p = _proj_simplex(p - eta * grad0)
            if eta > 1.0 / (tau * p.max() + 1e-12):
                eta = 1.0 / (tau * p.max() + 1e-12)

        # occasionally nudge with flip-average and tiny jitter to escape plateaus

            pf = p[::-1].copy()
            p = _proj_simplex(0.5 * (p + pf))
        if itr % 400 == 0:
            p = _proj_simplex(p + 1e-3 * rng.standard_normal(n))

        # focus the peak gradually

            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = p.copy()
                last_best_i = itr

    sc = _score(best * a_scale)
    if np.isfinite(sc) and sc < best_score:

        best = best.copy()
    return best, best_score

    win = np.clip(win, 0, None)
    if win.sum() > 0:

    inits.append(("jitter-uniform",
                  _proj_simplex(np.full(n, 1.0 / n)
                                + 0.05 * rng.standard_normal(n))))

    # ---- run optimisations, split the budget among inits ----
    nruns = len(inits)
    per_run = (deadline - start) / max(1, nruns)
    best_a = None
    best_score = np.inf

    for k, (name, p0) in enumerate(inits):
        run_end = start + per_run * (k + 1)
        if k == nruns - 1:
            run_end = deadline
        # ensure a little headroom between runs
        run_end = min(run_end, deadline - 1.0)
        try:
            pb, sc = _run_lse(p0, n, run_end, np.random.default_rng(seed + 1000 + k))
            a = pb * a_scale
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a
        except Exception:
            continue

    # ---- guaranteed valid fallback (uniform, score 2) if needed ----
    if best_a is None or not np.isfinite(best_score):
        a = np.full(n, 1.0 / n) * a_scale
        best_a = a
        best_score = _score(a)

    # ---- final exact validation ----
    a = np.clip(np.asarray(best_a, dtype=np.float64), 0.0, 1000.0)
    a = a[np.isfinite(a)]
    if a.size == 0 or a.sum() < 0.01:
        a = np.full(n, 1.0 / n) * a_scale
    final = evaluate_sequence(a.tolist())  # noqa: F821
    if not np.isfinite(final):
        # ultimate fallback
        a = np.full(n, 1.0 / n) * a_scale
        final = evaluate_sequence(a.tolist())  # noqa: F821
    if not np.isfinite(final):
        a = [1.0] * n
    assert np.isfinite(final)

    return [float(x) for x in np.asarray(a, dtype=np.float64)]
