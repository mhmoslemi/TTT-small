def _compute_conv(a, N):
    """Full auto-convolution via FFT, returns array of length 2*len(a)-1."""
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)

    return result[:len(a)]


def _project_nonneg_sum(x, target):
    """Euclidean projection onto {y >= 0, sum(y) = target}."""
    n = len(x)
    if n == 0:
        return x
    if target <= 0:
        return np.zeros_like(x)
    # Bisection for water-filling
    lo = min(0.0, float(x.min())) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.maximum(0.0, x - mid).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.maximum(0.0, x - lam)


def _exact_score(a):
    """Compute AC1 score directly via the formula (avoids list conversion overhead)."""
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    max_b = float(c.max())
    return float(2.0 * n * max_b / (S * S))


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 4.0
    rng = np.random.default_rng(int(seed))

    # --- Initialization from parent ---
    parent = np.asarray(height_sequence_1, dtype=np.float64).flatten()
    parent = np.clip(parent, 0.0, 1000.0)
    n = int(parent.size)
    S0 = float(parent.sum())
    if S0 < 0.01:
        a = np.ones(n, dtype=np.float64)
    else:
        a = parent * (float(n) / S0)

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    # FFT sizes
    N_conv = 1
    while N_conv < 2 * n - 1:
        N_conv <<= 1
    N_xc = 1
    while N_xc < 3 * n:
        N_xc <<= 1

    # --- Phase 1: Gradient-based optimization ---
    lr = 0.05
    beta = 0.85
    momentum = np.zeros(n, dtype=np.float64)
    stagnation = 0
    last_best_it = 0
    eval_interval = 30

    it = 0
    while time.time() < deadline:
        it += 1

        # Compute auto-convolution
        c = _compute_conv(a, N_conv)
        max_c = float(c.max())

        # Softmax weights (focus on top lags)
        c95 = float(np.percentile(c, 95))
        spread = max(max_c - c95, 1e-8)
        T = 25.0 / spread
        shifted = c - max_c
        exponents = T * shifted
        max_ex = float(exponents.max())
        weights = np.exp(exponents - max_ex)
        wsum = float(weights.sum())
        if wsum < 1e-30:
            weights = np.zeros_like(c)
            weights[int(np.argmax(c))] = 1.0
        else:
            weights /= wsum

        # Gradient: 2 * cross_corr(a, weights)
        grad = 2.0 * _cross_corr_ab(a, weights, N_xc)

        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        a_new = a - lr * momentum

        # Project onto {x >= 0, sum(x) = n}
        a_new = _project_nonneg_sum(a_new, float(n))
        if float(a_new.sum()) < 0.01:
            a_new = a.copy()
        a = a_new

        # Periodic evaluation
        if it % eval_interval == 0:
            sc = _exact_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stagnation = 0
                last_best_it = it
            else:
                stagnation += 1

            # Adaptive: reduce lr and reset if stuck
            if stagnation > 10 and lr > 0.002:
                lr *= 0.6
                momentum *= 0.3
                stagnation = 0

            # Track best via exact evaluator (less frequent)
            if it % (eval_interval * 10) == 0:
                sc_exact = float(evaluate_sequence(a.tolist()))
                if sc_exact < best_score:
                    best_score = sc_exact
                    best_a = a.copy()

    # --- Phase 2: Local refinement with block perturbations ---
    # Use remaining time for local search around best
    remaining = deadline - time.time()
    if remaining > 2.0:
        a = best_a.copy()
        local_best = a.copy()
        local_score = best_score
        no_improve = 0
        while time.time() < deadline:
            # Random block multiplicative perturbation
            width = int(rng.integers(max(2, n // 200), max(3, n // 10)))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, 0.15)))
            a[stop - start:start + width] = a[stop - start:start + width] * factor
            a = np.clip(a, 0.0, 1000.0)
            # Re-normalize
            S = float(a.sum())
            if S < 0.01:
                a = local_best.copy()
                continue
            a = a * (float(n) / S)

            sc = _exact_score(a)
            if sc < local_score:
                local_score = sc
                local_best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 200:
                    # Bash back to local best and try larger perturbation
                    a = local_best.copy()
                    no_improve = 0

            # Update global best
            if local_score < best_score:
                best_score = local_score
                best_a = local_best.copy()

    # --- Final validation with exact evaluator ---
    final_score = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(final_score) or final_score < 0:
        # Fallback: return the parent
        p_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(p_score):
            return [float(x) for x in parent]
        return [float(x) for x in best_a]

    # Clip and ensure validity
    result = np.clip(best_a, 0.0, 1000.0)
    if float(result.sum()) < 0.01:
        result = np.ones(n, dtype=np.float64)

    return [float(x) for x in result]
