import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _try_freeform_lp(a, n, S, deadline):
    """Run LP with only conv(a,b) <= M and b >= 0, letting amplitude float."""
    c = np.convolve(a, a)
    M = float(np.max(c))
    rows = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    vals = np.tile(a, n)
    C = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    res = linprog(-np.ones(n), A_ub=C, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        b = np.clip(res.x, 0, 1000)
        sb = np.sum(b)
        if sb > 1e-12:
            return b
    return None


def _trailing_lp(a, n, S, low_frac=0.35, hi_frac=0.90, nmax=1500, deadline=None):
    """Truncated LP keeping the last L elements free; front part fixed."""
    a_sorted = np.sort(a)
    L = int(n * hi_frac)
    if L > nmax:
        L = nmax
    Lo = int(L * low_frac)
    Lo = max(Lo, 1)
    # Keep the last L indices free
    free_idx = np.arange(n - L, n)
    f = a[free_idx].copy()
    Sf = sum(f)
    c_full = np.convolve(a, a)
    M = float(np.max(c_full))

    Nb = len(free_idx)
    rows = []
    vals = []
    for i in range(Nb):
        j = free_idx[i]
        k0 = max(0, j - n + 1)
        k1 = min(2 * n - 2, j + n - 1)
        ks = np.arange(k0, k1 + 1)
        v = a[max(0, k0 - j):n - max(0, k1 - j + 1) if k1 - j + 1 < n else n]
        v = np.zeros(len(ks))
        for idx, k in enumerate(ks):
            d = k - j
            if 0 <= d < n:
                v[idx] = a[d]
        rows.append((ks, v))

    all_r = np.concatenate([r[0] for r in rows])
    all_c = np.concatenate([np.full(len(r[0]), i, dtype=np.int64) for i, r in enumerate(rows)])
    all_v = np.concatenate([r[1] for r in rows])
    Cs = csr_matrix((all_v, (all_r, all_c)), shape=(2 * n - 1, Nb))

    if time is not None and deadline is not None and time.monotonic() > deadline:
        return None

    res = linprog(-np.ones(Nb), A_ub=Cs, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * Nb, method='highs')
    if not res.success or res.x is None:

    g = np.clip(res.x, 0, 1000)
    sg = np.sum(g)
    if sg < 1e-12:
        return None
    g = g * Sf / sg
    return free_idx, g, Lo
