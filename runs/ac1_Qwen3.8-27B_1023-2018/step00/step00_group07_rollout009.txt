```python
def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Water filling
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cssv - 1) / rho > 0
    if not cond.any():
        return np.ones(n) / n
    rho_hat = np.max(np.where(cond)[0]) + 1
    theta = (cssv[rho_hat-1] - 1) / rho_hat
    return np.maximum(v - theta, 0)
```

This is the standard O(n log n) projection.
