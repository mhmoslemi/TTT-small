def _fast_score(a):
    n = a.size
    c = np.convolve(a, a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def _make_inits(n):
    inits = []
    i = np.arange(n, dtype=np.float64)
    # constant
    inits.append(np.ones(n) / n)
    # linear increasing
    a = np.arange(1, n + 1, dtype=np.float64)
    inits.append(a / a.sum())
    # linear decreasing
    a = np.arange(n, 1, -1, dtype=np.float64)
    inits.append(a / a.sum())
    # quadratic ramp
    a = (np.arange(1, n + 1, dtype=np.float64)) ** 2
    inits.append(a / a.sum())
    # sqrt ramp
    a = np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    inits.append(a / a.sum())
    # Gaussian

    # triangle
    a = np.minimum(i + 1, n - i).astype(np.float64)
    a = np.maximum(a, 1.0)
    inits.append(a / a.sum())
    return inits

    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    a_new = a - lr * mh / (np.sqrt(vh) + eps)
    return a_new, m, v
