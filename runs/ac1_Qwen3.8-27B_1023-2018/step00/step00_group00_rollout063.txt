import numpy as np
from scipy.signal import fftconvolve
import time

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000).copy()
    n = a.size
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    # Normalize to a convenient scale
    scale = np.sqrt(2 * n)
    a = a * (scale / a.sum())
    
    alpha = 0.01  # initial step fraction
    
    for it in range(1000000):  # max iterations
        if time.monotonic() > deadline:
            break
        
        # Compute convolution via FFT
        b = fftconvolve(a, a, mode='full')
        max_b = b.max()
        sum_a = a.sum()
        score = 2 * n * max_b / (sum_a ** 2)
        
        if score < best_score:
            best = a.copy()
            best_score = score
        
        # Gradient at the peak
        k_star = int(np.argmax(b))
        # grad[j] = 2*a[k_star - j] for valid j
        idx = k_star - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        grad = np.zeros(n)
        grad[valid] = 2 * a[idx[valid]]
        
        # Step size: fraction of max allowable
        pos_grad = grad > 0
        if pos_grad.any():
            max_step = np.min(a[pos_grad] / grad[pos_grad]) * 0.9
        else:
            max_step = 0.0
        
        step = alpha * max_step
        a_new = a - step * grad
        a_new = np.clip(a_new, 0, 1000)
        
        # Renormalize
        if a_new.sum() > 0.01:
            a_new = a_new * (scale / a_new.sum())
        
        # Check improvement
        b_new = fftconvolve(a_new, a_new, mode='full')
        sum_new = a_new.sum()
        score_new = 2 * n * b_new.max() / (sum_new ** 2)
        
        if score_new < score:
            a = a_new
        else:
            alpha = max(0.001, alpha * 0.8)
        
        # Periodic perturbation to escape local minima
        if it % 100 == 0 and it > 0:
            mask = rng.random(n) < 0.05
            a[mask] *= np.exp(rng.normal(0, 0.1, mask.sum()))
            a = np.clip(a, 0, 1000)
            a = a * (scale / a.sum())
    
    return [float(x) for x in best]

import numpy as np
from scipy.signal import fftconvolve
import time


def _conv_and_score(a, n):
    """Compute full convolution and AC1 score."""
    b = fftconvolve(a, a, mode='full')
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return b, max_b, np.inf
    score = 2.0 * n * max_b / (sum_a ** 2)
    return b, max_b, score


def _gradient_at_peaks(a, b, n, top_k=5):
    """Compute sum of gradients at top-k peak positions."""
    max_b = b.max()
    # Find positions where b is within 0.5% of max
    threshold = 0.995 * max_b
    near = np.where(b >= threshold)[0]
    if len(near) > top_k:
        # Take the top_k highest
        top_idx = near[np.argsort(b[near])[-top_k:]]
    else:
        top_idx = near
    if len(top_idx) == 0:
        top_idx = np.array([int(np.argmax(b))])
    
    grad = np.zeros(n)
    for k in top_idx:
        idx = k - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        grad[valid] += a[idx[valid]]
    grad *= 2.0
    # Project onto simplex tangent (sum-zero)
    grad -= grad.sum() / n
    return grad
