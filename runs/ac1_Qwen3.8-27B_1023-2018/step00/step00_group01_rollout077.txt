    best_p = normalize(parent)
    if best_p is None:
        best_p = np.ones(N) / N
    best_score = fast_score(best_p)

    top = []

    def consider(p, force=False):
        nonlocal best_p, best_score
        pn = normalize(p)
        if pn is None:
            return
        sc = fast_score(pn)
        if not np.isfinite(sc):
            return
        if force or sc < best_score - 1e-14:
            best_score = sc
            best_p = pn.copy()
        if len(top) < 6:
            top.append((sc, pn.copy()))
        else:
            worst_i = int(np.argmax([s for s, _ in top]))
            if sc < top[worst_i][0] - 1e-14:
                top[worst_i] = (sc, pn.copy())
        top.sort(key=lambda z: z[0])

    consider(parent, force=False)
    consider(np.ones(N), force=False)

    # ------------------------------------------------------------------
    # Cheap structured seed pool.  These are not the final answer; they give
    # several basins for the local optimizers below.
    # ------------------------------------------------------------------
    if time_left() > 30.0:
        x = (np.arange(N) + 0.5) / float(N)
        eps = 1e-9

        for r in (0.25, 0.4, 0.6, 0.8, 1.2, 1.6, 2.2, 3.1, 4.7):
            p = np.power(r, x)
            consider(p)
            consider(p[::-1])

        for a in (-2.0, -1.25, -0.7, -0.35, 0.0, 0.35, 0.7, 1.25, 2.0, 3.0):
            p = np.power(x + eps, a)
            consider(p)
            p = np.power(1.0 - x + eps, a)
            consider(p)
            p = np.power(np.maximum(2.0 * x - 1.0, 0.05) ** a +
                         np.maximum(1.0 - 2.0 * x, 0.05) ** a, 1.0)
            consider(p)

        # Low-frequency smooth random profiles.
        for _ in range(350):
            if time_left() <= 25.0:
                break
            M = int(rng.integers(3, 7))
            v = np.zeros(N, dtype=np.float64)
            for m in range(1, M + 1):
                if rng.random() < 0.7:
                    v += float(rng.normal(0.0, 0.75)) * np.cos(np.pi * m * x)
                if rng.random() < 0.7:
                    v += float(rng.normal(0.0, 0.75)) * np.sin(np.pi * m * x)
            p = np.exp(np.clip(v, -2.5, 2.5))
            consider(p)

        # Random block / step profiles.
        for B in (4, 8, 16, 32, 64):
            for _ in range(35):
                if time_left() <= 20.0:
                    break
                h = rng.exponential(size=B) + rng.random(size=B) * 0.3
                idx = np.minimum(B - 1, (np.arange(N) * B) // N)
                p = h[idx]
                consider(p)

    consider(best_p, force=False)

    # ------------------------------------------------------------------
    # Shared objective machinery.
    # ------------------------------------------------------------------
    def objective_core(p, beta):
        """Return (soft_score_components, weights, Fp, center) for sum(p)=1."""
        F = np.fft.rfft(p, Nf)
        q = np.fft.irfft(F * F, Nf)[:L]
        q = np.maximum(q, 0.0)
        qmax = float(np.max(q)) if q.size else 0.0
        if qmax <= 0.0 or not np.isfinite(qmax):
            q = np.ones(L) / float(L)
            qmax = 1.0 / L

        sh = beta * (q - qmax)
        smax = float(np.max(sh))
        ew = np.exp(sh - smax)
        sw = float(ew.sum())
        if sw <= 0.0 or not np.isfinite(sw):
            w = np.ones(L) / float(L)
            soft = qmax
        else:
            w = ew / sw
            soft = qmax + (smax + math.log(sw)) / beta

        score = float(2.0 * N * soft)
        return score, w, F, q, qmax

    def gradient_from_weights(w, Fp):
        W = np.fft.rfft(w, Nf)
        r = np.fft.irfft(W * np.conj(Fp), Nf)[:N]
        r = np.maximum(r, 0.0)
        return r

    def theta_from_p(p):
        p = np.maximum(np.asarray(p, dtype=np.float64), 1e-300)
        th = np.log(p)
        th = th - float(th[0])
        return th[1:].astype(np.float64, copy=True)

    def p_from_theta_full(theta1):
        th = np.empty(N, dtype=np.float64)
        th[0] = 0.0
        th[1:] = theta1
        if not np.all(np.isfinite(th)):
            th = np.where(np.isfinite(th), th, 0.0)
        z = th - float(np.max(th))
        e = np.exp(z)
        s = float(e.sum())
        if s <= 0.0 or not np.isfinite(s):
            return np.ones(N) / N
        p = e / s
        p = np.clip(p, 0.0, None)
        s2 = float(p.sum())
        if s2 <= 0.0:
            return np.ones(N) / N
        return p / s2

    def run_full_lbfgs(p_start, beta, maxiter):
        if _sp_minimize is None or N <= 1:
            return normalize(p_start)

        x0 = theta_from_p(p_start)
        if x0.size == 0:
            return normalize(p_start)

        pen_coeff = 1e-6

        def fun(x):
            if not np.all(np.isfinite(x)):
                return 1e100, np.zeros_like(x)

            p = p_from_theta_full(x)
            score, w, F, q, qmax = objective_core(p, beta)
            if not np.isfinite(score):
                score = 1e100

            r = gradient_from_weights(w, F)
            # Score is 2N * soft(q).  Chain rule through softmax with theta0 fixed.
            G = 4.0 * N * r
            center = float(np.dot(G, p))
            grad = p * (G - center)
            grad[1:] += 2.0 * pen_coeff * x
            score += pen_coeff * float(np.dot(x, x))

            if not np.all(np.isfinite(grad)):
                grad = np.where(np.isfinite(grad), grad, 0.0)
            return float(score), grad.astype(np.float64, copy=False)

        try:
            res = _sp_minimize(
                fun,
                x0.astype(np.float64, copy=True),
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": int(maxiter),
                    "ftol": 1e-12,
                    "gtol": 1e-8,
                    "maxls": 25,
                    "maxcor": 20,
                    "max_step": 50.0,
                    "disp": False,
                },
            )
            p_new = p_from_theta_full(np.asarray(res.x, dtype=np.float64))
        except Exception:
            p_new = normalize(p_start)

        return normalize(p_new)

    def run_block_lbfgs(p_start, B, beta, maxiter):
        if _sp_minimize is None or N <= 4 or B < 2:
            return normalize(p_start)

        B = int(B)
        idx = np.minimum(B - 1, (np.arange(N) * B) // N).astype(np.int64)
        counts = np.bincount(idx, minlength=B).astype(np.float64)
        counts = np.maximum(counts, 1.0)

        # Initial theta from block averages.
        h = np.bincount(idx, weights=np.asarray(p_start, dtype=np.float64), minlength=B) / counts
        h = np.maximum(h, 1e-12)
        th0 = np.log(h)
        th0 = th0 - float(th0[0])
        x0 = th0[1:].astype(np.float64, copy=True)

        pen_coeff = 1e-5

        def block_p(ztheta):
            th = np.empty(B, dtype=np.float64)
            th[0] = 0.0
            th[1:] = ztheta
            z = th - float(np.max(th))
            v = np.exp(z)
            denom = float(np.dot(v, counts))
            if denom <= 0.0 or not np.isfinite(denom):
                return None, None
            p = v[idx] / denom
            p = np.clip(p, 0.0, None)
            s = float(p.sum())
            if s <= 0.0 or not np.isfinite(s):
                return None, None
            return p / s, (v, denom)

        def fun(x):
            if not np.all(np.isfinite(x)):
                return 1e100, np.zeros_like(x)

            th = np.empty(B, dtype=np.float64)
            th[0] = 0.0
            th[1:] = x
            z = th - float(np.max(th))
            v = np.exp(z)
            denom = float(np.dot(v, counts))
            if denom <= 0.0 or not np.isfinite(denom):
                return 1e100, np.zeros_like(x)

            p = v[idx] / denom
            p = np.clip(p, 0.0, None)
            s = float(p.sum())
            if s <= 0.0 or not np.isfinite(s):
                p = np.ones(N) / N
            else:
                p = p / s

            score, w, F, q, qmax = objective_core(p, beta)
            if not np.isfinite(score):
                score = 1e100

            r = gradient_from_weights(w, F)
            G = 4.0 * N * r
            center = float(np.dot(G, p))

            sumG = np.bincount(idx, weights=G, minlength=B).astype(np.float64)
            avgG = sumG / counts
            mass = counts * v / denom
            grad = mass * (avgG - center)
            grad[1:] += 2.0 * pen_coeff * x
            score += pen_coeff * float(np.dot(x, x))

            if not np.all(np.isfinite(grad)):
                grad = np.where(np.isfinite(grad), grad, 0.0)
            return float(score), grad.astype(np.float64, copy=False)

        try:
            res = _sp_minimize(
                fun,
                x0,
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": int(maxiter),
                    "ftol": 1e-12,
                    "gtol": 1e-8,
                    "maxls": 25,
                    "maxcor": 20,
                    "max_step": 50.0,
                    "disp": False,
                },
            )
            p_new, _ = block_p(np.asarray(res.x, dtype=np.float64))
        except Exception:
            p_new = normalize(p_start)

        return normalize(p_new)

    # ------------------------------------------------------------------
    # Coarse block refinement.  This matches the step-function viewpoint and
    # usually gives a much better basin than the parent.
    # ------------------------------------------------------------------
    if _sp_minimize is not None and N >= 8:
        for B in (16, 32, 64):
            if time_left() < 70.0:
                break
            p_b = run_block_lbfgs(best_p, B, beta=20, maxiter=110)
            consider(p_b)
            if time_left() < 45.0:
                break
            th_rand = np.log(np.maximum(rng.random(B) + 1e-3, 1e-3))
            th_rand = th_rand - float(th_rand[0])
            p_seed = p_from_theta_full(np.zeros(N - 1))
            p_seed = np.clip(p_seed, 0.0, None)
            p_seed /= p_seed.sum()
            # Use a simple random block seed by converting random heights to p.
            idx = np.minimum(B - 1, (np.arange(N) * B) // N)
            p_seed = rng.exponential(size=B)[idx]
            p_seed /= p_seed.sum()
            p_b2 = run_block_lbfgs(p_seed, B, beta=20, maxiter=90)
            consider(p_b2)

    for sc, ptop in list(top):
        if time_left() < 70.0:
            break
        p_b = run_block_lbfgs(ptop, 48, beta=20, maxiter=90)
        consider(p_b)

    consider(best_p, force=False)

    # ------------------------------------------------------------------
    # Full-resolution local refinement by softmax natural gradient L-BFGS.
    # Run a few betas so the objective sharpens toward the exact max.
    # ------------------------------------------------------------------
    starts = [best_p]
    for sc, ptop in top:
        if ptop is best_p:
            continue
        if len([z for z in starts if z is ptop]) == 0:
            starts.append(ptop)
        if len(starts) >= 4:
            break

    # Add a small noisy uniform start to break symmetric local minima.
    if time_left() > 90.0:
        u = np.ones(N) / N
        u = np.clip(u * np.exp(rng.normal(0.0, 0.02, N)), 0.0, None)
        u /= u.sum()
        starts.append(u)

    beta_schedule = (12, 30, 80, 180)
    for p_start in starts:
        if time_left() < 90.0:
            break
        p_cur = normalize(p_start)
        if p_cur is None:
            continue
        sc_cur = fast_score(p_cur)
        for beta in beta_schedule:
            if time_left() < 60.0:
                break
            maxiter = 90
            if beta >= 80:
                maxiter = 70
            p_new = run_full_lbfgs(p_cur, beta=beta, maxiter=maxiter)
            sc_new = fast_score(p_new)
            consider(p_new)
            if sc_new < sc_cur - 1e-12 and p_new is not None:
                p_cur = p_new
                sc_cur = sc_new

    # If time remains, gently perturb the current best and restart softly.
    if _sp_minimize is not None and time_left() > 120.0 and N >= 8:
        for _ in range(6):
            if time_left() < 60.0:
                break
            std = float(rng.choice((0.02, 0.05, 0.10, 0.20)))
            th = theta_from_p(best_p)
            th = th + std * rng.standard_normal(th.size)
            p_new = run_full_lbfgs(p_from_theta_full(th), beta=120, maxiter=45)
            consider(p_new)

    # ------------------------------------------------------------------
    # Small exact-ish polishing step using a hard/soft peak-shaving update.
    # ------------------------------------------------------------------
    if time_left() > 60.0 and N >= 8:
        p_cur = normalize(best_p)
        cur_sc = fast_score(p_cur)
        beta = 80.0
        stuck = 0
        for it in range(350):
            if time_left() < 15.0:
                break

            F = np.fft.rfft(p_cur, Nf)
            q = np.fft.irfft(F * F, Nf)[:L]
            q = np.maximum(q, 0.0)
            M = float(np.max(q)) if q.size else 0.0
            if M <= 0 or not np.isfinite(M):
                break

            sh = beta * (q - M)
            smax = float(np.max(sh))
            ew = np.exp(sh - smax)
            sw = float(ew.sum())
            if sw <= 0 or not np.isfinite(sw):
                w = np.ones(L) / float(L)
            else:
                w = ew / sw

            r = gradient_from_weights(w, F)
            h = r / M
            h = h - float(np.mean(h))
            hscale = float(np.max(np.abs(h)))
            if hscale <= 0:
                break
            h = h / hscale

            accepted = False
            for gamma in (0.15, 0.05, 0.35, 0.02):
                p_try = p_cur * np.exp(-gamma * h)
                p_try = np.clip(p_try, 1e-15, None)
                p_try /= p_try.sum()
                sc_try = fast_score(p_try)
                if np.isfinite(sc_try) and sc_try < cur_sc - 1e-15:
                    p_cur = p_try
                    cur_sc = sc_try
                    accepted = True
                    break

            if accepted:
                stuck = 0
            else:
                stuck += 1
                beta = min(400.0, beta * 1.25)
                if stuck == 8:
                    p_cur = p_cur * np.exp(rng.normal(0.0, 0.01, N))
                    p_cur = np.clip(p_cur, 1e-15, None)
                    p_cur /= p_cur.sum()
                    cur_sc = fast_score(p_cur)
                    stuck = 0
                if stuck > 30:
                    break

            if it % 20 == 0 or it == 349:
                consider(p_cur)

    consider(best_p, force=False)

    # ------------------------------------------------------------------
    # Final exact validation against the read-only evaluator.
    # ------------------------------------------------------------------
    final_p = best_p
    if final_p is None or final_p.size == 0 or not np.all(np.isfinite(final_p)):
        final_p = normalize(parent)
        if final_p is None:
            final_p = np.ones(64) / 64.0

    final_p = np.asarray(final_p, dtype=np.float64)
    s = float(final_p.sum())
    if not np.isfinite(s) or s <= 0:
        final_p = np.ones(N) / float(N)
        s = 1.0
    final_p = final_p / s
    final_p = np.clip(final_p, 0.0, 1000.0)

    try:
        final_exact = float(evaluate_sequence([float(v) for v in final_p]))
    except Exception:
        final_exact = float("inf")

    # Fallback to parent if exact evaluation unexpectedly gets worse.
    try:
        par_n = normalize(parent)
        if par_n is not None:
            par_exact = float(evaluate_sequence([float(v) for v in par_n]))
            if np.isfinite(par_exact) and (not np.isfinite(final_exact) or par_exact < final_exact):
                final_p = par_n
                final_exact = par_exact
    except Exception:
        pass

    # Absolute last safeguard.
    if not np.isfinite(final_exact):
        final_p = np.ones(256) / 256.0
        try:
            final_exact = float(evaluate_sequence([float(v) for v in final_p]))
        except Exception:
            pass

    final_p = np.where(np.isfinite(final_p), final_p, 0.0)
    final_p = np.clip(final_p, 0.0, 1000.0)
    if final_p.size == 0 or float(final_p.sum()) < 0.01:
        final_p = np.ones(256) / 256.0

    return [float(v) for v in final_p]
