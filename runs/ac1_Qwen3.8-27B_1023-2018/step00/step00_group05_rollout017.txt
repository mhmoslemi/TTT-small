def convolve_via_fft(a):

    n = len(a)
    c = np.convolve(a, a)
    k_star = np.argmax(c)
    # b[m] = a[k_star - m] if 0 <= k_star - m < n, else 0
    # This is a[k_star-m] for m in [max(0, k_star-n+1), min(n-1, k_star)]
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    b = np.zeros(n)
    # b[m] = a[k_star - m] for m in [lo, hi]
    # k_star - m ranges from k_star - lo to k_star - hi
    # = k_star - max(0, k_star-n+1) to k_star - min(n-1, k_star)
    # If k_star < n: lo=0, hi=k_star, so k_star-m goes from k_star to 0
    # If k_star >= n: lo=k_star-n+1, hi=n-1, so k_star-m goes from n-1 to k_star-n+1
    m_indices = np.arange(lo, hi + 1)
    a_indices = k_star - m_indices
    b[m_indices] = a[a_indices]
    # Fix diagonal: where k_star = 2*m
    diag_m = k_star // 2
    if k_star % 2 == 0 and 0 <= diag_m < n:

    """One subgradient step on max(c) with sum(a) fixed."""
    n = len(a)
    c = full_convolve(a)
    k_star = np.argmax(c)
    # Compute g[m] = d c[k_star] / d a[m]
    # g[m] = a[k_star - m] + a[m] (when k_star - m == m, i.e., k_star = 2m)
