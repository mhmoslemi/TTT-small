def _gradient_refine(p, scorer, time_limit, seed, consider):

    p=_project(p,rng)
    cur=scorer.score(p)
    consider(p, exact_if_close=True)
    lr=1.0
    topK = max(8, min(128, n//8))
    stall=0
    while time.monotonic() < deadline:
       if cur < 1.2: break? optional
       b=scorer.conv(p)
       M=float(np.max(b))
       if M <= 1e-15: break
       # top indices
       if topK >= len(b):
          order=np.arange(len(b))[::-1]
       else:
          order=np.argpartition(b, -topK)[-topK:][::-1]
       # weights
       vals=b[order]
       # if top all same? use uniform
       if vals[-1] > 0 and M > 0:
          w = np.exp(-3.0*(M - vals)/M)
       else:
          w=np.ones(topK)
       w/=w.sum()
       G=np.zeros(n)
       for k,wt in zip(order,w):

           k=int(k)

       G -= G.mean()
       # If gradient small, kick

import numpy as np
import time

_scorer_cache = {}

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

class _ConvScorer:
    def __init__(self, n):
        self.n = int(n)
        self.L = 2 * self.n - 1
        self.use_fft = self.n > 256
        if self.use_fft:
            self.N = _next_pow2(self.L)
        else:
            self.N = 0

    def conv(self, a):
        a = np.asarray(a, dtype=float)
        if self.use_fft:
            fa = np.fft.rfft(a, self.N)
            c = np.fft.irfft(fa * fa, self.N)[:self.L]
            c = np.maximum(c, 0.0)
            return c
        return np.maximum(np.convolve(a, a), 0.0)

    def score(self, a):
        return float(2.0 * self.n * np.max(self.conv(a)))

def _scorer(n):
    n = int(n)
    sc = _scorer_cache.get(n)
    if sc is None:
        sc = _ConvScorer(n)
        _scorer_cache[n] = sc
    return sc

def _project(a, rng=None):
    a = np.asarray(a, dtype=float)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 1e-12, None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-8:
        if rng is not None:
            a = rng.random(a.size) + 1.0
        else:
            a = np.ones(a.size, dtype=float)
        s = float(a.sum())
    return a / s

def _hermitian(P):
    N = len(P)
    idx = np.arange(N)
    partner = (-idx) % N
    return 0.5 * (P + np.conj(P[partner]))

def _sqrt_start(n):
    L = 2 * n - 1
    N = _next_pow2(L)
    target = np.zeros(N, dtype=float)
    target[:L] = 1.0 / L
    C = np.fft.fft(target)
    P = np.sqrt(np.abs(C) + 1e-30) * np.exp(0.5j * np.angle(C))
    P = _hermitian(P)
    q = np.fft.ifft(P).real
    p = np.maximum(q[:n], 0.0)
    return p

def _compute_l2_grad(p, e, N, n):
    L = 2 * n - 1
    P = np.fft.fft(p, N)
    Erev = np.fft.fft(e[::-1], N)
    h = np.fft.ifft(P * Erev).real
    return 2.0 * h[L - n:L][::-1]

def _compute_max_grad(p, scorer, n):
    b = scorer.conv(p)
    M = float(np.max(b))
    if M <= 1e-15:
        return np.zeros(n, dtype=float)
    K = min(64, max(8, int(2 * np.sqrt(n))))
    if K >= len(b):
        order = np.arange(len(b))
    else:
        order = np.argpartition(b, -K)[-K:]
    vals = b[order]
    w = np.exp(-3.0 * (M - vals) / (M + 1e-30))
    w /= w.sum()
    G = np.zeros(n, dtype=float)
    for k, wt in zip(order, w):

        else:

        if not np.isfinite(s) or s <= 0.0:
            return
        arr = arr / s
        arr = np.where(np.isfinite(arr), arr, 0.0)
        s2 = float(arr.sum())
        if not np.isfinite(s2) or s2 <= 0.0:
            return
        arr = arr / s2
        if sc is None:
            sc = _scorer(arr.size).score(arr)
        if not np.isfinite(sc):
            return
        sc = float(sc)
        if sc < best_score:
            best_score = sc
            best_seq = arr.copy()
        if len(top_candidates) < 5:
            top_candidates.append((sc, arr.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif sc < top_candidates[-1][0]:
            top_candidates.append((sc, arr.copy()))
            top_candidates.sort(key=lambda x: x[0])
            top_candidates.pop()

    parent_arr = None
    try:
        parent = height_sequence_1
        arr = np.asarray(parent, dtype=float)
        if arr.ndim == 1 and arr.size > 0 and np.all(np.isfinite(arr)):
            parent_arr = np.clip(arr, 0.0, 1000.0)
            s = float(parent_arr.sum())
            if s > 0.0:
                consider(parent_arr / s)
            else:
                parent_arr = None
    except Exception:
        parent_arr = None

    if best_seq is None:
        consider(np.ones(4, dtype=float))

    n_candidates = []
    def add_n(n):
        n = int(n)
        if 4 <= n <= 20000 and n not in n_candidates:
            n_candidates.append(n)

    if parent_arr is not None:
        add_n(parent_arr.size)
    if parent_arr is None:
        add_n(32)
    for n in (64, 128, 256, 512, 1024, 2048):
        add_n(n)

    def optimize_n(n, end_time, name_seed):
        scorer = _scorer(n)
        rng = np.random.default_rng(name_seed)
        starts = []
        if parent_arr is not None and parent_arr.size == n:
            starts.append(parent_arr)
        try:
            starts.append(_sqrt_start(n))
        except Exception:
            pass
        starts.append(np.ones(n, dtype=float))
        for _ in range(5):
            starts.append(rng.random(n) + 0.01)
        x = np.arange(n, dtype=float)
        denom = np.maximum(x + 1.0, n - x)
        starts.append(1.0 / (denom + 1e-3))
        starts.append(1.0 / (x + 1.0 + 1e-3))
        starts.append(np.exp(x / max(1.0, float(n - 1))))
        c = np.cos(np.pi * x / max(1.0, float(n - 1)))
        starts.append(1.0 + 0.8 * (c - c.mean()))
        psp = np.zeros(n, dtype=float)
        m = max(4, n // 4)
        idx = rng.choice(n, size=min(n, m), replace=False)
        psp[idx] = 1.0
        psp += 1e-3
        starts.append(psp)

        scored = []
        for p0 in starts:
            if time.monotonic() >= end_time:
                break
            p = _project(p0, rng)
            sc = scorer.score(p)
            if np.isfinite(sc):
                scored.append((sc, p))
                consider(p, sc)
        if not scored:
            return
        scored.sort(key=lambda z: z[0])
        now = time.monotonic()
        rem = max(0.0, end_time - now)
        if rem <= 5.0:
            return
        _run_start(scored[0][1], min(end_time, now + 0.65 * rem), name_seed, scorer, consider)
        if len(scored) > 1 and time.monotonic() < end_time:
            _run_start(scored[1][1], end_time, name_seed + 17, scorer, consider)

    parent_n = parent_arr.size if parent_arr is not None else None
    for idx, n in enumerate(n_candidates):
        now = time.monotonic()
        remaining = deadline - now
        if remaining < 35.0:
            break
        if parent_n is not None and n == parent_n and idx == 0:
            alloc = min(520.0, max(80.0, 0.42 * remaining))
        else:
            n_left = max(1, len(n_candidates) - idx)
            alloc = min(120.0, max(25.0, remaining / n_left))
        end_n = now + alloc
        try:
            optimize_n(n, end_n, int(seed) + 101 * (idx + 1))
        except Exception:
            pass

    final_list = None
    final_score = float("inf")
    if top_candidates:
        for _, arr in top_candidates:
            try:
                lst = [float(v) for v in arr.tolist()]
                if not lst or not all(np.isfinite(v) for v in lst):
                    continue
                sc = float(evaluate_sequence(lst))
                if np.isfinite(sc) and sc < final_score:
                    final_score = sc
                    final_list = lst
            except Exception:
                continue

    if final_list is None and best_seq is not None:
        lst = [float(v) for v in best_seq.tolist()]
        try:
            if lst and np.isfinite(evaluate_sequence(lst)):
                final_list = lst
        except Exception:
            pass

    if final_list is None:
        final_list = [1.0, 1.0, 1.0, 1.0]

    try:
        if evaluate_sequence(final_list) != float("inf"):
            return final_list
    except Exception:
        pass
    return [1.0, 1.0, 1.0, 1.0]
