import numpy as np

def _as_unit_nonnegative(a):
    try:
        a = np.asarray(a, dtype=np.float64)
    except Exception:
        return None
    if a.ndim != 1 or a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    if not np.isfinite(a).all():
        return None
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    p = a / s
    p = np.clip(p, 0.0, 1.0)
    s2 = float(p.sum())
    if not np.isfinite(s2) or s2 <= 0.0:
        return None
    p /= s2
    return p


def _central_binomial_nonblocked(n):
    if n <= 0:
        return None
    a = np.ones(n, dtype=np.float64)
    if n > 1:
        i = np.arange(1, n, dtype=np.float64)
        ratios = (2.0 * i - 1.0) / (2.0 * i)
        a[1:] = np.cumprod(ratios)
    return a


def _signed_flat_root_abs(alpha, n, m):
    """Nonnegative approximation to |sqrt((1-z^m)/(1-z))| truncated to length n."""
    b = np.array(alpha, dtype=np.float64, copy=True)
    if m is None or m <= 0 or m >= n:
        return b
    R = n // m
    c = 1.0
    for r in range(1, R + 1):
        c *= (r - 1.5) / r
        start = r * m
        if start >= n:
            break
        b[start:] += c * alpha[:n - start]
    return np.abs(b)


def _fft_score(p, L):
    n = p.size
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    M = float(np.max(c))
    if not np.isfinite(M) or M < 0.0:
        M = 0.0
    return 2.0 * float(n) * M


def _fft_score_grad(p, L, beta):
    n = p.size
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    M = float(np.max(c))
    score = 2.0 * float(n) * M

    t = beta * c
    t -= float(t.max())
    e = np.exp(t)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    Fw = np.fft.rfft(w[::-1], L)
    g = np.fft.irfft(F * Fw, L)
    grad = 2.0 * g[n - 1:2 * n - 1][::-1]
    grad = np.maximum(grad, 0.0)
    return score, grad


def _safe_mirror_update(p, grad, eta):
    gm = float(np.max(np.abs(grad)))

        return p.copy(), False
    d = grad / gm

    if not np.isfinite(s) or s <= 0.0:

    q /= s
    if not np.isfinite(q).all():

    return q, True

    add_seed(np.maximum(alpha[::-1] + alpha * 0.25, 0.0))

    for m in (max(16, n // 8), n // 4, n // 3, n // 2, 3 * n // 4):
        if 4 < m < n:
            add_seed(_signed_flat_root_abs(alpha, n, m))

    idx = np.arange(n, dtype=np.float64) + 0.5
    for pexp in (0.38, 0.44, 0.50, 0.56, 0.64):
        add_seed(idx ** (-pexp))

    x = np.arange(n, dtype=np.float64) / max(1.0, float(n - 1))
    for lam in (0.6, 1.3, 2.2, 3.7, 6.0):
        add_seed(np.exp(-lam * x))
    for lam in (0.8, 2.0, 5.0):
        add_seed(np.ones(n, dtype=np.float64) - lam * x)

    if parent is not None:
        if parent.size == n:
            add_seed(parent)
        elif parent.size > n:
            add_seed(parent[:n])

    seed_scores = []
    for p in seeds:
        if p.size != n:
            continue
        sc = _fft_score(p, L)
        if np.isfinite(sc):
            seed_scores.append((sc, p.copy()))

    if not seed_scores:
        seed_scores = [(2.0, alpha_p)]

    seed_scores.sort(key=lambda z: z[0])

    chosen = []
    chosen_scores = []
    for sc, p in seed_scores:
        keep = True
        for q in chosen:
            d = np.abs(p - q).sum()
            if d < 1e-3:
                keep = False
                break
        if keep:
            chosen.append(p)
            chosen_scores.append(sc)
            if len(chosen) >= 8:
                break

    global_best = chosen[0].copy() if chosen else alpha_p.copy()
    global_score = chosen_scores[0] if chosen_scores else _fft_score(global_best, L)

    if global_score <= 0.0 or not np.isfinite(global_score):
        global_score = _fft_score(global_best, L)
    if not np.isfinite(global_score):
        global_score = float("inf")

    opt_start = time.monotonic()
    opt_span = max(0.0, opt_end - opt_start)
    fracs = (0.30, 0.21, 0.15, 0.11, 0.09, 0.06, 0.05, 0.03)

    for k, p in enumerate(chosen):
        if k >= len(fracs):
            break
        end = opt_start + opt_span * fracs[k]
        if end <= time.monotonic():
            break
        if time.monotonic() >= opt_end - 0.2:
            break

        if k == 0:
            b0f, b1f, e0 = 1.0, 45.0, 0.22
        elif k <= 2:
            b0f, b1f, e0 = 2.0, 40.0, 0.20
        else:
            b0f, b1f, e0 = 4.0, 35.0, 0.18

        best_p, best_sc = _optimize_unit_vector(
            p, n, L, end, rng,
            beta0_factor=b0f, beta1_factor=b1f, eta0=e0
        )
        if best_sc < global_score:
            global_score = best_sc
            global_best = best_p.copy()

    if time.monotonic() < opt_end - 1.0:
        best_p, best_sc = _optimize_unit_vector(
            global_best, n, L, opt_end, rng,
            beta0_factor=10.0, beta1_factor=80.0, eta0=0.16
        )
        if best_sc < global_score:
            global_score = best_sc
            global_best = best_p.copy()

    exact_candidates = []

    def add_exact_candidate(p):
        if p is None or p.size != n:
            return
        q = np.asarray(p, dtype=np.float64)
        if not np.isfinite(q).all():
            return
        q = np.clip(q, 0.0, 1000.0)
        s = float(q.sum())
        if not np.isfinite(s) or s <= 0.0:
            return
        q = q / s
        sc = _fft_score(q, L)
        if not np.isfinite(sc):
            sc = 1e100
        exact_candidates.append((sc, q))

    add_exact_candidate(global_best)
    add_exact_candidate(alpha_p)
    add_exact_candidate(alpha_p[::-1])
    add_exact_candidate(alpha_p + alpha_p[::-1])
    for p in chosen[:4]:
        add_exact_candidate(p)
    if parent is not None:
        if parent.size == n:
            add_exact_candidate(parent)
        elif parent.size > n:
            add_exact_candidate(parent[:n])

    exact_candidates.sort(key=lambda z: z[0])
    # Remove very duplicate exact candidates cheaply.
    filtered = []
    for sc, p in exact_candidates:
        dup = False
        for q in filtered:
            if np.abs(p - q).sum() < 1e-4:
                dup = True
                break
        if not dup:
            filtered.append((sc, p))
        if len(filtered) >= 6:
            break
    exact_candidates = filtered

    best_exact_seq = None
    best_exact_score = float("inf")

    for _, p in exact_candidates:
        if time.monotonic() > eval_end - 0.5:
            break
        try:
            seq = [float(x) for x in p]
            sc = float(evaluate_sequence(seq))
        except Exception:
            sc = float("inf")
        if np.isfinite(sc) and sc < best_exact_score:
            best_exact_score = sc
            best_exact_seq = [float(x) for x in p]

    if best_exact_seq is None:
        try:
            best_exact_seq = [float(x) for x in global_best]
        except Exception:
            best_exact_seq = fallback_alpha_len(n)

    # Final sanity: ensure non-empty finite list.
    try:
        if not best_exact_seq or len(best_exact_seq) == 0 or not all(
            np.isfinite(float(x)) for x in best_exact_seq
        ):
            best_exact_seq = fallback_alpha_len(n)
    except Exception:
        best_exact_seq = fallback_alpha_len(n)

    return best_exact_seq
