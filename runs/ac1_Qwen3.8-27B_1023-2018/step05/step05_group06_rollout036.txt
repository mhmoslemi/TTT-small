from scipy.optimize import minimize

    """Compute full autocorrelation using FFT."""
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return c

    """Compute score from precomputed convolution and sum."""
    if s < 0.01:
        return math.inf
    return 2.0 * n * float(c.max()) / (s * s)

    """Compute score of a sequence."""
    p = np.asarray(p, dtype=np.float64)

    s = float(p.sum())

        return math.inf

    p_cur = best_p.copy()
    R_cur = best_R.copy()
    cur_score = best_score
    S = 1.0
    flat = 0
    sweep = 0
    perturbation_count = 0

    while time.monotonic() < hard - 20.0:
        sweep += 1

        # Precompute prefix/suffix max for O(1) max_const lookup
        prefix_max = np.empty(Ltot + 1, dtype=np.float64)
        prefix_max[0] = 0.0
        for k in range(1, Ltot + 1):
            prefix_max[k] = max(prefix_max[k - 1], R_cur[k - 1])
        suffix_max = np.empty(Ltot + 1, dtype=np.float64)
        suffix_max[Ltot] = 0.0
        for k in range(Ltot - 1, -1, -1):
            suffix_max[k] = max(suffix_max[k + 1], R_cur[k])

        improved = False
        k_peak = int(np.argmax(R_cur))

        # Build priority order: coordinates affecting the peak first
        lo_pk = max(0, k_peak - n + 1)
        hi_pk = min(n - 1, k_peak)
        peak_coords = np.arange(lo_pk, hi_pk + 1)
        other_coords = np.setdiff1d(np.arange(n), peak_coords)
        if len(other_coords) > 0:
            order = np.concatenate([
                peak_coords[rng.permutation(len(peak_coords))],
                other_coords[rng.permutation(len(other_coords))]
            ])
        else:
            order = peak_coords[rng.permutation(len(peak_coords))]

        for i in order:
            if time.monotonic() > hard - 20.0:
                break
            if S < 1e-300:
                break

            # Periodic renormalization
            if S > 1.3 or S < 0.77:
                inv = 1.0 / S
                p_cur *= inv
                R_cur *= inv * inv
                S = 1.0

            v = float(p_cur[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            infl = p_cur[:Ls]

            # max_const via prefix/suffix (O(1))
            max_const = 0.0
            if start > 0:
                max_const = max(max_const, float(prefix_max[start]))
            if end < Ltot:
                max_const = max(max_const, float(suffix_max[end]))

            diag_offset = 2 * i - start

            # Search range
            x_lo = 0.0
            x_hi = max(4.0 * v, 12.0 / n)
            x_hi = min(x_hi, 25.0 / n)
            if x_hi <= x_lo:
                x_hi = x_lo + 2.0 / n

            # Coarse grid evaluation
            pts = np.linspace(x_lo, x_hi, G_coarse + 1)
            ws = pts - v
            aff = R_cur[start:end][None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag_offset < Ls:
                aff[:, diag_offset] += ws * ws
            maff = aff.max(axis=1)
            maff = np.maximum(maff, max_const)
            denoms = S + ws
            vals = np.where(denoms > 1e-15,
                            2.0 * n * maff / np.maximum(denoms, 1e-30) ** 2, 1e100)

            if not np.any(np.isfinite(vals)):
                continue

            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Fine refinement around best
            if best_val < cur_score - 1e-14:
                dw = (x_hi - x_lo) / G_coarse
                lo_f = max(0.0, best_x - 1.5 * dw)
                hi_f = min(x_hi, best_x + 1.5 * dw)
                if hi_f - lo_f > 1e-15:
                    pts_f = np.linspace(lo_f, hi_f, G_fine + 1)
                    ws_f = pts_f - v
                    aff_f = R_cur[start:end][None, :] + ws_f[:, None] * infl[None, :]
                    if 0 <= diag_offset < Ls:
                        aff_f[:, diag_offset] += ws_f * ws_f
                    maff_f = aff_f.max(axis=1)
                    maff_f = np.maximum(maff_f, max_const)
                    denoms_f = S + ws_f
                    vals_f = np.where(denoms_f > 1e-15,
                                      2.0 * n * maff_f / np.maximum(denoms_f, 1e-30) ** 2, 1e100)
                    if np.any(np.isfinite(vals_f)):
                        imin_f = int(np.argmin(vals_f))
                        if float(vals_f[imin_f]) < best_val:
                            best_x = float(pts_f[imin_f])
                            best_val = float(vals_f[imin_f])

            # Apply update if improvement
            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and best_x >= 0.0:
                w = best_x - v
                if abs(w) > 1e-16:
                    R_cur[start:end] += 2.0 * w * infl
                    if 0 <= diag_offset < Ls:
                        R_cur[start + diag_offset] += w * w
                    p_cur[i] = best_x
                    S += w
                    cur_score = best_val
                    improved = True

        # End of sweep: renormalize
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p_cur *= inv
            R_cur *= inv * inv
            S = 1.0

        # Periodic exact recompute
        if sweep % 5 == 0:
            R_cur = _fft_conv(p_cur)
            R_cur = np.maximum(R_cur, 0.0)
            cur_score = float(2.0 * n * R_cur.max())

        # Update best
        if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
            best_score = cur_score
            best_p = p_cur.copy()
            best_R = R_cur.copy()

        # Stalling detection and perturbation
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 2 and time.monotonic() < hard - 40.0:
                perturbation_count += 1
                # Choose perturbation type
                ptype = perturbation_count % 4
                if ptype == 0:
                    # Single Fourier mode
                    mode = int(rng.integers(1, max(2, n // 4)))
                    phase = float(rng.uniform(0, 2 * math.pi))
                    eps = 0.025 + 0.01 * min(flat, 12)
                    p_cur *= (1.0 + eps * np.sin(2.0 * math.pi * mode * idx_arr / n + phase))
                elif ptype == 1:
                    # Two Fourier modes (broadband)
                    mode1 = int(rng.integers(1, max(2, n // 6)))
                    mode2 = int(rng.integers(1, max(2, n // 3)))
                    phase1 = float(rng.uniform(0, 2 * math.pi))
                    phase2 = float(rng.uniform(0, 2 * math.pi))
                    eps = 0.02 + 0.008 * min(flat, 12)
                    p_cur *= (1.0 + eps * (
                        np.sin(2.0 * math.pi * mode1 * idx_arr / n + phase1) +
                        0.7 * np.sin(2.0 * math.pi * mode2 * idx_arr / n + phase2)
                    ))
                elif ptype == 2:
                    # Multiplicative Gaussian noise
                    eps = 0.03 + 0.01 * min(flat, 10)
                    p_cur *= (1.0 + eps * rng.standard_normal(n))
                else:
                    # Localized block perturbation
                    blk_start = int(rng.integers(0, n - 100))
                    blk_len = int(rng.integers(50, min(200, n - blk_start)))
                    eps = 0.05
                    p_cur[blk_start:blk_start + blk_len] *= (
                        1.0 + eps * rng.standard_normal(blk_len))

                p_cur = np.maximum(p_cur, 0.0)
                s_new = float(p_cur.sum())
                if s_new > 0.01:
                    p_cur /= s_new
                    S = 1.0
                    R_cur = _fft_conv(p_cur)
                    R_cur = np.maximum(R_cur, 0.0)
                    cur_score = float(2.0 * n * R_cur.max())
                    if math.isfinite(cur_score) and cur_score < best_score:
                        best_score = cur_score
                        best_p = p_cur.copy()
                        best_R = R_cur.copy()
                else:
                    p_cur = best_p.copy()
                    R_cur = best_R.copy()
                    S = 1.0
                    cur_score = best_score
                flat = 0

            if flat >= 50:

                p_cur = best_p * (1.0 + 0.06 * rng.standard_normal(n))
                p_cur = np.maximum(p_cur, 0.0)
                s_new = float(p_cur.sum())
                if s_new > 0.01:
                    p_cur /= s_new
                    S = 1.0
                    R_cur = _fft_conv(p_cur)
                    R_cur = np.maximum(R_cur, 0.0)
                    cur_score = float(2.0 * n * R_cur.max())
                flat = 0
