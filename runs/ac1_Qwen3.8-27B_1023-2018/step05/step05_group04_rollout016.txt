                    cur_peak = max(float(aff_seg.max()), max_const)
                    S = S + w
                    improved = True

        s_new = float(p.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            inv = 1.0 / s_new
            p *= inv
            c = _conv_self(p)
            new_peak = float(c.max())
            new_score = 2.0 * n * new_peak
            if math.isfinite(new_score) and new_score < best_score - 1e-15:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat_cnt = 0
        else:
            flat_cnt += 1
            if flat_cnt >= max_flat:
                if time.monotonic() < stop_time - 30.0:
                    kick_size = 0.015 + 0.03 * min(1.0, (flat_cnt - max_flat) / 10.0)
                    p = p * np.exp(rng.normal(0.0, kick_size, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_peak = float(c.max())
                    flat_cnt = 0
                else:
                    break
        sweep += 1

    # Final check
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = 2.0 * n * float(_conv_self(p).max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _resample(src, n_target):
    src = np.asarray(src, dtype=np.float64).ravel()
    n_target = int(max(1, n_target))
    m = src.size
    if m == 0:
        return np.full(n_target, 1.0 / n_target, dtype=np.float64)
    if m == n_target:
        out = src.copy()
    else:
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n_target, dtype=np.float64) + 0.5) / n_target, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n_target, 1.0 / n_target, dtype=np.float64)
    return out / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    hard = t0 + budget - 35.0
    if hard <= t0 + 10.0:
        hard = t0 + max(5.0, budget - 3.0)

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float('inf')

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()

    consider(parent)

    n_work = int(best_arr.size)
    if n_work < 128:
        n_work = 3072
    if n_work > 4096:
        n_work = 3072

    # Phase 1: LSE optimization
    lse_time = min(100.0, 0.10 * budget)
    p_work = _resample(best_arr, n_work)
    lse_end = t0 + lse_time
    if lse_end > t0 + 20.0:
        try:
            p2 = _lse_optimize(p_work, lse_end, consider=consider)
            if p2 is not None and p2.size == n_work and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p_work = p2 / float(p2.sum())
                consider(p_work)
        except Exception:
            pass

    # Phase 2: Main coordinate descent
    cd_end = hard - 100.0
    if cd_end > time.monotonic() + 60.0:
        p_start = _resample(best_arr, n_work)
        try:
            p_cd, sc_cd = _coord_descent(p_start, cd_end, consider=consider, seed=int(seed + 7777))
            if p_cd is not None and len(p_cd) > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                consider(p_cd)
        except Exception:
            pass

    # Phase 3: Multi-start perturbations
    rem = hard - 60.0 - time.monotonic()
    if rem > 90.0:
        rng_ms = np.random.default_rng(int(seed + 55555))
        n_starts = min(5, int(rem // 60))
        for si in range(n_starts):
            if time.monotonic() > hard - 65.0:
                break
            base = _resample(best_arr, n_work)
            # Different perturbation strategies
            if si % 3 == 0:
                # Smooth multiplicative noise
                noise = rng_ms.normal(0.0, 0.04, n_work)
                noise -= noise.mean()
                p_try = base * np.exp(noise)
            elif si % 3 == 1:
                # Shift the shape
                shift = rng_ms.uniform(-0.05, 0.05)
                x = np.arange(n_work) / n_work
                shift_x = np.clip(x + shift, 0, 1)
                resampled = np.interp(shift_x, np.arange(n_work) / n_work, base)
                p_try = resampled
            else:
                # Blend with a different template
                x = (np.arange(n_work, dtype=np.float64) + 0.5) / n_work
                for shape_name in ['tent', 'trapezoid']:
                    if si % 3 == 2 and si % 4 == 2:
                        tent = np.maximum(0.0, 1.0 - 2.0 * np.abs(x - 0.5))
                        s_t = float(tent.sum())
                        if s_t > 0:
                            tent /= s_t
                        p_try = 0.4 * tent + 0.6 * base
                    else:
                        # Trapezoid: high in middle 60%
                        mid_lo, mid_hi = 0.2, 0.8
                        trap = np.ones(n_work)
                        n_lo = int(0.2 * n_work)
                        n_hi = n_work - int(0.2 * n_work)
                        if n_lo > 0:
                            trap[:n_lo] = np.linspace(0.3, 1.0, n_lo)
                        if n_hi > n_lo:
                            trap[n_hi:] = np.linspace(1.0, 0.3, n_work - n_hi)
                        s_t = float(trap.sum())
                        if s_t > 0:
                            trap /= s_t
                        p_try = 0.3 * trap + 0.7 * base
                    break
            p_try = np.maximum(p_try, 0.0)
            s_t = float(p_try.sum())
            if s_t > 1e-10:
                p_try /= s_t
            consider(p_try)
            # Run short CD on this
            cd_sub_end = min(hard - 65.0, time.monotonic() + rem / (n_starts + 1) - 5.0)
            if cd_sub_end > time.monotonic() + 20.0:
                try:
                    p_sub, _ = _coord_descent(p_try, cd_sub_end, consider=consider, seed=int(seed + 66666 + si * 111), max_flat=3)
                    if p_sub is not None and len(p_sub) > 0:
                        consider(p_sub)
                except Exception:
                    pass

    # Phase 4: Final polish on best
    rem = hard - 10.0 - time.monotonic()
    if rem > 30.0:
        p_final = _resample(best_arr, n_work)
        try:
            p_f, _ = _coord_descent(p_final, hard - 10.0, consider=consider, seed=int(seed + 99999), max_flat=4)
            if p_f is not None and len(p_f) > 0:
                consider(p_f)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
