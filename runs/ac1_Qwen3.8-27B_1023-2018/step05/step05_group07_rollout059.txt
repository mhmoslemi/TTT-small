import numpy as np
import time


def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 1:
        return np.array([max(0.0, float(v[0]))]) if n == 1 else np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    return w / s if s > 0 else np.ones(n) / n


def _grad_smooth(a, beta):
    """Smooth log-sum-exp objective and its gradient for max-conv problem."""
    n = len(a)
    c = np.convolve(a, a)
    cmax = c.max()
    shifted = beta * (c - cmax)
    m = shifted.max()
    exp_v = np.exp(shifted - m)
    psum = exp_v.sum()
    p = exp_v / psum
    # objective
    obj = cmax + (m + np.log(psum)) / beta
    # gradient: grad[j] = sum_k p[k]*a[k-j] for valid indices
    grad = np.zeros(n)
    for j in range(n):
        klo = j
        khi = min(2 * n, j + n)
        if khi > klo:
            grad[j] = np.dot(p[klo:khi], a[:khi - klo])
    return obj, grad, c

def _polish_exact(a0, deadline, seed=0):

    best_M = float(c.max())
    best_a = a.copy()
    rng = np.random.default_rng(seed)

    for _ in range(min(2000, int(max(0, deadline - time.monotonic()) * 100))):
        if time.monotonic() > deadline:
            break
        c = np.convolve(a, a)
        M = float(c.max())
        if M < best_M:
            best_M = M
            best_a = a.copy()

        # Active lags (top few)
        thr = M * 0.9995
        active = np.where(c >= thr)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Subgradient from active lags
        grad = np.zeros(n)
        for k_star in active:
            ks = int(k_star)
            lo = max(0, ks - n + 1)
            hi = min(n, ks + 1)
            for j in range(lo, hi):
                idx = ks - j
                if 0 <= idx < n:
                    grad[j] += a[idx]

        gt = grad - grad.mean()
        gn = np.linalg.norm(gt)
        if gn < 1e-15:
            a *= np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            continue

        # Try several step sizes
        improved = False
        for alpha in [0.05 / gn, 0.02 / gn, 0.01 / gn, 0.005 / gn, 0.002 / gn]:
            an = a - alpha * gt
            an = np.maximum(an, 0)
            an_s = an.sum()

                continue

            cn = np.convolve(an, an)
            Mn = float(cn.max())

                a = an
                improved = True
                break
        if not improved:
            a *= np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 1e-12)
            a /= a.sum()

    return best_a
