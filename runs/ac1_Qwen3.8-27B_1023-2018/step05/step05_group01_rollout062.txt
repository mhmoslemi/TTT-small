    if float(p.sum()) < 1e-12:
        p = np.ones(256, dtype=np.float64)
    return p


def _conv_fft(a):
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    sz = next_fast_len(L)
    F = rfft(a, sz)
    c = irfft(F * F, sz)[:L]
    return np.maximum(c, 0.0)


def _score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_fft(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s)) if math.isfinite(m) else math.inf


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    return float(2.0 * n * m / (s * s)) if math.isfinite(m) else math.inf


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    ss = float(e.sum())
    if ss <= 0 or not math.isfinite(ss):
        return np.full(u.size, 1.0 / u.size)
    return e / ss


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        return src.copy()
    m = src.size
    xp = (np.arange(m) + 0.5) / m
    fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
    out = np.maximum(fp, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def _lse_optimize(p, stop_time, seed=42, consider=None):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 5

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)
        q = c[n - 1:n + n - 2]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 1e-300:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    stages = [
        (0.12, 1.0), (0.06, 1.0), (0.03, 1.0), (0.012, 1.2),
        (0.005, 1.3), (0.002, 1.4), (0.0008, 1.5), (0.0003, 1.5),
        (0.0001, 1.4), (0.00003, 1.3), (0.00001, 1.2), (0.000003, 1.1),
    ]

    start = time.monotonic()
    avail = max(0.0, float(stop_time) - start - 8.0)
    if avail <= 10.0:
        return p

    best_p = p.copy()
    best_sc = _score_fast(p)
    total_w = float(sum(w for _, w in stages))
    rng = np.random.default_rng(seed)
    cum = 0.0

    for si, (frac, wt) in enumerate(stages):
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 4.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)

        best_u = u.copy()
        f0, _ = lse_func(best_u, T)
        if not math.isfinite(f0):
            continue
        best_f = f0
        stale = 0
        max_rounds = min(40, int(max(5, 80.0 * wt / (1.0 + 0.5 * si / len(stages)))))

        while time.monotonic() < end_stage - 1.0 and stale < 3:
            maxit = 30
            opts = {"maxiter": maxit, "gtol": 1e-12, "ftol": 1e-16,
                    "maxls": 25, "maxcor": 25, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-14:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        best_u = best_u + rng.normal(0, 0.003, best_u.size)
            if getattr(res, "status", 1) == 0 and stale == 0:
                if time.monotonic() < end_stage - 2.0:
                    best_u = best_u + rng.normal(0, 0.002, best_u.size)

        p_cur = _softmax(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fast(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
            u = p_to_u(p)
        elif math.isfinite(sc_cur) and sc_cur < best_sc + 0.01:
            p = p_cur.copy()
            u = p_to_u(p)
        else:
            p = best_p.copy()
            u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _subgrad_step(p, n_iters, stop_time, consider=None, seed=777):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    rng = np.random.default_rng(seed)
    best_p = p.copy()
    best_sc = _score_fast(p)
    step_base = 0.002
    decay = 0.97

    for it in range(n_iters):
        if time.monotonic() > stop_time - 1.0:
            break
        c = _conv_fft(p)
        cmax = float(c.max())
        if not math.isfinite(cmax):
            break
        # Find all lags within 1% of max
        thresh = cmax * 0.995
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        # Subgradient: average gradient over active lags
        g = np.zeros(n)
        for k in active:
            k = int(k)
            # gradient of c[k] w.r.t. a[i]: a[k-i] + a[i-k] (valid)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            partner = k - idx
            g[idx] += a_val_look(p, partner)
            partner2 = idx - k
            valid2 = (partner2 >= 0) & (partner2 < n)
            idx2 = idx[valid2]
            partner2v = partner2[valid2]
            if len(idx2) > 0:
                g[idx2] += p[partner2v]
        # Normalize gradient
        gnorm = np.sqrt(float(np.dot(g, g)))
        if gnorm < 1e-300:
            break
        step = step_base / gnorm
        p_new = p - step * g
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if ss < 1e-300:
            break
        p_new /= ss
        sc_new = _score_fast(p_new)
        if math.isfinite(sc_new) and sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_p = p_new.copy()
            p = p_new.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass
        step *= decay

    return best_p


def a_val_look(p, idx):
    idx = np.asarray(idx)
    valid = (idx >= 0) & (idx < p.size)
    out = np.zeros(idx.size)
    if np.any(valid):
        out[valid] = p[idx[valid]]
    return out


def _coord_polish(p, stop_time, G=64, seed=42, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    Ltot = 2 * n - 1
    c = _conv_fft(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    max_sweeps = 500

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(30.0 * v, 50.0 * S / float(n), 0.02 * S, 1e-14)
            x_hi = min(x_hi, 10.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, S / float(n))

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.unique(np.concatenate((pts, [v])))

            diag_i = i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[valid, None] * infl[None, :]
                dval = base[diag_i] + w[valid] * infl[diag_i] + w[valid] * w[valid]
                aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                dv = denom[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if G > 8 and 0 < best_x < x_hi:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 32, dtype=np.float64)
                fine = np.unique(np.concatenate((fine, [best_x])))
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val - 1e-15:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= 12.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_fft(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-16:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 10.0:
                    sigma = 0.015 + 0.02 * min(flat, 8) / 8.0
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    else:
                        break
                    c = _conv_fft(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    flat = 0
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        sc = _score_fast(p)
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 3.0:
        hard = t0 + max(3.0, budget * 0.9)
    final_reserve = max(10.0, 0.05 * (hard - t0))
    phase_limit = hard - final_reserve

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    try:
        best_score = _score_exact(best_arr)
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(128) / 128.0
        best_score = _score_exact(best_arr)

    n_best = best_arr.size

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s_v = float(a.sum())
        if s_v < 0.01:
            return
        try:
            sc = _score_exact(a)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            a = a / s_v
            best_arr = a.copy()
            best_score = sc

    def run_lse_coord(init_p, end_time, sd, do_lse=True, do_sg=False, do_coord=True, G=64):
        nonlocal best_arr, best_score
        p = _resample(init_p, init_p.size)
        if p.size == 0 or float(p.sum()) <= 0:
            return
        p = p / float(p.sum())
        consider(p)

        if time.monotonic() > end_time - 15.0:
            return
        phase_len = max(0.0, float(end_time) - time.monotonic())

        if do_lse and phase_len > 40.0:
            lse_stop = min(float(end_time) - max(25.0, 0.35 * phase_len),
                           time.monotonic() + 0.6 * phase_len)
            if lse_stop > time.monotonic() + 15.0:
                try:
                    p2 = _lse_optimize(p, lse_stop, consider=consider, seed=sd)
                    if p2 is not None and p2.size == p.size and float(p2.sum()) > 0:
                        p = p2 / float(p2.sum())
                        consider(p)
                except Exception:
                    pass

        if do_sg and time.monotonic() < end_time - 30.0:
            sg_stop = min(float(end_time) - 25.0, time.monotonic() + 0.15 * phase_len)
            if sg_stop > time.monotonic() + 10.0:
                try:
                    p_sg = _subgrad_step(p, 200, sg_stop, consider=consider, seed=sd + 999)
                    if p_sg is not None and p_sg.size == p.size:
                        consider(p_sg)
                        # blend
                        p = 0.5 * p + 0.5 * p_sg
                        p = np.maximum(p, 0)
                        p = p / float(p.sum())
                        consider(p)
                except Exception:
                    pass

        rem = float(end_time) - time.monotonic()
        if do_coord and rem > 20.0:
            cstop = min(float(end_time) - 2.0, time.monotonic() + rem - 3.0)
            if cstop > time.monotonic() + 10.0:
                try:
                    p3, _ = _coord_polish(p, cstop, G=G, seed=sd, consider=consider)
                    if p3 is not None and p3.size == p.size and float(p3.sum()) > 0:
                        p = p3 / float(p3.sum())
                        consider(p)
                except Exception:
                    pass
        return p

    # === Phase 1: Main optimization at parent length ===
    p1_end = min(phase_limit, t0 + 0.50 * (phase_limit - t0))
    run_lse_coord(best_arr, p1_end, int(seed + 11), do_lse=True, do_sg=True, do_coord=True, G=72)

    # === Phase 2: Second pass with perturbation ===
    if time.monotonic() < phase_limit - 60.0:
        rng2 = np.random.default_rng(seed + 7777)
        pert = best_arr * np.exp(rng2.normal(0, 0.03, best_arr.size))
        pert = np.maximum(pert, 0.0)
        ps = float(pert.sum())
        if ps > 0:
            pert /= ps
        sub_end = min(phase_limit - 40.0, time.monotonic() + 0.30 * (phase_limit - time.monotonic()))
        if sub_end > time.monotonic() + 30.0:
            run_lse_coord(pert, sub_end, int(seed + 3000), do_lse=True, do_sg=False, do_coord=True, G=64)

    # === Phase 3: Asymmetric initialization ===
    if time.monotonic() < phase_limit - 70.0:
        rng3 = np.random.default_rng(seed + 55555)
        n = n_best
        x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
        # Try a ramp-based init
        for ramp_seed in range(2):
            if time.monotonic() > phase_limit - 50.0:
                break
            if ramp_seed == 0:
                ramp = np.exp(2.5 * (x - 0.5))  # increasing
            else:
                ramp = np.exp(-3.0 * (x - 0.7))  # peak off-center
            init = best_arr * ramp
            init = np.maximum(init, 0)
            is_ = float(init.sum())
            if is_ > 0:
                init /= is_
            sub_end = min(phase_limit - 35.0, time.monotonic() + 0.20 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 25.0:
                run_lse_coord(init, sub_end, int(seed + 9000 + ramp_seed * 333),
                              do_lse=True, do_sg=False, do_coord=True, G=56)

    # === Phase 4: Multi-start with different random inits ===
    if time.monotonic() < phase_limit - 80.0:
        rng4 = np.random.default_rng(seed + 88888)
        for ms in range(3):
            if time.monotonic() > phase_limit - 50.0:
                break
            # Structured perturbation: add a sinusoidal component
            freq = 1.0 + 3.0 * rng4.random()
            phase = 2 * math.pi * rng4.random()
            mod = 1.0 + 0.3 * np.sin(freq * 2 * math.pi * np.arange(n_best) / n_best + phase)
            init = best_arr * mod
            init = np.maximum(init, 0)
            is_ = float(init.sum())
            if is_ > 0:
                init /= is_
            sub_end = min(phase_limit - 40.0, time.monotonic() + 0.15 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 20.0:
                run_lse_coord(init, sub_end, int(seed + 70000 + ms * 777),
                              do_lse=False, do_sg=True, do_coord=True, G=48)

    # === Phase 5: Final intensive polish ===
    final_end = min(hard - 15.0, t0 + budget - 10.0)
    if final_end > time.monotonic() + 20.0 and best_arr.size > 0:
        nb = int(best_arr.size)
        rem = final_end - time.monotonic()

        # Short LSE with very low T
        if rem > 60.0 and nb <= 6000:
            lstop = min(final_end - 40.0, time.monotonic() + 0.35 * rem)
            if lstop > time.monotonic() + 20.0:
                try:
                    p2 = _lse_optimize(best_arr, lstop, consider=consider, seed=int(seed + 999999))
                    if p2 is not None and p2.size == nb and float(p2.sum()) > 0:
                        p2 = p2 / float(p2.sum())
                        consider(p2)
                except Exception:
                    pass

        # Long coordinate polish
        cstop = min(final_end - 5.0, time.monotonic() + max(25.0, 0.55 * (final_end - time.monotonic())))
        if cstop > time.monotonic() + 15.0:
            try:
                p3, _ = _coord_polish(best_arr, cstop, G=80, seed=int(seed + 1234567), consider=consider)
                if p3 is not None and p3.size == nb and float(p3.sum()) > 0:
                    p3 = p3 / float(p3.sum())
                    consider(p3)
            except Exception:
                pass

        # One more short coord pass
        cstop2 = min(final_end - 2.0, time.monotonic() + max(10.0, 0.1 * (final_end - time.monotonic())))
        if cstop2 > time.monotonic() + 8.0:
            try:
                p4, _ = _coord_polish(best_arr, cstop2, G=96, seed=int(seed + 42424242), consider=consider)
                if p4 is not None and p4.size == nb and float(p4.sum()) > 0:
                    p4 = p4 / float(p4.sum())
                    consider(p4)
            except Exception:
                pass

    # === Final validation ===
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(128, 1.0 / 128.0)
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = evaluate_sequence(ls)
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
