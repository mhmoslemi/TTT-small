import numpy as np
import time
from scipy.optimize import linprog


def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)


def _conv_mat(n, a):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        A[k, jlo:jhi] = a[k - np.arange(jlo, jhi)]
    return A


def _lp_br(a, M, tl=2.0):
    n = len(a)
    A = _conv_mat(n, a)
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': tl,
                               'primal_feasibility_tolerance': 1e-12,
                               'dual_feasibility_tolerance': 1e-12})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def _lp_mint(a, tl=2.0):
    n = len(a)
    A = _conv_mat(n, a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.hstack([np.ones(n), 0.0])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',

                               'dual_feasibility_tolerance': 1e-12})
        if res.status == 0 and res.x is not None:
