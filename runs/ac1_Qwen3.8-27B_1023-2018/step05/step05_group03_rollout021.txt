The gradient:
∂f/∂u_i = 2n * ∂LSE/∂a_i * ∂a_i/∂u_i

∂a/∂u: a_i = exp(u_i)/S_softmax, so ∂a_j/∂u_i = a_j(δ_{ji} - a_i)

∂LSE/∂a_i = (1/T) * sum_k w_k * ∂c_k/∂a_i
where w_k = exp((c_k - cmax)/T) / sum(exp((c_k-cmax)/T))
and ∂c_k/∂a_i = 2*a_{k-i} (if 0≤k-i≤n-1 and k≠2i) or 2*a_i (if k=2i)

This is O(n^2) to compute the gradient, which for n=3072 is about 9.4M operations. That's fine per gradient evaluation.

The issue is that each gradient evaluation requires computing the full convolution (O(n log n) via FFT) and then the weighted sum (O(n^2) in the worst case, or O(n) if we use the structure).

Actually, ∂LSE/∂a_i = (1/T) * sum_{k: 0≤|k-i|≤n-1} w_k * 2 * a_{k-i}

This is a cross-correlation of w with a, evaluated at position i. So it's a convolution! We can compute all ∂LSE/∂a_i simultaneously via FFT:

h = (2/T) * conv(w, a) [cross-correlation]
Then ∂f/∂u_i = h[i] * (a[i] - a[i]^2) - a[i] * sum_j h[j] * a[j]
= a[i] * (h[i] - sum_j h[j] * a[j])

                a_cur = a_cur / S
                S = 1.0
                c_cur = full_conv(a_cur)
                cur_sc = score_from_conv(c_cur)

            if improved:
                if cur_sc < best_sc_local - 1e-14:
                    best_sc_local = cur_sc
                    best_local = a_cur.copy()
                    if cur_sc < best_score - 1e-14:
                        best_a = a_cur.copy()
                        best_score = cur_sc
                flat = 0
            else:
                flat += 1
                if flat >= 5:
                    # Perturbation to escape
                    sigma = min(0.2, 0.02 * flat)
                    noise = cd_rng.normal(0, sigma, n_loc)
                    a_pert = a_cur * np.exp(noise)
                    sp = float(a_pert.sum())
                    if sp > 1e-10:
                        a_pert /= sp
                    c_pert = full_conv(a_pert)
                    sc_pert = score_from_conv(c_pert)
                    if sc_pert < best_sc_local + 0.05:
                        a_cur = a_pert
                        c_cur = c_pert
                        S = 1.0
                        cur_sc = sc_pert
                    else:
                        a_cur = best_local.copy()
                        c_cur = full_conv(a_cur)
                        cur_sc = best_sc_local
                    flat = 0
                elif flat >= 30:
                    break

            sweep += 1

        return best_local, best_sc_local

    # LSE-LBFGS optimization
    def lse_lbfgs(a_cur, stop, T_frac, max_iter_chunk=12):
        nonlocal best_a, best_score
        n_loc = a_cur.size
        a_work = a_cur.copy()
        a_work = a_work / a_work.sum()

        conv_len = L
        csize = fft_size

        c_work = full_conv(a_work)
        cmax0 = float(c_work.max())
        best_f = 2.0 * n_loc * cmax0

        u = np.log(np.maximum(a_work, 1e-30))
        u -= u.mean()

        def lse_obj(uu, T):
            uu = np.asarray(uu, dtype=np.float64).ravel()
            if uu.size != n_loc:
                return 1e100, np.zeros(n_loc)
            if not np.all(np.isfinite(uu)):
                uu = np.where(np.isfinite(uu), uu, 0.0)
            uu -= uu.mean()
            m = uu.max()
            e = np.exp(uu - m)
            s = e.sum()
            if not math.isfinite(s) or s < 1e-300:
                aa = np.full(n_loc, 1.0 / n_loc)
            else:
                aa = e / s
            aa = np.maximum(aa, 0.0)
            sa = aa.sum()
            if sa < 1e-300:
                aa = np.full(n_loc, 1.0 / n_loc)
            else:
                aa /= sa

            F = rfft(aa, csize)
            cv = irfft(F * F, csize)[:conv_len]
            np.maximum(cv, 0.0, out=cv)
            cmax = float(cv.max())
            if not math.isfinite(cmax):
                return 1e100, np.zeros(n_loc)

            scale = (cv - cmax) / T
            exps = np.exp(np.minimum(scale, 50.0))
            se = exps.sum()
            if not math.isfinite(se) or se < 1e-300:
                f = 2.0 * n_loc * cmax
                w_cv = np.zeros(L)
                w_cv[int(np.argmax(cv))] = 1.0
            else:
                f = 2.0 * n_loc * (cmax + T * math.log(se / L))
                w_cv = exps / se

            # h[i] = sum_j a[j] * w_cv[j+i] (cross-correlation)
            # h = conv(a, reverse(w))[n-1:2n-1][::-1]
            w_rev = w_cv[::-1]
            F2 = rfft(aa, fft_size * 2)
            G2 = rfft(w_rev, fft_size * 2)
            h_full = irfft(F2 * np.conj(G2), fft_size * 2)
            h = h_full[n_loc - 1:2 * n_loc - 1][::-1]
            h = np.real(h[:n_loc])
            grad_a = 4.0 * n_loc * h
            gd = float(np.dot(grad_a, aa))
            grad_u = aa * (grad_a - gd)
            grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
            return float(f), grad_u

        from scipy.optimize import minimize as sp_minimize
        best_u = u.copy()
        best_f_local = best_f
        stale = 0

        while time.monotonic() < stop - 3:
            c_work = full_conv(_softmax_arr(best_u, n_loc))
            cmax_w = float(c_work.max())
            T = max(1e-10, T_frac * cmax_w)
            T_list = [T, T * 0.4, T * 0.15]

            for T_try in T_list:
                if time.monotonic() > stop - 3:
                    break
                try:
                    res = sp_minimize(
                        lse_obj, best_u, args=(T_try,),
                        method='L-BFGS-B', jac=True,
                        options={'maxiter': max_iter_chunk, 'gtol': 1e-10,
                                 'ftol': 1e-14, 'maxls': 16, 'maxcor': 12,
                                 'disp': False, 'maxdisp': 0}
                    )
                except Exception:
                    break
                if res.x is not None:
                    rx = np.asarray(res.x, dtype=np.float64).ravel()
                    if rx.size == n_loc and np.all(np.isfinite(rx)):
                        fv = float(res.fun) if hasattr(res, 'fun') else 1e100
                        if math.isfinite(fv) and fv < best_f_local - 1e-12:
                            # Convert back to a
                            aa = _softmax_arr(rx, n_loc)
                            aa = aa / aa.sum()
                            cv2 = full_conv(aa)
                            sc2 = score_from_conv(cv2)
                            if math.isfinite(sc2) and sc2 < best_f_local - 1e-12:
                                best_f_local = sc2
                                best_u = rx.copy()
                                a_work = aa.copy()
                                stale = 0
                                if sc2 < best_score - 1e-14:
                                    best_a = aa.copy()
                                    best_score = sc2
                                break
                        else:
                            stale += 1
                if stale >= 3:
                    break

            if stale >= 3:
                break
            T_frac *= 0.6

        aa = _softmax_arr(best_u, n_loc)
        aa = aa / aa.sum()
        return aa, best_f_local

    def _softmax_arr(uu, sz):
        uu = np.asarray(uu, dtype=np.float64).ravel()[:sz]
        uu -= uu.max()
        e = np.exp(uu)
        s = e.sum()
        if s < 1e-300:
            return np.full(sz, 1.0 / sz)
        return e / s

    # Main optimization loop
    # Phase 2: Alternating coord descent and LSE
    now = time.monotonic()
    p2_end = min(hard - 120, t0 + budget * 0.65)
    T_frac = 0.012
    cd_seed = 1000

    while now < p2_end:
        # Coordinate descent block
        cd_stop = min(p2_end, now + min(120.0, 0.5 * (p2_end - now)))
        if cd_stop - now > 15:
            a_cur = best_a.copy()
            a_cur = a_cur / a_cur.sum()
            c_cur = full_conv(a_cur)
            a_cur, cs = coord_sweeps(a_cur, c_cur, cd_stop,
                                     n_sweeps_max=3000, G=52, seed_cd=cd_seed)
            cd_seed += 7919
            if cs < best_score - 1e-14:
                best_a = a_cur.copy()
                best_score = cs
            now = time.monotonic()

        # LSE block
        lse_stop = min(p2_end, now + 45.0)
        if lse_stop - now > 20:
            a_cur = best_a.copy()
            a_cur, lsp = lse_lbfgs(a_cur, lse_stop, T_frac)
            if lsp < best_score - 1e-14:
                best_a = a_cur.copy()
                best_score = lsp
            T_frac = max(0.002, T_frac * 0.7)
            now = time.monotonic()

    # Phase 3: Fine LSE polish
    p3_end = min(hard - 60, now + min(200.0, 0.2 * budget))
    if p3_end - now > 20:
        a_cur = best_a.copy()
        a_cur = a_cur / a_cur.sum()
        for Tf in (0.005, 0.002, 0.0008, 0.0003):
            if time.monotonic() > p3_end - 10:
                break
            a_cur, _ = lse_lbfgs(a_cur, min(p3_end, time.monotonic() + 50), Tf, max_iter_chunk=15)

    # Phase 4: Final coordinate polish
    final_stop = min(hard - 5, time.monotonic() + max(20.0, 0.12 * budget))
    if final_stop - time.monotonic() > 15:
        a_cur = best_a.copy()
        a_cur = a_cur / a_cur.sum()
        c_cur = full_conv(a_cur)
        a_cur, cs = coord_sweeps(a_cur, c_cur, final_stop,
                                 n_sweeps_max=5000, G=64, seed_cd=cd_seed + 999)
        if cs < best_score - 1e-14:
            best_a = a_cur.copy()
            best_score = cs

    # Final exact evaluation and return
    out_a = best_a.copy()
    out_a = np.where(np.isfinite(out_a), out_a, 0.0)
    out_a = np.clip(out_a, 0.0, 1000.0)
    out_s = float(out_a.sum())
    if out_a.size == 0 or not math.isfinite(out_s) or out_s < 0.01:
        out_a = np.ones(64) / 64.0
        out_s = out_a.sum()
    out_a = out_a / out_s

    # Verify with exact evaluator
    try:
        result = [float(x) for x in out_a]
        final_sc = float(evaluate_sequence(result))
        if not math.isfinite(final_sc):
            # Fallback to parent
            result = [float(x) for x in p / ps]
            final_sc2 = float(evaluate_sequence(result))
            if not math.isfinite(final_sc2):
                return [1.0]
            return result
        return result
    except Exception:
        return [1.0]
