    s = float(a.sum())
    if s < 1e-12:
        return float('inf')
    a = a / s
    c = _conv_self(a)
    return _score_norm(c, n)


def _lse_phase(a_init, stop_time, consider):
    n = a_init.size
    a = np.maximum(a_init.copy(), 1e-30)
    a = a / a.sum()
    u = np.log(a)
    u = u - u.mean()

    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)

    def _cfast(pp):
        F = rfft(pp, conv_size)
        c = irfft(F * F, conv_size)[:2 * n - 1]
        return c

    def _grad_lse(pp, w):
        # g_i = (2/T) * sum_j w[i+j] * a[j]
        # Compute via: pad w to length 4n, correlate with pp
        w_pad = np.zeros(4 * n, dtype=np.float64)
        w_pad[:2 * n - 1] = w
        # cross-correlation: (w_pad * pp)[n-1 : n-1+n]
        cp_size = next_fast_len(5 * n)
        FW = rfft(w_pad, cp_size)
        FP = rfft(pp, cp_size)
        cc = irfft(FW * FP, cp_size)
        # cc[k] = sum_j w_pad[k-j]*pp[j], we want sum_j w[i+j]*pp[j] = cc[2n-1-i+... ]
        # Actually: sum_j w[i+j]*pp[j]. Let m = i + j - (i) = j. 
        # Define h[k] = w[k] for k in [0,2n-2]. Then sum_j h[i+j]*pp[j]
        # This is the cross-correlation of h and pp at offset i: (h * pp_reversed)
        # Simpler: use direct computation for n=3072 (O(n^2) ~ 9.4M, fast)
        # Or: g[i] = sum_{j=0}^{n-1} w[i+j]*pp[j]
        # Build via: g = correlate(w_full, pp) where w_full[k]=w[k] for 0<=k<2n-1
        # Actually let's just do it with a loop-free approach:
        # g[i] = sum_j w[i+j]*pp[j]
        # Create matrix: M[i,j] = w[i+j] for i in [0,n-1], j in [0,n-1]
        # M = w[0:n-1 : n-1 : n-1]... no, let's use the FFT approach properly.
        # sum_j w[i+j] * pp[j] = inner product of pp and rolling window of w
        # This equals the (n-1+i)-th element of convolution of reversed(pp) and w
        pp_rev = pp[::-1]
        conv_rw = np.convolve(pp_rev, w_pad[:2*n-1], mode='full')
        # conv_rw[k] = sum_j pp_rev[j] * w[k-j] = sum_j pp[n-1-j]*w[k-j]
        # We want sum_j w[i+j]*pp[j]. Let l = n-1-j, then j = n-1-l:
        # sum_l w[i+n-1-l]*pp[n-1-l] = sum_l pp[l]*w[i+n-1-l]
        # This is the cross-correlation of pp and w at lag (i+n-1):
        # = conv(pp, w_rev)[i+n-1] where w_rev[k]=w[2n-2-k]
        # Simpler: just use the fact that g[i] = (w * pp_flip)[i] in the right window
        # Actually the easiest: g = w[i:i+n] dot pp for each i
        # Use: g[i] = sum_j w[i+j] * pp[j]
        # Let's build the Toeplitz-like matrix
        # For n=3072, this is a 3072x3072 matrix multiply: g = T @ pp
        # T[i,j] = w[i+j]
        idx_i = np.arange(n)[:, None]
        idx_j = np.arange(n)[None, :]
        T = w[idx_i + idx_j]
        g = T @ pp
        return g

    stages = [
        (0.08, 0.15),
        (0.025, 0.15),
        (0.007, 0.20),
        (0.002, 0.25),
        (0.0006, 0.25),
    ]

    best_a_local = a.copy()
    best_sc_local = _score_norm(_conv_self(a), n)

    total_w = sum(w for _, w in stages)
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        cum += wt
        end_stage = min(stop_time - 3.0, (stop_time - (stop_time - (stop_time - time.monotonic() + time.monotonic() - (stop_time - stop_time + time.monotonic())))) * 0)  # placeholder
        # Better: distribute time proportionally
        elapsed_frac = (time.monotonic() - (stop_time - (stop_time - time.monotonic()))) / (stop_time - time.monotonic())
        end_stage = min(stop_time - 3.0, time.monotonic() + (stop_time - time.monotonic()) * (cum - wt) / total_w if False else stop_time - 3.0 - (stop_time - time.monotonic()) * (total_w - cum) / total_w)
        if end_stage <= time.monotonic() + 5.0:
            continue

        c = _cfast(a)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)

        def lse_func(uu):
            m = float(np.max(uu))
            e = np.exp(uu - m)
            s = e.sum()
            if s <= 0 or not math.isfinite(s):
                return 1e100, np.zeros_like(uu)
            pp = e / s
            c = _cfast(pp)
            cmax_l = float(c.max())
            if cmax_l <= 0 or not math.isfinite(cmax_l):
                return 1e100, np.zeros_like(uu)
            scaled = (c - cmax_l) / T
            exps = np.exp(scaled)
            se = float(exps.sum())
            if se <= 0 or not math.isfinite(se):
                w = np.zeros(c.size)
                w[int(np.argmax(c))] = 1.0
            else:
                w = exps / se
            f = cmax_l + T * float(np.log(se / c.size))
            gd = _grad_lse(pp, w) / T
            # softmax gradient
            grad = pp * (gd - float(np.dot(pp, gd)))
            grad = np.where(np.isfinite(grad), grad, 0.0)
            return f, grad

        best_u = u.copy()
        stale = 0
        while time.monotonic() < end_stage - 2.0:
            try:
                f0, _ = lse_func(best_u)
            except Exception:
                break
            try:
                res = minimize(lse_func, best_u, method='L-BFGS-B', jac=True,
                               options={'maxiter': 40, 'gtol': 1e-12, 'ftol': 1e-16,
                                        'maxls': 20, 'maxdisp': 0})
            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval):
                    best_u = res.x.copy()
                    pp = np.exp(best_u - best_u.max())
                    pp /= pp.sum()
                    sc = _score_norm(_cfast(pp), n)
                    if math.isfinite(sc) and sc < best_sc_local:
                        best_sc_local = sc
                        best_a_local = pp.copy()
                        a = pp.copy()
                        u = np.log(np.maximum(pp, 1e-30))
                        u -= u.mean()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if getattr(res, 'status', 1) == 0:
                break

        consider(best_a_local)

    return best_a_local


def _cd_phase(a_init, stop_time, seed, consider):
    n = int(a_init.size)
    a = np.maximum(np.asarray(a_init, dtype=np.float64).ravel(), 0.0).copy()
    s_sum = float(a.sum())
    if s_sum < 1e-12:
        return a
    a = a / s_sum
    c = _conv_self(a)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_a = a.copy()
    Ltot = 2 * n - 1

    rng = np.random.default_rng(int(seed) + 42)
    GR = (math.sqrt(5.0) - 1.0) / 2.0
    G = 48

    def eval_1d(w, base, infl, diag_idx, max_const):
        s_new = 1.0 + w
        if s_new <= 1e-15:
            return float('inf')
        aff = base + w * infl
        if diag_idx is not None and 0 <= diag_idx < len(base):
            dval = base[diag_idx] + w * infl[diag_idx] + w * w
            if dval > float(aff.max()):
                aff = aff.copy()
                aff[diag_idx] = dval
        m = max(float(aff.max()), max_const)
        return 2.0 * n * m / (s_new * s_new)

    flat = 0
    sweep = 0
    while sweep < 500000 and time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)

        for ii in order:
            i = int(ii)
            if time.monotonic() > stop_time - 1.0:
                break

            v = float(a[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end].copy()
            infl = 2.0 * a[:Ls]
            # diag: k = 2i, offset in base = 2i - start = 2i - i = i
            diag_off = i
            if diag_off < Ls:
                diag_idx = diag_off
            else:
                diag_idx = None

            # Unmodified max
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < Ltot:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

            # Search range
            x_lo = 0.0
            x_hi = max(50.0 * v, 40.0 / n, 0.005)
            x_hi = min(x_hi, 0.8)

            if x_hi <= x_lo:
                continue

            # Grid search
            pts = np.linspace(x_lo, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                ins = np.searchsorted(pts, v)
                pts = np.insert(pts, ins, v)

            ws = pts - v
            aff_mat = base[None, :] + ws[:, None] * infl[None, :]
            if diag_idx is not None:
                aff_mat[:, diag_idx] += ws * ws
            maff = np.maximum(aff_mat.max(axis=1), max_const)
            s_new = s_sum + ws  # but we normalized to sum=1, so s_sum~1
            s_new = 1.0 + ws
            valid = s_new > 1e-12
            scores = np.full(pts.shape, float('inf'), dtype=np.float64)
            if np.any(valid):
                scores[valid] = 2.0 * n * maff[valid] / (s_new[valid] ** 2)
            scores = np.where(np.isfinite(scores), scores, 1e100)

            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            # Golden section refinement
            lo = float(pts[max(0, imin - 1)])
            hi = float(pts[min(len(pts) - 1, imin + 1)])
            if hi - lo < 1e-14:
                lo = max(0.0, best_x - 1e-6)
                hi = best_x + 1e-6
            if hi <= lo:
                lo, hi = 0.0, x_hi

            a_gs, b_gs = lo, hi
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)
            f_c = eval_1d(c_gs - v, base, infl, diag_idx, max_const)
            f_d = eval_1d(d_gs - v, base, infl, diag_idx, max_const)
            for _ in range(28):
                if b_gs - a_gs < 1e-13 * max(1.0, abs(best_x)):
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = eval_1d(c_gs - v, base, infl, diag_idx, max_const)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_d = eval_1d(d_gs - v, base, infl, diag_idx, max_const)
            x_mid = (a_gs + b_gs) / 2.0
            f_mid = eval_1d(x_mid - v, base, infl, diag_idx, max_const)
            if f_mid < best_val:
                best_x = x_mid
                best_val = f_mid

            # Check endpoints
            for xp in (x_lo, x_hi):
                fp = eval_1d(xp - v, base, infl, diag_idx, max_const)
                if fp < best_val:
                    best_x = xp
                    best_val = fp

            if math.isfinite(best_val) and best_val < cur_score - 1e-15 and 0.0 <= best_x:
                w = best_x - v
                if abs(w) > 1e-14:
                    a[i] = best_x
                    aff_seg = base + w * infl
                    if diag_idx is not None:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag_idx] += w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    c[start:end] = aff_seg
                    cur_score = 2.0 * n * new_max / (1.0 + w) ** 2
                    improved = True

        # Renormalize
        s_new = float(a.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            inv = 1.0 / s_new
            a *= inv
            c = _conv_self(a)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-15:
                best_score = new_score
                best_a = a.copy()
                consider(best_a)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 20.0:
                    kick_size = 0.015 + 0.03 * min(1.0, flat / 30.0)
                    a = a * np.exp(rng.normal(0.0, kick_size, n))
                    a = np.maximum(a, 0.0)
                    ss = float(a.sum())
                    if ss > 0:
                        a /= ss
                    c = _conv_self(a)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break

        sweep += 1

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = min(float(budget_s), 1300.0)
    hard = t0 + budget - 12.0
    if hard <= t0 + 5.0:
        hard = t0 + max(3.0, budget - 2.0)

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(3072, dtype=np.float64)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    parent = np.clip(parent, 0.0, 1000.0)
    ps = float(parent.sum())
    if ps < 0.01:
        parent = np.ones(3072, dtype=np.float64) / 3072.0
        ps = 1.0
    n = int(parent.size)
    a = parent / ps

    best_a = a.copy()
    best_sc = _score_norm(_conv_self(a), n)

    def consider(x):
        nonlocal best_a, best_sc
        try:
            xx = np.maximum(np.asarray(x, dtype=np.float64).ravel(), 0.0)
        except Exception:
            return
        s = float(xx.sum())
        if s < 1e-10:
            return
        xx = xx / s
        c = _conv_self(xx)
        sc = _score_norm(c, n)
        if math.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = xx.copy()

    consider(a)

    # Phase 1: LSE optimization
    p1_dur = min(180.0, 0.2 * budget)
    p1_end = t0 + p1_dur
    if p1_end > t0 + 30.0:
        try:
            a2 = _lse_phase(a, p1_end, consider)
            if a2 is not None and a2.size == n and np.all(np.isfinite(a2)):
                a = np.maximum(a2, 0.0)
                ss = float(a.sum())
                if ss > 1e-10:
                    a /= ss
                consider(a)
        except Exception:
            pass

    # Phase 2: Long CD
    cd_end = hard - 8.0
    if cd_end > time.monotonic() + 30.0:
        try:
            a3 = _cd_phase(a, cd_end, int(seed) + 7777, consider)
            if a3 is not None and a3.size == n and np.all(np.isfinite(a3)):
                a = np.maximum(a3, 0.0)
                ss = float(a.sum())
                if ss > 1e-10:
                    a /= ss
                consider(a)
        except Exception:
            pass

    # Phase 3: If time remains, CD from a slightly different start
    rem = hard - 10.0 - time.monotonic()
    if rem > 60.0:
        try:
            rng2 = np.random.default_rng(int(seed) + 12345)
            a_alt = a * np.exp(rng2.normal(0.0, 0.01, n))
            a_alt = np.maximum(a_alt, 0.0)
            a_alt /= a_alt.sum()
            consider(a_alt)
            cd_end2 = min(hard - 8.0, time.monotonic() + rem * 0.8)
            if cd_end2 > time.monotonic() + 20.0:
                a4 = _cd_phase(a_alt, cd_end2, int(seed) + 9999, consider)
                if a4 is not None and a4.size == n and np.all(np.isfinite(a4)):
                    consider(a4)
        except Exception:
            pass

    # Final: one more CD squeeze on the overall best
    rem = hard - 10.0 - time.monotonic()
    if rem > 40.0:
        try:
            a5 = _cd_phase(best_a, hard - 8.0, int(seed) + 11111, consider)
            if a5 is not None:
                consider(a5)
        except Exception:
            pass

    # Final validation and return
    try:
        out = np.asarray(best_a, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.ones(64, dtype=np.float64) / 64.0
        result = [float(x) for x in out]
        sc = float(evaluate_sequence(result))
        if not math.isfinite(sc):
            result = [float(x) for x in parent]
            sc2 = float(evaluate_sequence(result))
            if not math.isfinite(sc2):
                return [1.0]
        return result
    except Exception:
        return [1.0]
