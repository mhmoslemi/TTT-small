        active = np.arange(c.size)
    else:

        active = np.where(c >= threshold)[0]

            top_idx = np.argsort(c[active])[::-1][:K]
            active = active[top_idx]

    if len(active) == 0:
        return None

    # Build constraint matrix: for each active lag k, row is [f[k-0], f[k-1], ..., f[k-(n-1)]]
    # (f*g)[k] = sum_j g[j] * f[k-j] (with f[m]=0 if m<0 or m>=n)
    rows = []
    for k in active:
        coeffs = np.zeros(n)
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            coeffs[j] = f[k - j]
        rows.append(coeffs)
    A = np.array(rows)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.CBC, time_limit=timeout)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK, time_limit=timeout)
        except Exception:
            return None
    if g.value is not None:
        gv = np.asarray(g.value, dtype=np.float64).ravel()
        if gv.size == n and np.all(np.isfinite(gv)) and gv.sum() > 1e-10:
            return np.maximum(gv, 0.0)
    return None


def _coord_polish(p, stop_time, max_sweeps=50000, G=None, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    if G is None:
        if n <= 500:
            G = 60
        elif n <= 1000:
            G = 50
        elif n <= 2000:
            G = 40
        else:
            G = 32
    G = int(max(16, G))

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.4:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.25 or S_total < 0.75:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls].copy()

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            # Newton initial guess
            k_diag = 2 * i
            if 0 <= k_diag < Ltot:
                cd = c[k_diag]
                ad = p[i]
                if S - ad > 1e-15:
                    newton_x = (cd - ad * S) / (S - ad)
                    if 0 <= newton_x <= 10 * S:
                        x_hi = max(20.0 * v, 60.0 * S / n, 0.05 * S, 1e-12)
                        x_hi = min(x_hi, max(10.0 * S, newton_x * 1.5 + 0.01 * S))
                    else:
                        x_hi = max(20.0 * v, 60.0 * S / n, 0.05 * S, 1e-12)
                        x_hi = min(x_hi, 10.0 * S)
                else:
                    x_hi = max(20.0 * v, 60.0 * S / n, 0.05 * S, 1e-12)
                    x_hi = min(x_hi, 10.0 * S)
            else:
                x_hi = max(20.0 * v, 60.0 * S / n, 0.05 * S, 1e-12)
                x_hi = min(x_hi, 10.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, S / n)

            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 <= v <= x_hi:
                pts = np.unique(np.concatenate((pts, [v])))

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                di = 0
                if di < Ls:
                    dval = base[di] + w * infl[di] + w * w
                    maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Refined search
            if G > 1:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 3.0 * dw)
                hi = min(x_hi, best_x + 3.0 * dw)
                if hi <= lo:
                    hi = lo + 1e-14
                fine_pts = np.linspace(lo, hi, 50)
                if 0.0 <= best_x <= hi:
                    fine_pts = np.unique(np.concatenate((fine_pts, [best_x])))
                vals2 = eval_xs(fine_pts)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine_pts[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2

                # Golden section
                gr = (math.sqrt(5.0) - 1.0) / 2.0
                g_lo = max(0.0, best_x - dw)
                g_hi = min(x_hi, best_x + dw)
                if g_hi - g_lo > 1e-16:
                    a_gs = g_hi - gr * (g_hi - g_lo)
                    b_gs = g_lo + gr * (g_hi - g_lo)
                    fa = float(eval_xs(np.array([a_gs]))[0])
                    fb = float(eval_xs(np.array([b_gs]))[0])
                    for _ in range(16):
                        if not math.isfinite(fa):
                            fa = 1e100
                        if not math.isfinite(fb):
                            fb = 1e100
                        if fa < fb:
                            g_hi = b_gs
                            b_gs = a_gs
                            fb = fa
                            a_gs = g_hi - gr * (g_hi - g_lo)
                            fa = float(eval_xs(np.array([a_gs]))[0])
                        else:
                            g_lo = a_gs
                            a_gs = b_gs
                            fa = fb
                            b_gs = g_lo + gr * (g_hi - g_lo)
                            fb = float(eval_xs(np.array([b_gs]))[0])
                        if g_hi - g_lo < 1e-15:
                            break
                    gs_x = (g_lo + g_hi) / 2.0
                    gv = float(eval_xs(np.array([gs_x]))[0])
                    if math.isfinite(gv) and gv < best_val:
                        best_x = gs_x
                        best_val = gv

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 10.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        np.maximum(c, 0.0, out=c)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-16:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 6.0:
                    try:
                        noise_amt = 0.02 + 0.01 * math.exp(-flat * 0.15)
                        p = p * np.exp(rng.normal(0.0, noise_amt, n))
                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss > 0.0:
                            p /= ss
                        c = _conv_self(p)
                        S_total = 1.0
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider is not None:
                                consider(best_p)
                        flat = 0
                    except Exception:
                        break
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _momentum_subgrad(p, stop_time, consider=None, max_iters=5000):
    """Subgradient method with Nesterov momentum for the max-autocorrelation problem."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    best_p = p.copy()
    best_score = float(2.0 * n * _conv_self(p).max())
    momentum = np.zeros(n)
    beta = 0.9
    base_step = 0.008

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break

        c = _conv_self(p)
        cmax = float(c.max())
        # Active set: lags where b[k] >= 0.98 * cmax
        threshold = cmax * 0.98
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Subgradient of max over active set
        grad = np.zeros_n = np.zeros(n)
        for k in active:
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                contrib = p[kj[valid]] + p[valid]
                grad[valid] += contrib
        grad /= len(active)
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-30:
            grad /= gnorm

        step = base_step / (1.0 + 0.01 * it)
        if step < 1e-9:
            break

        # Nesterov momentum
        momentum = beta * momentum - step * grad
        p_y = p - momentum
        p_y = np.maximum(p_y, 0.0)
        ss = float(p_y.sum())
        if ss < 1e-15:
            p_y = p.copy()
            ss = 1.0
        p_y /= ss

        # Line search: try a few scales
        for scale in (1.0, 0.5, 0.25, 0.1):
            p_cand = p - scale * momentum
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = float(2.0 * n * _conv_self(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-16:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                break
        else:
            base_step *= 0.95
            if base_step < 1e-12:
                break

        if it % 100 == 0 and consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 20.0
    usable = max(1.0, hard - t0)
    final_reserve = max(12.0, 0.08 * usable) if usable > 80.0 else 0.0
    phase_limit = hard - final_reserve

    parent = _get_parent()
    n = int(parent.size)
    best_arr = parent / float(parent.sum())
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_fft(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    # ========== Phase 1: L-S-E Optimization ==========
    now = time.monotonic()
    p1_end = min(phase_limit, now + 0.40 * (phase_limit - now))
    if p1_end > now + 20.0:
        p_work = best_arr.copy()
        try:
            p2 = _lsgs_optimize(p_work, n, p1_end, consider=consider, fine_only=False)
        except Exception:
            p2 = p_work
        if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0:
            p2 = p2 / float(p2.sum())
            consider(p2)

    # ========== Phase 2: LP Dual Steps ==========
    now = time.monotonic()
    p2_end = min(phase_limit, now + 0.20 * (phase_limit - now))
    if p2_end > now + 15.0:
        p_work = best_arr.copy() / float(best_arr.sum())
        n_lp = int(p_work.size)
        # Do a few LP steps
        for lp_iter in range(5):
            now = time.monotonic()
            if now > p2_end - 10.0:
                break
            if n_lp > 4000:
                break  # LP too large
            c = _conv_self(p_work)
            M = float(c.max())
            if not math.isfinite(M) or M <= 0:
                break
            K = min(100, 2 * n_lp - 1)
            try:
                g_star = _dual_lp_step(p_work, M, K=K, timeout=20)
            except Exception:
                g_star = None
            if g_star is not None and g_star.size == n_lp and g_star.sum() > 1e-10:
                g_norm = g_star / g_star.sum()
                # Blend
                for t_val in (0.3, 0.5, 0.7):
                    p_blend = (1 - t_val) * p_work + t_val * g_norm
                    p_blend = np.maximum(p_blend, 0.0)
                    ss = float(p_blend.sum())
                    if ss < 1e-15:
                        continue
                    p_blend /= ss
                    consider(p_blend)
                p_work = p_blend / float(p_blend.sum())
            else:
                break

    # ========== Phase 3: Fine L-S-E + Coordinate Descent ==========
    now = time.monotonic()
    p3_end = min(phase_limit, now + 0.25 * (phase_limit - now))
    if p3_end > now + 15.0:
        p_work = best_arr.copy() / float(best_arr.sum())
        # Fine L-S-E
        lse_stop = min(p3_end - 20.0, now + 0.5 * (p3_end - now))
        if lse_stop > now + 15.0:
            try:
                p2 = _lsgs_optimize(p_work, n, lse_stop, consider=consider, fine_only=True)
            except Exception:
                p2 = p_work
            if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p_work = p2 / float(p2.sum())
                consider(p_work)

        # Coordinate descent
        now = time.monotonic()
        if now < p3_end - 10.0:
            cd_stop = min(p3_end - 2.0, now + (p3_end - now) * 0.8)
            try:
                p3, _ = _coord_polish(p_work, cd_stop, max_sweeps=80000, seed=int(seed + 777), consider=consider)
            except Exception:
                p3 = p_work
            if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and p3.sum() > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)

    # ========== Phase 4: Perturbation + Re-optimize ==========
    now = time.monotonic()
    p4_end = min(phase_limit, now + 0.10 * (phase_limit - now))
    if p4_end > now + 15.0:
        rng = np.random.default_rng(seed + 4242)
        for try_i in range(3):
            now = time.monotonic()
            if now > p4_end - 10.0:
                break
            # Perturb
            noise_scale = 0.03 * (try_i + 1)
            p_pert = best_arr * np.exp(rng.normal(0.0, noise_scale, n))
            p_pert = np.maximum(p_pert, 0.0)
            ss = float(p_pert.sum())
            if ss < 1e-15:
                continue
            p_pert /= ss
            # Quick L-S-E + CD
            sub_end = min(p4_end - 5.0, now + (p4_end - now) / (3 - try_i) - 2.0)
            if sub_end < now + 10.0:
                continue
            try:
                p_t = _lsgs_optimize(p_pert, n, sub_end, consider=consider, fine_only=True)
            except Exception:
                p_t = p_pert
            if p_t is not None and p_t.size == n and p_t.sum() > 0:
                p_t = p_t / float(p_t.sum())
                consider(p_t)
                # Quick CD
                now = time.monotonic()
                if now < sub_end - 5.0:
                    try:
                        p_t2, _ = _coord_polish(p_t, sub_end, max_sweeps=20000, seed=int(seed + 5000 + try_i), consider=consider)
                    except Exception:
                        p_t2 = p_t
                    if p_t2 is not None and p_t2.size == n and p_t2.sum() > 0:
                        p_t2 = p_t2 / float(p_t2.sum())
                        consider(p_t2)

    # ========== Phase 5: Final Polish ==========
    now = time.monotonic()
    final_end = min(hard - 5.0, t0 + budget - 5.0)
    if final_end > now + 20.0 and best_arr.size > 0:
        nbest = int(best_arr.size)
        rem = final_end - now

        # Momentum subgradient
        if rem > 40.0 and nbest <= 5000:
            sg_stop = min(final_end - 30.0, now + 0.3 * rem)
            try:
                p_sg = _momentum_subgrad(best_arr, sg_stop, consider=consider, max_iters=5000)
            except Exception:
                p_sg = best_arr
            if p_sg is not None and p_sg.size == nbest and p_sg.sum() > 0:
                p_sg = p_sg / float(p_sg.sum())
                consider(p_sg)

        # Final coordinate descent
        now = time.monotonic()
        cstop = min(final_end - 3.0, now + max(10.0, 0.4 * (final_end - now)))
        if cstop > now + 8.0:
            try:
                p3, _ = _coord_polish(best_arr, cstop, max_sweeps=100000, seed=int(seed + 99999), consider=consider)
            except Exception:
                p3 = best_arr
            if p3 is not None and p3.size == nbest and p3.sum() > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)

    # ========== Final Validation ==========
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
