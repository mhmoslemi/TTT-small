def _coord_descent(p, stop_time, seed=42, consider=None, max_sweeps=None):
    """Coordinate descent with grid search for each coordinate."""
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p
    p = p / s  # normalize to sum=1

    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    
    if max_sweeps is None:
        max_sweeps = 500000
    sweep = 0
    flat = 0

    # Adaptive grid size
    G = 40  # grid points per coordinate

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.0:
        improved = False
        # Random order
        order = rng.permutation(n)
        
        for ii in range(n):
            i = int(order[ii])
            if time.monotonic() > stop_time - 0.5:
                break
            v = float(p[i])
            # Affected region: conv[k] for k in [i, i+n-1]
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue
            base = c[start_i:end_i].copy()
            # infl[k] = p[k-i] for k in [0, Ls-1], i.e., p[0], p[1], ..., p[Ls-1]
            infl = p[:Ls].copy()
            
            # Max outside affected region
            max_const = 0.0
            if start_i > 0:
                ml = float(c[:start_i].max())
                if ml > max_const:
                    max_const = ml
            if end_i < Ltot:
                mr = float(c[end_i:].max())
                if mr > max_const:
                    max_const = mr

            S = 1.0  # normalized sum
            avg = S / float(n)
            
            # Search range for x (new value of p[i])
            x_hi = max(30.0 * v, 100.0 * avg, 0.05 * S)
            x_hi = min(x_hi, 20.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, avg)

            # Grid search
            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 <= v <= x_hi:
                # Insert current value
                pts = np.unique(np.concatenate([pts, [v]]))

            # Evaluate score at each grid point
            w = pts - v
            denom = S + w
            vals = np.full(pts.shape, math.inf)
            valid = denom > 1e-15
            if np.any(valid):
                # aff[k] = base[k] + w * infl[k] for each point
                # We want max_k aff[k] for each w
                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                # Diagonal correction: c[2i] gets extra w^2 term
                diag = i - start_i
                if 0 <= diag < Ls:
                    dval = base[diag] + w * infl[diag] + w * w
                    maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                vals[valid] = (2.0 * n * maff[valid]) / (denom[valid] ** 2)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Fine refinement
            if G > 4:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 3.0 * dw)
                hi = min(x_hi, best_x + 3.0 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 50)
                fine = np.unique(np.concatenate([fine, [best_x]]))
                w2 = fine - v
                denom2 = S + w2
                vals2 = np.full(fine.shape, math.inf)
                valid2 = denom2 > 1e-15
                if np.any(valid2):
                    aff2 = base[None, :] + w2[:, None] * infl[None, :]
                    maff2 = np.max(aff2, axis=1)
                    diag = i - start_i
                    if 0 <= diag < Ls:
                        dval2 = base[diag] + w2 * infl[diag] + w2 * w2
                        maff2 = np.maximum(maff2, dval2)
                    maff2 = np.maximum(maff2, max_const)
                    vals2[valid2] = (2.0 * n * maff2[valid2]) / (denom2[valid2] ** 2)
                    if np.any(np.isfinite(vals2)):
                        imin2 = int(np.argmin(vals2))
                        vx2 = float(fine[imin2])
                        vv2 = float(vals2[imin2])
                        if vv2 < best_val:
                            best_x = vx2
                            best_val = vv2

            # Accept if improvement
            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    # Update convolution
                    S_new = S + w
                    # Affected region update
                    c[start_i:end_i] = base + w * infl
                    # Diagonal correction
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    # Re-normalize
                    if S_new > 1e-300:
                        inv = 1.0 / S_new
                        p *= inv
                        c *= inv * inv
                    cur_score = 2.0 * n * c[start_i:end_i].max()
                    if start_i > 0:
                        cur_score = max(cur_score, float(c[:start_i].max()))
                    if end_i < Ltot:
                        cur_score = max(cur_score, float(c[end_i:].max()))
                    improved = True

        # End of sweep: recompute convolution to avoid drift
        if sweep % 5 == 0:
            c = _conv_self(p)
            np.maximum(c, 0.0, out=c)
            cur_score = float(2.0 * n * c.max())
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 6:
                # Diversify
                if time.monotonic() < stop_time - 5.0:
                    p = p * np.exp(rng.normal(0.0, 0.03, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    if cur_score < best_score - 1e-13:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider:
                            consider(p)
                    flat = 0
                else:
                    break
        sweep += 1

    return best_p
