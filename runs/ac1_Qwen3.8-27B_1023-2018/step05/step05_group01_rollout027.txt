```python
    """One subgradient step to minimize max(a*a)."""
    n = len(a)
    L = 2*n - 1
    size = next_fast_len(L)
    
    # Compute c = a * a
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    
    # Find top-K
    top_k = np.argpartition(c, -K)[-K:]
    S = top_k[np.argsort(c[top_k][::-1])]  # indices of top-K values
    
    # Compute subgradient
    # g[i] = (1/|S|) sum_{k in S} (a[k-i] + a[i-k])
    # a[k-i] = flipped a, shifted by k
    a_flipped = a[::-1]  # a_flipped[j] = a[n-1-j]
    
    g = np.zeros(n)

        # a[k-i] for i = 0,...,n-1: this is a_flipped[n-1-(k-i)] = a_flipped[n-1-k+i]
        # = a[n-1-(n-1-k+i)] = a[k-i]
        # So a[k-i] = a_flipped shifted: a_flipped[n-1-k+i] for i=0,...,n-1
        # This is a_flipped[n-1-k : n-1-k+n] (padded)
        idx = n - 1 - k
        if idx >= 0:
            g[:n-idx] += a_flipped[idx:idx+n-idx]  # wait, this is getting confusing
        # Let me just do it directly
        for i in range(n):
            if 0 <= k-i < n:
                g[i] += a[k-i]
            if 0 <= i-k < n:
                g[i] += a[i-k]
    g /= len(S)
    
    return g, c.max()
```

Now let me think about the mirror descent more carefully. The key insight is:

We want to minimize f(a) = max_k c_k where c = a*a, subject to a in simplex.

The subgradient at a is: g = sum_{k in S} Q_k a / |S| where S = argmax(c) (or top-K).

Q_k a is the vector with (Q_k a)[i] = a[k-i] + a[i-k] (with out-of-range being 0).

Mirror descent update: a_new ∝ a * exp(-eta * g)

This is equivalent to: a_new[i] = a[i] * w[i] / sum_j a[j] * w[j] where w[i] = exp(-eta * g[i]).

The effect is to decrease a[i] where g[i] is large (i.e., where increasing a[i] would increase the max convolution value) and increase a[i] where g[i] is small or negative.

```python
    c = irfft(F * F, size)[:L]

    n = a.size
    c = _conv(a)
    return 2.0 * n * float(c.max())


def _compute_subgrad(a, S):
    """Compute subgradient of max_k c_k at a, using active set S."""
    n = a.size
    g = np.zeros(n)
    for k in S:
        k = int(k)
        # a[k-i] for valid i
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            g[idx] += a[k - idx]
        # a[i-k] for valid i
        i_lo2 = max(0, k)
        i_hi2 = min(n - 1, k + n - 1)
        if i_lo2 <= i_hi2:
            idx2 = np.arange(i_lo2, i_hi2 + 1)
            g[idx2] += a[idx2 - k]
    g /= len(S)
    return g

    n = a.size
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_normalized(a)
    
    fft_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1

        F = rfft(a, fft_size)
        c = irfft(F * F, fft_size)[:conv_len]
        return c, float(c.max())

        scaled = -eta * g
```

```python
def _subgrad(a, k_list):
    """Subgradient of max_k (a*a)_k using active set k_list."""
    n = a.size

        lo1 = max(0, k - n + 1)
        hi1 = min(k, n - 1)
        if lo1 <= hi1:

            g[idx] += a[k - idx]
        # a[i-k] for valid i: i in [max(0,k), min(n-1, k+n-1)]

        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:

            g[idx2] += a[idx2 - k]

        return c

        G = rfft(w, fft_size)

        np.maximum(q, 0.0, out=q)
        return q

    p1_end = min(hard_deadline, t0 + 0.50 * (hard_deadline - t0))
    if p1_end > time.monotonic() + 20.0:
        try:
            p1, s1 = _mirror_descent(best_arr, p1_end, K=10, base_eta=0.8, seed=int(seed + 1))
            consider(p1)
        except Exception:
            pass

    # Phase 2: LSE optimization from current best
    p2_end = min(hard_deadline, time.monotonic() + 0.35 * (hard_deadline - time.monotonic()))
    if p2_end > time.monotonic() + 15.0:
        try:
            p2 = _lse_optimize(best_arr, p2_end, consider=consider, seed=int(seed + 2))
            if p2 is not None and p2.size == n and p2.sum() > 0:
                p2 = p2 / p2.sum()
                consider(p2)
        except Exception:
            pass

    # Phase 3: Coordinate descent polish
    p3_end = min(hard_deadline - 10.0, time.monotonic() + 0.30 * (hard_deadline - time.monotonic()))
    if p3_end > time.monotonic() + 10.0:
        try:
            p3, s3 = _coord_descent(best_arr, p3_end, G=40, seed=int(seed + 3), consider=consider)
            if p3 is not None and p3.size == n and p3.sum() > 0:
                consider(p3)
        except Exception:
            pass

    # Phase 4: Second mirror descent pass with refined initialization
    if time.monotonic() < hard_deadline - 60.0:
        p4_end = min(hard_deadline - 30.0, time.monotonic() + 0.4 * (hard_deadline - time.monotonic()))
        if p4_end > time.monotonic() + 30.0:
            try:
                p4, s4 = _mirror_descent(best_arr, p4_end, K=15, base_eta=0.5, seed=int(seed + 4))
                consider(p4)
            except Exception:
                pass

    # Phase 5: Further LSE refinement
    if time.monotonic() < hard_deadline - 40.0:
        p5_end = min(hard_deadline - 20.0, time.monotonic() + 0.3 * (hard_deadline - time.monotonic()))
        if p5_end > time.monotonic() + 15.0:
            try:
                p5 = _lse_optimize(best_arr, p5_end, consider=consider, seed=int(seed + 5))
                if p5 is not None and p5.size == n and p5.sum() > 0:
                    p5 = p5 / p5.sum()
                    consider(p5)
            except Exception:
                pass

    # Phase 6: Final coordinate polish
    if time.monotonic() < hard_deadline - 10.0:
        p6_end = hard_deadline - 5.0
        if p6_end > time.monotonic() + 8.0:
            try:
                p6, s6 = _coord_descent(best_arr, p6_end, G=50, seed=int(seed + 6), consider=consider)
                if p6 is not None and p6.size == n and p6.sum() > 0:
                    consider(p6)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in np.clip(parent, 0, 1000)]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0 / 64.0] * 64
        return ls
    except Exception:
        return [1.0 / 64.0] * 64
```
