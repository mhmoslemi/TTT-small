                    mout2 = max(mout2, float(cf[:st2].max()))
                if en2 < L:
                    mout2 = max(mout2, float(cf[en2:].max()))
                base2 = cf[st2:en2]
                dp2 = 2 * i - st2
                sc2 = max(v2, 0.5 / n)
                xl2 = max(0.0, v2 - 1.2 * sc2)
                xh2 = v2 + 1.5 * sc2
                xh2 = min(xh2, max(xl2 + 0.3 * sc2, 3.0 / n))
                if xh2 - xl2 < 1e-13:
                    continue
                pts2 = np.linspace(xl2, xh2, G2 + 1)
                ts2 = pts2 - v2
                aff2 = base2[None, :] + ts2[:, None] * infl2[None, :]
                if 0 <= dp2 < Ls2:
                    aff2[:, dp2] += ts2 * ts2
                maf2 = np.maximum(aff2.max(axis=1), mout2)
                maf2 = np.maximum(maf2, 0.0)
                den2 = np.maximum(Sf + ts2, 1e-15)
                vals2 = 2.0 * n * maf2 / (den2 * den2)
                im2 = int(np.argmin(vals2))
                xb2 = float(pts2[im2])
                vb2 = float(vals2[im2])
                if vb2 < sff - 1e-13 and xb2 >= 0:
                    t2 = xb2 - v2
                    pf[i] = xb2
                    Sf += t2
                    cf[st2:en2] += t2 * infl2
                    if 0 <= dp2 < Ls2:
                        cf[st2 + dp2] += t2 * t2
                    sff = vb2
                    imp2 = True
            if Sf > 1e-15:
                inv2 = 1.0 / Sf
                pf *= inv2
                cf *= inv2 * inv2
                Sf = 1.0
            cf = conv(pf)
            np.maximum(cf, 0.0, out=cf)
            sff = float(2.0 * n * cf.max())
            if math.isfinite(sff) and sff < best_score - 1e-14:
                best_score = sff
                best_p = pf.copy()
            if not imp2:
                break

    # Final validation with exact evaluator
    try:
        out_list = [float(x) for x in best_p]
        fs = evaluate_sequence(out_list)
        if math.isfinite(fs) and fs > 0:
            return out_list
    except Exception:
        pass

    # Fallback
    try:
        out_list = [float(x) for x in p]
        fs2 = evaluate_sequence(out_list)
        if math.isfinite(fs2) and fs2 > 0:
            return out_list
    except Exception:
        pass

    return [1.0 / n] * n
