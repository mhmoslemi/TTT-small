Let me denote w = x - v (the change in a[i]).
c_k(w) = c_k + w*(a[k-i]+a[i]) + w²*δ_{k,2i}

And the objective to minimize: f(w) = 2n * max( C_const, max_{k in A_i} (c_k + w*g_k + w²*h_k) ) / (S+w)²

where A_i = {k : i ≤ k ≤ min(i+n-1, 2n-2)} and C_const = max_{k ∉ A_i} c_k.

This is a univariate function. To find the minimum:
- It's a ratio of a convex piecewise function (max of linear and one quadratic) to a quadratic.
- Not necessarily convex in w, but typically has a single minimum.
- Best approach: grid search + golden section refinement.

For the grid, I need to figure out the range of w. The optimal w should be such that the new a[i] is in [0, some upper bound]. A reasonable range is [0, 3*S/n] (at most 3x the average).

        return p

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:
        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(3072, dtype=np.float64)
    return p


def _conv_self(a):
    n = a.size
    if n <= 4096:
        c = np.convolve(a, a, mode="full")
    else:
        L = 2 * n - 1
        size = next_fast_len(L)
        F = rfft(a, size)
        c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * a.size * m / (s * s))


def _conv_fft(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _lse_optimize(p, stop_time, consider=None):
    """Brief LSE-based gradient optimization on the simplex."""
    n = p.size
    p = np.maximum(p.astype(np.float64), 0.0).copy()
    S = float(p.sum())
    if S < 1e-300:
        return p
    p = p / S

    best_p = p.copy()
    best_sc = float(2.0 * n * _conv_fft(p).max())

    # Precompute FFT sizes
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len((2 * n - 1) + n)
    w_len = 2 * n - 1

    def compute_grad(a, T):
        c = _conv_fft(a)
        cm = float(c.max())
        if cm < 1e-300:
            return np.zeros(n), cm, 0.0
        scaled = (c - cm) / T
        scaled = np.clip(scaled, -400.0, 400.0)
        exps = np.exp(scaled)
        sw = float(exps.sum())
        if sw < 1e-300:
            w = np.zeros(w_len)
            w[int(np.argmax(c))] = 1.0
        else:
            w = exps / sw
        L_val = cm + T * float(np.log(sw / w_len))
        if not math.isfinite(L_val):
            L_val = cm

        # Cross-correlation: R[i] = sum_j w[j+i]*a[j] for j in [0,n-1]
        a_pad = np.zeros(cross_size)
        a_pad[:n] = a
        Fw = rfft(w, cross_size)
        Fa = rfft(a_pad, cross_size)
        R = irfft(Fw * np.conj(Fa), cross_size)[:n]
        np.maximum(R, 0.0, out=R)

        # Sliding window sum
        cum = np.concatenate(([0.0], np.cumsum(w, dtype=np.float64)))
        hi_idx = np.minimum(np.arange(n) + n, w_len)
        win_w = cum[hi_idx] - cum[np.arange(n)]

        grad_L = R + a * win_w
        grad = (2.0 * n) * (grad_L / 1.0 - 2.0 * L_val)
        # Since S=1: dF/da_i = 2n*(grad_L[i] - 2L)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return grad, cm, L_val

    # Schedule: decreasing T
    n_iters = 200
    c0 = _conv_fft(p)
    cm0 = float(c0.max())
    T0 = 0.05 * cm0
    T_end = 0.001 * cm0

    for it in range(n_iters):
        if time.monotonic() > stop_time - 1.0:
            break
        frac = it / max(1, n_iters - 1)
        T = T0 * (T_end / T0) ** frac
        if T < 1e-300:
            T = 1e-300

        grad, _, _ = compute_grad(p, T)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step size
        lr = 0.01 / (1.0 + 0.05 * it)
        # Projected gradient step on simplex
        p_new = p - lr * grad
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if ss < 1e-15:
            continue
        p_new /= ss

        # Check improvement (use FFT score)
        sc = float(2.0 * n * _conv_fft(p_new).max())
        if math.isfinite(sc) and sc < best_sc - 1e-16:
            best_sc = sc
            best_p = p_new.copy()
            p = p_new.copy()
        elif math.isfinite(sc) and sc < float(2.0 * n * _conv_fft(p).max()):
            p = p_new.copy()
        else:
            # Try smaller step
            p_new2 = p - lr * 0.25 * grad
            p_new2 = np.maximum(p_new2, 0.0)
            ss2 = float(p_new2.sum())
            if ss2 > 1e-15:
                p_new2 /= ss2
                sc2 = float(2.0 * n * _conv_fft(p_new2).max())
                if math.isfinite(sc2) and sc2 < float(2.0 * n * _conv_fft(p).max()):
                    p = p_new2.copy()
                    if sc2 < best_sc:
                        best_sc = sc2
                        best_p = p_new2.copy()

        if it % 30 == 0 and consider:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p


def _coord_descent(p, stop_time, seed=12345, consider=None, max_sweeps=10000):
    """Efficient coordinate descent on the simplex for 2n*max(a*a)."""
    n = p.size
    p = np.maximum(p.astype(np.float64), 0.0).copy()
    S = float(p.sum())
    if S < 1e-300:
        return p
    p /= S  # work on simplex, S=1

    Ltot = 2 * n - 1
    c = _conv_fft(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    G = 48  # grid points
    flat_count = 0
    sweep = 0

    while sweep < max_sweeps:
        if time.monotonic() > stop_time - 0.5:
            break

        improved = False
        order = rng.permutation(n)

        for ii in range(n):
            i = int(order[ii])
            if time.monotonic() > stop_time - 0.2:
                break

            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end]  # (Ls,)
            infl = 2.0 * p[:Ls]  # (Ls,)

            # max_const: max of unaffected convolution entries
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < Ltot:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

            # Quadratic position (local index where k=2i)
            quad_pos = 2 * i - start
            if not (0 <= quad_pos < Ls):
                quad_pos = -1

            # Search range for x = new a[i]
            avg = 1.0 / n
            x_hi = max(3.0 * v, 5.0 * avg, 1e-6)
            x_hi = min(x_hi, 0.1)  # in simplex, no element > 0.1 typically
            if x_hi < 1e-8:
                x_hi = 1e-8

            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 < v < x_hi:
                pts = np.unique(np.concatenate([pts, [v]]))

            w_arr = pts - v  # (P,)
            denom = 1.0 + w_arr  # S+v = 1+v since S=1... wait
            # Actually S=1 and we change a[i] from v to x=v+w, so new S = 1+w
            valid = denom > 1e-12

            # Vectorized evaluation
            aff = base[np.newaxis, :] + w_arr[:, np.newaxis] * infl[np.newaxis, :]
            if quad_pos >= 0:
                aff[:, quad_pos] += w_arr * w_arr
            m_aff = np.max(aff, axis=1)
            m_total = np.maximum(m_aff, max_const)
            scores = np.full(pts.shape, math.inf)
            sv = denom[valid]
            scores[valid] = (2.0 * n * m_total[valid]) / (sv * sv)

            if not np.any(np.isfinite(scores)):
                continue

            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            # Golden section refinement
            if G > 4:
                dw = x_hi / G
                lo = max(0.0, best_x - 1.5 * dw)
                hi = min(x_hi, best_x + 1.5 * dw)
                if hi - lo > 1e-14:
                    gr = (math.sqrt(5.0) - 1.0) / 2.0
                    a_gs = hi - gr * (hi - lo)
                    b_gs = lo + gr * (hi - lo)

                    def eval1d(xv):
                        wv = xv - v
                        dv = 1.0 + wv
                        if dv < 1e-12:
                            return math.inf
                        av = base + wv * infl
                        if quad_pos >= 0:
                            av[quad_pos] += wv * wv
                        mv = max(float(np.max(av)), max_const)
                        mv = max(mv, 0.0)
                        return (2.0 * n * mv) / (dv * dv)

                    fa = eval1d(a_gs)
                    fb = eval1d(b_gs)
                    for _ in range(22):
                        if hi - lo < 1e-15:
                            break
                        if fa < fb:
                            hi = b_gs
                            fb = fa
                            b_gs = a_gs
                            a_gs = hi - gr * (hi - lo)
                            fa = eval1d(a_gs)
                        else:
                            lo = a_gs
                            fa = fb
                            a_gs = b_gs
                            b_gs = lo + gr * (hi - lo)
                            fb = eval1d(b_gs)
                    gx = 0.5 * (lo + hi)
                    gv = eval1d(gx)
                    if math.isfinite(gv) and gv < best_val:
                        best_x = gx
                        best_val = gv

            if math.isfinite(best_val) and best_val < cur_score - 1e-15 and 0.0 <= best_x <= 1.0:
                w_new = best_x - v
                if abs(w_new) > 1e-16:
                    p[i] = best_x
                    c[start:end] += w_new * infl
                    if quad_pos >= 0:
                        c[start + quad_pos] += w_new * w_new
                    cur_score = best_val
                    improved = True

        # Periodic recomputation to prevent drift
        if sweep % 8 == 0:
            c = _conv_fft(p)
            cur_score = float(2.0 * n * c.max())
            if math.isfinite(cur_score) and cur_score < best_score - 1e-16:
                best_score = cur_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        # Check sweep improvement
        np_max_c = float(c.max())
        sc_now = 2.0 * n * np_max_c
        if math.isfinite(sc_now) and sc_now < best_score - 1e-16:
            best_score = sc_now
            best_p = p.copy()
            if consider:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat_count = 0
        else:
            flat_count += 1
            if flat_count >= 4:
                if time.monotonic() < stop_time - 3.0:
                    sigma = 0.015 + 0.01 * math.exp(-0.05 * flat_count)
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss < 1e-15:
                        break
                    p /= ss
                    c = _conv_fft(p)
                    cur_score = float(2.0 * n * c.max())
                    flat_count = 0
                else:
                    break
        sweep += 1

    return best_p
