import time
import math
from scipy.optimize import linprog


def _conv(a):
    return np.convolve(a, a)


def _score_of(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0:
        return float("inf")
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    return float(2.0 * a.size * float(_conv(a).max()) / (s * s))


def _get_parent():
    try:
        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(200)
    p = np.clip(p, 0.0, 1000.0)
    if p.size == 0 or not np.all(np.isfinite(p)):
        p = np.ones(200)
    return p


def _build_conv_mat(n, a):
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[d[mask]]
    return A


def _solve_lp(A, M, n, tl=3.0):
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                      method="highs",
                      options={"time_limit": tl,
                               "dual_feasibility_tolerance": 1e-11,
                               "primal_feasibility_tolerance": 1e-11})
        if res.status == 0 and res.x is not None:
            return np.clip(res.x, 0.0, None)
    except Exception:
        pass
    return None


def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones_like(a)
    return a / s


def _convmax_unit(a):
    a = _normalize(a)
    return float(_conv(a).max())


def _line_search(a, g, curM):
    best_t, best_v = 0.0, curM
    for t in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0):
        v = _convmax_unit((1.0 - t) * a + t * g)
        if v < best_v - 1e-13:
            best_v, best_t = v, t
    if best_t <= 0.0:
        return None, curM
    lo = max(0.0, best_t - 0.12)
    hi = min(1.0, best_t + 0.12)
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    for _ in range(24):
        if hi - lo < 1e-7:
            break
        m1 = lo + gr * (hi - lo)
        m2 = hi - gr * (hi - lo)
        f1 = _convmax_unit((1.0 - m1) * a + m1 * g)
        f2 = _convmax_unit((1.0 - m2) * a + m2 * g)
        if f1 > f2:
            lo = m1
        else:
            hi = m2
    best_t = 0.5 * (lo + hi)
    cand = _normalize((1.0 - best_t) * a + best_t * g)
    return cand, float(_conv(cand).max())


def _smooth_grad(a, conv, alpha):
    n = a.size
    S = conv
    z = S - S.max()
    e = np.exp(alpha * z)
    p = e / e.sum()
    # grad_j = sum_k p_k * 2 * a[k-j]  (valid k-j)
    g = np.zeros(n)
    # S_k = sum_j a_j a_{k-j}; gradient w.r.t a_j of S_k is 2 a_{k-j}
    # reshape to correlate
    conv2 = np.convolve(a, p)  # sum_k p_k a_{k-j}? convolve a with p
    # (a ? p)_m = sum_j a_j p_{m-j}. We want for each j: sum_k p_k a_{k-j}
    # = (p ? a)_j with suitable. Directly: g_full[j] = sum_k p_k * 2 a[k-j]
    idx = np.arange(2 * n - 1)[:, None]
    jj = np.arange(n)[None, :]
    d = idx - jj          # d = k - j
    mask = (d >= 0) & (d < n)
    gmat = np.zeros((2 * n - 1, n))
    gmat[mask] = p[d[mask]]
    g = 2.0 * gmat.sum(axis=0)
    return g

    stuck = 0

    while time.monotonic() < deadline:
        conv = _conv(a)
        M = float(conv.max())

            stuck = 0

            stuck += 1

                break
            continue

        moved = False

                    break

                        moved = True
                        break

                        moved = True
