def _lse_phase(p, n, stop_time, consider):
    """Log-sum-exp smoothed optimization to enter basin."""
    p = p.copy()
    s = float(p.sum())
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    L = 2 * n - 1
    size = next_fast_len(L)
    cross_L = 3 * n - 2
    cross_size = next_fast_len(cross_L)

    def cfast(a):
        F = rfft(a, size)
        c = irfft(F * F, size)[:L]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_L]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax):
            return 1e100, np.zeros_like(uu)
        if T <= 0:
            T = 1e-4 * cmax
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if sum_exp <= 0 or not math.isfinite(sum_exp):
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    stages = [(0.05, 0.5), (0.015, 0.3), (0.004, 0.2)]

                x_hi = 2.0 * v + 1.0 / n

            # Vectorized grid evaluation
            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 <= v <= x_hi:
                pts = np.sort(np.unique(np.concatenate((pts, [v]))))
                G_eff = len(pts) - 1
            else:
                G_eff = G

            ws = pts - v
            wins = ws[:, None] * infl[None, :]
            aff = base[None, :] + wins
            if 0 <= diag_offset < Ls:
                aff[:, diag_offset] += ws[:, None] * ws[:, None]
            maff = aff.max(axis=1)
            maff = np.maximum(maff, max_const)
            denoms = S + ws
            vals = np.where(denoms > 1e-15, 2.0 * n * maff / (denoms * denoms), math.inf)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Golden section refinement
            if G_eff > 3:
                dw = x_hi / G_eff
                lo_x = max(0.0, best_x - 2.0 * dw)
                hi_x = min(x_hi, best_x + 2.0 * dw)
                if hi_x - lo_x > 1e-13:
                    gr = (math.sqrt(5.0) - 1.0) / 2.0
                    a_gs = lo_x
                    b_gs = hi_x
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    d_gs = a_gs + gr * (b_gs - a_gs)

                    def eval_x(x):
                        w = x - v
                        denom = S + w
                        if denom <= 1e-15:
                            return math.inf
                        nb = base + w * infl
                        if 0 <= diag_offset < Ls:
                            nb = nb.copy()
                            nb[diag_offset] += w * w
                        mv = float(nb.max())
                        if max_const > mv:
                            mv = max_const
                        return 2.0 * n * mv / (denom * denom)

                    f_c = eval_x(c_gs)
                    f_d = eval_x(d_gs)
                    for _ in range(14):
                        if f_c < f_d:
                            b_gs = d_gs
                            d_gs = c_gs
                            f_d = f_c
                            c_gs = b_gs - gr * (b_gs - a_gs)
                            f_c = eval_x(c_gs)
                        else:
                            a_gs = c_gs
                            c_gs = d_gs
                            f_c = f_d
                            d_gs = a_gs + gr * (b_gs - a_gs)
                            f_d = eval_x(d_gs)
                    x_opt = (a_gs + b_gs) / 2.0
                    f_opt = eval_x(x_opt)
                    if f_opt < best_val:
                        best_x, best_val = x_opt, f_opt

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and best_x >= 0:
                w = best_x - v
                if abs(w) > 1e-16:
                    # Store old values for correct update
                    a_old = p[:Ls].copy()
                    if i < Ls:
                        a_old[i] = v
                    p[i] = best_x
                    S += w
                    c[start:end] += 2.0 * w * a_old
                    if 0 <= diag_offset < Ls:
                        c[start + diag_offset] += w * w
                    cur_score = best_val
                    improved = True

        # Renormalize
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        # Update prefix/suffix
        prefix_max[0] = 0.0
        for j in range(1, Ltot + 1):
            prefix_max[j] = max(prefix_max[j-1], float(c[j-1]))
        suffix_max[Ltot] = 0.0
        for j in range(Ltot - 1, -1, -1):
            suffix_max[j] = max(suffix_max[j+1], float(c[j]))

        if sweep_count % 4 == 0:
            c = _conv_self(p)
            np.maximum(c, 0.0, out=c)

        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
            consider(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5 and time.monotonic() < stop_time - 60.0:
                sigma = 0.015 + 0.01 * min(flat, 10)
                noise = rng.normal(0.0, sigma, n)
                p *= np.exp(noise)
                p = np.maximum(p, 0.0)
                ss = float(p.sum())
                if ss > 1e-300:
                    p /= ss
                    S = 1.0
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score:
                        best_score = cur_score
                        best_p = p.copy()
                        consider(best_p)
                flat = 0

        if flat >= 25:
            break

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    return best_p, best_score


def _focus_polish(p, stop_time, consider, top_k=250):
    """Targeted polish on coordinates affecting near-max lags."""
    n = len(p)
    p = p.copy()
    S = float(p.sum())
    p /= S
    S = 1.0
    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(77777)

    cmax = float(c.max())
    active_thresh = cmax * 0.99
    active_lags = np.where(c >= active_thresh)[0]

    important = np.zeros(n, dtype=bool)
    counts = np.zeros(n, dtype=int)
    for lag in active_lags:
        lo = max(0, int(lag) - n + 1)
        hi = min(n - 1, int(lag))
        important[lo:hi+1] = True
        counts[lo:hi+1] += 1
    important_idx = np.where(important)[0]
    other_idx = np.where(~important)[0]
    if len(important_idx) > top_k:
        important_idx = important_idx[np.argsort(-counts[important_idx])][:top_k]

    G_fine = 32
    sweep = 0
    flat = 0
    prefix_max = np.zeros(Ltot + 1)
    prefix_max[0] = 0.0
    for j in range(1, Ltot + 1):
        prefix_max[j] = max(prefix_max[j-1], float(c[j-1]))
    suffix_max = np.zeros(Ltot + 2)
    suffix_max[Ltot] = 0.0
    for j in range(Ltot - 1, -1, -1):
        suffix_max[j] = max(suffix_max[j+1], float(c[j]))

    while time.monotonic() < stop_time - 3.0:
        sweep += 1
        improved = False
        if sweep % 4 < 3:
            order = rng.permutation(important_idx)
        else:
            order = rng.permutation(other_idx)

        for i in order:
            if time.monotonic() > stop_time - 1.5:
                break
            if S > 1.1 or S < 0.9:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0

            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            max_const = 0.0
            if start > 0:
                max_const = max(max_const, float(prefix_max[start]))
            if end < Ltot:
                max_const = max(max_const, float(suffix_max[end]))

            base = c[start:end].copy()
            diag_offset = 2 * i - start
            infl = 2.0 * p[:Ls].copy()

            # Narrower range for polish
            x_hi = max(6.0 * v, 15.0 / n, 0.005)
            x_hi = min(x_hi, v + 0.05)
            x_lo = max(0.0, v - 0.5 * x_hi)
            if x_hi <= 0:
                x_hi = v + 1e-4
                x_lo = max(0.0, v - 1e-4)

            pts = np.linspace(x_lo, x_hi, G_fine + 1)
            ws = pts - v
            wins = ws[:, None] * infl[None, :]
            aff = base[None, :] + wins
            if 0 <= diag_offset < Ls:
                aff[:, diag_offset] += ws[:, None] * ws[:, None]
            maff = aff.max(axis=1)
            maff = np.maximum(maff, max_const)
            denoms = S + ws
            vals = np.where(denoms > 1e-15, 2.0 * n * maff / (denoms * denoms), math.inf)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Tiny golden
            if G_fine > 4:
                dw = (x_hi - x_lo) / G_fine
                lo_gs = max(0.0, best_x - 1.0 * dw)
                hi_gs = min(x_hi, best_x + 1.0 * dw)
                if hi_gs - lo_gs > 1e-13:
                    gr = (math.sqrt(5.0) - 1.0) / 2.0
                    a_gs, b_gs = lo_gs, hi_gs
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    d_gs = a_gs + gr * (b_gs - a_gs)

                    def eval_x(x):
                        w = x - v
                        denom = S + w
                        if denom <= 1e-15:
                            return math.inf
                        nb = base + w * infl
                        if 0 <= diag_offset < Ls:
                            nb = nb.copy()
                            nb[diag_offset] += w * w
                        mv = float(nb.max())
                        if max_const > mv:
                            mv = max_const
                        return 2.0 * n * mv / (denom * denom)

                    f_c = eval_x(c_gs)
                    f_d = eval_x(d_gs)
                    for _ in range(10):
                        if f_c < f_d:
                            b_gs, d_gs, f_d = d_gs, c_gs, f_c
                            c_gs = b_gs - gr * (b_gs - a_gs)
                            f_c = eval_x(c_gs)
                        else:
                            a_gs, c_gs, f_c = c_gs, d_gs, f_d
                            d_gs = a_gs + gr * (b_gs - a_gs)
                            f_d = eval_x(d_gs)
                    x_opt = (a_gs + b_gs) / 2.0
                    f_opt = eval_x(x_opt)
                    if f_opt < best_val:
                        best_x, best_val = x_opt, f_opt

            if math.isfinite(best_val) and best_val < cur_score - 1e-15 and best_x >= 0:
                w = best_x - v
                if abs(w) > 1e-16:
                    a_old = p[:Ls].copy()
                    if i < Ls:
                        a_old[i] = v
                    p[i] = best_x
                    S += w
                    c[start:end] += 2.0 * w * a_old
                    if 0 <= diag_offset < Ls:
                        c[start + diag_offset] += w * w
                    cur_score = best_val
                    improved = True

        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        # Update prefix/suffix every sweep
        prefix_max[0] = 0.0
        for j in range(1, Ltot + 1):
            prefix_max[j] = max(prefix_max[j-1], float(c[j-1]))
        suffix_max[Ltot] = 0.0
        for j in range(Ltot - 1, -1, -1):
            suffix_max[j] = max(suffix_max[j+1], float(c[j]))

        if sweep % 3 == 0:
            c = _conv_self(p)
            np.maximum(c, 0.0, out=c)

        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
            consider(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 10:
                break

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 30.0

    # Get parent
    try:
        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(512, dtype=np.float64)
    if p.size == 0:
        p = np.ones(512, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(512, dtype=np.float64)
    n = len(p)
    p = p / float(p.sum())

    best_p = p.copy()
    best_score = _fft_score(p)

    def consider(arr):
        nonlocal best_p, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.clip(a, 0.0, 1000.0)
            a = np.where(np.isfinite(a), a, 0.0)
            if a.size == 0 or float(a.sum()) < 0.01:
                return
            sc = _fft_score(a)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                a = a / float(a.sum())
                best_p = a.copy()
                best_score = sc
        except Exception:
            pass

    # Verify initial with exact
    try:
        exact_sc = _exact_score(best_p)
        if math.isfinite(exact_sc):
            best_score = min(best_score, exact_sc)
    except Exception:
        pass

    # Phase 1: LSE smoothing (~80s)
    lse_stop = min(hard - 1150.0, t0 + 80.0)
    if lse_stop > time.monotonic() + 20.0:
        try:
            p2 = _lse_phase(p, n, lse_stop, consider)
            if p2 is not None and len(p2) == n and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p2 = p2 / float(p2.sum())
                consider(p2)
                if _fft_score(p2) < _fft_score(best_p):
                    p = p2.copy()
        except Exception:
            pass
        if _fft_score(p) > _fft_score(best_p) + 0.001:
            p = best_p.copy()

    # Phase 2: Main coordinate descent (~900s)
    coord_stop = min(hard - 250.0, t0 + 1050.0)
    if coord_stop > time.monotonic() + 40.0:
        try:
            p3, sc3 = _coord_descent(p, coord_stop, consider, seed=seed + 777)
            if p3 is not None and len(p3) == n and np.all(np.isfinite(p3)) and p3.sum() > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)
        except Exception:
            pass

    # Phase 3: Focus polish (~180s)
    pol_src = best_p.copy()
    polish_stop = min(hard - 50.0, t0 + 1230.0)
    if polish_stop > time.monotonic() + 30.0:
        try:
            p4, sc4 = _focus_polish(pol_src, polish_stop, consider, top_k=300)
            if p4 is not None and len(p4) == n and np.all(np.isfinite(p4)) and p4.sum() > 0:
                p4 = p4 / float(p4.sum())
                consider(p4)
        except Exception:
            pass

    # Phase 4: If time remains, another short coord descent
    rem_stop = hard - 35.0
    if rem_stop > time.monotonic() + 40.0:
        try:
            p5, sc5 = _coord_descent(best_p, rem_stop, consider, seed=seed + 54321)
            if p5 is not None and len(p5) == n and np.all(np.isfinite(p5)) and p5.sum() > 0:
                consider(p5)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_p, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(128, 1.0 / 128.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            try:
                pp = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
                pp = np.clip(pp, 0.0, 1000.0)
                ls = [float(x) for x in pp]
                final_sc = float(evaluate_sequence(ls))
            except Exception:
                return [1.0]
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
