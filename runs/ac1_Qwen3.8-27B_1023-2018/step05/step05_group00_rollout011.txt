            if hi2 <= lo2:
                hi2 = lo2 + 1e-14
            pts2 = np.linspace(lo2, hi2, G2 + 1, dtype=np.float64)
            if lo2 <= best_x <= hi2:
                pts2 = np.concatenate((pts2, [best_x]))
            vals2 = eval_xs(pts2)
            if np.any(np.isfinite(vals2)):
                imin2 = int(np.argmin(vals2))
                bx2 = float(pts2[imin2])
                bv2 = float(vals2[imin2])
                if bv2 < best_val:
                    best_x = bx2
                    best_val = bv2

            # Golden section refinement
            if G2 > 20 and math.isfinite(best_val):
                dw2 = (hi2 - lo2) / float(G2)
                glo = max(x_lo, best_x - 1.5 * dw2)
                ghi = min(x_hi, best_x + 1.5 * dw2)
                if ghi > glo + 1e-15:
                    gr = (math.sqrt(5.0) - 1.0) / 2.0
                    a_gs = glo
                    b_gs = ghi
                    fval = best_val
                    for _gs in range(8):
                        c_gs = b_gs - gr * (b_gs - a_gs)
                        d_gs = a_gs + gr * (b_gs - a_gs)
                        fc = float(np.min(eval_xs(np.array([c_gs]))))
                        fd = float(np.min(eval_xs(np.array([d_gs]))))
                        if not math.isfinite(fc) and not math.isfinite(fd):
                            break
                        if fc < fd:
                            b_gs = d_gs
                            if math.isfinite(fc) and fc < fval:
                                fval = fc
                                best_x = c_gs
                                best_val = fc
                        else:
                            a_gs = c_gs
                            if math.isfinite(fd) and fd < fval:
                                fval = fd
                                best_x = d_gs
                                best_val = fd
                        if abs(b_gs - a_gs) < 1e-16 * max(1.0, abs(best_x)):
                            break

            if (math.isfinite(best_val) and best_val < cur_score - 1e-14
                    and 0.0 <= best_x <= 20.0 * S):
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    c[start_i:end_i] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        # Renormalize periodically
        S = float(p.sum())
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        c = _conv_self(p)
        np.maximum(c, 0.0, out=c)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        if consider is not None and sweep % 10 == 0:
            try:
                nn = p.copy()
                nn /= nn.sum()
                consider(nn)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 10.0:
                    sig = 0.015 if fine else 0.03
                    p = p * np.exp(rng.normal(0.0, sig, n, dtype=np.float64))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
                        best_score = cur_score
                        best_p = p.copy()
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_self(p)
    sc = float(2.0 * n * c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 18.0

    parent = _get_parent()
    n = parent.size

    # Normalize parent
    work = parent.copy()
    work = np.where(np.isfinite(work), work, 0.0)
    work = np.clip(work, 0.0, 1000.0)
    if work.sum() < 0.01:
        work = np.ones(64, dtype=np.float64) / 64.0
        n = 64
    work = work / float(work.sum())

    best_arr = work.copy()
    best_score = float(_score_norm1(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        sc = float(_score_norm1(a / s))
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_arr = (a / s).copy()
            best_score = sc

    # === Phase 1: LSE optimization ===
    p1_end = min(hard - 300.0, t0 + 0.25 * budget)
    if p1_end > t0 + 20.0:
        try:
            p1 = _lse_lbfgs(work, p1_end, consider=consider, fine_only=False, seed=int(seed))
            if p1 is not None and p1.size == n and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 /= float(p1.sum())
                consider(p1)
        except Exception:
            pass

    work = best_arr.copy()

    # === Phase 2: Main coordinate descent ===
    p2_end = min(hard - 200.0, t0 + 0.62 * budget)
    if p2_end > time.monotonic() + 30.0:
        try:
            work = _coord_descent(work, p2_end, consider=consider,
                                  seed=int(seed + 1000), fine=False)
        except Exception:
            pass

    work = best_arr.copy()

    # === Phase 3: LSE again (finer) + coordinate descent ===
    p3a_end = min(hard - 120.0, time.monotonic() + 0.12 * budget)
    if p3a_end > time.monotonic() + 15.0:
        try:
            p3 = _lse_lbfgs(work, p3a_end, consider=consider, fine_only=True,
                            seed=int(seed + 2000))
            if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and p3.sum() > 0:
                p3 /= float(p3.sum())
                consider(p3)
        except Exception:
            pass

    work = best_arr.copy()
    p3b_end = min(hard - 100.0, t0 + 0.82 * budget)
    if p3b_end > time.monotonic() + 20.0:
        try:
            work = _coord_descent(work, p3b_end, consider=consider,
                                  seed=int(seed + 3000), fine=True)
        except Exception:
            pass

    work = best_arr.copy()

    # === Phase 4: Final fine coordinate descent ===
    p4_end = min(hard - 12.0, t0 + budget - 20.0)
    if p4_end > time.monotonic() + 15.0:
        try:
            work = _coord_descent(work, p4_end, consider=consider,
                                  seed=int(seed + 4000), fine=True)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
