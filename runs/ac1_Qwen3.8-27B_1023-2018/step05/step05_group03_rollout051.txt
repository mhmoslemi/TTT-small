def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.full(64, 1.0/64.0)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.full(a.size, 1.0/a.size)
    return a / s


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:

    p = np.where(np.isfinite(p), p, 0.0)

def _fft_conv(a):
    n = a.size
    L = 2*n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F*F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _fft_cross(p, w):
    n = p.size
    Lx = 3*n - 2
    size = next_fast_len(Lx)
    F = rfft(p[::-1], size)
    G = rfft(w, size)
    c = irfft(F*G, size)[:Lx]
    q = c[n-1:2*n-1]
    np.maximum(q, 0.0, out=q)
    return q


def _score_fft(a):
    n = a.size
    s = float(a.sum())
    if not math.isfinite(s) or s < 0.01:
        return math.inf
    a2 = a / s
    c = _fft_conv(a2)
    m = float(c.max())
    return float(2.0 * n * m)


def _score_exact(a):
    n = len(a)
    b = np.convolve(a, a)
    s = float(sum(a))
    if s < 0.01:
        return math.inf
    mb = float(max(b))
    return float(2.0 * n * mb / (s*s))


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0/u.size)
    return e / s


def _lse_optimize(p, stop_time, consider, stages):
    n = p.size
    conv_len = 2*n - 1
    cross_len = 3*n - 2
    conv_size = next_fast_len(conv_len)
    cross_size = next_fast_len(cross_len)

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F*F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F*G, cross_size)[:cross_len]
        q = c[n-1:2*n-1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        uu = np.asarray(uu, dtype=np.float64).ravel()
        if uu.size != n:
            return 1e100, np.zeros(n)
        uu = np.where(np.isfinite(uu), uu, 0.0)
        m = float(np.max(uu))
        e = np.exp(uu - m)
        s = float(e.sum())
        if not math.isfinite(s) or s <= 0:
            pp = np.full(n, 1.0/n)
        else:
            pp = e / s
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros(n)
        if not math.isfinite(T) or T <= 0:
            T = max(1e-14, 1e-4*cmax)
        T = max(1e-14, T)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(conv_len)
            im = int(np.argmax(c))
            w[im] = 1.0
            f = cmax
        else:
            w = exps / sum_exp
            ratio = sum_exp / float(c.size)
            if ratio <= 0:
                f = cmax
            else:
                f = cmax + T * float(math.log(ratio))
                if not math.isfinite(f):
                    f = cmax
        q = xfast(pp, w)
        gid = 2.0 * q
        gd = float(np.dot(pp, gid))
        grad = pp * (gid - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    pp = p.copy()
    u0 = np.log(np.maximum(pp, 1e-15))
    u0 -= float(np.mean(u0))
    uu = u0.copy()
    best_u = uu.copy()
    best_f = lse_func(uu, stages[0][0])[0]

    for frac, max_iter in stages:
        if time.monotonic() > stop_time - 3.0:
            break
        c = cfast(pp)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-14, frac * cmax)
        stale = 0
        while time.monotonic() < stop_time - 4.0 and stale < 3:
            opts = {"maxiter": max_iter, "gtol": 1e-10, "ftol": 1e-15,
                    "maxls": 10, "maxcor": 12, "disp": False, "maxdisp": 0}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    fval = float(res.fun) if hasattr(res, 'fun') else lse_func(x, T)[0]
                    if math.isfinite(fval) and fval < best_f - 1e-13:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        continue
            if getattr(res, "status", 1) == 0:
                break
        pu = np.max(best_u)
        e = np.exp(best_u - pu)
        pp = e / float(e.sum())
        pp = np.maximum(pp, 0.0)
        ps = float(pp.sum())
        if ps > 0:
            pp /= ps
        sc = _score_fft(pp)
        if math.isfinite(sc):
            consider(pp)
        best_u = np.log(np.maximum(pp, 1e-15))
        best_u -= float(np.mean(best_u))

    return pp


def _coord_descent(p, stop_time, G, seed, consider, max_sweeps=200):
    n = p.size
    L = 2*n - 1
    c = _fft_conv(p)
    c = np.maximum(c, 0.0)
    S = 1.0
    p = p.copy().astype(np.float64)
    cur_score = float(2.0*n*c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    if consider:
        consider(best_p)

    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)

        for pos in range(n):
            if time.monotonic() > stop_time - 1.0:
                break
            if S <= 1e-300:
                break

            i = int(order[pos])
            v = float(p[i])

            start = i
            end = min(i + n, L)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end].copy()
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < L:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

            di = 2*i - start
            Sloc = S

            # Candidate x values
            xhi = max(20.0*v, 20.0*Sloc/float(n), 0.001)
            xhi = min(xhi, max(0.1, 5.0*v))
            if not math.isfinite(xhi) or xhi <= 0:
                xhi = max(1e-8, 10.0*Sloc/float(n))

            pts = [0.0, v, xhi]

            m1 = max(16, int(G*0.4))
            minx = max(1e-10, 1e-4*v, Sloc/(5000.0*float(n)))
            if minx < xhi:
                pts.extend(np.geomspace(minx, xhi, m1).tolist())

            if v > 1e-12:
                m2 = max(10, int(G*0.3))
                factors = np.geomspace(0.3, 3.0, m2)
                pts.extend((v*factors).tolist())

            m3 = max(12, int(G*0.3))
            linhi = min(xhi, max(5.0*Sloc/float(n), 0.001))
            if linhi > 0:
                pts.extend(np.linspace(0.0, linhi, m3).tolist())
                # Extra points near current value
                if v > 0:
                    step = max(v*0.05, Sloc/(100.0*n))
                    for mult in np.linspace(-1, 1, 16):
                        pt = v + mult*step
                        if 0 <= pt <= xhi:
                            pts.append(pt)

            pts = np.unique(np.clip(np.asarray(pts, dtype=np.float64), 0.0, xhi))
            if pts.size == 0:
                pts = np.array([v])

            # Evaluate
            w = pts - v
            den = Sloc + w
            valid = den > 1e-14
            infl = 2.0 * p
            aff = base[None, :] + w[:, None] * infl[start:end][None, :]
            if 0 <= di < Ls:
                aff[:, di] = base[di] + w * infl[start+di] + w*w
            np.maximum(aff, max_const, out=aff)
            mv = np.max(aff, axis=1)
            vals = np.full(pts.size, math.inf)
            if np.any(valid):
                vals[valid] = (2.0*n*mv[valid]) / (den[valid]*den[valid])

            if not np.any(np.isfinite(vals)):
                continue
            im = int(np.argmin(vals))
            best_x = float(pts[im])
            best_val = float(vals[im])

            # Fine-tune around best
            if best_val < cur_score - 1e-12:
                if best_x <= 0:
                    lo, hi = 0.0, max(5.0*Sloc/float(n), 5.0*v, 1e-7)
                else:
                    step = max(v, Sloc/float(n), 1e-9)
                    lo = max(0.0, best_x*0.6, best_x - 2.0*step)
                    hi = min(xhi, max(best_x*1.4, best_x + 2.0*step))
                if hi > lo:
                    fine = np.linspace(lo, hi, 40)
                    if lo > 0:
                        fine = np.concatenate([fine, np.geomspace(lo, hi, 20)])
                    fine = np.unique(np.clip(fine, 0.0, xhi))
                    w2 = fine - v
                    den2 = Sloc + w2
                    valid2 = den2 > 1e-14
                    aff2 = base[None, :] + w2[:, None] * infl[start:end][None, :]
                    if 0 <= di < Ls:
                        aff2[:, di] = base[di] + w2 * infl[start+di] + w2*w2
                    np.maximum(aff2, max_const, out=aff2)
                    mv2 = np.max(aff2, axis=1)
                    vals2 = np.full(fine.size, math.inf)
                    if np.any(valid2):
                        vals2[valid2] = (2.0*n*mv2[valid2]) / (den2[valid2]*den2[valid2])
                    if np.any(np.isfinite(vals2)):
                        im2 = int(np.argmin(vals2))
                        if vals2[im2] < best_val:
                            best_x = float(fine[im2])
                            best_val = float(vals2[im2])

            if (math.isfinite(best_val) and best_val < cur_score - 1e-12
                    and 0.0 <= best_x <= xhi):
                d = best_x - v
                if abs(d) > 1e-16:
                    c[start:end] += d * 2.0 * p[start:end]
                    diag = 2*i
                    if 0 <= diag < L:
                        c[diag] += d*d
                    p[i] = best_x
                    S += d
                    cur_score = best_val
                    improved = True

        # Renormalize
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv*inv
            S = 1.0
        else:
            break

        c = _fft_conv(p)
        c = np.maximum(c, 0.0)
        desc = float(2.0*n*c.max())
        if math.isfinite(desc) and desc < best_score - 1e-14:
            best_score = desc
            best_p = p.copy()
            if consider:
                consider(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                # Perturbation to escape stagnation
                if time.monotonic() < stop_time - 30.0:
                    sigmas = [0.03, 0.06, 0.10, 0.15, 0.22]
                    sigma = sigmas[min(flat-5, len(sigmas)-1)]
                    noise = rng.normal(0.0, sigma, n)
                    # Weight perturbation toward peak region
                    peak_idx = int(np.argmax(c))
                    dist = np.abs(np.arange(n) - (peak_idx - n + 1))
                    weights = 1.0 + 0.5*np.exp(-dist.astype(float)/(0.3*n))
                    q = p * np.exp(noise * weights)
                    if sigma >= 0.10:
                        x = (np.arange(n, dtype=np.float64) + 0.5)/n
                        amp = 0.15 + 0.25*float(rng.random())
                        f1 = 1 + int(rng.integers(0, 5))
                        ph1 = 2.0*math.pi*float(rng.random())
                        q *= np.exp(amp*math.sin(2.0*math.pi*f1*x + ph1))
                    q = np.maximum(q, 0.0)
                    qs = float(q.sum())
                    if math.isfinite(qs) and qs > 0:
                        p = q / qs
                        c = _fft_conv(p)
                        c = np.maximum(c, 0.0)
                        S = 1.0
                        cur_score = float(2.0*n*c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider:
                                consider(best_p)
                    flat = 0
                else:
                    break
        sweep += 1

    if consider:
        consider(best_p)
    return best_p, best_score


def _lp_direction(a, M):
    if linprog is None or sp is None:
        return None
    n = a.size
    if n < 16 or n > 550:
        return None
    if not math.isfinite(M) or M <= 0:
        return None
    try:
        idx = np.arange(n, dtype=np.int32)
        cols = np.repeat(idx, n)
        off = np.tile(idx, n)
        rows = cols + off
        data = a[off].astype(np.float64)
        A = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
        cobj = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        res = linprog(cobj, A_ub=A, b_ub=b_ub, bounds=(0, None), method="highs")
        if res is not None and getattr(res, "success", False):
            b = np.asarray(res.x, dtype=np.float64).ravel()
            b = np.where(np.isfinite(b), b, 0.0)
            b = np.maximum(b, 0.0)
            if float(b.sum()) > 0:
                return b
    except Exception:
        pass
    return None


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.full(n, 1.0/n)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m) + 0.5)/m
        fp = np.interp((np.arange(n) + 0.5)/n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(n, 1.0/n)
    return out/s

    p2_stop = min(hard - 120, t0 + 0.60*budget)
    if time.monotonic() < p2_stop - 10:
        best_arr, _ = _coord_descent(best_arr, p2_stop, G=96, seed=seed+1,
                                     consider=consider, max_sweeps=300)

    # Phase 3: Fine L-BFGS + coordination (~20% of time)
    p3_stop = min(hard - 60, t0 + 0.82*budget)
    if time.monotonic() < p3_stop - 10:
        lse_stages_fine = [(0.001, 20), (0.0004, 20), (0.0001, 25)]
        lse_stop = min(p3_stop, time.monotonic() + min(120, 0.10*budget))
        if lse_stop > time.monotonic() + 15:
            best_arr = _lse_optimize(best_arr, lse_stop, consider, lse_stages_fine)

        cd_stop = min(p3_stop - 5, time.monotonic() + max(30, 0.08*budget))
        if cd_stop > time.monotonic() + 10:
            best_arr, _ = _coord_descent(best_arr, cd_stop, G=128, seed=seed+999,
                                         consider=consider, max_sweeps=200)

    # Phase 4: Final intensive coordinate polish
    p4_stop = hard - 5
    if time.monotonic() < p4_stop - 5:
        best_arr, _ = _coord_descent(best_arr, p4_stop, G=160, seed=seed+7777,
                                     consider=consider, max_sweeps=300)

    # Final exact validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = _normalize(parent)
            s = float(out.sum())
        out = out / s
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            ls2 = [float(x) for x in _normalize(parent)]
            sc2 = evaluate_sequence(ls2)
            if math.isfinite(sc2):
                return ls2
            return [1.0]
        return ls
    except Exception:
        return [1.0]
