            best_t = mid

    cand = (1 - best_t) * a + best_t * g
    cand = np.maximum(cand, 0.0)
    cs = cand.sum()
    if cs < 1e-15:
        return None, best_val
    cand = cand / cs
    return cand, best_val


def _geom_line_search(a, g, max_iter=40):
    n = a.size
    best_t = 0.0
    best_val = float(_eval(a))
    a_pos = np.maximum(a, 1e-15)
    g_pos = np.maximum(g, 1e-15)

    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
        cand = a_pos ** (1 - t) * g_pos ** t
        cs = cand.sum()
        if cs < 1e-15:
            continue
        cand = cand / cs
        v = float(_eval(cand))
        if v < best_val:
            best_val = v
            best_t = t

    if best_t == 0.0:
        return None, best_val

    lo = max(0.0, best_t - 0.1)
    hi = min(1.0, best_t + 0.1)
    for _ in range(max_iter):
        if hi - lo < 1e-9:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        c1 = a_pos ** (1 - m1) * g_pos ** m1
        s1 = c1.sum()
        if s1 < 1e-15:
            lo = m1
            continue
        c1 = c1 / s1
        v1 = float(_eval(c1))
        c2 = a_pos ** (1 - m2) * g_pos ** m2
        s2 = c2.sum()
        if s2 < 1e-15:
            hi = m2
            continue
        c2 = c2 / s2
        v2 = float(_eval(c2))
        if v1 < v2:
            hi = m2
        else:
            lo = m1
        mid = 0.5 * (lo + hi)
        cm = a_pos ** (1 - mid) * g_pos ** mid
        sm = cm.sum()
        if sm < 1e-15:
            continue
        cm = cm / sm
        vm = float(_eval(cm))
        if vm < best_val:
            best_val = vm
            best_t = mid

    cand = a_pos ** (1 - best_t) * g_pos ** best_t
    cs = cand.sum()
    if cs < 1e-15:
        return None, best_val
    return cand / cs, best_val


def _gradient_dir(a):
    n = a.size
    c = np.convolve(a, a)
    M = float(c.max())
    # Find all active lags
    active = np.where(c >= M * 0.999)[0]

    grad = np.zeros(n)
    for k_star in active:
        ks = int(k_star)
        for j in range(n):
            idx = ks - j
            if 0 <= idx < n:
                if ks == 2 * j:
                    grad[j] += a[j]
                else:
                    grad[j] += 2.0 * a[idx]

    # Project to zero-mean (invariance to scaling)
    grad = grad - grad.mean()
    gn = float(np.linalg.norm(grad))
    if gn < 1e-15:
        return None
    return grad / gn


def _grad_step(a, deadline):
    n = a.size
    grad = _gradient_dir(a)
    if grad is None:
        return None
    best_cand = None
    best_val = float(_eval(a))
    for alpha in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.1]:
        if time.monotonic() > deadline:
            break
        cand = a - alpha * grad
        cand = np.maximum(cand, 0.0)
        cs = cand.sum()
        if cs < 1e-12:
            continue
        cand = cand / cs
        v = float(_eval(cand))
        if v < best_val:
            best_val = v
            best_cand = cand
    return best_cand


def _optimize(a0, deadline, seed=42, max_stuck=120):
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = a.size

    best_a = a.copy()
    best_score = float(_eval(a))
    rng = np.random.default_rng(seed)
    stuck = 0
    pert_count = 0

    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        M = float(c.max())
        score = 2.0 * n * M  # sum(a)=1 so score = 2n*M

        if score < best_score * (1.0 - 1e-12):
            best_score = score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Try LP best response
        if stuck % 3 == 0 or stuck > 20:
            A = _build_A(n, a)
            g = _lp_solve(A, M * (1 + 1e-8), n, tl=1.5)
            if g is not None and g.sum() > 1e-12:
                g = g / g.sum()
                diff = float(np.max(np.abs(g - a)))

                if diff > 1e-9:
                    # Linear interpolation line search
                    cand, val = _line_search(a, g)
                    if cand is not None and val < score * (1 - 1e-11):
                        a = cand
                        stuck = 0
                        continue

                    # Geometric mean line search
                    cand_g, val_g = _geom_line_search(a, g)
                    if cand_g is not None and val_g < score * (1 - 1e-11):
                        a = cand_g
                        stuck = 0
                        continue

                    # Relaxed LP
                    g2 = _lp_solve(A, M * (1 + 5e-4), n, tl=1.0)
                    if g2 is not None and g2.sum() > 1e-12:
                        g2 = g2 / g2.sum()
                        cand2, val2 = _line_search(a, g2)
                        if cand2 is not None and val2 < score * (1 - 1e-11):
                            a = cand2
                            stuck = 0
                            continue

        # Gradient step when stuck a bit
        if stuck > 30 and stuck % 10 == 0:
            cand = _grad_step(a, deadline)
            if cand is not None:
                a = cand
                stuck = 0
                continue

        # Perturbation
        if stuck > max_stuck:
            sigma = min(0.015 + 0.008 * (stuck // max_stuck), 0.4)
            mode = pert_count % 6
            xx = np.arange(n, dtype=np.float64)

            if mode == 0:
                a *= np.exp(rng.normal(0, sigma, n))
            elif mode == 1:
                f1 = rng.uniform(1, 12)
                f2 = rng.uniform(1, 12)
                p1, p2 = rng.uniform(0, 2 * np.pi, 2)
                a *= (1.0 + sigma * np.sin(2 * np.pi * f1 * xx / n + p1) +
                      0.5 * sigma * np.cos(2 * np.pi * f2 * xx / n + p2))
                a = np.maximum(a, 1e-12)
            elif mode == 2:
                # Targeted: reduce contributions to peak lags
                c_full = np.convolve(a, a)
                M_full = float(c_full.max())
                peak_lags = np.where(c_full > 0.92 * M_full)[0]
                contrib = np.zeros(n)
                for ks in peak_lags:
                    ki = int(ks)
                    jmin = max(0, ki - n + 1)
                    jmax = min(n, ki + 1)
                    for j in range(jmin, jmax):
                        idx = ki - j
                        if 0 <= idx < n:
                            contrib[j] += a[idx]
                cmax = contrib.max()
                if cmax > 1e-15:
                    contrib = contrib / cmax
                    a *= (1.0 - 0.6 * sigma * contrib)
                    a = np.maximum(a, 1e-12)
            elif mode == 3:
                # Pair swap
                i, j = rng.integers(0, n, 2)
                if i != j:
                    tmp = a[i].copy()
                    a[i] = a[j]
                    a[j] = tmp
            elif mode == 4:
                # Block reflection
                block = max(n // 8, 1)
                start = int(rng.integers(0, max(1, n - block)))
                seg = a[start:start + block].copy()
                a[start:start + block] = seg[::-1]
            else:
                # Random coordinate jitter
                for _ in range(rng.integers(3, 10)):
                    idx = int(rng.integers(0, n))
                    a[idx] *= np.exp(rng.normal(0, 0.15))
                    a[idx] = max(a[idx], 1e-12)

            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            stuck = 0
            pert_count += 1

            if sigma > 0.35 and pert_count > 80:
                break

        if stuck > 600:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 25.0

    parent = _get_parent()
    n_p = parent.size

    best_seq = parent.copy()
    best_sc = float(_eval(best_seq))
    if not np.isfinite(best_sc):
        best_seq = np.ones(100)
        best_sc = float(_eval(best_seq))

    def try_update(cand):
        nonlocal best_seq, best_sc
        try:
            cand = np.clip(np.asarray(cand, dtype=np.float64).ravel(), 0, 1000)
            if cand.size == 0 or not np.all(np.isfinite(cand)):
                return
            sc = float(_eval(cand))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = cand.copy()
        except Exception:
            pass

    # Phase 1: Optimize parent (45% of time)
    t1_end = t0 + budget * 0.45
    try:
        r = _optimize(parent, t1_end, seed=1)
        try_update(r)
    except Exception:
        pass

    # Phase 2: Diverse initializations at parent length (20%)
    t2_end = t0 + budget * 0.65
    pn = parent / max(parent.sum(), 1e-15)
    x = np.linspace(0, 1, n_p)
    ctr = (n_p - 1) / 2.0

    inits = []
    # Gaussian profiles
    for sig in [0.1, 0.15, 0.2, 0.25, 0.35, 0.5, 0.7]:
        v = np.exp(-0.5 * ((x - 0.5) / sig) ** 2)
        inits.append(v / v.sum())
    # Symmetric peaks
    for w in [0.1, 0.15, 0.2, 0.3]:
        v = np.exp(-0.5 * ((x - w) / 0.08) ** 2) + np.exp(-0.5 * ((x - (1 - w)) / 0.08) ** 2)
        inits.append(v / v.sum())
    # Cosine
    for m in [1, 2, 3]:
        v = np.maximum(np.cos(m * np.pi * (x - 0.5)) ** 2 + 0.005, 0)
        inits.append(v / v.sum())
    # Power law from ends
    for p in [1.5, 2, 3]:
        v = x ** p + (1 - x) ** p
        inits.append(v / v.sum())
    # Random
    rng_init = np.random.default_rng(777)
    for _ in range(6):
        v = np.abs(rng_init.normal(1, 0.4, n_p)) + 0.01
        inits.append(v / v.sum())
    # Blends with parent
    for frac in [0.3, 0.5, 0.7]:
        for init in inits[:6]:
            v = frac * pn + (1 - frac) * init
            inits.append(v / v.sum())

    elapsed = time.monotonic() - t0
    remaining = max(t2_end - time.monotonic(), 5.0)
    per_init = max(2.0, remaining / max(1, len(inits)))

    for i, init in enumerate(inits):
        if time.monotonic() > t2_end - 3.0:
            break
        alloc = min(time.monotonic() + per_init, t2_end - 2.0)
        try:
            r = _optimize(init, alloc, seed=3000 + i)
            try_update(r)
        except Exception:
            pass

    # Phase 3: Try different lengths (20%)
    t3_end = t0 + budget * 0.85
    ab = np.clip(best_seq, 0, 1000).astype(np.float64)
    ab = ab / max(ab.sum(), 1e-15)
    n_best = ab.size

    length_tries = []
    for dn in [-50, -30, -10, 10, 30, 50, 80, 120]:
        nt = n_best + dn
        if 50 <= nt <= 600:
            length_tries.append(nt)
    # Also try some round numbers
    for nt in [150, 200, 250, 300, 400, 500]:
        if nt not in length_tries and 50 <= nt <= 600:
            length_tries.append(nt)
    length_tries = sorted(set(length_tries))

    x_old = np.linspace(0, 1, n_best)
    for nt in length_tries:
        if time.monotonic() > t3_end - 5.0:
            break
        x_new = np.linspace(0, 1, nt)
        init_new = np.interp(x_new, x_old, ab)
        init_new = np.clip(init_new, 0, 1000)
        sn = init_new.sum()
        if sn < 1e-10:
            continue
        init_new = init_new / sn
        alloc = min(time.monotonic() + 25, t3_end - 3.0)
        try:
            r = _optimize(init_new, alloc, seed=5000 + nt)
            try_update(r)
        except Exception:
            pass

    # Phase 4: Final polish (remaining time)
    if time.monotonic() < deadline - 5.0:
        af = np.clip(best_seq, 0, 1000).astype(np.float64)
        af = af / max(af.sum(), 1e-15)
        try:
            r = _optimize(af, deadline, seed=9999, max_stuck=80)
            try_update(r)
        except Exception:
            pass

    # Final validation
    try:
        fs = float(evaluate_sequence([float(v) for v in best_seq]))
        if not np.isfinite(fs) or fs < 0:
            raise ValueError
    except Exception:
        fallback = [1.0] * 100
        return fallback

    return [float(v) for v in best_seq]
