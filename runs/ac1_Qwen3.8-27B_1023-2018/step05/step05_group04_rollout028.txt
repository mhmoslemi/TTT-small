def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]

def _softmax_log(u, T):
    """Compute softmax weights from log-values with temperature."""
    cmax = float(np.max(u))
    if not math.isfinite(cmax):
        return np.zeros(u.shape), cmax
    scaled = (u - cmax) / T
    exps = np.exp(scaled, dtype=np.float64)
    s = float(exps.sum())
    if not math.isfinite(s) or s <= 0:
        w = np.zeros(u.shape)
        w[int(np.argmax(u))] = 1.0
        return w, cmax
    w = exps / s
    f = cmax + T * math.log(s / u.size)
    return w, f


def _lse_grad(a, w, n):
    """
    Compute gradient of LSE objective w.r.t. a (normalized, sum=1).
    w: softmax weights on convolution
    Returns: gradient in log-space (for a > 0), shape (n,)
    """
    L = 2 * n - 1
    cross_size = next_fast_len(n + L - 1)
    # Compute B^T w: for each i, sum_k w[k] * (a[k-i] + a[i-k])
    # Term 1: sum_k w[k] * a[k-i] = cross-correlation
    # Term 2: sum_k w[k] * a[i-k] = convolution-like
    # Combined: 2 * (reversed_a * w) centered
    F = rfft(a[::-1], cross_size)
    G = rfft(w, cross_size)
    prod = F * G
    c = irfft(prod, cross_size)[:n + L - 1]
    # Extract the relevant part: indices n-1 to 2n-2
    q = c[n - 1:2 * n - 1]
    q = np.maximum(q, 0.0)
    # Gradient in a-space: 2*q
    # Gradient in log-space (friction): a * (2*q - dot(a, 2*q))
    gdp = 2.0 * q
    tdp = float(np.dot(a, gdp))
    grad = a * (gdp - tdp)
    return grad

    flat = 0
    sweep = 0
    GR = (math.sqrt(5.0) - 1.0) / 2.0
    
    def eval_1d(w, base, infl, diag_idx, max_const, S_curr):
        """Evaluate score for changing a[i] by amount w."""
        s_new = S_curr + w
        if s_new <= 1e-15:
            return float('inf')
        aff_max = float(np.max(base + w * infl))
        if 0 <= diag_idx < len(base):
            dval = base[diag_idx] + w * infl[diag_idx] + w * w
            if dval > aff_max:
                aff_max = dval
        m = max(aff_max, max_const)
        return 2.0 * n * m / (s_new * s_new)
    
    max_sweeps = n_sweeps_limit
    while sweep < max_sweeps and time.monotonic() < stop_time - 3.0:
        if sweep % 500 == 0 and time.monotonic() > stop_time - 15.0:
            break
            
        improved = False
        S_curr = float(a.sum())
        
        # Order: prioritize coordinates near the current peak
        if sweep % 10 == 0:
            peak_idx = int(np.argmax(c))
            order = rng.permutation(n)
            # Sort: coordinates closer to peak first (with some randomness)
            dists = np.abs(order - peak_idx).astype(float) + 0.3 * (rng.random(n) - 0.5) * n
            order = order[np.argsort(dists)]
        else:
            order = rng.permutation(n)
        
        for ii in range(n):
            i = int(order[ii])
            if (ii % 256 == 0) and time.monotonic() > stop_time - 2.0:
                break
                
            v = float(a[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            
            base = c[start:end]
            infl = 2.0 * a[:Ls]
            diag = i  # position of diagonal in base
            
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < Ltot:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc
            
            # 1D search bounds
            x_hi = max(10.0 * v, 5.0 / n, 0.005)
            x_hi = min(x_hi, 0.3)
            x_hi = max(x_hi, 1e-10)
            
            # Grid search for good initial bracket
            G = 32
            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            ws = pts - v
            s_new = S_curr + ws
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag < Ls:
                aff[:, diag] += ws * ws
            maff = np.max(aff, axis=1)
            maff = np.maximum(maff, max_const)
            valid = s_new > 1e-12
            scores = np.full(pts.shape, float('inf'), dtype=np.float64)
            if np.any(valid):
                scores[valid] = 2.0 * n * maff[valid] / (s_new[valid] ** 2)
            
            if not np.any(np.isfinite(scores)):
                continue
            
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])
            
            # Golden section refinement
            lo = float(pts[max(0, imin - 2)])
            hi = float(pts[min(len(pts) - 1, imin + 2)])
            if hi - lo < 1e-14:
                lo = max(0.0, best_x - 1e-5)
                hi = best_x + 1e-5
            
            a_gs, b_gs = lo, hi
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)
            f_c = eval_1d(c_gs - v, base, infl, diag, max_const, S_curr)
            f_d = eval_1d(d_gs - v, base, infl, diag, max_const, S_curr)
            for _ in range(20):
                if b_gs - a_gs < 1e-13 * max(1.0, abs(best_x)):
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = eval_1d(c_gs - v, base, infl, diag, max_const, S_curr)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_d = eval_1d(d_gs - v, base, infl, diag, max_const, S_curr)
            x_mid = (a_gs + b_gs) / 2.0
            f_mid = eval_1d(x_mid - v, base, infl, diag, max_const, S_curr)
            if f_mid < best_val:
                best_x = x_mid
                best_val = f_mid
            for xp in (0.0, x_hi):
                fp = eval_1d(xp - v, base, infl, diag, max_const, S_curr)
                if fp < best_val:
                    best_x = xp
                    best_val = fp
            
            if math.isfinite(best_val) and best_val < cur_score - 1e-15 and 0.0 <= best_x <= x_hi:
                w = best_x - v
                if abs(w) > 1e-15:
                    a[i] = best_x
                    S_curr = float(a.sum())
                    aff_seg = base + w * infl
                    if 0 <= diag < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag] += w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    c[start:end] = aff_seg
                    cur_score = 2.0 * n * new_max / (S_curr * S_curr)
                    improved = True
        
        # Renormalize after each sweep
        s_new = float(a.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            a = a / s_new
            c = _conv_self(a)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-15:
                best_score = new_score
                best_a = a.copy()
                if consider:
                    try:
                        consider(best_a)
                    except Exception:
                        pass
        
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 6:
                if time.monotonic() < stop_time - 20.0:
                    kick_size = 0.01 + 0.03 * min(1.0, flat / 30.0)
                    a = a * np.exp(rng.normal(0.0, kick_size, n))
                    a = np.maximum(a, 0.0)
                    ss = float(a.sum())
                    if ss > 0:
                        a /= ss
                    c = _conv_self(a)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
        sweep += 1
    
    # Final normalization and check
    a = np.maximum(a, 0.0)
    ss = float(a.sum())
    if ss > 1e-300:
        a = a / ss
    sc = _score_normalized(a)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a.copy()
        if consider:
            try:
                consider(best_a)
            except Exception:
                pass
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 35.0  # 35s safety margin
    
    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float('inf')
    
    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_score = sc
            best_arr = a.copy()
    
    consider(parent)
    
    n_work = int(best_arr.size)
    if n_work < 128:
        n_work = 384
    if n_work > 6144:
        n_work = 4096
    
    # Resample to working length
    p_work = np.asarray(best_arr, dtype=np.float64).ravel()
    if p_work.size != n_work:
        m = p_work.size
        xp = (np.arange(m) + 0.5) / m
        fp = np.interp((np.arange(n_work) + 0.5) / n_work, xp, p_work)
        p_work = np.maximum(fp, 0.0)
    else:
        p_work = np.maximum(p_work.copy(), 0.0)
    s_w = float(p_work.sum())
    if s_w > 0:
        p_work /= s_w
    consider(p_work)
    
    # Phase 1: LSE with Adam (coarse to fine)
    lse_budget = min(180.0, 0.15 * budget)
    lse_end = t0 + lse_budget
    if lse_end > t0 + 20.0:
        try:
            p1 = _lse_adam_optimize(p_work, lse_end, consider=consider,
                                    initial_T_frac=0.08, min_T_frac=0.002)
            if p1 is not None and p1.size == n_work and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p_work = p1 / float(p1.sum())
                consider(p_work)
        except Exception:
            pass
    
    # Phase 2: Fine LSE
    fine_budget = min(120.0, 0.10 * budget)
    fine_end = min(hard - 200.0, time.monotonic() + fine_budget)
    if fine_end > time.monotonic() + 15.0:
        try:
            p2 = _lse_adam_optimize(p_work, fine_end, consider=consider,
                                    initial_T_frac=0.005, min_T_frac=0.0003)
            if p2 is not None and p2.size == n_work and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p_work = p2 / float(p2.sum())
                consider(p_work)
        except Exception:
            pass
    
    # Phase 3: Main coordinate descent (largest time budget)
    cd_end = hard - 15.0
    if cd_end > time.monotonic() + 30.0:
        p_start = np.asarray(best_arr, dtype=np.float64).ravel()
        if p_start.size != n_work:
            m = p_start.size
            xp = (np.arange(m) + 0.5) / m
            p_start = np.maximum(np.interp((np.arange(n_work) + 0.5) / n_work, xp, p_start), 0.0)
        p_start = p_start / float(p_start.sum())
        try:
            p_cd, _ = _coord_descent(p_start, cd_end, consider=consider, seed=int(seed + 7777))
            if p_cd is not None and p_cd.size > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                consider(p_cd)
        except Exception:
            pass
    
    # Phase 4: If time remains, another CD pass from best
    rem = hard - 15.0 - time.monotonic()
    if rem > 60.0 and best_score > 1.5028:
        p_f = np.asarray(best_arr, dtype=np.float64).ravel()
        if p_f.size != n_work:
            m = p_f.size
            xp = (np.arange(m) + 0.5) / m
            p_f = np.maximum(np.interp((np.arange(n_work) + 0.5) / n_work, xp, p_f), 0.0)
        p_f = p_f / float(p_f.sum())
        try:
            p_cd2, _ = _coord_descent(p_f, hard - 15.0, consider=consider, seed=int(seed + 9999))
            if p_cd2 is not None and p_cd2.size > 0 and np.all(np.isfinite(p_cd2)) and p_cd2.sum() > 0:
                consider(p_cd2)
        except Exception:
            pass
    
    # Phase 5: Final squeeze - very targeted CD
    rem = hard - 15.0 - time.monotonic()
    if rem > 30.0 and best_score > 1.5028:
        p_sq = np.asarray(best_arr, dtype=np.float64).ravel()
        if p_sq.size != n_work:
            m = p_sq.size
            xp = (np.arange(m) + 0.5) / m
            p_sq = np.maximum(np.interp((np.arange(n_work) + 0.5) / n_work, xp, p_sq), 0.0)
        p_sq = p_sq / float(p_sq.sum())
        try:
            # Reduce sweep limit for time-limited pass
            p_cd3, _ = _coord_descent(p_sq, hard - 15.0, consider=consider,
                                     seed=int(seed + 12345), n_sweeps_limit=100000)
            if p_cd3 is not None and p_cd3.size > 0 and np.all(np.isfinite(p_cd3)) and p_cd3.sum() > 0:
                consider(p_cd3)
        except Exception:
            pass
    
    # Final validation with exact evaluator
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.asarray(parent, dtype=np.float64).ravel()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in np.asarray(parent, dtype=np.float64).ravel()]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
