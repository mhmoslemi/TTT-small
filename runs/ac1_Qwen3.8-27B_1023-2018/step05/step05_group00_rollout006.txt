conv(a,a)[k] = (2n-1-k)/n^2 for k=n,...,2n-2
Max is at k = n-1: n/n^2 = 1/n.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a, size=None, clen=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if size is None:
        clen = 2 * n - 1
        size = next_fast_len(clen)
    F = rfft(a, size)
    c = irfft(F * F, size)[:clen]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a, c=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    if c is None:
        c = _conv_self(a)
    m = float(c.max()) if c.size > 0 else 0.0
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max()) if b.size > 0 else 0.0
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * a.size * m / (s * s))


def _softmax(u, T=1.0):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    m = float(np.max(u))
    e = np.exp((u - m) / T, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    return e / s


def _lse_optimize(p, stop_time, consider=None, fine_only=False):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p
    p = p / s

    u0 = np.log(np.maximum(p, 1e-300))
    u0 = u0 - float(np.mean(u0))

    conv_len = 2 * n - 1
    conv_size = next_fast_len(conv_len)

    def cfast(pp):
        F = rfft(pp, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def lse_obj(uu, T):
        pp = _softmax(uu, 1.0)
        pp = np.maximum(pp, 0.0)
        ss = float(pp.sum())
        if ss < 1e-300:
            return 1e100, np.zeros_like(uu)
        pp = pp / ss
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 1e-12:
            T = 1e-6 * cmax
        scaled = (c - cmax) / T
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size, dtype=np.float64)
            bak = int(np.argmax(c))
            w[bak] = 1.0
            f = cmax
        else:
            f = cmax + T * math.log(sum_exp / c.size)
            w = exps / sum_exp
        # Gradient: d/dpp max ≈ 2 * (pp * w convolved)
        # (pp * W)[k] = sum_j pp[j] * w[k-j]
        F = rfft(pp, conv_size)
        G = rfft(w, conv_size)
        cwp = irfft(F * G, conv_size)[:conv_len]
        q = cwp[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    def run_chunk(uu, T, end_stage):
        best_u = np.array(uu, dtype=np.float64, copy=True)
        try:
            f0, _ = lse_obj(best_u, T)
        except Exception:
            return best_u
        if not math.isfinite(f0):
            return best_u
        best_f = float(f0)
        stale = 0
        while time.monotonic() < end_stage - 1.0:
            maxit = 50 if n <= 2048 else 30
            opts = {
                "maxiter": maxit,
                "gtol": 1e-14,
                "ftol": 1e-16,
                "maxls": 30,
                "maxcor": 40,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(lse_obj, best_u, args=(T,),
                               method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == uu.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 4:
                            break
            if getattr(res, "status", 1) == 0:
                break
        return best_u

    if fine_only:
        stages = ((0.003, 0.25), (0.001, 0.35), (0.0003, 0.25), (0.00008, 0.15))
    else:
        stages = (
            (0.2, 0.06),
            (0.08, 0.08),
            (0.03, 0.10),
            (0.01, 0.12),
            (0.003, 0.14),
            (0.001, 0.15),
            (0.0003, 0.14),
            (0.00008, 0.12),
            (0.00002, 0.09),
        )

    start = time.monotonic()
    avail = float(stop_time - start - 10.0)
    if avail <= 5.0:
        return p

    best_p = p.copy()
    best_sc = _score_fast(p)
    total_w = float(sum(w for _, w in stages))
    if total_w <= 0:
        total_w = 1.0

    cum = 0.0
    u = u0.copy()
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            cmax = 1.0
        T = max(1e-12, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_cur = _softmax(u, 1.0)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        sc_cur = _score_fast(p_cur)
        if not math.isfinite(sc_cur):
            p = best_p.copy()
            u = np.log(np.maximum(p, 1e-300))
            u -= float(np.mean(u))
            continue
        if sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif sc_cur > best_sc + 0.002:
            p = best_p.copy()
        else:
            p = p_cur.copy()
        u = np.log(np.maximum(p, 1e-300))
        u -= float(np.mean(u))
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, max_sweeps=200000, seed=12345, consider=None):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p, math.inf
    p = p / s

    G = 36
    Gf = 30
    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.4 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue
            base = c[start_i:end_i]
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start_i > 0:
                ml = float(c[:start_i].max())
                if ml > max_const:
                    max_const = ml
            if end_i < Ltot:
                mr = float(c[end_i:].max())
                if mr > max_const:
                    max_const = mr

            avg = S / n
            x_hi = max(30.0 * avg, 0.03 * S, 1e-12)
            x_hi = min(x_hi, 12.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = avg

            pts = np.linspace(0.0, x_hi, G, dtype=np.float64)
            if 0.0 <= v <= x_hi:
                idx_v = int(np.argmin(np.abs(pts - v)))
                if idx_v < G:
                    pts[idx_v] = v
                else:
                    tmp = np.zeros(G + 1)
                    tmp[:G] = pts
                    tmp[G] = v
                    pts = tmp

            diag_offset = i - start_i  # = i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                dval = base[diag_offset] + w * infl[diag_offset] + w * w
                maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Fine search around best
            dw = x_hi / G
            lo = max(0.0, best_x - 2.0 * dw)
            hi = min(x_hi, best_x + 2.0 * dw)
            if hi <= lo:
                hi = lo + 1e-12
            fine = np.linspace(lo, hi, Gf, dtype=np.float64)
            if 0.0 <= best_x <= x_hi:
                fine = np.concatenate((fine, [v]))
                fine = np.unique(fine)
            vals2 = eval_xs(fine)
            if np.any(np.isfinite(vals2)):
                imin2 = int(np.argmin(vals2))
                vx2 = float(fine[imin2])
                vv2 = float(vals2[imin2])
                if vv2 < best_val:
                    best_x = vx2
                    best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 5e-14 and 0.0 <= best_x <= 12.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start_i:end_i] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        # Renormalize
        if S_total > 1e-300 and math.isfinite(S_total):
            inv = 1.0 / S_total
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 6.0:
                    # Perturb to escape local minimum
                    sigma = 0.02 if n > 1000 else 0.03
                    p *= np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score:
                        best_score = cur_score
                        best_p = p.copy()
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _peak_shave(p, stop_time, consider=None, n_iter=50):
    """Targeted peak reduction: identify active lags and reduce contributing coefficients."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    conv_len = 2 * n - 1
    conv_size = next_fast_len(conv_len)

    best_p = p.copy()
    best_sc = _score_fast(p)

    for it in range(n_iter):
        if time.monotonic() > stop_time - 2.0:
            break
        c = _conv_self(p, conv_size, conv_len)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            break

        # Find active lags (within 1% of max)
        threshold = cmax * 0.99
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            break

        # Build gradient: for each active lag k, d(c[k])/d(a[j]) = 2*a[k-j]
        grad_shave = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    grad_shave[j] += p[kj]
        grad_shave *= 2.0

        # Normalize: multiply gradient by p to get relative gradient
        rel_grad = grad_shave.copy()
        rg_mean = float(np.mean(rel_grad))
        scale = cmax  # normalize

        # Multiplicative update: p[j] *= exp(-eta * rel_grad[j])
        eta = 0.05
        p_new = p * np.exp(-eta * rel_grad / max(scale, 1e-12))
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if ss < 1e-300:
            break
        p_new /= ss

        sc_new = _score_fast(p_new)
        if math.isfinite(sc_new) and sc_new < best_sc:
            best_sc = sc_new
            best_p = p_new.copy()
            p = p_new
            if consider:
                try:
                    consider(best_p)
                except Exception:
                    pass
        else:
            # Try smaller step
            for eta2 in [0.02, 0.01, 0.005]:
                p_new2 = p * np.exp(-eta2 * rel_grad / max(scale, 1e-12))
                p_new2 = np.maximum(p_new2, 0.0)
                ss2 = float(p_new2.sum())
                if ss2 < 1e-300:
                    continue
                p_new2 /= ss2
                sc_new2 = _score_fast(p_new2)
                if math.isfinite(sc_new2) and sc_new2 < best_sc:
                    best_sc = sc_new2
                    best_p = p_new2.copy()
                    p = p_new2
                    if consider:
                        try:
                            consider(best_p)
                        except Exception:
                            pass
                    break

    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 20.0
    if hard <= t0 + 5.0:
        hard = t0 + max(3.0, budget - 2.0)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(128, dtype=np.float64) / 128.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_arr = a.copy()
            best_score = sc

    n_par = int(best_arr.size)

    # === Phase 1: LSE optimization (coarse) ===
    phase1_end = min(hard - 120.0, t0 + 0.22 * budget)
    if phase1_end > t0 + 20.0 and n_par >= 64:

    phase5_end = min(hard - 60.0, t0 + 0.82 * budget)

            work, _ = _coord_polish(work, phase5_end, max_sweeps=400000,

    # === Phase 6: Peak shaving again ===
    phase6_end = min(hard - 50.0, time.monotonic() + 25.0)

    phase7_end = min(hard - 20.0, t0 + budget - 15.0)

        half = (phase7_end - time.monotonic()) * 0.45

                    consider(p7)
            except Exception:

        work = best_arr.copy()
        work = work / float(work.sum())

        coord7_end = phase7_end
        if coord7_end > time.monotonic() + 10.0:
            try:
                work, _ = _coord_polish(work, coord7_end, max_sweeps=200000,
                                        seed=int(seed + 987654), consider=consider)
            except Exception:
                pass

    # === Final validation and return ===
    try:

        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:

        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
