            pass

    # ===== Phase 1: Optimize parent with LP method =====
    t1 = t0 + min(budget * 0.35, 450.0)
    try:
        r = _optimize_lp(parent, t1, seed=1, max_iters=2000)
        try_update(r)
    except Exception:
        pass

    # ===== Phase 2: Second pass on current best (different seed) =====
    t2 = t0 + min(budget * 0.50, 650.0)
    if time.monotonic() < t2:
        ab = np.clip(best_seq, 0, 1000).astype(np.float64)
        ab = ab / max(ab.sum(), 1e-15)
        try:
            r = _optimize_lp(ab, t2, seed=123, max_iters=1500)
            try_update(r)
        except Exception:
            pass

    # ===== Phase 3: Alternative initializations =====
    t3 = t0 + min(budget * 0.70, 910.0)
    if time.monotonic() < t3:
        x = np.linspace(0, 1, n_parent)
        pn = parent / max(parent.sum(), 1e-15)
        inits = []

        # Smooth shapes
        for sigma in [0.08, 0.12, 0.18, 0.25, 0.35, 0.5]:
            v = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2) + 0.001
            inits.append(v / v.sum())

        # Off-center Gaussians
        for c in [0.25, 0.35, 0.65, 0.75]:
            for w in [0.1, 0.15]:
                v = np.exp(-0.5 * ((x - c) / w) ** 2) + 0.001
                inits.append(v / v.sum())

        # Cosine powers
        for m in [1, 2, 3, 4, 5]:
            v = np.maximum(np.cos(m * np.pi * (x - 0.5)), 0.0) ** 2 + 0.001
            inits.append(v / v.sum())

        # Blend with parent
        for frac in [0.4, 0.6, 0.8]:
            for init in inits[:4]:
                v = frac * pn + (1.0 - frac) * init
                inits.append(v / v.sum())

        # Seeded random
        rng_init = np.random.default_rng(777)
        for _ in range(6):
            v = np.abs(rng_init.normal(1.0, 0.4, n_parent)) + 0.001
            inits.append(v / v.sum())

        remaining = max(t3 - time.monotonic(), 3.0)
        per_init = max(4.0, remaining / max(1, len(inits)))

        for i, init in enumerate(inits):
            if time.monotonic() > t3 - 5.0:
                break
            alloc = min(time.monotonic() + per_init, t3 - 3.0)
            try:
                r = _optimize_lp(init, alloc, seed=500 + i)
                try_update(r)
            except Exception:
                pass

    # ===== Phase 4: Different sequence lengths =====
    t4 = t0 + min(budget * 0.85, 1105.0)
    if time.monotonic() < t4:
        x_old = np.linspace(0, 1, n_parent)
        pn = parent / max(parent.sum(), 1e-15)
        # Try shorter and longer
        lengths = [n_parent - 60, n_parent - 40, n_parent - 20,
                   n_parent + 30, n_parent + 60, n_parent + 100,
                   n_parent + 150, n_parent + 200]
        for nt in lengths:
            if time.monotonic() > t4 - 20.0:
                break
            if nt < 30 or nt > 700:
                continue
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, pn)
            init_new = np.clip(init_new, 0, 1000)
            s = init_new.sum()
            if s < 1e-10:
                continue
            init_new = init_new / s
            alloc = min(time.monotonic() + 35.0, t4 - 10.0)
            try:
                r = _optimize_lp(init_new, alloc, seed=2000 + nt)
                try_update(r)
            except Exception:
                pass

    # ===== Phase 5: Final polish on best =====
    if time.monotonic() < deadline - 10.0:
        af = np.clip(best_seq, 0, 1000).astype(np.float64)
        af = af / max(af.sum(), 1e-15)
        try:
            r = _optimize_lp(af, deadline, seed=9999, max_iters=3000)
            try_update(r)
        except Exception:
            pass

    # ===== Final validation =====
    try:
        seq_out = [float(v) for v in best_seq]
        fs = float(evaluate_sequence(seq_out))
        if not np.isfinite(fs) or fs < 0:
            raise ValueError("invalid")
    except Exception:
        seq_out = [1.0] * 100
        try:
            if not np.isfinite(float(evaluate_sequence(seq_out))):
                seq_out = [1.0]
        except Exception:
            seq_out = [1.0]

    return seq_out
