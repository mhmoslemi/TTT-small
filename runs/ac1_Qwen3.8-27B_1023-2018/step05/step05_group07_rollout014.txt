```python
def build_conv_mat(n, a):
    """Build (2n-1) x n matrix where A[k,j] = a[k-j] if valid."""
    k = np.arange(2*n-1)[:, None]
    j = np.arange(n)[None, :]
```

```python
def solve_minimax_lp(A, n):
    """Solve: min t s.t. A@q <= t*1, sum(q)=1, q>=0, t>=0"""
    # Variables: q[0..n-1], t
    # c = [0, ..., 0, 1] (minimize t)
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    # A_ub @ [q, t] <= b_ub
    # A[k,j]*q[j] - t <= 0 for all k
    # -sum(q) <= -1 and sum(q) <= 1 (equality)
    A_ub = np.zeros((2*n-1+2, n+1))
    # Inequality constraints: A[k,:] @ q - t <= 0
    A_ub[:2*n-1, :n] = A
    A_ub[:2*n-1, -1] = -1
    b_ub = np.zeros(2*n-1)
    # sum(q) = 1: -sum(q) <= -1 and sum(q) <= 1
    A_ub[2*n-1, :n] = -1
    b_ub[2*n-1] = -1
    A_ub[2*n, :n] = 1
    b_ub[2*n] = 1
    
    bounds = [(0, None)]*n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 3.0})
    if res.status == 0:
        return res.x[:n], res.x[-1]
    return None, None
```
