import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)

    if float(p.sum()) < 1e-12:
        p = np.ones(3072, dtype=np.float64)
    return p


def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(np.asarray(c, dtype=np.float64), 0.0)


def _score_fft(a):
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _coord_descent(p, stop_time, consider, seed=42):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    S = float(p.sum())
    if S < 1e-300 or n == 0:
        return p, math.inf
    p = p / S
    S = 1.0
    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    flat = 0
    sweep = 0
    GOLD = 0.6180339887498949

    def eval_1d(i, v, w):
        start = i
        end = start + n
        denom = S + w
        if denom < 1e-12:
            return math.inf
        base = c[start:end]
        infl = 2.0 * p[:n]
        aff = base + w * infl
        diag = 2 * i - start
        if 0 <= diag < n:
            aff[diag] = base[diag] + w * (2.0 * v) + w * w
        mv = float(aff.max())
        max_const = 0.0
        if start > 0:
            max_const = max(max_const, float(c[:start].max()))
        if end < Ltot:
            max_const = max(max_const, float(c[end:].max()))
        mv = max(mv, max_const)
        return 2.0 * n * mv / (denom * denom)

    while time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for idx in order:

            if time.monotonic() > stop_time - 0.3:
                break

            v = float(p[i])
            start = i
            end = start + n
            diag = i  # 2i - start = 2i - i = i

            # Compute max_const
            max_const = 0.0
            if start > 0:

            if end < Ltot:

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_self(p)
    sc = 2.0 * n * float(c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _lse_optimize(p, stop_time, consider=None):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p
    p = p / s

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return np.maximum(c, 0.0)

    def lse_func(uu, T):
        uu_max = float(np.max(uu))
        e = np.exp(uu - uu_max)
        ss = float(e.sum())
        if not math.isfinite(ss) or ss <= 0:
            return 1e100, np.zeros_like(uu)
        pp = e / ss
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / conv_len))
            w = exps / sum_exp

        # Gradient: grad_u[i] = pp[i]*(1-pp[i]) * 2 * h[i]
        # where h[i] = sum_{j=0}^{n-1} w[i+j]*pp[j]
        # h = (w * pp_rev)[i+n-1] via FFT
        x_size = next_fast_len(3 * n - 2)
        Fw = rfft(w, x_size)
        Frev = rfft(pp[::-1], x_size)
        conv = irfft(Fw * Frev, x_size)
        h = conv[n - 1:2 * n - 1]
        gdp = 2.0 * h
        grad_u = pp * (1.0 - pp) * gdp
        return float(f), np.where(np.isfinite(grad_u), grad_u, 0.0)

    def p_to_u(pp):
        ll = np.log(np.maximum(pp, 1e-300))
        return ll - float(np.mean(ll))

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score_fft(p)

    stages = (
        (0.12, 1.0),
        (0.05, 1.0),
        (0.02, 1.5),
        (0.008, 1.5),
        (0.003, 1.0),
        (0.001, 1.0),
    )
    total_w = sum(wt for _, wt in stages)

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        stage_t = max(3.0, (wt / total_w) * (stop_time - now - 5.0))
        stage_end = now + stage_t
        cmax_cur = float(cfast(p).max())
        T = max(1e-15, frac * cmax_cur)
        stale = 0

        while time.monotonic() < stage_end - 1.0 and stale < 5:
            opts = {"maxiter": 40, "gtol": 1e-10, "ftol": 1e-16,
                    "maxls": 30, "maxcor": 30, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                stale += 1
                continue

            if res.x is not None and np.all(np.isfinite(res.x)):
                p_cur = np.exp(res.x - np.max(res.x))
                p_cur /= float(p_cur.sum())
                sc_cur = _score_fft(p_cur)
                if math.isfinite(sc_cur) and sc_cur < best_sc:
                    best_sc = sc_cur
                    best_p = p_cur.copy()
                    p = p_cur.copy()
                    u = p_to_u(p)
                    stale = 0
                else:
                    stale += 1
            if getattr(res, "status", 1) == 0:
                break

    return best_p
