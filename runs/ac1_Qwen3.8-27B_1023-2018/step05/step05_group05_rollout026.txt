                if sc_check < best_score - 1e-14:
                    best_score = sc_check
                    best_p = p.copy()
                    if consider:
                        try:
                            consider(best_p)
                        except Exception:
                            pass

        if improved:
            flat_count = 0
        else:
            flat_count += 1
            if flat_count >= max_flat:
                # Perturb
                if time.monotonic() < stop_time - 10.0:
                    noise = rng.normal(0, 0.03, n)
                    p = np.maximum(p * np.exp(noise), 0)
                    ss = float(p.sum())
                    if ss > 1e-300:
                        p /= ss
                        S = 1.0
                        c = _conv_self(p)
                        cur_score = 2.0 * n * float(c.max())
                        flat_count = 0
                    else:
                        break
                else:
                    break

        sweep += 1

    # Final
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_self(p)
    sc = 2.0 * n * float(c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 5.0:
        hard = t0 + max(5.0, budget - 2.0)

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = math.inf
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_arr]))
    except Exception:
        pass
    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(128)
        best_score = float(evaluate_sequence([float(x) for x in best_arr]))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            sc = float(evaluate_sequence([float(x) for x in a]))
            if math.isfinite(sc) and sc < best_score - 1e-15:
                a = a / s
                best_arr = a.copy()
                best_score = sc
        except Exception:
            pass

    n_best = int(best_arr.size)

    # Phase 1: LSE optimization
    now = time.monotonic()
    lse_time = min(200.0, 0.20 * budget)
    lse_stop = t0 + lse_time
    if lse_stop > now + 30.0:
        try:
            p1 = _lse_phase(best_arr, lse_stop, consider=consider, seed=int(seed))
            if p1 is not None and p1.size == n_best and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    # Phase 2: Main coordinate descent
    now = time.monotonic()
    cd_time = min(600.0, 0.45 * budget)
    cd_stop = min(t0 + cd_time, hard - 100.0)
    if cd_stop > now + 30.0:
        try:
            p2, _ = _coord_descent(best_arr, cd_stop, consider=consider, seed=int(seed + 101))
            if p2 is not None and p2.size == n_best and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p2 = p2 / float(p2.sum())
                consider(p2)
        except Exception:
            pass

    # Phase 3: Second LSE at lower temperature
    now = time.monotonic()
    rem = hard - 20.0 - now
    if rem > 50.0:
        lse2_stop = now + max(30.0, 0.3 * rem)
        try:
            p3 = _lse_phase(best_arr, lse2_stop, consider=consider, seed=int(seed + 202))
            if p3 is not None and p3.size == n_best and np.all(np.isfinite(p3)) and p3.sum() > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)
        except Exception:
            pass

    # Phase 4: Final coordinate descent polish
    now = time.monotonic()
    cd2_stop = hard - 15.0
    if cd2_stop - now > 30.0:
        try:
            p4, _ = _coord_descent(best_arr, cd2_stop, consider=consider, seed=int(seed + 999))
            if p4 is not None and p4.size == n_best and np.all(np.isfinite(p4)) and p4.sum() > 0:
                p4 = p4 / float(p4.sum())
                consider(p4)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
