        # = 0+0+3*w[2]+2*w[1]+1*w[0] = 3*w[2]+2*w[1]+w[0]
        # 2*conv[4] = 2*(3*w[2]+2*w[1]+w[0]) ✓
        
        # grad[1] should be 2*sum_j w[1+j]*a[j] = 2*(w[1]*a[0]+w[2]*a[1]+w[3]*a[2]) = 2*(w[1]+2*w[2]+3*w[3])
        # Using formula: grad[1] = 2*conv[1+2*3-2] = 2*conv[5]
        # conv[5] = a_flip[1]*w[4]+a_flip[2]*w[3]+a_flip[3]*w[2]+a_flip[4]*w[1]+(a_flip[5]*w[0])
        # a_flip has length 5, so a_flip[5] doesn't exist. conv[5] uses m from max(0,5-4)=1 to min(4,5)=4
        # = a_flip[1]*w[4]+a_flip[2]*w[3]+a_flip[3]*w[2]+a_flip[4]*w[1]
        # = 0*w[4]+3*w[3]+2*w[2]+1*w[1] = 3*w[3]+2*w[2]+w[1] ✓
        
        # Great, so grad[i] = 2*conv[i+2n-2] where conv = convolve(a_flip, w)
        # a_flip = a_pad[::-1] where a_pad = [a, 0, ..., 0] of length 2n-1
        
        # For n=3072, conv has length 4*3072-3 = 12285. This is fine.
        
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    stages = (
        (0.10, 1.0),
        (0.03, 1.0),
        (0.01, 1.0),
        (0.003, 1.0),
        (0.001, 1.0),
        (0.0003, 1.0),
        (0.0001, 1.0),
        (0.00003, 1.0),
    )

    start = time.monotonic()
    avail = max(0.0, float(stop_time) - start - 3.0)
    if avail <= 5.0:
        return p

    best_p = p.copy()
    best_sc = _score_fft(p)
    total_w = float(sum(w for _, w in stages))
    rng = np.random.default_rng(seed)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 2.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 1.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue
        c = np.convolve(p, p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)

        best_u = u.copy()
        f0, _ = lse_func(best_u, T)
        if not math.isfinite(f0):
            continue
        best_f = f0
        stale = 0

        while time.monotonic() < end_stage - 0.5:
            maxit = 30
            opts = {"maxiter": maxit, "gtol": 1e-12, "ftol": 1e-16,
                    "maxls": 30, "maxcor": 20, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-15:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            best_u = best_u + rng.normal(0, 0.003, best_u.size)
                            continue
            if getattr(res, "status", 1) == 0 and stale < 2:
                if time.monotonic() < end_stage - 1.0:
                    best_u = best_u + rng.normal(0, 0.002, best_u.size)
                else:
                    break

        p_cur = _softmax_fn(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        else:
            p = best_p.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, G=None, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    if G is None:
        if n <= 500:
            G = 64
        elif n <= 1500:
            G = 48
        elif n <= 4000:
            G = 40
        else:
            G = 32
    G = int(max(16, G))

    Ltot = 2 * n - 1
    c = np.convolve(p, p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < 500000 and time.monotonic() < stop_time - 0.5:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.2:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(20.0 * v, 40.0 * S / float(n), 0.02 * S, 1e-14)
            x_hi = min(x_hi, 8.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, S / float(n))

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag_i = i

            # Vectorized evaluation
            w = pts - v
            denom = S + w
            valid = (denom > 1e-15) & (w >= -v)  # a[i] >= 0 means pts >= 0
            vals = np.full(pts.shape, math.inf, dtype=np.float64)
            if np.any(valid):
                wv = w[valid]
                aff = base[None, :] + wv[:, None] * infl[None, :]
                dval = base[diag_i] + wv * infl[diag_i] + wv * wv
                aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                dv = denom[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Fine grid around best
            if G > 4 and best_x > 0 and best_x < x_hi:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 32, dtype=np.float64)
                fine = np.concatenate((fine, [best_x]))
                fine = np.unique(fine)
                w2 = fine - v
                denom2 = S + w2
                valid2 = (denom2 > 1e-15) & (fine >= 0)
                vals2 = np.full(fine.shape, math.inf, dtype=np.float64)
                if np.any(valid2):
                    wv2 = w2[valid2]
                    aff2 = base[None, :] + wv2[:, None] * infl[None, :]
                    dval2 = base[diag_i] + wv2 * infl[diag_i] + wv2 * wv2
                    aff2 = np.maximum(aff2, dval2[:, None])
                    aff2 = np.maximum(aff2, max_const)
                    mv2 = aff2.max(axis=1)
                    dv2 = denom2[valid2]
                    vals2[valid2] = (2.0 * n * mv2) / (dv2 * dv2)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val - 1e-15:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 10.0 * S:
                w_delta = best_x - v
                if abs(w_delta) > 1e-15:
                    p[i] = best_x
                    S_total += w_delta
                    c[start:end] += w_delta * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w_delta * w_delta
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = np.convolve(p, p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 5.0:
                    sigma = 0.015 + 0.015 * min(flat, 15) / 15.0
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    else:
                        break
                    c = np.convolve(p, p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    flat = 0
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        sc = float(_score_fft(p))
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 20.0
    if hard <= t0 + 2.0:
        hard = t0 + max(2.0, budget * 0.9)

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    n_parent = int(best_arr.size)

    # Phase 1: LSE warmup on parent
    p1_end = t0 + 0.20 * (budget - 20.0)
    if p1_end > t0 + 10:
        p1 = _lse_optimize(best_arr, p1_end, consider=consider, seed=int(seed+1))

    # Phase 2: Coordinate polish (main effort)
    p2_end = t0 + 0.65 * (budget - 20.0)
    rem2 = p2_end - time.monotonic()
    if rem2 > 15:
        p2, _ = _coord_polish(best_arr, min(p2_end, t0+budget-25.0), seed=int(seed+2), consider=consider)
        if p2 is not None and p2.size == n_parent:
            consider(p2)

    # Phase 3: LSE refinement on polished result
    p3_end = t0 + 0.80 * (budget - 20.0)
    rem3 = p3_end - time.monotonic()
    if rem3 > 10:
        p3 = _lse_optimize(best_arr, min(p3_end, t0+budget-30.0), consider=consider, seed=int(seed+3))

    # Phase 4: Final coordinate polish
    p4_end = t0 + budget - 25.0
    rem4 = p4_end - time.monotonic()
    if rem4 > 10:
        p4, _ = _coord_polish(best_arr, min(p4_end, t0+budget-22.0), seed=int(seed+4), consider=consider)
        if p4 is not None and p4.size == n_parent:
            consider(p4)

    # Phase 5: Random perturb + quick polish if time remains
    rem5 = t0 + budget - 23.0 - time.monotonic()
    if rem5 > 20:
        rng5 = np.random.default_rng(seed + 999)
        for attempt in range(3):
            rem5 = t0 + budget - 23.0 - time.monotonic()
            if rem5 < 10:
                break
            pert = best_arr * np.exp(rng5.normal(0, 0.02, n_parent))
            pert = np.maximum(pert, 0.0)
            ps = float(pert.sum())
            if ps > 0:
                pert /= ps
                consider(pert)
            sub_end = time.monotonic() + rem5 * 0.4
            if sub_end > time.monotonic() + 5:
                pp, _ = _coord_polish(pert, sub_end, seed=int(seed+100+attempt*7), consider=consider)
                if pp is not None and pp.size == n_parent:
                    consider(pp)

        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum())
