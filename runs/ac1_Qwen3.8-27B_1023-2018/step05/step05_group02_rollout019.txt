        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:
        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(3072, dtype=np.float64)
    return p

def _conv_fft(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score_arr(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_fft(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))

def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0/u.size)
    return e / s

def _lbfgs_phase(p, stop_time, consider):
    """LSE-smoothed L-BFGS optimization."""
    n = p.size
    p = np.maximum(p, 0.0).copy()
    s = float(p.sum())
    if s < 1e-12:
        return p
    p = p / s
    
    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))
    
    u = p_to_u(p)
    conv_size = next_fast_len(2*n - 1)
    cross_size = next_fast_len(3*n - 2)
    conv_len = 2*n - 1
    cross_len = 3*n - 2
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c
    
    def grad_cross(pp, ww):
        """Compute (w ⊛ a)[i] for i=0..n-1."""
        W = rfft(ww, cross_size)
        A = rfft(pp[::-1], cross_size)
        R = irfft(W * A, cross_size)[:cross_len]
        return R[n-1:2*n-1]
    
    def lse_obj(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -400, 400)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
        else:
            w = exps / sum_exp
        f = cmax + T * math.log(sum_exp / c.size)
        if not math.isfinite(f):
            f = cmax
        
        q = grad_cross(pp, w)
        tdp = float(np.dot(pp, q))
        grad = pp * (2.0*q - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad
    
    # Stages with decreasing temperature
    stages = [
        (0.05, 0.15),
        (0.02, 0.20),
        (0.008, 0.20),
        (0.003, 0.20),
        (0.001, 0.25),
    ]
    
    start = time.monotonic()
    total_w = sum(w for _, w in stages)
    cum = 0.0
    best_p = p.copy()
    best_sc = _score_arr(p)
    
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3:
            break
        cum += wt
        end_stage = min(stop_time - 2, start + (stop_time - start - 5) * (cum / total_w))
        if end_stage <= now + 2:
            continue
        
        c = cfast(p)
        cmax = float(c.max())
        T = max(1e-10, frac * cmax)
        
        best_u = u.copy()
        best_f = 1e100
        try:
            f0, _ = lse_obj(u, T)
            if math.isfinite(f0):
                best_f = f0
        except Exception:
            pass
        
        stale = 0
        while time.monotonic() < end_stage - 1:
            try:
                res = minimize(lse_obj, best_u, args=(T,), method='L-BFGS-B', jac=True,
                              options={'maxiter': 40, 'gtol': 1e-12, 'ftol': 1e-15,
                                       'maxcor': 25, 'maxdisp': 0, 'disp': False})
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)) and res.x.size == u.size:
                fv = float(res.fun)
                if math.isfinite(fv) and fv < best_f - 1e-16:
                    best_f = fv
                    best_u = res.x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if getattr(res, 'status', 1) == 0:
                break
        
        u = best_u
        p_new = _softmax(u)
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if ss > 1e-300:
            p_new /= ss
        sc_new = _score_arr(p_new)
        if math.isfinite(sc_new):
            if sc_new < best_sc:
                best_p = p_new.copy()
                best_sc = sc_new
                p = p_new.copy()
            u = p_to_u(p)
            try:
                consider(best_p)
            except Exception:
                pass
    
    return best_p
