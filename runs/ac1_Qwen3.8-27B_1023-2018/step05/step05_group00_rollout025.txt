    p = np.maximum(p, 0.0)
    S_final = float(p.sum())
    if S_final > 1e-300:
        c_f = _conv_self(p)
        sf = float(2.0 * n * c_f.max() / (S_final * S_final))
        if math.isfinite(sf) and sf < best_score:
            best_score = sf
            best_p = p.copy()
    return best_p, best_score


def _lse_optimize(p, stop_time, consider=None, fine_only=False):
    """L-BFGS-B on log-sum-exp smoothed objective."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = conv_size

    def cfast(a):
        F = rfft(a, conv_size)
        cn = irfft(F * F, conv_size)[:conv_len]
        np.maximum(cn, 0.0, out=cn)
        return cn

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        cn = irfft(F * G, cross_size)[:conv_len]
        q = cn[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def softmax(uu):
        m = float(np.max(uu))
        e = np.exp(uu - m)
        ss = float(e.sum())
        if not math.isfinite(ss) or ss <= 0:
            e = np.full(uu.size, 1.0 / uu.size)
            ss = 1.0
        return e / ss

    def lse_func(uu, T):
        pp = softmax(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * max(cmax, 1e-300))
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    if fine_only:
        stages = ((0.001, 0.30), (0.0003, 0.35), (0.00008, 0.35))
    else:
        stages = (
            (0.10, 0.08),
            (0.04, 0.10),
            (0.012, 0.12),
            (0.003, 0.15),
            (0.0008, 0.18),
            (0.0002, 0.17),
            (0.00005, 0.15),
            (0.00001, 0.15),
        )

    start = time.monotonic()
    avail = max(0.0, float(stop_time) - start - 8.0)
    if avail < 10.0:
        return p
    total_w = sum(wt for _, wt in stages)
    cum = 0.0
    best_u = u.copy()
    best_f = math.inf
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:
            continue
        c_cur = cfast(softmax(best_u))
        cmax = float(c_cur.max()) if c_cur.size else 1.0
        T = max(1e-300, frac * cmax)
        stale = 0
        while time.monotonic() < end_stage - 1.0 and stale < 4:
            maxit = 30 if n > 2000 else 50
            opts = {"maxiter": maxit, "gtol": 1e-14, "ftol": 1e-16,
                    "maxls": 25, "maxcor": 30, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_f:
                    best_f = fval
                    best_u = np.array(res.x, dtype=np.float64, copy=True)
                    stale = 0
                else:
                    stale += 1
            if getattr(res, "status", 1) == 0:
                break
    pp = softmax(best_u)
    pp = np.maximum(pp, 0.0)
    ss = float(pp.sum())
    if ss > 1e-300:
        pp /= ss
    if consider is not None:
        try:
            consider(pp)
        except Exception:
            pass
    return pp


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 25.0

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_general(best_arr))
    except Exception:
        best_score = math.inf
    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(128, dtype=np.float64) / 128.0
        best_score = float(_score_general(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_general(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_arr = a.copy()
            best_score = sc

    n_par = int(best_arr.size)

    # Phase 1: Initial coordinate descent
    end1 = min(hard_deadline - 500.0, t0 + 150.0)
    if end1 > t0 + 20.0:
        work = best_arr.copy()
        work /= float(work.sum())
        try:
            work, _ = _coord_descent(work, end1, seed=int(seed), consider=consider)
        except Exception:
            pass

    # Phase 2: LSE optimization (coarse to fine)
    end2 = min(hard_deadline - 400.0, t0 + 400.0)
    if end2 > time.monotonic() + 30.0:
        work = best_arr.copy()
        work /= float(work.sum())
        try:
            work = _lse_optimize(work, end2, consider=consider, fine_only=False)
            if work is not None and work.size == n_par:
                work = np.maximum(work, 0.0)
                ws = float(work.sum())
                if ws > 0:
                    work /= ws
                    consider(work)
        except Exception:
            pass

    # Phase 3: Coordinate descent after LSE
    end3 = min(hard_deadline - 300.0, t0 + 700.0)
    if end3 > time.monotonic() + 30.0:
        work = best_arr.copy()
        work /= float(work.sum())
        try:
            work, _ = _coord_descent(work, end3, seed=int(seed + 1000), consider=consider)
        except Exception:
            pass

    # Phase 4: Fine LSE
    end4 = min(hard_deadline - 200.0, t0 + 950.0)
    if end4 > time.monotonic() + 30.0:
        work = best_arr.copy()
        work /= float(work.sum())
        try:
            work = _lse_optimize(work, end4, consider=consider, fine_only=True)
            if work is not None and work.size == n_par:
                work = np.maximum(work, 0.0)
                ws = float(work.sum())
                if ws > 0:
                    work /= ws
                    consider(work)
        except Exception:
            pass

    # Phase 5: Final coordinate descent polish
    end5 = min(hard_deadline - 30.0, t0 + budget - 40.0)
    if end5 > time.monotonic() + 15.0:
        work = best_arr.copy()
        work /= float(work.sum())
        try:
            work, _ = _coord_descent(work, end5, seed=int(seed + 99999), consider=consider)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
