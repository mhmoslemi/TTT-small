import numpy as np
import time
from scipy.optimize import linprog


def _conv_fft(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_fast(a):
    """Score assuming sum(a)=1, a>=0."""
    c = _conv_fft(a)
    return 2.0 * len(a) * float(c.max())


def _score_general(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_fft(a)
    return float(2.0 * len(a) * c.max() / (s * s))


def _build_A(a):
    n = len(a)
    nrow = 2 * n - 1
    A = np.zeros((nrow, n), dtype=np.float64)
    for k in range(nrow):
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        cols = np.arange(jlo, jhi)
        A[k, cols] = a[k - cols]
    return A


def _solve_lp(A, b_ub, n, time_limit=2.0):
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit,
                               'primal_feasibility_tolerance': 1e-12,
                               'dual_feasibility_tolerance': 1e-12,
                               'dual_bound_strategy': 'DualHeuristic'})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _line_search(a, d, n_fine=25):
    def f(t):

        v = f(t)

            best_v, best_t = v, t

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = max(0.0, best_t - 0.08)
    hi = min(1.0, best_t + 0.08)

        if hi - lo < 1e-9:

        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        f1, f2 = f(m1), f(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t_star = 0.5 * (lo + hi)
    v_star = f(t_star)
    if v_star < best_v:
        best_t, best_v = t_star, v_star

    cand = (1.0 - best_t) * a + best_t * d
    cand = np.maximum(cand, 0.0)
