    """Try to improve coordinate i by searching over [0, x_hi] with G points."""
    v = a[i]
    base = c[i:i+n]  # affected convolution values

    max_const = 0.0

    diag_idx = i  # offset in base array for k=2i
    
    # Vectorized: for each delta, compute max of (base + delta*infl + diag_correction)
    aff = base[None, :] + deltas[:, None] * infl[None, :]
    # Add quadratic term at diagonal
    aff[:, diag_idx] += deltas ** 2
    maff = aff.max(axis=1)

    new_S = S + deltas

    imin = np.argmin(vals)
    return deltas[imin], vals[imin]
