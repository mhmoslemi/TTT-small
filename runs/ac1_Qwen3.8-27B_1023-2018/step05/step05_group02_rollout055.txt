def _subgradient_polish(p, stop_time, consider=None, max_iters=5000):
    """Subgradient method with momentum for flattening autocorrelation."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    best_p = p.copy()
    best_score = float(2.0 * n * _conv_self(p).max())
    momentum = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    t_count = 0
    base_lr = 0.004

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break

        c = _conv_self(p)
        cmax = float(c.max())
        # Use soft active set (top few lags)
        threshold = cmax - 0.01 * cmax
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        if len(active) > 20:
            # Keep top 20
            idx = np.argsort(c[active])[-20:]
            active = active[idx]

        # Compute gradient of soft max
        grad = np.zeros(n)
        for k in active:
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                contrib = p[int(kj[valid])] + p[valid]
                grad[valid] += contrib
        grad /= len(active)
        # Normalize gradient
        gnorm = float(np.linalg.norm(grad))
        if gnorm > 1e-300:
            grad /= gnorm

        t_count += 1
        lr = base_lr / (1.0 + 0.01 * t_count)
        m_t = beta1 * m_t + (1 - beta1) * grad
        v_t = beta2 * v_t + (1 - beta2) * grad * grad
        m_hat = m_t / (1 - beta1**t_count)
        v_hat = v_t / (1 - beta2**t_count)
        update = m_hat / (np.sqrt(v_hat) + 1e-8)

        improved_this_iter = False
        for scale in (1.0, 0.5, 0.25, 0.1, 0.05):
            p_cand = p - scale * lr * update
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = float(2.0 * n * _conv_self(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-16:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                improved_this_iter = True
                break

        if not improved_this_iter and it % 100 == 0:
            base_lr *= 0.95
            if base_lr < 1e-10:
                break

        if it % 80 == 0 and consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p

        p = _resample_mid(src, n)

            q = p * np.exp(side * amp * (2.0 * x - 1.0))

        q = np.maximum(q, 0.0)
        s = float(q.sum())
        if s > 0.0:
            q /= s
        else:
            q = p
        return q
