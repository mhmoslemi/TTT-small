def _conv_self(a):

    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(c, 0.0)

def _score(a):
    """Compute AC1 score."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    return float(2.0 * a.size * m / (s * s))

def _cosine_optimize(p, m_params, stop_time, consider=None):
    """Optimize a cosine-parameterized sequence."""
    n = p.size
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    
    # Initial cosine coefficients from DFT of p
    F = rfft(p)
    # F[j] = sum p[i] * exp(-2πji/n)
    # For a[i] = (1 + sum α_k cos(2πki/n))/n:
    # F[j] = α_j/2 for j = 1,...,m (and F[0] = 1)
    # So α_j = 2*F[j]
    alpha = np.zeros(m_params)
    for j in range(m_params):
        if j < len(F):
            alpha[j] = 2.0 * float(F[j].real) / n  # scaled by 1/n for the parameterization
    
    def a_from_alpha(al):
        """Reconstruct sequence from cosine coefficients."""
        i = np.arange(n, dtype=np.float64)
        val = np.ones(n)
        for j in range(len(al)):
            val += al[j] * np.cos(2.0 * math.pi * j * i / n)
        return val / n  # normalize to sum ≈ 1
    
    def objective(al):
        a = a_from_alpha(al)
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s < 1e-300:
            return 1e100
        c = _conv_self(a)
        m = float(c.max())
        return float(2.0 * n * m / (s * s))
    
    # Use L-BFGS-B to optimize
    best_alpha = alpha.copy()
    best_score = objective(alpha)
    
    # Multiple restarts with small perturbations
    rng = np.random.default_rng(12345)
    
    for restart in range(3):
        if time.monotonic() > stop_time - 10:
            break
        try:
            res = minimize(objective, best_alpha, method='L-BFGS-B', 
                          options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12, 'maxdisp': 0, 'disp': False})
            if np.all(np.isfinite(res.x)):
                sc = objective(res.x)
                if sc < best_score:
                    best_score = sc
                    best_alpha = res.x.copy()
        except Exception:
            pass
    
    a_out = a_from_alpha(best_alpha)
    a_out = np.maximum(a_out, 0.0)

    if ss > 1e-300:
        a_out /= ss
    if consider:
