def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _pairwise_cd(p, stop_time, consider=None, seed=99999):
    """Optimize pairs of adjacent coordinates simultaneously."""
    n = int(p.size)
    if n < 2:
        return p, float('inf')

    cur_score = float(2.0 * n * c.max())

    Ltot = 2 * n - 1
    flat = 0

    rounds = 0
    while rounds < 50 and time.monotonic() < stop_time - 5.0:
        improved = False
        for _ in range(n // 2):
            if time.monotonic() > stop_time - 3.0:
                break
            i = int(rng.integers(0, n - 1))
            j = i + 1
            vi = float(p[i])
            vj = float(p[j])

            # Determine affected conv range
            start_c = min(i, j)
            end_c = max(i + n, j + n)
            end_c = min(end_c, Ltot)
            if start_c >= end_c:
                continue

            # Affected indices
            mask_i = np.zeros(Ltot, dtype=bool)
            mask_i[max(0, i):min(Ltot, i + n)] = True
            mask_j = np.zeros(Ltot, dtype=bool)
            mask_j[max(0, j):min(Ltot, j + n)] = True

            base = c[start_c:end_c].copy()
            infl_i = np.zeros(end_c - start_c)
            infl_j = np.zeros(end_c - start_c)

            # For index k in conv, the coefficient of w_i (change in a[i]) is:
            # a[k-i] if 0<=k-i<n, plus a[i] if 0<=k-i<n (for the a[i]*a[k-i] term)
            # Actually: conv[k] includes a[i]*a[k-i] (if valid) and a[k-i]*a[i] (same)
            # When we change a[i] by dw: contribution is dw * a[k-i] (if k-i in [0,n-1])
            # Plus for the self-term when k=2i: dw*(2*a[i]+dw)
            
            # Simpler: just do grid search in 2D
            # This is expensive, so limit to a small grid
            G2 = 12
            wi_range = (-0.5 * vi, max(3.0 * vi, 5.0/n))
            wj_range = (-0.5 * vj, max(3.0 * vj, 5.0/n))

            wis = np.linspace(wi_range[0], wi_range[1], G2)
            wjs = np.linspace(wj_range[0], wj_range[1], G2)

            best_2d = (0.0, 0.0, cur_score)

                    # + (vj+wj)*a[k-j] - vj*a[k-j] for k in [j, j+n-1]

                    # This is complex; just compute the full local max
                    c_temp = c.copy()
                    # Update conv for index i
                    for k in range(max(0, i), min(Ltot, i + n)):
                        a_ki = p[k - i] if 0 <= k - i < n else 0.0
                        c_temp[k] += wi * a_ki
                    # Update conv for index j
                    for k in range(max(0, j), min(Ltot, j + n)):
                        a_kj = p[k - j] if 0 <= k - j < n else 0.0
                        c_temp[k] += wj * a_kj
                    # Self terms
                    c_temp[i + i] += wi * vi + wi * wi  # a[i] changed
                    c_temp[j + j] += wj * vj + wj * wj
                    # Cross terms: a[i]*a[j] appears in conv[i+j] and conv[j+i]
                    # conv[i+j] gets (vi+wi)(vj+wj) - vi*vj = vi*wj + vj*wi + wi*wj
                    if i + j < Ltot:
                        cross_old = vi * vj
                        cross_new = new_i * new_j
                        # conv[i+j] = sum a[ll]*a[i+j-ll], includes a[i]*a[j] and a[j]*a[i]
                        # Actually conv[i+j] includes a[i]*a[j] once if i != j
                        # Let's just be safe and add the difference
                        c_temp[i + j] += cross_new - cross_old
                    # Diagonal corrections (a[i]^2 term in conv[2i])
                    # Already handled above
                    
                    m = float(c_temp.max())
                    s_total = s + wi + wj
                    if s_total < 1e-12:
                        continue
                    sc = 2.0 * n * m / (s_total * s_total)
                    if math.isfinite(sc) and sc < best_2d[2]:

            if best_2d[2] < cur_score - 1e-12:
                p[i] = best_2d[0]
                p[j] = best_2d[1]
                c = _conv_self(p)
                cur_score = float(2.0 * n * c.max())
                improved = True

        # Renormalize
        s_new = float(p.sum())
        if s_new > 1e-300:
            p /= s_new
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())

                best_p = p.copy()

                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                break
        rounds += 1

    return best_p, best_score

    best_score = float('inf')

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()

    consider(parent)
    n_work = int(best_arr.size)
    if n_work < 128:
        n_work = 384
    if n_work > 4096:
        n_work = 3072

    # Phase 1: Quick LSE optimization
    lse_time = min(80.0, 0.08 * budget)
    p_work = _resample(best_arr, n_work)
    lse_end = t0 + lse_time
    if lse_end > t0 + 10.0:
        try:
            p2 = _lse_optimize(p_work, lse_end, consider=consider)
            if p2 is not None and p2.size == n_work and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p_work = p2 / float(p2.sum())
                consider(p_work)
        except Exception:
            pass

    # Phase 2: Main coordinate descent
    cd_end = hard - 30.0
    if cd_end > time.monotonic() + 30.0:
        p_start = _resample(best_arr, n_work)
        try:
            p_cd, _ = _coord_descent(p_start, cd_end, consider=consider, seed=int(seed + 777))
            if p_cd is not None and p_cd.size > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                consider(p_cd)
        except Exception:
            pass

    # Phase 3: Pairwise refinement
    rem = hard - 15.0 - time.monotonic()
    if rem > 20.0:
        p_fp = _resample(best_arr, n_work)
        try:
            p_pw, _ = _pairwise_cd(p_fp, hard - 10.0, consider=consider, seed=int(seed + 55555))
            if p_pw is not None and p_pw.size > 0 and np.all(np.isfinite(p_pw)) and p_pw.sum() > 0:
                consider(p_pw)
        except Exception:
            pass

    # Phase 4: Final CD squeeze
    rem = hard - 8.0 - time.monotonic()
    if rem > 15.0:
        p_final = _resample(best_arr, n_work)
        try:
            p_f, _ = _coord_descent(p_final, hard - 5.0, consider=consider, seed=int(seed + 8888), G=48)
            if p_f is not None and p_f.size > 0 and np.all(np.isfinite(p_f)) and p_f.sum() > 0:
                consider(p_f)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
