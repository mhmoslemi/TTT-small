            maff = np.maximum(maff, max_const)
            denoms = S + ws
            vals = np.where(denoms > 1e-14, 2.0 * n * maff / (denoms * denoms), math.inf)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Tiny golden
            dw = (x_hi - x_lo) / max(G, 1)
            lo_gs = max(0.0, best_x - 1.0 * dw)
            hi_gs = min(x_hi, best_x + 1.0 * dw)
            if hi_gs - lo_gs > 1e-15:
                gr = (math.sqrt(5.0) - 1.0) / 2.0
                a_gs, b_gs = lo_gs, hi_gs

                def ev_x(x):
                    w = x - v
                    denom = S + w
                    if denom <= 1e-14:
                        return math.inf
                    nb = base + w * infl
                    if has_diag:
                        nb = nb.copy()
                        nb[diag_off] += w * w
                    mv = max(float(nb.max()), max_const)
                    return 2.0 * n * mv / (denom * denom)

                c_gs = b_gs - gr * (b_gs - a_gs)
                d_gs = a_gs + gr * (b_gs - a_gs)
                f_c, f_d = ev_x(c_gs), ev_x(d_gs)
                for _ in range(10):
                    if f_c < f_d:
                        b_gs = d_gs; d_gs = c_gs; f_d = f_c
                        c_gs = b_gs - gr * (b_gs - a_gs); f_c = ev_x(c_gs)
                    else:
                        a_gs = c_gs; c_gs = d_gs; f_c = f_d
                        d_gs = a_gs + gr * (b_gs - a_gs); f_d = ev_x(d_gs)
                x_opt = (a_gs + b_gs) * 0.5
                f_opt = ev_x(x_opt)
                if f_opt < best_val:
                    best_x, best_val = x_opt, f_opt

            if math.isfinite(best_val) and best_val < cur_sc - 1e-15 and best_x >= 0:
                w = best_x - v
                if abs(w) > 1e-16:
                    a_old = p[:Ls].copy()
                    p[i] = best_x
                    S += w
                    c[start:end] += 2.0 * w * a_old
                    if has_diag:
                        c[start + diag_off] += w * w
                    cur_sc = best_val
                    improved = True

        if S > 1e-15 and abs(S - 1.0) > 1e-10:
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0

        if sweep % 3 == 0:
            c = _conv_fft(p)
            cur_sc = float(2.0 * n * c.max())

        if cur_sc < best_sc - 1e-14:
            best_sc = cur_sc
            best_p = p.copy()
            consider(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 10:
                break

    return best_p, best_sc
