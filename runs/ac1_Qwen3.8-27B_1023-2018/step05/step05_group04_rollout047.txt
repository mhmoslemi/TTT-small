        # f[k-j] for j = j_lo, j_lo+1, ..., j_hi
        # = f[k-j_lo], f[k-j_lo-1], ..., f[k-j_hi]
        # Since j_lo <= j_hi, k-j_lo >= k-j_hi
        # So we want f[k-j_lo], f[k-j_lo-1], ..., f[k-j_hi] which is f[k-j_hi : k-j_lo+1][::-1]
        offset += cnt
```

Hmm, this Python loop over 6143 rows with vectorized inner operations should be fast (6143 iterations, each O(n) at most). Let me estimate: 6143 * ~3000 ops = ~18M ops, which is maybe 50ms in Python. That's fine.

Actually, let me reconsider. Building the matrix this way in Python is going to be slow because of the loop. Let me use a smarter approach.

The matrix A has the property that A[k, j] = f[k-j] when 0 <= k-j < n and 0 <= j < n. This is essentially the "skip matrix" or "convolution matrix". 

I can build it as:
- For each diagonal d = k - j (the "lag"), A[k, k-d] = f[d] for valid k.
- The valid range for d is 0 to n-1.
- For a given d, the valid rows k are d to d+n-1 (but constrained to 0 to 2n-2), and columns j = k-d are 0 to n-1.

So for each d in 0..n-1:
- Rows: max(d, 0) to min(d+n-1, 2n-2)
- Cols: max(d,0)-d to min(d+n-1, 2n-2)-d = 0 to min(n-1, 2n-2-d)

Wait, this is getting complicated. Let me just use the straightforward approach but vectorized:

```python
def _build_conv_matrix(f, n):
    L = 2 * n - 1
    # Total non-zeros = n^2
    total_nnz = n * n
    
    # For row k (0 to L-1), non-zeros at columns j_lo to j_hi
    # j_lo = max(0, k-n+1), j_hi = min(k, n-1)
    
    # Build row pointers for CSR
    # Row k has nnz_per_row[k] elements
    nnz_row = np.zeros(L, dtype=np.int64)
    for k in range(L):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        nnz_row[k] = j_hi - j_lo + 1
    
    row_ptr = np.zeros(L + 1, dtype=np.int64)
    row_ptr[1:] = np.cumsum(nnz_row)
    
    # Build col_idx and data
    col_idx = np.empty(total_nnz, dtype=np.int32)
    data = np.empty(total_nnz, dtype=np.float64)
    
    for k in range(L):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        start = row_ptr[k]
        end = row_ptr[k+1]
        col_idx[start:end] = range(j_lo, j_hi + 1)
        # data: f[k-j] for j in j_lo..j_hi
        # = f[k-j_lo], f[k-j_lo-1], ..., f[k-j_hi]
        # Since j_lo <= j_hi, k-j_lo >= k-j_hi
        # f[k-j_lo], f[k-j_lo-1], ..., f[k-j_hi] = f[k-j_hi : k-j_lo+1][::-1]
        data[start:end] = f[k-j_hi : k-j_lo+1][::-1]
    
    A = csr_matrix((data, col_idx, row_ptr), shape=(L, n))
    return A
```

This should work. The loop is 6143 iterations, each doing a small numpy slice assignment. Should be fast enough (< 0.1s).

Now, the LP:
- Maximize c^T g where c = (1, 1, ..., 1)
- Subject to A @ g <= M (where M = max(conv(f,f)))
- g >= 0

Using scipy.optimize.linprog:
- c = -ones(n) (since linprog minimizes)
- A_ub = A (sparse)
- b_ub = M * ones(L)
- bounds = [(0, None)] * n
- method = 'highs'

This should work. Let me estimate solve time: 6143 constraints, 3072 variables, ~9.4M non-zeros. HiGHS should handle this in 1-5 seconds.

After solving, we get g*. Then we rescale: g = g* / sum(g*) (to have unit sum).
Update: f_new = (1-t)*f + t*g, then renormalize to unit sum.

The step size t should be small (0.01-0.1) to avoid oscillation.

Let me also think about whether this LP is well-posed. The constraint A @ g <= M means (f*g)_k <= M for all k. Since f >= 0 and g >= 0, all (f*g)_k >= 0. The constraint is feasible (g=0 works). It's bounded above (g_j <= M/f_max for each j, roughly). So the LP has a finite optimum.

One concern: if f has a single non-zero element, the LP might not be well-posed. But for practical purposes with the parent initialization, f should be well-distributed.

Let me now write the complete program:

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p

def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.asarray(c, dtype=np.float64)

def _score_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float('inf')
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return float('inf')
    return float(2.0 * n * m / (s * s))

def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.shape, 1.0 / u.size, dtype=np.float64)
    return e / s

def _build_conv_matrix(f, n):
    L = 2 * n - 1
    row_ptr = np.zeros(L + 1, dtype=np.int64)
    for k in range(L):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        row_ptr[k + 1] = row_ptr[k] + (j_hi - j_lo + 1)
    total_nnz = int(row_ptr[-1])
    col_idx = np.empty(total_nnz, dtype=np.int32)
    data = np.empty(total_nnz, dtype=np.float64)
    for k in range(L):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        s = row_ptr[k]
        e = row_ptr[k + 1]
        col_idx[s:e] = np.arange(j_lo, j_hi + 1, dtype=np.int32)
        data[s:e] = f[k - j_hi : k - j_lo + 1][::-1]
    return csr_matrix((data, col_idx, row_ptr), shape=(L, n))

def _lp_dual_step(f, M, stop_time):
    """Solve the LP dual problem and return the LP solution g (unit sum)."""
    n = f.size
    L = 2 * n - 1
    A = _build_conv_matrix(f, n)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(L, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None
    
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x, 0.0)
        s = float(g.sum())
        if s > 1e-30:
            return g / s
    return None

def _lse_optimize(p, stop_time, consider=None):
    n = p.size
    p = np.maximum(np.asarray(p, dtype=np.float64), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    u = np.log(np.maximum(p, 1e-30), dtype=np.float64)
    u = u - float(u.mean())
    
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c
    
    def xfast(a, w):
        F = rfft(a[::-1], cross_size)
        G = rfft(w, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q
    
    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f_val = cmax
        else:
            f_val = cmax + T * float(np.log(sum_exp / c.size))
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return f_val, grad
    
    stages = [
        (0.15, 0.15),
        (0.04, 0.2),
        (0.01, 0.25),
        (0.003, 0.25),
        (0.0008, 0.15),
    ]
    
    best_p = p.copy()
    best_sc = _score_arr(p)
    start = time.monotonic()
    avail = max(0.0, stop_time - start - 12.0)
    total_w = sum(w for _, w in stages)
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 8.0:
            break
        cum += wt
        end_stage = min(stop_time - 6.0, start + avail * (cum / total_w))
        if end_stage <= now + 4.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)
        best_u = u.copy()
        try:
            f0, _ = lse_func(u, T)
        except Exception:
            f0 = 1e100
        best_fstage = f0
        stale = 0
        while time.monotonic() < end_stage - 2.0:
            maxit = 30 if n > 2000 else 45
            opts = {"maxiter": maxit, "gtol": 1e-11, "ftol": 1e-15, "maxls": 15, "maxdisp": 0}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_fstage - 1e-14:
                    best_fstage = fval
                    best_u = res.x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 2:
                        break
            if getattr(res, "status", 1) == 0:
                break
        p_cur = _softmax(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        sc = _score_arr(p_cur)
        if math.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best_p = p_cur.copy()
            p = p_cur.copy()
            u = np.log(np.maximum(p, 1e-30), dtype=np.float64)
            u = u - float(u.mean())
        else:
            p = best_p.copy()
            u = np.log(np.maximum(p, 1e-30), dtype=np.float64)
            u = u - float(u.mean())
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p

# Coordinate descent - faster version
def _coord_descent(p, stop_time, consider=None, seed=12345):
    n = int(p.size)
    if n == 0:
        return p, float('inf')
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p, float('inf')
    p = p / s
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    Ltot = 2 * n - 1
    flat = 0
    sweep = 0
    GR = (math.sqrt(5.0) - 1.0) / 2.0
    G = 32  # Reduced grid size for speed
    
    max_sweeps = 100000
    while sweep < max_sweeps and time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 1.0:
                break
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]
            diag = i
            
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < Ltot:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc
            
            # Quick grid search
            x_hi = max(20.0 * v, 10.0 / n, 0.005)
            x_hi = min(x_hi, 0.3)
            x_hi = max(x_hi, 1e-8)
            
            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            ws = pts - v
            s_new = 1.0 + ws  # normalized sum = 1
            valid = s_new > 1e-12
            if not np.any(valid):
                continue
            
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag < Ls:
                aff[:, diag] += ws * ws
            maff = np.max(aff, axis=1)
            maff = np.maximum(maff, max_const)
            
            scores = np.full(pts.shape, float('inf'), dtype=np.float64)
            scores[valid] = 2.0 * n * maff[valid] / (s_new[valid] ** 2)
            
            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_val = float(scores[imin])
            
            # Golden section (fewer iterations for speed)
            if 0 < imin < len(pts) - 1:
                lo = float(pts[max(0, imin - 1)])
                hi = float(pts[min(len(pts) - 1, imin + 1)])
            else:
                lo = 0.0
                hi = x_hi
            
            def eval_1d(w):
                s_n = 1.0 + w
                if s_n <= 1e-15:
                    return float('inf')
                aff_m = float(np.max(base + w * infl))
                if 0 <= diag < Ls:
                    dval = base[diag] + w * infl[diag] + w * w
                    if dval > aff_m:
                        aff_m = dval
                m = max(aff_m, max_const)
                return 2.0 * n * m / (s_n * s_n)
            
            a_gs = lo
            b_gs = hi
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)
            f_c = eval_1d(c_gs - v)
            f_d = eval_1d(d_gs - v)
            for _ in range(15):  # Reduced from 22
                if b_gs - a_gs < 1e-10:
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = eval_1d(c_gs - v)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_d = eval_1d(d_gs - v)
            x_mid = (a_gs + b_gs) / 2.0
            f_mid = eval_1d(x_mid - v)
            best_x = x_mid
            if f_mid < best_val:
                best_val = f_mid
            # Check endpoints
            for xp in (0.0, x_hi):
                fp = eval_1d(xp - v)
                if fp < best_val:
                    best_x = xp
                    best_val = fp
            
            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= x_hi:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    aff_seg = base + w * infl
                    if 0 <= diag < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag] += w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    c[start:end] = aff_seg
                    cur_score = 2.0 * n * new_max
                    improved = True
        
        # Renormalize
        s_new = float(p.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            inv = 1.0 / s_new
            p *= inv
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-15:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass
        
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 20.0:
                    kick_size = 0.02 + 0.03 * min(1.0, flat / 15.0)
                    p = p * np.exp(rng.normal(0.0, kick_size, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                elif time.monotonic() < stop_time - 5.0:
                    flat = 0
                    # Try another kick
                    p = p * np.exp(rng.normal(0.0, 0.05, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                else:
                    break
        sweep += 1
    
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m,
