                if flat3 >= 20:
                    break

    # ============================================================
    # Phase 4: One more full CD round if time permits
    # ============================================================
    phase4_end = deadline - 15.0
    if phase4_end > time.monotonic() + 30:
        p = best_p.copy() / float(best_p.sum())
        c = conv_self(p)
        S = 1.0
        cur_score = float(2.0 * n * float(c.max()))
        rng3 = np.random.default_rng(seed + 31337)
        G4 = 14

        while time.monotonic() < phase4_end - 2:
            cmax = float(c.max())
            thresh = cmax * 0.97
            active_lags = np.where(c >= thresh)[0]
            imp = np.zeros(n, dtype=bool)
            for lag in active_lags:
                lo = max(0, int(lag) - n + 1)
                hi = min(n - 1, int(lag))
                imp[lo:hi + 1] = True
            imp_idx = np.where(imp)[0]
            oth_idx = np.where(~imp)[0]
            if len(oth_idx) > 150:
                sample_o = rng3.choice(oth_idx, 150, replace=False)
            else:
                sample_o = oth_idx
            order = rng3.permutation(np.concatenate([imp_idx, sample_o]))

            improved4 = False
            for i in order:
                if time.monotonic() > phase4_end - 1:
                    break
                v = float(p[i])
                start = i
                end = min(i + n, L)
                Ls = end - start
                if Ls <= 0:
                    continue
                max_const = 0.0
                if start > 0:
                    max_const = max(max_const, float(c[:start].max()))
                if end < L:
                    max_const = max(max_const, float(c[end:].max()))
                base = c[start:end]
                infl = 2.0 * p[:Ls]
                diag_off = 2 * i - start
                x_hi = min(max(3.0 * v + 0.003, 6.0 / n), 1.0)
                pts = np.linspace(0.0, x_hi, G4 + 1)
                ws = pts - v
                aff = base[None, :] + ws[:, None] * infl[None, :]
                if 0 <= diag_off < Ls:
                    aff[:, diag_off] += ws[:, None] * ws[:, None]
                maff = np.maximum(aff.max(axis=1), max_const)
                denoms = S + ws
                vals = np.where(denoms > 0.5, 2.0 * n * maff / (denoms * denoms), 1e100)
                if not np.any(np.isfinite(vals)):
                    continue
                imin = int(np.argmin(vals))
                best_x = float(pts[imin])
                best_val = float(vals[imin])
                if math.isfinite(best_val) and best_val < cur_score - 1e-15 and best_x >= 0:
                    w = best_x - v
                    if abs(w) > 1e-17:
                        c[start:end] += w * infl
                        if 0 <= diag_off < Ls:
                            c[start + diag_off] += w * w
                        p[i] = best_x
                        S += w
                        cur_score = best_val
                        improved4 = True

            if S > 0.5 and S < 1.5:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
            elif S > 1e-10:
                p /= S
                c /= S * S
            S = 1.0
            cur_score = float(2.0 * n * float(c.max()))
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_p = p.copy()

            if not improved4:
                p[imp_idx[:50]] *= np.exp(rng3.normal(0, 0.03, min(50, len(imp_idx))))
                p = np.maximum(p, 0)
                ps = float(p.sum())
                if ps > 1e-10:
                    p /= ps
                c = conv_self(p)
                cur_score = float(2.0 * n * float(c.max()))
                if cur_score < best_score:
                    best_score = cur_score
                    best_p = p.copy()

    # ============================================================
    # Final validation with exact evaluator
    # ============================================================
    try:
        out = np.asarray(best_p, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        ls = [float(x) for x in out]
        sc = float(evaluate_sequence(ls))
        if not math.isfinite(sc) or sc > 2.0:
            # Fallback: return parent
            try:
                pp = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
                pp = np.clip(pp, 0.0, 1000.0)
                pp = np.where(np.isfinite(pp), pp, 0.0)
                ls = [float(x) for x in pp]
                sc2 = float(evaluate_sequence(ls))
                if not math.isfinite(sc2):
                    ls = [1.0]
            except:
                ls = [1.0]
        return ls
    except:
        return [1.0]
