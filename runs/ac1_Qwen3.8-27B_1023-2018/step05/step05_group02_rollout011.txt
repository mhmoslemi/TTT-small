    return float(2 * n * max_b / (sum_a**2))

            denom = S + w_arr
            valid = denom > 1e-15
            vals = np.full(pts.shape, math.inf)
            if np.any(valid):
                aff = base[None, :] + w_arr[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                dval = base[0] + w_arr * infl[0] + w_arr * w_arr
                maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                vals[valid] = (2.0 * n * maff[valid]) / (denom[valid] ** 2)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Refine around best
            dw = x_hi / G
            lo = max(0.0, best_x - 2.0 * dw)
            hi = min(x_hi, best_x + 2.0 * dw)
            if hi > lo:
                fine_pts = np.linspace(lo, hi, 30)
                w_arr = fine_pts - v
                denom = S + w_arr
                valid = denom > 1e-15
                vals2 = np.full(fine_pts.shape, math.inf)
                if np.any(valid):
                    aff = base[None, :] + w_arr[:, None] * infl[None, :]
                    maff = np.max(aff, axis=1)
                    dval = base[0] + w_arr * infl[0] + w_arr * w_arr
                    maff = np.maximum(maff, dval)
                    maff = np.maximum(maff, max_const)
                    vals2[valid] = (2.0 * n * maff[valid]) / (denom[valid] ** 2)
                if np.any(np.isfinite(vals2)):
                    im2 = int(np.argmin(vals2))
                    if vals2[im2] < best_val:
                        best_x = float(fine_pts[im2])
                        best_val = float(vals2[im2])

            if math.isfinite(best_val) and best_val < cur_score - 1e-13:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_fft(p)
        new_score = float(2.0 * n * c.max())
        if new_score < best_score - 1e-16:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 5.0:
                    noise = 0.03 + 0.02 * math.exp(-flat * 0.15)
                    p = p * np.exp(rng.normal(0.0, noise, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_fft(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if cur_score < best_score:
                        best_score = cur_score
                        best_p = p.copy()
                    flat = 0
                else:
                    break
        sweep += 1

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p, best_score


def _subgrad_polish(p, stop_time, consider=None, max_iters=2000):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p /= s
    best_p = p.copy()
    best_score = float(_sc(p))
    step = 0.004

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break
        c = _conv_fft(p)
        cmax = float(c.max())
        threshold = cmax * 0.998
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        grad = np.zeros(n)
        for k in active:
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                contrib = p[kj[valid]] + p[valid]
                grad[valid] += contrib
        grad /= len(active)

        for scale in (1.0, 0.5, 0.25, 0.1):
            p_c = p - scale * step * grad
            p_c = np.maximum(p_c, 0.0)
            ss = float(p_c.sum())
            if ss < 1e-15:
                continue
            p_c /= ss
            sc = float(_sc(p_c))
            if math.isfinite(sc) and sc < best_score - 1e-16:
                best_score = sc
                best_p = p_c.copy()
                p = p_c.copy()
                break
        else:
            step *= 0.92
            if step < 1e-12:
                break

        if it % 100 == 0 and consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 18.0

    parent = _get_parent()
    n = int(parent.size)
    best_arr = parent.copy()
    best_score = float(_sc_exact(best_arr))

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = float(_sc_exact(best_arr))

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size == 0:
            return
        s = float(arr.sum())
        if s < 0.01:
            return
        sc = float(_sc_exact(arr))
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = arr.copy()

    def make_variant(src, n_t, sd):
        p = _resample(src, n_t)
        rng = np.random.default_rng(sd)
        r = rng.random()
        if r < 0.33:
            # Skew
            x = (np.arange(n_t) + 0.5) / n_t
            side = 1.0 if rng.random() < 0.5 else -1.0
            amp = 0.3 + 0.6 * rng.random()
            q = p * np.exp(side * amp * (2.0 * x - 1.0))
        elif r < 0.66:
            # Bimodal
            x = (np.arange(n_t) + 0.5) / n_t
            sigma = 0.05 + 0.1 * rng.random()
            q = p * (np.exp(-((x - 0.3) ** 2) / (2 * sigma ** 2)) +
                     np.exp(-((x - 0.7) ** 2) / (2 * sigma ** 2)))
        else:
            # Noise
            q = p * np.exp(rng.normal(0, 0.3, n_t))
        q = np.maximum(q, 0.0)
        s = float(q.sum())
        return q / s if s > 0 else p

    T_coarse = (0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004, 0.0001, 0.00002)
    T_fine = (0.00008, 0.00003, 0.00001, 0.000003)

    # Phase 1: LSE from parent (40% of usable time)
    usable = hard - t0 - 10.0
    p1_end = t0 + 0.40 * usable
    if p1_end > time.monotonic() + 30:
        p1 = _lse_opt(best_arr, p1_end, T_coarse, consider=consider)
        if p1.size == n and p1.sum() > 0:
            p1 /= p1.sum()
            consider(p1)

    # Phase 2: LSE from diverse initializations (20%)
    p2_end = t0 + 0.60 * usable
    if p2_end > time.monotonic() + 20:
        rng_top = np.random.default_rng(seed)
        variants = []
        for vi in range(3):
            v = make_variant(best_arr, n, int(seed + 1000 + vi * 77))
            variants.append(v)
        sub_end = p2_end
        rem = sub_end - time.monotonic()
        if rem > 30:
            chunk = rem / len(variants)
            for vi, v in enumerate(variants):
                vend = min(sub_end, time.monotonic() + chunk - 2)
                if vend > time.monotonic() + 10:
                    pv = _lse_opt(v, vend, T_coarse, consider=consider)
                    if pv.size == n and pv.sum() > 0:
                        pv /= pv.sum()
                        consider(pv)

    # Phase 3: Coordinate descent (30%)
    p3_end = t0 + 0.90 * usable
    if p3_end > time.monotonic() + 30:
        p3, _ = _coord_descend(best_arr, p3_end, consider=consider, seed=int(seed + 55555), G=30)
        consider(p3)

    # Phase 4: Fine LSE (10%)
    now = time.monotonic()
    p4_end = min(hard - 30, t0 + usable)
    if p4_end > now + 30:
        pf = _lse_opt(best_arr, p4_end, T_fine, consider=consider)
        if pf.size == n and pf.sum() > 0:
            pf /= pf.sum()
            consider(pf)

    # Final subgradient + coord polish
    now = time.monotonic()
    if now < hard - 15:
        sg_end = min(hard - 8, now + max(5, 0.3 * (hard - now)))
        if sg_end > now + 5:
            ps = _subgrad_polish(best_arr, sg_end, consider=consider, max_iters=1000)
            consider(ps)

    now = time.monotonic()
    if now < hard - 5:
        c_end = min(hard - 3, now + max(3, 0.2 * (hard - now)))
        if c_end > now + 3:
            pc, _ = _coord_descend(best_arr, c_end, consider=consider, seed=int(seed + 99999), G=20)
            consider(pc)

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
