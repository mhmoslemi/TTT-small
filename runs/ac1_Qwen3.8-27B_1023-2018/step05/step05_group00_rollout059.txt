        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            f = cmax
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
        else:
            f = cmax + T * math.log(sum_exp / c.size)
            w = exps / sum_exp
            if not math.isfinite(f):
                f = cmax
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    if fine_only:
        stages = ((0.001, 0.35), (0.0003, 0.35), (0.00005, 0.30))
    else:
        stages = (
            (0.12, 0.08),
            (0.04, 0.10),
            (0.01, 0.12),
            (0.003, 0.14),
            (0.0008, 0.16),
            (0.0002, 0.15),
            (0.00004, 0.14),
            (0.000005, 0.11),
        )

    start = time.monotonic()
    avail = float(stop_time - start - 8.0)
    if avail <= 5.0:
        return p

    best_p = p.copy()
    best_sc = _score_fast(p)
    total_w = float(sum(w for _, w in stages))
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)

        best_u = u.copy()
        best_f = None
        stale = 0
        max_iter = 0
        while time.monotonic() < end_stage - 1.0 and stale < 4:
            max_iter += 1
            if max_iter > 25:
                break
            opts = {
                "maxiter": 40,
                "gtol": 1e-13,
                "ftol": 1e-16,
                "maxls": 25,
                "maxcor": 40,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(lse_func, best_u, args=(T,),
                               method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    fval = float(res.fun) if math.isfinite(res.fun) else math.inf
                    if math.isfinite(fval):
                        if best_f is None or fval < best_f:
                            best_f = fval
                            best_u = x.copy()
                            stale = 0
                        else:
                            stale += 1
                        if getattr(res, "status", 1) == 0 and stale == 0:
                            break
                    else:
                        stale += 1
            else:
                stale += 1

        p_cur = _softmax(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fast(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif math.isfinite(sc_cur):
            p = p_cur.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, max_sweeps=200000, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    G = 36 if n <= 1200 else 28
    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    G_cur = G

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue
            base = c[start_i:end_i].copy()
            infl = 2.0 * np.zeros(Ls)
            # infl[d] = 2*a[d] for d in [0, n-1], except at d=i where it's handled separately
            valid_a = min(n, Ls)
            infl[:valid_a] = 2.0 * p[:valid_a]
            # The diagonal term (d=i, j=2i) is handled separately

            max_const = 0.0
            if start_i > 0:
                ml = float(c[:start_i].max())
                if ml > max_const:
                    max_const = ml
            if end_i < Ltot:
                mr = float(c[end_i:].max())
                if mr > max_const:
                    max_const = mr

            avg = 1.0 / float(n)
            x_hi = max(20.0 * v, 60.0 * avg, 0.03, 1e-12)
            x_hi = min(x_hi, 12.0)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, avg)

            pts = np.linspace(0.0, x_hi, G_cur + 1)
            if 0.0 <= v <= x_hi:
                pts = np.unique(np.concatenate((pts, [v])))

            diag_idx = i - start_i  # = i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S_total + w
                vals = np.full(xs.shape, math.inf)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                # Fix diagonal: conv[2i] additional term is w*(2v + w) - 2w*a[i] = w^2
                # Actually: for d=i, the correct update is w*(2v) + w^2, but infl[i] = 2*a[i] = 2v
                # So aff at diag = base[diag] + w*2v, and we need to add w^2
                # This is already handled by infl[diag] = 2*v, and the w^2 is the extra
                dval = base[diag_idx] + w * infl[diag_idx] + w * w
                aff = np.maximum(aff, dval[:, None])
                maff = np.max(aff, axis=1)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if G_cur > 4:
                dw = x_hi / float(G_cur)
                lo = max(0.0, best_x - 3.0 * dw)
                hi = min(x_hi, best_x + 3.0 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 48)
                if 0.0 <= best_x <= x_hi:
                    fine = np.unique(np.concatenate((fine, [best_x])))
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= 15.0:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start_i:end_i] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        np.maximum(c, 0.0, out=c)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
            if flat < 3 and G_cur > 16:
                G_cur = max(16, G_cur - 2)
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 6.0:
                    sigma = 0.02 + 0.02 * min(flat, 10)
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            consider(best_p)
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _subgradient_step(p, n_steps=50, lr=None, stop_time=None):
    """Simple subgradient method on the max of convolution."""
    n = p.size
    p = p.copy()
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s < 1e-300:
        return p
    p /= s
    c = _conv_self(p)
    M = float(c.max())
    if lr is None:
        lr = 0.01 / n
    for _ in range(n_steps):
        if stop_time is not None and time.monotonic() > stop_time - 2:
            break
        # Subgradient: uniform over argmax positions
        thresh = M * (1 - 1e-10)
        mask = c >= thresh
        w = mask.astype(np.float64)
        wsum = w.sum()
        if wsum < 1:
            w[np.argmax(c)] = 1.0
            wsum = 1.0
        w /= wsum
        # Gradient of max(conv) w.r.t. a[i]: sum_j w_j * d(conv_j)/d(a_i)
        # d(conv_j)/d(a_i) = a[(j-i)%n] + a[i-(j)%n] (circular approx)
        # For linear: d(conv_j)/d(a_i) = a[j-i] (if 0<=j-i<n) + a[j-i] (if j-i≠i, else a[i])
        # More precisely: for j in [i, i+n-1], d conv[j]/da[i] = 2*a[j-i] (or 2*a[i]+w if j=2i)
        # Use circular approximation for speed
        from scipy.fft import fft, ifft
        W = fft(w, 2 * n - 1)
        A = fft(p, 2 * n - 1)
        # Gradient in freq domain
        grad_freq = 2.0 * (A[::-1] * W[::-1] + np.conj(A) * W)  # approximate
        # Actually simpler: grad[i] = sum_j w[j] * (a[j-i] + a[i-j] if j!=i else 2a[i])
        # This is a cross-correlation of w with a (circular, but we use linear)
        # Use FFT cross-correlation
        L = 2 * n - 1
        size = next_fast_len(L)
        Gw = fft(w, size)
        Ga = fft(p, size)
        cross = ifft(Gw[:, None] * Ga[None, :])  # not quite right
        # Simpler: just do it with numpy
        # grad[i] = sum over j of w[j] * d(conv[j])/d(a[i])
        # For efficiency, just do the full computation
        grad = np.zeros(n)
        for j in range(L):
            if w[j] < 1e-12:
                continue
            lo = max(0, j - n + 1)
            hi = min(n, j + 1)
            for k in range(lo, hi):
                # d conv[j] / d a[k] = a[j-k] + a[k-j] (if j-k != k) or 2*a[k] (if j-k==k)
                d1 = j - k
                if 0 <= d1 < n:
                    val = a_k_jk = p[d1]
                    if d1 == k:
                        grad[k] += w[j] * 2.0 * p[k]
                    else:
                        d2 = k  # the other term: a[k] * a[j-k], derivative w.r.t a[k] gives a[j-k]
                        # Actually d(a[k]*a[j-k])/da[k]: if j-k != k, it's a[j-k]; the term a[j-k]*a[k] also depends on a[k] with derivative a[j-k]
                        # So total: 2*a[j-k] if j-k != k, else 2*a[k]
                        grad[k] += w[j] * 2.0 * p[d1]
        # Projected gradient step
        p_new = p - lr * grad
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if ss > 1e-300:
            p_new /= ss
        c_new = _conv_self(p_new)
        M_new = float(c_new.max())
        if M_new < M * 1.01:
            p = p_new
            c = c_new
            M = M_new
            lr *= 0.95
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return p
