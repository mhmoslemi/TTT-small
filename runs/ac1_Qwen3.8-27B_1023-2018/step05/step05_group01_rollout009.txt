            sweep += 1
        
        # Final
        ss = float(p.sum())
        if ss > 1e-300:
            p = p / ss
        consider(p)
        consider(best_local_p)
        return best_local_p
    
    # LP-based direction finding
    def lp_step(p, M, stop_time):
        """Solve LP: max sum(b) s.t. conv(p,b) <= M, b >= 0"""
        nn = p.size
        Ltot = 2*nn - 1
        # Build constraint matrix: C[k, j] = p[j] if valid
        # C has shape (Ltot, nn), banded
        from scipy.sparse import dia_matrix, csc_matrix
        # For each j, p[j] appears in C[k, j] for k = j, ..., j+nn-1 (clamped to 0..Ltot-1)
        # This is a banded matrix
        # Row k, col j: p[j] if max(0, k-nn+1) <= j <= min(k, nn-1)
        # Equivalently: k is in [j, j+nn-1] and 0 <= k <= Ltot-1
        # Diagonal offset: k - j ranges from 0 to nn-1
        
        # Actually, C[k,j] = p[j] if 0<=k-j<nn (i.e., j <= k <= j+nn-1)
        # So for diagonal offset d = k-j, C[k,j] = p[j] for d in {0,1,...,nn-1}
        # This means C is a lower triangular banded matrix with nn diagonals
        
        # Let's build it as a sparse matrix
        # Row indices: for each j, rows j to min(j+nn-1, Ltot-1)
        row_idx = []
        col_idx = []
        vals = []
        for j in range(nn):
            r_start = j
            r_end = min(j + nn, Ltot)
            for k in range(r_start, r_end):
                row_idx.append(k)
                col_idx.append(j)
                vals.append(float(p[j]))
        
        C = csc_matrix((vals, (row_idx, col_idx)), shape=(Ltot, nn))
        
        # LP: max 1^T b s.t. C @ b <= M, b >= 0
        # In standard form: min -1^T b s.t. C @ b <= M, b >= 0
        c_obj = -np.ones(nn)
        
        try:
            from scipy.optimize import linprog
            res = linprog(c_obj, A_ub=C, b_ub=np.full(Ltot, M),
                         bounds=[(0, None)]*nn, method='highs')
            if res.success and res.x is not None:
                b = res.x
                b = np.maximum(b, 0)
                bsum = float(b.sum())
                if bsum > 1e-10:
                    b = b / bsum
                    return b
        except Exception:
            pass
        return None

    best_arr = parent.copy()

    phase1_end = t0 + 0.38 * (hard_stop - t0)
