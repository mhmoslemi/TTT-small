        pp = np.maximum(pp, 1e-15)
        uu = np.log(pp)
        uu -= float(np.mean(uu))
        return uu

    def obj(uu, T):
        uu = np.asarray(uu, dtype=np.float64).ravel()
        if uu.size != n:
            return 1e100, np.zeros(n)
        uu = np.where(np.isfinite(uu), uu, 0.0)
        m = float(np.max(uu))
        e = np.exp(uu - m)
        s = float(e.sum())
        if not math.isfinite(s) or s <= 0:
            pp = np.full(n, 1.0/n)
        else:
            pp = e/s
        pp = np.where(np.isfinite(pp), pp, 0.0)
        ss = float(pp.sum())
        if ss > 0:
            pp /= ss

        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax):
            return 1e100, np.zeros(n)

        T = max(1e-14, T)
        scaled = (c - cmax)/T
        w = np.exp(scaled)
        se = float(w.sum())
        if not math.isfinite(se) or se <= 0:
            im = int(np.argmax(c))
            w = np.zeros(conv_len)
            w[im] = 1.0
            val = cmax
        else:
            w = w/se
            val = cmax + T*math.log(se/conv_len)

        q = xfast(pp, w)
        gid = 2.0*q
        gd = float(np.dot(pp, gid))
        grad = pp*(gid - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(val), grad

    def run_chunk(uu, T, end):
        best_u = uu.copy()
        best_f = obj(uu, T)[0]
        stale = 0
        while time.monotonic() < end - 1.5:
            maxit = 12 if n > 2000 else 18
            opts = {"maxiter": maxit, "gtol": 1e-9, "ftol": 1e-14, "maxls": 12, "maxcor": 12, "disp": False, "maxdisp": 0}
            try:
                res = minimize(obj, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                xx = np.asarray(res.x, dtype=np.float64).ravel()
                if xx.size == n and np.all(np.isfinite(xx)):
                    fv = float(res.fun)
                    if math.isfinite(fv) and fv < best_f - 1e-13:
                        best_f = fv
                        best_u = xx.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 2:
                            break
            else:
                break
        return best_u

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score_fft(p)
    if consider:
        try:
            consider(best_p)
        except Exception:
            pass

    stages = (
        (0.012, 0.14),
        (0.005, 0.16),
        (0.0015, 0.18),
        (0.0005, 0.20),
        (0.00015, 0.22),
        (0.00005, 0.24),
    )

    start = time.monotonic()
    avail = float(stop_time - start - 4.0)
    if avail <= 5.0:
        return best_p
    total_w = sum(w for _, w in stages)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += wt
        end_stage = min(stop_time - 3.0, start + avail*(cum/total_w))
        if end_stage <= now + 2.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-14, frac*cmax)
        u = run_chunk(u, T, end_stage)
        p_new = _softmax(u)
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if math.isfinite(ss) and ss > 1e-300:
            p_new /= ss
        else:
            p_new = best_p.copy()
        sc_new = _score_fft(p_new)
        if math.isfinite(sc_new) and sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_p = p_new.copy()
            p = p_new.copy()
            if consider:
                try:
                    consider(best_p)
                except Exception:
                    pass
        else:
            p = best_p.copy()
        u = p_to_u(p)

    return best_p


def _fin_coord_eval(pts, base, infl_vec, max_const, v, Sloc, n, di):
    """Evaluate score for candidate values of one coordinate."""
    w = pts - v
    den = Sloc + w
    valid = den > 1e-14
    aff = base[None, :] + w[:, None]*infl_vec[None, :]
    Ls = base.size
    if 0 <= di < Ls:
        aff[:, di] = base[di] + w*infl_vec[di] + w*w
    np.maximum(aff, max_const, out=aff)
    mv = np.max(aff, axis=1)
    vals = np.full(pts.size, math.inf)
    if np.any(valid):
        vals[valid] = (2.0*n*mv[valid])/(den[valid]*den[valid])
    vals = np.where(np.isfinite(vals), vals, math.inf)
    return vals


def _coord_polish(p, stop_time, G=72, seed=12345, consider=None):
    p = _norm_nn(p)
    n = int(p.size)
    if n <= 1:
        return p, math.inf

    c = _fft_conv(p)
    c = np.maximum(c, 0.0)
    S = 1.0
    infl = 2.0*p
    cur_score = float(2.0*n*c.max())
    best_score = cur_score
    best_p = p.copy()
    if consider:
        try:
            consider(best_p)
        except Exception:
            pass

    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    L = 2*n - 1

    while time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)

        for pos in range(n):
            if time.monotonic() > stop_time - 0.4:
                break
            if S <= 1e-300:
                break
            if S > 1.20 or S < 0.80:
                iv = 1.0/S
                p *= iv
                c *= iv*iv
                infl *= iv
                S = 1.0
                c = np.maximum(c, 0.0)
                cur_score = float(2.0*n*c.max())

            i = int(order[pos])
            v = float(p[i])
            start = i
            end = min(i+n, L)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end].copy()
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < L:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

            Sloc = S
            xhi = max(20.0*v, 20.0*Sloc/n, 0.003)
            xhi = min(xhi, max(0.05, v*3.0))
            if not math.isfinite(xhi) or xhi <= 0:
                xhi = max(1e-8, 10.0*Sloc/n)

            di = 2*i - start

            # Build candidate set
            pts_list = [0.0, v, xhi]
            m1 = max(10, int(G*0.4))
            minx = max(1e-10, Sloc/(3000.0*n))
            if minx < xhi:
                pts_list.extend(np.geomspace(minx, xhi, m1).tolist())
            if v > 1e-12:
                m2 = max(6, int(G*0.25))
                factors = np.geomspace(0.2, 5.0, m2)
                pts_list.extend((v*factors).tolist())
            m3 = max(8, int(G*0.30))
            linhi = min(xhi, max(6.0*Sloc/n, 0.001))
            if linhi > 0:
                pts_list.extend(np.linspace(0.0, linhi, m3).tolist())

            pts = np.unique(np.clip(np.asarray(pts_list, dtype=np.float64), 0.0, xhi))
            if pts.size == 0:
                pts = np.array([v])

            vals = _fin_coord_eval(pts, base, infl, max_const, v, Sloc, n, di)
            if not np.any(np.isfinite(vals)):
                continue
            im = int(np.argmin(vals))
            best_x = float(pts[im])
            best_val = float(vals[im])

            # Fine search around best
            if best_val < cur_score + 1e-9:
                step = max(v, Sloc/n, 1e-10)
                lo = max(0.0, best_x - 3.0*step)
                hi = min(xhi, best_x + 3.0*step)
                if hi <= lo:
                    hi = lo + 1e-10
                fine = np.linspace(lo, hi, 32)
                if lo > 0:
                    fine = np.unique(np.concatenate([fine, np.geomspace(lo, hi, 16)]))
                else:
                    fine = np.unique(np.concatenate([fine, [0.0]]))
                fine = np.clip(fine, 0.0, xhi)
                vals2 = _fin_coord_eval(fine, base, infl, max_const, v, Sloc, n, di)
                if np.any(np.isfinite(vals2)):
                    im2 = int(np.argmin(vals2))
                    vx2 = float(fine[im2])
                    vv2 = float(vals2[im2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-11 and 0.0 <= best_x <= xhi:
                d = best_x - v
                if abs(d) > 1e-15:
                    c[start:end] += d*infl[start:end]
                    diag = 2*i
                    if 0 <= diag < L:
                        c[diag] += d*d
                    p[i] = best_x
                    infl[i] = 2.0*best_x
                    S += d
                    cur_score = best_val
                    improved = True

        if S > 1e-300 and math.isfinite(S):
            iv = 1.0/S
            p *= iv
            c *= iv*iv
            infl *= iv
            S = 1.0
        else:
            break
        c = np.maximum(c, 0.0)
        desc = float(2.0*n*c.max())
        if math.isfinite(desc) and desc < best_score - 1e-13:
            best_score = desc
            best_p = p.copy()
            if consider:
                try:
                    consider(best_p)
                except Exception:
                    pass
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 30.0:
                    sigma = 0.04 + 0.03*min(flat-5, 4)
                    noise = rng.normal(0.0, sigma, n)
                    q = p*np.exp(noise)
                    if flat >= 9:
                        x = (np.arange(n)+0.5)/n
                        amp = 0.15 + 0.25*rng.random()
                        f1 = 1 + int(rng.integers(0, 4))
                        ph1 = 2*math.pi*rng.random()
                        q *= np.exp(amp*math.sin(2*math.pi*f1*x + ph1))
                    q = np.maximum(q, 0.0)
                    qs = float(q.sum())
                    if math.isfinite(qs) and qs > 0:
                        p = q/qs
                        c = _fft_conv(p)
                        c = np.maximum(c, 0.0)
                        S = 1.0
                        infl = 2.0*p
                        cur_score = float(2.0*n*c.max())
                        if math.isfinite(cur_score) and cur_score < best_score:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider:
                                try:
                                    consider(best_p)
                                except Exception:
                                    pass
                    flat = 0
                else:
                    break
        sweep += 1

    return best_p, best_score


def _lp_full(p, M, time_limit=60.0):
    """Full-size LP direction at n=3072."""
    if linprog is None or sp is None:
        return None
    n = len(p)
    if n < 128 or n > 4096:
        return None
    t_start = time.monotonic()

    # Build sparse convolution matrix A of shape (2n-1, n)
    # A[k, j] = p[k-j] if 0 <= k-j < n
    idx = np.arange(n)
    cols = np.repeat(idx, n)
    rows = cols + np.tile(np.arange(n), n)
    valid = (rows >= 0) & (rows < 2*n-1)
    rows = rows[valid]
    cols = cols[valid]
    data = p[cols[valid] - rows[valid] + idx[np.searchsorted(idx, cols[valid])]] # This is wrong
    # Let me redo this properly
    # A[k,j] = p[k-j] when 0<=k-j<n
    # For fixed j, k ranges from j to j+n-1, and A[k,j] = p[k-j] = p[0], p[1], ..., p[n-1]
    nnz = 0
    for j in range(n):
        nnz += min(n, 2*n-1-j)  # k from j to min(j+n-1, 2n-2)
    # Total nnz ~ n^2
    if nnz > 20_000_000:  # too many
        return None

    # Build properly
    row_list = []
    col_list = []
    data_list = []
    for j in range(n):
        k_start = j
        k_end = min(j+n, 2*n-1)
        if k_start >= k_end:
            continue
        ks = np.arange(k_start, k_end)
        vs = p[ks - j]
        row_list.append(ks)
        col_list.append(np.full(k_end-k_start, j, dtype=np.int32))
        data_list.append(vs)
        if time.monotonic() - t_start > time_limit*0.5:
            return None

    rows_arr = np.concatenate(row_list)
    cols_arr = np.concatenate(col_list)
    data_arr = np.concatenate(data_list)

    try:
        A = sp.coo_matrix((data_arr, (rows_arr, cols_arr)), shape=(2*n-1, n)).tocsr()
    except Exception:
        return None

    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method="highs",
                      options={"time_limit": time_limit})
    except Exception:
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None))
        except Exception:
            return None

    if res is not None and getattr(res, "success", False):
        b = np.asarray(res.x, dtype=np.float64).ravel()
        b = np.where(np.isfinite(b), b, 0.0)
        b = np.maximum(b, 0.0)
        if b.sum() > 0:
            return b
    return None


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(20.0, budget)
    reserve = 20.0 if budget > 300 else 10.0 if budget > 60 else 3.0
    hard = t0 + budget - reserve
    if hard <= t0 + 2:
        hard = t0 + max(1.5, budget - 0.5)

    parent = _get_parent()
    n = int(parent.size) if parent.size > 0 else 3072
    rng = np.random.default_rng(seed)

    best_arr = _norm_nn(parent.copy())
    best_score = _score_fft(best_arr)
    if not math.isfinite(best_score):
        best_arr = np.full(n, 1.0/n)
        best_score = _score_fft(best_arr)

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return
        if arr.size == 0:
            return
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if not math.isfinite(s) or s < 0.01:
            return
        arr = arr/s
        sc = _score_fft(arr)
        if math.isfinite(sc) and sc < best_score - 1e-13:
            best_arr = arr.copy()
            best_score = sc

    # Phase 1: Try diverse initializations
    phase1_stop = min(hard - 60, t0 + max(1.0, 0.08*budget))
    shapes = _init_shapes(n, rng)
    candidates = [best_arr.copy()]
    for sh in shapes:
        if time.monotonic() > phase1_stop:
            break
        # Try time-reversed too
        candidates.append(sh)
        candidates.append(sh[::-1].copy())
    # Also try power transforms of parent
    for q in (0.6, 0.8, 1.2, 1.5):
        cand = np.maximum(best_arr, 0.0)**q
        candidates.append(_norm_nn(cand))

    # Pick the best candidate
    for cand in candidates:
        if time.monotonic() > phase1_stop:
            break
        cand = _norm_nn(cand)
        consider(cand)

    # Phase 2: L-BFGS on best
    phase2_stop = min(hard - 40, t0 + max(5.0, 0.18*budget))
    if phase2_stop > time.monotonic() + 15:
        p2 = _lbfgs_optimize(best_arr.copy(), phase2_stop, consider=consider)
        consider(p2)

    # Phase 3: LP direction at full size (if time permits)
    lp_stop = min(hard - 30, time.monotonic() + min(90.0, 0.10*budget))
    if lp_stop > time.monotonic() + 20 and n <= 4096:
        c_cur = _fft_conv(best_arr)
        M_cur = float(c_cur.max())
        b_dir = _lp_full(best_arr, M_cur, time_limit=min(60.0, lp_stop - time.monotonic() - 5))
        if b_dir is not None:
            s_dir = float(b_dir.sum())
            if s_dir > 0:
                g = b_dir/s_dir
                for t in (0.5, 0.35, 0.25, 0.15, 0.10, 0.05):
                    cand = (1-t)*best_arr + t*g
                    cand = _norm_nn(cand)
                    consider(cand)

    # Phase 4: More L-BFGS passes
    if time.monotonic() < hard - 60:
        rem = hard - 60 - time.monotonic()
        if rem > 30:
            p4 = _lbfgs_optimize(best_arr.copy(), time.monotonic() + min(120.0, 0.5*rem), consider=consider)
            consider(p4)

    # Phase 5: Coordinate descent (main workhorse)
    coord_stop = min(hard - 10, time.monotonic() + max(30.0, 0.55*(hard - time.monotonic())))
    if coord_stop > time.monotonic() + 20:
        p5, _ = _coord_polish(best_arr.copy(), coord_stop, G=64, seed=seed+777, consider=consider)
        consider(p5)

    # Phase 6: Fourier perturbations + refine
    if time.monotonic() < hard - 40:
        for _ in range(3):
            if time.monotonic() > hard - 40:
                break
            # Fourier perturbation
            F_p = rfft(best_arr, next_fast_len(2*n-1))
            F_p = F_p * (1 + 0.05*rng.standard_normal(F_p.size) + 0.03j*rng.standard_normal(F_p.size))
            # Ensure real and positive
            q = irfft(F_p, next_fast_len(2*n-1))[:n]
            q = np.maximum(q, 0.0)
            q = _norm_nn(q)
            consider(q)
            # Short coord refinement
            if time.monotonic() < hard - 30:
                p6, _ = _coord_polish(best_arr.copy(), time.monotonic() + min(60.0, 0.1*(hard-time.monotonic())), G=48, seed=seed+int(1e6*_), consider=consider)
                consider(p6)

    # Phase 7: Final high-res coordinate polish
    if time.monotonic() < hard - 15:
        final_stop = hard - 5.0
        p7, _ = _coord_polish(best_arr.copy(), final_stop, G=96, seed=seed+424242, consider=consider)
        consider(p7)

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = _norm_nn(parent)
            s = float(out.sum())
        if out.size == 0 or not math.isfinite(s) or s < 0.01:
            out = np.full(64, 1.0/64)
        out = out/float(out.sum())
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            fallback = _norm_nn(parent)
            ls2 = [float(x) for x in fallback]
            sc2 = float(evaluate_sequence(ls2))
            if math.isfinite(sc2):
                return ls2
            return [1.0]
        return ls
    except Exception:
        return [1.0]
