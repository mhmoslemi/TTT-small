```python
def _simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    # v is 1D array
    # We want max(0, v - tau) such that sum = target_sum
    # Sort v in descending order
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    # Find the threshold
    # sum(max(v - tau, 0)) = target_sum
    # For the top k elements above tau: sum(v_sorted[:k]) - k*tau = target_sum
    # tau = (sum(v_sorted[:k]) - target_sum) / k
    # We need v_sorted[k-1] >= tau > v_sorted[k] (0-indexed)
    w = np.cumsum(v_sorted)

    ks = np.arange(1, n+1)

    # i.e., k*v_sorted[k-1] - w[k-1] + target_sum > 0

    valid_idx = np.where(valid)[0]
    if len(valid_idx) == 0:
        return np.full(n, target_sum / n)
    k = valid_idx[-1] + 1
```

```python
def _conv_mat(n, a):
    """Build (2n-1) x n convolution matrix."""
    # Use the structure: A[k,j] = a[k-j] if 0<=k-j<n
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        lo = max(0, j)  # k-j >= 0 => k >= j
        hi = min(2*n-1, j+n)  # k-j < n => k < j+n
        for k in range(lo, hi):
            A[k, j] = a[k-j]
    return A

def _lp(A, M, n, tl):
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(A.shape[0], M),
                      bounds=[(0, None)]*n, method='highs',
                      options={'time_limit': tl, 'presolve': True})
        if res.status == 0 and res.x is not None and np.all(np.isfinite(res.x)):
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None

def _ls(a, g, M):
    n = a.size
    bt, bv = 0.0, M
    for t in np.linspace(0, 1, 13):
        c = np.maximum((1-t)*a + t*g, 0)
        s = c.sum()
        if s < 1e-12: continue
        c /= s
        v = np.convolve(c, c).max()
        if v < bv: bv, bt = v, t
    if bt == 0: return None, M
    lo, hi = max(0, bt-.04), min(1, bt+.04)
    for _ in range(20):
        if hi-lo < 1e-9: break
        m = .5*(lo+hi)
        c = np.maximum((1-m)*a + m*g, 0)
        s = c.sum()
        if s < 1e-12: lo = m; continue
        c /= s
        v = np.convolve(c, c).max()
        if v < bv: bv, bt, hi = v, m, m
        else: lo = m
    c = np.maximum((1-bt)*a + bt*g, 0)
    s = c.sum()
    if s < 1e-12: return None, M
    c /= s
    return c, bv
```

**Key new idea: Use the log-sum-exp smoothing to create a differentiable surrogate, then use scipy.optimize.minimize with BFGS or L-BFGS-B.**

The smooth objective:
f_t(a) = 2n * (1/μ) * log(Σ_k exp(μ * (a*a)_k)) / (sum a)²

For large μ, this approximates the true objective. The gradient is well-defined.

With scipy.optimize.minimize:
- Method: 'L-BFGS-B' (handles box constraints 0 <= a_i <= 1000)
- Objective: f_t(a) + penalty for sum(a)
- Gradient: computed analytically

But the scale invariance means I need to handle the normalization. Let me fix sum(a) = 1 by penalizing deviations:

f(a) = 2n * S_μ(a) + λ * (sum(a) - 1)²

where S_μ(a) = (1/μ) log(Σ_k exp(μ (a*a)_k)).

The gradient:
∂f/∂a_m = 2n * 2/μ * Σ_k p_k a_{k-m} + 2λ(sum(a)-1)

where p_k = exp(μ(a*a)_k) / Σ_j exp(μ(a*a)_j) (softmax weights).

This is computable in O(n²).

Let me try this approach:

```python
def _smooth_obj_grad(a, mu, lam):
    b = np.convolve(a, a)
    u = mu * b
    um = u.max()
    eu = np.exp(u - um)
    Z = eu.sum()
    p = eu / Z  # softmax
    S = (um + np.log(Z)) / mu  # smooth max
    
    n = len(a)
    # Gradient of S wrt a
    # dS/da_m = 2 * sum_k p[k] * a[k-m] (if 0 <= k-m < n)
    # This is a Toeplitz product
    # Use convolution: result[m] = sum_j a[j] * p[m+j]
    # = (a * p_rev)[2n-2-m]
    p_rev = p[::-1]
    conv = np.convolve(a, p_rev)
    grad_S = 2.0 * conv[2*n-2 : n-1 : -1]  # length n
    
    grad = 2*n*grad_S + 2*lam*(a.sum() - 1)
    return 2*n*S + lam*(a.sum()-1)**2, grad
```
