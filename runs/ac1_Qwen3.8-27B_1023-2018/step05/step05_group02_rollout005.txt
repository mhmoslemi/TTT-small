    L = 2*n - 1
    sz = next_fast_len(L)
    F = rfft(a, sz)
    c = irfft(F*F, sz)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(np.where(np.isfinite(a), a, 0.0), 0.0, 1000.0)
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv(a)
    m = float(c.max())
    return float(2.0*n*m/(s*s))

def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.ones(n)/n
    if src.size == n:
        out = src.copy()
    else:
        xp = (np.arange(src.size)+0.5)/src.size
        out = np.interp((np.arange(n)+0.5)/n, xp, src)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    return out/s if s > 1e-300 else np.ones(n)/n

def _step_search(n, time_limit):
    """Grid search over two-level step functions."""
    best_p = None
    best_sc = math.inf
    t_end = time.monotonic() + time_limit
    alphas = np.linspace(0.05, 0.95, 200)
    log_rs = np.linspace(-3, 3, 200)  # r = h1/h2 in [e^-3, e^3]
    
    n_alpha = len(alphas)
    n_r = len(log_rs)
    total = n_alpha * n_r
    
    L = 2*n - 1
    sz = next_fast_len(L)
    
    count = 0
    for ia, alpha in enumerate(alphas):
        if time.monotonic() > t_end:
            break
        m = max(1, min(n-1, int(round(alpha * n))))
        for ir, lr in enumerate(log_rs):
            if time.monotonic() > t_end:
                break
            r = math.exp(lr)

                best_p = a.copy()
            count += 1
