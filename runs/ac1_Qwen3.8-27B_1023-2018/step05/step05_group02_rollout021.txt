                    p[i] = bx
                    S += w
                    conv[start_k:end_k] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        conv[di] += w * w
                    np.maximum(conv, 0.0, out=conv)
                    cur_sc = bv
                    improved = True

        # Rescale
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            conv *= inv * inv
        S = 1.0
        # Re-convolve periodically for accuracy
        if sweep % 20 == 0:
            conv = _conv_self(p)
        ns = float(2.0 * n * conv.max())
        if math.isfinite(ns) and ns < best_sc - 1e-16:
            best_sc = ns
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 8.0:
                    noise = 0.03 + 0.02 * math.exp(-0.05 * flat)
                    p = p * np.exp(rng.normal(0, noise, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                        S = 1.0
                    conv = _conv_self(p)
                    cur_sc = float(2.0 * n * conv.max())
                    if math.isfinite(cur_sc) and cur_sc < best_sc - 1e-15:
                        best_sc = cur_sc
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    flat = 0
                else:
                    break

        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_p = p.copy()
    return best_p, best_sc


def _subgrad_polish(p, stop_time, consider=None, max_iters=500):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    best_p = p.copy()
    best_sc = _score_arr(p)
    step0 = 0.003
    avg_p = p.copy()
    avg_w = 1.0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break
        c = _conv_self(p)
        cmax = float(c.max())
        thresh = cmax * 0.9995
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        active = active[:80]  # cap active set

        grad = np.zeros(n)
        for k in active:
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                grad[valid] += p[kj[valid]] + p[valid]
        grad /= float(len(active))

        step = step0 / (1.0 + 0.01 * it)
        p_cand = p - step * grad
        p_cand = np.maximum(p_cand, 0.0)
        ss = float(p_cand.sum())
        if ss < 1e-15:
            break
        p_cand /= ss
        sc = _score_arr(p_cand)
        if math.isfinite(sc):
            avg_w += 1.0
            avg_p += p_cand
            avg_p /= avg_w
            sc_avg = _score_arr(avg_p)
            if math.isfinite(sc_avg) and sc_avg < best_sc - 1e-15:
                best_sc = sc_avg
                best_p = avg_p.copy()
                if consider is not None:
                    try:
                        consider(best_p)
                    except Exception:
                        pass
            if math.isfinite(sc) and sc < best_sc - 1e-15:
                best_sc = sc
                best_p = p_cand.copy()
                if consider is not None:
                    try:
                        consider(best_p)
                    except Exception:
                        pass
            p = p_cand
        else:
            step0 *= 0.8
            if step0 < 1e-10:
                break

    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 12.0

    parent = _get_parent()
    n_par = parent.size
    p_norm = parent / parent.sum()

    best_arr = p_norm.copy()
    best_score = _score_arr(best_arr)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.clip(a, 0.0, 1000.0)
        a = np.where(np.isfinite(a), a, 0.0)
        if a.size == 0:
            return
        ss = float(a.sum())
        if ss < 0.01:
            return
        sc = _score_arr(a)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_arr = (a / ss).copy()
            best_score = sc

    # Time allocation
    now = time.monotonic()
    total_rem = max(1.0, hard - now)

    # Phase 1: L-BFGS on parent (30%)
    p1_end = now + 0.30 * total_rem
    p = _lbfgs_lse(p_norm, p1_end, consider=consider, fine=False)
    consider(p)

    # Phase 2: Coordinate descent (35%)
    now = time.monotonic()
    p2_end = min(hard - 20.0, now + 0.35 * (total_rem - (p1_end - t0)))
    if p2_end > now + 10.0:
        p, _ = _coord_descent(p, p2_end, seed=int(seed + 11), consider=consider, G=56, max_sweeps=80000)
        consider(p)

    # Phase 3: Fine L-BFGS (15%)
    now = time.monotonic()
    rem = max(0.0, hard - now)
    if rem > 30.0:
        p3_end = now + 0.15 * rem
        p = _lbfgs_lse(p, p3_end, consider=consider, fine=True)
        consider(p)

    # Phase 4: More coordinate descent (20%)
    now = time.monotonic()
    rem = max(0.0, hard - now)
    if rem > 20.0:
        p4_end = now + 0.20 * rem
        p, _ = _coord_descent(p, p4_end, seed=int(seed + 777), consider=consider, G=64, max_sweeps=100000)
        consider(p)

    # Phase 5: Subgradient + final coord (remaining)
    now = time.monotonic()
    rem = max(0.0, hard - now - 5.0)
    if rem > 20.0:
        sg_end = now + 0.35 * rem
        p = _subgrad_polish(p, sg_end, consider=consider, max_iters=800)
        consider(p)

        now = time.monotonic()
        rem2 = max(0.0, hard - now - 3.0)
        if rem2 > 10.0:
            p, _ = _coord_descent(p, min(hard - 3.0, now + rem2), seed=int(seed + 9999),
                                  consider=consider, G=48, max_sweeps=50000)
            consider(p)

    # Try alternate initialization if time permits
    now = time.monotonic()
    rem3 = max(0.0, hard - now - 5.0)
    if rem3 > 60.0:
        rng = np.random.default_rng(int(seed + 5555))
        n_alt = n_par
        x = (np.arange(n_alt) + 0.5) / n_alt
        # Try a "bump" initialization
        bump = 1.0 + 0.5 * np.exp(-((2*x - 1)**2) / 0.3)
        p_alt = parent / parent.sum() * bump
        p_alt = np.maximum(p_alt, 0.0)
        p_alt /= p_alt.sum()
        consider(p_alt)
        alt_end = now + 0.5 * rem3
        p_alt = _lbfgs_lse(p_alt, alt_end, consider=consider, fine=False)
        consider(p_alt)
        now = time.monotonic()
        rem4 = max(0.0, hard - now - 3.0)
        if rem4 > 15.0:
            p_alt, _ = _coord_descent(p_alt, min(hard - 3.0, now + rem4),
                                      seed=int(seed + 2222), consider=consider, G=48, max_sweeps=30000)
            consider(p_alt)

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = evaluate_sequence(ls)
            if not math.isfinite(final_sc):
                return [1.0 / max(1, int(parent.size))]
        return ls
    except Exception:
        return [1.0]
