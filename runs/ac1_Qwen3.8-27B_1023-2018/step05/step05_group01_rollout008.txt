            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * g
            f = np.maximum(f, 0.0)
            f = f / float(f.sum())
            if best_sc < best_score:
                best_score = best_sc
                best_f = f.copy()
                consider(best_f)
        else:
            # No improvement, try a small perturbation
            rng = np.random.default_rng(it + 42)
            f_pert = f * np.exp(rng.normal(0, 0.01, n))
            f_pert = np.maximum(f_pert, 0.0)
            f_pert /= float(f_pert.sum())
            sc_pert = _score_fft(f_pert)
            if math.isfinite(sc_pert) and sc_pert < best_score:
                best_score = sc_pert
                best_f = f_pert.copy()
                consider(best_f)
            f = f_pert
    
    return best_f
