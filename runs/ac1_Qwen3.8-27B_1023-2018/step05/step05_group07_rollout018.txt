            continue
        c /= s
        v = np.convolve(c, c).max()
        if v < bv:
            bv, bt = v, m
            hi = m
        else:
            lo = m
    c = np.maximum((1.0 - bt) * a + bt * g, 0.0)
    s = c.sum()
    if s < 1e-12:
        return None, M
    return c / s, bv


def _smooth_grad(a, beta):
    n = len(a)
    c = np.convolve(a, a)
    mc = c.max()
    ec = np.exp(beta * (c - mc))
    w = ec / ec.sum()
    g = np.zeros(n)
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        m = end - j
        if m > 0:
            g[j] = w[j:end] @ a[:m]
    g *= 2.0
    g -= g.mean()
    return g


def _optimize(a0, deadline, seed=42):
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a)
    a /= a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0
    
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        M = float(c.max())
        
        if M < best_M * (1.0 - 1e-13):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 200:
            sig = min(0.02 + 0.01 * (stuck // 50), 0.3)
            a *= np.exp(rng.normal(0.0, sig, n))
            a = np.maximum(a, 1e-10)
            a /= a.sum()
            stuck = 0
            continue
        
        A = _band_mat(n, a)
        moved = False
        
        # LP best response
        for rel in (1.0 + 1e-10, 1.0 + 1e-7, 1.0 + 1e-4):
            g = _lp(A, M * rel, n, tl=3.0)
            if g is None or g.sum() < 1e-10:
                continue
            g /= g.sum()
            if np.max(np.abs(g - a)) < 1e-9:
                continue
            cand, val = _linesearch(a, g, M)
            if cand is not None and val < M * (1.0 - 1e-12):
                a = cand
                moved = True
                break
        
        # Gradient
        if not moved:
            for beta in (100.0, 250.0, 600.0):
                grad = _smooth_grad(a, beta)
                gn = float(np.linalg.norm(grad))
                if gn < 1e-15:
                    break
                for alpha in (8.0/n, 4.0/n, 2.0/n, 1.0/n, 0.5/n):
                    cand = np.maximum(a - alpha * grad / gn, 0.0)
                    cs = cand.sum()
                    if cs < 1e-12:
                        continue
                    cand /= cs
                    v = float(np.convolve(cand, cand).max())
                    if v < M * (1.0 - 1e-12):
                        a = cand
                        moved = True
                        break
                if moved:
                    break
        
        # Small perturbation
        if not moved:
            a *= np.exp(rng.normal(0.0, 0.005, n))
            a = np.maximum(a, 1e-10)
            a /= a.sum()
    
    return best_a


def _polish(a0, deadline, seed=777):
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, 1000).copy()
    a /= max(a.sum(), 1e-15)
    n = len(a)
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        M = float(c.max())
        if M < best_M * (1.0 - 1e-14):
            best_M = M
            best_a = a.copy()
        
        # Try coordinate pair adjustment
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        
        d = a[i] - a[j]
        for frac in (0.05, 0.1, 0.2, 0.3):
            delta = frac * d
            if abs(delta) < 1e-18:
                continue
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 1e-12:
                continue
            cand /= s
            v = float(np.convolve(cand, cand).max())
            if v < M * (1.0 - 1e-14):
                a = cand
                break
        
        # Also try tiny gradient step
        grad = _smooth_grad(a, 500.0)
        gn = float(np.linalg.norm(grad))
        if gn > 1e-15:
            for alpha in (0.1/n, 0.05/n, 0.01/n):
                cand = np.maximum(a - alpha * grad / gn, 0.0)
                cs = cand.sum()
                if cs < 1e-12:
                    continue
                cand /= cs
                v = float(np.convolve(cand, cand).max())
                if v < M * (1.0 - 1e-14):
                    a = cand
                    break
    
    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    hard_deadline = t0 + budget - 5.0  # 5s headroom
    
    p = _get_parent()
    n0 = len(p)
    
    best = np.clip(p.astype(np.float64), 0, 1000).copy()
    bs = _eval(best)
    
    def upd(cand):
        nonlocal best, bs
        try:
            sc = _eval(cand)
            if np.isfinite(sc) and sc < bs:
                bs = sc
                best = np.clip(np.asarray(cand, dtype=np.float64), 0, 1000).copy()
        except Exception:
            pass
    
    # Phase 1: LP optimize parent (~45% of budget)
    t1 = t0 + min(580.0, budget * 0.45)
    try:
        upd(_optimize(p, t1, seed=1))
    except Exception:
        pass
    
    # Phase 2: Nearby lengths (~25% of budget)
    t2 = t0 + min(900.0, budget * 0.72)
    deltas = [-20, -10, -5, 5, 10, 20, 30]
    for dn in deltas:
        if time.monotonic() > t2 - 20:
            break
        nt = n0 + dn
        if nt < 50 or nt > 500:
            continue
        x0 = np.linspace(0, 1, n0)
        x1 = np.linspace(0, 1, nt)
        init = np.clip(np.interp(x1, x0, p), 0, 1000)
        if init.sum() < 1e-10:
            continue
        init /= init.sum()
        alloc = min(time.monotonic() + 40.0, t2 - 5.0)
        try:
            upd(_optimize(init, alloc, seed=500 + nt))
        except Exception:
            pass
