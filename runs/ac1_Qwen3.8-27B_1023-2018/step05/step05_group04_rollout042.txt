                if flat >= 3:
                    if time.monotonic() < stop_time - 20.0:
                        kick_size = 0.015 + 0.025 * min(1.0, flat / 15.0)
                        p = p * np.exp(rng.normal(0.0, kick_size, n_local))
                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss > 0:
                            p /= ss
                        c = autocorr(p)
                        cur_score = 2.0 * n_local * float(c.max())
                        flat = 0
                    else:
                        break
            sweep += 1

        return best_p, best_score

    # Run coordinate descent
    cd_end = hard - 8.0
    if cd_end > time.monotonic() + 30.0:
        p_cd, _ = coord_descent(best_p, cd_end)
        consider(p_cd)

    # If time remains, try another CD pass from a slightly perturbed start
    rem = hard - 10.0 - time.monotonic()
    if rem > 60.0:
        rng2 = np.random.default_rng(int(seed + 777))
        p_pert = best_p * np.exp(rng2.normal(0.0, 0.03, n))
        p_pert = np.maximum(p_pert, 0.0)
        p_pert /= float(p_pert.sum())
        consider(p_pert)
        p_cd2, _ = coord_descent(p_pert, hard - 10.0)
        consider(p_cd2)

    # Final validation
    out = np.asarray(best_p, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    if out.size == 0 or float(out.sum()) < 0.01:
        out = np.array([1.0])
    result = [float(x) for x in out]
    try:
        final_sc = float(evaluate_sequence(result))
        if not math.isfinite(final_sc):
            result = [1.0]
    except Exception:
        result = [1.0]
    return result
