    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def _subgrad_step(a, step, deadline):
    """One subgradient step on M = max_k (a*a)[k], projected to simplex."""
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    # Active set: lags where c[k] is within tolerance of M
    threshold = M - 1e-10 * M
    active = np.where(c >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    # Subgradient: average of gradients over active lags
    grad = np.zeros(n)
    for k in active:
        k_int = int(k)
        # d(a*a)[k]/da[m] = 2*a[k-m] if 0<=k-m<n
        k_minus_m = k_int - np.arange(n)
        valid = (k_minus_m >= 0) & (k_minus_m < n)
        grad[valid] += 2.0 * a[k_minus_m[valid]]
    grad /= len(active)
    
    # Step
    a_new = a - step * grad
    a_new = _project_simplex(a_new)
    return a_new, M, active

def _lp_sweep(a, deadline, max_iters=200):
    """Subgradient + LP hybrid optimization."""
    n = len(a)
    a = _project_simplex(a.copy())
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    
    # Adaptive step size
    step = 0.1
    step_decay = 0.9995
    step_min = 0.001
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        a_new, M, active = _subgrad_step(a, step, deadline)
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a_new.copy()
        
        a = a_new
        step *= step_decay
        if step < step_min:
            step = max(step_min, 0.5 * step + 0.5 * 0.01)  # decay to 0.01
        
        # Periodic LP refinement
        if it % 50 == 49 and step < 0.02:
            # Build LP matrix
            k_idx = np.arange(2*n-1)[:, None]
            j_idx = np.arange(n)[None, :]
            d = k_idx - j_idx
            mask = (d >= 0) & (d < n)
            A_lp = np.zeros((2*n-1, n))
            A_lp[mask] = a[d[mask]]
            
            M_lp = float(np.convolve(a, a).max())
            res = linprog(-np.ones(n), A_ub=A_lp, b_ub=np.full(2*n-1, M_lp*(1+1e-10)),
                         bounds=[(0, None)]*n, method='highs',
                         options={'time_limit': 3.0})
            if res.status == 0 and res.x is not None:
                g = np.clip(res.x, 0, None)
                if g.sum() > 1e-12:
                    g = _project_simplex(g)
                    # Line search between a and g
                    for t in [0.3, 0.5, 0.7, 1.0]:
                        cand = _project_simplex((1-t)*a + t*g)
                        c_val = np.convolve(cand, cand).max()
                        if c_val < best_M:
                            a = cand
                            best_M = c_val
                            best_a = cand.copy()
                            break
    
    return best_a

Hmm, but this subgradient approach might be slow because:
1. The step size decays, so progress slows
2. The function is non-convex, so subgradient may not converge to the global minimum
3. The active set changes, making the subgradient noisy

Let me think about a better approach. The LP best-response is actually quite powerful because it finds an exact solution to a related problem. Let me focus on that.

Actually, let me reconsider the problem structure. We want to minimize:
  f(a) = 2n * max_k (a*a)[k] / (sum a)^2

By homogeneity, we can set sum(a) = 1 and minimize 2n * max_k (a*a)[k].

The function g(a) = max_k (a*a)[k] is the max of quadratic forms. It's quasiconvex (sublevel sets {a: (a*a)[k] <= t for all k} are convex if each quadratic form is convex... but as we noted, they're not all convex).

However, the LP approach avoids this non-convexity by linearizing. Given a current point a with value M, the LP finds the best g that satisfies the linear constraint (a*g)[k] <= M. This is a "tangent plane" approach.

Let me think about the convergence of the best-response dynamics. In the bilinear game:
- Player 1 (row) chooses a, Player 2 (column) chooses g
- Payoff: max_k (a*g)[k]
- Player 1 maximizes, Player 2 minimizes

Wait, that's not quite right. Let me re-derive.

The LP: max 1'g s.t. A_a * g <= M * 1, g >= 0.
where A_a is the matrix with (A_a)_{k,j} = a_{k-j} (the convolution matrix).

This is equivalent to: max 1'g s.t. max_k (a*g)_k <= M, g >= 0.

By strong duality, this equals: min M' * 1'y s.t. A_a' * y >= 1, y >= 0. (where y has 2n-1 components)

At the optimum, if g* is the optimal solution and g* != a/sum(a), then we can move from a toward g* to reduce the self-correlation peak.

The key property: at a Nash equilibrium of the game where both players use the same distribution, a = g*, and the value M* is the min-max value. This is related to the von Neumann minimax theorem for the bilinear game.

But wait, the game max_k (a*g)_k is not bilinear in (a,g) because of the max. However, we can reformulate:
  max_k (a*g)_k = max_{y >= 0, 1'y=1} sum_k y_k (a*g)_k = max_{y in simplex} a' (C_y) g

where C_y = sum_k y_k C_k is a convex combination of the Toeplitz matrices. So the game is:
  Player 1 chooses a (simplex), Player 2 chooses g (simplex)
  Payoff: max_{y in simplex} a' C_y g

This is a "max over y of a bilinear form", which is the upper value of a bilinear game. By minimax, the value v satisfies:
  v = max_a min_g max_y a' C_y g

Hmm, this is getting complicated. Let me just focus on the practical algorithm.

I think the best approach is:
1. Use the LP best-response as the main optimization step (it's exact and strong)
2. Use subgradient as a fallback when LP doesn't improve
3. Try multiple lengths
4. Use the parent as the main initialization

Let me also consider: for n=281, the LP is 561 constraints x 281 variables. The matrix A has about 281*281 ≈ 79K nonzeros (each row has at most n nonzeros, but on average about n/2). This is very small for an LP solver.

Actually, let me count more carefully. Row k of A has nonzeros at columns j where 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= min(n-1, k). The number of nonzeros in row k is min(n, k+1, 2n-1-k). For k=0: 1 nonzero. For k=n-1: n nonzeros. For k=n: n-1 nonzeros. Total nonzeros = sum_{k=0}^{2n-2} min(n, k+1, 2n-1-k) = roughly n^2.
