        # = 2*(w * a_reversed_conv)[appropriate position]
        # In FFT: if W = FFT(w), A = FFT(a), then (w*a)[k] = ifft(W*A)[k]
        # And sum_k w[k]*a[k-i] = (w*a)[i]... no.
        # (w*a)[k] = sum_i w[i]*a[k-i]

        # Let j = k-i: = sum_j w[j+i]*a[j] = cross-correlation of w and a at lag i
        # = (a * w)[i+n-1] ... 
        # OK I'll just use: grad_a[i] = 2 * (cross_corr_of_w_and_a_at_lag_i)
        # cross_corr[i] = sum_j a[j]*w[j+i]
        # Which equals: (a_full * w_full)[n-1+i] where *_full is full convolution? No...
        # (a*w)[k] = sum_j a[j]*w[k-j]. Set k = n-1+i: (a*w)[n-1+i] = sum_j a[j]*w[n-1+i-j]
        # That's not quite what I want either.
        # 
        # I'll just directly compute using the FFT of a flipped w:
        # flip_w = w[::-1], FFT flip_w, multiply by FFT(a), IFFT -> gives (a * flip_w)[k] = sum_j a[j]*flip_w[k-j] = sum_j a[j]*w[n-1-k+j]
        # Hmm this is getting nowhere. Let me just do it with a simple O(n) per element approach
        # but vectorized over the "u" parameter. Actually for the LSE phase I can afford to be
        # a bit less efficient since it's only 10% of time. Let me just compute the gradient
        # by finite differences or by a simple loop-free formula.
        #
        # SIMPLIFICATION: For the LSE phase, I'll use the subgradient of max(conv) which is
        # 2*convolve(p, e_j) where j argmax. For the soft version with weights w, it's
        # 2*convolve(p, w) at the right indices. But convolve(p,w) has length 2n-1 and I
        # need to extract the n "diagonal" entries.
        #
        # Actually the simplest correct formula:
        # grad_f/da[i] = 2 * sum_{k} w[k] * (a[k-i] if 0<=k-i<n else 0 + a[i-k] if 0<=i-k<n else 0)
        # The first term: sum_k w[k]*a[k-i] for k-i in [0,n-1] = sum_j w[i+j]*a[j] (j=0..n-1, i+j < 2n-1)
        # The second term: sum_k w[k]*a[i-k] for i-k in [0,n-1] = sum_j w[i-j]*a[j] (j=0..i, i-j < 2n-1)
        # 
        # Let me define: C[i] = sum_j a[j]*w[i+j] (first term) and D[i] = sum_j a[j]*w[i-j] (second term)
        # Then grad_a[i] = 2*(C[i] + D[i])
        # C is the cross-correlation of a and w: C[i] = <a, shift_i(w)>
        # In numpy: C = np.correlate(w, a, 'full')[n-1:] ... let me verify
        # np.correlate(a, v, 'full')[k] = sum_j a[j+k-1]*conj(v[j]) for 'full' mode... 
        # Actually numpy docs: for 1D, correlate(a, v)[k] = sum_n a[n+k]*conj(v[n]) for k=-(M-1)..(N-1)
        # where len(a)=N, len(v)=M. With both length n: k ranges from -(n-1) to n-1.
        # So correlate(a, w, 'full') has length 2n-1, indexed -n+1 to n-1 -> 0 to 2n-2.
        # correlate(a, w, 'full')[n-1+i] = sum_n a[n+i]*w[n] = sum_j a[j+i]*w[j] for valid j
        # That's C[i]! (with C[i] = sum_j a[j+i]*w[j], which equals sum_j a[j]*w[j-i]... wait)
        # Hmm: sum_j a[j+i]*w[j] vs sum_j a[j]*w[j-i]. These are the same by reindexing (let k=j+i).
        # So C[i] = correlate(a, w, 'full')[n-1+i]
        # Similarly D[i] = sum_j a[j]*w[i-j] = sum_j a[j]*w[j-i] (by symmetry, since w is real)
        # D[i] = correlate(a, w, 'full')[n-1-i]
        # So grad_a[i] = 2*(correlate(a,w,'full')[n-1+i] + correlate(a,w,'full')[n-1-i])
        # Great!
        corr_aw = np.correlate(a, w, mode='full')
        # corr_aw has length 2n-1, corr_aw[k] = sum_j a[j+k-(n-1)]*w[j]
        # I need: C[i] = sum_j a[j+i]*w[j] = corr_aw at index (n-1+i)
        # D[i] = sum_j a[j]*w[j-i] = corr_aw at index (n-1-i) ... 
        # Let me verify with small example. a=[a0,a1], w=[w0,w1], n=2.
        # corr_aw[0] = a[0-1]*w[0] + ... hmm numpy: 
        # correlate([a0,a1],[w0,w1],'full') = 
        # index 0: a0*w0 (only a[0]*w[0] since k=0 means a[j+0-(2-1)]*w[j] = a[j-1]*w[j])
        # Actually let me just test: np.correlate([1,2,3],[0,1,0],'full') = [0,1,2,3,0,1,0]? No.
        # np.correlate(a,v) = sum_n a[n+k]*conj(v[n]) for k from -(len(v)-1) to len(a)-1
        # For a=[a0,a1], v=[w0,w1]: k from -1 to 1.
        # k=-1: sum_n a[n-1]*v[n] = a[-1]*v[0] + a[0]*v[1] = 0 + a0*w1 = a0*w1
        # k=0: a0*w0 + a1*w1
        # k=1: a1*w0
        # So correlate = [a0*w1, a0*w0+a1*w1, a1*w0], length 3 = 2n-1.
        # corr[1] = a0*w0+a1*w1 = center.
        # I need C[i] = sum_j a[j+i]*w[j]:
        # C[0] = a0*w0+a1*w1 = corr[1] = corr[n-1]  ✓
        # C[1] = a1*w0 = corr[2] = corr[n-1+1]  ✓
        # D[i] = sum_j a[j]*w[i-j]:
        # D[0] = a0*w0+a1*w1 = corr[1]
        # D[1] = a0*w1 = corr[0] = corr[n-1-1]  ✓
        # So: grad_a[i] = 2*(corr[n-1+i] + corr[n-1-i])
        # But we need to handle bounds: corr has length 2n-1, indices 0 to 2n-2.
        # n-1+i is valid for i=0..n-1 (gives n-1 to 2n-2) ✓
        # n-1-i is valid for i=0..n-1 (gives n-1 to 0) ✓
        
        # But wait, this formula assumes the "full" correlation. Let me just use it:
        ga = 2.0 * (corr_aw[n-1:n-1+n] + corr_aw[n-1:-1 if n > 1 else None][::-1])
        # Actually: corr_aw[n-1-i] for i=0..n-1 gives corr_aw[n-1], corr_aw[n-2], ..., corr_aw[0]
        # Which is corr_aw[n-1::-1]
        
        # Now chain rule to log-space: da_i/du_j = a_i*(delta_ij - a_j)
        # grad_u_j = sum_i ga[i] * a_j*(delta_ij - a_j) = ga[j]*a_j - a_j*sum_i ga[i]*a_i
        gu = ga * a - a * float(np.dot(ga, a))
        gu = np.where(np.isfinite(gu), gu, 0.0)
        return float(f), gu

    # Run L-BFGS with decreasing T
    u = np.log(np.maximum(a, 1e-15))
    u -= np.mean(u)
    best_p = a.copy()
    best_sc = _score_arr(a)

    stages = [(0.008, 30), (0.002, 40), (0.0005, 50)]
    for T_frac, maxit in stages:
        if time.monotonic() > stop_time - 3:
            break
        c = cfast(a)
        T = T_frac * max(1e-10, float(c.max()))
        opts = {"maxiter": maxit, "gtol": 1e-10, "ftol": 1e-14, "maxls": 10, "maxcor": 10, "disp": False}
        try:
            res = minimize(obj_grad, u, args=(T,), method='L-BFGS-B', jac=True, options=opts)
            if res.x is not None and np.all(np.isfinite(res.x)):
                p_new = logsof_to_prob(res.x)
                sc = _score_arr(p_new)
                if math.isfinite(sc) and sc < best_sc - 1e-12:
                    best_sc = sc
                    best_p = p_new.copy()
                    a = p_new.copy()
                    u = np.log(np.maximum(a, 1e-15))
                    u -= np.mean(u)
                    consider(a)
        except Exception:
            pass

    return best_p


def _coord_descent(a, stop_time, G=80, seed=42, consider=None, verbose=False):
    """Aggressive coordinate descent with incremental convolution tracking."""
    a = _normalize(a)
    n = a.size
    if n <= 1:
        return a, math.inf

    L = 2 * n - 1
    conv = _fft_conv(a)
    cur_score = float(2.0 * n * conv.max())
    best_score = cur_score
    best_a = a.copy()
    S = 1.0  # sum of a (should stay ~1)

    if consider is not None:
        consider(best_a)

    rng = np.random.default_rng(seed)
    flat_sweeps = 0
    max_flat = 6
    noise_sigs = [0.03, 0.06, 0.10, 0.15, 0.22]
    
    sweep_num = 0
    while time.monotonic() < stop_time - 1.0:
        sweep_num += 1
        improved = False
        order = rng.permutation(n)

        for pos in range(n):
            now = time.monotonic()
            if now > stop_time - 0.3:
                return best_a, best_score

            i = int(order[pos])
            v = float(a[i])

            # Determine affected range in conv
            # Changed entries: k where d*a[k-i] or d*a[i-k] is nonzero, plus k=2i for d^2
            # k range for d*a[k-i]: i <= k <= i+n-1, but k must be in [0, L-1]
            # k range for d*a[i-k]: i-(n-1) <= k <= i, but k must be in [0, L-1]
            k_lo = max(0, i - (n - 1))
            k_hi = min(L - 1, i + (n - 1))
            if k_hi <= k_lo:
                continue

            # Build influence vector for affected range
            # infl[k] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
            segments = []
            # a[k-i] for k in [max(i, k_lo), min(i+n-1, k_hi)] -> indices k-i in [0, n-1]
            seg1_start = max(i, k_lo)
            seg1_end = min(i + n - 1, k_hi)
            if seg1_end >= seg1_start:
                offsets1 = np.arange(seg1_start - i, seg1_end - i + 1)
                segments.append((seg1_start, a[offsets1]))
            # a[i-k] for k in [max(k_lo, i-n+1), min(i, k_hi)] -> indices i-k in [0, n-1]
            seg2_start = max(k_lo, i - n + 1)
            seg2_end = min(i, k_hi)
            if seg2_end >= seg2_start:
                offsets2 = i - np.arange(seg2_start, seg2_end + 1)
                segments.append((seg2_start, a[offsets2]))

            # Combine into a single influence array over [k_lo, k_hi]
            rng_len = k_hi - k_lo + 1
            infl = np.zeros(rng_len, dtype=np.float64)
            for start, vals in segments:
                idx = np.arange(start, start + len(vals)) - k_lo
                infl[idx] += vals

            # Diagonal position within the range
            diag_pos = 2 * i - k_lo if k_lo <= 2 * i <= k_hi else -1

            # Unaffected max
            max_unaff = 0.0
            if k_lo > 0:
                mu = float(conv[:k_lo].max())
                if mu > max_unaff:
                    max_unaff = mu
            if k_hi < L - 1:
                mu = float(conv[k_hi + 1:].max())
                if mu > max_unaff:
                    max_unaff = mu

            base = conv[k_lo:k_hi + 1].copy()

            # Generate candidate values for a[i]
            candidates = [0.0, v]
            # Geometric range around v
            if v > 1e-12:
                geo_lo = max(1e-10, v * 0.01)
                geo_hi = max(v * 50.0, S / n * 50.0)
                geo_hi = min(geo_hi, S * 0.5)  # Don't make any single element dominate
                m_geo = max(8, G // 3)
                candidates.extend(np.geomspace(geo_lo, geo_hi, m_geo).tolist())
            
            # Linear range
            lin_hi = min(S * 0.3, max(S / n * 20.0, v * 10.0, 0.001))
            m_lin = max(8, G // 3)
            if lin_hi > 0:
                candidates.extend(np.linspace(0.0, lin_hi, m_lin).tolist())

            # Also try values near the current best of the affected range
            # The optimal new value balances the affected entries
            # Quick estimate: if we set a[i]=x, affected conv[k] = base[k] + (x-v)*infl[k] + (x-v)^2*delta_{diag}
            # We want to minimize max of these. Top two candidates:
            # Make the max of affected equal to the max of unaffected
            # Or make two affected entries equal
            if infl.max() > 1e-15:
                # Try to set a[i] so that peak affected = max_unaff
                # base[k] + d*infl[k] + d^2 (if diag) = max_unaff
                # For the k where base+ d*infl is largest:
                # This is complex, so just add a few smart candidates
                peak_idx = int(np.argmax(base + v * infl))
                target = max_unaff
                if base[peak_idx] > 0 and infl[peak_idx] > 0:
                    x_est = (target - base[peak_idx]) / infl[peak_idx] + v
                    if 0 <= x_est <= S:
                        candidates.append(x_est)
                        candidates.append(x_est * 0.8)
                        candidates.append(x_est * 1.2)

            candidates = np.unique(np.clip(np.asarray(candidates, dtype=np.float64), 0.0, S))
            if candidates.size <= 2:
                candidates = np.array([0.0, v, min(S, 2.0 * S / n)])

            # Evaluate all candidates vectorized
            # For candidate x: d = x - v
            # new_conv[k] = base[k] + d*infl[k] + d^2 (if k is diag)
            # score candidate = 2*n*max(new_conv_max, max_unaff) / S_new^2
            # S_new = S - v + x (but we normalize, so effectively S_new/S)
            # Actually since we normalize at the end, and score is homogeneous, let's track S.
            # After change: S_new = S - v + x
            # Normalized a_new has sum 1, conv scales by (1/S_new)^2
            # So effective max = max(new_conv, max_unaff) / S_new^2 (if S_new > 0)
            
            d = candidates - v  # shape (G,)
            # new_vals[k, j] = base[k] + d[j]*infl[k] + d[j]^2 (if k==diag)
            # Shape: (rng_len, G)
            new_vals = base[:, None] + d[None, :] * infl[:, None]
            if 0 <= diag_pos < rng_len:
                new_vals[diag_pos, :] += d * d

            new_max_aff = new_vals.max(axis=0)  # shape (G,)
            eff_max = np.maximum(new_max_aff, max_unaff)  # shape (G,)
            S_new = S - v + candidates  # shape (G,)
            valid = S_new > 1e-15
            scores = np.full(candidates.size, math.inf, dtype=np.float64)
            if np.any(valid):
                scores[valid] = 2.0 * n * eff_max[valid] / (S_new[valid] * S_new[valid])

            if not np.any(np.isfinite(scores)):
                continue

            im = int(np.argmin(scores))
            best_x = float(candidates[im])
            best_sc = float(scores[im])

            # Local refinement around best_x
            if best_sc < cur_score - 1e-13:
                lo = max(0.0, best_x - max(0.1 * best_x, S / (100.0 * n), 1e-8))
                hi = min(S, best_x + max(0.1 * best_x, S / (100.0 * n), 1e-8))
                fine = np.linspace(lo, hi, 24)
                fd = fine - v
                fnv = base[:, None] + fd[None, :] * infl[:, None]
                if 0 <= diag_pos < rng_len:
                    fnv[diag_pos, :] += fd * fd
                fmax = fnv.max(axis=0)
                feff = np.maximum(fmax, max_unaff)
                fS = S - v + fine
                fvalid = fS > 1e-15
                fscores = np.full(fine.size, math.inf)
                if np.any(fvalid):
                    fscores[fvalid] = 2.0 * n * feff[fvalid] / (fS[fvalid] * fS[fvalid])
                if np.any(np.isfinite
