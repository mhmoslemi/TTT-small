```python
def _lbfgs_smooth(p, n, fft_size, stop_time, consider, seed):
    p = p.copy()
    S = float(p.sum())
    p = p / S
    
    # Work in log-space: a_i = exp(u_i) / sum(exp(u))
    # This ensures a_i > 0 and sum(a) = 1
    
    u = np.log(p + 1e-30)  # log-space
    u = u - u.mean()  # center for numerical stability
    
    L = 2 * n - 1
    
    def softmax(u):
        m = u.max()
        e = np.exp(u - m)
        return e / e.sum()
    
    def obj_and_grad(u, T):
        a = softmax(u)
        F = rfft(a, fft_size)
        c = irfft(F * F, fft_size)[:L]
        cmax = c.max()
        
        # Log-sum-exp: T * log(sum(exp((c - cmax)/T)))
        if T < 1e-10:
            T = 1e-10 * max(cmax, 1)
        scaled = (c - cmax) / T
        # For numerical stability
        log_sum = np.log(np.exp(scaled).sum() / L)
        f = cmax + T * log_sum  # this is the smoothed max
        
        # Gradient: ∂f/∂a_i = sum_k w_k * ∂c_k/∂a_i
        # where w_k = exp(scaled_k) / sum(exp(scaled))
        w = np.exp(scaled) / np.exp(scaled).sum()
        
        # ∂c_k/∂a_i = 2 * a[k-i] (for valid k-i)
        # So ∂f/∂a = 2 * conv(w, a) (with appropriate truncation)
        # conv(w, a)[i] = sum_k w[k] * a[k-i]... 
        # Actually: ∂f/∂a_i = sum_k w_k * 2*a[k-i]
        # = 2 * sum_k w_k * a[k-i]
        # = 2 * (w reversed * a)[i]... 
        # Let me think: sum_k w[k] * a[k-i] for k = 0, ..., 2n-2
        # Let j = k-i: sum_j w[i+j] * a[j] for j = 0, ..., n-1
        # = correlation of w with a
        # This is the cross-correlation, which can be computed via FFT
        
        # Actually, let me just compute it directly for clarity
        # grad_a[i] = 2 * sum_{k=i}^{min(i+n-1, 2n-2)} w[k] * a[k-i]
        # = 2 * sum_{j=0}^{min(n-1, 2n-2-i)} w[i+j] * a[j]
        
        # Using FFT: this is the cross-correlation of w (padded) and a
        cross_size = next_fast_len(3 * n - 2)
        Fw = rfft(np.concatenate([w, np.zeros(n - 1)]), cross_size)
        Fa = rfft(a, cross_size)
```

```python
        if math.isfinite(new_sc) and new_sc < best_score - 1e-14:
            best_score = new_sc
            best_a = a.copy()
            consider(best_a)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4 and time.monotonic() < stop_time - 80.0:
                sigma = 0.015 + 0.005 * min(flat, 8)
                noise = rng.normal(0.0, sigma, n)
                a *= np.exp(noise)
                a = np.maximum(a, 0.0)
                S = float(a.sum())
                if S > 1e-300:
                    b = conv_full(a)
                    cur_score = 2.0 * n * float(b.max()) / (S * S)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                        consider(best_a)
                flat = 0

        if flat >= 30:
            break

    return best_a


def _fine_polish(a, n, fft_size, L, stop_time, consider, seed):
    """Polish with finer grid on important coordinates."""
    a = np.maximum(a, 0.0).copy()
    S = float(a.sum())
    b = conv_full(a)
    cur_score = 2.0 * n * float(b.max()) / (S * S)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng(seed + 777)

    # Find active lags
    bmax = float(b.max())
    thresh = bmax * 0.99
    active = np.where(b >= thresh)[0]

    # Important coordinates
    important = np.zeros(n, dtype=bool)
    for lag in active:
        lo = max(0, int(lag) - n + 1)
        hi = min(n - 1, int(lag))
        important[lo:hi + 1] = True
    imp_idx = np.where(important)[0]
    if len(imp_idx) > 200:
        counts = np.zeros(n, dtype=int)
        for lag in active:
            lo = max(0, int(lag) - n + 1)
            hi = min(n - 1, int(lag))
            counts[lo:hi + 1] += 1
        imp_idx = imp_idx[np.argsort(-counts[imp_idx])][:200]
    other_idx = np.where(~important)[0]

    G = 64
    flat = 0
    sweep = 0

    while time.monotonic() < stop_time - 2.0:
        sweep += 1
        improved = False
        if sweep % 3 != 0:
            order = rng.permutation(imp_idx)
        else:
            order = rng.permutation(other_idx)

        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            if S <= 1e-300:
                break

            v = float(a[i])
            base = b[i:i + n].copy()
            max_out = 0.0
            if i > 0:
                max_out = float(b[:i].max())
            if i + n < L:
                mo2 = float(b[i + n:L].max())
                if mo2 > max_out:
                    max_out = mo2

            # Narrow range for polishing
            x_hi = v + max(4.0 * v, 15.0 / n, 0.001)
            x_lo = max(0.0, v - max(2.0 * v, 7.0 / n, 0.0005))
            x_hi = min(x_hi, v * 3.0 + 0.01)

            pts = np.linspace(x_lo, x_hi, G + 1)
            d = pts - v
            aff = base[None, :] + d[:, None] * (2.0 * a)[None, :]
            aff[:, i] += d * d
            maff = np.maximum(aff.max(axis=1), max_out)
            denoms = S + d
            vals = np.where(denoms > 1e-15, 2.0 * n * maff / (denoms * denoms), np.inf)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Fine golden section
            dw = (x_hi - x_lo) / G
            lo_gs = max(0.0, best_x - 1.5 * dw)
            hi_gs = min(x_hi, best_x + 1.5 * dw)
            if hi_gs - lo_gs > 1e-15:
                gr = (math.sqrt(5.0) - 1.0) / 2.0
                ag, bg = lo_gs, hi_gs
                cg = bg - gr * (bg - ag)
                dg = ag + gr * (bg - ag)

                def ev(x):
                    dd = x - v
                    nb = base + dd * (2.0 * a)
                    nb = nb.copy()
                    nb[i] += dd * dd
                    m = max(float(nb.max()), max_out)
                    de = S + dd
                    if de <= 1e-15:
                        return float('inf')
                    return 2.0 * n * m / (de * de)

                fc, fd = ev(cg), ev(dg)
                for _ in range(25):
                    if fc < fd:
                        bg, dg, fd = dg, cg, fc
                        cg = bg - gr * (bg - ag)
                        fc = ev(cg)
                    else:
                        ag, cg, fc = cg, dg, fd
                        dg = ag + gr * (bg - ag)
                        fd = ev(dg)
                x_opt = (ag + bg) * 0.5
                f_opt = ev(x_opt)
                if f_opt < best_val:
                    best_x, best_val = x_opt, f_opt

            if math.isfinite(best_val) and best_val < cur_score * (1.0 - 1e-13) and best_x >= 0.0:
                dd = best_x - v
                if abs(dd) > 1e-18:
                    b[i:i + n] += dd * (2.0 * a)[:n]
                    b[2 * i] += dd * dd
                    a[i] = best_x
                    S += dd
                    cur_score = best_val
                    improved = True

        if sweep % 3 == 0:
            b = conv_full(a)
            S = float(a.sum())
            cur_score = 2.0 * n * float(b.max()) / (S * S)

        new_sc = 2.0 * n * float(b.max()) / (S * S)
        if math.isfinite(new_sc) and new_sc < best_score - 1e-14:
            best_score = new_sc
            best_a = a.copy()
            consider(best_a)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 10:
                break

    return best_a
```
