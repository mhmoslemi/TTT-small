```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.asarray(c, dtype=np.float64)


def _score_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float('inf')
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return float('inf')
    return float(2.0 * n * m / (s * s))


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.shape, 1.0 / u.size, dtype=np.float64)
    return e / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 30.0  # leave 30s headroom
    
    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float('inf')
    
    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()
    
    consider(parent)
    
    n = int(best_arr.size)
    # Work with normalized sequence (sum = 1)
    p = best_arr.copy()
    p = p / float(p.sum())
    
    # Precompute FFT sizes
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)  # for cross-correlation of length n with window of length 2n-1
    cross_len = 3 * n - 2
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return c
    
    def xfast(a, w):
        # q[i] = sum_k w[k] * a[(k-i) mod?] -- need to be careful with indexing
        # Actually: q[i] = sum_k w[k] * a[k-i] where a[j]=0 for j outside [0,n-1]
        # This is a cross-correlation. Using FFT:
        # Flip a and do convolution with w
        F = rfft(a[::-1], cross_size)
        G = rfft(w, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        return q
    
    # ============ LSE Optimization ============
    def lse_phase(p_start, end_time):
        """Run LSE optimization from p_start until end_time."""
        p = np.maximum(p_start, 0.0).copy()
        s = float(p.sum())
        if s < 1e-300:
            return p
        p = p / s
        u = np.log(np.maximum(p, 1e-30))
        u -= float(u.mean())
        
        c = cfast(p)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return p
        
        # Temperature schedule: geometric from 0.1*cmax to 1e-4*cmax
        n_stages = 8
        T_scales = np.array([0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004])
        
        def lse_func(uu, T):
            pp = _softmax(uu)
            c = cfast(pp)
            cmax_val = float(c.max())
            if not math.isfinite(cmax_val) or cmax_val <= 0:
                return 1e100, np.zeros_like(uu)
            scaled = (c - cmax_val) / T
            # Clip for numerical stability
            scaled = np.clip(scaled, -700, 700)
            exps = np.exp(scaled, dtype=np.float64)
            sum_exp = float(exps.sum())
            if not math.isfinite(sum_exp) or sum_exp <= 0:
                w = np.zeros(c.size)
                w[int(np.argmax(c))] = 1.0
                f = cmax_val
            else:
                f = cmax_val + T * float(np.log(sum_exp / c.size))
                w = exps / sum_exp
            q = xfast(pp, w)
            gdp = 2.0 * q
            tdp = float(np.dot(pp, gdp))
            grad = pp * (gdp - tdp)
            grad = np.where(np.isfinite(grad), grad, 0.0)
            return f, grad
        
        best_u = u.copy()
        best_f = 1e100
        for stage_idx, T_scale in enumerate(T_scales):
            now = time.monotonic()
            if now > end_time - 5.0:
                break
            T = T_scale * cmax
            # Run L-BFGS-B for this temperature
            max_calls = 5000
            opts = {"maxiter": max_calls, "gtol": 1e-10, "ftol": 1e-14, "maxls": 20, "maxdisp": 0}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                continue
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_f:
                    best_f = fval
                    best_u = res.x.copy()
        
        p_out = _softmax(best_u)
        p_out = np.maximum(p_out, 0.0)
        ss = float(p_out.sum())
        if ss > 0:
            p_out /= ss
        return p_out
    
    # ============ Coordinate Descent ============
    def coord_descent(p_start, end_time, seed_cd=12345):
        n_cd = int(p_start.size)
        p = np.maximum(np.asarray(p_start, dtype=np.float64).ravel(), 0.0).copy()
        s = float(p.sum())
        if s < 1e-300:
            return p, float('inf')
        p = p / s
        c = cfast(p)
        # Score with sum=1: 2*n*max(c)
        cur_score = float(2.0 * n_cd * c.max())
        best_score_cd = cur_score
        best_p = p.copy()
        rng = np.random.default_rng(seed_cd)
        Ltot = 2 * n_cd - 1
        G = 32  # grid points (reduced from 48 for speed)
        GR = (math.sqrt(5.0) - 1.0) / 2.0
        flat = 0
        sweep = 0
        
        while sweep < 20000 and time.monotonic() < end_time - 2.0:
            improved = False
            order = rng.permutation(n_cd)
            for idx_i in order:
                i = int(idx_i)
                if time.monotonic() > end_time - 0.5:
                    break
                v = float(p[i])
                start = i
                end = min(i + n_cd, Ltot)
                Ls = end - start
                if Ls <= 0:
                    continue
                base = c[start:end].copy()
                infl = 2.0 * p[:Ls]
                diag = i  # position of 2i in [start, end)
                
                # Max of conv outside the affected range
                max_const = 0.0
                if start > 0:
                    mc = float(c[:start].max())
                    if mc > max_const:
                        max_const = mc
                if end < Ltot:
                    mc = float(c[end:].max())
                    if mc > max_const:
                        max_const = mc
                
                # 1D search: x is the new value of p[i]
                # Score(x) = 2*n*max(base + (x-v)*infl + (x-v)^2*[diag], max_const) / (1 + (x-v))^2
                # (since sum was 1)
                S = 1.0
                x_hi = max(10.0 * v, 5.0 / n_cd, 0.001)
                x_hi = min(x_hi, 0.1)
                x_hi = max(x_hi, 1e-8)
                
                # Grid search
                pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
                ws = pts - v
                s_new = S + ws
                aff = base[None, :] + ws[:, None] * infl[None, :]
                if 0 <= diag < Ls:
                    aff[:, diag] += ws * ws
                maff = np.max(aff, axis=1)
                maff = np.maximum(maff, max_const)
                valid = s_new > 1e-12
                scores = np.full(pts.shape, float('inf'), dtype=np.float64)
                if np.any(valid):
                    scores[valid] = 2.0 * n_cd * maff[valid] / (s_new[valid] ** 2)
                
                if not np.any(np.isfinite(scores)):
                    continue
                imin = int(np.argmin(scores))
                best_x = float(pts[imin])
                best_val = float(scores[imin])
                
                # Golden section refinement (only if not at boundary)
                if 0 < imin < G:
                    lo = float(pts[max(0, imin - 1)])
                    hi = float(pts[min(G, imin + 1)])
                    if hi - lo > 1e-14:
                        a_gs = lo
                        b_gs = hi
                        c_gs = b_gs - GR * (b_gs - a_gs)
                        d_gs = a_gs + GR * (b_gs - a_gs)
                        
                        def f1d(x_val):
                            w = x_val - v
                            sn = S + w
                            if sn <= 1e-15:
                                return float('inf')
                            aff_m = float(np.max(base + w * infl))
                            if 0 <= diag < Ls:
                                dv = base[diag] + w * infl[diag] + w * w
                                if dv > aff_m:
                                    aff_m = dv
                            m = max(aff_m, max_const)
                            return 2.0 * n_cd * m / (sn * sn)
                        
                        f_c = f1d(c_gs)
                        f_d = f1d(d_gs)
                        for _ in range(18):
                            if b_gs - a_gs < 1e-13:
                                break
                            if f_c < f_d:
                                b_gs = d_gs
                                d_gs = c_gs
                                f_d = f_c
                                c_gs = b_gs - GR * (b_gs - a_gs)
                                f_c = f1d(c_gs)
                            else:
                                a_gs = c_gs
                                c_gs = d_gs
                                f_c = f_d
                                d_gs = a_gs + GR * (b_gs - a_gs)
                                f_d = f1d(d_gs)
                        x_mid = (a_gs + b_gs) / 2.0
                        f_mid = f1d(x_mid)
                        if f_mid < best_val:
                            best_x = x_mid
                            best_val = f_mid
                
                if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x:
                    w = best_x - v
                    if abs(w) > 1e-15:
                        p[i] = best_x
                        aff_seg = base + w * infl
                        if 0 <= diag < Ls:
                            aff_seg = aff_seg.copy()
                            aff_seg[diag] += w * w
                        new_max = max(float(aff_seg.max()), max_const)
                        c[start:end] = aff_seg
                        cur_score = 2.0 * n_cd * new_max  # approx (sum changed)
                        improved = True
            
            # Renormalize
            s_new = float(p.sum())
            if s_new > 1e-300 and math.isfinite(s_new):
                inv = 1.0 / s_new
                p *= inv
                c = cfast(p)
                new_score = float(2.0 * n_cd * c.max())
                if math.isfinite(new_score) and new_score < best_score_cd - 1e-15:
                    best_score_cd = new_score
                    best_p = p.copy()
            
            if improved:
                flat = 0
            else:
                flat += 1
                if flat >= 3:
                    if time.monotonic() < end_time - 10.0:
                        # Kick: add random noise
                        kick = 0.01 + 0.03 * min(1.0, flat / 10.0)
                        p = p * np.exp(rng.normal(0.0, kick, n_cd))
                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss > 0:
                            p /= ss
                        c = cfast(p)
                        cur_score = float(2.0 * n_cd * c.max())
                        flat = 0
                    else:
                        break
            sweep += 1
        
        return best_p, best_score_cd
    
    # ============ Main Search Loop ============
    # Phase 1: LSE on parent (30% of budget)
    lse_end = t0 + 0.3 * (budget - 30.0)
    if lse_end > t0 + 15.0:
        try:
            p_lse = lse_phase(p, lse_end)
            if p_lse is not None and p_lse.size == n and np.all(np.isfinite(p_lse)) and p_lse.sum() > 0:
                p_lse = p_lse / float(p_lse.sum())
                consider(p_lse)
        except Exception:
            pass
    
    # Phase 2: CD on best found (40% of budget)
    cd_end = t0 + 0.7 * (budget - 30.0)
    if cd_end > time.monotonic() + 20.0:
        p_cd_input = best_arr / float(best_arr.sum())
        try:
            p_cd, sc_cd = coord_descent(p_cd_input, cd_end, seed_cd=int(seed + 1111))
            if p_cd is not None and p_cd.size > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                consider(p_cd)
        except Exception:
            pass
    
    # Phase 3: Another LSE from current best (15% of budget)
    lse2_end = t0 + 0.85 * (budget - 30.0)
    if lse2_end > time.monotonic() + 15.0:
        p_start2 = best_arr / float(best_arr.sum())
        # Perturb slightly
        rng3 = np.random.default_rng(int(seed + 2222))
        p_start2 = p_start2 * np.exp(rng3.normal(0, 0.02, n))
        p_start2 = np.maximum(p_start2, 0)
        p_start2 /= float(p_start2.sum())
        try:
            p_lse2 = lse_phase(p_start2, lse2_end)
            if p_lse2 is not None and p_lse2.size == n and np.all(np.isfinite(p_lse2)) and p_lse2.sum() > 0:
                p_lse2 /= float(p_lse2.sum())
                consider(p_lse2)
        except Exception:
            pass
    
    # Phase 4: Final CD polish (remaining time minus safety margin)
    final_end = hard - 2.0
    if final_end > time.monotonic() + 15.0:
        p_final_input = best_arr / float(best_arr.sum())
        try:
            p_f, _ = coord_descent(p_final_input, final_end, seed_cd=int(seed + 3333))
            if p_f is not None and p_f.size > 0 and np.all(np.isfinite(p_f)) and p_f.sum() > 0:
                consider(p_f)
        except Exception:
            pass
    
    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0
