def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)

def _conv_matrix(n, a):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        A[k, jlo:jhi] = a[k - jhi + 1:k - jlo + 1:-1]
    return A


def _lp_best_response(a, M, n, time_limit=1.5):
    A = _conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * (1 + 1e-9))
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0 and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None
