            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, avg)

            G = 36
            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag = i - start_i

            w_arr = pts - v
            denom = S_val + w_arr
            base_row = base[None, :]
            infl_row = infl[None, :]
            aff = base_row + w_arr[:, None] * infl_row
            dval = base[diag] + w_arr * infl[diag] + w_arr * w_arr
            aff = np.maximum(aff, dval[:, None])
            aff = np.maximum(aff, max_const)
            vals = np.full(pts.shape, math.inf, dtype=np.float64)
            valid = denom > 1e-15
            dv = denom[valid]
            mv = np.max(aff[valid], axis=1)
            vals[valid] = (2.0 * n * mv) / (dv * dv)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Fine refinement around best_x
            dw = x_hi / G
            lo = max(0.0, best_x - 3.0 * dw)
            hi = min(x_hi, best_x + 3.0 * dw)
            if hi <= lo:
                hi = lo + 1e-12
            fine = np.linspace(lo, hi, 50, dtype=np.float64)
            if 0.0 <= best_x <= x_hi:
                fine = np.concatenate((fine, [best_x]))
                fine = np.unique(fine)
            w2 = fine - v
            denom2 = S_val + w2
            aff2 = base_row + w2[:, None] * infl_row
            dval2 = base[diag] + w2 * infl[diag] + w2 * w2
            aff2 = np.maximum(aff2, dval2[:, None])
            aff2 = np.maximum(aff2, max_const)
            vals2 = np.full(fine.shape, math.inf, dtype=np.float64)
            valid2 = denom2 > 1e-15
            dv2 = denom2[valid2]
            mv2 = np.max(aff2[valid2], axis=1)
            vals2[valid2] = (2.0 * n * mv2) / (dv2 * dv2)
            if np.any(np.isfinite(vals2)):
                imin2 = int(np.argmin(vals2))
                if float(vals2[imin2]) < best_val:
                    best_x = float(fine[imin2])
                    best_val = float(vals2[imin2])

            if math.isfinite(best_val) and best_val < cur_score - 1e-14:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start_i:end_i] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        # End of sweep: renormalize
        S_val = float(S_total)
        if S_val > 1e-300 and math.isfinite(S_val):
            inv = 1.0 / S_val
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        np.maximum(c, 0.0, out=c)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
            consider(p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                # Restart with perturbation
                if time.monotonic() < end_time - 10.0:
                    # Fourier-based perturbation
                    size = next_fast_len(n)
                    F = rfft(p, size)
                    phase = rng.uniform(0, 2 * math.pi, F.size)
                    F_pert = F * np.exp(1j * phase * 0.08)
                    p_new = irfft(F_pert, size)[:n]
                    p_new = np.maximum(p_new, 0.0)
                    ss = float(p_new.sum())
                    if ss > 0:
                        p_new /= ss
                        c = _conv_self(p_new)
                        S_total = 1.0
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
                            best_score = cur_score
                            best_p = p_new.copy()
                            consider(p_new)
                        p = p_new
                    flat = 0
                else:
                    break
        sweep += 1

    p_out = np.maximum(p, 0.0)
    ss = float(p_out.sum())
    if ss > 1e-300:
        p_out /= ss
    sc = _score_fast(p_out)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p_out.copy()
        consider(best_p)
    return best_p


def _lp_blend(p, end_time, consider, n):
    """LP-based direction finding and blending."""
    n = int(np.asarray(p).size)
    if n < 32:
        return
    c = _conv_self(p)
    M = float(c.max())
    if M <= 0 or not math.isfinite(M):
        return
    try:
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = p[j]
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        constraints = [A @ g <= b]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.CLARABEL, verbose=False)
        if g.value is not None:
            g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
            gv = float(g_val.sum())
            if gv > 1e-10:
                g_val /= gv
                work = np.asarray(p, dtype=np.float64)
                work = work / float(work.sum())
                for alpha in [0.3, 0.15, 0.07, 0.03]:
                    blended = (1.0 - alpha) * work + alpha * g_val
                    blended = np.maximum(blended, 0.0)
                    bs = float(blended.sum())
                    if bs > 0:
                        blended /= bs
                        consider(blended)
    except Exception:
        pass


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 30.0

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = _score_fast(best_arr)
    if not math.isfinite(best_score):
        best_arr = np.ones(256, dtype=np.float64) / 256.0
        best_score = _score_fast(best_arr)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            sc = _score_fast(a)
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_arr = a.copy()
                best_score = sc
        except Exception:
            pass

    n = int(best_arr.size)
    work = best_arr.copy()
    work = np.maximum(work, 0.0)
    work /= float(work.sum())

    # Phase 1: LSE optimization
    p1_end = min(hard - 700.0, t0 + 0.22 * budget)
    if p1_end > t0 + 25.0:
        try:
            p1 = _do_lse(work, p1_end, consider, n, fine_only=False)
            if p1 is not None and p1.size == n:
                consider(p1)
        except Exception:
            pass

    work = best_arr.copy()
    work = np.maximum(work, 0.0)
    work /= float(work.sum())

    # Phase 2: Main coordinate descent
    p2_end = min(hard - 300.0, t0 + 0.60 * budget)
    if p2_end > time.monotonic() + 25.0:
        try:
            _coord_descent(work, p2_end, consider, n, int(seed) + 111)
        except Exception:
            pass

    work = best_arr.copy()
    work = np.maximum(work, 0.0)
    work /= float(work.sum())

    # Phase 3: LP blend
    p3_end = min(hard - 200.0, time.monotonic() + 80.0)
    if p3_end > time.monotonic() + 15.0:
        try:
            _lp_blend(work, p3_end, consider, n)
        except Exception:
            pass

    work = best_arr.copy()
    work = np.maximum(work, 0.0)
    work /= float(work.sum())

    # Phase 4: More coordinate descent
    p4_end = min(hard - 120.0, t0 + 0.80 * budget)
    if p4_end > time.monotonic() + 25.0:
        try:
            _coord_descent(work, p4_end, consider, n, int(seed) + 222)
        except Exception:
            pass

    work = best_arr.copy()
    work = np.maximum(work, 0.0)
    work /= float(work.sum())

    # Phase 5: Fine LSE
    p5_end = min(hard - 60.0, t0 + budget - 60.0)
    if p5_end > time.monotonic() + 30.0:
        lse_e = min(p5_end - 40.0, time.monotonic() + 0.4 * (p5_end - time.monotonic()))
        if lse_e > time.monotonic() + 15.0:
            try:
                _do_lse(work, lse_e, consider, n, fine_only=True)
            except Exception:
                pass
        work = best_arr.copy()
        work = np.maximum(work, 0.0)
        work /= float(work.sum())

    # Phase 6: Final coordinate polish
    p6_end = min(hard - 20.0, t0 + budget - 35.0)
    if p6_end > time.monotonic() + 15.0:
        try:
            _coord_descent(work, p6_end, consider, n, int(seed) + 333)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.maximum(parent, 0.0).astype(np.float64)
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        fs = float(evaluate_sequence(ls))
        if not math.isfinite(fs):
            ls = [float(x) for x in np.clip(parent, 0, 1000).ravel()]
            if not ls or not all(math.isfinite(x) for x in ls):
                ls = [1.0]
    except Exception:
        ls = [1.0]
    return ls
