import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize, linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:
        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(3072, dtype=np.float64)
    return p


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _score_norm(a):
    """Assume a is non-negative, sum-normalized to 1."""
    n = a.size
    b = np.convolve(a, a)
    return float(2.0 * n * b.max())


def _coord_descent(p, stop_time, seed=42, consider=None, G=28, max_sweeps=999999):
    n = p.size
    p = np.maximum(p, 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    b = np.convolve(p, p)
    b = np.maximum(b, 0.0)
    Lt = 2 * n - 1
    S = 1.0
    cur_score = float(2.0 * n * b.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat_count = 0
    t0 = time.monotonic()

    while sweep < max_sweeps:
        if time.monotonic() > stop_time - 3.0:
            break

        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break

            # Rescale if needed
            if S > 1.25 or S < 0.75:
                inv = 1.0 / S
                p *= inv
                b *= inv * inv
                S = 1.0

            v = p[i]
            ks = i
            ke = min(i + n, Lt)
            Ls = ke - ks
            if Ls <= 0:
                continue

            base = b[ks:ke].copy()
            # influence: b_k += 2w * a_{k-i} for k in [i, i+n-1]
            # a_{k-i} for k in [i, i+n-1] is a[0] to a[min(n-1, Ls-1)]
            infl = 2.0 * p[:Ls].copy()
            # self-term: b_{2i} += w^2, position in base is 2i - i = i
            self_pos = 2 * i - i  # = i

            # constant max outside affected region
            const_max = 0.0
            if ks > 0:
                const_max = max(const_max, float(b[:ks].max()))
            if ke < Lt:
                const_max = max(const_max, float(b[ke:].max()))

            # Search range for w (new value = v + w)
            x_lo = 0.0
            x_hi = max(50.0 * v, 30.0 * S / n, 0.02, 1e-12)
            x_hi = min(x_hi, 5.0 * S)
            if not math.isfinite(x_hi) or x_hi <= x_lo:
                x_hi = max(1e-8, S / n) * 2

            # Evaluate on grid
            npts = G
            pts = np.linspace(0.0, x_hi, npts + 1)
            if 0.0 <= v <= x_hi:
                pts = np.unique(np.concatenate((pts, [v])))

            # Vectorized evaluation
            W = pts - v  # offset
            # Affected: base + W*infl + W^2 * (self at position self_pos)
            # Shape: (npts, Ls)
            aff = base[None, :] + W[:, None] * infl[None, :]
            if 0 <= self_pos < Ls:
                aff[:, self_pos] += W * W
            row_max = aff.max(axis=1)
            total_max = np.maximum(row_max, const_max)
            denom = (S + W)
            valid = denom > 1e-15
            vals = np.full(pts.shape, 2.0 * n)
            if np.any(valid):
                vals[valid] = 2.0 * n * total_max[valid] / (denom[valid] * denom[valid])

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Refine with golden section
            dw = x_hi / float(npts)
            g_lo = max(0.0, best_x - 1.5 * dw)
            g_hi = min(x_hi, best_x + 1.5 * dw)
            if g_hi - g_lo > 1e-18:
                gr = (math.sqrt(5.0) - 1.0) / 2.0

                def _eval_single(x):
                    w = x - v
                    aff_s = base + w * infl
                    if 0 <= self_pos < Ls:
                        aff_s[self_pos] += w * w
                    rm = float(aff_s.max())
                    tm = max(rm, const_max)
                    d = S + w
                    if d <= 1e-15:
                        return 2.0 * n
                    return 2.0 * n * tm / (d * d)

                a_gs = g_lo + (1.0 - gr) * (g_hi - g_lo)
                b_gs = g_lo + gr * (g_hi - g_lo)
                fa = _eval_single(a_gs)
                fb = _eval_single(b_gs)
                for _ in range(20):
                    if not math.isfinite(fa):
                        fa = 1e100
                    if not math.isfinite(fb):
                        fb = 1e100
                    if fa < fb:
                        g_hi = b_gs
                        b_gs = a_gs
                        fb = fa
                        a_gs = g_lo + (1.0 - gr) * (g_hi - g_lo)
                        fa = _eval_single(a_gs)
                    else:
                        g_lo = a_gs
                        a_gs = b_gs
                        fa = fb
                        b_gs = g_lo + gr * (g_hi - g_lo)
                        fb = _eval_single(b_gs)
                    if g_hi - g_lo < 1e-16:
                        break
                gs_x = (g_lo + g_hi) / 2.0
                gv = _eval_single(gs_x)
                if math.isfinite(gv) and gv < best_val:
                    best_x = gs_x
                    best_val = gv

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S += w
                    b[ks:ke] += w * infl
                    di = 2 * i
                    if 0 <= di < Lt:
                        b[di] += w * w
                    cur_score = best_val
                    improved = True

        # Periodic exact recalculation and rescale
        S = float(S)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            b *= inv * inv
            S = 1.0
        else:
            break

        # Recompute b exactly every few sweeps to avoid drift
        if sweep % 10 == 9:
            b = np.convolve(p, p)
            b = np.maximum(b, 0.0)
            cur_score = float(2.0 * n * b.max())

        if math.isfinite(cur_score) and cur_score < best_score - 1e-16:
            best_score = cur_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat_count = 0
        else:
            flat_count += 1
            if flat_count >= 5:
                # Perturbation to escape local minimum
                if time.monotonic() < stop_time - 10.0:
                    noise = 0.015 + 0.01 * math.exp(-0.05 * flat_count)
                    rng2 = np.random.default_rng(seed + sweep)
                    p = p * np.exp(rng2.normal(0.0, noise, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 1e-15:
                        p /= ss
                        S = 1.0
                    else:
                        p = best_p.copy()
                        S = 1.0
                    b = np.convolve(p, p)
                    b = np.maximum(b, 0.0)
                    cur_score = float(2.0 * n * b.max())
                    flat_count = 0
                else:
                    break

        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    return p


def _lp_direction(p, M, stop_time):
    """Solve LP: max sum(g) s.t. (p*g)_k <= M, g >= 0. Returns g or None."""
    n = p.size
    if time.monotonic() > stop_time - 5.0:
        return None
    # Build constraint matrix A: (2n-1) x n, A[k,j] = p_{k-j} if valid
    # To save memory, build row by row
    c = np.ones(n)  # objective: maximize c^T g
    A_ub = np.zeros((2 * n - 1, n), dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = p[k - j]
    # g >= 0
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
        if res.status == 0:
            g = res.x
            g = np.maximum(g, 0.0)
            gs = float(g.sum())
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def _lse_polish(p, stop_time, consider=None):
    """Short L-BFGS polish with log-sum-exp smoothing."""
    n = p.size
    if stop_time - time.monotonic() < 30.0:
        return p
    p = np.maximum(p, 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)

    def _softmax(uu):
        m = float(np.max(uu))
        e = np.exp(uu - m)
        ss = float(e.sum())
        if not math.isfinite(ss) or ss <= 0:
            return np.full(uu.size, 1.0 / uu.size)
        return e / ss

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -400.0, 400.0)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    c = cfast(p)
    cmax = float(c.max())
    T = max(1e-300, 0.001 * cmax)

    best_u = u.copy()
    best_f = float(lse_func(u, T)[0])
    stale = 0

    while time.monotonic() < stop_time - 5.0:
        try:
            res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                           jac=True, options={"maxiter": 40, "gtol": 1e-14,
                                              "ftol": 1e-16, "maxls": 25,
                                              "maxcor": 25, "disp": False})
        except Exception:
            break
        if res.x is not None:
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if x.size == n and np.all(np.isfinite(x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_f - 1e-16:
                    best_f = fval
                    best_u = x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
        if getattr(res, "status", 1) == 0:
            break

    pp = _softmax(best_u)
    pp = np.maximum(pp, 0.0)
    ss = float(pp.sum())
    if ss > 1e-15:
        pp /= ss
    if consider is not None:
        try:
            consider(pp)
        except Exception:
            pass
    return pp

            return

        except Exception:
            return

    # Phase 1: Intensive coordinate descent from parent
    p = parent.copy()
    ps = float(p.sum())

    consider(p)

    try:

    if p is not None and p.size == n:
        consider(p)

    # Phase 2: LP direction + coordinate descent restart
    rem_to_lse = t0 + 0.88 * (budget - 15.0)
    if time.monotonic() < rem_to_lse - 60.0:
        # Try LP direction
        try:

                    p_cand = (1 - t_mix) * pn + t_mix * g
                    p_cand = np.maximum(p_cand, 0.0)
                    sc = float(p_cand.sum())

                        p_cand /= sc
                        consider(p_cand)

        try:

        except Exception:

        except Exception:
            pass

    except Exception:
