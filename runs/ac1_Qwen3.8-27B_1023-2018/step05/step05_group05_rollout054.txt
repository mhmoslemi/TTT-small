import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(np.asarray(c, dtype=np.float64), 0.0)


def _score_fft(a):
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0 / u.size)
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _lse_optimize(p, stop_time, consider=None):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p
    p = p / s

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return np.maximum(c, 0.0)

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        return np.maximum(q, 0.0)

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = 1e-4 * cmax
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        return float(f), np.where(np.isfinite(grad), grad, 0.0)

    def p_to_u(pp):
        ll = np.log(np.maximum(pp, 1e-300))
        return ll - float(np.mean(ll))

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score_fft(p)

    stages = (
        (0.05, 0.10),
        (0.015, 0.15),
        (0.005, 0.20),
        (0.002, 0.25),
        (0.0007, 0.30),
    )

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        end_stage = min(stop_time - 4.0, now + max(5.0, wt * 25.0))
        cmax = float(cfast(p).max())
        T = max(1e-15, frac * cmax)
        stale = 0
        while time.monotonic() < end_stage - 1.0:
            maxit = 40
            opts = {"maxiter": maxit, "gtol": 1e-12, "ftol": 1e-16,
                    "maxls": 30, "maxcor": 25, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval):
                    p_cur = _softmax(res.x)
                    sc_cur = _score_fft(p_cur)
                    if math.isfinite(sc_cur) and sc_cur < best_sc:
                        best_sc = sc_cur
                        best_p = p_cur.copy()
                        p = p_cur.copy()
                        u = p_to_u(p)
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 4:
                            break
            if getattr(res, "status", 1) == 0:
                break

    p_out = best_p.copy()
    ss = float(p_out.sum())
    if ss > 1e-300:
        p_out = p_out / ss
    return p_out


def _coord_descent(p, stop_time, consider=None, seed=42, max_sweeps=500000):

        if sweep % 20 == 0:

        if math.isfinite(new_sc) and new_sc < best_score - 1e-15:

                    ss = float(p.sum())
                    if ss > 0:

                    c = _conv_self(p)
                    S = 1.0
                    cur_score = 2.0 * n * float(c.max())

                else:

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_self(p)
    sc = 2.0 * n * float(c.max())

def _subgradient_steps(p, stop_time, nsteps=200, consider=None):

            break
        peak_idx = int(np.argmax(c))

            if 0 <= j < n:

                    consider(best_p)
                except Exception:
                    pass
        else:

            for trial in range(3):
                eta /= 3.0

def _pair_coord_descent(p, stop_time, consider=None, seed=777, max_pairs=50000):

    for pair_iter in range(max_pairs):
        if time.monotonic() > stop_time - 2.0:

        j = int(rng.integers(0, n))

        vj = float(p[j])
        total_v = vi + vj
        if total_v < 1e-15:
            continue

        # w_i = x - vi, w_j = (total_v - x) - vj = -(x - vi) = -w_i
        # So the change is: p[i] += w, p[j] -= w

        # This is getting complex. Let's just do a simple 1D search over x.

                p5 = p5 / float(p5.sum())
                consider(p5)
        except Exception:
            pass

    # Final coordinate descent polish
    rem = hard - 15.0 - time.monotonic()
    if rem > 15.0:
        p = best_arr.copy()
        ps = float(p.sum())
        if ps > 1e-12:
            p /= ps
        try:
            p6, _ = _coord_descent(p, hard - 15.0, consider=consider, seed=int(seed + 99999), max_sweeps=100000)
            if p6 is not None and p6.size == n_best and np.all(np.isfinite(p6)) and p6.sum() > 0:
                p6 = p6 / float(p6.sum())
                consider(p6)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
