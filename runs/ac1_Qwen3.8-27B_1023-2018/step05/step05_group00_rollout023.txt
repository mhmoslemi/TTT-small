            hi = max(v * 3, 0.04 * S / n, 1e-10)
            hi = min(hi, 10.0 * S / n)
            G = 12
            deltas = np.linspace(-v, hi, G+1)
            
            best_d = 0.0
            best_v = score  # current score
            
            for d in deltas:
                if d == 0: continue
                Sd = S + d
                if Sd < 0.01: continue
                new_seg = c_seg + d * infl
                new_seg[diag_pos] += d * d
                seg_max = float(new_seg.max())
                tot_max = max(seg_max, max_oth)
                sc = 2.0 * n * tot_max / (Sd * Sd)
                if sc < best_v:
                    best_v = sc
                    best_d = d
            
            # Fine search around best_d
            if best_d != 0 and G > 4:
                span = hi - (-v)
                lo = best_d - span/G
                hi2 = best_d + span/G
                fine = np.linspace(max(lo, -v*1.001), hi2, 20)
                for d in fine:
                    Sd = S + d
                    if Sd < 0.01: continue
                    new_seg = c_seg + d * infl
                    new_seg[diag_pos] += d * d
                    seg_max = float(new_seg.max())
                    tot_max = max(seg_max, max_oth)
                    sc = 2.0 * n * tot_max / (Sd * Sd)
                    if sc < best_v:
                        best_v = sc
