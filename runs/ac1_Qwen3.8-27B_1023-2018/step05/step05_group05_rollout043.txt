import time
import math
from scipy.fft import rfft, irfft, next_fast_len

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = None
    if p is None or p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p.copy()

def _fft_convolve(a):
    """Full self-convolution via FFT."""
    n = a.size
    L = 2*n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(c, 0.0)  # small negative values from FFT roundoff

def _score_norm(a):
    """Score assuming sum(a) = 1: 2n * max(conv)."""
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())

def frank_wolfe(a, stop_time, check_fn=None, seed=42):
    """Frank-Wolfe method for min max_k c[k] s.t. sum a = 1, a >= 0."""
    n = a.size
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-300:
        return a
    a /= s
    
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_val = _score_norm(a)

            if i_lo > i_hi:

New contribution with $x$:
At $k = 2i$: $x^2$
At $k \neq 2i$, $0 \leq k-i \leq n-1$: $2x a[k-i]$
Otherwise: 0.

So:
$c''[k] = c[k] - 2a_i a[k-i] + 2x a[k-i] + \mathbb{1}[k=2i](x^2 - a_i^2)$ for $k \in [i, i+n-1]$
$c''[k] = c[k]$ otherwise

And $c'[k] = c''[k] / (s+x)^2$ where $s = 1 - a_i$.

The objective is $\max_k c'[k] = \frac{1}{(s+x)^2} \max_k c''[k]$.

Let $M(x) = \max_k c''[k]$. We want to minimize $M(x)/(s+x)^2$ over $x \geq 0$.

$M(x) = \max(\max_{k \in [i, i+n-1]} (c[k] + 2(x-a_i)a[k-i] + \mathbb{1}[k=2i](x^2 - a_i^2)), \max_{k \notin [i,i+n-1]} c[k])$

Let $C = \max_{k \notin [i,i+n-1]} c[k]$ (the "constant floor" - not affected by the change).
Let $W(x) = \max_{k \in [i,i+n-1]} (c[k] + 2(x-a_i)a[k-i] + \mathbb{1}[k=2i](x^2 - a_i^2))$

Then $M(x) = \max(W(x), C)$ and objective is $\max(W(x), C)/(s+x)^2$.
