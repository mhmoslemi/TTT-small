def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _fft_cross_rev(p, w):
    """Cross-correlation: (rev(p) * w)[n-1 : 2n-1] gives (p * w)[0..n-1] shifted."""
    p = np.asarray(p, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = p.size
    L = 3 * n - 2
    size = next_fast_len(L)
    F = rfft(p[::-1], size)
    G = rfft(w, size)
    c = irfft(F * G, size)[:L]
    q = c[n - 1:2 * n - 1]
    q = np.asarray(q, dtype=np.float64)
    np.maximum(q, 0.0, out=q)
    return q


def _score(a):

def _lbfgs_smooth(p, stop_time, consider=None):
    """LBFGS on log-space with log-sum-exp smoothing of max-conv."""
    p = _normalize(p)
    n = int(p.size)
    if n <= 1 or time.monotonic() > stop_time:
        return p

    conv_len = 2 * n - 1
    cross_len = 3 * n - 2
    conv_size = next_fast_len(conv_len)
    cross_size = next_fast_len(cross_len)

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def p_to_u(pp):
        pp = np.maximum(pp, 1e-15)
        uu = np.log(pp, dtype=np.float64)
        uu -= float(np.mean(uu))
        return uu

    def lse_func(uu, T):
        uu = np.asarray(uu, dtype=np.float64).ravel()
        if uu.size != n:
            return 1e100, np.zeros(n, dtype=np.float64)
        uu = np.where(np.isfinite(uu), uu, 0.0)
        m = float(np.max(uu))
        e = np.exp(uu - m, dtype=np.float64)
        s = float(e.sum())
        if not math.isfinite(s) or s <= 0.0:
            pp = np.full(n, 1.0/n, dtype=np.float64)
        else:
            pp = e / s
        ss = float(pp.sum())
        if not math.isfinite(ss) or ss <= 0.0:
            pp = np.full(n, 1.0/n, dtype=np.float64)
        else:
            pp /= ss

        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0.0:
            return 1e100, np.zeros(n, dtype=np.float64)

        if not math.isfinite(T) or T <= 0.0:
            T = 1e-4 * cmax
        T = max(1e-14, T)

        scaled = (c - cmax) / T
        # Clip to avoid overflow
        scaled = np.clip(scaled, -50.0, 50.0)
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(conv_len, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            w = exps / sum_exp
            ratio = sum_exp / float(c.size)
            f = cmax + T * float(math.log(max(ratio, 1e-300)))

        q = xfast(pp, w)
        gid = 2.0 * q
        gd = float(np.dot(pp, gid))
        grad = pp * (gid - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score(p)

    stages = (
        (0.012, 0.16),
        (0.004, 0.18),
        (0.0012, 0.20),
        (0.0003, 0.22),
        (0.00008, 0.24),
    )

    start = time.monotonic()
    avail = float(stop_time - start - 4.0)
    if avail <= 5.0:
        return best_p

    total_w = sum(w for _, w in stages)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        cum += wt
        end_stage = min(stop_time - 4.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:
            continue

        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-14, frac * cmax)

        # Run LBFGS in this stage
        best_f = lse_func(u, T)[0]
        if not math.isfinite(best_f):
            continue

        max_iters = 8 if n > 2000 else 12
        stale_count = 0
        while time.monotonic() < end_stage - 1.5 and stale_count < 3:
            opts = {"maxiter": max_iters, "gtol": 1e-10, "ftol": 1e-15,
                    "maxls": 12, "maxcor": 12, "disp": False, "maxdisp": 0}
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    fval = float(res.fun)
                    if math.isfinite(fval) and fval < best_f - 1e-14:
                        best_f = fval
                        u = x.copy()
                        stale_count = 0
                    else:
                        stale_count += 1
                else:
                    stale_count += 1
            if getattr(res, "status", 1) == 0:
                break

        p_new = np.exp(u - float(np.max(u)))
        s_new = float(p_new.sum())
        if math.isfinite(s_new) and s_new > 0:
            p_new /= s_new
        else:
            p_new = best_p.copy()

        sc_new = _score(p_new)
        if math.isfinite(sc_new) and sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_p = p_new.copy()
            if consider:
                try:
                    consider(best_p)
                except Exception:
                    pass
        p = best_p.copy()
        u = p_to_u(p)

    return best_p

        return p, math.inf

    best_p = p.copy()

        try:
            consider(best_p)
        except Exception:
            pass

    sweep = 0
    flat = 0

        improved = False
        order = rng.permutation(n)

        # Identify top-k hot spots for priority
        if sweep % 4 == 0:
            top_lags = np.argpartition(c, -top_k)[-top_k:] if c.size > top_k else np.arange(c.size)
        else:
            top_lags = np.argmax(c)
            top_lags = np.array([int(top_lags)]) if np.isscalar(top_lags) else np.asarray(top_lags)

            if S <= 1e-300:

                c *= inv * inv
                infl *= inv
                S = 1.0
                c = np.maximum(c, 0.0)
                cur_score = float(2.0 * n * c.max())

            i = int(order[pos])
            v = float(p[i])
            start = i
            end = min(i + n, L)
            Ls = end - start

            infl_seg = infl[start:end]

            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < L:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

            Sloc = float(S)
            # Search range
            xhi = max(25.0 * v, 25.0 * Sloc / float(n), 0.003)

def _lp_direction(a, M):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0."""
    if linprog is None or sp is None:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n < 32 or n > 600:
        return None
    if not math.isfinite(M) or M <= 0.0:
        return None
    try:
        idx = np.arange(n, dtype=np.int32)
        cols = np.repeat(idx, n)
        off = np.tile(idx, n)
        rows = cols + off
        data = a[off].astype(np.float64)
        A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        cobj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        res = linprog(cobj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method="highs")
        if res is not None and getattr(res, "success", False):
            b = np.asarray(res.x, dtype=np.float64).ravel()
            b = np.where(np.isfinite(b), b, 0.0)
            b = np.maximum(b, 0.0)
            if float(b.sum()) > 0.0:
                return b
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1300.0
    reserve = 20.0 if budget > 200 else 10.0
    hard = t0 + budget - reserve

    parent = _get_parent()
    n = int(parent.size)
    best_arr = parent.copy()
    best_score = _score(best_arr)

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
            if arr.size == 0:
                return
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            s = float(arr.sum())
            if not math.isfinite(s) or s < 0.01:
                return
            arr = arr / s
            sc = _score(arr)
            if math.isfinite(sc) and sc < best_score - 1e-13:
                best_score = sc
                best_arr = arr.copy()
        except Exception:
            pass

    rng_top = np.random.default_rng(seed)
    explored = [best_arr.copy()]

    def add_explore(cand):
        cand = _normalize(cand)
        if cand.size != n:
            cand = _resample(cand, n)
        sc = _score(cand)
        if math.isfinite(sc) and sc < _score(best_arr) + 0.08:
            explored.append(cand)

    # === Phase 1: Diverse initializations ===
    phase1_end = min(hard - 60.0, t0 + max(2.0, 0.06 * budget))

    # Power transforms
    for q in (0.4, 0.6, 0.8, 1.4, 1.8, 2.5):
        if time.monotonic() > phase1_end:
            break
        cand = np.power(np.maximum(best_arr, 0.0), float(q))
        cand = _normalize(cand)
        consider(cand)
        add_explore(cand)

    # Gaussian / window shapes
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    for sigma in (0.1, 0.15, 0.2, 0.3, 0.4):
        if time.monotonic() > phase1_end:
            break
        cand = np.exp(-((x - 0.5) / sigma) ** 2)
        cand = _normalize(cand)
        consider(cand)
        add_explore(cand)

    # Raised cosine / Hann
    cand = 0.5 * (1 + np.cos(2 * np.pi * x))
    cand = _normalize(cand)
    consider(cand)
    add_explore(cand)

    # Periodic patterns
    for per in (3, 4, 5, 7):
        if time.monotonic() > phase1_end:
            break
        cand = 0.5 + 0.4 * np.cos(2 * np.pi * per * x)
        cand = np.maximum(cand, 0.01)
        cand = _normalize(cand)
        consider(cand)
        add_explore(cand)

    # Multi-hump
    for nh in (2, 3, 4):
        if time.monotonic() > phase1_end:
            break
        centers = np.linspace(0.15, 0.85, nh)
        cand = np.zeros(n, dtype=np.float64)
        for ct in centers:
            cand += np.exp(-((x - ct) / 0.12) ** 2)
        cand = _normalize(cand)
        consider(cand)
        add_explore(cand)

    # Random shaped perturbations
    for _ in range(6):
        if time.monotonic() > phase1_end:
            break
        amp = 0.3 + 0.8 * float(rng_top.random())
        f1 = 1 + int(rng_top.integers(0, 8))
        ph1 = 2.0 * math.pi * float(rng_top.random())
        f2 = 2 + int(rng_top.integers(0, 12))
        ph2 = 2.0 * math.pi * float(rng_top.random())
        shape = 1.0 + amp * math.sin(2.0 * math.pi * f1 * x + ph1) + 0.5 * amp * math.sin(2.0 * math.pi * f2 * x + ph2)
        cand = best_arr * np.maximum(shape, 0.05)
        cand = _normalize(cand)
        consider(cand)
        add_explore(cand)

    # LP at small scale for structured direction
    if budget > 300 and time.monotonic() < phase1_end - 30:
        for ns in (256, 384, 512):
            if time.monotonic() > phase1_end - 15:
                break
            if ns >= n:
                continue
            small = _resample(best_arr, ns)
            small_c = _fft_conv(small)
            M = float(small_c.max()) if small_c.size else 0
            if math.isfinite(M) and M > 0:
                b = _lp_direction(small, M)
                if b is not None:
                    g = b / float(b.sum())
                    for t in (0.4, 0.25, 0.15):
                        cand = (1 - t) * small + t * g
                        cand = _normalize(cand)
                        full_cand = _resample(cand, n)
                        consider(full_cand)
                        add_explore(full_cand)
                        break

    # === Phase 2: LBFGS smoothing on best ===
    phase2_end = min(hard - 80.0, t0 + max(20.0, 0.30 * budget))
    if phase2_end > time.monotonic() + 15:
        try:
            p2 = best_arr.copy()
            p2 = _lbfgs_smooth(p2, phase2_end, consider=consider)
            consider(p2)
        except Exception:
            pass

    # Try second-best explored point if significantly different
    if len(explored) > 1 and phase2_end > time.monotonic() + 30:
        alt = explored[-1]
        sc_alt = _score(alt)
        sc_best = _score(best_arr)
        if math.isfinite(sc_alt) and math.isfinite(sc_best) and sc_alt < sc_best + 0.03:
            alt_stop = min(phase2_end, time.monotonic() + max(15.0, 0.15 * (phase2_end - time.monotonic())))
            if alt_stop > time.monotonic() + 12:
                try:
                    p_alt = _lbfgs_smooth(alt, alt_stop, consider=consider)
                    consider(p_alt)
                except Exception:
                    pass

    # === Phase 3: Coordinate descent (main workhorse) ===
    phase3_end = min(hard - 30.0, t0 + max(30.0, 0.75 * budget))
    if phase3_end > time.monotonic() + 10:
        try:
            p3 = best_arr.copy()
            p3, _ = _coord_polish(p3, phase3_end, G=72, seed=seed + 1, consider=consider, top_k=5)
            consider(p3)
        except Exception:
            pass

    # === Phase 4: Fine polish ===
    phase4_end = min(hard - 5.0, t0 + budget - reserve)
    if phase4_end > time.monotonic() + 8:
        try:
            p4 = best_arr.copy()
            # Quick LBFGS with fine temperature
            fine_stop = min(phase4_end - 5.0, time.monotonic() + max(5.0, 0.25 * max(0.0, phase4_end - time.monotonic())))
            if fine_stop > time.monotonic() + 8:
                p4 = _lbfgs_smooth(p4, fine_stop, consider=consider)
                consider(p4)
            # Final coordinate descent
            cstop = min(phase4_end - 2.0, time.monotonic() + max(5.0, 0.70 * max(0.0, phase4_end - time.monotonic())))
            if cstop > time.monotonic() + 5:
                p4, _ = _coord_polish(p4, cstop, G=90, seed=seed + 7777, consider=consider, top_k=7)
                consider(p4)
        except Exception:
            pass

    # === Final validation and return ===
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or not math.isfinite(s) or s < 0.01:
            out = parent.copy()
            s = float(out.sum())
        if out.size == 0 or not math.isfinite(s) or s < 0.01:
            out = np.full(64, 1.0/64, dtype=np.float64)
        out = out / float(out.sum())
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls2 = [float(x) for x in parent / float(parent.sum())]
            sc2 = float(evaluate_sequence(ls2))
            if math.isfinite(sc2):
                return ls2
            return [1.0]
        return ls
    except Exception:
        return [1.0]
