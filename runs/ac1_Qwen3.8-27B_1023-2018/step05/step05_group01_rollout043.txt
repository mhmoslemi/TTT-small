        cum += wt
        end_stage = min(float(stop_time) - 2.0, start + avail * (cum / total_w))
        if end_stage <= now + 1.0:
            continue

        T = max(1e-300, frac * cmax_init)
        best_u = u.copy()
        f0, _ = lse_func(best_u, T)
        if not math.isfinite(f0):
            continue
        best_f = f0
        stale = 0

        while time.monotonic() < end_stage - 0.5:
            maxit = 30
            opts = {"maxiter": maxit, "gtol": 1e-12, "ftol": 1e-16,
                    "maxls": 30, "maxcor": 25, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-16:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 2:
                            best_u = best_u + rng.normal(0, 0.003, n)
                            continue
            if getattr(res, "status", 1) == 0 and time.monotonic() < end_stage - 1.0:
                if stale < 2:
                    best_u = best_u + rng.normal(0, 0.002, n)
                    f_chk, _ = lse_func(best_u, T)
                    if math.isfinite(f_chk) and f_chk < best_f:
                        best_f = f_chk
                        best_u = res.x.copy()
                    stale += 1
                else:
                    break

        p_cur = _softmax_fn(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = float(2.0 * n * _conv_self(p_cur).max())
        if math.isfinite(sc_cur) and sc_cur < best_score:
            best_score = sc_cur
            p = p_cur.copy()
            best_p = p_cur.copy()
        elif math.isfinite(sc_cur) and sc_cur < best_score + 0.01:
            p = p_cur.copy()
        else:
            p = best_p.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(p)
            except Exception:
                pass

    p = _softmax_fn(u)
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    return p


def _coord_descent(p, stop_time, G=None, seed=12345, consider=None, use_golden=True):
    """Fast coordinate descent with optional golden section refinement."""
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    if G is None:
        G = 32 if n <= 4096 else 24
    G = int(max(12, G))

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    phi = (1.0 + math.sqrt(5.0)) / 2.0  # golden ratio

    while sweep < 1000000 and time.monotonic() < stop_time - 1.0:
        improved = False
        # Use random order
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S_total <= 1e-300:
                break
            # Periodic renormalization
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0

            S = float(S_total)
            v = float(p[i])
            start = max(0, i)  # min lag index affected
            end = min(i + n, Ltot)  # max lag index affected (exclusive)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end].copy()
            # Influence: how c[k] changes when p[i] changes by w
            # c[k] += w * (p[k-i] + p[i-k]) + (if k == 2i) w^2
            # For k in [start, end): k-i is in [0, n-1] or i-k is in [0, n-1]
            # Actually: c[k] = sum_j p[j]*p[k-j]. Changing p[i] by w:
            #   c[k] += w*p[k-i] (if 0<=k-i<n) + w*p[i-k] (if 0<=i-k<n, i.e., k<i)
            #   + w^2 (if k == 2i)
            # So for k in [start, end):
            #   If k >= i: contribution is w * p[k-i]
            #   If k < i: contribution is w * p[i-k]  (but k < i means k-i < 0, so p[k-i]=0)
            #   Wait, let me be more careful.
            # c[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} p[j]*p[k-j]
            # d(c[k])/dp[i] = p[k-i] (if 0<=k-i<=n-1) + p[i-k] (if 0<=i-k<=n-1)
            # Note: k-i in [0,n-1] means k in [i, i+n-1]
            # i-k in [0,n-1] means k in [i-(n-1), i]
            # So the affected range is [max(0, i-(n-1)), min(Ltot, i+n-1)]
            # But since Ltot = 2n-1, i+n-1 <= 2n-1 always for i < n
            # And i-(n-1) >= -(n-1), so max(0, ...) = 0 for i < n-1, or i-n+1 for i >= n-1
            # Actually for i in [0, n-1]: affected k is [max(0, i-n+1), min(2n-1, i+n-1)]
            # = [0, i+n-1] if i < n-1, or [i-n+1, 2n-1] if i >= n-1
            # Hmm, this is getting complicated. Let me just use the original approach.

            # Original approach: c[k] for k in [i, i+n-1] gets w*p[k-i]
            # c[k] for k in [max(0, i-n+1), i-1] gets w*p[i-k]
            # c[2i] additionally gets w^2

            # Let me simplify: the range is [max(0, i-(n-1)), min(2n-2, i+(n-1))]
            start_i = max(0, i - (n - 1))
            end_i = min(2 * n - 1, i + n)  # exclusive
            Ls = end_i - start_i
            if Ls <= 0:
                continue

            base = c[start_i:end_i].copy()
            # infl[k] = d(c[start_i + k])/dp[i] (linear part)
            # For lag l = start_i + k:
            #   contribution from p[i] as first factor: p[l - i] if 0 <= l-i < n
            #   contribution from p[i] as second factor: p[i - l] if 0 <= i-l < n (i.e., l <= i)
            infl = np.zeros(Ls)
            for k in range(Ls):
                l = start_i + k
                val = 0.0
                ji = l - i  # j = l - i, need 0 <= ji < n
                if 0 <= ji < n:
                    val += p[ji]
                ji2 = i - l  # j = i - l, need 0 <= ji2 < n
                if 0 <= ji2 < n:
                    val += p[ji2]
                infl[k] = val

            # But this O(Ls) loop is slow. Let me vectorize.
            # l = start_i + arange(Ls)
            lags = start_i + np.arange(Ls)
            ji = lags - i
            contrib1 = np.where((ji >= 0) & (ji < n), p[ji[(ji >= 0) & (ji < n)]], 0.0)
            ji = i - lags
            contrib2 = np.where((ji >= 0) & (ji < n), p[ji[(ji >= 0) & (ji < n)]], 0.0)
            infl = contrib1 + contrib2

            # Actually the above has an issue with indexing. Let me redo:
            lags = start_i + np.arange(Ls, dtype=np.int64)
            idx1 = lags - i  # indices into p for first term
            valid1 = (idx1 >= 0) & (idx1 < n)
            idx2 = i - lags  # indices into p for second term
            valid2 = (idx2 >= 0) & (idx2 < n)
            infl = np.zeros(Ls)
            if np.any(valid1):
                infl[valid1] += p[idx1[valid1]]
            if np.any(valid2):
                infl[valid2] += p[idx2[valid2]]

            # max_const: max of c outside the affected range
            max_const = 0.0
            if start_i > 0:
                max_const = max(max_const, float(c[:start_i].max()))
            if end_i < Ltot:
                max_const = max(max_const, float(c[end_i:].max()))

            # The diagonal term: lag = 2*i
            diag_k = 2 * i - start_i  # index within [0, Ls)
            if 0 <= diag_k < Ls:
                pass  # will be handled in evaluation

            x_hi = max(20.0 * v, 40.0 * S / float(n), 0.01 * S, 1e-14)
            x_hi = min(x_hi, 10.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, S / float(n))

            def eval_single(x):
                """Evaluate score for candidate value x of p[i]."""
                w = x - v
                if abs(w) < 1e-18:
                    return cur_score if S_total == 1.0 else 2.0*n*c.max()/(S_total*S_total)
                denom = S + w
                if denom <= 1e-15:
                    return math.inf
                # New c values in affected range
                new_seg = base + w * infl
                if 0 <= diag_k < Ls:
                    new_seg[diag_k] += w * w
                new_seg = np.maximum(new_seg, max_const)
                mv = float(new_seg.max())
                return (2.0 * n * mv) / (denom * denom)

            # Step 1: Coarse grid
            grid = np.linspace(0.0, x_hi, G + 1)
            if 0.0 < v < x_hi:
                grid = np.unique(np.concatenate([grid, [v]]))
            vals = np.array([eval_single(float(x)) for x in grid])
            valid_mask = np.isfinite(vals)
            if not np.any(valid_mask):
                continue
            imin = int(np.argmax(valid_mask * vals - 1e100))  # this is wrong
            # Fix: find min among valid
            vals_safe = np.where(valid_mask, vals, 1e100)
            imin = int(np.argmin(vals_safe))
            best_x = float(grid[imin])
            best_val = float(vals_safe[imin])

            # Step 2: Golden section refinement
            if use_golden and best_x > 0 and best_x < x_hi:
                # Search in [grid[imin-1], grid[imin+1]] (clamped)
                lo = float(grid[max(0, imin - 1)])
                hi = float(grid[min(len(grid) - 1, imin + 1)])
                if hi - lo < 1e-15:
                    pass
                else:
                    # Golden section
                    gr = (3.0 - math.sqrt(5.0)) / 2.0  # ~0.382
                    a_g = lo
                    b_g = hi
                    c_g = b_g - gr * (b_g - a_g)
                    d_g = a_g + gr * (b_g - a_g)
                    fc = eval_single(c_g)
                    fd = eval_single(d_g)
                    for _ in range(8):
                        if fc < fd:
                            b_g = d_g
                            d_g = c_g
                            fd = fc
                            c_g = b_g - gr * (b_g - a_g)
                            fc = eval_single(c_g)
                        else:
                            a_g = c_g
                            c_g = d_g
                            fc = fd
                            d_g = a_g + gr * (b_g - a_g)
                            fd = eval_single(d_g)
                    best_x_candidate = (a_g + b_g) / 2.0
                    best_val_candidate = eval_single(best_x_candidate)
                    if best_val_candidate < best_val - 1e-15:
                        best_x = best_x_candidate
                        best_val = best_val_candidate

            if (math.isfinite(best_val) and best_val < cur_score - 1e-12
                    and 0.0 <= best_x <= 50.0 * S):
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start_i:end_i] += w * infl
                    if 0 <= 2 * i < Ltot:
                        c[2 * i] += w * w
                    cur_score = best_val
                    improved = True

        # End of sweep
        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0

        # Recompute c exactly (to correct drift)
        c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 5.0:
                    sigma = 0.01 + 0.01 * min(flat, 15) / 15.0
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    else:
                        break
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    flat = 0
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = float(2.0 * n * _conv_self(p).max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _subspace_opt(p, stop_time, m_size=3, consider=None, seed=999):
    """Optimize small subspaces of coordinates jointly."""
    n = int(p.size)
    p = p.copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p /= s

    Ltot = 2 * n - 1
    c = _conv_self(p)
    best_score = float(2.0 * n * c.max())
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    # Find the lag index of the max
    max_lag = int(np.argmax(c))

    # Coordinates that most influence c[max_lag]:
    # c[max_lag] = sum_j p[j] * p[max_lag - j]
    # These are the pairs (j, max_lag - j) for valid j
    influence = np.zeros(n)
    for j in range(n):
        partner = max_lag - j
        if 0 <= partner < n:
            influence[j] = p[partner]
        else:
            influence[j] = 0.0

    # Pick top m_size by influence, plus some random
    top_idx = np.argsort(-influence)[:m_size]
    for idx in range(int(stop_time) - time.monotonic() > 20 and top_idx.size):  # time check
        pass

    # Simple 2D optimization for pairs
    pair_count = 0
    while time.monotonic() < stop_time - 2.0 and pair_count < 500:
        # Pick a random pair of indices
        i1 = int(rng.integers(0, n))
        i2 = int(rng.integers(0, n))
        if i1 == i2:
            continue
        pair_count += 1

        v1, v2 = float(p[i1]), float(p[i2])
        # The affected lags are in [min(i1,i2)-n+1, max(i1,i2)+n-1]
        start = max(0, min(i1, i2) - n + 1)
        end = min(Ltot, max(i1, i2) + n)
        Ls = end - start
        if Ls <= 0 or Ls > 2 * n:
            continue

        base = c[start:end].copy()
        lags = start + np.arange(Ls, dtype=np.int64)

        # Compute infl1 and infl2 (linear part for p[i1] and p[i2])
        infl1 = np.zeros(Ls)
        infl2 = np.zeros(Ls)
        idx1 = lags - i1
        valid1 = (idx1 >= 0) & (idx1 < n)
        if np.any(valid1):
            infl1[valid1] += p[idx1[valid1]]
        idx1b = i1 - lags
        valid1b = (idx1b >= 0) & (idx1b < n)
        if np.any(valid1b):
            infl1[valid1b] += p[idx1b[valid1b]]

        idx2 = lags - i2
        valid2 = (idx2 >= 0) & (idx2 < n)
        if np.any(valid2):
            infl2[valid2] += p[idx2[valid2]]
        idx2b = i2 - lags
        valid2b = (idx2b >= 0) & (idx2b < n)
        if np.any(valid2b):
            infl2[valid2b] += p[idx2b[valid2b]]

        # Cross-influence: d(c[k])/dp[i1]/dp[i2] contributes w1*w2 where both i1 and i2 are in the sum
        # c[k] = ... + p[i1]*p[k-i1] + p[i2]*p[k-i2] + ...
        # If k-i1 == i2 (i.e., k = i1+i2): extra term w1*w2
        # If k-i2 == i1 (same): already counted
        # Also if i1 == i2 (excluded)
        cross_diag = i1 + i2  # lag where cross-term appears
        cross_k = cross_diag - start
        has_cross = 0 <= cross_k < Ls

        max_const = 0.0
        if start > 0:
            max_const = max(max_const, float(c[:start].max()))
        if end < Ltot:
            max_const = max(max_const, float(c[end:].max()))

        S = float(p.sum())
        x1_hi = max(20.0 * v1, 10.0 / n, 1e-14)
        x2_hi = max(20.0 * v2, 10.0 / n, 1e-14)
        x1_hi = min(x1_hi, 5.0 * S)
        x2_hi = min(x2_hi, 5.0 * S)

        # 3x3 grid
        g1 = np.array([0.0, v1 * 0.5, v1, v1 * 1.5, x1_hi])
        g2 = np.array([0.0, v2 * 0.5, v2, v2 * 1.5, x2_hi])
        g1 = np.unique(g1)
        g2 = np.unique(g2)

        best_val = float(2.0 * n * c.max()) / (S * S)  # current
        best_x1, best_x2 = v1, v2

        for x1 in g1:
            w1 = float(x1) - v1
            for x2 in g2:
                w2 = float(x2) - v2
                denom = S + w1 + w2
                if denom <= 1e-15:
                    continue
                new_seg = base + w1 * infl1 + w2 * infl2
                if has_cross:
                    new_seg[cross_k] += w1 * w2
                # Diagonal terms (w1^2 at lag 2*i1, w2^2 at lag 2*i2)
                k1 = 2 * i1 - start
                if 0 <= k1 < Ls:
                    new_seg[k1] += w1 * w1
                k2 = 2 * i2 - start
                if 0 <= k2 < Ls:
                    new_seg[k2] += w2 * w2
                new_seg = np.maximum(new_seg, max_const)
                mv = float(new_seg.max())
                val = (2.0 * n * mv) / (denom * denom)
                if val < best_val - 1e-15:
                    best_val = val
                    best_x1, best_x2 = float(x1), float(x2)

        if best_val < float(2.0 * n * _conv_self(p).max()) / (S * S) - 1e-13:
            # Apply update
            p[i1] = best_x1
            p[i2] = best_x2
            c[start:end] += (best_x1 - v1) * infl1 + (best_x2 - v2) * infl2
            if has_cross:
                c[cross_diag] += (best_x1 - v1) * (best_x2 - v2)
            k1 = 2 * i1
            if 0 <= k1 < Ltot:
                c[k1] += (best_x1 - v1) ** 2
            k2 = 2 * i2
            if 0 <= k2 < Ltot:
                c[k2] += (best_x2 - v2) ** 2

            # Renormalize occasionally
            s_new = float(p.sum())
            if s_new > 0 and (s_new > 1.2 or s_new < 0.8):
                p /= s_new
                c /= (s_new * s_new)

    s_final = float(p.sum())
    if s_final > 0:
        p /= s_final
    sc = float(2.0 * n * _conv_self(p).max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p


def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        return src.copy()
    m = src.size
    xp = (np.arange(m) + 0.5) / m
    fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
    out = np.maximum(fp, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 2.0:
        hard = t0 + max(2.0, budget * 0.9)

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss

    def _exact_score(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        n = a.size
        s = float(a.sum())
        if s < 0.01:
            return math.inf
        b = np.convolve(a, a)
        m = float(b.max())
        if not math.isfinite(m):
            return math.inf
        return 2.0 * n * m / (s * s)

    try:
        best_score = _exact_score(best_arr)
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = _exact_score(best_arr)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0 or float(a.sum()) < 0.01:
            return
        try:
            sc = _exact_score(a)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            a_norm = a / float(a.sum())
            best_arr = a_norm.copy()
            best_score = sc

    n_parent = int(best_arr.size)

    # Phase 1: LSE optimization at parent length
    phase1_end = min(hard - 60.0, t0 + 0.25 * (hard - t0))
    if phase1_end > t0 + 20.0:
        try:
            p1 = _lse_optimize(best_arr, phase1_end, consider=consider, seed=int(seed))
            if p1 is not None and p1.size == n_parent and float(p1.sum()) > 0:
                p1 /= float(p1.sum())
                consider(p1)
        except Exception:
            pass

    # Phase 2: Coordinate descent (main workhorse)

    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    u = _p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax_fn(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    c_init = cfast(p)
    cmax_init = float(c_init.max()) if c_init.size else 1.0
    if not math.isfinite(cmax_init) or cmax_init <= 0:
        cmax_init = 1.0

    stages = [
        (0.08, 1.0),
        (0.03, 1.0),
        (0.01, 1.2),
        (0.003, 1.5),
        (0.001, 1.8),
        (0.0003, 2.0),
        (0.00008, 2.5),
        (0.00002, 3.0),
    ]

    start = time.monotonic()
    avail = max(0.0, float(stop_time) - start - 5.0)
    if avail <= 5.0:
        return p

    best_p = p.copy()
    best_sc = float(2.0 * n * cmax_init)
    total_w = sum(w for _, w in stages)
    rng = np.random.default_rng(seed)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        cum += wt
        end_stage = min(float(stop_time) - 2.0, start + avail * (cum / total_w))
        if end_stage <= now + 1.0:
            continue

        T = max(1e-300, frac * cmax_init)
        best_u = u.copy()
        f0, _ = lse_func(best_u, T)
        if not math.isfinite(f0):
            continue
        best_f = f0
        stale = 0

        while time.monotonic() < end_stage - 0.5:
            opts = {"maxiter": 30, "gtol": 1e-12, "ftol": 1e-16,
                    "maxls": 30, "maxcor": 25, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-16:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 2:
                            best_u = best_u + rng.normal(0, 0.003, n)
                            continue
            if getattr(res, "status", 1) == 0 and time.monotonic() < end_stage - 1.0:
                if stale < 3:
                    best_u = best_u + rng.normal(0, 0.002, n)
                    f_chk, _ = lse_func(best_u, T)
                    stale += 1
                else:
                    break

        p_cur = _u_to_p(best_u)
        sc_cur = _score_norm(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_sc = sc_cur
            p = p_cur.copy()
            best_p = p_cur.copy()
        elif math.isfinite(sc_cur) and sc_cur < best_sc + 0.01:
            p = p_cur.copy()
        else:
            p = best_p.copy()
        u = _p_to_u(p)
        if consider is not None:
            try:
                consider(p)
            except Exception:
                pass

    return best_p


def _coord_descent(p, stop_time, G=40, seed=12345, consider=None):
    n = int(p.size)
    p = np.maximum(p.copy(), 0.0)
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p /= s

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S = 1.0
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    flat = 0

    gr_phi = (3.0 - math.sqrt(5.0)) / 2.0

    while time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break

            if S > 1.25 or S < 0.8:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0
                cur_score = 2.0 * n * float(c.max())

            v = float(p[i])
            start_i = max(0, i - (n - 1))
            end_i = min(Ltot, i + n)
            Ls = end_i - start_i
            if Ls <= 0:
                continue

            base = c[start_i:end_i].copy()

            lags = start_i + np.arange(Ls, dtype=np.int64)
            infl = np.zeros(Ls)
            idx1 = lags - i
            valid1 = (idx1 >= 0) & (idx1 < n)
            if np.any(valid1):
                infl[valid1] += p[idx1[valid1]]
            idx2 = i - lags
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid2):
                infl[valid2] += p[idx2[valid2]]

            max_const = 0.0
            if start_i > 0:
                ml = float(c[:start_i].max())
                if ml > max_const:
                    max_const = ml
            if end_i < Ltot:
                mr = float(c[end_i:].max())
                if mr > max_const:
                    max_const = mr

            diag_k = 2 * i - start_i
            has_diag = (0 <= diag_k < Ls)

            x_hi = max(25.0 * v, 50.0 / n, 0.02 / n, 1e-12)
            x_hi = min(x_hi, 8.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-10, 1.0 / n)

            # Vectorized grid evaluation
            xs = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            ws = xs - v
            denoms = S + ws
            valid_d = denoms > 1e-15

            if not np.any(valid_d):
                continue

            # Build (G+1, Ls) matrix but do it in a memory-efficient way
            # Use chunking if needed
            mat = base[None, :] + ws[:, None] * infl[None, :]
            if has_diag:
                mat[:, diag_k] += ws * ws
            np.maximum(mat, max_const, out=mat)
            row_max = mat.max(axis=1)

            vals = np.full(len(xs), 1e100)
            vals[valid_d] = (2.0 * n * row_max[valid_d]) / (denoms[valid_d] ** 2)

            vals_safe = np.where(np.isfinite(vals), vals, 1e100)
            imin = int(np.argmin(vals_safe))
            best_x = float(xs[imin])
            best_val = float(vals_safe[imin])

            # Golden section refinement
            if best_x > 0.5 * x_hi / G and best_x < x_hi - 0.5 * x_hi / G:
                lo_g = float(max(0.0, xs[max(0, imin - 1)]))
                hi_g = float(xs[min(G, imin + 1)])
                if hi_g - lo_g > 1e-14:
                    def eval_pt(x_pt):
                        w_pt = x_pt - v
                        d_pt = S + w_pt
                        if d_pt <= 1e-15:
                            return 1e100
                        seg = base + w_pt * infl
                        if has_diag:
                            seg[diag_k] += w_pt * w_pt
                        mx = float(max(seg.max(), max_const))
                        return (2.0 * n * mx) / (d_pt * d_pt)

                    a_g, b_g = lo_g, hi_g
                    c_g = b_g - gr_phi * (b_g - a_g)
                    d_g = a_g + gr_phi * (b_g - a_g)
                    fc = eval_pt(c_g)
                    fd = eval_pt(d_g)
                    for _ in range(7):
                        if fc < fd:
                            b_g = d_g
                            d_g = c_g
                            fd = fc
                            c_g = b_g - gr_phi * (b_g - a_g)
                            fc = eval_pt(c_g)
                        else:
                            a_g = c_g
                            c_g = d_g
                            fc = fd
                            d_g = a_g + gr_phi * (b_g - a_g)
                            fd = eval_pt(d_g)
                    mid = (a_g + b_g) / 2.0
                    fm = eval_pt(mid)
                    if fm < best_val - 1e-14:
                        best_x, best_val = mid, fm

            if best_val < cur_score - 1e-13 and 0.0 <= best_x <= 20.0 * S:
                w = best_x - v
                if abs(w) > 1e-16:
                    p[i] = best_x
                    S += w
                    c[start_i:end_i] += w * infl
                    if has_diag:
                        c[2 * i] += w * w
                    cur_score = best_val
                    improved = True

        # End of sweep
        if S > 1e-10:
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        c = _conv_self(p)
        new_score = 2.0 * n * float(c.max())
        if new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if not improved:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 5.0:
                    sigma = 0.012 + 0.008 * min(flat, 12) / 12.0
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    psum = float(p.sum())
                    if psum > 0:
                        p /= psum
                    else:
                        break
                    c = _conv_self(p)
                    S = 1.0
                    cur_score = 2.0 * n * float(c.max())
                    flat = 0
                else:
                    break
        else:
            flat = 0

    psum = float(p.sum())
    if psum > 0:
        p /= psum
    sc = 2.0 * n * float(_conv_self(p).max())
    if sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _pair_opt(p, stop_time, consider=None, seed=777, max_pairs=2000):
    """Optimize pairs of coordinates for larger moves."""
    n = int(p.size)
    p = np.maximum(p.copy(), 0.0)
    s = float(p.sum())
    if s < 1e-300:
        return p
    p /= s

    Ltot = 2 * n - 1
    c = _conv_self(p)
    best_score = 2.0 * n * float(c.max())
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    S = 1.0
    pairs_done = 0
    no_improve = 0

    while time.monotonic() < stop_time - 2.0 and pairs_done < max_pairs:
        if no_improve > 50:
            sigma = 0.02
            p = p * np.exp(rng.normal(0, sigma, n))
            p = np.maximum(p, 0)
            p /= float(p.sum())
            c = _conv_self(p)
            S = 1.0
            no_improve = 0

        i1 = int(rng.integers(0, n))
        i2 = int(rng.integers(0, n))
        if i1 == i2:
            pairs_done += 1
            continue
        pairs_done += 1

        if S > 1.2 or S < 0.85:
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0

        v1, v2 = float(p[i1]), float(p[i2])
        start = max(0, min(i1, i2) - (n - 1))
        end = min(Ltot, max(i1, i2) + n)
        Ls = end - start
        if Ls <= 0 or Ls > 3 * n:
            continue

        base = c[start:end].copy()
        lags = start + np.arange(Ls, dtype=np.int64)

        infl1 = np.zeros(Ls)
        idx1 = lags - i1
        va1 = (idx1 >= 0) & (idx1 < n)
        if np.any(va1):
            infl1[va1] += p[idx1[va1]]
        idx1b = i1 - lags
        va1b = (idx1b >= 0) & (idx1b < n)
        if np.any(va1b):
            infl1[va1b] += p[idx1b[va1b]]

        infl2 = np.zeros(Ls)
        idx2 = lags - i2
        va2 = (idx2 >= 0) & (idx2 < n)
        if np.any(va2):
            infl2[va2] += p[idx2[va2]]
        idx2b = i2 - lags
        va2b = (idx2b >= 0) & (idx2b < n)
        if np.any(va2b):
            infl2[va2b] += p[idx2b[va2b]]

        cross_lag = i1 + i2
        cross_k = cross_lag - start
        has_cross = 0 <= cross_k < Ls

        k1_pos = 2 * i1 - start
        has_k1 = 0 <= k1_pos < Ls
        k2_pos = 2 * i2 - start
        has_k2 = 0 <= k2_pos < Ls

        max_const = 0.0
        if start > 0:
            max_const = max(max_const, float(c[:start].max()))
        if end < Ltot:
            max_const = max(max_const, float(c[end:].max()))

        x1_hi = max(20.0 * v1, 30.0 / n, 1e-12)
        x2_hi = max(20.0 * v2, 30.0 / n, 1e-12)
        x1_hi = min(x1_hi, 5.0 * S)
        x2_hi = min(x2_hi, 5.0 * S)

        # 4x4 grid
        g1 = np.array([0.0, v1 * 0.5, v1, v1 * 1.5, x1_hi])
        g2 = np.array([0.0, v2 * 0.5, v2, v2 * 1.5, x2_hi])

        best_val = 2.0 * n * float(c.max()) / (S * S)
        best_x1, best_x2 = v1, v2

        for x1 in g1:
            w1 = float(x1) - v1
            for x2 in g2:
                w2 = float(x2) - v2
                denom = S + w1 + w2
                if denom <= 1e-15:
                    continue
                seg = base + w1 * infl1 + w2 * infl2
                if has_k1:
                    seg[k1_pos] += w1 * w1
                if has_k2:
                    seg[k2_pos] += w2 * w2
                if has_cross:
                    seg[cross_k] += w1 * w2
                mx = float(max(seg.max(), max_const))
                val = (2.0 * n * mx) / (denom * denom)
                if val < best_val - 1e-14:
                    best_val = val
                    best_x1, best_x2 = float(x1), float(x2)

        if best_val < 2.0 * n * float(c.max()) / (S * S) - 1e-13:
            w1 = best_x1 - v1
            w2 = best_x2 - v2
            p[i1] = best_x1
            p[i2] = best_x2
            S += w1 + w2
            c[start:end] += w1 * infl1 + w2 * infl2
            if has_k1:
                c[2 * i1] += w1 * w1
            if has_k2:
                c[2 * i2] += w2 * w2
            if has_cross:
                c[cross_lag] += w1 * w2
            no_improve = 0
        else:
            no_improve += 1

    s_final = float(p.sum())
    if s_final > 0:
        p /= s_final
    sc = 2.0 * n * float(_conv_self(p).max())
    if sc < best_score:
        best_score = sc
        best_p = p.copy()
    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p


def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        return src.copy()
    m = src.size
    xp = (np.arange(m) + 0.5) / m
    fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
    out = np.maximum(fp, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 2.0:
        hard = t0 + max(2.0, budget * 0.88)

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    else:
        best_arr = np.ones(64) / 64.0
        ss = 1.0

    def _exact(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        n = a.size
        s = float(a.sum())
        if s < 0.01:
            return math.inf
        b = np.convolve(a, a)
        m = float(b.max())
        if not math.isfinite(m):
            return math.inf
        return 2.0 * n * m / (s * s)

    try:
        best_score = _exact(best_arr)
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score):
        best_arr = np.ones(64) / 64.0
        best_score = _exact(best_arr)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0 or float(a.sum()) < 0.01:
                return
            sc = _exact(a)
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_arr = (a / float(a.sum())).copy()
                best_score = sc
        except Exception:
            pass

    n_p = int(best_arr.size)

    # Phase 1: LSE optimization (~20% of time)
    p1_end = min(hard - 60.0, t0 + 0.20 * (hard - t0))
    if p1_end > t0 + 15.0:
        try:
            p1 = _lse_optimize(best_arr, p1_end, consider=consider, seed=int(seed))
            if p1 is not None and p1.size == n_p and float(p1.sum()) > 0:
                p1 /= float(p1.sum())
                consider(p1)
        except
