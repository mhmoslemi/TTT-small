                            p /= ss
                        c = cfast(p)
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < local_best:
                            local_best = cur_score
                            local_best_p = p.copy()
                            consider(p)
                        flat = 0
                    else:
                        break
            sweep += 1

        return local_best_p, local_best

    # Run coordinate descent
    now = time.monotonic()
    cd_stop = min(hard_deadline - 15.0, now + 0.35 * (hard_deadline - t0))
    if cd_stop > now + 20.0:
        p_cd, sc_cd = coordinate_descent(best_arr, cd_stop, max_sweeps=100000,
                                          G=24, seed_cd=seed + 55555)
        if p_cd is not None and p_cd.size == n and np.all(np.isfinite(p_cd)):
            consider(p_cd)

    # === Phase 3: Fine LSE + coordinate descent ===
    now = time.monotonic()
    if now < hard_deadline - 30.0:
        # Fine LSE
        u = p_to_u(best_arr)
        c = cfast(best_arr)
        cmax = float(c.max())
        if math.isfinite(cmax) and cmax > 0:
            fine_stages = [(0.000005, 0.4), (0.000002, 0.35), (0.000001, 0.25)]
            ft_end = min(hard_deadline - 25.0, now + 15.0)
            ftot = sum(w for _, w in fine_stages)
            fcum = 0.0
            for ffrac, fwt in fine_stages:
                if time.monotonic() > ft_end - 2.0:
                    break
                fcum += fwt
                fend = min(ft_end, now + 15.0 * (fcum / ftot))
                T = max(1e-300, ffrac * cmax)
                best_u_f = u.copy()
                best_f_f = float(lse_func(u, T)[0])
                stale_f = 0
                while time.monotonic() < fend - 0.5:
                    opts = {"maxiter": 25, "gtol": 1e-15, "ftol": 1e-18,
                            "maxls": 20, "maxcor": 15, "maxdisp": 0, "disp": False}
                    try:
                        res = minimize(lse_func, best_u_f, args=(T,),
                                       method="L-BFGS-B", jac=True, options=opts)
                    except Exception:
                        break
                    if res.x is not None:
                        x = np.asarray(res.x, dtype=np.float64).ravel()
                        if x.size == u.size and np.all(np.isfinite(x)):
                            fval = float(res.fun)
                            if math.isfinite(fval) and fval < best_f_f - 1e-17:
                                best_f_f = fval
                                best_u_f = x.copy()
                                stale_f = 0
                            else:
                                stale_f += 1
                                if stale_f >= 3:
                                    break
                    if getattr(res, "status", 1) == 0:
                        break
                p_f = softmax(best_u_f)
                p_f = np.maximum(p_f, 0.0)
                ss = float(p_f.sum())
                if ss > 1e-300:
                    p_f /= ss
                best_arr = p_f.copy()
                u = p_to_u(best_arr)
                consider(best_arr)
                c = cfast(best_arr)
                cmax = float(c.max()) if c.size else 0.0

        # Final coordinate descent
        now = time.monotonic()
        fcd_stop = min(hard_deadline - 8.0, now + 0.4 * (hard_deadline - now))
        if fcd_stop > now + 10.0:
            p_fcd, _ = coordinate_descent(best_arr, fcd_stop, max_sweeps=50000,
                                           G=20, seed_cd=seed + 88888)
            if p_fcd is not None and p_fcd.size == n and np.all(np.isfinite(p_fcd)):
                consider(p_fcd)

    # === Final validation ===
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.ones(64) / 64
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        try:
            return [float(x) for x in parent[:64]]
        except Exception:
            return [1.0]
