    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.

    if not isinstance(sequence, list):

        return np.inf

        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf

def _subgradient_polish(p, stop_time, consider=None, max_iters=3000):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    best_p = p.copy()
    best_score = float(2.0 * n * _conv_self(p).max())
    base_step = 0.005
    min_step = 1e-8

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break

        c = _conv_self(p)
        cmax = float(c.max())
        threshold = cmax * 0.999
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Compute gradient of max (subgradient = average over active lags)
        grad = np.zeros(n)
        for k in active:
            # d/dx_j (a*a)[k] = a[k-j] + a[j] if 0 <= k-j < n
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                contrib = p[kj[valid]] + p[valid]
                grad[valid] += contrib
        grad /= len(active)

        step = base_step / (1.0 + 0.02 * it)
        if step < min_step:
            break

        for scale in (1.0, 0.5, 0.25, 0.1):
            p_cand = p - scale * step * grad
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = float(2.0 * n * _conv_self(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-16:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                break
        else:
            base_step *= 0.9
            if base_step < 1e-12:
                break

        if it % 50 == 0 and consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p
