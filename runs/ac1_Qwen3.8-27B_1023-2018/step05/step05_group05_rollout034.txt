  - The sum becomes $S' = S - a_i + x$
  - The convolution values become as above
  - The score is $2n \cdot \max_k (a'a')_k / (S')^2$
  
  To minimize this over $x \ge 0$, we can:
  1. Identify the "active" set of lags where $(a*a)_k$ is near the maximum
  2. For the active set, the function is piecewise: each piece corresponds to a different lag being the maximizer
  3. On each piece, the score is a rational function (ratio of a quadratic in $x$ to a quadratic in $x$), which can be minimized analytically
  
  For the piece where lag $k^*$ is the maximizer:
  $f(x) = 2n \cdot [(a*a)_{k^*} + 2(x-a_i)a_{k^*-i} + (x^2-a_i^2)\mathbf{1}[k^*=2i]] / (S-a_i+x)^2$
  
  This is $2n \cdot (Ax^2 + Bx + C) / (x+D)^2$ for constants $A, B, C, D$.
  
  Taking derivative: $\frac{d}{dx} \frac{Ax^2 + Bx + C}{(x+D)^2} = 0$
  $\Rightarrow \frac{(2Ax+B)(x+D)^2 - (Ax^2+Bx+C) \cdot 2(x+D)}{(x+D)^4} = 0$
  $\Rightarrow (2Ax+B)(x+D) - 2(Ax^2+Bx+C) = 0$
  $\Rightarrow 2Ax^2 + 2ADx + Bx + BD - 2Ax^2 - 2Bx - 2C = 0$
  $\Rightarrow (2AD - B)x + (BD - 2C) = 0$
  $\Rightarrow x^* = (2C - BD) / (2AD - B)$
  
  So for each active lag, we can compute the optimal $x$ analytically! Then we check which lag is actually the maximizer at that $x$, and iterate.

                            ki = k - i
                            if 0 <= ki < n:
                                if k == 2 * i:
                                    c2[k] = old_c[k] - ai * ai + try_x * try_x
                                else:
                                    c2[k] = old_c[k] + 2.0 * w2 * p[ki]
                        ns = Sp + try_x
                        if ns <= 0:
                            continue
                        rs = 2.0 * n * float(c2.max()) / (ns * ns)
                        if rs < cur_score - 1e-12:
                            p[i] = try_x
                            s = 1.0
                            p /= (Sp + try_x)
                            c[:] = _conv_fft(p)
                            cur_score = 2.0 * n * float(c.max())
                            if cur_score < best_score:
                                best_score = cur_score
                                best_p = p.copy()
                            improved = True
                            found = True
                            break
                    if not found:
                        break

        if not improved:
            no_improve += 1
            if no_improve >= 80:
                break
            # Perturb to escape local minimum
            if no_improve % 10 == 0:
                eps = 0.01 * no_improve / 80
                p = p * np.exp(rng.normal(0, eps, n))
                p = np.maximum(p, 0)
                ps = float(p.sum())
                if ps > 0:
                    p /= ps
                c = _conv_fft(p)
                cur_score = 2.0 * n * float(c.max())
                no_improve = 0
        else:
            no_improve = 0
        sweep += 1

    return best_p, best_score


def _subgradient_step(p, step_size, n_top=5):
    """One averaged-subgradient step with top-k lags."""
    n = p.size
    c = _conv_fft(p)
    # Get top lags
    top_k = np.argsort(c)[-n_top:]
    s = float(p.sum())
    # Subgradient: for each top lag k*, grad_i += weight * 2*a[k*-i]
    grad = np.zeros(n)
    for k in top_k:
        ki = k
        # grad contribution from lag k: 2 * a[max(0, k-(n-1)) : min(n, k+1)] shifted
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        idx = np.arange(lo, hi)
        vals = k - idx  # these are the a[j] where j+k-i... 
        # Actually: d(dk)/da_i = 2*a[k-i] for 0<=k-i<n
        j_vals = k - np.arange(n)
        valid = (j_vals >= 0) & (j_vals < n)
        if np.any(valid):
            grad[valid] += 2.0 * p[j_vals[valid]]
    # Normalize gradient
    gn = float(np.sqrt(np.sum(grad**2)))
    if gn > 1e-15:
        grad /= gn
    # Projected step
    p_new = p - step_size * grad / n
    # Project to simplex: p >= 0, sum(p) = 1
    p_new = np.maximum(p_new, 0)
    ps = float(p_new.sum())
    if ps > 1e-300:
        p_new /= ps
    else:
        p_new = p.copy()
    return p_new


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 30.0

    parent = _get_parent()
    n = parent.size
    n = min(n, 3072)

    best_arr = parent[:n].copy()
    best_arr = np.maximum(best_arr, 0)
    bs = float(best_arr.sum())
    if bs > 0:
        best_arr /= bs
    best_score = math.inf
    try:
        best_score = evaluate_sequence([float(x) for x in best_arr])
    except Exception:
        pass

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            sc = float(evaluate_sequence([float(x) for x in a]))
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_arr = a.copy()
        except Exception:
            pass

    # Phase 1: Coordinate descent on parent
    cd_stop1 = t0 + min(400.0, 0.4 * budget)
    if cd_stop1 > t0 + 30:
        p1, sc1 = _cd_analytical(best_arr, cd_stop1, seed=int(seed))
        if p1 is not None and p1.size == n and np.all(np.isfinite(p1)):
            consider(p1)

    # Phase 2: Subgradient + restart cycles
    rng = np.random.default_rng(int(seed) + 999)
    n_restarts = 4
    rem = hard_deadline - time.monotonic()
    per_restart = max(20, rem / max(1, n_restarts) - 10)

    current = best_arr.copy()
    for r in range(n_restarts):
        phase_stop = time.monotonic() + per_restart
        if phase_stop < time.monotonic() + 15:
            break

        # Perturb current best
        if r > 0:
            current = best_arr * np.exp(rng.normal(0, 0.05, n))
            current = np.maximum(current, 0)
            cs = float(current.sum())
            if cs > 0:
                current /= cs

        # Subgradient phase (short)
        sg_stop = min(phase_stop - 40, time.monotonic() + 30)
        eta = 0.02
        it = 0
        while time.monotonic() < sg_stop:
            current = _subgradient_step(current, eta, n_top=3)
            it += 1
            if it % 50 == 0:
                consider(current)
                eta = max(0.001, eta * 0.98)

        # CD polish
        cd_stop2 = phase_stop - 5
        p2, sc2 = _cd_analytical(current, cd_stop2, seed=int(seed) + r * 7)
        if p2 is not None and p2.size == n and np.all(np.isfinite(p2)):
            consider(p2)

    # Phase 3: Final intensive CD on best found
    final_stop = hard_deadline
    if time.monotonic() < final_stop - 20:
        p3, sc3 = _cd_analytical(best_arr, final_stop, seed=int(seed) + 7777)
        if p3 is not None and p3.size == n and np.all(np.isfinite(p3)):
            consider(p3)

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        os_ = float(out.sum())
        if os_ < 0.01:
            out = np.ones(64) / 64
        ls = [float(x) for x in out]
        fs = float(evaluate_sequence(ls))
        if not math.isfinite(fs):
            ls = [float(x) for x in parent[:n]]
            fs = float(evaluate_sequence(ls))
            if not math.isfinite(fs):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
