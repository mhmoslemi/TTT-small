        s = float(a.sum())
        if s < 0.01:
            return
        sc = _score(a)
        if math.isfinite(sc) and sc < best_sc - 1e-15:
            best_sc = sc
            best_p = (a / s).copy()

    p = parent.copy()
    consider(p)

    # Phase 1: L-BFGS (25% of usable time)
    usable = max(1.0, hard_end - t0 - 10.0)
    p1_end = t0 + 0.25 * usable
    if time.monotonic() < p1_end - 5.0:
        try:
            p = _lbfgs_opt(p, n, p1_end, consider)
            consider(p)
        except Exception:
            pass
        p = best_p.copy()

    # Phase 2: Main coordinate descent (55% of usable time)
    p2_end = t0 + 0.80 * usable
    if time.monotonic() < p2_end - 5.0:
        try:
            p = _coord_descent(p, p2_end, seed + 1, consider)
            consider(p)
        except Exception:
            pass
        p = best_p.copy()

    # Phase 3: Fine L-BFGS (10%)
    p3_end = t0 + 0.90 * usable
    if time.monotonic() < p3_end - 5.0:
        try:
            p = _lbfgs_opt(p, n, p3_end, consider)
            consider(p)
        except Exception:
            pass
        p = best_p.copy()

    # Phase 4: Final coordinate descent (10%)
    p4_end = hard_end - 5.0
    if time.monotonic() < p4_end - 3.0:
        try:
            p = _coord_descent(p, p4_end, seed + 999, consider)
            consider(p)
        except Exception:
            pass

    # Final validation with exact evaluator
    out = np.asarray(best_p, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    if out.size == 0 or float(out.sum()) < 0.01:
        out = np.ones(64, dtype=np.float64) / 64.0
    result = [float(x) for x in out]
    final_sc = evaluate_sequence(result)
    if not math.isfinite(final_sc):
        fallback = [float(x) for x in parent]
        fs2 = evaluate_sequence(fallback)
        if not math.isfinite(fs2):
            return [1.0]
        return fallback
    return result
