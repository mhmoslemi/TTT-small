    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if p.sum() < 1e-12:
        p = np.ones(3072)
    return p

def _conv(a):
    n = a.size
    L = 2*n - 1
    sz = next_fast_len(L)
    F = rfft(a, sz)
    return np.real(irfft(F*F, sz)[:L])

def _score(a):
    n = a.size
    s = a.sum()
    if s < 0.01:
        return math.inf
    return 2.0 * n * _conv(a).max() / (s*s)

def _lse_grad(a, T):
    """Compute LSE objective and its gradient."""
    n = a.size
    L = 2*n - 1
    c = _conv(a)
    cmax = c.max()
    scaled = (c - cmax) / T
    exps = np.exp(scaled)
    se = exps.sum()
    if se <= 0 or not math.isfinite(se):
        return cmax, np.zeros(n)
    obj = cmax + T * math.log(se)
    w = exps / se  # Softmax weights
    
    # Gradient: g_j = sum_k w_k * (a_{k-j} + a_{j-k})
    # f1_j = sum_k w_k * a_{k-j} = np.correlate(w, a, 'full')[:n]... 
    # Actually let me use FFT for both terms.
    # f2 = conv(a, w) [:n]
    # f1 = correlate(w, a) [n-1:n-1+n]... 
    
    # Let me just use direct FFT:
    sz2 = next_fast_len(L + n - 1)  # We need size >= L + n - 1 for full convolution
    # Actually for conv(a, w): length n + L - 1 = 3n - 2
    # For correlate(w, a): same length
    
    # f2 = full convolution of a and w: length n + L - 1 = 3n - 2
    sz_fw = next_fast_len(n + L - 1)
    Fa = rfft(a, sz_fw)
    Fw = rfft(w, sz_fw)
    f2_full = np.real(irfft(Fa * Fw, sz_fw))[:n]
    
    # f1_j = sum_m w[m+j] * a[m] for m=0..n-1
    # This is the cross-correlation of w and a at lag j
    # = (reverse(w) * a) reversed... 
    # Let me use: f1 = ifft(fft(w) * conj(fft(a_padded)))... 
    
    # Actually, the simplest: f1_j = sum_m w[m+j]*a[m]
    # = <w[j:j+n], a> (dot product)
    # For all j: this is a "sliding inner product"
    # f1 = [np.dot(w[j:j+n], a) for j in range(n)]
    # Vectorized via FFT:
    # f1 = ifft(fft(w_ext) * conj(fft(a_ext)))... 
    
    # Let me think about this differently.
    # f1_j = sum_{m=0}^{n-1} w[m+j] * a[m]
    # Define: x[p] = w[p] for p in [0, L-1], 0 otherwise
    #         y[m] = a[m] for m in [0, n-1], 0 otherwise
    # f1_j = sum_m x[m+j] * y[m]
    # This is the (j)-th element of the cross-correlation CTFT:
    # In DFT terms: if X = DFT(x), Y = DFT(y) (both zero-padded to size >= L+n-1)
    # Then: ifft(X * conj(Y))[j] = sum_n x[n] * y[n-j] (circular)
    #      ifft(conj(X) * Y)[j] = sum_n x[n-j] * y[n] (circular)
    #      Both are circular, so we need linear version.
    
    # What we want is: sum_m x[m+j] * y[m]
    # Let's define z[k] = x[k+j] (x shifted left by j)
    # f1_j = <z, y> = <x shifted left by j, y>
    # = DFT-based: ifft(Fft(x) * exp(-i*2*pi*j/N) * conj(Fft(y)))... 
    
    # This is getting too complicated. Let me just use the O(n^2) direct method
    # since n=3072 gives only ~10M operations.
    
    # Actually, I realize there's a much simpler way:
    # Note that g_j = sum_k w_k * (a_{k-j} + a_{j-k})
    # The term a_{k-j} + a_{j-k} is the "cross-correlation kernel" at lag |k-j|
    # (when a is symmetric, it's 2*a_{|k-j|})
    
    # But a is NOT necessarily symmetric. So we need to compute this properly.
    
    # Simplest correct approach:
    # For each j, g_j = sum_k w_k * a[k-j] (where a[k-j]=0 if k-j not in [0,n-1])
    #              + sum_k w_k * a[j-k] (where a[j-k]=0 if j-k not in [0,n-1])
    
    # First sum: for fixed j, the valid k range is [j, j+n-1] (intersected with [0, L-1])
    # = dot(w[j:min(j+n, L)], a[0:min(n, L-j)])
    # Second sum: for fixed j, valid k range is [max(0, j-n+1), min(j, L-1)]
    # = dot(w[max(0,j-n+1):min(j+1,L)], a[j-max(0,j-n+1):n])
    
    # This is still O(n^2) total if done naively. Let me just use the matrix approach:
    # Create an n x n matrix M where M[j, m] = w[m+j] (for the first sum)
    # Then f1 = M @ a
    # Similarly for the second sum.
    # Memory: n^2 * 8 = 75MB for n=3072. OK.
    # Time: n^2 = 10M multiply-adds. Fast.
    
    # But actually, even simpler: just use scipy.signal.fftconvolve
    
    # f2 = conv(a, w)[:n]  -- full convolution, take first n elements
    # f1: we need sum_m w[m+j]*a[m] for j=0..n-1
    #     = (w flipped and convolved with a flipped)[...]
    
    # You know what, let me just use the O(n^2) approach with numpy broadcasting:
    # f1_j = sum_m w[m+j]*a[m]
    # Create indices: for j in 0..n-1, m in 0..n-1: index into w is m+j
    # w_idx = np.arange(n)[:, None] + np.arange(n)[None, :]  -- shape (n, n), w_idx[j,m] = m+j
    # But we need w[m+j] to be valid: 0 <= m+j < L
    # Since m < n and j < n, m+j < 2n = L+1, so m+j can be up to 2n-2 = L-1. Valid!
    # So w_matrix = w[w_idx]  -- shape (n, n)
    # f1 = w_matrix.T @ a  -- f1[j] = sum_m w[m+j]*a[m]
    
    # Similarly: f2_j = sum_m a[m]*w[j-m] where j-m in [0, L-1]
    # w_idx2[j, m] = j - m, valid when 0 <= j-m < L, i.e., m <= j < m+L
    # Since m < n and j < n, j-m > -n and j-m < n < L. So valid when j >= m.
    # For j < m: w[j-m] = 0 (out of bounds)
    # So f2_j = sum_{m=0}^{min(j, n-1)} a[m]*w[j-m]
    # This is just (a * w)[j] = conv(a, w)[j] for j < n
    # Which is np.convolve(a, w)[:n]
    
    # OK let me just implement it cleanly:
    
    sz = next_fast_len(3*n - 2)
    Fa = rfft(a, sz)
    Fw = rfft(w, sz)
    # conv(a, w) full:
    conv_aw = np.real(irfft(Fa * Fw, sz))
    f2 = conv_aw[:n]  # f2[j] = sum_m a[m]*w[j-m]
    
    # For f1: we need the cross-correlation
    # f1[j] = sum_m w[m+j] * a[m]
    # = conv(reverse(w), reverse(a))[L-1-j]... let me verify
    # conv(rw, ra)[k] = sum_m rw[m] * ra[k-m] = sum_m w[L-1-m] * a[n-1-k+m]
    # For k = L-1-j: = sum_m w[L-1-m] * a[n-1-j+m]
    # Let p = L-1-m: = sum_p w[p] * a[n-1-j+L-1-p] = sum_p w[p] * a[2n-3-j-p]... 
    # This is getting messy. Let me just use the O(n^2) approach.
    
    # O(n^2) approach for f1:
    # w_idx[j, m] = m + j for j,m in [0, n-1]
    # All valid since 0 <= m+j <= 2n-2 = L-1
    w_idx = np.arange(n)[None, :] + np.arange(n)[:, None]  # w_idx[j,m] = m+j
    f1 = (w[w_idx]).T @ a  # f1[j] = sum_m w[m+j]*a[m]
    
    # Wait, w_idx has shape (n, n) with w_idx[j, m] = m + j
    # w[w_idx] has shape (n, n) with element [j, m] = w[m+j]
    # (w[w_idx]).T @ a: element j = sum_m w[w_idx[j,m]] * a[m] = sum_m w[m+j]*a[m] = f1[j] ✓
    
    g = f1 + f2
    
    # Rescale gradient by 1/T (since obj = cmax + T*log(sum_exp), and d(obj)/da = sum_k (exp(c_k-T)/T / sum_exp) * dc_k/da
    # Actually: d/d a [T * log(sum_k exp(c_k/T))]
    # = T * (1/sum_k exp(c_k/T)) * sum_k exp(c_k/T) * (1/T) * dc_k/da
    # = sum_k (exp(c_k/T)/sum_k exp(c_k/T)) * dc_k/da
    # = sum_k w_k * dc_k/da
    # So the gradient is already correctly scaled! No extra 1/T factor needed.
    
    return obj, g
