        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros(n)

        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        se = float(exps.sum())
        if not math.isfinite(se) or se <= 0:
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            w = exps / se
            ratio = se / conv_len
            f = cmax + T * math.log(max(ratio, 1e-15))

        q = xfast(p, w)
        gid = 2.0 * q
        gd = float(np.dot(p, gid))
        grad = p * (gid - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    u = p_to_u(a)
    best_p = a.copy()
    best_sc = _score(a)

    c = cfast(a)
    cmax = float(c.max())
    stages = [(0.008, 0.3), (0.002, 0.35), (0.0005, 0.35)]
    total_w = sum(w for _, w in stages)
    cum = 0.0
    avail = max(0.0, stop_time - time.monotonic() - 3.0)

    for frac, wt in stages:
        if time.monotonic() > stop_time - 3.0:
            break
        cum += wt
        end_stage = min(stop_time - 2.0, time.monotonic() + avail * wt / total_w)
        if end_stage <= time.monotonic() + 5.0:
            continue

        p_curr = best_p
        c_curr = cfast(p_curr)
        T = max(1e-14, frac * float(c_curr.max()))
        u = p_to_u(p_curr)

        while time.monotonic() < end_stage - 1.0:
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B",
                               jac=True, options={"maxiter": 20, "gtol": 1e-9,
                                                  "ftol": 1e-14, "maxls": 12,
                                                  "maxcor": 12, "disp": False})
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    m2 = float(np.max(x))
                    e2 = np.exp(x - m2)
                    s2 = float(e2.sum())
                    if s2 > 0 and math.isfinite(s2):
                        p_new = e2 / s2
                        p_new = np.maximum(p_new, 0.0)
                        sc_new = _score(p_new)
                        if math.isfinite(sc_new) and sc_new < best_sc - 1e-12:
                            best_sc = sc_new
                            best_p = p_new
                            c_curr = cfast(p_new)
                            T = max(1e-14, frac * float(c_curr.max()))
                            u = p_to_u(p_new)
                    else:
                        break
            if getattr(res, "status", 1) == 0:
                break

    return best_p


def _shaped_perturbations(base, n, rng):
    """Generate a few shaped perturbation candidates."""
    cands = []
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)

    # Power shapes
    for q in (0.6, 0.8, 1.3, 1.7, 2.2):
        cand = np.power(np.maximum(base, 0), q)
        s = float(cand.sum())
        if s > 0:
            cands.append(cand / s)

    # Exponential tilts
    for amp in (0.5, 1.0, 2.0):
        for side in (1, -1):
            cand = base * np.exp(side * amp * (2 * x - 1))
            cand = np.maximum(cand, 0)
            s = float(cand.sum())
            if s > 0:
                cands.append(cand / s)

    # Gaussian bump (asymmetric)
    for center in (0.25, 0.5, 0.75):
        for width in (0.1, 0.2, 0.4):
            cand = np.exp(-((x - center) / width) ** 2)
            s = float(cand.sum())
            if s > 0:
                cands.append(cand / s)

    # Sine-modulated
    for f in (1, 2, 3):
        for amp in (0.3, 0.7):
            cand = base * (1.0 + amp * np.sin(2 * math.pi * f * x))
            cand = np.maximum(cand, 0)
            s = float(cand.sum())
            if s > 0:
                cands.append(cand / s)

    return cands


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(20.0, budget)
    reserve = 15.0 if budget > 200 else 5.0
    hard = t0 + budget - reserve
    if hard <= t0 + 2:
        hard = t0 + max(1.0, budget - 0.5)

    parent = _get_parent()
    n = int(parent.size)
    rng = np.random.default_rng(seed)

    # Initialize: parent
    best_a = parent.copy()
    best_score = _score(best_a)

    def consider(a):
        nonlocal best_a, best_score
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size != n:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s < 0.01:
            return
        a = a / s
        sc = _score(a)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

    # Phase 1: Try shaped perturbations (quick, ~10s)
    p1_stop = min(hard - 60, t0 + max(5, 0.04 * budget))
    cands = _shaped_perturbations(parent, n, rng)
    for cand in cands:
        if time.monotonic() > p1_stop:
            break
        consider(cand)

    # Phase 2: L-BFGS refinement on best candidate (~20-30% of time)
    p2_start = time.monotonic()
    p2_stop = min(hard - 40, p2_start + max(10, 0.22 * budget))
    if p2_stop > p2_start + 8:
        try:
            a_lb = _lbfgs_phase(best_a.copy(), p2_stop, seed=seed)
            consider(a_lb)
        except Exception:
            pass

    # Phase 3: Coordinate descent (main workhorse, ~55-65% of time)
    p3_stop = min(hard - 20, time.monotonic() + max(10, 0.55 * budget))
    try:
        a_cd, sc_cd = _coord_descent(best_a.copy(), p3_stop, seed=seed + 100)
        consider(a_cd)
    except Exception:
        pass

    # Phase 4: If time remains, multiple restarts with perturbations
    remaining = max(0.0, hard - 5.0 - time.monotonic())
    if remaining > 30:
        n_rounds = int(remaining / 40)
        n_rounds = max(1, min(n_rounds, 8))
        round_budget = remaining / n_rounds
        for r in range(n_rounds):
            if time.monotonic() > hard - 5:
                break
            sig = 0.04 + 0.02 * r
            noise = rng.normal(0, sig, n)
            a_new = best_a * np.exp(noise)
            a_new = np.maximum(a_new, 0)
            s = float(a_new.sum())
            if s > 0:
                a_new /= s
                consider(a_new)
            # Coordinate descent from this point
            rd_stop = min(hard - 5, time.monotonic() + round_budget - 3)
            if rd_stop > time.monotonic() + 5:
                try:
                    a_rd, _ = _coord_descent(best_a.copy(), rd_stop, seed=seed + 1000 + r * 7)
                    consider(a_rd)
                except Exception:
                    pass

    # Final validation
    try:
        out = np.asarray(best_a, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = _get_parent()
        out = out / float(out.sum())
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            return [1.0]
        return ls
    except Exception:
        return [1.0]
