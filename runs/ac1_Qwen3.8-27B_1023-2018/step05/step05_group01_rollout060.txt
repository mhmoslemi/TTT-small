    if s < 0.01:
        return math.inf
    c = np.convolve(a, a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _coord_descent(p, stop_time, seed=42, G=64, consider=None):
    n = int(p.size)
    if n == 0:
        return p, math.inf
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p /= s

    c = _conv_full(p)
    Ltot = 2 * n - 1
    cur_score = 2.0 * n * float(c.max())
    best_p = p.copy()
    best_score = cur_score
    rng = np.random.default_rng(seed)
    flat = 0
    total_w = 1.2

    for sweep in range(200000):
        if time.monotonic() > stop_time - 1.5:
            break
        # Rescale if drifted
        S = float(p.sum())
        if S > 1.5 or S < 0.67:
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0

        improved = False
        order = rng.permutation(n)
        sweep_start = time.monotonic()

        for i in order:
            if time.monotonic() > stop_time - 0.8:
                break
            if S <= 1e-300:
                break

            v = float(p[i])
            k_start = i
            k_end = min(i + n, Ltot)
            if k_end <= k_start:
                continue
            n_aff = k_end - k_start

            base = c[k_start:k_end]
            lin = p[:n_aff] + v
            has_quad = (i < n_aff)
            quad_idx = i

            unchanged_max = 0.0
            if k_start > 0:
                unchanged_max = float(c[:k_start].max())
            if k_end < Ltot and unchanged_max < float(c[k_end:].max()):
                unchanged_max = float(c[k_end:].max())

            # Quick check: x=0 and x=2v
            def quick_eval(x):
                d = x - v
                ns = S + d
                if ns < 1e-15:
                    return math.inf
                na = base + d * lin
                if has_quad:
                    na[quad_idx] += d * d
                na = np.maximum(na, unchanged_max)
                return 2.0 * n * float(na.max()) / (ns * ns)

            sc0 = quick_eval(0.0)
            sc2v = quick_eval(min(2.0 * v, 4.0 / n))
            if sc0 >= cur_score - 1e-14 and sc2v >= cur_score - 1e-14 and v > 0:
                continue

            # Full grid
            x_max = max(20.0 * v, 40.0 / float(n), 0.01 * S)
            x_max = min(x_max, 8.0 * S, 4.0)
            if x_max < 1e-12:
                x_max = max(1e-8, S / float(n))

            pts = np.linspace(0.0, x_max, G)
            if 0.0 < v < x_max:
                pts = np.unique(np.concatenate([pts, [v]]))

            ds = pts - v
            aff = base[np.newaxis, :] + ds[:, np.newaxis] * lin[np.newaxis, :]
            if has_quad:
                aff[:, quad_idx] += ds ** 2
            aff = np.maximum(aff, unchanged_max)
            aff_max = aff.max(axis=1)
            nss = S + ds
            valid = nss > 1e-15
            scs = np.full(pts.shape, math.inf)
            scs[valid] = 2.0 * n * aff_max[valid] / (nss[valid] ** 2)

            if not np.any(np.isfinite(scs)):
                continue
            imin = int(np.argmin(scs))
            bx = float(pts[imin])
            bsc = float(scs[imin])

            # Refinement 1
            if 0.0 < bx < x_max:
                dw = x_max / G
                lo = max(0.0, bx - 2.0 * dw)
                hi = min(x_max, bx + 2.0 * dw)
                fine = np.linspace(lo, hi, 40)
                fine_ds = fine - v
                fa = base[np.newaxis, :] + fine_ds[:, np.newaxis] * lin[np.newaxis, :]
                if has_quad:
                    fa[:, quad_idx] += fine_ds ** 2
                fa = np.maximum(fa, unchanged_max)
                fm = fa.max(axis=1)
                fns = S + fine_ds
                fv = fns > 1e-15
                fsc = np.full(fine.shape, math.inf)
                fsc[fv] = 2.0 * n * fm[fv] / (fns[fv] ** 2)
                if np.any(np.isfinite(fsc)):
                    fi = int(np.argmin(fsc))
                    if fsc[fi] < bsc - 1e-16:
                        bx = float(fine[fi])
                        bsc = float(fsc[fi])

            # Refinement 2 (very fine)
            if 0.0 < bx < x_max and bsc < cur_score:
                dw2 = x_max / (G * 4)
                lo2 = max(0.0, bx - dw2)
                hi2 = min(x_max, bx + dw2)
                fine2 = np.linspace(lo2, hi2, 24)
                f2_ds = fine2 - v
                f2a = base[np.newaxis, :] + f2_ds[:, np.newaxis] * lin[np.newaxis, :]
                if has_quad:
                    f2a[:, quad_idx] += f2_ds ** 2
                f2a = np.maximum(f2a, unchanged_max)
                f2m = f2a.max(axis=1)
                f2ns = S + f2_ds
                f2v = f2ns > 1e-15
                f2sc = np.full(fine2.shape, math.inf)
                f2sc[f2v] = 2.0 * n * f2m[f2v] / (f2ns[f2v] ** 2)
                if np.any(np.isfinite(f2sc)):
                    f2i = int(np.argmin(f2sc))
                    if f2sc[f2i] < bsc - 1e-16:
                        bx = float(fine2[f2i])
                        bsc = float(f2sc[f2i])

            if math.isfinite(bsc) and bsc < cur_score - 1e-14 and 0.0 <= bx <= 10.0 * S:
                d = bx - v
                p[i] = bx
                S += d
                c[k_start:k_end] += d * lin
                if has_quad:
                    c[k_start + quad_idx] += d * d
                cur_score = bsc
                improved = True

        S = float(p.sum())
        if S > 1e-300:
            p /= S
            c /= (S * S)
        S = 1.0
        c = _conv_full(p)
        cur_score = 2.0 * n * float(c.max())

        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 25:
                if time.monotonic() < stop_time - 10.0:
                    sigma = 0.015 + 0.005 * min(flat // 50, 8)
                    p *= np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    else:
                        break
                    c = _conv_full(p)
                    cur_score = 2.0 * n * float(c.max())
                    S = 1.0
                    flat = 0
                else:
                    break

    return best_p, best_score


def _lse_optimize(p, stop_time, seed=42, consider=None):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p /= s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def softmax_fn(uu):
        m = float(np.max(uu))
        e = np.exp(uu - m)
        ssum = float(e.sum())
        if ssum <= 0 or not math.isfinite(ssum):
            return np.full(uu.size, 1.0 / uu.size)
        return e / ssum

    def lse_func(uu, T):
        pp = softmax_fn(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    stages = (
        (0.10, 0.25),
        (0.03, 0.25),
        (0.008, 0.25),
        (0.002, 0.25),
    )

    start = time.monotonic()
    avail = max(0.0, float(stop_time) - start - 5.0)
    if avail <= 10.0:
        return p

    rng = np.random.default_rng(seed)
    best_u = u.copy()
    total_w = sum(w for _, w in stages)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        cum += wt
        end_stage = min(float(stop_time) - 2.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)
        f0, _ = lse_func(best_u, T)
        if not math.isfinite(f0):
            continue
        best_f = f0
        stale = 0

        while time.monotonic() < end_stage - 0.5:
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True,
                               options={"maxiter": 30, "gtol": 1e-12, "ftol": 1e-16,
                                        "maxls": 20, "maxcor": 20, "maxdisp": 0, "disp": False})
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    fval = float(res.fun)
                    if math.isfinite(fval) and fval < best_f - 1e-15:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            best_u += rng.normal(0, 0.003, best_u.size)
                            continue
            if getattr(res, "status", 1) == 0 and stale < 3:
                if time.monotonic() < end_stage - 1.0:
                    best_u += rng.normal(0, 0.002, best_u.size)
                else:
                    break

        p_new = softmax_fn(best_u)
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if ss > 1e-300:
            p_new /= ss
        p = p_new.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(p)
            except Exception:
                pass

    return p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 20.0
    final_reserve = max(8.0, 0.05 * (hard - t0))
    phase_limit = hard - final_reserve

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    try:
        best_score = float(_score_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = float(_score_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    n_parent = int(best_arr.size)

    # Phase 1: Coordinate descent (main effort)
    p1_end = min(phase_limit, t0 + 0.60 * (phase_limit - t0))
    if p1_end > t0 + 20.0:
        try:
            p1, sc1 = _coord_descent(best_arr, p1_end, seed=int(seed + 100), G=64, consider=consider)
            if p1 is not None and p1.size == n_parent:
                best_arr = p1 / float(p1.sum())
                consider(best_arr)
        except Exception:
            pass

    # Phase 2: LSE optimization
    rem2 = phase_limit - time.monotonic()
    if rem2 > 40.0:
        lse_end = min(phase_limit - 10.0, time.monotonic() + 0.5 * rem2)
        if lse_end > time.monotonic() + 20.0:
            try:
                p2 = _lse_optimize(best_arr, lse_end, seed=int(seed + 2000), consider=consider)
                if p2 is not None and p2.size == n_parent and float(p2.sum()) > 0:
                    p2 = p2 / float(p2.sum())
                    consider(p2)
            except Exception:
                pass

    # Phase 3: More coordinate descent
    rem3 = phase_limit - time.monotonic()
    if rem3 > 25.0:
        p3_end = min(phase_limit - 5.0, time.monotonic() + rem3 - 3.0)
        if p3_end > time.monotonic() + 15.0:
            try:
                p3, sc3 = _coord_descent(best_arr, p3_end, seed=int(seed + 3000), G=80, consider=consider)
                if p3 is not None and p3.size == n_parent:
                    best_arr = p3 / float(p3.sum())
                    consider(best_arr)
            except Exception:
                pass

    # Phase 4: Try a perturbation + quick CD
    rem4 = phase_limit - time.monotonic()
    if rem4 > 30.0:
        rng4 = np.random.default_rng(seed + 9999)
        pert = best_arr * np.exp(rng4.normal(0, 0.03, best_arr.size))
        pert = np.maximum(pert, 0.0)
        ps = float(pert.sum())
        if ps > 0:
            pert /= ps
            p4_end = min(phase_limit - 3.0, time.monotonic() + 0.5 * rem4)
            if p4_end > time.monotonic() + 15.0:
                try:
                    p4, _ = _coord_descent(pert, p4_end, seed=int(seed + 7777), G=64, consider=consider)
                except Exception:
                    pass

    # Final LSE squeeze
    rem5 = hard - time.monotonic()
    if rem5 > 30.0 and best_arr.size > 0:
        lstop = min(hard - 10.0, time.monotonic() + 0.4 * rem5)
        if lstop > time.monotonic() + 10.0:
            try:
                p5 = _lse_optimize(best_arr, lstop, seed=int(seed + 12345), consider=consider)
                if p5 is not None and p5.size == int(best_arr.size) and float(p5.sum()) > 0:
                    p5 = p5 / float(p5.sum())
                    consider(p5)
            except Exception:
                pass

    # Final CD squeeze
    rem6 = hard - time.monotonic()
    if rem6 > 20.0 and best_arr.size > 0:
        cstop = min(hard - 5.0, time.monotonic() + max(10.0, 0.5 * rem6))
        if cstop > time.monotonic() + 10.0:
            try:
                p6, _ = _coord_descent(best_arr, cstop, seed=int(seed + 88888), G=80, consider=consider)
                if p6 is not None and p6.size == int(best_arr.size):
                    p6 = p6 / float(p6.sum())
                    consider(p6)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
