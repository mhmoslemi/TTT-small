import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize

def _fft_conv(a):
    """Compute a*a (self-convolution) via FFT."""
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2*n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F*F, size)[:L]
    return np.maximum(c, 0.0)

def _score_ff(a):
    """Fast score using FFT convolution."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = a.sum()
    if s < 0.01:
        return math.inf
    c = _fft_conv(a)
    m = c.max()
    return float(2.0*n*m/(s*s))

def _lse_obj_grad(u, T, conv_size, conv_len, cross_size, cross_len):
    """LSE objective and gradient."""
    # softmax
    u_max = u.max()
    e = np.exp(u - u_max)
    s = e.sum()
    p = e / s
    
    # conv
    F = rfft(p, conv_size)
    c = irfft(F*F, conv_size)[:conv_len]
    c = np.maximum(c, 0.0)
    cmax = c.max()
    
    if T < 1e-10:
        # Hard max
        score = cmax
        w = np.zeros_like(c)
        w[c == cmax] = 1.0 / (c == cmax).sum()
    else:
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = exps.sum()
        score = cmax + T * math.log(sum_exp / c.size)
        w = exps / sum_exp
    
    # Cross-correlation for gradient
    Fp = rfft(p[::-1], cross_size)
    Fw = rfft(w, cross_size)
    cc = irfft(Fp * Fw, cross_size)[:cross_len]
    q = cc[conv_len-1 : 2*conv_len-2]  # positions n-1 to 2n-2
    q = np.maximum(q, 0.0)
    
    # Gradient in p-space: 2*q - p*sum(2*q)
    # In u-space: p * (2*q - p*(2*q).sum() / p)... 
    # Actually: df/du[i] = p[i] * (df/dp[i] - p·(df/dp))
    # df/dp[i] = 2*q[i]
    
    dp = 2.0 * q
    # Project to sum-zero (since p changes on the simplex)
    # Actually with softmax parameterization, the gradient is:
    # grad_u[i] = p[i] * (dp[i] - p @ dp)
    pdp = p @ dp
    grad = p * (dp - pdp)
    
    # But wait: the objective is 2n * LSE / s^2 where s = sum(p) = 1 (since softmax)
    # So we just need to minimize LSE (with sum(p) = 1 constraint handled by softmax)
    # Score = 2n * LSE / 1^2 = 2n * LSE
    # So minimizing LSE is equivalent to minimizing score
    
    return float(score), grad

    n = p.size

    p = np.maximum(p, 0)

    if S < 1e-300:
        return p

    cur_score = 2.0 * n * c.max()  # since S=1

    max_flat = 6

        improved = False
        # Random order
        order = rng.permutation(n)

                return p
            i = int(idx)

            start = i

            Ls = end - start

            base = c[start:end].copy()
            # Influence: 2 * p[0:Ls]

            # Diagonal position (k=2i in full conv -> position 2i-start in slice)
            diag_pos = 2*i - start  # = i (since start=i)

            max_const = 0.0
            if start > 0:

            # New score: 2n * max(max_const, max_k(new_conv[k])) / (S+d)^2
            # S = 1 (normalized)

            candidates = [0.0]
            
            # Identify top entries in base
            top_k = min(10, Ls)
            top_indices = np.argsort(base)[-top_k:]

                    ina = infl[ia]

                    bb = base[ib]
                    if abs(ina - inb) > 1e-15:

                if abs(ina) > 1e-15 and max_const > ba:

                    if -v*0.9 < d_mc < v*2.0:
                        candidates.append(d_mc)

            for frac in [0.5, 0.7, 1.0, -0.3, -0.5, -0.7]:
                candidates.append(v * frac)

            best_d = 0.0
            best_f = cur_score

                if nv < 0 or nv > 10.0:  # reasonable bounds

                if S + d < 0.1:  # sum must stay positive
                    continue

                new_vals = base + d * infl
                if diag_pos < Ls:

                f = 2.0 * n * new_max / ((S + d) * (S + d))

                    best_d = d

                    if f < best_f - 1e-12:
                        best_f = f
                        best_d = dd

            if abs(best_d) > 1e-14:
                p[i] = v + best_d
                S += best_d
                # Update conv
                c[start:end] += best_d * infl
                if diag_pos < Ls:
                    c[start + diag_pos] += best_d * best_d
                cur_score = best_f
                improved = True
        
        # Rescale
        if S > 1e-300:

            p *= inv
            c *= inv * inv
            S = 1.0
        
        # Verify periodically

            flat_count = 0
        else:
            flat_count += 1
            if flat_count >= max_flat:
                # Perturb

                    p = p * np.exp(noise)

                    c = _fft_conv(p)
                    cur_score = 2.0 * n * c.max()
                    flat_count = 0
                else:
                    break
    
    # Final rescale
    S = p.sum()
    if S > 1e-300:
        p /= S
    return p
