                v1 = float(np.convolve(c1/s1, c1/s1).max()) if s1 > 1e-12 else float('inf')
                c2 = np.maximum(a + m2 * d, 0.0)
                s2 = c2.sum()
                v2 = float(np.convolve(c2/s2, c2/s2).max()) if s2 > 1e-12 else float('inf')
                if v1 < best_v:
                    best_v, best_t = v1, m1
                if v2 < best_v:
                    best_v, best_t = v2, m2
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
        cand = np.maximum(a + best_t * d, 0.0)
        s = cand.sum()
        if s < 1e-12:
            return a.copy(), best_v
        return cand / s, best_v

    def optimize(a0, dl, seed=42):
        a = np.clip(np.asarray(a0, dtype=np.float64), 0, 1000).copy()

        a /= a.sum()

        rng = np.random.default_rng(seed)
        stuck = 0

                sigma = min(0.03 + 0.02 * (stuck // 100), 0.6)
                mode = (stuck // 30) % 4
                if mode == 0:
                    a *= np.exp(rng.normal(0, sigma, nn))
                elif mode == 1:
                    xx = np.arange(nn, dtype=np.float64)
                    f = rng.uniform(1, 6)
                    a *= (1 + sigma * np.sin(2*np.pi*f*xx/nn + rng.uniform(0, 6.28)))
                    a = np.maximum(a, 1e-10)
                elif mode == 2:

                    s_ = rng.integers(0, nn - blk)
                    tmp = a[s_:s_+blk].copy()
                    a[s_:s_+blk] = a[s_+blk:s_+2*blk] if s_ + 2*blk <= nn else tmp
                    a = np.maximum(a, 1e-10)
                else:
                    a *= (1 + sigma * rng.uniform(-1, 1, nn))
                    a = np.maximum(a, 1e-10)
                a /= a.sum()
                stuck = 0

                    break

            if g is not None and g.sum() > 1e-8:
                g /= g.sum()
                d = g - a
                if float(np.max(np.abs(d))) > 1e-10:
                    cand, val = line_search(a, d)
                    if val < M * (1 - 1e-11):
                        a = cand
                        moved = True
                        stuck = 0
            
            # 2: Relaxed LP
            if not moved and g is not None:
                g2 = do_lp(A, M * (1 + 5e-4), tl=2.0)
                if g2 is not None and g2.sum() > 1e-8:
                    g2 /= g2.sum()
                    d = g2 - a
                    if float(np.max(np.abs(d))) > 1e-10:
                        cand, val = line_search(a, d)
                        if val < M * (1 - 1e-11):
                            a = cand
                            moved = True
                            stuck = 0
            
            # 3: Subgradient step
            if not moved:
                active = np.where(c_full >= M * 0.9999)[0]
                grad = np.zeros(nn)
                for ks in active:
                    ks_i = int(ks)
                    lo = max(0, ks_i - nn + 1)
                    hi = min(nn, ks_i + 1)
                    for jj in range(lo, hi):
                        idx = ks_i - jj
                        if 0 <= idx < nn:
                            grad[jj] += 2.0 * a[idx]
                gn = float(np.linalg.norm(grad))
                if gn > 1e-14:
                    for lr in [0.3, 0.1, 0.05, 0.02, 0.01, 0.005]:
                        cand = np.maximum(a - lr * grad / gn, 0.0)
                        cs = cand.sum()
                        if cs < 1e-12:
                            continue
                        cand /= cs
                        v = float(np.convolve(cand, cand).max())
                        if v < M * (1 - 1e-11):
                            a = cand
                            moved = True
                            stuck = 0
                            break
            
            # 4: Small random kick
            if not moved:
                a *= np.exp(rng.normal(0, 0.015, nn))
                a = np.maximum(a, 1e-10)
                a /= a.sum()
        
        return a
