            try:
                from scipy.optimize import minimize
                res = minimize(lse_func, best_u, args=(T,), method='L-BFGS-B',
                              jac=True, options={'maxiter': 30, 'gtol': 1e-10,
                              'ftol': 1e-14, 'maxls': 20, 'maxcor': 15,
                              'maxdisp': 0, 'disp': False})
            except Exception:
                break
            
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    fval = float(res.fun)
                    if math.isfinite(fval) and fval < best_f - 1e-15:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            best_u = best_u + rng.normal(0, 0.003, best_u.size)
                            continue
            
            if time.monotonic() >= end_stage - 1.0:
                break
            # Small perturbation to escape local minima
            best_u = best_u + rng.normal(0, 0.002, best_u.size)
            f_chk, _ = lse_func(best_u, T)
            if math.isfinite(f_chk) and f_chk < best_f:
                best_f = f_chk
                stale = 0
        
        p_cur = _softmax(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif math.isfinite(sc_cur) and sc_cur < best_sc + 0.002:
            p = p_cur.copy()
        else:
            p = best_p.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass
    
    return best_p

def _coord_descent(p, stop_time, consider=None, seed=12345):
    """Fine coordinate descent with adaptive grid."""
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p/s
    
    Ltot = 2*n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0*n*c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    
    # Adaptive grid size
    G_coarse = 32
    G_fine = 32
    
    while sweep < 50000 and time.monotonic() < stop_time - 0.5:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0/S_total
                p *= inv
                c *= inv*inv
                S_total = 1.0
            
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i+n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0*p[:Ls]  # influence of p[i] on b[start:end]
            
            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr
            
            # Search range
            x_hi = max(50.0*v, 80.0*S/float(n), 0.01*S)
            x_hi = min(x_hi, 10.0*S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, S/float(n))
            
            # Coarse grid
            pts = np.linspace(0.0, x_hi, G_coarse+1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.unique(np.concatenate((pts, [v])))
            
            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[valid, None]*infl[None, :]
                diag_i = i
                if start <= diag_i < end:
                    dval = base[diag_i - start] + w[valid]*infl[diag_i - start] + w[valid]*w[valid]
                    aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                dv = denom[valid]
                vals[valid] = (2.0*n*mv)/(dv*dv)
                return vals
            
            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])
            
            # Fine search around best
            if best_x > 0 and best_x < x_hi:
                dw = x_hi/float(G_coarse)
                lo = max(0.0, best_x - 3.0*dw)
                hi = min(x_hi, best_x + 3.0*dw)
                fine = np.linspace(lo, hi, G_fine, dtype=np.float64)
                fine = np.unique(np.concatenate((fine, [best_x])))
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val - 1e-15:
                        best_x = vx2
                        best_val = vv2
            
            if math.isfinite(best_val) and best_val < cur_score - 1e-12 and 0.0 <= best_x <= 10.0*S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w*infl
                    di = 2*i
                    if 0 <= di < Ltot:
                        c[di] += w*w
                    cur_score = best_val
                    improved = True
        
        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0/S
            p *= inv
            c *= inv*inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        new_score = float(2.0*n*c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass
        
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 6.0:
                    sigma = 0.015 + 0.015*min(flat, 15)/15.0
                    p = p*np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    else:
                        break
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0*n*c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_p = p.copy()
                    flat = 0
                else:
                    break
        sweep += 1
    
    ss = float(p.sum())
    if ss > 1e-300:
        p = p/ss
    try:
        sc = float(_score(p))
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score

def _subgrad_step(p, alpha, stop_time):
    """One subgradient step: a_i -= alpha * partial(max_b)/partial(a_i) / s_i"""
    n = p.size
    c = _conv_self(p)
    cmax = c.max()
    # Subgradient of max: uniform over active set
    active = np.where(c >= cmax - 1e-10*cmax)[0]
    w = np.zeros(c.size)
    w[active] = 1.0/len(active)
    # Gradient: 2 * (a * w_middle) where w_middle is the middle n elements
    # Actually, d/d(a[i]) max(b) = sum_k w[k] * db[k]/da[i]
    # db[k]/da[i] = a[k-i] + a[k-i] (with appropriate bounds)
    # This is the cross-correlation of a with w (reversed)
    # Use FFT
    size = next_fast_len(3*n-2)
    F = rfft(p[::-1], size)
    G = rfft(w, size)
    c2 = irfft(F*G, size)[:3*n-2]
    q = c2[n-1:2*n-1]
    p_new = p - alpha*q
    p_new = np.maximum(p_new, 0.0)
    s = float(p_new.sum())
    if s > 1e-300:
        p_new /= s
    return p_new
