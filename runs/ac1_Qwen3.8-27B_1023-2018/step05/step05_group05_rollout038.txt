def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)

        p = np.ones(384, dtype=np.float64)

def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(np.asarray(c, dtype=np.float64), 0.0)

def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))

                actual_sc = _score_unit(p_norm)
                if math.isfinite(actual_sc) and actual_sc < best_score - 1e-15:
                    best_score = actual_sc
                    best_p = p_norm.copy()
                    if consider is not None:
                        try:
                            consider(p_norm)
                        except Exception:
                            pass
            # Renormalize to prevent drift
            s_actual = float(p.sum())
            if s_actual > 1e-10 and abs(s_actual - 1.0) > 1e-5:
                p /= s_actual
                S = 1.0
                c = _conv_fast(p)
                cur_score = 2.0 * n * float(c.max())

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 25.0:
                    noise_scale = 0.04 if flat < 8 else 0.08
                    p = p * np.exp(rng.normal(0, noise_scale, n))
                    p = np.maximum(p, 0.0)
                    s = float(p.sum())
                    if s > 1e-300:
                        p /= s
                    else:
                        p = np.ones(n) / n
                        s = 1.0
                    S = 1.0
                    c = _conv_fast(p)
                    cur_score = 2.0 * n * float(c.max())
                    flat = 0
                else:
                    break

        sweep += 1

    s = float(best_p.sum())
    if s > 1e-10:
        best_p = best_p / s
    return best_p, best_score

def _lp_direction(f, stop_time):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0. Returns g normalized or None."""
    from scipy.sparse import csr_matrix
    from scipy.optimize import linprog

    n = f.size
    M = float(_conv_fast(f).max())
    if M <= 0:
        return None

    # Build sparse constraint matrix A (2n-1 x n)
    # A[k, j] = f[k-j] if 0 <= k-j < n
    row_parts = []
    col_parts = []
    val_parts = []
    for j in range(n):
        nr = min(n, 2 * n - 1 - j)
        if nr > 0:
            row_parts.append(np.arange(j, j + nr))
            col_parts.append(np.full(nr, j, dtype=np.int32))
            val_parts.append(f[:nr])

    if time.monotonic() > stop_time - 5.0:
        return None

    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    c_obj = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs',
                      options={'time_limit': max(5, stop_time - time.monotonic() - 5)})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = float(g.sum())
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

def _line_search(f, g, stop_time):
    """Find best t in [0,1] for f_t = (1-t)*f + t*g (both unit sum)."""
    best_t = 0.0
    best_sc = _score_unit(f)
    # Golden section
    a, b = 0.0, 1.0
    gr = (math.sqrt(5) - 1) / 2
    steps = 0
    while (b - a) > 1e-6 and time.monotonic() < stop_time - 3.0 and steps < 60:
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        fc1 = _score_unit((1 - c1) * f + c1 * g)
        fc2 = _score_unit((1 - c2) * f + c2 * g)
        if fc1 < fc2:
            b = c2
        else:
            a = c1
        steps += 1
    t_mid = (a + b) / 2
    sc_mid = _score_unit((1 - t_mid) * f + t_mid * g)
    if sc_mid < best_sc:
        best_sc = sc_mid
        best_t = t_mid
    return (1 - best_t) * f + best_t * g, best_sc

def _resample_length(src, n_target):
    """Resample a unit-sum sequence to a new length, returning unit-sum result."""
    m = src.size
    if m == n_target:
        return src.copy()
    if m == 0:
        return np.ones(n_target) / n_target
    xp = (np.arange(m) + 0.5) / m
    fp = (np.arange(n_target) + 0.5) / n_target
    out = np.interp(fp, xp, src)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.ones(n_target) / n_target
    return out / s

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_parent = parent.size

    best_p = np.maximum(parent.astype(np.float64), 0.0)
    bs = float(best_p.sum())
    if bs < 1e-300:
        best_p = np.ones(64) / 64
        bs = 1.0
    best_p = best_p / bs
    best_score = _score_unit(best_p)

    def consider(arr):
        nonlocal best_p, best_score
        a = np.asarray(arr, dtype=np.float64).ravel()
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        an = a / s
        sc = _score_unit(an)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = an.copy()

    consider(best_p)

    # Phase 1: Coordinate descent on parent
    cd1_stop = t0 + min(250.0, 0.22 * budget)
    if cd1_stop > time.monotonic() + 30.0:
        p1, _ = _coord_descent(best_p, cd1_stop, seed=int(seed), consider=consider)
        if p1.size > 0:
            consider(p1)

    # Phase 2: LP direction
    if time.monotonic() < hard_deadline - 200.0:
        lp_stop = time.monotonic() + min(40.0, 0.04 * budget)
        g = _lp_direction(best_p, lp_stop)
        if g is not None and g.size == best_p.size:
            new_p, _ = _line_search(best_p, g, time.monotonic() + 10.0)
            consider(new_p)

    # Phase 3: More coordinate descent
    cd2_stop = min(hard_deadline - 150.0, time.monotonic() + 300.0)
    if cd2_stop > time.monotonic() + 30.0:
        p3, _ = _coord_descent(best_p, cd2_stop, seed=int(seed + 100), consider=consider)
        if p3.size > 0:
            consider(p3)

    # Phase 4: LP direction again
    if time.monotonic() < hard_deadline - 100.0:
        lp_stop2 = time.monotonic() + min(40.0, 0.04 * budget)
        g2 = _lp_direction(best_p, lp_stop2)
        if g2 is not None and g2.size == best_p.size:
            new_p2, _ = _line_search(best_p, g2, time.monotonic() + 10.0)
            consider(new_p2)

    # Phase 5: Try alternative lengths
    remaining = hard_deadline - time.monotonic()
    if remaining > 120.0:
        # Try n = 2048
        try:
            alt_n = 2048
            p_alt = _resample_length(best_p, alt_n)
            alt_stop = time.monotonic() + min(80.0, 0.25 * remaining)
            p_alt_opt, _ = _coord_descent(p_alt, alt_stop, seed=int(seed + 200), consider=consider)
            if p_alt_opt.size > 0:
                consider(p_alt_opt)
        except Exception:
            pass

        remaining = hard_deadline - time.monotonic()
        if remaining > 80.0:
            # Try n = 4096
            try:
                alt_n2 = 4096
                p_alt2 = _resample_length(best_p, alt_n2)
                alt_stop2 = time.monotonic() + min(70.0, 0.4 * remaining)
                p_alt2_opt, _ = _coord_descent(p_alt2, alt_stop2, seed=int(seed + 300), consider=consider)
                if p_alt2_opt.size > 0:
                    consider(p_alt2_opt)
            except Exception:
                pass

    # Phase 6: Final polish on best
    remaining = hard_deadline - time.monotonic()
    if remaining > 40.0:
        polish_stop = hard_deadline - 10.0
        p_fin, _ = _coord_descent(best_p, polish_stop, seed=int(seed + 500), consider=consider)
        if p_fin.size > 0:
            consider(p_fin)

    # Final validation with exact evaluator
    try:
        out = np.asarray(best_p, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            # Fallback to parent
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
