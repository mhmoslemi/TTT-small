        if T <= 0.0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    def run_chunk(uu, T, end_stage):
        try:
            f0, _ = lse_func(uu, T)
        except Exception:
            return uu
        if not math.isfinite(f0):
            return uu
        best_u = np.array(uu, dtype=np.float64, copy=True)
        best_f = float(f0)
        stale = 0

        while time.monotonic() < end_stage - 1.0:
            maxit = 80 if n > 2000 else 120
            opts = {
                "maxiter": maxit,
                "gtol": 1e-14,
                "ftol": 1e-18,
                "maxls": 30,
                "maxcor": 50,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(
                    lse_func, best_u, args=(T,),
                    method="L-BFGS-B", jac=True, options=opts,
                )
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == uu.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 4:
                            break
            if getattr(res, "status", 1) == 0:
                break
        return best_u

    if fine_only:
        stages = (
            (0.003, 0.20),
            (0.001, 0.25),
            (0.0003, 0.25),
            (0.00008, 0.30),
        )
    else:
        stages = (
            (0.12, 0.08),
            (0.04, 0.10),
            (0.01, 0.12),
            (0.003, 0.14),
            (0.0008, 0.16),
            (0.0002, 0.16),
            (0.00005, 0.14),
            (0.00001, 0.10),
            (0.000002, 0.10),
        )

    start = time.monotonic()
    avail = float(stop_time - start - 8.0)
    if avail <= 5.0:
        return p

    best_p = p.copy()
    best_sc = _score_fft(p)
    total_w = float(sum(w for _, w in stages))
    if total_w <= 0.0:
        total_w = 1.0

    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        sc_cur = _score_fft(p_cur)
        if not math.isfinite(sc_cur):
            p = best_p.copy()
            u = p_to_u(p)
            continue
        if sc_cur < best_sc:
            best_sc = sc_cur
            best_p = p_cur.copy()
            p = p_cur.copy()
        else:
            p = best_p.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, max_sweeps=500000, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p, math.inf
    p = p / s

    G = 40
    G_fine = 50
    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue
            base = c[start_i:end_i].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start_i > 0:
                ml = float(c[:start_i].max())
                if ml > max_const:
                    max_const = ml
            if end_i < Ltot:
                mr = float(c[end_i:].max())
                if mr > max_const:
                    max_const = mr

            avg = S / float(n)
            x_hi = max(30.0 * v, 100.0 * avg, 0.05 * S, 1e-12)
            x_hi = min(x_hi, 20.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, avg)

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 <= v <= x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag = i - start_i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                dval = base[diag] + w * infl[diag] + w * w
                maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Fine golden-section-like refinement
            if G > 4:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 3.0 * dw)
                hi = min(x_hi, best_x + 3.0 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, G_fine, dtype=np.float64)
                if 0.0 <= best_x <= x_hi:
                    fine = np.concatenate((fine, [best_x]))
                    fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= 20.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start_i:end_i] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        np.maximum(c, 0.0, out=c)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 6.0:
                    try:
                        sigma = 0.015 + 0.02 * min(flat / 20.0, 1.0)
                        p = p * np.exp(rng.normal(0.0, sigma, n, dtype=np.float64))
                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss > 0.0:
                            p /= ss
                        c = _conv_self(p)
                        S_total = 1.0
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider is not None:
                                consider(best_p)
                        flat = 0
                    except Exception:
                        break
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _lp_direction(f, stop_time):
    n = int(f.size)
    if n < 10 or n > 2000:
        return None
    M = float(_conv_self(f).max())
    if M <= 0 or not math.isfinite(M):
        return None

    # Build A matrix: A[k,j] = f[j] for j <= k <= j+n-1
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f[j]

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        constraints = [A @ g <= b]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        for slv in [cp.SCS, cp.ECOS, cp.CLP]:
            try:
                prob.solve(solver=slv, verbose=False)
                if time.monotonic() > stop_time - 3:
                    break
                break
            except Exception:
                continue
        if g.value is None:
            return None
        g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        gv = float(g_val.sum())
        if gv < 1e-10 or not math.isfinite(gv):
            return None
        return g_val / gv
    except Exception:
        return None


def _targeted_repair(a, stop_time, consider=None):
    """Targeted repair: identify peak lags and reduce contribution."""
    n = int(a.size)
    if n == 0:
        return a
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0).copy()
    s = float(a.sum())
    if s < 1e-300:
        return a
    a = a / s
    Ltot = 2 * n - 1

    for _ in range(50):
        if time.monotonic() > stop_time - 2.0:
            break
        c = _conv_self(a)
        cmax = float(c.max())
        # Find all lags within 5% of max
        thresh = 0.95 * cmax
        peak_lags = np.where(c >= thresh)[0]
        if len(peak_lags) == 0:
            break

        # For each peak lag, find which a[i] contributes most
        # and try to reduce it
        improved_any = False
        for k in peak_lags[:5]:
            if time.monotonic() > stop_time - 1.5:
                break
            # b[k] = sum_j a[j]*a[k-j] for valid j
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_hi < j_lo:
                continue
            js = np.arange(j_lo, j_hi + 1)
            contribs = a[js] * a[k - js]
            total = float(contribs.sum())
            if total <= 0:
                continue
            # Reduce the largest contributor by a small fraction
            imax = js[int(np.argmax(contribs))]
            frac = 0.05 + 0.05 * (len(peak_lags) - 1) / max(1, len(peak_lags))
            delta = a[imax] * frac
            if delta < 1e-15:
                continue
            # Move mass to the least-contributing position
            jmin = js[int(np.argmin(contribs))]
            a[imax] -= delta
            a[jmin] += delta
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-300:
                break
            a /= ss
            improved_any = True

        if not improved_any:
            break

    if consider is not None:
        try:
            consider(a)
        except Exception:
            pass
    return a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 15.0
    if hard <= t0 + 3.0:
        hard = t0 + max(2.0, budget - 1.0)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64)
        best_arr /= 64.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 5e-15:
            best_arr = a.copy()
            best_score = sc

    n_par = int(best_arr.size)

    # Phase 1: LSE optimization on the parent
    phase1_end = min(hard - 80.0, t0 + 0.22 * budget)
    if phase1_end > t0 + 20.0 and n_par >= 128:
        p1 = best_arr.copy()
        p1 = p1 / float(p1.sum())
        try:
            p1 = _lse_optimize(p1, phase1_end, consider=consider, fine_only=False)
            if p1 is not None and p1.size == n_par and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # Phase 2: Coordinate descent (main polishing)
    phase2_end = min(hard - 60.0, t0 + 0.50 * budget)
    if phase2_end > time.monotonic() + 15.0:
        try:
            work, _ = _coord_polish(work, phase2_end, max_sweeps=500000,
                                    seed=int(seed + 7777), consider=consider)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # Phase 3: Targeted repair + another LSE round
    phase3_end = min(hard - 60.0, t0 + 0.65 * budget)
    if phase3_end > time.monotonic() + 15.0:
        try:
            repaired = _targeted_repair(work, min(time.monotonic() + 10.0, phase3_end - 10.0), consider=consider)
            if repaired is not None and repaired.size == n_par:
                work = repaired.copy()
                work = work / float(work.sum())
        except Exception:
            pass

        # Fine LSE on the repaired/Polished result
        lse3_end = min(phase3_end - 10.0, time.monotonic() + 0.4 * (phase3_end - time.monotonic()))
        if lse3_end > time.monotonic() + 10.0:
            try:
                p3 = _lse_optimize(work, lse3_end, consider=consider, fine_only=True)
                if p3 is not None and p3.size == n_par and np.all(np.isfinite(p3)) and p3.sum() > 0:
                    p3 = p3 / float(p3.sum())
                    consider(p3)
                    work = best_arr.copy()
                    work = work / float(work.sum())
            except Exception:
                pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # Phase 4: LP direction for diversity
    lp_end = min(hard - 40.0, time.monotonic() + 30.0)
    if lp_end > time.monotonic() + 10.0 and n_par <= 2000:
        g_dir = _lp_direction(work, lp_end)
        if g_dir is not None and g_dir.size == n_par:
            for t_alpha in [0.3, 0.15, 0.08, 0.04]:
                blended = (1.0 - t_alpha) * work + t_alpha * g_dir
                blended = np.maximum(blended, 0.0)
                bs = float(blended.sum())
                if bs > 0:
                    blended /= bs
                    consider(blended)

    work = best_arr.copy()
    work = work / float(work.sum())

    # Phase 5: Final coordinate descent (most time for polishing)
    phase5_end = min(hard - 20.0, t0 + budget - 25.0)
    if phase5_end > time.monotonic() + 15.0:
        try:
            work, _ = _coord_polish(work, phase5_end, max_sweeps=500000,
                                    seed=int(seed + 987654), consider=consider)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # Phase 6: Another targeted repair + quick polish
    rem = hard - 15.0 - time.monotonic()
    if rem > 20.0:
        try:
            repaired = _targeted_repair(work, time.monotonic() + min(10.0, rem * 0.3), consider=consider)
            if repaired is not None and repaired.size == n_par:
                work = repaired.copy()
                work = work / float(work.sum())
                consider(work)
        except Exception:
            pass

    # Phase 7: Final final coordinate descent squeeze
    fin_end = min(hard - 8.0, t0 + budget - 10.0)
    if fin_end > time.monotonic() + 10.0:
        work = best_arr.copy()
        work = work / float(work.sum())
        try:
            work, _ = _coord_polish(work, fin_end, max_sweeps=300000,
                                    seed=int(seed + 555555), consider=consider)
        except Exception:
            pass

    # Final validation and return
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
