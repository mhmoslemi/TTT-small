        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0)
    if n <= 512:
        c = np.convolve(a, a, mode="full")
    else:
        L = 2 * n - 1
        size = next_fast_len(L)
        F = rfft(a, size)
        c = irfft(F * F, size)[:L]
    return np.maximum(c, 0.0)


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _score_exact_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _softmax_fn(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        return src.copy()
    m = src.size
    xp = (np.arange(m) + 0.5) / m
    fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
    out = np.maximum(fp, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def _lse_optimize(p, stop_time, consider=None, seed=42):
    """LSE-based optimization with adaptive temperature."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax_fn(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    start = time.monotonic()
    avail = max(0.0, float(stop_time) - start - 6.0)
    if avail <= 5.0:
        return p

    best_p = p.copy()
    best_sc = _score_fft(p)
    rng = np.random.default_rng(seed)

    # Adaptive temperature schedule
    c = cfast(p)
    cmax = float(c.max()) if c.size else 1.0
    if not math.isfinite(cmax) or cmax <= 0:
        cmax = 1.0
    T = 0.1 * cmax
    T_min = 1e-7 * cmax
    stale_count = 0
    max_stale = 3
    best_lse_f = math.inf

    while time.monotonic() < stop_time - 4.0 and T > T_min:
        maxit = 30
        opts = {"maxiter": maxit, "gtol": 1e-12, "ftol": 1e-16,
                "maxls": 25, "maxcor": 25, "maxdisp": 0, "disp": False}
        try:
            res = minimize(lse_func, u, args=(T,), method="L-BFGS-B",
                           jac=True, options=opts)
        except Exception:
            break

        if res.x is not None:
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if x.size == n and np.all(np.isfinite(x)):
                try:
                    fval = float(res.fun)
                except Exception:
                    fval = math.inf
                if math.isfinite(fval) and fval < best_lse_f - 1e-14:
                    best_lse_f = fval
                    u = x.copy()
                    stale_count = 0
                    # Reduce temperature
                    T *= 0.5
                    p_cur = _softmax_fn(u)
                    sc_cur = _score_fft(p_cur)
                    if math.isfinite(sc_cur) and sc_cur < best_sc:
                        best_p = p_cur.copy()
                        best_sc = sc_cur
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                else:
                    stale_count += 1
                    if stale_count >= max_stale:
                        # Perturb and reset stagnation
                        u = u + rng.normal(0, 0.01, n)
                        stale_count = 0
                        if T > T_min * 2:
                            T *= 0.7  # Reduce less aggressively
                    elif T > T_min:
                        T *= 0.6
        else:
            break

    # Final extraction
    p_final = _softmax_fn(u)
    p_final = np.maximum(p_final, 0.0)
    ss = float(p_final.sum())
    if ss > 1e-300:
        p_final /= ss
    sc_final = _score_fft(p_final)
    if math.isfinite(sc_final) and sc_final < best_sc:
        best_p = p_final.copy()
        best_sc = sc_final
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p


def _coord_polish(p, stop_time, G=None, seed=12345, consider=None):
    """Coordinate descent with fine grid and diversification."""
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    if G is None:
        if n <= 300:
            G = 64
        elif n <= 600:
            G = 48
        elif n <= 1500:
            G = 36
        elif n <= 4000:
            G = 28
        else:
            G = 20
    G = int(max(16, G))

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < 50000 and time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.5 or S_total < 0.67:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(30.0 * v, 60.0 * S / float(n), 0.05 * S, 1e-14)
            x_hi = min(x_hi, 10.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, S / float(n))

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag_i = i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[valid, None] * infl[None, :]
                dval = base[diag_i] + w[valid] * infl[diag_i] + w[valid] * w[valid]
                aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                dv = denom[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Refine with finer grid around best
            if G > 4 and best_x > 0 and best_x < x_hi:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.5 * dw)
                hi = min(x_hi, best_x + 2.5 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 32, dtype=np.float64)
                if 0.0 <= best_x <= x_hi:
                    fine = np.concatenate((fine, [best_x]))
                    fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val - 1e-15:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 10.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 8.0:
                    # Diversification: multiplicative perturbation
                    sigma = 0.03 + 0.03 * min(flat, 15) / 15.0
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    else:
                        break
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    flat = 0
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        sc = float(_score_fft(p))
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _peak_focus(p, stop_time, consider=None, seed=999):
    """Targeted perturbation to reduce specific peak lags."""
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p
    p = p / s
    
    best_p = p.copy()
    best_sc = _score_fft(p)
    rng = np.random.default_rng(seed)
    
    for it in range(200):
        if time.monotonic() > stop_time - 2.0:
            break
        
        c = _conv_self(p)
        # Get top 10 peak lags
        top_k = min(10, c.size)
        peak_indices = np.argsort(c)[-top_k:][::-1]
        peak_vals = c[peak_indices]
        
        # For each peak lag, try to reduce it by adjusting coordinates
        # that contribute most to that peak
        for k_idx in range(min(5, len(peak_indices))):
            k = int(peak_indices[k_idx])
            # c[k] = sum_i a[i]*a[k-i], the "influence" of a[j] on c[k] is a[k-j] + a[j-k]
            infl = np.zeros(n)
            for j in range(n):
                val = 0.0
                if 0 <= k - j < n:
                    val += p[k - j]
                if 0 <= j - k < n:
                    val += p[j - k]
                infl[j] = val
            
            # Try decreasing the most influential coordinates
            top_infl = np.argsort(infl)[-5:][::-1]
            for j in top_infl:
                if time.monotonic() > stop_time - 1.0:
                    break
                # Try reducing p[j] by a small amount
                for frac in [0.1, 0.05, 0.02]:
                    p_test = p.copy()
                    p_test[int(j)] *= (1.0 - frac)
                    p_test = np.maximum(p_test, 0.0)
                    ps = float(p_test.sum())
                    if ps > 0:
                        p_test /= ps
                    sc_test = _score_fft(p_test)
                    if math.isfinite(sc_test) and sc_test < best_sc:
                        p = p_test.copy()
                        best_p = p.copy()
                        best_sc = sc_test
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                        break
    
    return best_p
