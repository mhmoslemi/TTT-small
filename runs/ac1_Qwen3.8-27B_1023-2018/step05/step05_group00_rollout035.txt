def _lp_step(f, stop_time, n_constraints_max=4096):
    n = int(f.size)
    if n < 10 or n > n_constraints_max:
        return None
    M = float(_conv_self(f).max())
    if M <= 0 or not math.isfinite(M):
        return None

    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f[j]

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        b = np.full(L, M, dtype=np.float64)
        constraints = [A @ g <= b]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        t_start = time.monotonic()
        solved = False
        for slv in [cp.SCS, cp.ECOS, cp.CBC, cp.CLP]:
            try:
                prob.solve(solver=slv, verbose=False)
                solved = True
                break
            except Exception:
                continue
            if time.monotonic() > stop_time - 5:
                break
        if not solved or g.value is None:
            return None
        g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        g_val = np.where(np.isfinite(g_val), g_val, 0.0)
        gv = float(g_val.sum())
        if gv < 1e-10 or not math.isfinite(gv):
            return None
        return g_val / gv
    except Exception:
        return None

            break

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    return p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 3.0:
        hard = t0 + max(2.0, budget - 1.0)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64)
        best_arr /= 64.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s_val = float(a.sum())
        if s_val < 0.01:
            return
        try:
            sc = float(_score_fft(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_arr = a.copy()
            best_score = sc

    n_par = int(best_arr.size)

    # === Phase 1: LSE optimization on the parent length ===
    phase1_end = min(hard - 80.0, t0 + 0.25 * budget)
    if phase1_end > t0 + 20.0 and n_par >= 64:
        p1 = best_arr.copy()
        p1 = p1 / float(p1.sum())
        try:
            p1 = _lbfgs_optimize(p1, phase1_end, consider=consider, fine_only=False)
            if p1 is not None and p1.size == n_par and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 2: Coordinate descent ===
    phase2_end = min(hard - 100.0, t0 + 0.48 * budget)
    if phase2_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord_polish(work, phase2_end, max_sweeps=500000,
                                    seed=int(seed + 7777), consider=consider)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 3: LP step for direction finding ===
    lp_end = min(hard - 80.0, time.monotonic() + 80.0)
    if lp_end > time.monotonic() + 20.0 and n_par <= 4096:
        g_dir = _lp_step(work, lp_end)
        if g_dir is not None and g_dir.size == n_par:
            for t_alpha in [0.3, 0.2, 0.12, 0.06]:
                blended = (1.0 - t_alpha) * work + t_alpha * g_dir
                blended = np.maximum(blended, 0.0)
                bs = float(blended.sum())
                if bs > 0:
                    blended /= bs
                    consider(blended)
            # Use best blended as new work for next phase
            best_blend = None
            best_blend_sc = math.inf
            for t_alpha in [0.3, 0.2, 0.12, 0.06]:
                blended = (1.0 - t_alpha) * work + t_alpha * g_dir
                blended = np.maximum(blended, 0.0)
                bs = float(blended.sum())
                if bs > 0:
                    blended /= bs
                    sc = _score_fft(blended)
                    if sc < best_blend_sc:
                        best_blend_sc = sc
                        best_blend = blended.copy()
            if best_blend is not None:
                work = best_blend

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 4: More coordinate descent ===
    phase4_end = min(hard - 60.0, t0 + 0.68 * budget)
    if phase4_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord_polish(work, phase4_end, max_sweeps=500000,
                                    seed=int(seed + 123456), consider=consider)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 5: Fine LSE + coordinate polish ===
    phase5_end = min(hard - 30.0, t0 + budget - 40.0)
    if phase5_end > time.monotonic() + 25.0:
        lse5_end = min(phase5_end - 25.0, time.monotonic() + 0.30 * (phase5_end - time.monotonic()))
        if lse5_end > time.monotonic() + 10.0:
            try:
                p5 = _lbfgs_optimize(work, lse5_end, consider=consider, fine_only=True)
                if p5 is not None and p5.size == n_par and np.all(np.isfinite(p5)) and p5.sum() > 0:
                    p5 = p5 / float(p5.sum())
                    consider(p5)
                    work = best_arr.copy()
                    work = work / float(work.sum())
            except Exception:
                pass

        coord5_end = phase5_end - 5.0
        if coord5_end > time.monotonic() + 10.0:
            try:
                work, _ = _coord_polish(work, coord5_end, max_sweeps=300000,
                                        seed=int(seed + 987654), consider=consider)
            except Exception:
                pass

    # === Phase 6: Block coordinate descent for local refinement ===
    work = best_arr.copy()
    work = work / float(work.sum())
    rem6 = hard - 25.0 - time.monotonic()
    if rem6 > 30.0:
        block_end = min(hard - 20.0, time.monotonic() + rem6 * 0.6)
        if block_end > time.monotonic() + 10.0:
            try:
                work, _ = _block_coord_polish(work, block_end, seed=int(seed + 31313),
                                              consider=consider, block_size=2)
            except Exception:
                pass

    # === Phase 7: Random restart exploration ===
    work = best_arr.copy()
    work = work / float(work.sum())
    rem7 = hard - 20.0 - time.monotonic()
    if rem7 > 40.0:
        rng = np.random.default_rng(int(seed + 555000))
        n_restarts = 2
        for r in range(n_restarts):
            if time.monotonic() > hard - 20.0:
                break
            perturbed = work * np.exp(rng.normal(0.0, 0.05, n_par, dtype=np.float64))
            perturbed = np.maximum(perturbed, 0.0)
            ps = float(perturbed.sum())
            if ps > 0:
                perturbed /= ps
                consider(perturbed)
                restart_end = min(hard - 20.0, time.monotonic() + rem7 / (n_restarts + 1) - 5.0)
                if restart_end > time.monotonic() + 8.0:
                    try:
                        perturbed, _ = _coord_polish(perturbed, restart_end, max_sweeps=50000,
                                                     seed=int(seed + 555000 + r * 111),
                                                     consider=consider)
                    except Exception:
                        pass

    # === Phase 8: Final coordinate squeeze ===
    work = best_arr.copy()
    work = work / float(work.sum())
    fin_end = min(hard - 10.0, t0 + budget - 12.0)
    if fin_end > time.monotonic() + 10.0:
        try:
            work, _ = _coord_polish(work, fin_end, max_sweeps=200000,
                                    seed=int(seed + 777777), consider=consider)
        except Exception:
            pass

    # Final validation and return
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
