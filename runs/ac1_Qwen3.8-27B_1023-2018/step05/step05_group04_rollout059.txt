        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)
        best_u = u.copy()
        stale = 0
        while time.monotonic() < end_stage - 1.0:
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True,
                               options={"maxiter": 35, "gtol": 1e-13, "ftol": 1e-16,
                                        "maxls": 10, "maxdisp": 0})
            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < 1e99 and fval < (best_u[0] * 0 + 1e100):
                    p_cur = _softmax(best_u)
                    sc_cur = _score_arr(p_cur)
                    if math.isfinite(sc_cur) and sc_cur < best_sc - 1e-13:
                        best_sc = sc_cur
                        best_p = p_cur.copy()
                        if consider:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    stale = 0
                    best_u = res.x.copy()
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if getattr(res, "status", 1) == 0:
                break

    p_final = _softmax(best_u)
    p_final = np.maximum(p_final, 0.0)
    sc_final = _score_arr(p_final)
    if math.isfinite(sc_final) and sc_final < best_sc - 1e-13:
        best_sc = sc_final
        best_p = p_final.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p


def _coord_descent(p, stop_time, consider=None, seed=12345, n_grid=56, gs_iter=24):
    n = int(p.size)
    if n == 0:
        return p, float('inf')
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p, float('inf')
    p = p / s
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    Ltot = 2 * n - 1
    GR = (math.sqrt(5.0) - 1.0) / 2.0
    flat = 0
    sweep = 0

    def eval_1d(w, base, infl, diag_pos, max_const, n):
        s_new = 1.0 + w
        if s_new <= 1e-15:
            return float('inf')
        aff_max = float(np.max(base + w * infl))
        if 0 <= diag_pos < len(base):
            dval = base[diag_pos] + w * infl[diag_pos] + w * w
            if dval > aff_max:
                aff_max = dval
        m = max(aff_max, max_const)
        return 2.0 * n * m / (s_new * s_new)

    max_sweeps = 100000
    while sweep < max_sweeps and time.monotonic() < stop_time - 3.0:
        improved = False
        n_imp = 0
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 1.0:
                break
            v = float(p[i])
            start_lag = i
            end_lag = min(i + n, Ltot)
            Ls = end_lag - start_lag
            if Ls <= 0:
                continue
            base = c[start_lag:end_lag]
            infl = 2.0 * p[:Ls]
            diag = i  # 2i - start_lag = i

            max_const = 0.0
            if start_lag > 0:
                mc = float(c[:start_lag].max())
                if mc > max_const:
                    max_const = mc
            if end_lag < Ltot:
                mc = float(c[end_lag:].max())
                if mc > max_const:
                    max_const = mc

            S = 1.0
            x_hi = max(35.0 * v, 25.0 / n, 0.01)
            x_hi = min(x_hi, 0.5)
            x_hi = max(x_hi, 1e-8)

            pts = np.linspace(0.0, x_hi, n_grid + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                ins = np.searchsorted(pts, v)
                pts = np.insert(pts, ins, v)

            ws = pts - v
            s_new = S + ws
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag < Ls:
                aff[:, diag] = aff[:, diag] + ws[:, None] * ws[:, None]
            maff = np.max(aff, axis=1)
            maff = np.maximum(maff, max_const)
            valid = s_new > 1e-12
            scores = np.full(pts.shape, float('inf'), dtype=np.float64)
            if np.any(valid):
                scores[valid] = 2.0 * n * maff[valid] / (s_new[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            # Golden section refinement
            gp = max(0, imin - 2)
            gp2 = min(len(pts) - 1, imin + 2)
            lo = float(pts[gp])
            hi = float(pts[gp2])
            if hi - lo < 1e-14:
                lo = max(0.0, best_x - 1e-6)
                hi = best_x + 1e-6

            a_gs = lo
            b_gs = hi
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)
            f_c = eval_1d(c_gs - v, base, infl, diag, max_const, n)
            f_d = eval_1d(d_gs - v, base, infl, diag, max_const, n)
            for _ in range(gs_iter):
                if b_gs - a_gs < 1e-13 * max(1.0, abs(best_x)):
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = eval_1d(c_gs - v, base, infl, diag, max_const, n)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_d = eval_1d(d_gs - v, base, infl, diag, max_const, n)
            x_mid = (a_gs + b_gs) / 2.0
            f_mid = eval_1d(x_mid - v, base, infl, diag, max_const, n)
            if f_mid < best_val:
                best_x = x_mid
                best_val = f_mid
            for xp in (0.0, x_hi, v * 0.5, v * 2.0):
                xp = min(max(xp, 0.0), x_hi)
                fp = eval_1d(xp - v, base, infl, diag, max_const, n)
                if fp < best_val:
                    best_x = xp
                    best_val = fp

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= x_hi:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    aff_seg = base + w * infl
                    if 0 <= diag < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag] += w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    c[start_lag:end_lag] = aff_seg
                    cur_score = 2.0 * n * new_max
                    improved = True
                    n_imp += 1

        # Renormalize
        s_new = float(p.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            inv = 1.0 / s_new
            p *= inv
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-15:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 12.0:
                    kick_size = 0.015 + 0.03 * min(1.0, flat / 15.0)
                    if rng.random() < 0.3:
                        idxs = rng.choice(n, size=max(1, n // 50), replace=False)
                        p[idxs] = 0.0
                    else:
                        p = p * np.exp(rng.normal(0.0, kick_size, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _smooth_and_rescore(p, n, sigma, rng):
    """Apply a small Gaussian perturbation and renormalize."""
    q = p + rng.normal(0.0, sigma, n)
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if s > 1e-300:
        q /= s
    return q


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 5.0:
        hard = t0 + max(3.0, budget - 2.0)

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float('inf')

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()

    consider(parent)
    n_work = int(best_arr.size)
    if n_work < 128:
        n_work = 384
    if n_work > 4096:
        n_work = 3072

    rng = np.random.default_rng(seed)

    # Phase 1: LSE optimization
    p_work = _resample(best_arr, n_work)
    lse_time = min(100.0, 0.10 * budget)
    lse_end = t0 + lse_time
    if lse_end > t0 + 10.0:
        try:
            p2 = _lse_optimize(p_work, lse_end, consider=consider)
            if p2 is not None and p2.size == n_work and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p_work = p2 / float(p2.sum())
                consider(p_work)
        except Exception:
            pass

    # Phase 2: Main coordinate descent
    cd_end = hard - 10.0
    if cd_end > time.monotonic() + 25.0:
        p_start = _resample(best_arr, n_work)
        try:
            p_cd, _ = _coord_descent(p_start, cd_end, consider=consider, seed=int(seed + 7777))
            if p_cd is not None and p_cd.size > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                consider(p_cd)
        except Exception:
            pass

    # Phase 3: Restart with different perturbation if time remains
    rem = hard - 10.0 - time.monotonic()
    if rem > 80.0:
        # Try a blended initialization
        p_blend = 0.7 * _resample(best_arr, n_work) + 0.3 * _resample(p_work, n_work)
        p_blend = np.maximum(p_blend, 0.0)
        sb = float(p_blend.sum())
        if sb > 0:
            p_blend /= sb
        consider(p_blend)

        # Add a small random perturbation
        p_blend = p_blend * np.exp(rng.normal(0.0, 0.02, n_work))
        p_blend = np.maximum(p_blend, 0.0)
        sb = float(p_blend.sum())
        if sb > 0:
            p_blend /= sb
        consider(p_blend)

        blend_end = min(hard - 8.0, time.monotonic() + rem * 0.55)
        if blend_end > time.monotonic() + 20.0:
            try:
                p_b, _ = _coord_descent(p_blend, blend_end, consider=consider, seed=int(seed + 8888))
                if p_b is not None and p_b.size > 0 and np.all(np.isfinite(p_b)) and p_b.sum() > 0:
                    consider(p_b)
            except Exception:
                pass

    # Phase 4: Try a "two-lobe" initialization if time remains
    rem = hard - 8.0 - time.monotonic()
    if rem > 60.0:
        x_idx = np.arange(n_work, dtype=np.float64)
        x_norm = x_idx / n_work
        # Two-lobe: peaks at 1/3 and 2/3
        lobe1 = np.exp(-8.0 * (x_norm - 1.0/3.0)**2)
        lobe2 = np.exp(-8.0 * (x_norm - 2.0/3.0)**2)
        two_lobe = lobe1 + lobe2
        s_t = float(two_lobe.sum())
        if s_t > 0:
            two_lobe /= s_t
        p_tl = 0.6 * _resample(best_arr, n_work) + 0.4 * two_lobe
        p_tl = np.maximum(p_tl, 0.0)
        s_tl = float(p_tl.sum())
        if s_tl > 0:
            p_tl /= s_tl
        consider(p_tl)

        tl_end = min(hard - 6.0, time.monotonic() + rem * 0.5)
        if tl_end > time.monotonic() + 15.0:
            try:
                p_tl2, _ = _coord_descent(p_tl, tl_end, consider=consider, seed=int(seed + 9999))
                if p_tl2 is not None and p_tl2.size > 0 and np.all(np.isfinite(p_tl2)) and p_tl2.sum() > 0:
                    consider(p_tl2)
            except Exception:
                pass

    # Phase 5: Final squeeze on the best
    rem = hard - 6.0 - time.monotonic()
    if rem > 30.0:
        p_final = _resample(best_arr, n_work)
        try:
            p_f, _ = _coord_descent(p_final, hard - 5.0, consider=consider, seed=int(seed + 11111))
            if p_f is not None and p_f.size > 0 and np.all(np.isfinite(p_f)) and p_f.sum() > 0:
                consider(p_f)
        except Exception:
            pass

    # Phase 6: Very final micro-squeeze
    rem = hard - 4.0 - time.monotonic()
    if rem > 15.0:
        p_sq = _resample(best_arr, n_work)
        # Tiny perturbation to break symmetry
        p_sq = p_sq * np.exp(rng.normal(0.0, 0.005, n_work))
        p_sq = np.maximum(p_sq, 0.0)
        ss = float(p_sq.sum())
        if ss > 0:
            p_sq /= ss
        consider(p_sq)
        try:
            p_sq2, _ = _coord_descent(p_sq, hard - 3.0, consider=consider, seed=int(seed + 22222))
            if p_sq2 is not None and p_sq2.size > 0 and np.all(np.isfinite(p_sq2)) and p_sq2.sum() > 0:
                consider(p_sq2)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
