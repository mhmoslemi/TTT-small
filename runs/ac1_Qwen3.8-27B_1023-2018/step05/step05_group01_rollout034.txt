    n = a.size

def _score_fft(a):

                    break

        p_cur = _softmax_fn(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif math.isfinite(sc_cur) and sc_cur < best_sc + 0.003:
            p = p_cur.copy()
        else:
            p = best_p.copy()
        u = _p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, G=48, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s0 = float(p.sum())
    if s0 < 1e-300:
        return p, math.inf
    p = p / s0

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < 200000 and time.monotonic() < stop_time - 0.3:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.15:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0

            S = float(S_total)
            v = float(p[i])
            start = i
            end = i + n
            base = c[start:end]
            diag_idx = i  # 2i - start = 2i - i = i

            # max_const: max outside [start, end)
            mc = 0.0
            if start > 0:
                ml = float(np.max(c[:start]))
                if ml > mc:
                    mc = ml
            if end < Ltot:
                mr = float(np.max(c[end:]))
                if mr > mc:
                    mc = mr

            # Grid setup
            x_hi = max(15.0 * v, 30.0 * S / float(n), 1e-4 * S, 1e-14)
            x_hi = min(x_hi, 6.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, S / float(n))

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            w_all = pts - v
            denom = S + w_all
            two_p = 2.0 * p
            aff = base[None, :] + w_all[:, None] * two_p[None, :]
            aff[:, diag_idx] += w_all * w_all
            aff = np.maximum(aff, mc)
            mv = aff.max(axis=1)
            valid = denom > 1e-15
            vals = np.full(len(pts), math.inf)
            vals[valid] = (2.0 * n * mv[valid]) / (denom[valid] ** 2)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Golden section refinement
            if best_x > 0 and best_x < x_hi and G > 8:
                gap = x_hi / G
                lo = max(0.0, best_x - gap)
                hi = min(x_hi, best_x + gap)
                if hi - lo > 1e-16:
                    def f1d(x):
                        w = x - v
                        d = S + w
                        if d <= 1e-15:
                            return math.inf
                        aff_s = base + w * two_p
                        aff_s[diag_idx] += w * w
                        aff_s = np.maximum(aff_s, mc)
                        mval = float(aff_s.max())
                        return (2.0 * n * mval) / (d * d)
                    gs_x = _golden_section_min(f1d, lo, hi, tol=1e-15, max_iter=25)
                    gs_val = f1d(gs_x)
                    if gs_val < best_val - 1e-15:
                        best_x = gs_x
                        best_val = gs_val

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 8.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * two_p
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        # End of sweep: rescale and recompute
        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 4.0:
                    sigma = 0.015 + 0.025 * min(flat, 12) / 12.0
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    else:
                        break
                    c = _conv_self(p)
                    S_total = 1.0
                    new_score = float(2.0 * n * c.max())
                    if math.isfinite(new_score) and new_score < best_score - 1e-14:
                        best_score = new_score
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    cur_score = new_score if math.isfinite(new_score) else cur_score
                    flat = 0
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        sc = float(_score_fft(p))
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_end = t0 + budget - 25.0

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    def run_opt(n, init_arr, end_time, sd, do_lse=True, do_coord=True):
        nonlocal best_arr, best_score
        p = _resample_mid(init_arr, n)
        if p.size != n or float(p.sum()) <= 0:
            p = np.full(n, 1.0 / n)
        p = p / float(p.sum())
        consider(p)

        if time.monotonic() > end_time - 8.0:
            return
        phase_len = max(0.0, float(end_time) - time.monotonic())

        if do_lse and phase_len > 25.0:
            lse_stop = min(float(end_time) - max(15.0, 0.35 * phase_len),
                           time.monotonic() + 0.50 * phase_len)
            if lse_stop > time.monotonic() + 10.0:
                try:
                    p2 = _lse_optimize(p, lse_stop, consider=consider, seed=sd)
                    if p2 is not None and p2.size == n and float(p2.sum()) > 0:
                        p = p2 / float(p2.sum())
                        consider(p)
                except Exception:
                    pass

        rem = float(end_time) - time.monotonic()
        if do_coord and rem > 12.0:
            cstop = min(float(end_time) - 1.0, time.monotonic() + rem - 1.5)
            if cstop > time.monotonic() + 8.0:
                try:
                    p3, _ = _coord_polish(p, cstop, seed=sd, consider=consider)
                    if p3 is not None and p3.size == n and float(p3.sum()) > 0:
                        p = p3 / float(p3.sum())
                        consider(p)
                except Exception:
                    pass
        return p

    n_parent = int(best_arr.size)
    total_avail = hard_end - t0

    # Phase 1: Main optimization at parent length (LSE + coord)
    p1_end = t0 + 0.55 * total_avail
    run_opt(n_parent, best_arr, p1_end, int(seed + 100), do_lse=True, do_coord=True)

    # Phase 2: Perturbation restarts at parent length
    if time.monotonic() < hard_end - 60.0:
        rng2 = np.random.default_rng(seed + 777)
        for pr in range(3):
            if time.monotonic() > hard_end - 40.0:
                break
            sigma_perturb = 0.03 + 0.02 * pr
            pert = best_arr * np.exp(rng2.normal(0, sigma_perturb, best_arr.size))
            pert = np.maximum(pert, 0.0)
            ps = float(pert.sum())
            if ps > 0:
                pert /= ps
            sub_end = min(hard_end - 30.0, time.monotonic() + 0.25 * (hard_end - time.monotonic()))
            if sub_end > time.monotonic() + 20.0:
                run_opt(n_parent, pert, sub_end, int(seed + 2000 + pr * 100),
                        do_lse=True, do_coord=True)

    # Phase 3: Different initializations at parent length
    if time.monotonic() < hard_end - 50.0:
        rng3 = np.random.default_rng(seed + 5000)
        for init_seed in range(2):
            if time.monotonic() > hard_end - 35.0:
                break
            n_try = n_parent
            x = (np.arange(n_try) + 0.5) / float(n_try)
            side = 1.0 if rng3.random() < 0.5 else -1.0
            amp = 0.3 + 1.0 * float(rng3.random())
            curv = 0.5 + 1.5 * float(rng3.random())
            base_shape = 1.0 + amp * side * (2.0 * x - 1.0) + curv * (2.0 * x - 1.0) ** 2
            base_shape = np.maximum(base_shape, 0.0)
            bs = float(base_shape.sum())
            if bs > 0:
                base_shape /= bs
            init = _resample_mid(best_arr, n_try) * 0.5 + base_shape * 0.5
            init = np.maximum(init, 0.0)
            init /= float(init.sum())
            sub_end = min(hard_end - 25.0, time.monotonic() + 0.2 * (hard_end - time.monotonic()))
            if sub_end > time.monotonic() + 15.0:
                run_opt(n_try, init, sub_end, int(seed + 6000 + init_seed * 333),
                        do_lse=False, do_coord=True)

    # Phase 4: Try different lengths
    if time.monotonic() < hard_end - 70.0:
        for n_alt_ratio in [0.75, 1.5]:
            if time.monotonic() > hard_end - 40.0:
                break
            n_alt = int(n_parent * n_alt_ratio)
            n_alt = max(512, min(n_alt, 8192))
            if n_alt == n_parent:
                continue
            sub_end = min(hard_end - 20.0, time.monotonic() + 0.15 * (hard_end - time.monotonic()))
            if sub_end > time.monotonic() + 20.0:
                run_opt(n_alt, best_arr, sub_end, int(seed + 8000 + int(n_alt_ratio * 100)),
                        do_lse=True, do_coord=True)

    # Phase 5: Final squeeze on best
    final_end = min(hard_end - 10.0, t0 + budget - 10.0)
    if final_end > time.monotonic() + 20.0 and best_arr.size > 0:
        nbest = int(best_arr.size)
        rem = final_end - time.monotonic()

        if rem > 40.0 and nbest <= 6000:
            lstop = min(final_end - 25.0, time.monotonic() + 0.3 * rem)
            if lstop > time.monotonic() + 10.0:
                try:
                    p2 = _lse_optimize(best_arr, lstop, consider=consider, seed=int(seed + 99999))
                    if p2 is not None and p2.size == nbest and float(p2.sum()) > 0:
                        p2 = p2 / float(p2.sum())
                        consider(p2)
                except Exception:
                    pass

        cstop = min(final_end - 3.0, time.monotonic() + max(10.0, 0.6 * (final_end - time.monotonic())))
        if cstop > time.monotonic() + 8.0:
            try:
                p3, _ = _coord_polish(best_arr, cstop, seed=int(seed + 123456789), consider=consider)
                if p3 is not None and p3.size == nbest and float(p3.sum()) > 0:
                    p3 = p3 / float(p3.sum())
                    consider(p3)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
