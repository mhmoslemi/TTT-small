            p = best_p.copy()
            continue
        p = p / S
        
        if (step + 1) % check_interval == 0:
            c = _conv_fft(p)
            M = float(c.max())
            k_star = int(c.argmax())
            sc = 2.0 * n * M
            if sc < best_score - 1e-12:
                best_score = sc
                best_p = p.copy()
                patience = 0
            else:
                patience += 1
                if patience > 50:
                    eta = max(eta * 0.7, 1e-5)
                    patience = 0
                    restarts += 1
                    if restarts > 5:
                        break
                    # Perturb
                    p = best_p * (1.0 + 0.05 * rng.standard_normal(n))
                    p = np.maximum(p, 0.0)
                    S = float(p.sum())
                    if S > 1e-300:
                        p = p / S
                        c = _conv_fft(p)
                        M = float(c.max())
                        k_star = int(c.argmax())
    
    return best_p


def _lse_optimize(p, stop_time, consider=None):
    """LSE smoothed optimization using L-BFGS-B."""
    p = np.maximum(p.astype(np.float64).copy(), 0.0)
    n = p.size
    if n == 0:
        return p
    S = float(p.sum())
    if S < 1e-300:
        return p
    p = p / S
    
    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))
    
    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c
    
    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        # Clip to avoid overflow
        scaled = np.clip(scaled, -50, 50)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * math.log(sum_exp / c.size)
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        
        # Gradient: sum_k w_k * 2*a[k-i] - 2*f (for S=1)
        # This is a weighted sum of shifted copies of a
        # Use FFT: FFT(w) * FFT(a_reversed) → gives the weighted shift
        # Actually, let's compute directly for clarity
        # grad[i] = sum_k w_k * 2 * a[k-i] - 2*f
        # = 2 * (w * a)[i] - 2*f  where (w*a) is cross-correlation
        
        # Compute cross-correlation of w and a via FFT
        F = rfft(a, conv_size)  # Wait, I need a here but a = pp
        G = rfft(w, conv_size)
        # cross-correlation: ifft(conj(F) * G) or ifft(F * conj(G))
        # Actually, sum_k w[k] * a[k-i] = (w * a_flipped)[i]
        # Let me just do it directly for n up to ~4096
        if n <= 4096:
            # Direct computation
            grad = np.zeros(n, dtype=np.float64)
            for k in range(conv_len):
                if w[k] < 1e-15:
                    continue
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                if hi >= lo:
                    idx = np.arange(lo, hi + 1)
                    src = k - idx
                    grad[idx] += 2.0 * w[k] * pp[src]
            grad -= 2.0 * f
        else:
            # FFT-based
            # sum_k w[k] * a[k-i] for i = 0, ..., n-1
            # This is the cross-correlation of w and a
            # = ifft(conj(fft(a)) * fft(w)) but with proper indexing
            F_a = rfft(pp, conv_size)
            F_w = rfft(w, conv_size)
            # Cross-correlation: ifft(conj(F_a) * F_w)
            # Wait, I need: sum_k w[k] * a[k-i]
            # Let flip_a = a[::-1]
            # Then sum_k w[k] * a[k-i] = (w * flip_a)[n-1-i]
            F_flip = rfft(pp[::-1], conv_size)
            cross = irfft(F_w * F_flip, conv_size)
            # cross[m] = sum_k w[k] * flip_a[m-k] = sum_k w[k] * a[n-1-(m-k)]
            # We want sum_k w[k] * a[k-i]
            # Let j = k-i, then k = j+i: sum_j w[j+i] * a[j]
            # This is (w_shifted * a)[i] where w_shifted depends on i...
            # This is getting complicated. Let me just use the direct method.
            grad = np.zeros(n, dtype=np.float64)
            # Use the fact that w is spiky (mostly concentrated near argmax)
            top_k = np.argsort(w)[-min(50, n):]
            for k in top_k:
                if w[k] < 1e-15:
                    continue
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                if hi >= lo:
                    idx = np.arange(lo, hi + 1)
                    src = k - idx
                    grad[idx] += 2.0 * w[k] * pp[src]
            grad -= 2.0 * f
        
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad
    
    # Stages: (temperature_fraction, time_fraction)
    stages = [
        (0.10, 0.12),
        (0.03, 0.15),
        (0.01, 0.20),
        (0.003, 0.25),
        (0.001, 0.28),
    ]
    
    start = time.monotonic()
    avail = max(0.0, stop_time - start - 5.0)
    if avail < 10:
        return p
    
    best_p = p.copy()
    best_sc = _score_fft(p)
    total_w = sum(w for _, w in stages)
    
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3:
            break
        cum += wt
        end_stage = min(stop_time - 3, start + avail * (cum / total_w))
        if end_stage <= now + 2:
            continue
        
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)
        
        # Run L-BFGS for this temperature
        u_cur = u.copy()
        f0, _ = lse_func(u_cur, T)
        if not math.isfinite(f0):
            continue
        best_u = u_cur.copy()
        best_f = f0
        
        max_iters = 8
        for it in range(max_iters):
            if time.monotonic() > end_stage - 1:
                break
            opts = {
                "maxiter": 25,
                "gtol": 1e-11,
                "ftol": 1e-14,
                "maxls": 15,
                "maxcor": 20,
                "disp": False,
            }
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                             jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                try:
                    val = float(res.fun)
                except Exception:
                    val = math.inf
                if math.isfinite(val) and val < best_f - 1e-14:
                    best_f = val
                    best_u = res.x.copy()
            if getattr(res, "status", 1) == 0:
                break
        
        u = best_u
        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur):
            if sc_cur < best_sc:
                best_p = p_cur.copy()
                best_sc = sc_cur
            p = p_cur.copy()
        u = p_to_u(p)
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    
    return best_p
