def _fft_conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()

    if n == 0:

            best_p = p.copy()
            try:
                consider(p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= max_flat:
                if time.monotonic() < deadline - 10.0:
                    # Restart with perturbation
                    sigma = 0.03 + 0.02 * min(flat / 10.0, 1.0)
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ps = float(p.sum())
                    if ps < 1e-300:
                        p = best_p.copy()
                    else:
                        p /= ps
                    S = 1.0
                    c = _fft_conv_self(p)
                    infl = 2.0 * p
                    cur_score = 2.0 * n * float(c.max())
                else:
                    break
                flat = 0

    p_final = np.maximum(best_p, 0.0)
    ps = float(p_final.sum())
    if ps > 1e-300:
        p_final /= ps
    return p_final, best_score


def _lse_smooth_step(p, deadline, consider, rng):
    """Short LSE-based smooth optimization to make coordinated moves."""
    n = int(p.size)
    if n < 32 or n > 4096 or time.monotonic() > deadline - 15:
        return p

    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p /= s

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def lse_obj(u, T):
        pp = np.exp(u - u.max())
        ss = pp.sum()
        if ss <= 0 or not math.isfinite(ss):
            pp = np.ones(n) / n
        else:
            pp /= ss
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros(n)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            f = cmax
        else:
            f = cmax + T * math.log(sum_exp / conv_len)
        # Gradient: w * pp where w = exps/sum_exp convolved with pp
        w = exps / sum_exp if sum_exp > 0 else np.zeros(conv_len)
        # Gradient of f w.r.t. pp: 2*w * pp (cross correlation)
        F_p = rfft(pp[::-1], conv_size)
        F_w = rfft(w, conv_size)
        cross = irfft(F_p * F_w, conv_size)[:conv_len]
        q = cross[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        grad = 2.0 * q
        # Push to softmax parameter u: grad_u = pp * (grad - pp·grad)
        tdp = float(np.dot(pp, grad))
        grad_u = pp * (grad - tdp)
        grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
        return float(f), grad_u

    # Temperature schedule
    c0 = cfast(p)
    cmax0 = float(c0.max()) if c0.size else 1.0
    T_start = max(1e-6, 0.01 * cmax0)
    T_end = max(1e-8, 1e-5 * cmax0)

    # Use scipy L-BFGS-B for a few iterations
    from scipy.optimize import minimize

    u0 = np.log(np.maximum(p, 1e-300))
    u0 = u0 - u0.mean()

    best_p = p.copy()
    best_f = lse_obj(u0, T_start)[0]
    stale = 0

    T = T_start
    n_stages = 5
    for stage in range(n_stages):
        if time.monotonic() > deadline - 5:
            break
        T = T_start * (T_end / T_start) ** (stage / max(1, n_stages - 1))
        T = max(T, 1e-10)

        maxit = 20
        opts = {"maxiter": maxit, "gtol": 1e-10, "ftol": 1e-14,
                "maxls": 15, "maxcor": 20, "maxdisp": 0, "disp": False}
        try:
            res = minimize(lse_obj, u0, args=(T,), method="L-BFGS-B",
                           jac=True, options=opts)
        except Exception:
            break

        if res.x is not None:
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if x.size == n and np.all(np.isfinite(x)):
                pp = np.exp(x - x.max())
                ss = pp.sum()
                if ss > 1e-300:
                    pp /= ss
                    fval = float(res.fun)
                    sc = 2.0 * n * float(cfast(pp).max())
                    if math.isfinite(sc) and sc < best_f:
                        best_f = sc
                        best_p = pp.copy()
                        u0 = x.copy()
                        try:
                            consider(pp)
                        except Exception:
                            pass
                        stale = 0
                    else:
                        stale += 1
        if stale >= 3:
            break

    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 15.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel().copy()
    except Exception:
        parent = None

    if parent is None or parent.size == 0:
        parent = np.ones(512, dtype=np.float64)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.sum() < 0.01:
        parent = np.ones(512, dtype=np.float64)

    n_par = parent.size
    p_norm = parent.copy()
    p_norm = p_norm / float(p_norm.sum())

    # Track best
    best_list = [float(x) for x in p_norm]
    best_score = evaluate_sequence(best_list)
    if not math.isfinite(best_score):
        best_score = 2.0
        best_list = [1.0 / 64.0] * 64

    def consider(arr):
        nonlocal best_list, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0 or a.sum() < 0.01:
                return
            ls = [float(x) for x in a]
            sc = evaluate_sequence(ls)
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_list = ls
                best_score = sc
        except Exception:
            pass

    rng = np.random.default_rng(int(seed))

    # === Phase 1: Coordinate descent on parent (40% of budget) ===
    phase1_end = t0 + 0.40 * (budget - 15.0)
    phase1_end = min(phase1_end, deadline - 60.0)
    if phase1_end > t0 + 10.0 and n_par >= 100:
        try:
            work = p_norm.copy()
            work, _ = _coord_descent(work, phase1_end, consider, rng,
                                     max_flat=6, G_coarse=40, G_fine=32)
        except Exception:
            pass

    # Use best found so far
    work_arr = np.array(best_list, dtype=np.float64)
    work_arr = np.maximum(work_arr, 0.0)
    ws = float(work_arr.sum())
    if ws > 0:
        work_arr /= ws

    # === Phase 2: LSE smooth step + coordinate descent (25%) ===
    phase2_end = t0 + 0.65 * (budget - 15.0)
    phase2_end = min(phase2_end, deadline - 40.0)
    if phase2_end > time.monotonic() + 20.0 and n_par <= 4096:
        lse_end = min(phase2_end - 30.0, time.monotonic() + 40.0)
        if lse_end > time.monotonic() + 10.0:
            try:
                work = _lse_smooth_step(work_arr, lse_end, consider, rng)
                if work is not None and work.size == n_par:
                    ws2 = float(work.sum())
                    if ws2 > 0:
                        work /= ws2
                        consider(work)
                        work_arr = work.copy()
            except Exception:
                pass

        # Coordinate descent after LSE
        cd_end = phase2_end
        if cd_end > time.monotonic() + 15.0:
            try:
                work_arr, _ = _coord_descent(work_arr, cd_end, consider, rng,
                                             max_flat=5, G_coarse=44, G_fine=36)
            except Exception:
                pass

    # === Phase 3: Try different lengths (20%) ===
    work_arr = np.array(best_list, dtype=np.float64)
    work_arr = np.maximum(work_arr, 0.0)
    ws = float(work_arr.sum())
    if ws > 0:
        work_arr /= ws

    phase3_end = t0 + 0.85 * (budget - 15.0)
    phase3_end = min(phase3_end, deadline - 30.0)

    if phase3_end > time.monotonic() + 60.0:
        # Try a longer length
        try_longer = min(8192, max(n_par * 2, 4096))
        if try_longer != n_par and try_longer <= 8192:
            p_long = _resample(work_arr, try_longer)
            consider(p_long)
            seg_end = min(phase3_end - 30.0, time.monotonic() + 0.5 * (phase3_end - time.monotonic()))
            if seg_end > time.monotonic() + 20.0:
                try:
                    p_long, _ = _coord_descent(p_long, seg_end, consider, rng,
                                               max_flat=5, G_coarse=36, G_fine=28)
                except Exception:
                    pass

        # Try a shorter length
        try_shorter = max(512, n_par // 2)
        if try_shorter != n_par and try_shorter != try_longer:
            p_short = _resample(work_arr, try_shorter)
            consider(p_short)
            seg_end = min(phase3_end - 15.0, time.monotonic() + 0.4 * (phase3_end - time.monotonic()))
            if seg_end > time.monotonic() + 15.0:
                try:
                    p_short, _ = _coord_descent(p_short, seg_end, consider, rng,
                                                max_flat=5, G_coarse=36, G_fine=28)
                except Exception:
                    pass

    # === Phase 4: Final polish on best (10%) ===
    work_arr = np.array(best_list, dtype=np.float64)
    work_arr = np.maximum(work_arr, 0.0)
    ws = float(work_arr.sum())
    if ws > 0:
        work_arr /= ws

    phase4_end = min(deadline - 10.0, t0 + budget - 12.0)
    if phase4_end > time.monotonic() + 10.0:
        try:
            work_arr, _ = _coord_descent(work_arr, phase4_end, consider, rng,
                                         max_flat=4, G_coarse=48, G_fine=40)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_list, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            # Fallback to parent
            ls = [float(x) for x in p_norm]
            final_sc = evaluate_sequence(ls)
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
