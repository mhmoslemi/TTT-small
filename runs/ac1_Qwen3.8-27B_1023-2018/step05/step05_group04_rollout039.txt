    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 20.0  # 20s buffer for safety
    
    # Get parent
    try:
        parent = np.asarray(globals()['height_sequence_1'], dtype=np.float64).ravel()
    except:
        parent = np.ones(3072, dtype=np.float64)
    parent = np.where(np.isfinite(parent), 0.0, parent)
    parent = np.clip(parent, 0, 1000)
    if parent.size == 0 or parent.sum() < 0.01:
        parent = np.ones(3072, dtype=np.float64)
    
    n = parent.size
    rng = np.random.default_rng(seed)
    
    # Normalize
    p = parent / parent.sum()
    c = _conv(p)
    best_score = 2.0 * n * float(c.max())
    best_p = p.copy()
    
    def consider(p_trial):
        nonlocal best_p, best_score
        p_trial = np.asarray(p_trial, dtype=np.float64)
        p_trial = np.maximum(p_trial, 0)
        s = p_trial.sum()
        if s < 1e-15 or not np.isfinite(s):
            return
        p_trial = p_trial / s
        if not np.all(np.isfinite(p_trial)):
            return
        c_trial = _conv(p_trial)
        sc = 2.0 * n * float(c_trial.max())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p_trial.copy()
    
    # Phase 1: LSE optimization
    lse_budget = min(80.0, 0.08 * budget)
    lse_end = t0 + lse_budget
    if lse_end > t0 + 10:
        p = _lse_optimize(p, lse_end, n)
        consider(p)
    
    # Phase 2: Main coordinate descent
    p = best_p.copy()
    cd_end = deadline - 5
    p = _cd(p, cd_end, n, rng, consider)
    consider(p)
    
    # Phase 3: Random restarts if time permits
    rem = deadline - 5 - time.monotonic()
    restarts_done = 0
    while rem > 45 and restarts_done < 10:
        # Generate a perturbed version
        sigma = 0.03 * (1 + restarts_done * 0.3)
        p_rand = best_p * np.exp(rng.normal(0, sigma, n))
        p_rand = np.maximum(p_rand, 0)
        p_rand /= p_rand.sum()
        consider(p_rand)
        
        alloc = min(rem * 0.4, 30.0)
        restarts_done += 1
        if alloc > 5:
            p2 = _cd(p_rand, time.monotonic() + alloc, n, rng, consider)
            consider(p2)
        rem = deadline - 5 - time.monotonic()
    
    # Final return
    out = np.clip(best_p, 0, 1000)
    if out.size == 0 or out.sum() < 0.01:
        return [1.0, 1.0, 1.0, 1.0]
    return [float(x) for x in out]
