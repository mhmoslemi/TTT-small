$\min_{x \geq 0} \frac{\max(M_{const}, \max_{j=0}^{n-1} [b_j + w \cdot f_j])}{(1+w)^2}$

    p1 = lbfgs_lse(best_arr, phase1_end, temps)

    # Phase 2: Coordinate descent (50% of time)
    phase2_end = t0 + 0.90 * (hard - t0)
    p2 = coord_desc(best_arr, phase2_end, G=45, seed=seed)
    consider(p2)

    # Phase 3: Final polish (remaining time)
    now = time.monotonic()
    if now < hard - 30.0 and best_arr.size == n:
        rem = hard - now
        # Fine LSE
        if rem > 60.0:
            lstop = min(hard - 40.0, now + 0.40 * rem)
            try:
                p3 = lbfgs_lse(best_arr, lstop, [0.002, 0.0008, 0.0003, 0.0001])
                consider(p3)
            except Exception:
                pass

        # Subgradient
        now = time.monotonic()
        if now < hard - 20.0:
            sg_stop = min(hard - 10.0, now + max(15.0, 0.15 * (hard - now)))
            try:
                p4 = subgradient_polish(best_arr, sg_stop, max_iters=2000)
                consider(p4)
            except Exception:
                pass

        # Final coordinate descent
        now = time.monotonic()
        if now < hard - 10.0:
            cstop = min(hard - 3.0, now + max(8.0, 0.25 * (hard - now)))
            try:
                p5 = coord_desc(best_arr, cstop, G=50, seed=seed + 999)
                consider(p5)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = evaluate_sequence(ls)
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
