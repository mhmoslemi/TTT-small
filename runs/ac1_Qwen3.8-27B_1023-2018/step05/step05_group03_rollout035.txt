def _lse_gradient(p, c, T, conv_len, n):
    """Compute gradient of T*log(sum exp(c/T)) w.r.t. p using FFT cross-correlation."""
    if T <= 0 or not math.isfinite(T):

    scaled = (c - c.max()) / T

    G_w = rfft(w, cross_size)
    cross = irfft(F_p * G_w, cross_size)[:3*n-2]
    grad = 2.0 * cross[n-1:2*n]
    return grad, float(c.max() + T * math.log(s_exp / conv_len)) if s_exp > 0 else float(c.max())

    # Final validation
    final_ls = [float(max(0.0, min(1000.0, x))) for x in best_p]
    try:
        fs = evaluate_sequence(final_ls)
    except Exception:
        fs = math.inf
    
    if not math.isfinite(fs):
        # Fallback to parent
        final_ls = [float(max(0.0, min(1000.0, x))) for x in p0]
        try:
            fs = evaluate_sequence(final_ls)
        except Exception:
            fs = math.inf
    
    if not math.isfinite(fs):
        return [1.0]
    
    return final_ls
