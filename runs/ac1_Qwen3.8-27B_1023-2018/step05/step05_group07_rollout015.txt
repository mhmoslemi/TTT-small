class _LPSolver:
    """Prebuild LP structure for a given length."""
    def __init__(self, n):
        self.n = n
        k = np.arange(2 * n - 1)[:, None]
        j = np.arange(n)[None, :]
        d = k - j
        self.mask = (d >= 0) & (d < n)
        self.flat_idx = np.where(self.mask)
        self.d_vals = d[self.mask]
        self.nnz = len(self.flat_idx)

    def matvec(self, a, x):
        """Compute A @ x where A is the conv matrix for coefficients a."""
        out = np.zeros(2 * self.n - 1)
        vals = a[self.d_vals] * x[self.flat_idx[1]]
        np.add.at(out, self.flat_idx[0], vals)
        return out

    def solve_lp(self, a, M, time_limit=4.0):
        n = self.n
        A = np.zeros((2 * n - 1, n))
        A[self.flat_idx] = a[self.d_vals]
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                          method='highs',
                          options={'time_limit': time_limit,
                                   'primal_feasibility_tolerance': 1e-12,
                                   'dual_feasibility_tolerance': 1e-12})
            if res.status == 0 and res.x is not None:
                return np.clip(res.x, 0, None)
        except Exception:
            pass

def _smooth_grad(conv, a, eps=0.01):
    """Softmax-weighted gradient of max(conv)."""
    n = len(a)
    mc = conv.max()
    log_p = (conv - mc) / eps
    log_p -= log_p.max()
    p = np.exp(log_p)
    p /= p.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for m in range(lo, hi):
            idx = k - m
            if 0 <= idx < n:
                grad[m] += 2.0 * pk * a[idx]
    return grad

        hi = min(n, k + 1)
        for m in range(lo, hi):

            if 0 <= idx < n:

    momentum = 0.0
    stuck = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 8:
            break

        conv = np.convolve(a, a)
        M = float(conv.max())

        if M < best_M:
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 80:
            sigma = min(0.04 + 0.01 * (stuck // 80), 0.3)
            a = a * np.exp(rng.normal(0, sigma, n))
            a = _project_simplex(a)
            momentum = 0.0
            stuck = 0
            if sigma > 0.25:
                break
            continue

        moved = False

        # --- LP best response ---
        if it % 2 == 0 or stuck > 10:
            g = lp.solve_lp(a, M * (1 + 1e-9), time_limit=3.0)
            if g is not None and g.sum() > 1e-10:
                g /= g.sum()
                if float(np.max(np.abs(g - a))) > 1e-8:
                    cand, val = _line_search_dir(a, g, M)
                    if cand is not None and val < M * (1 - 1e-9):
                        momentum = 0.7 * momentum + 0.3 * (cand - a)
                        a = _project_simplex(a + momentum)
                        moved = True

        # --- Smooth gradient step ---
        if not moved:
            eps = 0.005 * M * (1 + 0.01 * it)
            grad = _smooth_grad(conv, a, eps=eps)
            grad_tan = grad - np.mean(grad)
            gn = float(np.linalg.norm(grad_tan))
            if gn > 1e-14:
                grad_tan /= gn
                for eta in [0.04, 0.02, 0.01, 0.005, 0.002]:
                    cand = _project_simplex(a - eta * grad_tan)
                    v = float(np.convolve(cand, cand).max())
                    if v < M * (1 - 1e-9):
                        a = cand
                        momentum = 0.5 * momentum + 0.5 * (cand - a) * (eta / 0.04)
                        moved = True
                        break

        # --- Hard subgradient step ---
        if not moved:
            grad_h = _hard_subgrad(conv, a, frac=0.995)
            grad_h -= np.mean(grad_h)
            gn = float(np.linalg.norm(grad_h))
            if gn > 1e-14:
                grad_h /= gn
                for eta in [0.03, 0.015, 0.008, 0.004]:
                    cand = _project_simplex(a - eta * grad_h)
                    v = float(np.convolve(cand, cand).max())
                    if v < M * (1 - 1e-9):
                        a = cand
                        moved = True
                        break

        # --- Small random perturbation as last resort ---
        if not moved:
            a = a * np.exp(rng.normal(0, 0.01, n))
            a = _project_simplex(a)
            momentum = 0.0

    return best_a, best_M


def _make_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(parent.copy() if len(parent) == n else np.ones(n) / n)

    x = np.linspace(0, 1, n)
    # Gaussian bumps at various positions and widths
    for pos in [0.5, 0.3, 0.7, 0.2, 0.8]:
        for w in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            v = np.exp(-0.5 * ((x - pos) / w) ** 2)
            v /= v.sum()
            inits.append(v)
    # Double peaks
    for d in [0.05, 0.1, 0.15, 0.2, 0.25]:
        v = np.exp(-0.5 * ((x - 0.5 + d) / 0.12) ** 2) + np.exp(-0.5 * ((x - 0.5 - d) / 0.12) ** 2)
        v /= v.sum()
        inits.append(v)
    # Cosine windows
    for m in [1, 2, 3]:
        v = np.maximum(np.cos(m * np.pi * (x - 0.5)) ** 2, 0) + 1e-6
        v /= v.sum()
        inits.append(v)
    # Power-law decay from one end
    for alpha in [1, 1.5, 2, 3]:
        v = (1 - x) ** alpha + 1e-6
        v /= v.sum()
        inits.append(v)
        v2 = x ** alpha + 1e-6
        v2 /= v2.sum()
        inits.append(v2)
    # Random (seeded)
    for _ in range(6):
        v = np.abs(rng.normal(1, 0.5, n)) + 0.01
        v /= v.sum()
        inits.append(v)
    # Smooth random (low-pass)
    for _ in range(4):
        v = rng.uniform(0.5, 1.5, n)
        # Simple moving average smoothing
        for _ in range(3):
            v = np.convolve(v, np.ones(5)/5, mode='same')
        v = np.maximum(v, 0.01)
        v /= v.sum()
        inits.append(v)
    return inits


def _interpolate_to_len(parent, new_len):
    """Interpolate parent to a new length."""
    old_len = len(parent)
    if old_len == new_len:
        return parent.copy()
    x_old = np.linspace(0, 1, old_len)
    x_new = np.linspace(0, 1, new_len)
    new_vals = np.interp(x_new, x_old, parent)
    new_vals = np.clip(new_vals, 0, 1000)
    s = new_vals.sum()
    if s < 1e-10:
        return np.ones(new_len) / new_len
    return new_vals / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 20.0

    # Get parent
    try:
        parent = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(200)
    parent = np.clip(parent, 0, 1000)
    if parent.size == 0 or not np.all(np.isfinite(parent)):
        parent = np.ones(200)

    n_par = len(parent)
    p_sum = parent.sum()
    pn = parent / p_sum if p_sum > 1e-15 else np.ones(n_par) / n_par

    best_a = pn.copy()
    best_score = _score(pn)

    def try_update(cand):
        nonlocal best_a, best_score
        try:
            cand = np.clip(np.asarray(cand, dtype=np.float64), 0, 1000)
            s = cand.sum()
            if s < 0.01:
                return
            sc = 2.0 * len(cand) * float(np.convolve(cand, cand).max()) / (s * s)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = cand.copy()
        except Exception:
            pass

    rng = np.random.default_rng(seed)

    # ===== Phase 1: Optimize parent (n_par) =====
    lp_par = _LPSolver(n_par)
    t1 = t0 + min(200.0, budget_s * 0.18)
    try:
        r, _ = _optimize_core(pn, lp_par, t1, seed=1, max_iters=400)
        try_update(r)
    except Exception:
        pass

    # ===== Phase 2: Diverse initializations at parent length =====
    inits = _make_inits(n_par, pn, rng)
    t2 = t0 + min(500.0, budget_s * 0.42)
    remaining = max(t2 - time.monotonic(), 5.0)
    per_init = max(4.0, remaining / len(inits))

    for i, init in enumerate(inits):
        if time.monotonic() > t2 - 5.0:
            break
        alloc = min(time.monotonic() + per_init, t2 - 3.0)
        try:
            r, _ = _optimize_core(init, lp_par, alloc, seed=1000 + i, max_iters=200)
            try_update(r)
        except Exception:
            pass
        # Also try from current best
        if i % 5 == 4 and time.monotonic() < t2 - 10:
            try:
                cur_best = _project_simplex(best_a.copy())
                if len(cur_best) == n_par:
                    alloc2 = min(time.monotonic() + per_init * 0.5, t2 - 3.0)
                    r2, _ = _optimize_core(cur_best, lp_par, alloc2, seed=5000 + i, max_iters=150)
                    try_update(r2)
            except Exception:
                pass

    # ===== Phase 3: Deep polish on best =====
    t3 = t0 + min(800.0, budget_s * 0.65)
    if len(best_a) == n_par:
        ab = _project_simplex(best_a.copy())
        try:
            r, _ = _optimize_core(ab, lp_par, t3, seed=9999, max_iters=500)
            try_update(r)
        except Exception:
            pass

    # ===== Phase 4: Try different lengths =====
    if time.monotonic() < deadline - 100:
        dn_list = [5, -5, 10, -10, 15, -15, 20, -20, 30, -30, 40, -40, 50, -50, 70, 100]
        for dn in dn_list:
            if time.monotonic() > deadline - 40:
                break
            nt = n_par + dn
            if nt < 50 or nt > 600:
                continue
            init_new = _interpolate_to_len(pn, nt)
            lp_new = _LPSolver(nt)
            alloc = min(time.monotonic() + 40.0, deadline - 30)
            try:
                r, _ = _optimize_core(init_new, lp_new, alloc, seed=7000 + nt, max_iters=200)
                try_update(r)
            except Exception:
                pass

    # ===== Phase 5: Final polish =====
    if time.monotonic() < deadline - 20:
        nf = len(best_a)
        lf = _LPSolver(nf)
        af = _project_simplex(best_a.copy())
        try:
            r, _ = _optimize_core(af, lf, deadline, seed=424242, max_iters=400)
            try_update(r)
        except Exception:
            pass

    # ===== Final validation with exact evaluator =====
    try:
        result = [float(v) for v in best_a]
        fs = evaluate_sequence(result)
        if not np.isfinite(fs) or fs <= 0:
            raise ValueError("bad score")
    except Exception:
        # Fallback
        result = [1.0] * 100
        fs = evaluate_sequence(result)
        if not np.isfinite(fs):
            result = [1.0]
    return result
