import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p

def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.asarray(c, dtype=np.float64)

def _score_norm(p):
    """Score for normalized p (sum=1): 2*n*max(conv(p,p))"""
    n = p.size
    if n == 0:
        return float('inf')
    c = _conv_self(p)
    m = float(c.max())
    if not math.isfinite(m):
        return float('inf')
    return float(2.0 * n * m)

def _lse_step(p, T, steps, stop_time):
    """Run L-BFGS on the LSE-smoothed objective for a fixed temperature T."""
    n = p.size
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    
    p = np.maximum(p, 1e-30).copy()
    s = p.sum()
    p = p / s
    u = np.log(p)
    u -= u.mean()
    
    def softmax(uu):
        m = uu.max()
        e = np.exp(uu - m)
        return e / e.sum()
    
    def lse_obj(uu, T):
        pp = softmax(uu)
        F = rfft(pp, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        cmax = c.max()
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        s_exp = exps.sum()
        f = cmax + T * np.log(s_exp / conv_len)
        
        # Gradient
        w = exps / s_exp
        # Cross-correlation: q_j = sum_k w_k * pp_{k-j} (for j in [0, n-1])
        # This is conv(pp_flipped, w)[n-1:2n-1]
        cross_size = next_fast_len(3 * n - 2)
        Fp = rfft(pp[::-1], cross_size)
        Gw = rfft(w, cross_size)
        cr = irfft(Fp * Gw, cross_size)[:3*n-2]
        q = cr[n-1:2*n-1]
        
        gdp = 2.0 * q
        tdp = np.dot(pp, gdp)
        grad = pp * (gdp - tdp)
        return f, grad
    
    best_f = 1e100
    best_u = u.copy()
    
    # Simple gradient descent with momentum (faster than L-BFGS in Python)
    m = np.zeros(n)
    lr = 0.1
    for step in range(steps):
        if time.monotonic() > stop_time:
            break
        f, g = lse_obj(u, T)
        if not math.isfinite(f):
            break
        m = 0.9 * m + 0.1 * g
        u = u - lr * m
        # Renormalize u (mean zero for softmax invariance)
        u -= u.mean()
        pp = softmax(u)
        sc = _score_norm(pp)
        if math.isfinite(sc) and sc < best_f:
            best_f = sc
            best_u = u.copy()
    
    return softmax(best_u)

    return best_p, best_score


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 30.0
    if hard <= t0 + 5.0:
        hard = t0 + max(3.0, budget - 2.0)

    parent = _get_parent()
    n_work = int(parent.size)
    if n_work < 128:
        n_work = 384
        parent = np.full(n_work, 1.0 / n_work)
    if n_work > 4096:
        n_work = 3072
        parent = _resample(parent, n_work)

    best_arr = parent.copy()
    best_score = float('inf')

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()

    consider(parent)

    # Phase 1: LSE warm-start (short)
    lse_time = min(40.0, 0.05 * budget)
    lse_end = t0 + lse_time
    if lse_end > time.monotonic() + 8.0:
        try:
            p_lse = _lse_refine(_resample(best_arr, n_work), lse_end)
            if p_lse is not None and p_lse.size == n_work and p_lse.sum() > 0:
                consider(p_lse)
        except Exception:
            pass

    # Phase 2: Main coordinate descent
    # Use the best found so far as starting point
    p_main = _resample(best_arr, n_work)
    cd_end = hard - 40.0
    if cd_end > time.monotonic() + 60.0:
        try:
            p_cd, _ = _cd_optimize(p_main, cd_end, consider=consider, seed=int(seed) + 12345)
            if p_cd is not None and p_cd.size > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                consider(p_cd)
        except Exception:
            pass

    # Phase 3: Try a different initialization + CD
    rem = hard - 35.0 - time.monotonic()
    if rem > 80.0:
        # Create a perturbed version of the best
        rng3 = np.random.default_rng(int(seed) + 777)
        p_pert = _resample(best_arr, n_work)
        # Add a small structured perturbation
        n_osc = rng3.integers(2, 8)
        phase = rng3.uniform(0, 2 * math.pi)
        osc = np.sin(2.0 * math.pi * n_osc * np.arange(n_work) / n_work + phase)
        p_pert = p_pert * np.exp(0.02 * osc)
        p_pert = np.maximum(p_pert, 0.0)
        p_pert /= p_pert.sum()
        consider(p_pert)
        
        cd3_end = min(hard - 35.0, time.monotonic() + rem * 0.55)
        if cd3_end > time.monotonic() + 30.0:
            try:
                p_3, _ = _cd_optimize(p_pert, cd3_end, consider=consider, seed=int(seed) + 8888)
                if p_3 is not None and p_3.size > 0 and np.all(np.isfinite(p_3)) and p_3.sum() > 0:
                    consider(p_3)
            except Exception:
                pass

    # Phase 4: Final polish CD on the overall best
    rem = hard - 10.0 - time.monotonic()
    if rem > 25.0:
        p_final = _resample(best_arr, n_work)
        try:
            p_f, _ = _cd_optimize(p_final, hard - 10.0, consider=consider, seed=int(seed) + 9999)
            if p_f is not None and p_f.size > 0 and np.all(np.isfinite(p_f)) and p_f.sum() > 0:
                consider(p_f)
        except Exception:
            pass

    # Final validation with the exact evaluator
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
