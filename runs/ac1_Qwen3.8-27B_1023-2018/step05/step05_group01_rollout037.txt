                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val - 1e-15:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= 12.0 * S:
                w = best_x - v
                if abs(w) > 1e-16:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-16:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 4.0:
                    sigma = 0.015 + 0.015 * min(flat, 12) / 12.0
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    else:
                        break
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    flat = 0
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        sc = float(_score_fft(p))
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 20.0
    if hard <= t0 + 2.0:
        hard = t0 + max(2.0, budget * 0.9)
    final_reserve = max(10.0, 0.05 * (hard - t0))
    phase_limit = hard - final_reserve

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    def make_perturb(src, n, sigma, sd):
        rng = np.random.default_rng(sd)
        p = _resample_mid(src, n)
        p = p * np.exp(rng.normal(0, sigma, n))
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if s > 0:
            p /= s
        else:
            p = _resample_mid(src, n)
        return p

    def run_opt(n, init_arr, end_time, sd, lse_frac=0.4, do_lse=True, do_coord=True):
        nonlocal best_arr, best_score
        p = _resample_mid(init_arr, n)
        if p.size != n or float(p.sum()) <= 0:
            p = np.full(n, 1.0 / n)
        p = p / float(p.sum())
        consider(p)

        if time.monotonic() > end_time - 15.0:
            return
        phase_len = max(0.0, float(end_time) - time.monotonic())

        if do_lse and phase_len > 40.0:
            lse_stop = time.monotonic() + lse_frac * phase_len
            lse_stop = min(lse_stop, float(end_time) - 15.0)
            if lse_stop > time.monotonic() + 15.0:
                try:
                    p2 = _lse_optimize(p, lse_stop, consider=consider, seed=sd)
                    if p2 is not None and p2.size == n and float(p2.sum()) > 0:
                        p = p2 / float(p2.sum())
                        consider(p)
                except Exception:
                    pass

        rem = float(end_time) - time.monotonic()
        if do_coord and rem > 15.0:
            cstop = min(float(end_time) - 1.0, time.monotonic() + rem - 1.0)
            if cstop > time.monotonic() + 10.0:
                try:
                    p3, _ = _coord_polish(p, cstop, seed=sd, consider=consider)
                    if p3 is not None and p3.size == n and float(p3.sum()) > 0:
                        p = p3 / float(p3.sum())
                        consider(p)
                except Exception:
                    pass
        return p

    n_parent = int(best_arr.size)

    # Phase 1: Main optimization at parent length (50% of available time)
    p1_end = t0 + 0.50 * (phase_limit - t0)
    p1_end = min(p1_end, phase_limit)
    if p1_end > time.monotonic() + 30.0:
        run_opt(n_parent, best_arr, p1_end, int(seed + 100), lse_frac=0.45)

    # Phase 2: Multi-start perturbations (25%)
    if time.monotonic() < phase_limit - 60.0:
        rng_m = np.random.default_rng(seed + 777)
        for pr in range(3):
            if time.monotonic() > phase_limit - 30.0:
                break
            sigma = 0.03 + 0.02 * pr
            pert = make_perturb(best_arr, n_parent, sigma, int(seed + 3000 + pr * 500))
            sub_end = min(phase_limit - 25.0, time.monotonic() + 0.28 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 30.0:
                run_opt(n_parent, pert, sub_end, int(seed + 5000 + pr * 700), lse_frac=0.4)

    # Phase 3: Structured initialization (15%)
    if time.monotonic() < phase_limit - 50.0:
        # Try a triangular/bump initialization
        x = (np.arange(n_parent, dtype=np.float64) + 0.5) / n_parent
        bump = np.exp(-4.0 * (x - 0.5) ** 2)
        bump_sm = _resample_mid(best_arr, n_parent) * 0.7 + bump / bump.sum() * 0.3
        bump_sm = np.maximum(bump_sm, 0.0)
        bump_sm /= bump_sm.sum()
        sub_end3 = min(phase_limit - 20.0, time.monotonic() + 0.2 * (phase_limit - time.monotonic()))
        if sub_end3 > time.monotonic() + 30.0:
            run_opt(n_parent, bump_sm, sub_end3, int(seed + 9000), lse_frac=0.45)

    # Phase 4: Final squeeze on best (10%)
    final_end = min(hard - 10.0, t0 + budget - 6.0)
    if final_end > time.monotonic() + 20.0 and best_arr.size > 0:
        nbest = int(best_arr.size)
        rem = final_end - time.monotonic()
        if rem > 40.0 and nbest <= 6000:
            cstop = min(final_end - 3.0, time.monotonic() + rem * 0.8)
            if cstop > time.monotonic() + 15.0:
                try:
                    p3, _ = _coord_polish(best_arr, cstop, seed=int(seed + 123456789), consider=consider)
                    if p3 is not None and p3.size == nbest and float(p3.sum()) > 0:
                        p3 = p3 / float(p3.sum())
                        consider(p3)
                except Exception:
                    pass
            # Short LSE refinement
            rem2 = final_end - time.monotonic()
            if rem2 > 30.0:
                lstop = min(final_end - 2.0, time.monotonic() + rem2 * 0.6)
                if lstop > time.monotonic() + 15.0:
                    try:
                        p4 = _lse_optimize(best_arr, lstop, consider=consider, seed=int(seed + 99999))
                        if p4 is not None and p4.size == nbest and float(p4.sum()) > 0:
                            p4 = p4 / float(p4.sum())
                            consider(p4)
                    except Exception:
                        pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
