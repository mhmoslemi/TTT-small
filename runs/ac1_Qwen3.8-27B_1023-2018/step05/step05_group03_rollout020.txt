import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _fft_conv(a, cache_size=None):
    n = a.size
    L = 2 * n - 1
    size = cache_size if cache_size is not None else next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return c


def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1, returns 2n*max_conv."""
    n = a.size
    c = _fft_conv(a)
    return float(2.0 * n * c.max())


def _softmax(u):
    u = u - u.max()
    e = np.exp(u)
    s = e.sum()
    if s < 1e-300:
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.full(a.size, 1.0 / a.size)
    return a / s


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:
        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if not math.isfinite(s) or s <= 1e-12:
        p = np.ones(3072, dtype=np.float64)
    return p


def _lse_objective(u, T, n, conv_size):
    a = _softmax(u)
    c = _fft_conv(a, conv_size)
    cmax = float(c.max())
    if T < 1e-15:
        T = 1e-15
    scaled = (c - cmax) / T
    exps = np.exp(scaled)
    sum_exp = float(exps.sum())
    if not math.isfinite(sum_exp) or sum_exp <= 0:
        im = int(np.argmax(c))
        exps = np.zeros_like(c)
        exps[im] = 1.0
        sum_exp = 1.0
    w = exps / sum_exp
    f = cmax + T * math.log(sum_exp / c.size)
    
    # Gradient: g_i = 2 * sum_j w[i+j] * a[j]
    N3 = next_fast_len(3 * n - 2)
    B = rfft(a[::-1], N3)
    W3 = rfft(w, N3)
    conv_bw = irfft(B * W3, N3)[:3 * n - 2]
    g = 2.0 * conv_bw[n - 1:2 * n - 1]
    
    dot = float(np.dot(g, a))
    grad_u = a * (g - dot)
    grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
    return f, grad_u


def _lse_optimize(p, stop_time, consider, fine_only=False):
    n = p.size
    conv_size = next_fast_len(2 * n - 1)
    u = np.log(np.maximum(p, 1e-15))
    u -= u.mean()
    
    if fine_only:
        stages = ((0.0008, 0.25), (0.0003, 0.30), (0.0001, 0.45))
    else:
        stages = (
            (0.012, 0.12),
            (0.004, 0.15),
            (0.0012, 0.18),
            (0.0004, 0.20),
            (0.00012, 0.22),
            (0.00004, 0.13),
        )

    start = time.monotonic()
    avail = stop_time - start - 5.0
    if avail <= 10:
        return p

    total_w = sum(w for _, w in stages)
    cum = 0.0
    for frac, wt in stages:
        if time.monotonic() > stop_time - 6.0:
            break
        cum += wt
        end_stage = min(stop_time - 5.0, start + avail * (cum / total_w))
        if end_stage <= time.monotonic() + 5.0:
            continue

        c = _fft_conv(p, conv_size)
        cmax = float(c.max())
        T = max(1e-15, frac * cmax)
        
        best_u = u.copy()
        best_f = None
        try:
            best_f, _ = _lse_objective(u, T, n, conv_size)
        except Exception:
            best_f = math.inf

        stale = 0
        while time.monotonic() < end_stage - 2.0:
            maxit = 12 if n > 2000 else 20
            try:
                res = minimize(
                    _lse_objective, best_u, args=(T, n, conv_size),
                    method="L-BFGS-B", jac=True,
                    options={"maxiter": maxit, "gtol": 1e-10, "ftol": 1e-15,
                             "maxls": 12, "maxcor": 15, "disp": False, "maxdisp": 0}
                )
            except Exception:
                break
            if res.x is not None and math.isfinite(res.fun) and res.fun < best_f - 1e-14:
                best_f = float(res.fun)
                best_u = np.asarray(res.x, dtype=np.float64).ravel().copy()
                stale = 0
            else:
                stale += 1
                if stale >= 2:
                    break
            if getattr(res, "status", 1) == 0:
                break

        p_new = _softmax(best_u)
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if math.isfinite(ss) and ss > 1e-300:
            p_new /= ss
            sc_new = _score(p_new)
            if math.isfinite(sc_new):
                consider(p_new)
                p = p_new.copy()
        u = np.log(np.maximum(p, 1e-15))
        u -= u.mean()

    return p


def _coord_descent(p, stop_time, G, seed, consider):

    rng = np.random.default_rng(seed)
    p = _normalize(p.copy())
    c = _fft_conv(p)
    c = np.maximum(c, 0.0)
    S = 1.0
    cur_score = float(2.0 * n * c.max())
    
    sweep = 0
    flat = 0
    while sweep < 999999 and time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)

        for pos in range(n):
            if time.monotonic() > stop_time - 0.5:
                break
            i = int(order[pos])
            v = float(p[i])
            L = 2 * n - 1
            start = i
            end = min(i + n, L)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end].copy()
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())

                    max_const = mc
            if end < L:

                    max_const = mc

            m3 = max(12, int(G * 0.35))
            linhi = min(xhi, max(6.0 * S / n, 0.002))

            np.maximum(aff, max_const, out=aff)

                    np.concatenate([
                        [best_x],
                        np.linspace(lo, hi, 40),

                    ]), 0.0, xhi))
                den2 = S + (fine - v)
                valid2 = den2 > 1e-14
                aff2 = base[None, :] + fine[:, None] * dv[None, start:end]
                if 0 <= di < aff2.shape[1]:
                    aff2[:, di] = base[di] + fine * dv[start + di] + fine * fine
                np.maximum(aff2, max_const, out=aff2)
                mv2 = np.max(aff2, axis=1)

                d = best_x - v

                    c[start:end] += d * dv[start:end]

                        c[diag] += d * d
                    p[i] = best_x
                    dv[i] = 2.0 * best_x
                    S += d
                    cur_score = best_val
                    improved = True

        # Renormalize
