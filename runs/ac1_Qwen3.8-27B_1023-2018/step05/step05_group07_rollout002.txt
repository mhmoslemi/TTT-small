    # ===== Phase 2: Diverse initializations (35% time) =====
    t2 = t0 + min(750.0, budget * 0.58)
    x = np.linspace(0, 1, n_parent)
    pn = parent / max(parent.sum(), 1e-15)

    inits = []
    # Gaussian bumps at various positions
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sig in [0.05, 0.1, 0.15, 0.25, 0.4]:
            v = np.exp(-0.5 * ((x - center) / sig) ** 2)
            inits.append(v / v.sum())
    # Symmetric pairs
    for w in [0.15, 0.25, 0.35, 0.42]:
        v = np.exp(-0.5 * ((x - w) / 0.07) ** 2) + np.exp(-0.5 * ((x - (1 - w)) / 0.07) ** 2)
        inits.append(v / v.sum())
    # Cosine windows
    for m in [1, 2, 3, 5]:
        v = np.maximum(np.cos(m * np.pi * (x - 0.5)) ** 2 + 0.005, 0)
        inits.append(v / v.sum())
    # Ramp
    for slope in [0.5, 1.0, 2.0]:
        v = np.maximum(1 - slope * np.abs(x - 0.5), 0.001)
        inits.append(v / v.sum())
    # Blends with parent
    for frac in [0.3, 0.5, 0.7]:
        for init in inits[:4]:
            v = frac * pn + (1 - frac) * init
            inits.append(v / v.sum())
    # Random
    rng_init = np.random.default_rng(7777)
    for _ in range(10):
        v = np.abs(rng_init.normal(1, 0.6, n_parent)) + 0.01
        inits.append(v / v.sum())

    # Deduplicate
    seen = set()
    unique_inits = []
    for v in inits:
        h = tuple((v * 1000).astype(int)[:20])
        if h not in seen:
            seen.add(h)
            unique_inits.append(v)
    inits = unique_inits

    elapsed = time.monotonic() - t0
    remaining = max(t2 - time.monotonic(), 2.0)
    per_init = max(4.0, remaining / max(1, len(inits)))

    for i, init in enumerate(inits):
        if time.monotonic() > t2 - 8:
            break
        alloc = min(time.monotonic() + per_init, t2 - 5)
        if alloc - time.monotonic() < 1:
            break
        try:
            r = _optimize_core(init, alloc, seed=1000 + i, max_stuck=60)
            try_update(r)
        except Exception:
            pass

    # ===== Phase 3: Length exploration (18% time) =====
    t3 = t0 + min(1000.0, budget * 0.77)
    if time.monotonic() < t3 - 30:
        dn_list = [-30, -20, -10, -5, 5, 10, 20, 30, 50]
        for dn in dn_list:
            if time.monotonic() > t3 - 15:
                break
            nt = n_parent + dn
            if nt < 20 or nt > 600:
                continue
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, parent)
            init_new = np.clip(init_new, 0, 1000)
            sn = init_new.sum()
            if sn < 1e-10:
                continue
            init_new = init_new / sn
            alloc = min(time.monotonic() + 40, t3 - 10)
            try:
                r = _optimize_core(init_new, alloc, seed=5000 + nt, max_stuck=50)
                try_update(r)
            except Exception:
                pass

    # ===== Phase 4: Final deep polish (remaining time) =====
    if time.monotonic() < deadline - 15:
        af = np.clip(best_seq, 0, 1000).astype(np.float64)
        sf = af.sum()
        if sf > 1e-15:
            af = af / sf
        try:
            r = _optimize_core(af, deadline - 5, seed=9999, max_stuck=200)
            try_update(r)
        except Exception:
            pass

    # ===== Final validation =====
    try:
        fs = float(evaluate_sequence([float(v) for v in best_seq]))
        if not np.isfinite(fs):
            best_seq = np.ones(100)
            best_score = _score(best_seq)
    except Exception:
        best_seq = np.ones(100)

    return [float(v) for v in best_seq]
