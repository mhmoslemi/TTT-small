So (f*f)[k] = f^T M_k f where M_k[i,j] = δ_{i+j,k}.

Is M_k PSD? M_k is a "reversed identity" matrix (anti-diagonal). Its eigenvalues are ±1 (for even n) or ±1 and 0 (for odd n). So M_k is NOT PSD in general!

This means (f*f)[k] is not a convex function of f! It's indefinite. So the problem is NOT convex in general.

However, for k near the center (k = n-1), M_k is the "exchange matrix" which has eigenvalues +1 and -1. For k near the edges, M_k is more "shifted" and might have different properties.

So the problem is non-convex. This is why the parent uses the LP-based heuristic (best response) rather than solving a QP directly.

import numpy as np
import time
from scipy.optimize import linprog


def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)


def _get_parent():
    try:
        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(200)
    p = np.clip(p, 0, 1000)
    if p.size == 0 or not np.all(np.isfinite(p)):
        p = np.ones(200)
    return p


def _convolve_mat(n, a):
    """Build (2n-1) x n matrix A where (A @ x)[k] = sum_j a[k-j]*x[j]."""
    rows = np.arange(2 * n - 1)[:, None]
    cols = np.arange(n)[None, :]
    diff = rows - cols
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[diff[mask]]
    return A


def _lp_solve(A, M, n, time_limit):
    """Solve max sum(g) s.t. A@ g <= M, g >= 0."""
    c_obj = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0 and res.x is not None:
            x = np.maximum(res.x, 0.0)
            if x.sum() > 1e-12:
                return x
    except Exception:
        pass
    return None


def _gradient_step(n, a, conv, M):
    """Compute subgradient of max_k (f*f)[k] and return step direction."""
    active = np.where(conv >= M * 0.9999)[0]
    if len(active) == 0:
        return None
    subgrad = np.zeros(n)
    use_active = active[:8]
    for k in use_active:
        ks = int(k)
        js = np.arange(n)
        idx = ks - js
        valid = (idx >= 0) & (idx < n)
        subgrad[valid] += 2.0 * a[idx[valid]]
    subgrad /= len(use_active)
    proj = subgrad - subgrad.mean()
    pn = float(np.linalg.norm(proj))
    if pn < 1e-14:
        return None
    return proj / pn


def _line_search_conv(a, direction, n, current_M):
    """Line search: find best t in a + t*direction (renormalized)."""
    best_t = 0.0
    best_val = current_M
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        cand = a + t * direction
        cand = np.maximum(cand, 0.0)
        cs = cand.sum()
        if cs < 1e-12:
            continue
        cand = cand / cs
        v = float(np.convolve(cand, cand).max())
        if v < best_val:
            best_val = v
            best_t = t
    if best_t == 0.0:
        return None, current_M
    return a + best_t * direction, best_val

                            best_t = t
                    if best_t > 0 and best_val < M - 1e-12:

                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        moved = True

        # --- Gradient step (if LP didn't help) ---

                    if cs < 1e-12:
                        continue
                    cand = cand / cs

                    if v < M - 1e-12:
                        a = cand
                        moved = True
                        break

        # --- LP with relaxed constraint (if still stuck) ---
        if not moved and remaining > 4:
            g2 = _lp_solve(A, M * (1 + 1e-4), n, min(2.0, remaining - 5))
            if g2 is not None:

                if g2s > 1e-10:

                            cand = np.maximum(cand, 0.0)

                            if cs < 1e-12:

                                best_t = t
                        if best_t > 0 and best_val < M - 1e-12:
                            a = (1.0 - best_t) * a + best_t * g2
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()
                            moved = True

        if not moved:
            a = a * np.exp(rng.normal(0, 0.015, n))
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
