def _lp_best_response(f, M, max_time_s=45.0):
    """Solve LP dual: min M*sum(y) s.t. A*y >= 1, y >= 0.
    A is n x (2n-1), A[j,j+k]=f[k]. Returns primal g (n-vector) or None."""
    n = len(f)
    L = 2 * n - 1
    t_start = time.monotonic()

    # Build sparse A: n rows, L cols. Row j has f[0..n-1] at cols j..j+n-1.
    # Use block construction for efficiency
    row_idx = np.repeat(np.arange(n), n)
    col_idx = np.arange(n) + np.repeat(np.arange(n), n)
    vals = np.tile(f, n)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(n, L))

    # Dual: min c^T y s.t. -A*y <= -1, y >= 0
    c = np.full(L, M)
    A_ub = -A
    b_ub = np.full(n, -1.0)
    bounds = [(0.0, None)] * L

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': max(1.0, max_time_s - (time.monotonic() - t_start)),
                               'presolve': True})
    except Exception:
        return None

    if time.monotonic() - t_start > max_time_s:
        return None

    if not res.success:
        return None

    # Get primal from dual via complementary slackness, or solve primal directly
    # Primal: max 1^T b s.t. A_p * b <= M*1, b >= 0; A_p is L x n
    # A_p[i,j] = f[i-j] if 0<=i-j<n
    # This is just the transpose structure. Let's solve primal directly.
    # Actually, easier: use the dual solution to reconstruct primal.
    # Complementary slackness: y[i] > 0 => (A_p b)[i] = M; b[j] > 0 => (A y)[j] = 1
    # This is complex. Let's just solve the primal.
    # Primal: max sum(b) s.t. A_p b <= M*ones(L), b >= 0
    # A_p is L x n with A_p[i,j] = f[i-j] if 0<=i-j<n else 0

    # Build A_p sparse (L x n)
    row_idx_p = np.repeat(np.arange(n), n) 
    # For column j, nonzero rows are j, j+1, ..., j+n-1 with values f[0], f[1], ..., f[n-1]
    row_idx_p = np.arange(n) + np.repeat(np.arange(n), n)  # row = j+k for j in 0..n-1, k in 0..n-1
    col_idx_p = np.repeat(np.arange(n), n)
    vals_p = np.tile(f, n)
    A_p = csr_matrix((vals_p, (row_idx_p, col_idx_p)), shape=(L, n))

    try:
        res_p = linprog(-np.ones(n), A_ub=A_p, b_ub=np.full(L, M),
                        bounds=[(0.0, None)] * n, method='highs',
                        options={'time_limit': max(1.0, max_time_s - 5.0 - (time.monotonic() - t_start)),
                                 'presolve': True})
    except Exception:
        return None

    if time.monotonic() - t_start > max_time_s:
        return None

    if not res_p.success or res_p.x is None:
        return None

    g = np.asarray(res_p.x, dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s < 1e-300:
        return None
    return g / s


def _lbfgs_lse(p, stop_time, consider=None, fine_only=False):

    n = int(p.size)
    if n == 0:
        return p

        return p

        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

        c = cfast(pp)
        cmax = float(c.max())

            return 1e100, np.zeros_like(uu)

        scaled = np.clip(scaled, -400.0, 400.0)

        sum_exp = float(exps.sum())

            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f_val = cmax
        else:
            f_val = cmax + T * float(np.log(sum_exp / c.size))
            if not math.isfinite(f_val):
                f_val = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)

                "disp": False,

            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == uu.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-15:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 4:
                            break
            if getattr(res, "status", 1) == 0:
                break
        return best_u

    if fine_only:
        stages = (
            (0.001, 0.35),
            (0.0003, 0.35),

    else:
        stages = (

            (0.01, 0.15),

    start = time.monotonic()
    avail = float(stop_time - start - 6.0)
    if avail <= 3.0:
        return p

    best_p = p.copy()
    best_sc = _score_fft(p)
    total_w = float(sum(w for _, w in stages))
    if total_w <= 0.0:
        total_w = 1.0

    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 2.0, start + avail * (cum / total_w))

        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0

def _subgradient_polish(p, stop_time, consider=None, max_iters=5000):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    best_p = p.copy()
    best_score = float(2.0 * n * _conv_self(p).max())
    base_step = 0.008
    min_step = 1e-9
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 1.5:
            break

        c = _conv_self(p)
        cmax = float(c.max())
        # Use a small set of active lags (top few)
        n_active = min(5, c.size)
        top_idx = np.argsort(c)[::-1][:n_active]
        # Restrict to those within 99.9% of max
        active = top_idx[c[top_idx] >= cmax * 0.995]
        if len(active) == 0:
            active = top_idx[:1]

        grad = np.zeros(n)
        for k in active:
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                contrib = p[kj[valid]] + p[valid]
                grad[valid] += contrib
        grad /= len(active)
        # Normalize gradient
        gn = float(np.linalg.norm(grad))
        if gn > 1e-300:
            grad /= gn

        step = base_step / (1.0 + 0.03 * it)
        if step < min_step:
            break

        improved_this = False
        for scale in (1.0, 0.5, 0.25, 0.125, 0.0625):
            p_cand = p - scale * step * grad
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = float(2.0 * n * _conv_self(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                improved_this = True
                break

        if not improved_this:
            stagnant += 1
            base_step *= 0.85
            if stagnant > 80 or base_step < 1e-6:
                break
        else:
            stagnant = max(0, stagnant - 2)

        if it % 100 == 0 and consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p

    best_sc = _score_fft(p)

    for it in range(n_iters):
        if time.monotonic() > stop_time - 8.0:
            break
        c = _conv_self(p)
        M = float(c.max())
        if not math.isfinite(M) or M <= 0:
            break

        g = _lp_best_response(p, M, max_time_s=min(30.0, stop_time - time.monotonic() - 10.0))
        if g is None or g.size != n:
            break

        # Blend
        g_norm = g / float(g.sum())
        blend = (1.0 - blend_t) * p + blend_t * g_norm
        blend = np.maximum(blend, 0.0)
        bs = float(blend.sum())
        if bs < 1e-300:
            break
        blend /= bs
        sc = _score_fft(blend)
        if math.isfinite(sc) and sc < best_sc - 1e-14:
            best_sc = sc
            best_p = blend.copy()
            p = blend.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass
        else:
            # Try smaller blend
            for t_try in (0.15, 0.08, 0.04):
                blend2 = (1.0 - t_try) * p + t_try * g_norm
                blend2 = np.maximum(blend2, 0.0)
                bs2 = float(blend2.sum())
                if bs2 < 1e-300:
                    continue
                blend2 /= bs2
                sc2 = _score_fft(blend2)
                if math.isfinite(sc2) and sc2 < best_sc - 1e-14:
                    best_sc = sc2
                    best_p = blend2.copy()
                    p = blend2.copy()
                    if consider is not None:
                        try:
                            consider(best_p)
                        except Exception:
                            pass
                    break
            else:
                break
        blend_t *= 0.7

    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 20.0

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(256, dtype=np.float64) / 256.0
        best_score = float(_score_exact_arr(best_arr))

    n_parent = int(best_arr.size)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    consider(parent)

    # Phase 1: LSE L-BFGS on parent length
    phase1_end = min(hard - 400.0, t0 + 280.0)
    if phase1_end > time.monotonic() + 30.0:
        try:
            p1 = _lbfgs_lse(parent, phase1_end, consider=consider, fine_only=False)
            if p1 is not None and p1.size == n_parent:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    # Phase 2: LP best response blending
    now = time.monotonic()
    phase2_end = min(hard - 300.0, now + 150.0)
    if phase2_end > now + 20.0:
        try:
            p2 = _lp_blend_step(best_arr, phase2_end, consider=consider, n_iters=4, blend_t=0.25)
            if p2 is not None:
                consider(p2)
        except Exception:
            pass

    # Phase 3: Heavy coordinate descent
    now = time.monotonic()
    phase3_end = min(hard - 200.0, now + 500.0)
    if phase3_end > now + 30.0:
        try:
            G_val = 36 if n_parent > 2000 else 48
            p3, _ = _coord_polish(best_arr, phase3_end, max_sweeps=200000, G=G_val,
                                  seed=int(seed + 7777), consider=consider)
            if p3 is not None:
                consider(p3)
        except Exception:
            pass

    # Phase 4: Subgradient + LP + final coord
    now = time.monotonic()
    phase4_end = hard - 15.0
    if phase4_end > now + 30.0:
        # Subgradient
        sg_end = min(phase4_end - 100.0, now + 80.0)
        if sg_end > now + 15.0:
            try:
                p4 = _subgradient_polish(best_arr, sg_end, consider=consider, max_iters=4000)
                if p4 is not None:
                    consider(p4)
            except Exception:
                pass

        # Another LP attempt
        now = time.monotonic()
        lp_end = min(phase4_end - 80.0, now + 60.0)
        if lp_end > now + 15.0:
            try:
                p5 = _lp_blend_step(best_arr, lp_end, consider=consider, n_iters=3, blend_t=0.15)
                if p5 is not None:
                    consider(p5)
            except Exception:
                pass

        # Final coordinate polish
        now = time.monotonic()
        cf_end = min(phase4_end - 10.0, now + 120.0)
        if cf_end > now + 15.0:
            try:
                p6, _ = _coord_polish(best_arr, cf_end, max_sweeps=200000, G=40,
                                      seed=int(seed + 99999), consider=consider)
                if p6 is not None:
                    consider(p6)
            except Exception:
                pass

    # Phase 5: Try a different length if time permits
    now = time.monotonic()
    if now < hard - 120.0 and n_parent > 1000:
        alt_n = n_parent
        # Try the main length with more fine LSE
        lse_end = min(hard - 60.0, now + 80.0)
        if lse_end > now + 20.0:
            try:
                p7 = _lbfgs_lse(best_arr, lse_end, consider=consider, fine_only=True)
                if p7 is not None:
                    consider(p7)
            except Exception:
                pass

        now = time.monotonic()
        cf_end2 = min(hard - 20.0, now + 60.0)
        if cf_end2 > now + 10.0:
            try:
                p8, _ = _coord_polish(best_arr, cf_end2, max_sweeps=100000, G=44,
                                      seed=int(seed + 55555), consider=consider)
                if p8 is not None:
                    consider(p8)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
