        pass
    return None

    grad -= grad.mean()

Actually, the LP-based approach is essentially a form of **mirror descent** on the constraint set {g: (a*g)_k <= M, g >= 0}. The mirror is the negative entropy, and the "distance" is the KL divergence. The step is:
g = argmax {<c, g> - D(a||g)} subject to constraints

where c = -1 and D(a||g) = sum a_j log(a_j/g_j) (KL divergence). This is a relative entropy step, not a Euclidean step. The resulting g is of the form g_j = a_j * exp(λ_j) where λ_j are determined by the dual variables.

In the Euclidean case (what I have), the step is:
g = argmax {<c, g>} subject to (a*g)_k <= M, g >= 0, g in simplex (or without simplex constraint)

This is a linear program, which is what I'm solving.

The Frank-Wolfe method uses:
g = argmin {<∇f(a), g> : g in C}

where C is the feasible set. In our case, f(a) = max_k (a*a)_k and C = {g: g >= 0, sum g = 1}. The linearized step would be:
g = argmin {<∇f(a), g> : g >= 0, sum g = 1}

where ∇f(a) = (1/|K*|) sum_{k in K*} 2 * a[·-k] (subgradient).

This is just: g_j = 1/|K*| if j minimizes the subgradient component, 0 otherwise. This is a vertex of the simplex, which is a very crude step.

The LP step I'm using is different: it's not a Frank-Wolfe step. It's more like a "best response" in a game-theoretic sense. The LP finds the g that maximizes sum(g) subject to (a*g) <= M. This is equivalent to: among all g with (a*g)_k <= M for all k, find the one with the largest sum. If the current a is not optimal, this g will have sum > 1 (assuming a is normalized), and moving toward g will reduce the peak.

import numpy as np
import time
from scipy.optimize import linprog


def _score_fn(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)


def _get_parent():
    try:
        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(200)
    p = np.clip(p, 0, 1000)
    if p.size == 0 or not np.all(np.isfinite(p)):
        p = np.ones(200)
    return p


def _norm(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        m = a.max()
        return a / m if m > 1e-15 else a
    return a / s


def _build_A(n, a):
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[d[mask]]
    return A


def _lp_solve(A, M, n, tl=4.0):
    c = -np.ones(n, dtype=np.float64)
    b = np.full(A.shape[0], M, dtype=np.float64)
    try:
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tl,
                               'dual_feasibility_tolerance': 1e-13,
                               'primal_feasibility_tolerance': 1e-13})
        if res.status in [0, 2] and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None


def _line_search(a, g):
    a_n = _norm(a)
    g_n = _norm(g)

    def _v(t):
        cand = (1 - t) * a_n + t * g_n
        cand = np.maximum(cand, 0)
        s = cand.sum()
        if s < 1e-15:
            return np.inf
        return float(np.convolve(cand / s, cand / s).max())

    best_t, best_v = 0.0, _v(0.0)
    for t in np.linspace(0.02, 1.0, 40):
        v = _v(t)
        if v < best_v:
            best_v, best_t = v, t

    if best_t > 0.005:
        lo = max(0.0, best_t - 0.025)
        hi = min(1.0, best_t + 0.025)
        for _ in range(60):
            if hi - lo < 1e-12:
                break
            mid = 0.5 * (lo + hi)
            v = _v(mid)
            if v < best_v:
                best_v, best_t = v, mid
                hi = mid
            else:
                lo = mid

    if best_t < 0.005:
        return None, best_v
    cand = (1 - best_t) * a_n + best_t * g_n
    cand = np.maximum(cand, 0)
    s = cand.sum()
    if s < 1e-12:
        return None, best_v
    return cand / s, best_v


def _subgrad_dir(a, M):
    c = np.convolve(a, a)
    n = a.size
    active = np.where(c >= M - 1e-13)[0]
    if len(active) == 0:
        active = np.where(c >= 0.999 * M)[0]
    grad = np.zeros(n)
    for ks in active:
        ks = int(ks)
        lo = max(0, ks - n + 1)
        hi = min(n, ks + 1)
        for jj in range(lo, hi):
            idx = ks - jj
            if 0 <= idx < n:
                grad[jj] += 2.0 * a[idx]
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-14:
        return None
    return grad / gn


def _coord_descent(a, M):
    """Try to reduce peak by redistributing mass at active lags."""
    c = np.convolve(a, a)
    n = a.size
    active = np.where(c >= M - 1e-12)[0]
    if len(active) == 0:
        active = np.where(c >= 0.999 * M)[0]

    for ks in active[:8]:
        ks = int(ks)
        lo = max(0, ks - n + 1)
        hi = min(n, ks + 1)
        best_ci, best_cj, best_prod = -1, -1, -1.0
        for i in range(lo, hi):
            j = ks - i
            if j != i and 0 <= j < n:
                prod = a[i] * a[j]
                if prod > best_prod:
                    best_prod, best_ci, best_cj = prod, i, j
        if best_ci < 0:
            continue
        d0 = 0.15 * min(a[best_ci], a[best_cj])
        for frac in [0.3, 0.5, 0.8]:
            d = frac * d0
            for sign in [1, -1]:
                cand = a.copy()
                cand[best_ci] -= sign * d
                cand[best_cj] += sign * d
                cand = np.maximum(cand, 0)
                s = cand.sum()
                if s < 1e-12:
                    continue
                cand /= s
                v = float(np.convolve(cand, cand).max())
                if v < M - 1e-13:
                    return cand
    return None


def _opt(a0, deadline, seed=42):
    a = _norm(np.clip(np.asarray(a0, dtype=np.float64), 0, 1000).copy())
    n = a.size
    if n < 2:
        return a
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 3:
            break

        c_full = np.convolve(a, a)
        M = float(c_full.max())

        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 80:
            sig = min(0.025 + 0.02 * (stuck // 80), 0.5)
            mode = int(rng.integers(0, 5))
            xx = np.arange(n, dtype=np.float64) / n
            if mode == 0:
                a *= np.exp(sig * rng.normal(0, 1, n))
            elif mode == 1:
                f = rng.uniform(1.0, 5.0)
                a *= (1.0 + sig * np.sin(2 * np.pi * f * xx + rng.uniform(0, 6.2832)))
            elif mode == 2:
                f1, f2 = rng.uniform(0.5, 3.0), rng.uniform(1.0, 7.0)
                a *= (1.0 + sig * (0.6 * np.sin(2 * np.pi * f1 * xx) + 0.4 * np.cos(2 * np.pi * f2 * xx)))
            elif mode == 3:
                a *= (1.0 + sig * rng.uniform(-1, 1, n))
            else:
                perm = np.arange(n)
                rng.shuffle(perm)
                a = a[perm]
            a = np.maximum(a, 1e-12)
            a = _norm(a)
            stuck = 0
            if sig > 0.48:
                break
            continue

        A = _build_A(n, a)
        moved = False

        for relax in [1e-12, 1e-8, 1e-5]:
            tl = max(1.0, min(4.0, 0.20 * rem))
            g = _lp_solve(A, M * (1.0 + relax), n, tl=tl)
            if g is None or g.sum() < 1e-12:
                continue
            g_n = g / g.sum()
            if np.max(np.abs(g_n - a)) < 1e-10:
                continue
            cand, val = _line_search(a, g)
            if cand is not None and val < M - 1e-13:
                a = cand
                moved = True
                break

        if not moved:
            grad = _subgrad_dir(a, M)
            if grad is not None:
                for alpha in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
                    cand = a - alpha * grad
                    cand = np.maximum(cand, 0)
                    s = cand.sum()
                    if s < 1e-12:
                        continue
                    cand /= s
                    v = float(np.convolve(cand, cand).max())
                    if v < M - 1e-14:
                        a = cand
                        moved = True
                        break

        if not moved:
            cand = _coord_descent(a, M)
            if cand is not None:
                a = cand
                moved = True

        if not moved:
            a *= np.exp(rng.normal(0, 0.012, n))
            a = np.maximum(a, 1e-12)
            a = _norm(a)

    return best_a


def _interp_to_length(seq, new_n):
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    xo = np.linspace(0, 1, old_n)
    xn = np.linspace(0, 1, new_n)
    return np.clip(np.interp(xn, xo, seq), 0, 1000)
