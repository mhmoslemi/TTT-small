                hi = lo + 1e-12
            fine = np.linspace(lo, hi, 28, dtype=np.float64)
            if 0.0 < best_x < x_hi:
                fine = np.unique(np.concatenate([fine, [best_x]]))

            w2 = fine - v
            S2 = 1.0 + w2
            aff2 = base[None, :] + w2[:, None] * infl[None, :]
            aff2[:, diag] += w2 * w2
            max_local2 = aff2.max(axis=1)
            max_new2 = np.maximum(max_local2, max_const)
            scores2 = np.full(fine.shape, math.inf, dtype=np.float64)
            valid2 = S2 > 1e-15
            if np.any(valid2):
                scores2[valid2] = (2.0 * n * max_new2[valid2]) / (S2[valid2] ** 2)

            if np.any(np.isfinite(scores2)):
                imin2 = int(np.argmin(scores2))
                if scores2[imin2] < best_val:
                    best_x = float(fine[imin2])
                    best_val = float(scores2[imin2])

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= 8.0:
                delta = best_x - v
                if abs(delta) > 1e-15:
                    p[i] = best_x
                    c[i:local_end] += 2.0 * delta * p
                    two_i = 2 * i
                    if 0 <= two_i < Ltot:
                        c[two_i] -= delta * delta
                    cur_score = best_val
                    improved = True

        # Normalize
        ss = float(p.sum())
        if ss > 1e-300 and math.isfinite(ss):
            inv = 1.0 / ss
            p *= inv
            c *= inv * inv
        else:
            break

        # Periodic recompute to fix accumulated error
        if sweep % 5 == 4:
            c = _conv_self(p)

        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat_count = 0
        else:
            flat_count += 1
            if flat_count >= 3 and time.monotonic() < stop_time - 10.0:
                # Random restart
                noise_scale = 0.03 + 0.02 * min(flat_count - 3, 10)
                p = p * np.exp(rng.normal(0.0, noise_scale, n, dtype=np.float64))
                p = np.maximum(p, 0.0)
                ss = float(p.sum())
                if ss > 0:
                    p /= ss
                c = _conv_self(p)
                cur_score = float(2.0 * n * c.max())
                if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
                    best_score = cur_score
                    best_p = p.copy()
                    if consider is not None:
                        try:
                            consider(best_p)
                        except Exception:
                            pass
                flat_count = 0
                if flat_count > 20:
                    break

        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        sc = float(_score_fft(p))
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p, best_score


def _lse_optimize(p, stop_time, consider=None):
    """LSE-smoothed optimization using L-BFGS-B."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300), dtype=np.float64)
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], conv_size)
        G = rfft(ww, conv_size)
        c = irfft(F * G, conv_size)[:conv_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu, dtype=np.float64)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0.0:
            return 1e100, np.zeros_like(uu, dtype=np.float64)
        if T <= 0.0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    def _softmax(uu):
        if not np.all(np.isfinite(uu)):
            return np.full(uu.size, 1.0 / uu.size, dtype=np.float64)
        m = float(np.max(uu))
        e = np.exp(uu - m, dtype=np.float64)
        s_ = float(e.sum())
        if not math.isfinite(s_) or s_ <= 0.0:
            return np.full(uu.size, 1.0 / uu.size, dtype=np.float64)
        return e / s_

    stages = (
        (0.05, 0.12),
        (0.02, 0.15),
        (0.008, 0.18),
        (0.003, 0.20),
        (0.001, 0.25),
        (0.0003, 0.25),
        (0.0001, 0.20),
    )

    best_p = p.copy()
    best_sc = _score_fft(p)
    total_w = float(sum(w for _, w in stages))
    cum = 0.0
    start = time.monotonic()
    avail = max(5.0, float(stop_time - start - 5.0))

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)
        best_u = np.array(u, dtype=np.float64, copy=True)
        best_f, _ = lse_func(best_u, T)
        stale = 0
        while time.monotonic() < end_stage - 1.0:
            maxit = 30
            opts = {"maxiter": maxit, "gtol": 1e-12, "ftol": 1e-15,
                    "maxls": 15, "maxcor": 25, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,),
                               method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_f - 1e-15:
                    best_f = fval
                    best_u = res.x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if getattr(res, "status", 1) == 0:
                break
        p_cur = _softmax(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur):
            if sc_cur < best_sc:
                best_sc = sc_cur
                best_p = p_cur.copy()
                p = p_cur.copy()
            else:
                p = best_p.copy()
            u = p_to_u(p)
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

    return best_p


def _lp_direction(f, stop_time):
    """LP-based direction: maximize sum(g) s.t. f*g <= M, g >= 0."""
    n = int(f.size)
    if n < 16 or n > 2048:
        return None
    c = _conv_self(f)
    M = float(c.max())
    if M <= 0 or not math.isfinite(M):
        return None
    try:
        import cvxpy as cp
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f[j]
        g = cp.Variable(n, nonneg=True)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        constraints = [A @ g <= b]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        solved = False
        for slv in [cp.ECOS, cp.SCS, cp.CLPU, cp.PDBLP]:
            try:
                prob.solve(solver=slv, verbose=False)
                solved = True
                break
            except Exception:
                continue
            if time.monotonic() > stop_time - 5:
                break
        if not solved or g.value is None:
            return None
        g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        gv = float(g_val.sum())
        if gv < 1e-10 or not math.isfinite(gv):
            return None
        return g_val / gv
    except Exception:
        return None


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 3.0:
        hard = t0 + max(2.0, budget - 1.0)

    parent = _get_parent()
    best_arr = parent.copy()
    best_arr = np.maximum(best_arr, 0.0)
    best_arr = np.clip(best_arr, 0.0, 1000.0)
    bs = float(best_arr.sum())
    if bs > 0:
        best_arr = best_arr / bs
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score):
        best_arr = np.ones(256, dtype=np.float64) / 256.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 5e-14:
            best_arr = a.copy()
            best_score = sc

    n_par = int(best_arr.size)

    # Phase 1: LSE optimization
    phase1_end = min(hard - 80.0, t0 + 0.25 * budget)
    if phase1_end > t0 + 30.0 and n_par >= 128:
        try:
            p1 = _lse_optimize(best_arr.copy(), phase1_end, consider=consider)
            if p1 is not None and p1.size == n_par and np.all(np.isfinite(p1)):
                p1 = np.maximum(p1, 0.0)
                ss = float(p1.sum())
                if ss > 0:
                    p1 = p1 / ss
                    consider(p1)
        except Exception:
            pass

    work = best_arr.copy()

    # Phase 2: Coordinate descent (main phase)
    phase2_end = min(hard - 60.0, t0 + 0.55 * budget)
    if phase2_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord_descent(work, phase2_end, max_sweeps=50000,
                                     seed=int(seed + 7777), consider=consider)
        except Exception:
            pass

    work = best_arr.copy()

    # Phase 3: LP direction + blend
    lp_end = min(hard - 50.0, time.monotonic() + 40.0)
    if lp_end > time.monotonic() + 10.0 and n_par <= 2048:
        g_dir = _lp_direction(work, lp_end)
        if g_dir is not None and g_dir.size == n_par:
            for t_alpha in [0.3, 0.2, 0.1]:
                blended = (1.0 - t_alpha) * work + t_alpha * g_dir
                blended = np.maximum(blended, 0.0)
                bs = float(blended.sum())
                if bs > 0:
                    blended /= bs
                    consider(blended)

    work = best_arr.copy()

    # Phase 4: More coordinate descent
    phase4_end = min(hard - 30.0, t0 + 0.82 * budget)
    if phase4_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord_descent(work, phase4_end, max_sweeps=50000,
                                     seed=int(seed + 123456), consider=consider)
        except Exception:
            pass

    work = best_arr.copy()

    # Phase 5: Try a different length
    rem = hard - 20.0 - time.monotonic()
    if rem > 100.0:
        # Try length 2048
        n2 = 2048
        xp = (np.arange(n_par, dtype=np.float64) + 0.5) / float(n_par)
        fp = np.interp((np.arange(n2, dtype=np.float64) + 0.5) / float(n2), xp, best_arr)
        p2 = np.maximum(fp, 0.0)
        p2s = float(p2.sum())
        if p2s > 0:
            p2 /= p2s
            consider(p2)
            if rem > 80.0:
                p2_end = min(hard - 20.0, time.monotonic() + rem - 30.0)
                if p2_end > time.monotonic() + 15.0:
                    try:
                        p2, _ = _coord_descent(p2, p2_end, max_sweeps=30000,
                                               seed=int(seed + 424242), consider=consider)
                    except Exception:
                        pass

    # Phase 6: Final squeeze
    work = best_arr.copy()
    fin_end = min(hard - 10.0, t0 + budget - 8.0)
    if fin_end > time.monotonic() + 10.0:
        try:
            work, _ = _coord_descent(work, fin_end, max_sweeps=30000,
                                     seed=int(seed + 555555), consider=consider)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
