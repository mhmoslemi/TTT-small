    n = v.size

def _score_arr(a):

def _subgradient_steps(p0, stop_time, consider=None, max_iters=5000):
    n = p0.size
    p = np.maximum(p0.copy(), 0.0)
    s = float(p.sum())
    if s < 1e-300:
        return p0
    p /= s
    
    conv_len = 2 * n - 1
    conv_size = next_fast_len(conv_len)
    
    def cfast(a):
        F = rfft(a, conv_size)
        return irfft(F * F, conv_size)[:conv_len]
    
    b = cfast(p)
    cur_score = float(2.0 * n * b.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(12345)
    
    eta = 0.5 / n  # initial step size
    momentum = 0.9
    vel = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > stop_time - 3.0:
            break
        
        # Find max
        k_star = int(np.argmax(b))
        b_max = b[k_star]
        
        # Compute subgradient
        g = np.zeros(n)
        # g_j = p_{k*-j} * (1 + delta_{2j, k*})
        # Valid j: 0 <= k*-j < n => j in [k*-n+1, k*] intersect [0, n-1]
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        for j in range(j_lo, j_hi + 1):
            g[j] = p[k_star - j]
            if 2 * j == k_star:
                g[j] *= 2.0
        
        # Damped update with momentum
        vel = momentum * vel - eta * g
        p_new = p + vel
        
        # Project onto simplex
        # Simplex projection
        n_ = n
        v = np.sort(p_new)[::-1]
        csum = np.cumsum(v)
        idx = np.arange(1, n_ + 1)
        mask = v * idx > (csum - 1)
        if np.any(mask):
            rho = int(mask.nonzero()[0][-1])  # last True
            theta = (csum[rho] - 1.0) / (rho + 1.0)
            p_new = np.maximum(p_new - theta, 0.0)
        else:
            p_new = np.zeros(n)
            p_new[0] = 1.0
        
        ss = float(p_new.sum())
        if ss < 1e-300:
            break
        p_new /= ss  # ensure exact sum=1 (projection should give this, but round-off)
        
        # Update
        p = p_new
        b = cfast(p)
        new_score = float(2.0 * n * b.max())
        
        if math.isfinite(new_score):
            if new_score < best_score:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass
        else:
            p = best_p.copy()
            break
        
        # Adaptive step size
        if it % 10 == 0 and it > 0:
            eta *= 0.99  # slowly decay
        if it < 50:
            eta *= 0.995
    
    return best_p

def _simplex_proj(v):

    w = np.maximum(v - theta, 0)
    s = w.sum()

            if getattr(res, "status", 1) == 0:
                break

        p_cur = _softmax_vec(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc = _score_arr(p_cur)
        if math.isfinite(sc) and sc < best_sc - 1e-14:
            best_sc = sc
            best_p = p_cur.copy()
            p = p_cur.copy()
            u = np.log(np.maximum(p, 1e-30))
            u -= float(u.mean())
        else:
            p = best_p.copy()
            u = np.log(np.maximum(p, 1e-30))
            u -= float(u.mean())

        if consider:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _subgradient_optimize(p0, stop_time, consider=None, seed=999):
    n = p0.size
    p = np.maximum(p0.copy(), 0.0)
    s = float(p.sum())
    if s < 1e-300:
        return p0
    p /= s
    b = _conv_self(p)
    cur_score = float(2.0 * n * b.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    conv_len = 2 * n - 1
    conv_size = next_fast_len(conv_len)

    def cfast(a):
        F = rfft(a, conv_size)
        return irfft(F * F, conv_size)[:conv_len]

    eta0 = 1.0 / (2.0 * n)
    it = 0
    max_iters = 20000
    while it < max_iters and time.monotonic() < stop_time - 2.0:
        k_star = int(np.argmax(b))
        b_max = float(b[k_star])

        g = np.zeros(n, dtype=np.float64)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        if j_hi >= j_lo:
            idxs = np.arange(j_lo, j_hi + 1)
            g[idxs] = p[k_star - idxs]
            diag_mask = (2 * idxs == k_star)
            if np.any(diag_mask):
                g[idxs[diag_mask]] *= 2.0

        eta = eta0 * (1.0 + 0.3 * math.cos(0.01 * it))
        p_new = p - eta * g
        p_new = _simplex_proj(p_new)

        b = cfast(p_new)
        new_score = float(2.0 * n * b.max())
        if math.isfinite(new_score) and new_score < cur_score:
            p = p_new
            cur_score = new_score
            if new_score < best_score - 1e-15:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass
        else:
            p = p + eta * g
            p = _simplex_proj(p)

        if it % 200 == 199:
            eta0 *= 0.97
        it += 1

    return best_p


def _coord_descent(p0, stop_time, consider=None, seed=7777):
    n = p0.size
    if n == 0:
        return p0, float('inf')
    p = np.maximum(np.asarray(p0, dtype=np.float64).ravel(), 0.0)
    s = float(p.sum())
    if s < 1e-300:
        return p0, float('inf')
    p /= s
    b = _conv_self(p)
    cur_score = float(2.0 * n * b.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    Ltot = 2 * n - 1
    G = 32
    flat = 0
    sweep = 0
    GR = (math.sqrt(5.0) - 1.0) / 2.0

    def eval_1d(w, base, infl, diag_pos, max_const, v, n, Ls):
        s_new = 1.0 + w - v
        if s_new <= 1e-15:
            return float('inf')
        mx = float(np.max(base + w * infl))
        if 0 <= diag_pos < Ls:
            dv = base[diag_pos] + w * w
            if dv > mx:
                mx = dv
        if max_const > mx:
            mx = max_const
        return 2.0 * n * mx / (s_new * s_new)

    max_sweeps = 100000
    while sweep < max_sweeps and time.monotonic() < stop_time - 2.5:
        improved = False
        order = rng.permutation(n)
        for ii in range(n):
            i = int(order[ii])
            if time.monotonic() > stop_time - 1.0:
                break
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            base = b[start:end].copy()
            infl = p[:Ls].copy()
            base -= v * infl
            diag_pos = i if i < Ls else -1

            max_const = 0.0
            if start > 0:
                mc = float(np.max(b[:start]))
                if mc > max_const:
                    max_const = mc
            if end < Ltot:
                mc = float(np.max(b[end:]))
                if mc > max_const:
                    max_const = mc

            x_hi = max(40.0 * v, 8.0 / n, 0.003)
            x_hi = min(x_hi, 0.4)
            x_hi = max(x_hi, 1e-10)

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            ws = pts - v
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag_pos < Ls:
                aff[:, diag_pos] = base[diag_pos] + pts * pts
            maff = np.maximum(aff.max(axis=1), max_const)
            s_new = 1.0 + ws
            valid = s_new > 1e-12
            scores = np.full(G + 1, float('inf'), dtype=np.float64)
            if np.any(valid):
                scores[valid] = 2.0 * n * maff[valid] / (s_new[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            if 0 < imin < G:
                lo = float(pts[max(0, imin - 2)])
                hi = float(pts[min(G, imin + 2)])
            else:
                lo = 0.0
                hi = x_hi
            if hi - lo > 1e-14:
                a_gs, b_gs = lo, hi
                c_gs = b_gs - GR * (b_gs - a_gs)
                d_gs = a_gs + GR * (b_gs - a_gs)
                f_c = eval_1d(c_gs, base, infl, diag_pos, max_const, v, n, Ls)
                f_d = eval_1d(d_gs, base, infl, diag_pos, max_const, v, n, Ls)
                for _ in range(18):
                    if b_gs - a_gs < 1e-13 * max(1.0, abs(hi)):
                        break
                    if f_c < f_d:
                        b_gs = d_gs
                        d_gs = c_gs
                        f_d = f_c
                        c_gs = b_gs - GR * (b_gs - a_gs)
                        f_c = eval_1d(c_gs, base, infl, diag_pos, max_const, v, n, Ls)
                    else:
                        a_gs = c_gs
                        c_gs = d_gs
                        f_c = f_d
                        d_gs = a_gs + GR * (b_gs - a_gs)
                        f_d = eval_1d(d_gs, base, infl, diag_pos, max_const, v, n, Ls)
                x_mid = (a_gs + b_gs) / 2.0
                f_mid = eval_1d(x_mid, base, infl, diag_pos, max_const, v, n, Ls)
                if f_mid < best_val:
                    best_x = x_mid
                    best_val = f_mid

            for xp in (0.0, x_hi):
                fp = eval_1d(xp, base, infl, diag_pos, max_const, v, n, Ls)
                if fp < best_val:
                    best_x = xp
                    best_val = fp

            if math.isfinite(best_val) and best_val < cur_score * (1.0 - 1e-13) and 0.0 <= best_x <= x_hi:
                w = best_x
                if abs(w - v) > 1e-16:
                    p[i] = w
                    aff_seg = base + w * infl
                    if 0 <= diag_pos < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag_pos] = base[diag_pos] + w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    b[start:end] = aff_seg
                    s_new_ = 1.0 + w - v
                    cur_score = 2.0 * n * new_max / (s_new_ * s_new_)
                    improved = True

        s_new_total = float(p.sum())
        if s_new_total > 1e-300 and math.isfinite(s_new_total):
            inv = 1.0 / s_new_total
            p *= inv
            b = _conv_self(p)
            new_score = float(2.0 * n * b.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-15:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 8.0:
                    kick = 0.008 + 0.025 * min(1.0, flat / 20.0)
                    p *= np.exp(rng.normal(0.0, kick, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    b = _conv_self(p)
                    cur_score = float(2.0 * n * b.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s
