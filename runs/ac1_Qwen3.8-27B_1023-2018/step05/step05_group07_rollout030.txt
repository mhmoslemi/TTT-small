def _fft_conv(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for efficient FFT
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(fa * fa, fft_size)
    return c[:size]

    return 2.0 * len(a) * float(c.max()) / (s * s)

def _build_conv_mat(n, a):
    """(2n-1) x n matrix for convolution: (A @ b)_k = (a*b)_k"""
    # Row k, col j: a[k-j] if 0 <= k-j < n, else 0
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[d[mask]]
    return A

def _lp_direction(a, M, time_limit=2.0):
    """Find LP best response direction."""
    n = len(a)
    A = _build_conv_mat(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * (1 + 1e-10))
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0 and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None

def _line_search_convex(a, direction, M_current, n_trials=12):
    """Find best convex combination along direction."""
    n = len(a)
    best_t, best_v = 0.0, M_current
    best_cand = None
    
    # Coarse grid
    ts = np.linspace(0, 1, n_trials)
    for t in ts:
        cand = (1 - t) * a + t * direction
        cand = np.maximum(cand, 0)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s
        v = float(_fft_conv(cand).max())
        if v < best_v - 1e-14:
            best_v = v
            best_t = t
            best_cand = cand
    
    if best_cand is None:
        return None, M_current
    
    # Fine grid around best
    lo = max(0, best_t - 1.0 / (n_trials - 1))
    hi = min(1, best_t + 1.0 / (n_trials - 1))
    for _ in range(30):
        if hi - lo < 1e-10:
            break
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        c1 = (1 - m1) * a + m1 * direction
        c1 = np.maximum(c1, 0) / max(c1.sum(), 1e-15)
        v1 = float(_fft_conv(c1).max())
        c2 = (1 - m2) * a + m2 * direction
        c2 = np.maximum(c2, 0) / max(c2.sum(), 1e-15)
        v2 = float(_fft_conv(c2).max())
        if v1 < v2:
            hi = m2
        else:
            lo = m1
        mid = 0.5 * (lo + hi)
        candid = (1 - mid) * a + mid * direction
        candid = np.maximum(candid, 0) / max(candid.sum(), 1e-15)
        vmid = float(_fft_conv(candid).max())
        if vmid < best_v:
            best_v = vmid
            best_t = mid
            best_cand = candid
    
    return best_cand, best_v

def _gradient_step(a, c, M, eta_options=None):
    """Subgradient step using analytical gradient structure."""
    n = len(a)
    if eta_options is None:
        eta_options = [0.01, 0.005, 0.002, 0.001, 0.0005]
    
    # Find active lags
    active = np.where(c >= M * 0.999)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    
    best_M = M
    best_cand = None
    
    for k_star in active[:8]:  # Try up to 8 active lags
        k_star = int(k_star)
        # Gradient: grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n
        grad = np.zeros(n)
        # k_star - i in [0, n) means i in [k_star - n + 1, k_star]
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] = 2 * a[k_star - i_hi:k_star - i_lo - 1:-1]
        
        # Normalize gradient for step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            continue
        
        for eta in eta_options:
            # Projected subgradient step
            cand = a - eta * grad / gnorm
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-12:
                continue
            cand = cand / s
            v = float(_fft_conv(cand).max())
            if v < best_M - 1e-14:
                best_M = v
                best_cand = cand
    
    return best_cand, best_M

def _mirror_descent_step(a, c, M, eta_options=None):
    """Mirror descent (exponentiated gradient) step."""
    n = len(a)
    if eta_options is None:
        eta_options = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    active = np.where(c >= M * 0.999)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    
    best_M = M
    best_cand = None
    
    for k_star in active[:5]:
        k_star = int(k_star)
        grad = np.zeros(n)
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] = 2 * a[k_star - i_hi:k_star - i_lo - 1:-1]
        
        # Project gradient to zero-mean (to preserve sum)
        grad = grad - grad.mean()
        
        for eta in eta_options:
            # Exponentiated gradient (mirror descent with KL)
            cand = a * np.exp(-eta * grad)
            s = cand.sum()
            if s < 1e-12 or not np.all(np.isfinite(cand)):
                continue
            cand = cand / s
            v = float(_fft_conv(cand).max())
            if v < best_M - 1e-14:
                best_M = v
                best_cand = cand
    
    return best_cand, best_M

def _optimize(a0, deadline, seed=42):
    """Main optimization loop combining multiple strategies."""
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, 1000).copy()

    if s < 1e-15:
        a = np.ones_like(a)
    a = a / s
    n = a.size
    
    best_a = a.copy()
    best_M = float(_fft_conv(a).max())
    rng = np.random.default_rng(seed)
    
    stuck = 0
    max_stuck = 100
    
    # Precompute FFT size
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2
    
    def conv_fast(x):
        fx = np.fft.rfft(x, fft_size)
        return np.fft.irfft(fx * fx, fft_size)[:2*n-1]
    
    while time.monotonic() < deadline:
        c = conv_fast(a)
        M = float(c.max())
        
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > max_stuck:
            # Perturb and reset
            sigma = min(0.02 + 0.01 * (stuck // max_stuck), 0.3)
            a *= np.exp(sigma * rng.normal(0, 1, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck = 0
            if sigma > 0.28:
                break
            continue
        
        moved = False
        
        # Strategy 1: LP direction
        if time.monotonic() < deadline - 5:
            g = _lp_direction(a, M, time_limit=1.5)
            if g is not None and g.sum() > 1e-12:
                g = g / g.sum()
                if np.max(np.abs(g - a)) > 1e-10:
                    cand, val = _line_search_convex(a, g, M)
                    if cand is not None and val < M - 1e-12:
                        a = cand
                        moved = True
        
        # Strategy 2: Gradient step
        if not moved:
            cand, val = _gradient_step(a, c, M)
            if cand is not None and val < M - 1e-12:
                a = cand
                moved = True
        
        # Strategy 3: Mirror descent
        if not moved:
            cand, val = _mirror_descent_step(a, c, M)
            if cand is not None and val < M - 1e-12:
                a = cand
                moved = True
        
        # Strategy 4: Coordinate descent on the max lag
        if not moved:
            k_max = int(np.argmax(c))
            # Try moving mass from the pair that contributes most to c[k_max]
            i_lo = max(0, k_max - n + 1)
            i_hi = min(n - 1, k_max)
            # The convolution at k_max: c[k_max] = sum_{i=i_lo}^{i_hi} a[i] * a[k_max - i]
            prods = np.array([a[i] * a[k_max - i] for i in range(i_lo, i_hi + 1)])
            if len(prods) > 0:
                top_idx = int(np.argmax(prods))
                i_top = i_lo + top_idx
                j_top = k_max - i_top
                if i_top != j_top:
                    # Try transferring mass
                    for frac in [0.1, 0.2, 0.3, 0.5]:
                        d = frac * abs(a[i_top] - a[j_top])
                        for sign in [1, -1]:
                            cand = a.copy()
                            cand[i_top] -= sign * d
                            cand[j_top] += sign * d
                            cand = np.maximum(cand, 0)
                            cs = cand.sum()
                            if cs < 1e-12:
                                continue
                            cand = cand / cs
                            v = float(conv_fast(cand).max())
                            if v < M - 1e-12:
                                a = cand
                                moved = True
                                break
                        if moved:
                            break
        
        if not moved:
            # Small random perturbation
            a *= np.exp(0.01 * rng.normal(0, 1, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
    
    return best_a

    return 2.0 * n * float(c.max()) / (s * s)

def _get_parent():
    try:
        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(200)
    p = np.clip(p, 0, 1000)
    if p.size == 0 or not np.all(np.isfinite(p)):
        p = np.ones(200)
    return p

def _build_conv_mat(n, a):
    """(2n-1) x n matrix."""
    k = np.arange(2*n-1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    A[mask] = a[d[mask]]
    return A

def _lp_dir(a, M, tl=2.0):
    n = len(a)
    A = _build_conv_mat(n, a)
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M*(1+1e-10)),
                      bounds=[(0,None)]*n, method='highs',
                      options={'time_limit': tl})
        if res.status == 0 and res.x is not None:
            return np.clip(res.x, 0, None)
    except:
        pass
    return None

def _compute_grad(a, k_star, n):
    """Analytical subgradient of c_k at a for lag k_star."""
    grad = np.zeros(n)
    i_lo = max(0, k_star - n + 1)
    i_hi = min(n-1, k_star)
    if i_lo <= i_hi:
        grad[i_lo:i_hi+1] = 2.0 * a[k_star-i_hi : k_star-i_lo+1][::-1]
    return grad

def _ternary_line_search(a, direction, M_cur, fft_size, n):
    """Ternary search for best convex combination."""
    size = 2*n-1
    def eval_t(t):
        cand = (1-t)*a + t*direction
        cand = np.maximum(cand, 0)
        s = cand.sum()
        if s < 1e-12:
            return float('inf')
        cand = cand / s
        c = _fft_conv(cand, fft_size)
        return float(c.max())
    
    lo, hi = 0.0, 1.0
    best_t, best_v = 0.0, M_cur
    for _ in range(40):
        m1 = lo + (hi-lo)/3
        m2 = hi - (hi-lo)/3
        v1, v2 = eval_t(m1), eval_t(m2)
        if v1 < v2:
            hi = m2
            if v1 < best_v: best_v, best_t = v1, m1
        else:
            lo = m1
            if v2 < best_v: best_v, best_t = v2, m2
        if hi - lo < 1e-10:
            break
    
    if best_t == 0.0:
        return None, M_cur
    cand = (1-best_t)*a + best_t*direction
    cand = np.maximum(cand, 0)
    s = cand.sum()
    if s < 1e-12:
        return None, M_cur
    cand = cand / s
    return cand, best_v
