        S_total = 1.0
        c = _conv_self(p)
        np.maximum(c, 0.0, out=c)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-16:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 5.0:
                    try:
                        noise_amt = 0.02 + 0.01 * math.exp(-flat * 0.15)
                        p = p * np.exp(rng.normal(0.0, noise_amt, n))
                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss > 0.0:
                            p /= ss
                        c = _conv_self(p)
                        S_total = 1.0
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score):
                            best_score = min(best_score, cur_score)
                            flat = 0
                    except Exception:
                        break
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 20.0
    usable = max(1.0, hard - t0)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64)
        best_arr /= 64.0
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    def make_skew(src, n, sd):
        p = _resample_mid(src, n)
        rng = np.random.default_rng(sd)
        x = (np.arange(n) + 0.5) / n
        side = 1.0 if rng.random() < 0.5 else -1.0
        amp = 0.3 + 0.7 * float(rng.random())
        q = p * np.exp(side * amp * (2.0 * x - 1.0))
        q = np.maximum(q, 0.0)
        s = float(q.sum())
        if s > 0.0:
            q /= s
        else:
            q = p
        return q

    def make_bump(src, n, sd):
        p = _resample_mid(src, n)
        rng = np.random.default_rng(sd)
        x = (np.arange(n) + 0.5) / n
        center = 0.3 + 0.4 * rng.random()
        width = 0.1 + 0.2 * rng.random()
        q = p * (1.0 + 0.5 * np.exp(-((x - center) / width) ** 2))
        q = np.maximum(q, 0.0)
        s = float(q.sum())
        if s > 0.0:
            q /= s
        else:
            q = p
        return q

    n = int(best_arr.size)
    if n == 0:
        n = 3072

    # Phase 1: L-BFGS on parent (70% of L-BFGS budget)
    phase1_end = t0 + 0.45 * usable
    now = time.monotonic()
    if phase1_end > now + 60.0:
        try:
            p2 = _lbfgs_optimize(best_arr, n, phase1_end, consider=consider, fine_only=False)
            if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0.0:
                p2 = p2 / float(p2.sum())
                consider(p2)
        except Exception:
            pass

    # Phase 2: Coordinate descent (main)
    now = time.monotonic()
    phase2_end = min(t0 + 0.75 * usable, phase1_end + 0.30 * usable)
    if phase2_end > now + 30.0:
        try:
            p3, _ = _coord_polish(best_arr, phase2_end, max_sweeps=200000, seed=seed, consider=consider)
            if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and p3.sum() > 0.0:
                p3 = p3 / float(p3.sum())
                consider(p3)
        except Exception:
            pass

    # Phase 3: L-BFGS fine-tune
    now = time.monotonic()
    phase3_end = min(t0 + 0.88 * usable, phase2_end + 0.12 * usable)
    if phase3_end > now + 20.0:
        try:
            p4 = _lbfgs_optimize(best_arr, n, phase3_end, consider=consider, fine_only=True)
            if p4 is not None and p4.size == n and np.all(np.isfinite(p4)) and p4.sum() > 0.0:
                p4 = p4 / float(p4.sum())
                consider(p4)
        except Exception:
            pass

    # Phase 4: Final coordinate descent
    now = time.monotonic()
    phase4_end = min(hard - 10.0, phase3_end + 0.10 * usable)
    if phase4_end > now + 15.0:
        try:
            p5, _ = _coord_polish(best_arr, phase4_end, max_sweeps=200000, seed=int(seed + 77777), consider=consider)
            if p5 is not None and p5.size == n and np.all(np.isfinite(p5)) and p5.sum() > 0.0:
                p5 = p5 / float(p5.sum())
                consider(p5)
        except Exception:
            pass

    # Try a skewed initialization if time remains
    now = time.monotonic()
    if now < hard - 120.0 and n <= 4096:
        skew_src = make_skew(best_arr, n, 999)
        skew_end = min(hard - 60.0, now + 0.30 * (hard - now))
        if skew_end > now + 30.0:
            try:
                p6 = _lbfgs_optimize(skew_src, n, skew_end, consider=consider, fine_only=False)
                if p6 is not None and p6.size == n and np.all(np.isfinite(p6)) and p6.sum() > 0.0:
                    p6 = p6 / float(p6.sum())
                    consider(p6)
                    # Quick coord polish on the skewed result
                    now2 = time.monotonic()
                    ce = min(hard - 30.0, now2 + 0.15 * (hard - now2))
                    if ce > now2 + 10.0:
                        p7, _ = _coord_polish(p6, ce, max_sweeps=50000, seed=555, consider=consider)
                        if p7 is not None and p7.size == n and np.all(np.isfinite(p7)) and p7.sum() > 0.0:
                            consider(p7)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
