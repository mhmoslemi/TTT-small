    p = p.copy()

                break

            k_end = min(i + n, 2*n - 1)  # exclusive

                continue

            base = c[k_start:k_end].copy()
            # Influence vector: a[k-i] for k in [k_start, k_end)
            # j = k - i, ranges from 0 to Ls-1
            infl = p[:Ls].copy()

            diag_pos = 2 * i - k_start  # position in the affected range

            fixed_max = 0.0
            if k_start > 0:
                fixed_max = max(fixed_max, float(c[:k_start].max()))
            if k_end < 2*n - 1:
                fixed_max = max(fixed_max, float(c[k_end:].max()))

            # where c'[k] = base[j] + 2*(x-v)*infl[j] for j != diag_pos
            # and c'[diag_pos] = base[diag_pos] + (x-v)*(x+v)

                w = x - v
                S_new = S + w
                if S_new < 1e-15:
                    return math.inf

                new_vals = base + 2.0 * w * infl
                if 0 <= diag_pos < Ls:

                # max of new values
                mv = float(np.maximum(new_vals.max(), fixed_max))
                return (2.0 * n * mv) / (S_new * S_new)

            pts = np.linspace(0.0, x_max, 40)

            if imin > 0 and imin < len(pts) - 1:

def _get_parent():

def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = np.convolve(a, a)
    m = float(np.max(c))
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _lse_optimize(p, stop_time, consider, seed):

    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())

        return p

        uu = np.log(np.maximum(pp, 1e-300))

        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]

            diag_pos = i  # position of k=2i within [k_start, k_end)

                ml = float(c[:k_start].max())
                if ml > fixed_max:
                    fixed_max = ml
            if k_end < Ltot:
                mr = float(c[k_end:].max())

                mv = max(float(nv.max()), fixed_max)
                return (2.0 * n * mv) / (S_new * S_new)

            coarse_pts = np.linspace(0.0, x_hi, 50)

        return [1.0]
