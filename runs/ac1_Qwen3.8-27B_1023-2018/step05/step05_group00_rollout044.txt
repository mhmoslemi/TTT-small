            maxit = 40
            opts = {"maxiter": maxit, "gtol": 1e-13, "ftol": 1e-16,
                    "maxls": 30, "maxcor": 40, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,),
                               method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-16:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 4:
                            break
            if getattr(res, "status", 1) == 0:
                break
        u = best_u
        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fft(p_cur)
        if not math.isfinite(sc_cur):
            p = best_p.copy()
            u = p_to_u(p)
            continue
        if sc_cur < best_sc:
            best_sc = sc_cur
            best_p = p_cur.copy()
            p = p_cur.copy()
        elif sc_cur > best_sc * (1.001):
            p = best_p.copy()
        else:
            p = p_cur.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p


def _coord_descent(p, stop_time, seed=12345, consider=None):
    """Aggressive coordinate descent with analytic bounds."""
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p, math.inf
    p = p / s

    G = 32  # grid points
    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < 500000 and time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            S = float(S_total)
            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue
            base = c[start_i:end_i].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start_i > 0:
                ml = float(c[:start_i].max())
                if ml > max_const:
                    max_const = ml
            if end_i < Ltot:
                mr = float(c[end_i:].max())
                if mr > max_const:
                    max_const = mr

            avg = S / float(n)
            # Tighter upper bound using active peak
            cmax_now = float(c.max())
            # The peak must not exceed cmax_now (for same score with S>= old S)
            # Find the lag of the max
            k_peak = int(np.argmax(c))
            # For coord i, at lag k_peak: base[k_peak - start_i] + w*infl[k_peak - start_i] <= cmax_now
            wk_peak = k_peak - start_i
            x_hi = max(20.0 * v, 60.0 * avg, 0.03 * S, 1e-12)
            if 0 <= wk_peak < Ls and infl[wk_peak] > 1e-300:
                # Constraint: base[wk_peak] + w*infl[wk_peak] <= cmax_now
                if base[wk_peak] < cmax_now:
                    w_max = (cmax_now - base[wk_peak]) / infl[wk_peak]
                    x_hi = min(x_hi, v + max(0, w_max) * 1.02)
            x_hi = min(x_hi, 12.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, avg * 2)

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            diag = i - start_i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                dval = base[diag] + w * infl[diag] + w * w
                maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Golden section refinement
            dw = x_hi / float(G)
            lo = max(0.0, best_x - 2.0 * dw)
            hi = min(x_hi, best_x + 2.0 * dw)
            if hi > lo:
                # Golden section
                gr = (math.sqrt(5.0) - 1.0) / 2.0
                a_g = lo
                b_g = hi
                c_g = b_g - gr * (b_g - a_g)
                d_g = a_g + gr * (b_g - a_g)
                fc = float(eval_xs(np.array([c_g]))[0])
                fd = float(eval_xs(np.array([d_g]))[0])
                for _ in range(30):
                    if time.monotonic() > stop_time - 0.3:
                        break
                    if fc < fd:
                        b_g = d_g
                        d_g = c_g
                        fd = fc
                        c_g = b_g - gr * (b_g - a_g)
                        fc = float(eval_xs(np.array([c_g]))[0])
                    else:
                        a_g = c_g
                        c_g = d_g
                        fc = fd
                        d_g = a_g + gr * (b_g - a_g)
                        fd = float(eval_xs(np.array([d_g]))[0])
                gx = (a_g + b_g) / 2.0
                gv = float(eval_xs(np.array([gx]))[0])
                if math.isfinite(gv) and gv < best_val:
                    best_x = gx
                    best_val = gv

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= 12.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start_i:end_i] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        # Rescale
        if S_total > 1e-300 and math.isfinite(S_total):
            inv = 1.0 / S_total
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        np.maximum(c, 0.0, out=c)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 5.0:
                    # Diversification
                    try:
                        scale = rng.choice([0.01, 0.02, 0.05, 0.1])
                        p = p * np.exp(rng.normal(0.0, scale, n, dtype=np.float64))
                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss > 0:
                            p /= ss
                        c = _conv_self(p)
                        S_total = 1.0
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
                            best_score = cur_score
                            best_p = p.copy()
                        flat = 0
                    except Exception:
                        break
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    c = _conv_self(p)
    sc = float(2.0 * n * c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _lp_direction(f, stop_time):
    """LP-based direction: maximize sum(g) s.t. f*g <= M, g >= 0."""
    n = int(f.size)
    if n < 10 or n > 5000:
        return None
    M = float(_conv_self(f).max())
    if M <= 0 or not math.isfinite(M):
        return None

    # Build constraint matrix A: A[k,j] = f[j] for j <= k <= j+n-1
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f[j]

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        constraints = [A @ g <= b]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        t_start = time.monotonic()
        for slv_name in ['ECOS', 'SCS', 'CLP']:
            try:
                if time.monotonic() > stop_time - 5.0:
                    break
                prob.solve(solver=slv_name, verbose=False,
                          **({'max_iters': 20000} if slv_name == 'SCS' else {}))
                if g.value is not None:
                    break
            except Exception:
                continue
        if g.value is None:
            return None
        g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        gv = float(g_val.sum())
        if gv < 1e-10 or not math.isfinite(gv):
            return None
        return g_val / gv
    except Exception:
        return None


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 5.0:
        hard = t0 + max(2.0, budget - 1.0)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64) / 64.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_arr = a.copy()
            best_score = sc

    n_par = int(best_arr.size)
    work = best_arr.copy() / float(best_arr.sum())

    # Phase 1: LSE optimization
    p1_end = min(hard - 80.0, t0 + 0.30 * budget)
    if p1_end > t0 + 30.0 and n_par >= 128:
        p1 = work.copy()
        try:
            p1 = _lbfgs_lse(p1, p1_end, consider=consider)
            if p1 is not None and p1.size == n_par and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass
    work = best_arr.copy() / float(best_arr.sum())

    # Phase 2: Aggressive coordinate descent
    p2_end = min(hard - 100.0, t0 + 0.55 * budget)
    if p2_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord_descent(work, p2_end, seed=int(seed + 7777), consider=consider)
        except Exception:
            pass
    work = best_arr.copy() / float(best_arr.sum())

    # Phase 3: LP direction + blend
    lp_end = min(hard - 80.0, time.monotonic() + 80.0)
    if lp_end > time.monotonic() + 15.0:
        try:
            g_dir = _lp_direction(work, lp_end)
            if g_dir is not None and g_dir.size == n_par:
                for t_alpha in [0.3, 0.2, 0.12, 0.06]:
                    blended = (1.0 - t_alpha) * work + t_alpha * g_dir
                    blended = np.maximum(blended, 0.0)
                    bs = float(blended.sum())
                    if bs > 0:
                        blended /= bs
                        consider(blended)
                        if blended.sum() > 0:
                            consider(blended)
        except Exception:
            pass
    work = best_arr.copy() / float(best_arr.sum())

    # Phase 4: More coordinate descent (different seed)
    p4_end = min(hard - 60.0, t0 + 0.78 * budget)
    if p4_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord_descent(work, p4_end, seed=int(seed + 99999), consider=consider)
        except Exception:
            pass
    work = best_arr.copy() / float(best_arr.sum())

    # Phase 5: Fine LSE
    p5_end = min(hard - 30.0, t0 + budget - 40.0)
    if p5_end > time.monotonic() + 20.0:
        try:
            p5 = _lbfgs_lse(work, p5_end, consider=consider)
            if p5 is not None and p5.size == n_par and np.all(np.isfinite(p5)) and p5.sum() > 0:
                p5 = p5 / float(p5.sum())
                consider(p5)
        except Exception:
            pass
    work = best_arr.copy() / float(best_arr.sum())

    # Phase 6: Final coordinate descent
    p6_end = min(hard - 15.0, t0 + budget - 15.0)
    if p6_end > time.monotonic() + 10.0:
        try:
            work, _ = _coord_descent(work, p6_end, seed=int(seed + 314159), consider=consider)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
