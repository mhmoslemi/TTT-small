def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0/n, dtype=np.float64)
    if src.size == n:
        return src.copy()
    m = src.size
    xp = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
    xq = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    out = np.interp(xq, xp, src)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0/n, dtype=np.float64)
    return out / s

    parent = _get_parent()
    n = int(parent.size)

    best_arr = _normalize(parent.copy())
    try:
        best_score = float(_score(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.full(64, 1.0/64.0, dtype=np.float64)
        best_score = 2.0

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
            if arr.size == 0:
                return
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            s = float(arr.sum())
            if not math.isfinite(s) or s < 0.01:
                return
            arr = arr / s
            sc = _score(arr)
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_arr = arr.copy()
                best_score = sc
        except Exception:
            pass

    hard_end = t0 + budget - reserve

    # Phase 1: LSE annealing
    l1_end = min(hard_end - 120.0, t0 + 0.35 * (budget - reserve))
    if l1_end > t0 + 30.0:
        try:
            p1 = _lse_anneal(best_arr.copy(), l1_end, consider, seed=seed)
            consider(p1)
        except Exception:
            pass

    # Phase 2: Coordinate polish
    c2_end = min(hard_end - 60.0, t0 + 0.65 * (budget - reserve))
    if c2_end > time.monotonic() + 20.0:
        try:
            p2, _ = _coord_polish(best_arr.copy(), c2_end, G=80, seed=seed+100, consider=consider)
            consider(p2)
        except Exception:
            pass

    # Phase 3: Another LSE + coord cycle
    l3_end = min(hard_end - 40.0, t0 + 0.82 * (budget - reserve))
    if l3_end > time.monotonic() + 15.0:
        try:
            p3 = _lse_anneal(best_arr.copy(), l3_end, consider, seed=seed+500)
            consider(p3)
        except Exception:
            pass

    # Phase 4: Final tight coordinate polish
    if time.monotonic() < hard_end - 10.0:
        try:
            p4, _ = _coord_polish(best_arr.copy(), hard_end - 2.0, G=90, seed=seed+999, consider=consider)
            consider(p4)
        except Exception:
            pass

    # Final validation and return
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = _normalize(parent)
            s = float(out.sum())
        if out.size == 0 or not math.isfinite(s) or s < 0.01:
            out = np.full(64, 1.0/64.0, dtype=np.float64)
        out = out / float(out.sum())
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            return [1.0]
        return ls
    except Exception:
        return [1.0]
