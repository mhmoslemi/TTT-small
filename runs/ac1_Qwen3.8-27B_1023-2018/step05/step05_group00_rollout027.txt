def _score_fft(a):

def _lp_direction(f, stop_time):
    n = int(f.size)
    if n < 20:
        return None
    c = _conv_self(f)
    M = float(c.max())
    if M <= 0 or not math.isfinite(M):
        return None

    # Build banded constraint matrix: A[k,j] = f[j] for max(0,k-n+1) <= j <= min(k,n-1)
    # This is a banded matrix of size (2n-1) x n with bandwidth n
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f[j]

    b = np.full(2 * n - 1, M, dtype=np.float64)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= b]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)

        solved = False
        for slv_name in [cp.CLIP, cp.SCS, cp.ECOS, cp.CBC]:
            if time.monotonic() > stop_time - 3.0:
                break
            try:
                prob.solve(solver=slv_name, verbose=False)
                solved = True
                break
            except Exception:
                continue

        if g.value is None:
            return None
        g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        gv = float(g_val.sum())
        if gv < 1e-10 or not math.isfinite(gv):
            return None
        return g_val / gv
    except Exception:
        return None


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 3.0:
        hard = t0 + max(2.0, budget - 1.0)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64)
        best_arr /= 64.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_arr = a.copy()
            best_score = sc

    n_par = int(best_arr.size)

    # === Phase 1: LSE optimization (coarse to fine) ===
    phase1_end = min(hard - 120.0, t0 + 0.25 * budget)
    if phase1_end > t0 + 20.0 and n_par >= 128:
        p1 = best_arr.copy()
        p1 = p1 / float(p1.sum())
        try:
            p1 = _lse_optimize(p1, phase1_end, consider=consider, fine_only=False)
            if p1 is not None and p1.size == n_par and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 2: Main coordinate descent ===
    phase2_end = min(hard - 150.0, t0 + 0.50 * budget)
    if phase2_end > time.monotonic() + 30.0:
        try:
            work, _ = _coord_descent(work, phase2_end, seed=int(seed + 1),
                                     consider=consider)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 3: LP direction for diversification ===
    lp_end = min(hard - 100.0, time.monotonic() + 80.0)
    if lp_end > time.monotonic() + 20.0 and np.all(np.isfinite(work)):
        g_dir = _lp_direction(work, lp_end)
        if g_dir is not None and g_dir.size == n_par:
            for t_alpha in [0.3, 0.2, 0.1, 0.05]:
                blended = (1.0 - t_alpha) * work + t_alpha * g_dir
                blended = np.maximum(blended, 0.0)
                bs = float(blended.sum())
                if bs > 0:
                    blended /= bs
                    consider(blended)

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 4: Second coordinate descent with different seed ===
    phase4_end = min(hard - 80.0, t0 + 0.72 * budget)
    if phase4_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord_descent(work, phase4_end, seed=int(seed + 9999),
                                     consider=consider)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 5: Fine LSE ===
    phase5_end = min(hard - 40.0, t0 + 0.85 * budget)
    if phase5_end > time.monotonic() + 20.0:
        lse5_end = min(phase5_end - 20.0, time.monotonic() + 0.4 * (phase5_end - time.monotonic()))
        if lse5_end > time.monotonic() + 15.0:
            try:
                p5 = _lse_optimize(work, lse5_end, consider=consider, fine_only=True)
                if p5 is not None and p5.size == n_par and np.all(np.isfinite(p5)) and p5.sum() > 0:
                    p5 = p5 / float(p5.sum())
                    consider(p5)
            except Exception:
                pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 6: Final coordinate polish ===
    phase6_end = min(hard - 20.0, t0 + budget - 35.0)
    if phase6_end > time.monotonic() + 15.0:
        try:
            work, _ = _coord_descent(work, phase6_end, seed=int(seed + 77777),
                                     consider=consider)
        except Exception:
            pass

    # === Phase 7: Try alternative length if time permits ===
    rem = hard - 25.0 - time.monotonic()
    if rem > 100.0:
        # Try a shorter sequence (downsample)
        n2 = max(512, n_par // 2)
        xp = (np.arange(n_par, dtype=np.float64) + 0.5) / float(n_par)
        fp = np.interp((np.arange(n2, dtype=np.float64) + 0.5) / float(n2), xp, best_arr)
        p2 = np.maximum(fp, 0.0)
        p2s = float(p2.sum())
        if p2s > 0:
            p2 /= p2s
            consider(p2)
            if rem > 80.0:
                p2_end = min(hard - 30.0, time.monotonic() + rem - 40.0)
                if p2_end > time.monotonic() + 20.0:
                    try:
                        p2, _ = _coord_descent(p2, p2_end, seed=int(seed + 424242),
                                               consider=consider)
                    except Exception:
                        pass

    # === Phase 8: One more coordinate polish on overall best ===
    fin_end = min(hard - 15.0, t0 + budget - 15.0)
    if fin_end > time.monotonic() + 10.0:
        work = best_arr.copy()
        work = work / float(work.sum())
        try:
            work, _ = _coord_descent(work, fin_end, seed=int(seed + 314159),
                                     consider=consider)
        except Exception:
            pass

    # === Final validation ===
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0 / 64.0] * 64
        return ls
    except Exception:
        return [1.0 / 64.0] * 64
