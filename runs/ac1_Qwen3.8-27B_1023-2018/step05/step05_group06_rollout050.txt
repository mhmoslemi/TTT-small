import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _conv_self(a):
    """FFT-based self-convolution."""
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _exact_score(p):
    """Exact score matching evaluator."""
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(p, p)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _fft_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(p)
    return float(2.0 * n * c.max() / (s * s))


def _coord_descent(p, stop_time, consider_callback, seed=12345):

    Coordinate descent minimizing 2n*max(c)/S^2.
    Maintains convolution c incrementally.
    """
    n = len(p)

    S = float(p.sum())
    if S < 1e-300:
        return p, math.inf
    p = p / S
    S = 1.0

    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    G = 32  # grid points

        sweep_count += 1
        improved = False
        order = rng.permutation(n)

                break
            if S <= 1e-300:

                c *= inv * inv

            start = i
            end = min(i + n, Ltot)

            if Ls <= 0:

            infl = 2.0 * p[:Ls]

            if start > 0:
                mc1 = float(c[:start].max())
                if mc1 > max_const:
                    max_const = mc1
            if end < Ltot:
                mc2 = float(c[end:].max())
                if mc2 > max_const:
                    max_const = mc2

            base = c[start:end].copy()
            diag_offset = 2 * i - start  # position in window where diagonal correction applies

            # Search range for new value x (absolute, not delta)
            x_hi = max(12.0 * v, 30.0 / n, 0.01)
            x_hi = min(x_hi, 3.0)
            if not math.isfinite(x_hi) or x_hi <= 0:

            # Evaluate on grid in x-space (absolute new value)
            delta = x_hi - v  # max delta

            deltas = pts - v  # deltas corresponding to each x

            # Window update: base + delta * infl + delta^2 * e_diag
            wins = deltas[:, None] * infl[None, :]  # (G+1, Ls)

            maff = aff.max(axis=1)

            maff = np.maximum(maff, 0.0)
            denoms = S + deltas
            vals = np.where(denoms > 1e-15, 2.0 * n * maff / (denoms * denoms), math.inf)

            imin = int(np.argmin(vals))

            # Apply update if improved

                if abs(d) > 1e-16:

            else:

                    a_old = p[:Ls].copy()
                    c[start:end] += 2.0 * d * a_old
                    if 0 <= diag_offset < Ls:
                        c[start + diag_offset] += d * d
                    p[i] = best_x
                    S += d
                    cur_score = best_val
                    improved = True

        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        if sweep_count % 5 == 0:
            c = _conv_self(p)
            np.maximum(c, 0.0, out=c)
            cur_score = float(2.0 * n * c.max())

        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
            consider_callback(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3 and time.monotonic() < stop_time - 60.0:
                sigma = 0.01 + 0.005 * min(flat, 20)
                noise = rng.normal(0.0, sigma, n)
                p *= np.exp(noise)
                p = np.maximum(p, 0.0)
                ss = float(p.sum())
                if ss > 1e-300:
                    p /= ss
                    S = 1.0
                    c = _conv_self(p)
                    np.maximum(c, 0.0, out=c)
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score:
                        best_score = cur_score
                        best_p = p.copy()
                        consider_callback(best_p)
                flat = 0

        if flat >= 20:
            sigma = 0.04
            noise = rng.normal(0.0, sigma, n)
            p *= np.exp(noise)
            p = np.maximum(p, 0.0)
            ss = float(p.sum())
            if ss > 1e-300:
                p /= ss
                S = 1.0
                c = _conv_self(p)
                np.maximum(c, 0.0, out=c)
                cur_score = float(2.0 * n * c.max())
            else:
                break
            flat = 0
            if flat >= 30:
                break

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    return best_p, best_score


def _focus_polish(p, stop_time, consider_callback, top_k=400):
    n = len(p)
    p = np.maximum(p, 0.0).astype(np.float64).copy()
    S = float(p.sum())
    if S < 1e-300:
        return p, math.inf
    p = p / S
    S = 1.0

    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(99999)

    cmax = float(c.max())
    active_thresh = cmax * 0.99
    active_lags = np.where(c >= active_thresh)[0]

    important = np.zeros(n, dtype=bool)
    for lag in active_lags:
        lo = max(0, int(lag) - n + 1)
        hi = min(n - 1, int(lag))
        important[lo:hi + 1] = True
    important_idx = np.where(important)[0]
    other_idx = np.where(~important)[0]

    if len(important_idx) > top_k:
        counts = np.zeros(n, dtype=np.int32)
        for lag in active_lags:
            lo = max(0, int(lag) - n + 1)
            hi = min(n - 1, int(lag))
            counts[lo:hi + 1] += 1
        important_idx = important_idx[np.argsort(-counts[important_idx])][:top_k]

    G_fine = 48
    sweep = 0
    flat = 0

    while time.monotonic() < stop_time - 2.0:
        sweep += 1
        improved = False

        if sweep % 3 < 2:
            order = rng.permutation(important_idx)
        else:
            order = rng.permutation(other_idx)

        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            if S <= 1e-300:
                break
            if S > 1.15 or S < 0.87:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0

            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            a_old_window = p[:Ls].copy()
            infl = 2.0 * a_old_window

            max_const = 0.0
            if start > 0:
                mc1 = float(c[:start].max())
                if mc1 > max_const:
                    max_const = mc1
            if end < Ltot:
                mc2 = float(c[end:].max())
                if mc2 > max_const:
                    max_const = mc2

            base = c[start:end].copy()
            diag_offset = 2 * i - start

            x_hi = max(6.0 * v, 15.0 / n, 0.005)
            x_hi = min(x_hi, max(2.0 * v, 0.01) + 0.02)
            x_lo = max(0.0, v - x_hi * 0.4)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = v + 5e-4
                x_lo = max(0.0, v - 5e-4)

            pts = np.linspace(x_lo, x_hi, G_fine + 1)
            ws = pts - v
            wins = ws[:, None] * infl[None, :]
            aff = base[None, :] + wins
            if 0 <= diag_offset < Ls:
                aff[:, diag_offset] += ws[:, None] * ws[:, None]
            maff = aff.max(axis=1)
            maff = np.maximum(maff, max_const)
            maff = np.maximum(maff, 0.0)
            denoms = S + ws
            vals = np.where(denoms > 1e-15, 2.0 * n * maff / (denoms * denoms), math.inf)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if G_fine > 4 and math.isfinite(best_val):
                dw = (x_hi - x_lo) / G_fine
                lo_gs = max(0.0, best_x - 1.5 * dw)
                hi_gs = min(x_hi, best_x + 1.5 * dw)
                if hi_gs - lo_gs > 1e-14:
                    gr = (math.sqrt(5.0) - 1.0) / 2.0
                    a_gs, b_gs = lo_gs, hi_gs
                    c_pt = b_gs - gr * (b_gs - a_gs)
                    d_pt = a_gs + gr * (b_gs - a_gs)

                    def eval_x2(x):
                        d = x - v
                        denom = S + d
                        if denom <= 1e-15:
                            return math.inf
                        nb = base + d * infl
                        if 0 <= diag_offset < Ls:
                            nb = nb.copy()
                            nb[diag_offset] += d * d
                        mv = float(nb.max())
                        if max_const > mv:
                            mv = max_const
                        return 2.0 * n * mv / (denom * denom)

                    f_c = eval_x2(c_pt)
                    f_d = eval_x2(d_pt)
                    for _ in range(14):
                        if f_c < f_d:
                            b_gs = d_pt
                            d_pt = c_pt
                            f_d = f_c
                            c_pt = b_gs - gr * (b_gs - a_gs)
                            f_c = eval_x2(c_pt)
                        else:
                            a_gs = c_pt
                            c_pt = d_pt
                            f_c = f_d
                            d_pt = a_gs + gr * (b_gs - a_gs)
                            f_d = eval_x2(d_pt)
                    x_opt = (a_gs + b_gs) * 0.5
                    f_opt = eval_x2(x_opt)
                    if f_opt < best_val:
                        best_x, best_val = x_opt, f_opt

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and best_x >= 0:
                d = best_x - v
                if abs(d) > 1e-16:
                    c[start:end] += 2.0 * d * a_old_window
                    if 0 <= diag_offset < Ls:
                        c[start + diag_offset] += d * d
                    p[i] = best_x
                    S += d
                    cur_score = best_val
                    improved = True

        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        if sweep % 3 == 0:
            c = _conv_self(p)
            np.maximum(c, 0.0, out=c)

        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
            consider_callback(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 6:
                break

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    return best_p, best_score


def _lbfgs_warm(p, n, stop_time, consider_callback):
    from scipy.optimize import minimize

    p = p.copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    pp_safe = np.maximum(p, 1e-30)
    u = np.log(pp_safe)
    u -= float(np.mean(u))

    L = 2 * n - 1
    size = next_fast_len(L)
    cross_size = next_fast_len(2 * n - 1)

    def softmax(uu):
        m = float(np.max(uu))
        e = np.exp(uu - m)
        s2 = float(e.sum())
        if s2 <= 0 or not math.isfinite(s2):
            return np.full(n, 1.0 / n)
        return e / s2

    def lse_func(uu, T):
        a = softmax(uu)
        F = rfft(a, size)
        c = irfft(F * F, size)[:L]
        np.maximum(c, 0.0, out=c)
        cmax = float(c.max())
        if not math.isfinite(cmax):
            return 1e100, np.zeros(n)
        if T <= 0:
            T = 1e-4 * max(cmax, 1e-10)
        scaled = np.clip((c - cmax) / T, -400, 400)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if sum_exp <= 0 or not math.isfinite(sum_exp):
            w = np.zeros(L)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * math.log(sum_exp / L)
            w = exps / sum_exp

        cd = soft:: np.zeros(n)  # placeholder, computed below
        # grad_a[i] = 2 * sum_{m=0}^{n-1} a[m] * w[i+m]
        corr_va = np.correlate(w, a, mode='valid')
        grad_a = 2.0 * corr_va
        tdp = float(np.dot(a, grad_a))
        grad_u = a * (grad_a - tdp)
        grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
        return float(f), grad_u

    best_u = u.copy()
    stages_temps = [0.03, 0.008, 0.002]
    elapsed_stage = 0
    total_plan = 90

    for T in stages_temps:
        if time.monotonic() > stop_time - 5.0:
            break
        stage_time = min(total_plan / len(stages_temps), stop_time - time.monotonic() - 5.0)
        if stage_time < 5.0:
            break
        try:
            res = minimize(lse_func, best_u, args=(T,), method='L-BFGS-B', jac=True,
                           options={'maxiter': 60, 'gtol': 1e-10, 'ftol': 1e-14,
                                    'maxls': 30, 'maxcor': 20, 'disp': False})
            if res.x is not None and np.all(np.isfinite(res.x)):
                fv, _ = lse_func(res.x, T)
                if math.isfinite(fv):
                    best_u = res.x.copy()
        except Exception:
            pass

    pp = softmax(best_u)
    pp = np.maximum(pp, 0.0)
    ss = float(pp.sum())
    if ss > 1e-300:
        pp /= ss
    else:
        pp = p
    consider_callback(pp)
    return pp


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 20.0

    # Get parent
    try:
        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(256, dtype=np.float64)
    if p.size == 0:
        p = np.ones(256, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(256, dtype=np.float64)
    n = len(p)
    p = p / float(p.sum())

    # Track best
    best_p = p.copy()
    best_score = _fft_score(p)

    # Validate with exact evaluator
    try:
        ls_init = [float(x) for x in best_p]
        exact_sc = float(evaluate_sequence(ls_init))
        if math.isfinite(exact_sc):
            best_score = exact_sc
    except Exception:
        pass

    def consider(arr):
        nonlocal best_p, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.clip(a, 0.0, 1000.0)
            a = np.where(np.isfinite(a), a, 0.0)
            if a.size == 0 or float(a.sum()) < 0.01:
                return
            sc = _fft_score(a)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                a = a / float(a.sum())
                best_p = a.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 1: L-BFGS warm-up (~90s)
    lbfgs_stop = min(hard - 1100.0, t0 + 90.0)
    if lbfgs_stop > time.monotonic() + 15.0:
        try:
            p2 = _lbfgs_warm(p, n, lbfgs_stop, consider)
            if p2 is not None and len(p2) == n and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p2 = p2 / float(p2.sum())
                consider(p2)
                if _fft_score(p2) < _fft_score(best_p):
                    p = p2.copy()
                else:
                    p = best_p.copy()
            else:
                p = best_p.copy()
        except Exception:
            p = best_p.copy()

    # Phase 2: Main coordinate descent (~950s)
    coord_stop = min(hard - 150.0, t0 + 1050.0)
    if coord_stop > time.monotonic() + 30.0:
        try:
            p3 = best_p.copy()
            p3, sc3 = _coord_descent(p3, coord_stop, consider, seed=seed + 777)
            if p3 is not None and len(p3) == n and np.all(np.isfinite(p3)) and p3.sum() > 0:
                consider(p3)
        except Exception:
            pass

    # Phase 3: Focus polish (~150s)
    polish_stop = min(hard - 50.0, t0 + 1200.0)
    if polish_stop > time.monotonic() + 25.0:
        try:
            p4 = best_p.copy()
            p4, sc4 = _focus_polish(p4, polish_stop, consider, top_k=300)
            if p4 is not None and len(p4) == n and np.all(np.isfinite(p4)) and p4.sum() > 0:
                consider(p4)
        except Exception:
            pass

    # Phase 4: Second coordinate descent round if time remains
    rem = hard - 35.0
    if rem > time.monotonic() + 40.0:
        try:
            p5 = best_p.copy()
            p5, sc5 = _coord_descent(p5, rem, consider, seed=seed + 99999)
            if p5 is not None and len(p5) == n and np.all(np.isfinite(p5)) and p5.sum() > 0:
                consider(p5)
        except Exception:
            pass

    # Phase 5: Final short polish
    fin_stop = hard - 15.0
    if fin_stop > time.monotonic() + 15.0:
        try:
            p6 = best_p.copy()
            p6, sc6 = _focus_polish(p6, fin_stop, consider, top_k=200)
            if p6 is not None and len(p6) == n and np.all(np.isfinite(p6)) and p6.sum() > 0:
                consider(p6)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        out = np.asarray(best_p, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(128, 1.0 / 128.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            # Fallback to parent
            try:
                pp = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
                pp = np.clip(pp, 0.0, 1000.0)
                pp = np.where(np.isfinite(pp), pp, 0.0)
                ls = [float(x) for x in pp]
                final_sc = float(evaluate_sequence(ls))
            except Exception:
                return [1.0]
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
