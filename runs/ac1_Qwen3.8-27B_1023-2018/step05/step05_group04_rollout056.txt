    sweep = 0
    GR = (math.sqrt(5.0) - 1.0) / 2.0

    def eval_1d(w, base, infl, diag, max_const, S):
        s_new = S + w
        if s_new <= 1e-15:
            return float('inf')
        aff_max = float(np.max(base + w * infl))
        if 0 <= diag < len(base):
            dval = base[diag] + w * infl[diag] + w * w
            if dval > aff_max:
                aff_max = dval
        m = max(aff_max, max_const)
        return 2.0 * n * m / (s_new * s_new)

    max_sweeps = 20000
    G = 40
    while sweep < max_sweeps and time.monotonic() < stop_time - 4.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if (idx_i & 127) == 0 and time.monotonic() > stop_time - 1.0:
                break
            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue
            base = c[start_i:end_i].copy()
            infl = 2.0 * p[:Ls]
            diag = i

            max_const = 0.0
            if start_i > 0:
                mc = float(c[:start_i].max())
                if mc > max_const:
                    max_const = mc
            if end_i < Ltot:
                mc = float(c[end_i:].max())
                if mc > max_const:
                    max_const = mc

            S = 1.0
            x_hi = max(20.0 * v, 10.0 / n, 0.005)
            x_hi = min(x_hi, 0.4)
            x_hi = max(x_hi, 1e-8)

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                ins = np.searchsorted(pts, v)
                pts = np.insert(pts, ins, v)

            ws = pts - v
            s_new = S + ws
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag < Ls:
                aff[:, diag] += ws * ws
            maff = np.max(aff, axis=1)
            maff = np.maximum(maff, max_const)
            valid = s_new > 1e-12
            scores = np.full(pts.shape, float('inf'), dtype=np.float64)
            if np.any(valid):
                scores[valid] = 2.0 * n * maff[valid] / (s_new[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            if 0 < imin < len(pts) - 1:
                lo = float(pts[max(0, imin - 2)])
                hi = float(pts[min(len(pts) - 1, imin + 2)])
            else:
                lo = 0.0
                hi = x_hi
            if hi - lo < 1e-14:
                lo = max(0.0, best_x - 1e-6)
                hi = best_x + 1e-6

            a_gs = lo
            b_gs = hi
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)
            f_c = eval_1d(c_gs - v, base, infl, diag, max_const, S)
            f_d = eval_1d(d_gs - v, base, infl, diag, max_const, S)
            for _ in range(18):
                if b_gs - a_gs < 1e-13 * max(1.0, abs(best_x)):
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = eval_1d(c_gs - v, base, infl, diag, max_const, S)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_f = eval_1d(d_gs - v, base, infl, diag, max_const, S)
                    f_d = f_f
            x_mid = (a_gs + b_gs) / 2.0
            f_mid = eval_1d(x_mid - v, base, infl, diag, max_const, S)
            if f_mid < best_val:
                best_x = x_mid
                best_val = f_mid

            if math.isfinite(best_val) and best_val < cur_score - 5e-15 and 0.0 <= best_x:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    aff_seg = base + w * infl
                    if 0 <= diag < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag] += w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    c[start_i:end_i] = aff_seg
                    cur_score = 2.0 * n * new_max
                    improved = True

        s_new = float(p.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            p *= (1.0 / s_new)
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-15:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 20.0:
                    kick_size = 0.015 + 0.03 * min(1.0, flat / 15.0)
                    p = p * np.exp(rng.normal(0.0, kick_size, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 28.0
    if hard <= t0 + 5.0:
        hard = t0 + max(3.0, budget - 2.0)

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float('inf')

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()

    consider(parent)

    n_work = int(best_arr.size)
    if n_work < 128:
        n_work = 384
    if n_work > 6144:
        n_work = 4096

    p_work = _resample(best_arr, n_work)

    # Phase 1: LSE optimization
    lse_time = min(150.0, 0.12 * budget)
    lse_end = t0 + lse_time
    if lse_end > time.monotonic() + 20.0:
        try:
            p2 = _lse_optimize(p_work, lse_end, consider=consider)
            if p2 is not None and p2.size == n_work and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p_work = p2 / float(p2.sum())
                consider(p_work)
        except Exception:
            pass

    # Phase 2: Main coordinate descent
    cd_end = hard - 8.0
    if cd_end > time.monotonic() + 20.0:
        p_start = _resample(best_arr, n_work)
        try:
            p_cd, _ = _coord_descent(p_start, cd_end, consider=consider,
                                     seed=int(seed + 7777))
            if p_cd is not None and p_cd.size > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                consider(p_cd)
        except Exception:
            pass

    # Phase 3: Different initialization + CD
    rem = hard - 6.0 - time.monotonic()
    if rem > 80.0:
        x = (np.arange(n_work, dtype=np.float64) + 0.5) / n_work
        # Smooth bump: raised cosine
        bump = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5)))
        bump = np.maximum(bump, 1e-6)
        s_b = float(bump.sum())
        if s_b > 0:
            bump /= s_b
        p_blend = 0.4 * bump + 0.6 * _resample(best_arr, n_work)
        p_blend = np.maximum(p_blend, 0.0)
        p_blend /= float(p_blend.sum())
        consider(p_blend)

        blend_end = min(hard - 6.0, time.monotonic() + rem * 0.65)
        if blend_end > time.monotonic() + 25.0:
            try:
                p_b, _ = _coord_descent(p_blend, blend_end, consider=consider,
                                        seed=int(seed + 8888))
                if p_b is not None and p_b.size > 0 and np.all(np.isfinite(p_b)) and p_b.sum() > 0:
                    consider(p_b)
            except Exception:
                pass

    # Phase 4: Final squeeze on overall best
    rem = hard - 10.0 - time.monotonic()
    if rem > 25.0:
        p_final = _resample(best_arr, n_work)
        try:
            p_f, _ = _coord_descent(p_final, hard - 10.0, consider=consider,
                                    seed=int(seed + 9999))
            if p_f is not None and p_f.size > 0 and np.all(np.isfinite(p_f)) and p_f.sum() > 0:
                consider(p_f)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
