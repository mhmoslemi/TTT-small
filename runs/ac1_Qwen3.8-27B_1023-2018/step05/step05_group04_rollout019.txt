    L = 2 * n - 1
    size = next_fast_len(L)

    p = np.maximum(np.asarray(p, dtype=np.float64), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    u = np.log(np.maximum(p, 1e-30), dtype=np.float64)
    u = u - float(u.mean())
    
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

        F = rfft(a[::-1], cross_size)
        G = rfft(w, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]

        np.maximum(q, 0.0, out=q)
        return q

        c = cfast(pp)
        cmax = float(c.max())

            w[int(np.argmax(c))] = 1.0
        else:

        q = xfast(pp, w)
        grad = pp * (2.0 * q - float(np.dot(pp, 2.0 * q)))
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return f, grad

        (0.08, 0.12),
        (0.03, 0.14),
        (0.01, 0.16),
        (0.004, 0.18),
        (0.0015, 0.20),
        (0.0005, 0.20),

    cum = 0.0

        if now > stop_time - 8.0:
            break
        cum += wt
        end_stage = min(stop_time - 6.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:

        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)
        best_u = u.copy()
        f0, _ = lse_func(u, T)
        best_fstage = f0
        stale = 0
        while time.monotonic() < end_stage - 1.5:

            opts = {"maxiter": maxit, "gtol": 1e-12, "ftol": 1e-16, "maxls": 12, "maxdisp": 0}
            try:

            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_fstage - 1e-14:
                    best_fstage = fval
                    best_u = res.x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if getattr(res, "status", 1) == 0:

        p_cur = _softmax(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        sc = _score_arr(p_cur)
        if math.isfinite(sc) and sc < best_sc - 1e-13:

            u = np.log(np.maximum(p, 1e-30), dtype=np.float64)
            u = u - float(u.mean())
        if consider:
            consider(best_p)
    return best_p

def _coord_descent(p, stop_time, consider=None, seed=12345):
    n = int(p.size)
    if n == 0:
        return p, float('inf')
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p, float('inf')

    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()

    Ltot = 2 * n - 1
    G = 32  # grid points (reduced from 48)
    GR = (math.sqrt(5.0) - 1.0) / 2.0

        s_new = S + w
        if s_new <= 1e-15:
            return float('inf')
        aff = np.max(base + w * infl)
        if 0 <= diag < len(base):
            dval = base[diag] + w * infl[diag] + w * w
            if dval > aff:
                aff = dval
        m = max(aff, max_const)
        return 2.0 * n * m / (s_new * s_new)
    
    sweep = 0
    flat = 0
    while sweep < 100000 and time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 1.0:
                break

            start_k = i
            end_k = min(i + n, Ltot)
            Ls = end_k - start_k
            if Ls <= 0:
                continue
            
            base = c[start_k:end_k]
            infl = 2.0 * p[:Ls]

            max_const = 0.0

            if end_k < Ltot:
                mc = float(c[end_k:].max())
                if mc > max_const:
                    max_const = mc
            
            S = 1.0
            x_hi = max(15.0 * v, 10.0 / n, 0.005)
            x_hi = min(x_hi, 0.5)
            x_hi = max(x_hi, 1e-8)
            
            # Grid scan
            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                ins = np.searchsorted(pts, v)
                pts = np.insert(pts, ins, v)
            
            ws = pts - v
            s_news = S + ws
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag < Ls:
                aff[:, diag] += ws * ws
            maff = np.max(aff, axis=1)
            maff = np.maximum(maff, max_const)
            valid = s_news > 1e-12
            scores = np.full(pts.shape, float('inf'), dtype=np.float64)
            if np.any(valid):

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])
            
            # Golden section refinement
            if 0 < imin < len(pts) - 1:
                lo = float(pts[max(0, imin - 1)])
                hi = float(pts[min(len(pts) - 1, imin + 1)])
            else:
                lo, hi = 0.0, x_hi
            
            a_gs, b_gs = lo, hi
            if b_gs - a_gs > 1e-14:
                c_gs = b_gs - GR * (b_gs - a_gs)
                d_gs = a_gs + GR * (b_gs - a_gs)
                f_c = eval_1d(c_gs - v, base, infl, diag, max_const, S, n)

                        b_gs = d_gs; d_gs = c_gs; f_d = f_c
                        c_gs = b_gs - GR * (b_gs - a_gs)
                        f_c = eval_1d(c_gs - v, base, infl, diag, max_const, S, n)
                    else:
                        a_gs = c_gs; c_gs = d_gs; f_c = f_d
                        d_gs = a_gs + GR * (b_gs - a_gs)
                        f_d = eval_1d(d_gs - v, base, infl, diag, max_const, S, n)
                x_mid = (a_gs + b_gs) / 2.0
                f_mid = eval_1d(x_mid - v, base, infl, diag, max_const, S, n)
                if f_mid < best_val:
                    best_x, best_val = x_mid, f_mid
            
            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x:

                    p = p * np.exp(rng.normal(0.0, kick_size, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
            if flat > 40:
                break
        sweep += 1
    
    # Final check
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            consider(best_p)
    return best_p, best_score
