def _fft_conv(a):

    L = 2 * n - 1

    return c

            consecutive_no_improve += 1

        # Perturbation when stuck
        if consecutive_no_improve >= 12 and time.monotonic() < hard_end - 30.0:
            sigma = 0.05 + 0.04 * min(consecutive_no_improve - 12, 10)
            noise = rng.normal(0, sigma, n)
            q = a * np.exp(noise)
            qs = q.sum()
            if qs > 0 and math.isfinite(qs):
                q /= qs
                qc = _fft_conv(q)
                qc = np.maximum(qc, 0.0)
                qsc = 2.0 * n * qc.max()
                if math.isfinite(qsc):
                    a = q.copy()
                    c = qc
                    S = 1.0
                    infl = 2.0 * a
                    cur_score = qsc
            consecutive_no_improve = 0

        sweep += 1

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(30.0, budget)
    reserve = 25.0 if budget > 300 else 10.0
    hard_stop = t0 + budget - reserve

    parent = _get_parent()
    n = parent.size
    a = parent.copy()

    best_a = a.copy()
    best_score = _score_fft(a)

    def consider(a_new):
        nonlocal best_a, best_score
        try:
            arr = np.asarray(a_new, dtype=np.float64).ravel()
        except Exception:
            return
        if arr.size == 0:
            return
        arr = np.clip(np.where(np.isfinite(arr), arr, 0.0), 0.0, 1000.0)
        s = arr.sum()
        if not math.isfinite(s) or s < 0.01:
            return
        arr = arr / s
        sc = _score_fft(arr)
        if math.isfinite(sc) and sc < best_score - 1e-13:
            best_score = sc
            best_a = arr.copy()

    consider(a)

    # Phase 1: LSE optimization
    lse_stop = min(hard_stop - 100.0, t0 + max(30.0, 0.18 * budget))
    if time.monotonic() < lse_stop:
        try:
            a = _lse_phase(a, lse_stop, consider)
        except Exception:
            a = best_a.copy()

    # Phase 2: Coordinate descent (main workhorse)
    coord_stop = hard_stop - 30.0
    if time.monotonic() < coord_stop:
        try:
            a = best_a.copy()
            a, _ = _coord_descend(a, coord_stop, consider, seed=seed + 7)
        except Exception:
            pass

    # Phase 3: Final LSE polish
    final_lse_stop = hard_stop - 10.0
    if time.monotonic() < final_lse_stop:
        try:
            a = best_a.copy()
            a2 = _lse_phase(a, final_lse_stop, consider)
        except Exception:
            pass

    # Phase 4: Quick final coordinate polish
    if time.monotonic() < hard_stop - 5.0:
        try:
            a = best_a.copy()
            a, _ = _coord_descend(a, hard_stop - 3.0, consider, seed=seed + 999)
        except Exception:
            pass

    # Final return
    out = best_a.copy()
    out = np.clip(np.where(np.isfinite(out), out, 0.0), 0.0, 1000.0)
    s = float(out.sum())
    if out.size == 0 or not math.isfinite(s) or s < 0.01:
        out = parent.copy()
        s = float(out.sum())
    if out.size == 0 or s < 0.01:
        return [1.0]
    out = out / s
    ls = [float(x) for x in out]
    try:
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            ls2 = [float(x) for x in parent]
            if math.isfinite(evaluate_sequence(ls2)):
                return ls2
            return [1.0]
    except Exception:
        return [1.0]
    return ls
