                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= 12.0 * S:
                w = best_x - v
                if abs(w) > 1e-16:
                    p[i] = best_x
                    S_total += w
                    c[start_i:end_i] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_fft(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 6.0:
                    try:
                        p = p * np.exp(rng.normal(0.0, 0.02, n, dtype=np.float64))
                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss > 0.0:
                            p /= ss
                        c = _conv_fft(p)
                        S_total = 1.0
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider is not None:
                                consider(best_p)
                        elif math.isfinite(cur_score):
                            p = best_p.copy()
                            c = _conv_fft(p)
                            S_total = 1.0
                            cur_score = best_score
                        flat = 0
                    except Exception:
                        break
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_fft(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _pair_coord(p, stop_time, consider=None, seed=999):
    """Update pairs of coordinates simultaneously."""
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n < 4:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s
    Ltot = 2 * n - 1
    c = _conv_fft(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    G = 16
    iters = 0
    max_iters = max(100, int(60.0 / max(1.0, n / 100.0)))

    while iters < max_iters and time.monotonic() < stop_time - 1.0:
        improved = False
        for _ in range(n // 2):
            if time.monotonic() > stop_time - 0.5:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                j = (i + 1) % n
            if j < i:
                i, j = j, i
            vi = float(p[i])
            vj = float(p[j])
            S = 1.0

            # Determine affected range
            start_k = min(i, j)
            end_k = max(i, j) + n
            end_k = min(end_k, Ltot)
            Ls = end_k - start_k
            if Ls <= 0:
                continue

            max_const = 0.0
            if start_k > 0:
                ml = float(c[:start_k].max())
                if ml > max_const:
                    max_const = ml
            if end_k < Ltot:
                mr = float(c[end_k:].max())
                if mr > max_const:
                    max_const = mr

            base = c[start_k:end_k].copy()

            # Influence for changing i and j
            infl_i = 2.0 * np.zeros(Ls, dtype=np.float64)
            infl_j = 2.0 * np.zeros(Ls, dtype=np.float64)
            for kk in range(Ls):
                k = start_k + kk
                ki = k - i
                kj = k - j
                if 0 <= ki < n:
                    infl_i[kk] = 2.0 * p[ki]
                if 0 <= kj < n:
                    infl_j[kk] = 2.0 * p[kj]

            avg = S / float(n)
            xi_hi = max(15.0 * vi, 40.0 * avg, 0.02 * S, 1e-12)
            xj_hi = max(15.0 * vj, 40.0 * avg, 0.02 * S, 1e-12)
            xi_hi = min(xi_hi, 10.0 * S)
            xj_hi = min(xj_hi, 10.0 * S)

            xi_pts = np.linspace(0.0, xi_hi, G + 1)
            xj_pts = np.linspace(0.0, xj_hi, G + 1)
            if 0.0 <= vi <= xi_hi:
                xi_pts = np.unique(np.concatenate((xi_pts, [vi])))
            if 0.0 <= vj <= xj_hi:
                xj_pts = np.unique(np.concatenate((xj_pts, [vj])))

            best_xi = vi
            best_xj = vj
            best_val = cur_score

            for xi in xi_pts:
                wi = xi - vi
                for xj in xj_pts:
                    wj = xj - vj
                    denom = S + wi + wj
                    if denom < 1e-12:
                        continue
                    aff = base + wi * infl_i + wj * infl_j
                    maff = float(np.max(aff))
                    # Self terms
                    di = 2 * i - start_k
                    if 0 <= di < Ls:
                        maff = max(maff, base[di] + wi * infl_i[di] + wi * wi + wj * infl_j[di] + wj * wj * (1.0 if i == j else 0.0))
                    dj = 2 * j - start_k
                    if 0 <= dj < Ls and i != j:
                        maff = max(maff, base[dj] + wi * infl_i[dj] + wi * wi * (1.0 if i == j else 0.0) + wj * infl_j[dj] + wj * wj)
                    maff = max(maff, max_const)
                    val = (2.0 * n * maff) / (denom * denom)
                    if val < best_val:
                        best_val = val
                        best_xi = float(xi)
                        best_xj = float(xj)

            if best_val < cur_score - 1e-14:
                wi = best_xi - vi
                wj = best_xj - vj
                p[i] = best_xi
                p[j] = best_xj
                c[start_k:end_k] += wi * infl_i + wj * infl_j
                di = 2 * i
                if 0 <= di < Ltot:
                    c[di] += wi * wi
                dj = 2 * j
                if 0 <= dj < Ltot:
                    c[dj] += wj * wj
                cur_score = best_val
                improved = True

        # Renormalize
        ss = float(p.sum())
        if ss > 1e-300:
            p /= ss
            c /= ss * ss
        c = _conv_fft(p)
        new_s = float(2.0 * n * c.max())
        if math.isfinite(new_s) and new_s < best_score - 1e-15:
            best_score = new_s
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        iters += 1
        if not improved and iters > 3:
            break

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 15.0
    if hard <= t0 + 3.0:
        hard = t0 + max(2.0, budget - 1.0)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64)
        best_arr /= 64.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_arr = a.copy()
            best_score = sc

    n_par = int(best_arr.size)

    # === Phase 1: LSE optimization ===
    p1_end = min(hard - 80.0, t0 + 0.22 * budget)
    if p1_end > t0 + 25.0 and n_par >= 128:
        p1 = best_arr.copy()
        p1 = p1 / float(p1.sum())
        try:
            p1 = _lse_optimize(p1, p1_end, consider=consider, fine_only=False)
            if p1 is not None and p1.size == n_par and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 2: Coordinate descent (main) ===
    p2_end = min(hard - 100.0, t0 + 0.50 * budget)
    if p2_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord_polish(work, p2_end, max_sweeps=500000,
                                    seed=int(seed + 7777), consider=consider)
        except Exception:
            pass

    # === Phase 3: LSE refinement ===
    p3_end = min(hard - 80.0, time.monotonic() + 0.15 * (hard - time.monotonic()))
    if p3_end > time.monotonic() + 20.0:
        work = best_arr.copy()
        work = work / float(work.sum())
        try:
            p3 = _lse_optimize(work, p3_end, consider=consider, fine_only=True)
            if p3 is not None and p3.size == n_par and np.all(np.isfinite(p3)) and p3.sum() > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 4: More coordinate descent ===
    p4_end = min(hard - 80.0, t0 + 0.72 * budget)
    if p4_end > time.monotonic() + 15.0:
        try:
            work, _ = _coord_polish(work, p4_end, max_sweeps=400000,
                                    seed=int(seed + 123456), consider=consider)
        except Exception:
            pass

    # === Phase 5: Pair coordinate updates ===
    p5_end = min(hard - 50.0, time.monotonic() + 40.0)
    if p5_end > time.monotonic() + 10.0 and n_par <= 4096:
        work = best_arr.copy()
        work = work / float(work.sum())
        try:
            work, _ = _pair_coord(work, p5_end, consider=consider,
                                  seed=int(seed + 54321))
        except Exception:
            pass

    # === Phase 6: Final LSE + coordinate polish ===
    p6_end = min(hard - 20.0, t0 + budget - 20.0)
    if p6_end > time.monotonic() + 15.0:
        work = best_arr.copy()
        work = work / float(work.sum())
        lse6_end = min(p6_end - 20.0, time.monotonic() + 0.3 * (p6_end - time.monotonic()))
        if lse6_end > time.monotonic() + 10.0:
            try:
                p6 = _lse_optimize(work, lse6_end, consider=consider, fine_only=True)
                if p6 is not None and p6.size == n_par and np.all(np.isfinite(p6)) and p6.sum() > 0:
                    p6 = p6 / float(p6.sum())
                    consider(p6)
            except Exception:
                pass

        work = best_arr.copy()
        work = work / float(work.sum())
        coord6_end = p6_end
        if coord6_end > time.monotonic() + 10.0:
            try:
                work, _ = _coord_polish(work, coord6_end, max_sweeps=200000,
                                        seed=int(seed + 987654), consider=consider)
            except Exception:
                pass

    # === Phase 7: Try a different length ===
    rem = hard - 15.0 - time.monotonic()
    if rem > 100.0:
        n2 = min(2048, max(1024, int(n_par * 0.65)))
        xp = (np.arange(n_par, dtype=np.float64) + 0.5) / float(n_par)
        fp = np.interp((np.arange(n2, dtype=np.float64) + 0.5) / float(n2), xp, best_arr)
        p_alt = np.maximum(fp, 0.0)
        ps = float(p_alt.sum())
        if ps > 0:
            p_alt /= ps
            consider(p_alt)
            if rem > 80.0:
                alt_end = min(hard - 20.0, time.monotonic() + rem - 25.0)
                if alt_end > time.monotonic() + 15.0:
                    try:
                        p_alt, _ = _coord_polish(p_alt, alt_end, max_sweeps=80000,
                                                 seed=int(seed + 424242), consider=consider)
                    except Exception:
                        pass

    # === Final squeeze ===
    fin_end = min(hard - 10.0, t0 + budget - 8.0)
    if fin_end > time.monotonic() + 10.0:
        work = best_arr.copy()
        work = work / float(work.sum())
        try:
            work, _ = _coord_polish(work, fin_end, max_sweeps=80000,
                                    seed=int(seed + 555555), consider=consider)
        except Exception:
            pass

    # Final validation and return
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
