        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp))
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        return float(f), np.where(np.isfinite(grad), grad, 0.0)

    def p_to_u(pp):
        ll = np.log(np.maximum(pp, 1e-300))
        return ll - float(np.mean(ll))

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score_fft(p, conv_size, conv_len)

    stages = (
        (0.15, 0.10),
        (0.05, 0.12),
        (0.015, 0.15),
        (0.005, 0.18),
        (0.0015, 0.20),
        (0.0005, 0.18),
        (0.00015, 0.17),
    )

    total_w = sum(w for _, w in stages)
    elapsed_total = stop_time - time.monotonic() - 5.0
    if elapsed_total < 10:
        return best_p

    for idx, (frac, wt) in enumerate(stages):
        now = time.monotonic()
        if now > stop_time - 6.0:
            break
        end_stage = min(stop_time - 5.0, now + max(5.0, (wt / total_w) * elapsed_total))
        cmax = float(cfast(p).max())
        T = max(1e-15, frac * cmax)
        stale = 0

        while time.monotonic() < end_stage - 2.0:
            maxit = 40
            opts = {"maxiter": maxit, "gtol": 1e-13, "ftol": 1e-17,
                    "maxls": 30, "maxcor": 30, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval):
                    p_cur = _softmax(res.x)
                    sc_cur = _score_fft(p_cur, conv_size, conv_len)
                    if math.isfinite(sc_cur) and sc_cur < best_sc:
                        best_sc = sc_cur
                        best_p = p_cur.copy()
                        p = p_cur.copy()
                        u = p_to_u(p)
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 4:
                            break
            if getattr(res, "status", 1) == 0:
                break

    p_out = best_p.copy()
    ss = float(p_out.sum())
    if ss > 1e-300:
        p_out = p_out / ss
    return p_out


def _coord_descent(p, stop_time, consider, seed_val=42):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p, math.inf
    p = p / s

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    c = _conv_fft(p, conv_size)[:conv_len]
    S = 1.0
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed_val)
    flat = 0
    sweep = 0
    G = max(20, min(60, int(200000 / max(n, 1)) + 20))

    while sweep < 50000 and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S <= 1e-300:
                break
            if S > 1.3 or S < 0.7:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0

            v = float(p[i])
            start = i
            end = min(i + n, conv_len)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start > 0:
                ml = float(np.max(c[:start]))
                max_const = max(max_const, ml)
            if end < conv_len:
                mr = float(np.max(c[end:]))
                max_const = max(max_const, mr)

            x_hi = max(30.0 * v, 80.0 / n, 0.02)
            x_hi = min(x_hi, 5.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = 1.0 / n

            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 <= v <= x_hi:
                pts = np.unique(np.concatenate([pts, [v]]))

            w_arr = pts - v
            denom = S + w_arr
            valid = denom > 1e-14
            if np.any(valid):
                aff = base[None, :] + w_arr[None, :] * infl[None, :]
                diag_idx = i - start
                if 0 <= diag_idx < Ls:
                    dvals = base[diag_idx] + w_arr * infl[diag_idx] + w_arr * w_arr
                    aff = np.maximum(aff, dvals[:, None])
                aff = np.maximum(aff, max_const)
                mv = np.max(aff, axis=1)
                out = np.full(pts.shape, math.inf)
                dv = denom[valid]
                out[valid] = (2.0 * n * mv[valid]) / (dv * dv)

                if np.any(np.isfinite(out)):
                    imin = int(np.argmin(out))
                    bx = float(pts[imin])
                    bv = float(out[imin])

                    dw = x_hi / G
                    lo = max(0.0, bx - 4.0 * dw)
                    hi = min(x_hi, bx + 4.0 * dw)
                    if hi > lo:
                        fine = np.linspace(lo, hi, 50)
                        w_f = fine - v
                        denom_f = S + w_f
                        valid_f = denom_f > 1e-14
                        if np.any(valid_f):
                            aff_f = base[None, :] + w_f[None, :] * infl[None, :]
                            if 0 <= diag_idx < Ls:
                                dvals_f = base[diag_idx] + w_f * infl[diag_idx] + w_f * w_f
                                aff_f = np.maximum(aff_f, dvals_f[:, None])
                            aff_f = np.maximum(aff_f, max_const)
                            mv_f = np.max(aff_f, axis=1)
                            out_f = np.full(fine.shape, math.inf)
                            dv_f = denom_f[valid_f]
                            out_f[valid_f] = (2.0 * n * mv_f[valid_f]) / (dv_f * dv_f)
                            if np.any(np.isfinite(out_f)):
                                im2 = int(np.argmin(out_f))
                                if float(out_f[im2]) < bv:
                                    bv = float(out_f[im2])
                                    bx = float(fine[im2])

                    if math.isfinite(bv) and bv < cur_score - 1e-14 and 0.0 <= bx <= 10.0 * S:
                        w = bx - v
                        if abs(w) > 1e-15:
                            p[i] = bx
                            S += w
                            c[start:end] += w * infl
                            if 0 <= diag_idx < Ls:
                                c[start + diag_idx] += w * w
                            cur_score = bv
                            improved = True

        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        if sweep % 5 == 0:
            c = _conv_fft(p, conv_size)[:conv_len]
        new_sc = 2.0 * n * float(c.max())
        if math.isfinite(new_sc) and new_sc < best_score - 1e-14:
            best_score = new_sc
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 20.0:
                    noise_scale = max(0.005, 0.05 * (flat / 10.0))
                    p = p * np.exp(rng.normal(0, noise_scale, n))
                    p = np.maximum(p, 0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                        S = 1.0
                    c = _conv_fft(p, conv_size)[:conv_len]
                    cur_score = 2.0 * n * float(c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_fft(p, conv_size)[:conv_len]
    sc = 2.0 * n * float(c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _antisym_project(a):
    """Project onto the anti-symmetric subspace: a[i] + a[n-1-i] = constant."""
    n = a.size
    C = 2.0 * float(a.sum()) / n
    b = np.zeros(n)
    for i in range(n // 2):
        b[i] = a[i] - C / 2.0
        b[n - 1 - i] = -b[i]
    if n % 2 == 1:
        b[n // 2] = 0.0
    result = C / 2.0 + b
    return np.maximum(result, 0.0)


def _antisym_perturb(a, rng, scale):
    n = a.size
    b = np.zeros((n // 2) + 1)
    b[:n // 2] = rng.normal(0, scale * max(1.0, float(a.mean())), size=n // 2)
    result = a.copy()
    for i in range(n // 2):
        result[i] += b[i]
        result[n - 1 - i] -= b[i]
    return np.maximum(result, 0.0)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 5.0:
        hard = t0 + max(5.0, budget - 1.0)

    parent = _get_parent()
    n = int(parent.size)
    best_arr = parent.copy()
    best_score = math.inf
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_arr]))
    except Exception:
        pass
    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(128)
        n = best_arr.size
        best_score = float(evaluate_sequence([float(x) for x in best_arr]))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            sc = float(evaluate_sequence([float(x) for x in a]))
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_arr = a / s
                best_score = sc
        except Exception:
            pass

    rng = np.random.default_rng(int(seed))

    # Phase 1: LSE optimization on parent
    lse_stop = t0 + min(200.0, 0.20 * budget)
    if lse_stop > t0 + 30.0:
        try:
            p1 = _lse_optimize(best_arr, lse_stop, consider=consider, seed_val=int(seed))
            if p1 is not None and p1.size == n and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    # Phase 2: Coordinate descent (main)
    coord_stop = hard - 120.0
    if coord_stop > time.monotonic() + 60.0:
        try:
            p2, _ = _coord_descent(best_arr, coord_stop, consider=consider, seed_val=int(seed + 7))
            if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p2 = p2 / float(p2.sum())
                consider(p2)
        except Exception:
            pass

    # Phase 3: Anti-symmetric projection + LSE refinement
    rem = hard - 20.0 - time.monotonic()
    if rem > 50.0:
        # Try anti-symmetric projection
        try:
            pa = _antisym_project(best_arr)
            ps = float(pa.sum())
            if ps > 0.01:
                pa = pa / ps
                consider(pa)
        except Exception:
            pass

        # LSE on current best
        lse2_stop = time.monotonic() + max(25.0, 0.35 * rem)
        if lse2_stop < hard - 30.0:
            try:
                p3 = _lse_optimize(best_arr, lse2_stop, consider=consider, seed_val=int(seed + 100))
                if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and p3.sum() > 0:
                    p3 = p3 / float(p3.sum())
                    consider(p3)
            except Exception:
                pass

    # Phase 4: Final coordinate descent polish
    coord2_stop = hard - 15.0
    if time.monotonic() < coord2_stop - 30.0:
        try:
            p4, _ = _coord_descent(best_arr, coord2_stop, consider=consider, seed_val=int(seed + 99999))
            if p4 is not None and p4.size == n and np.all(np.isfinite(p4)) and p4.sum() > 0:
                p4 = p4 / float(p4.sum())
                consider(p4)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
