Actually, the previous code's approach is cleaner: change a[i], then rescale the whole vector. The score after rescaling by factor α = S_total/(S_total+δ) is:

where c' is the convolution with the unscaled modified vector. But since score is scale-invariant, we can equivalently compute:

The score of a'/(S+δ) (normalized) is the same: 2n * max((a'*a')/(S+δ)²) / 1 = same.

The previous code does this correctly with the eval_xs function. Let me make it faster and more precise.

Key optimization: Instead of evaluating the max over all 2n-1 points for each trial δ, we can:
1. Precompute which points are "near the max" (within, say, 1% of current max)
2. For trial values, only check those points plus the directly affected ones
3. Use the structure: c'[k] = c[k] + δ*d[k] + δ²*1_{k=2i} where d[k] = a[k-i]+a[i-k]

The max of {c[k] + δ*d[k] + δ²*1_{k=2i} for all k} / (S+δ)²

For fixed δ, this is just the max over k of (c[k] + δ*d[k] + δ²*1_{k=2i}), divided by (S+δ)².

- The numerator is a convex piecewise-linear (plus quadratic at one point) function of δ
- The denominator (S+δ)² is convex and increasing for δ > -S
- The ratio is not convex in general

                    if abs(w) > 1e-15:
                        a_local[i] = best_x
                        c_local[k_lo:k_hi + 1] += w * d
                        if diag_off >= 0:
                            c_local[2 * i] += w * w
                        score = best_val
                        improved = True
                        if score < best_local_score:
                            best_local_score = score
                            best_local_a = a_local.copy()

            # End of sweep
            sweep += 1
            if improved:
                flat_count = 0
            else:
                flat_count += 1
                if flat_count >= 4:
                    if time.monotonic() < max_time - 15.0:
                        sigma = 0.015 + 0.015 * min(flat_count, 12) / 12.0
                        a_local = a_local * np.exp(rng_local.normal(0.0, sigma, n_local))
                        a_local = np.maximum(a_local, 0.0)
                        ss = float(a_local.sum())
                        if ss > 1e-300:
                            a_local /= ss
                        else:
                            break
                        c_local = np.convolve(a_local, a_local)
                        score = 2.0 * n_local * float(c_local.max())
                        if score < best_local_score:
                            best_local_score = score
                            best_local_a = a_local.copy()
                        flat_count = 0
                        c_local = np.maximum(c_local, 0.0)
                    else:
                        break

        return best_local_a, best_local_score

    # === LSE Optimization ===
    def lse_optimize(p_in, max_time, rng_local, T_frac_init=0.05):
        n_ls = p_in.size
        L_ls = 2 * n_ls - 1
        conv_size = next_fast_len(L_ls)

        def cfast(pp):
            F = rfft(pp, conv_size)
            cc = irfft(F * F, conv_size)[:L_ls]
            np.maximum(cc, 0.0, out=cc)
            return cc

        def lse_grad(uu, T):
            pp = np.exp(uu - uu.max())
            pp /= pp.sum()
            cc = cfast(pp)
            cmax = float(cc.max())
            if cmax < 1e-300:
                return cmax, np.zeros_like(uu)
            scaled = (cc - cmax) / T
            exps = np.exp(scaled)
            s_exp = float(exps.sum())
            if not math.isfinite(s_exp) or s_exp <= 0:
                f = cmax
                w = np.zeros(cc.size)
                w[int(np.argmax(cc))] = 1.0
            else:
                f = cmax + T * math.log(s_exp / cc.size)
                w = exps / s_exp
            f *= 2.0 * n_ls
            # Gradient: d/d(u_j) of 2n * logsumexp((c - cmax)/T)
            # dc/d(pp_j) is the cross-conv with w
            # dpp_j/du_j = pp_j*(delta_jk - pp_k)
            # Use FFT for cross-conv
            F_w = rfft(w, conv_size)
            grad_pp = np.zeros(n_ls)
            # (convolve of pp with w) at specific indices
            Fp = rfft(pp, conv_size)
            cross = irfft(Fp * F_w, conv_size)[:L_ls]
            # The gradient w.r.t. pp is: for each j, d(sum_k c[k]*w[k])/dpp[j]
            # c[k] = sum_i pp[i]*pp[k-i], dc[k]/dpp[j] = pp[k-j] + pp[j-k] (with bounds)
            # So grad_pp[j] = sum_k w[k]*(pp[k-j]+pp[j-k])
            # This is 2*(convolution of pp with w) at index j, but only the "full" part
            # Actually: sum_k w[k]*pp[k-j] + sum_k w[k]*pp[j-k]
            # First term: (w * pp)[j] in convolution sense, but w has length L_ls
            # Let's use FFT: sum_k w[k]*pp[k-j] for valid k
            # This equals the cross-correlation
            # Simpler: grad_pp = 2 * (circular_convolve(pp, w_reversed))[:n]
            w_rev = w[::-1]
            F_wr = rfft(w_rev, conv_size)
            grad_cross = irfft(Fp * F_wr, conv_size)[:n_ls]
            grad_pp = 2.0 * grad_cross
            # Chain rule through softmax
            tdp = float(np.dot(pp, grad_pp))
            grad_u = pp * (grad_pp - tdp) * (2.0 * n_ls) / s_exp * (cc.size / T)
            # Simpler correct gradient:
            # f = 2n * (cmax + T*ln(sum exp((c-cmax)/T)))
            # df/du_j = 2n * T * (1/sum_exp) * sum_k exp((c[k]-cmax)/T)/T * dc[k]/du_j
            # = 2n * (1/sum_exp) * sum_k exps[k] * dc[k]/du_j
            # dc[k]/du_j = dc[k]/dpp_j * dpp_j/du_j
            # = (pp[k-j]+pp[j-k]) * pp_j*(delta_jj - pp)... 
            # This is getting complex. Use numerical or the simpler form:
            # grad_u_j = pp[j] * (sum_k w[k] * (pp[k-j]+pp[j-k]) - sum_k w[k]*(pp[k-j]+pp[j-k])*pp_k)*2n
            # Let me just use: grad_pp (the gradient of f w.r.t. pp), then project
            grad_f_pp = (2.0 * n_ls / s_exp) * grad_cross
            tdp2 = float(np.dot(pp, grad_f_pp))
            grad_u = pp * (grad_f_pp - tdp2)
            return float(2.0 * n_ls * f / (2.0 * n_ls)), grad_u  # return f (without 2n factor for clarity)

        from scipy.optimize import minimize

        u = np.log(np.maximum(p_in, 1e-300))
        u = u - u.mean()
        cc0 = cfast(p_in)
        T = T_frac_init * float(cc0.max())
        best_u = u.copy()
        best_f = 2.0 * n_ls * float(cc0.max())

        stages = [0.10, 0.04, 0.015, 0.005, 0.002, 0.0008, 0.0003]
        start_t = time.monotonic()
        avail = max_time - start_t - 2.0
        if avail < 10:
            return p_in

        total_w = sum(1.0 for _ in stages)
        cum = 0.0
        for frac in stages:
            cum += 1.0
            stage_end = min(max_time - 1.0, start_t + avail * (cum / total_w))
            if stage_end <= time.monotonic() + 3.0:
                break
            T = max(1e-10, frac * float(cc0.max()))

            def obj(uu):
                pp = np.exp(uu - uu.max())
                pp /= pp.sum()
                cc = cfast(pp)
                cmax = float(cc.max())
                if cmax < 1e-300:
                    return cmax
                scaled = (cc - cmax) / T
                exps = np.exp(scaled)
                s_e = float(exps.sum())
                return cmax + T * math.log(s_e / cc.size)

            def obj_grad(uu):
                pp = np.exp(uu - uu.max())
                pp /= pp.sum()
                cc = cfast(pp)
                cmax = float(cc.max())
                if cmax < 1e-300:
                    return cmax, np.zeros_like(uu)
                scaled = (cc - cmax) / T
                exps = np.exp(scaled)
                s_e = float(exps.sum())
                f = cmax + T * math.log(s_e / cc.size)
                w_vec = exps / s_e
                w_rev = w_vec[::-1]
                Fp = rfft(pp, conv_size)
                Fwr = rfft(w_rev, conv_size)
                gcross = irfft(Fp * Fwr, conv_size)[:n_ls]
                g_pp = 2.0 * gcross * (1.0 / s_e) * s_e / T * T  # simplify: 2*gcross/s_e * (s_e/T)*T
                g_pp = 2.0 * gcross
                tgp = float(np.dot(pp, g_pp))
                gu = pp * (g_pp - tgp) / s_e * (cc.size / T)
                # Correct: df/dpp_j = (1/s_e)*sum_k exps[k]/T * dc[k]/dpp_j = (1/T)*(w_vec * dc/dpp)
                # dc[k]/dpp_j = pp[k-j]+pp[j-k] (the gcross before the 2x)
                # So df/dpp_j = (1/T) * gcross_j
                # Then df/du_j = pp_j*(df/dpp_j - sum_k pp_k*df/dpp_k)
                g_pp2 = gcross / T
                tgp2 = float(np.dot(pp, g_pp2))
                gu = pp * (g_pp2 - tgp2)
                return f, gu

            for _ in range(5):
                if time.monotonic() > stage_end - 1.0:
                    break
                try:
                    res = minimize(obj_grad, best_u, method='L-BFGS-B', jac=True,
                                   options={'maxiter': 40, 'gtol': 1e-10, 'ftol': 1e-14,
                                            'maxls': 30, 'maxcor': 20, 'disp': False})
                except Exception:
                    break
                if res.x is not None and np.all(np.isfinite(res.x)):
                    f_new = float(res.fun)
                    if f_new < best_f - 1e-15:
                        best_f = f_new
                        best_u = res.x.copy()
                        best_score_lse = 2.0 * n_ls * best_f
                        pp_new = np.exp(best_u - best_u.max())
                        pp_new /= pp_new.sum()
                        sc_check = 2.0 * n_ls * float(cfast(pp_new).max())
                        if sc_check < best_local_a is not None and False:
                            pass
                        cc0 = cfast(pp_new)

            cc0 = cfast(np.exp(best_u - best_u.max()) / np.exp(best_u - best_u.max()).sum())

        pp_out = np.exp(best_u - best_u.max())
        ss = pp_out.sum()
        if ss > 0:
            pp_out /= ss
        return np.maximum(pp_out, 0.0)

    # === Main optimization loop ===
    # Phase 1: Coordinate descent (primary)
    phase1_end = min(deadline - 60.0, t0 + budget * 0.55)
    if phase1_end > t0 + 10:
        a, bs = coord_descent(a, c, phase1_end, rng)
        consider(a)
        s = float(a.sum())
        if s > 0:
            a /= s
            c = np.convolve(a, a)

    # Phase 2: LSE for global structure
    if time.monotonic() < deadline - 120.0:
        lse_end = min(deadline - 60.0, time.monotonic() + 80.0)
        if lse_end > time.monotonic() + 15.0:
            try:
                p2 = lse_optimize(a, lse_end, np.random.default_rng(seed + 1))
                if p2 is not None and p2.size == n and float(p2.sum()) > 0:
                    p2 = p2 / float(p2.sum())
                    consider(p2)
            except Exception:
                pass

    # Phase 3: More coordinate descent on best
    if time.monotonic() < deadline - 50.0:
        a_cur = best_a.copy()
        c_cur = np.convolve(a_cur, a_cur)
        phase3_end = min(deadline - 20.0, time.monotonic() + (deadline - time.monotonic()) * 0.5)
        if phase3_end > time.monotonic() + 10.0:
            a, bs = coord_descent(a_cur, c_cur, phase3_end, np.random.default_rng(seed + 200))
            consider(a)

    # Phase 4: Perturb and re-optimize
    if time.monotonic() < deadline - 40.0:
        rng_p = np.random.default_rng(seed + 500)
        for pr in range(2):
            if time.monotonic() > deadline - 25.0:
                break
            pert = best_a * np.exp(rng_p.normal(0, 0.03, n))
            pert = np.maximum(pert, 0)
            ps = float(pert.sum())
            if ps > 0:
                pert /= ps
            sub_end = min(deadline - 15.0, time.monotonic() + 20.0)
            if sub_end > time.monotonic() + 8.0:
                c_p = np.convolve(pert, pert)
                a_p, _ = coord_descent(pert, c_p, sub_end, np.random.default_rng(seed + 800 + pr * 100))
                consider(a_p)

    # Phase 5: Final coordinate polish
    if time.monotonic() < deadline - 10.0:
        a_f = best_a.copy()
        c_f = np.convolve(a_f, a_f)
        final_end = min(deadline - 4.0, time.monotonic() + max(5.0, deadline - time.monotonic() - 4.0))
        if final_end > time.monotonic() + 3.0:
            a_f, _ = coord_descent(a_f, c_f, final_end, np.random.default_rng(seed + 9999))
            consider(a_f)

    # Final validation
    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    if out.size == 0 or float(out.sum()) < 0.01:
        out = parent.copy()
        out = np.clip(out, 0.0, 1000.0)
    if out.size == 0 or float(out.sum()) < 0.01:
        out = np.full(64, 1.0 / 64.0)

    ls = [float(x) for x in out]
    try:
        sc = float(evaluate_sequence(ls))
    except Exception:
        sc = math.inf
    if not math.isfinite(sc):
        ls = [1.0]
    return ls
