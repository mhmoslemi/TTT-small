from scipy.optimize import linprog


def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)

def _build_conv_mat(n, a):
    """Build (2n-1) x n convolution matrix A where A[k,j] = a[k-j] if valid."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        js = np.arange(jlo, jhi)
        A[k, js] = a[k - js]
    return A


def _lp_best_response(a, M, time_limit=2.5):
    n = len(a)
    A = _build_conv_mat(n, a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def _line_search(a, g, M):

        cand = np.maximum(cand, 0)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s

        if v < best_v:

        c2 = np.maximum((1 - m2) * a + m2 * g, 0)

def _gradient_step(a, c_full, M):
    """Log-sum-exp gradient descent step."""
    n = len(a)
    for temp in [0.005, 0.01, 0.02]:
        cs = c_full / temp
        cs = cs - cs.max()
        p = np.exp(cs)
        ps = p.sum()
        if ps < 1e-30:
            continue
        p = p / ps
        grad = np.zeros(n)
        for k in range(len(c_full)):
            pk = p[k]
            if pk < 1e-10:
                continue
            jlo = max(0, int(k) - n + 1)
            jhi = min(n, int(k) + 1)
            js = np.arange(jlo, jhi)
            idx = int(k) - js
            grad[js] += 2.0 * pk * a[idx]
        grad = grad - grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            continue

            cand = np.maximum(a - alpha * grad / gn, 0)

                continue
            cand = cand / cs
            v = float(np.convolve(cand, cand).max())
            if v < M * (1 - 5e-11):
