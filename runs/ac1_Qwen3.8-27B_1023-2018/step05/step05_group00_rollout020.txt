- Where c_affected is c[i:i+n] and c_outside is the max of c outside [i, i+n-1]

For the grid search per coordinate:

- Total per coordinate: O(n)
- Per sweep: O(n^2)
- For n=3072: ~9.4M per sweep

        s_out = float(p_out.sum())

        # sum_k w[k]*p[k-i] for k-i in [0,n-1]: this is a cross-correlation

        # d(sum_k w[k]*c[k])/dp[i] = 2 * sum_k w[k] * (p[k-i] + p[i-k])

        # This is getting too complex. Let me use a finite difference for the key coordinates

        # grad[i] = 2 * sum_k w[k] * (p[k-i] + p[i-k]) where terms are 0 if index out of range
        # = 2 * [sum_{k=i}^{i+n-1} w[k]*p[k-i] + sum_{k=0}^{i} w[k]*p[i-k]]  (clamped to [0,2n-2])

        # and cross_b = sum_m w[i-m]*p[m] = correlation of w[:i+1] reversed with p
        # Both can be computed via FFT but it's complex. Let me just do it in O(n) per element.
        # For n=3072, the full gradient is O(n^2) which is ~9M. Acceptable for a few iterations.
        
        grad = np.zeros(n, dtype=np.float64)
        # Vectorized: for each i, grad[i] = 2*(sum_{j=0}^{n-1} w[i+j]*p[j] + sum_{j=0}^{i} w[i-j]*p[j])
        # First term: for i=0,...,n-1: sum_j w[i+j]*p[j] where i+j < 2n-1
        # This is a "sliding dot product" = (w * p_flipped)[2n-2+i] ... 
        # Let's just use np.correlate or manual
        w_ext = np.zeros(2 * n, dtype=np.float64)
        w_ext[:L] = w
        p_ext = np.zeros(2 * n, dtype=np.float64)
        p_ext[:n] = p
        # cross1[i] = sum_j p[j] * w[i+j] for j=0..n-1, i+j < 2n-1
        # = sum_j p[j] * w_ext[i+j]
        # This is like a cross-correlation
        for j in range(n):
            pass  # too slow per-element, let me use FFT
        
        # FFT approach:
        # cross1[i] = sum_j p[j]*w[i+j] = sum_j p[j]*w[i+j]
        # Let v[j] = p[j], and we want for each i: sum_j v[j]*w[i+j]
        # = (v * w_rev)[i + (2n-2)] ... not exactly
        # Actually: sum_j p[j]*w[i+j]. Let k=i+j: sum_k p[k-i]*w[k] = (p_shifted * w)[i]
        # where p_shifted[i-k] = p[k-i]... 
        # This is: cross1 = correlate(w, p, mode='full')[n-1:n-1+n]... 
        # np.correlate(w, p, 'full')[k] = sum_j w[j]*p[j+k-n+1]... 
        # Let me just use the direct O(n^2) for the gradient since we only do a few L-BFGS iterations
        
        # Simple O(n^2) gradient
        for ii in range(n):
            # sum_j p[j]*w[ii+j] for j=0..n-1, ii+j < 2n-1
            jmax = min(n, 2 * n - 1 - ii)
            if jmax > 0:
                term1 = float(np.dot(p[:jmax], w[ii:ii + jmax]))
            else:
                term1 = 0.0
            # sum_j p[j]*w[ii-j] for j=0..min(ii,2n-2-ii+... ) -> j=0..ii, ii-j in [0,2n-2]
            jmax2 = min(ii + 1, 2 * n - 1)
            if jmax2 > 0:
                term2 = float(np.dot(p[:jmax2], w[ii - (jmax2 - 1):ii + 1][::-1]))
            else:
                term2 = 0.0
            grad[ii] = 2.0 * (term1 + term2)
        
        # Map grad_p to grad_u: grad_u = pp * (grad_p - pp.dot(grad_p))
        gp = float(np.dot(pp, grad))
        grad_u = pp * (grad - gp)
        grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
        return 2.0 * n * f, grad_u

    # Run L-BFGS for a few iterations
    best_u = u.copy()
    best_f = 2.0 * n * _convolve_fft(softmax(u)).max()
    
    for T_stage in [0.01, 0.003, 0.001]:
        if time.monotonic() > stop_time - 5.0:
            break
        try:
            res = minimize(obj, best_u, method='L-BFGS-B', jac=True,
                           options={'maxiter': 20, 'gtol': 1e-10, 'maxls': 15, 'disp': False})
            if res.x is not None and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_f:
                    best_f = fval
                    best_u = res.x.copy()
        except Exception:
            break

    p_out = softmax(best_u)
    if consider is not None:
        try:
            consider(p_out)
        except Exception:
            pass
    return p_out


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64) / 64.0
        best_score = float(_score_exact(best_arr))

    # Also check uniform
    n_uni = best_arr.size
    uni = np.ones(n_uni, dtype=np.float64) / n_uni
    uni_sc = float(_score_exact(uni))
    if math.isfinite(uni_sc) and uni_sc < best_score:
        best_arr = uni.copy()
        best_score = uni_sc

    # Also check parent as-is
    par_sc = float(_score_exact(best_arr)) if best_arr.size != n_uni else best_score

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        so = float(a.sum())
        if so < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_arr = a.copy()
            best_score = sc

    n = int(best_arr.size)
    work = best_arr.copy()
    ws = float(work.sum())
    if ws > 0:
        work /= ws

    # Phase 1: Main coordinate descent (bulk of time)
    p1_end = min(hard - 60.0, t0 + 0.65 * budget)
    if p1_end > time.monotonic() + 30.0:
        try:
            work, _ = _coord_descent(work, p1_end, seed=int(seed + 1), consider=consider,
                                     max_sweeps=300000)
        except Exception:
            pass

    work = best_arr.copy()
    ws = float(work.sum())
    if ws > 0:
        work /= ws

    # Phase 2: LSE/L-BFGS refinement
    p2_end = min(hard - 40.0, time.monotonic() + 0.15 * budget)
    if p2_end > time.monotonic() + 20.0:
        try:
            p5 = _lse_optimize(work, p2_end, consider=consider)
            if p5 is not None and p5.size == n and np.all(np.isfinite(p5)):
                consider(p5)
        except Exception:
            pass

    work = best_arr.copy()
    ws = float(work.sum())
    if ws > 0:
        work /= ws

    # Phase 3: Second coordinate descent round
    p3_end = min(hard - 30.0, t0 + 0.85 * budget)
    if p3_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord_descent(work, p3_end, seed=int(seed + 777), consider=consider,
                                     max_sweeps=150000)
        except Exception:
            pass

    work = best_arr.copy()
    ws = float(work.sum())
    if ws > 0:
        work /= ws

    # Phase 4: Final short coordinate descent
    p4_end = min(hard - 15.0, t0 + budget - 15.0)
    if p4_end > time.monotonic() + 10.0:
        try:
            work, _ = _coord_descent(work, p4_end, seed=int(seed + 99999), consider=consider,
                                     max_sweeps=50000)
        except Exception:
            pass

    # Final: try blending with uniform (sometimes helps at the end)
    for frac in [0.1, 0.3, 0.5]:
        blended = (1.0 - frac) * best_arr + frac * (1.0 / n)
        blended = np.maximum(blended, 0.0)
        bs = float(blended.sum())
        if bs > 0:
            blended /= bs
            consider(blended)
