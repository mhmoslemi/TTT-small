import numpy as np

import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    if c.size == 0:
        return math.inf
    m = float(c.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    if b.size == 0:
        return math.inf
    m = float(b.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * a.size * m / (s * s))


def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        e = np.full(u.size, 1.0 / u.size, dtype=np.float64)
        s = 1.0
    return e / s


def _p_to_u(p):
    u = np.log(np.maximum(p, 1e-300), dtype=np.float64)
    return u - float(np.mean(u))


def _lse_optimize(p, stop_time, consider=None, fine_only=False):

    n = int(p.size)

        return p
    s = float(p.sum())

        return p
    p = p / s
    u = _p_to_u(p)

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1].copy(), conv_size)
        G = rfft(ww, conv_size)
        c = irfft(F * G, conv_size)[:conv_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())

        scaled = (c - cmax) / T

        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp

        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))

    total_w = float(sum(w for _, w in stages))

    for frac, wt in stages:

                "disp": False,

                break

                            except Exception:

    flat = 0

    G_FINE = 48

        improved = False
        order = rng.permutation(n)

            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.5 or S_total < 0.66:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0

            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue

            base = c[start_i:end_i].copy()
            infl = 2.0 * p[:Ls].copy()
            diag = min(i, Ls - 1)  # position of c[2i] within base

            # max_const: max of c outside [start_i, end_i)
            max_const = 0.0
            if start_i > 0:
                ml = float(c[:start_i].max())
                if ml > max_const:
                    max_const = ml
            if end_i < Ltot:
                mr = float(c[end_i:].max())
                if mr > max_const:
                    max_const = mr

            avg = 1.0 / float(n)
            x_hi = max(30.0 * v, 100.0 * avg, 0.03, 1e-12)
            x_hi = min(x_hi, 20.0)
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, avg)

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = 1.0 + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-14
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                dval = base[diag] + w * infl[diag] + w * w
                aff[:, diag] = dval
                maff = np.max(aff, axis=1)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 <= v <= x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Fine search around best
            if G > 4:

                lo = max(0.0, best_x - 3.0 * dw)
                hi = min(x_hi, best_x + 3.0 * dw)

                    fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):

                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])

                        best_val = vv2

            # Second refinement (golden-section-like)

                    vv3 = float(vals3[imin3])
                    if vv3 < best_val - 1e-16:

            if (math.isfinite(best_val) and best_val < cur_score - 1e-14

                except Exception:
                    pass

                    scale = rng.choice([0.01, 0.02, 0.04, 0.06])
                    noise = rng.normal(0.0, scale, n, dtype=np.float64)
                    p_new = p * np.exp(noise)
                    p_new = np.maximum(p_new, 0.0)
                    ss = float(p_new.sum())
                    if ss > 0:
                        p_new /= ss
                        c_new = _conv_self(p_new)
                        sc_new = float(2.0 * n * c_new.max())
                        if math.isfinite(sc_new):
                            if sc_new < best_score - 1e-13:
                                best_score = sc_new

                                p = p_new.copy()

                                    except Exception:

                            else:
                                p = p_new.copy()
                                c = c_new
                                cur_score = sc_new

                else:

        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:

    try:

    except Exception:

        return ls
    except Exception:
        return [1.0]
