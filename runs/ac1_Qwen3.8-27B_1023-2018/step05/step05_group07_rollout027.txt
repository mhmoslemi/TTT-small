def _optimize_lp_fast(a0, deadline, seed=42):
    """Faster variant: skip matrix rebuild when possible, use smaller time limit."""
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0
    max_stuck = 50
    pert_level = 0

    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        M = float(c.max())
        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > max_stuck:
            sigma = min(0.02 + 0.01 * pert_level, 0.2)
            xx = np.arange(n, dtype=np.float64)
            mode = pert_level % 4
            if mode == 0:
                active = np.where(c >= M * 0.90)[0]
                p = np.zeros(n)
                for k in active:
                    lo = max(0, int(k) - n + 1)
                    hi = min(n, int(k) + 1)
                    for j in range(lo, hi):
                        idx = int(k) - j
                        if 0 <= idx < n:
                            p[j] += a[idx]
                pmax = p.max() + 1e-15
                a *= np.exp(sigma * (1.0 - 2.0 * p / pmax))
            elif mode == 1:
                m = rng.integers(1, max(2, n // 10))
                phi = rng.uniform(0, 2 * np.pi)
                a *= (1.0 + sigma * np.sin(2 * np.pi * m * xx / n + phi))
            elif mode == 2:
                a *= np.exp(rng.normal(0, sigma, n))
            else:
                center = n / 2.0
                w = 1.0 + sigma * np.exp(-0.5 * ((xx - center) / (n / 4.0)) ** 2)
                a /= np.maximum(w, 0.5)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck = 0
            pert_level += 1
            if pert_level > 60:
                break
            continue

        A = _build_A(n, a)
        moved = False
        g = _solve_lp(A, M * (1 + 1e-10), n, tl=1.5)
        if g is not None and g.sum() > 1e-12:
            g = g / g.sum()
            if float(np.max(np.abs(g - a))) > 1e-10:
                cand, val = _ls(a, g)
                if val < M * (1 - 1e-10):
                    a = cand
                    moved = True
        if not moved:
            a *= np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

    return best_a

def _make_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    pn = parent / max(parent.sum(), 1e-15)
    x = np.linspace(0, 1, n)
    inits = []

    # Interpolated parent
    if n != len(parent):
        p_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        p_interp = np.clip(p_interp, 0, 1000)
        inits.append(p_interp / p_interp.sum())
    else:
        inits.append(pn.copy())

    # Gaussian bumps
    for sigma in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
        v = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(v / v.sum())

    # Bathtub (U-shape)
    for width in [0.3, 0.4, 0.5, 0.6]:
        v = 1.0 - width * np.exp(-0.5 * ((x - 0.5) / 0.15) ** 2)
        v = np.maximum(v, 0.01)
        inits.append(v / v.sum())

    # Cosine windows
    for m in [1, 2, 3, 4, 5]:
        v = np.maximum(np.cos(m * np.pi * (x - 0.5)) ** 2 + 0.01, 0)
        inits.append(v / v.sum())

    # Two-peak
    for pos in [0.15, 0.2, 0.25, 0.3, 0.35]:
        v = np.exp(-0.5 * ((x - pos) / 0.08) ** 2) + np.exp(-0.5 * ((x - (1 - pos)) / 0.08) ** 2)
        inits.append(v / v.sum())

    # Sine/square
    for m in [1, 2, 3]:
        v = np.sin(m * np.pi * x) ** 2 + 0.01
        inits.append(v / v.sum())

    # Random (seeded)
    for _ in range(6):
        v = np.abs(rng.normal(1.0, 0.6, n)) + 0.01
        inits.append(v / v.sum())

    # Blends with parent
    for frac in [0.3, 0.5, 0.7]:
        for init in inits[:4]:
            v = frac * pn + (1 - frac) * init
            inits.append(v / v.sum())

    return inits

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 25.0

    parent = _get_parent()
    n_parent = parent.size

    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(100)
        best_score = _score(best_seq)

    def try_update(cand):
        nonlocal best_seq, best_score
        try:
            cand = np.clip(np.asarray(cand, dtype=np.float64), 0, 1000)
            if cand.size == 0 or not np.all(np.isfinite(cand)):
                return
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand.copy()
        except Exception:
            pass

    rng_main = np.random.default_rng(seed)

    # ===== Phase 1: Deep optimization on parent (40% budget) =====
    t1 = t0 + (budget - 25) * 0.40
    if t1 > t0 + 5:
        try:
            r = _optimize_lp(parent, t1, seed=42)
            try_update(r)
        except Exception:
            pass

    # ===== Phase 2: Diverse initializations (35% budget) =====
    t2 = t0 + (budget - 25) * 0.75
    inits = _make_inits(n_parent, parent, rng_main)

    # Sort by initial score (skip worst ones)
    scored_inits = []
    for init in inits:
        sc = _score(init)
        if np.isfinite(sc):
            scored_inits.append((sc, init))
    scored_inits.sort(key=lambda x: x[0])
    scored_inits = [s[1] for s in scored_inits[:25]]  # top 25

    elapsed = time.monotonic() - t0
    remaining_p2 = max(t2 - time.monotonic(), 2.0)
    per_init = max(2.0, remaining_p2 / max(1, len(scored_inits)))

    for i, init in enumerate(scored_inits):
        if time.monotonic() > t2 - 5.0:
            break
        alloc = min(time.monotonic() + per_init, t2 - 3.0)
        if alloc < time.monotonic() + 1.0:
            break
        try:
            r = _optimize_lp_fast(init, alloc, seed=1000 + i)
            try_update(r)
        except Exception:
            pass

    # ===== Phase 3: Polish best found (20% budget) =====
    t3 = deadline
    ab = best_seq.copy()
    try:
        r = _optimize_lp(ab, t3, seed=7777)
        try_update(r)
    except Exception:
        pass

    # ===== Phase 4: Try slightly different lengths if time remains =====
    if time.monotonic() < deadline - 60:
        for dn in [1, -1, 2, -2, 5, -5, 10, -10]:
            if time.monotonic() > deadline - 30:
                break
            nt = n_parent + dn
            if nt < 20 or nt > 600:
                continue
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, parent)
            init_new = np.clip(init_new, 0, 1000)
            sn = init_new.sum()
            if sn < 1e-10:
                continue
            init_new = init_new / sn
            alloc = min(time.monotonic() + 20.0, deadline - 15.0)
            try:
                r = _optimize_lp_fast(init_new, alloc, seed=5000 + nt)
                try_update(r)
            except Exception:
                pass

    # ===== Final validation =====
    try:
        fs = float(evaluate_sequence([float(v) for v in best_seq]))
        if not np.isfinite(fs):
            raise ValueError("invalid final")
    except Exception:
        fallback = [1.0] * 100
        return fallback

    return [float(v) for v in best_seq]
