import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 20.0
    
    def get_parent():
        try:
            h = globals()["height_sequence_1"]
            p = np.asarray(h, dtype=np.float64).ravel()
        except Exception:
            p = np.ones(3072, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0.0, 1000.0)
        if p.size == 0 or p.sum() < 0.01:
            p = np.ones(3072, dtype=np.float64)
        return p
    
    parent = get_parent()
    n = len(parent)
    L = 2 * n - 1
    cs = next_fast_len(L)
    
    def fft_conv(x):
        F = rfft(x, cs)
        return irfft(F * F, cs)[:L]
    
    def score_of(a):
        s = float(a.sum())
        if s < 1e-100:
            return math.inf
        c = fft_conv(a)
        return 2.0 * n * float(c.max()) / (s * s)
    
    # Generate initializations
    inits = []

    p = parent.copy()
    p = p / p.sum()

        a_exp /= a_exp.sum()
        inits.append(a_exp)

    a_step /= a_step.sum()
    inits.append(a_step)

    r = 0.95
    a_geo = r ** np.arange(n, dtype=np.float64)
    a_geo /= a_geo.sum()
    a_mix = 0.5 * parent/parent.sum() + 0.5 * a_geo
    a_mix /= a_mix.sum()

            if time.monotonic() > deadline - 5.0:

            # Build candidate values

            cands = {v, 0.0, mean_a, 2.0*mean_a, 4.0*mean_a}
            if v > 0:

            cands.update([mean_a*0.1, mean_a*10.0])
            cands = [max(0.0, c) for c in cands if math.isfinite(c)]
            cands = sorted(set(cands))

                if abs(delta) < 1e-15:
