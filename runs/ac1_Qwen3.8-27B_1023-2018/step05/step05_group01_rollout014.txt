def _get_parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)

    p = np.where(np.isfinite(p), p, 0.0)

def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0)
    if n <= 512:
        c = np.convolve(a, a, mode="full")
    else:
        L = 2 * n - 1
        size = next_fast_len(L)
        F = rfft(a, size)
        c = irfft(F * F, size)[:L]
    return np.maximum(c, 0.0)


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _score_exact_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)

def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        return src.copy()
    m = src.size
    xp = (np.arange(m) + 0.5) / m
    fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
    out = np.maximum(fp, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s

    n = int(p.size)

        return p
    p = p / s

    def p_to_u(pp):

    cross_size = next_fast_len(3 * n - 2)

    best_p = None
    best_sc = _score_fft(init_arr)

    segment_counts = [8, 16, 32, 64]

            break
        if n % m != 0:
            # Adjust: use floor(n/m) for full segments and distribute remainder
            base_len = n // m
            rem = n % m
            lens = [base_len + (1 if i < rem else 0) for i in range(m)]
        else:
            base_len = n // m
            lens = [base_len] * m

        for i in range(m):

            e_i = s_i + lens[i]
            h0[i] = float(init_arr[s_i:e_i].mean()) if lens[i] > 0 else 1.0/m
        h0 = np.maximum(h0, 1e-8)
        h0 = h0 / h0.sum()

        # Try a few random perturbations
        attempts = 3
        for att in range(attempts):
            if time.monotonic() > stop_time - 5:
                break
            h_init = h0.copy()
            if att > 0:
                h_init = h_init * np.exp(rng.normal(0, 0.1 + 0.05*att, m))
                h_init = np.maximum(h_init, 1e-8)
                h_init /= h_init.sum()

            def extended(h):
                h = np.maximum(h, 0.0)
                hs = float(h.sum())
                if hs < 1e-300:
                    return np.full(n, 1.0/n)
                h = h / hs
                arr = np.empty(n)
                for i in range(m):
                    s_i = sum(lens[:i])
                    e_i = s_i + lens[i]
                    arr[s_i:e_i] = h[i]
                return arr

            def obj(hv):
                hv = np.maximum(hv, 1e-12)
                hs = float(hv.sum())

                    return 1e100

                arr = extended(hv)
                c = _conv_self(arr)
                cm = float(c.max())
                return 2.0 * n * cm  # sum=1, so score = 2n*max_b

            def obj_and_grad(hv):

                hs = float(hv.sum())
                if hs < 1e-300:
                    return 1e100, np.zeros_like(hv)

                c = _conv_self(arr)

            if seg_stop <= time.monotonic() + 3:
                continue
            try:
                res = minimize(obj_and_grad, h_init.copy(), method="L-BFGS-B",
                               jac=True, options={"maxiter": 80, "gtol": 1e-9,
                                                   "ftol": 1e-14, "maxls": 30,
                                                   "maxcor": 10, "disp": False})
            except Exception:
                continue
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_best = np.maximum(res.x, 0.0)
                hs = float(h_best.sum())
                if hs > 1e-300:
                    h_best /= hs
                    arr = extended(h_best)
                    sc = _score_fft(arr)
                    if math.isfinite(sc) and (best_p is None or sc < best_sc):
                        best_sc = sc
                        best_p = arr.copy()
                        if consider is not None:
                            try:
                                consider(arr)
                            except Exception:
                                pass

    return best_p


def _coord_polish(p, stop_time, G=None, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    if G is None:
        if n <= 500:
            G = 56
        elif n <= 1000:
            G = 48
        elif n <= 3000:
            G = 40
        elif n <= 6000:
            G = 32
        else:
            G = 24
    G = int(max(16, G))

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < 500000 and time.monotonic() < stop_time - 0.5:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.2:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(25.0 * v, 50.0 * S / float(n), 0.03 * S, 1e-14)
            x_hi = min(x_hi, 10.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, S / float(n))

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag_i = i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[valid, None] * infl[None, :]
                dval = base[diag_i] + w[valid] * infl[diag_i] + w[valid] * w[valid]
                aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                dv = denom[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if G > 4 and best_x > 0 and best_x < x_hi:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 32, dtype=np.float64)
                if 0.0 <= best_x <= x_hi:
                    fine = np.concatenate((fine, [best_x]))
                    fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val - 1e-15:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 10.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        # Re-verify convolution periodically
        if sweep % 50 == 0:
            c_new = _conv_self(p)
            c = c_new
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        if consider is not None and sweep % 200 == 0:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 10.0:
                    sigma = 0.015 + 0.015 * min(flat, 12) / 12.0
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    else:
                        break
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_p = p.copy()
                    flat = 0
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        sc = float(_score_fft(p))
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 2.0:
        hard = t0 + max(2.0, budget * 0.9)
    final_reserve = max(10.0, 0.08 * (hard - t0))
    phase_limit = hard - final_reserve

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    n_parent = int(best_arr.size)

    # Phase 1: LSE optimization at parent length (first ~15%)
    p1_end = min(phase_limit, t0 + 0.15 * (phase_limit - t0))
    if p1_end > time.monotonic() + 20.0:
        try:
            p1 = _lse_optimize(best_arr, p1_end, consider=consider, seed=int(seed + 100))
            if p1 is not None and p1.size == n_parent and float(p1.sum()) > 0:
                best_arr = p1 / float(p1.sum())
                consider(best_arr)
        except Exception:
            pass

    # Phase 2: Heavy coordinate descent (next ~45%)
    p2_end = min(phase_limit - (phase_limit - t0) * 0.40, t0 + 0.60 * (phase_limit - t0))
    if p2_end > time.monotonic() + 30.0:
        try:
            p2, _ = _coord_polish(best_arr, p2_end, seed=int(seed + 200), consider=consider)
            if p2 is not None and p2.size == n_parent and float(p2.sum()) > 0:
                best_arr = p2 / float(p2.sum())
                consider(best_arr)
        except Exception:
            pass

    # Phase 3: Step-function structural search (next ~20%)
    p3_end = min(phase_limit - (phase_limit - t0) * 0.20, t0 + 0.80 * (phase_limit - t0))
    if p3_end > time.monotonic() + 30.0:
        try:
            p3 = _step_func_search(n_parent, p3_end, consider, int(seed + 3000), best_arr)
            if p3 is not None and p3.size == n_parent and float(p3.sum()) > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)
        except Exception:
            pass

    # Phase 4: Second round of coordinate descent on best (next ~10%)
    p4_end = min(phase_limit - 20.0, t0 + 0.92 * (phase_limit - t0))
    if p4_end > time.monotonic() + 20.0:
        # Try a perturbation for diversity
        rng4 = np.random.default_rng(int(seed + 4000))
        pert = best_arr * np.exp(rng4.normal(0, 0.03, best_arr.size))
        pert = np.maximum(pert, 0.0)
        ps = float(pert.sum())
        if ps > 0:
            pert /= ps
        # Coordinate polish the perturbed version briefly, then the best
        sub_end = p4_end - (p4_end - time.monotonic()) * 0.4
        if sub_end > time.monotonic() + 10.0:
            try:
                pp, _ = _coord_polish(pert, sub_end, seed=int(seed + 5000), consider=consider)
                if pp is not None and pp.size == n_parent and float(pp.sum()) > 0:
                    consider(pp / float(pp.sum()))
            except Exception:
                pass
        try:
            p4, _ = _coord_polish(best_arr, p4_end, seed=int(seed + 6000), consider=consider)
            if p4 is not None and p4.size == n_parent and float(p4.sum()) > 0:
                best_arr = p4 / float(p4.sum())
                consider(best_arr)
        except Exception:
            pass

    # Phase 5: Try a slightly different length
    if time.monotonic() < phase_limit - 40.0:
        n_alt = n_parent * 2 if n_parent <= 4096 else max(n_parent // 2, 1024)
        if n_alt != n_parent and n_alt <= 8192:
            sub_end = min(phase_limit - 10.0, time.monotonic() + 0.25 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 20.0:
                try:
                    p_alt = _resample_mid(best_arr, n_alt)
                    consider(p_alt)
                    pa, _ = _coord_polish(p_alt, sub_end, seed=int(seed + 7000), consider=consider)
                    if pa is not None and pa.size > 0 and float(pa.sum()) > 0:
                        consider(pa / float(pa.sum()))
                except Exception:
                    pass

    # Phase 6: Final squeeze LSE + coordinate
    final_end = min(hard - 15.0, t0 + budget - 10.0)
    if final_end > time.monotonic() + 20.0 and best_arr.size > 0:
        nbest = int(best_arr.size)
        rem = final_end - time.monotonic()
        if rem > 40.0 and nbest <= 4000:
            lstop = min(final_end - 25.0, time.monotonic() + 0.35 * rem)
            if lstop > time.monotonic() + 10.0:
                try:
                    p2 = _lse_optimize(best_arr, lstop, consider=consider, seed=int(seed + 99999))
                    if p2 is not None and p2.size == nbest and float(p2.sum()) > 0:
                        p2 = p2 / float(p2.sum())
                        consider(p2)
                except Exception:
                    pass

        cstop = min(final_end - 3.0, time.monotonic() + max(15.0, 0.55 * (final_end - time.monotonic())))
        if cstop > time.monotonic() + 8.0:
            try:
                p3, _ = _coord_polish(best_arr, cstop, seed=int(seed + 123456789), consider=consider)
                if p3 is not None and p3.size == nbest and float(p3.sum()) > 0:
                    p3 = p3 / float(p3.sum())
                    consider(p3)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
