def _lse_gradient(pp, c, T):
    """Compute gradient of LSE objective w.r.t. p (normalized, sum=1).
    Returns the gradient in the simplex (sum = 0) and the softmax weights w."""
    n = pp.size
    conv_len = c.size
    # LSE weights
    scaled = (c - float(c.max())) / T
    exps = np.exp(scaled, dtype=np.float64)
    sum_exp = float(exps.sum())
    if not math.isfinite(sum_exp) or sum_exp <= 0:
        w = np.zeros(conv_len)
        w[int(np.argmax(c))] = 1.0
    else:
        w = exps / sum_exp

    # C1[j] = sum_l a[l] * w[l+j] : cross-correlation
    # C2[j] = sum_l a[l] * w[j-l] : convolution
    size = next_fast_len(3 * n - 2)
    # C2: conv(pp, w)[:n]
    F_pp = rfft(pp, size)
    F_w = rfft(w, size)
    C2 = irfft(F_pp * F_w, size)[:n]

    # C1: conv(pp_rev, w)[n-1:2n-1]
    pp_rev = pp[::-1].copy()
    F_ppr = rfft(pp_rev, size)
    C1 = irfft(F_ppr * F_w, size)[n - 1:2 * n - 1]

    # Raw gradient (not projected to simplex)
    raw_grad = C1 + C2

    # Project to simplex: subtract pp * dot(pp, raw_grad)
    tdp = float(np.dot(pp, raw_grad))
    grad = pp * (raw_grad - tdp)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad, w

def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _sca_lp_step(a, stop_time, max_iters=5):
    """One round of SCA: linearize the max-convolution constraint and solve LP."""
    try:
        import cvxpy as cp
    except ImportError:
        return a

    n = a.size
    a = np.asarray(a, dtype=np.float64).copy()
    s = float(a.sum())
    if s < 1e-300:
        return a
    a = a / s

    c = _conv_self(a)
    M_cur = float(c.max())
    if not math.isfinite(M_cur) or M_cur <= 0:
        return a

    # Active set: indices where b[k] is within 2% of max
    thresh = 0.98 * M_cur
    active = np.where(c >= thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])

    # Build gradient matrix G[k, j] = a[k-j] + a[j-k] for k in active, j in 0..n-1
    # With a[m]=0 for m not in [0,n-1]
    G = np.zeros((len(active), n), dtype=np.float64)
    for idx, k in enumerate(active):
        for j in range(n):
            val = 0.0
            if 0 <= k - j < n:
                val += a[k - j]
            if 0 <= j - k < n:
                val += a[j - k]
            G[idx, j] = val

    # Linearized constraints: b_k(a_t) + G[idx,:] @ (a - a_t) <= M
    # => G[idx,:] @ a <= M - b_k(a_t) + G[idx,:] @ a_t
    rhs = np.zeros(len(active), dtype=np.float64)
    for idx, k in enumerate(active):
        constant = M_cur - c[k] + float(G[idx] @ a)
        rhs[idx] = constant

    # LP: minimize M
    # s.t. G @ a <= M + rhs_adjusted  => G @ a - M <= rhs_adjusted
    #      sum(a) == 1
    #      a >= 0
    #      M >= 0

    # Variables: a (n) + M (1) => n+1 vars
    x = cp.Variable(n + 1)
    a_var = x[:n]
    M_var = x[n]

    objective = cp.Minimize(M_var)

    constraints = []
    # G @ a - M <= rhs  => [G, -1] @ x <= rhs
    A = np.hstack([G, -np.ones((len(active), 1))])
    b_vec = rhs.copy()
    # Add: for inactive k, use b_k(a_t) + G_k @ (a-a_t) <= M too (upper bound)
    # Just use active set for now
    constraints.append(A @ x <= b_vec)
    constraints.append(cp.sum(a_var) == 1)
    constraints.append(a_var >= 0)
    constraints.append(M_var >= 0)

    prob = cp.Problem(objective, constraints)

    # Try solving
    t_start = time.monotonic()
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
        if prob.status not in ("optimal", "optimal_inaccurate"):
            # Try PIPS
            prob.solve(solver=cp.PIPS, verbose=False)
    except Exception:
        try:
            prob.solve(verbose=False)
        except Exception:
            return a

    if time.monotonic() - t_start > (stop_time - time.monotonic() - 5.0) or prob.status != "optimal":
        return a

    if x.value is None:
        return a

    a_new = np.maximum(x.value[:n], 0.0)
    s_new = float(a_new.sum())
    if s_new < 0.01 or not math.isfinite(s_new):
        return a
    a_new /= s_new

    # Verify
    sc_new = _score_arr(a_new)
    sc_old = _score_arr(a)
    if math.isfinite(sc_new) and sc_new < sc_old:
        return a_new
    return a

    lse2_end = min(now + min(200.0, 0.25 * budget), hard - 80.0)
    if lse2_end > now + 15.0:
        p_start = _resample(best_arr, n_work)
        try:
            p3 = _lse_optimize(p_start, lse2_end, consider=consider)
            if p3 is not None and p3.size == n_work and np.all(np.isfinite(p3)) and p3.sum() > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)
                if _score_arr(p3) < _score_arr(p_work):
                    p_work = p3
        except Exception:
            pass

    # Phase 4: Coordinate descent on the best
    now = time.monotonic()
    cd_end = min(now + min(400.0, 0.3 * budget), hard - 20.0)
    if cd_end > now + 30.0:
        p_start = _resample(best_arr, n_work)
        try:
            p_cd, _ = _coord_descent(p_start, cd_end, consider=consider, seed=int(seed + 7777))
            if p_cd is not None and p_cd.size > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                consider(p_cd)
                if _score_arr(p_cd) < _score_arr(p_work):
                    p_work = p_cd
        except Exception:
            pass

    # Phase 5: Second LSE polish
    now = time.monotonic()
    lse3_end = min(now + 100.0, hard - 30.0)
    if lse3_end > now + 10.0:
        p_start = _resample(best_arr, n_work)
        try:
            p4 = _lse_optimize(p_start, lse3_end, consider=consider)
            if p4 is not None and p4.size == n_work and np.all(np.isfinite(p4)) and p4.sum() > 0:
                p4 = p4 / float(p4.sum())
                consider(p4)
        except Exception:
            pass

    # Phase 6: Final CD squeeze
    now = time.monotonic()
    cd2_end = min(now + 80.0, hard - 10.0)
    if cd2_end > now + 15.0:
        p_start = _resample(best_arr, n_work)
        try:
            p_cd2, _ = _coord_descent(p_start, cd2_end, consider=consider, seed=int(seed + 9999))
            if p_cd2 is not None and p_cd2.size > 0 and np.all(np.isfinite(p_cd2)) and p_cd2.sum() > 0:
                consider(p_cd2)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = a_par.copy()
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in a_par]
            if float(evaluate_sequence(ls)) > final_sc:
                ls = [1.0]
        return ls
    except Exception:
        return [1.0]
