        # Using the FFT cross-conv trick: (pp * w)_full has length 3n-2, and we want
        # the middle n elements.
        
        q = xfast(pp, w)  # This gives (pp * w)_k for k = 0..n-1
        # But we need the full cross-correlation... 
        # Actually w has length 2n-1, pp has length n.
        # (pp * w) has length 3n-2.
        # The gradient component i is: sum_k w_k * (dpk/dpp_i) where pk = (pp*pp)_k
        # dpk/dpp_i = pp_{i+k} (if valid) + pp_{i-k} (if valid) ... 
        # = (w * pp)_i where (w*pp)_i = sum_k w_k * pp_{i+k} + sum_k w_k * pp_{i-k}
        # = (w * pp)_i + (w * pp)_i (by symmetry of w)
        # Wait, w is NOT symmetric in general (it's the soft-max weights on the convolution).
        # But c = pp*pp IS symmetric: c_k = c_{2n-2-k}.
        # So w is symmetric: w_k = w_{2n-2-k}.
        # Therefore the gradient is 2 * sum_k w_k * pp_{i+k} (with boundary)
        # = 2 * (w * pp)_i in the cross-conv sense.
        
        # The xfast function computes (pp[::-1] * w) and extracts [n-1:2n-1]
        # pp[::-1] * w has length 3n-2
        # Element at position j: sum_i pp_rev[i] * w[j-i] = sum_i pp[n-1-i] * w[j-i]
        # At position n-1+i: sum_i pp[n-1-i] * w[n-1+i-i] = sum_i pp[n-1-i] * w[n-1+i]
        # Let j = n-1-i: = sum_j pp[j] * w[i+j]
        # = sum_j pp[j] * w[i+j] = (shifted cross-correlation)
        
        # Hmm, I think the gradient is:
        # grad_i = 2 * sum_{k=0}^{2n-2} w[k] * (contribution of pp_i to c_k)
        # c_k = sum_{i: max(0,k-n+1) <= i <= min(n-1, k)} pp[i] * pp[k-i]
        # dc_k/dpp_i = pp[k-i] + pp[i-k] (if i != k-i, i.e., 2i != k)
        # dc_k/dpp_i = 2*pp[k-i] if 2i != k, or pp[k-i] = pp[i]^... 
        # Actually c_k = sum_i pp[i]*pp[k-i], so dc_k/dpp_i = pp[k-i] + pp[i-k] (the two terms that contain pp_i)
        # But when i = k-i, i.e., 2i=k, it's just 2*pp[i] (the derivative of pp_i^2).
        # So dc_k/dpp_i = pp[k-i] + pp[i-k] with the convention that out-of-range terms are 0.
        # = pp[k-i] + pp[(k-i) mod ...] no, just 0 if out of range.
        
        # grad_i = sum_k w_k * (pp[k-i] + pp[i-k])  (out of range = 0)
        # = sum_k w_k * pp[k-i] + sum_k w_k * pp[i-k]
        # = sum_m w_{m+i} * pp[m] + sum_m w_{m-i} * pp[m]  (where m = k-i, m = i-k)
        # = (w_shifted * pp)_i + (w_shifted * pp)_i  by symmetry
        # = 2 * (w * pp)_i  where (w*pp)_i = sum_m w[m+i] * pp[m] (valid m)
        
        # This is the cross-correlation of w and pp, which is what xfast computes!
        # xfast(pp, w) = (pp[::-1] * w)[n-1:2n-1]
        # Element i (0-indexed): (pp[::-1] * w)[n-1+i] = sum_j pp_rev[j] * w[n-1+i-j]
        # = sum_j pp[n-1-j] * w[n-1+i-j]
        # Let m = n-1-j: = sum_m pp[m] * w[n-1+i-(n-1-m)] = sum_m pp[m] * w[i+m]
        # = sum_m pp[m] * w[m+i] ✓
        
        # So q = xfast(pp, w) gives us sum_m pp[m]*w[m+i] for i=0..n-1
        # And the gradient is 2*q (since w is symmetric by c = pp*pp symmetry)
        
        # But we also need to account for the normalization: pp = exp(u)/sum(exp(u))
        # d(pp)/du_i = pp_i * (delta_{ij} - pp_j)
        # So the gradient in u-space is: pp_i * (2*q_i - sum(pp * 2*q))
        
        gd = float(np.dot(pp, 2.0 * q))
        grad = pp * (2.0 * q - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad
    
    u = p_to_u(p)
    best_u = u.copy()
    best_f = lse_func(u, 0.01)[0]
    
    # Temperature schedule
    stages = [
        (0.008, 0.15),
        (0.003, 0.15),
        (0.001, 0.18),
        (0.0003, 0.20),
        (0.0001, 0.22),
        (0.00003, 0.25),
        (0.00001, 0.25),
    ]
    
    start = time.monotonic()
    avail = stop_time - start - 5.0
    if avail <= 5:
        return p
    
    total_w = sum(w for _, w in stages)
    cum = 0.0
    
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 6:
            break
        cum += wt
        end = min(stop_time - 5, start + avail * (cum / total_w))
        if end <= now + 3:
            continue
        
        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        T = max(1e-14, frac * cmax)
        
        # Run L-BFGS
        maxiter = 18 if n > 2000 else 25
        stale = 0
        while time.monotonic() < end - 1.0 and stale < 3:
            try:
                res = minimize(
                    lse_func, best_u, args=(T,),
                    method='L-BFGS-B', jac=True,
                    options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-15,
                             'maxls': 20, 'maxcor': 20, 'disp': False}
                )
            except Exception:
                break
            
            if res.x is not None and res.x.size == n:
                x = np.asarray(res.x, dtype=np.float64)
                if np.all(np.isfinite(x)):
                    fv = float(res.fun)
                    if math.isfinite(fv) and fv < best_f - 1e-14:
                        best_f = fv
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
            else:
                stale += 1
            if getattr(res, 'status', 1) == 0:
                break
        
        p_new = u_to_p(best_u)
        p_new = np.maximum(p_new, 0.0)
        s = p_new.sum()
        if s > 1e-300:
            p_new /= s
        
        sc = _score(p_new)
        cur_sc = _score(p)
        if math.isfinite(sc) and sc < cur_sc:
            p = p_new
            u = p_to_u(p)
            best_u = u.copy()
            best_f = lse_func(u, T)[0]
            if consider:
                consider(p)
        else:
            u = p_to_u(p)
            best_u = u.copy()
    
    return p


def _coord_polish(p, stop_time, consider=None, seed=12345, G=80):
    """Coordinate descent with adaptive evaluation."""
    n = p.size
    c = _fft_conv(p)
    c = np.maximum(c, 0.0)
    S = 1.0
    infl = 2.0 * p  # 2*p since c = p*p, dc_i/dp_i contains 2*p contribution
    cur_sc = float(2.0 * n * c.max())
    best_sc = cur_sc
    best_p = p.copy()
    
    if consider:
        consider(best_p)
    
    rng = np.random.default_rng(seed)
    flat = 0
    
    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        
        for pos in range(n):
            if time.monotonic() > stop_time - 0.5:
                break
            if S <= 1e-300:
                break
            # Renormalize periodically
            if S > 1.20 or S < 0.80:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                infl *= inv
                S = 1.0
                np.maximum(c, 0.0, out=c)
                
            i = int(order[pos])
            v = float(p[i])
            L = 2*n - 1
            start = i
            end = min(i + n, L)
            Ls = end - start
            if Ls <= 0:
                continue
            
            base = c[start:end].copy()
            # max of unaffected parts
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                max_const = max(max_const, mc)
            if end < L:
                mc = float(c[end:].max())
                max_const = max(max_const, mc)
            
            Sloc = float(S)
            
            # Evaluate at candidate x values
            # f(x) = 2*n*max(max_const, max(base + (x-v)*infl_seg + (x-v)^2 * diag_contribution)) / (Sloc + x - v)^2
            # where infl_seg = infl[start:end], and the (x-v)^2 term only affects position 2i-start
            
            di = 2*i - start  # index within base where the quadratic term appears
            xhi = max(10.0 * max(v, Sloc/n), Sloc * 0.01, 0.001)
            xhi = min(xhi, max(0.1, 5.0 * v) if v > 1e-10 else Sloc * 0.05)
            xhi = max(xhi, 1e-6)
            
            # Generate test points
            pts = np.array([0.0, v, xhi])
            m1 = max(16, G // 2)
            lo1 = max(1e-12, 0.01 * v if v > 0 else Sloc / (10000.0 * n))
            if lo1 < xhi:
                pts = np.concatenate([pts, np.geomspace(lo1, xhi, m1)])
            if v > 1e-12:
                m2 = max(10, G // 4)
                fcts = np.geomspace(0.3, 4.0, m2)
                pts = np.concatenate([pts, v * fcts])
            m3 = max(12, G // 3)
            linhi = min(xhi, max(5.0 * Sloc / n, v * 0.5))
            if linhi > 0:
                pts = np.concatenate([pts, np.linspace(0, linhi, m3)])
            pts = np.unique(np.clip(pts, 0, xhi))
            
            # Evaluate
            w = pts - v
            den = Sloc + w
            valid = den > 1e-14
            aff = base[None, :] + w[:, None] * infl[start:end][None, :]
            if 0 <= di < Ls:
                aff[:, di] += w * w
            np.maximum(aff, max_const, out=aff)
            mv = np.max(aff, axis=1)
            vals = np.full(pts.size, math.inf)
            vals[valid] = (2.0 * n * mv[valid]) / (den[valid] * den[valid])
            vals = np.where(np.isfinite(vals), vals, math.inf)
            
            if not np.any(np.isfinite(vals)):
                continue
            im = int(np.argmin(vals))
            best_x = float(pts[im])
            best_val = float(vals[im])
            
            # Refine around best
            if best_x > 0:
                step = max(v, Sloc/n, 1e-9)
                lo = max(0.0, best_x - 2*step)
                hi = min(xhi, best_x + 2*step)
            else:
                lo = 0.0
                hi = max(3*Sloc/n, 1e-6)
            if hi > lo:
                fine = np.linspace(lo, hi, 40)
                if lo > 0:
                    fine = np.unique(np.concatenate([fine, np.geomspace(lo, hi, 16)]))
                w2 = fine - v
                den2 = Sloc + w2
                valid2 = den2 > 1e-14
                aff2 = base[None, :] + w2[:, None] * infl[start:end][None, :]
                if 0 <= di < Ls:
                    aff2[:, di] += w2 * w2
                np.maximum(aff2, max_const, out=aff2)
                mv2 = np.max(aff2, axis=1)
                vals2 = np.full(fine.size, math.inf)
                vals2[valid2] = (2.0 * n * mv2[valid2]) / (den2[valid2] * den2[valid2])
                vals2 = np.where(np.isfinite(vals2), vals2, math.inf)
                if np.any(np.isfinite(vals2)):
                    im2 = int(np.argmin(vals2))
                    if vals2[im2] < best_val:
                        best_x = float(fine[im2])
                        best_val = float(vals2[im2])
            
            if math.isfinite(best_val) and best_val < cur_sc - 1e-12 and 0 <= best_x <= xhi:
                d = best_x - v
                if abs(d) > 1e-16:
                    c[start:end] += d * infl[start:end]
                    diag = 2*i
                    if 0 <= diag < L:
                        c[diag] += d * d
                    p[i] = best_x
                    infl[i] = 2.0 * best_x
                    S += d
                    cur_sc = best_val
                    improved = True
        
        # Renormalize
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            infl *= inv
            S = 1.0
        np.maximum(c, 0.0, out=c)
        desc = float(2.0 * n * c.max())
        
        if math.isfinite(desc) and desc < best_sc - 1e-14:
            best_sc = desc
            best_p = p.copy()
            if consider:
                consider(best_p)
        
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                # Perturbation to escape local minimum
                if time.monotonic() < stop_time - 10:
                    sigmas = [0.02, 0.04, 0.08, 0.12, 0.18]
                    sigma = sigmas[min(flat - 3, len(sigmas) - 1)]
                    noise = rng.normal(0, sigma, n)
                    q = p * np.exp(noise)
                    qs = q.sum()
                    if qs > 0:
                        p = q / qs
                        c = _fft_conv(p)
                        np.maximum(c, 0.0, out=c)
                        S = 1.0
                        infl = 2.0 * p
                        cur_sc = float(2.0 * n * c.max())
                        flat = 0
                else:
                    break
    
    if consider:
        consider(best_p)
    return best_p, best_sc


def _lp_direction(a, M, n_max=2048):
    """LP to find a good direction."""
    if linprog is None or sp is None:
        return None
    n = a.size
    if n < 64 or n > n_max:
        return None
    
    try:
        idx = np.arange(n, dtype=np.int32)
        # Build sparse constraint matrix
        # For each lag k (0..2n-2), constraint: sum_i a[i] * b[i+k] <= M (where valid)
        # Actually: (a*b)_k = sum_i a[i] * b[k-i] for |k-i| < n
        # Let me just build it directly for moderate n
        
        if n > 2048:
            return None
        
        # For lag k, (a*b)_k = sum_{i: max(0,k-n+1)<=i<=min(n-1,k)} a[i] * b[i] * ... 
        # Wait: (a*b)_k = sum_i a[i] * b[k-i] where i ranges so both indices are valid
        # = sum_i a[i] * b[k-i] for i in [max(0, k-(n-1)), min(n-1, k)]
        
        # Sparse construction
        rows = []
        cols = []
        datas = []
        
        for k in range(2*n - 1):
            i_lo = max(0, k - (n-1))
            i_hi = min(n-1, k)
            for i in range(i_lo, i_hi + 1):
                rows.append(k)
                cols.append(i)
                datas.append(a[i])  # Wait, this isn't right...
        
        # Actually (a*b)_k = sum_i a[i]*b[k-i]
        # But b is the variable, a is fixed.
        # So the coefficient of b[j] in constraint k is a[k-j] (when valid).
        # For b[j], it appears in constraint k when k-j is in [0, n-1], i.e., k in [j, j+n-1]
        # with coefficient a[k-j].
        
        # Let me redo: for variable b[j], it appears in constraints k = j, j+1, ..., j+n-1
        # with coefficient a[k-j] = a[0], a[1], ..., a[n-1]
        
        # So for each j, the non-zeros in column j are at rows j, j+1, ..., j+n-1
        # with values a[0], a[1], ..., a[n-1]
        
        # Total non-zeros: n * n = n^2 (for dense) but with the constraint that row < 2n-1
        # For j, rows j to j+n-1, but max row is 2n-2, so it's min(n-1, 2n-2-j) entries
        # Total: for j=0: n entries, j=1: n entries, ..., j=n-1: n entries (since 2n-2 >= n-1+n-1=2n-2)
        # Actually for j=n-1: rows n-1 to 2n-2, that's n entries. So total is n^2.
        
        # For n=2048, that's 4M non-zeros. Might be tight but let's try.
        if n > 1500:
            return None
            
        rows = np.repeat(np.arange(2*n-1), 0)  # will fill
        # Actually let me build it more efficiently
        r_parts = []
        c_parts = []
        d_parts = []
        for j in range(n):
            r_lo = j
            r_hi = min(j + n - 1, 2*n - 2)
            r_parts.append(np.arange(r_lo, r_hi + 1))
            c_parts.append(np.full(r_hi - r_lo + 1, j))
            d_parts.append(a[:r_hi - r_lo + 1])
        
        rows = np.concatenate(r_parts)
        cols = np.concatenate(c_parts)
        datas = np.concatenate(d_parts)
        
        A = sp.coo_matrix((datas, (rows, cols)), shape=(2*n-1, n)).tocsr()
        
        res = linprog(
            -np.ones(n),
            A_ub=A,
            b_ub=np.full(2*n-1, M),
            bounds=[(0, None)] * n,
            method='highs',
            options={'time_limit': 60, 'presolve': True}
        )
        
        if res is not None and getattr(res, 'success', False):
            b = np.asarray(res.x, dtype=np.float64)
            b = np.where(np.isfinite(b), b, 0.0)
            b = np.maximum(b, 0)
            if b.sum() > 0:
                return b / b.sum()
    except Exception:
        pass
    return None


def _lp_subproblem(a, n_small, stop_time):
    """Solve LP on a reduced problem."""
    n = a.size
    a_small = _resample(a, n_small)
    a_small = _normalize(a_small)
    best = a_small.copy()
    best_sc = _score(a_small)
    
    for _ in range(5):
        if time.monotonic() > stop_time:
            break
        c = _fft_conv(a_small)
        M = float(c.max())
        if M <= 0:
            break
        b = _lp_direction(a_small, M, n_max=max(n_small, 100))
        if b is None:
            break
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1]:
            cand = (1-t)*a_small + t*b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs > 0:
                cand /= cs
                sc = _score(cand)
                if sc < best_sc - 1e-10:
                    a_small = cand
                    best_sc = sc
                    best = cand.copy()
                    improved = True
                    break
        if not improved:
            break
    
    return best


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s) if budget_s else 1300.0
    budget = max(10.0, budget)
    reserve = 20.0 if budget > 200 else 10.0
    hard = t0 + budget - reserve
    
    parent = _get_parent()
    n = parent.size
    
    best = _normalize(parent.copy())
    try:
        best_sc = float(evaluate_sequence([float(x) for x in best]))
    except Exception:
        best_sc = math.inf
    
    if not math.isfinite(best_sc):
        best = np.full(n, 1.0/n) if n > 0 else np.full(64, 1.0/64)
        n = best.size
        best_sc = math.inf
    
    def consider(a):
        nonlocal best, best_sc
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
            if arr.size == 0:
                return
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.clip(arr, 0, 1000)
            s = arr.sum()
            if s < 0.01:
                return
            arr = arr / s
            sc = float(_score(arr))
            # Verify with exact evaluator periodically
            if math.isfinite(sc) and sc < best_sc - 1e-12:
                # Quick check
                ls = [float(x) for x in arr]
                try:
                    sc_exact = float(evaluate_sequence(ls))
                except Exception:
                    sc_exact = sc
                if math.isfinite(sc_exact) and sc_exact < best_sc - 1e-12:
                    best = arr.copy()
                    best_sc = sc_exact
        except Exception:
            pass
    
    consider(best)
    
    # Phase 1: Exploration (first 10% of time)
    exp_stop = t0 + 0.10 * (budget - reserve)
    rng = np.random.default_rng(seed)
    
    # Try power-law shapes
    for q in [0.7, 1.3, 0.5, 1.5, 2.0]:
        if time.monotonic() > exp_stop:
            break
        x = np.arange(n, dtype=np.float64) / n
        cand = (1 - x + 1e-6) ** q * (x + 1e-6) ** q  # symmetric-ish
        cand = _normalize(cand)
        consider(cand)
    
    # Try sinusoidal modulation
    for f in [1, 2, 3, 5]:
        if time.monotonic() > exp_stop:
            break
        x = np.arange(n, dtype=np.float64) / n
        for phase in [0, math.pi/2, math.pi]:
            cand = 1.0 + 0.3 * math.sin(2*math.pi*f*x + phase)
            cand = _normalize(cand)
            consider(cand)
    
    # LP sub-problem for shape
    if budget > 300 and time.monotonic() < exp_stop:
        for ns in [256, 384]:
            if time.monotonic() > exp_stop - 30:
                break
            try:
                small = _lp_subproblem(best, ns, min(exp_stop, time.monotonic() + 50))
                if small is not None:
                    cand = _resample(small, n)
                    consider(cand)
            except Exception:
                pass
    
    # Phase 2: LSE optimization
    lse_stop = min(hard - 60, t0 + 0.40 * (budget - reserve))
    if lse_stop > time.monotonic() + 20:
        try:
            best = _lse_optimize(best, lse_stop, consider=consider)
        except Exception:
            pass
    
    # Phase 3: Coordinate descent
    cd_stop = min(hard - 30, t0 + 0.75 * (budget - reserve))
    if cd_stop > time.monotonic() + 15:
        try:
            best, _ = _coord_polish(best, cd_stop, consider=consider, seed=seed+1, G=80)
        except Exception:
            pass
    
    # Phase 4: Second LSE pass (finer)
    if time.monotonic() < hard - 60:
        lse2_stop = min(hard - 30, time.monotonic() + min(80, 0.1*(budget-reserve)))
        if lse2_stop > time.monotonic() + 15:
            try:
                best = _lse_optimize(best, lse2_stop, consider=consider)
            except Exception:
                pass
    
    # Phase 5: Final coordinate polish
    if time.monotonic() < hard - 10:
        try:
            best, _ = _coord_polish(best, hard - 5, consider=consider, seed=seed+999, G=100)
        except Exception:
            pass
    
    # Final validation
    try:
        out = np
