                    mv2 = np.max(aff2, axis=1)
                    out2 = np.full(fine.shape, math.inf)
                    d2v = d2[v2]
                    out2[v2] = (2.0 * n * mv2[v2]) / (d2v * d2v)
                    if np.any(np.isfinite(out2)):
                        im2 = int(np.argmin(out2))
                        if float(out2[im2]) < bv:
                            bv = float(out2[im2])
                            bx = float(fine[im2])

            if math.isfinite(bv) and bv < cur_score - 1e-14 and 0.0 <= bx <= 12.0 * S:
                w = bx - v
                if abs(w) > 1e-15:
                    p[i] = bx
                    S += w
                    c[start:end] += w * infl
                    c[start + (i - start)] += w * w
                    cur_score = bv
                    improved = True

        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        if sweep % 5 == 0:
            c = _conv_self(p)
        new_sc = 2.0 * n * float(c.max())
        if math.isfinite(new_sc) and new_sc < best_score - 1e-14:
            best_score = new_sc
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(p)
                except Exception:
                    pass

        if improved:
            flat_count = 0
        else:
            flat_count += 1
            if flat_count >= 3:
                if time.monotonic() < stop_time - 20.0:
                    sigma = 0.01 + 0.005 * flat_count
                    p = p * np.exp(rng.normal(0, sigma, n))
                    p = np.maximum(p, 0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    S = 1.0
                    c = _conv_self(p)
                    cur_score = 2.0 * n * float(c.max())
                    flat_count = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_self(p)
    sc = 2.0 * n * float(c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    n = int(parent.size)

    s_par = float(parent.sum())
    if s_par < 1e-12:
        parent = np.ones(n, dtype=np.float64)
        s_par = float(n)
    p_init = np.maximum(parent, 0.0) / s_par

    # Alternative: linearly decreasing
    lin = np.arange(n, 0, -1, dtype=np.float64)
    lin = lin / lin.sum()

    sc_par = _score_fft(p_init)
    sc_lin = _score_fft(lin)

    if sc_par <= sc_lin:
        p_work = p_init.copy()
        best_score = sc_par
    else:
        p_work = lin.copy()
        best_score = sc_lin
    best_arr = p_work.copy()

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            sc = float(evaluate_sequence([float(x) for x in a]))
            if math.isfinite(sc) and sc < best_score:
                best_score = sc
                best_arr = (a / s).copy()
        except Exception:
            pass

    # Phase 1: LSE optimization (initial shaping)
    lse1_stop = min(t0 + 150.0, deadline - 600.0)
    if lse1_stop > time.monotonic() + 30.0:
        try:
            p1 = _lse_optimize(p_work, lse1_stop, consider=consider)
            if p1 is not None and p1.size == n and np.all(np.isfinite(p1)):
                ss = float(p1.sum())
                if ss > 1e-300:
                    p1 = p1 / ss
                p_work = p1
                consider(p1)
        except Exception:
            pass

    # Phase 2: Coordinate descent (main)
    coord1_stop = min(deadline - 250.0, time.monotonic() + 500.0)
    if coord1_stop > time.monotonic() + 60.0:
        try:
            p2, _ = _coord_descent(p_work, coord1_stop, consider=consider, seed=int(seed + 7))
            if p2 is not None and p2.size == n and np.all(np.isfinite(p2)):
                ss = float(p2.sum())
                if ss > 1e-300:
                    p2 = p2 / ss
                p_work = p2
                consider(p2)
        except Exception:
            pass

    # Phase 3: LSE at low T (refinement)
    rem3 = deadline - 150.0 - time.monotonic()
    if rem3 > 40.0:
        lse2_stop = time.monotonic() + max(30.0, 0.45 * rem3)
        try:
            p3 = _lse_optimize(p_work, lse2_stop, consider=consider)
            if p3 is not None and p3.size == n and np.all(np.isfinite(p3)):
                ss = float(p3.sum())
                if ss > 1e-300:
                    p3 = p3 / ss
                p_work = p3
                consider(p3)
        except Exception:
            pass

    # Phase 4: Final coordinate descent (polishing)
    coord2_stop = deadline - 20.0
    if time.monotonic() < coord2_stop - 30.0:
        try:
            p4, _ = _coord_descent(p_work, coord2_stop, consider=consider, seed=int(seed + 999))
            if p4 is not None and p4.size == n and np.all(np.isfinite(p4)):
                ss = float(p4.sum())
                if ss > 1e-300:
                    p4 = p4 / ss
                p_work = p4
                consider(p4)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(max(n, 1), 1.0 / max(n, 1))
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in np.maximum(parent, 0.0)]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
