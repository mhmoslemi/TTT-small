from scipy.fft import rfft, irfft, next_fast_len

def _fft_conv(a):
    """FFT-based self-convolution."""
    n = len(a)
    L = 2*n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F*F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):

    n = len(a)
    c = _fft_conv(a)

                    break
                per_stage = max(5, (lbfgs_end - time.monotonic()) / (len(stages)))
                if per_stage < 5:
                    break
                try:
                    res = minimize(lse_obj, cur_u, args=(T,), method='L-BFGS-B',
                                   jac=True, options={'maxiter': max_it, 'gtol': 1e-10,
                                                      'disp': False, 'maxls': 20})
                    if res.x is not None and np.all(np.isfinite(res.x)):
                        cur_u = res.x
                except Exception:
                    pass
            # Convert back
            m = float(np.max(cur_u))
            pp = np.exp(cur_u - m)
            ss = float(pp.sum())
            if ss > 1e-300:
                pp = pp / ss
                consider(pp)
                if 2.0 * n * float(_fft_conv(pp, fft_size).max()) < cur_score:
                    p = pp.copy()
                    c = _fft_conv(p, fft_size)
                    cur_score = 2.0 * n * float(c.max())
        except Exception:
            pass

    # ---- Phase 2: Main coordinate descent ----
    G = 20  # grid points for coarse search
    flat = 0
    sweep = 0
    S = 1.0

    while time_ok(5.0):
        sweep += 1
        improved = False
        order = rng.permutation(n)

        for i in order:
            if not time_ok(3.0):
                break
            if S < 1e-15:
                break
            # Renormalize periodically
            if S > 1.3 or S < 0.77:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0

            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            # Search range
            x_hi = max(20.0 * v, 60.0 / n, 0.005)
            x_hi = min(x_hi, 8.0 / n)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(v * 2, 1.0 / n)

            # Influence
            infl = 2.0 * p[:Ls]
            diag_pos = 2 * i - start
            has_diag = (0 <= diag_pos < Ls)

            # Max outside window
            max_out = 0.0
            if start > 0:
                max_out = float(c[:start].max())
            if end < Ltot:
                mo2 = float(c[end:].max())
                if mo2 > max_out:
                    max_out = mo2

            base = c[start:end]

            # Vectorized coarse grid
            pts = np.linspace(0.0, x_hi, G + 1)
            if v > 0 and v < x_hi:
                # Ensure v is a grid point for continuity
                idx = int(np.round(v / x_hi * G))
                pts[idx] = v
                pts = np.sort(pts)

            ws = pts - v
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if has_diag:
                aff[:, diag_pos] += ws * ws
            maff = np.maximum(aff.max(axis=1), max_out)
            new_S = S + ws
            with np.errstate(divide='ignore', invalid='ignore'):
                vals = 2.0 * n * maff / (new_S * new_S)
            vals[~np.isfinite(vals)] = np.inf
            vals[new_S < 1e-15] = np.inf

            if not np.any(np.isfinite(vals)):
                continue

            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Golden section refinement (only if interior)
            if 1 <= imin <= G - 1:
                dw = x_hi / G
                lo_gs = max(0.0, best_x - 2.0 * dw)
                hi_gs = min(x_hi, best_x + 2.0 * dw)
                if hi_gs - lo_gs > 1e-14:
                    gr = 0.6180339887498949
                    ga = lo_gs + gr * (hi_gs - lo_gs)
                    gb = hi_gs - gr * (hi_gs - lo_gs)

                    def _ev(x):
                        w = x - v
                        nc = base + w * infl
                        if has_diag:
                            nc = nc.copy()
                            nc[diag_pos] += w * w
                        mv = float(nc.max())
                        if max_out > mv:
                            mv = max_out
                        ns = S + w
                        if ns < 1e-15:
                            return np.inf
                        return 2.0 * n * mv / (ns * ns)

                    fga = _ev(ga)
                    fgb = _ev(gb)
                    for _ in range(12):
                        if fga < fgb:
                            hi_gs = gb
                            gb = ga
                            fgb = fga
                            ga = lo_gs + gr * (hi_gs - lo_gs)
                            fga = _ev(ga)
                        else:
                            lo_gs = ga
                            ga = gb
                            fga = fgb
                            gb = lo_gs + gr * (hi_gs - lo_gs)
                            fgb = _ev(gb)
                    x_opt = (lo_gs + hi_gs) * 0.5
                    f_opt = _ev(x_opt)
                    if f_opt < best_val:
                        best_x, best_val = x_opt, f_opt

            # Apply update
            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and best_x >= 0:
                w = best_x - v
                if abs(w) > 1e-18:
                    p[i] = best_x
                    S += w
                    c[start:end] += w * infl
                    if has_diag:
                        c[start + diag_pos] += w * w
                    cur_score = best_val
                    improved = True

        # End of sweep
        if S > 1e-15:
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        # Periodic full recompute for accuracy
        if sweep % 5 == 0:
            c = _fft_conv(p, fft_size)
            cur_score = 2.0 * n * float(c.max())

        new_score = 2.0 * n * float(c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-13:
            best_score = new_score
            best_p = p.copy()
            consider(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4 and time_ok(80.0):
                # Adaptive perturbation
                sigma = 0.003 + 0.004 * min(flat, 12)
                noise = rng.normal(0.0, sigma, n)
                p *= np.exp(noise)
                p = np.maximum(p, 0.0)
                ss = float(p.sum())
                if ss > 1e-15:
                    p /= ss
                    S = 1.0
                    c = _fft_conv(p, fft_size)
                    cur_score = 2.0 * n * float(c.max())
                flat = 0

        if flat >= 25:
            break

    # ---- Phase 3: Quick polish on active coordinates (if time remains) ----
    polish_end = min(hard - 10.0, t0 + budget - 40.0)
    if time.monotonic() < polish_end - 60.0 and best_score < 1.51:
        try:
            p2 = best_p.copy()
            c2 = _fft_conv(p2, fft_size)
            cmax = float(c2.max())
            # Active lags: those contributing to the max
            thresh = cmax * 0.99
            active = np.where(c2 >= thresh)[0]
            # Important coordinates
            important = np.zeros(n, dtype=bool)
            for lag in active[:200]:
                lo = max(0, int(lag) - n + 1)
                hi = min(n - 1, int(lag))
                important[lo:hi + 1] = True
            imp_idx = np.where(important)[0]
            if imp_idx.size > 400:
                # Keep top 400 by centrality
                centers = imp_idx
                mid = (imp_idx[0] + imp_idx[-1]) // 2
                centers = centers[np.abs(centers - mid).argmin(400) if centers.size > 400 else slice(None)]
                imp_idx = centers[:400] if centers.size > 0 else imp_idx[:400]

            G_f = 40
            S2 = 1.0
            cur2 = 2.0 * n * float(c2.max())
            flat2 = 0
            rsp = np.random.default_rng(seed + 999)

            while time.monotonic() < polish_end - 10.0 and flat2 < 10:
                flat2 += 1
                imp2 = False
                for i in rsp.permutation(imp_idx):
                    if not time_ok(15.0):
                        break
                    v = float(p2[i])
                    st = i
                    en = min(i + n, Ltot)
                    Ls2 = en - st
                    if Ls2 <= 0:
                        continue
                    infl2 = 2.0 * p2[:Ls2]
                    dp = 2 * i - st
                    hd = (0 <= dp < Ls2)
                    mo = 0.0
                    if st > 0:
                        mo = float(c2[:st].max())
                    if en < Ltot:
                        m2 = float(c2[en:].max())
                        if m2 > mo:
                            mo = m2
                    base2 = c2[st:en]

                    # Narrow range for polish
                    xhi2 = max(10.0 * v, 20.0 / n, 0.002)
                    xhi2 = min(xhi2, v * 3.0 + 0.005)
                    xlo2 = max(0.0, v - 0.5 * xhi2)

                    pts2 = np.linspace(xlo2, max(xhi2, xlo2 + 1e-10), G_f + 1)
                    ws2 = pts2 - v
                    aff2 = base2[None, :] + ws2[:, None] * infl2[None, :]
                    if hd:
                        aff2[:, dp] += ws2 * ws2
                    ma2 = np.maximum(aff2.max(axis=1), mo)
                    ns2 = S2 + ws2
                    with np.errstate(divide='ignore', invalid='ignore'):
                        vals2 = 2.0 * n * ma2 / (ns2 * ns2)
                    vals2[~np.isfinite(vals2)] = np.inf
                    vals2[ns2 < 1e-15] = np.inf
                    if not np.any(np.isfinite(vals2)):
                        continue
                    im2 = int(np.argmin(vals2))
                    bx = float(pts2[im2])
                    bv = float(vals2[im2])

                    if math.isfinite(bv) and bv < cur2 - 1e-14 and bx >= 0:
                        w2 = bx - v
                        if abs(w2) > 1e-18:
                            p2[i] = bx
                            S2 += w2
                            c2[st:en] += w2 * infl2
                            if hd:
                                c2[st + dp] += w2 * w2
                            cur2 = bv
                            imp2 = True

                if S2 > 1e-15:
                    iv2 = 1.0 / S2
                    p2 *= iv2
                    c2 *= iv2 * iv2
                    S2 = 1.0
                if not imp2:
                    flat2 += 1
                else:
                    flat2 = 0
                c2 = _fft_conv(p2, fft_size)
                sc2 = 2.0 * n * float(c2.max())
                if sc2 < best_score - 1e-13:
                    best_p = p2.copy()
                    best_score = sc2
                    consider(best_p)
        except Exception:
            pass

    # ---- Final validation and return ----
    out = best_p.copy()
    out = np.clip(out, 0.0, 1000.0)
    out = np.where(np.isfinite(out), out, 0.0)
    if out.size == 0 or float(out.sum()) < 0.01:
        out = np.ones(64)

    ls = [float(x) for x in out]
    try:
        fs = float(evaluate_sequence(ls))
        if not math.isfinite(fs):
            ls = [1.0]
    except Exception:
        ls = [1.0]

    return ls
