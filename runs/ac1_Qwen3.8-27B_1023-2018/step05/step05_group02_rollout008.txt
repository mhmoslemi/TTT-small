def _lp_geometric_step(f, M, max_time=30):
    """Solve the LP to find g that maximizes sum(g) subject to f*g <= M, g >= 0"""
    n = len(f)
    # Build the convolution constraint matrix
    # (f * g)[k] = sum_j f[k-j] * g[j] for valid j
    # This is a matrix product C @ g <= M where C[k,j] = f[k-j] if valid
    
    # Actually, it's easier to just express it as:
    # For each k, sum over j: f[max(0,k-n+1)..min(k,n-1)] * g[j]
    
    # Let me use scipy to build the matrix
    from scipy.sparse import csr_matrix
    
    row_idx = []
    col_idx = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            row_idx.append(k)
            col_idx.append(j)
            vals.append(f[k - j])
    
    C = csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
    
    import cvxpy as cp
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [C @ g <= M, g >= 0]
    prob = cp.Problem(objective, constraints)

    if g.value is not None:
        best_g = np.maximum(np.array(g.value).flatten(), 0)
    else:
        best_g = None
    
    return best_g

def _conv_self(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))

def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)

    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)

import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize, linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:

def _conv_self(a):
    n = a.size
    if n == 0:
        return np.array([0.0])
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * a.size * m / (s * s))


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == 1:
        return np.full(n, float(src[0]) / n)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m) + 0.5) / m
        fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _lp_geometric_step(f, M, time_limit=45.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Returns g or None."""
    n = f.size
    if n < 2:
        return None
    # Build convolution matrix C: C[k,j] = f[k-j] if 0<=k-j<n
    # Column j: C[j:j+n, j] = f
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    # Solve: minimize -sum(g) s.t. C@g <= M, g >= 0
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    t_start = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)

        G = rfft(ww, cross_size)

        q = c[n - 1:2 * n - 1]

            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 6.0:
                    try:
                        noise_amt = 0.02 + 0.01 * math.exp(-flat * 0.1)
                        p = p * np.exp(rng.normal(0.0, noise_amt, n))
                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss > 0.0:
                            p /= ss
                        c = _conv_self(p)
                        S_total = 1.0
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider is not None:
                                consider(best_p)
                        flat = 0
                    except Exception:
                        break
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    usable = max(1.0, hard - t0)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64)
        best_arr /= 64.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    # Phase allocation
    # Phase 1: LP geometric steps (20% if time allows)
    # Phase 2: LSE + L-BFGS (45%)
    # Phase 3: Coordinate descent (30%)
    # Final polish (5%)

    phase1_end = t0 + 0.20 * usable
    phase2_end = t0 + 0.65 * usable
    phase3_end = t0 + 0.95 * usable

    # === Phase 1: LP geometric steps ===
    n_best = int(best_arr.size)
    if n_best >= 4 and n_best <= 5000 and time.monotonic() < phase1_end - 5.0:
        f_cur = best_arr.copy()
        f_cur = f_cur / float(f_cur.sum())
        for lp_iter in range(4):
            if time.monotonic() > phase1_end - 5.0:
                break
            c_f = _conv_self(f_cur)
            M_val = float(c_f.max())
            if M_val <= 0:
                break
            # Normalize for LP
            S_f = float(f_cur.sum())
            f_norm = f_cur / S_f * math.sqrt(2.0 * n_best)
            c_norm = _conv_self(f_norm)
            M_norm = float(c_norm.max())
            if M_norm <= 0 or not math.isfinite(M_norm):
                break
            remaining_lp = min(50.0, float(phase1_end - time.monotonic() - 6.0))
            if remaining_lp < 5.0:
                break
            g_sol = _lp_geometric_step(f_norm, M_norm, time_limit=remaining_lp)
            if g_sol is None or g_sol.size != n_best:
                break
            g_sol = g_sol / float(g_sol.sum()) * math.sqrt(2.0 * n_best)
            # Try different mixing ratios
            improved_here = False
            for t in (0.15, 0.25, 0.35, 0.50):
                f_mix = (1.0 - t) * f_norm + t * g_sol
                f_mix = np.maximum(f_mix, 0.0)
                ss = float(f_mix.sum())
                if ss < 1e-15:
                    continue
                f_mix /= ss
                sc_mix = _score(f_mix)
                if math.isfinite(sc_mix) and sc_mix < best_score - 1e-12:
                    best_arr = f_mix.copy()
                    best_score = sc_mix
                    f_cur = f_mix.copy()
                    improved_here = True
                    consider(f_mix)
            if not improved_here:
                # Even if no improvement, try the LP direction with small step
                for t in (0.05, 0.10):
                    f_mix = (1.0 - t) * f_norm + t * g_sol
                    f_mix = np.maximum(f_mix, 0.0)
                    ss = float(f_mix.sum())
                    if ss < 1e-15:
                        continue
                    f_mix /= ss
                    sc_mix = _score(f_mix)
                    if math.isfinite(sc_mix) and sc_mix < _score(f_cur) - 1e-10:
                        f_cur = f_mix.copy()
                        break

    # === Phase 2: LSE + L-BFGS ===
    now = time.monotonic()
    if now < phase2_end - 15.0 and best_arr.size > 0:
        p_opt = best_arr.copy()
        p_opt = p_opt / float(p_opt.sum())
        n_opt = int(p_opt.size)
        lbfgs_stop = min(float(phase2_end) - 10.0, now + 0.9 * (phase2_end - now))
        if lbfgs_stop > now + 15.0:
            try:
                p2 = _lse_lbfgs(p_opt, lbfgs_stop, consider=consider, do_fine=False)
                if p2 is not None and p2.size == n_opt and np.all(np.isfinite(p2)) and p2.sum() > 0.0:
                    p2 = p2 / float(p2.sum())
                    consider(p2)
                    best_arr = best_arr.copy()  # consider already updated best_arr
            except Exception:
                pass

        # Fine LSE
        now = time.monotonic()
        if now < phase2_end - 10.0 and best_arr.size > 0:
            p_fine = best_arr.copy()
            p_fine = p_fine / float(p_fine.sum())
            n_fine = int(p_fine.size)
            fine_stop = min(float(phase2_end - 5.0), now + max(20.0, 0.4 * (phase2_end - now)))
            if fine_stop > now + 10.0 and n_fine <= 4000:
                try:
                    p3 = _lse_lbfgs(p_fine, fine_stop, consider=consider, do_fine=True)
                    if p3 is not None and p3.size == n_fine and np.all(np.isfinite(p3)) and p3.sum() > 0.0:
                        p3 = p3 / float(p3.sum())
                        consider(p3)
                except Exception:
                    pass

    # === Phase 3: Coordinate descent ===
    now = time.monotonic()
    if now < phase3_end - 10.0 and best_arr.size > 0:
        n_cd = int(best_arr.size)
        p_cd = best_arr.copy()
        p_cd = p_cd / float(p_cd.sum())
        # Choose G based on n
        if n_cd <= 450:
            G_val = 50
        elif n_cd <= 900:
            G_val = 40
        elif n_cd <= 2000:
            G_val = 32
        else:
            G_val = 26
        cd_stop = min(float(phase3_end) - 5.0, now + max(20.0, 0.9 * (phase3_end - now)))
        if cd_stop > now + 10.0:
            try:
                p_cd_res, _ = _coord_polish(p_cd, cd_stop, max_sweeps=100000,
                                            G=G_val, seed=int(seed + 777), consider=consider)
                if p_cd_res is not None and p_cd_res.size == n_cd and p_cd_res.sum() > 0.0:
                    consider(p_cd_res)
            except Exception:
                pass

    # === Final polish ===
    now = time.monotonic()
    final_end = min(hard, t0 + budget - 5.0)
    if now < final_end - 8.0 and best_arr.size > 0:
        n_fp = int(best_arr.size)
        p_fp = best_arr.copy()
        p_fp = p_fp / float(p_fp.sum())

        # Quick LSE fine
        if final_end - now > 30.0 and n_fp <= 4000:
            fs = min(final_end - 20.0, now + 0.4 * (final_end - now))
            if fs > now + 8.0:
                try:
                    p_fs = _lse_lbfgs(p_fp, fs, consider=consider, do_fine=True)
                    if p_fs is not None and p_fs.size == n_fp and p_fs.sum() > 0.0:
                        consider(p_fs)
                except Exception:
                    pass

        # Final coordinate
        now = time.monotonic()
        if now < final_end - 5.0:
            p_fcd = best_arr.copy()
            p_fcd = p_fcd / float(p_fcd.sum())
            n_fcd = int(p_fcd.size)
            G_final = 30 if n_fcd > 1500 else 40
            fcd_stop = min(final_end - 3.0, now + max(8.0, 0.5 * (final_end - now)))
            if fcd_stop > now + 5.0:
                try:
                    p_res, _ = _coord_polish(p_fcd, fcd_stop, max_sweeps=80000,
                                             G=G_final, seed=int(seed + 999999), consider=consider)
                    if p_res is not None and p_res.size == n_fcd and p_res.sum() > 0.0:
                        consider(p_res)
                except Exception:
                    pass

    # === Final validation ===
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
