def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)


def _build_A(n, a):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        A[k, jlo:jhi] = a[k - jhi + 1:k - jlo + 1][::-1]
    return A


def _ls_val(a, g, t):
    cand = (1.0 - t) * a + t * g
    np.maximum(cand, 0.0, out=cand)
    cs = cand.sum()
    if cs < 1e-15:
        return 1e10
    cand /= cs
    return float(np.convolve(cand, cand).max())


def _gradient_step(a, c, M, n, rng):
    """Subgradient descent step on active lags."""
    active = np.where(c >= M * 0.99999)[0]
    grad = np.zeros(n)
    for ks in active:
        ks = int(ks)
        jlo = max(0, ks - n + 1)
        jhi = min(n, ks + 1)
        for j in range(jlo, jhi):
            idx = ks - j
            if idx == j:
                grad[j] += 2.0 * a[idx]
            else:
                grad[j] += a[idx]
    grad /= max(len(active), 1)
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None
    for alpha in [0.02, 0.01, 0.005, 0.002, 0.001]:
        cand = a - alpha * grad / gn
        cand = np.maximum(cand, 0)
        cs = cand.sum()
        if cs > 1e-10:

            v = float(np.convolve(cand, cand).max())
