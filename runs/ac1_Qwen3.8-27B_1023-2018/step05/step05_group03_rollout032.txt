import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:
        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if not math.isfinite(s) or s <= 1e-12:
        p = np.ones(3072, dtype=np.float64)
    return p


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if not math.isfinite(s) or s < 0.01:
        return math.inf
    a = a / s
    c = _fft_conv(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m)


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if not math.isfinite(s) or s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _lbfgs_run(p, stop_time, consider=None):
    """Run LSE-smoothed optimization for a short burst."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n = int(p.size)
    if n <= 1 or time.monotonic() > stop_time:
        return p

    conv_len = 2 * n - 1
    cross_len = 3 * n - 2
    conv_size = next_fast_len(conv_len)
    cross_size = next_fast_len(cross_len)

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def p_to_u(pp):
        pp = np.maximum(pp, 1e-15)
        uu = np.log(pp, dtype=np.float64)
        uu -= float(np.mean(uu))
        return uu

    def lse_func(uu, T):
        if uu.size != n:
            return 1e100, np.zeros(n, dtype=np.float64)
        if not np.all(np.isfinite(uu)):
            uu = np.where(np.isfinite(uu), uu, 0.0)
        m = float(np.max(uu))
        e = np.exp(uu - m, dtype=np.float64)
        s = float(e.sum())
        if not math.isfinite(s) or s <= 0:
            pp = np.full(n, 1.0 / n)
        else:
            pp = e / s
        pp = np.where(np.isfinite(pp), pp, 1.0 / n)

        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros(n)

        T = max(1e-14, T)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            w = exps / sum_exp
            ratio = sum_exp / float(c.size)
            f = cmax + T * float(math.log(max(1e-300, ratio)))

        q = xfast(pp, w)
        gid = 2.0 * q
        gd = float(np.dot(pp, gid))
        grad = pp * (gid - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score_fft(p)

    c = cfast(p)
    cmax = float(c.max()) if c.size else 1.0

    stages = [(0.008, 0.35), (0.002, 0.35), (0.0005, 0.30)]
    total_w = sum(w for _, w in stages)
    start = time.monotonic()
    avail = max(1.0, stop_time - start - 5.0)

    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 10.0:
            break
        cum += wt
        end_stage = min(stop_time - 8.0, start + avail * (cum / total_w))
        if end_stage <= now + 5.0:
            continue

        T = max(1e-14, frac * cmax)
        stale = 0
        while time.monotonic() < end_stage - 2.0:
            opts = {"maxiter": 12, "gtol": 1e-10, "ftol": 1e-14,
                    "maxls": 12, "maxcor": 12, "disp": False, "maxdisp": 0}
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    fval = float(res.fun)
                    if math.isfinite(fval):
                        pp = _softmax_u(x)
                        sc = _score_fft(pp)
                        if math.isfinite(sc) and sc < best_sc - 1e-13:
                            best_sc = sc
                            best_p = pp.copy()
                            p = pp.copy()
                            u = p_to_u(p)
                            stale = 0
                            if consider:
                                try:
                                    consider(best_p)
                                except Exception:
                                    pass
                        else:
                            stale += 1
                    else:
                        stale += 1
                else:
                    stale += 1
            else:
                stale += 1
            if getattr(res, "status", 1) == 0 or stale >= 3:
                break
        c = cfast(best_p)
        cmax = float(c.max()) if c.size else 1.0

    return best_p


def _softmax_u(u):
    u = np.asarray(u, dtype=np.float64).ravel()
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _coord_polish(p, stop_time, consider=None, seed=777, G_base=80):
    """Efficient coordinate descent with adaptive granularity."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n = int(p.size)
    if n <= 1:
        return p, math.inf

    c = _fft_conv(p)
    L = 2 * n - 1
    infl = 2.0 * p
    S = float(p.sum())
    if S <= 0:
        return p, math.inf
    p = p / S
    c = _fft_conv(p)
    S = 1.0
    infl = 2.0 * p
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()

    if consider:
        try:
            consider(best_p)
        except Exception:
            pass

    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    G = G_base

    while time.monotonic() < stop_time - 1.5:
        # Periodically increase granularity
        if sweep % 50 == 0 and sweep > 100:
            G = min(160, G_base + sweep // 50 * 4)

        improved = False
        order = rng.permutation(n)

        for pos in range(n):
            if time.monotonic() > stop_time - 0.5:
                break
            if S <= 1e-300:
                break

            i = int(order[pos])
            v = float(p[i])

            start = i
            end = min(i + n, L)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end]
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < L:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc
            if not math.isfinite(max_const) or max_const < 0:
                max_const = 0.0

            di = 2 * i - start
            Sloc = float(S)

            # Generate candidate points
            xhi = max(20.0 * v, 0.004)
            xhi = min(xhi, 0.1)  # cap for large n

            pts_list = [0.0, v]
            # Geometric around v
            if v > 1e-12:
                factors = np.geomspace(0.05, 20.0, G * 2, dtype=np.float64)
                pts_list.extend((v * factors).tolist())
            # Linear range
            linhi = min(xhi, max(0.01, 8.0 / n))
            pts_list.extend(np.linspace(0.0, linhi, G, dtype=np.float64).tolist())
            # Very small values
            if v > 1e-8:
                pts_list.extend(np.geomspace(max(1e-10, v * 1e-4), v, G // 2,
                                            dtype=np.float64).tolist())

            pts = np.unique(np.clip(np.asarray(pts_list, dtype=np.float64), 0.0, xhi))
            if pts.size > 256:
                idx = np.linspace(0, pts.size - 1, 256, dtype=int)
                pts = pts[idx]

            # Vectorized evaluation
            w = pts - v
            den = Sloc + w
            valid = den > 1e-14
            aff = base[None, :] + w[:, None] * (infl[start:end])[None, :]
            if 0 <= di < Ls:
                aff[:, di] = base[di] + w * infl[di] + w * w
            np.maximum(aff, max_const, out=aff)
            mv = np.max(aff, axis=1)
            vals = np.full(pts.size, math.inf, dtype=np.float64)
            if np.any(valid):
                vals[valid] = (2.0 * n * mv[valid]) / (den[valid] * den[valid])
            vals = np.where(np.isfinite(vals), vals, math.inf)

            if not np.any(np.isfinite(vals)):
                continue
            im = int(np.argmin(vals))
            best_x = float(pts[im])
            best_val = float(vals[im])

            # Local refinement around best
            if best_val < cur_score - 1e-12 and 0.0 <= best_x <= xhi:
                step = max(v * 0.5, Sloc / n * 2.0, 1e-9)
                lo = max(0.0, best_x - step)
                hi = min(xhi, best_x + step)
                if hi > lo:
                    fine = np.linspace(lo, hi, 48, dtype=np.float64)
                    if lo > 0:
                        fine = np.concatenate([fine, np.geomspace(lo, hi, 24)])
                    fine = np.unique(np.clip(fine, 0.0, xhi))
                    if fine.size > 1:
                        wf = fine - v
                        denf = Sloc + wf
                        validf = denf > 1e-14
                        afff = base[None, :] + wf[:, None] * (infl[start:end])[None, :]
                        if 0 <= di < Ls:
                            afff[:, di] = base[di] + wf * infl[di] + wf * wf
                        np.maximum(afff, max_const, out=afff)
                        mvf = np.max(afff, axis=1)
                        valsf = np.full(fine.size, math.inf)
                        if np.any(validf):
                            valsf[validf] = (2.0 * n * mvf[validf]) / (denf[validf] * denf[validf])
                        valsf = np.where(np.isfinite(valsf), valsf, math.inf)
                        if np.any(np.isfinite(valsf)):
                            imf = int(np.argmin(valsf))
                            if valsf[imf] < best_val:
                                best_x = float(fine[imf])
                                best_val = float(valsf[imf])

            if (math.isfinite(best_val) and best_val < cur_score - 1e-12
                    and 0.0 <= best_x <= xhi):
                d = best_x - v
                if abs(d) > 1e-16:
                    c[start:end] += d * infl[start:end]
                    diag = 2 * i
                    if 0 <= diag < L:
                        c[diag] += d * d
                    p[i] = best_x
                    infl[i] = 2.0 * best_x
                    S += d
                    cur_score = best_val
                    improved = True

        # Renormalize
        if S > 1e-300 and math.isfinite(S):
            if S > 1.001 or S < 0.999:
                inv = 1.0 / S
                p *= inv
                c = c * inv * inv
                infl *= inv
                S = 1.0
                c = np.maximum(c, 0.0)
                cur_score = float(2.0 * n * c.max())

        # Recompute from FFT periodically to avoid drift
        if sweep % 20 == 0:
            c = _fft_conv(p)
            c = np.maximum(c, 0.0)
            infl = 2.0 * p
            S = float(p.sum())
            if S > 1e-300:
                p = p / S
                c = _fft_conv(p)
                c = np.maximum(c, 0.0)
                infl = 2.0 * p
                S = 1.0
                cur_score = float(2.0 * n * c.max())

        if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            if consider:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 8:
                # Escape: multiplicative noise
                if time.monotonic() < stop_time - 30.0:
                    sigmas = [0.03, 0.06, 0.10, 0.15]
                    sigma = sigmas[min(flat - 8, len(sigmas) - 1)]
                    noise = rng.normal(0.0, sigma, n)
                    q = p * np.exp(noise)
                    q = np.maximum(q, 0.0)
                    qs = float(q.sum())
                    if math.isfinite(qs) and qs > 0:
                        p = q / qs
                        c = _fft_conv(p)
                        c = np.maximum(c, 0.0)
                        infl = 2.0 * p
                        S = 1.0
                        cur_score = float(2.0 * n * c.max())
                        if cur_score < best_score:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider:
                                try:
                                    consider(best_p)
                                except Exception:
                                    pass
                    flat = 0
                else:
                    break

        sweep += 1

    if consider:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(10.0, budget)

    reserve = 20.0
    hard = t0 + budget - reserve

    parent = _get_parent()
    n_target = int(parent.size)

    best_arr = np.asarray(parent, dtype=np.float64).ravel().copy()
    best_arr = np.where(np.isfinite(best_arr), best_arr, 0.0)
    best_arr = np.clip(best_arr, 0.0, 1000.0)
    s = float(best_arr.sum())
    if not math.isfinite(s) or s < 0.01:
        best_arr = np.full(n_target if n_target > 0 else 64, 1.0)
    best_arr = best_arr / best_arr.sum()

    try:
        best_score = _score_exact(best_arr)
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score):
        best_arr = np.full(64, 1.0 / 64.0)
        best_score = _score_exact(best_arr)

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
            if arr.size == 0:
                return
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            s = float(arr.sum())
            if not math.isfinite(s) or s < 0.01:
                return
            arr = arr / s
            sc = _score_exact(arr)
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_arr = arr.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 1: Brief LSE optimization (up to 180s)
    now = time.monotonic()
    lse_stop = min(hard - 60.0, now + 180.0)
    if lse_stop > now + 20.0:
        try:
            p_lse = _lbfgs_run(best_arr, lse_stop, consider=consider)
            consider(p_lse)
        except Exception:
            pass

    # Phase 2: Coordinate descent (bulk of the time)
    now = time.monotonic()
    cd_stop = min(hard - 10.0, now + (hard - now - 15.0) * 0.80)
    if cd_stop > now + 20.0:
        try:
            p_cd, _ = _coord_polish(best_arr, cd_stop, consider=consider,
                                    seed=seed + 1, G_base=90)
            consider(p_cd)
        except Exception:
            pass

    # Phase 3: Second coordinate descent with different seed (if time)
    now = time.monotonic()
    if now < hard - 60.0:
        cd_stop2 = min(hard - 8.0, now + (hard - now - 10.0) * 0.85)
        if cd_stop2 > now + 30.0:
            try:
                p_cd2, _ = _coord_polish(best_arr, cd_stop2, consider=consider,
                                         seed=seed + 999, G_base=120)
                consider(p_cd2)
            except Exception:
                pass

    # Final validation and return
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = _get_parent()
            s = float(out.sum())
        if out.size == 0 or not math.isfinite(s) or s < 0.01:
            out = np.full(64, 1.0 / 64.0)
        out = out / float(out.sum())
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            return [1.0]
        return ls
    except Exception:
        try:
            ls = [float(x) for x in best_arr]
            if math.isfinite(evaluate_sequence(ls)):
                return ls
        except Exception:
            pass
        return [1.0]
