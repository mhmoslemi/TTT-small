def _coord_descent(p, stop_time, consider=None, seed=12345):
    n = int(p.size)
    if n == 0:
        return p, float('inf')
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p, float('inf')
    p = p / s
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()

    Ltot = 2 * n - 1
    flat = 0
    sweep = 0
    GR = (math.sqrt(5.0) - 1.0) / 2.0

    def eval_1d(w, base, infl, diag, max_const, S):
        s_new = S + w
        if s_new <= 1e-15:
            return float('inf')
        aff_max = float(np.max(base + w * infl))
        if 0 <= diag < len(base):
            dval = base[diag] + w * infl[diag] + w * w
            if dval > aff_max:
                aff_max = dval
        m = max(aff_max, max_const)
        return 2.0 * n * m / (s_new * s_new)

    max_sweeps = 200000
    while sweep < max_sweeps and time.monotonic() < stop_time - 3.5:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 1.5:
                break
            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue
            base = c[start_i:end_i]
            infl = 2.0 * p[:Ls]
            diag = i

            max_const = 0.0
            if start_i > 0:
                mc = float(c[:start_i].max())
                if mc > max_const:
                    max_const = mc
            if end_i < Ltot:
                mc = float(c[end_i:].max())
                if mc > max_const:
                    max_const = mc

            S = 1.0
            x_hi = max(60.0 * v, 40.0 / n, 0.02)
            x_hi = min(x_hi, 1.0)

            G = 64
            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                ins = np.searchsorted(pts, v)
                pts = np.insert(pts, ins, v)

            ws = pts - v
            s_new = S + ws
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag < Ls:
                aff[:, diag] += ws * ws
            maff = np.max(aff, axis=1)
            maff = np.maximum(maff, max_const)
            valid = s_new > 1e-12
            scores = np.full(pts.shape, float('inf'), dtype=np.float64)
            if np.any(valid):
                scores[valid] = 2.0 * n * maff[valid] / (s_new[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            if 0 < imin < len(pts) - 1:
                lo = float(pts[max(0, imin - 2)])
                hi = float(pts[min(len(pts) - 1, imin + 2)])
            else:
                lo = 0.0
                hi = x_hi
            if hi - lo < 1e-14:
                lo = max(0.0, best_x - 1e-6)
                hi = best_x + 1e-6

            a_gs = lo
            b_gs = hi
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)
            f_c = eval_1d(c_gs - v, base, infl, diag, max_const, S)
            f_d = eval_1d(d_gs - v, base, infl, diag, max_const, S)
            for _ in range(30):
                if b_gs - a_gs < 1e-13 * max(1.0, abs(best_x)):
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = eval_1d(c_gs - v, base, infl, diag, max_const, S)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_d = eval_1d(d_gs - v, base, infl, diag, max_const, S)
            x_mid = (a_gs + b_gs) / 2.0
            f_mid = eval_1d(x_mid - v, base, infl, diag, max_const, S)
            if f_mid < best_val:
                best_x = x_mid
                best_val = f_mid
            for xp in (0.0, x_hi):
                fp = eval_1d(xp - v, base, infl, diag, max_const, S)
                if fp < best_val:
                    best_x = xp
                    best_val = fp

            if math.isfinite(best_val) and best_val < cur_score - 1e-15 and 0.0 <= best_x <= x_hi:
                w = best_x - v
                if abs(w) > 1e-16:
                    p[i] = best_x
                    aff_seg = base + w * infl
                    if 0 <= diag < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag] += w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    c[start_i:end_i] = aff_seg
                    cur_score = 2.0 * n * new_max
                    improved = True

        s_new = float(p.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            inv = 1.0 / s_new
            p *= inv
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-16:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)

                        pass

                    if kick_type == 0:
                        ks = 0.01 + 0.03 * min(1.0, flat / 30.0)
                        p = p * np.exp(rng.normal(0.0, ks, n))
                    elif kick_type == 1:
                        ks = 0.005 + 0.01 * min(1.0, flat / 30.0)
                        even_mask = np.arange(n) % 2 == 0
                        p[even_mask] *= np.exp(rng.normal(0.0, ks, even_mask.sum()))
                        p[~even_mask] *= np.exp(rng.normal(0.0, ks, (~even_mask).sum()))
                    else:
                        ks = 0.008 + 0.02 * min(1.0, flat / 30.0)
                        perm = rng.permutation(n)
                        k_size = max(4, n // 64)
                        p[perm[:k_size]] *= np.exp(rng.normal(0.0, ks * 2, k_size))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _fine_anneal(p, stop_time, consider=None, seed=777):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    c = _conv_self(p)
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    scale = 0.02
    no_improve = 0
    while time.monotonic() < stop_time - 2.0:
        idx = int(rng.integers(0, n))
        delta = rng.normal(0.0, scale)
        v_new = p[idx] + delta
        if v_new < 0:
            v_new = 0.0
        p[idx] = v_new
        ss = float(p.sum())
        if ss < 1e-15:
            p[idx] = v_new = 0.0
            ss = 1.0
        p /= ss
        c = _conv_self(p)
        new_score = 2.0 * n * float(c.max())
        if new_score < cur_score - 1e-15:
            cur_score = new_score
            no_improve = 0
            if new_score < best_score - 1e-16:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass
        else:
            p[idx] = p[idx] / (1.0 + delta / p[idx]) if p[idx] > 0 else 0.0
            p[idx] = max(0.0, p[idx] - delta * ss / n)
            p[idx] = max(0.0, v_new if new_score >= cur_score else p[idx])
            no_improve += 1
            if no_improve > 500:
                scale *= 0.5
                no_improve = 0
                if scale < 1e-8:
                    break

    p = best_p
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    return p

        x = (np.arange(n_work, dtype=np.float64) + 0.5) / n_work
        # Cosine-modulated alternating
        even_w = 1.0 + 0.3 * np.cos(2.0 * np.pi * x * 3.0)
        odd_w = 1.0 - 0.3 * np.cos(2.0 * np.pi * x * 3.0 + 0.5)
        alt = np.empty(n_work, dtype=np.float64)
        alt[0::2] = even_w[0::2]
        alt[1::2] = odd_w[1::2]
        alt = np.maximum(alt, 0.0)
        s_alt = float(alt.sum())
        if s_alt > 0:
            alt /= s_alt
        # Blend with parent
        p_blend = 0.4 * alt + 0.6 * (best_arr / float(best_arr.sum()))
        p_blend = np.maximum(p_blend, 0.0)
        p_blend /= float(p_blend.sum())
        consider(p_blend)

        blend_end = min(hard - 50.0, time.monotonic() + rem * 0.5)
        if blend_end > time.monotonic() + 25.0:
            try:
                p_b, _ = _coord_descent(p_blend, blend_end, consider=consider, seed=int(seed + 8888))
                if p_b is not None and p_b.size > 0 and np.all(np.isfinite(p_b)) and p_b.sum() > 0:
                    consider(p_b)
            except Exception:
                pass

    # Phase 4: Final coordinate descent squeeze on the overall best
    rem = hard - 15.0 - time.monotonic()
    if rem > 30.0:
        p_final = best_arr.copy()
        p_final = np.maximum(p_final, 0.0)
        s_f = float(p_final.sum())
        if s_f > 0:
            p_final /= s_f
        try:
            p_f, _ = _coord_descent(p_final, hard - 15.0, consider=consider, seed=int(seed + 9999))
            if p_f is not None and p_f.size > 0 and np.all(np.isfinite(p_f)) and p_f.sum() > 0:
                consider(p_f)
        except Exception:
            pass

    # Phase 5: Fine anneal on the absolute best
    rem = hard - 10.0 - time.monotonic()
    if rem > 15.0:
        p_fin = best_arr.copy()
        p_fin = np.maximum(p_fin, 0.0)
        s_ff = float(p_fin.sum())
        if s_ff > 0:
            p_fin /= s_ff
        try:
            p_a = _fine_anneal(p_fin, hard - 10.0, consider=consider, seed=int(seed + 5555))
            if p_a is not None and p_a.size > 0 and np.all(np.isfinite(p_a)) and p_a.sum() > 0:
                consider(p_a)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
