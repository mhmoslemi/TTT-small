        c = cfast(pp)
        if c.size != conv_len:
            return 1e100, np.zeros(n, dtype=np.float64)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros(n, dtype=np.float64)

        T = max(1e-14, T)
        scaled = (c - cmax) / T
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(conv_len, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            w = exps / sum_exp
            ratio = sum_exp / float(c.size)
            f = cmax + T * float(math.log(max(ratio, 1e-300)))
            if not math.isfinite(f):
                f = cmax

        q = xfast(pp, w)
        gid = 2.0 * q
        gd = float(np.dot(pp, gid))
        grad = pp * (gid - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    def run_chunk(uu, T, end):
        try:
            f0, _ = lse_func(uu, T)
        except Exception:
            return uu
        if not math.isfinite(f0):
            return uu
        best_u = np.array(uu, dtype=np.float64, copy=True)
        best_f = float(f0)
        stale = 0
        while time.monotonic() < end - 2.0:
            maxit = 12 if n > 2000 else 18
            opts = {"maxiter": maxit, "gtol": 1e-10, "ftol": 1e-15,
                    "maxls": 12, "maxcor": 12, "disp": False, "maxdisp": 0}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-14:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, "status", 1) == 0:
                break
        return best_u

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score_fft(p)
    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass

    if fine_only:
        stages = ((0.0008, 0.25), (0.0002, 0.35), (0.00005, 0.40))
    else:
        stages = ((0.015, 0.12), (0.004, 0.15), (0.001, 0.20),
                  (0.0003, 0.22), (0.00008, 0.31))

    start = time.monotonic()
    avail = float(stop_time - start - 4.0)
    if avail <= 5.0:
        return best_p
    total_w = float(sum(wt for _, wt in stages))
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 4.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-14, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_new = _softmax(u)
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if math.isfinite(ss) and ss > 1e-300:
            p_new /= ss
        else:
            p_new = best_p.copy()
        sc_new = _score_fft(p_new)
        if math.isfinite(sc_new) and sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_p = p_new.copy()
            p = p_new.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass
        else:
            p = best_p.copy()
        u = p_to_u(p)

    return best_p


def _coord_descent(p, stop_time, consider, G=72, seed=12345, max_sweeps=500):
    p = _norm_nonneg(p)
    n = int(p.size)
    if n <= 1:
        return p, _score_fft(p)

    L = 2 * n - 1
    c = _fft_conv(p)
    c = np.maximum(c, 0.0)
    S = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass

    rng = np.random.default_rng(seed)
    flat = 0
    sweep = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)

        for pos in range(n):
            if time.monotonic() > stop_time - 0.3:
                break
            if S <= 1e-300:
                break
            if S > 1.15 or S < 0.85:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0
                c = np.maximum(c, 0.0)
                cur_score = float(2.0 * n * c.max())

            i = int(order[pos])
            v = float(p[i])

            # Determine affected range
            start = max(0, i)
            end = min(L, i + n)
            # Affected m values: m where 0 <= m-i < n, i.e., i <= m < i+n
            m_lo = i
            m_hi = min(L, i + n)
            Ls = m_hi - m_lo
            if Ls <= 0:
                continue

            # Base convolution excluding coordinate i's contribution
            # infl[m] = 2*a[m-i] for m in [m_lo, m_hi), m != 2i
            idx = np.arange(m_lo, m_hi, dtype=np.int32)
            shift = idx - i  # 0 to n-1
            infl_vec = 2.0 * p[shift].copy()

            # Diagonal position
            diag = 2 * i
            di = diag - m_lo  # local index of diagonal

            # Precompute max outside affected range
            max_const = 0.0
            if m_lo > 0:
                mc = float(c[:m_lo].max())
                if mc > max_const:
                    max_const = mc
            if m_hi < L:
                mc = float(c[m_hi:].max())
                if mc > max_const:
                    max_const = mc

            base = c[m_lo:m_hi].copy()

            # Build candidate x values
            xhi = max(20.0 * v, 20.0 * S / float(n), 0.003)
            xhi = min(xhi, max(0.04, 2.0 * v))
            if not math.isfinite(xhi) or xhi <= 0:
                xhi = max(1e-8, 10.0 * S / float(n))

            pts = [0.0, v, xhi]
            minx = max(1e-9, 1e-4 * v, S / (3000.0 * float(n)))
            if minx < xhi:
                m1 = max(14, int(G * 0.45))
                pts.extend(np.geomspace(minx, xhi, m1, dtype=np.float64).tolist())
            if v > 1e-12:
                m2 = max(8, int(G * 0.25))
                factors = np.geomspace(0.15, 6.0, m2, dtype=np.float64)
                pts.extend((v * factors).tolist())
            m3 = max(8, int(G * 0.30))
            linhi = min(xhi, max(6.0 * S / float(n), 0.001))
            if linhi > 0:
                pts.extend(np.linspace(0.0, linhi, m3, dtype=np.float64).tolist())

            pts = np.unique(np.clip(np.asarray(pts, dtype=np.float64), 0.0, xhi))
            if pts.size == 0:
                pts = np.array([v], dtype=np.float64)

            # Evaluate f(x) for all candidate x
            w = pts - v
            den = S + w
            valid = den > 1e-14

            # aff[j, k] = base[k] + w[j] * infl_vec[k]
            aff = base[None, :] + w[:, None] * infl_vec[None, :]
            # Handle diagonal: add w^2 (since x^2 - v^2 = (x-v)^2 + 2v(x-v) = w^2 + 2vw)
            # Actually: x^2 - v^2 = (x-v)(x+v) = w*(v+w+v) = w*(2v+w)
            # But infl_vec already has 0 at diagonal (we set it to 2*a[shift] but at shift=i, a[shift]=a[i]=v)
            # Wait: at m=diag=2i, shift = 2i - i = i, so infl_vec[di] = 2*a[i] = 2v
            # The linear part contributes w*2v, and the quadratic part is x^2 - v^2 - 2v*w = w^2
            if 0 <= di < Ls:
                aff[:, di] = base[di] + w * 2.0 * v + w * w

            np.maximum(aff, max_const, out=aff)
            mv = np.max(aff, axis=1)
            vals = np.full(pts.size, math.inf, dtype=np.float64)
            if np.any(valid):
                vals[valid] = (2.0 * n * mv[valid]) / (den[valid] * den[valid])
            vals = np.where(np.isfinite(vals), vals, math.inf)

            if not np.any(np.isfinite(vals)):
                continue
            im = int(np.argmin(vals))
            best_x = float(pts[im])
            best_val = float(vals[im])

            # Refine around best
            if best_val < cur_score - 1e-12:
                if best_x <= 0:
                    lo = 0.0
                    hi = max(5.0 * S / float(n), 5.0 * v, 1e-7)
                else:
                    step = max(v, S / float(n), 1e-9)
                    lo = max(0.0, best_x * 0.6, best_x - 3.0 * step)
                    hi = min(xhi, max(best_x * 1.4, best_x + 3.0 * step))
                if hi <= lo:
                    hi = lo + 1e-9
                fine = np.linspace(lo, hi, 40, dtype=np.float64)
                if lo > 0:
                    fine = np.concatenate([fine, np.geomspace(lo, hi, 20, dtype=np.float64)])
                fine = np.unique(np.clip(fine, 0.0, xhi))

                wf = fine - v
                denf = S + wf
                validf = denf > 1e-14
                afff = base[None, :] + wf[:, None] * infl_vec[None, :]
                if 0 <= di < Ls:
                    afff[:, di] = base[di] + wf * 2.0 * v + wf * wf
                np.maximum(afff, max_const, out=afff)
                mvf = np.max(afff, axis=1)
                valsf = np.full(fine.size, math.inf, dtype=np.float64)
                if np.any(validf):
                    valsf[validf] = (2.0 * n * mvf[validf]) / (denf[validf] * denf[validf])
                valsf = np.where(np.isfinite(valsf), valsf, math.inf)
                if np.any(np.isfinite(valsf)):
                    imf = int(np.argmin(valsf))
                    if valsf[imf] < best_val:
                        best_x = float(fine[imf])
                        best_val = float(valsf[imf])

            if (math.isfinite(best_val) and best_val < cur_score - 1e-11
                    and 0.0 <= best_x <= xhi):
                d = best_x - v
                if abs(d) > 1e-16:
                    c[m_lo:m_hi] += d * infl_vec
                    if 0 <= diag < L:
                        c[diag] += d * d
                    p[i] = best_x
                    S += d
                    cur_score = best_val
                    improved = True

        # End of sweep
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        c = _fft_conv(p)
        c = np.maximum(c, 0.0)
        desc = float(2.0 * n * c.max())
        if math.isfinite(desc) and desc < best_score - 1e-14:
            best_score = desc
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 20.0:
                    # Perturbation to escape local minimum
                    lvl = min(flat - 3, 4)
                    sigmas = (0.02, 0.05, 0.10, 0.18, 0.30)
                    sigma = sigmas[lvl]
                    noise = rng.normal(0.0, sigma, n, dtype=np.float64)
                    q = p * np.exp(noise)
                    if sigma >= 0.10:
                        x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
                        amp = 0.15 + 0.4 * float(rng.random())
                        f1 = 1 + int(rng.integers(0, 6))
                        ph1 = 2.0 * math.pi * float(rng.random())
                        q *= np.exp(amp * np.sin(2.0 * math.pi * f1 * x + ph1))
                    q = np.maximum(q, 0.0)
                    qs = float(q.sum())
                    if math.isfinite(qs) and qs > 0:
                        p = q / qs
                        c = _fft_conv(p)
                        c = np.maximum(c, 0.0)
                        S = 1.0
                        cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p, best_score


def _subgradient_step(p, stop_time, consider, n_iter=50, lr_base=0.002):
    """Simple subgradient method using softmax-weighted gradient of max."""
    p = _norm_nonneg(p)
    n = int(p.size)
    if n <= 1:
        return p

    conv_len = 2 * n - 1
    conv_size = next_fast_len(conv_len)
    cross_len = 3 * n - 2
    cross_size = next_fast_len(cross_len)

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    c = cfast(p)
    best_p = p.copy()
    best_sc = float(2.0 * n * c.max())
    if consider:
        try:
            consider(best_p)
        except Exception:
            pass

    for it in range(n_iter):
        if time.monotonic() > stop_time - 2.0:
            break
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            break

        # Softmax weighting with temperature
        T = max(1e-6, 0.005 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if sum_exp <= 0 or not math.isfinite(sum_exp):
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
        else:
            w = exps / sum_exp

        # Gradient direction
        q = xfast(p, w)
        gid = 2.0 * q
        gd = float(np.dot(p, gid))
        grad = p * (gid - gd)  # gradient of log-sum-exp w.r.t. log params, scaled by p
        grad = np.where(np.isfinite(grad), grad, 0.0)

        # Step size with decay
        lr = lr_base / (1.0 + 0.01 * it)
        # Move in negative gradient direction
        p_new = p * np.exp(-lr * grad / max(cmax, 1e-14))
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if not math.isfinite(ss) or ss <= 0:
            break
        p_new /= ss

        c = cfast(p_new)
        sc = float(2.0 * n * c.max())
        if math.isfinite(sc) and sc < best_sc - 1e-14:
            best_sc = sc
            best_p = p_new.copy()
            p = p_new.copy()
            if consider:
                try:
                    consider(best_p)
                except Exception:
                    pass
        else:
            p = best_p.copy()
            c = cfast(p)

    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(10.0, budget)
    reserve = 20.0 if budget > 200 else 8.0
    hard = t0 + budget - reserve
    if hard <= t0 + 2.0:
        hard = t0 + max(1.0, budget - 0.5)

    parent = _get_parent()
    n_target = int(parent.size) if parent.size > 0 else 3072

    best_arr = _norm_nonneg(parent.copy())
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.full(64, 1.0/64.0, dtype=np.float64)
        best_score = 2.0

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return
        if arr.size == 0:
            return
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if not math.isfinite(s) or s < 0.01:
            return
        arr = arr / s
        try:
            sc = float(_score_exact_arr(arr))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_arr = arr.copy()
            best_score = sc

    # Phase 1: Quick exploratory perturbations
    phase1_end = t0 + max(5.0, 0.03 * budget)
    rng_exp = np.random.default_rng(seed)

    # Power-law reshaping
    for q in (0.6, 0.85, 1.15, 1.5):
        if time.monotonic() > phase1_end:
            break
        cand = np.power(np.maximum(best_arr, 1e-12), float(q))
        cand = _norm_nonneg(cand)
        consider(cand)

    # Linear tilt
    for side in (1.0, -1.0):
        if time.monotonic() > phase1_end:
            break
        for amp in (0.3, 0.7, 1.2):
            x = (np.arange(n_target, dtype=np.float64) + 0.5) / float(n_target)
            cand = best_arr * np.exp(side * amp * (2.0 * x - 1.0))
            cand = np.maximum(cand, 0.0)
            cand = _norm_nonneg(cand)
            consider(cand)

    # Sine perturbations
    for _ in range(8):
        if time.monotonic() > phase1_end:
            break
        amp = 0.1 + 0.5 * float(rng_exp.random())
        f1 = 1 + int(rng_exp.integers(0, 5))
        ph = 2.0 * math.pi * float(rng_exp.random())
        x = (np.arange(n_target, dtype=np.float64) + 0.5) / float(n_target)
        cand = best_arr * np.exp(amp * np.sin(2.0 * math.pi * f1 * x + ph))
        cand = np.maximum(cand, 0.0)
        cand = _norm_nonneg(cand)
        consider(cand)

    # Phase 2: L-BFGS with medium smoothing
    phase2_end = min(hard - 300.0, t0 + 0.18 * budget)
    if phase2_end > time.monotonic() + 15.0:
        try:
            p2 = _lbfgs_smoothed(best_arr, phase2_end, consider, fine_only=False)
            consider(p2)
            if p2 is not None and p2.size == n_target:
                best_arr = _norm_nonneg(p2)
        except Exception:
            pass

    # Phase 3: Main coordinate descent (bulk of time)
    phase3_end = min(hard - 100.0, t0 + 0.55 * budget)
    if phase3_end > time.monotonic() + 20.0:
        try:
            p3, _ = _coord_descent(
                best_arr, phase3_end, consider,
                G=80, seed=seed + 100, max_sweeps=800
            )
            consider(p3)
            if p3 is not None and p3.size == n_target:
                best_arr = _norm_nonneg(p3)
        except Exception:
            pass

    # Phase 4: Subgradient step to escape any local minimum
    phase4_end = min(hard - 60.0, time.monotonic() + max(20.0, 0.08 * budget))
    if phase4_end > time.monotonic() + 15.0:
        try:
            p4 = _subgradient_step(best_arr, phase4_end, consider,
                                   n_iter=80, lr_base=0.003)
            consider(p4)
            if p4 is not None and p4.size == n_target:
                best_arr = _norm_nonneg(p4)
        except Exception:
            pass

    # Phase 5: Fine L-BFGS
    phase5_end = min(hard - 80.0, time.monotonic() + max(20.0, 0.12 * budget))
    if phase5_end > time.monotonic() + 15.0:
        try:
            p5 = _lbfgs_smoothed(best_arr, phase5_end, consider, fine_only=True)
            consider(p5)
            if p5 is not None and p5.size == n_target:
                best_arr = _norm_nonneg(p5)
        except Exception:
            pass

    # Phase 6: Final coordinate descent polish
    phase6_end = hard - 5.0
    if phase6_end > time.monotonic() + 15.0:
        try:
            p6, _ = _coord_descent(
                best_arr, phase6_end, consider,
                G=96, seed=seed + 9999, max_sweeps=600
            )
            consider(p6)
            if p6 is not None and p6.size == n_target:
                best_arr = _norm_nonneg(p6)
        except Exception:
            pass

    # Final validation and return
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or not math.isfinite(s) or s < 0.01:
            out = _norm_nonneg(parent)
            s = float(out.sum())
        if out.size == 0 or not math.isfinite(s) or s < 0.01:
            return [1.0]
        out = out / s
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            fallback = _norm_nonneg(parent)
            ls2 = [float(x) for x in fallback]
            sc2 = float(evaluate_sequence(ls2))
            if math.isfinite(sc2):
                return ls2
            return [1.0]
        return ls
    except Exception:
        return [1.0]


def _score_exact_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if not math.isfinite(s) or s < 0.01:
        return math.inf
    a = a / s
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return float(2.0 * n * m)
