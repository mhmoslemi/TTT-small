    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)

    c = irfft(F * F, size)[:L]

def _score_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()

        return float('inf')

                        aff_seg = aff_seg.copy()
                        aff_seg[diag] += w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    c[start:end] = aff_seg
                    cur_score = 2.0 * n * new_max
                    improved = True

        s_new = float(p.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            inv = 1.0 / s_new
            p *= inv
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-15:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 15.0:
                    kick_size = 0.02 + 0.06 * min(1.0, flat / 12.0)
                    p = p * np.exp(rng.normal(0.0, kick_size, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 30.0
    if hard <= t0 + 5.0:
        hard = t0 + max(3.0, budget - 2.0)

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float('inf')

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()

    consider(parent)

    n_work = int(best_arr.size)
    if n_work < 128:
        n_work = 384
    if n_work > 4096:
        n_work = 3072

    # Phase 1: LSE optimization (aggressive)
    lse_time = min(200.0, 0.16 * budget)
    p_work = _resample(best_arr, n_work)
    lse_end = t0 + lse_time
    if lse_end > t0 + 20.0 and time.monotonic() < hard - 100.0:
        try:
            p2 = _lse_optimize(p_work, lse_end, consider=consider)
            if p2 is not None and p2.size == n_work and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p_work = p2 / float(p2.sum())
                consider(p_work)
        except Exception:
            pass

    # Phase 2: Main coordinate descent (bulk of time)
    cd_budget = 0.60 * budget
    cd_end = min(hard - 80.0, time.monotonic() + cd_budget)
    if cd_end > time.monotonic() + 40.0:
        p_start = _resample(best_arr, n_work)
        try:
            p_cd, _ = _coord_descent(p_start, cd_end, consider=consider, seed=int(seed + 7777))
            if p_cd is not None and p_cd.size > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                consider(p_cd)
        except Exception:
            pass

    # Phase 3: Alternative initialization + CD
    rem = hard - 60.0 - time.monotonic()
    if rem > 80.0:
        rng3 = np.random.default_rng(int(seed + 55555))
        x = (np.arange(n_work, dtype=np.float64) + 0.5) / n_work
        # Gaussian-like window
        sigma = n_work / 5.0
        gauss = np.exp(-0.5 * ((x - 0.5) * 2.0) ** 2 * (n_work / (4.0 * sigma)) ** 2)
        gauss = np.maximum(gauss, 0.0)
        s_g = float(gauss.sum())
        if s_g > 0:
            gauss /= s_g
        p_blend = 0.4 * gauss + 0.6 * _resample(best_arr, n_work)
        p_blend = np.maximum(p_blend, 0.0)
        ss_b = float(p_blend.sum())
        if ss_b > 0:
            p_blend /= ss_b
        consider(p_blend)

        blend_end = min(hard - 60.0, time.monotonic() + rem * 0.55)
        if blend_end > time.monotonic() + 20.0:
            try:
                p_b, _ = _coord_descent(p_blend, blend_end, consider=consider, seed=int(seed + 8888))
                if p_b is not None and p_b.size > 0 and np.all(np.isfinite(p_b)) and p_b.sum() > 0:
                    consider(p_b)
            except Exception:
                pass

    # Phase 4: Final CD squeeze on the best
    rem = hard - 25.0 - time.monotonic()
    if rem > 40.0:
        p_final = _resample(best_arr, n_work)
        try:
            p_f, _ = _coord_descent(p_final, hard - 25.0, consider=consider, seed=int(seed + 9999))
            if p_f is not None and p_f.size > 0 and np.all(np.isfinite(p_f)) and p_f.sum() > 0:
                consider(p_f)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
