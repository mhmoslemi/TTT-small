def _cross_fft(a, w, cross_size):
    """Compute q[j] = sum_m a[m] * w[j+m] for j=0..n-1"""
    n = a.size
    F = rfft(a[::-1], cross_size)
    G = rfft(w, cross_size)
    c = irfft(F * G, cross_size)
    return c[n-1:2*n-1]

def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = a.size
    c = _fft_conv(a)
    return float(2.0 * n * c.max())

def _lse_optimize(p, stop_time, consider):
    """LSE optimization in log-space."""
    p = _normalize(p.copy())
    n = p.size
    conv_len = 2*n - 1
    cross_len = 3*n - 2
    conv_size = next_fast_len(conv_len)
    cross_size = next_fast_len(cross_len)
    
    def lse_func(u, T):
        # Convert u to p = softmax(u)
        m = float(np.max(u))
        e = np.exp(u - m)
        s = float(e.sum())
        if s <= 0:
            return 1e100, np.zeros(n)
        pp = e / s
        
        # Compute convolution
        F = rfft(pp, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        cmax = float(c.max())
        
        # LSE value
        if T <= 0:
            T = 1e-14
        scaled = (c - cmax) / T
        max_scaled = float(np.max(scaled))
        if max_scaled > 500:
            # Overflow protection
            exps = np.exp(scaled - max_scaled)
            sum_exp = float(exps.sum())
            if sum_exp <= 0:
                w = np.zeros(conv_len)
                w[int(np.argmax(c))] = 1.0
            else:
                w = exps / sum_exp
            f = cmax + T * (max_scaled + math.log(sum_exp / conv_len))
        else:
            exps = np.exp(scaled)
            sum_exp = float(exps.sum())
            if sum_exp <= 0:
                w = np.zeros(conv_len)
                w[int(np.argmax(c))] = 1.0
                f = cmax
            else:
                w = exps / sum_exp
                f = cmax + T * math.log(sum_exp / conv_len)
        
        # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j]
        # = 2 * cross_correlation(w, a)[j]
        # Using FFT:
        F2 = rfft(pp[::-1], cross_size)
        G2 = rfft(w, cross_size)
        c2 = irfft(F2 * G2, cross_size)
        q = c2[n-1:2*n-1]  # q[j] = sum_m pp[m] * w[j+m]
        
        # Gradient of LSE w.r.t. u
        # dLSE/du[j] = sum_k w[k] * dc[k]/du[j]
        # dc[k]/du[j] = dc[k]/da[j] * da[j]/du[j] + sum_{j'≠j} dc[k]/da[j'] * da[j']/du[j]
        # = 2*a[k-j]*pp[j] - pp[j]*sum_{j'} 2*a[k-j']*pp[j']
        # = pp[j] * (2*q[j] - 2*sum_j' pp[j']*q[j'])
        # = pp[j] * 2 * (q[j] - dot(pp, q))
        
        gd = float(np.dot(pp, q))
        grad = pp * 2.0 * (q - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        
        return float(f), grad
    
    # Convert to log-space
    p_clipped = np.maximum(p, 1e-15)
    u = np.log(p_clipped)
    u -= np.mean(u)
    
    best_u = u.copy()
    best_f = lse_func(u, 0.01)[0]
    
    # Temperature schedule
    c = _fft_conv(p)
    cmax0 = float(c.max())
    
    stages = [
        (0.010, 0.15),
        (0.004, 0.15),
        (0.0015, 0.20),
        (0.0005, 0.25),
        (0.0002, 0.25),
    ]
    
    t_start = time.monotonic()
    avail = max(0.0, stop_time - t_start - 3.0)
    total_w = sum(w for _, w in stages)
    cum = 0.0
    
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        cum += wt
        end_stage = t_start + avail * (cum / total_w)
        end_stage = min(end_stage, stop_time - 3.0)
        if end_stage <= now + 2.0:
            continue
        
        T = max(1e-14, frac * cmax0)
        
        # Run L-BFGS-B
        while time.monotonic() < end_stage - 1.0:
            try:
                res = minimize(lse_func, best_u, args=(T,), method='L-BFGS-B', jac=True,
                              options={'maxiter': 20, 'gtol': 1e-10, 'disp': False, 'maxls': 15, 'maxcor': 12})
            except Exception:
                break
            
            if res.x is not None and np.all(np.isfinite(res.x)):
                f_new = float(res.fun) if np.isfinite(res.fun) else 1e100
                if f_new < best_f - 1e-12:
                    best_f = f_new
                    best_u = res.x.copy()
                else:
                    break
    
    # Convert back
    p_new = _normalize(_softmax(best_u))
    return p_new

def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0:
        return np.full(u.size, 1.0/u.size)
    return e / s
