```python
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.asarray(c, dtype=np.float64)

    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size

        return float('inf')

        return float('inf')

def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.shape, 1.0 / u.size, dtype=np.float64)
    return e / s

def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s
```

```python
def _subgradient_kick(p, eta=0.05):
    """Take a subgradient step to reduce the max autocorrelation."""
    n = p.size
    c = _conv_self(p)
    k_star = int(np.argmax(c))
    # Gradient of c[k_star] wrt a[i] = 2*a[k_star - i] (for valid i)
    # Valid i: max(0, k_star-n+1) <= i <= min(k_star, n-1)
    i_lo = max(0, k_star - n + 1)
    i_hi = min(k_star, n - 1)
    grad = np.zeros(n)
    # For i in [i_lo, i_hi]: grad[i] = 2*p[k_star - i]
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] = 2.0 * p[k_star - idx]
    # Step: a <- max(0, a - eta * grad / (n * max_grad))
    max_g = float(grad.max())
    if max_g > 0:
        step = eta * grad / (n * max_g)
        p_new = np.maximum(0.0, p - step)
        s = float(p_new.sum())
        if s > 0:
            p_new /= s
        else:
            p_new = p.copy()
        return p_new
    return p
```

```python
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _lse_adam(p, stop_time, consider=None, T_start=0.05, T_end=0.0005):
    n = p.size
    p = np.maximum(np.asarray(p, dtype=np.float64), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    u = np.log(np.maximum(p, 1e-30), dtype=np.float64)
    u = u - float(u.mean())

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def lse_grad(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = 1e-6
        scaled = (c - cmax) / T
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            w = exps / sum_exp
        # Gradient wrt a: 2 * cross-correlation of pp and w
        F_a = rfft(pp[::-1]

                   , cross_size)
        F_w = rfft(w, cross_size
                  )
        cross = irfft(F_a * F_w, cross_size)
        grad_wrt_a = 2.0 * cross[n - 1:2 * n - 1]
        tdp = float(np.dot(pp, grad_wrt_a))
        grad = pp * (grad_wrt_a - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return f, grad

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_ = np.zeros_like(u)
    v_ = np.zeros_like(u)
    t = 0

    best_p = p.copy()
    best_sc = _score_arr(p)
    start = time.monotonic()
    total_time = max(1.0, stop_time - start)

    while time.monotonic() < stop_time - 5.0:
        frac = min(1.0, (time.monotonic() - start) / (total_time * 0.8))
        lr = 0.008 * (1.0 - 0.7 * frac)
        T = T_start * (T_end / T_start) ** frac
        T = max(T, 1e-7)

        f, grad = lse_grad(u, T)
        if not math.isfinite(f):
            break

        t += 1
        m_ = beta1 * m_ + (1 - beta1) * grad
        v_ = beta2 * v_ + (1 - beta2) * (grad * grad)
        m_hat = m_ / (1 - beta1 ** t)
        v_hat = v_ / (1 - beta2 ** t)
        u = u - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 100 == 0:
            p_cur = _softmax(u)
            sc = _score_arr(p_cur)
            if math.isfinite(sc) and sc < best_sc - 1e-14:
                best_sc = sc
                best_p = p_cur.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

    p_final = _softmax(u)
    sc = _score_arr(p_final)
    if math.isfinite(sc) and sc < best_sc:
        best_p = p_final
    return best_p


def _coord_descent_v2(p, stop_time, consider=None, seed=12345):
    n = int(p.size)
    if n == 0:
        return p
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    Ltot = 2 * n - 1
    GR = (math.sqrt(5.0) - 1.0) / 2.0

    def eval_1d(w, base, infl, diag, max_const):
        s_new = 1.0 + w
        if s_new <= 1e-15:
            return float('inf')
        aff_max = float(np.max(base + w * infl))
        if 0 <= diag < len(base):
            dval = base[diag] + w * infl[diag] + w * w
            if dval > aff_max:
                aff_max = dval
        m = max(aff_max, max_const)
        return 2.0 * n * m / (s_new * s_new)

    flat = 0
    sweep = 0
    G = 64

    while time.monotonic() < stop_time - 3.0:
        if flat < 2:
            x_hi_mult = 2.5
        elif flat < 6:
            x_hi_mult = 1.0
        else:
            x_hi_mult = 0.4

        improved = False
        order = rng.permutation(n)

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 1.0:
                break
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end]
            infl = 2.0 * p[:Ls]
            diag = i

            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < Ltot:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

            x_hi = max(x_hi_mult * 8.0 * max(v, 1e-6), 4.0 / n, 0.003)
            x_hi = min(x_hi, 0.8)

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                ins = np.searchsorted(pts, v)
                if ins < len(pts):
                    pts = np.insert(pts, ins, v)

            ws = pts - v
            s_new = 1.0 + ws
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag < Ls:
                aff = aff.copy()
                aff[:, diag] += ws * ws
            maff = np.max(aff, axis=1)
            maff = np.maximum(maff, max_const)
            valid = s_new > 1e-12
            scores = np.full(pts.shape, float('inf'), dtype=np.float64)
            if np.any(valid):
                scores[valid] = 2.0 * n * maff[valid] / (s_new[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            if 0 < imin < len(pts) - 1:
                lo = float(pts[max(0, imin - 1)])
                hi = float(pts[min(len(pts) - 1, imin + 1)])
            else:
                lo, hi = 0.0, x_hi
            if hi - lo < 1e-14:
                lo = max(0.0, best_x - 1e-8)
                hi = best_x + 1e-8

            a_gs, b_gs = lo, hi
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)
            f_c = eval_1d(c_gs - v, base, infl, diag, max_const)
            f_d = eval_1d(d_gs - v, base, infl, diag, max_const)
            for _ in range(22):
                if b_gs - a_gs < 1e-13:
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = eval_1d(c_gs - v, base, infl, diag, max_const)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_d = eval_1d(d_gs - v, base, infl, diag, max_const)
            x_mid = (a_gs + b_gs) / 2.0
            f_mid = eval_1d(x_mid - v, base, infl, diag, max_const)
            if f_mid < best_val:
                best_x = x_mid
                best_val = f_mid
            for xp in (0.0, x_hi):
                fp = eval_1d(xp - v, base, infl, diag, max_const)
                if fp < best_val:
                    best_x = xp
                    best_val = fp

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= x_hi:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    aff_seg = base + w * infl
                    if 0 <= diag < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag] += w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    c[start:end] = aff_seg
                    cur_score = 2.0 * n * new_max / ((1.0 + w) ** 2)
                    improved = True

        s_new = float(p.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            p *= 1.0 / s_new
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-15:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 15.0:
                    # Structure-aware kick
                    k_star = int(np.argmax(c))
                    i_lo = max(0, k_star - n + 1)
                    i_hi = min(k_star, n - 1)
                    kick_size = 0.04 + 0.06 * min(1.0, flat / 8.0)
                    kick = rng.normal(0.0, kick_size * 0.4, n)
                    kick[i_lo:i_hi + 1] += rng.normal(0.0, kick_size, i_hi - i_lo + 1)
                    p = p * np.exp(kick)
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 35.0
    if hard <= t0 + 5.0:
        hard = t0 + max(3.0, budget - 2.0)

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float('inf')

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()

    consider(parent)
    n_work = int(best_arr.size)
    if n_work < 128:
        n_work = 384
    if n_work > 4096:
        n_work = 3072

    # Phase 1: LSE-Adam from parent
    lse_time = min(80.0, 0.08 * budget)
    p_work = _resample(best_arr, n_work)
    lse_end = t0 + lse_time
    if lse_end > t0 + 15.0:
        try:
            p2 = _lse_adam(p_work, lse_end, consider=consider, T_start=0.05, T_end=0.001)
            if p2 is not None and p2.size == n_work and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p_work = p2 / float(p2.sum())
                consider(p_work)
        except Exception:
            pass

    # Phase 2: Main CD
    cd_end = hard - 50.0
    if cd_end > time.monotonic() + 30.0:
        p_start = _resample(best_arr, n_work)
        try:
            p_cd = _coord_descent_v2(p_start, cd_end, consider=consider, seed=int(seed + 7777))
            if p_cd is not None and p_cd.size > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                consider(p_cd)
        except Exception:
            pass

    # Phase 3: Second LSE pass from best
    rem = hard - 50.0 - time.monotonic()
    if rem > 50.0:
        p_for_lse = _resample(best_arr, n_work)
        lse_end2 = min(hard - 50.0, time.monotonic() + rem * 0.6)
        if lse_end2 > time.monotonic() + 20.0:
            try:
                p3 = _lse_adam(p_for_lse, lse_end2, consider=consider, T_start=0.02, T_end=0.0003)
                if p3 is not None:
                    consider(p3)
            except Exception:
                pass

    # Phase 4: Final CD squeeze
    rem = hard - 10.0 - time.monotonic()
    if rem > 20.0:
        p_final = _resample(best_arr, n_work)
        try:
            p_f = _coord_descent_v2(p_final, hard - 10.0, consider=consider, seed=int(seed + 9999))
            if p_f is not None and p_f.size > 0 and np.all(np.isfinite(p_f)) and p_f.sum() > 0:
                consider(p_f)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
```
