import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        h = np.ones(3072, dtype=np.float64)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1000.0)
    if h.size == 0 or h.sum() < 1e-12:
        h = np.ones(3072, dtype=np.float64)
    return h


def _score_norm(a, n, conv_size):
    """Score assuming sum(a)=1: 2*n*max(a*a)"""
    F = rfft(a, conv_size)
    c = irfft(F * F, conv_size)[:2 * n - 1]
    m = float(np.max(np.maximum(c, 0.0)))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m


def _score_exact_list(seq):
    try:
        return float(evaluate_sequence([float(x) for x in seq]))
    except Exception:
        return math.inf


def _adam_step(a, g, m_s, v_s, t_s, lr, b1=0.9, b2=0.999, eps=1e-8):
    m_s = b1 * m_s + (1 - b1) * g
    v_s = b2 * v_s + (1 - b2) * g * g
    m_hat = m_s / (1 - b1 ** t_s)
    v_hat = v_s / (1 - b2 ** t_s)
    a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-300:
        a = a / s
    else:
        a = np.full(a.size, 1.0 / a.size)
    return a, m_s, v_s, t_s + 1


def _optimize_lse_adam(a, n, conv_size, cross_size, T_frac, n_iter, lr_scale, stop_t, seed):
    """LSE-Adam optimization at a fixed temperature fraction."""
    T = T_frac * (_score_norm(a, n, conv_size) / (2.0 * n))
    if T < 1e-20:
        T = 1e-20
    lr = lr_scale * T
    m_s = np.zeros(n)
    v_s = np.zeros(n)
    t_s = 0
    for _ in range(n_iter):
        if time.monotonic() > stop_t:
            break
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        cmax = float(c.max())
        if cmax < 1e-300:
            break
        w = np.exp((c - cmax) / T)
        ws = w.sum()
        if ws < 1e-300 or not math.isfinite(ws):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= ws
        # Gradient: g_i = 2 * sum_k w_k * a[k-i] (valid k-i)
        # Cross-correlation via FFT
        Fw = rfft(w, cross_size)
        Fa = rfft(a, cross_size)
        cc = irfft(Fw * np.conj(Fa), cross_size)
        g = 2.0 * cc[:n]
        g = g - g.mean()  # project to sum constraint
        if not np.all(np.isfinite(g)):
            break
        a, m_s, v_s, t_s = _adam_step(a, g, m_s, v_s, t_s, lr)
    return a


def _coord_descent(a, n, stop_t, seed, check_fn=None):
    """Coordinate descent to polish the solution."""
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-300:
        a = np.full(n, 1.0 / n)
        s = 1.0
    a = a / s
    Ltot = 2 * n - 1
    conv_s = next_fast_len(Ltot)

    def conv_self(x):
        F = rfft(x, conv_s)
        c = irfft(F * F, conv_s)[:Ltot]
        return np.maximum(c, 0.0)

    c = conv_self(a)
    cur_sc = 2.0 * n * float(c.max())
    best_sc = cur_sc
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    G = max(20, min(60, int(80000 // max(n, 1))))
    flat = 0
    sweep = 0

    while time.monotonic() < stop_t and flat < 6:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            if time.monotonic() > stop_t:
                break
            i = int(idx_i)
            v = float(a[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * a[:Ls]
            max_const = 0.0
            if start > 0:
                max_const = max(max_const, float(np.max(c[:start])))
            if end < Ltot:
                max_const = max(max_const, float(np.max(c[end:])))

            x_hi = max(40.0 * v, 100.0 / n, 0.01)
            x_hi = min(x_hi, 4.0)
            pts = np.linspace(0.0, x_hi, G + 1)
            if 0 <= v <= x_hi:
                pts = np.unique(np.concatenate([pts, [v]]))

            def ev(xs):
                w = xs - v
                aff = base[None, :] + w[:, None] * infl[None, :]
                dv = base[i - start] + w * infl[i - start] + w * w
                aff = np.maximum(aff, dv[:, None])
                aff = np.maximum(aff, max_const)
                mv = np.max(aff, axis=1)
                denom = s + w
                out = np.full(xs.shape, 1e100)
                valid = denom > 1e-14
                out[valid] = (2.0 * n * mv[valid]) / (denom[valid] ** 2)
                return out

            vals = ev(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            bx = float(pts[imin])
            bv = float(vals[imin])

            dw = x_hi / G
            lo = max(0.0, bx - 3.0 * dw)
            hi = min(x_hi, bx + 3.0 * dw)
            if hi > lo:
                fine = np.linspace(lo, hi, 50)
                if 0 <= bx <= hi:
                    fine = np.unique(np.concatenate([fine, [bx, v]]))
                v2 = ev(fine)
                if np.any(np.isfinite(v2)):
                    i2 = int(np.argmin(v2))
                    if v2[i2] < bv:
                        bv = float(v2[i2])
                        bx = float(fine[i2])

            if math.isfinite(bv) and bv < cur_sc - 1e-14 and 0 <= bx:
                w = bx - v
                if abs(w) > 1e-15:
                    a[i] = bx
                    s += w
                    c[start:end] += w * infl
                    diag = i - start
                    c[start + diag] += w * w
                    cur_sc = bv
                    improved = True

        # Periodic recompute
        if sweep % 5 == 4:
            c = conv_self(a)
            cur_sc = 2.0 * n * float(c.max())
        if math.isfinite(cur_sc) and cur_sc < best_sc - 1e-14:
            best_sc = cur_sc
            best_a = a.copy()
            if check_fn:
                try:
                    check_fn(best_a)
                except Exception:
                    pass
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3 and time.monotonic() < stop_t - 30:
                a = a * np.exp(rng.normal(0, 0.015, n))
                a = np.maximum(a, 0)
                aa = a.sum()
                if aa > 0:
                    a /= aa
                c = conv_self(a)
                cur_sc = 2.0 * n * float(c.max())
                flat = 0
        sweep += 1

    if best_sc > cur_sc:
        best_sc = cur_sc
        best_a = a.copy()
    return best_a, best_sc


def _lse_lbfgs(a, n, conv_size, cross_size, T_frac, stop_t, max_calls=200):
    """L-BFGS-B on LSE surrogate."""
    T = T_frac * (_score_norm(a, n, conv_size) / (2.0 * n))

    def lse_obj(u):
        pp = np.exp(u - u.max())
        ss = pp.sum()
        if ss < 1e-300:
            return 1e100, np.zeros_like(u)
        pp = pp / ss
        F = rfft(pp, conv_size)
        c = irfft(F * F, conv_size)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        cmax = float(c.max())
        if cmax < 1e-30:
            return 1e100, np.zeros_like(u)
        w = np.exp((c - cmax) / T)
        ws = w.sum()
        if ws < 1e-300:
            return cmax, np.zeros_like(u)
        w /= ws
        # objective
        fval = cmax + T * float(np.log(ws / (2 * n - 1)))
        # gradient
        Fw = rfft(w, cross_size)
        Fp = rfft(pp, cross_size)
        cc = irfft(Fw * np.conj(Fp), cross_size)
        gpp = 2.0 * cc[:n]
        gpp = gpp - gpp.mean()
        grad = pp * (gpp - np.dot(pp, gpp) * 0)
        # Actually gradient of LSE w.r.t. log-params:
        # dLSE/dpp_i = 2*sum_k w_k*a_{k-i} (the gpp above, but w.r.t. pp)
        # dLSE/du_i = pp_i * (gpp_i - mean-adjustment for the simplex)
        grad = pp * gpp
        grad = grad - pp * np.dot(pp, gpp)  # project off the simplex
        return fval, np.where(np.isfinite(grad), grad, 0.0)

    u = np.log(np.maximum(a, 1e-20))
    u = u - u.mean()
    opts = {"maxiter": 30, "gtol": 1e-10, "maxls": 20, "maxcor": 20, "disp": False}
    call_count = [0]
    orig_obj = lse_obj

    def obj_with_count(u):
        if time.monotonic() > stop_t:
            raise Exception("timeout")
        call_count[0] += 1
        if call_count[0] > max_calls:
            raise Exception("max calls")
        return orig_obj(u)

    try:
        res = minimize(obj_with_count, u, method="L-BFGS-B", jac=True, options=opts)
        if res.x is not None and np.all(np.isfinite(res.x)):
            pp = np.exp(res.x - res.x.max())
            pp /= pp.sum()
            return pp
    except Exception:
        pass
    return a
