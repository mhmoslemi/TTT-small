def _build_conv_mat(n, a):
    """Build (2n-1) x n convolution matrix where A[k,j] = a[k-j] if valid."""
    idx = np.arange(2 * n - 1)[:, None] - np.arange(n)[None, :]
    mask = (idx >= 0) & (idx < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[idx[mask]]
    return A


def _lp_best_response(a, M, tl=5.0):
    """Solve LP: max sum(g) s.t. a*g <= M, g >= 0."""
    n = len(a)
    A = _build_conv_mat(n, a)
    c_obj = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tl,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0 and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None


def _line_search_score(a, direction, n_trials=40):
    """Line search along a + t*(direction - a) for t in [0,1], normalize after."""
    n = len(a)
    best_t, best_v = 0.0, float('inf')
    ts = np.linspace(0, 1, n_trials)
    for t in ts:
        cand = (1 - t) * a + t * direction
        cand = np.maximum(cand, 0)
        cs = cand.sum()
        if cs < 1e-15:
            continue
        cand = cand / cs
        v = float(np.convolve(cand, cand).max())
        if v < best_v:
            best_v, best_t = v, t
    # Refine
    if best_t > 0 and best_t < 1:
        lo, hi = max(0, best_t - 1.0 / n_trials), min(1, best_t + 1.0 / n_trials)
        for _ in range(40):
            if hi - lo < 1e-9:
                break
            mid = 0.5 * (lo + hi)
            cand = (1 - mid) * a + mid * direction
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                lo = mid
                continue
            cand = cand / cs
            v = float(np.convolve(cand, cand).max())
            if v < best_v:
                best_v, best_t = v, mid
                hi = mid
            else:
                lo = mid
    cand = (1 - best_t) * a + best_t * direction
    cand = np.maximum(cand, 0)
    cs = cand.sum()
    if cs < 1e-15:
        return None, best_v
    return cand / cs, best_v

            stuck = 0

def _gradient_descent(a0, deadline, lr_schedule=None):
    """Projected gradient descent with subgradient of max convolution."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_cmax = float(np.convolve(a, a).max())
    rng = np.random.default_rng(123)
    lr = 0.02
    stuck = 0
    it = 0

    while time.monotonic() < deadline and it < 50000:
        it += 1
        c = np.convolve(a, a)
        M = float(c.max())

        if M < best_cmax * (1 - 1e-13):
            best_cmax = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 500:
            lr = max(lr * 0.95, 0.001)
            if stuck > 2000:
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                stuck = 0
            continue

        # Subgradient: for each active lag k, add 2*shifted(a)
        active = np.where(c >= M * (1 - 1e-9))[0]
        grad = np.zeros(n)
        for k_s in active:
            k = int(k_s)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * a[idx]
        # Average over active set
        grad /= max(len(active), 1)
        # Project gradient to sum-zero direction (preserve normalization)
        grad = grad - grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            a = a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            continue

        # Try a few step sizes
        moved = False
        for alpha in [lr, lr * 0.5, lr * 0.1, lr * 2]:
            cand = a - alpha * grad / gn
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            v = float(np.convolve(cand, cand).max())
            if v < M * (1 - 1e-12):
                a = cand
                moved = True
                break
        if not moved:
            # Small random step
            a = a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            lr = max(lr * 0.99, 0.001)

    return best_a


def _try_length_change(parent, nt, deadline, time_per=15.0):
    """Interpolate parent to new length and optimize briefly."""
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, nt)
    init = np.interp(x_new, x_old, parent)
    init = np.clip(init, 0, 1000)
    sn = init.sum()
    if sn < 1e-10:
        return None
    init = init / sn
    alloc = time.monotonic() + time_per
    result = _refine_smooth(init, min(alloc, deadline))
    return result

        ab = ab / max(ab.sum(), 1e-15)
        # Short smooth minimize
        try:
            r = _refine_smooth(ab, deadline - 5, max_total_iter=1000)
            try_update(r)
        except Exception:
            pass
        # Short LP
        if time.monotonic() < deadline - 5:
            ab = np.clip(best_seq, 0, 1000).astype(np.float64)
            ab = ab / max(ab.sum(), 1e-15)
            try:
                r = _lp_refine(ab, deadline - 5, max_iters=50)
                try_update(r)
            except Exception:
                pass

    # ===== Final validation =====
    try:
        fs = float(evaluate_sequence([float(v) for v in best_seq]))
        if not np.isfinite(fs):
            raise ValueError("invalid")
    except Exception:
        return [1.0] * 100

    return [float(v) for v in best_seq]
