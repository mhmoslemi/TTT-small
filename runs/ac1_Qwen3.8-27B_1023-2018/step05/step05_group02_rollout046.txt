def _conv_self(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fft(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))

def _score_exact(arr):
    a = np.asarray(arr, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)

    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)

        return math.inf
    return float(2.0 * a.size * m / (s * s))

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:
        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(3072, dtype=np.float64)
    return p


def _conv_fft(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_fft(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _score_exact(arr):
    a = np.asarray(arr, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if a.size == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * a.size * m / (s * s))


def _softmax_u(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _p_to_u(p):
    uu = np.log(np.maximum(p, 1e-300))
    return uu - float(np.mean(uu))


def _lse_optimize(p, stop_time, T_list, seed=0):
    """L-BFGS optimization on LSE surrogates with sequential temperatures."""
    n = int(p.size)
    p = np.maximum(p, 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    conv_len = 2 * n - 1
    cross_len = 3 * n - 2
    conv_size = next_fast_len(conv_len)
    cross_size = next_fast_len(cross_len)

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax_u(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0.0:
            return 1e100, np.zeros_like(uu)
        T = max(T, 1e-300)
        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -500.0, 500.0)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if sum_exp <= 0.0 or not math.isfinite(sum_exp):
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * math.log(sum_exp / c.size)
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    u = _p_to_u(p)
    best_u = u.copy()
    best_f, _ = lse_func(u, T_list[0])

    for T in T_list:
        if time.monotonic() >= stop_time - 2.0:
            break
        u = best_u.copy()
        stale = 0
        max_inner = 60
        while time.monotonic() < stop_time - 2.0 and stale < 4:
            try:
                res = minimize(
                    lse_func, u, args=(T,), method="L-BFGS-B", jac=True,
                    options={"maxiter": max_inner, "gtol": 1e-14, "ftol": 1e-18,
                             "maxls": 40, "maxcor": 40, "disp": False}
                )
            except Exception:
                break
            if res.x is not None and res.x.size == u.size:
                x = np.asarray(res.x, dtype=np.float64)
                if np.all(np.isfinite(x)):
                    fv = float(res.fun)
                    if math.isfinite(fv) and fv < best_f - 1e-15:
                        best_f = fv
                        best_u = x.copy()
                        u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        break
            else:
                break

    p_out = _softmax_u(best_u)
    p_out = np.maximum(p_out, 0.0)
    ss = float(p_out.sum())
    if ss > 0:
        p_out /= ss
    return p_out


def _coord_descent(p, stop_time, seed=42, consider=None, max_sweeps=999999):
    """Coordinate descent with incremental convolution update."""
    n = int(p.size)
    p = np.maximum(p, 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    c = _conv_fft(p)
    S = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    Ltot = 2 * n - 1
    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S > 1.25 or S < 0.75:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0
            Sv = float(S)
            vi = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls].copy()

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            # Analytical candidates for w
            candidates = [0.0]
            # Diagonal optimum
            di = 2 * i
            if 0 <= di < Ltot and di - start >= 0 and di - start < Ls:
                di_local = di - start
                aii = vi
                if abs(Sv - aii) > 1e-12:
                    wd = (c[di] - aii * Sv) / (Sv - aii)
                    if -aii * 0.999 < wd < 10.0 * Sv:
                        candidates.append(wd)
            # Most critical linear lag optimum
            if Ls > 0:
                crit_idx = int(np.argmax(infl))
                if infl[crit_idx] > 1e-15:
                    wl = Sv - base[crit_idx] / (0.5 * infl[crit_idx])
                    if -vi * 0.999 < wl < 10.0 * Sv:
                        candidates.append(wl)
            # Boundaries
            candidates.append(-vi * 0.999)
            x_hi = min(5.0 * Sv / n * 5.0, 10.0 * Sv)
            candidates.append(x_hi)

            # Evaluate candidates and pick best region
            def eval_w(w):
                denom = Sv + w
                if denom < 1e-15:
                    return math.inf
                aff = base + w * infl
                if di >= 0 and di < Ltot:
                    di_local = di - start
                    if 0 <= di_local < Ls:
                        aff[di_local] += w * w
                maff = float(np.max(aff))
                if max_const > maff:
                    maff = max_const
                return (2.0 * n * maff) / (denom * denom)

            best_cand_w = 0.0
            best_cand_v = eval_w(0.0)
            for w in candidates:
                if w < -vi or w > 10.0 * Sv:
                    continue
                v = eval_w(w)
                if math.isfinite(v) and v < best_cand_v:
                    best_cand_v = v
                    best_cand_w = w

            # Golden section around best candidate
            g_lo = max(-vi, best_cand_w - 3.0 * Sv / n)
            g_hi = min(10.0 * Sv, best_cand_w + 3.0 * Sv / n)
            if g_hi - g_lo > 1e-14:
                gr = (math.sqrt(5.0) - 1.0) / 2.0
                a_gs = g_hi - gr * (g_hi - g_lo)
                b_gs = g_lo + gr * (g_hi - g_lo)
                fa = eval_w(a_gs)
                fb = eval_w(b_gs)
                for _ in range(28):
                    if not math.isfinite(fa):
                        fa = 1e100
                    if not math.isfinite(fb):
                        fb = 1e100
                    if fa < fb:
                        g_hi = b_gs
                        b_gs = a_gs
                        fb = fa
                        a_gs = g_hi - gr * (g_hi - g_lo)
                        fa = eval_w(a_gs)
                    else:
                        g_lo = a_gs
                        a_gs = b_gs
                        fa = fb
                        b_gs = g_lo + gr * (g_hi - g_lo)
                        fb = eval_w(b_gs)
                    if g_hi - g_lo < 1e-15:
                        break
                gs_w = (g_lo + g_hi) / 2.0
                gs_v = eval_w(gs_w)
                if math.isfinite(gs_v) and gs_v < best_cand_v:
                    best_cand_v = gs_v
                    best_cand_w = gs_w

            if math.isfinite(best_cand_v) and best_cand_v < cur_score - 1e-14:
                w = best_cand_w
                if abs(w) > 1e-15:
                    p[i] = vi + w
                    S += w
                    c[start:end] += w * infl
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_cand_v
                    improved = True

        if S > 1e-300:
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        S = 1.0
        # Re-verify with full convolution periodically
        if sweep % 20 == 0:
            c = _conv_fft(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 4.0:
                    noise_amt = 0.02 + 0.01 * math.exp(-flat * 0.08)
                    p = p * np.exp(rng.normal(0.0, noise_amt, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                        S = 1.0
                    c = _conv_fft(p)
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score:
                        best_score = cur_score
                        best_p = p.copy()
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    sc = float(2.0 * n * _conv_fft(p).max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    t_end = t0 + budget - 25.0

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = _score_exact(best_arr)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        if float(a.sum()) < 0.01:
            return
        try:
            sc = _score_exact(a)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / float(a.sum())
            best_arr = a.copy()
            best_score = sc

    n = int(best_arr.size)
    p0 = best_arr / float(best_arr.sum())
    consider(p0)

    # Phase 1: LSE optimization from parent (coarse to fine)
    if time.monotonic() < t_end - 60.0:
        T_coarse = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
        stop1 = min(t_end - 120.0, t0 + 0.18 * (t_end - t0))
        try:
            p1 = _lse_optimize(p0, stop1, T_coarse, seed=seed)
            if p1 is not None and p1.size == n and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 /= float(p1.sum())
            else:
                p1 = p0.copy()
        except Exception:
            p1 = p0.copy()
        consider(p1)

    # Phase 2: Coordinate descent (main workhorse)
    now = time.monotonic()
    cd_stop = min(t_end - 80.0, now + 0.45 * (t_end - t0))
    if cd_stop > now + 30.0:
        try:
            p2, _ = _coord_descent(p1, cd_stop, seed=seed + 1, consider=consider)
            if p2 is not None and p2.size == n and p2.sum() > 0:
                p2 /= float(p2.sum())
            else:
                p2 = p1.copy()
        except Exception:
            p2 = p1.copy()
        consider(p2)

    # Phase 3: LSE fine-tune from CD result
    now = time.monotonic()
    if now < t_end - 80.0:
        T_fine = [0.001, 0.0005, 0.0002, 0.0001]
        stop2 = min(t_end - 60.0, now + 0.20 * (t_end - t0))
        if stop2 > now + 20.0:
            try:
                p3 = _lse_optimize(p2, stop2, T_fine, seed=seed + 100)
                if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and p3.sum() > 0:
                    p3 /= float(p3.sum())
                else:
                    p3 = p2.copy()
            except Exception:
                p3 = p2.copy()
            consider(p3)
        else:
            p3 = p2.copy()
    else:
        p3 = p2.copy()

    # Phase 4: Final coordinate descent
    now = time.monotonic()
    if now < t_end - 15.0:
        cd_stop2 = min(t_end - 8.0, now + 0.25 * (t_end - t0))
        if cd_stop2 > now + 15.0:
            try:
                p4, _ = _coord_descent(p3, cd_stop2, seed=seed + 500, consider=consider)
                if p4 is not None and p4.size == n and p4.sum() > 0:
                    p4 /= float(p4.sum())
                else:
                    p4 = p3.copy()
            except Exception:
                p4 = p3.copy()
            consider(p4)
        else:
            p4 = p3.copy()
    else:

    # Phase 5: One more quick CD pass if time allows
    now = time.monotonic()
    if now < t_end - 10.0:
        cd_stop3 = min(t_end - 5.0, now + 0.10 * (t_end - t0))
        if cd_stop3 > now + 8.0:

                if p5 is not None and p5.size == n and p5.sum() > 0:

                    p5 = p4.copy()
            except Exception:
                p5 = p4.copy()

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()

            final_sc = evaluate_sequence(ls)
            if not math.isfinite(final_sc):

        return ls
    except Exception:
