            if Ls <= 0:
                continue

            base = b[start_k:end_k].copy()
            infl = 2.0 * p

            m_unaff = 0.0
            if start_k > 0:
                m_unaff = max(m_unaff, float(np.max(b[:start_k])))
            if end_k < 2 * n - 1:
                m_unaff = max(m_unaff, float(np.max(b[end_k:])))

            x_hi = max(20.0 * v, 40.0 / n, 0.01)
            x_hi = min(x_hi, 5.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = 1.0 / n

            # Fast 1D search: evaluate objective on grid
            n_coarse = 30
            xs = np.linspace(0.0, x_hi, n_coarse)
            v_off = v - xs  # w = x - v, so w = -v_off... let's use w = x - v
            # Actually: w = x - v, x in [0, x_hi], so w in [-v, x_hi - v]
            ws = xs - v  # shape (n_coarse,)

            # Vectorized: for each w, compute max(base + w*infl) and diag correction
            # base + w[:,None]*infl[None,:] → shape (n_coarse, Ls)
            aff = base[None, :] + ws[:, None] * infl[None, :]
            # Diagonal correction: at position i (within the slice), add w²
            diag_idx = i - start_k
            if 0 <= diag_idx < Ls:
                aff[:, diag_idx] += ws * ws
            # Combine with m_unaff
            aff_max = np.maximum(aff, m_unaff)
            row_max = np.max(aff_max, axis=1)  # shape (n_coarse,)
            denoms = S + ws  # shape (n_coarse,)
            valid = denoms > 1e-14
            if not np.any(valid):
                continue
            scores = np.full(ws.shape, math.inf)
            scores[valid] = 2.0 * n * row_max[valid] / (denoms[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            bx = float(xs[imin])
            bv = float(scores[imin])

            # Fine search around best
            dw = x_hi / n_coarse
            lo = max(0.0, bx - 4.0 * dw)
            hi = min(x_hi, bx + 4.0 * dw)
            if hi > lo:
                fine = np.linspace(lo, hi, 40)
                ws_f = fine - v
                aff_f = base[None, :] + ws_f[:, None] * infl[None, :]
                if 0 <= diag_idx < Ls:
                    aff_f[:, diag_idx] += ws_f * ws_f
                aff_f_max = np.maximum(aff_f, m_unaff)
                row_max_f = np.max(aff_f_max, axis=1)
                denoms_f = S + ws_f
                valid_f = denoms_f > 1e-14
                if np.any(valid_f):
                    scores_f = np.full(ws_f.shape, math.inf)
                    scores_f[valid_f] = 2.0 * n * row_max_f[valid_f] / (denoms_f[valid_f] ** 2)
                    if np.any(np.isfinite(scores_f)):
                        im2 = int(np.argmin(scores_f))
                        if float(scores_f[im2]) < bv:
                            bv = float(scores_f[im2])
                            bx = float(fine[im2])

            if math.isfinite(bv) and bv < cur_score - 1e-14 and 0.0 <= bx <= 10.0 * S:
                w = bx - v
                if abs(w) > 1e-15:
                    p[i] = bx
                    S += w
                    b[start_k:end_k] += w * infl
                    if 0 <= diag_idx < Ls:
                        b[start_k + diag_idx] += w * w
                    cur_score = max(bv, 2.0 * n * m_unaff / (S * S))
                    improved = True

        # Rescale
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            b *= inv * inv
            S = 1.0
        else:
            break

        # Verify periodically
        if sweep % 5 == 0 or improved:
            b = _conv_self(p)
            new_sc = 2.0 * n * float(b.max())
            if math.isfinite(new_sc) and new_sc < best_score - 1e-14:
                best_score = new_sc
                best_p = p.copy()
                if consider is not None:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 20.0:
                    perturbed = p * np.exp(rng.normal(0, 0.015, n))
                    perturbed = np.maximum(perturbed, 0)
                    ps = float(perturbed.sum())
                    if ps > 1e-300:
                        perturbed /= ps
                    b_new = _conv_self(perturbed)
                    sc_new = 2.0 * n * float(b_new.max())
                    if math.isfinite(sc_new):
                        p = perturbed
                        b = b_new
                        S = 1.0
                        cur_score = sc_new
                else:
                    break
                flat = 0

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    sc_final = _score_fft(p)
    if math.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_p = p.copy()
    return best_p, best_score


def _peak_shave(p, stop_time, consider=None, seed=99):
    """Targeted search: reduce the peak bins specifically."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p /= s
    b = _conv_self(p)
    best_p = p.copy()
    best_sc = 2.0 * n * float(b.max())
    rng = np.random.default_rng(seed)

    for it in range(200):
        if time.monotonic() > stop_time - 3.0:
            break
        # Find top-K peak bins
        k_peak = int(np.argmax(b))
        # The peak b[k] = sum_i a[i]*a[k-i]. Try to reduce it.
        # Strategy: find the i that maximizes a[i]*a[k-i] and try to reduce that pair.
        k = k_peak
        # Valid i range for this k: max(0, k-n+1) to min(k, n-1)
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo > i_hi:
            break

        # Try small reductions in a[i] for the top contributors
        prods = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            other = k - i
            if 0 <= other < n:
                prods[i] = p[i] * p[other]
        top_indices = [i for i in range(i_lo, i_hi + 1) if prods[i] > 0]
        if not top_indices:
            break
        top_indices.sort(key=lambda x: -prods[x])
        target_i = top_indices[0]

        # Try reducing p[target_i] slightly
        delta = 0.005 * p[target_i]
        p_trial = p.copy()
        p_trial[target_i] -= delta
        if p_trial[target_i] < 0:
            p_trial[target_i] = 0.0
        st = float(p_trial.sum())
        if st < 0.01:
            continue
        b_trial = _conv_self(p_trial)
        sc_trial = 2.0 * n * float(b_trial.max()) / (st * st)
        sc_cur = 2.0 * n * float(b.max()) / (s * s)
        if math.isfinite(sc_trial) and sc_trial < sc_cur - 1e-15:
            p = p_trial
            b = b_trial
            s = st
            sc_new = 2.0 * n * float(b.max())
            if sc_new / (s * s) < best_sc / (float(best_p.sum()) ** 2):
                best_p = p.copy()
                best_sc = sc_new
                if consider is not None:
                    try:
                        consider(p)
                    except Exception:
                        pass
        else:
            # Try a tiny random perturbation
            idx = rng.integers(0, n)
            p_alt = p.copy()
            p_alt[idx] *= (1.0 + rng.normal(0, 0.01))
            p_alt = np.maximum(p_alt, 0)
            sa = float(p_alt.sum())
            if sa > 0.01:
                b_alt = _conv_self(p_alt)
                sc_alt = 2.0 * n * float(b_alt.max()) / (sa * sa)
                sc_cur2 = 2.0 * n * float(b.max()) / (s * s)
                if math.isfinite(sc_alt) and sc_alt < sc_cur2:
                    p = p_alt
                    b = b_alt
                    s = sa

    p_out = np.maximum(p, 0.0)
    ss = float(p_out.sum())
    if ss > 1e-300:
        p_out /= ss
    return p_out


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = math.inf
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_arr]))
    except Exception:
        pass
    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(128)
        best_score = float(evaluate_sequence([float(x) for x in best_arr]))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            sc = float(evaluate_sequence([float(x) for x in a]))
            if math.isfinite(sc) and sc < best_score - 1e-15:
                s = float(a.sum())
                if s > 0.01:
                    best_arr = (a / s).copy()
                    best_score = sc
        except Exception:
            pass

    n_best = int(best_arr.size)

    # Phase 1: LSE optimization (~200s)
    lse_stop = t0 + min(200.0, 0.18 * budget)
    if lse_stop > time.monotonic() + 30.0:
        try:
            p1 = _lse_optimize(best_arr, lse_stop, consider=consider)
            if p1 is not None and p1.size == n_best and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
                best_arr = p1.copy()
        except Exception:
            pass

    # Phase 2: Main coordinate descent (~600s)
    coord_stop = t0 + min(850.0, 0.65 * budget)
    coord_stop = max(coord_stop, time.monotonic() + 60.0)
    coord_stop = min(coord_stop, hard - 60.0)
    if coord_stop > time.monotonic() + 40.0:
        try:
            p2, _ = _coord_descent(best_arr, coord_stop, consider=consider,
                                   seed=int(seed), max_sweeps=800)
            if p2 is not None and p2.size == n_best and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p2 = p2 / float(p2.sum())
                consider(p2)
                best_arr = p2.copy()
        except Exception:
            pass

    # Phase 3: LSE polish (~200s)
    rem3 = hard - 50.0 - time.monotonic()
    if rem3 > 60.0:
        lse2_stop = time.monotonic() + min(180.0, 0.5 * rem3)
        try:
            p3 = _lse_optimize(best_arr, lse2_stop, consider=consider)
            if p3 is not None and p3.size == n_best and np.all(np.isfinite(p3)) and p3.sum() > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)
                best_arr = p3.copy()
        except Exception:
            pass

    # Phase 4: Final coordinate descent (~150s)
    coord2_stop = hard - 30.0
    if time.monotonic() < coord2_stop - 40.0:
        try:
            p4, _ = _coord_descent(best_arr, coord2_stop, consider=consider,
                                   seed=int(seed + 777), max_sweeps=500)
            if p4 is not None and p4.size == n_best and np.all(np.isfinite(p4)) and p4.sum() > 0:
                p4 = p4 / float(p4.sum())
                consider(p4)
                best_arr = p4.copy()
        except Exception:
            pass

    # Peak shaving with remaining time
    shave_stop = hard - 15.0
    if time.monotonic() < shave_stop - 20.0:
        try:
            p5 = _peak_shave(best_arr, shave_stop, consider=consider, seed=int(seed + 31337))
            if p5 is not None and p5.size == n_best and np.all(np.isfinite(p5)) and p5.sum() > 0:
                p5 = p5 / float(p5.sum())
                consider(p5)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
