            denom_v = denom[valid]

            # Evaluate: new_c[j] = base[j] + w*infl[j], add w^2 at diag
            new_vals = base[None, :] + wv[:, None] * infl[None, :]
            new_vals[:, diag_pos] += wv[:, None] * wv[:, None]
            new_max_region = new_vals.max(axis=1)
            new_max = np.maximum(new_max_region, max_const)

            vals = 2.0 * n * new_max / (denom_v * denom_v)
            vals = np.where(np.isfinite(vals), vals, 1e100)

            imin = int(np.argmin(vals))
            best_val = float(vals[imin])
            best_w = float(wv[imin])

            # Fine refinement around best_w
            if best_val < cur_score - 1e-14 and abs(best_w) > 1e-15:
                span = max(0.1 * S, abs(best_w) * 0.5, v * 0.3)
                fine = np.linspace(max(-v + 1e-15, best_w - span),
                                   min(20 * S, best_w + span), 8)
                if best_w > -v and best_w < 20 * S:
                    fine = np.insert(fine, int(np.argmin(np.abs(fine - best_w))), best_w)
                fdenom = S + fine
                fvalid = fdenom > 1e-12
                if np.any(fvalid):
                    fv = fine[fvalid]
                    fd = fdenom[fvalid]
                    fnv = base[None, :] + fv[:, None] * infl[None, :]
                    fnv[:, diag_pos] += fv[:, None] * fv[:, None]
                    fnmax = np.maximum(fnv.max(axis=1), max_const)
                    fvals = 2.0 * n * fnmax / (fd * fd)
                    fvals = np.where(np.isfinite(fvals), fvals, 1e100)
                    fimin = int(np.argmin(fvals))
                    if float(fvals[fimin]) < best_val:
                        best_val = float(fvals[fimin])
                        best_w = float(fv[fimin])

            if math.isfinite(best_val) and best_val < cur_score - 1e-14:
                w = best_w
                p[i] = v + w
                S += w
                c[start:end] += w * infl
                c[2 * i] += w * w
                cur_score = best_val
                improved = True

        # Renormalize
        if S > 1e-300:
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0

        real_score = float(2.0 * n * c.max())
        if real_score < best_score - 1e-14:
            best_score = real_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() > stop_time - 3.0:
                    break
                sigma = 0.015 + 0.01 * min(flat, 8)
                p = p * np.exp(rng.normal(0.0, sigma, n))
                p = np.maximum(p, 0.0)
                ss = float(p.sum())
                if ss > 1e-300:
                    p /= ss
                    S = 1.0
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    if cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                else:
                    break
                flat = 0

        sweep += 1

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 15.0

    parent = _get_parent()
    n = int(parent.size)
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    else:
        best_arr = np.ones(n) / n
        ss = 1.0

    best_score = float(_score_fast(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0 or float(a.sum()) < 0.01:
                return
            sc = float(_score_exact(a))
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_arr = (a / float(a.sum())).copy()
                best_score = sc
        except Exception:
            pass

    # Phase 1: Brief LSE for global shape (30% of time)
    lse_end = t0 + 0.25 * (hard - t0)
    if lse_end > t0 + 20.0:
        try:
            p1 = _lse_optimize(best_arr, lse_end, seed=int(seed + 100), consider=consider)
            if p1 is not None and p1.size == n and float(p1.sum()) > 0:
                p1 = p1 / float(p1.sum())
                sc1 = _score_fast(p1)
                if math.isfinite(sc1) and sc1 < best_score:
                    best_arr = p1.copy()
                    best_score = sc1
        except Exception:
            pass

    # Phase 2: Intensive coordinate descent (50% of time)
    cd_end = t0 + 0.75 * (hard - t0)
    if cd_end > time.monotonic() + 30.0:
        c1 = _conv_self(best_arr)
        try:
            p2, sc_cd = _coord_descend(best_arr, c1, cd_end, seed=int(seed + 200),
                                       consider=consider)
            if p2 is not None and p2.size == n and float(p2.sum()) > 0:
                p2 = p2 / float(p2.sum())
                sc2 = _score_fast(p2)
                if math.isfinite(sc2) and sc2 < best_score:
                    best_arr = p2.copy()
                    best_score = sc2
        except Exception:
            pass

    # Phase 3: Perturb and re-optimize
    if time.monotonic() < hard - 60.0:
        rng3 = np.random.default_rng(int(seed + 999))
        for pr in range(3):
            if time.monotonic() > hard - 40.0:
                break
            pert = best_arr * np.exp(rng3.normal(0, 0.03, n))
            pert = np.maximum(pert, 0.0)
            ps = float(pert.sum())
            if ps > 0:
                pert /= ps
            sub_end = min(hard - 30.0, time.monotonic() + 0.2 * (hard - time.monotonic()))
            if sub_end > time.monotonic() + 20.0:
                c_p = _conv_self(pert)
                try:
                    p3, _ = _coord_descend(pert, c_p, sub_end, seed=int(seed + 3000 + pr * 100),
                                           consider=consider)
                    if p3 is not None and p3.size == n and float(p3.sum()) > 0:
                        p3 = p3 / float(p3.sum())
                        sc3 = _score_fast(p3)
                        if math.isfinite(sc3) and sc3 < best_score:
                            best_arr = p3.copy()
                            best_score = sc3
                except Exception:
                    pass

    # Phase 4: Final intensive coordinate polish
    final_end = hard - 5.0
    if final_end > time.monotonic() + 20.0 and best_arr.size > 0:
        c_final = _conv_self(best_arr)
        try:
            p4, _ = _coord_descend(best_arr, c_final, final_end,
                                   seed=int(seed + 123456789), consider=consider)
            if p4 is not None and p4.size == n and float(p4.sum()) > 0:
                p4 = p4 / float(p4.sum())
                sc4 = _score_fast(p4)
                if math.isfinite(sc4) and sc4 < best_score:
                    best_arr = p4.copy()
                    best_score = sc4
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
