```python
def _lse_grad(u, T):
    """Compute LSE value and gradient w.r.t. u (log-prob parameterization)."""
    n = u.size
    a = _softmax(u)  # probability vector, sum=1
    # conv = a * a
    conv_size = next_fast_len(2*n-1)
    F = rfft(a, conv_size)
    conv = irfft(F*F, conv_size)[:2*n-1]
    np.maximum(conv, 0.0, out=conv)
    cmax = float(conv.max())
    if cmax <= 0 or not math.isfinite(cmax):
        return 1e100, np.zeros_like(u)
    # LSE weights
    scaled = (conv - cmax) / T
    exps = np.exp(scaled)
    sum_exp = float(exps.sum())
    if not math.isfinite(sum_exp) or sum_exp <= 0:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w = exps / sum_exp
    # LSE value
    lse_val = cmax + T * math.log(sum_exp / (2*n-1))
    # Gradient: grad_a[i] = 2 * cross_corr(a, w)[i]
    # cross_corr(a, w)[i] = sum_j a[j] * w[i+j]
    # In FFT: ifft(fft(a,N) * conj(fft(w,N)))[:n], N >= n + (2n-1) - 1 = 3n-2
    N = next_fast_len(3*n-2)
    Fa = fft(a, N)
    Fw = fft(w, N)
    cross = np.real(ifft(Fa * np.conj(Fw)))[:n]
    grad_a = 2.0 * cross
    # Transform to u-space: grad_u = a * (grad_a - dot(a, grad_a))
    ta = float(np.dot(a, grad_a))
    grad_u = a * (grad_a - ta)
    grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
    return lse_val, grad_u
```
