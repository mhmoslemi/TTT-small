```python
import cvxpy as cp

def lp_update(f, M):
    n = len(f)
    b = cp.Variable(n)
    # f * b has length 2n-1
    # (f*b)[k] = sum_j f[j] * b[k-j]
    # We need (f*b)[k] <= M for all k
    
    constraints = [b >= 0]
    for k in range(2*n - 1):
        # (f*b)[k] = sum_{j} f[j] * b[k-j] where 0 <= j < n and 0 <= k-j < n
        # So j ranges from max(0, k-n+1) to min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            f_vals = f[indices]
            b_indices = k - indices
            constraints.append(f_vals @ b[b_indices] <= M)
    
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)  # or other solver
    return b.value
```

```
    if sum(b) <= sum(f) * (1 + eps): break  # no improvement possible
    g = b / sum(b) * sum(f)  # rescale to match normalization
    f = (1-t)*f + t*g  # interpolate
    run a few sweeps of coordinate descent on f
```

The interpolation step t is important - too large and we might increase the peak, too small and progress is slow. The paper suggests small t.

Let me also think about the coordinate descent more carefully. The key insight is that for a single coordinate change, the score function is:
score(w) = 2n * max_k (base_k + w * infl_k + (w^2 if k == 2i else 0)) / (S + w)^2

where base_k is the current conv[k] (excluding the contribution from a_i), infl_k = 2*a[k-i] (the linear coefficient), S is the current sum.

This is a piecewise function (the max creates pieces), but within each piece it's a ratio of a quadratic (or linear) to a quadratic, which is unimodal. So the 1D minimization is well-behaved.

The current code uses a grid + golden section. I think we can do better by:
1. Identifying which piece the optimum is in (by finding which constraint is active at the optimum)
2. Directly solving the resulting 1D problem analytically

For a single active constraint k*, the problem is:
minimize (base_k* + w * infl_k* + δ_k*(w^2)) / (S+w)^2
where δ_k* = 1 if k* = 2i, else 0.

If δ = 0: minimize (a + b*w) / (S+w)^2 where a = base_k*, b = infl_k*
Taking derivative: [b(S+w)^2 - (a+bw)*2(S+w)] / (S+w)^4 = 0
=> b(S+w) - 2(a+bw) = 0
=> bS + bw - 2a - 2bw = 0
=> -bw + bS - 2a = 0
=> w = (bS - 2a) / b = S - 2a/b (if b ≠ 0)

If δ = 1: minimize (a + b*w + w^2) / (S+w)^2
Taking derivative: [(b+2w)(S+w)^2 - (a+bw+w^2)*2(S+w)] / (S+w)^4 = 0
=> (b+2w)(S+w) - 2(a+bw+w^2) = 0
=> bS + bw + 2wS + 2w^2 - 2a - 2bw - 2w^2 = 0
=> bS - bw + 2wS - 2a = 0
=> w(2S - b) = 2a - bS
=> w = (2a - bS) / (2S - b) (if 2S - b ≠ 0)

So we can find the exact optimum for each active constraint, then check which constraint is actually the max at that point, and iterate.

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize, linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:
        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(3072, dtype=np.float64)
    return p

def _conv_self(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))

def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * a.size * m / (s * s))

def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0 / u.size)
    return e / s

# LSE-based L-BFGS optimization
def _lbfgs_optimize(p, n, stop_time, consider=None, fine_only=False):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0.0:
            return 1e100, np.zeros_like(uu)
        if T <= 0.0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -400.0, 400.0)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    def run_chunk(uu, T, end_stage):
        try:
            f0, _ = lse_func(uu, T)
        except Exception:
            return uu
        if not math.isfinite(f0):
            return uu
        best_u = np.array(uu, dtype=np.float64, copy=True)
        best_f = float(f0)
        stale = 0
        while time.monotonic() < end_stage - 1.5:
            maxit = 40
            opts = {
                "maxiter": maxit,
                "gtol": 1e-14,
                "ftol": 1e-18,
                "maxls": 30,
                "maxcor": 30,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == uu.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-16:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, "status", 1) == 0:
                break
        return best_u

    if fine_only:
        stages = ((0.0005, 0.30), (0.0001, 0.35), (0.00002, 0.35))
    else:
        stages = (
            (0.08, 0.06),
            (0.03, 0.10),
            (0.01, 0.12),
            (0.003, 0.15),
            (0.001, 0.20),
            (0.0003, 0.20),
            (0.00008, 0.17),
        )

    start = time.monotonic()
    avail = float(stop_time - start - 8.0)
    if avail <= 4.0:
        return p

    best_p = p.copy()
    best_sc = _score(p)
    total_w = float(sum(w for _, w in stages))
    if total_w <= 0.0:
        total_w = 1.0

    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 4.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        sc_cur = _score(p_cur)
        if not math.isfinite(sc_cur):
            p = best_p.copy()
            u = p_to_u(p)
            continue
        if sc_cur < best_sc:
            best_sc = sc_cur
            best_p = p_cur.copy()
            p = p_cur.copy()
        elif sc_cur > best_sc + max(0.002, 2e-4 * best_sc):
            p = best_p.copy()
        else:
            p = p_cur.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p
```

```python
def _get_parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()

def _conv_self(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()

    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())

        return np.full(u.size, 1.0 / u.size)

            x_hi = min(x_hi, 5.0)
            x_hi = max(x_hi, 1e-12)

            # Coarse grid
            npts = G + 1
            pts = np.linspace(0.0, x_hi, npts)

            def eval_scores(pts_arr):
                w = pts_arr.reshape(-1, 1)
                denom = S - v + w
                valid = denom > 1e-15
                result = np.full(w.shape[0], math.inf)
                if not np.any(valid):
                    return result
                wv = w[valid]
                dv = denom[valid]
                aff = base[None, :] + wv * infl[None, :]
                maff = np.max(aff, axis=1)
                if di_local >= 0 and di_local < Ls:
                    dval = base[di_local] + wv * wv
                    maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                result[valid] = (2.0 * n * maff) / (dv * dv)
                return result

            vals = eval_scores(pts)
            if not np.any(np.isfinite(vals)):
                continue

            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Refined search
            if G > 1:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 3.0 * dw)
                hi = min(x_hi, best_x + 3.0 * dw)
                if hi - lo < 1e-15:
                    hi = lo + 1e-12
                fine_pts = np.linspace(lo, hi, 32)
                if 0.0 <= v <= x_hi:
                    fine_pts = np.unique(np.concatenate((fine_pts, [v])))
                vals2 = eval_scores(fine_pts)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine_pts[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2

                # Golden section
                gr = (math.sqrt(5.0) - 1.0) / 2.0
                g_lo = max(0.0, best_x - dw * 1.5)
                g_hi = min(x_hi, best_x + dw * 1.5)
                if g_hi - g_lo > 1e-15:
                    a_gs = g_hi - gr * (g_hi - g_lo)
                    b_gs = g_lo + gr * (g_hi - g_lo)
                    fa = float(eval_scores(np.array([a_gs]))[0])
                    fb = float(eval_scores(np.array([b_gs]))[0])
                    for _ in range(14):
                        if not math.isfinite(fa):
                            fa = 1e100
                        if not math.isfinite(fb):
                            fb = 1e100
                        if fa < fb:
                            g_hi = b_gs
                            b_gs = a_gs
                            fb = fa
                            a_gs = g_hi - gr * (g_hi - g_lo)
                            fa = float(eval_scores(np.array([a_gs]))[0])
                        else:
                            g_lo = a_gs
                            a_gs = b_gs
                            fa = fb
                            b_gs = g_lo + gr * (g_hi - g_lo)
                            fb = float(eval_scores(np.array([b_gs]))[0])
                        if g_hi - g_lo < 1e-15:
                            break
                    gs_x = (g_lo + g_hi) / 2.0
                    gv = float(eval_scores(np.array([gs_x]))[0])
                    if math.isfinite(gv) and gv < best_val:
                        best_x = gs_x
                        best_val = gv

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x < 10.0:
                w = best_x - v
                if abs(w) > 1e-18:
                    p[i] = best_x
                    S_total += w
                    c[start_k:end_k] += w * infl
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        # End of sweep
        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0

        # Recompute from scratch every sweep to avoid drift
        c = _conv_self(p)
        cur_score = float(2.0 * n * c.max())

        if math.isfinite(cur_score) and cur_score < best_score - 1e-16:
            best_score = cur_score
            best_p = p.copy()

        if consider is not None and sweep % 3 == 0:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 6:
                if time.monotonic() < stop_time - 8.0:
                    noise_amt = 0.025 + 0.01 * math.exp(-flat * 0.05)
                    p = p * np.exp(rng.normal(0.0, noise_amt, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0.0:
                        p /= ss
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
                        best_score = cur_score
                        best_p = p.copy()
                    flat = 0
                else:
                    break
        sweep += 1

    # Final state
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _subgradient_polish(p, stop_time, consider=None, max_iters=2000):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    best_p = p.copy()
    best_score = float(2.0 * n * _conv_self(p).max())
    base_step = 0.003
    min_step = 1e-9

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break

        c = _conv_self(p)
        cmax = float(c.max())
        threshold = cmax * 0.9995
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        grad = np.zeros(n)
        for k in active:
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                contrib = p[kj[valid]] + p[valid]
                grad[valid] += contrib
        grad /= float(len(active))

        # Project gradient onto tangent space of simplex (sum = 0)
        grad -= float(np.mean(grad))

        step = base_step / (1.0 + 0.01 * it)
        if step < min_step:
            break

        for scale in (1.0, 0.5, 0.25, 0.1, 0.05):
            p_cand = p - scale * step * grad
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = float(2.0 * n * _conv_self(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-16:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                break
        else:
            base_step *= 0.92
            if base_step < 1e-13:
                break

        if it % 30 == 0 and consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 15.0

    parent = _get_parent()
    best_arr = parent.copy()
    psum = float(best_arr.sum())
    if psum < 1e-12:
        best_arr = np.ones(64, dtype=np.float64) / 64.0
    else:
        best_arr = best_arr / psum
    best_score = _score(best_arr)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = _score(a)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    n = int(best_arr.size)
    total_useful = max(1.0, hard_deadline - t0)

    # Phase 1: LSE + L-BFGS (~15% of budget)
    phase1_frac = 0.15
    lbfgs_end = t0 + phase1_frac * total_useful
    p = best_arr.copy()
    if lbfgs_end > t0 + 15.0:
        p = _lbfgs_optimize(p, n, lbfgs_end, consider=consider)
        p = np.maximum(p, 0.0)
        ps = float(p.sum())
        if ps > 0:
            p /= ps
        consider(p)

    # Phase 2: Main coordinate descent (~65% of budget)
    phase2_end = t0 + 0.80 * total_useful
    if phase2_end > time.monotonic() + 20.0:
        p, _ = _coord_polish(p, phase2_end, max_sweeps=200000, seed=seed, consider=consider)
        p = np.maximum(p, 0.0)
        ps = float(p.sum())
        if ps > 0:
            p /= ps
        consider(p)

    # Phase 3: LSE fine-tune on best found
    now = time.monotonic()
    if now < hard_deadline - 30.0:
        lstop = min(hard_deadline - 15.0, now + 0.12 * total_useful)
        if lstop > now + 10.0:
            p2 = _lbfgs_optimize(best_arr, n, lstop, consider=consider, fine_only=True)
            p2 = np.maximum(p2, 0.0)
            ps = float(p2.sum())
            if ps > 0:
                p2 /= ps
            consider(p2)

    # Phase 4: Subgradient polish
    now = time.monotonic()
    if now < hard_deadline - 15.0:
        sg_stop = min(hard_deadline - 8.0, now + 0.08 * total_useful)
        if sg_stop > now + 8.0:
            p_sg = _subgradient_polish(best_arr, sg_stop, consider=consider, max_iters=2000)
            ps = float(p_sg.sum())
            if ps > 0:
                p_sg /= ps
            consider(p_sg)

    # Phase 5: Final coordinate descent squeeze
    now = time.monotonic()
    cstop = min(hard_deadline - 3.0, now + 0.10 * total_useful)
    if cstop > now + 8.0:
        p3, _ = _coord_polish(best_arr, cstop, max_sweeps=100000, seed=int(seed) + 77777, consider=consider)
        p3 = np.maximum(p3, 0.0)
        ps = float(p3.sum())
        if ps > 0:
            p3 /= ps
        consider(p3)

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.asarray(parent, dtype=np.float64).ravel()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.array([1.0])
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
```
