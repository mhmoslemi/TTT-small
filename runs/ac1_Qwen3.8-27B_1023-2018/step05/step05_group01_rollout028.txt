def _subgradient_stir(p, stop_time, consider=None, seed=777):
    """Take projected subgradient steps to reduce top convolution values."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    rng = np.random.default_rng(seed)
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_p = p.copy()
    best_score = cur_score
    step_sizes = [0.05, 0.04, 0.03, 0.02, 0.015, 0.01, 0.008]
    it = 0
    while time.monotonic() < stop_time - 1.0 and it < 200:
        c = _conv_self(p)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            break
        # Find top k active constraints
        k = min(10, c.size)
        top_idx = np.argpartition(c, -k)[-k:]
        # Compute subgradient: average of gradients of top-k active constraints
        grad = np.zeros(n)
        for ki in top_idx:
            ki = int(ki)
            # Gradient of c[ki] wrt a[j] = 2*a[ki-j] (with valid indices)
            j_min = max(0, ki - (n - 1))
            j_max = min(n - 1, ki)
            if j_max >= j_min:
                src_start = ki - j_max
                src_end = ki - j_min + 1
                grad[j_min:j_max + 1] += 2.0 * p[src_start:src_end]
        grad /= float(k)
        # Projected gradient step on simplex
        eta = step_sizes[min(it // 30, len(step_sizes) - 1)]
        p_new = p - eta * grad
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if ss > 1e-300:
            p_new = p_new / ss
        sc = float(2.0 * n * _conv_self(p_new).max())
        if math.isfinite(sc) and sc < cur_score:
            p = p_new
            cur_score = sc
            if sc < best_score:
                best_score = sc
                best_p = p.copy()
                if consider is not None:
                    try:
                        consider(best_p)
                    except Exception:
                        pass
        it += 1
    return best_p

            sub_end = min(phase_limit - 30.0, time.monotonic() + 0.2 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 30.0:
                run_opt(n_alt, best_arr, sub_end, int(seed + 8000 + n_alt),
                        do_lse=True, do_subgrad=False, do_coord=True)

    # Phase 5: Final coordinate polish on best
    final_end = min(hard - 15.0, t0 + budget - 10.0)
    if final_end > time.monotonic() + 20.0 and best_arr.size > 0:
        nbest = int(best_arr.size)
        cstop = min(final_end - 3.0, time.monotonic() + max(15.0, 0.6 * (final_end - time.monotonic())))
        if cstop > time.monotonic() + 10.0:
            try:
                p3, _ = _coord_polish(best_arr, cstop, seed=int(seed + 99999), consider=consider, G=48)
                if p3 is not None and p3.size == nbest and float(p3.sum()) > 0:
                    p3 = p3 / float(p3.sum())
                    consider(p3)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
