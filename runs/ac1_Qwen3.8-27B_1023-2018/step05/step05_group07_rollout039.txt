```python
    n = a.size
    a = a / a.sum()  # normalize
    t_start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break
        
        c_full = np.convolve(a, a)
        M = float(c_full.max())
        
        A = _build_A(n, a)
        
        # Solve LP: max sum(b) s.t. A@b <= M, b >= 0
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)]*n, method='highs',
                      options={'time_limit': 5, 'mip_rel_gap': 0})
        
        if res.status != 0:
            break
        
        b = res.x
        b_sum = b.sum()
        
        if b_sum <= 1.0 + 1e-9:
            # At (local) optimum
            break
        
        # Line search for t
        best_t, best_M = 0, M
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:

            new_M = float(np.convolve(new_a, new_a).max())
            if new_M < best_M:

        a = np.maximum(a, 0)
        a = a / a.sum()
```

Wait, I realize there might be an issue with the line search. When t=1, we're setting a = b/b_sum. The new M might be larger than the old M even though sum(b) > 1. Let me think about why.

The LP ensures conv(a, b)_k <= M for all k. But conv(b, b) might have values larger than M * sum(b). So setting a = b/b_sum might increase the max convolution.

This is why the line search is important: we need to find a t where the new max convolution is actually smaller.

Let me think about this differently. The score is `2n * M / s^2` where s = sum(a). Since we normalize s=1, the score is `2n * M`. We want to minimize M.

The update a -> (1-t)*a + t*b/s_b gives:
conv(a_new, a_new) = (1-t)^2 conv(a,a) + 2t(1-t)/(s_b) conv(a,b) + t^2/s_b^2 conv(b,b)

The max of this is <= (1-t)^2 M + 2t(1-t)/s_b * M + t^2/s_b^2 * max(conv(b,b))

= M * [(1-t)^2 + 2t(1-t)/s_b] + t^2/s_b^2 * max(conv(b,b))

For this to be < M, we need:
(1-t)^2 + 2t(1-t)/s_b + t^2 * max(conv(b,b))/(s_b^2 * M) < 1

Let R = max(conv(b,b))/(M). Then:
(1-t)^2 + 2t(1-t)/s_b + t^2 * R/s_b^2 < 1

Expanding:
1 - 2t + t^2 + 2t/s_b - 2t^2/s_b + t^2*R/s_b^2 < 1
t * (-2 + 2/s_b) + t^2 * (1 - 2/s_b + R/s_b^2) < 0
t * 2(1/s_b - 1) + t^2 * (1 - 2/s_b + R/s_b^2) < 0

Since s_b > 1, (1/s_b - 1) < 0, so the linear term is negative.
The quadratic coefficient: 1 - 2/s_b + R/s_b^2 = (s_b - 1)^2/s_b^2 + (R - 1)/s_b^2

If R < 1 (i.e., max(conv(b,b)) < M), then the quadratic coefficient is positive (since (s_b-1)^2 >= 0 and R-1 < 0 but (s_b-1)^2 dominates for large s_b).

Actually, let me just compute the coefficient: (s_b - 1)^2 + R - 1 = s_b^2 - 2*s_b + R. For this to be positive, we need s_b^2 - 2*s_b + R > 0, i.e., (s_b-1)^2 > 1-R.

If s_b is close to 1 (say s_b = 1.01), then (s_b-1)^2 = 0.0001, and we need R < 0.9999 for the quadratic to be positive. If R > 0.9999, the quadratic coefficient could be negative, meaning the objective is concave in t and the maximum is at t=1.

In practice, R = max(conv(b,b))/M is likely close to 1 (since b is "similar" to a in some sense). So the improvement from the update might be small.

Actually, let me try a completely different approach: **direct minimization of the max convolution using subgradient methods**.

The objective (with normalization sum(a)=1, a>=0) is:
minimize: max_k conv(a,a)_k

This is a convex optimization problem in the sense that the objective is the max of convex functions (each conv(a,a)_k is a quadratic form, which is convex). Wait, is conv(a,a)_k convex in a?

conv(a,a)_k = sum_j a[j] * a[k-j] = a^T Q_k a where Q_k is a symmetric matrix.

Q_k is positive semidefinite iff conv(a,a)_k >= 0 for all a, which is true for non-negative a. But Q_k might not be PSD in general (it could have negative eigenvalues for arbitrary a).

Actually, for k = n-1 (the center): conv(a,a)_{n-1} = sum_j a[j]*a[n-1-j] = a^T J a where J is the reversal matrix. J is a permutation matrix, so J^2 = I. The eigenvalues of J are ±1, so J is not PSD. Thus conv(a,a)_{n-1} is not convex in general.

So the problem is non-convex. The LP fixed-point iteration is a natural approach for non-convex problems (it's like a "benders decomposition" or "column generation" approach).

Let me think about another approach: **alternating minimization**.

We want to minimize max_k sum_j a[j]*a[k-j] subject to sum(a)=1, a>=0.

Introduce M as an auxiliary variable:
minimize M
subject to: sum_j a[j]*a[k-j] <= M for all k
             sum(a) = 1
             a >= 0

This is still non-convex (quadratic constraints). But we can use the "semidefinite relaxation" or "rank-1 factorization" approach.

Alternatively, we can use a **penalty method**: minimize max_k conv(a,a)_k + mu * (sum(a) - 1)^2 + penalty for a < 0.

Or we can use **scipy.optimize.minimize** with the SLSQP or trust-constr method, using the subgradient of the max.

The subgradient of f(a) = max_k conv(a,a)_k at a is:
∇f(a) = sum_{k in K} lambda_k * ∇_a conv(a,a)_k

where K = {k : conv(a,a)_k = M} and lambda_k >= 0, sum lambda_k = 1.

∇_a conv(a,a)_k = 2 * (Q_k @ a) where Q_k is the matrix for the k-th convolution term.

But computing this is O(n^2) per subgradient evaluation, and the optimization might need many iterations.

```python
import numpy as np
import time
from scipy.optimize import linprog

def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)

def _build_A(n, a):
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    return A

def _lp_solve(A, M, n):
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)]*n, method='highs',
                      options={'time_limit': 5,
                               'primal_feasibility_tolerance': 1e-12,
                               'dual_feasibility_tolerance': 1e-12})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def _optimize_lp(a0, deadline, seed=42):
    """LP-based fixed-point iteration."""
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = a.size
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    while time.monotonic() < deadline:
        M = float(np.convolve(a, a).max())
        
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Perturb
            sigma = min(0.05 + 0.01 * (no_improve // 200), 0.3)
            a = a * np.exp(rng.normal(0, sigma, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve = 0
            if sigma > 0.29:
                break
            continue
        
        A = _build_A(n, a)
        b = _lp_solve(A, M, n)
        
        if b is None:
            break
        
        b_sum = b.sum()
        
        if b_sum <= 1.0 + 1e-6:
            # Near equilibrium
            # Try small perturbation and re-check
            if no_improve > 50:
                a = a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                no_improve -= 30
            else:
                break
        
        # Line search
        best_t, best_new_M = 0.0, M
        for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            new_a = (1-t)*a + t*b/b_sum
            new_a = np.maximum(new_a, 0)
            ns = new_a.sum()
            if ns < 1e-12:
                continue
            new_a = new_a / ns
            new_M = float(np.convolve(new_a, new_a).max())
            if new_M < best_new_M - 1e-12:
                best_new_M = new_M
                best_t = t
        
        if best_t > 0:
            a = (1-best_t)*a + best_t*b/b_sum
            a = np.maximum(a, 0)
            a = a / a.sum()
        else:
            # No improvement; try subgradient step
            c_full = np.convolve(a, a)
            active = np.where(c_full >= M * 0.9999)[0]
            grad = np.zeros(n)
            for k_star in active:
                k_star = int(k_star)
                lo = max(0, k_star - n + 1)
                hi = min(n, k_star + 1)
                for j in range(lo, hi):
                    idx = k_star - j
                    if 0 <= idx < n:
                        grad[j] += 2.0 * a[idx]
            grad = grad - grad.mean()  # zero mean (tangent to simplex)
            gn = np.linalg.norm(grad)
            if gn > 1e-14:
                for alpha in [0.01, 0.005, 0.001]:
                    new_a = a - alpha * grad / gn
                    new_a = np.maximum(new_a, 0)
                    ns = new_a.sum()
                    if ns < 1e-12:
                        continue
                    new_a = new_a / ns
                    new_M = float(np.convolve(new_a, new_a).max())
                    if new_M < M - 1e-12:
                        a = new_a
                        break
    
    return best_a

def _make_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.linspace(0, 1, n)
    
    # Parent (interpolated to length n if needed)
    if parent.size != n:
        p = np.interp(x, np.linspace(0, 1, parent.size), parent)
    else:
        p = parent.copy()
    p = np.clip(p, 0, None)
    if p.sum() > 0:
        inits.append(p / p.sum())
    
    # Hann window
    w = 0.5 * (1 - np.cos(2*np.pi*x))
    inits.append(w / w.sum())
    
    # Gaussian windows
    for sigma in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
        w = np.exp(-0.5 * ((x - 0.5)/sigma)**2)
        inits.append(w / w.sum())
    
    # Cosine power windows
    for p in [1, 2, 3, 4]:
        w = np.cos(np.pi * (x - 0.5))**p
        w = np.maximum(w, 0)
        if w.sum() > 0:
            inits.append(w / w.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random
    for _ in range(5):
        w = np.abs(rng.normal(1, 0.5, n)) + 0.01
        inits.append(w / w.sum())
    
    # Blends with parent
    if p.sum() > 0:
        pn = p / p.sum()
        for frac in [0.3, 0.5, 0.7]:
            for init in inits[:6]:
                w = frac * pn + (1-frac) * init
                inits.append(w / w.sum())
    
    return inits
```

```python
import numpy as np
import time
from scipy.optimize import linprog

def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)

def _build_A(n, a):
    """Build (2n-1) x n matrix where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    return A

def _solve_lp(A, M, n):
    """Solve: max sum(b) s.t. A@b <= M, b >= 0. Returns b or None."""
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(A.shape[0], M),
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': 5})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def _optimize(a0, deadline, seed=42):

    a = np.clip(np.asarray(a0, dtype=np.float64), 0, 1000).copy()

        a = np.ones_like(a)

    stall = 0

        M = float(np.convolve(a, a).max())

            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

            xx = np.arange(n, dtype=np.float64)

            if mode == 0:

            elif mode == 1:

            a = np.maximum(a, 1e-15)
```
