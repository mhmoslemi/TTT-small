    p_out = best_p.copy()
    ss = float(p_out.sum())
    if ss > 1e-300:
        p_out = p_out / ss
    return p_out


def _coord_descent(p, stop_time, consider=None, seed=42, max_sweeps=200000):
    n = int(p.size)
    p = np.maximum(p.copy(), 0.0)
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p, math.inf
    p = p / s

    Ltot = 2 * n - 1
    c = _fft_conv(p)
    S = 1.0

    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S <= 1e-300:
                break
            
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]  # influence: c[start+k] changes by w * p[k] (times 2 initially)
            # Wait: c[start+k] = ... + p[i]*p[start+k-i] + ... 
            # If we change p[i] from v to v+w:
            # c[start+k] changes by 2*w*p[k] (for k != i-start) or w*(2*p[i]+w) (for k = i-start)
            # Actually: c[start+k] = sum_j p[j]*p[start+k-j]
            # The terms involving p[i] (j=i): p[i]*p[start+k-i] = p[i]*p[k] (if 0<=k<n)
            # The terms involving p[i] (start+k-j=i, j=start+k-i): p[start+k-i]*p[i] = p[k]*p[i] (same)
            # So: delta c[start+k] = 2*w*p[k] for k != i-start, and w*(2*p[i]+w) for k = i-start
            # Wait, if start+k-i = k, then k = 0... no.
            # start = i, so c[i+k] for k=0,...,n-1
            # c[i+k] = sum_j p[j] * p[i+k-j]
            # Terms with j=i: p[i] * p[k]

            # If k != i: two terms, delta = 2*w*p[k]

                    b_gs = a_gs

                    a_gs = hi - gr * (hi - lo)
                    fa = eval_x(a_gs)
