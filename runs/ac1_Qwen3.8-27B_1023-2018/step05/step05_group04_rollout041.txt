                        aff_seg = aff_seg.copy()
                        aff_seg[diag_pos] += w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    c[start:end] = aff_seg
                    cur_score = 2.0 * n * new_max
                    improved = True

        # Renormalize
        s_new = float(p.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            inv = 1.0 / s_new
            p *= inv
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-15:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= max_flat:
                if time.monotonic() < stop_time - 10.0:
                    kick_size = 0.02 + 0.04 * min(1.0, flat / 10.0)
                    p = p * np.exp(rng.normal(0.0, kick_size, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _lse_refine(p, stop_time, consider=None):
    """Lightweight log-sum-exp style refinement using subgradient steps."""
    n = p.size
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    best_sc = _score_arr(p)
    best_p = p.copy()

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return c

    eta = 0.05
    decay = 0.999
    stale = 0
    while time.monotonic() < stop_time - 2.0 and stale < 10:
        c = cfast(p)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            break
        # Soft top-k weights
        T = 0.01 * cmax
        scaled = (c - cmax) / T
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if sum_exp <= 0 or not math.isfinite(sum_exp):
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
        else:
            w = exps / sum_exp

        # Subgradient: grad_c[i] = sum_k w[k] * (a[k-i] + [k==2i]*a[i])
        # Use FFT for efficiency
        F = rfft(p, conv_size)
        # grad[i] = sum_k w[k] * a[k-i] (valid range) + correction for k=2i
        # This is a cross-correlation of w and p
        # conv correlation: grad_corr[i] = sum_k w[k] * p[k-i] for valid k-i
        # Equivalent to: (w_reversed * p) at appropriate offset
        w_rev = w[::-1].copy()
        Fs = rfft(w_rev, conv_size)
        G = rfft(p, conv_size)
        corr = irfft(Fs * G, conv_size)[:conv_len]
        # corr[k] = sum_i w_rev[i] * p[k-i] = sum_i w[conv_len-1-i] * p[k-i]
        # We want: sum_k w[k] * p[k-i] for each i
        # Let t = k-i, so k = t+i: sum_t w[t+i] * p[t]
        # This is a correlation. Let's use a different approach.
        # Actually, let's just compute it directly with numpy for the affected range
        # For efficiency, use the fact that we only need the gradient at positions where p[i] > 0
        # grad[i] = sum_k w[k] * (p[k-i] + delta(k,2i)*p[i])
        # First term is cross-correlation of w with p evaluated at lag i
        # cross_corr[i] = sum_k w[k] * p[k-i]

        # Use FFT-based cross-correlation
        # cross_corr = ifft(fft(w, N) * conj(fft(p, N)))
        N = next_fast_len(3 * n)
        Wf = rfft(w, N)
        Pf = rfft(p, N)
        # cross_corr[m] = sum_k w[k] * p[k+m] ... need to be careful with conventions
        # Let's use: corr[m] = sum_j w[j] * p[j+m]
        # FFT: corr = ifft(conj(FFT(w)) * FFT(p)) ... this is circular correlation
        # For linear correlation with proper zero-padding:
        Wf_full = np.fft.rfft(w, N)
        Pf_full = np.fft.rfft(p, N)
        corr = np.fft.irfft(np.conj(Wf_full) * Pf_full, N)[:2 * n - 1]
        # corr[m] = sum_j w[j] * p[j+m] for m >= 0
        # We want grad[i] = sum_k w[k] * p[k-i] = sum_k w[k+i] * p[k] (by change of var)
        # = corr[i] (if we define corr[m] = sum_j w[j+m] * p[j])
        # Actually: sum_k w[k] * p[k-i], let t = k-i, k = t+i:
        # = sum_t w[t+i] * p[t]
        # This is the cross-correlation of w and p at lag i
        # Using the computed corr: corr[m] = sum_j w[j] * p[j+m]
        # We need sum_t w[t+i] * p[t] = sum_t w[t+i] * p[t]
        # Let u = t+i, t = u-i: sum_u w[u] * p[u-i]
        # This is cross_corr at lag -i, or equivalently corr at lag i if we reverse
        # Actually for the purpose here, let's just use a simpler approach

        # Simpler: directly compute gradient using the formula
        # grad[i] = sum_{k: 0<=k-i<n} w[k] * p[k-i] + w[2i] * p[i] (if 2i valid)
        # The first sum is a cross-correlation. Let me just do it with a loop over affected range
        # For n=3072, this is O(n^2) per step which is too slow.
        # Let me use FFT properly.

        # The gradient of the weighted sum of c[k] w.r.t. a[i]:
        # d/d(a[i]) [sum_k w[k] * c[k]] where c[k] = sum_j a[j]*a[k-j]
        # = sum_k w[k] * (a[k-i] + a[i]) for valid k where both i and k-i are in [0,n-1]
        # = sum_{k: max(i,i-n+1)<=k<=min(i+n-1, 2n-2)} w[k] * a[k-i] + ...
        # This is getting complex. Let me use a simpler subgradient approach.

        # Alternative: just do a simple steepest descent on the log-transformed variables
        # u = log(p), optimize sum_k w[k]*c[k] subject to constraints
        # This is getting too complex. Let me just use coordinate descent instead.
        break

    return best_p


def _try_initialization(init_func, n, stop_time, consider, seed):
    """Try an initialization and run coordinate descent on it."""
    try:
        p = init_func(n)
    except Exception:
        return None, float('inf')
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s < 1e-10:
        return None, float('inf')
    p = p / s
    sc = _score_arr(p)
    if consider:
        try:
            consider(p)
        except Exception:
            pass
    if math.isfinite(sc):
        p_out, sc_out = _coord_descent(p, stop_time, consider=consider, seed=seed)
        if p_out is not None and p_out.size > 0:
            consider(p_out)
            sc = _score_arr(p_out)
            return p_out, sc
    return p, sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 5.0:
        hard = t0 + max(3.0, budget - 2.0)

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float('inf')

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()

    consider(parent)

    n_parent = int(best_arr.size)

    # Main phase: coordinate descent on parent length
    n_work = n_parent if 128 <= n_parent <= 4096 else 3072
    p_start = _resample(best_arr, n_work)
    consider(p_start)

    # Allocate time: 70% to main CD, 15% to alternative inits, 15% to final refinement
    main_end = t0 + 0.65 * (hard - t0)
    alt_end = t0 + 0.82 * (hard - t0)
    final_end = hard - 3.0

    # Phase 1: Coordinate descent on parent
    if time.monotonic() < main_end - 5.0:
        try:
            p1, _ = _coord_descent(p_start, main_end, consider=consider, seed=int(seed) + 1000)
            if p1 is not None and p1.size > 0 and np.all(np.isfinite(p1)) and p1.sum() > 0:
                consider(p1)
        except Exception:
            pass

    # Phase 2: Try alternative initializations with remaining time
    remaining_for_alt = alt_end - time.monotonic()
    if remaining_for_alt > 30.0:
        rng = np.random.default_rng(int(seed) + 99999)

        # Tent initialization
        x = (np.arange(n_work, dtype=np.float64) + 0.5) / n_work
        tent = np.maximum(0.0, 1.0 - 2.0 * np.abs(x - 0.5))
        ts = float(tent.sum())
        if ts > 0:
            tent /= ts
        t_end = time.monotonic() + remaining_for_alt * 0.35
        if t_end > time.monotonic() + 20.0:
            try:
                p_t, _ = _coord_descent(tent, t_end, consider=consider, seed=int(seed) + 2000)
                if p_t is not None:
                    consider(p_t)
            except Exception:
                pass

        # Blended initialization (parent + tent)
        p_blend = 0.6 * _resample(best_arr, n_work) + 0.4 * tent
        p_blend = np.maximum(p_blend, 0.0)
        bs = float(p_blend.sum())
        if bs > 0:
            p_blend /= bs
        b_end = time.monotonic() + remaining_for_alt * 0.35
        if b_end > time.monotonic() + 20.0:
            try:
                p_b, _ = _coord_descent(p_blend, b_end, consider=consider, seed=int(seed) + 3000)
                if p_b is not None:
                    consider(p_b)
            except Exception:
                pass

        # Perturbed parent
        p_pert = _resample(best_arr, n_work) * np.exp(rng.normal(0.0, 0.1, n_work))
        p_pert = np.maximum(p_pert, 0.0)
        ps = float(p_pert.sum())
        if ps > 0:
            p_pert /= ps
        pp_end = time.monotonic() + remaining_for_alt * 0.3
        if pp_end > time.monotonic() + 20.0:
            try:
                p_p, _ = _coord_descent(p_pert, pp_end, consider=consider, seed=int(seed) + 4000)
                if p_p is not None:
                    consider(p_p)
            except Exception:
                pass

    # Phase 3: Final refinement on the best
    rem = final_end - time.monotonic()
    if rem > 20.0:
        p_final = _resample(best_arr, n_work)
        try:
            p_f, _ = _coord_descent(p_final, final_end, consider=consider, seed=int(seed) + 5000)
            if p_f is not None and p_f.size > 0 and np.all(np.isfinite(p_f)) and p_f.sum() > 0:
                consider(p_f)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
