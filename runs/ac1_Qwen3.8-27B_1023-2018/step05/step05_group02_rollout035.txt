        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        sc_cur = _score_fft(p_cur)
        if not math.isfinite(sc_cur):
            p = best_p.copy()
            u = p_to_u(p)
            continue
        if sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif sc_cur > best_sc + max(0.003, 3e-4 * best_sc):
            p = best_p.copy()
        else:
            p = p_cur.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_descent(p, stop_time, seed=42, consider=None):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    L = 2 * n - 1
    c = _conv_self(p)
    cur_max = float(c.max())
    best_max = cur_max
    best_p = p.copy()

    rng = np.random.default_rng(seed)
    sweep = 0
    flat_count = 0
    G_coarse = 25
    G_fine = 50

    while sweep < 200000 and time.monotonic() < stop_time - 0.8:
        improved = False
        order = rng.permutation(n)

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.2:
                break

            v = float(p[i])
            # Affected lags: k in [i, i+n-1]
            k_start = i
            k_end = min(i + n, L)
            Ls = k_end - k_start
            if Ls <= 0:
                continue

            # base = c[k_start:k_end] - v * 2 * p[0:Ls] (adjust for k=2i)
            # linear coeff = 2 * p[0:Ls]
            # At k=2i (index 2i-k_start = i within the slice): quad term
            base = c[k_start:k_end].copy()
            lin = 2.0 * p[:Ls].copy()
            # base = c[k] - v*lin[k-k_start] - v^2*(1 if k==2i else 0)
            base -= v * lin
            if k_start <= 2 * i < k_end:
                pos_in_slice = 2 * i - k_start
                base[pos_in_slice] -= v * v
                # lin at this position: the quadratic replaces the linear
                # Actually for k=2i: c_k = base + 2*x*p[k-i] + x^2 but since k-i=i:
                # c_k = base_old + x*p[i] + p[i]*x + ... no.
                # Let me redo: c_k = sum_j a_j * a_{k-j}
                # a_i appears as: a_i * a_{k-i} (first factor) + a_{k-i} * a_i (second factor)
                # If k-i = i (k=2i): a_i * a_i appears only ONCE = x^2
                # If k-i != i: a_i * a_{k-i} appears TWICE = 2*x*a_{k-i}
                # So for k=2i: c_k(x) = (c_k - v^2) + x^2
                #   base = c_k - v*(2*p[i]) - v^2 + v*(2*p[i]) ... 
                # Let me just be explicit:
                # c_k_current = c_k (with a_i = v)
                # For k=2i: c_k = (sum_{j!=i} a_j*a_{k-j}) + v*v
                #   base = c_k - v*v (this is the constant part)
                #   c_k(x) = base + x*x
                # For k != 2i (but affected): c_k = (sum_{j: j!=i, k-j!=i} a_j*a_{k-j}) + v*a_{k-i} + a_{k-i}*v
                #   = const + 2*v*a_{k-i}
                #   base = c_k - 2*v*a_{k-i}
                #   c_k(x) = base + 2*x*a_{k-i}
                #
                # So my formula: base = c_slice - v*lin works for k!=2i.
                # For k=2i: base should be c_k - v^2, and c_k(x) = base + x^2.
                # But v*lin at k=2i is v*2*p[i] = 2v^2.
                # So base = c_k - 2v^2 - v^2 = c_k - 3v^2. That's wrong.
                #
                # Let me fix: for k=2i, the linear coefficient should be 0 (not 2*p[i]).
                # Because a_i only appears as x*x, not as 2*x*a_{k-i}.
                
                # Fix: set lin at position 2i to 0, and the base already has -v*2*p[i] subtracted.
                # We need: base_correct = c_k - v^2 (for the x^2 formula)
                # We have: base_current = c_k - v*2*p[i] - v^2 = c_k - 2v^2 - v^2 = c_k - 3v^2
                # We need: base_correct = c_k - v^2
                # So: base_correct = base_current + 2v^2
                
                base[pos_in_slice] += 2.0 * v * v
                lin[pos_in_slice] = 0.0  # No linear term at k=2i

            # Unaffected max
            if k_start > 0:
                ml = float(c[:k_start].max())
            else:
                ml = 0.0
            if k_end < L:
                mr = float(c[k_end:].max())
            else:
                mr = 0.0
            max_const = max(ml, mr)

            # x_hi: upper bound for the search
            x_hi = max(30.0 * v, 80.0 / n, 0.01, 1e-12)
            x_hi = min(x_hi, 5.0)  # since sum=1, no single element should exceed ~1
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, 1.0 / n)

            # Coarse grid
            pts = np.linspace(0.0, x_hi, G_coarse + 1)
            if 0.0 <= v <= x_hi:
                pts = np.unique(np.concatenate((pts, [v])))

            w = pts - v
            denom = 1.0 + w  # (sum = 1, so new sum = 1 + (x - v))
            # c_k(x) = base + w*lin + w^2*(1 if k==2i else 0)
            aff = base[None, :] + w[:, None] * lin[None, :]
            if k_start <= 2 * i < k_end:
                pos = 2 * i - k_start
                aff[:, pos] += w * w
            maff = np.maximum(aff, max_const, axis=0)
            # maff has shape (len(pts),) after max over axis... wait
            # aff has shape (len(pts), Ls), maff should be max over Ls axis
            maff = np.max(aff, axis=1)
            maff = np.maximum(maff, max_const)
            # Score = 2n * maff / denom^2
            dv = denom
            vals = np.where(dv > 1e-15, 2.0 * n * maff / (dv * dv), 1e100)
            vals = np.where(np.isfinite(vals), vals, 1e100)

            if not np.any(np.isfinite(vals)):
                continue

            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Fine refinement: golden section
            dw = x_hi / float(G_coarse)
            lo = max(0.0, best_x - 2.0 * dw)
            hi = min(x_hi, best_x + 2.0 * dw)
            if hi - lo > 1e-14:
                # Fine grid
                fpts = np.linspace(lo, hi, G_fine)
                fw = fpts - v
                fd = 1.0 + fw
                faff = base[None, :] + fw[:, None] * lin[None, :]
                if k_start <= 2 * i < k_end:
                    pos = 2 * i - k_start
                    faff[:, pos] += fw * fw
                fmaff = np.maximum(np.max(faff, axis=1), max_const)
                fvals = np.where(fd > 1e-15, 2.0 * n * fmaff / (fd * fd), 1e100)
                fvals = np.where(np.isfinite(fvals), fvals, 1e100)
                if np.any(np.isfinite(fvals)):
                    fi = int(np.argmin(fvals))
                    if fvals[fi] < best_val:
                        best_x = float(fpts[fi])
                        best_val = float(fvals[fi])

                # Golden section
                gr = (math.sqrt(5.0) - 1.0) / 2.0
                g_lo = max(0.0, best_x - dw * 0.5)
                g_hi = min(x_hi, best_x + dw * 0.5)
                if g_hi - g_lo > 1e-16:
                    def eval_single(xx):
                        ww = xx - v
                        dd = 1.0 + ww
                        if dd <= 1e-15:
                            return 1e100
                        aa = base + ww * lin
                        if k_start <= 2 * i < k_end:
                            pos = 2 * i - k_start
                            aa[pos] += ww * ww
                        mm = float(np.max(aa))
                        if mm < max_const:
                            mm = max_const
                        return 2.0 * n * mm / (dd * dd)

                    a_gs = g_hi - gr * (g_hi - g_lo)
                    b_gs = g_lo + gr * (g_hi - g_lo)
                    fa = eval_single(a_gs)
                    fb = eval_single(b_gs)
                    for _ in range(16):
                        if not math.isfinite(fa):
                            fa = 1e100
                        if not math.isfinite(fb):
                            fb = 1e100
                        if fa < fb:
                            g_hi = b_gs
                            b_gs = a_gs
                            fb = fa
                            a_gs = g_hi - gr * (g_hi - g_lo)
                            fa = eval_single(a_gs)
                        else:
                            g_lo = a_gs
                            a_gs = b_gs
                            fa = fb
                            b_gs = g_lo + gr * (g_hi - g_lo)
                            fb = eval_single(b_gs)
                        if g_hi - g_lo < 1e-15:
                            break
                    gs_x = (g_lo + g_hi) / 2.0
                    gv = eval_single(gs_x)
                    if math.isfinite(gv) and gv < best_val:
                        best_x = gs_x
                        best_val = gv

            if math.isfinite(best_val) and best_val < cur_max - 1e-14 and 0.0 <= best_x <= 5.0:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    c[k_start:k_end] = base + w * lin
                    if k_start <= 2 * i < k_end:
                        pos = 2 * i - k_start
                        c[k_start + pos] += w * w
                    # Scale c since sum changed: new_sum = 1 + w
                    # We'll renormalize at the end of each sweep
                    cur_max = best_val
                    improved = True

        # Renormalize
        ss = float(p.sum())
        if ss > 1e-300 and math.isfinite(ss):
            inv = 1.0 / ss
            p *= inv
            c = _conv_self(p)
            np.maximum(c, 0.0, out=c)
            new_max = float(c.max())
            if math.isfinite(new_max) and new_max < best_max - 1e-16:
                best_max = new_max
                best_p = p.copy()
                if consider is not None:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat_count = 0
        else:
            flat_count += 1
            if flat_count >= 5:
                if time.monotonic() < stop_time - 8.0:
                    noise_amt = 0.02 + 0.01 * math.exp(-flat_count * 0.08)
                    p = p * np.exp(rng.normal(0.0, noise_amt, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0.0:
                        p /= ss
                    c = _conv_self(p)
                    cur_max = float(c.max())
                    if math.isfinite(cur_max) and cur_max < best_max - 1e-15:
                        best_max = cur_max
                        best_p = p.copy()
                        if consider is not None:
                            consider(best_p)
                    flat_count = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    c = _conv_self(p)
    sc = float(2.0 * n * c.max())
    if math.isfinite(sc) and sc < best_max:
        best_max = sc
        best_p = p.copy()
    return best_p, 2.0 * n * best_max


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 15.0

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(128, dtype=np.float64) / 128.0
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    n = int(best_arr.size)

    # Phase 1: L-BFGS optimization (use ~30% of time)
    phase1_end = min(hard_deadline - 60.0, t0 + 0.30 * (hard_deadline - t0))
    if phase1_end > t0 + 15.0:
        try:
            p2 = _lbfgs_optimize(best_arr, n, phase1_end, consider=consider)
            if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0.0:
                p2 = p2 / float(p2.sum())
                consider(p2)
        except Exception:
            pass

    # Phase 2: Coordinate descent (use ~60% of time)
    now = time.monotonic()
    phase2_end = min(hard_deadline - 30.0, t0 + 0.90 * (hard_deadline - t0))
    if phase2_end > now + 20.0:
        try:
            p3, _ = _coord_descent(best_arr, phase2_end, seed=int(seed + 7777), consider=consider)
            if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and p3.sum() > 0.0:
                p3 = p3 / float(p3.sum())
                consider(p3)
        except Exception:
            pass

    # Phase 3: Final coordinate descent squeeze
    now = time.monotonic()
    phase3_end = hard_deadline - 8.0
    if phase3_end > now + 10.0:
        try:
            p4, _ = _coord_descent(best_arr, phase3_end, seed=int(seed + 99999), consider=consider)
            if p4 is not None and p4.size == n and np.all(np.isfinite(p4)) and p4.sum() > 0.0:
                p4 = p4 / float(p4.sum())
                consider(p4)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
