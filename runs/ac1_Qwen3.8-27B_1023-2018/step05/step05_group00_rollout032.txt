        # W = np.lib.stride_tricks.sliding_window_view(w, n)  # shape (n, n)

        # (dc/da)[k, i] = 2*a[k-i] if 0<=k-i<n else 0

        # = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]

        # q = W @ a where W is the sliding window view of w
        # But w has length 2n-1, and we want windows of size n starting at positions 0 to n-1.
        # sliding_window_view(w, n) gives windows at positions 0 to (2n-1)-n = n-1. That's exactly what we want!

        if np.lib.stride_tricks.has_stride_tricks:

                W = np.lib.stride_tricks.sliding_window_view(w, n)
                q = W @ pp  # (n,) array
                grad = (2.0 / T) * q
            except Exception:

        grad_u = grad * pp
        # Subtract mean to maintain the zero-mean constraint on u

        grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
        
        return float(f), grad_u

    from scipy.optimize import minimize

            best_score = cur_score
            best_p = p.copy()
            if consider is not None:
                consider(best_p)
    
    return best_p
