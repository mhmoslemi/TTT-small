                    c = _fft_conv(q)
                    S = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if cur_score < best_score * 1.05:
                        p = q
                    else:
                        p = best_p.copy()
                        c = _fft_conv(p)
                        cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
                continue

        sweep += 1

    consider(best_p)
    return best_p


def _lp_direction(a, M, method="highs"):
    """Find LP direction: max sum(b) s.t. a*b <= M, b >= 0."""
    if linprog is None or sp is None:
        return None
    n = int(a.size)
    if n < 16 or n > 2048:
        return None
    if not math.isfinite(M) or M <= 0:
        return None

    try:
        idx = np.arange(n, dtype=np.int64)
        cols = np.repeat(idx, n)
        off = np.tile(idx, n)
        rows = cols + off
        data = a[off].astype(np.float64)
        A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        cobj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        try:
            res = linprog(cobj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method=method)
        except Exception:
            res = linprog(cobj, A_ub=A, b_ub=b_ub, bounds=(0.0, None))
        if res is not None and getattr(res, "success", False):
            b = np.asarray(res.x, dtype=np.float64).ravel()
            b = np.where(np.isfinite(b), b, 0.0)
            b = np.maximum(b, 0.0)
            if float(b.sum()) > 0:
                return b / b.sum()
    except Exception:
        pass
    return None


def _resample(src, n_out):
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.full(n_out, 1.0 / n_out, dtype=np.float64)
    if src.size == n_out:
        return src.copy()
    m = src.size
    xp = (np.arange(m, dtype=np.float64) + 0.5) / m
    fp = np.interp((np.arange(n_out, dtype=np.float64) + 0.5) / n_out, xp, src)
    return _normalize(fp)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(20.0, budget)
    reserve = 20.0 if budget > 200 else 5.0
    hard = t0 + budget - reserve

    parent = _get_parent()
    n = int(parent.size)

    best_arr = _normalize(parent.copy())
    best_score = math.inf
    try:
        best_score = float(_score_norm(best_arr))
    except Exception:
        pass

    def consider(a):
        nonlocal best_arr, best_score
        if a.size != n:
            a = _resample(a, n)
        a = _normalize(a)
        try:
            sc = float(_score_norm(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()

    consider(best_arr)

    # Phase 1: L-BFGS optimization
    lse_stop = min(hard, t0 + max(30.0, 0.12 * budget))
    if lse_stop > t0 + 20.0:
        try:
            p = _lbfgs_optimize(best_arr.copy(), lse_stop, consider)
            consider(p)
        except Exception:
            pass

    # Phase 2: LP-generated shapes at small scales
    lp_stop = min(hard - 300.0, time.monotonic() + max(20.0, 0.06 * budget))
    if lp_stop > time.monotonic() + 15.0:
        small_sizes = [128, 256, 512]
        for ns in small_sizes:
            if time.monotonic() > lp_stop - 10.0:
                break
            if ns >= n:
                continue
            try:
                small = _resample(best_arr, ns)
                c_small = _fft_conv(small)
                M = float(c_small.max())
                if math.isfinite(M) and M > 0:
                    b = _lp_direction(small, M)
                    if b is not None:
                        # Interpolate
                        for t in (0.3, 0.5, 0.7, 0.9):
                            cand_small = (1 - t) * small + t * b
                            cand_small = _normalize(cand_small)
                            cand_full = _resample(cand_small, n)
                            consider(cand_full)
            except Exception:
                pass

    # Phase 3: Main coordinate descent
    main_stop = min(hard - 100.0, time.monotonic() + max(60.0, 0.55 * budget))
    if main_stop > time.monotonic() + 30.0:
        try:
            p = _coord_descent(best_arr.copy(), main_stop, consider, n_cand=20, max_sweeps=200000)
            consider(p)
        except Exception:
            pass

    # Phase 4: Another L-BFGS pass with fine temperatures
    if time.monotonic() < hard - 80.0:
        fine_stop = min(hard - 50.0, time.monotonic() + min(80.0, 0.08 * budget))
        if fine_stop > time.monotonic() + 15.0:
            try:
                p = _lbfgs_optimize(best_arr.copy(), fine_stop, consider)
                consider(p)
            except Exception:
                pass

    # Phase 5: Final intensive coordinate descent
    if time.monotonic() < hard - 30.0:
        try:
            p = _coord_descent(best_arr.copy(), hard - 5.0, consider, n_cand=24, max_sweeps=200000, reshape_interval=25)
            consider(p)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        if out.size == 0:
            out = parent.copy()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = parent / parent.sum()
            s = float(out.sum())
        if s > 0:
            out = out / s
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if math.isfinite(final_sc):
            return ls
    except Exception:
        pass

    # Fallback
    try:
        out = parent / parent.sum()
        ls = [float(x) for x in out]
        sc = float(evaluate_sequence(ls))
        if math.isfinite(sc):
            return ls
    except Exception:
        pass
    return [1.0]
