                        if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                            best_score = cur_score
                            best_p = p.copy()
                    else:
                        break
                    flat = 0
                else:
                    break
        sweep += 1

    return best_p, best_score


def _lse_optimize(p, stop_time, consider=None, T_frac=0.005, max_outer=8):
    n = int(p.size)
    if n == 0 or n < 32:
        return p
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    def u_to_p(uu):
        m = float(np.max(uu))
        e = np.exp(uu - m)
        ss = float(e.sum())
        if ss <= 0 or not math.isfinite(ss):
            return np.full(uu.size, 1.0 / uu.size)
        return e / ss

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def obj(uu, T):
        pp = u_to_p(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        scaled = (c - cmax) / T
        exps = np.exp(np.minimum(scaled, 50.0))
        sum_exp = float(exps.sum())
        if sum_exp <= 0 or not math.isfinite(sum_exp):
            f = cmax
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
        else:
            f = cmax + T * math.log(sum_exp / c.size)
            w = exps / sum_exp
        # Gradient: 2 * (pp * (cross - dot(pp, cross)))
        # cross = convolution of pp with w (at the "second half")
        F = rfft(pp[::-1], conv_size)
        G = rfft(w, conv_size)
        cc = irfft(F * G, conv_size)[:conv_len]
        np.maximum(cc, 0.0, out=cc)
        q = cc[n - 1:2 * n - 1]
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    c0 = cfast(p)
    cmax0 = float(c0.max())
    T = max(1e-10, T_frac * cmax0)

    best_u = u.copy()
    best_f = obj(u, T)[0]

    for outer in range(max_outer):
        if time.monotonic() > stop_time - 3.0:
            break
        opts = {"maxiter": 50, "gtol": 1e-10, "ftol": 1e-14, "maxls": 20, "disp": False, "maxcor": 30}
        try:
            res = minimize(obj, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
        except Exception:
            break
        if res.x is not None and res.x.size == n:
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if np.all(np.isfinite(x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_f - 1e-12:
                    best_f = fval
                    best_u = x.copy()
                    p_cur = u_to_p(best_u)
                    if consider is not None:
                        try:
                            consider(p_cur)
                        except Exception:
                            pass
                if getattr(res, "status", 1) == 0:
                    break
    p_out = u_to_p(best_u)
    return p_out


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 15.0

    parent = _get_parent()
    n_par = int(parent.size)
    best_arr = parent.copy()
    best_arr = np.where(np.isfinite(best_arr), best_arr, 0.0)
    best_arr = np.clip(best_arr, 0.0, 1000.0)
    best_score = math.inf
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0 or float(best_arr.sum()) < 0.01:
        best_arr = np.ones(128, dtype=np.float64) / 128.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0 or float(a.sum()) < 0.01:
            return
        try:
            sc = float(_score_fast(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 5e-14:
            sa = float(a.sum())
            best_arr = a.copy()
            best_score = sc

    work = best_arr.copy()
    ws = float(work.sum())
    if ws > 1e-15:
        work = work / ws

    # === Phase 1: Main coordinate descent ===
    p1_end = min(hard - 120.0, t0 + 0.50 * budget)
    if p1_end > time.monotonic() + 30.0:
        try:
            work, _ = _coord_descent(work, p1_end, seed=int(seed + 100),
                                     consider=consider, G=48, perturb_every=4)
        except Exception:
            pass
    work = best_arr.copy()
    ws = float(work.sum())
    if ws > 1e-15:
        work = work / ws

    # === Phase 2: LSE gradient step ===
    p2_end = min(hard - 80.0, time.monotonic() + 0.15 * (hard - t0))
    if p2_end > time.monotonic() + 20.0 and n_par >= 64:
        try:
            p2 = _lse_optimize(work, p2_end, consider=consider, T_frac=0.003, max_outer=6)
            if p2 is not None and p2.size == n_par and np.all(np.isfinite(p2)) and p2.sum() > 1e-15:
                p2 = p2 / float(p2.sum())
                consider(p2)
        except Exception:
            pass

    work = best_arr.copy()
    ws = float(work.sum())
    if ws > 1e-15:
        work = work / ws

    # === Phase 3: Second coordinate descent (finer) ===
    p3_end = min(hard - 60.0, t0 + 0.78 * budget)
    if p3_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord_descent(work, p3_end, seed=int(seed + 99999),
                                     consider=consider, G=60, perturb_every=6)
        except Exception:
            pass

    work = best_arr.copy()
    ws = float(work.sum())
    if ws > 1e-15:
        work = work / ws

    # === Phase 4: Try different length ===
    rem4 = hard - 30.0 - time.monotonic()
    if rem4 > 80.0 and n_par != 2048:
        n2 = 2048
        xp = (np.arange(n_par, dtype=np.float64) + 0.5) / float(n_par)
        fp = np.interp((np.arange(n2, dtype=np.float64) + 0.5) / float(n2), xp, best_arr)
        p2 = np.maximum(fp, 0.0)
        ps2 = float(p2.sum())
        if ps2 > 0.01:
            p2 /= ps2
            consider(p2)
            if rem4 > 60.0:
                p2_end = min(hard - 30.0, time.monotonic() + rem4 - 20.0)
                if p2_end > time.monotonic() + 15.0:
                    try:
                        p2, _ = _coord_descent(p2, p2_end, seed=int(seed + 55555),
                                               consider=consider, G=48, perturb_every=5)
                    except Exception:
                        pass

    # === Phase 5: Final LSE on best ===
    rem5 = hard - 25.0 - time.monotonic()
    if rem5 > 20.0 and n_par >= 64:
        work = best_arr.copy()
        ws = float(work.sum())
        if ws > 1e-15:
            work = work / ws
        try:
            p5 = _lse_optimize(work, hard - 25.0, consider=consider, T_frac=0.001, max_outer=5)
            if p5 is not None and p5.size == n_par:
                consider(p5)
        except Exception:
            pass

    # === Phase 6: Final coordinate descent squeeze ===
    rem6 = hard - 18.0 - time.monotonic()
    if rem6 > 15.0:
        work = best_arr.copy()
        ws = float(work.sum())
        if ws > 1e-15:
            work = work / ws
        try:
            work, _ = _coord_descent(work, hard - 18.0, seed=int(seed + 777777),
                                     consider=consider, G=55, perturb_every=8)
        except Exception:
            pass

    # === Final validation ===
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
