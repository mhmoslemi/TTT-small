def _cd_sweep(p, c, n, S, G=32, stop_time=None, rng=None):
    """One coordinate descent sweep. Returns improved p, c, S, score, improved_flag."""
    Ltot = 2 * n - 1
    score = 2.0 * n * float(c.max())
    best_score = score
    improved = False
    best_p = p.copy()
    
    if rng is None:
        rng = np.random.default_rng(0)
    
    order = rng.permutation(n)
    
    for i in order:
        if stop_time is not None and time.monotonic() > stop_time:
            break
        
        v = p[i]
        start = i
        end = min(i + n, Ltot)
        Ls = end - start
        if Ls <= 0:

        # j-i ranges from 0 to Ls-1
        infl = 2.0 * p[:Ls]  # 2*a[0], 2*a[1], ..., 2*a[Ls-1]

        if start > 0:

        if end < Ltot:

        if has_diag:

        m_total = np.maximum(m_win, c_max_out)

def _lp_update(a, n, M, budget_s=25.0):
    """LP-based update. Returns new sequence or None."""
    if not HAS_CVXPY:
        return None
    
    t0 = time.monotonic()
    if time.monotonic() - t0 > budget_s:
        return None
    
    try:
        # Build constraint matrix
        # (a*b)_j <= M for j = 0, ..., 2n-2
        # (a*b)_j = sum_{l=0}^{n-1} A[j,l] * b[l]
        # A[j,l] = a[j-l] if 0 <= j-l < n, else 0
        
        b_var = cp.Variable(n, nonneg=True)
        obj = cp.Maximize(cp.sum(b_var))
        constraints = []
        
        # Build rows of A
        # For efficiency, build as sparse
        import scipy.sparse as sp
        
        rows = []
        cols = []
        data = []
        
        for j in range(2 * n - 1):
            l_min = max(0, j - n + 1)
            l_max = min(j, n - 1)
            for l in range(l_min, l_max + 1):
                val = a[j - l]
                if val > 0:
                    rows.append(j)
                    cols.append(l)
                    data.append(val)
        
        A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        
        # For non-negativity of a (which it is), we can simplify
        # Only add constraints where A row has nonzeros
        # But actually all rows have nonzeros for 0 <= j < 2n-1
        
        constraints.append(A @ b_var <= M)
        
        prob = cp.Problem(obj, constraints)
        
        # Set time limit
        solver = cp.CBC
        try:
            prob.solve(solver=solver, time_limit=budget_s)
        except:
            # Try another solver
            prob.solve(solver=cp.CBC)
        
        if b_var.value is not None and np.all(np.isfinite(b_var.value)):
            b_val = np.maximum(b_var.value, 0.0)
            s_b = float(b_val.sum())
            if s_b > 1e-10:
                # Rescale to match sum(a)
                s_a = float(a.sum())
                b_rescaled = b_val * (s_a / s_b)
                return b_rescaled
    except Exception as e:
        pass
    
    return None

def _conv_self(a):
    n = len(a)

    size = next_fast_len(L)

    c = irfft(F * F, size)[:L]

def _build_lp_matrix(a, n):
    """Build sparse matrix A where (A @ b)[j] = (a*b)[j] for full convolution."""
    # Column l has entries a[0], a[1], ..., a[min(n-1, 2n-2-l)] at rows l, l+1, ...
    # For l in [0, n-1]: valid rows are j = l to min(l+n-1, 2n-2)
    # A[j, l] = a[j - l]
    total = n * n  # each column has exactly n entries (since 2n-1-l >= n for l <= n-1)
    rows = np.zeros(total, dtype=np.int32)
    cols = np.zeros(total, dtype=np.int32)
    data = np.zeros(total, dtype=np.float64)
    offset = 0
    for l in range(n):

        rows[offset:offset + L_col] = np.arange(l, l + L_col, dtype=np.int32)
        cols[offset:offset + L_col] = l
        data[offset:offset + L_col] = a[:L_col]
        offset += L_col
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A

            if flat_count >= 3 and time.monotonic() < deadline - 80.0:
                sigma = 0.004 + 0.002 * min(flat_count, 20)
                noise = rng.normal(0.0, sigma, n)
                p *= np.exp(noise)
                p = np.maximum(p, 0.0)
                s2 = float(p.sum())
                if s2 > 1e-10:
                    p /= s2
                    S = 1.0
                    c = _conv_self(p)
                flat_count = 0
            if flat_count >= 20:
                break

        # Periodic LP update (every 50 sweeps, only if plenty of time left)
        if (HAS_CVXPY and sweep % 50 == 0 and
                time.monotonic() < deadline - 120.0 and remaining > 250):
            M = float(c.max())
            b_lp = _try_lp(p, n, M, time_limit=min(40.0, remaining * 0.1))
            if b_lp is not None:
                c_cur = _conv_self(p)
                sc_cur = 2.0 * n * float(c_cur.max())
                for t_blend in [0.3, 0.5, 0.7]:
                    p_try = (1.0 - t_blend) * p + t_blend * b_lp
                    p_try = np.maximum(p_try, 0.0)
                    s_try = float(p_try.sum())
                    if s_try > 1e-10:
                        p_try /= s_try
                        c_try = _conv_self(p_try)
                        sc_try = 2.0 * n * float(c_try.max())
                        if sc_try < best_score - 1e-13:
                            best_score = sc_try
                            best_p = p_try.copy()
                        if sc_try < sc_cur:
                            p = p_try.copy()
                            c = c_try
                            S = 1.0
                            break

    # Final validation with exact evaluator
    out = np.clip(best_p, 0.0, 1000.0)
    out = np.where(np.isfinite(out), out, 0.0)
    if float(out.sum()) < 0.01:
        out = np.ones(n) / n

    ls = [float(x) for x in out]
    try:
        sc = float(evaluate_sequence(ls))
        if not math.isfinite(sc):
            try:
                pp = np.asarray(height_sequence_1, dtype=np.float64).ravel()
                pp = np.clip(pp, 0.0, 1000.0)
                ls = [float(x) for x in pp]
            except:
                ls = [1.0]
    except:
        pass

    return ls
