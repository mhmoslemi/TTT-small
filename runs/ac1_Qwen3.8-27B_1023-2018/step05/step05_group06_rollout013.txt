                if math.isfinite(best_val) and best_val < cur_sc - 1e-16 and best_x >= 0.0:
                    w = best_x - v
                    if abs(w) > 1e-18:
                        c[start:end] += w * infl
                        if has_diag:
                            c[start + diag_offset] += w * w
                        p[i] = best_x
                        S += w
                        improved = True

            # End of sweep: renormalize
            if S > 1e-300 and math.isfinite(S):
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0

            # Periodic exact recompute
            if sweep % 4 == 0:
                c = conv_self(p)

            new_sc = float(2.0 * n * np.max(c))
            if math.isfinite(new_sc) and new_sc < best_score - 1e-14:
                best_p = p.copy()
                best_score = new_sc

            if improved:
                flat = 0
            else:
                flat += 1
                if flat >= 4 and time.monotonic() < stop_time - 30.0:
                    sigma = perturb_sigma * min(1.0 + 0.3 * flat, 3.0)
                    p *= np.exp(rng_cd.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 1e-15:
                        p /= ss
                        S = 1.0
                        c = conv_self(p)
                        flat = 0
                    else:
                        break

            if flat >= 25:
                break

        p = np.maximum(p, 0.0)
        ss = float(p.sum())
        if ss > 1e-300:
            p /= ss
        return p, float(2.0 * n * np.max(conv_self(p)))

    def _subgradient_step(p, stop_time, n_iter=200):
        """Subgradient descent on max(conv) with projection onto simplex-like set."""
        p = p.copy().astype(np.float64)
        S = float(p.sum())
        if S < 1e-300:
            return p
        p /= S
        best_local = p.copy()
        best_local_sc = eval_score_normalized(p)

        for it in range(n_iter):
            if time.monotonic() > stop_time - 3.0:
                break
            c = conv_self(p)
            cmax = float(c.max())
            # Active lags: where c is within 5% of max
            thresh = 0.95 * cmax
            active = np.where(c >= thresh)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(c))])

            # Subgradient: for each active lag k, grad_i += 2*a[k-i]/len(active)
            grad = np.zeros(n)
            for k in active:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                indices = np.arange(lo, hi + 1)
                pairs = k - indices  # k - i for i in indices
                grad[indices] += 2.0 * p[pairs]
            grad /= len(active)

            # Step size
            sc = float(2.0 * n * cmax)
            lr = 0.1 * (1.0 / it + 0.1) if it > 0 else 0.1
            p -= lr * grad

            # Project: clip to [0, inf] and renormalize
            p = np.maximum(p, 0.0)
            s = float(p.sum())
            if s < 1e-15:
                p = best_local.copy()
                break
            p /= s

            sc_new = eval_score_normalized(p)
            if math.isfinite(sc_new) and sc_new < best_local_sc:
                best_local = p.copy()
                best_local_sc = sc_new

        return best_local

    # === Main optimization loop ===
    p_work = p.copy()

    # Phase 1: Coordinate descent (main workhorse)
    cd1_stop = min(hard - 200.0, t0 + 600.0)
    if cd1_stop > t0 + 60.0:
        try:
            p1, sc1 = _coord_descent(p_work, cd1_stop, seed_cd=seed + 111, G=36)
            if p1 is not None and len(p1) == n and np.all(np.isfinite(p1)) and p1.sum() > 1e-12:
                p_work = p1 / float(p1.sum())
                consider(p_work)
        except Exception:
            pass

    # Restore best if we degraded
    if eval_score_normalized(best_p) < eval_score_normalized(p_work) - 1e-10:
        p_work = best_p.copy()

    # Phase 2: Subgradient refinement on active lags
    sg_stop = min(hard - 100.0, t0 + 750.0)
    if sg_stop > time.monotonic() + 30.0:
        try:
            p2 = _subgradient_step(p_work, sg_stop, n_iter=500)
            if p2 is not None and len(p2) == n and np.all(np.isfinite(p2)) and p2.sum() > 1e-12:
                p2 = p2 / float(p2.sum())
                consider(p2)
        except Exception:
            pass

    # Phase 3: Second round of coordinate descent (finer)
    cd2_stop = min(hard - 60.0, t0 + 900.0)
    if cd2_stop > time.monotonic() + 40.0:
        try:
            p3, sc3 = _coord_descent(p_work, cd2_stop, seed_cd=seed + 7777, G=48, perturb_sigma=0.018)
            if p3 is not None and len(p3) == n and np.all(np.isfinite(p3)) and p3.sum() > 1e-12:
                p_work = p3 / float(p3.sum())
                consider(p_work)
        except Exception:
            pass

    # Restore best if degraded
    if eval_score_normalized(best_p) < eval_score_normalized(p_work) - 1e-10:
        p_work = best_p.copy()

    # Phase 4: Final polished coordinate descent
    cd3_stop = hard - 30.0
    if cd3_stop > time.monotonic() + 30.0:
        try:
            p4, sc4 = _coord_descent(p_work, cd3_stop, seed_cd=seed + 31337, G=56, perturb_sigma=0.012)
            if p4 is not None and len(p4) == n and np.all(np.isfinite(p4)) and p4.sum() > 1e-12:
                p_work = p4 / float(p4.sum())
                consider(p_work)
        except Exception:
            pass

    # Final: ensure we return the best found
    final = best_p.copy()
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)
    fs = float(final.sum())
    if fs < 0.01:
        final = np.full(n, 1.0 / n)
    else:
        final = final / fs

    # Exact final validation
    ls = [float(x) for x in final]
    try:
        final_sc = evaluate_sequence(ls)
    except Exception:
        final_sc = math.inf

    if not math.isfinite(final_sc):
        # Fallback to parent
        try:
            pp = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
            pp = np.clip(pp, 0.0, 1000.0)
            pp = pp / float(pp.sum())
            ls = [float(x) for x in pp]
            final_sc = evaluate_sequence(ls)
        except Exception:
            return [1.0]
    if not math.isfinite(final_sc):
        return [1.0]

    return ls
