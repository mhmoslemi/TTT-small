import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:
        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(3072, dtype=np.float64)
    return p


def _conv_self(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _score_from_conv(p, c):
    n = p.size
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _coord_descent(p, c, S_total, n, rng, stop_time, G=45, gs_iters=30):
    """Run coordinate descent with incremental convolution updates."""
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    flat_count = 0
    max_flat = 5
    sweep = 0

    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            if S_total <= 1e-300:
                break

            # Periodic renormalization to prevent drift
            if S_total > 1.2 or S_total < 0.8:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0

            v = float(p[i])
            start = i
            end = min(i + n, 2 * n - 1)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end]
            infl = 2.0 * p[:Ls]

            # Maximum of convolution outside [start, end)
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < 2 * n - 1:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

            # Search bounds

            # Grid evaluation
            xs = np.linspace(x_lo, x_hi, G + 1)

            ns = xs.size

            w = xs - v
            denom = S_total + w
            valid = denom > 1e-15

            # Vectorized: aff[j, k] = base[k] + w[j] * infl[k]
            aff = base[None, :] + w[:, None] * infl[None, :]

            # Add w^2 at diagonal position
            di = 2 * i - start
            if 0 <= di < Ls:
                aff[:, di] += w * w

            maff = np.max(aff, axis=1)
            if max_const > 0.0:
                maff = np.maximum(maff, max_const)

            vals = np.full(ns, math.inf)
            if np.any(valid):
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)

            if not np.any(np.isfinite(vals)):
                continue

            imin = int(np.argmin(vals))
            best_x = float(xs[imin])
            best_val = float(vals[imin])

            # Golden section refinement around grid minimum
            if best_val < cur_score - 1e-16:
                dw = x_hi / G
                lo = max(0.0, best_x - 2.5 * dw)

                if hi - lo > 1e-15:
                    gr = (math.sqrt(5.0) - 1.0) / 2.0
                    a_gs = hi - gr * (hi - lo)
                    b_gs = lo + gr * (hi - lo)

                    def eval_one(x0):
                        w0 = x0 - v
                        d0 = S_total + w0
                        if d0 <= 1e-15:
                            return math.inf
                        aff0 = base + w0 * infl
                        if 0 <= di < Ls:
                            aff0[di] += w0 * w0
                        m0 = float(np.max(aff0))
                        if max_const > m0:
                            m0 = max_const
                        return (2.0 * n * m0) / (d0 * d0)

                    fa = eval_one(a_gs)
                    fb = eval_one(b_gs)
                    for _ in range(gs_iters):
                        if hi - lo < 1e-16:

                        if fa < fb:

                            b_gs = a_gs
                            fb = fa
                            a_gs = hi - gr * (hi - lo)
                            fa = eval_one(a_gs)
                        else:
                            lo = a_gs
                            a_gs = b_gs
                            fa = fb
                            b_gs = lo + gr * (hi - lo)
                            fb = eval_one(b_gs)

                    gs_x = (lo + hi) * 0.5

                    if math.isfinite(gv) and gv < best_val:
                        best_x = gs_x

            p *= inv
            c *= inv * inv
            S_total = 1.0

        if sweep % 8 == 7:
            c = _conv_self(p)
            cur_score = float(2.0 * n * c.max())

        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = p.copy()

        if improved:
            flat_count = 0
        else:
            flat_count += 1

        if flat_count >= max_flat:
            if time.monotonic() < stop_time - 5.0:
                amt = 0.015 + 0.01 * math.exp(-flat_count * 0.15)
                p = np.maximum(p * np.exp(rng.normal(0.0, amt, n)), 0.0)
                ss = float(p.sum())
                if ss > 0:
                    p /= ss
                    S_total = 1.0
                else:
                    break
                c = _conv_self(p)
                cur_score = float(2.0 * n * c.max())
                flat_count = 0
            else:
                break

        sweep += 1

    return best_p, best_score, p, c, S_total
