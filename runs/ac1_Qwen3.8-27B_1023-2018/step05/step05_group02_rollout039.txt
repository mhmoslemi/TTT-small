        # Actually: sum_k w[k] * p[k-j] where 0<=k-j<n means j<=k<=j+n-1
        # = sum_{m=0}^{n-1} w[m+j] * p[m] (where m+j < conv_len)
        # This equals the (n-1+j)-th element of np.convolve(w, p)
        # np.convolve(w, p)[j + (n-1)] = sum_m w[m] * p[j+n-1-m]... 
        # Hmm, let me just use the direct formula
        # w has length conv_len = 2n-1
        # For each j: grad[j] = 2*T * dot(w[j:j+n], p)
        
        # Vectorized: create matrix W where W[j, m] = w[j+m]
        # That's a (n x n) Hankel matrix. For n=3072, that's 9.4M entries = 75MB.
        # Might be too much. Let's use FFT.
        
        # Alternative: grad[j] = 2T * (w * p)[j+n-1] where * denotes cross-correlation
        # Using FFT: if we compute the convolution w_conv_p = np.convolve(w, p), then
        # (w * p)[j + n - 1] = w_conv_p[j + n - 1 + (n-1)] ... this is getting confusing.
        
        # Let me just use the property: sum_k w[k] * p[k-j] = <w[j:], p> (inner product)
        # = sum_{m=0}^{n-1} w[j+m] * p[m]
        # This is the j-th "lag" of the cross-correlation of w and p.
        # Using FFT: cross_corr = irfft(fft(w, L) * conjugate(fft(p, L)))
        # where L >= len(w) + n - 1... 
        # Actually, let's just use a simple approach:
        # grad = 2T * (cross-correlation of w and p, offset appropriately)
        
        L2 = next_fast_len(2 * n)
        Wf = rfft(w, L2)
        Pf = rfft(p, L2)
        cc = irfft(Wf * np.conj(Pf), L2)
        # cc[k] = sum_m w[m] * p[m-k] ... depending on convention
        # Actually irfft(FFT(w) * conj(FFT(p))) gives the cross-correlation
        # cc[k] = sum_m w[m+k] * p[m] (for the appropriate shift)
        # The standard result: if C = ifft(FFT(w) * conj(FFT(p))), then
        # C[k] = sum_m w[(m+k) mod L] * p[m]
        # For our purposes: we want sum_m w[m+j] * p[m] for m=0..n-1
        # This is C[j] if L is large enough (no wrap-around)
        # With L=2n, and w having length 2n-1, p having length n:
        # C[j] = sum_m w[m+j] * p[m] for m such that m+j < 2n-1 and m < n
        # = sum_{m=0}^{n-1} w[m+j] * p[m] (since j < n, m+j < n + n-1 = 2n-1)
        # Perfect, so C[j] is exactly what we want.
        
        grad = 2.0 * T * cc[:n]
        grad = np.where(np.isfinite(grad), grad, 0.0)
        
        # Normalization gradient: we want to minimize f subject to sum(p)=1
        # Using softmax parameterization, the constraint is automatically satisfied
        # But we need the gradient w.r.t. u (the log parameters)
        # p = softmax(u), so dp_j/du_k = p_j * (delta_{jk} - p_k)
        # du_k gradient = sum_j (df/dp_j) * dp_j/du_k = p_k * (sum_j df/dp_j * p_j... 
        # Wait: sum_j grad_p[j] * dp_j/du_k = sum_j grad_p[j] * p_j * (delta_{jk} - p_k)
        # = p_k * grad_p[k] - p_k * sum_j grad_p[j] * p_j
        # = p_k * (grad_p[k] - dot(p, grad_p))
        
        gp = grad
        tdp = float(np.dot(pp, gp))
        grad_u = pp * (gp - tdp)
        grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
        return float(f), grad_u

    u = p_to_u(p)
    c0 = cfast(p)
    cmax0 = float(c0.max())
    
    stages = [(0.01, 0.25), (0.003, 0.35), (0.0008, 0.40)]
    start_t = time.monotonic()
    avail = max(0.0, float(stop_time) - start_t - 3.0)
    
    cum = 0.0
    total_w = sum(w2 for _, w2 in stages)
    best_u = u.copy()
    best_f = 1e100
    
    for frac, wt in stages:
        cum += wt
        end_stage = min(float(stop_time) - 2.0, start_t + avail * (cum / total_w))
        if end_stage <= time.monotonic() + 3.0:
            continue
        T = max(1e-300, frac * cmax0)
        maxit = 40
        opts = {"maxiter": maxit, "gtol": 1e-14, "ftol": 1e-16, "maxls": 25,
                "maxcor": 25, "maxdisp": 0, "disp": False}
        try:
            f0, _ = lse_obj(best_u, T)
        except Exception:
            continue
        if not math.isfinite(f0):
            continue
        best_f_local = f0
        while time.monotonic() < end_stage - 1.0:
            try:
                res = minimize(lse_obj, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and res.x.size == u.size and np.all(np.isfinite(res.x)):
                try:
                    fval = float(res.fun)
                except Exception:
                    fval = math.inf
                if math.isfinite(fval) and fval < best_f_local - 1e-15:
                    best_f_local = fval
                    best_u = res.x.copy()
                else:
                    break
            if getattr(res, "status", 1) == 0:
                break
        p_cur = softmax(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        if consider is not None:
            try:
                consider(p_cur)
            except Exception:
                pass
        c_cur = cfast(p_cur)
        cmax0 = float(c_cur.max()) if c_cur.size else cmax0

    p_final = softmax(best_u)
    p_final = np.maximum(p_final, 0.0)
    ss = float(p_final.sum())
    if ss > 1e-300:
        p_final /= ss
    return p_final


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.full(n, 1.0 / n)
    m = src.size
    if m == n:
        out = src.copy()
    elif m == 1:
        out = np.full(n, float(src[0]) / n)
    else:
        xp = (np.arange(m) + 0.5) / m
        fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
        out = np.maximum(fp, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 25.0

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float(_score_exact(best_arr))

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64) / 64.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            sc = float(2.0 * a.size * float(np.convolve(a, a).max()) / (s * s))
            if math.isfinite(sc) and sc < best_score - 1e-14:
                a = a / s
                best_arr = a.copy()
                best_score = sc
        except Exception:
            pass

    rng = np.random.default_rng(seed)

    # Phase 1: Coordinate descent on parent (majority of time)
    n_parent = int(best_arr.size)
    phase1_end = t0 + min(budget * 0.55, 700.0)
    phase1_end = min(phase1_end, hard - 200.0)

    if phase1_end > t0 + 30.0:
        rng1 = np.random.default_rng(seed + 1)
        p1, _ = _coord_descent(best_arr, phase1_end, rng1, G=32, consider=consider)
        if p1 is not None and p1.size == n_parent and np.all(np.isfinite(p1)) and p1.sum() > 0:
            consider(p1)

    # Phase 2: L-BFGS refinement
    phase2_end = min(hard - 100.0, t0 + budget * 0.75)
    now = time.monotonic()
    if phase2_end > now + 40.0 and n_parent <= 5000:
        try:
            p2 = _lbfgs_refine(best_arr, phase2_end, consider=consider)
            if p2 is not None and p2.size == n_parent and np.all(np.isfinite(p2)) and p2.sum() > 0:
                consider(p2)
        except Exception:
            pass

    # Phase 3: More coordinate descent on best
    phase3_end = min(hard - 30.0, t0 + budget * 0.92)
    now = time.monotonic()
    if phase3_end > now + 30.0:
        rng3 = np.random.default_rng(seed + 999)
        p3, _ = _coord_descent(best_arr, phase3_end, rng3, G=28, consider=consider)
        if p3 is not None and p3.size == n_parent and np.all(np.isfinite(p3)) and p3.sum() > 0:
            consider(p3)

    # Phase 4: Try a different length if time remains
    now = time.monotonic()
    if hard - now > 80.0:
        for alt_n in [2048, 1536]:
            if hard - time.monotonic() < 50.0:
                break
            alt_p = _resample(best_arr, alt_n)
            consider(alt_p)
            rng4 = np.random.default_rng(seed + alt_n)
            alt_end = min(hard - 20.0, time.monotonic() + 40.0)
            p_alt, _ = _coord_descent(alt_p, alt_end, rng4, G=28, consider=consider)
            if p_alt is not None and np.all(np.isfinite(p_alt)) and p_alt.sum() > 0:
                consider(p_alt)

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
