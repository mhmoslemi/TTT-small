def _conv_self(a):
    """FFT-based self-convolution."""
    n = a.size
    L = next_fast_len(2*n - 1)
    F = rfft(a, L)
    c = irfft(F * F, L)[:2*n-1]
    np.maximum(c, 0.0, out=c)  # numerical cleanup

def _score(a):
