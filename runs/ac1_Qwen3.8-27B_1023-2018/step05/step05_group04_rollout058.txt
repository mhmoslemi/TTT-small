Let me also think about the LSE approach. The key issue with LSE is that the gradient is:
`∇_u log(sum_j exp(b_j(u)/T)) = (1/T) * (B^T w) / (sum w)`
where `w_j = exp(b_j/T) / sum exp(b_k/T)` and `B` is the Jacobian of `b` with respect to `a`.

For the convolution, `∂b_j/∂a_i = a_{j-i} + a_{j-i}` (the two terms I identified earlier) = `2*a_{j-i}` (for `j≠2i`) or `2*a_i + 2w` (at the diagonal, but in the gradient evaluation, it's `2*a_i`).

Wait, `∂b_j/∂a_i`:
- If `0 <= j-i < n`: `b_j` contains `a_i * a_{j-i}` (as first factor) and `a_{j-i} * a_i` (as second factor)
- Actually, in the sum `b_j = sum_k a_k * a_{j-k}`:
  - Term `k=i`: `a_i * a_{j-i}`, derivative w.r.t. `a_i` is `a_{j-i}`
  - Term `k=j-i`: `a_{j-i} * a_i`, derivative w.r.t. `a_i` is `a_{j-i}`
  - (These are the same term only when `i = j-i`, i.e., `j = 2i`)
- So `∂b_j/∂a_i = 2*a_{j-i}` if `0 <= j-i < n`, else 0
- At `j = 2i`: `∂b_j/∂a_i = 2*a_i` (both terms are `a_i * a_i`, derivative is `2*a_i`)

This is consistent. The Jacobian `J_{j,i} = 2*a_{j-i}` (for valid indices).

In the LSE code, the gradient computation uses:

tdp = float(np.dot(pp, gdp))
grad = pp * (gdp - tdp)

The `xfast` function computes the cross-correlation of `pp[::-1]` with `w`, taking the relevant slice. This gives `q_i = sum_j w_j * pp[j-i]` (or similar). Then `gdp_i = 2*q_i` and `grad_i = pp_i * (gdp_i - tdp)`.

The `tdp` term accounts for the normalization constraint `sum pp = 1` (projecting the gradient onto the tangent space of the simplex).

            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = 1e-4 * cmax
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return f, grad

    stages = [(0.2, 0.10), (0.05, 0.15), (0.012, 0.20), (0.003, 0.25)]
    best_p = p.copy()
    best_sc = _score_arr(p)
    start = time.monotonic()
    avail = max(0.0, stop_time - start - 10.0)
    total_w = sum(w for _, w in stages)
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 8.0:
            break
        cum += wt
        end_stage = min(stop_time - 6.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)
        best_u = u.copy()
        stale = 0
        while time.monotonic() < end_stage - 1.5:
            maxit = 35 if n <= 2000 else 25
            opts = {"maxiter": maxit, "gtol": 1e-11, "ftol": 1e-15, "maxls": 12, "maxdisp": 0}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_u is None and False:
                    pass
                if math.isfinite(fval):
                    p_cur = softmax(res.x)
                    p_cur = np.maximum(p_cur, 0.0)
                    sc = _score_arr(p_cur)
                    if math.isfinite(sc) and sc < best_sc - 1e-13:
                        best_sc = sc
                        best_p = p_cur.copy()
                        p = p_cur.copy()
                        u = np.log(np.maximum(p, 1e-30))
                        u -= float(u.mean())
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
                best_u = res.x.copy()
            if getattr(res, "status", 1) == 0:
                break
    if consider:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p

def _cd_optimize(p, stop_time, consider=None, seed=123):
    n = int(p.size)
    if n == 0:
        return p, float('inf')
    p = _normalize(p.copy())
    c = _conv_self(p)
    cur = float(2.0 * n * c.max())
    best_sc = cur
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    Ltot = 2 * n - 1
    flat = 0
    sweep = 0
    GR = (math.sqrt(5.0) - 1.0) / 2.0
    G = 40

    def _eval1d(w, base, infl, diag, max_const, n):
        s_new = 1.0 + w
        if s_new < 1e-15:
            return float('inf')
        aff_max = float(np.max(base + w * infl))
        if 0 <= diag < len(base):
            dval = base[diag] + w * infl[diag] + w * w
            if dval > aff_max:
                aff_max = dval
        m = aff_max if aff_max > max_const else max_const
        return 2.0 * n * m / (s_new * s_new)

    max_sweeps = 100000
    while sweep < max_sweeps and time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 1.0:
                break
            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue
            base = c[start_i:end_i]
            infl = 2.0 * p[:Ls]
            diag = i

            max_const = 0.0
            if start_i > 0:
                mc = float(c[:start_i].max())
                if mc > max_const:
                    max_const = mc
            if end_i < Ltot:
                mc = float(c[end_i:].max())
                if mc > max_const:
                    max_const = mc

            x_hi = max(30.0 * v, 20.0 / n, 0.01)
            x_hi = min(x_hi, 0.5)
            x_hi = max(x_hi, 1e-8)

            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 < v < x_hi:
                ins = np.searchsorted(pts, v)
                pts = np.insert(pts, ins, v)

            ws = pts - v
            s_new = 1.0 + ws
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag < Ls:
                aff[:, diag] += ws * ws
            maff = np.max(aff, axis=1)
            if max_const > 0:
                maff = np.maximum(maff, max_const)
            valid = s_new > 1e-12
            scores = np.full(pts.shape, 1e100, dtype=np.float64)
            if np.any(valid):
                scores[valid] = 2.0 * n * maff[valid] / (s_new[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            # Terzen section refinement
            if 0 < imin < len(pts) - 1:
                lo = float(pts[max(0, min(imin - 2, len(pts)-1))])
                hi = float(pts[min(len(pts) - 1, imin + 2)])
            else:
                lo, hi = 0.0, x_hi
            if hi - lo < 1e-14:
                lo = max(0.0, best_x - 1e-6)
                hi = best_x + 1e-6

            a_gs, b_gs = lo, hi
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)
            f_c = _eval1d(c_gs - v, base, infl, diag, max_const, n)
            f_d = _eval1d(d_gs - v, base, infl, diag, max_const, n)
            for _ in range(20):
                if b_gs - a_gs < 1e-12 * max(1.0, abs(best_x)):
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = _eval1d(c_gs - v, base, infl, diag, max_const, n)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_d = _eval1d(d_gs - v, base, infl, diag, max_const, n)
            x_mid = (a_gs + b_gs) / 2.0
            f_mid = _eval1d(x_mid - v, base, infl, diag, max_const, n)
            if f_mid < best_val:
                best_x, best_val = x_mid, f_mid
            for xp in (0.0, x_hi):
                fp = _eval1d(xp - v, base, infl, diag, max_const, n)
                if fp < best_val:
                    best_x, best_val = xp, fp

            if math.isfinite(best_val) and best_val < cur - 1e-14 and 0.0 <= best_x:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    aff_seg = base + w * infl
                    if 0 <= diag < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag] += w * w
                    c[start_i:end_i] = aff_seg
                    new_max_seg = float(aff_seg.max())
                    cur = 2.0 * n * max(new_max_seg, max_const)
                    improved = True

        s_new = float(p.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            p /= s_new
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_sc - 1e-15:
                best_sc = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 12.0:
                    strategy = rng.integers(4)
                    if strategy == 0:
                        kick = 0.02 + 0.04 * min(1.0, flat / 15.0)
                        p = p * np.exp(rng.normal(0, kick, n))
                    elif strategy == 1:
                        p = p + 0.005 * rng.standard_normal(n)
                    elif strategy == 2:
                        mask = rng.random(n) < 0.1
                        p[mask] *= (0.3 + 0.4 * rng.random(mask.sum()))
                    else:
                        phase = rng.uniform(0, 2 * math.pi)
                        freq = rng.uniform(0.5, 3.0)
                        p = p + 0.01 * np.sin(2 * math.pi * freq * np.arange(n) / n + phase)
                    p = _normalize(p)
                    c = _conv_self(p)
                    cur = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = _normalize(p)
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_sc

def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m) + 0.5) / m
        fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n)
    return out / s
