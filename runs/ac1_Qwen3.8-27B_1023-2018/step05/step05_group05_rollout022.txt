def _lp_direction(a, M, seed):
    """Try LP-based direction find. Returns None on failure."""
    n = a.size
    conv_len = 2 * n - 1
    # Build constraint matrix A: A[k,j] = a[|k-j|] if |k-j|<n else 0
    # (a*g)[k] = sum_j a[j] * g[k-j]
    # A[k, j] = a[k-j] if 0<=k-j<n, else 0
    try:
        # Build sparse
        from scipy.sparse import csr_matrix
        rows = []
        cols = []
        vals = []
        for j in range(n):
            # (a*g)[k] has term a[j]*g[k-j] for k in [j, j+n-1]
            ks = np.arange(j, j + n)
            idx = np.arange(n)
            rows.extend(ks)
            cols.extend([j] * n)
            vals.extend(a[idx])
        A = csr_matrix((vals, (rows, cols)), shape=(conv_len, n))
        from scipy.optimize import linprog
        c_obj = -np.ones(n)
        b_ub = np.full(conv_len, M)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None
