def _lse_optimize(p, stop_time, consider=None):

        e = np.exp(uu, dtype=np.float64)
        return e / e.sum()

    u0 = np.log(np.maximum(p, 1e-300))
    u0 = u0 - u0.mean()

    L = 2 * n - 1
    conv_size = next_fast_len(L)
    grad_size = next_fast_len(3 * n)

    def obj(uu):

        F = rfft(pp, conv_size)
        c = irfft(F * F, conv_size)[:L]
        np.maximum(c, 0.0, out=c)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0.0:
            return 1e100, np.zeros_like(uu)
        T = max(0.002 * cmax, 1e-8)

        exps = np.exp(scaled, dtype=np.float64)

        if sum_e > 0 and math.isfinite(sum_e):
            f = cmax + T * math.log(sum_e)
        else:
            f = cmax

        Fw = rfft(w, grad_size)

    best_u = u0.copy()

def _lp_diversify(f, stop_time, consider=None):
    n = int(f.size)

        return None
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        A = np.zeros((L := 2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f[j]
        b = np.full(L, M, dtype=np.float64)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= b])
        t_start = time.monotonic()
        solved = False
        for slv in [cp.ECOS, cp.SCS, cp.CLP]:
            try:

                break

                continue

    if ws > 0:
        work /= ws

    # Phase 5: Final LSE + coordinate (remaining time minus safety)
    p5_end = min(hard - 15.0, t0 + budget - 15.0)
    if p5_end > time.monotonic() + 15.0:
        lse_end = min(p5_end - 15.0, time.monotonic() + 0.3 * (p5_end - time.monotonic()))
        if lse_end > time.monotonic() + 10.0:
            try:
                p5 = _lse_optimize(work, lse_end, consider=consider)
                if p5 is not None and p5.size == n:
                    consider(p5)
                    work = best_arr.copy()
                    ws = float(work.sum())
                    if ws > 0:
                        work /= ws
            except Exception:
                pass

        coord_end = p5_end
        if coord_end > time.monotonic() + 10.0:
            try:
                work, _ = _coord_descent(work, coord_end, seed=int(seed + 99999),
                                         consider=consider, max_sweeps=50000)
            except Exception:
                pass

    # Final blending attempts
    best_n = int(best_arr.size)
    if best_n > 0:
        for frac in [0.05, 0.1, 0.2]:
            blended = (1.0 - frac) * best_arr + frac * (1.0 / best_n)
            blended = np.maximum(blended, 0.0)
            bs = float(blended.sum())
            if bs > 0:
                blended /= bs
                consider(blended)

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0:
            out = np.full(n, 1.0 / n, dtype=np.float64)
        so = float(out.sum())
        if so < 0.01:
            out = np.full(n, 1.0 / n, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
