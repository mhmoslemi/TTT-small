        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0:
            T = 1e-4 * cmax
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        return float(f), np.where(np.isfinite(grad), grad, 0.0)

    def p_to_u(pp):
        ll = np.log(np.maximum(pp, 1e-300))
        return ll - float(np.mean(ll))

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = 2.0 * n * float(cfast(p).max())

    stages = ((0.1, 0.3), (0.04, 0.3), (0.01, 0.4))
    for frac, wt in stages:
        if time.monotonic() > stop_time - 5.0:
            break
        cmax = float(cfast(p).max())
        T = max(1e-15, frac * cmax)
        stale = 0
        while time.monotonic() < stop_time - 4.0 and stale < 3:
            opts = {"maxiter": 25, "gtol": 1e-11, "ftol": 1e-15,
                    "maxls": 20, "maxcor": 20, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                p_cur = _softmax(res.x)
                c_cur = cfast(p_cur)
                sc_cur = 2.0 * n * float(c_cur.max())
                if math.isfinite(sc_cur) and sc_cur < best_sc:
                    best_sc = sc_cur
                    best_p = p_cur.copy()
                    p = p_cur.copy()
                    u = p_to_u(p)
                    stale = 0
                else:
                    stale += 1
            if getattr(res, "status", 1) == 0:
                break

    ss = float(best_p.sum())
    if ss > 1e-300:
        best_p = best_p / ss
    return best_p


def _coord_descent(p, stop_time, consider, seed=42):
    """Fast coordinate descent with periodic verification and perturbation."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    S = float(p.sum())
    if S < 1e-300 or n == 0:
        return p, math.inf
    p = p / S
    S = 1.0
    L = 2 * n - 1
    b = _conv_self(p)
    cur_score = 2.0 * n * float(b.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    G1 = 40
    G2 = 25
    flat = 0
    sweeps = 0

    while time.monotonic() < stop_time - 5.0:
        # Prefix and suffix max for unaffected regions
        prefix_max = np.maximum.accumulate(b)
        suffix_max = np.flip(np.maximum.accumulate(np.flip(b)))

        max_unaff = np.zeros(n)
        if n > 1:
            max_unaff[1:] = prefix_max[:n - 1]
        right_parts = np.zeros(n)
        if n > 1:
            right_parts[:n - 1] = suffix_max[n:2 * n - 1]
        max_unaff = np.maximum(max_unaff, right_parts)

        sweep_improved = False
        order = rng.permutation(n)

        for ii in range(n):
            i = int(order[ii])
            if time.monotonic() > stop_time - 2.0:
                break

            v = p[i]

            base = b[i:i + n]
            infl = 2.0 * p
            mc = max_unaff[i]
            quad_idx = i

            if v < 1e-14:
                # Small positive trial
                delta = 0.3 / n
                aff_trial = base + delta * infl
                aff_trial[i] += delta * delta
                new_max = max(float(aff_trial.max()), mc)
                new_score = 2.0 * n * new_max / ((S + delta) ** 2)
                if new_score < cur_score * (1.0 - 1e-12):
                    p[i] = delta
                    S += delta
                    b[i:i + n] += delta * infl
                    b[2 * i] += delta * delta
                    cur_score = new_score
                    sweep_improved = True
                continue

            # Grid search
            d_hi = max(3.0 * v, 5.0 / n)
            d_hi = min(d_hi, 3.0 * S / n + v)
            if d_hi < 1e-14:
                d_hi = 1.0 / n

            deltas = np.linspace(-v, d_hi, G1)
            aff = base[None, :] + deltas[:, None] * infl[None, :]
            aff[:, quad_idx] += deltas ** 2
            max_aff = np.maximum(aff.max(axis=1), mc)
            denom = (S + deltas) ** 2
            valid = denom > 1e-30
            scores = np.full(G1, np.inf)
            scores[valid] = 2.0 * n * max_aff[valid] / denom[valid]

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_d = float(deltas[imin])
            best_f = float(scores[imin])

            # Fine search
            dw = (d_hi + v) / G1
            lo = max(-v, best_d - 3 * dw)
            hi = min(d_hi, best_d + 3 * dw)
            if hi - lo > 1e-12:
                df = np.linspace(lo, hi, G2)
                aff_f = base[None, :] + df[:, None] * infl[None, :]
                aff_f[:, quad_idx] += df ** 2
                max_aff_f = np.maximum(aff_f.max(axis=1), mc)
                denom_f = (S + df) ** 2
                valid_f = denom_f > 1e-30
                scores_f = np.full(G2, np.inf)
                scores_f[valid_f] = 2.0 * n * max_aff_f[valid_f] / denom_f[valid_f]
                if np.any(np.isfinite(scores_f)):
                    imf = int(np.argmin(scores_f))
                    if scores_f[imf] < best_f:
                        best_d = float(df[imf])
                        best_f = float(scores_f[imf])

            if math.isfinite(best_f) and best_f < cur_score * (1.0 - 1e-13):
                delta = best_d
                p[i] = v + delta
                S += delta
                b[i:i + n] += delta * infl
                b[2 * i] += delta * delta
                cur_score = best_f
                sweep_improved = True

        # End of sweep
        if sweeps % 30 == 29:
            b = _conv_self(p)
            cur_score = 2.0 * n * float(b.max()) / (S * S)
            if math.isfinite(cur_score) and cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()
                consider(best_p)

        # Rescale if drift
        if S > 1.6 or S < 0.6:
            p = p / S
            b = b / (S * S)
            S = 1.0
            cur_score = 2.0 * n * float(b.max())

        if sweep_improved:
            flat = 0
        else:
            flat += 1
            if flat >= 6 and time.monotonic() < stop_time - 40.0:
                # Perturb to escape
                scale = 0.03 + 0.01 * min(flat, 5)
                p = p * np.exp(rng.normal(0, scale, n))
                p = np.maximum(p, 0.0)
                S = float(p.sum())
                if S > 1e-10:
                    p = p / S
                    S = 1.0
                b = _conv_self(p)
                cur_score = 2.0 * n * float(b.max())
                flat = 0

        if flat >= 12:
            break

        sweeps += 1

    # Final verify
    b = _conv_self(p)
    sc = 2.0 * n * float(b.max()) / (S * S)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        consider(best_p)

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 20.0
    if hard <= t0 + 5.0:
        hard = t0 + max(5.0, budget - 1.0)

    parent = _get_parent()
    best_arr = np.clip(np.where(np.isfinite(parent), parent, 0.0), 0.0, 1000.0).copy()
    best_score = float(evaluate_sequence([float(x) for x in best_arr]))
    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(128)
        best_score = 2.0

    def consider(arr):
        nonlocal best_arr, best_score
        a = np.asarray(arr, dtype=np.float64).ravel()
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0 or float(a.sum()) < 0.01:
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in a]))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score:
            best_arr = a.copy()
            best_score = sc

    n = len(best_arr)
    p = best_arr.copy()
    p = p / float(p.sum())

    # Phase 1: LSE optimization (short, ~15% of budget)
    lse_stop = t0 + min(150.0, 0.15 * budget)
    if lse_stop > t0 + 20.0:
        try:
            p1 = _lse_phase(p, lse_stop, consider)
            if p1 is not None and p1.size == n and np.all(np.isfinite(p1)) and float(p1.sum()) > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
                p = p1.copy()
        except Exception:
            pass

    # Phase 2: Main coordinate descent (~65% of budget)
    coord_stop = hard - 0.25 * (hard - time.monotonic())
    if coord_stop > time.monotonic() + 30.0:
        try:
            p2, _ = _coord_descent(p, coord_stop, consider, seed=int(seed))
            if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and float(p2.sum()) > 0:
                p2 = p2 / float(p2.sum())
                consider(p2)
                p = p2.copy()
        except Exception:
            pass

    # Phase 3: Second round of coordinate descent with different seed
    rem = hard - 15.0 - time.monotonic()
    if rem > 30.0:
        try:
            p3, _ = _coord_descent(p, hard - 15.0, consider, seed=int(seed + 999))
            if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and float(p3.sum()) > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)
                p = p3.copy()
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
