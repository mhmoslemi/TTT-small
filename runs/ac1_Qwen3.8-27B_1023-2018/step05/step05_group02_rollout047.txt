import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize

def _fft_conv_self(a):
    """Self-convolution via FFT, returns array of length 2n-1."""
    n = a.size
    L = 2 * n - 1
    if L <= 4096:
        return np.maximum(np.convolve(a, a), 0.0)
    sz = next_fast_len(L)
    F = rfft(a, sz)
    c = irfft(F * F, sz)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _exact_score_arr(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = np.convolve(a, a)
    m = float(c.max())
    return 2.0 * n * m / (s * s)

def _lse_optimize(p0, n, end_time, consider, fine=False):
    """Optimize using log-sum-exp approximation of max, with L-BFGS-B."""
    p = np.maximum(p0.copy(), 0.0)
    s = float(p.sum())
    if s < 1e-30:
        return p
    p = p / s
    
    # Parameterize as log
    def to_log(x):
        return np.log(np.maximum(x, 1e-30))
    def from_log(u):
        e = np.exp(u - u.max())
        return e / e.sum()
    
    u = to_log(p)
    
    conv_sz = next_fast_len(2 * n - 1)
    cross_sz = next_fast_len(3 * n - 2)
    L_conv = 2 * n - 1
    L_cross = 3 * n - 2
    
    def func_grad(u_var, T):
        pp = from_log(u_var)
        # Self-conv
        F = rfft(pp, conv_sz)
        c = irfft(F * F, conv_sz)[:L_conv]
        np.maximum(c, 0.0, out=c)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(u_var)
        
        # LSE approximation
        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -400, 400)
        exps = np.exp(scaled)
        s_exp = float(exps.sum())
        if s_exp <= 0 or not math.isfinite(s_exp):
            w = np.zeros(L_conv)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * math.log(s_exp / L_conv)
            w = exps / s_exp
        
        # Cross-correlation (gradient)
        Fp = rfft(pp[::-1], cross_sz)
        Gw = rfft(w, cross_sz)
        corr = irfft(Fp * Gw, cross_sz)[:L_cross]
        q = corr[n-1:2*n-1]
        np.maximum(q, 0.0, out=q)
        
        # Gradient: 2*q - dot(pp, 2*q) * pp (for sum-normalized)
        gq = 2.0 * q
        tdp = float(pp @ gq)
        grad = pp * (gq - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        
        return float(f), grad
    
    # Stages

        while time.monotonic() < stage_end - 1.0:
