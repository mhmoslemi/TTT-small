            c_t = _conv_self_fft(a)
            sc = float(2.0 * n * c_t.max())
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_p = a.copy()
                best_score = sc
        except Exception:
            pass

    def run_cd(p, c, stop_time, rng, G, max_flat=30, perturb_sigma_base=0.01,
              recompute_every=5, golden_iters=14, focus_active=False):
        """Efficient coordinate descent for minimizing 2n*max(conv(a,a))."""
        n_local = len(p)
        Ltot_local = 2 * n_local - 1
        S_local = float(p.sum())
        cur_score = float(2.0 * n_local * c.max())
        sweep = 0
        flat = 0

        while sweep < 1000000 and time.monotonic() < stop_time - 5.0:
            sweep += 1
            improved = False

            # Compute prefix/suffix max arrays (O(n) in C)
            pm = np.maximum.accumulate(c)
            sm = np.maximum.accumulate(c[::-1])[::-1]

            # Determine which coordinates to visit
            if focus_active and c.size > 0:
                cmax_local = float(c.max())
                thresh = cmax_local * 0.95
                active_lags = np.where(c >= thresh)[0]
                important = np.zeros(n_local, dtype=bool)
                for lag in active_lags:
                    lo = max(0, int(lag) - n_local + 1)
                    hi = min(n_local - 1, int(lag))
                    important[lo:hi + 1] = True
                imp_count = int(important.sum())
                if imp_count > 0 and imp_count < n_local:
                    # Visit important coords first, then others
                    imp_idx = np.where(important)[0]
                    other_idx = np.where(~important)[0]
                    order = np.concatenate([rng.permutation(imp_idx), rng.permutation(other_idx)])
                else:
                    order = rng.permutation(n_local)
            else:
                order = rng.permutation(n_local)

            for i in order:
                if time.monotonic() > stop_time - 3.0:
                    return p, c, cur_score
                if S_local <= 1e-300:
                    return p, c, cur_score

                # Renormalize if drift is large
                if S_local > 1.4 or S_local < 0.71:
                    inv = 1.0 / S_local
                    p *= inv
                    c *= inv * inv
                    S_local = 1.0
                    cur_score = float(2.0 * n_local * c.max())
                    pm = np.maximum.accumulate(c)
                    sm = np.maximum.accumulate(c[::-1])[::-1]

                v = float(p[i])
                start = i
                end = min(i + n_local, Ltot_local)
                Ls = end - start
                if Ls <= 0:
                    continue

                # Max outside window [start, end)
                max_const = 0.0
                if start > 0 and pm[start - 1] > max_const:
                    max_const = pm[start - 1]
                if end < Ltot_local and sm[end] > max_const:
                    max_const = sm[end]

                infl = 2.0 * p[:Ls]
                base = c[start:end]
                diag_offset = 2 * i - start
                has_diag = (0 <= diag_offset < Ls)

                # Search range
                x_hi = max(10.0 * v, 25.0 / n_local, 0.003)
                x_hi = min(x_hi, 3.0 / (n_local * 0.5) + 0.005)
                x_lo = 0.0

                # Grid search (vectorized)
                pts = np.linspace(x_lo, x_hi, G + 1)
                ws = pts - v
                aff = base[None, :] + ws[:, None] * infl[None, :]
                if has_diag:
                    aff[:, diag_offset] += ws[:, None] * ws[:, None]
                maff = aff.max(axis=1)
                np.maximum(maff, max_const, out=maff)
                denoms = S_local + ws
                vals = np.where(denoms > 1e-15, 2.0 * n_local * maff / (denoms * denoms), math.inf)

                if not np.any(np.isfinite(vals)):
                    continue

                imin = int(np.argmin(vals))
                best_x = float(pts[imin])
                best_val = float(vals[imin])

                # Golden section refinement
                if G > 6:
                    dw = (x_hi - x_lo) / G
                    lo_gs = max(0.0, best_x - 2.0 * dw)
                    hi_gs = min(x_hi, best_x + 2.0 * dw)
                    if hi_gs - lo_gs > 1e-14:
                        gr = (math.sqrt(5.0) - 1.0) / 2.0
                        a_gs = lo_gs
                        b_gs = hi_gs
                        c_gs_v = b_gs - gr * (b_gs - a_gs)
                        d_gs_v = a_gs + gr * (b_gs - a_gs)

                        def _eval_x(xv):
                            wv = xv - v
                            if abs(wv) < 1e-18:
                                nb_max = float(base.max())
                            else:
                                nb = base + wv * infl
                                if has_diag:
                                    nb = nb.copy()
                                    nb[diag_offset] += wv * wv
                                nb_max = float(nb.max())
                            if max_const > nb_max:
                                nb_max = max_const
                            dn = S_local + wv
                            if dn <= 1e-15:
                                return math.inf
                            return 2.0 * n_local * nb_max / (dn * dn)

                        f_c = _eval_x(c_gs_v)
                        f_d = _eval_x(d_gs_v)
                        for _ in range(golden_iters):
                            if f_c < f_d:
                                b_gs = d_gs_v
                                d_gs_v = c_gs_v
                                f_d = f_c
                                c_gs_v = b_gs - gr * (b_gs - a_gs)
                                f_c = _eval_x(c_gs_v)
                            else:
                                a_gs = c_gs_v
                                c_gs_v = d_gs_v
                                f_c = f_d
                                d_gs_v = a_gs + gr * (b_gs - a_gs)
                                f_d = _eval_x(d_gs_v)
                        x_opt = (a_gs + b_gs) / 2.0
                        f_opt = _eval_x(x_opt)
                        if math.isfinite(f_opt) and f_opt < best_val:
                            best_x = x_opt
                            best_val = f_opt

                # Apply update if better
                if math.isfinite(best_val) and best_val < cur_score - 1e-14 and best_x >= 0:
                    w = best_x - v
                    if abs(w) > 1e-18:
                        old_v = v
                        p[i] = best_x
                        S_local += w
                        # Update convolution incrementally
                        a_old = p[:Ls].copy()
                        if i < Ls:
                            a_old[i] = old_v
                        c[start:end] += 2.0 * w * a_old
                        if has_diag:
                            c[start + diag_offset] += w * w
                        cur_score = best_val
                        improved = True

            # End of sweep
            # Renormalize
            if S_local > 1e-300 and math.isfinite(S_local) and abs(S_local - 1.0) > 1e-7:
                inv = 1.0 / S_local
                p *= inv
                c *= inv * inv
                S_local = 1.0
                cur_score = float(2.0 * n_local * c.max())

            # Periodic exact FFT recompute to prevent drift
            if sweep % recompute_every == 0:
                c_exact = _conv_self_fft(p)
                c = c_exact
                cur_score = float(2.0 * n_local * c.max())

            if improved:
                flat = 0
            else:
                flat += 1
                if flat >= perturb_sigma_base and flat < max_flat:
                    # Perturb to escape local minimum
                    sigma = perturb_sigma_base * flat
                    noise = rng.normal(0.0, sigma, n_local)
                    p *= np.exp(noise)
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 1e-300:
                        p /= ss
                        S_local = 1.0
                        c = _conv_self_fft(p)
                        cur_score = float(2.0 * n_local * c.max())
                    flat = 0
                elif flat >= max_flat:
                    break

        return p, c, cur_score

    # ==================== MAIN OPTIMIZATION ====================

    # Phase 1: Coarse coordinate descent (35% of time)
    stop1 = t0 + budget * 0.30
    if time.monotonic() < stop1 - 10:
        try:
            p, c, _ = run_cd(p, c, stop1, rng, G=20, max_flat=25,
                             perturb_sigma_base=0.015, recompute_every=4,
                             golden_iters=10, focus_active=False)
            consider(p)
        except Exception:
            pass

    # Phase 2: Medium coordinate descent (30% of time)
    stop2 = t0 + budget * 0.60
    if time.monotonic() < stop2 - 10:
        try:
            p2 = best_p.copy()
            c2 = _conv_self_fft(p2)
            p2, c2, _ = run_cd(p2, c2, stop2, rng, G=35, max_flat=30,
                               perturb_sigma_base=0.01, recompute_every=5,
                               golden_iters=14, focus_active=False)
            consider(p2)
        except Exception:
            pass

    # Phase 3: Focused coordinate descent on active coordinates (25% of time)
    stop3 = t0 + budget * 0.87
    if time.monotonic() < stop3 - 10:
        try:
            p3 = best_p.copy()
            c3 = _conv_self_fft(p3)
            p3, c3, _ = run_cd(p3, c3, stop3, rng, G=50, max_flat=40,
                               perturb_sigma_base=0.005, recompute_every=4,
                               golden_iters=16, focus_active=True)
            consider(p3)
        except Exception:
            pass

    # Phase 4: Final fine polish if time remains
    if time.monotonic() < deadline - 15:
        try:
            p4 = best_p.copy()
            c4 = _conv_self_fft(p4)
            p4, c4, _ = run_cd(p4, c4, deadline - 10, rng, G=70, max_flat=50,
                               perturb_sigma_base=0.003, recompute_every=3,
                               golden_iters=18, focus_active=True)
            consider(p4)
        except Exception:
            pass

    # Final validation with exact evaluator
    out = np.asarray(best_p, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    if out.size == 0 or float(out.sum()) < 0.01:
        out = np.full(128, 1.0 / 128.0)
    ls = [float(x) for x in out]
    final_sc = float(evaluate_sequence(ls))
    if not math.isfinite(final_sc):
        # Fallback to parent
        try:
            pp = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
            pp = np.clip(pp, 0.0, 1000.0)
            pp = pp / float(pp.sum())
            ls = [float(x) for x in pp]
            final_sc = float(evaluate_sequence(ls))
        except Exception:
            return [1.0]
        if not math.isfinite(final_sc):
            return [1.0]

    return ls
