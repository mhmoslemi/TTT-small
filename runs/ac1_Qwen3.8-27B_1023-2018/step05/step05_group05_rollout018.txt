        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(3072, dtype=np.float64)
    return p


def _conv_self(a):
    n = int(a.size)
    if n == 0:
        return np.zeros(1, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(c, 0.0)


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s < 1e-300:
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _p2u(p):
    p = np.maximum(np.asarray(p, dtype=np.float64), 1e-300)
    ll = np.log(p)
    return ll - float(np.mean(ll))
