def _conv_self(a):
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _fft_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    s = float(p.sum())
    if s < 1e-15:
        return math.inf
    c = _conv_self(p)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _exact_score(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    n = len(p)
    s = float(p.sum())
    if s < 1e-15:
        return math.inf
    b = np.convolve(p, p)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(len(u), 1.0 / len(u))
    return e / s

def propose_candidate(seed=42, budget_s=1300, **kwargs):

    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 5.0:
        hard = t0 + max(3.0, budget * 0.9)

    try:
        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(512, dtype=np.float64)
    if p.size == 0:
        p = np.ones(512, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(512, dtype=np.float64)
    n = len(p)
    p = p / float(p.sum())

    best_p = p.copy()
    best_score = _fft_score(p)

    def consider(arr):
        nonlocal best_p, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.clip(a, 0.0, 1000.0)
            a = np.where(np.isfinite(a), a, 0.0)
            if a.size == 0 or float(a.sum()) < 0.01:
                return
            sc = _fft_score(a)
            if math.isfinite(sc) and sc < best_score - 1e-13:
                a = a / float(a.sum())
                best_p = a.copy()
                best_score = sc
        except Exception:
            pass

    try:
        exact_sc = _exact_score(best_p)
        if math.isfinite(exact_sc):
            best_score = exact_sc
            consider(best_p)
    except Exception:
        pass

    # Phase 1: L-BFGS warm-up
    lbfgs_stop = min(hard - (budget * 0.55), t0 + min(120.0, budget * 0.12))
    if lbfgs_stop > time.monotonic() + 25.0:
        try:
            p2 = _lbfgs_phase(p, n, lbfgs_stop, consider)
            if p2 is not None and len(p2) == n and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p = p2 / float(p2.sum())
                consider(p)
        except Exception:
            pass
        if _fft_score(p) > _fft_score(best_p) + 0.001:
            p = best_p.copy()

    # Phase 2: main coordinate descent
    coord_stop = min(hard - max(200.0, budget * 0.18), t0 + (budget - 300.0))
    if coord_stop > time.monotonic() + 30.0:
        try:
            p3, _ = _coord_descent(p, coord_stop, consider, seed=seed + 777)
            if p3 is not None and len(p3) == n and np.all(np.isfinite(p3)) and p3.sum() > 0:
                consider(p3 / float(p3.sum()))
        except Exception:
            pass

    # Multi-start perturbation + short coordinate descent rounds
    rng = np.random.default_rng(seed)
    explor_runtime = max(60.0, min(240.0, (hard - 120.0) - time.monotonic()))
    if explor_runtime > 40.0:
        for rep in range(6):
            round_stop = min(hard - 120.0, time.monotonic() + explor_runtime / 6.0)
            if round_stop <= time.monotonic() + 15.0:
                break
            try:
                base = best_p.copy()
                sigma = 0.03 + 0.02 * rep
                base = base * np.exp(rng.normal(0.0, sigma, n))
                base = np.maximum(base, 0.0)
                ss = float(base.sum())
                if ss <= 1e-300:
                    continue
                base /= ss
                consider(base)
                p6, _ = _coord_descent(base, round_stop, consider, seed=seed + 1000 + rep)
                if p6 is not None and len(p6) == n and np.all(np.isfinite(p6)) and p6.sum() > 0:
                    consider(p6 / float(p6.sum()))
            except Exception:
                continue
            if time.monotonic() > hard - 150.0:
                break

    # Phase 3: focus polish
    polish_stop = min(hard - 60.0, t0 + (budget - 70.0))
    if polish_stop > time.monotonic() + 20.0:
        try:
            p4, _ = _focus_polish(best_p, polish_stop, consider, top_k=300)
            if p4 is not None and len(p4) == n and np.all(np.isfinite(p4)) and p4.sum() > 0:
                consider(p4 / float(p4.sum()))
        except Exception:
            pass

    # Phase 4: final coordinate round if time remains
    rem_stop = hard - 30.0
    if rem_stop > time.monotonic() + 30.0:
        try:
            p5, _ = _coord_descent(best_p, rem_stop, consider, seed=seed + 54321)
            if p5 is not None and len(p5) == n and np.all(np.isfinite(p5)) and p5.sum() > 0:
                consider(p5 / float(p5.sum()))
        except Exception:
            pass

    # Final validation with the authoritative evaluator
    try:
        out = np.asarray(best_p, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(128, 1.0 / 128.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            try:
                pp = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
                pp = np.clip(pp, 0.0, 1000.0)
                ls2 = [float(x) for x in pp]
                if math.isfinite(float(evaluate_sequence(ls2))):
                    ls = ls2
            except Exception:
                pass
            if not math.isfinite(float(evaluate_sequence(ls))):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
