                        # Try moving mass from i to j or vice versa
                        for d in [0.01, 0.005, 0.001]:
                            for sign in [1, -1]:
                                cand = a_c.copy()
                                cand[i] -= sign * d * a_c[i]
                                cand[j] += sign * d * a_c[i]
                                cand = np.maximum(cand, 1e-15)
                                cs = cand.sum()
                                if cs < 1e-12: continue
                                cand /= cs
                                v = float(np.convolve(cand, cand).max())
                                if v < M - 1e-14:
                                    if M - v > best_improve:
                                        best_improve = M - v
                                        best_pair = (i, j, sign, d)
                if best_pair is not None:
                    i, j, sign, d = best_pair
                    a_c[i] -= sign * d * a_c[i]
                    a_c[j] += sign * d * a_c[i]
                    a_c = np.maximum(a_c, 1e-15)
                    a_c /= a_c.sum()
                else:
                    break
            sc = 2.0 * n * float(np.convolve(a_c, a_c).max())
            if sc < best_score:
                best_score = sc
                best_a = a_c.copy()
            a = a_c.copy()
        
        phase += 1
    
    return best_a
