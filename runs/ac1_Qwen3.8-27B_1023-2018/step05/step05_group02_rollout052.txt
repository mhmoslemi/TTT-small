            best = a.copy()

    def lse_obj(u, T):
        a = np.exp(u - u.max())
        a /= a.sum()
        L = 2*n - 1
        sz = next_fast_len(L)
        F = rfft(a, sz)
        c = irfft(F*F, sz)[:L]
        cmax = c.max()
        w = np.exp((c - cmax) / T)
        wsum = w.sum()
        f = cmax + T * math.log(wsum / L)
        # Gradient
        # d(cmax)/du_i = ... (approximate: cross_corr(w, a) * 2)
        # Actually: f = T*log(sum exp((c_k - cmax)/T)) + cmax
        # df/da_i = sum_k (exp((c_k-cmax)/T) / wsum) * 2*a_{k-i}
        # = 2 * cross_corr(w/wsum, a)_i
        # df/du_j = df/da_j * (a_j - a_j * a_j) ... (softmax Jacobian)
        # Actually, du -> da: da_j = a_j * (du_j - avg(du))
        # So df/du_j = sum_i (df/da_i) * da_i/du_j = sum_i (df/da_i) * a_j * (delta_ij - a_i)
        # = a_j * (df/da_j - sum_i df/da_i * a_i)
        
        grad_a = np.zeros(n)
        # Cross-correlation of w/wsum with a
        wz = w / wsum  # length L = 2n-1
        # (convolution or cross-correlation)
        # grad_a[i] = 2 * sum_k wz[k] * a[k-i] for valid k
        # This is the cross-correlation of wz and a
        Fwz = rfft(wz, sz)
        Fa = rfft(a, sz)
        # Cross-correlation: irfft(Fwz * conj(Fa)) gives sum wz[k]*a[k+i]... 
        # Actually, let me use a different approach
        # sum_k wz[k] * a[k-i] = (wz * a)[2n-2-i] ... no
        # Let me just compute it directly for small n
        # Actually n=3072, so let's use FFT
        # cross_corr[wz, a][i] = sum_k wz[k] * a[k-i] (with proper indexing)
        # = irfft(conj(Fwz) * Fa) ... need to be careful
        # 
        # Let's use: np.convolve(wz, a[::-1]) and extract the right part
        # Or: cross-correlation wz ⊛ a at lag i = sum_k wz[k+i] * a[k]
        #     = sum_k wz[k] * a[k-i] (with substitution)
        # In FFT: ifft(conj(Fwz) * Fa) gives circular cross-correlation
        # For linear: pad to at least 3n-2
        sz2 = next_fast_len(3*n - 2)
        Fwz2 = rfft(wz, sz2)
        Fa2 = rfft(a, sz2)
        cc = irfft(Fwz2 * np.conj(Fa2), sz2)  # Not quite right
        # Actually: cross-correlation (wz ⊛ a)[m] = sum_k wz[k+m] a[k]
        # In frequency: ifft(conj(Fwz) * Fa)
        # Let me verify: ifft(conj(Fwz) * Fa)[m] = sum_k wz[k] a[k-m] (circular)
        # For linear: pad so no wraparound
        # sum_k wz[k] a[k-m] for k where both are in range
        # This gives grad_a[m] = 2 * sum_k wz[k] a[k-m]
        
        # Hmm, this is getting complicated. Let me just do it directly.
        # For each i, grad_a[i] = 2 * sum_{k=i}^{i+n-1} wz[k] * a[k-i]
        # This is the cross-correlation of wz (length 2n-1) and a (length n) at lags 0 to n-1
        
        # Using FFT: pad wz to size 3n-2, pad a to size 3n-2
        # ifft(F_wz * conj(F_a))[n-1-1 : n-1+n-1] ... 
        # I'll just use np.convolve for simplicity since n=3072
        # np.convolve(wz, a[::-1], 'full') gives cross-correlation
        cc_full = np.convolve(wz, a[::-1], 'full')
        # cc_full[m] = sum_k wz[k] * a[m-1-k] ... no
        # np.convolve(x, y)[k] = sum_i x[i] y[k-i]
        # So np.convolve(wz, a[::-1])[k] = sum_i wz[i] * a[::-1][k-i] = sum_i wz[i] * a[n-1-k+i]
        # We want: sum_k wz[k] * a[k-i] = sum_j wz[j] * a[j-i] (renaming)
        # Let m = j-i: = sum_m wz[m+i] * a[m]
        # = cross-correlation of wz and a at lag i
        # np.correlate(wz, a, 'full')[i + n - 1] should give this
        cc = np.correlate(wz, a, 'full')  # cc[m] = sum_k wz[k+m-(n-1)] * a[k]
        # Hmm, let me just verify the indexing
        # np.correlate(x, v, 'full')[k] = sum_j x[j+k] * v[j] (for appropriate ranges)
        # We want sum_k wz[k] * a[k-i] = sum_j wz[j+i] * a[j]
        # This is np.correlate(wz, a)[i] (with 'full' mode, offset by n-1)
        # Actually np.correlate(wz, a, 'full')[i + (len(a)-1)] ... 
        # Let me just use a simpler approach
        
        # Direct computation (O(n^2) but n=3072 so ~10M ops, acceptable)
        # Actually, let me use FFT properly
        sz_cc = next_fast_len(2*n)
        Fwz_cc = rfft(wz, sz_cc)
        Fa_cc = rfft(a, sz_cc)
        # Circular cross-correlation: ifft(conj(Fwz) * Fa)
        # This gives (wz ⊛ a)[m] = sum_k wz[k] * a[(k+m) mod N]
        # For linear (no wrap): need N >= len(wz) + len(a) - 1 = 3n - 2
        sz_cc = next_fast_len(3*n - 2)
        Fwz_cc = rfft(wz, sz_cc)
        Fa_cc = rfft(a, sz_cc)
        cc_linear = irfft(Fwz_cc * np.conj(Fa_cc), sz_cc)
        # cc_linear[m] = sum_k wz[k] * a[(k+m)] (circular, but linear since padded)
        # We want: grad_a[i] = 2 * sum_k wz[k] * a[k-i] for k in [i, i+n-1]
        # = 2 * sum_j wz[j+i] * a[j] (letting j = k-i)
        # This is the cross-correlation at lag i
        # In our cc_linear: cc_linear[m] = sum_k wz[k+m] * a[k] (for valid k)
        # So cc_linear[i] = sum_k wz[k+i] * a[k] = what we want!
        # But we need to check the index range
        # wz has length 2n-1, a has length n
        # cc_linear[i] for i = 0, ..., n-1: sum_k wz[k+i] * a[k]
        #   where 0 <= k < n and 0 <= k+i < 2n-1
        #   i.e., k < n and k < 2n-1-i
        #   Since i >= 0, 2n-1-i >= n-1, so k < n is the binding constraint
        #   So cc_linear[i] = sum_{k=0}^{n-1} wz[k+i] * a[k] ✓
        
        grad_a = 2.0 * cc_linear[:n]
        
        # Now convert to gradient w.r.t. u
        # a = softmax(u), so da_j/du_i = a_j(δ_ij - a_i)
        # df/du_j = sum_i (df/da_i) * (da_i/du_j) = sum_i grad_a[i] * a_j(δ_ij - a_i)
        # = a_j * (grad_a[j] - sum_i grad_a[i] * a_i)
        tda = np.dot(grad_a, a)
        grad_u = a * (grad_a - tda)
        
        return float(f), grad_u
    
    # Run LSE optimization
    u = np.log(np.maximum(p, 1e-300))
    u -= u.mean()
    
    c = np.convolve(p, p, 'full')
    cmax0 = c.max()
    
    stages = [
        (0.10, 0.10),
        (0.04, 0.12),
        (0.015, 0.15),
        (0.005, 0.20),
        (0.002, 0.20),
        (0.0008, 0.23),
    ]
    
    total_w = sum(w for _, w in stages)
    elapsed_start = time.monotonic()
    lse_time = 0.60 * (deadline - t0)
    opt_end = t0 + lse_time
    cum = 0.0
    
    for frac, wt in stages:
        if time.monotonic() > opt_end - 3:
            break
        cum += wt
        stage_end = min(opt_end - 2, t0 + lse_time * (cum / total_w))
        if stage_end <= time.monotonic() + 3:
            continue
        T = frac * cmax0
        while time.monotonic() < stage_end - 1:
            try:
                res = minimize(lse_obj, u, args=(T,), method='L-BFGS-B', jac=True,
                              options={'maxiter': 50, 'gtol': 1e-12, 'disp': False, 'maxcor': 40})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    f, _ = lse_obj(res.x, T)
                    if np.isfinite(f):
                        u = res.x.copy()
                        a_new = np.exp(u - u.max())
                        a_new /= a_new.sum()
                        consider(a_new)
            except Exception:
                break
            # Check if converged
            if hasattr(res, 'status') and res.status == 0:
                break
    
    # Update best from LSE
    if time.monotonic() < opt_end - 2:
        a_lse = np.exp(u - u.max())
        a_lse /= a_lse.sum()
        consider(a_lse)
    
    # Coordinate descent
    cd_start = time.monotonic()
    cd_end = min(deadline - 30, cd_start + 0.30 * (deadline - t0))
    
    def coord_descent(a, end_time, seed_cd):
        a = np.asarray(a, dtype=np.float64).copy()
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 1e-15:
            return a
        a /= s
        
        n_local = a.size
        L = 2 * n_local - 1
        c = np.convolve(a, a, 'full')
        c = np.maximum(c, 0.0)
        S = 1.0
        cur_score = 2.0 * n_local * c.max()
        best_local = a.copy()
        best_score_local = cur_score
        
        rng = np.random.default_rng(seed_cd)
        G = 32
        sweep = 0
        flat = 0
        
        while sweep < 50000 and time.monotonic() < end_time - 2:
            improved = False
            order = rng.permutation(n_local)
            for idx in order:
                i = int(idx)
                if time.monotonic() > end_time - 1:
                    break
                if S > 1.25 or S < 0.75:
                    inv = 1.0 / S
                    a *= inv
                    c *= inv * inv
                    S = 1.0
                
                v = a[i]
                start = i
                end_i = min(i + n_local, L)
                Ls = end_i - start
                if Ls <= 0:
                    continue
                base = c[start:end_i].copy()
                infl = 2.0 * a[:Ls].copy()
                
                # Max of unaffected entries
                max_const = 0.0
                if start > 0:
                    max_const = max(max_const, c[:start].max())
                if end_i < L:
                    max_const = max(max_const, c[end_i:].max())
                
                x_hi = min(20.0 * v if v > 1e-15 else 0.001, 0.05)
                x_hi = max(x_hi, 1e-15)
                
                pts = np.linspace(0.0, x_hi, G + 1)
                
                # Vectorized evaluation
                w = pts - v
                denom = S + w
                aff = base[None, :] + w[:, None] * infl[None, :]
                di = 2 * i
                if start <= di < end_i:
                    aff[:, di - start] += w * w
                maff = np.maximum(aff, max_const)
                vals = (2.0 * n_local * maff.max(axis=1)) / (denom * denom)
                
                imin = int(np.argmin(vals))
                bx = float(pts[imin])
                bv = float(vals[imin])
                
                # Refine
                dw = x_hi / G
                lo = max(0.0, bx - 2*dw)
                hi = min(x_hi, bx + 2*dw)
                if hi > lo:
                    fp = np.linspace(lo, hi, 32)
                    wf = fp - v
                    df = S + wf
                    afff = base[None, :] + wf[:, None] * infl[None, :]
                    if start <= di < end_i:
                        afff[:, di - start] += wf * wf
                    mafff = np.maximum(afff, max_const)
                    valsf = (2.0 * n_local * mafff.max(axis=1)) / (df * df)
                    iminf = int(np.argmin(valsf))
                    if valsf[iminf] < bv:
                        bx = float(fp[iminf])
                        bv = float(valsf[iminf])
                
                if bv < cur_score - 1e-14 and 0 <= bx:
                    a[i] = bx
                    w_new = bx - v
                    if abs(w_new) > 1e-15:
                        S += w_new
                        c[start:end_i] += w_new * infl
                        if 0 <= di < L:
                            c[di] += w_new * w_new
                    cur_score = bv
                    improved = True
            
            # Rescale
            if S > 1e-30:
                inv = 1.0 / S
                a *= inv
                c *= inv * inv
            S = 1.0
            c = np.convolve(a, a, 'full')
            c = np.maximum(c, 0.0)
            new_sc = 2.0 * n_local * c.max()
            if new_sc < best_score_local - 1e-15:
                best_score_local = new_sc
                best_local = a.copy()
            
            if improved:
                flat = 0
            else:
                flat += 1
                if flat >= 3:
                    # Perturb
                    a = a * np.exp(rng.normal(0, 0.03, n_local))
                    a = np.maximum(a, 0)
                    ss = a.sum()
                    if ss > 0:
                        a /= ss
                    c = np.convolve(a, a, 'full')
                    c = np.maximum(c, 0.0)
                    cur_score = 2.0 * n_local * c.max()
                    S = 1.0
                    flat = 0
            sweep += 1
        
        if cur_score < best_score_local:
            best_local = a / a.sum()
            best_score_local = cur_score
        return best_local
    
    # Run coordinate descent
    p_cd = coord_descent(best, cd_end, seed + 1)
    consider(p_cd)
    
    # Final polish
    final_end = deadline - 10
    if time.monotonic() < final_end - 20:
        # More LSE at very low T
        u2 = np.log(np.maximum(best, 1e-300))
        u2 -= u2.mean()
        c2 = np.convolve(best, best, 'full')
        T2 = 0.0003 * c2.max()
        lse_end2 = min(final_end - 15, time.monotonic() + 0.2 * (final_end - time.monotonic()))
        if lse_end2 > time.monotonic() + 10:
            while time.monotonic() < lse_end2 - 2:
                try:
                    res = minimize(lse_obj, u2, args=(T2,), method='L-BFGS-B', jac=True,
                                  options={'maxiter': 40, 'gtol': 1e-13, 'disp': False})
                    if res.x is not None:
                        u2 = res.x.copy()
                        a2 = np.exp(u2 - u2.max())
                        a2 /= a2.sum()
                        consider(a2)
                except:
                    break
                if hasattr(res, 'status') and res.status == 0:
                    break
        
        # Final coordinate descent
        if time.monotonic() < final_end - 5:
            p_final = coord_descent(best, final_end, seed
