        # = (p_rev * w)[j + n - 1] where p_rev[i] = p[n-1-i]
        # Because (p_rev * w)[k] = sum_i p_rev[i] * w[k-i] = sum_i p[n-1-i] * w[k-i]
        # Let m = n-1-i: = sum_m p[m] * w[k-(n-1-m)] = sum_m p[m] * w[k-n+1+m]
        # Set k-n+1 = j: k = j+n-1. So corr[j] = (p_rev * w)[j+n-1].
        
        p_rev = pp[::-1]
        F = rfft(p_rev, cross_size)
        G = rfft(w, cross_size)
        corr = irfft(F * G, cross_size)
        grad_p = 2.0 * T * corr[n - 1:2 * n - 1]
        
        # Normalize gradient for softmax parameterization
        u_grad = pp * (grad_p - float(np.dot(pp, grad_p)))
        return float(f), np.where(np.isfinite(u_grad), u_grad, 0.0)

    u = _p_to_u(p)
    best_p = p.copy()
    best_sc = _fft_score(p)

    if fine_only:
        stages = ((0.003, 0.35), (0.001, 0.35), (0.0004, 0.30))
    else:
        stages = (
            (0.30, 0.06),
            (0.12, 0.10),
            (0.04, 0.14),
            (0.012, 0.18),
            (0.004, 0.20),
            (0.0012, 0.18),
            (0.0004, 0.14),
        )

    elapsed_start = time.monotonic()
    total_avail = max(5.0, stop_time - elapsed_start)
    
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 8.0:
            break
        stage_time = wt * total_avail
        end_stage = min(stop_time - 6.0, now + stage_time)
        
        cmax = float(cfast(p).max())
        T = max(1e-15, frac * cmax)
        stale = 0
        
        while time.monotonic() < end_stage - 2.0:
            maxit = 40
            opts = {"maxiter": maxit, "gtol": 1e-14, "ftol": 1e-16,
                    "maxls": 30, "maxcor": 25, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval):
                    p_cur = _softmax(res.x)
                    sc_cur = _fft_score(p_cur)
                    if math.isfinite(sc_cur) and sc_cur < best_sc - 1e-14:
                        best_sc = sc_cur
                        best_p = p_cur.copy()
                        p = p_cur.copy()
                        u = _p_to_u(p)
                        stale = 0
                        if consider is not None:
                            try:
                                consider(p_cur)
                            except Exception:
                                pass
                    else:
                        stale += 1
                        if stale >= 4:
                            break
            if getattr(res, "status", 1) == 0:
                break

    p_out = best_p.copy()
    ss = float(p_out.sum())
    if ss > 1e-300:
        p_out = p_out / ss
    return p_out


def _coord_descent(p, stop_time, consider=None, seed=42):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p, math.inf
    p = p / s

    Ltot = 2 * n - 1
    c = _conv_fft(p)
    S = 1.0
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    flat = 0
    sweep = 0

    while time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 1.0:
                break
            if S <= 1e-300:
                break
            if S > 1.3 or S < 0.7:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0

            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start > 0:
                ml = float(np.max(c[:start]))
                max_const = max(max_const, ml)
            if end < Ltot:
                mr = float(np.max(c[end:]))
                max_const = max(max_const, mr)

            x_hi = max(40.0 * v, 100.0 / n, 0.01)
            x_hi = min(x_hi, 10.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = 1.0 / n

            n_coarse = 60
            pts = np.linspace(0.0, x_hi, n_coarse + 1)
            if 0.0 <= v <= x_hi:
                pts = np.unique(np.concatenate([pts, [v]]))

            # Vectorized evaluation
            w = pts - v
            denom = S + w
            valid = denom > 1e-14
            if not np.any(valid):
                continue
            aff = base[None, :] + w[:, None] * infl[None, :]
            diag_idx = i - start
            if 0 <= diag_idx < Ls:
                dvals = base[diag_idx] + w * infl[diag_idx] + w * w
                aff = np.maximum(aff, dvals[:, None])
            aff = np.maximum(aff, max_const)
            mv = np.max(aff, axis=1)
            out = np.full(xs.shape if False else pts.shape, math.inf)
            dv = denom[valid]
            out[valid] = (2.0 * n * mv[valid]) / (dv * dv)

            if not np.any(np.isfinite(out)):
                continue
            imin = int(np.argmin(out))
            bx = float(pts[imin])
            bv = float(out[imin])

            # Fine search
            dw = x_hi / n_coarse
            lo = max(0.0, bx - 3.0 * dw)
            hi = min(x_hi, bx + 3.0 * dw)
            if hi > lo + 1e-15:
                fine = np.linspace(lo, hi, 40)
                if 0.0 <= v <= hi:
                    fine = np.unique(np.concatenate([fine, [v, bx]]))
                w2 = fine - v
                denom2 = S + w2
                valid2 = denom2 > 1e-14
                if np.any(valid2):
                    aff2 = base[None, :] + w2[:, None] * infl[None, :]
                    if 0 <= diag_idx < Ls:
                        dvals2 = base[diag_idx] + w2 * infl[diag_idx] + w2 * w2
                        aff2 = np.maximum(aff2, dvals2[:, None])
                    aff2 = np.maximum(aff2, max_const)
                    mv2 = np.max(aff2, axis=1)
                    out2 = np.full(fine.shape, math.inf)
                    dv2 = denom2[valid2]
                    out2[valid2] = (2.0 * n * mv2[valid2]) / (dv2 * dv2)
                    if np.any(np.isfinite(out2)):
                        im2 = int(np.argmin(out2))
                        if float(out2[im2]) < bv:
                            bv = float(out2[im2])
                            bx = float(fine[im2])

            if math.isfinite(bv) and bv < cur_score - 1e-12 and 0.0 <= bx <= 20.0 * S:
                w_delta = bx - v
                if abs(w_delta) > 1e-15:
                    p[i] = bx
                    S += w_delta
                    c[start:end] += w_delta * infl
                    if 0 <= diag_idx < Ls:
                        c[start + diag_idx] += w_delta * w_delta
                    cur_score = bv
                    improved = True

        # Rescale
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        # Verify periodically
        if sweep % 8 == 0:
            c = _conv_fft(p)
        new_sc = 2.0 * n * float(c.max())
        if math.isfinite(new_sc) and new_sc < best_score - 1e-14:
            best_score = new_sc
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 30.0:
                    # Perturb
                    sigma = 0.03 * flat / 5.0
                    p = p * np.exp(rng.normal(0, sigma, n))
                    p = np.maximum(p, 0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_fft(p)
                    S = 1.0
                    cur_score = 2.0 * n * float(c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_fft(p)
    sc = 2.0 * n * float(c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = math.inf
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_arr]))
    except Exception:
        pass
    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(128)
        best_score = float(evaluate_sequence([float(x) for x in best_arr]))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s_val = float(a.sum())
            if s_val < 0.01:
                return
            sc = float(evaluate_sequence([float(x) for x in a]))
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_arr = a.copy()
                best_score = sc
        except Exception:
            pass

    n_best = int(best_arr.size)

    # Phase 1: LSE optimization on parent
    lse_stop = t0 + min(200.0, 0.20 * budget)
    if lse_stop > t0 + 40.0:
        try:
            p1 = _lse_optimize(best_arr.copy(), lse_stop, consider=consider)
            if p1 is not None and p1.size == n_best and np.all(np.isfinite(p1)) and float(p1.sum()) > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    # Phase 2: Coordinate descent (main workhorse)
    coord_stop = hard - 120.0
    if coord_stop > time.monotonic() + 80.0:
        try:
            p2, _ = _coord_descent(best_arr.copy(), coord_stop, consider=consider, seed=int(seed))
            if p2 is not None and p2.size == n_best and np.all(np.isfinite(p2)) and float(p2.sum()) > 0:
                p2 = p2 / float(p2.sum())
                consider(p2)
        except Exception:
            pass

    # Phase 3: Second LSE at low temperature
    rem = hard - 20.0 - time.monotonic()
    if rem > 50.0:
        lse2_stop = time.monotonic() + max(25.0, 0.35 * rem)
        try:
            p3 = _lse_optimize(best_arr.copy(), lse2_stop, consider=consider, fine_only=True)
            if p3 is not None and p3.size == n_best and np.all(np.isfinite(p3)) and float(p3.sum()) > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)
        except Exception:
            pass

    # Phase 4: Final coordinate descent polishing
    coord2_stop = hard - 15.0
    if time.monotonic() < coord2_stop - 25.0:
        try:
            p4, _ = _coord_descent(best_arr.copy(), coord2_stop, consider=consider, seed=int(seed + 999))
            if p4 is not None and p4.size == n_best and np.all(np.isfinite(p4)) and float(p4.sum()) > 0:
                p4 = p4 / float(p4.sum())
                consider(p4)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
