    if S < 1e-12:

def _softmax(u):
    u = np.asarray(u, dtype=np.float64).ravel()
    if u.size == 0:
        return u
    u = np.where(np.isfinite(u), u, 0.0)
    m = u.max()
    e = np.exp(u - m)
    s = e.sum()
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0/u.size, dtype=np.float64)

def _coord_solve(c, a, i, S, n, K=25):
    """
    Solve 1D subproblem for coordinate i.
    Returns (best_x, best_score).
    """
    v = a[i]
    L = 2*n - 1
    start = i
    end = min(i+n, L)
    Ls = end - start
    if Ls <= 0:
        return v, 2.0*n*c.max()/(S*S)
    
    base = c[start:end]
    di = i  # diagonal local index (2i - i = i)
    
    # Influence vector: 2*a[0:Ls]
    # (since k = i + j, influence = 2*a[k-i] = 2*a[j])
    # But we only have a[0:n], and Ls = n (typically), so:
    infl = 2.0 * a[:Ls]
    
    # Max of unaffected
    max_const = 0.0
    if start > 0:
        mc = c[:start].max()
        if mc > max_const:
            max_const = mc
    if end < L:
        mc = c[end:].max()
        if mc > max_const:
            max_const = mc
    
    # Current score
    cur_score = 2.0*n*c.max()/(S*S)
    
    # Generate candidates
    w_max = min(1000.0 - v, max(5.0*S/n + 3.0*v, 0.01))
    if w_max < 0:
        w_max = 0.0
    
    cand_ws = [-v, 0.0]
    
    # Top K linear terms
    if Ls > 1 and K > 0:
        # Get values at w=0 for all positions
        # Exclude diagonal for linear terms
        lin_mask = np.ones(Ls, dtype=bool)
        lin_mask[di] = False
        lin_vals = base[lin_mask]
        lin_infl = infl[lin_mask]
        
        # Top K by value
        k_eff = min(K, len(lin_vals))
        if k_eff > 0:
            top_local = np.argpartition(-lin_vals, k_eff-1)[:k_eff]
            tb = lin_vals[top_local]
            ti = lin_infl[top_local]
            
            # Pairs of linear terms
            for p in range(k_eff):
                for q in range(p+1, k_eff):
                    d_s = ti[p] - ti[q]
                    if abs(d_s) > 1e-15:
                        wc = (tb[q] - tb[p]) / d_s
                        if -v <= wc <= w_max:
                            cand_ws.append(wc)
                
                # Linear vs constant
                if ti[p] > 1e-15:
                    wc = (max_const - tb[p]) / ti[p]
                    if -v <= wc <= w_max:
                        cand_ws.append(wc)
                
                # Linear vs quadratic
                # tb[p] + ti[p]*w = base[di] + 2*v*w + w^2
                # w^2 + (2v - ti[p])*w + (base[di] - tb[p]) = 0
                B = 2.0*v - ti[p]
                D = base[di] - tb[p]
                disc = B*B - 4.0*D
                if disc >= 0:
                    sd = math.sqrt(disc)
                    w1 = (-B - sd) / 2.0
                    w2 = (-B + sd) / 2.0
                    for wc in (w1, w2):
                        if -v <= wc <= w_max:
                            cand_ws.append(wc)
            
            # Constant vs quadratic
            B = 2.0*v
            D = base[di] - max_const
            disc = B*B - 4.0*D
            if disc >= 0:
                sd = math.sqrt(disc)
                w1 = (-B - sd) / 2.0
                w2 = (-B + sd) / 2.0
                for wc in (w1, w2):
                    if -v <= wc <= w_max:
                        cand_ws.append(wc)
    
    # Add a few heuristic points
    if v > 1e-10:
        for f in (0.3, 0.6, 0.8, 1.2, 1.5):
            wc = v*(1/f - 1)  # x = v/f
            if -v < wc < w_max:
                cand_ws.append(wc)
    
    # Convert to array
    w_arr = np.array(cand_ws, dtype=np.float64)
    w_arr = np.unique(w_arr)
    w_arr = w_arr[(w_arr >= -v - 1e-12) & (w_arr <= w_max + 1e-12)]
    w_arr = np.clip(w_arr, -v, w_max)
    
    if w_arr.size == 0:
        return v, cur_score
    
    # Vectorized evaluation
    aff = base[None, :] + w_arr[:, None] * infl[None, :]
    aff[:, di] = base[di] + w_arr * infl[di] + w_arr * w_arr
    np.maximum(aff, max_const, out=aff)
    mv = aff.max(axis=1)
    den = S + w_arr
    valid = den > 1e-14
    scores = np.full(w_arr.size, math.inf, dtype=np.float64)
    if np.any(valid):
        scores[valid] = 2.0*n*mv[valid] / (den[valid]*den[valid])
    
    # Find best
    finite_mask = np.isfinite(scores)
    if not np.any(finite_mask):
        return v, cur_score
    im = np.argmin(np.where(finite_mask, scores, math.inf))
    best_w = w_arr[im]
    best_x = max(0.0, v + best_w)
    best_sc = scores[im]
    
    return best_x, best_sc

def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))

        return np.full(n, 1.0 / n, dtype=np.float64)

        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m) + 0.5) / m
        fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
        out = fp

    s = out.sum()
    if not math.isfinite(s) or s < 1e-12:

def _lse_optimize(p, stop_time, consider=None, fine_only=False):
    p = _norm_nonneg(p)
    n = p.size
    if n <= 1 or time.monotonic() > stop_time:
        return p

    conv_len = 2 * n - 1
    cross_len = 3 * n - 2
    conv_size = next_fast_len(conv_len)
    cross_size = next_fast_len(cross_len)

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def p_to_u(pp):
        pp = np.maximum(pp, 1e-15)
        uu = np.log(pp)
        uu -= uu.mean()
        return uu

    def lse_func(uu, T):
        uu = np.asarray(uu, dtype=np.float64).ravel()
        if uu.size != n:
            return 1e100, np.zeros(n)
        uu = np.where(np.isfinite(uu), uu, 0.0)
        m = uu.max()
        e = np.exp(uu - m)
        s = e.sum()
        if not math.isfinite(s) or s <= 0:
            pp = np.full(n, 1.0 / n)
        else:
            pp = e / s
        pp = np.where(np.isfinite(pp), pp, 0.0)
        ss = pp.sum()
        if ss <= 0:
            pp = np.full(n, 1.0 / n)
            ss = 1.0
        pp /= ss

        c = cfast(pp)
        cmax = c.max()
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros(n)

        if not math.isfinite(T) or T <= 0:
            T = max(1e-14, 1e-4 * cmax)
        T = max(1e-14, T)

        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = exps.sum()
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(conv_len)
            w[int(c.argmax())] = 1.0
            f = cmax
        else:
            w = exps / sum_exp
            f = cmax + T * math.log(sum_exp / conv_len)
            if not math.isfinite(f):
                f = cmax

        q = xfast(pp, w)
        gid = 2.0 * q
        gd = pp @ gid
        grad = pp * (gid - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score_fft(p)

    if fine_only:
        stages = ((0.0012, 0.30), (0.0003, 0.40), (0.00008, 0.50))
    else:
        stages = ((0.02, 0.15), (0.007, 0.18), (0.002, 0.20),
                  (0.0005, 0.22), (0.0001, 0.28))

    start = time.monotonic()
    avail = max(0.0, stop_time - start - 5.0)
    if avail <= 5.0:
        return best_p

    total_w = sum(wt for _, wt in stages)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 6.0:
            break
        cum += wt
        end_stage = min(stop_time - 5.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:
            continue

        c = cfast(p)
        cmax = c.max() if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-14, frac * cmax)

        stale = 0
        while time.monotonic() < end_stage - 2.0 and stale < 3:
            maxit = 12 if n > 2000 else 18
            opts = {"maxiter": maxit, "gtol": 1e-10, "ftol": 1e-14,
                    "maxls": 12, "maxcor": 12, "disp": False, "maxdisp": 0}
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    fval = float(res.fun)
                    if math.isfinite(fval) and fval < best_sc - 1e-12:
                        u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                    if getattr(res, "status", 1) == 0:
                        break
                else:
                    stale += 1
            else:
                stale += 1

        p_new = _softmax(u)
        p_new = np.maximum(p_new, 0.0)
        ss = p_new.sum()
        if math.isfinite(ss) and ss > 1e-12:
            p_new /= ss
        else:
            p_new = best_p.copy()

        sc_new = _score_fft(p_new)
        if math.isfinite(sc_new) and sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_p = p_new.copy()
            p = p_new.copy()
            if consider:
                try:
                    consider(best_p.copy())
                except Exception:
                    pass
        else:
            p = best_p.copy()
        u = p_to_u(p)

    return best_p


def _lp_improve_small(src, n_small, stop_time, max_iter=3):
    if linprog is None or sp is None:
        return None
    n_small = int(n_small)
    if n_small < 32 or n_small > 576 or time.monotonic() > stop_time:
        return None

    a = _resample(src, n_small)
    a = _norm_nonneg(a)
    best_a = a.copy()
    best_sc = _score_fft(a)
    if not math.isfinite(best_sc):
        return None

    for _ in range(max_iter):
        if time.monotonic() > stop_time - 5.0:
            break
        c = _fft_conv(a)
        M = c.max() if c.size else 0.0
        if not math.isfinite(M) or M <= 0:
            break
        try:
            idx = np.arange(n_small)
            cols = np.repeat(idx, n_small)
            off = np.tile(idx, n_small)
            rows = cols + off
            data = a[off].astype(np.float64)
            A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n_small - 1, n_small)).tocsr()
            cobj = -np.ones(n_small)
            b_ub = np.full(2 * n_small - 1, M)
            res = linprog(cobj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method="highs")
        except Exception:
            break
        if res is not None and getattr(res, "success", False):
            b = np.asarray(res.x, dtype=np.float64).ravel()
            b = np.where(np.isfinite(b), b, 0.0)
            b = np.maximum(b, 0.0)
            s = b.sum()
            if s <= 0:
                break
            g = b / s
            improved = False
            for t in (0.5, 0.3, 0.15, 0.08):
                cand = (1 - t) * a + t * g
                cand = np.maximum(cand, 0.0)
                cs = cand.sum()
                if cs <= 0:
                    continue
                cand /= cs
                sc = _score_fft(cand)
                if math.isfinite(sc) and sc < best_sc - 1e-10:
                    a = cand
                    best_sc = sc
                    best_a = cand.copy()
                    improved = True
                    break
            if not improved:
                break
    if math.isfinite(_score_fft(best_a)):
        return best_a
    return None


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(10.0, budget)
    reserve = 20.0 if budget > 200 else 8.0
    hard = t0 + budget - reserve

    parent = _get_parent()
    n = parent.size
    best_arr = _norm_nonneg(parent.copy())
    best_score = _score_exact(best_arr)
    if not math.isfinite(best_score):
        best_arr = np.full(64, 1.0 / 64.0)
        best_score = _score_exact(best_arr)

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return
        if arr.size == 0:
            return
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = arr.sum()
        if not math.isfinite(s) or s < 0.01:
            return
        arr = arr / s
        sc = _score_exact(arr)
        if math.isfinite(sc) and sc < best_score - 1e-13:
            best_arr = arr.copy()
            best_score = sc

    rng = np.random.default_rng(seed)

    # Phase 1: Exploration
    phase1_stop = min(hard - 60.0, t0 + min(100.0, 0.12 * budget))
    base = best_arr.copy()

    for q in (0.6, 0.85, 1.15, 1.45, 1.9):
        if time.monotonic() > phase1_stop - 20:
            break
        cand = np.power(np.maximum(base, 0), q)
        cand = _norm_nonneg(cand)
        consider(cand)

    for _ in range(6):
        if time.monotonic() > phase1_stop - 15:
            break
        side = 1.0 if rng.random() < 0.5 else -1.0
        amp = 0.3 + 0.8 * rng.random()
        x = (np.arange(n) + 0.5) / n
        f = 1 + int(rng.integers(1, 5))
        ph = 2 * math.pi * rng.random()
        cand = base * np.exp(side * amp * np.sin(2 * math.pi * f * x + ph))
        cand = _norm_nonneg(np.maximum(cand, 0))
        consider(cand)

    # Small LP exploration
    if budget > 300 and time.monotonic() < phase1_stop - 30:
        for ns in (256, 384):
            if time.monotonic() > phase1_stop - 15:
                break
            stop_lp = min(phase1_stop - 5, time.monotonic() + (25.0 if ns <= 256 else 40.0))
            try:
                small = _lp_improve_small(best_arr, ns, stop_lp, max_iter=2)
            except Exception:
                small = None
            if small is not None:
                cand = _resample(small, n)
                cand = _norm_nonneg(cand)
                consider(cand)

    # Phase 2: First coordinate descent
    phase2_stop = min(hard - 80.0, time.monotonic() + min(500.0, 0.40 * (budget - 60)))
    if time.monotonic() < phase2_stop - 10:
        a2, _ = _coord_descent(best_arr, phase2_stop, consider=consider,
                               seed=seed + 7, K=30)
        consider(a2)

    # Phase 3: L-BFGS escape if stuck
    phase3_stop = min(hard - 60.0, time.monotonic() + min(250.0, 0.20 * (budget - 80)))
    if time.monotonic() < phase3_stop - 20:
        p3 = _lse_optimize(best_arr, phase3_stop, consider=consider, fine_only=False)
        consider(p3)
        if _score_fft(p3) < _score_fft(best_arr) + 0.01:
            a3, _ = _coord_descent(p3, min(hard - 40, phase3_stop + 100),
                                   consider=consider, seed=seed + 997, K=30)
            consider(a3)

    # Phase 4: Extended coordinate descent
    phase4_stop = min(hard - 30.0, time.monotonic() + max(100.0, 0.35 * max(0.0, hard - time.monotonic())))
    if time.monotonic() < phase4_stop - 10:
        a4, _ = _coord_descent(best_arr, phase4_stop, consider=consider,
                               seed=seed + 4242, K=35)
        consider(a4)

    # Phase 5: Final fine-tune L-BFGS + coord
    if time.monotonic() < hard - 50:
        final_stop = min(hard - 10, time.monotonic() + min(120.0, 0.25 * max(0.0, hard - time.monotonic())))
        if final_stop > time.monotonic() + 20:
            p5 = _lse_optimize(best_arr, final_stop - 30, consider=consider, fine_only=True)
            consider(p5)
            a5, _ = _coord_descent(p5, hard - 8, consider=consider,
                                   seed=seed + 99999, K=40)
            consider(a5)

    # Final validation and return
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = out.sum()
        if out.size == 0 or not math.isfinite(s) or s < 0.01:
            out = _norm_nonneg(parent)
            s = out.sum()
        if out.size == 0 or not math.isfinite(s) or s < 0.01:
            return [1.0]
        out = out / s
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            fallback = _norm_nonneg(parent)
            ls2 = [float(x) for x in fallback]
            sc2 = evaluate_sequence(ls2)
            if math.isfinite(sc2):
                return ls2
            return [1.0]
        return ls
    except Exception:
        return [1.0]
