```python
    s = float(a.sum())
    if s < 1e-300:
        return a
    a = a / s
    
    conv_sz = next_fast_len(2 * n - 1)
    conv_L = 2 * n - 1
    cross_sz = next_fast_len(3 * n - 2)
    cross_L = 3 * n - 2
    
    def cfast(p):
        F = rfft(p, conv_sz)
        c = irfft(F * F, conv_sz)[:conv_L]
        np.maximum(c, 0, out=c)
        return c
    
    def xfast(p, w):
        F = rfft(p[::-1], cross_sz)
        G = rfft(w, cross_sz)
        c = irfft(F * G, cross_sz)[:cross_L]

        np.maximum(q, 0, out=q)
        return q

        c = cfast(p)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:

        exps = np.exp(scaled)
        se = float(exps.sum())
        if not math.isfinite(se) or se <= 0:

            f = cmax
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0

            w = exps / se
        q = xfast(p, w)
        gdp = 2.0 * q
        tdp = float(np.dot(p, gdp))
        grad = p * (gdp - tdp)
        return float(f), np.where(np.isfinite(grad), grad, 0.0)

    a_safe = np.maximum(a, 1e-300)
    log_a = np.log(a_safe)
    u = log_a - float(np.mean(log_a))
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(256, dtype=np.float64)
    if p.size == 0:
        p = np.ones(256, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(256, dtype=np.float64)
    return p.copy()


def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    sz = next_fast_len(L)
    F = rfft(a, sz)
    c = irfft(F * F, sz)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0 / u.size)
    m = float(u.max())
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _lse_optimize(a, stop_time, n, consider, fine_only=False):
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0).copy()
    s = float(a.sum())
    if s < 1e-300 or n == 0:
        return a
    a = a / s

    conv_sz = next_fast_len(2 * n - 1)
    conv_L = 2 * n - 1
    cross_sz = next_fast_len(3 * n - 2)
    cross_L = 3 * n - 2

    def cfast(p):
        F = rfft(p, conv_sz)
        c = irfft(F * F, conv_sz)[:conv_L]
        c = np.asarray(c, dtype=np.float64)
        np.maximum(c, 0.0, out=c)
        return c

    def xcor(p, w):
        F = rfft(p[::-1], cross_sz)
        G = rfft(w, cross_sz)
        c = irfft(F * G, cross_sz)[:cross_L]
        q = c[n - 1:2 * n - 1]
        q = np.asarray(q, dtype=np.float64)
        np.maximum(q, 0.0, out=q)
        return q

    def lse_obj(u, T):
        p = _softmax(u)
        c = cfast(p)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros(u.shape)
        if T <= 0 or not math.isfinite(T):
            T = 1e-4 * cmax
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        se = float(exps.sum())
        if not math.isfinite(se) or se <= 0:
            f = cmax
            w = np.zeros(conv_L)
            w[int(np.argmax(c))] = 1.0
        else:
            f = cmax + T * math.log(se / conv_L)
            w = exps / se
        q = xcor(p, w)
        gdp = 2.0 * q
        tdp = float(np.dot(p, gdp))
        grad = p * (gdp - tdp)
        return float(f), np.where(np.isfinite(grad), grad, 0.0)

    # u init
    a_safe = np.maximum(a, 1e-300)
    log_a = np.log(a_safe)
    u = log_a - float(np.mean(log_a))

    if fine_only:
        stages = [(0.003, 40), (0.001, 40), (0.0003, 30), (0.0001, 30)]
    else:
        stages = [(0.1, 25), (0.03, 25), (0.01, 25), (0.003, 25), (0.001, 25), (0.0003, 25)]

    cu = u.copy()
    for T_frac, maxit in stages:
        if time.monotonic() > stop_time - 5.0:
            break
        cmax_current = float(cfast(a).max())
        T = max(1e-15, T_frac * cmax_current)
        stale = 0
        while time.monotonic() < stop_time - 3.0 and stale < 4:
            opts = {"maxiter": maxit, "gtol": 1e-14, "ftol": 1e-18,
                    "maxls": 25, "maxcor": 20, "disp": False}
            try:
                res = minimize(lse_obj, cu, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            ok = (res.x is not None and np.all(np.isfinite(res.x)))
            if ok:
                p_new = _softmax(res.x)
                ps = float(p_new.sum())
                if ps > 1e-12:
                    p_new = p_new / ps
                    c_new = cfast(p_new)
                    sc_new = 2.0 * n * float(c_new.max())
                    cur_a_sc = 2.0 * n * float(cfast(a).max())
                    if math.isfinite(sc_new) and sc_new < cur_a_sc * (1 - 1e-12):
                        a = p_new
                        cu = res.x.copy()
                        stale = 0
                        consider(a)
                    else:
                        stale += 1
            else:
                stale += 1
            if getattr(res, "status", 1) == 0:
                break

    s = float(a.sum())
    if s > 1e-300:
        a = a / s
    return a


def _coord_descent(a, stop_time, n, consider, seed_val=42):
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0).copy()
    S = float(a.sum())
    if S < 1e-300 or n == 0:
        return a, math.inf
    a = a / S
    S = 1.0

    Ltot = 2 * n - 1
    c = _conv_self(a)
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_a = a.copy()
    rng = np.random.default_rng(seed_val)

    G_coarse = 40
    G_fine = 30
    flat = 0
    sweep = 0
    max_sweeps = 50000

    while sweep < max_sweeps and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)

        for idx in order:
            i = int(idx)
            if time.monotonic() > stop_time - 1.0:
                break

            S = float(a.sum())
            if S <= 1e-300 or not math.isfinite(S):
                break
            if S > 1.01 or S < 0.99:
                inv = 1.0 / S
                a *= inv
                c *= inv * inv
                S = 1.0

            v = float(a[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end]  # view, not copy needed for reading
            infl = 2.0 * a[:Ls]  # influence vector

            max_const = 0.0
            if start > 0:
                max_const = max(max_const, float(c[:start].max()))
            if end < Ltot:
                max_const = max(max_const, float(c[end:].max()))

            # Grid
            x_hi = max(3.0 * v, 8.0 / n, 0.005)
            x_hi = min(x_hi, 5.0)  # cap to prevent huge values
            pts = np.linspace(0.0, x_hi, G_coarse + 1)
            if 0.0 <= v <= x_hi:
                # Add v to grid
                idx_v = round(v / x_hi * G_coarse)
                pts[idx_v] = v  # replace nearest grid point with exact v

            Δ = pts - v
            aff = base[None, :] + Δ[:, None] * infl[None, :]
            # Diagonal correction
            diag_off = i if i < Ls else -1
            if diag_off >= 0:
                aff[:, diag_off] += Δ * Δ

            aff = np.maximum(aff, max_const)
            mv = aff.max(axis=1)
            denom = S + Δ
            valid = denom > 1e-14
            scores = np.full(G_coarse + 1, math.inf)
            scores[valid] = (2.0 * n * mv[valid]) / (denom[valid] * denom[valid])

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            bx = float(pts[imin])
            bv = float(scores[imin])

            # Fine refine
            dw = x_hi / G_coarse
            lo = max(0.0, bx - 2.5 * dw)
            hi = min(x_hi, bx + 2.5 * dw)
            if hi > lo + 1e-15:
                fine_pts = np.linspace(lo, hi, G_fine)
                fine_pts = np.unique(np.concatenate([fine_pts, [v, 0.0]]))
                fine_pts = np.clip(fine_pts, 0.0, x_hi)
                Δf = fine_pts - v
                afff = base[None, :] + Δf[:, None] * infl[None, :]
                if diag_off >= 0:
                    afff[:, diag_off] += Δf * Δf
                afff = np.maximum(afff, max_const)
                mvf = afff.max(axis=1)
                denomf = S + Δf
                validf = denomf > 1e-14
                scoresf = np.full(fine_pts.size, math.inf)
                scoresf[validf] = (2.0 * n * mvf[validf]) / (denomf[validf] * denomf[validf])
                if np.any(np.isfinite(scoresf)):
                    im2 = int(np.argmin(scoresf))
                    if float(scoresf[im2]) < bv:
                        bv = float(scoresf[im2])
                        bx = float(fine_pts[im2])

            if math.isfinite(bv) and bv < cur_score - 1e-14:
                Δb = bx - v
                if abs(Δb) > 1e-15:
                    a[i] = bx
                    S += Δb
                    c[start:end] += Δb * infl
                    if diag_off >= 0:
                        c[start + diag_off] += Δb * Δb
                    cur_score = bv
                    improved = True

        # End of sweep
        S = float(a.sum())
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            a *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        # Periodic FFT recompute
        if sweep % 20 == 0:
            c = _conv_self(a)
        new_sc = 2.0 * n * float(c.max())
        if math.isfinite(new_sc) and new_sc < best_score - 1e-14:
            best_score = new_sc
            best_a = a.copy()
            consider(best_a)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 20.0:
                    # Perturb
                    a = a * np.exp(0.01 * rng.standard_normal(n))
                    a = np.maximum(a, 0.0)
                    ss = float(a.sum())
                    if ss > 1e-12:
                        a /= ss
                    c = _conv_self(a)
                    S = 1.0
                    cur_score = 2.0 * n * float(c.max())
                    flat = 0
                else:
                    break

        sweep += 1

    # Final
    ss = float(a.sum())
    if ss > 1e-300:
        a = a / ss
    c = _conv_self(a)
    sc = 2.0 * n * float(c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score
```
