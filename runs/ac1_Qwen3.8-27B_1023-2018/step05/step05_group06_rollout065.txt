    n = len(a)

    c = irfft(F * F, size)[:L]

def _softmax(u):

def _solve_lp(f, M, n, time_limit=8.0):
    t0 = time.monotonic()
    try:
        nnz = n * n
        row_idx = np.empty(nnz, dtype=np.int32)
        col_idx = np.empty(nnz, dtype=np.int32)
        vals = np.empty(nnz, dtype=np.float64)
        for j in range(n):
            s0 = j * n
            row_idx[s0:s0 + n] = np.arange(j, j + n)
            col_idx[s0:s0 + n] = j
            vals[s0:s0 + n] = f
        A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
        if time.monotonic() - t0 > time_limit * 0.4:
            return None
        from scipy.optimize import linprog
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={"time_limit": max(1.0, time_limit * 0.5)})
        if getattr(res, "success", False) and res.x is not None:
            b = np.maximum(res.x, 0.0)
            sb = float(b.sum())
            if sb > 1e-10:
                return b / sb
        return None
    except Exception:
        return None

    m_aff = float(c[i:i + n].max())
    m_val = max_const if max_const > m_aff else m_aff
    if S <= 1e-15:
        return S, math.inf
    return S, 2.0 * n * m_val / (S * S)

    # Final validation
    try:
        out = np.asarray(best_p, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(128, 1.0 / 128.0)
        ls = [float(x) for x in out]
        sc = float(evaluate_sequence(ls))
        if not math.isfinite(sc):
            try:
                pp = np.asarray(globals()["height_sequence_1"],
                                dtype=np.float64).ravel()
                pp = np.clip(pp, 0.0, 1000.0)
                sc = float(evaluate_sequence([float(x) for x in pp]))
                if math.isfinite(sc):
                    ls = [float(x) for x in pp]
            except Exception:
                pass
        if not math.isfinite(sc):
            ls = [1.0]
        return ls
    except Exception:
        return [1.0]
