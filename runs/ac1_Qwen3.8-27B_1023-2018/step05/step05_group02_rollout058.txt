            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    def run_chunk(uu, T, end_stage):
        try:
            f0, _ = lse_func(uu, T)
        except Exception:
            return uu
        if not math.isfinite(f0):
            return uu
        best_u = np.array(uu, dtype=np.float64, copy=True)
        best_f = float(f0)
        stale = 0
        while time.monotonic() < end_stage - 1.0:
            maxit = 30 if n > 2000 else 50
            opts = {
                "maxiter": maxit,
                "gtol": 1e-14,
                "ftol": 1e-16,
                "maxls": 30,
                "maxcor": 20,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == uu.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-16:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 4:
                            break
            if getattr(res, "status", 1) == 0:
                break
        return best_u

    stages = (
        (0.1, 0.08),
        (0.04, 0.10),
        (0.012, 0.12),
        (0.004, 0.14),
        (0.001, 0.18),
        (0.0003, 0.20),
        (0.0001, 0.18),
    )

    start = time.monotonic()
    avail = float(stop_time - start - 4.0)
    if avail <= 3.0:
        return p

    best_p = p.copy()
    best_sc = _score_fft(p)
    total_w = float(sum(wt for _, wt in stages))
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 2.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 2.0, start + avail * (cum / total_w))
        if end_stage <= now + 1.5:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        sc_cur = _score_fft(p_cur)
        if not math.isfinite(sc_cur):
            p = best_p.copy()
            u = p_to_u(p)
            continue
        if sc_cur < best_sc:
            best_sc = sc_cur
            best_p = p_cur.copy()
            p = p_cur.copy()
        else:
            p = best_p.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, max_sweeps=50000, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    G = 40 if n <= 1024 else 30 if n <= 4000 else 24
    G = int(max(16, G))

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls].copy()

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(30.0 * v, 80.0 * S / n, 0.05 * S, 1e-14)
            x_hi = min(x_hi, 12.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, S / n)

            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 <= v <= x_hi:
                pts = np.unique(np.concatenate((pts, [v])))

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                di = 2 * i - start
                if 0 <= di < Ls:
                    dval = base[di] + w * infl[di] + w * w
                    maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Golden section refinement
            dw = x_hi / float(G)
            lo = max(0.0, best_x - 2.0 * dw)
            hi = min(x_hi, best_x + 2.0 * dw)
            if hi <= lo:
                hi = lo + 1e-14
            gr = (math.sqrt(5.0) - 1.0) / 2.0
            g_lo = lo
            g_hi = hi
            a_gs = g_hi - gr * (g_hi - g_lo)
            b_gs = g_lo + gr * (g_hi - g_lo)
            fa = float(eval_xs(np.array([a_gs]))[0])
            fb = float(eval_xs(np.array([b_gs]))[0])
            for _ in range(16):
                if not math.isfinite(fa):
                    fa = 1e100
                if not math.isfinite(fb):
                    fb = 1e100
                if fa < fb:
                    g_hi = b_gs
                    b_gs = a_gs
                    fb = fa
                    a_gs = g_hi - gr * (g_hi - g_lo)
                    fa = float(eval_xs(np.array([a_gs]))[0])
                else:
                    g_lo = a_gs
                    a_gs = b_gs
                    fa = fb
                    b_gs = g_lo + gr * (g_hi - g_lo)
                    fb = float(eval_xs(np.array([b_gs]))[0])
                if g_hi - g_lo < 1e-16:
                    break
            gs_x = (g_lo + g_hi) / 2.0
            gv = float(eval_xs(np.array([gs_x]))[0])
            if math.isfinite(gv) and gv < best_val:
                best_x = gs_x
                best_val = gv

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= 10.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        np.maximum(c, 0.0, out=c)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-16:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 4.0:
                    noise_amt = 0.02 + 0.02 * math.exp(-flat * 0.15)
                    p = p * np.exp(rng.normal(0.0, noise_amt, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0.0:
                        p /= ss
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
                        best_score = cur_score
                        best_p = p.copy()
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _subgradient_polish(p, stop_time, consider=None, max_iters=2000):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    best_p = p.copy()
    best_score = float(2.0 * n * _conv_self(p).max())
    base_step = 0.004
    min_step = 1e-10
    stagnation = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 1.5:
            break

        c = _conv_self(p)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            break
        threshold = cmax * 0.9995
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        grad = np.zeros(n)
        for k in active:
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                contrib = p[kj[valid]] + p[valid]
                grad[valid] += contrib
        grad /= len(active)

        # Normalize gradient
        gn = float(np.linalg.norm(grad))
        if gn > 1e-300:
            grad /= gn

        step = base_step / (1.0 + 0.01 * it)
        if step < min_step:
            break

        found_improve = False
        for scale in (1.0, 0.5, 0.25, 0.1, 0.05):
            p_cand = p - scale * step * grad
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = float(2.0 * n * _conv_self(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-16:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                found_improve = True
                break

        if found_improve:
            stagnation = 0
        else:
            stagnation += 1
            base_step *= 0.92
            if stagnation > 50:
                break

        if it % 100 == 0 and consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 15.0
    usable = max(1.0, hard - t0)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(128, dtype=np.float64)
        best_arr /= 128.0
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    if best_score >= 2.0:
        # Parent is bad; use uniform as fallback
        best_arr = np.ones(int(parent.size if parent.size > 0 else 128), dtype=np.float64)
        best_arr /= best_arr.size
        best_score = float(_score_exact_arr(best_arr))

    n_best = int(best_arr.size)

    # === Phase 1: L-BFGS optimization (30% of budget) ===
    phase1_end = t0 + 0.30 * usable
    try:
        p2 = _lbfgs_optimize(best_arr.copy(), phase1_end, consider=consider)
        if p2 is not None and p2.size == n_best and np.all(np.isfinite(p2)) and p2.sum() > 0.0:
            p2 = p2 / float(p2.sum())
            consider(p2)
            best_arr = best_arr.copy()  # consider already updated
    except Exception:
        pass

    # === Phase 2: Coordinate descent (35% of budget) ===
    now = time.monotonic()
    phase2_end = min(t0 + 0.65 * usable, hard - 60.0)
    if phase2_end > now + 20.0:
        try:
            p3, _ = _coord_polish(best_arr, phase2_end, max_sweeps=100000, seed=int(seed), consider=consider)
            if p3 is not None and p3.size == n_best and p3.sum() > 0.0:
                p3 = p3 / float(p3.sum())
                consider(p3)
        except Exception:
            pass

    # === Phase 3: Subgradient + coord (15% of budget) ===
    now = time.monotonic()
    phase3_end = min(t0 + 0.80 * usable, hard - 30.0)
    if phase3_end > now + 15.0:
        # Subgradient
        sg_end = min(phase3_end - 15.0, now + 0.5 * (phase3_end - now))
        if sg_end > now + 10.0:
            try:
                p_sg = _subgradient_polish(best_arr, sg_end, consider=consider, max_iters=3000)
                if p_sg is not None and p_sg.size == n_best and p_sg.sum() > 0.0:
                    p_sg = p_sg / float(p_sg.sum())
                    consider(p_sg)
            except Exception:
                pass

        # Coord after subgradient
        now = time.monotonic()
        if phase3_end > now + 10.0:
            try:
                p3b, _ = _coord_polish(best_arr, phase3_end, max_sweeps=100000, seed=int(seed + 7777), consider=consider)
                if p3b is not None and p3b.size == n_best and p3b.sum() > 0.0:
                    p3b = p3b / float(p3b.sum())
                    consider(p3b)
            except Exception:
                pass

    # === Phase 4: Random perturbation + re-optimize (10% of budget) ===
    now = time.monotonic()
    phase4_end = min(t0 + 0.92 * usable, hard - 15.0)
    if phase4_end > now + 20.0:
        rng_top = np.random.default_rng(int(seed + 31337))
        n_pert = n_best
        # Try 2 perturbations
        n_tries = 2
        trial_budget = (phase4_end - now) / n_tries
        for trial in range(n_tries):
            t_start = time.monotonic()
            t_end = min(phase4_end, t_start + trial_budget)
            if t_end <= t_start + 10.0:
                break
            noise_amt = 0.03 + 0.02 * rng_top.random()
            p_pert = best_arr * np.exp(rng_top.normal(0.0, noise_amt, n_pert))
            p_pert = np.maximum(p_pert, 0.0)
            s_ptr = float(p_pert.sum())
            if s_ptr > 0:
                p_pert /= s_ptr
            consider(p_pert)
            # Quick L-BFGS
            lb_end = min(t_end - 8.0, t_start + 0.5 * (t_end - t_start))
            if lb_end > time.monotonic() + 8.0:
                try:
                    p_lb = _lbfgs_optimize(p_pert, lb_end, consider=consider)
                    if p_lb is not None and p_lb.size == n_pert:
                        p_pert = p_lb / float(p_lb.sum())
                        consider(p_pert)
                except Exception:
                    pass
            # Quick coord
            c_end = min(t_end - 1.0, time.monotonic() + max(10.0, 0.4 * (t_end - time.monotonic())))
            if c_end > time.monotonic() + 8.0:
                try:
                    p_c, _ = _coord_polish(p_pert, c_end, max_sweeps=50000, seed=int(seed + 50000 + trial * 100), consider=consider)
                    if p_c is not None and p_c.size == n_pert:
                        consider(p_c)
                except Exception:
                    pass

    # === Phase 5: Final polish (8% of budget) ===
    now = time.monotonic()
    final_end = min(hard - 5.0, t0 + budget - 8.0)
    if final_end > now + 10.0:
        # Final coordinate descent
        try:
            p_final, _ = _coord_polish(best_arr, final_end, max_sweeps=100000, seed=int(seed + 999999), consider=consider)
            if p_final is not None and p_final.size == n_best and p_final.sum() > 0.0:
                p_final = p_final / float(p_final.sum())
                consider(p_final)
        except Exception:
            pass

    # === Final validation ===
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
