Now for the coordinate descent, let me improve it with the analytical 1D solution:

For coordinate i with value v, the affected convolution segment is m ∈ [i, i+n-1].
Let C = current convolution (length 2n-1).
Let a = current sequence (length n, sum = S ≈ 1 after normalization).

The change: we set a[i] = w (instead of v).
- New sum: S' = S + (w - v)
- For m in [i, i+n-1]: new_c[m] = C[m] - v*a[m-i] + w*a[m-i] = C[m] + (w-v)*a[m-i]
  Wait, that's not right. Let me re-derive.
  
  C[m] = sum_j a[j]*a[m-j]. The terms involving a[i] are:
  - j = i: a[i]*a[m-i]
  - j = m-i: a[m-i]*a[i] (same term if we count both)
  
  Actually, in np.convolve, C[m] = sum_{j=0}^{n-1} a[j]*a[m-j] where a is zero-padded outside [0,n-1].
  
  So C[m] = a[i]*a[m-i] + sum_{j≠i} a[j]*a[m-j] = v * a[m-i] + C̄[m]
  
  where C̄[m] = C[m] - v * a[m-i] (the "partial" convolution excluding the a[i] term).
  
  But wait: a[m-i] might equal a[i] itself when m-i = i, i.e., m = 2i. In that case:
  C[m=2i] = v * a[i] + C̄[2i] = v² + C̄[2i].
  
  So if we change a[i] from v to w:
  new_C[m] = C̄[m] + w * a[m-i] for m in [i, i+n-1]
  new_C[m] = C[m] for m outside [i, i+n-1]
  
  But when m-i = i (m = 2i), a[m-i] = a[i] = w (the new value!), so:
  new_C[2i] = C̄[2i] + w * w = C̄[2i] + w²
  
  This is the diagonal term that makes it non-linear!
  
  For m ≠ 2i in the affected range: new_C[m] = C̄[m] + w * a[m-i] (linear in w)
  For m = 2i: new_C[2i] = C̄[2i] + w² (quadratic in w)

So the 1D subproblem is:
minimize f(w) = 2n * max(M_ext, max_{m≠2i, m∈[i,i+n-1]} (C̄[m] + w*b[m]), C̄[2i] + w²) / (S - v + w)²

where b[m] = a[m-i] for m ≠ 2i, and S' = S - v + w.

This is still complex but solvable. For the analytical approach:
- The quadratic w² + C̄[2i] is always convex and increasing for w > 0
- The linear terms C̄[m] + w*b[m] are linear
- M_ext is constant

The max of all these is convex in w. The score f(w) = 2n * (convex) / (linear)² is not necessarily unimodal, but in practice it should be close to unimodal.

For speed, I'll use a hybrid:
1. Compute the "natural" optimum w from the diagonal: w_cand = sqrt(max(0, target - C̄[2i])) where target is the desired max level
2. Compute the "balance" optimum from the top linear constraint
3. Check a few candidate w values
4. Pick the best

def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.asarray(c, dtype=np.float64)


def _score_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()

    s = float(e.sum())

        c = irfft(F * F, conv_size)[:conv_len]

def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s

                best_x = float(pts[imin])
                best_val = float(scores[imin])

                if math.isfinite(best_val) and best_val < cur_score - 1e-15 and 0.0 <= best_x <= x_hi:
                    w_new = best_x - v
                    if abs(w_new) > 1e-15:
                        p[i] = best_x
                        aff_seg = base + w_new * infl
                        if 0 <= diag < Ls:
                            aff_seg = aff_seg.copy()
                            aff_seg[diag] += w_new * w_new
                        new_max = max(float(aff_seg.max()), max_const)
                        c[start_m:end_m] = aff_seg
                        cur_score = 2.0 * n * new_max
                        improved = True

        s_new = float(p.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            p /= s_new
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-14:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 8:
                if time.monotonic() < stop_time - 15.0:
                    p = p * np.exp(rng.normal(0.0, 0.02, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 5.0:
        hard = t0 + max(3.0, budget - 2.0)

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float('inf')

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()

    consider(parent)
    rng = np.random.default_rng(seed)

    n_work = int(best_arr.size)
    if n_work < 128:
        n_work = 384
    if n_work > 4096:
        n_work = 3072

    # Phase 1: LSE optimization
    lse_time = min(150.0, 0.14 * budget)
    p_work = _resample(best_arr, n_work)
    lse_end = t0 + lse_time
    if lse_end > t0 + 15.0:
        try:
            p2 = _lse_optimize(p_work, lse_end, consider=consider)
            if p2 is not None and p2.size == n_work and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p_work = p2 / float(p2.sum())
                consider(p_work)
        except Exception:
            pass

    # Phase 2: Main coordinate descent (bulk of time)
    cd_end1 = t0 + 0.42 * budget
    if cd_end1 > time.monotonic() + 60.0:
        p_start = _resample(best_arr, n_work)
        try:
            pc, _ = _coord_descent(p_start, cd_end1, consider=consider, seed=int(seed + 1111))
            if pc is not None and pc.size > 0 and np.all(np.isfinite(pc)) and pc.sum() > 0:
                consider(pc)
        except Exception:
            pass

    # Phase 3: Block coordinate descent for different locality
    cd_end2 = t0 + 0.62 * budget
    if cd_end2 > time.monotonic() + 30.0:
        p_start = _resample(best_arr, n_work)
        try:
            pb, _ = _block_cd(p_start, cd_end2, consider=consider, seed=int(seed + 2222), block_size=3)
            if pb is not None and pb.size > 0 and np.all(np.isfinite(pb)) and pb.sum() > 0:
                consider(pb)
        except Exception:
            pass

    # Phase 4: Perturbation and re-optimize
    rem = t0 + 0.78 * budget - time.monotonic()
    if rem > 40.0:
        p_pert = _resample(best_arr, n_work) * np.exp(rng.normal(0.0, 0.04, n_work))
        p_pert = np.maximum(p_pert, 0.0)
        p_pert /= float(p_pert.sum())
        consider(p_pert)
        perturb_end = min(t0 + 0.78 * budget, time.monotonic() + rem * 0.7)
        if perturb_end > time.monotonic() + 20.0:
            try:
                pp, _ = _coord_descent(p_pert, perturb_end, consider=consider, seed=int(seed + 3333))
                if pp is not None and pp.size > 0 and np.all(np.isfinite(pp)) and pp.sum() > 0:
                    consider(pp)
            except Exception:
                pass

    # Phase 5: Final CD squeeze on overall best
    final_end = hard - 5.0
    rem = final_end - time.monotonic()
    if rem > 30.0:
        p_final = _resample(best_arr, n_work)
        try:
            pf, _ = _coord_descent(p_final, final_end, consider=consider, seed=int(seed + 4444))
            if pf is not None and pf.size > 0 and np.all(np.isfinite(pf)) and pf.sum() > 0:
                consider(pf)
        except Exception:
            pass

    # Phase 6: Quick block CD if time remains
    final_end2 = hard - 5.0
    rem = final_end2 - time.monotonic()
    if rem > 20.0:
        p_final = _resample(best_arr, n_work)
        try:
            pf, _ = _block_cd(p_final, final_end2, consider=consider, seed=int(seed + 5555), block_size=2)
            if pf is not None and pf.size > 0 and np.all(np.isfinite(pf)) and pf.sum() > 0:
                consider(pf)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
