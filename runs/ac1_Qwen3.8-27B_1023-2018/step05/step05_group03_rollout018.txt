import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize, linprog
from scipy import sparse as sp

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:
        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if not math.isfinite(s) or s <= 1e-30:
        p = np.ones(3072, dtype=np.float64)
    return p / s

def _fft_conv(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = a.size
    c = _fft_conv(a)
    m = float(c.max())
    return 2.0 * n * m

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s < 0.01:
        return None
    return a / s

def _softmax(u):
    u = np.asarray(u, dtype=np.float64).ravel()
    u = np.where(np.isfinite(u), u, 0.0)
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size)
    out = e / s
    return np.maximum(out, 1e-300)

def _psu(u):
    p = np.maximum(u, 1e-15)
    uu = np.log(p)
    uu -= float(np.mean(uu))
    return uu

def _lse_func(u, T, n, conv_size, xcorr_size):
    p = _softmax(u)
    F = rfft(p, conv_size)
    c = irfft(F * F, conv_size)[:2*n-1]
    c = np.maximum(c, 0.0)
    cmax = float(c.max())
    if not math.isfinite(cmax) or cmax <= 0:
        return 1e100, np.zeros(n)
    scaled = (c - cmax) / max(T, 1e-300)
    w = np.exp(scaled)
    wsum = float(w.sum())
    if not math.isfinite(wsum) or wsum <= 0:
        w = np.zeros(2*n-1)
        w[int(np.argmax(c))] = 1.0
        wsum = 1.0
    else:
        w /= wsum
    f = cmax + T * math.log(wsum / (2*n-1))
    Wf = rfft(w, xcorr_size)
    Pf = rfft(p, xcorr_size)
    corr = irfft(np.conj(Wf) * Pf, xcorr_size)[:n]
    grad = 2.0 * corr
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(f), grad

def _lse_optimize(p, stop_time, n, fine=False):
    conv_len = 2 * n - 1
    conv_size = next_fast_len(conv_len)
    xcorr_size = next_fast_len(3 * n - 2)
    u = _psu(p)
    best_u = u.copy()
    best_f = _lse_func(u, 0.05, n, conv_size, xcorr_size)[0]

    if fine:
        stages = [(0.005, 0.20), (0.002, 0.25), (0.0007, 0.30), (0.0002, 0.25)]
    else:
        stages = [(0.04, 0.15), (0.015, 0.15), (0.005, 0.18), (0.0015, 0.20), (0.0004, 0.22), (0.0001, 0.10)]

    total_w = sum(w for _, w in stages)
    cum = 0.0
    start = time.monotonic()
    avail = max(0.0, stop_time - start - 5.0)

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 8.0:
            break
        cum += wt
        end_stage = min(stop_time - 5.0, start + avail * (cum / total_w))
        if end_stage <= now + 5.0:
            continue
        c = _fft_conv(p)
        cmax = float(c.max()) if c.size else 1.0
        T = max(1e-12, frac * cmax)

        stale = 0
        while time.monotonic() < end_stage - 3.0 and stale < 3:
            maxit = 20 if n > 2000 else 30
            try:
                res = minimize(
                    _lse_func, best_u, args=(T, n, conv_size, xcorr_size),
                    method="L-BFGS-B", jac=True,
                    options={"maxiter": maxit, "gtol": 1e-10, "ftol": 1e-15, "maxls": 15, "maxcor": 20}
                )
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fval = float(res.fun) if hasattr(res, 'fun') and np.isfinite(res.fun) else float('inf')
                if fval < best_f - 1e-12:
                    best_f = fval
                    best_u = np.array(res.x, copy=True)
                    stale = 0
                else:
                    stale += 1
            else:
                break

        new_p = _softmax(best_u)
        new_p = np.maximum(new_p, 0.0)
        ss = float(new_p.sum())
        if math.isfinite(ss) and ss > 0:
            p = new_p / ss
        u = _psu(p)

    return p

def _lp_direction(a, M, n, time_limit=30.0):
    if linprog is None or sp is None:
        return None
    if n < 16 or n > 5000:
        return None
    if not math.isfinite(M) or M <= 0:
        return None
    t_start = time.monotonic()
    try:
        idx = np.arange(n, dtype=np.int32)
        cols_idx = np.repeat(idx, n)
        off = np.tile(idx, n)
        rows_idx = cols_idx + off - (n - 1)
        valid = (rows_idx >= 0) & (rows_idx < 2*n-1)
        cols_idx = cols_idx[valid]
        rows_idx = rows_idx[valid]
        data = a[off[valid]].astype(np.float64)
        A = sp.coo_matrix((data, (rows_idx, cols_idx)), shape=(2*n-1, n)).tocsr()
        cobj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        res = linprog(cobj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if time.monotonic() - t_start > time_limit:
            return None
        if res is not None and getattr(res, 'success', False):
            b = np.asarray(res.x, dtype=np.float64).ravel()
            b = np.where(np.isfinite(b), b, 0.0)
            b = np.maximum(b, 0.0)
            s = float(b.sum())
            if s > 1e-15:
                return b / s
    except Exception:
        pass
    return None

def _lp_refine(a, stop_time, n, max_rounds=6):
    best = a.copy()
    best_sc = _score(a)
    for _ in range(max_rounds):
        if time.monotonic() > stop_time - 15.0:
            break
        c = _fft_conv(a)
        M = float(c.max())
        if not math.isfinite(M) or M <= 0:
            break
        b = _lp_direction(a, M, n, time_limit=min(25.0, max(5.0, stop_time - time.monotonic() - 20.0)))
        if b is None:
            break
        improved = False
        for t in (0.6, 0.4, 0.25, 0.15, 0.08, 0.04):
            cand = (1.0 - t) * a + t * b
            cand = np.maximum(cand, 0.0)
            ss = float(cand.sum())
            if not math.isfinite(ss) or ss < 0.01:
                continue
            cand /= ss
            sc = _score(cand)
            if math.isfinite(sc) and sc < best_sc - 1e-11:
                best_sc = sc
                best = cand.copy()
                a = cand
                improved = True
                break
        if not improved:
            break
    return best, best_sc

def _coord_descent(p, stop_time, n, seed=12345, G=80):
    p = p.copy()
    c = _fft_conv(p)
    cur = 2.0 * n * float(c.max())
    best = p.copy()
    best_sc = cur
    S = 1.0
    infl = 2.0 * p  # influence vector: dc[k]/dp[i] contribution
    rng = np.random.default_rng(seed)
    L = 2 * n - 1
    flat = 0
    sweep = 0

    while sweep < 500000 and time.monotonic() < stop_time - 2.0:
        improved_any = False
        order = rng.permutation(n)
        for pos in range(n):
            if time.monotonic() > stop_time - 0.5:
                return best, best_sc
            if S <= 1e-300:
                break
            if abs(S - 1.0) > 0.15:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                infl *= inv
                S = 1.0
                np.maximum(c, 0.0, out=c)

            i = int(order[pos])
            v = float(p[i])
            start_i = i
            end_i = min(i + n, L)
            Ls = end_i - start_i
            if Ls <= 0:
                continue
            base = c[start_i:end_i].copy()
            max_rest = 0.0
            if start_i > 0:
                mr = float(c[:start_i].max())
                max_rest = max(max_rest, mr)
            if end_i < L:
                mr = float(c[end_i:].max())
                max_rest = max(max_rest, mr)

            def eval_x(x):
                w = x - v
                den = S + w
                if den <= 1e-14:
                    return 1e100
                aff = base + w * infl[start_i:end_i]
                di = i - start_i
                aff[di] = base[di] + w * infl[di] + w * w
                np.maximum(aff, max_rest, out=aff)
                mv = float(np.max(aff))
                return 2.0 * n * mv / (den * den)

            # Geometric + linear search
            x_hi = max(25.0 * max(v, 1e-12), 25.0 * S / n, 0.002)
            x_hi = min(x_hi, max(0.02, 5.0 * v))
            x_hi = min(x_hi, 1.0)
            pts = [0.0, v]
            pts.extend(np.geomspace(max(1e-10, x_hi * 1e-6), x_hi, G).tolist())
            if v > 1e-10:
                pts.extend((v * np.geomspace(0.1, 10.0, G // 2)).tolist())
            pts.extend(np.linspace(0.0, min(x_hi, 3.0 * S / n), G // 2).tolist())
            pts = np.unique(np.clip(np.asarray(pts), 0.0, x_hi))
            vals = np.array([eval_x(x) for x in pts])
            finite = np.where(np.isfinite(vals))[0]
            if len(finite) == 0:
                continue
            im = finite[np.argmin(vals[finite])]
            bx = float(pts[im])
            bv = float(vals[im])

            # Fine search around best
            step = max(v, S / n, 1e-9)
            lo = max(0.0, bx - 3.0 * step)
            hi = min(x_hi, bx + 3.0 * step)
            fine_pts = np.linspace(lo, hi, 40)
            if lo > 0:
                fine_pts = np.concatenate([fine_pts, np.geomspace(lo, hi, 20)])
            else:
                fine_pts = np.concatenate([fine_pts, np.linspace(0.0, hi, 20)])
            fine_pts = np.unique(np.clip(fine_pts, 0.0, x_hi))
            fvals = np.array([eval_x(x) for x in fine_pts])
            ffinite = np.where(np.isfinite(fvals))[0]
            if len(ffinite) > 0:
                fim = ffinite[np.argmin(fvals[ffinite])]
                if fvals[fim] < bv:
                    bx = float(fine_pts[fim])
                    bv = float(fvals[fim])

            if math.isfinite(bv) and bv < cur - 1e-12 and 0.0 <= bx:
                d = bx - v
                if abs(d) > 1e-15:
                    c[start_i:end_i] += d * infl[start_i:end_i]
                    diag = 2 * i
                    if 0 <= diag < L:
                        c[diag] += d * d
                    p[i] = bx
                    infl[i] = 2.0 * bx
                    S += d
                    cur = bv
                    improved_any = True

        # Renormalize
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            np.maximum(c, 0.0, out=c)
            infl *= inv
            S = 1.0

        check = 2.0 * n * float(c.max())
        if math.isfinite(check) and check < best_sc - 1e-12:
            best_sc = check
            best = p.copy()

        if improved_any:
            flat = 0
        else:
            flat += 1
            if flat >= 5 and time.monotonic() < stop_time - 40.0:
                sig = 0.04 + 0.04 * min(flat - 5, 4)
                noise = rng.normal(0.0, sig, n)
                q = p * np.exp(noise)
                q = np.maximum(q, 0.0)
                qs = float(q.sum())
                if math.isfinite(qs) and qs > 0:
                    p = q / qs
                    c = _fft_conv(p)
                    infl = 2.0 * p
                    S = 1.0
                    cur = 2.0 * n * float(c.max())
                    if math.isfinite(cur) and cur < best_sc:
                        best_sc = cur
                        best = p.copy()
                flat = 0
            elif flat >= 10:
                break
        sweep += 1

    return best, best_sc

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    hard = t0 + budget - 12.0

    parent = _get_parent()
    n = int(parent.size)
    best_arr = parent.copy()
    best_score = _score(best_arr)

    def consider(a):
        nonlocal best_arr, best_score
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if not math.isfinite(s) or s < 0.01:
            return
        a = a / s
        try:
            sc = _score(a)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()

    # Phase 1: Quick shaped perturbations
    if time.monotonic() < hard - 200.0:
        base = best_arr.copy()
        for q in (0.6, 0.8, 1.1, 1.4):
            cand = np.power(np.maximum(base, 0.0), q)
            cand = _normalize(cand)
            if cand is not None:
                consider(cand)
        rng_top = np.random.default_rng(seed)
        for _ in range(4):
            if time.monotonic() > hard - 200.0:
                break
            side = 1.0 if rng_top.random() < 0.5 else -1.0
            amp = 0.4 + 1.0 * rng_top.random()
            x = (np.arange(n) + 0.5) / n
            cand = base * np.exp(side * amp * (2.0 * x - 1.0))
            cand = _normalize(cand)
            if cand is not None:
                consider(cand)

    # Phase 2: LSE optimization (main)
    lse_stop = min(hard - 250.0, t0 + 0.45 * (budget - 20.0))
    if lse_stop > time.monotonic() + 30.0:
        p = _lse_optimize(best_arr, lse_stop, n, fine=False)
        consider(p)

    # Phase 3: LP refinement
    lp_stop = min(hard - 150.0, time.monotonic() + 0.22 * (budget - 20.0))
    if lp_stop > time.monotonic() + 20.0:
        p_lp, _ = _lp_refine(best_arr, lp_stop, n, max_rounds=5)
        consider(p_lp)

    # Phase 4: Coordinate descent (primary polish)
    cd_stop = min(hard - 80.0, time.monotonic() + 0.22 * (budget - 20.0))
    if cd_stop > time.monotonic() + 20.0:
        p_cd, _ = _coord_descent(best_arr, cd_stop, n, seed=seed + 777, G=72)
        consider(p_cd)

    # Phase 5: Fine LSE
    fl_stop = min(hard - 120.0, time.monotonic() + 0.12 * (budget - 20.0))
    if fl_stop > time.monotonic() + 20.0:
        p_fl = _lse_optimize(best_arr, fl_stop, n, fine=True)
        consider(p_fl)

    # Phase 6: Final coordinate polish
    fc_stop = min(hard - 15.0, time.monotonic() + 0.80 * max(0.0, hard - 15.0 - time.monotonic()))
    if fc_stop > time.monotonic() + 15.0:
        p_fc, _ = _coord_descent(best_arr, fc_stop, n, seed=seed + 99999, G=88)
        consider(p_fc)

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = parent / float(parent.sum())
        out = out / float(out.sum())
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent / float(parent.sum())]
            sc2 = evaluate_sequence(ls)
            if not math.isfinite(sc2):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
