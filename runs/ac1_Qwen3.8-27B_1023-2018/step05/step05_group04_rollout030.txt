**LP-based approach (Frank-Wolfe variant):**

Given current a (normalized, sum=1), compute M = max((a*a)).
Solve LP: max sum(b) s.t. b ≥ 0, (a*b)[k] ≤ M for all k.

The constraint (a*b)[k] ≤ M means: sum_j a[j]*b[k-j] ≤ M for k = 0, 1, ..., 2n-2.

This is a linear program with n variables and 2n-1 constraints. For n=3072, that's 3072 variables and 6143 constraints. CBC or GLPK should handle this in a few seconds.

After solving, get b*. Normalize: g = b* / sum(b*). Then blend: a ← (1-t)*a + t*g for some step size t.

The step size can be chosen by line search: minimize score((1-t)*a + t*g) over t ∈ [0, 1].

This is essentially a conditional gradient (Frank-Wolfe) method where the linear oracle is replaced by an LP oracle. The LP gives a better direction than just the minimum coordinate.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.linalg import toeplitz
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.asarray(c, dtype=np.float64)


def _score_normalized(a):
    """Score for normalized sequence (sum=1): 2*n*max(conv)."""
    n = a.size
    if n == 0:
        return float('inf')
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return float('inf')
    return 2.0 * n * m


def _resample(src, n_target):
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.ones(n_target) / n_target
    if src.size == n_target:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m) + 0.5) / m
        xp_new = (np.arange(n_target) + 0.5) / n_target
        out = np.interp(xp_new, xp, src)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.ones(n_target) / n_target
    return out / s

    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard_deadline = t0 + budget - 15.0

    parent = _get_parent()
    n = int(parent.size)

    p = parent.copy().astype(np.float64)
    s = float(p.sum())
    if s < 1e-12:
        p = np.ones(n) / n
    else:
        p = p / s

    best_p = p.copy()
    best_score = _score_normalized(p)

    def consider(arr):
        nonlocal best_p, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 0.01 or a.size == 0:
                return
            a = a / ss
            sc = _score_normalized(a)
            if math.isfinite(sc) and sc < best_score:
                best_score = sc
                best_p = a.copy()
        except Exception:
            pass

    def cd_optimize(p_init, stop_time, gs_iters=35, seed_cd=42):
        """Coordinate descent with golden section. Returns (best_p, best_score)."""
        nn = int(p_init.size)
        if nn == 0:
            return p_init.copy(), float('inf')
        p = p_init.copy()
        c = _conv_self(p)
        Ltot = 2 * nn - 1
        S = 1.0
        cur_score = _score_normalized(p)
        best_local = cur_score
        best_local_p = p.copy()
        GR = (math.sqrt(5.0) - 1.0) / 2.0
        rng = np.random.default_rng(seed_cd)
        flat_count = 0

        while time.monotonic() < stop_time - 5.0:
            order = rng.permutation(nn)
            improved_sweep = False

            for idx in order:
                if time.monotonic() > stop_time - 3.0:
                    break
                i = int(idx)
                v = float(p[i])

                start = i
                end = min(i + nn, Ltot)
                Ls = end - start
                if Ls <= 0:
                    continue

                base = c[start:end]
                infl_total = p[:Ls] + v
                diag_idx = 2 * i - start

                max_const = 0.0
                if start > 0:
                    mc = float(c[:start].max())
                    if mc > max_const:
                        max_const = mc
                if end < Ltot:
                    mc = float(c[end:].max())
                    if mc > max_const:
                        max_const = mc

                w_hi = max(3.0 * v + 2.0 / nn, 5.0 / nn)

                if w_hi - w_lo < 1e-15:
                    continue

                    aff = base + w * infl_total
                    aff_max = float(aff.max())
                    if 0 <= diag_idx < Ls:
                        dq = base[diag_idx] + w * infl_total[diag_idx] + w * w
                        if dq > aff_max:

                    m = max(aff_max, max_const)
                    s_new = S + w
                    if s_new <= 1e-15:
                        return float('inf')
                    return (2.0 * nn * m) / (s_new * s_new)

                a_gs = w_lo
                b_gs = w_hi
                c_gs = b_gs - GR * (b_gs - a_gs)
                d_gs = a_gs + GR * (b_gs - a_gs)
                f_c = phi(c_gs)
                f_d = phi(d_gs)

                for _ in range(gs_iters):
                    if b_gs - a_gs < 1e-11 * max(1.0, abs(w_hi)):
                        break
                    if f_c < f_d:
                        b_gs = d_gs
                        d_gs = c_gs
                        f_d = f_c
                        c_gs = b_gs - GR * (b_gs - a_gs)
                        f_c = phi(c_gs)
                    else:
                        a_gs = c_gs
                        c_gs = d_gs
                        f_c = f_d
                        d_gs = a_gs + GR * (b_gs - a_gs)
                        f_d = phi(d_gs)

                best_w = (a_gs + b_gs) / 2.0
                best_phi = phi(best_w)

                # Check endpoints

                if f_lo < best_phi:
                    best_phi = f_lo
                    best_w = w_lo
                f_hi = phi(w_hi)
                if f_hi < best_phi:
                    best_phi = f_hi
                    best_w = w_hi

                if math.isfinite(best_phi) and best_phi < cur_score - 1e-14:
                    p[i] = v + best_w
                    S += best_w
                    aff_seg = base + best_w * infl_total
                    if 0 <= diag_idx < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag_idx] += best_w * best_w
                    c[start:end] = aff_seg
                    cur_score = best_phi
                    improved_sweep = True

            # Renormalize
            s_new = float(p.sum())
            if s_new > 1e-300:
                p = p / s_new
                c = _conv_self(p)
                S = 1.0
                cur_score = _score_normalized(p)

            if cur_score < best_local - 1e-15:
                best_local = cur_score
                best_local_p = p.copy()
                flat_count = 0
            else:
                flat_count += 1
                if flat_count >= 8 and time.monotonic() < stop_time - 30.0:
                    # Kick to escape local optimum
                    kick_sigma = 0.03 + 0.03 * min(1.0, flat_count / 30.0)
                    p = p * np.exp(rng.normal(0.0, kick_sigma, nn))
                    p = np.maximum(p, 0.0)
                    ps = float(p.sum())
                    if ps > 1e-300:
                        p = p / ps
                        c = _conv_self(p)
                        cur_score = _score_normalized(p)
                    flat_count = 0

        return best_local_p, best_local

    def lp_global_move(a, stop_time):
        """LP-based global move. Returns improved candidate or None."""
        nn = int(a.size)
        if nn > 2048:
            return None
        a = a.copy()
        s = float(a.sum())
        if s < 1e-12:
            return None
        a = a / s
        c = _conv_self(a)
        M = float(c.max())
        if not math.isfinite(M) or M <= 0:
            return None

        # Build convolution matrix A[k,l] = a[k-l] if 0 <= k-l < nn
        col = np.zeros(2 * nn - 1)
        col[:nn] = a
        row = np.zeros(nn)
        row[0] = a[0]
        A = toeplitz(col, row)

        c_obj = -np.ones(nn)
        b_ub = np.full(2 * nn - 1, M)
        bounds = [(0.0, None)] * nn

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': max(1, stop_time - time.monotonic() - 10)})
        except Exception:
            return None

        if res.status != 0 or res.x is None:
            return None

        b = np.maximum(res.x, 0.0)
        bsum = float(b.sum())
        if bsum < 1e-12:
            return None
        g = b / bsum

        # Line search for blend
        best_blend = a.copy()
        best_blend_sc = _score_normalized(a)
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:

            blend = np.maximum(blend, 0.0)
            bs = float(blend.sum())

                continue
            blend = blend / bs
            sc = _score_normalized(blend)
            if math.isfinite(sc) and sc < best_blend_sc:
                best_blend_sc = sc
                best_blend = blend

        return best_blend

    # === Main optimization loop ===

    # Phase 1: Initial CD on parent (60% of budget)
    phase1_end = t0 + 0.60 * budget
    if phase1_end > t0 + 30.0:
        try:
            p1, sc1 = cd_optimize(p, min(phase1_end, hard_deadline - 60.0), gs_iters=35, seed_cd=int(seed))
            consider(p1)
        except Exception:
            pass

    # Phase 2: LP-based global moves on a shorter version (20% of budget)
    phase2_end = t0 + 0.80 * budget
    rem2 = max(0, phase2_end - time.monotonic() - 30.0)

        while time.monotonic() < lp_stop:

            except Exception:

    rem3 = max(0, hard_deadline - 20.0 - time.monotonic())

    if rem4 > 15.0:

            pass

    try:
        out = np.asarray(best_p, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
