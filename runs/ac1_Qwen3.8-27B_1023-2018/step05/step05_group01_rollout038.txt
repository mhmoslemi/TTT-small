def _cross_corr(w, a):
    """Compute cross-correlation (w ⊛ a)[i] = sum_k w[k]*a[k-i] for i=0..n-1."""
    n = a.size
    cross_size = next_fast_len(3 * n - 2)
    F = rfft(a[::-1], cross_size)
    G = rfft(w, cross_size)
    cc = irfft(F * G, cross_size)[:3*n-2]

def _lse_optimize(p, stop_time, consider=None, seed=42):

    n = p.size

        return p

        m = float(np.max(u))
        e = np.exp(u - m)

        if s <= 0 or not math.isfinite(s):
            return np.full(u.size, 1.0/u.size)
        return e / s
