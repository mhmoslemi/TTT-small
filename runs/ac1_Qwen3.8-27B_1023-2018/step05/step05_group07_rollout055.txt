Gradient of $M_\tau(a)$:

$\frac{\partial M_\tau}{\partial a_i} = \frac{1}{\sum_k e^{c_k/\tau}} \sum_k e^{c_k/\tau} \frac{\partial c_k}{\partial a_i}$

$\frac{\partial c_k}{\partial a_i} = 2a[k-i]$ (if $0 \leq k-i < n$)

import numpy as np
import time
from scipy.optimize import linprog


def _build_A(n, a):
    """Build (2n-1) x n convolution matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_solve(A, M, n, tl=6.0):
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tl,
                               'dual_feasibility_tolerance': 1e-11,
                               'primal_feasibility_tolerance': 1e-11})
        if res.status == 0 and res.x is not None:
            x = np.clip(res.x, 0, None)
            if x.sum() > 1e-10:
                return x
    except Exception:
        pass
    return None


def _line_search(a, g, M_init):
    """Best t in [0,1] for cand = (1-t)*a + t*g, normalized."""
    n = a.size

            continue

def _subgrad_step(a, c, M, alphas):
    n = a.size
    active = np.where(c >= M * 0.9995)[0]
    if len(active) == 0:
        return None
    grad = np.zeros(n)
    for k_s in active:
        ks = int(k_s)
        lo = max(0, ks - n + 1)
        hi = min(n, ks + 1)
        for j in range(lo, hi):
            idx = ks - j
            if 0 <= idx < n:
                grad[j] += 2.0 * a[idx]
    grad -= grad.mean()
    gn = float(np.linalg.norm(grad))
    if gn < 1e-14:
        return None

        cand = np.maximum(a - alpha * grad / gn, 0.0)

            continue
        cand /= cs

    return None
