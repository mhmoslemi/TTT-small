Now for the coordinate descent, I'll improve the previous version:

Key improvements:
1. Precompute the "outside max" more efficiently
2. Use a larger grid for the initial search
3. Add a "swap" move (exchange values between two positions)
4. Better kick mechanism

Actually, let me think about the "swap" move. If we swap a[i] and a[j], the convolution changes in a complex way. It might not be worth the implementation effort. Let me stick with the 1D coordinate update.

Another idea: "Block coordinate descent" where we update 2 consecutive coordinates (a[i], a[i+1]) simultaneously. The effect on the convolution is a 2-variable quadratic, which we can optimize in 2D. But this is more complex to implement correctly.

Let me focus on making the 1D CD as fast as possible and run many sweeps.

For n=3072:
- Each CD sweep: 3072 updates, each O(n) = O(3072^2) ≈ 9.4M ops
- numpy can do ~100M ops/sec for simple vector ops
- So one sweep: ~0.1 sec
- In 200 seconds: ~2000 sweeps

That's a lot of sweeps. The previous program might not be using all the time effectively.

Let me also think about the "flat" detection. If no improvement in 4 sweeps, the previous program kicks. This might be too aggressive. Let me use 8-10 sweeps before kicking.

Also, the kick in the previous program multiplies by exp(Noises). A better kick might be:
- Add a small random vector: p += sigma * randn(n)
- Or: p = p * (1 + sigma * randn(n)) (multiplicative noise)
- Or: p = p + sigma * (rand(n) - 0.5) (additive noise)
- Then clip to >= 0 and renormalize

The multiplicative noise preserves the "shape" better. Let me use that.

Let me also add a "cooling" schedule for the kick size: start with larger kicks and decrease over time.

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p

def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.asarray(c, dtype=np.float64)

def _score_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float('inf')
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return float('inf')
    return float(2.0 * n * m / (s * s))

def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.shape, 1.0 / u.size, dtype=np.float64)
    return e / s

def _lse_optimize(p, stop_time, consider=None):
    n = p.size
    p = np.maximum(np.asarray(p, dtype=np.float64), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    u = np.log(np.maximum(p, 1e-30), dtype=np.float64)
    u = u - float(u.mean())

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(a, w):
        # Cross-correlation: result[i] = sum_j w[i+j] * a[j]
        # This is the cross-correlation of w and a
        F = rfft(a[::-1], cross_size)
        G = rfft(w, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return f, grad

    stages = [
        (0.2, 0.10),
        (0.08, 0.15),
        (0.03, 0.20),
        (0.01, 0.22),
        (0.004, 0.20),
        (0.001, 0.13),
    ]

    best_p = p.copy()
    best_sc = _score_arr(p)
    start = time.monotonic()
    avail = max(0.0, stop_time - start - 10.0)
    total_w = sum(w for _, w in stages)
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        cum += wt
        end_stage = min(stop_time - 4.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)
        best_u = u.copy()
        try:
            f0, _ = lse_func(u, T)
        except Exception:
            f0 = 1e100
        best_fstage = f0
        stale = 0
        while time.monotonic() < end_stage - 1.5:
            maxit = 30 if n > 2000 else 45
            opts = {"maxiter": maxit, "gtol": 1e-11, "ftol": 1e-15, "maxls": 10, "maxdisp": 0}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_fstage - 1e-14:
                    best_fstage = fval
                    best_u = res.x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if getattr(res, "status", 1) == 0:
                break
        p_cur = _softmax(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        sc = _score_arr(p_cur)
        if math.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best_p = p_cur.copy()
            p = p_cur.copy()
            u = np.log(np.maximum(p, 1e-30), dtype=np.float64)
            u = u - float(u.mean())
        else:
            p = best_p.copy()
            u = np.log(np.maximum(p, 1e-30), dtype=np.float64)
            u = u - float(u.mean())
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p


def _coord_descent(p, stop_time, consider=None, seed=12345, max_flat=8):
    n = int(p.size)
    if n == 0:
        return p, float('inf')
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p, float('inf')
    p = p / s
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    Ltot = 2 * n - 1
    G = 64  # grid points
    flat = 0
    sweep = 0
    GR = (math.sqrt(5.0) - 1.0) / 2.0
    kick_sigma = 0.03

    while sweep < 100000 and time.monotonic() < stop_time - 3.0:
        improved = False
        # Use random order for diversity, but also try sequential every 5th sweep
        if sweep % 5 == 0:
            order = np.arange(n)
        else:
            order = rng.permutation(n)
        
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 1.0:
                break
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end]
            infl = 2.0 * p[:Ls]
            diag = i  # 2i - start = i

            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < Ltot:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

            S = 1.0
            x_hi = max(30.0 * v, 20.0 / n, 0.01)
            x_hi = min(x_hi, 0.5)
            x_hi = max(x_hi, 1e-8)

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                ins = np.searchsorted(pts, v)
                pts = np.insert(pts, ins, v)

            ws = pts - v
            s_new = S + ws
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag < Ls:
                aff[:, diag] += ws * ws
            maff = np.max(aff, axis=1)
            maff = np.maximum(maff, max_const)
            valid = s_new > 1e-12
            scores = np.full(pts.shape, float('inf'), dtype=np.float64)
            if np.any(valid):
                scores[valid] = 2.0 * n * maff[valid] / (s_new[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            # Golden section refinement
            if 0 < imin < len(pts) - 1:
                lo = float(pts[max(0, imin - 2)])
                hi = float(pts[min(len(pts) - 1, imin + 2)])
            else:
                lo = 0.0
                hi = x_hi
            if hi - lo < 1e-14:
                lo = max(0.0, best_x - 1e-6)
                hi = best_x + 1e-6

            a_gs = lo
            b_gs = hi
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)

            def eval_1d(w):
                s_new = S + w
                if s_new <= 1e-15:
                    return float('inf')
                aff_max = float(np.max(base + w * infl))
                if 0 <= diag < Ls:
                    dval = base[diag] + w * infl[diag] + w * w
                    if dval > aff_max:
                        aff_max = dval
                m = max(aff_max, max_const)
                return 2.0 * n * m / (s_new * s_new)

            f_c = eval_1d(c_gs - v)
            f_d = eval_1d(d_gs - v)
            for _ in range(20):
                if b_gs - a_gs < 1e-12 * max(1.0, abs(best_x)):
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = eval_1d(c_gs - v)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_d = eval_1d(d_gs - v)
            x_mid = (a_gs + b_gs) / 2.0
            f_mid = eval_1d(x_mid - v)
            if f_mid < best_val:
                best_x = x_mid
                best_val = f_mid
            for xp in (0.0, x_hi, v):
                fp = eval_1d(xp - v)
                if fp < best_val:
                    best_x = xp
                    best_val = fp

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= x_hi:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    aff_seg = base + w * infl
                    if 0 <= diag < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag] += w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    c[start:end] = aff_seg
                    cur_score = 2.0 * n * new_max
                    improved = True

        # Renormalize
        s_new = float(p.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            inv = 1.0 / s_new
            p *= inv
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-15:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= max_flat:
                if time.monotonic() < stop_time - 10.0:
                    # Multiplicative kick with decreasing sigma
                    sigma = kick_sigma * math.exp(-sweep / 2000.0)
                    noise = rng.normal(1.0, sigma * n / 100.0, n)
                    p = p * noise
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                    kick_sigma *= 0.8
                else:
                    break
        sweep += 1

    # Final check
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):

    hard = t0 + budget - 25.0

    n_work = int(parent.size)
    if n_work < 64:
        n_work = 384
        p_start = np.ones(n_work) / n_work
    else:
        p_start = parent.copy() / float(parent.sum())

    best_arr = parent.copy()
    best_score = float('inf')

        except Exception:
            return

            return

            best_arr = a.copy()

    if lse_end > time.monotonic() + 20.0:
        try:
            p2 = _lse_optimize(best_arr.copy() / float(best_arr.sum()), lse_end, consider=consider)
            if p2 is not None and p2.size == n_work:
                consider(p2)
        except Exception:
            pass

    #
