                aff[:, diag_off] += ws * ws

            maff = np.maximum(maff, max_const)
            denoms = S + ws
            vals = np.where(denoms > 1e-15, 2.0 * n * maff / (denoms * denoms), np.inf)

            if not np.any(np.isfinite(vals)):
                continue

            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if best_val < cur_score - 1e-13 and best_x >= 0:
                # Golden section refinement
                dw = (x_hi - x_lo) / G
                lo_gs = max(0.0, best_x - 2.5 * dw)
                hi_gs = min(x_hi, best_x + 2.5 * dw)

                if hi_gs - lo_gs > 1e-12:
                    gr = (math.sqrt(5.0) - 1.0) / 2.0
                    a_g, b_g = lo_gs, hi_gs
                    c_g = b_g - gr * (b_g - a_g)
                    d_g = a_g + gr * (b_g - a_g)

                    def ev(x):
                        w = x - v
                        dn = S + w
                        if dn <= 1e-15:
                            return math.inf
                        nb = base + w * infl
                        if 0 <= diag_off < Ls:
                            nb = nb.copy()
                            nb[diag_off] += w * w
                        mv = float(nb.max())
                        if max_const > mv:
                            mv = max_const
                        return 2.0 * n * mv / (dn * dn)

                    f_c = ev(c_g)
                    f_d = ev(d_g)
                    for _ in range(14):
                        if f_c < f_d:
                            b_g = d_g; d_g = c_g; f_d = f_c
                            c_g = b_g - gr * (b_g - a_g); f_c = ev(c_g)
                        else:
                            a_g = c_g; c_g = d_g; f_c = f_d
                            d_g = a_g + gr * (b_g - a_g); f_d = ev(d_g)
                    x_opt = (a_g + b_g) * 0.5
                    f_opt = ev(x_opt)
                    if f_opt < best_val:
                        best_x, best_val = x_opt, f_opt

                if best_val < cur_score - 1e-13 and best_x >= 0:
                    w = best_x - v
                    if abs(w) > 1e-17:
                        a_old = p[:Ls].copy()
                        if i < Ls:
                            a_old[i] = v
                        p[i] = best_x
                        S += w
                        c[start:end] += 2.0 * w * a_old
                        if 0 <= diag_off < Ls:
                            c[start + diag_off] += w * w
                        cur_score = best_val
                        improved = True

        # Renormalize
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        # Periodic exact recompute
        if sweep % 4 == 0:
            c = _fft_conv_self(p)
            cur_score = float(2.0 * n * c.max())

        new_sc = float(2.0 * n * c.max())
        if math.isfinite(new_sc) and new_sc < best_score - 5e-13:
            best_score = new_sc
            best_p = p.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1

        if improved:
            flat = 0
        else:
            flat += 1

        # Perturbation to escape local minima
        if flat >= (12 if progress < 0.5 else 8):
            sigma = 0.008 * (1.0 + 0.5 * progress) * min(flat * 0.1, 2.0)
            if no_improve_sweeps > 40:
                sigma *= 2.0
            noise = rng.normal(0, sigma, n)
            p *= np.exp(noise)
            p = np.maximum(p, 0.0)
            ss = float(p.sum())
            if ss > 1e-300:
                p /= ss
                S = 1.0
                c = _fft_conv_self(p)
                cur_score = float(2.0 * n * c.max())
                if cur_score < best_score:
                    best_score = cur_score
                    best_p = p.copy()
            flat = 0

        if flat >= 60:
            break

    # Final exact validation
    out = np.asarray(best_p, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    if out.size == 0 or float(out.sum()) < 0.01:
        out = np.full(128, 1.0 / 128.0)

    result = [float(x) for x in out]
    try:
        fs = float(evaluate_sequence(result))
        if not math.isfinite(fs):
            try:
                pp = np.asarray(height_sequence_1, dtype=np.float64).ravel()
                pp = np.clip(pp, 0, 1000)
                result = [float(x) for x in pp]
            except Exception:
                result = [1.0]
    except Exception:
        result = [1.0]

    return result
