def _coord_descent_vec(p, stop_time, seed=42, consider=None, G=64):

        return p, math.inf
    p = p / s

    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score

    sweep = 0
    flat = 0
    
    while time.monotonic() < stop_time - 0.5:
        improved = False
        order = rng.permutation(n)
        
        for ii in order:
            i = int(ii)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            
            infl = 2.0 * p[:Ls]
            diag_i = i  # position in [0, Ls-1] where 2i = start + diag_i → diag_i = i
            # Check: start + diag_i = i + i = 2i. Yes.
            
            # Unaffected max
            max_const = 0.0

                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < Ltot:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

                best_val = float(vals[imin])
