`max(c_const, max_{k in [i, i+n-1]} (c[k] + 2w*a[k-i] + w^2 * [k=2i]))`

where `c_const` is the max of `c` outside the affected range.

The function to minimize (as a function of `w`) is:
`2*n * max(c_const, max_k (c[k] + 2w*a[k-i] + w^2*[k=2i])) / (S + w)^2`

This is a univariate function of `w`. We can minimize it using golden section search or by evaluating on a grid.

The function is the max of:
- `f_0(w) = 2*n*c_const / (S+w)^2` (decreasing in w for w > -S)
- `f_k(w) = 2*n*(c[k] + 2w*a[k-i] + w^2*[k=2i]) / (S+w)^2` for each k

Each `f_k` is a rational function. The max is piecewise. Finding the exact minimum is complicated, but a grid search + local refinement should work well.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize, linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if p.sum() < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p

def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(c, 0.0)

def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0 / u.size)
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0 / u.size)
    return e / s

def _lse_optimize(p, stop_time):
    """LSE optimization using L-BFGS-B on log-sum-exp smoothing."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p
    p = p / s
    
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return np.maximum(c, 0.0)
    
    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        return np.maximum(q, 0.0)
    
    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = 1e-4 * cmax
        scaled = (c - cmax) / T
        # Clip to avoid overflow
        scaled = np.clip(scaled, -500, 500)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        return float(f), np.where(np.isfinite(grad), grad, 0.0)
    
    def p_to_u(pp):
        ll = np.log(np.maximum(pp, 1e-300))
        return ll - float(np.mean(ll))
    
    u = p_to_u(p)
    best_p = p.copy()
    best_sc = 2.0 * n * float(_conv_self(p).max())  # since sum=1
    
    stages = (
        (0.1, 0.10),
        (0.04, 0.15),
        (0.01, 0.15),
        (0.003, 0.20),
        (0.001, 0.20),
        (0.0003, 0.20),
    )
    
    total_w = sum(w for _, w in stages)
    
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 8.0:
            break
        end_stage = min(stop_time - 5.0, now + max(5.0, wt * 25.0))
        cmax = float(_conv_self(p).max())
        T = max(1e-15, frac * cmax)
        stale = 0
        while time.monotonic() < end_stage - 2.0:
            maxit = 50
            opts = {"maxiter": maxit, "gtol": 1e-13, "ftol": 1e-18,
                    "maxls": 30, "maxcor": 30, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval):
                    p_cur = _softmax(res.x)
                    sc_cur = 2.0 * n * float(_conv_self(p_cur).max())
                    if math.isfinite(sc_cur) and sc_cur < best_sc:
                        best_sc = sc_cur
                        best_p = p_cur.copy()
                        p = p_cur.copy()
                        u = p_to_u(p)
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 4:
                            break
            if getattr(res, "status", 1) == 0:
                break
    
    p_out = best_p.copy()
    ss = float(p_out.sum())
    if ss > 1e-300:
        p_out = p_out / ss
    return p_out

def _build_conv_matrix(f):
    """Build the (2n-1) x n matrix A where (A @ b)[k] = (f*b)[k]."""
    n = len(f)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        l0 = max(0, k - n + 1)
        l1 = min(n, k + 1)
        A[k, l0:l1] = f[k-l1+1:k-l0+1][::-1]
    return A

def _lp_direction(f, m):
    """Solve LP: max sum(b) s.t. f*b <= m, b >= 0. Returns b or None."""
    n = len(f)
    A = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, m)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            if b.sum() > 1e-10:
                return b
    except Exception:
        pass
    return None

def _coord_descent(p, stop_time, seed=42, max_sweeps=500000):

    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p
    p = p / s
    
    G = max(20, min(60, int(120000 / max(n, 1)) + 20))
    Ltot = 2 * n - 1
    c = _conv_self(p)
    S = 1.0
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    flat = 0
    sweep = 0
    
    while sweep < max_sweeps and time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S <= 1e-300:
                break
            
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]
            
            max_const = 0.0
            if start > 0:
                max_const = max(max_const, float(np.max(c[:start])))
            if end < Ltot:
                max_const = max(max_const, float(np.max(c[end:])))
            
            x_hi = max(20.0 * v, 60.0 / n, 0.01)
            x_hi = min(x_hi, 10.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = 1.0 / n
            
            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 < v < x_hi:
                pts = np.unique(np.concatenate([pts, [v]]))
            
            # Vectorized evaluation
            w_coords = pts[:, None]
            dv = S + w_coords[:, 0]
            valid = dv > 1e-14
            
            aff = base[None, :] + w_coords * infl[None, :]
            diag_idx = i - start
            if 0 <= diag_idx < Ls:
                aff[:, diag_idx] += w_coords[:, 0] ** 2
            aff = np.maximum(aff, 0.0)
            aff = np.maximum(aff, max_const)
            mv = np.max(aff, axis=1)
            
            out = np.full(pts.shape, math.inf)
            out[valid] = (2.0 * n * mv[valid]) / (dv[valid] ** 2)
            
            if not np.any(np.isfinite(out)):
                continue
            
            imin = int(np.argmin(out))
            bx = float(pts[imin])
            bv = float(out[imin])
            
            # Fine search
            dw = x_hi / G
            lo = max(0.0, bx - 3.0 * dw)
            hi = min(x_hi, bx + 3.0 * dw)
            if hi > lo:
                fine = np.linspace(lo, hi, 60)
                if 0.0 < bx < x_hi:
                    fine = np.unique(np.concatenate([fine, [bx, v]]))
                w_coords2 = fine[:, None]
                dv2 = S + w_coords2[:, 0]
                valid2 = dv2 > 1e-14
                aff2 = base[None, :] + w_coords2 * infl[None, :]
                diag_idx2 = i - start
                if 0 <= diag_idx2 < Ls:
                    aff2[:, diag_idx2] += w_coords2[:, 0] ** 2
                aff2 = np.maximum(aff2, 0.0)
                aff2 = np.maximum(aff2, max_const)
                mv2 = np.max(aff2, axis=1)
                out2 = np.full(fine.shape, math.inf)
                out2[valid2] = (2.0 * n * mv2[valid2]) / (dv2[valid2] ** 2)
                if np.any(np.isfinite(out2)):
                    im2 = int(np.argmin(out2))
                    if float(out2[im2]) < bv:
                        bv = float(out2[im2])
                        bx = float(fine[im2])
            
            if math.isfinite(bv) and bv < cur_score - 1e-14 and 0.0 <= bx <= 20.0 * S:
                w = bx - v
                if abs(w) > 1e-16:
                    p[i] = bx
                    S += w
                    c[start:end] += w * infl
                    c[start + diag_idx] += w * w
                    cur_score = bv
                    improved = True
                elif bx != v:
                    p[i] = bx
                    S += w
                    cur_score = bv
                    improved = True
        
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break
        
        if sweep % 5 == 0:
            c = _conv_self(p)
        new_sc = 2.0 * n * float(c.max())
        if math.isfinite(new_sc) and new_sc < best_score - 1e-15:
            best_score = new_sc
            best_p = p.copy()
        
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 20.0:
                    p = p * np.exp(rng.normal(0, 0.03, n))
                    p = np.maximum(p, 0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    S = 1.0
                    cur_score = 2.0 * n * float(c.max())
                    flat = 0
                else:
                    break
        sweep += 1
    
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_self(p)
    sc = 2.0 * n * float(c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score
