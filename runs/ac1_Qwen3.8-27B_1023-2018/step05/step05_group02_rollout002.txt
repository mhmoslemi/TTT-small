    sweep = 0
    flat = 0

    while time.monotonic() < stop_time - 0.5 and sweep < 100000:
        improved = False
        order = rng.permutation(n)

        for idx in range(n):
            i = int(order[idx])
            if time.monotonic() > stop_time - 0.1:
                break

            if S > 1.25 or S < 0.75:
                p /= S
                c /= S * S
                S = 1.0

            v = p[i]
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            infl = 2.0 * p[:Ls]
            base = c[start:end]
            di = 2 * i - start

            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < Ltot:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

            x_hi = max(30.0 * v, 80.0 * S / n, 0.2 * S / n * 10)
            x_hi = min(x_hi, 50.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 1e-15:
                x_hi = S

            pts = np.linspace(0.0, x_hi, G)
            deltas = pts - v

            aff = base[None, :] + deltas[:, None] * infl[None, :]
            if 0 <= di < Ls:
                aff[:, di] += deltas * deltas

            m_aff = np.max(aff, axis=1)
            m_final = np.maximum(m_aff, max_const)
            denom = (S + deltas) * (S + deltas)
            valid = denom > 1e-20
            scores = np.full(G, 1e100)
            scores[valid] = 2.0 * n * m_final[valid] / denom[valid]

            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_sc = float(scores[imin])

            # Fine grid refinement
            if best_sc < cur_score - 1e-13:
                dw = x_hi / G
                lo = max(0.0, best_x - 2.5 * dw)
                hi = min(x_hi, best_x + 2.5 * dw)
                if hi - lo < 1e-14:
                    hi = lo + 1e-14
                fpts = np.linspace(lo, hi, 80)
                fd = fpts - v
                faff = base[None, :] + fd[:, None] * infl[None, :]
                if 0 <= di < Ls:
                    faff[:, di] += fd * fd
                fm_a = np.max(ffaff, axis=1)
                fm_f = np.maximum(fm_a, max_const)
                fden = (S + fd) * (S + fd)
                fval_i = fden > 1e-20
                fscores = np.full(80, 1e100)
                fscores[fval_i] = 2.0 * n * fm_f[fval_i] / fden[fval_i]
                imin2 = int(np.argmin(fscores))
                if fscores[imin2] < best_sc:
                    best_x = float(fpts[imin2])
                    best_sc = float(fscores[imin2])

            if best_sc < cur_score - 1e-13:
                delta = best_x - v
                if abs(delta) > 1e-15:
                    p[i] = best_x
                    S += delta
                    c[start:end] += delta * infl
                    if 0 <= 2 * i < Ltot:
                        c[2 * i] += delta * delta
                    cur_score = best_sc
                    improved = True

        if S > 1e-300:
            p /= S
            c /= S * S
        S = 1.0
        c = _conv_self(p)
        cur_score = 2.0 * n * c.max()

        if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = p.copy()
            if consider:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 10.0:
                    noise_amt = 0.02 + 0.015 * math.exp(-flat * 0.08)
                    p = p * np.exp(rng.normal(0.0, noise_amt, n))
                    p = np.maximum(p, 0.0)
                    ps = float(p.sum())
                    if ps > 1e-300:
                        p /= ps
                        S = 1.0
                        c = _conv_self(p)
                        cur_score = 2.0 * n * c.max()
                        if math.isfinite(cur_score) and cur_score < best_score:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider:
                                try:
                                    consider(best_p)
                                except Exception:
                                    pass
                    flat = 0
                else:
                    break
        sweep += 1

    return best_p, best_score


def _subgradient_polish(p, stop_time, consider=None, max_iters=2000):
    n = p.size
    p = np.maximum(p.astype(np.float64), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    best_p = p.copy()
    best_score = float(2.0 * n * _conv_self(p).max())
    base_step = 0.003
    min_step = 1e-9

    for it in range(max_iters):
        if time.monotonic() > stop_time - 1.0:
            break

        c = _conv_self(p)
        cmax = float(c.max())
        thresh = cmax * 0.999
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        grad = np.zeros(n)
        for k in active:
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                grad[valid] += p[kj[valid]] + p[valid]
        grad /= len(active)

        for scale in (1.0, 0.5, 0.25, 0.1):
            step = scale * base_step / (1.0 + 0.01 * it)
            if step < min_step:
                break
            p_c = p - step * grad
            p_c = np.maximum(p_c, 0.0)
            ps = float(p_c.sum())
            if ps < 1e-15:
                continue
            p_c /= ps
            sc = float(2.0 * n * _conv_self(p_c).max())
            if math.isfinite(sc) and sc < best_score - 1e-16:
                best_score = sc
                best_p = p_c.copy()
                p = p_c.copy()
                break
        else:
            base_step *= 0.92
            if base_step < 1e-13:
                break

        if it % 100 == 0 and consider:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64) / 64.0
        best_score = float(_score_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_arr = a.copy()
            best_score = sc

    n = best_arr.size

    # Phase 1: LP-based Frank-Wolfe (first ~120s)
    lp_end = min(hard - 60.0, t0 + 120.0)
    if lp_end > t0 + 30.0:
        p_work = best_arr / float(best_arr.sum())
        try:
            p_lpfw = _lp_frank_wolfe(p_work, max_iters=6, stop_time=lp_end, consider=consider)
            if p_lpfw is not None and p_lpfw.size == n and np.all(np.isfinite(p_lpfw)):
                p_lpfw = p_lpfw / float(p_lpfw.sum())
                consider(p_lpfw)
        except Exception:
            pass

    # Phase 2: Coordinate descent (main phase, ~1000s)
    cd_end = min(hard - 60.0, t0 + budget - 60.0)
    now = time.monotonic()
    if cd_end > now + 30.0:
        p_cd = best_arr / float(best_arr.sum())
        # First pass: coordinate descent
        cd1_end = min(cd_end, now + 0.55 * (cd_end - now))
        if cd1_end > now + 20.0:
            try:
                p_cd, _ = _coord_descent(p_cd, cd1_end, seed=int(seed + 1000), consider=consider)
            except Exception:
                pass

        # Second pass: continue from best
        now = time.monotonic()
        if cd_end > now + 20.0:
            p_cd2 = best_arr / float(best_arr.sum())
            try:
                p_cd2, _ = _coord_descent(p_cd2, cd_end, seed=int(seed + 2000), consider=consider)
            except Exception:
                pass

    # Phase 3: Final polishing (last ~50-80s)
    now = time.monotonic()
    fin_end = hard - 5.0
    if fin_end > now + 15.0:
        p_fin = best_arr / float(best_arr.sum())
        # Subgradient
        sg_end = min(fin_end, now + 0.4 * (fin_end - now))
        if sg_end > now + 8.0:
            try:
                p_sg = _subgradient_polish(p_fin, sg_end, consider=consider, max_iters=500)
                if p_sg is not None and p_sg.size == n and np.all(np.isfinite(p_sg)):
                    consider(p_sg)
            except Exception:
                pass

        # Final coordinate descent
        now = time.monotonic()
        if fin_end > now + 10.0:
            p_fin2 = best_arr / float(best_arr.sum())
            try:
                p_fin2, _ = _coord_descent(p_fin2, fin_end, seed=int(seed + 99000), consider=consider)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        if time.monotonic() > hard:
            pass  # still return, harness handles it
        return ls
    except Exception:
        return [1.0]
