        a = np.maximum(a, 1e-14)
        u = np.log(a)
        return u - u.mean()
    
    # L-BFGS
    def lbf(a0, tS, fin=False):
        a0 = np.maximum(a0, 1e-14)
        a0 = a0 / a0.sum()
        u = tl(a0)
        bu = u.copy()
        
        if fin:
            stg = [(0.0008, 0.30), (0.0002, 0.35), (0.00005, 0.35)]
        else:
            stg = [(0.015, 0.10), (0.005, 0.14), (0.0015, 0.18), (0.0004, 0.22), (0.0001, 0.36)]
        
        tw = sum(w for _, w in stg)
        cm2 = 0.0
        for tf, wt in stg:
            if now() > tS - 3: break
            aC = smx(bu)
            cC = fc(aC)
            cM = cC.max()
            if cM <= 0: break
            T = max(1e-12, tf * cM)
            fr = (cm2 + wt) / tw
            cm2 += wt
            se = min(tS - 3, t0 + budget * 0.85 * fr)
            if se <= now() + 5: continue
            
            def lf(uu):
                pp = smx(uu)
                F = rfft(pp, cSz)
                cc = np.ascontiguousarray(irfft(F * F, cSz)[:cLen])
                np.maximum(cc, 0.0, out=cc)
                cm = cc.max()
                sc = (cc - cm) / T
                ex = np.exp(sc)
                sE = ex.sum()
                if not np.isfinite(sE) or sE < 1e-300:
                    ww = np.zeros(cLen)
                    ww[np.argmax(cc)] = 1.0
                    fv = cm
                else:
                    ww = ex / sE
                    fv = cm + T * math.log(sE / cLen)
                W = rfft(ww, xSz)
                A = rfft(pp, xSz)
                xc = irfft(np.conj(W) * A, xSz)[:n]
                gc = 2.0 * xc
                dd = float(pp @ gc)
                gr = 2.0 * n * pp * (gc - dd)
                return 2.0 * n * fv, gr
            
            while now() < se - 2:
                try:
                    rs = minimize(lf, bu, method='L-BFGS-B', jac=True,
                                  options={'maxiter': 25, 'gtol': 1e-10,
                                           'ftol': 1e-14, 'maxls': 20,
                                           'maxcor': 15, 'disp': False})
                except Exception:
                    break
                if rs is None or rs.x is None: break
                x = np.asarray(rs.x, dtype=np.float64)
                if x.size == n and np.all(np.isfinite(x)):
                    fn = float(rs.fun) if np.isfinite(rs.fun) else 1e100
                    bn = sn(smx(x))
                    if np.isfinite(bn) and bn < sn(smx(bu)) - 1e-12:
                        bu = x.copy()
                    else:
                        break
                if getattr(rs, 'status', 1) == 0: break
            
            an = smx(bu)
            an = an / an.sum()
            trk(an)
        
        r = smx(bu)
        return r / r.sum()
