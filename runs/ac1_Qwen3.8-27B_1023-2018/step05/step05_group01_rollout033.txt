import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _softmax_fn(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.shape, 1.0 / u.size)
    return e / s


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    m = float(np.max(c))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _score_exact_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(np.max(b))
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _conv_self(a, conv_size=None, conv_len=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    if conv_size is None:
        conv_size = next_fast_len(L)
    if conv_len is None:
        conv_len = L
    F = rfft(a, conv_size)
    c = irfft(F * F, conv_size)[:conv_len]
    np.maximum(c, 0.0, out=c)
    return c


class _AC1State:
    __slots__ = ('p', 'n', 'conv_size', 'conv_len', 'cross_size', 'cross_len', 'rng')

    def __init__(self, p, seed):
        self.p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
        self.n = self.p.size
        s = float(self.p.sum())
        if s < 1e-300:
            self.p = np.full(self.n, 1.0 / self.n)
        else:
            self.p = self.p / s
        self.conv_size = next_fast_len(2 * self.n - 1)
        self.conv_len = 2 * self.n - 1
        self.cross_size = next_fast_len(3 * self.n - 2)
        self.cross_len = 3 * self.n - 2
        self.rng = np.random.default_rng(seed)

    def _c(self, a):
        F = rfft(a, self.conv_size)
        c = irfft(F * F, self.conv_size)[:self.conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def _x(self, pp, w):
        F = rfft(pp[::-1], self.cross_size)
        G = rfft(w, self.cross_size)
        c = irfft(F * G, self.cross_size)[:self.cross_len]
        q = c[self.n - 1:2 * self.n - 1]
        np.maximum(q, 0.0, out=q)
        return q


def _lse_optimize(st, stop_time, consider, T_stages, max_lbfgs_per_stage=4,
                  perturb_scale=0.004, seed_offset=0):
    n = st.n
    p = st.p.copy()
    u = np.log(np.max(p, 1e-300))
    u = u - float(u.mean())
    rng = np.random.default_rng(st.seed + seed_offset)
    best_p = p.copy()
    best_sc = _score_fft(p)

    def lse_func(uu, T):
        pp = _softmax_fn(uu)
        c = st._c(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        Tsafe = max(T, 1e-300)
        scaled = (c - cmax) / Tsafe
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + Tsafe * float(np.log(sum_exp / c.size))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = st._x(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    for stage_i, T in enumerate(T_stages):
        if time.monotonic() > stop_time - 3.0:
            break
        best_u = u.copy()
        f0, _ = lse_func(best_u, T)
        if not math.isfinite(f0):
            continue
        best_f = f0
        stale = 0
        max_calls = max_lbfgs_per_stage
        for call_i in range(max_calls):
            if time.monotonic() > stop_time - 2.0:
                break
            maxit = 35
            opts = {"maxiter": maxit, "gtol": 1e-13, "ftol": 1e-16,
                    "maxls": 25, "maxcor": 15, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            improved = False
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    fval = float(res.fun)
                    if math.isfinite(fval) and fval < best_f - 1e-14:
                        best_f = fval
                        best_u = x.copy()
                        improved = True
                        stale = 0
                    else:
                        stale += 1
            if not improved and stale >= 2:
                best_u = best_u + rng.normal(0, perturb_scale, n)
                stale = 0
            if getattr(res, "status", 1) == 0 and improved:
                if call_i < max_calls - 1 and time.monotonic() < stop_time - 5.0:
                    best_u = best_u + rng.normal(0, perturb_scale * 0.5, n)

        p_cur = _softmax_fn(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_sc = sc_cur
            best_p = p_cur.copy()
            p = p_cur.copy()
        elif math.isfinite(sc_cur) and sc_cur < best_sc + 0.01:
            p = p_cur.copy()
        else:
            p = best_p.copy()
        u = np.log(np.maximum(p, 1e-300))
        u = u - float(u.mean())
        if consider is not None:
            try:
                consider(p)
            except Exception:
                pass
        st.p = p.copy()

    st.p = best_p.copy()
    return best_p, best_sc


def _coord_polish(st, stop_time, consider, G=56, seed_offset=0):
    n = st.n
    p = np.maximum(st.p.copy(), 0.0)
    s = float(p.sum())
    if s < 1e-300:
        return st.p, math.inf
    p = p / s

    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(st.seed + seed_offset)
    sweep = 0
    flat = 0
    max_sweeps = 500000

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]
            max_const = 0.0
            if start > 0:
                ml = float(np.max(c[:start]))
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(np.max(c[end:]))
                if mr > max_const:
                    max_const = mr
            S = float(p.sum())
            x_hi = max(20.0 * v, 40.0 / float(n), 0.02)
            x_hi = min(x_hi, 8.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, 1.0 / n)

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag_i = i
            S_total = S

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S_total + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                wv = w[valid]
                aff = base[None, :] + wv[:, None] * infl[None, :]
                dval = base[diag_i] + wv * infl[diag_i] + wv * wv
                aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                dv = denom[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if G > 4 and best_x > 0 and best_x < x_hi:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 32, dtype=np.float64)
                if 0.0 <= best_x <= x_hi:
                    fine = np.concatenate((fine, [best_x]))
                    fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val - 1e-15:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 10.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = S_total
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 5.0:
                    sigma = 0.015 + 0.015 * min(flat, 8) / 8.0
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    else:
                        break
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    flat = 0
                    improved = True
                else:
                    break
        sweep += 1

    st.p = best_p.copy()
    return best_p, best_score


def _hardmax_gradient_step(st, stop_time, consider, n_steps=500, lr_scale=0.5):
    n = st.n
    p = st.p.copy()
    best_p = p.copy()
    best_sc = _score_fft(p)
    rng = np.random.default_rng(st.seed + 77777)

    for step_i in range(n_steps):
        if time.monotonic() > stop_time - 2.0:
            break
        c = st._c(p)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            break
        k_star = int(np.argmax(c))

        # Gradient of c[k_star] wrt p: g[i] = 2*p[k_star - i] for valid (k_star - i)
        g = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                g[i] = 2.0 * p[j]

        S = float(p.sum())
        # Score = 2n * cmax / S^2
        # d(score)/d(p[i]) = 2n * (g[i]/S - 2*cmax/S^2)  [with chain rule through S]
        # = 2n*(g[i]*S - 2*cmax) / S^3
        # For softmax parameterization: grad_u[i] = p[i]*(grad_p[i] - dot(p, grad_p))
        grad_p = 2.0 * n * (g - 2.0 * cmax) / (S * S)
        mean_gp = float(np.dot(p, grad_p))
        grad_u = p * (grad_p - mean_gp)
        grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)

        # Adaptive learning rate
        gnorm = float(np.linalg.norm(grad_u))
        if gnorm < 1e-30:
            break
        lr = lr_scale * cmax / gnorm
        lr = min(lr, 0.05)

        u = np.log(np.maximum(p, 1e-300))
        u = u - float(u.mean())
        u_new = u - lr * grad_u
        p_new = _softmax_fn(u_new)
        p_new = np.maximum(p_new, 0.0)
        p_new = p_new / p_new.sum()
        sc_new = _score_fft(p_new)
        if math.isfinite(sc_new) and sc_new < best_sc:
            best_sc = sc_new
            best_p = p_new.copy()
            p = p_new.copy()
            if consider is not None:
                try:
                    consider(p)
                except Exception:
                    pass
        else:
            # Backtrack or break
            if step_i > 50:
                # Add perturbation
                p = p * np.exp(rng.normal(0, 0.01, n))
                p = np.maximum(p, 0)
                p = p / p.sum()
            else:
                lr *= 0.5
                if lr < 1e-10:
                    break

    st.p = best_p.copy()
    return best_p, best_sc
