import numpy as np

from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize
from scipy.sparse import csr_matrix

def _conv_self(a):
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return c

def _fft_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(p)
    m = float(c.max())
    return float(2.0 * n * m / (s * s)) if math.isfinite(m) else math.inf

def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(len(u), 1.0 / len(u))
    return e / s

def _lbfgs_warmup(p, n, stop_time, consider):
    """L-BFGS with log-sum-exp smoothing."""
    p = p.copy()
    s = float(p.sum())
    p = p / s
    
    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))
    
    u = p_to_u(p)
    L = 2 * n - 1
    size = next_fast_len(L)
    
    def cfast(a):
        F = rfft(a, size)
        c = irfft(F * F, size)[:L]
        return c
    
    # Cross-correlation for gradient
    cross_L = 3 * n - 2
    cross_size = next_fast_len(cross_L)
    
    def cross_corr(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_L]
        q = c[n - 1:2 * n - 1]
        return q
    
    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax):
            return 1e100, np.zeros_like(uu)
        if T <= 0:
            T = 1e-4 * cmax
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if sum_exp <= 0 or not math.isfinite(sum_exp):
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            w = exps / sum_exp
        q = cross_corr(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad
    
    stages = [(0.05, 1.0), (0.01, 1.0)]
    best_u = u.copy()
    best_f = 1e100
    
    for T, wt in stages:
        end_stage = min(stop_time - 5.0, time.monotonic() + 30.0)
        while time.monotonic() < end_stage - 2.0:
            opts = {"maxiter": 50, "gtol": 1e-12, "ftol": 1e-16, "maxls": 30, "maxcor": 20, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fv = lse_func(res.x, T)[0]
                if math.isfinite(fv) and fv < best_f:
                    best_f = fv
                    best_u = res.x.copy()
    
    pp = _softmax(best_u)
    pp = np.maximum(pp, 0.0)
    ss = float(pp.sum())
    if ss > 0:
        pp /= ss
    consider(pp)
    return pp


def _solve_lp_direction(f, M, n, time_limit=10.0):
    """Solve the LP: max sum(b) s.t. (f*b)[k] <= M, b >= 0.
    Returns b or None if infeasible/too slow."""
    t_start = time.monotonic()
    
    # Build sparse constraint matrix A: A[k, j] = f[k-j] if 0 <= k-j < n
    # Shape: (2n-1) x n
    # For column j: non-zeros at rows j, j+1, ..., j+n-1 with values f[0], f[1], ..., f[n-1]
    
    # Efficient construction using the structure
    # Total non-zeros: sum over j of min(n, 2n-1-j) = n^2 (since j+n-1 <= 2n-2 for j <= n-1)
    # Actually for j=0: rows 0 to n-1 (n entries)
    # For j=n-1: rows n-1 to 2n-2 (n entries)
    # So each column has exactly n non-zeros, total = n^2
    
    nnz = n * n
    row_idx = np.empty(nnz, dtype=np.int32)
    col_idx = np.empty(nnz, dtype=np.int32)
    vals = np.empty(nnz, dtype=np.float64)
    
    for j in range(n):
        start = j * n
        row_idx[start:start+n] = np.arange(j, j+n)
        col_idx[start:start+n] = j
        vals[start:start+n] = f
    
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
    
    # Check time
    if time.monotonic() - t_start > time_limit * 0.3:
        return None
    
    # Full constraint matrix includes -b <= 0 (b >= 0)
    # linprog: min c @ b s.t. A_ub @ b <= b_ub, A_eq @ b == b_eq, lb <= b <= ub
    # We want: max sum(b) => min -sum(b)
    # Constraints: A @ b <= M (already), b >= 0 (via bounds)
    
    c_obj = -np.ones(n)
    A_ub = A
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    from scipy.optimize import linprog
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            sb = float(b.sum())
            if sb > 1e-10:
                return b / sb
            else:
                return None
        return None
    except:
        return None
