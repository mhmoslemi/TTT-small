One approach: make a have a "ring" shape - concentrate mass away from the center of the sequence. This way, at the peak lag k=n-1, the overlapping pairs (i, n-1-i) tend to be near the edges where a is small.

Wait, no. If a is concentrated at the edges, then at k=n-1, we have pairs (i, n-1-i). If a_0 and a_{n-1} are large, then a_0 * a_{n-1} contributes to c_{n-1}. So concentrating at both edges actually HURTS the peak.

What about concentrating at ONE edge? If a_0 is large and a_{n-1} is small, then at k=n-1, the contribution from i=0 is a_0 * a_{n-1} which is small. But at k=0, it's a_0^2 which could be large.

The constraint is max over ALL k, not just k=n-1. So we need to balance all lags.

            jlo = max(0, k - len(a) + 1)
            jhi = min(len(a), k + 1)
            j = np.arange(jlo, jhi)
            idx = k - j
            grad[j] += 2.0 * a[idx]
        return grad, M, active

    # Main optimization loop
    def optimize(a0, end_time, max_stuck=300):
        a = np.asarray(a0, dtype=np.float64).copy()
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones_like(a)
            s = a.size
        a = a / s
        n_local = a.size

        local_best = a.copy()
        local_best_M = raw_peak(a)
        stuck = 0

        while time.monotonic() < end_time:
            c = np.convolve(a, a)
            M = float(c.max())

            if M < local_best_M * (1.0 - 1e-13):
                local_best_M = M
                local_best = a.copy()
                stuck = 0
            else:
                stuck += 1

            # Perturb if stuck too long
            if stuck > max_stuck:
                level = min((stuck - max_stuck) // (max_stuck // 4 + 1), 10)
                sigma = 0.02 + 0.02 * level
                xx = np.arange(n_local, dtype=np.float64)
                mode = level % 5
                if mode == 0:
                    a *= np.exp(rng.normal(0, sigma, n_local))
                elif mode == 1:
                    f = rng.uniform(2, 12)
                    a *= (1.0 + sigma * np.sin(2 * np.pi * f * xx / n_local + rng.uniform(0, 6.28)))
                elif mode == 2:
                    a *= (1.0 + sigma * rng.standard_normal(n_local))
                elif mode == 3:
                    # Mirror perturbation
                    a *= (1.0 + sigma * rng.standard_normal(n_local))
                    a = np.maximum(a, 1e-10)
                    a /= a.sum()
                    a_rev = a[::-1]
                    a = 0.5 * a + 0.5 * a_rev
                else:
                    # Block redistribution
                    block = max(n_local // 12, 2)
                    st = int(rng.integers(0, max(1, n_local - 2 * block)))
                    a2 = a.copy()
                    a2[st:st+block] = a[st+block:st+2*block]
                    a2[st+block:st+2*block] = a[st:st+block]
                    a = a2
                a = np.maximum(a, 1e-10)
                a /= a.sum()
                stuck = 0
                if sigma > 0.30:
                    break
                continue

            moved = False
            A = build_A(a, n_local)

            # Strategy 1: LP best response (tight)
            g = lp_best_response(a, M, n_local, tl=4.0, relax=1.0)
            if g is not None:
                gs = g.sum()
                g = g / gs
                if np.max(np.abs(g - a)) > 1e-9:
                    cand, val = line_search(a, g, M)
                    if cand is not None and val < M * (1.0 - 1e-11):
                        a = cand
                        moved = True

            # Strategy 2: LP with slight relaxation
            if not moved:
                g2 = lp_best_response(a, M, n_local, tl=3.0, relax=1.005)
                if g2 is not None:
                    g2s = g2.sum()
                    g2 = g2 / g2s
                    if np.max(np.abs(g2 - a)) > 1e-9:
                        cand, val = line_search(a, g2, M)
                        if cand is not None and val < M * (1.0 - 1e-11):
                            a = cand
                            moved = True

            # Strategy 3: Averaged LP direction
            if not moved and g is not None:
                gs = g.sum()
                if gs > 1e-12:
                    g_avg = np.maximum(0.5 * g / gs + 0.5 * a, 0)
                    g_avg /= g_avg.sum()
                    cand, val = line_search(a, g_avg, M)
                    if cand is not None and val < M * (1.0 - 1e-11):
                        a = cand
                        moved = True

            # Strategy 4: Analytic subgradient descent
            if not moved:
                grad, M2, active = compute_subgradient(a)
                gn = np.linalg.norm(grad)
                if gn > 1e-14:
                    grad_normed = grad / gn
                    for alpha in [0.08, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]:
                        cand = np.maximum(a - alpha * grad_normed, 0)
                        s2 = cand.sum()
                        if s2 < 1e-12:
                            continue
                        cand /= s2
                        v = float(np.convolve(cand, cand).max())
                        if v < M2 * (1.0 - 1e-11):
                            a = cand
                            moved = True
                            break

            # Strategy 5: Coordinate-level pair adjustment at active lags
            if not moved:
                active = np.where(c >= M * 0.9999)[0]
                found = False
                for k_star in active[:10]:
                    if found:
                        break
                    k_star = int(k_star)
                    jlo = max(0, k_star - n_local + 1)
                    jhi = min(n_local, k_star + 1)
                    j_arr = np.arange(jlo, jhi)
                    i_arr = k_star - j_arr
                    prods = a[j_arr] * a[i_arr]
                    imax = int(np.argmax(prods))
                    pi, pj = int(j_arr[imax]), int(i_arr[imax])
                    if pi == pj:
                        continue
                    # Try to equalize
                    diff = abs(a[pi] - a[pj])
                    if diff < 1e-12:
                        continue
                    larger, smaller = (pi, pj) if a[pi] > a[pj] else (pj, pi)
                    for frac in [0.2, 0.4, 0.6, 0.8]:
                        d = frac * 0.5 * diff
                        cand = a.copy()
                        cand[larger] -= d
                        cand[smaller] += d
                        cand = np.maximum(cand, 0)
                        s2 = cand.sum()
                        if s2 < 1e-12:
                            continue
                        cand /= s2
                        v = float(np.convolve(cand, cand).max())
                        if v < M * (1.0 - 1e-11):
                            a = cand
                            moved = True
                            found = True
                            break

            # Strategy 6: Tiny random nudge
            if not moved:
                a *= np.exp(rng.normal(0, 0.004, n_local))
                a = np.maximum(a, 1e-10)
                a /= a.sum()

        return local_best

    # === PHASE 1: Polish parent (longest budget) ===
    end1 = t0 + min(350.0, budget * 0.27)
    try:
        r = optimize(parent, end1, max_stuck=350)
        try_update(r)
    except Exception:
        pass

    # === PHASE 2: Diverse initializations ===
    inits = []
    x = np.linspace(0, 1, n)
    pn = parent / max(parent.sum(), 1e-15)

    # Various shapes
    for sig in [0.06, 0.1, 0.15, 0.2, 0.3, 0.4, 0.55]:
        for ctr in [0.35, 0.5, 0.65, 0.5]:
            v = np.exp(-0.5 * ((x - ctr) / sig) ** 2)
            v = np.maximum(v, np.exp(-0.5 * ((x - (1 - ctr)) / sig) ** 2))
            v = np.maximum(v, 0.003)
            v /= v.sum()
            inits.append(v.copy())

    # Cosine windows
    for m in [1, 2, 3, 4, 5, 6, 8, 10]:
        v = np.cos(m * np.pi * (x - 0.5)) ** 2 + 0.008
        v = np.maximum(v, 0)
        v /= v.sum()
        inits.append(v.copy())

    # S-shaped (tanh)
    for steep in [3, 5, 8, 12]:
        v = 1.0 / (1.0 + np.exp(-steep * (x - 0.5)))
        v = np.maximum(v, 0.005)
        v /= v.sum()
        inits.append(v.copy())

    # Random
    for s2 in range(6):
        r2 = np.random.default_rng(7000 + s2)
        v = r2.exponential(1.0, n) + 0.008
        v /= v.sum()
        inits.append(v.copy())

    # Blends with parent
    for fr in [0.3, 0.5, 0.7]:
        for v in inits[:10]:
            b = fr * pn + (1.0 - fr) * v
            b = np.maximum(b, 0)
            b /= b.sum()
            inits.append(b.copy())

    end2 = t0 + min(720.0, budget * 0.56)
    remaining = max(1.0, end2 - time.monotonic())
    per_init = max(4.0, remaining / max(1, len(inits)))
    for i, init in enumerate(inits):
        if time.monotonic() > end2 - 8:
            break
        alloc = min(time.monotonic() + per_init, end2 - 5.0)
        try:
            r = optimize(init, alloc, max_stuck=80)
            try_update(r)
        except Exception:
            pass

    # === PHASE 3: Deep polish on best ===
    end3 = t0 + min(1080.0, budget * 0.84)
    ab = best_a.copy()
    s_ab = ab.sum()
    if s_ab > 1e-15:
        ab /= s_ab
    try:
        r = optimize(ab, end3, max_stuck=500)
        try_update(r)
    except Exception:
        pass

    # === PHASE 4: Different lengths ===
    if time.monotonic() < deadline - 60:
        for dn in [-20, 10, -10, 20, 30, -30]:
            if time.monotonic() > deadline - 18:
                break
            nt = n + dn
            if nt < 50 or nt > 600:
                continue
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, nt)
            inew = np.interp(x_new, x_old, best_a)
            inew = np.clip(inew, 0, 1000)
            sn = inew.sum()
            if sn < 1e-10:
                continue
            inew /= sn
            alloc = min(time.monotonic() + 30, deadline - 12)
            try:
                r = optimize(inew, alloc, max_stuck=100)
                try_update(r)
            except Exception:
                pass

    # === PHASE 5: Final micro-polish ===
    if time.monotonic() < deadline - 3:
        af = best_a.copy()
        s_af = af.sum()
        if s_af > 1e-15:
            af /= s_af
        try:
            r = optimize(af, deadline, max_stuck=600)
            try_update(r)
        except Exception:
            pass

    # === Final validation ===
    result = [float(v) for v in best_a]
    try:
        fs = float(evaluate_sequence(result))
        if not np.isfinite(fs):
            raise ValueError("Non-finite score")
    except Exception:
        return [1.0] * 100
    return result
