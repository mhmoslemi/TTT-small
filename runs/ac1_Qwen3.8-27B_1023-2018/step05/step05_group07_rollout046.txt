def _score_fast(a):
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * c.max() / (s * s)

def _project_simplex(v):
    """Project v onto {x : x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def _lp_dir(a, M, tol=1e-10):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    # Build constraint matrix: A[k, j] = a[k-j] if 0<=k-j<n
    # Use the fact that it's Toeplitz
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[k, j] = a[k - j]
    
    c = -np.ones(n)
    b = np.full(2*n-1, M * (1 + tol))
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs',
                      options={'time_limit': 5, 'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except:
        pass
    return None

def _gradient_step(a, alpha):
    """Projected subgradient step on the simplex."""
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    # Active lags
    active = np.where(c >= M * 0.9999)[0]
    # Subgradient: average over active lags
    grad = np.zeros(n)
    for k in active:

        for j in range(lo, hi):

    grad /= len(active)
    # Projected gradient step
    v = a - alpha * grad
    a_new = _project_simplex(v)
    return a_new, M

import numpy as np
import time
from scipy.optimize import linprog


def _get_parent():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(200)
    p = np.clip(p, 0, 1000)
    if p.size == 0 or not np.all(np.isfinite(p)):
        p = np.ones(200)
    return p


def _score(a):
    """Compute the AC1 score. a is normalized (sum=1) for internal use."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * float(c.max()) / (s * s)


def _build_toeplitz(n, a):
    """Build (2n-1) x n Toeplitz convolution matrix."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    A[mask] = a[d[mask]]
    return A


def _lp_direction(a, M, time_limit=5.0):
    """Solve max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    A = _build_toeplitz(n, a)
    c = -np.ones(n)
    b = np.full(A.shape[0], M * (1 + 1e-12))
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-13,
                               'primal_feasibility_tolerance': 1e-13})
        if res.status in (0, 1) and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-14:
                return g
    except Exception:
        pass
    return None


def _golden_search(a, g, M_current, n_evals=40):
    """Golden section search for t in [0,1] minimizing max(conv((1-t)a+tg, ...))."""
    n = len(a)
    g = np.maximum(g, 0)
    gs = g.sum()
    if gs < 1e-14:
        return a, M_current

    def fval(t):
        if t < 1e-14:
            return M_current
        cand = (1 - t) * a + t * g
        cs = cand.sum()
        if cs < 1e-14:
            return M_current
        cand = cand / cs
        return float(np.convolve(cand, cand).max())

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc, fd = fval(c), fval(d)

    for _ in range(n_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = fval(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = fval(d)
        if hi - lo < 1e-10:
            break

    best_t = 0.5 * (lo + hi)
    best_val = fval(best_t)

    # Also check t=1 directly
    v1 = fval(1.0)
    if v1 < best_val:
        best_t, best_val = 1.0, v1

    if best_t < 1e-8:
        return a, M_current

    new_a = (1 - best_t) * a + best_t * g
    new_a = np.maximum(new_a, 0.0)
    new_a = new_a / new_a.sum()
    return new_a, best_val


def _subgradient_direction(a):
    """Compute subgradient of max_k (a*a)_k averaged over active lags."""
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    active = np.where(c >= M * 0.9995)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])

    grad = np.zeros(n)
    for k_star in active:
        ks = int(k_star)
        lo = max(0, ks - n + 1)
        hi = min(n, ks + 1)
        for j in range(lo, hi):
            idx = ks - j
            if 0 <= idx < n:
                grad[j] += 2.0 * a[idx]
    grad /= len(active)
    return grad, M, active


def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _optimize_core(a_init, deadline, seed=42, max_iter=2000):
    """Core optimization: LP iterations + gradient escape + perturbations."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-14:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0
    grad_stuck = 0

    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        M = float(c.max())

        if M < best_M * (1.0 - 1e-13):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # --- LP-based step ---
        moved = False
        g = _lp_direction(a, M, time_limit=3.0)
        if g is not None and g.sum() > 1e-12:
            g = g / g.sum()
            diff = float(np.max(np.abs(g - a)))
            if diff > 1e-11:
                new_a, new_M = _golden_search(a, g, M, n_evals=35)
                if new_M < M * (1.0 - 1e-12):
                    a = new_a
                    moved = True
                else:
                    # Try half step
                    g2 = 0.5 * a + 0.5 * g
                    g2 = np.maximum(g2, 0)
                    if g2.sum() > 1e-14:
                        g2 /= g2.sum()
                        new_a2, new_M2 = _golden_search(a, g2, M, n_evals=20)
                        if new_M2 < M * (1.0 - 1e-12):
                            a = new_a2
                            moved = True

        if moved:
            grad_stuck = 0
            continue

        # --- Stuck: try gradient / perturbation ---
        if stuck > 30:
            sigma = min(0.02 + 0.01 * (stuck // 30), 0.4)
            mode = (stuck // 30) % 6
            xx = np.linspace(0, 1, n)
            if mode == 0:
                a *= np.exp(sigma * rng.standard_normal(n))
            elif mode == 1:
                freq = rng.uniform(1.0, 10.0)
                a *= (1.0 + sigma * np.sin(2 * np.pi * freq * xx + rng.uniform(0, 6.28)))
            elif mode == 2:
                # Antisymmetric perturbation around center
                half = n // 2
                pert = rng.standard_normal(n)
                pert[:half] = -pert[n - half - 1:n - 1:-1] if n - 1 >= half else 0
                a += sigma * pert * a
            elif mode == 3:
                # Block modification
                blk = max(n // 8, 3)
                pos = rng.integers(0, n - blk)
                a[pos:pos + blk] *= np.exp(sigma * rng.standard_normal(blk))
            elif mode == 4:
                # Smooth perturbation
                a = np.convolve(a, np.ones(5) / 5.0, mode='same')
                a *= (1.0 + 0.05 * rng.standard_normal(n))
            else:
                # Gradient escape
                grad, M_g, _ = _subgradient_direction(a)
                for alpha in [0.05, 0.03, 0.02, 0.01, 0.005]:
                    v = a - alpha * grad
                    v = _project_simplex(v)
                    cv = np.convolve(v, v).max()
                    if cv < M * (1.0 - 1e-10):
                        a = v
                        moved = True
                        break
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck = 0
            grad_stuck += 1
            if grad_stuck > 50:
                break
            continue

        # Small perturbation to avoid exact fixed point
        if stuck > 5:
            a *= (1.0 + 0.005 * rng.standard_normal(n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

    return best_a


def _coordinate_descent(a, deadline, max_rounds=50):
    """Try coordinate-wise adjustments to reduce max autocorrelation."""
    a = np.maximum(a.copy(), 0.0)
    a /= a.sum()
    n = len(a)
    best_M = float(np.convolve(a, a).max())
    best_a = a.copy()
    rng = np.random.default_rng(12345)

    for _round in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = np.convolve(a, a)
        M = c.max()
        active = np.where(c >= M * 0.999)[0]

        for k_star in active[:8]:
            if time.monotonic() > deadline:
                break
            ks = int(k_star)
            lo = max(0, ks - n + 1)
            hi = min(n, ks + 1)
            # Find pairs contributing to this lag
            if hi - lo < 2:
                continue
            for i in range(lo, hi):
                j = ks - i
                if j < 0 or j >= n or j <= i:
                    continue
                # Try shifting mass between a[i] and a[j]
                for delta_frac in [0.05, 0.1, 0.2]:
                    delta = delta_frac * min(a[i], a[j])
                    for sign in [1, -1]:
                        cand = a.copy()
                        cand[i] -= sign * delta
                        cand[j] += sign * delta
                        cand = np.maximum(cand, 0)
                        cs = cand.sum()
                        if cs < 0.01:
                            continue
                        cand /= cs
                        v = float(np.convolve(cand, cand).max())
                        if v < best_M * (1.0 - 1e-10):
                            best_M = v
                            best_a = cand.copy()
                            a = cand.copy()
                            break

    return best_a
