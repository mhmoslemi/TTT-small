    stages = [
        (0.10, 0.12),
        (0.03, 0.15),
        (0.008, 0.18),
        (0.002, 0.22),
        (0.0005, 0.22),
    ]

    best_p = p.copy()
    best_sc = _score_arr(p)
    start = time.monotonic()
    avail = max(0.0, stop_time - start - 8.0)
    total_w = sum(w for _, w in stages)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        cum += wt
        end_stage = min(stop_time - 4.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)
        best_u = u.copy()
        try:
            f0, _ = lse_func(u, T)
        except Exception:
            f0 = 1e100
        best_fstage = f0
        stale = 0
        while time.monotonic() < end_stage - 1.5:
            maxit = 35 if n > 2000 else 50
            opts = {"maxiter": maxit, "gtol": 1e-12, "ftol": 1e-16, "maxls": 12, "maxdisp": 0}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_fstage - 1e-14:
                    best_fstage = fval
                    best_u = res.x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
            else:
                stale += 1
                if stale >= 3:
                    break
            if getattr(res, "status", 1) == 0:
                break

        p_cur = _softmax_safe(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        np.maximum(p_cur, 0.0, out=p_cur)
        p_cur /= float(p_cur.sum())
        sc = _score_arr(p_cur)
        if math.isfinite(sc) and sc < best_sc - 1e-14:
            best_sc = sc
            best_p = p_cur.copy()
            p = p_cur.copy()
            u = np.log(np.maximum(p, 1e-30), dtype=np.float64)
            u = u - float(u.mean())
        else:
            p = best_p.copy()
            u = np.log(np.maximum(p, 1e-30), dtype=np.float64)
            u = u - float(u.mean())
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p


def _coord_descent(p, stop_time, consider=None, seed=12345, active_k=64):
    n = int(p.size)
    if n == 0:
        return p, float('inf')
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p, float('inf')
    p = p / s
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    Ltot = 2 * n - 1
    GR = (math.sqrt(5.0) - 1.0) / 2.0
    flat = 0
    sweep = 0
    max_sweeps = 200000

    G = 24
    GS_ITER = 12

    def eval_1d(w, base, infl, diag_local, max_const):
        aff_max = float(np.max(base + w * infl))
        if 0 <= diag_local < len(base):
            dval = base[diag_local] + w * infl[diag_local] + w * w
            if dval > aff_max:
                aff_max = dval
        return max(aff_max, max_const)

    def do_sweep():
        nonlocal p, c, cur_score
        order = rng.permutation(n)
        for oi in order:
            i = int(oi)
            v = float(p[i])
            if v < 1e-15:
                v = 0.0
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            infl = 2.0 * p[:Ls]
            diag_local = i  # 2i - start = 2i - i = i

            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < Ltot:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

            base = c[start:end]

            x_hi = max(3.0 * v, 1.0 / n, 1e-8)
            x_hi = min(x_hi, 0.5 / n * 4.0)
            x_hi = max(x_hi, 1e-10)
            x_lo = 0.0

            pts = np.linspace(x_lo, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                ins = np.searchsorted(pts, v)
                pts = np.insert(pts, ins, v)

            S = 1.0
            ws = pts - v
            s_new = S + ws
            valid = s_new > 1e-12
            if not np.any(valid):
                continue

            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag_local < Ls:
                aff[:, diag_local] += ws * ws
            maff = np.max(aff, axis=1)
            maff = np.maximum(maff, max_const)
            scores = np.full(pts.shape, float('inf'), dtype=np.float64)
            scores[valid] = 2.0 * n * maff[valid] / (s_new[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            if 0 < imin < len(pts) - 1:
                lo = float(pts[max(0, imin - 2)])
                hi = float(pts[min(len(pts) - 1, imin + 2)])
            else:
                lo = x_lo
                hi = x_hi
            if hi - lo < 1e-15:
                lo = max(0.0, best_x - 1e-7)
                hi = best_x + 1e-7

            a_gs, b_gs = lo, hi
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)
            f_c = eval_1d(c_gs - v, base, infl, diag_local, max_const)
            f_c = 2.0 * n * f_c / ((1.0 + c_gs - v) ** 2)
            f_d = eval_1d(d_gs - v, base, infl, diag_local, max_const)
            f_d = 2.0 * n * f_d / ((1.0 + d_gs - v) ** 2)
            for _ in range(GS_ITER):
                if b_gs - a_gs < 1e-14 * max(1.0, abs(best_x)):
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = eval_1d(c_gs - v, base, infl, diag_local, max_const)
                    f_c = 2.0 * n * f_c / ((1.0 + c_gs - v) ** 2)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_d = eval_1d(d_gs - v, base, infl, diag_local, max_const)
                    f_d = 2.0 * n * f_d / ((1.0 + d_gs - v) ** 2)
            x_mid = (a_gs + b_gs) * 0.5
            fm = eval_1d(x_mid - v, base, infl, diag_local, max_const)
            fm = 2.0 * n * fm / ((1.0 + x_mid - v) ** 2)
            if fm < best_val:
                best_x = x_mid
                best_val = fm
            for xp in (x_lo, x_hi):
                fp = eval_1d(xp - v, base, infl, diag_local, max_const)
                fp = 2.0 * n * fp / ((1.0 + xp - v) ** 2)
                if fp < best_val:
                    best_x = xp
                    best_val = fp

            if math.isfinite(best_val) and best_val < cur_score / (1.0 ** 2) - 1e-15 and 0.0 <= best_x <= x_hi:
                w = best_x - v
                if abs(w) > 1e-16:
                    p[i] = best_x
                    aff_seg = base + w * infl
                    if 0 <= diag_local < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag_local] += w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    c[start:end] = aff_seg
                    ss = 1.0 + w
                    cur_score = 2.0 * n * new_max / (ss * ss)

    while sweep < max_sweeps and time.monotonic() < stop_time - 3.0:
        improved = False
        pre_max = float(c.max())
        do_sweep()
        post_max = float(c.max())
        ss = float(p.sum())
        if ss > 1e-300 and math.isfinite(ss):
            p /= ss
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
        else:
            new_score = float('inf')

        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
            improved = True
            if consider:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3 and time.monotonic() < stop_time - 15.0:
                kick = 0.015 + 0.03 * min(1.0, flat / 30.0)
                p = p * np.exp(rng.normal(0.0, kick, n))
                p = np.maximum(p, 1e-15)
                p /= float(p.sum())
                c = _conv_self(p)
                cur_score = float(2.0 * n * c.max())
                flat = 0

        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 35.0
    if hard <= t0 + 5.0:
        hard = t0 + max(3.0, budget - 2.0)

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float('inf')

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_score = sc
            best_arr = a.copy()

    consider(parent)
    n_work = int(best_arr.size)
    if n_work < 128:
        n_work = 3072
    if n_work > 4096:
        n_work = 3072

    p_work = _resample(best_arr, n_work)

    # Phase 1: LSE optimization
    lse_time = min(60.0, 0.08 * budget)
    lse_end = t0 + lse_time
    if lse_end > t0 + 15.0 and time.monotonic() < lse_end - 5.0:
        try:
            p2 = _lse_optimize(p_work, lse_end, consider=consider)
            if p2 is not None and p2.size == n_work and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p_work = p2 / float(p2.sum())
                consider(p_work)
        except Exception:
            pass

    # Phase 2: Main coordinate descent
    cd_end = min(hard - 10.0, t0 + budget * 0.55)
    if cd_end > time.monotonic() + 30.0:
        p_start = _resample(best_arr, n_work)
        try:
            p_cd, _ = _coord_descent(p_start, cd_end, consider=consider, seed=int(seed + 7777))
            if p_cd is not None and p_cd.size > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                consider(p_cd)
        except Exception:
            pass

    # Phase 3: Second CD from best with different seed
    rem = hard - 10.0 - time.monotonic()
    if rem > 40.0:
        p_start2 = _resample(best_arr, n_work)
        cd2_end = t0 + budget * 0.80
        if cd2_end > time.monotonic() + 20.0:
            try:
                p_cd2, _ = _coord_descent(p_start2, cd2_end, consider=consider, seed=int(seed + 8888))
                if p_cd2 is not None and p_cd2.size > 0 and np.all(np.isfinite(p_cd2)) and p_cd2.sum() > 0:
                    consider(p_cd2)
            except Exception:
                pass

    # Phase 4: Try a different initialization (smooth window)
    rem = hard - 10.0 - time.monotonic()
    if rem > 50.0:
        x = (np.arange(n_work, dtype=np.float64) + 0.5) / n_work
        # Raised cosine window
        rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
        rc = np.maximum(rc, 1e-10)
        rc /= float(rc.sum())
        p_blend = 0.4 * rc + 0.6 * _resample(best_arr, n_work)
        p_blend = np.maximum(p_blend, 0.0)
        p_blend /= float(p_blend.sum())
        consider(p_blend)
        blend_end = min(hard - 10.0, time.monotonic() + rem * 0.5)
        if blend_end > time.monotonic() + 20.0:
            try:
                p_b, _ = _coord_descent(p_blend, blend_end, consider=consider, seed=int(seed + 9999))
                if p_b is not None and p_b.size > 0 and np.all(np.isfinite(p_b)) and p_b.sum() > 0:
                    consider(p_b)
            except Exception:
                pass

    # Phase 5: Final CD squeeze on overall best
    rem = hard - 8.0 - time.monotonic()
    if rem > 25.0:
        p_final = _resample(best_arr, n_work)
        try:
            p_f, _ = _coord_descent(p_final, hard - 8.0, consider=consider, seed=int(seed + 11111))
            if p_f is not None and p_f.size > 0 and np.all(np.isfinite(p_f)) and p_f.sum() > 0:
                consider(p_f)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
