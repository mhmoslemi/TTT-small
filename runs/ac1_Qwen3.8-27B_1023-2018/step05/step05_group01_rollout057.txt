    if math.isfinite(sc) and consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p, best_local


def _lse_optimize(p, stop_time, seed=42, consider=None):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        Gf = rfft(ww, cross_size)
        c = irfft(F * Gf, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def softmax_fn(uu):
        m = float(np.max(uu))
        ex = np.exp(uu - m)
        sm = float(ex.sum())
        if sm <= 0 or not math.isfinite(sm):
            return np.full(uu.size, 1.0 / uu.size)
        return ex / sm

    def lse_func(uu, T):
        pp = softmax_fn(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    stages = (
        (0.10, 0.10),
        (0.04, 0.12),
        (0.012, 0.14),
        (0.003, 0.16),
        (0.0008, 0.16),
        (0.0002, 0.16),
        (0.00005, 0.16),
    )

    start = time.monotonic()
    avail = max(0.0, float(stop_time) - start - 5.0)
    if avail <= 5.0:
        return p

    best_p = p.copy()
    total_w = float(sum(w for _, w in stages))
    rng = np.random.default_rng(seed)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 2.0, start + avail * (cum / total_w))
        if end_stage <= now + 1.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)

        best_u = u.copy()
        f0, _ = lse_func(best_u, T)
        if not math.isfinite(f0):
            continue
        best_f = f0
        stale = 0

        while time.monotonic() < end_stage - 0.5:
            opts = {"maxiter": 30, "gtol": 1e-11, "ftol": 1e-14,
                    "maxls": 25, "maxcor": 20, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-14:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            best_u = best_u + rng.normal(0, 0.004, best_u.size)
                            continue
            if getattr(res, "status", 1) == 0 and stale < 3:
                best_u = best_u + rng.normal(0, 0.002, best_u.size)

        p_cur = softmax_fn(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur):
            p = p_cur.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(p)
            except Exception:
                pass

    return p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 18.0
    if deadline <= t0 + 2.0:
        deadline = t0 + max(2.0, budget * 0.9)

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    n = best_arr.size

    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        n = 64
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    def make_tilted(src, n_out, sd, amp_range=(0.2, 1.5)):
        p = _resample_mid(src, n_out)
        rng = np.random.default_rng(sd)
        x = (np.arange(n_out, dtype=np.float64) + 0.5) / float(n_out)
        side = 1.0 if rng.random() < 0.5 else -1.0
        amp = amp_range[0] + (amp_range[1] - amp_range[0]) * float(rng.random())
        curv = 0.0 + 2.5 * float(rng.random())
        cub = -1.5 + 3.0 * float(rng.random())
        q = p * np.exp(side * amp * (2.0 * x - 1.0) + curv * (2.0 * x - 1.0) ** 2 + cub * (2.0 * x - 1.0) ** 3)
        q = np.maximum(q, 1e-15)
        s = float(q.sum())
        if s > 0:
            q /= s
        else:
            q = p
        return q

    def make_gauss_tilted(n_out, sd):
        rng = np.random.default_rng(sd)
        x = (np.arange(n_out, dtype=np.float64) + 0.5) / float(n_out)
        mu = 0.3 + 0.4 * float(rng.random())
        sigma = 0.1 + 0.3 * float(rng.random())
        tilt = (rng.random() - 0.5) * 0.8
        q = np.exp(-0.5 * ((x - mu) / sigma) ** 2 + tilt * (x - 0.5))
        q = np.maximum(q, 1e-15)
        s = float(q.sum())
        if s > 0:
            q /= s
        return q

    def run_cd(p_init, end_time, sd, G=48, do_lse=True, lse_frac=0.4):
        nonlocal best_arr, best_score
        p = _resample_mid(p_init, n)
        if p.size != n or float(p.sum()) <= 0:
            p = np.full(n, 1.0 / n)
        p = p / float(p.sum())
        consider(p)

        if time.monotonic() > end_time - 12.0:
            return p

        rem = float(end_time) - time.monotonic()
        if do_lse and rem > 40.0:
            lse_stop = time.monotonic() + lse_frac * rem
            lse_stop = min(lse_stop, end_time - max(20.0, 0.5 * rem))
            if lse_stop > time.monotonic() + 15.0:
                try:
                    p2 = _lse_optimize(p, lse_stop, seed=sd, consider=consider)
                    if p2 is not None and p2.size == n and float(p2.sum()) > 0:
                        p = p2 / float(p2.sum())
                        consider(p)
                except Exception:
                    pass

        cd_stop = float(end_time) - 2.0
        if cd_stop > time.monotonic() + 10.0:
            try:
                p3, _ = _coord_descent(p, cd_stop, seed=sd, G=G, consider=consider)
                if p3 is not None and p3.size == n and float(p3.sum()) > 0:
                    p = p3 / float(p3.sum())
                    consider(p)
            except Exception:
                pass
        return p

    # Phase 1: Main coordinate descent on parent
    p1_end = t0 + 0.45 * (deadline - t0)
    run_cd(best_arr, p1_end, int(seed + 1), G=56, do_lse=True, lse_frac=0.35)

    # Phase 2: Perturb and re-optimize
    if time.monotonic() < deadline - 60.0:
        rng2 = np.random.default_rng(seed + 555)
        pert = best_arr * np.exp(rng2.normal(0, 0.03, best_arr.size))
        pert = np.maximum(pert, 0.0)
        ps = float(pert.sum())
        if ps > 0:
            pert /= ps
        p2_end = min(deadline - 45.0, time.monotonic() + 0.25 * (deadline - time.monotonic()))
        if p2_end > time.monotonic() + 25.0:
            run_cd(pert, p2_end, int(seed + 1111), G=56, do_lse=False)

    # Phase 3: Different shape initializations
    if time.monotonic() < deadline - 80.0:
        for init_idx in range(3):
            if time.monotonic() > deadline - 50.0:
                break
            if init_idx == 0:
                init = make_tilted(best_arr, n, int(seed + 3000 + init_idx * 77), amp_range=(0.3, 1.2))
            elif init_idx == 1:
                init = make_gauss_tilted(n, int(seed + 4000 + init_idx * 77))
            else:
                init = make_tilted(best_arr, n, int(seed + 5000 + init_idx * 77), amp_range=(0.5, 2.0))
            sub_end = min(deadline - 40.0, time.monotonic() + 0.2 * (deadline - time.monotonic()))
            if sub_end > time.monotonic() + 20.0:
                run_cd(init, sub_end, int(seed + 6000 + init_idx * 111), G=48, do_lse=(init_idx == 0))

    # Phase 4: Try alternate length
    if time.monotonic() < deadline - 100.0:
        n_alt = n // 2 if n > 2048 else n * 2
        n_alt = min(n_alt, 4096)
        if n_alt != n:
            p_alt = _resample_mid(best_arr, n_alt)
            sub_end = min(deadline - 30.0, time.monotonic() + 0.15 * (deadline - time.monotonic()))
            if sub_end > time.monotonic() + 30.0:
                # Run CD at alternate length
                p_alt_cur = np.maximum(p_alt, 0.0).copy()
                p_alt_cur /= float(p_alt_cur.sum())
                n_alt_int = int(p_alt_cur.size)
                Ltot_a = 2 * n_alt_int - 1
                c_a = _conv_self(p_alt_cur)
                sc_a = 2.0 * n_alt_int * float(c_a.max())
                cd_stop_alt = float(sub_end) - 2.0
                if cd_stop_alt > time.monotonic() + 15.0:
                    try:
                        p_alt_opt, _ = _coord_descent(p_alt_cur, cd_stop_alt, seed=int(seed + 9000), G=48, consider=consider)
                        if p_alt_opt is not None and p_alt_opt.size == n_alt_int and float(p_alt_opt.sum()) > 0:
                            consider(p_alt_opt)
                    except Exception:
                        pass

    # Phase 5: Final intensive CD on best
    final_end = min(deadline - 10.0, t0 + budget - 10.0)
    if final_end > time.monotonic() + 20.0 and best_arr.size > 0:
        nbest = int(best_arr.size)
        rem = final_end - time.monotonic()
        if rem > 50.0:
            # LSE polish
            if nbest <= 4096:
                lstop = min(final_end - 30.0, time.monotonic() + 0.3 * rem)
                if lstop > time.monotonic() + 12.0:
                    try:
                        p2 = _lse_optimize(best_arr, lstop, seed=int(seed + 99999), consider=consider)
                        if p2 is not None and p2.size == nbest and float(p2.sum()) > 0:
                            p2 = p2 / float(p2.sum())
                            consider(p2)
                    except Exception:
                        pass
            # Coordinate descent polish
            cstop = min(final_end - 5.0, time.monotonic() + max(15.0, 0.6 * (final_end - time.monotonic())))
            if cstop > time.monotonic() + 10.0:
                try:
                    p3, _ = _coord_descent(best_arr, cstop, seed=int(seed + 88888), G=64, consider=consider)
                    if p3 is not None and p3.size == nbest and float(p3.sum()) > 0:
                        p3 = p3 / float(p3.sum())
                        consider(p3)
                except Exception:
                    pass

    # Final validation with exact evaluator
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.where(np.isfinite(parent), parent, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
