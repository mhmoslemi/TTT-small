def _lp_step(a, M):
    n = len(a)
    L = 2 * n - 1
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for x in range(L):
        j_lo = max(0, x - n + 1)
        j_hi = min(n - 1, x)
        for j in range(j_lo, j_hi + 1):
            aj = a[x - j]
            if aj > 0:
                rows.append(x)
                cols.append(j)
                vals.append(aj)
    A = csr_matrix((vals, (rows, cols)), shape=(L, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(L, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, 
                      method='highs', options={'time_limit': 30})
        if res.status == 0:
