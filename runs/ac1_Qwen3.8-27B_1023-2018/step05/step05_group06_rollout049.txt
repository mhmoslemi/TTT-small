        gr = (math.sqrt(5.0) - 1.0) / 2.0

        while remaining() > 5.0 and time.monotonic() < stop_time:
            sweep += 1
            improved = False
            order = rng.permutation(n)

            for idx, i in enumerate(order):
                if time.monotonic() > stop_time - 2.0 or remaining() < 3.0:
                    break
                if S < 1e-300:
                    break
                if S > 1.4 or S < 0.72:
                    p /= S
                    c /= (S * S)
                    S = 1.0

                v = float(p[i])
                start = i
                end = min(i + n, Ltot)
                Ls = end - start
                if Ls <= 0:
                    continue

                # infl[j] = 2 * p[j] for j in 0..Ls-1 (since c[start+k] += 2w*p[k])
                infl = 2.0 * p[:Ls]

                # Max outside window
                max_const = 0.0
                if start > 0:
                    mc1 = float(c[:start].max())
                    if mc1 > max_const:
                        max_const = mc1
                if end < Ltot:
                    mc2 = float(c[end:].max())
                    if mc2 > max_const:
                        max_const = mc2

                base = c[start:end]
                diag_off = 2 * i - start

                # Search range
                x_hi = max(8.0 * v, 25.0 / n, 0.005)
                x_hi = min(x_hi, 2.5)
                x_lo = 0.0
                if not math.isfinite(x_hi) or x_hi <= x_lo:
                    x_hi = max(2.0 * v, 1e-4) + 1e-6
                    x_lo = 0.0

                # Grid evaluation (vectorized)
                pts = np.linspace(x_lo, x_hi, grid_size + 1)
                ws = pts - v  # (G+1,)

                # aff[k, j] = base[j] + ws[k] * infl[j] (+ws[k]^2 at diag)
                aff = base[None, :] + ws[:, None] * infl[None, :]
                if 0 <= diag_off < Ls:
                    aff_diag = aff[:, diag_off].copy()
                    aff[:, diag_off] = aff_diag + ws * ws
                maff = aff.max(axis=1)
                maff = np.maximum(maff, max_const)
                maff = np.maximum(maff, 0.0)
                dens = S + ws
                valid = dens > 1e-15
                vals = np.where(valid, 2.0 * n * maff / (dens * dens), np.inf)

                if not np.any(np.isfinite(vals)):
                    continue

                imin = int(np.argmin(vals))
                best_x = float(pts[imin])
                best_val = float(vals[imin])

                # Golden section refinement
                if grid_size > 3:
                    dw = (x_hi - x_lo) / grid_size
                    lo_gs = max(0.0, best_x - 2.0 * dw)
                    hi_gs = min(x_hi, best_x + 2.0 * dw)
                    if hi_gs - lo_gs > 1e-13:
                        a_g, b_g = lo_gs, hi_gs
                        c_g = b_g - gr * (b_g - a_g)
                        d_g = a_g + gr * (b_g - a_g)

                        def f_of_x(x):
                            w = x - v
                            d = S + w
                            if d < 1e-15:
                                return float('inf')
                            # Compute max in window
                            nb = base + w * infl
                            if 0 <= diag_off < Ls:
                                nb_diag = nb[diag_off] + w * w
                                if nb_diag > nb.max():
                                    m = nb_diag
                                else:
                                    m = float(nb.max())
                            else:
                                m = float(nb.max())
                            if m < max_const:
                                m = max_const
                            if m < 0.0:
                                m = 0.0
                            return 2.0 * n * m / (d * d)

                        f_c = f_of_x(c_g)
                        f_d = f_of_x(d_g)
                        for _ in range(refine_iter):
                            if f_c < f_d:
                                b_g = d_g
                                d_g = c_g
                                f_d = f_c
                                c_g = b_g - gr * (b_g - a_g)
                                f_c = f_of_x(c_g)
                            else:
                                a_g = c_g
                                c_g = d_g
                                f_c = f_d
                                d_g = a_g + gr * (b_g - a_g)
                                f_d = f_of_x(d_g)
                        x_opt = 0.5 * (a_g + b_g)
                        f_opt = f_of_x(x_opt)
                        if f_opt < best_val:
                            best_x, best_val = x_opt, f_opt

                # Apply update
                if math.isfinite(best_val) and best_val < cur_score - 1e-14 and best_x >= 0.0:
                    w = best_x - v
                    if abs(w) > 1e-16:
                        p[i] = best_x
                        S += w
                        c[start:end] += w * infl
                        if 0 <= diag_off < Ls:
                            c[start + diag_off] += w * w
                        cur_score = best_val
                        improved = True

            # End of sweep
            if S > 1e-300 and math.isfinite(S):
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0
            else:
                break

            # Periodic exact recompute
            if sweep % recompute_every == 0:
                c = fft_conv(p)

            cur_max = float(c.max())
            new_score = 2.0 * n * cur_max
            if math.isfinite(new_score) and new_score < best_local - 1e-14:
                best_local = new_score
                best_p_local = p.copy()

            if improved:
                flat = 0
            else:
                flat += 1
                if flat >= 4 and remaining() > 20.0:
                    break
            if flat >= 12:
                break

        return best_p_local, best_local

    # === Main optimization ===
    p_work = p.copy()

    # Phase 1: Coarse coordinate descent (25% time)
    t1_stop = min(deadline - 0.65 * budget, t0 + 0.25 * budget)
    if remaining() > 15.0:
        r1, _ = coord_descent(p_work, t1_stop, grid_size=25, refine_iter=10, recompute_every=6)
        if r1 is not None:
            p_work = r1
            consider(p_work)

    # Phase 2: Fine coordinate descent (45% time)
    t2_stop = min(deadline - 0.25 * budget, t0 + 0.70 * budget)
    if remaining() > 15.0:
        r2, _ = coord_descent(p_work, t2_stop, grid_size=50, refine_iter=16, recompute_every=8)
        if r2 is not None:
            p_work = r2
            consider(p_work)

    # Phase 3: Restart with perturbation (18% time)
    t3_stop = deadline - 0.12 * budget
    if remaining() > 20.0:
        n_restarts = 4
        pert_sigmas = [0.04, 0.08, 0.12, 0.06]
        for r in range(n_restarts):
            if time.monotonic() > t3_stop - 5.0 or remaining() < 10.0:
                break
            sigma = pert_sigmas[r % len(pert_sigmas)]
            p_pert = best_p * np.exp(rng.normal(0.0, sigma, n))
            p_pert = np.maximum(p_pert, 0.0)
            ss = float(p_pert.sum())
            if ss > 1e-12:
                p_pert /= ss
            stop_r = min(t3_stop - 3.0, time.monotonic() + max(5.0, remaining() * 0.2))
            r3, _ = coord_descent(p_pert, stop_r, grid_size=40, refine_iter=13, recompute_every=8)
            if r3 is not None:
                consider(r3)

    # Phase 4: Final fine polish (12% time)
    if remaining() > 10.0:
        t4_stop = deadline - 5.0
        r4, _ = coord_descent(best_p, t4_stop, grid_size=65, refine_iter=20, recompute_every=6)
        if r4 is not None:
            consider(r4)

    # Final validation
    out = best_p.copy()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    ss = float(out.sum())
    if ss < 0.01:
        out = np.full(128, 1.0 / 128.0)

    result_list = [float(x) for x in out]
    final_sc = evaluate_sequence(result_list)
    if not math.isfinite(final_sc):
        try:
            pp = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
            pp = np.clip(pp, 0.0, 1000.0)
            result_list = [float(x) for x in pp]
            final_sc = evaluate_sequence(result_list)
        except:
            result_list = [1.0]
        if not math.isfinite(final_sc):
            result_list = [1.0]

    return result_list
