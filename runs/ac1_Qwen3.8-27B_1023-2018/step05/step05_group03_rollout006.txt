def _fft_cross(p, w):

    L = 3 * n - 2

        valid = (rows >= 0) & (rows < 2 * n - 1)
        cols = cols[valid]
        rows = rows[valid]
        data = a[off[valid]].astype(np.float64)
        A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        cobj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        t_start = time.monotonic()

            return None
        if res is not None and getattr(res, "success", False):
            b = np.asarray(res.x, dtype=np.float64).ravel()

            b = np.maximum(b, 0.0)
            if float(b.sum()) > 1e-10:

        pass
    return None

            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 30.0:
                    sigmas = (0.02, 0.05, 0.08, 0.12, 0.18)
                    sigma = sigmas[min(flat - 3, len(sigmas) - 1)]
                    noise = rng.normal(0.0, sigma, n)
                    q = p * np.exp(noise)
                    if sigma >= 0.08:
                        x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
                        amp = 0.15 + 0.35 * float(rng.random())
                        f1 = 1 + int(rng.integers(0, 6))
                        ph1 = 2.0 * math.pi * float(rng.random())
                        q *= np.exp(amp * math.sin(2.0 * math.pi * f1 * x + ph1))
                    q = np.maximum(q, 0.0)
                    qs = float(q.sum())
                    if math.isfinite(qs) and qs > 0.0:
                        p = q / qs
                        c = _fft_conv(p)
                        c = np.maximum(c, 0.0)
                        S = 1.0
                        infl = 2.0 * p
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 5e-14:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider is not None:
                                try:
                                    consider(best_p)
                                except Exception:
                                    pass
                    flat = 0
                else:
                    break
        sweep += 1

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(20.0, budget)
    reserve = 30.0 if budget > 200 else 15.0
    hard = t0 + budget - reserve
    if hard <= t0 + 3.0:
        hard = t0 + max(2.0, budget - 1.0)

    parent = _get_parent()
    n = int(parent.size)
    best_arr = _normalize_nonneg(parent.copy())
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.full(64, 1.0 / 64.0)
        best_score = float(_score_exact(best_arr))

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return
        if arr.size == 0:
            return
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if not math.isfinite(s) or s < 0.01:
            return
        arr = arr / s
        try:
            sc = float(_score_exact(arr))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-13:
            best_arr = arr.copy()
            best_score = sc

    # Quick perturbations
    rng = np.random.default_rng(seed)
    base = best_arr.copy()
    for q in (0.6, 0.8, 1.3, 1.7):
        if time.monotonic() > hard - 50.0:
            break
        cand = np.power(np.maximum(base, 1e-15), float(q))
        cand = _normalize_nonneg(cand)
        consider(cand)

    for _ in range(4):
        if time.monotonic() > hard - 40.0:
            break
        side = 1.0 if rng.random() < 0.5 else -1.0
        amp = 0.25 + 0.8 * float(rng.random())
        x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
        cand = base * np.exp(side * amp * (2.0 * x - 1.0))
        cand = _normalize_nonneg(cand)
        consider(cand)

    # Phase 1: LSE optimization
    p1_stop = min(hard - 30.0, t0 + max(60.0, 0.28 * budget))
    if p1_stop > time.monotonic() + 30.0:
        try:
            p = best_arr.copy()
            p = _lse_optimize(p, p1_stop, consider=consider, stages=(
                (0.008, 0.12),
                (0.003, 0.15),
                (0.001, 0.18),
                (0.0003, 0.20),
                (0.0001, 0.22),
                (0.00003, 0.25),
                (0.00001, 0.30),
            ))
            consider(p)
        except Exception:
            pass

    # Phase 2: LP iterations
    now = time.monotonic()
    lp_stop = min(hard - 60.0, now + max(40.0, 0.22 * budget))
    if lp_stop > now + 25.0 and n >= 64 and n <= 8192:
        try:
            lp_budget_per = min(90.0, max(20.0, 0.20 * budget))
            p = best_arr.copy()
            p = _lp_iterate(p, lp_stop, consider=consider, max_iter=5, lp_time_budget=lp_budget_per)
            consider(p)
        except Exception:
            pass

    # Phase 3: Coordinate descent
    now = time.monotonic()
    cd_stop = min(hard - 40.0, now + max(60.0, 0.30 * budget))
    if cd_stop > now + 30.0:
        try:
            p, _ = _coord_polish(best_arr, cd_stop, G=80, seed=seed + 777, consider=consider)
            consider(p)
        except Exception:
            pass

    # Phase 4: Second LSE round (fine)
    now = time.monotonic()
    p4_stop = min(hard - 50.0, now + max(40.0, 0.15 * budget))
    if p4_stop > now + 25.0:
        try:
            p = best_arr.copy()
            p = _lse_optimize(p, p4_stop, consider=consider, stages=(
                (0.0003, 0.25),
                (0.0001, 0.35),
                (0.00003, 0.40),
                (0.00001, 0.50),
            ))
            consider(p)
        except Exception:
            pass

    # Phase 5: More LP if time
    now = time.monotonic()
    lp2_stop = min(hard - 50.0, now + max(30.0, 0.12 * budget))
    if lp2_stop > now + 20.0 and n >= 64 and n <= 8192:
        try:
            p = best_arr.copy()
            p = _lp_iterate(p, lp2_stop, consider=consider, max_iter=3, lp_time_budget=60.0)
            consider(p)
        except Exception:
            pass

    # Phase 6: Final coordinate polish
    now = time.monotonic()
    cd2_stop = min(hard - 8.0, now + max(20.0, 0.15 * budget))
    if cd2_stop > now + 10.0:
        try:
            p, _ = _coord_polish(best_arr, cd2_stop, G=100, seed=seed + 99999, consider=consider)
            consider(p)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = _normalize_nonneg(parent)
        out = out / float(out.sum())
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls2 = [float(x) for x in _normalize_nonneg(parent)]
            sc2 = float(evaluate_sequence(ls2))
            if math.isfinite(sc2):
                return ls2
            return [1.0]
        return ls
    except Exception:
        return [1.0]
