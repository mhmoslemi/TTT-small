                    a[start:start + block], a[start + block:start + 2 * block] = \
                        a[start + block:start + 2 * block].copy(), a[start:start + block].copy()
                    a *= np.exp(rng.normal(0, 0.02, n))
            elif mode == 3:
                freq = rng.uniform(1.0, 6.0)
                a *= (1.0 + sigma * 0.5 * np.sin(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 6.28)))
            else:
                a = _spectral_perturb(a, rng, strength=sigma)
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            stall = 0
            if sigma > 0.45:
                break
            continue

        C = _conv_mat(n, a)
        moved = False

        # LP oracle at multiple M levels
        for M_factor, tl in [(1.0, 3.0), (1 + 1e-6, 3.0), (1 + 1e-4, 2.0), (1 + 1e-3, 2.0), (1 + 5e-3, 2.0)]:
            if time.monotonic() > deadline:
                break
            M_use = M * M_factor
            g = _lp_solve(C, M_use, n, tl=tl)
            if g is not None and g.sum() > 1e-12:
                g = g / g.sum()
                if float(np.max(np.abs(g - a))) > 1e-9:
                    cand, val = _line_search(a, g, n)
                    if cand is not None and val < M * (1 - 1e-11):
                        a = cand
                        moved = True
                        break
            if time.monotonic() > deadline - 1.0:
                break

        # Subgradient step
        if not moved and time.monotonic() < deadline - 0.5:
            cand = _subgrad_step(a, c, M, n)
            if cand is not None:
                a = cand
                moved = True

        # Coordinate descent
        if not moved and time.monotonic() < deadline - 0.5:
            cand = _coord_descent(a, c, M, n, rng)
            if cand is not None:
                a = cand
                moved = True

        # Small random kick
        if not moved:
            a *= np.exp(rng.normal(0, 0.012, n))
            a = np.maximum(a, 1e-12)
            a = a / a.sum()

    return best_a


def _optimize_aggressive(a0, deadline, seed=42):
    """More aggressive version with averaged LP directions."""
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stall = 0
    lp_hist = []
    max_stall = 300

    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        M = float(c.max())

        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            sigma = min(0.025 + 0.015 * (stall // max_stall), 0.45)
            a = _spectral_perturb(a, rng, strength=sigma)
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            stall = 0
            if sigma > 0.40:
                break
            continue

        C = _conv_mat(n, a)
        moved = False
        g_best = None

        for M_factor in [1.0, 1 + 1e-5, 1 + 2e-4, 1e-2 + 1]:
            if time.monotonic() > deadline - 1.0:
                break
            M_use = M * M_factor
            g = _lp_solve(C, M_use, n, tl=3.0)
            if g is not None and g.sum() > 1e-12:
                g = g / g.sum()
                if g_best is None or float(np.max(np.abs(g - g_best))) > 1e-9:
                    if g_best is not None:
                        lp_hist.append(g)
                        if len(lp_hist) > 5:
                            lp_hist.pop(0)
                    g_best = g
        if time.monotonic() > deadline - 1.0:
            break

        if g_best is not None and float(np.max(np.abs(g_best - a))) > 1e-9:
            # Try average of recent LP solutions
            if len(lp_hist) >= 2:
                g_avg = np.mean([g_best] + lp_hist, axis=0)
                g_avg = np.maximum(g_avg, 0)
                gs = g_avg.sum()
                if gs > 1e-12:
                    g_avg /= gs
                    cand, val = _line_search(a, g_avg, n)
                    if cand is not None and val < M * (1 - 1e-11):
                        a = cand
                        moved = True
            if not moved:
                g_try = 0.7 * g_best + 0.3 * a
                g_try = np.maximum(g_try, 0)
                gs = g_try.sum()
                if gs > 1e-12:
                    g_try /= gs
                    cand, val = _line_search(a, g_try, n)
                    if cand is not None and val < M * (1 - 1e-11):
                        a = cand
                        moved = True
            if not moved:
                cand, val = _line_search(a, g_best, n)
                if cand is not None and val < M * (1 - 1e-11):
                    a = cand
                    moved = True

        if not moved and time.monotonic() < deadline - 0.5:
            cand = _subgrad_step(a, c, M, n)
            if cand is not None:
                a = cand
                moved = True

        if not moved and time.monotonic() < deadline - 0.5:
            cand = _coord_descent(a, c, M, n, rng)
            if cand is not None:
                a = cand
                moved = True

        if not moved:
            a *= np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 1e-12)
            a = a / a.sum()

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0

    parent = _get_parent()
    n_parent = parent.size

    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(100)
        best_score = _score(best_seq)

    def try_update(cand):
        nonlocal best_seq, best_score
        try:
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.clip(np.asarray(cand, dtype=np.float64), 0, 1000).copy()
        except Exception:
            pass

    # Phase 1: Aggressive optimization of parent
    t1 = t0 + budget_s * 0.40
    try:
        r = _optimize_aggressive(parent, t1, seed=1)
        try_update(r)
    except Exception:
        pass

    # Phase 2: Try nearby lengths
    if time.monotonic() < deadline - 80:
        for dl in [-20, -10, -5, 5, 10, 20, 30, 50]:
            if time.monotonic() > deadline - 30:
                break
            nl = n_parent + dl
            if nl < 30 or nl > 700:
                continue
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, nl)
            init = np.interp(x_new, x_old, parent)
            init = np.clip(init, 0, 1000)
            init = init / init.sum()
            alloc = min(time.monotonic() + 25, deadline - 20)
            try:
                r = _optimize(init, alloc, seed=500 + nl)
                try_update(r)
            except Exception:
                pass

    # Phase 3: Diverse initializations at parent length
    if time.monotonic() < deadline - 50:
        pn = parent / parent.sum()
        x = np.linspace(0, 1, n_parent)
        rng_i = np.random.default_rng(42)
        inits = []
        for mu in [0.3, 0.4, 0.5, 0.6, 0.7]:
            for sig in [0.08, 0.15, 0.25]:
                v = np.exp(-0.5 * ((x - mu) / sig) ** 2) + 0.001
                inits.append(v / v.sum())
        for w in [0.15, 0.25, 0.35]:
            v = np.exp(-((x - w) / 0.06) ** 2) + np.exp(-((x - (1 - w)) / 0.06) ** 2) + 0.001
            inits.append(v / v.sum())
        for _ in range(6):
            v = np.abs(rng_i.normal(1, 0.4, n_parent)) + 0.01
            inits.append(v / v.sum())
        for f in [0.3, 0.5, 0.7]:
            for ini in inits[:3]:
                v = f * pn + (1 - f) * ini
                inits.append(v / v.sum())

        for i, ini in enumerate(inits):
            if time.monotonic() > deadline - 15:
                break
            alloc = min(time.monotonic() + 10, deadline - 10)
            try:
                r = _optimize(ini, alloc, seed=3000 + i)
                try_update(r)
            except Exception:
                pass

    # Phase 4: Final polish on best found so far
    if time.monotonic() < deadline - 10:
        ab = np.clip(best_seq, 0, 1000).astype(np.float64)
        ab = ab / max(ab.sum(), 1e-12)
        try:
            r = _optimize_aggressive(ab, deadline, seed=8888)
            try_update(r)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        fs = float(evaluate_sequence([float(v) for v in best_seq]))
        if not np.isfinite(fs):
            raise ValueError("invalid score")
    except Exception:
        best_seq = np.ones(100)

    return [float(v) for v in best_seq]
