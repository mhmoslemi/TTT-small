                    fvals[j] = 2.0 * n * float(arr.max()) / (S + wj) ** 2
                if np.any(np.isfinite(fvals)):
                    im2 = int(np.argmin(fvals))
                    if fvals[im2] < bv:
                        bv = fvals[im2]
                        bw = fine_pts[im2]
            
            # Accept if improvement
            if math.isfinite(bv) and bv < cur_score - 1e-15:
                w = bw
                p[i] += w
                S += w
                # Update convolution
                c[start:end] += w * infl
                if 0 <= diag_pos < Ls:
                    c[start + diag_pos] += w * w
                cur_score = bv
                improved = True
        
        # Rescale after sweep
        if S > 1e-300:
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            cur_score = 2.0 * n * float(c.max())  # recompute after rescale
            S = 1.0
            check_update(p)
        
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5 and perturb_count < 50:
                # Perturb
                noise = rng.normal(0, 0.01 * (1 + 0.1 * perturb_count), n)
                p = p * np.exp(noise)
                p = np.maximum(p, 0)
                ss = float(p.sum())
                if ss > 0:
                    p /= ss
                    S = 1.0
                else:
                    p = parent.copy()
                    S = 1.0
                c = np.convolve(p, p)
                cur_score = 2.0 * n * float(c.max())
                flat = 0
                perturb_count += 1
            elif flat >= 10:
                # Try a completely different perturbation
                p = p * rng.uniform(0.9, 1.1, n)
                p = np.maximum(p, 0)
                ss = float(p.sum())
                if ss > 0:
                    p /= ss
                c = np.convolve(p, p)
                cur_score = 2.0 * n * float(c.max())
                flat = 0
                perturb_count += 1
    
    # Final: check and return
    check_update(p)
    
    # Build output list
    out = np.asarray(best_arr, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    if out.size == 0:
        return [1.0]
    ls = [float(x) for x in out]
    
    try:
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
    except Exception:
        pass
    
    return ls

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:
        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(3072, dtype=np.float64)
    return p


def _conv(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(c, 0.0)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 20.0

    parent = _get_parent()
    n = parent.size

    S0 = float(parent.sum())
    if S0 < 0.01:
        parent = np.ones(n) / n
    else:
        parent = parent / S0

    best_arr = parent.copy()
    c0 = np.convolve(parent, parent)
    best_score = 2.0 * n * float(c0.max())

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            b = np.convolve(a, a)
            m = float(b.max())
            if not math.isfinite(m) or m < 0:
                return
            sc = 2.0 * a.size * m / (s * s)
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_arr = a.copy()
        except Exception:
            pass

    # Main coordinate descent
    p = parent.copy()
    S = 1.0
    c = _conv(p)
    Ltot = 2 * n - 1

    rng = np.random.default_rng(int(seed))

    G_coarse = 20
    G_fine = 12

    sweep = 0
    flat = 0
    perturb_count = 0
    last_verify = t0

    while time.monotonic() < hard - 25.0:
        sweep += 1
        improved = False

        # Periodic exact verification
        if sweep % 80 == 0 or (time.monotonic() - last_verify > 60.0):
            c = np.convolve(p, p)
            cur_score = 2.0 * n * float(c.max())
            consider(p)
            last_verify = time.monotonic()

        center = max(max_pos - n // 2, 0)
        order = list(range(center, min(center + n, n)))
        remaining = [i for i in range(n) if i not in set(order)]

                c[start:end] += w * infl
                if 0 <= diag_pos < Ls:

                cur_score = bv
                improved = True

        # Rescale
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv

            S = 1.0

            S = 1.0

            cur_score = 2.0 * n * float(c.max())

            break

            flat = 0

                else:

                    S = 1.0

                cur_score = 2.0 * n * float(c.max())
                flat = 0

    out = np.asarray(best_arr, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)

    try:
        fsc = float(evaluate_sequence(ls))
        if not math.isfinite(fsc):
            ls = [float(x) for x in parent]
            evaluate_sequence(ls)
    except Exception:
        ls = [float(x) for x in parent]

    return ls
