    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf

        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf

    sequence = [min(1000.0, x) for x in sequence]

def _lp_step(f, stop_time):
    """LP-based direction: maximize sum(g) s.t. f*g <= M, g >= 0."""
    n = int(f.size)
    if n < 10:
        return None
    M = float(_conv_self(f).max())
    if M <= 0 or not math.isfinite(M):
        return None

    # Build constraint matrix A: A[k,j] = f[j] for j <= k <= j+n-1
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f[j]

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        constraints = [A @ g <= b]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        t_start = time.monotonic()
        solver = None
        for slv in [cp.SCS, cp.ECOS, cp.CLP]:
            try:
                prob.solve(solver=slv, verbose=False)
                solver = slv
                break
            except Exception:
                continue
            if time.monotonic() > stop_time - 5:
                break
        if g.value is None:
            return None
        g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        gv = float(g_val.sum())
        if gv < 1e-10 or not math.isfinite(gv):
            return None
        return g_val / gv
    except Exception:
        return None
