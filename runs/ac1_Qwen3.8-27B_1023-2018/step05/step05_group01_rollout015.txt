def _lp_polish(f, B, n, max_iter=50, t_init=0.1):
    """LP-based polishing step."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix, lil_matrix
    
    f = f.copy()
    s = f.sum()
    f = f / s  # normalize
    
    for it in range(max_iter):
        # Recompute B
        b = np.convolve(f, f)
        B = b.max()
        
        if B <= 0:
            break
        
        # Build sparse constraint matrix A
        # A[m, j] = f[m-j] if 0 <= m-j < n
        rows, cols, vals = [], [], []
        for j in range(n):
            m_start = j
            m_end = min(2*n-1, j+n)
            for m in range(m_start, m_end):
                if 0 <= m - j < n:
                    rows.append(m)
                    cols.append(j)
                    vals.append(f[m-j])
        
        A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        
        # Solve LP: max sum(g) s.t. A g <= B, g >= 0
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, B), 
                      bounds=[(0, None)]*n, method='highs',
                      options={'maxiter': 10000, 'presolve': True})
        
        if res.status != 0:
            break
        
        g = res.x
        g_sum = g.sum()
        if g_sum <= 0:
            break
        
        # Rescale g
        g = g * (s / g_sum)
        
        # Line search for step size
        best_t = 0
        best_B_new = B
        for t in [t_init, t_init*0.5, t_init*0.25, t_init*0.1]:
            f_new = (1-t)*f + t*g
            b_new = np.convolve(f_new, f_new)

            if B_new < best_B_new - 1e-12:
                best_B_new = B_new
                best_t = t
        
        if best_t > 0:
            f = (1-best_t)*f + best_t*g
        else:

    return f

But the (f*d)_m term is a bilinear term (f and d). Wait, no: (f*d)_m = sum_j f[j] d[m-j]. For fixed f, this is LINEAR in d. So the constraint b_m + 2*sum_j f[j] d[m-j] <= t is linear in (d, t).

So the LP is:
minimize t
s.t. sum_j f[j] d[m-j] - 1/2 * (t - b_m) <= 0 for m = 0,...,2n-2
     sum_j d[j] = 0
     d[j] >= -f[j] for j = 0,...,n-1
     (t is unbounded)

This is an LP with n+1 variables (d and t) and 2n-1 + 1 + n constraints.

For n=3072: 3073 variables, ~9216 constraints. The constraint matrix is (2n-1) x (n+1) for the first group, plus the sum constraint and the non-negativity constraints.

The sparsity of the first group: each row m has n nonzeros (from the convolution with f) plus 1 for t. So the total nonzeros in the first group is (2n-1)*(n+1) ≈ 2n^2 ≈ 19M for n=3072. That's a lot.

The gradient of e_m w.r.t. d is: 2*f reversed, convolved... it's 2*(f^R * f)[d index]... no.

(f*d)_m = sum_j f[j] d[m-j]. d(f*d)_m/d d[i] = f[m-i] (if 0 <= m-i < n).

So the gradient of e_m w.r.t. d is: 2*f[m-i] (if valid), which is the same as before.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:
        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(3072, dtype=np.float64)
    return p

def _conv_nt(a):
    """Self-convolution via FFT for large n, direct for small n."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2*n - 1
    if n <= 512:
        return np.convolve(a, a)
    size = next_fast_len(L)
    F = rfft(a, size)
    return irfft(F*F, size)[:L]

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_nt(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0*n*m/(s*s)

def _softmax(u):
    m = u.max()
    e = np.exp(u - m)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(u.size, 1.0/u.size)
    return e/s

def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.full(n, 1.0/n)
    m = src.size
    if m == n:
        return src.copy()
    xp = (np.arange(m) + 0.5)/m
    fp = np.interp((np.arange(n) + 0.5)/n, xp, src)
    out = np.maximum(fp, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.full(n, 1.0/n)
    return out/s

def _lse_optimize(p, stop_time, consider=None, seed=42):
    """LSE-based optimization with adaptive temperature."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = p.size
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p/s  # normalize to sum=1
    
    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - uu.mean()
    
    u = p_to_u(p)
    conv_size = next_fast_len(2*n - 1)
    cross_size = next_fast_len(3*n - 2)
    conv_len = 2*n - 1
    cross_len = 3*n - 2
    
    def cfast(pp):
        F = rfft(pp, conv_size)
        c = irfft(F*F, conv_size)[:conv_len]
        return c
    
    def xfast(pp, ww):
        """Cross-correlation: q[i] = sum_k pp[k] * ww[i+k]"""
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F*G, cross_size)[:cross_len]
        q = c[n-1:2*n-1]
        return q
    
    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not np.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        T_eff = max(T, 1e-15)
        scaled = (c - cmax)/T_eff
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not np.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f_val = cmax
        else:
            f_val = cmax + T_eff * math.log(sum_exp/c.size)
            if not np.isfinite(f_val):
                f_val = cmax
            w = exps/sum_exp
        q = xfast(pp, w)
        gdp = 2.0*q
        tdp = float(pp@gdp)
        grad = pp*(gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return f_val, grad
    
    # Adaptive temperature schedule
    c0 = cfast(p)
    c0max = float(c0.max())
    T_start = 0.1 * c0max
    T_end = 0.00001 * c0max
    
    n_stages = 15
    log_T = np.linspace(math.log10(T_start), math.log10(T_end), n_stages)
    
    t_start = time.monotonic()
    avail = max(0.0, stop_time - t_start - 5.0)
    if avail < 10:
        return p
    
    best_p = p.copy()
    best_sc = 2.0*n*c0max  # score with sum=1
    rng = np.random.default_rng(seed)
    
    # Allocate time proportionally to stage index (later stages get more time)
    weights = np.linspace(1.0, 2.0, n_stages)
    weights /= weights.sum()
    cumw = np.cumsum(weights)
    
    for si in range(n_stages):
        now = time.monotonic()
        stage_end = t_start + avail * (cumw[si] if si < n_stages else 1.0)
        stage_end = min(stage_end, float(stop_time) - 2.0)
        if stage_end < now + 3.0:
            break
        
        T = 10**log_T[si]
        
        # Optimization loop for this T
        f0, _ = lse_func(u, T)
        if not np.isfinite(f0):
            continue
        best_f_stage = f0
        best_u_stage = u.copy()
        stale = 0
        
        while time.monotonic() < stage_end - 1.0:
            maxit = 50
            opts = {
                "maxiter": maxit, "gtol": 1e-12, "ftol": 1e-16,
                "maxls": 30, "maxcor": 25, "maxdisp": 0, "disp": False
            }
            try:
                res = minimize(lse_func, best_u_stage, args=(T,),
                              method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    fval = float(res.fun)
                    if np.isfinite(fval) and fval < best_f_stage - 1e-14:
                        best_f_stage = fval
                        best_u_stage = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            best_u_stage = best_u_stage + rng.normal(0, 0.003, n)
                            continue
            
            if getattr(res, "status", 1) != 0 or stale >= 2:
                break
        
        p_cur = _softmax(best_u_stage)
        p_cur = np.maximum(p_cur, 0.0)
        ss = p_cur.sum()
        if ss > 1e-300:
            p_cur /= ss
        
        c_cur = cfast(p_cur)
        sc_cur = 2.0 * n * float(c_cur.max())
        if np.isfinite(sc_cur) and sc_cur < best_sc - 1e-14:
            best_sc = sc_cur
            best_p = p_cur.copy()
            p = p_cur.copy()
        u = p_to_u(p)
        
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    
    return best_p

def _coord_descent(p, stop_time, G=48, seed=123, consider=None):
    """Coordinate descent with random ordering."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = p.size
    if n == 0:
        return p, math.inf
    s = p.sum()
    if s < 1e-300:
        return p, math.inf
    p = p/s
    
    Ltot = 2*n - 1
    c = _conv_nt(p)
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    
    while time.monotonic() < stop_time:
        improved = False
        order = rng.permutation(n)
        for idx in order:
            i = int(idx)
            if time.monotonic() > stop_time - 0.5:
                break
            
            S = float(p.sum())
            if S < 1e-300:
                break
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]
            diag_i = i  # position diag_i in base corresponds to c[2i]
            
            # Find max outside the affected region
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < Ltot:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc
            
            # Grid search for optimal p[i]
            x_hi = max(20*v + 1e-10, 0.1*S/n, 1e-10)
            x_hi = min(x_hi, 5.0*S)
            
            pts = np.linspace(0.0, x_hi, G+1)
            if 0 < v < x_hi:
                pts = np.unique(np.concatenate([pts, [v]]))
            
            def eval_x(x):
                w = x - v
                denom = S + w
                if denom < 1e-15:
                    return math.inf
                new_vals = base + w * infl
                dval = base[diag_i] + w * infl[diag_i] + w*w
                new_vals = np.maximum(new_vals, dval)
                new_vals = np.maximum(new_vals, max_const)
                mv = float(new_vals.max())
                return 2.0 * n * mv / (denom*denom)
            
            vals = np.array([eval_x(x) for x in pts])
            valid = np.isfinite(vals)
            if not np.any(valid):
                continue
            imin = int(np.argmin(np.where(valid, vals, np.inf)))
            best_x = float(pts[imin])
            best_val = float(vals[imin])
            
            # Local refinement
            if 0 < best_x < x_hi:
                dw = x_hi / G
                lo = max(0, best_x - 2*dw)
                hi = min(x_hi, best_x + 2*dw)
                if hi > lo:
                    fine = np.linspace(lo, hi, 20)
                    if 0 <= best_x <= x_hi:
                        fine = np.unique(np.concatenate([fine, [best_x]]))
                    vals2 = np.array([eval_x(x) for x in fine])
                    valid2 = np.isfinite(vals2)
                    if np.any(valid2):
                        imin2 = int(np.argmin(np.where(valid2, vals2, np.inf)))
                        if vals2[imin2] < best_val - 1e-15:
                            best_x = float(fine[imin2])
                            best_val = float(vals2[imin2])
            
            if np.isfinite(best_val) and best_val < cur_score - 1e-13:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    c[start:end] += w * infl
                    di = 2*i
                    if 0 <= di < Ltot:
                        c[di] += w*w
                    cur_score = best_val
                    improved = True
        
        # Rescan after full pass
        if time.monotonic() > stop_time - 1.0:
            break
        c = _conv_nt(p)
        ss = p.sum()
        if ss > 1e-300:
            p /= ss
            c /= ss*ss
        new_score = 2.0 * n * float(c.max())
        if np.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
            if consider:
                try:
                    consider(best_p)
                except Exception:
                    pass
        
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 5.0:
                    sigma = 0.02 + 0.01 * min(flat, 10)
                    p *= np.exp(rng.normal(0, sigma, n))
                    p = np.maximum(p, 0)
                    ss = p.sum()
                    if ss > 0:
                        p /= ss
                    else:
                        break
                    c = _conv_nt(p)
                    cur_score = 2.0 * n * float(c.max())
                    flat = 0
                else:
                    break
        
        sweep += 1
    
    return best_p, best_score
