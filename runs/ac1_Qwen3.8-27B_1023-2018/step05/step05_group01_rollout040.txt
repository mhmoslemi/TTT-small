def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())

        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _score_exact_arr(a):

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.full(n, 1.0 / n)
    r = int(rho[-1])
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def _pgd_optimize(b, stop_time, seed=42, n_nearmax=3):
    """Projected gradient descent with averaged subgradient over near-max lags."""
    b = np.maximum(np.asarray(b, dtype=np.float64).ravel(), 0.0).copy()
    n = b.size
    s = float(b.sum())
    if s < 1e-300:
        return b
    b = b / s

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1

    def conv(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    c = conv(b)
    cmax = float(c.max())
    best_b = b.copy()
    best_val = cmax

    start_tm = time.monotonic()
    alpha = 1.0 / (2.0 * n)
    patience = 0
    max_patience = 5000

    for it in range(500000):
        if time.monotonic() > stop_time - 2.0:
            break
        if patience > max_patience:
            break

        cmax = float(c.max())
        if cmax < best_val - 1e-15:
            best_val = cmax
            best_b = b.copy()
            patience = 0
        else:
            patience += 1

        # Find near-max lags
        thresh = cmax * 0.995 - 1e-15
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        if len(active) > n_nearmax:
            order = np.argsort(c[active])[::-1]
            active = active[order[:n_nearmax]]

        # Compute averaged subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, int(k) - n + 1)
            hi = min(int(k), n - 1)
            if hi >= lo:
                g[lo:hi + 1] += b[lo:int(k) + 1][::-1]
        g /= len(active)
        g *= 4.0 * n

        # Project gradient onto tangent space
        g = g - float(np.mean(g))

        # Line search (simple backtracking)
        step = alpha
        for _ls in range(6):
            b_new = _project_simplex(b - step * g)
            c_new = conv(b_new)
            cmax_new = float(c_new.max())
            if cmax_new < cmax - 1e-12:
                break
            step *= 0.5

        b = b_new
        c = c_new

        # Decay step size slowly
        if it % 200 == 0 and it > 0:
            alpha *= 0.95

    s_best = float(best_b.sum())
    if s_best > 1e-300:
        best_b = best_b / s_best
    return best_b


def _softmax_fn(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _lse_optimize(p, stop_time, seed=42, n_nearmax=5):
    """LSE-based smooth optimization via L-BFGS-B."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax_fn(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    stages = (

        (0.04, 0.08),
        (0.015, 0.10),
        (0.005, 0.12),
        (0.0015, 0.14),

    total_w = float(sum(w for _, w in stages))
    rng = np.random.default_rng(seed)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:

        if end_stage <= now + 1.0:

        cmax_v = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax_v) or cmax_v <= 0:
            break
        T = max(1e-300, frac * cmax_v)

        best_u = u.copy()
        f0, _ = lse_func(best_u, T)
        if not math.isfinite(f0):
            continue
        best_f = f0
        stale = 0

        while time.monotonic() < end_stage - 0.5:

            opts = {"maxiter": maxit, "gtol": 1e-12, "ftol": 1e-16,
                    "maxls": 25, "maxcor": 25, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-15:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            best_u = best_u + rng.normal(0, 0.003, best_u.size)
                            continue
            if getattr(res, "status", 1) == 0:
                if time.monotonic() < end_stage - 1.0 and stale < 3:
                    best_u = best_u + rng.normal(0, 0.002, best_u.size)
                    f_chk, _ = lse_func(best_u, T)
                    if math.isfinite(f_chk):
                        if f_chk < best_f:
                            best_f = f_chk
                            best_u = best_u.copy()
                        stale += 1
                else:
                    break

        p_cur = _softmax_fn(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif math.isfinite(sc_cur) and sc_cur < best_sc + 0.003:
            p = p_cur.copy()
        else:
            p = best_p.copy()
        u = p_to_u(p)

    return best_p


def _coord_polish(p, stop_time, G=None, seed=12345, consider=None):

                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])

            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:

def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        return src.copy()
    m = src.size
    xp = (np.arange(m) + 0.5) / m
    fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
    out = np.maximum(fp, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def _make_parabolic(n):
    x = np.arange(n, dtype=np.float64) + 0.5
    t = x / n
    a = t * (1.0 - t)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s > 0:
        a /= s
    return a


def _make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n, dtype=np.float64) + 0.5
    t = x / n
    center = 0.5
    a = np.exp(-0.5 * ((t - center) / sigma_frac) ** 2)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s > 0:
        a /= s
    return a


def _make_asym_init(src, n, sd):
    p = _resample_mid(src, n)
    rng = np.random.default_rng(sd)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    side = 1.0 if rng.random() < 0.5 else -1.0
    amp = 0.2 + 0.8 * float(rng.random())
    curv = 0.3 + 1.5 * float(rng.random())
    q = p * np.exp(side * amp * (2.0 * x - 1.0) + curv * (2.0 * x - 1.0) ** 2)
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if s > 0:
        q /= s
    else:
        q = p
    return q

        hard = t0 + max(2.0, budget * 0.9)
    final_reserve = max(10.0, 0.06 * (hard - t0))
    phase_limit = hard - final_reserve

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    n_parent = int(best_arr.size)

    def run_pipeline(init_arr, end_time, sd, do_pgd=True, do_lse=True, do_coord=True):
        nonlocal best_arr, best_score
        p = _resample_mid(init_arr, n_parent)
        if p.size != n_parent or float(p.sum()) <= 0:
            p = np.full(n_parent, 1.0 / n_parent)
        p = p / float(p.sum())
        consider(p)

        if time.monotonic() > end_time - 10.0:
            return

        # PGD phase
        if do_pgd and time.monotonic() < end_time - 30.0:
            pgd_stop = min(end_time - max(20.0, 0.4 * (end_time - t0)),
                           time.monotonic() + 0.3 * (end_time - time.monotonic()))
            if pgd_stop > time.monotonic() + 10.0:
                try:
                    p2 = _pgd_optimize(p, pgd_stop, seed=sd)
                    if p2 is not None and p2.size == n_parent and float(p2.sum()) > 0:
                        p = p2 / float(p2.sum())
                        consider(p)
                except Exception:
                    pass

        # LSE phase
        if do_lse and time.monotonic() < end_time - 20.0:
            rem_lse = end_time - time.monotonic()
            lse_stop = min(end_time - max(15.0, 0.25 * (end_time - t0)),
                           time.monotonic() + 0.5 * rem_lse)
            if lse_stop > time.monotonic() + 15.0:
                try:
                    p2 = _lse_optimize(p, lse_stop, seed=sd)
                    if p2 is not None and p2.size == n_parent and float(p2.sum()) > 0:
                        p = p2 / float(p2.sum())
                        consider(p)
                except Exception:
                    pass

        # Coordinate descent phase
        if do_coord and time.monotonic() < end_time - 5.0:
            cstop = min(end_time - 2.0, time.monotonic() + (end_time - time.monotonic()) - 2.0)
            if cstop > time.monotonic() + 10.0:
                try:
                    p3, _ = _coord_polish(p, cstop, seed=sd, consider=consider)
                    if p3 is not None and p3.size == n_parent and float(p3.sum()) > 0:
                        p = p3 / float(p3.sum())
                        consider(p)
                except Exception:
                    pass
        return p

    # Phase 1: Main optimization from parent
    p1_end = min(phase_limit, t0 + 0.45 * (phase_limit - t0))
    run_pipeline(best_arr, p1_end, int(seed + 100))

    # Phase 2: PGD + LSE + coord from perturbed parent
    if time.monotonic() < phase_limit - 60.0:
        rng2 = np.random.default_rng(seed + 777)
        for pr in range(2):
            if time.monotonic() > phase_limit - 30.0:
                break
            pert = best_arr * np.exp(rng2.normal(0, 0.03, best_arr.size))
            pert = np.maximum(pert, 0.0)
            ps = float(pert.sum())
            if ps > 0:
                pert /= ps
            sub_end = min(phase_limit - 20.0, time.monotonic() + 0.3 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 20.0:
                run_pipeline(pert, sub_end, int(seed + 2000 + pr * 100))

    # Phase 3: Try parabolic and Gaussian initializations
    if time.monotonic() < phase_limit - 50.0:
        n_par = n_parent
        par = _make_parabolic(n_par)
        sub_end = min(phase_limit - 30.0, time.monotonic() + 0.25 * (phase_limit - time.monotonic()))
        if sub_end > time.monotonic() + 20.0:
            run_pipeline(par, sub_end, int(seed + 5000), do_pgd=True, do_lse=True, do_coord=True)

    if time.monotonic() < phase_limit - 40.0:
        gau = _make_gaussian(n_parent, sigma_frac=0.35)
        sub_end = min(phase_limit - 20.0, time.monotonic() + 0.2 * (phase_limit - time.monotonic()))
        if sub_end > time.monotonic() + 15.0:
            run_pipeline(gau, sub_end, int(seed + 6000), do_pgd=True, do_lse=True, do_coord=True)

    # Phase 4: Asymmetric perturbation of best
    if time.monotonic() < phase_limit - 30.0:
        for init_seed in range(2):
            if time.monotonic() > phase_limit - 15.0:
                break
            init = _make_asym_init(best_arr, n_parent, int(seed + 7000 + init_seed * 777))
            sub_end = min(phase_limit - 10.0, time.monotonic() + 0.2 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 15.0:
                run_pipeline(init, sub_end, int(seed + 8000 + init_seed * 333),
                             do_pgd=True, do_lse=True, do_coord=True)

    # Phase 5: Final squeeze - LSE + coord on the best
    final_end = min(hard - 10.0, t0 + budget - 8.0)
    if final_end > time.monotonic() + 20.0 and best_arr.size > 0:
        nbest = int(best_arr.size)
        rem = final_end - time.monotonic()
        if rem > 50.0 and nbest <= 5000:
            lstop = min(final_end - 30.0, time.monotonic() + 0.35 * rem)
            if lstop > time.monotonic() + 15.0:
                try:
                    p2 = _lse_optimize(best_arr, lstop, seed=int(seed + 99999))
                    if p2 is not None and p2.size == nbest and float(p2.sum()) > 0:
                        p2 = p2 / float(p2.sum())
                        consider(p2)
                except Exception:
                    pass

        cstop = min(final_end - 4.0, time.monotonic() + max(15.0, 0.45 * (final_end - time.monotonic())))
        if cstop > time.monotonic() + 10.0:
            try:
                p3, _ = _coord_polish(best_arr, cstop, seed=int(seed + 123456789), consider=consider)
                if p3 is not None and p3.size == nbest and float(p3.sum()) > 0:
                    p3 = p3 / float(p3.sum())
                    consider(p3)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
