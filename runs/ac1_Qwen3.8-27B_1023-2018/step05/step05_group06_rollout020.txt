```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def _conv_self(a):
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _exact_score(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(p, p)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))

def _fft_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(p)
    return float(2.0 * n * c.max() / (s * s))
```

```python
def _solve_lp(f, M, n):
    """Solve: max sum(b) s.t. A@b <= M, b >= 0"""
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    # A has shape (2n-1, n)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # max 1^T b s.t. A b <= M, b >= 0
    # linprog requires min c^T x s.t. A_ub x <= b_ub, x >= 0
    c = -np.ones(n)  # maximize = minimize negative
    A_ub = A.tocsr()
    b_ub = np.full(2*n-1, M)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    return result
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csc_matrix
from scipy.optimize import minimize

def _conv_self(a):
    """FFT-based self-convolution (linear, length 2n-1)."""
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _exact_score(p):
    """Exact score using np.convolve (matches evaluator)."""
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(p, p)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))

def _fft_score(p):
    """FFT-based approximate score (fast)."""
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(p)
    return float(2.0 * n * c.max() / (s * s))

def _build_A_indices(n):
    """Precompute row/col indices for the (f*b) constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def _lp_step(f, M, rows, cols, n, time_limit=20):
    """
    Solve LP: max sum(b) s.t. A@b <= M, b >= 0
    where A[k,j] = f[k-j] if 0 <= k-j <= n-1.
    Returns optimal b or None.
    """
    vals = f[rows - cols]
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    Coordinate descent: for each i, find best a_i via 1D golden section.
    Normalization: sum(p) maintained at 1 throughout (renormalize periodically).
    """

    S = float(p.sum())
    if S < 1e-15:
        return p, math.inf
    p /= S
    S = 1.0
    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())

                    if max_const > mv:
                        mv = max_const
                    return 2.0 * n * mv / (denom * denom)

                f_c = eval_x(c_gs)
                f_d = eval_x(d_gs)
                for _ in range(14):
                    if f_c < f_d:
                        b_gs = d_gs
                        d_gs = c_gs
                        f_d = f_c
                        c_gs = b_gs - gr * (b_gs - a_gs)
                        f_c = eval_x(c_gs)
                    else:
                        a_gs = c_gs
                        c_gs = d_gs
                        f_c = f_d
                        d_gs = a_gs + gr * (b_gs - a_gs)
                        f_d = eval_x(d_gs)
                x_opt = (a_gs + b_gs) / 2.0
                f_opt = eval_x(x_opt)
                if f_opt < best_val:

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and best_x >= 0:

                if abs(w) > 1e-16:

                    old_v = v
                    p[i] = best_x
                    S += w
                    # Update c using old p values (p[i] is now new, so use old_v)
                    a_old = p[:Ls].copy()
                    if i < Ls:

                    improved = True

            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        if sweep_count % 3 == 0:
            c = _conv_self(p)
            np.maximum(c, 0.0, out=c)

        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()

            flat = 0
        else:

                noise = rng.normal(0.0, sigma, n)
                p *= np.exp(noise)
                p = np.maximum(p, 0.0)
                ss = float(p.sum())
                if ss > 1e-300:
                    p /= ss
                    S = 1.0
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score:

                        best_p = p.copy()
                        consider_callback(best_p)
```
