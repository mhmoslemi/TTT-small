def _eval_coord(pts, base, infl_vec, max_const, v, Sloc, n, di):
    pts = np.asarray(pts, dtype=np.float64).ravel()

        return np.zeros(0), pts
    w = pts - v
    den = Sloc + w
    valid = den > 1e-14
    aff = base[None,:] + w[:,None]*infl_vec[None,:]
    if 0 <= di < base.size:
        aff[:,di] = base[di] + w*infl_vec[di] + w*w
    np.maximum(aff, max_const, out=aff)
    mv = np.max(aff, axis=1)
    vals = np.full(pts.size, math.inf)
    if np.any(valid):
        vals[valid] = (2.0*n*mv[valid])/(den[valid]**2)
    vals = np.where(np.isfinite(vals), vals, math.inf)
    return vals, pts
