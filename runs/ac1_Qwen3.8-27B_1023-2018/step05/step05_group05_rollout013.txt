import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p

def _conv_self_fft(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(c, 0.0)

def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self_fft(a)
    m = float(c.max())
    return 2.0 * n * m / (s * s)

def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0 / u.size)
    return e / s

def _lse_optimize(p, stop_time, consider=None):
    """LSE smooth optimization using L-BFGS-B"""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p
    p = p / s
    
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return np.maximum(c, 0.0)
    
    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n-1:2*n-1]
        return np.maximum(q, 0.0)
    
    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = 1e-4 * cmax
        scaled = (c - cmax) / T
        exps = np.exp(np.minimum(scaled, 0))  # prevent overflow
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        return float(f), np.where(np.isfinite(grad), grad, 0.0)
    
    def p_to_u(pp):
        ll = np.log(np.maximum(pp, 1e-300))
        return ll - float(np.mean(ll))
    
    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score(p)
    
    stages = (
        (0.25, 0.10),
        (0.08, 0.12),
        (0.025, 0.14),
        (0.008, 0.16),
        (0.0025, 0.20),
        (0.0007, 0.18),
    )
    
    total_w = sum(w for _, w in stages)
    time_per = max(5.0, (stop_time - time.monotonic() - 5.0) / total_w)
    
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        end_stage = min(stop_time - 4.0, now + wt * time_per)
        cmax = float(cfast(p).max())
        T = max(1e-15, frac * cmax)
        stale = 0
        while time.monotonic() < end_stage - 1.0:
            opts = {"maxiter": 30, "gtol": 1e-12, "ftol": 1e-14,
                    "maxls": 25, "maxcor": 20, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval):
                    p_cur = _softmax(res.x)
                    sc_cur = _score(p_cur)
                    if math.isfinite(sc_cur) and sc_cur < best_sc - 1e-15:
                        best_sc = sc_cur
                        best_p = p_cur.copy()
                        p = p_cur.copy()
                        u = p_to_u(p)
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, "status", 1) == 0:
                break
    
    p_out = best_p.copy()
    ss = float(p_out.sum())
    if ss > 1e-300:
        p_out /= ss
    return p_out

def _coord_descent(p, stop_time, consider=None, seed=42):
    """Fast coordinate descent with analytical 1D minimization"""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p, math.inf
    p = p / s  # normalize so S=1
    
    c = _conv_self_fft(p)
    S = 1.0
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    flat = 0
    sweep = 0
    Ltot = 2 * n - 1
    
    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for idx in order:
            if time.monotonic() > stop_time - 0.5:
                break
            i = int(idx)
            
            if S > 1.3 or S < 0.7:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0
            
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            
            base = c[start:end]  # (Ls,)
            infl = 2.0 * p[:Ls]  # (Ls,)
            diag_idx = i - start  # index of diagonal in affected range
            
            # max of unaffected entries
            max_const = 0.0
            if start > 0:
                max_const = max(max_const, float(c[:start].max()))
            if end < Ltot:
                max_const = max(max_const, float(c[end:].max()))
            
            # Analytical candidate points
            # For each j in [0, Ls): objective_j(w) = (base[j] + w*infl[j] + (w^2 if j==diag_idx else 0)) / (S+w)^2
            # Plus: max_const / (S+w)^2
            # Optimal w for each: 
            #   linear case: w = S - 2*base[j]/infl[j] (if infl[j] != 0)
            #   quadratic (diag): w = (2*base[di] - infl[di]*S) / (2*S - infl[di])
            #   max_const: w = -S (invalid, S+w>0 needed)
            
            # We'll evaluate at a small set of candidate w values
            # Plus a coarse grid for safety
            
            x_hi = max(30.0 * v, 40.0 / n, 0.01)
            x_hi = min(x_hi, 4.0 * S, 1000.0 / n)
            if x_hi <= 0:
                x_hi = 1.0 / n
            
            # Grid + analytical points
            n_grid = min(32, max(16, int(200000 / max(n, 1)) + 16))
            grid = np.linspace(0.0, x_hi, n_grid + 1)
            
            # Add analytical candidates
            with np.errstate(divide='ignore', invalid='ignore'):
                # For linear parts (j != diag_idx)
                lin_w = S - 2.0 * (base / infl[np.abs(infl) > 1e-300])
                lin_w = lin_w[np.abs(infl) > 1e-300]
                # Filter to [0, x_hi]
                lin_w = lin_w[(lin_w >= 0) & (lin_w <= x_hi)]
                
                # For diagonal
                if 0 <= diag_idx < Ls and abs(infl[diag_idx]) > 1e-300:
                    denom = 2.0 * S - infl[diag_idx]
                    if abs(denom) > 1e-300:
                        diag_w = (2.0 * base[diag_idx] - infl[diag_idx] * S) / denom
                        if 0 <= diag_w <= x_hi:
                            lin_w = np.append(lin_w, diag_w)
            
            if len(lin_w) > 0:
                lin_w = lin_w[lip if False else np.isfinite(lin_w)]
                if len(lin_w) > 20:
                    lin_w = lin_w[:20]  # cap
                grid = np.unique(np.concatenate([grid, lin_w]))
            grid = np.unique(np.concatenate([grid, [v]]))  # include current
            
            # Evaluate all candidates
            w = grid - v  # displacement
            # Affected: base + w*infl + (w^2 at diag)
            aff = base[None, :] + w[:, None] * infl[None, :]
            aff[:, diag_idx] += w * w
            aff = np.maximum(aff, max_const)
            mv = np.max(aff, axis=1)
            dv = S + w
            valid = dv > 1e-14
            if not np.any(valid):
                continue
            obj = np.full(w.shape, math.inf)
            obj[valid] = mv[valid] / (dv[valid] ** 2)
            
            # Only consider w where 0 <= v + w <= 1000 (bounds)
            obj[w < -v + 1e-14] = math.inf
            obj[(v + w) > 1000.0] = math.inf
            
            if not np.any(np.isfinite(obj)):
                continue
            imin = int(np.argmin(obj))
            bx = float(grid[imin])
            bv = float(obj[imin])
            
            # Fine search around best
            if len(grid) > 2:
                # Find spacing
                dw_typical = x_hi / n_grid
                lo = max(0.0, bx - 3.0 * dw_typical)
                hi = min(x_hi, bx + 3.0 * dw_typical)
                fine = np.linspace(lo, hi, 24)
                fine = np.unique(np.concatenate([fine, [bx, v]]))
                w2 = fine - v
                aff2 = base[None, :] + w2[:, None] * infl[None, :]
                aff2[:, diag_idx] += w2 * w2
                aff2 = np.maximum(aff2, max_const)
                mv2 = np.max(aff2, axis=1)
                dv2 = S + w2
                valid2 = dv2 > 1e-14
                obj2 = np.full(w2.shape, math.inf)
                if np.any(valid2):
                    obj2[valid2] = mv2[valid2] / (dv2[valid2] ** 2)
                obj2[w2 < -v + 1e-14] = math.inf
                obj2[(v + w2) > 1000.0] = math.inf
                if np.any(np.isfinite(obj2)):
                    im2 = int(np.argmin(obj2))
                    if obj2[im2] < bv:
                        bv = float(obj2[im2])
                        bx = float(fine[im2])
            
            if math.isfinite(bv) and bv < cur_score - 1e-14 and 0.0 <= bx <= 1000.0:
                w = bx - v
                if abs(w) > 1e-15:
                    p[i] = bx
                    S += w
                    c[start:end] += w * infl
                    c[start + diag_idx] += w * w
                    cur_score = 2.0 * n * bv
                    improved = True
        
        # Rescale
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break
        
        # Verify periodically
        if sweep % 8 == 0:
            c = _conv_self_fft(p)
        new_sc = 2.0 * n * float(c.max())
        if math.isfinite(new_sc) and new_sc < best_score - 1e-14:
            best_score = new_sc
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass
        
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 10.0:
                    # Perturb
                    n_pert = max(10, n // 100)
                    idxs = rng.choice(n, size=n_pert, replace=False)
                    p[idxs] *= np.exp(rng.normal(0, 0.05, n_pert))
                    p = np.maximum(p, 0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self_fft(p)
                    S = 1.0
                    cur_score = 2.0 * n * float(c.max())
                    flat = 0
                else:
                    break
        sweep += 1
    
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_self_fft(p)
    sc = 2.0 * n * float(c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score

                if math.isfinite(fval):
                    p_cur = _softmax(res.x)
                    sc_cur = _score_fft(p_cur)
                    if math.isfinite(sc_cur) and sc_cur < best_sc - 1e-15:
                        best_sc = sc_cur
                        best_p = p_cur.copy()
                        p = p_cur.copy()
                        u = _p_to_u(p)
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 4:
                            break
            if getattr(res, "status", 1) == 0:
                break

    p_out = best_p.copy()
    ss = float(p_out.sum())
    if ss > 1e-300:
        p_out /= ss
    return p_out


def _coord_descent(p, stop_time, consider=None, seed=42):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p, math.inf
    p = p / s

    c = _conv_fft(p)
    S = 1.0
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    flat = 0
    sweep = 0
    Ltot = 2 * n - 1

    # Precompute ranges
    starts = np.arange(n, dtype=np.int64)
    ends = np.minimum(starts + n, Ltot)
    lens = ends - starts
    diag_idxs = np.zeros(n, dtype=np.int64)  # always i - i = 0 in local frame? No.
    # Actually start = i, so diag position in local range = (2i) - i = i
    # But we need 2i < end = min(i+n, Ltot)
    # 2i < i+n => i < n (always true)
    # 2i < Ltot = 2n-1 => i < n-0.5 (always true for i < n)
    # So diag_idx_local = i for all i
    # Wait: the affected range is k in [i, min(i+n-1, 2n-2)]
    # The diagonal k=2i is at local index 2i - i = i
    # And i < Ls = min(i+n, 2n-1) - i
    # For i < n/2: Ls = n, and i < n ✓
    # For i >= n/2: Ls = 2n-1-i, and i < 2n-1-i iff i < n-0.5 ✓
    # So diag_local = i, and i < Ls always

    n_grid_base = max(20, min(40, int(180000 / max(n, 1)) + 20))

    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for idx in order:
            if time.monotonic() > stop_time - 0.5:
                break
            i = int(idx)
            if S <= 1e-300:
                break
            if S > 1.4 or S < 0.65:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0

            v = float(p[i])
            st = i
            en = min(i + n, Ltot)
            Ls = en - st
            if Ls <= 0:
                continue

            base = c[st:en]
            infl = 2.0 * p[:Ls]
            di = st  # local index of diagonal = i - st = 0? No!
            # local index for k=2i is: 2i - st = 2i - i = i
            # But we need i < Ls. Since st=i, local index of k is k - i.
            # k=2i => local index = i. And Ls = en - i. i < en - i iff 2i < en.
            # en = min(i+n, 2n-1). 2i < i+n iff i < n ✓. 2i < 2n-1 iff i < n-0.5 ✓.
            di = i  # local index within [st, en) of the diagonal term

            # Max of unaffected
            max_const = 0.0
            if st > 0:
                mc_l = float(c[:st].max())
                max_const = max(max_const, mc_l)
            if en < Ltot:
                mc_r = float(c[en:].max())
                max_const = max(max_const, mc_r)

            x_hi = max(40.0 * v, 60.0 / n, 0.005)
            x_hi = min(x_hi, 5.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = 1.0 / n

            # Build candidate grid
            grid = np.linspace(0.0, x_hi, n_grid_base + 1)

            # Add analytical optimum for each element (if infl != 0)
            nz = np.abs(infl) > 1e-12
            if np.any(nz):
                with np.errstate(divide='ignore', invalid='ignore'):
                    w_lin = S - 2.0 * base[nz] / infl[nz]
                w_lin = w_lin[np.isfinite(w_lin) & (w_lin >= 0) & (w_lin <= x_hi)]
                # Diagonal correction
                if 0 <= di < Ls:
                    with np.errstate(divide='ignore', invalid='ignore'):
                        denom_d = 2.0 * S - infl[di]
                        if abs(denom_d) > 1e-12:
                            w_d = (2.0 * base[di] - infl[di] * S) / denom_d
                            if np.isfinite(w_d) and 0 <= w_d <= x_hi:
                                w_lin = np.concatenate([w_lin, [w_d]])
                if w_lin.size > 0:
                    if w_lin.size > 12:
                        w_lin = w_lin[np.argsort(np.abs(w_lin - v))[:12]]
                    grid = np.unique(np.concatenate([grid, w_lin]))

            grid = np.unique(np.concatenate([grid, [v]]))
            grid = grid[(grid >= 0) & (grid <= x_hi)]

            if grid.size == 0:
                continue

            # Evaluate
            w = grid - v
            aff = base[None, :] + w[:, None] * infl[None, :]
            aff[:, di] += w * w
            aff = np.maximum(aff, max_const)
            mv = np.max(aff, axis=1)
            dv = S + w
            valid = dv > 1e-14
            obj = np.full(w.shape, math.inf)
            if np.any(valid):
                obj[valid] = mv[valid] / (dv[valid] * dv[valid])
            obj[w < -v] = math.inf

            if not np.any(np.isfinite(obj)):
                continue
            imin = int(np.argmin(obj))
            bx = float(grid[imin])
            bv = float(obj[imin])

            # Fine search
            if n_grid_base > 4:
                dw_typ = x_hi / n_grid_base
                lo = max(0.0, bx - 4.0 * dw_typ)
                hi = min(x_hi, bx + 4.0 * dw_typ)
                fine = np.linspace(lo, hi, 20)
                fine = np.unique(np.concatenate([fine, [bx, v]]))
                w2 = fine - v
                aff2 = base[None, :] + w2[:, None] * infl[None, :]
                aff2[:, di] += w2 * w2
                aff2 = np.maximum(aff2, max_const)
                mv2 = np.max(aff2, axis=1)
                dv2 = S + w2
                valid2 = dv2 > 1e-14
                obj2 = np.full(w2.shape, math.inf)
                if np.any(valid2):
                    obj2[valid2] = mv2[valid2] / (dv2[valid2] * dv2[valid2])
                obj2[w2 < -v] = math.inf
                if np.any(np.isfinite(obj2)):
                    im2 = int(np.argmin(obj2))
                    if obj2[im2] < bv:
                        bv = float(obj2[im2])
                        bx = float(fine[im2])

            if math.isfinite(bv) and bv < cur_score / (2.0 * n) - 1e-15 and bx >= 0.0:
                w_opt = bx - v
                if abs(w_opt) > 1e-15:
                    p[i] = bx
                    S += w_opt
                    c[st:en] += w_opt * infl
                    c[st + di] += w_opt * w_opt
                    cur_score = 2.0 * n * bv
                    improved = True

        # Rescale
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        # Periodic verification
        if sweep % 6 == 0 or not improved:
            c = _conv_fft(p)
        new_sc = 2.0 * n * float(c.max())
        if math.isfinite(new_sc) and new_sc < best_score - 1e-14:
            best_score = new_sc
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 12.0:
                    n_pert = max(8, n // 80)
                    idxs = rng.choice(n, size=n_pert, replace=False)
                    scale = np.exp(rng.normal(0, 0.04, n_pert))
                    p[idxs] *= scale
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_fft(p)
                    S = 1.0
                    cur_score = 2.0 * n * float(c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_fft(p)
    sc = 2.0 * n * float(c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _block_descent(p, stop_time, consider=None, seed=123):
    """Adjust pairs of adjacent coordinates together."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n < 2:
        return p, math.inf
    p = p / s
    c = _conv_fft(p)
    S = 1.0
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    Ltot = 2 * n - 1
    sweeps = 0

    while time.monotonic() < stop_time - 2.0:
        improved = False
        pairs = np.arange(n - 1)
        rng.shuffle(pairs)
        for pr in pairs[:n // 2]:
            if time.monotonic() > stop_time - 0.5:
                break
            i = int(pr)
            j = i + 1
            if S > 1.3 or S < 0.7:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0

            vi = float(p[i])
            vj = float(p[j])

            # Affected range: [i, i+n-1] U [j, j+n-1] = [i, min(j+n-1, 2n-2)]
            st = i
            en = min(j + n, Ltot)
            Ls = en - st
            if Ls <= 0:
                continue

            base = c[st:en].copy()

            # Change: p[i] += w, p[j] -= w (mass transfer, sum preserved)
            # Effect on c[k] for k in affected range:
            # From p[i]: 2*w*p[k-i] (cross) + w^2 (diag at k=2i)
            # From p[j]: -2*w*p[k-j] (cross) - w^2 (diag at k=2j)
            # Net: 2w*(p[k-i] - p[k-j]) + w^2*(delta_{k,2i} - delta_{k,2j})
            # For k in [i, i+n-1]: p[k-i] is valid
            # For k in [j, j+n-1]: p[k-j] is valid

            infl_i = np.zeros(Ls)
            infl_j = np.zeros(Ls)
            # i contributes to k in [i, min(i+n-1, Ltot-1)]
            ii_st = max(0, i - st)
            ii_en = min(Ls, i + n - st)
            infl_i[ii_st:ii_en] = 2.0 * p[:ii_en - ii_st]
            # j contributes (with negative sign) to k in [j, min(j+n-1, Ltot-1)]
            jj_st = max(0, j - st)
            jj_en = min(Ls, j + n - st)
            infl_j[jj_st:jj_en] = 2.0 * p[:jj_en - jj_st]
            net_infl = infl_i - infl_j

            # Diagonal corrections: at k=2i (local idx 2i-st) add w^2, at k=2j subtract w^2
            di_i = 2 * i - st
            di_j = 2 * j - st

            max_const = 0.0
            if st > 0:
                max_const = max(max_const, float(c[:st].max()))
            if en < Ltot:
                max_const = max(max_const, float(c[en:].max()))

            # 1D: minimize max(base + w*net_infl + w^2*(δ_di_i - δ_di_j), max_const) / (S+w-S+w)^2
            # Since sum is preserved, denom = S^2 (constant!)
            # So minimize max(base + w*net_infl + w^2*(δ_di_i - δ_di_j), max_const)
            # Subject to: 0 <= vi + w and 0 <= vj - w, i.e., -vi <= w <= vj

            w_lo = -vi
            w_hi = vj
            if w_hi - w_lo < 1e-14:
                continue

            n_pt = 32
            w_grid = np.linspace(w_lo, w_hi, n_pt)

            aff = base[None, :] + w_grid[:, None] * net_infl[None, :]
            if 0 <= di_i < Ls:
                aff[:, di_i] += w_grid * w_grid
            if 0 <= di_j < Ls:
                aff[:, di_j] -= w_grid * w_grid
            aff = np.maximum(aff, max_const)
            mv = np.max(aff, axis=1)
            # Also enforce non-negativity of p[i], p[j]
            # Already handled by w in [w_lo, w_hi]
            obj = mv / (S * S)  # since sum unchanged

            if not np.any(np.isfinite(obj)):
                continue
            imin = int(np.argmin(obj))
            bw = float(w_grid[imin])

            if obj[imin] < cur_score / (2.0 * n) - 1e-15 and abs(bw) > 1e-14:
                p[i] += bw
                p[j] -= bw
                c[st:en] += bw * net_infl
                if 0 <= di_i < Ls:
                    c[st + di_i] += bw * bw
                if 0 <= di_j < Ls:
                    c[st + di_j] -= bw * bw
                # S unchanged (mass transfer)
                cur_score = 2.0 * n * float(obj[imin]) * S * S / S * S
                cur_score = 2.0 * n * float(obj[imin])
                improved = True

        if sweep % 10 == 0 or not improved:
            c = _conv_fft(p)
        new_sc = 2.0 * n * float(c.max())
        if math.isfinite(new_sc) and new_sc < best_score - 1e-14:
            best_score = new_sc
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        sweeps += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_fft(p)
    sc = 2.0 * n * float(c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 15.0

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = math.inf
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_arr]))
    except Exception:
        pass
    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(128)
        best_score = float(evaluate_sequence([float(x) for x in best_arr]))

    n_best = int(best_arr.size)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            sc = float(evaluate_sequence([float(x) for x in a]))
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_arr = a.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 1: LSE optimization (shorter)
    lse_stop = t0 + min(120.0, 0.18 * budget)
    if lse_stop > t0 + 15.0:
        try:
            p1 = _lse_optimize(best_arr, lse_stop, consider=consider, fine_only=False)
            if p1 is not None and p1.size == n_best and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    # Phase 2: Main coordinate descent (bulk of time)
    coord_stop = hard - 80.0
    if coord_stop > time.monotonic() + 30.0:
        try:
            p2, _ = _coord_descent(best_arr, coord_stop, consider=consider,
                                   seed=int(seed + 7))
            if p2 is not None and p2.size == n_best and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p2 = p2 / float(p2.sum())
                consider(p2)
        except Exception:
            pass

    # Phase 3: LSE fine + coordinate polish
    rem = hard - 12.0 - time.monotonic()
    if rem > 25.0:
        lse2_stop = time.monotonic() + max(12.0, 0.30 * rem)
        try:
            p3 = _lse_optimize(best_arr, lse2_stop, consider=consider, fine_only=True)
            if p3 is not None and p3.size == n_best and np.all(np.isfinite(p3)) and p3.sum() > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)
        except Exception:
            pass

        # Phase 4: Block descent (pairs)
        blk_stop = hard - 30.0
        if time.monotonic() < blk_stop - 10.0:
            try:
                p4, _ = _block_descent(best_arr, blk_stop, consider=consider,
                                       seed=int(seed + 42))
                if p4 is not None and p4.size == n_best and np.all(np.isfinite(p4)) and p4.sum() > 0:
                    p4 = p4 / float(p4.sum())
                    consider(p4)
            except Exception:
                pass

        # Phase 5: Final coordinate polish
        coord2_stop = hard - 8.0
        if time.monotonic() < coord2_stop - 10.0:
            try:
                p5, _ = _coord_descent(best_arr, coord2_stop, consider=consider,
                                       seed=int(seed + 777))
                if p5 is not None and p5.size == n_best and np.all(np.isfinite(p5)) and p5.sum() > 0:
                    p5 = p5 / float(p5.sum())
                    consider(p5)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
