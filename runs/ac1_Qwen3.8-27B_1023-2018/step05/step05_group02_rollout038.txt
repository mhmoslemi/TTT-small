    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c

def _score_arr(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))

def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * a.size * m / (s * s))

def _build_conv_matrix(a):
    n = a.size
    i_idx = np.arange(n)[:, None]
    j_idx = np.arange(n)[None, :]
    rows = (i_idx + j_idx).ravel()
    cols = np.broadcast_to(j_idx, (n, n)).ravel()
    vals = np.broadcast_to(a[:, None], (n, n)).ravel()
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def _lp_solve(a, max_time=45.0):
    n = a.size
    c = _conv_self(a)
    M = float(c.max())
    if M <= 0 or not math.isfinite(M):
        return None
    M_scaled = M * 1.0
    A = _build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraint = A @ b_var <= M_scaled
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, [constraint])
    solvers_to_try = [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.CBC]
    for sol in solvers_to_try:
        try:
            t_start = time.monotonic()
            prob.solve(solver=sol, verbose=False)
            if time.monotonic() - t_start > max_time:
                continue
            if b_var.value is not None:
                bv = np.asarray(b_var.value, dtype=np.float64).ravel()
                bv = np.where(np.isfinite(bv), bv, 0.0)
                bv = np.maximum(bv, 0.0)
                if bv.sum() > 1e-10:
                    return bv
        except Exception:
            continue
    return None

def _lp_phase(a, stop_time, consider, max_iters=200):
    a = np.maximum(a.copy(), 0.0)
    s0 = float(a.sum())
    if s0 < 1e-10:
        return a
    a = a / s0
    best_a = a.copy()
    best_sc = _score_arr(a)
    t_blend = 0.6
    no_improve = 0
    failed = 0
    for it in range(max_iters):
        if time.monotonic() > stop_time - 3.0:
            break
        b = _lp_solve(a, max_time=35.0)
        if b is None:
            failed += 1
            if failed > 3:
                break
            continue
        failed = 0
        b_sum = float(b.sum())
        b_norm = b * (1.0 / b_sum)
        a_new = (1.0 - t_blend) * a + t_blend * b_norm
        a_new = np.maximum(a_new, 0.0)
        sn = float(a_new.sum())
        if sn < 1e-10:
            no_improve += 1
            continue
        a_new = a_new / sn
        sc_new = _score_arr(a_new)
        if math.isfinite(sc_new):
            if sc_new < best_sc - 1e-12:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new.copy()
                no_improve = 0
                consider(a_new)
                t_blend = max(0.3, t_blend * 0.97)
            else:
                a = a_new.copy()
                no_improve += 1
                t_blend = max(0.2, t_blend * 0.9)
                if no_improve >= 12:
                    break
    return best_a

def _softmax(u):
    m = float(np.max(u))
    e = np.exp(np.clip(u - m, -500, 500))
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size)
    return e / s

def _lse_optimize(p, stop_time, consider, fine_only=False):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0.0:
            return 1e100, np.zeros_like(uu)
        if T <= 0.0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -400.0, 400.0)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0
0.0)
        return float(f), grad

    def run_chunk(uu, T, end_t):
        try:
            f0, _ = lse_func(uu, T)
        except Exception:
            return uu
        if not math.isfinite(f0):
            return uu
        best_u = np.array(uu, dtype=np.float64, copy=True)
        best_f = float(f0)
        stale = 0
        while time.monotonic() < end_t - 1.0:
            maxit = 30 if n > 2000 else 50
            opts = {"maxiter": maxit, "gtol": 1e-12, "ftol": 1e-16,
                    "maxls": 30, "maxcor": 30, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == uu.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-15:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, "status", 1) == 0:
                break
        return best_u

    if fine_only:
        stages = ((0.001, 0.30), (0.0002, 0.40), (0.00005, 0.30))
    else:
        stages = ((0.10, 0.08), (0.03, 0.12), (0.01, 0.14), (0.003, 0.18),
                  (0.0008, 0.22), (0.0002, 0.26))

    start = time.monotonic()
    avail = max(0.0, stop_time - start - 5.0)
    if avail <= 3.0:
        return p
    best_p = p.copy()
    best_sc = _score_fft_safe(p)
    total_w = float(sum(w for _, w in stages))
    if total_w <= 0.0:
        total_w = 1.0
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        cum += float(wt)
        end_stage = min(stop_time - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 1.5:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        sc_cur = _score_fft_safe(p_cur)
        if not math.isfinite(sc_cur):
            p = best_p.copy()
            u = p_to_u(p)
            continue
        if sc_cur < best_sc:
            best_sc = sc_cur
            best_p = p_cur.copy()
            p = p_cur.copy()
        elif sc_cur > best_sc + max(0.002, 2e-4 * best_sc):
            p = best_p.copy()
        else:
            p = p_cur.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p

def _score_fft_safe(a):
    try:
        n = a.size
        s = float(a.sum())
        if s < 0.01:
            return math.inf
        c = _conv_self(a)
        m = float(c.max())
        if not math.isfinite(m):
            return math.inf
        return float(2.0 * n * m / (s * s))
    except Exception:
        return math.inf

def _coord_polish(p, stop_time, max_sweeps=100000, G=None, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    if G is None:
        if n <= 450:
            G = 50
        elif n <= 900:
            G = 38
        elif n <= 1800:
            G = 30
        else:
            G = 24
    G = int(max(12, G))

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.4:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.25 or S_total < 0.75:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls].copy()
            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr
            x_hi = max(20.0 * v, 60.0 * S / n, 0.05 * S, 1e-12)
            x_hi = min(x_hi, 10.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, S / n)
            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 <= v <= x_hi:
                pts = np.unique(np.concatenate((pts, [v])))

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                di = 0
                if di < Ls:
                    dval = base[di] + w * infl[di] + w * w
                    maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])
            if G > 1:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.5 * dw)
                hi = min(x_hi, best_x + 2.5 * dw)
                if hi <= lo:
                    hi = lo + 1e-14
                fine_pts = np.linspace(lo, hi, 40)
                if 0.0 <= best_x <= hi:
                    fine_pts = np.unique(np.concatenate((fine_pts, [best_x])))
                vals2 = eval_xs(fine_pts)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine_pts[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2
                gr = (math.sqrt(5.0) - 1.0) / 2.0
                g_lo = max(0.0, best_x - dw)
                g_hi = min(x_hi, best_x + dw)
                if g_hi - g_lo > 1e-16:
                    a_gs = g_hi - gr * (g_hi - g_lo)
                    b_gs = g_lo + gr * (g_hi - g_lo)
                    fa = float(eval_xs(np.array([a_gs]))[0])
                    fb = float(eval_xs(np.array([b_gs]))[0])
                    for _ in range(14):
                        if not math.isfinite(fa):
                            fa = 1e100
                        if not math.isfinite(fb):
                            fb = 1e100
                        if fa < fb:
                            g_hi = b_gs
                            b_gs = a_gs
                            fb = fa
                            a_gs = g_hi - gr * (g_hi - g_lo)
                            fa = float(eval_xs(np.array([a_gs]))[0])
                        else:
                            g_lo = a_gs
                            a_gs = b_gs
                            fa = fb
                            b_gs = g_lo + gr * (g_hi - g_lo)
                            fb = float(eval_xs(np.array([b_gs]))[0])
                        if g_hi - g_lo < 1e-16:
                            break
                    gs_x = (g_lo + g_hi) / 2.0
                    gv = float(eval_xs(np.array([gs_x]))[0])
                    if math.isfinite(gv) and gv < best_val:
                        best_x = gs_x
                        best_val = gv
            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 10.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True
        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
