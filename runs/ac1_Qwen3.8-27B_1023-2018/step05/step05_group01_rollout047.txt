                        if stale >= 2:
                            best_u = best_u + rng.normal(0, 0.008, best_u.size)
                            continue
            if getattr(res, "status", 1) == 0:
                if time.monotonic() < end_stage - 1.0 and stale < 3:
                    best_u = best_u + rng.normal(0, 0.004, best_u.size)
                    f_chk, _ = lse_func(best_u, T)
                    if math.isfinite(f_chk):
                        if f_chk < best_f:
                            best_f = f_chk
                        stale += 1
                else:
                    break

        p_cur = _softmax_fn(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif math.isfinite(sc_cur) and sc_cur < best_sc + 0.005:
            p = p_cur.copy()
        else:
            p = best_p.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, G=None, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    if G is None:
        if n <= 450:
            G = 56
        elif n <= 900:
            G = 44
        elif n <= 2000:
            G = 36
        elif n <= 4096:
            G = 28
        else:
            G = 24
    G = int(max(16, G))

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < 300000 and time.monotonic() < stop_time - 1.0:
        improved = False
        # Peak-aware ordering: coordinates contributing most to the peak go first
        peak_pos = int(np.argmax(c))
        if peak_pos >= n:
            peak_pos = 2 * n - 2 - peak_pos
        # Coordinates that affect position peak_pos are indices [peak_pos-n+1, peak_pos]
        lo_idx = max(0, peak_pos - n + 1)
        hi_idx = min(n - 1, peak_pos)
        affected = np.arange(lo_idx, hi_idx + 1)
        if affected.size > 0:
            order = np.concatenate([affected, rng.permutation(n)[~np.isin(
                np.arange(n), affected)] if n <= 5000 else rng.permutation(n)])
        else:
            order = rng.permutation(n)

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(20.0 * v, 40.0 * S / float(n), 0.02 * S, 1e-14)
            x_hi = min(x_hi, 8.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, S / float(n))

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag_i = i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[valid, None] * infl[None, :]
                dval = base[diag_i] + w[valid] * infl[diag_i] + w[valid] * w[valid]
                aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                dv = denom[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Fine search
            if G > 4 and best_x > 0 and best_x < x_hi:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.5 * dw)
                hi = min(x_hi, best_x + 2.5 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 32, dtype=np.float64)
                if 0.0 <= best_x <= x_hi:
                    fine = np.concatenate((fine, [best_x]))
                    fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val - 1e-15:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 10.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 8.0:
                    # Smarter perturbation: focus on peak-adjacent coordinates
                    sigma = 0.015 + 0.015 * min(flat, 12) / 12.0
                    pert = p * np.exp(rng.normal(0.0, sigma, n))
                    # Extra perturbation on peak-adjacent coords
                    peak_pos = int(np.argmax(c))
                    if peak_pos >= n:
                        peak_pos = 2 * n - 2 - peak_pos
                    lo_idx = max(0, peak_pos - n + 1)
                    hi_idx = min(n - 1, peak_pos)
                    pert[lo_idx:hi_idx+1] *= np.exp(rng.normal(0.0, sigma * 1.5, hi_idx - lo_idx + 1))
                    p = np.maximum(pert, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    else:
                        break
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    flat = 0
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        sc = float(_score_fft(p))
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _fine_polish(p, stop_time, seed=999, consider=None):
    """Very fine coordinate descent with larger G for the best candidate."""
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s
    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    # Use finer grid for near-converged optimization
    G_fine = 64 if n <= 2000 else 48

    for outer in range(50):
        if time.monotonic() > stop_time - 1.0:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > stop_time - 0.5:
                break
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]
            max_const = 0.0
            if start > 0:
                max_const = max(max_const, float(c[:start].max()))
            if end < Ltot:
                max_const = max(max_const, float(c[end:].max()))

            # Narrow search around current value
            v_lo = max(0.0, v * 0.7 - S / n * 0.5)
            v_hi = v * 1.3 + S / n * 0.5
            v_hi = min(v_hi, 4.0 * S / n)
            if v_hi <= v_lo:
                v_hi = v_lo + max(1e-10, v * 0.1)

            pts = np.linspace(v_lo, v_hi, G_fine + 1, dtype=np.float64)
            if 0.0 < v < v_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag_i = i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[valid, None] * infl[None, :]
                dval = base[diag_i] + w[valid] * infl[diag_i] + w[valid] * w[valid]
                aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                dv = denom[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if math.isfinite(best_val) and best_val < cur_score - 1e-14:
                w = best_x - v
                if abs(w) > 1e-16:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass
        if not improved:
            break

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        sc = float(_score_fft(p))
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 2.0:
        hard = t0 + max(2.0, budget * 0.9)
    final_reserve = max(10.0, 0.05 * (hard - t0))
    phase_limit = hard - final_reserve

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    def make_asym_init(src, n, sd):
        p = _resample_mid(src, n)
        rng = np.random.default_rng(sd)
        x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
        side = 1.0 if rng.random() < 0.5 else -1.0
        amp = 0.3 + 1.2 * float(rng.random())
        curv = 0.5 + 2.0 * float(rng.random())
        q = p * np.exp(side * amp * (2.0 * x - 1.0) + curv * (2.0 * x - 1.0) ** 2)
        q = np.maximum(q, 0.0)
        s = float(q.sum())
        if s > 0:
            q /= s
        else:
            q = p
        return q

    def make_geometric_init(n, sd):
        rng = np.random.default_rng(sd)
        r = 0.92 + 0.06 * float(rng.random())
        x = np.arange(n, dtype=np.float64)
        if rng.random() < 0.5:
            a = r ** x
        else:
            a = r ** (n - 1 - x)
        s = float(a.sum())
        if s > 0:
            a /= s
        return a

    def run_full_opt(n, init_arr, end_time, sd, do_lse=True, do_coord=True):
        nonlocal best_arr, best_score
        p = _resample_mid(init_arr, n)
        if p.size != n or float(p.sum()) <= 0:
            p = np.full(n, 1.0 / n)
        p = p / float(p.sum())
        consider(p)

        if time.monotonic() > end_time - 10.0:
            return
        phase_len = max(0.0, float(end_time) - time.monotonic())

        if do_lse and phase_len > 30.0:
            lse_stop = min(float(end_time) - max(20.0, 0.3 * phase_len),
                           time.monotonic() + 0.60 * phase_len)
            if lse_stop > time.monotonic() + 10.0:
                try:
                    p2 = _lse_optimize(p, lse_stop, consider=consider, seed=sd)
                    if p2 is not None and p2.size == n and float(p2.sum()) > 0:
                        p = p2 / float(p2.sum())
                        consider(p)
                except Exception:
                    pass

        rem = float(end_time) - time.monotonic()
        if do_coord and rem > 15.0:
            cstop = min(float(end_time) - 1.5, time.monotonic() + rem - 2.0)
            if cstop > time.monotonic() + 10.0:
                try:
                    p3, _ = _coord_polish(p, cstop, seed=sd, consider=consider)
                    if p3 is not None and p3.size == n and float(p3.sum()) > 0:
                        p = p3 / float(p3.sum())
                        consider(p)
                except Exception:
                    pass
        return p

    n_parent = int(best_arr.size)

    # Phase 1: Optimize at parent length (main effort - 50%)
    p1_end = min(phase_limit, t0 + 0.50 * (phase_limit - t0))
    p_opt = run_full_opt(n_parent, best_arr, p1_end, int(seed + 100), do_lse=True, do_coord=True)

    # Phase 2: Perturb and re-optimize (20%)
    if time.monotonic() < phase_limit - 40.0:
        rng2 = np.random.default_rng(seed + 777)
        for pr in range(3):
            if time.monotonic() > phase_limit - 25.0:
                break
            pert = best_arr * np.exp(rng2.normal(0, 0.05, best_arr.size))
            pert = np.maximum(pert, 0.0)
            ps = float(pert.sum())
            if ps > 0:
                pert /= ps
            sub_end = min(phase_limit - 20.0, time.monotonic() + 0.35 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 20.0:
                run_full_opt(n_parent, pert, sub_end, int(seed + 2000 + pr * 100),
                             do_lse=True, do_coord=True)

    # Phase 3: Different initializations (20%)
    if time.monotonic() < phase_limit - 30.0:
        for init_seed in range(3):
            if time.monotonic() > phase_limit - 20.0:
                break
            init = make_asym_init(best_arr, n_parent, int(seed + 5000 + init_seed * 777))
            sub_end = min(phase_limit - 15.0, time.monotonic() + 0.2 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 20.0:
                run_full_opt(n_parent, init, sub_end, int(seed + 6000 + init_seed * 333),
                             do_lse=True, do_coord=True)

    # Phase 4: Try geometric init
    if time.monotonic() < phase_limit - 30.0:
        geo = make_geometric_init(n_parent, int(seed + 9000))
        sub_end = min(phase_limit - 10.0, time.monotonic() + 0.15 * (phase_limit - time.monotonic()))
        if sub_end > time.monotonic() + 15.0:
            run_full_opt(n_parent, geo, sub_end, int(seed + 9500),
                         do_lse=True, do_coord=True)

    # Phase 5: Final squeeze with fine polish
    final_end = min(hard - 12.0, t0 + budget - 8.0)
    if final_end > time.monotonic() + 20.0 and best_arr.size > 0:
        nbest = int(best_arr.size)
        rem = final_end - time.monotonic()

        # LSE polish on best
        if rem > 40.0 and nbest <= 5000:
            lstop = min(final_end - max(30.0, 0.3 * rem), time.monotonic() + 0.35 * rem)
            if lstop > time.monotonic() + 10.0:
                try:
                    p2 = _lse_optimize(best_arr, lstop, consider=consider, seed=int(seed + 99999))
                    if p2 is not None and p2.size == nbest and float(p2.sum()) > 0:
                        p2 = p2 / float(p2.sum())
                        consider(p2)
                except Exception:
                    pass

        # Standard coord polish
        cstop = min(final_end - max(15.0, 0.15 * (final_end - time.monotonic())),
                    time.monotonic() + 0.45 * (final_end - time.monotonic()))
        if cstop > time.monotonic() + 10.0:
            try:
                p3, _ = _coord_polish(best_arr, cstop, seed=int(seed + 123456789), consider=consider)
                if p3 is not None and p3.size == nbest and float(p3.sum()) > 0:
                    p3 = p3 / float(p3.sum())
                    consider(p3)
            except Exception:
                pass

        # Fine polish (very narrow search around optimum)
        fstop = min(hard - 15.0, final_end - 3.0)
        if fstop > time.monotonic() + 10.0:
            try:
                p4, _ = _fine_polish(best_arr, fstop, seed=int(seed + 777777777), consider=consider)
                if p4 is not None and p4.size == nbest and float(p4.sum()) > 0:
                    p4 = p4 / float(p4.sum())
                    consider(p4)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
