import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(64, dtype=np.float64)
    if p.size == 0:
        p = np.ones(64, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(64, dtype=np.float64)
    return p

def _conv_fft(a):
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fft(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_fft(a)
    m = float(c.max()) if c.size else 0.0
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _softmax(u):
    m = np.max(u)
    e = np.exp(u - m)
    s = e.sum()
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0/u.size)
    return e / s

def _lse_lbfgs(p, stop_time, consider=None, stages=None):
    """LSE-smoothed L-BFGS optimization."""
    p = np.maximum(p.astype(np.float64), 0.0)
    n = p.size
    s = p.sum()
    if s < 1e-300:
        return p
    p = p / s
    
    def p2u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - uu.mean()
    
    u = p2u(p)
    conv_size = next_fast_len(2*n - 1)
    conv_len = 2*n - 1
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F*F, conv_size)[:conv_len]
        np.maximum(c, 0, out=c)
        return c
    
    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = c.max()
        if T <= 0:
            T = 1e-4 * cmax
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = exps.sum()
        if sum_exp <= 0 or not math.isfinite(sum_exp):
            w = np.zeros(c.size)
            w[c.argmax()] = 1.0
            f = cmax
        else:
            w = exps / sum_exp
            f = cmax + T * np.log(sum_exp / c.size)
        # Gradient
        F_a = rfft(pp, conv_size)
        F_w = rfft(w, conv_size)
        # cross-conv: reverse(pp) * w
        cross_size = next_fast_len(conv_len)
        F_rev = rfft(pp[::-1], cross_size)
        F_w2 = rfft(w, cross_size)
        cc = irfft(F_rev * F_w2, cross_size)[:conv_len]
        q = cc[n-1:2*n-1]
        np.maximum(q, 0, out=q)
        gdp = 2.0 * q
        tdp = float(pp @ gdp)
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad
    
    if stages is None:
        stages = [(0.05, 0.15), (0.01, 0.20), (0.003, 0.25), (0.001, 0.25), (0.0003, 0.15)]
    
    best_u = u.copy()
    best_f = 1e100
    total_w = sum(w for _, w in stages)
    start = time.monotonic()
    avail = stop_time - start - 5.0
    if avail <= 3:
        return p
    
    cum = 0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3:
            break
        cum += wt
        end_stage = min(stop_time - 2, start + avail * cum / total_w)
        if end_stage <= now + 1:
            continue
        c = cfast(_softmax(u))
        cmax = c.max()
        T = frac * cmax
        
        best_u_local = u.copy()
        best_f_local = 1e100
        try:
            f0, _ = lse_func(u, T)
            best_f_local = f0
        except:
            continue
        
        stale = 0
        while time.monotonic() < end_stage - 1:
            try:
                res = minimize(lse_func, best_u_local, args=(T,),
                             method='L-BFGS-B', jac=True,
                             options={'maxiter': 30, 'gtol': 1e-12, 'maxls': 20, 'disp': False})
            except:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                f = res.fun
                if math.isfinite(f) and f < best_f_local - 1e-15:
                    best_f_local = f
                    best_u_local = res.x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if res.status == 0:
                break
        
        p_cur = _softmax(best_u_local)
        p_cur = np.maximum(p_cur, 0)
        sc = _score_fft(p_cur)
        if math.isfinite(sc):
            consider(p_cur) if consider else None
        u = best_u_local.copy()
    
    p_out = _softmax(u)
    p_out = np.maximum(p_out, 0)
    ss = p_out.sum()
    if ss > 1e-300:
        p_out /= ss
    return p_out

def _coord_descent(p, stop_time, seed=42, consider=None, max_sweeps=50000):
    n = int(p.size)
    p = np.maximum(p.astype(np.float64).copy(), 0.0)
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p /= s
    S = 1.0
    
    Ltot = 2 * n - 1
    c = _conv_fft(p)
    cur_score = 2.0 * n * float(c.max())
    best_p = p.copy()
    best_score = cur_score
    rng = np.random.default_rng(seed)
    
    if n <= 512:
        G = 48
    elif n <= 2048:
        G = 36
    else:
        G = 28
    
    sweep = 0
    flat = 0
    
    while sweep < max_sweeps and time.monotonic() < stop_time - 2:
        improved = False
        order = rng.permutation(n)
        
        for ii in order:
            i = int(ii)
            if time.monotonic() > stop_time - 1:
                break
            
            v = p[i]
            si = i
            ei = min(i + n, Ltot)
            Ls = ei - si
            if Ls <= 0:
                continue
            
            base = c[si:ei]
            infl = 2.0 * p[:Ls]
            
            max_const = 0.0
            if si > 0:
                mc = float(c[:si].max())
                if mc > max_const: max_const = mc
            if ei < Ltot:
                mc = float(c[ei:].max())
                if mc > max_const: max_const = mc
            
            avg = S / n
            x_hi = max(25.0 * v, 80.0 * avg, 0.04 * S, 1e-10)
            x_hi = min(x_hi, 10.0 * S)
            
            pts = np.linspace(0.0, x_hi, G + 1)
            if 0 <= v <= x_hi:
                pts = np.concatenate([pts, [v]])
                pts = np.unique(pts)
            
            w = pts - v
            denom = S + w
            valid = denom > 1e-15
            
            aff = base[None, :] + w[:, None] * infl[None, :]
            maff = np.max(aff, axis=1)
            diag = i - si
            dval = base[diag] + w * infl[diag] + w * w
            maff = np.maximum(maff, dval)
            maff = np.maximum(maff, max_const)
            
            vals = np.full(pts.size, math.inf)
            vals[valid] = (2.0 * n * maff[valid]) / (denom[valid] ** 2)
            
            imin = int(np.argmin(vals))
            best_x = pts[imin]
            best_val = vals[imin]
            
            # Parabolic refinement
            if 0 < imin < len(pts) - 1 and math.isfinite(best_val):
                try:
                    y0, y1, y2 = vals[imin-1], vals[imin], vals[imin+1]
                    if all(math.isfinite(y) for y in [y0, y1, y2]):
                        denom_p = 2.0 * (2.0*y1 - y0 - y2)
                        if abs(denom_p) > 1e-30:
                            step = pts[1] - pts[0]
                            x_par = best_x + 0.5 * denom_p * (y0 - y2) / step
                            x_par = max(0.0, min(x_hi, x_par))
                            wp = x_par - v
                            dp = S + wp
                            if dp > 1e-15:
                                affp = base + wp * infl
                                mp = max(float(np.max(affp)), float(base[diag] + wp*infl[diag] + wp*wp), max_const)
                                vp = 2.0 * n * mp / (dp * dp)
                                if vp < best_val:
                                    best_x = x_par
                                    best_val = vp
                except:
                    pass
            
            if math.isfinite(best_val) and best_val < cur_score - 1e-14:
                dw = best_x - v
                if abs(dw) > 1e-15:
                    p[i] = best_x
                    S += dw
                    c[si:ei] += dw * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += dw * dw
                    cur_score = best_val
                    improved = True
        
        # Periodic renormalization and recompute
        if S > 1.05 or S < 0.95:
            inv = 1.0 / S if S > 1e-300 else 1.0
            p *= inv
            c = _conv_fft(p)
            S = 1.0
        
        new_score = 2.0 * n * float(c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
            if consider:
                try: consider(best_p)
                except: pass
        
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 5:
                    sigma = 0.03 / math.sqrt(n)
                    p = p * np.exp(rng.normal(0, sigma, n))
                    p = np.maximum(p, 0)
                    ss = p.sum()
                    if ss > 1e-300:
                        p /= ss
                        S = 1.0
                    c = _conv_fft(p)
                    cur_score = 2.0 * n * float(c.max())
                    flat = 0
                else:
                    break
        
        sweep += 1
    
    return best_p, best_score

def propose_candidate(seed=42, budget_s=1300, **kwargs):

    parent = _get_parent()

        best_score = float(_score_fft(best_arr))

        best_score = math.inf

            return

            return

            return
