            # Actually h_j = sum_k w_k * a[k-j] (with bounds)
            # This is a correlation: h = np.correlate(w, a, mode='full')... no
            # h_j = sum_k w_k * a[k-j] for k in 0..2n-2, j in 0..n-1
            # = convolution of w (size 2n-1) with reversed a (size n)
            # Let me just compute it directly for n=281
            h = np.zeros(n)
            for j in range(n):
                # h_j = sum_k w_k * a[k-j] where 0<=k-j<n i.e. j<=k<=j+n-1
                k_lo = j
                k_hi = min(j+n, 2*n-1)
                if k_lo < k_hi:
                    h[j] = np.dot(w[k_lo:k_hi], a[0:k_hi-k_lo])
            # d lse / d a_j = h_j
            # d lse / d y_l = sum_j h_j * a_j * (delta_jl - a_l)
            # = h_l * a_l - a_l * sum_j h_j * a_j
            grad_a = h  # d lse / d a_j
            grad_y = a * (grad_a - np.dot(a, grad_a))
            return grad_y
        
        try:
            res_opt = minimize(obj, np.log(ab + 1e-10), jac=obj_grad, method='L-BFGS-B',
                             options={'maxiter': 500, 'maxfun': 2000, 'ftol': 1e-15, 'gtol': 1e-12})
            if res_opt.success or res_opt.fun < obj(np.log(ab + 1e-10)):
                a_new = np.exp(res_opt.x)
                a_new /= a_new.sum()
                c_new = np.max(np.convolve(a_new, a_new))
                if c_new < best_M:
                    best_M = c_new
                    best_a = a_new
        except:
            pass
    
    # Final validation
    final_score = float(evaluate_sequence([float(v) for v in best_a]))
    if not np.isfinite(final_score):
        return [1.0] * 100
    
    return [float(v) for v in best_a]

import numpy as np
import time
from scipy.optimize import linprog

def _make_a_matrix(n, a):
    """Build (2n-1) x n convolution matrix: A[k,j] = a[k-j] if 0<=k-j<n."""
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    d = k_idx - j_idx
    mask = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[d[mask]]
    return A

def _score_normalized(M, n):
    """Score = 2n * M (when sum(a)=1)."""
    return 2.0 * n * M

def _lp_solve(A, M, n, tl=3.0, eps=1e-12):
    """Solve: max sum(b) s.t. A@b <= M*(1+eps), b >= 0."""
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M * (1.0 + eps))
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tl,
                               'dual_feasibility_tolerance': 1e-13,
                               'primal_feasibility_tolerance': 1e-13})
        if res.status == 0 and res.x is not None:
            return np.clip(res.x, 0.0, None)
    except Exception:
        pass
    return None

def _line_search(a, b, M_init):
    """Line search: minimize max(conv((1-t)a+tb, (1-t)a+tb)) for t in [0,1]."""
    best_t, best_val = 0.0, M_init
    # Coarse
    ts = np.array([0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4,
                   0.5, 0.6, 0.7, 0.8, 0.9, 1.0])
    for t in ts:
        cand = (1.0 - t) * a + t * b
        val = float(np.max(np.convolve(cand, cand)))
        if val < best_val - 1e-16:
            best_val, best_t = val, t
    if best_t == 0.0:
        return None, M_init
    # Fine (bisection)
    idx = np.argmin(np.abs(ts - best_t))
    lo = max(0.0, ts[max(0, idx - 1)])
    hi = min(1.0, ts[min(len(ts) - 1, idx + 1)])
    for _ in range(60):
        if hi - lo < 1e-12:
            break
        mid = 0.5 * (lo + hi)
        cand = (1.0 - mid) * a + mid * b
        val = float(np.max(np.convolve(cand, cand)))
        if val < best_val - 1e-16:
            best_val, best_t = val, mid
            hi = mid
        else:
            lo = mid
    cand = (1.0 - best_t) * a + best_t * b
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-14:
        return None, M_init
    cand /= s
    return cand, best_val

def _lp_optimize(a0, deadline, rng, max_iter=50000):
    """Main LP-based optimization loop."""
    a = np.clip(a0.copy(), 0, None)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    n = a.size
    best_a = a.copy()
    best_M = float(np.max(np.convolve(a, a)))
    stuck = 0
    polyak_a = a.copy()
    num_avg = 0
    
    for iteration in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        conv_a = np.convolve(a, a)
        M = float(np.max(conv_a))
        
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Update Polyak average
        num_avg += 1
        alpha_p = 2.0 / (num_avg + 2.0)
        polyak_a = (1.0 - alpha_p) * polyak_a + alpha_p * a
        # Check if Polyak average is better
        conv_p = np.max(np.convolve(polyak_a, polyak_a))
        if conv_p < best_M - 1e-15:
            best_M = float(conv_p)
            best_a = polyak_a.copy()
        
        # Escape mechanism
        if stuck > 200:
            sigma = min(0.015 + 0.005 * (stuck // 200), 0.25)
            mode = (iteration // 200) % 4
            if mode == 0:
                a *= np.exp(sigma * rng.normal(0, 1, n))
            elif mode == 1:
                xx = np.arange(n) / n
                a *= 1.0 + sigma * np.sin(2 * np.pi * rng.uniform(1, 6) * xx + rng.uniform(0, 6.28))
            elif mode == 2:
                a *= 1.0 + sigma * rng.uniform(-1, 1, n)
            else:
                half = n // 2
                a[half:] *= np.exp(sigma * rng.normal(0, 1, n - half))
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            stuck = 0
            if sigma > 0.2:
                # Reset Polyak
                polyak_a = a.copy()
                num_avg = 0
                if iteration > 50000:
                    break
            continue
        
        # Build LP matrix
        A = _make_a_matrix(n, a)
        moved = False
        
        # Primary LP
        b = _lp_solve(A, M, n, tl=2.5, eps=1e-12)
        if b is not None and b.sum() > 1e-12:
            b /= b.sum()
            if np.max(np.abs(b - a)) > 1e-10:
                cand, val = _line_search(a, b, M)
                if cand is not None and val < M - 1e-14:
                    a = cand
                    moved = True
        
        # Relaxed LP if primary didn't move
        if not moved:
            b2 = _lp_solve(A, M, n, tl=2.0, eps=1e-8)
            if b2 is not None and b2.sum() > 1e-12:
                b2 /= b2.sum()
                if np.max(np.abs(b2 - a)) > 1e-10:
                    for t_try in [0.01, 0.02, 0.05, 0.1]:
                        cand = (1.0 - t_try) * a + t_try * b2
                        cand = np.maximum(cand, 0)
                        cs = cand.sum()
                        if cs < 1e-14:
                            continue
                        cand /= cs
                        val = float(np.max(np.convolve(cand, cand)))
                        if val < M - 1e-14:
                            a = cand
                            moved = True
                            break
        
        # Subgradient step
        if not moved:
            active = np.where(conv_a >= M * 0.9998)[0]
            if len(active) > 0:
                grad = np.zeros(n)
                for k_s in active[:8]:  # Limit active set for speed
                    lo = max(0, int(k_s) - n + 1)
                    hi = min(n, int(k_s) + 1)
                    for j in range(lo, hi):
                        idx = int(k_s) - j
                        if 0 <= idx < n:
                            grad[j] += a[idx]
                grad -= grad.mean()
                gn = float(np.linalg.norm(grad))
                if gn > 1e-14:
                    for alpha in [0.02, 0.01, 0.005, 0.002, 0.001]:
                        cand = a - alpha * grad / gn
                        cand = np.maximum(cand, 0.0)
                        cs = cand.sum()
                        if cs < 1e-14:
                            continue
                        cand /= cs
                        val = float(np.max(np.convolve(cand, cand)))
                        if val < M - 1e-14:
                            a = cand
                            moved = True
                            break
        
        if not moved:
            # Tiny perturbation to escape exact fixed point
            a *= np.exp(0.003 * rng.normal(0, 1, n))
            a = np.maximum(a, 1e-15)
            a /= a.sum()
    
    return best_a, best_M

def _smooth_polish(a0, deadline):
    """Polish using smooth log-sum-exp surrogate with L-BFGS-B."""
    n = a0.size
    a = a0.copy()
    
    beta = 50.0
    max_c_init = float(np.max(np.convolve(a, a)))
    
    def _obj(y):
        a_try = np.exp(y - np.max(y))  # numerical stability
        a_try /= a_try.sum()
        c = np.convolve(a_try, a_try)
        mc = np.max(c)
        if mc < 1e-30:
            return 1e-30
        lse = mc + (1.0 / beta) * np.log(np.sum(np.exp(beta * (c - mc)))) / beta
        return float(lse)
    
    def _obj_grad(y):
        a_try = np.exp(y - np.max(y))
        a_try /= a_try.sum()
        c = np.convolve(a_try, a_try)
        mc = np.max(c)
        w = np.exp(beta * (c - mc))
        w /= w.sum()
        # h_j = sum_k w_k * a[k-j] for valid range
        h = np.zeros(n)
        for j in range(n):
            k_lo = j
            k_hi = min(j + n, 2 * n - 1)
            if k_lo < k_hi:
                h[j] = np.dot(w[k_lo:k_hi], a_try[:k_hi - k_lo])
        # d lse / d y_l = a_l * (h_l - sum_j h_j * a_j)
        sh = np.dot(h, a_try)
        return a_try * (h - sh)
    
    try:
        from scipy.optimize import minimize
        y0 = np.log(np.maximum(a, 1e-15))
        for trial in range(3):
            res = minimize(_obj, y0, jac=_obj_grad, method='L-BFGS-B',
                         options={'maxiter': 200, 'maxfun': 500,
                                  'ftol': 1e-16, 'gtol': 1e-12,
                                  'maxls': 50})
            y_new = res.x
            a_new = np.exp(y_new - np.max(y_new))
            a_new /= a_new.sum()
            M_new = float(np.max(np.convolve(a_new, a_new)))
            if M_new < max_c_init - 1e-16:
                max_c_init = M_new
                a = a_new.copy()
                y0 = y_new.copy()
            if time.monotonic() > deadline:
                break
            beta = min(beta * 2.0, 500.0)  # Sharpen
    except Exception:
        pass
    
    return a, max_c_init
