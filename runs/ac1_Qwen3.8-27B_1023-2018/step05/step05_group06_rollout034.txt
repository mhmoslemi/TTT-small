        p = p.copy()
        c = c.copy()
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            
            infl = 2.0 * p[:Ls].copy()
            diag_pos = 2 * i - start
            has_diag = (0 <= diag_pos < Ls)
            
            max_out = 0.0
            if start > 0:
                max_out = max(max_out, float(c[:start].max()))
            if end < Ltot:
                max_out = max(max_out, float(c[end:].max()))
            
            base = c[start:end].copy()
            
            # Find candidate w values analytically
            # g(w) = max(base + w*infl + w^2*delta, max_out)
            # f(w) = 2n * g(w) / (S+w)^2
            # For constraint j active: f_j(w) = 2n*(base[j]+w*infl[j]+w^2*dj)/(S+w)^2
            # df_j/dw = 0 => (infl[j]+2w*dj)(S+w) - 2(base[j]+w*infl[j]+w^2*dj) = 0
            # => w*(2*dj - infl[j]) = 2*base[j] - S*infl[j]
            
            # Get top active constraints
            base_max = float(base.max())
            # Sort indices by base value (descending)
            top_k = min(8, Ls)
            idx_order = np.argsort(base)[::-1][:top_k]
            
            candidates = [0.0, -v]  # no change, and set to 0
            
            for idx in idx_order:
                j = int(idx)
                bj = float(base[j])
                inj = float(infl[j])
                dj = 1.0 if (has_diag and j == diag_pos) else 0.0
                
                denom = 2.0 * dj - inj
                if abs(denom) > 1e-15:
                    w_cand = (2.0 * bj - S * inj) / denom
                else:
                    continue
                
                # Clamp to valid range
                w_cand = max(w_cand, -v + 1e-15)
                # Limit magnitude
                max_step = 5.0 * max(v, 1.0 / n) + 1.0 / n
                if w_cand > max_step:
                    w_cand = max_step
                if w_cand > 1.0:
                    w_cand = 1.0
                candidates.append(w_cand)
            
            # Also add a few fractional attempts of the best candidate
            # Evaluate all candidates
            best_w = 0.0
            best_f = 2.0 * n * max(base_max, max_out) / (S * S)  # current score
            
            for w in candidates:
                if abs(w) < 1e-15:
                    continue
                if S + w <= 1e-10:
                    continue
                new_base = base + w * infl
                if has_diag:
                    new_base[diag_pos] += w * w
                m_val = float(new_base.max())
                if max_out > m_val:
                    m_val = max_out
                f_val = 2.0 * n * m_val / ((S + w) * (S + w))
                if math.isfinite(f_val) and f_val < best_f - 1e-15:
                    best_f = f_val
                    best_w = w
            
            # Refine around best_w with a few extra points
            if abs(best_w) > 1e-12:
                for mult in [0.5, 0.75, 1.25, 1.5]:
                    w_try = best_w * mult
                    if w_try < -v + 1e-15:
                        w_try = -v + 1e-15
                    if S + w_try <= 1e-10:
                        continue
                    new_base = base + w_try * infl
                    if has_diag:
                        new_base[diag_pos] += w_try * w_try
                    m_val = float(new_base.max())
                    if max_out > m_val:
                        m_val = max_out
                    f_val = 2.0 * n * m_val / ((S + w_try) * (S + w_try))
                    if math.isfinite(f_val) and f_val < best_f - 1e-15:
                        best_f = f_val
                        best_w = w_try
            
            if abs(best_w) > 1e-13:
                p[i] += best_w
                S += best_w
                c[start:end] += best_w * infl
                if has_diag:
                    c[start + diag_pos] += best_w * best_w
                improved = True
        
        return p, c, S, improved
    
    def lp_based_update(p, c, S, n, Ltot):
        """LP-based direction update. Returns updated p, c, S and score."""
        # Normalize p to sum=1 for the LP
        p_norm = p / S
        
        c_norm = _fft_conv_self(p_norm)
        M = float(c_norm.max())
        
        if M <= 0 or not math.isfinite(M):
            return p, c, S, None
        
        # Solve LP
        g = _solve_lp_direction(p_norm, M)
        if g is None:
            return p, c, S, None
        
        g_sum = float(g.sum())
        if g_sum < 1e-10:
            return p, c, S, None
        
        # Rescale g to match normalization
        g = g * (1.0 / g_sum)  # now sum(g) = 1
        
        # Line search: find best t in [0,1] for h(t) = (1-t)*p_norm + t*g
        # score(t) = 2n * max((h*h)[k]) / (sum(h))^2
        # Since p_norm and g both sum to 1, h(t) also sums to 1
        # (h*h)[k] = (1-t)^2*c_norm[k] + t(1-t)*(f*g + g*f)[k] + t^2*(g*g)[k]
        
        fg = _fft_cross(p_norm, g)
        gg = _fft_conv_self(g)
        
        # (h*h)[k] = (1-t)^2 * c_norm[k] + t(1-t) * fg[k] + t(1-t) * fg[k] + t^2 * gg[k]
        # Wait: (h*h)[k] = (h*h)[k] where h = (1-t)f + tg
        # = (1-t)^2*(f*f)[k] + 2t(1-t)*(f*g)[k] + t^2*(g*g)[k]
        # Note: f*g = g*f for convolution (it's commutative)
        
        best_t = 0.0
        best_score_t = 2.0 * n * M  # current score (sum=1)
        
        # Coarse grid
        for t in np.linspace(0.0, 1.0, 40):
            h = (1-t)*c_norm + 2*t*(1-t)*fg + t*t*gg
            score_t = 2.0 * n * float(np.max(h))
            if score_t < best_score_t:
                best_score_t = score_t
                best_t = t
        
        # Fine grid around best_t
        if 0 < best_t < 1:
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.05)
            for t in np.linspace(lo, hi, 30):
                h = (1-t)*c_norm + 2*t*(1-t)*fg + t*t*gg
                score_t = 2.0 * n * float(np.max(h))
                if score_t < best_score_t:
                    best_score_t = score_t
                    best_t = t
        
        if best_t > 0.01:
            p_new = (1.0 - best_t) * p_norm + best_t * g
            p_new = np.maximum(p_new, 0.0)
            s_new = float(p_new.sum())
            if s_new > 1e-10:
                p_new = p_new / s_new
                c_new = _fft_conv_self(p_new)
                return p_new, c_new, 1.0, best_score_t
        
        return p, c, S, None
    
    # Main optimization loop
    sweep = 0
    flat_count = 0
    lp_interval = 30  # do LP every 30 sweeps
    
    while time.monotonic() < deadline - 60:
        # Coordinate descent phase
        cd_stop = min(deadline - 60, time.monotonic() + 40.0)
        sweeps_this_phase = 0
        
        while time.monotonic() < cd_stop:
            p, c, S, improved = coord_descent_sweep(p, c, S, n, Ltot)
            sweeps_this_phase += 1
            sweep += 1
            
            # Renormalize
            if S > 1.3 or S < 0.7:
                inv = 1.0 / S
                p = p * inv
                c = c * inv * inv
                S = 1.0
            
            # Check score periodically
            if sweep % 5 == 0:
                c = _fft_conv_self(p)
                cmax = float(c.max())
                S = float(p.sum())
                cur_score = 2.0 * n * cmax / (S * S)
                if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_p = (p / S).copy()
                    consider(best_p)
                
                if improved:
                    flat_count = 0
                else:
                    flat_count += 1
            
            if not improved:
                flat_count += 1
        
        # LP-based update periodically
        if sweep % lp_interval < sweeps_this_phase and time.monotonic() < deadline - 90:
            try:
                p, c, S, lp_score = lp_based_update(p, c, S, n, Ltot)
                if lp_score is not None:
                    S = float(p.sum())
                    if S > 1e-10:
                        p = p / S
                        c = _fft_conv_self(p)
                        cur_score = _fft_score(p)
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                            best_score = cur_score
                            best_p = p.copy()
                            consider(p)
            except:
                pass
        
        # Perturbation if stuck
        if flat_count >= 15 and time.monotonic() < deadline - 120:
            sigma = 0.03 + 0.02 * min(flat_count - 15, 10)
            p = p * np.exp(rng.normal(0.0, sigma, n))
            p = np.maximum(p, 0.0)
            S = float(p.sum())
            if S > 1e-10:
                p = p / S
                S = 1.0
                c = _fft_conv_self(p)
            flat_count = 0
        
        if flat_count >= 40:
            # Try a different perturbation
            if time.monotonic() < deadline - 90:
                # Random restart from best_p with perturbation
                p = best_p * (1.0 + 0.1 * rng.standard_normal(n))
                p = np.maximum(p, 0.0)
                S = float(p.sum())
                if S > 1e-10:
                    p = p / S
                    S = 1.0
                    c = _fft_conv_self(p)
                flat_count = 0
            else:
                break
    
    # Final coordinate descent polish
    polish_stop = deadline - 40.0
    if time.monotonic() < polish_stop:
        pass_count = 0
        while time.monotonic() < polish_stop and pass_count < 50:
            p, c, S, improved = coord_descent_sweep(p, c, S, n, Ltot)
            pass_count += 1
            if S > 1.3 or S < 0.7:
                inv = 1.0 / S
                p = p * inv
                c = c * inv * inv
                S = 1.0
            if pass_count % 3 == 0:
                c = _fft_conv_self(p)
                S = float(p.sum())
                cur_score = 2.0 * n * float(c.max()) / (S * S)
                if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_p = (p / S).copy()
            if not improved:
                if pass_count >= 10:
                    break
    
    # Final validation
    try:
        out = np.asarray(best_p, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(128, 1.0/128)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            try:
                pp = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
                pp = np.clip(pp, 0.0, 1000.0)
                ls = [float(x) for x in pp]
                final_sc = float(evaluate_sequence(ls))
            except:
                return [1.0]
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except:
        return [1.0]
