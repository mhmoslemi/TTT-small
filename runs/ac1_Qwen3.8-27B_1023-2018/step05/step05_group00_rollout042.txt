    n = a.size

            # With the convention that if k-i == i-k (both equal i), we still count both terms.
            # For k=2i, i>0: k-i=i (valid), i-k=-i (invalid). So dc[2i]/da[i] = a[i] + 0 = a[i].
            # But the actual derivative is 2a[i]!
            # The resolution: when k=2i, the term in the sum is a[i]*a[i] (j=i, k-j=i).
            # d/d(a[i]) [a[i]*a[i]] = 2a[i].
            # In our formula: a[k-i] = a[i] (from j=i term) + a[i-k] = a[-i] = 0 (from j=k-i term, but j=k-i=i, same as j=i, so it's not a separate term).
            # The issue is that j=i and j=k-i are the SAME index when k=2i. So we should only count it once.
            # But the derivative of a[i]^2 is 2a[i], not a[i].
            # I think the correct approach is:
            # dc[k]/da[i] = (a[k-i] if 0<=k-i<n) + (a[i-k] if 0<=i-k<n and i-k != k-i)
            # i.e., the second term is included only if k-i != i-k, i.e., k != 2i.
            # For k=2i: dc[2i]/da[i] = a[i] (from the first term only, since the "second" term is the same).
            # But that gives a[i], not 2a[i]. Contradiction!
            # 
            # Let me just compute directly: c[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
            # dc[k]/da[i] = sum_{j: ...} [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
            # = a[k-i] (if 0<=k-i<n and 0<=i<n) + a[k-i] (if 0<=k-i<n and 0<=k-i<n and k-i != i)
            # Wait, delta(k-j,i) means j=k-i. This is valid if 0<=k-i<n.
            # And delta(j,i) means j=i. This is valid if 0<=i<n and 0<=k-i<n.
            # So: dc[k]/da[i] = a[k-i]*[0<=k-i<n] + a[k-i]*[0<=k-i<n] = 2*a[k-i]*[0<=k-i<n]
            # when k != 2i (so j=i and j=k-i are different indices).
            # When k = 2i: j=i gives a[k-i]=a[i]. j=k-i=i gives a[j]=a[i].
            # But j=i and j=k-i are the SAME j! So the sum only has this term once.
            # dc[2i]/da[i] = a[i] + a[i] = 2a[i]? Or just a[i]?
            # The sum is over j. When k=2i, the valid j values include j=i (once).
            # For j=i: a[j]*a[k-j] = a[i]*a[i].
            # d/d(a[i]) of this single term = 2a[i].
            # In the formula: delta(j,i)*a[k-j] = 1*a[i] = a[i].
            # a[j]*delta(k-j,i) = a[i]*1 = a[i].
            # Sum = a[i] + a[i] = 2a[i]. ✓
            # So the formula dc[k]/da[i] = a[k-i]*[0<=k-i<n] + a[k-i]*[0<=k-i<n] = 2*a[k-i]*[0<=k-i<n]
            # is ALWAYS correct (including k=2i), as long as we interpret it as a sum over the valid j values.
            # Wait but that gives 2*a[k-i] for ALL k where 0<=k-i<n, not just k=2i.
            # Let me re-examine. For k=1, i=1, n=3: 0<=k-i=0<n. dc[1]/da[1] = 2*a[0] = 2.
            # Check: c[1] = a[0]*a[1] + a[1]*a[0] = 2a[0]a[1]. d/d(a[1]) = 2a[0]. ✓
            # For k=2, i=1, n=3: 0<=k-i=1<n. dc[2]/da[1] = 2*a[1] = 2a[1].
            # Check: c[2] = a[0]*a[2] + a[1]*a[1] + a[2]*a[0]. d/d(a[1]) = 2a[1]. ✓
            # Great! So the general formula is: dc[k]/da[i] = 2 * a[k-i] if 0<=k-i<n, else 0.
            # (with the understanding that a[j]=0 for j outside [0,n-1])
            #
            # So: df/da[i] = 2 * sum_k w[k] * a[k-i] * [0<=k-i<n] / L * (normalization)
            # = (2/L) * sum_{j: 0<=j<n} w[i+j] * a[j]  (where j = k-i, k = i+j)
            # = (2/L) * (w_shifted * a)[i] where w_shifted[i] = w[i+0]+... 
            # Actually: sum_k w[k]*a[k-i] = sum_j w[i+j]*a[j] (with j=k-i, w defined on [0,2n-2])
            # This is a cross-correlation of w and a.
            # Let's just compute it via FFT:
            # corr[i] = sum_j w[i+j] * a[j] for j in [0,n-1], i in [0,n-1]
            # = (w_padded * a_padded) cross-correlation
            # Using FFT: corr = irfft(rfft(w_full) * conj(rfft(a_padded_to_full)))... 
            # This is getting complicated. Let me just use the simpler approach:
            # grad_a[i] = 2/L * sum_{j=0}^{n-1} w[i+j] * a[j]
            # We can compute this as: for each i, it's a dot product of w[i:i+n] and a.
            # Or: it's the "reversed" convolution. Let w_rev[k] = w[2n-2-k].
            # Then corr[i] = sum_j w[i+j]*a[j] = (a * w_rev)[i+n-1] ... let me just use direct FFT.
            # 
            # Alternative: use the fact that sum_k w[k] * (a * delta_shifted_k)[i]
            # where delta_shifted_k is a shifted delta.
            # 
            # Simplest correct approach: 
            # grad_a = (2/L) * np.array([sum(w[i:i+n] * a) for i in range(n)])
            # But this is O(n^2). For n=3072, that's ~9.4M per evaluation, which is too slow for L-BFGS.
            #
            # FFT approach: corr[i] = sum_j w[i+j]*a[j] for j=0..n-1
            # Let W[k] = rfft(w, size=2n-1 padded to fast length)
            # Let A[k] = rfft(a, size=same)
            # Then the cross-correlation is: corr = irfft(W * conj(A), size)[:n] ... need to be careful.
            # Actually: ifft(FFT(w) * conj(FFT(a)))[m] = sum_k w[k+m]*a[k] (circular correlation)
            # For linear: pad to size >= (2n-1) + n - 1 = 3n-2... that's a lot.
            # 
            # Alternative: corr[i] = sum_j w[i+j]*a[j]
            # = sum_j a[j]*w[i+j]
            # Let's define u[j] = a[j], v[j] = w[j]. Then corr[i] = sum_j u[j]*v[i+j].
            # This is the cross-correlation (u ⊛ v)[i] = sum_j u[j]*v[i+j].
            # In terms of convolution: (u ⊛ v)[i] = (u * v_rev)[i + len(v) - 1] where v_rev = v[::-1].
            # len(v) = 2n-1. So corr[i] = (u * v_rev)[i + 2n-2].
            # (u * v_rev) has length n + (2n-1) - 1 = 3n-2. We need indices 2n-2 to 3n-3.
            # Using FFT: size >= 3n-2. For n=3072, that's ~9214, next_fast_len ~ 16384. Fine.
            
            # For efficiency in L-BFGS, let me use a simpler gradient approximation:
            # The gradient of LSE w.r.t. a is proportional to:
            #   2 * sum_k w[k] * (shifted a)
            # which is 2 * cross-correlation of w with a.
            # Let me compute it via the "flip and convolve" trick:
            # corr[i] = sum_j a[j] * w[i+j] = (a * w_flipped)[i + (2n-2)]
            # where w_flipped = w[::-1] and * denotes linear convolution.
            
            corr_size = next_fast_len(n + 2 * n - 1)
            A_fft = rfft(pp, corr_size)
            Wf_fft = rfft(w[::-1], corr_size)
            corr_full = irfft(A_fft * Wf_fft, corr_size)
            grad_a = 2.0 * corr_full[n - 1:2 * n - 1] / conv_len  # indices (2n-2)-i... 
            # Wait, I need to be careful with the indexing.
            # (a * w_flipped)[m] = sum_j a[j] * w_flipped[m-j] = sum_j a[j] * w[2n-2-(m-j)] = sum_j a[j] * w[2n-2-m+j]
            # I want corr[i] = sum_j a[j] * w[i+j].
            # Set 2n-2-m+j = i+j -> m = 2n-2-i.
            # So corr[i] = (a * w_flipped)[2n-2-i].
            # corr_full[m] = (a * w_flipped)[m]. We need m = 2n-2-i for i=0..n-1.
            # So m ranges from 2n-2 down to n-1.
            grad_a = 2.0 * corr_full[n-1:2*n-1][::-1] / conv_len
            
            # Also need to account for the softmax: d(a)/d(u) = a*(e_i - a)
            # So d(f)/d(u) = grad_a * a - a * (grad_a @ a)
            ta = float(np.dot(grad_a, pp))
            grad_u = pp * (grad_a - ta)
            grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
            return float(f), grad_u

        from scipy.optimize import minimize as sp_minimize
        u = p_to_u(a_in)
        best_u = u.copy()
        
        if fine_only:
            stages = [(0.003, 0.35), (0.0008, 0.35), (0.0002, 0.30)]
        else:
            stages = [(0.1, 0.10), (0.03, 0.12), (0.01, 0.14), (0.003, 0.16),
                      (0.001, 0.18), (0.0003, 0.15), (0.00008, 0.15)]
        
        total_w = sum(w for _, w in stages)
        avail = end_time - time.monotonic() - 5.0
        if avail < 10:
            return a_in
        cum = 0.0
        for frac, wt in stages:
            now = time.monotonic()
            if now > end_time - 3:
                break
            cum += wt
            stage_end = min(end_time - 2, t0 + (end_time - t0) * (cum / total_w))
            if stage_end <= now + 3:
                continue
            cc = compute_c(u_to_p(best_u))
            cmax = cc.max()
            T = max(1e-12, frac * cmax)
            f0, _ = lse_func(best_u, T)
            if not math.isfinite(f0):
                continue
            best_f = f0
            stale = 0
            while time.monotonic() < stage_end - 1:
                try:
                    res = sp_minimize(lse_func, best_u, args=(T,), method='L-BFGS-B',
                                      jac=True, options={'maxiter': 40, 'gtol': 1e-12,
                                                          'ftol': 1e-15, 'maxls': 20,
                                                          'maxdisp': 0, 'disp': False})
                except Exception:
                    break
                if res.fun < best_f:
                    best_u = res.x.copy()
                    best_f = float(res.fun)
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
                if res.status == 0:
                    break
            pp = u_to_p(best_u)
            if pp.sum() < 1e-10:
                continue
            pp /= pp.sum()
            sc = float(2.0 * n_local * compute_c(pp).max())
            if math.isfinite(sc):
                consider(pp)
            best_u = p_to_u(consider(pp) or pp)  # keep the LSE iterate
            # Actually update u from current best
            aa = np.maximum(pp, 1e-30)
            best_u = p_to_u(aa)
        
        pp = u_to_p(best_u)
        pp = np.maximum(pp, 0.0)
        s = pp.sum()
        if s > 0:
            pp /= s
        return pp

    # Run LSE optimization
    lse_end = min(deadline - 300.0, t0 + 0.30 * budget)
    if lse_end > t0 + 30:
        try:

                p_lse /= p_lse.sum()
                consider(p_lse)
                a = best_a.copy()
                a /= a.sum()
                c = _conv_self(a)

    # === Phase 2: Coordinate Descent ===
    def coord_descent(a_init, end_time, seed_cd, G_coarse=40, G_fine=60):
        aa = np.array(a_init, dtype=np.float64, copy=True)
        aa = np.maximum(aa, 0.0)
        ss = aa.sum()
        if ss < 1e-15:
            return aa
        aa /= ss
        cc = _conv_self(aa)
        n_l = aa.size
        L_tot = 2 * n_l - 1
        rng_cd = np.random.default_rng(seed_cd)
        sweep = 0
        flat = 0
        cur_cmax = float(cc.max())
        cur_score = 2.0 * n_l * cur_cmax

        while sweep < 100000 and time.monotonic() < end_time - 1.0:
            improved = False
            order = rng_cd.permutation(n_l)
            
            for ii in order:
                i = int(ii)
                if time.monotonic() > end_time - 0.5:
                    break
                
                v = aa[i]
                # Affected range: k in [i, i+n_l-1] clamped to [0, L_tot-1]
                k_s = i
                k_e = min(i + n_l, L_tot)
                L_aff = k_e - k_s
                if L_aff <= 0:
                    continue
                
                a_vals = aa[:L_aff]  # a[k-i] for k in [k_s, k_e-1], i.e., a[0:L_aff]
                base = cc[k_s:k_e].copy()
                diag_idx = 2 * i - k_s
                has_diag = (0 <= diag_idx < L_aff)
                
                # Unaffected max
                if k_s > 0:
                    ml = float(cc[:k_s].max())
                else:
                    ml = 0.0
                if k_e < L_tot:
                    mr = float(cc[k_e:].max())
                else:
                    mr = 0.0
                c_out = max(ml, mr)
                
                # Grid
                avg = 1.0 / n_l
                x_hi = min(max(30.0 * v, 60.0 * avg, 0.02), 8.0)
                if x_hi < 1e-10:
                    x_hi = max(1e-8, avg)
                
                x_grid = np.linspace(0.0, x_hi, G_coarse + 1, dtype=np.float64)
                w_grid = x_grid - v
                denom = (1.0 + w_grid) ** 2  # (S + w)^2, S=1
                
                delta = base[None, :] + 2.0 * w_grid[:, None] * a_vals[None, :]
                if has_diag:
                    delta[:, diag_idx] += w_grid[:, None] * w_grid[:, None]
                
                c_max_aff = delta.max(axis=1)
                c_max_tot = np.maximum(c_max_aff, c_out)
                f_vals = 2.0 * n_l * c_max_tot / denom
                f_vals = np.where(np.isfinite(f_vals), f_vals, np.inf)
                
                imin = int(np.argmin(f_vals))
                best_x = float(x_grid[imin])
                best_val = float(f_vals[imin])
                
                # Fine refinement
                if best_val < cur_score - 1e-13:
                    dw = x_hi / G_coarse
                    lo = max(0.0, best_x - 2.0 * dw)
                    hi = min(x_hi, best_x + 2.0 * dw)
                    if hi <= lo:
                        hi = lo + 1e-12
                    x_f = np.linspace(lo, hi, G_fine + 1, dtype=np.float64)
                    w_f = x_f - v
                    denom_f = (1.0 + w_f) ** 2
                    delta_f = base[None, :] + 2.0 * w_f[:, None] * a_vals[None, :]
                    if has_diag:
                        delta_f[:, diag_idx] += w_f[:, None] * w_f[:, None]
                    c_max_af = np.maximum(delta_f.max(axis=1), c_out)
                    f_f = 2.0 * n_l * c_max_af / denom_f
                    f_f = np.where(np.isfinite(f_f), f_f, np.inf)
                    imin_f = int(np.argmin(f_f))
                    if f_f[imin_f] < best_val:
                        best_x = float(x_f[imin_f])
                        best_val = float(f_f[imin_f])
                
                # Update
                if math.isfinite(best_val) and best_val < cur_score - 1e-13:
                    w = best_x - v
                    aa[i] = best_x
                    cc[k_s:k_e] = base + 2.0 * w * a_vals
                    if has_diag:
                        cc[2 * i] += w * w
                    cc = np.maximum(cc, 0.0)
                    new_S = 1.0 + w
                    if new_S > 1e-10:
                        aa /= new_S
                        cc /= (new_S * new_S)
                        cur_score = best_val
                        improved = True
                    else:
                        # Revert
                        aa[i] = v
                        cc[k_s:k_e] = base
                        if has_diag:
                            cc[2 * i] -= w * w
            
            if improved:
                flat = 0
            else:
                flat += 1
                if flat >= 4:
                    if time.monotonic() < end_time - 10.0:
                        aa = aa * np.exp(rng_cd.normal(0, 0.02, n_l))
                        aa = np.maximum(aa, 0.0)
                        s2 = aa.sum()
                        if s2 > 0:
                            aa /= s2
                        cc = _conv_self(aa)
                        cur_score = 2.0 * n_l * cc.max()
                        flat = 0
                    else:
                        break
            sweep += 1
        
        aa = np.maximum(aa, 0.0)
        s_final = aa.sum()
        if s_final > 0:
            aa /= s_final
        return aa

    # Run coordinate descent
    cd1_end = min(deadline - 200.0, t0 + 0.65 * budget)
    if cd1_end > time.monotonic() + 30:
        try:
            a_cd = coord_descent(a, cd1_end, int(seed + 111), G_coarse=40, G_fine=60)
            if a_cd is not None and a_cd.sum() > 0.01:
                a_cd /= a_cd.sum()
                consider(a_cd)
        except Exception:
            pass

    # === Phase 3: Fine LSE ===
    a_work = best_a.copy()
    a_work /= a_work.sum()
    lse2_end = min(deadline - 150.0, time.monotonic() + 0.15 * budget)
    if lse2_end > time.monotonic() + 20:
        try:
            p2 = lse_optimize(a_work, lse2_end, fine_only=True)
            if p2 is not None and p2.size == n and p2.sum() > 0.01:
                p2 /= p2.sum()
                consider(p2)
        except Exception:
            pass

    # === Phase 4: More coordinate descent ===
    a_work = best_a.copy()
    a_work /= a_work.sum()
    cd2_end = min(deadline - 80.0, time.monotonic() + 0.20 * budget)
    if cd2_end > time.monotonic() + 20:
        try:
            a_cd2 = coord_descent(a_work, cd2_end, int(seed + 222), G_coarse=48, G_fine=80)
            if a_cd2 is not None and a_cd2.sum() > 0.01:
                a_cd2 /= a_cd2.sum()
                consider(a_cd2)
        except Exception:
            pass

    # === Phase 5: Final fine coordinate descent ===
    a_work = best_a.copy()
    a_work /= a_work.sum()
    cd3_end = deadline - 20.0
    if cd3_end > time.monotonic() + 15:
        try:
            a_cd3 = coord_descent(a_work, cd3_end, int(seed + 333), G_coarse=56, G_fine=100)
            if a_cd3 is not None and a_cd3.sum() > 0.01:
                a_cd3 /= a_cd3.sum()
                consider(a_cd3)
        except Exception:
            pass

    # Final validation
    try:
        out_arr = np.asarray(best_a, dtype=np.float64).ravel()
        out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
        out_arr = np.clip(out_arr, 0.0, 1000.0)
        if out_arr.size == 0 or out_arr.sum() < 0.01:
            out_arr = parent.copy()
        out_list = [float(x) for x in out_arr]
        final_sc = evaluate_sequence(out_list)
        if not math.isfinite(final_sc):
            out_list = [float(x) for x in parent]
            final_sc = evaluate_sequence(out_list)
            if not math.isfinite(final_sc):
                out_list = [1.0]
        return out_list
    except Exception:
        return [1.0]
