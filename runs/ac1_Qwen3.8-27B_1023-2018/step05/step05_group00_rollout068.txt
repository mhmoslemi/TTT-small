def _lp_step(f, stop_time):
    n = int(f.size)
    if n < 10 or n > 4096:
        return None
    M = float(_conv_self(f).max())
    if M <= 0 or not math.isfinite(M):
        return None
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f[j]
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        constraints = [A @ g <= b]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        solved = False
        for slv in [cp.SCS, cp.ECOS, cp.CLP]:
            try:
                prob.solve(solver=slv, verbose=False)
                solved = True
                break
            except Exception:
                continue
            if time.monotonic() > stop_time - 5:
                break
        if not solved or g.value is None:
            return None
        g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        gv = float(g_val.sum())
        if gv < 1e-10 or not math.isfinite(gv):
            return None
        return g_val / gv
    except Exception:
        return None

    if lp_end > time.monotonic() + 10.0:

        if g_dir is not None and g_dir.size == n_par:
            for t_alpha in [0.3, 0.2, 0.1, 0.05]:
                blended = (1.0 - t_alpha) * work + t_alpha * g_dir
                blended = np.maximum(blended, 0.0)
                bs = float(blended.sum())
                if bs > 0:
                    blended /= bs
                    consider(blended)

    work = best_arr.copy()
    work = work / float(work.sum())

    phase4_end = min(hard - 100.0, t0 + 0.72 * budget)
    if phase4_end > time.monotonic() + 30.0:
        try:
            work, _ = _coord_polish(work, phase4_end, max_sweeps=500000,
                                    seed=int(seed + 123456), consider=consider)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    phase5_end = min(hard - 40.0, t0 + budget - 35.0)
    if phase5_end > time.monotonic() + 25.0:
        lse5_end = min(phase5_end - 50.0, time.monotonic() + 0.25 * (phase5_end - time.monotonic()))
        if lse5_end > time.monotonic() + 10.0:
            try:
                p5 = _lbfgs_optimize(work, lse5_end, consider=consider, fine_only=True)
                if p5 is not None and p5.size == n_par and np.all(np.isfinite(p5)) and p5.sum() > 0:
                    p5 = p5 / float(p5.sum())
                    consider(p5)
                    work = best_arr.copy()
                    work = work / float(work.sum())
            except Exception:
                pass
        coord5_end = phase5_end
        if coord5_end > time.monotonic() + 15.0:
            try:
                work, _ = _coord_polish(work, coord5_end, max_sweeps=300000,
                                        seed=int(seed + 987654), consider=consider)
            except Exception:
                pass

    rem = hard - 30.0 - time.monotonic()
    if rem > 80.0:
        for n2 in [n_par // 2, n_par * 2 // 3, n_par * 3 // 4]:
            if n2 < 64 or n2 >= n_par:
                continue
            xp = (np.arange(n_par, dtype=np.float64) + 0.5) / float(n_par)
            fp = np.interp((np.arange(n2, dtype=np.float64) + 0.5) / float(n2), xp, best_arr)
            p2 = np.maximum(fp, 0.0)
            p2s = float(p2.sum())
            if p2s > 0:
                p2 /= p2s
                consider(p2)
                rem2 = hard - 35.0 - time.monotonic()
                if rem2 > 40.0:
                    p2_end = min(hard - 35.0, time.monotonic() + rem2 - 10.0)
                    if p2_end > time.monotonic() + 15.0:
                        try:
                            p2, _ = _coord_polish(p2, p2_end, max_sweeps=100000,
                                                  seed=int(seed + 424242), consider=consider)
                        except Exception:
                            pass
                if time.monotonic() > hard - 40.0:
                    break

    fin_end = min(hard - 15.0, t0 + budget - 12.0)
    if fin_end > time.monotonic() + 15.0:
        work = best_arr.copy()
        work = work / float(work.sum())
        try:
            work, _ = _coord_polish(work, fin_end, max_sweeps=200000,
                                    seed=int(seed + 555555), consider=consider)
        except Exception:
            pass

    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
