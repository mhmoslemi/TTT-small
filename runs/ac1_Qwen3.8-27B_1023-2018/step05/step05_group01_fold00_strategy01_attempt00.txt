    """Score based on harmonic distribution penalty"""

    s_old = np.sqrt(3 * np.dot(arr, arr))

    loss_base = s_current / (s_old ** harmonic_weight)

    diff = np.diff(arr)
    tail_removed = np.min(diff) * np.size(diff) ** 0.5
    if tail_removed > 0:
        s_current += tail_removed

def sequence_performance(seq: List[float], target: float = 0.001) -> float:
    """Compute performance score"""
    arr = np.array(seq, dtype=np.float64)
    
    # Compute convolution
    convolve_result = np.convolve(arr, arr, 'full')
    max_val = float(np.max(convolve_result))
    score_val = 2 * len(arr) * max_val / (np.sum(arr) ** 2) if np.sum(arr) > 0.01 else 1e12
    
    penalty = calculate_lower_harmonic_bound([x * 0.5 * target for x in seq]) if score_val > 1e10 else 1.0
    score_val *= (1 + penalty)
