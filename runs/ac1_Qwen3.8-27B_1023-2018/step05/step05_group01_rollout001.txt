    sweep = 0
    flat = 0
    max_sweeps = 50000

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.0:
        improved = False
        # Determine ordering
        if sweep % 4 == 0:
            order = np.arange(n)
        elif sweep % 4 == 1:
            order = np.arange(n-1, -1, -1)
        elif sweep % 4 == 2:
            order = rng.permutation(n)
        else:
            # Sort by "sensitivity": coordinates that are near the max convolution
            # The max is at argmax(c); coordinates near that index are more sensitive
            kmax = int(np.argmax(c))
            # Sensitivity ~ |f[kmax - i]| (the subgradient magnitude)
            kmax_clipped = min(max(kmax, 0), n-1)
            sens = np.zeros(n)
            lo = max(0, kmax - n + 1)
            hi = min(n, kmax + 1)
            for i in range(lo, hi):
                sens[i] = abs(p[kmax - i])
            order = np.argsort(-sens)  # high sensitivity first

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            S = 1.0  # normalized
            v = float(p[i])

            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end].copy()
            # infl[k-start] = 2 * p[k-i] for k in [start, end), i.e., p[0..Ls-1]
            infl = 2.0 * p[:Ls]  # p[k - i] for k from i to i+Ls-1

            # Constant: max of c outside [start, end)
            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            # Search range for v
            x_hi = max(20.0 * v, 40.0 / float(n), 0.02, 1e-14)
            x_hi = min(x_hi, 10.0)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, 1.0 / n)

            diag_i = i - start  # index of 2i in the base array

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            w = pts - v
            denom = 1.0 + w  # S + w = 1 + (v_new - v_old)
            
            vals = np.full(pts.shape, math.inf, dtype=np.float64)
            valid = denom > 1e-15
            if not np.any(valid):
                continue

            # Vectorized evaluation
            aff = base[None, :] + w[valid, None] * infl[None, :]
            
            # The diagonal term (k=2i): base[diag_i] + w*infl[diag_i] + w^2
            if 0 <= diag_i < Ls:
                dval = base[diag_i] + w[valid] * infl[diag_i] + w[valid] * w[valid]
                aff = np.maximum(aff, dval[:, None])
            
            aff = np.maximum(aff, max_const)
            mv = aff.max(axis=1)
            vals[valid] = (2.0 * n * mv) / (denom[valid] * denom[valid])

            if not np.any(np.isfinite(vals)):
                continue

            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Refinement
            if G > 10 and 0.0 < best_x < x_hi:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 1.5 * dw)
                hi = min(x_hi, best_x + 1.5 * dw)
                if hi <= lo:
                    hi = lo + 1e-14
                fine = np.linspace(lo, hi, 32, dtype=np.float64)
                if 0.0 <= best_x <= x_hi:
                    fine = np.concatenate((fine, [best_x]))
                    fine = np.unique(fine)
                wf = fine - v
                denomf = 1.0 + wf
                vals_f = np.full(fine.shape, math.inf, dtype=np.float64)
                validf = denomf > 1e-15
                if np.any(validf):
                    aff_f = base[None, :] + wf[validf, None] * infl[None, :]
                    if 0 <= diag_i < Ls:
                        dval_f = base[diag_i] + wf[validf] * infl[diag_i] + wf[validf] * wf[validf]
                        aff_f = np.maximum(aff_f, dval_f[:, None])
                    aff_f = np.maximum(aff_f, max_const)
                    mv_f = aff_f.max(axis=1)
                    vals_f[validf] = (2.0 * n * mv_f) / (denomf[validf] ** 2)
                    if np.any(np.isfinite(vals_f)):
                        imin_f = int(np.argmin(vals_f))
                        vx_f = float(fine[imin_f])
                        vv_f = float(vals_f[imin_f])
                        if vv_f < best_val - 1e-15:
                            best_x = vx_f
                            best_val = vv_f

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 10.0:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    c[start:end] += w * infl
                    if 0 <= diag_i < Ls:
                        c[start + diag_i] += w * w
                    cur_score = best_val
                    improved = True

        # Renormalize (should already be ~1 since we normalized)
        ss = float(p.sum())
        if ss > 1e-300 and math.isfinite(ss):
            if abs(ss - 1.0) > 1e-6:
                p /= ss
                c *= (1.0 / ss) ** 2
                cur_score = float(2.0 * n * c.max())

        # Clip very small values
        if sweep % 10 == 0:
            small = p < clip_thresh
            if np.any(small):
                p[small] = 0.0

        # Re-evaluate periodically to correct drift
        if sweep % 5 == 4:
            c = _conv_self(p)
            real_score = float(2.0 * n * c.max())
            cur_score = real_score
            if real_score < best_score - 1e-15:
                best_score = real_score
                best_p = p.copy()
                if consider is not None:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 10.0:
                    # Perturbation to escape
                    sigma = 0.01 + 0.01 * min(flat, 15) / 15.0
                    mask = p > clip_thresh
                    p_pert = p.copy()
                    p_pert[mask] *= np.exp(rng.normal(0, sigma, mask.sum()))
                    p_pert = np.maximum(p_pert, 0.0)
                    p_s = float(p_pert.sum())
                    if p_s > 0:
                        p_pert /= p_s
                        c = _conv_self(p_pert)
                        ps = float(2.0 * n * c.max())
                        if ps < best_score:
                            p = p_pert
                            cur_score = ps
                        elif ps < cur_score + 0.01:
                            p = p_pert
                            cur_score = ps
                    flat = 0
                else:
                    break

        sweep += 1

    # Final
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_self(p)
    sc = float(2.0 * n * c.max())
    if sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _lse_optimize(p, stop_time, consider=None, seed=42):
    """LSE-based optimization (improved from parent)."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def softmax_fn(uu):
        m = float(np.max(uu))
        e = np.exp(uu - m)
        ss = float(e.sum())
        if ss <= 0 or not math.isfinite(ss):
            return np.full(uu.size, 1.0 / uu.size)
        return e / ss

    def lse_func(uu, T):
        pp = softmax_fn(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        scaled = (c - cmax) / T
        exps = np.exp(np.minimum(scaled, 50.0))  # prevent overflow
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    stages = (
        (0.1, 0.10),
        (0.03, 0.10),
        (0.01, 0.10),
        (0.003, 0.10),
        (0.001, 0.12),
        (0.0003, 0.12),
    )

    start = time.monotonic()
    best_u = u.copy()
    best_sc = _score_fft(p)

    for frac_h, _wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac_h * cmax)

        best_f_st = math.inf
        max_inner = 8
        for _ in range(max_inner):
            if time.monotonic() > stop_time - 3.0:
                break
            opts = {"maxiter": 30, "gtol": 1e-10, "ftol": 1e-14,
                    "maxls": 20, "maxcor": 20, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    fval = float(res.fun)
                    if math.isfinite(fval) and fval < best_f_st - 1e-15:
                        best_f_st = fval
                        best_u = x.copy()
                    else:
                        break

        p_cur = softmax_fn(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_sc = sc_cur
            p = p_cur.copy()
            if consider is not None:
                try:
                    consider(p)
                except Exception:
                    pass
        u = p_to_u(p)

    return p
