                    break
                i = int(order[pos])
                v = float(a[i])
                start = i
                end = min(i + nn, L)
                Ls = end - start
                if Ls <= 0:
                    continue
                base_c = c[start:end]
                # Max of unaffected parts
                max_const = 0.0
                if start > 0:
                    mc = float(c[:start].max())
                    if mc > max_const:
                        max_const = mc
                if end < L:
                    mc = float(c[end:].max())
                    if mc > max_const:
                        max_const = mc

                # Evaluate new score for candidate x
                def new_score(x):
                    w = x - v
                    aff = base_c + w * infl[start:end]
                    di = 2 * i - start
                    if 0 <= di < Ls:
                        aff[di] += w * w
                    np.maximum(aff, max_const, out=aff)
                    mv = float(np.max(aff))
                    S_new = 1.0 + w  # since sum=1
                    if S_new <= 1e-14:
                        return math.inf
                    return float(2.0 * nn * mv) / (S_new * S_new)

                # Dense grid search
                xhi = max(10.0 * v, 0.02 / nn)
                xhi = min(xhi, 0.5 / nn * 6)
                if not math.isfinite(xhi) or xhi <= 0:
                    xhi = 5.0 / nn
                pts = np.concatenate([
                    np.array([0.0, v]),
                    np.geomspace(max(1e-12, 0.01 * v if v > 0 else 1e-12),
                                 xhi, 18),
                    np.linspace(0.0, min(xhi, 3.0/nn), 12),
                ])
                pts = np.unique(np.clip(pts, 0.0, xhi))
                vals = np.array([new_score(float(x)) for x in pts])
                fin_mask = np.isfinite(vals)
                if not np.any(fin_mask):
                    continue
                im = int(np.argmin(np.where(fin_mask, vals, np.inf)))
                bx = float(pts[im])
                bv = float(vals[im])

                # Local refinement
                if bx > 0 and bv < cur_score:
                    step = max(v * 0.3, 1e-6 / nn)
                    lo = max(0.0, bx - 2 * step)
                    hi = min(xhi, bx + 2 * step)
                    fine_pts = np.linspace(lo, hi, 24)
                    vals2 = np.array([new_score(float(x)) for x in fine_pts])
                    fin2 = np.isfinite(vals2)
                    if np.any(fin2):
                        im2 = int(np.argmin(np.where(fin2, vals2, np.inf)))
                        if vals2[im2] < bv:
                            bx = float(fine_pts[im2])
                            bv = float(vals2[im2])

                if math.isfinite(bv) and bv < cur_score - 1e-13:
                    d = bx - v
                    c[start:end] += d * infl[start:end]
                    diag = 2 * i
                    if 0 <= diag < L:
                        c[diag] += d * d
                    a[i] = bx
                    infl[i] = 2.0 * bx
                    cur_score = bv
                    improved = True

            # Renormalize
            S = float(a.sum())
            if S > 1e-10 and math.isfinite(S):
                a /= S
                c /= (S * S)
                infl /= S
            c = _fft_conv(a)
            desc = float(2.0 * nn * c.max())
            if math.isfinite(desc) and desc < best_local_sc - 1e-14:
                best_local_sc = desc
                best_local = a.copy()

            if improved:
                flat = 0
            else:
                flat += 1
                if flat >= 6 and time.monotonic() < stop_time - 30:
                    # Perturb to escape
                    sigma = 0.02 + 0.04 * min(flat - 6, 5)
                    noise = rng_cd.normal(0, sigma, nn)
                    q = a * np.exp(noise)
                    qs = q.sum()
                    if qs > 1e-10:
                        a = q / qs
                        c = _fft_conv(a)
                        infl = 2.0 * a
                        cur_score = float(2.0 * nn * c.max())
                    flat = 0
                elif flat >= 12:
                    break

        consider(best_local)
        return best_local

    # Coordinate descent phase
    now = time.monotonic()
    if now < hard - 30:
        cd_end = min(hard - 10.0, now + 0.45 * (hard - now))
        p3 = _coord_polish(best_arr, cd_end, seed_cd=seed + 777)
        consider(p3)

    # Final squeeze: more coordinate descent + a short L-BFGS
    now = time.monotonic()
    if now < hard - 25:
        fin_end = hard - 8.0
        p4 = _coord_polish(best_arr, fin_end, seed_cd=seed + 4242, max_sweeps=80000)
        consider(p4)

    # Validate with exact evaluator
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = np.ones(n)
            s = float(n)
        out = out / s
        ls = [float(x) for x in out]
        fs = evaluate_sequence(ls)
        if not math.isfinite(fs):
            # Fallback
            ls = [1.0]
        return ls
    except Exception:
        return [1.0]
