            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, avg)

            G = 64 if n <= 2048 else 48
            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 <= v <= x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-12
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                if 0 <= diag < Ls:
                    dval = base[diag] + w * infl[diag] + w * w
                    maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_ext)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            dw = x_hi / float(G)
            lo = max(0.0, best_x - 3.0 * dw)
            hi = min(x_hi, best_x + 3.0 * dw)
            if hi <= lo:
                hi = lo + 1e-12
            fine = np.linspace(lo, hi, 60, dtype=np.float64)
            if 0.0 <= best_x <= x_hi:
                fine = np.concatenate((fine, [best_x]))
                fine = np.unique(fine)
            vals2 = eval_xs(fine)
            if np.any(np.isfinite(vals2)):
                imin2 = int(np.argmin(vals2))
                vx2 = float(fine[imin2])
                vv2 = float(vals2[imin2])
                if vv2 < best_val:
                    best_x = vx2
                    best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-12 and 0.0 <= best_x <= 20.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    a[i] = best_x
                    S += w
                    c[start_k:end_k] += w * infl
                    if 0 <= diag < Ls:
                        c[start_k + diag] += w * w
                    cur_score = best_val
                    improved = True

        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            a *= inv
            c *= inv * inv
        else:
            break
        S = 1.0
        c = _conv_self(a)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = a.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 10.0:
                    try:
                        a = a * np.exp(rng.normal(0.0, 0.03, n, dtype=np.float64))
                        a = np.maximum(a, 0.0)
                        ss = float(a.sum())
                        if ss > 0:
                            a = a / ss
                        S = 1.0
                        c = _conv_self(a)
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
                            best_score = cur_score
                            best_p = a.copy()
                        flat = 0
                    except Exception:
                        break
                else:
                    break
        sweep += 1

    a = np.maximum(a, 0.0)
    ss = float(a.sum())
    if ss > 1e-300:
        a = a / ss
    c = _conv_self(a)
    sc = float(2.0 * n * c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = a.copy()
    return best_p, best_score


def _lse_optimize(a, stop_time, consider=None):
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0).copy()
    n = int(a.size)
    if n == 0:
        return a
    s = float(a.sum())
    if s < 1e-300:
        return a
    a = a / s

    def softmax(u):
        m = float(np.max(u))
        e = np.exp(u - m)
        s_e = float(e.sum())
        if not math.isfinite(s_e) or s_e <= 0:
            return np.full(u.size, 1.0 / u.size)
        return e / s_e

    u = np.log(np.maximum(a, 1e-300))
    u = u - float(np.mean(u))

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1

    def cfast(pp):
        F = rfft(pp, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return np.maximum(c, 0.0)

    def lse_func(uu, T):
        pp = softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e50, np.zeros_like(uu)
        if T <= 0:
            T = 1e-4 * cmax
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            w = exps / sum_exp
        F_pp = rfft(pp, conv_size)
        F_w = rfft(w, conv_size)
        cross = irfft(F_w * F_pp.conj(), conv_size)[:conv_len]
        q = cross[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    best_u = u.copy()
    stages = (0.05, 0.02, 0.008, 0.003, 0.001, 0.0004, 0.0001, 0.00003)

    start = time.monotonic()
    avail = stop_time - start - 5.0
    if avail < 10:
        return a

    best_p = a.copy()
    best_sc = _score_fast(a)
    n_stages = len(stages)

    for si, frac in enumerate(stages):
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        stage_end = min(stop_time - 3.0, start + avail * (si + 1) / n_stages - 0.5)
        if stage_end <= now + 3:
            continue
        c = cfast(softmax(best_u))
        cmax = float(c.max())
        T = frac * cmax
        maxit = 50 if n > 2000 else 80
        for _ in range(5):
            if time.monotonic() > stage_end - 1.0:
                break
            try:
                res = minimize(lse_func, best_u, args=(T,),
                               method='L-BFGS-B', jac=True,
                               options={'maxiter': maxit, 'gtol': 1e-10,
                                        'ftol': 1e-14, 'maxls': 30, 'maxcor': 30,
                                        'maxdisp': 0, 'disp': False})
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                try:
                    fv = float(res.fun)
                except Exception:
                    fv = math.inf
                if math.isfinite(fv):
                    best_u = res.x.copy()
            if getattr(res, 'status', 1) == 0:
                break

        p_cur = softmax(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        sc_cur = _score_fast(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_sc = sc_cur
            best_p = p_cur.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0

    parent = _get_parent()
    best_arr = parent.copy()
    best_arr = np.where(np.isfinite(best_arr), best_arr, 0.0)
    best_arr = np.clip(best_arr, 0.0, 1000.0)
    best_score = math.inf
    try:
        sc = float(evaluate_sequence([float(x) for x in best_arr]))
        if math.isfinite(sc):
            best_score = sc
    except Exception:
        pass

    if not math.isfinite(best_score) or best_arr.size == 0 or float(best_arr.sum()) < 0.01:
        best_arr = np.ones(256, dtype=np.float64) / 256.0
        best_score = float(_score_fast(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0 or float(a.sum()) < 0.01:
                return
            sc = float(_score_fast(a))
            if math.isfinite(sc) and sc < best_score - 1e-12:
                best_arr = a.copy()
                best_score = sc
        except Exception:
            pass

    n_par = int(best_arr.size)
    work = best_arr / float(best_arr.sum())

    # Phase 1: LSE optimization
    p1_end = min(hard - 100.0, t0 + 0.20 * budget)
    if p1_end > time.monotonic() + 15.0:
        try:
            p1 = _lse_optimize(work, p1_end, consider=consider)
            if p1 is not None and p1.size == n_par and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass
    work = best_arr / float(best_arr.sum())

    # Phase 2: Coordinate descent (main workhorse)
    p2_end = min(hard - 120.0, t0 + 0.50 * budget)
    if p2_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord_descent(work, p2_end, seed=int(seed + 1111), consider=consider)
        except Exception:
            pass
    work = best_arr / float(best_arr.sum())

    # Phase 3: Fine LSE
    p3_end = min(hard - 100.0, t0 + 0.65 * budget)
    if p3_end > time.monotonic() + 15.0:
        try:
            p3 = _lse_optimize(work, p3_end, consider=consider)
            if p3 is not None and p3.size == n_par and p3.sum() > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)
        except Exception:
            pass
    work = best_arr / float(best_arr.sum())

    # Phase 4: More coordinate descent
    p4_end = min(hard - 60.0, t0 + 0.82 * budget)
    if p4_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord_descent(work, p4_end, seed=int(seed + 2222), consider=consider)
        except Exception:
            pass
    work = best_arr / float(best_arr.sum())

    # Phase 5: Try shorter length
    rem = hard - 30.0 - time.monotonic()
    if rem > 120.0 and n_par > 1024:
        n2 = n_par // 2
        xp = (np.arange(n_par) + 0.5) / n_par
        fp = np.interp((np.arange(n2) + 0.5) / n2, xp, best_arr)
        p2 = np.maximum(fp, 0.0)
        ps = float(p2.sum())
        if ps > 0.01:
            p2 = p2 / ps
            consider(p2)
            if rem > 150.0:
                p2_end = min(hard - 25.0, time.monotonic() + rem - 30.0)
                if p2_end > time.monotonic() + 20.0:
                    try:
                        best_cur = best_arr.copy()
                        work2 = p2.copy()
                        work2, _ = _coord_descent(work2, p2_end, seed=int(seed + 3333), consider=consider)
                    except Exception:
                        pass

    # Phase 6: Final squeeze
    fin_end = min(hard - 15.0, t0 + budget - 12.0)
    if fin_end > time.monotonic() + 15.0:
        work = best_arr / float(best_arr.sum())
        try:
            work, _ = _coord_descent(work, fin_end, seed=int(seed + 4444), consider=consider)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
