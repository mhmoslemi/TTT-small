            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag_i = i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[valid, None] * infl[None, :]
                dval = base[diag_i] + w[valid] * infl[diag_i] + w[valid] * w[valid]
                aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                dv = denom[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if G > 4 and best_x > 0 and best_x < x_hi:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 32, dtype=np.float64)
                if 0.0 <= best_x <= x_hi:
                    fine = np.concatenate((fine, [best_x]))
                    fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val - 1e-15:
                        best_x = vx2
                        best_val = vv2

            if (math.isfinite(best_val) and best_val < cur_score - 1e-14
                    and 0.0 <= best_x <= 10.0 * S):
                w = best_x - v
                if abs(w) > 1e-16:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-16:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 5.0:
                    sigma = 0.015 + 0.015 * min(flat, 8) / 8.0
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    else:
                        break
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    flat = 0
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        sc = float(_score(p))
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 30.0
    if hard <= t0 + 3.0:
        hard = t0 + max(3.0, budget * 0.9)

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    best_score = _score(best_arr)

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(256) / 256.0
        best_score = _score(best_arr)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_arr = a / s
            best_score = sc

    def make_asym_init(src, n, sd):
        src = np.asarray(src, dtype=np.float64).ravel()
        if src.size != n:
            xp = (np.arange(src.size) + 0.5) / src.size
            fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
            src = np.maximum(fp, 0.0)
            s = src.sum()
            if s > 0:
                src = src / s
            else:
                src = np.full(n, 1.0 / n)
        rng = np.random.default_rng(sd)
        x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
        side = 1.0 if rng.random() < 0.5 else -1.0
        amp = 0.3 + 1.5 * float(rng.random())
        curv = 0.3 + 1.5 * float(rng.random())
        q = src * np.exp(side * amp * (2.0 * x - 1.0) + curv * (2.0 * x - 1.0) ** 2)
        q = np.maximum(q, 0.0)
        s = float(q.sum())
        if s > 0:
            q /= s
        else:
            q = src.copy()
        return q

    n_parent = int(best_arr.size)
    total_time = max(10.0, hard - t0)

    # Phase 1: LSE optimization at parent length
    p1_end = t0 + 0.45 * total_time
    if p1_end > time.monotonic() + 25.0:
        try:
            best_arr = _lse_optimize(best_arr, p1_end, seed=int(seed + 10), consider=consider)
        except Exception:
            pass
    best_arr = np.maximum(best_arr, 0.0)
    ss = float(best_arr.sum())
    if ss > 0:
        best_arr = best_arr / ss
    consider(best_arr)

    # Phase 2: Coarse coordinate descent
    p2_end = t0 + 0.68 * total_time
    if p2_end > time.monotonic() + 15.0:
        try:
            best_arr, _ = _coord_polish(best_arr, p2_end, G=36, seed=int(seed + 200),consider=consider)
        except Exception:
            pass
    consider(best_arr)

    # Phase 3: Short LSE refinement at lower temperature
    rem3 = t0 + 0.80 * total_time - time.monotonic()
    if rem3 > 20.0 and time.monotonic() < t0 + 0.80 * total_time:
        p3_end = t0 + 0.80 * total_time
        try:
            best_arr = _lse_optimize(best_arr, p3_end, seed=int(seed + 500), consider=consider)
        except Exception:
            pass
    best_arr = np.maximum(best_arr, 0.0)
    ss = float(best_arr.sum())
    if ss > 0:
        best_arr = best_arr / ss
    consider(best_arr)

    # Phase 4: Fine coordinate descent
    p4_end = t0 + 0.93 * total_time
    if p4_end > time.monotonic() + 10.0:
        try:
            best_arr, _ = _coord_polish(best_arr, p4_end, G=56, seed=int(seed + 900),
                                        consider=consider)
        except Exception:
            pass
    consider(best_arr)

    # Phase 5: Try a perturbed restart
    p5_end = hard - 15.0
    rem5 = p5_end - time.monotonic()
    if rem5 > 30.0:
        rng5 = np.random.default_rng(int(seed + 7777))
        pert = best_arr * np.exp(rng5.normal(0, 0.02, best_arr.size))
        pert = np.maximum(pert, 0.0)
        ps = float(pert.sum())
        if ps > 0:
            pert /= ps
        sub_end = min(p5_end, time.monotonic() + 0.5 * rem5)
        if sub_end > time.monotonic() + 15.0:
            try:
                p_pert = _lse_optimize(pert, sub_end, seed=int(seed + 8888), consider=consider)
                if p_pert is not None and p_pert.size == n_parent:
                    p_pert = np.maximum(p_pert, 0.0)
                    sp = float(p_pert.sum())
                    if sp > 0:
                        p_pert /= sp
                        consider(p_pert)
            except Exception:
                pass

    # Phase 6: Final micro-polish
    p6_end = hard - 8.0
    if p6_end > time.monotonic() + 8.0:
        try:
            best_arr, _ = _coord_polish(best_arr, p6_end, G=80, seed=int(seed + 99999),
                                        consider=consider)
        except Exception:
            pass
    consider(best_arr)

    # Final validation with exact evaluator
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0 / 64.0] * 64
        return ls
    except Exception:
        try:
            return [float(x) for x in parent]
        except Exception:
            return [1.0]
