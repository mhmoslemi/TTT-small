    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)

                        aff[idx, diag_local] = max(aff[idx, diag_local], dval[idx])
                maff = np.maximum(aff, max_const)
                vals[valid] = (2.0 * n * maff[valid]) / (denom[valid] ** 2)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            bx = float(pts[imin])
            bv = float(vals[imin])

            if G > 4:
                dw = x_hi / float(G)
                lo = max(0.0, bx - 3.0 * dw)
                hi = min(x_hi, bx + 3.0 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 50, dtype=np.float64)
                if 0.0 < bx < x_hi:
                    fine = np.insert(fine, 25, bx)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < bv:
                        bx = vx2
                        bv = vv2

            if math.isfinite(bv) and bv < cur_score - 1e-14 and 0.0 <= bx <= 20.0 * S:
                w_change = bx - v
                if abs(w_change) > 1e-15:
                    p[i] = bx
                    S_total += w_change
                    c[start_i:end_i] += w_change * infl_vec
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w_change * w_change
                    cur_score = bv
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0

        if sweep % cheap_interval == 0:
            c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 8.0:
                    sigma = 0.02 + 0.03 * min(flat / 10.0, 2.0)
                    try:
                        p = p * np.exp(rng.normal(0.0, sigma, n, dtype=np.float64))
                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss > 0:
                            p /= ss
                        c = _conv_self(p)
                        S_total = 1.0
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider is not None:
                                consider(best_p)
                        flat = 0
                    except Exception:
                        break
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _lp_step(f, stop_time):
    n = int(f.size)
    if n < 20:
        return None
    M = float(_conv_self(f).max())
    if M <= 0 or not math.isfinite(M):
        return None
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f[j]
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        constraints = [A @ g <= b]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        for slv in [cp.SCS, cp.ECOS, cp.CL
P]:
            try:
                if time.monotonic() > stop_time - 3:
                    break
                prob.solve(solver=slv, verbose=False)
                break
            except Exception:
                continue
        if g.value is None:
            return None
        g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        gv = float(g_val.sum())
        if gv < 1e-10 or not math.isfinite(gv):
            return None
        return g_val / gv
    except Exception:
        return None


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 20.0
    if hard <= t0 + 3.0:
        hard = t0 + max(2.0, budget - 1.0)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_fft(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64)
        best_arr /= 64.0
        best_score = float(_score_fft(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_fft(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()

    n_par = int(best_arr.size)

    # Phase 1: LSE on parent length
    p1_end = min(hard - 80.0, t0 + 0.25 * budget)
    if p1_end > t0 + 20.0 and n_par >= 64:
        p1 = best_arr.copy() / float(best_arr.sum())
        try:
            p1 = _lbfgs_optimize(p1, p1_end, consider=consider, fine_only=False)
            if p1 is not None and p1.size == n_par and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 /= float(p1.sum())
                consider(p1)
        except Exception:
            pass

    work = best_arr.copy()
    work /= float(work.sum())

    # Phase 2: Coordinate descent
    p2_end = min(hard - 80.0, t0 + 0.50 * budget)
    if p2_end > time.monotonic() + 15.0:
        try:
            work, _ = _coord_polish(work, p2_end, max_sweeps=500000,
                                    seed=int(seed + 7777), consider=consider)
        except Exception:
            pass
    work = best_arr.copy()
    work /= float(work.sum())

    # Phase 3: LP step
    lp_end = min(hard - 60.0, time.monotonic() + 40.0)
    if lp_end > time.monotonic() + 12.0:
        g_dir = _lp_step(work, lp_end)
        if g_dir is not None and g_dir.size == n_par:
            for ta in [0.3, 0.15, 0.08, 0.04]:
                blended = (1.0 - ta) * work + ta * g_dir
                blended = np.maximum(blended, 0.0)
                bs = float(blended.sum())
                if bs > 0:
                    blended /= bs
                    consider(blended)

    work = best_arr.copy()
    work /= float(work.sum())

    # Phase 4: More coordinate descent
    p4_end = min(hard - 50.0, t0 + 0.75 * budget)
    if p4_end > time.monotonic() + 15.0:
        try:
            work, _ = _coord_polish(work, p4_end, max_sweeps=500000,
                                    seed=int(seed + 123456), consider=consider)
        except Exception:
            pass

    # Phase 5: Fine LSE + coordinate
    p5_end = min(hard - 25.0, t0 + budget - 30.0)
    if p5_end > time.monotonic() + 25.0:
        l5_end = min(p5_end - 25.0, time.monotonic() + 0.4 * max(0.0, p5_end - time.monotonic()))
        if l5_end > time.monotonic() + 12.0:
            work = best_arr.copy() / float(best_arr.sum())
            try:
                p5 = _lbfgs_optimize(work, l5_end, consider=consider, fine_only=True)
                if p5 is not None and p5.size == n_par and np.all(np.isfinite(p5)) and p5.sum() > 0:
                    p5 /= float(p5.sum())
                    consider(p5)
            except Exception:
                pass

        c5_end = p5_end
        if c5_end > time.monotonic() + 12.0:
            work = best_arr.copy()
            work /= float(work.sum())
            try:
                work, _ = _coord_polish(work, c5_end, max_sweeps=300000,
                                        seed=int(seed + 987654), consider=consider)
            except Exception:
                pass

    # Phase 6: Multi-restart with diverse perturbations
    rem = hard - 25.0 - time.monotonic()
    n_restarts = max(1, int(rem // 40)) if rem > 40 else 0
    for r in range(n_restarts):
        rt_end = min(hard - 20.0, time.monotonic() + max(15, rem / max(1, n_restarts) - 5))
        if rt_end <= time.monotonic() + 8:
            break
        rng_r = np.random.default_rng(int(seed + 777777 + r * 13))
        pert = best_arr.copy() * np.exp(rng_r.normal(0, 0.08, n_par, dtype=np.float64))
        pert = np.maximum(pert, 0)
        ps = float(pert.sum())
        if ps > 0:
            pert /= ps
            consider(pert)
            try:
                work_r, _ = _coord_polish(pert, rt_end, max_sweeps=100000,
                                          seed=int(seed + r * 999 + 5000), consider=consider)
            except Exception:
                pass

    # Phase 7: Final coordinate polish on best
    fin_end = min(hard - 15.0, t0 + budget - 12.0)
    if fin_end > time.monotonic() + 10.0:
        work = best_arr.copy() / float(best_arr.sum())
        try:
            work, _ = _coord_polish(work, fin_end, max_sweeps=100000,
                                    seed=int(seed + 555555), consider=consider)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ps = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ps))
            if not math.isfinite(final_sc):
                return [1.0]
            ls = ps
        return ls
    except Exception:
        return [1.0]
