        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    best_p = p.copy()
    best_sc = _score_fft(p)

    stages = (
        (0.08, 0.10),
        (0.03, 0.12),
        (0.01, 0.14),
        (0.004, 0.16),
        (0.0015, 0.18),
        (0.0006, 0.20),
        (0.0002, 0.20),
    )

    start = time.monotonic()
    avail = float(stop_time - start - 6.0)
    if avail <= 5.0:
        return p

    total_w = sum(w for _, w in stages)
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        cum += wt
        end_stage = min(stop_time - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)
        best_u = u.copy()
        best_f = float(lse_func(u, T)[0])
        stale = 0
        while time.monotonic() < end_stage - 1.0:
            maxit = 40
            opts = {
                "maxiter": maxit, "gtol": 1e-14, "ftol": 1e-16,
                "maxls": 30, "maxcor": 30, "maxdisp": 0, "disp": False,
            }
            try:
                res = minimize(lse_func, best_u, args=(T,),
                               method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    fval = float(res.fun)
                    if math.isfinite(fval) and fval < best_f - 1e-16:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, "status", 1) == 0:
                break
        p_cur = _softmax(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur):
            if sc_cur < best_sc:
                best_p = p_cur.copy()
                best_sc = sc_cur
                p = p_cur.copy()
            elif sc_cur > best_sc + 0.003:
                p = best_p.copy()
            else:
                p = p_cur.copy()
            u = p_to_u(p)
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

    return best_p


def _coord_polish(p, stop_time, consider=None, seed=12345):
    """Aggressive coordinate descent with golden section refinement."""
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p /= s

    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    flat = 0
    sweep = 0

    G = 48 if n <= 4000 else 36
    gr = (math.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end].copy()
            infl = 2.0 * p[:Ls].copy()

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            S = float(p.sum())
            x_hi = max(25.0 * v, 50.0 * S / n, 0.04 * S, 1e-14)
            x_hi = min(x_hi, 12.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, S / n)

            # Grid search
            pts = np.linspace(0.0, x_hi, G + 1)
            dw = x_hi / float(G)

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                di_local = 2 * i - start
                if 0 <= di_local < Ls:
                    dval = base[di_local] + w * infl[di_local] + w * w
                    maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Refined grid + golden section
            lo = max(0.0, best_x - 3.0 * dw)
            hi = min(x_hi, best_x + 3.0 * dw)
            if hi > lo:
                fine_pts = np.linspace(lo, hi, 30)
                vals2 = eval_xs(fine_pts)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    best_x = float(fine_pts[imin2])
                    best_val = float(vals2[imin2])

            # Golden section (20 iters)
            g_lo = max(0.0, best_x - dw)
            g_hi = min(x_hi, best_x + dw)
            if g_hi - g_lo > 1e-15:
                a_gs = g_hi - gr * (g_hi - g_lo)
                b_gs = g_lo + gr * (g_hi - g_lo)
                fa = float(eval_xs(np.array([a_gs]))[0])
                fb = float(eval_xs(np.array([b_gs]))[0])
                for _ in range(20):
                    if not math.isfinite(fa):
                        fa = 1e100
                    if not math.isfinite(fb):
                        fb = 1e100
                    if fa < fb:
                        g_hi = b_gs
                        b_gs = a_gs
                        fb = fa
                        a_gs = g_hi - gr * (g_hi - g_lo)
                        fa = float(eval_xs(np.array([a_gs]))[0])
                    else:
                        g_lo = a_gs
                        a_gs = b_gs
                        fa = fb
                        b_gs = g_lo + gr * (g_hi - g_lo)
                        fb = float(eval_xs(np.array([b_gs]))[0])
                    if g_hi - g_lo < 1e-15:
                        break
                gs_x = (g_lo + g_hi) / 2.0
                gv = float(eval_xs(np.array([gs_x]))[0])
                if math.isfinite(gv) and gv < best_val:
                    best_x = gs_x
                    best_val = gv

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= 12.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        # Residual renormalization
        S = float(p.sum())
        if S > 1e-300:
            inv = 1.0 / S
            p *= inv
            c = _conv_self(p)
            np.maximum(c, 0.0, out=c)
            cur_score = float(2.0 * n * c.max())
        else:
            break

        if math.isfinite(cur_score) and cur_score < best_score - 1e-16:
            best_score = cur_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 4.0:
                    # Targeted perturbation: add noise weighted by convolution peak contribution
                    cpeak = float(c.max())
                    pk = int(np.argmax(c))
                    # Perturb coordinates that contribute to peak
                    noise_amt = 0.02 + 0.01 * math.exp(-flat * 0.05)
                    noise = rng.normal(0.0, noise_amt, n)
                    p2 = p * np.exp(noise)
                    p2 = np.maximum(p2, 0.0)
                    ss = float(p2.sum())
                    if ss > 0.0:
                        p2 /= ss
                        sc2 = float(2.0 * n * _conv_self(p2).max())
                        if math.isfinite(sc2) and sc2 < best_score * 1.02:
                            p = p2
                            c = _conv_self(p)
                            cur_score = sc2
                            flat = 0
                        # Also try small harmonic perturbation
                        elif time.monotonic() < stop_time - 6.0:
                            for M_h in (2, 3, 4, 6):
                                x_h = 2.0 * np.pi * np.arange(n) / M_h
                                p3 = p * (1.0 + 0.05 * np.cos(x_h + rng.uniform(0, 2*np.pi)))
                                p3 = np.maximum(p3, 0.0)
                                ss3 = float(p3.sum())
                                if ss3 > 0:
                                    p3 /= ss3
                                    sc3 = float(2.0 * n * _conv_self(p3).max())
                                    if math.isfinite(sc3) and sc3 < best_score * 1.01:
                                        p = p3
                                        c = _conv_self(p)
                                        cur_score = sc3
                                        flat = 0
                                        break
                else:
                    break
        sweep += 1

    # Final recompute
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    try:
        sc = float(2.0 * n * _conv_self(p).max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _subgrad_polish(p, stop_time, consider=None, max_iters=2000):
    """Subgradient + projected gradient descent on max-conv."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p /= s

    best_p = p.copy()
    best_score = float(2.0 * n * _conv_self(p).max())
    base_step = 0.004

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break
        c = _conv_self(p)
        cmax = float(c.max())
        # Subgradient: average over lags within 0.1% of max
        threshold = cmax * 0.9995
        active = np.where(c >= threshold)[0]
        if len(active) > 50:
            active = np.where(c >= cmax * 0.999)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        grad = np.zeros(n)
        for k in active:
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                contrib = p[kj[valid]] + p[valid]
                grad[valid] += contrib
        grad /= len(active)
        # Center to maintain sum
        grad -= np.sum(grad) / n

        step = base_step / (1.0 + 0.01 * it)
        found = False
        for scale in (1.0, 0.5, 0.25, 0.1):
            p_cand = p - scale * step * grad
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = float(2.0 * n * _conv_self(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                found = True
                break
        if not found:
            base_step *= 0.85
            if base_step < 1e-10:
                break
        if it % 100 == 0 and consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(30.0, float(budget_s))
    hard = t0 + budget - 25.0  # safety margin

    parent = _get_parent()
    n = int(parent.size)
    if n < 2:
        parent = np.ones(3072, dtype=np.float64)
        n = 3072
    parent = parent / float(parent.sum())

    best_arr = parent.copy()
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf
    if not math.isfinite(best_score):
        best_arr = np.full(n, 1.0 / n)
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_arr = a.copy()
            best_score = sc

    # Phase 1: LSE optimization (40% of time)
    phase1_end = t0 + 0.38 * (hard - t0)
    now = time.monotonic()
    if phase1_end > now + 20.0:
        try:
            p1 = _lse_optimize(best_arr, phase1_end, consider=consider)
            if p1 is not None and p1.size == n and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 /= float(p1.sum())
                consider(p1)
                best_arr = p1.copy()
        except Exception:
            pass

    # Phase 2: Coordinate descent (50% of time)
    phase2_end = t0 + 0.88 * (hard - t0)
    now = time.monotonic()
    if phase2_end > now + 20.0:
        try:
            p2, _ = _coord_polish(best_arr, phase2_end, consider=consider, seed=int(seed))
            if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p2 /= float(p2.sum())
                consider(p2)
                best_arr = p2.copy()
        except Exception:
            pass

    # Phase 3: Subgradient polish + final coord (10%)
    phase3_end = hard - 5.0
    now = time.monotonic()
    if phase3_end > now + 10.0:
        sg_end = min(phase3_end - 0.4 * (phase3_end - now), now + 0.4 * (phase3_end - now))
        if sg_end > now + 8.0:
            try:
                p3 = _subgrad_polish(best_arr, sg_end, consider=consider, max_iters=1500)
                if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and p3.sum() > 0:
                    p3 /= float(p3.sum())
                    consider(p3)
                    best_arr = p3.copy()
            except Exception:
                pass

        now = time.monotonic()
        c_end = min(phase3_end - 2.0, now + max(10.0, 0.55 * (phase3_end - now)))
        if c_end > now + 5.0:
            try:
                p4, _ = _coord_polish(best_arr, c_end, consider=consider, seed=int(seed + 7777))
                if p4 is not None and p4.size == n and np.all(np.isfinite(p4)) and p4.sum() > 0:
                    p4 /= float(p4.sum())
                    consider(p4)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in out]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                ls = [1.0]
        return ls
    except Exception:
        return [1.0]
