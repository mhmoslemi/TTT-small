def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:

    p = np.where(np.isfinite(p), p, 0.0)

def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(np.asarray(c, dtype=np.float64), 0.0)

def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))

def _score_exact(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    return float(2.0 * n * m / (s * s))

    # sum_j w[i+j] * a[j] = <w[i:], a[0:n]> (dot product of w[i:i+n] and a)
    
    # This is a sliding dot product, which is a cross-correlation
    # cross_corr[i] = sum_j w[i+j] * a[j] for j=0..n-1
    # = (w rev convolve a)... 
    
    # Let me just compute it directly for now (O(n^2)) and optimize later
    # Actually for n=3072, O(n^2) = 9.4M operations, which is fast enough
    
    # Or use FFT:
    # cross_corr = sum_j w[i+j] * a[j]
    # Let b[j] = a[n-1-j] (reversed a)
    # Then sum_j w[i+j] * b[n-1-j]
    # = (w * b)[i + n - 1] (convolution)
    # So cross_corr[i] = conv(w, reversed_a)[i + n - 1]
    
    size = next_fast_len(L + n - 1)
    Fw = rfft(w, size)
    Fa = rfft(a, size)  # a has length n, zero-padded to size
    # Hmm, this isn't quite right for cross-correlation
    # Let me use a different approach
    
    # Actually, the simplest: 
    # sum_j w[i+j] * a[j] for j=0..n-1
    # If I define v = w (length L), and a (length n)
    # Then for each i in 0..n-1: sum_{j=0}^{n-1} v[i+j] * a[j]
    # This is the "sliding dot product" or "cross-correlation"
    # = (v * a_rev)[i + n - 1] where a_rev = a[::-1]
    
    a_rev = a[::-1]
    conv = irfft(Fw * rfft(a_rev, size), size)[:L + n - 1]
    cross_corr = conv[n - 1 : L - 1]  # length n
    
    grad = (2.0 / L) * cross_corr
    return grad

For the coordinate descent, the key insight is:
- When changing a[i] by delta, only c[k] for k in [i, i+n-1] change
- The new score is 2n * max(new_affected_max, unaffected_max) / (S+delta)^2
- We want to find delta that minimizes this

The optimal delta can be found by:
1. Computing the "break points" where the identity of the max element changes
2. Solving the 1D optimization in each interval

But for efficiency, I'll use a set of smart candidate points:
- delta = 0
- delta values that make each affected element equal to the unaffected max
- delta values that minimize the quadratic (at k=2i)
- delta = -a[i] (set to zero)
