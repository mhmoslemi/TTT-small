Now let me think about the LSE optimization. The LSE approximation to the max is:
`LSE(c, T) = T * log(sum(exp(c/T)))` (shifted by max for numerical stability)

Its gradient w.r.t. c is `softmax(c/T)` (the probability weights). And the gradient w.r.t. a is:
`grad_a[i] = sum_k w[k] * dc[k]/da[i]` where `w = softmax(c/T)`

`dc[k]/da[i] = 2*a[k-i]` (if valid)

So `grad_a[i] = 2 * sum_k w[k] * a[k-i]` (valid range)

This is a weighted sum of reversed slices of a, which can be computed via FFT (cross-correlation of w with a).

For the L-BFGS optimization, we parameterize a as softmax(u) to stay on the simplex, and minimize LSE(softmax(u) convolved, T).

The chain rule gives:
`d/d u_j [LSE] = sum_i (da_i/du_j) * dLSE/da_i`

For softmax: `da_i/du_j = a_i * (delta_ij - a_j)`

So the gradient in u-space is: `g_u[j] = a_j * (dLSE/da_j - sum_i a_i * dLSE/da_i)`

                    flat = 0
                else:
                    break

        sweep += 1

    # Final correction
    c = _fft_conv_self(p)
    sc = 2.0 * n * float(c.max())
    if sc < best_score:
        best_score = sc
        best_p = p.copy()

    return best_p, best_score


def lse_optimize(p, stop_time, T_list=None, consider=None):
    """LSE-smoothed optimization on the simplex via L-BFGS-B."""
    n = len(p)
    p = np.maximum(p.astype(np.float64).copy(), 0.0)
    s = float(p.sum())
    if s < 1e-30:
        return p
    p = p / s

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1

    def _cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def _u_from_p(pp):
        uu = np.log(np.maximum(pp, 1e-30))
        return uu - float(np.mean(uu))

    def p_to_u(pp):
        return _u_from_p(pp)

    def lse_objective(uu, T):
        pp = _softmax(uu)
        c = _cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        scaled = (c - cmax) / T
        # Clip for numerical stability
        scaled = np.clip(scaled, -50.0, 50.0)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if sum_exp <= 0 or not math.isfinite(sum_exp):
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            fval = cmax
        else:
            fval = cmax + T * math.log(sum_exp / float(c.size))
            w = exps / sum_exp
        # Gradient of LSE w.r.t. pp: grad_pp = 2 * (w * pp reversed / cross-corr)
        # dc[k]/da[i] = 2*a[k-i] for valid range
        # grad_pp[i] = sum_k w[k] * 2*pp[k-i]
        # This is 2 * cross_correlation(w, pp) with appropriate indexing
        # cross[k] = sum_i w[i] * pp[k-i] ... let's compute directly
        # grad_pp[i] = 2 * sum_{k: 0<=k-i<L, 0<=k<L} w[k] * pp[k-i] for valid k-i in [0,n-1]
        # = 2 * (cross_corr of w and pp) at shift i
        # Use FFT: convolve(w, pp_reversed) gives the cross-correlation
        w_pad = np.zeros(conv_size)
        w_pad[:len(w)] = w
        pp_pad = np.zeros(conv_size)
        pp_pad[:len(pp)] = pp[::-1]  # reversed for cross-correlation
        Wf = rfft(w_pad, conv_size)
        Pf = rfft(pp_pad, conv_size)
        cross = irfft(Wf * Pf, conv_size)[:conv_len]
        # cross[k] = sum_j w[j] * pp_rev[k-j] = sum_j w[j] * pp[n-1-(k-j)]
        # Hmm, this is getting complicated. Let me just do it directly for the gradient.
        # grad_pp[i] = sum_k w[k] * dc[k]/da[i] = sum_k w[k] * 2*pp[k-i] (valid)
        # = 2 * sum_k w[k] * pp[k-i] where 0<=k-i<n and 0<=k<2n-1
        # For each i: sum over k from max(0,i) to min(2n-2, i+n-1) of w[k]*pp[k-i]
        # This is a cross-correlation: grad_pp = 2 * np.correlate(w, pp, 'full')[n-1:2n-1]
        # Actually: np.correlate(w, pp, 'full')[k] = sum_j w[k+j-n+1] * pp[j] ... it's messy
        # Let me just use the FFT approach properly.
        # conv(w, pp) with w of length 2n-1, pp of length n:
        # conv_result[k] = sum_j w[j]*pp[k-j] for valid j,k
        # We want: for each i, sum_k w[k]*pp[k-i] = conv(w, pp)[2n-1-i] ??? 
        # Actually let me just compute it directly since |w| support is ~n and we need n outputs.
        
        # Direct computation using slicing:
        w_arr = w  # length 2n-1
        pp_arr = pp  # length n
        grad_pp = np.zeros(n)
        for k in range(2 * n - 1):
            if w_arr[k] < 1e-18:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad_pp[lo:hi + 1] += 2.0 * w_arr[k] * pp_arr[k - hi:k - lo + 1][::-1]
        
        # Chain rule: grad_u = pp * (grad_pp - dot(pp, grad_pp))
        tdp = float(np.dot(pp_arr, grad_pp))
        grad_u = pp_arr * (grad_pp - tdp)
        grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
        return float(fval), grad_u

    if T_list is None:
        c0 = _cfast(p)
        cmax0 = float(c0.max())
        T_list = [0.004 * cmax0, 0.001 * cmax0, 0.0003 * cmax0, 0.00008 * cmax0, 0.00002 * cmax0]

    u = p_to_u(p)
    best_u = u.copy()
    best_f = math.inf
    best_p_out = p.copy()

    now = time.monotonic()
    total_time = stop_time - now - 3.0
    if total_time <= 5:
        return best_p_out

    n_stages = len(T_list)
    per_stage = total_time / n_stages

    for si, T in enumerate(T_list):
        stage_end = min(stop_time - 2.0, now + (si + 1) * per_stage)
        if stage_end <= time.monotonic() + 3.0:
            continue
        f0, _ = lse_objective(u, T)
        if not math.isfinite(f0):
            continue
        best_f_local = f0
        stale = 0
        max_iters = 8

        for it in range(max_iters):
            if time.monotonic() > stage_end - 1.0:
                break
            try:
                res = minimize(
                    lse_objective, u, args=(T,),
                    method='L-BFGS-B', jac=True,
                    options={'maxiter': 40, 'gtol': 1e-10, 'ftol': 1e-14,
                             'maxls': 20, 'maxcor': 25, 'disp': False, 'maxdisp': 0}
                )
            except Exception:
                break
            if res.x is None:
                break
            x_new = np.asarray(res.x, dtype=np.float64)
            if x_new.size != n or not np.all(np.isfinite(x_new)):
                break
            f_new, _ = lse_objective(x_new, T)
            if math.isfinite(f_new) and f_new < best_f_local - 1e-12:
                best_f_local = f_new
                u = x_new.copy()
                stale = 0
            else:
                stale += 1
                if stale >= 3:
                    break
            if getattr(res, 'status', 1) == 0:
                break

        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        pc = float(p_cur.sum())
        if pc > 1e-30:
            p_cur /= pc
        sc_cur = _score_simplex(p_cur)
        if math.isfinite(sc_cur):
            if consider is not None:
                try:
                    consider(p_cur)
                except Exception:
                    pass
            p = p_cur.copy()
            u = p_to_u(p)

    return _softmax(best_u) if best_u is not None else p


def mirror_descent(p, n_iters=200, eta_init=0.5, n_top=8):
    """Mirror descent on simplex for fast coarse optimization."""
    n = len(p)
    p = np.maximum(p.copy(), 0.0)
    s = float(p.sum())
    if s < 1e-30:
        p = np.ones(n) / n
        return p
    p = p / s

    for it in range(n_iters):
        c = _fft_conv_self(p)
        cmax = float(c.max())
        # Active set: top n_top coefficients
        if n_top < len(c):
            top_idx = np.argpartition(c, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(c))
        # Gradient
        g = np.zeros(n)
        for k in top_idx:
            if c[k] < 0.9 * cmax:  # skip if not truly active
                continue
            lo = max(0, int(k) - n + 1)
            hi = min(n - 1, int(k))
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * p[int(k) - hi:int(k) - lo + 1][::-1]
        gmax = float(np.max(g))
        if gmax < 1e-30:
            break
        eta = eta_init / (1.0 + 0.01 * it)
        eta = min(eta, 0.3 / gmax)
        p = p * np.exp(-eta * g)
        p = np.maximum(p, 0.0)
        p = p / p.sum()

    return p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    hard = t0 + float(budget_s) - 15.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(3072, dtype=np.float64)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.sum() < 0.01 or parent.size == 0:
        parent = np.ones(3072, dtype=np.float64)

    n = len(parent)
    p = parent / parent.sum()

    best_p = p.copy()
    best_score = _score_simplex(p)

    def consider(arr):
        nonlocal best_p, best_score
        a = np.asarray(arr, dtype=np.float64).ravel()
        a = np.clip(np.where(np.isfinite(a), a, 0.0), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        a_norm = a / a.sum()
        sc = _score_simplex(a_norm)
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_p = a_norm.copy()
            best_score = sc

    # Phase 1: Quick mirror descent (30s)
    phase1_end = min(hard - 100, t0 + 30)
    if phase1_end > t0 + 10 and n >= 128:
        try:
            p_md = mirror_descent(p, n_iters=300, eta_init=0.4, n_top=6)
            sc_md = _score_simplex(p_md)
            if math.isfinite(sc_md) and sc_md < best_score:
                consider(p_md)
            # Blend
            for alpha in [0.3, 0.5, 0.7]:
                blend = (1 - alpha) * p + alpha * p_md
                blend = np.maximum(blend, 0)
                bs = blend.sum()
                if bs > 0:
                    blend /= bs
                    consider(blend)
        except Exception:
            pass

    p = best_p.copy()

    # Phase 2: Coordinate descent (main, ~350s)
    phase2_end = min(hard - 200, t0 + 400)
    if phase2_end > time.monotonic() + 30:
        try:
            p, _ = coord_descent(p, phase2_end, seed=int(seed + 1000))
            consider(p)
        except Exception:
            pass

    p = best_p.copy()

    # Phase 3: LSE optimization (~150s)
    phase3_end = min(hard - 150, time.monotonic() + 150)
    if phase3_end > time.monotonic() + 20:
        try:
            p_lse = lse_optimize(p, phase3_end, consider=consider)
            if p_lse is not None and np.all(np.isfinite(p_lse)) and p_lse.sum() > 0:
                consider(p_lse)
        except Exception:
            pass

    p = best_p.copy()

    # Phase 4: Coordinate descent again (~250s)
    phase4_end = min(hard - 100, time.monotonic() + 250)
    if phase4_end > time.monotonic() + 20:
        try:
            p, _ = coord_descent(p, phase4_end, seed=int(seed + 2000))
            consider(p)
        except Exception:
            pass

    p = best_p.copy()

    # Phase 5: Fine LSE (~80s)
    phase5_end = min(hard - 60, time.monotonic() + 80)
    if phase5_end > time.monotonic() + 15:
        try:
            c0 = _fft_conv_self(p)
            cmax0 = float(c0.max())
            T_list = [0.0003 * cmax0, 0.0001 * cmax0, 0.00003 * cmax0, 0.00001 * cmax0]
            p_lse = lse_optimize(p, phase5_end, T_list=T_list, consider=consider)
            if p_lse is not None:
                consider(p_lse)
        except Exception:
            pass

    p = best_p.copy()

    # Phase 6: Final coordinate squeeze
    phase6_end = min(hard - 20, t0 + budget_s - 20)
    if phase6_end > time.monotonic() + 15:
        try:
            p, _ = coord_descent(p, phase6_end, seed=int(seed + 3000))
            consider(p)
        except Exception:
            pass

    # Final: ensure validity and return
    out = np.asarray(best_p, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s_out = float(out.sum())
    if s_out < 0.01 or out.size == 0:
        out = parent.copy()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.sum() < 0.01:
            out = np.full(64, 1.0 / 64.0)

    # Scale to reasonable magnitude (avoids underflow/overflow in evaluator)
    # The score is scale-invariant, so any positive scaling works.
    # Use scaling so that max(out) ~ 1 to stay well within [0, 1000].
    if out.max() > 0:
        scale = 1.0 / out.max()
        out = out * scale
        if out.max() > 1000:
            out = out * (1000.0 / out.max())

    # Final validation with exact evaluator
    lst = [float(x) for x in out]
    try:
        final_sc = evaluate_sequence(lst)
        if not math.isfinite(final_sc):
            # Fallback to parent
            lst = [float(x) for x in parent]
            final_sc = evaluate_sequence(lst)
            if not math.isfinite(final_sc):
                return [1.0]
    except Exception:
        lst = [float(x) for x in parent]
        try:
            evaluate_sequence(lst)
        except Exception:
            return [1.0]

    return lst
