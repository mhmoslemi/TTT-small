```python
def _lp_dir(f, M, timeout=60):
    """Solve LP: max 1'g s.t. A g <= M*1, g >= 0. Return normalized g or None."""
    n = f.size
    if n < 32 or n > 5000: return None
    if not math.isfinite(M) or M <= 0: return None
    try:
        # Build sparse A: A[k,m] = f[k-m] if 0<=k-m<n else 0
        rows = []
        cols = []
        data = []
        for k in range(2*n-1):
            lo = max(0, k-n+1)
            hi = min(n-1, k)
            if lo > hi: continue
            j = k - np.arange(lo, hi+1)  # f indices
            rows.extend([k]*(hi-lo+1))
            cols.extend(range(lo, hi+1))
            data.extend(f[j].tolist())
        A = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
        cobj = -np.ones(n)
        b_ub = np.full(2*n-1, M)
```

Let me reconsider the LP approach. The LP:
max 1'g
s.t. A g <= M * 1

The dual is:
min M * 1'y
s.t. A'y >= 1
     y >= 0

where y has 2n-1 components. This dual has 2n-1 variables and n constraints. The constraint matrix A' is n x (2n-1), with the same nonzeros.

Either way, the LP is the same size. Let me stick with the primal.

Now, for the Frank-Wolfe iteration:
1. Given current a (normalized), compute c = a*a, M = max(c)
2. Solve LP to get g
3. a_new = (1-t)*a + t*g, normalize
4. Check if score improved

The step size t can be chosen by line search or by the FW rule t = 2/(k+2).

Let me also think about a simpler but potentially very effective approach: **iterative water-filling on the autocorrelation**. 

The idea: we want b_k to be as flat as possible. At each step:
1. Compute b = a*a

3. Adjust a to reduce energy at those lags

This is related to the "least mean" autocorrelation method but adapted for nonnegative sequences.

```python
def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        return src.copy()
    m = src.size
    out = np.interp((np.arange(n) + 0.5) / n, (np.arange(m) + 0.5) / m, src)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def _softmax(u):
    u = np.asarray(u, dtype=np.float64).ravel()

        return u
    u = np.where(np.isfinite(u), u, 0.0)
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0 / u.size)
    return e / s
```

```python
def _lbfgs_opt(p, stop_time, consider=None, fine_only=False):
    p = _norm(p)
    n = int(p.size)
    if n <= 1 or time.monotonic() > stop_time:
        return p

    conv_len = 2 * n - 1
    conv_size = next_fast_len(conv_len)
    cross_len = 3 * n - 2
    cross_size = next_fast_len(cross_len)

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return np.maximum(c, 0.0)

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        return np.maximum(q, 0.0)

    def lse_func(uu, T):
        if uu.size != n:
            return 1e100, np.zeros(n)
        uu = np.where(np.isfinite(uu), uu, 0.0)
        m = float(np.max(uu))
        e = np.exp(uu - m)
        s = float(e.sum())
        if not math.isfinite(s) or s <= 0:
            pp = np.full(n, 1.0 / n)
        else:
            pp = e / s
            ss = float(pp.sum())
            pp = pp / ss if math.isfinite(ss) and ss > 0 else np.full(n, 1.0 / n)

        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros(n)

        if not math.isfinite(T) or T <= 0:
            T = max(1e-14, 1e-4 * cmax)
        T = max(1e-14, T)

        scaled = (c - cmax) / T
        # Clip to avoid overflow
        scaled = np.clip(scaled, -50, 50)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            w = exps / sum_exp
            f = cmax + T * math.log(sum_exp / conv_len)
            if not math.isfinite(f):
                f = cmax

        q = xfast(pp, w)
        gid = 2.0 * q
        gd = float(np.dot(pp, gid))
        grad = pp * (gid - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    u = _p2u(p)
    best_p = p.copy()
    best_sc = _fscore(p)
    if consider:
        try: consider(best_p)
        except Exception: pass

    if fine_only:
        stages = ((0.0010, 0.22), (0.00025, 0.30), (0.00006, 0.48))
    else:
        stages = (
            (0.018, 0.12),
            (0.006, 0.15),
            (0.0016, 0.18),
            (0.00045, 0.20),
            (0.00012, 0.35),
        )

    start = time.monotonic()
    avail = float(stop_time - start - 5.0)
    if avail <= 5.0:
        return best_p

    total_w = sum(w for _, w in stages)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 6.0:
            break
        cum += wt
        end_stage = min(stop_time - 5.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:
            continue

        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-14, frac * cmax)

        # Run L-BFGS chunk
        try:
            f0, _ = lse_func(u, T)
        except Exception:
            continue
        if not math.isfinite(f0):
            continue
        best_u = u.copy()
        best_f = f0
        stale = 0

        while time.monotonic() < end_stage - 2.0:
            maxit = 16 if n > 2000 else 25
            try:
                res = minimize(lse_func, best_u, args=(T,), method='L-BFGS-B', jac=True,
                               options={'maxiter': maxit, 'gtol': 1e-9, 'ftol': 1e-14,
                                        'maxls': 16, 'maxcor': 15, 'disp': False})
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    fval = float(res.fun) if hasattr(res, 'fun') else math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-13:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 2:
                            break
            if getattr(res, 'status', 1) == 0:
                break

        u = best_u
        p_new = _softmax(u)
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if math.isfinite(ss) and ss > 0:
            p_new /= ss
        else:
            p_new = best_p.copy()

        sc_new = _fscore(p_new)
        if math.isfinite(sc_new) and sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_p = p_new.copy()
            p = p_new.copy()
            if consider:
                try: consider(best_p)
                except Exception: pass
        else:
            p = best_p.copy()
        u = _p2u(p)

    return best_p
```

```python
def _coord_polish(p, stop_time, G=64, seed=12345, consider=None):
    p = _norm(p)
    n = int(p.size)
    if n <= 1:
        return p, math.inf

    c = _fconv(p)
    S = 1.0
    infl = 2.0 * p  # influence vector: if we add d to p[i], c changes by d * 2p shifted by i
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    if consider:
        try: consider(best_p)
        except Exception: pass

    rng = np.random.default_rng(seed)
    flat = 0

    while time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)

        for pos in range(n):
            if time.monotonic() > stop_time - 0.5:
                break
            if S <= 1e-300:
                break

            # Renormalize if drift
            if S > 1.2 or S < 0.8:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                infl *= inv
                S = 1.0
                c = np.maximum(c, 0.0)
                cur_score = float(2.0 * n * c.max())

            i = int(order[pos])
            v = float(p[i])
            L = 2 * n - 1
            start = i
            end = min(i + n, L)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end]
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const: max_const = mc
            if end < L:
                mc = float(c[end:].max())
                if mc > max_const: max_const = mc
            if not math.isfinite(max_const): max_const = 0.0

            # Build candidate x values
            xhi = max(20.0 * v, 20.0 / float(n), 0.004)
            xhi = min(xhi, max(0.05, v))
            if not math.isfinite(xhi) or xhi <= 0:
                xhi = max(1e-8, 10.0 / float(n))

            pts = [0.0, v, xhi]
            minx = max(1e-9, 1e-5 * v, 1.0 / (2000.0 * n))
            if minx < xhi:
                m1 = max(12, int(G * 0.4))
                pts.extend(np.geomspace(minx, xhi, m1).tolist())
            if v > 1e-12:
                m2 = max(8, int(G * 0.25))
                pts.extend((v * np.geomspace(0.2, 5.0, m2)).tolist())
            m3 = max(10, int(G * 0.35))
            linhi = min(xhi, max(8.0 / float(n), 0.0015))
            pts.extend(np.linspace(0.0, linhi, m3).tolist())

            pts = np.unique(np.clip(np.concatenate([pts]), 0, xhi))
            if pts.size == 0:
                pts = np.array([v])

            # Evaluate score for each candidate x
            di = 2 * i - start
            den = S + (pts - v)
            valid = den > 1e-14
            aff = base[None, :] + (pts - v)[:, None] * infl[None, :]
            if 0 <= di < aff.shape[1]:
                aff[:, di] = base[di] + (pts - v) * infl[di] + (pts - v) ** 2
            np.maximum(aff, max_const, out=aff)
            mv = np.max(aff, axis=1)
            vals = np.full(pts.size, math.inf)
            vals[valid] = (2.0 * n * mv[valid]) / (den[valid] ** 2)
            vals = np.where(np.isfinite(vals), vals, math.inf)

            if not np.any(np.isfinite(vals)):
                continue
            im = int(np.argmin(vals))
            best_x = float(pts[im])
            best_val = float(vals[im])

            # Refine around best
            if best_val < cur_score + max(1e-10, 1e-5 * cur_score):
                if best_x <= 0:
                    lo, hi = 0.0, max(5.0 / n, 5.0 * v, 1e-7)
                else:
                    step = max(v, 1.0 / n, 1e-9)
                    lo = max(0.0, best_x * 0.5, best_x - 2.5 * step)
                    hi = min(xhi, max(best_x * 1.5, best_x + 2.5 * step))
                if hi <= lo: hi = lo + 1e-9
                fine = np.linspace(lo, hi, 40)
                if lo > 0:
                    fine = np.concatenate([fine, np.geomspace(lo, hi, 20)])
                else:
                    fine = np.concatenate([fine, np.linspace(0, hi, 20)])
                fine = np.unique(np.clip(fine, 0, xhi))

                den2 = S + (fine - v)
                valid2 = den2 > 1e-14
                aff2 = base[None, :] + (fine - v)[:, None] * infl[None, :]
                if 0 <= di < aff2.shape[1]:
                    aff2[:, di] = base[di] + (fine - v) * infl[di] + (fine - v) ** 2
                np.maximum(aff2, max_const, out=aff2)
                mv2 = np.max(aff2, axis=1)
                vals2 = np.full(fine.size, math.inf)
                vals2[valid2] = (2.0 * n * mv2[valid2]) / (den2[valid2] ** 2)
                vals2 = np.where(np.isfinite(vals2), vals2, math.inf)
                if np.any(np.isfinite(vals2)):
                    im2 = int(np.argmin(vals2))
                    if vals2[im2] < best_val:
                        best_x = float(fine[im2])
                        best_val = float(vals2[im2])

            if (math.isfinite(best_val) and best_val < cur_score - 1e-11
                    and 0 <= best_x <= xhi):
                d = best_x - v
                if abs(d) > 1e-16:
                    c[start:end] += d * infl
                    diag = 2 * i
                    if 0 <= diag < L:
                        c[diag] += d * d
                    p[i] = best_x
                    infl[i] = 2.0 * best_x
                    S += d
                    cur_score = best_val
                    improved = True

        # End full sweep
```

```python
def _lp_iterate(a, stop_time, consider=None, max_iters=100):
    """Frank-Wolfe iterations using LP."""
    a = _norm(a)
    n = a.size
    best_a = a.copy()
    best_sc = _fscore(a)
    if consider:
        try: consider(best_a)
        except Exception: pass

        if time.monotonic() > stop_time - 5:
            break

        c = _fconv(a)
        M = float(c.max())

            break

        if g is None:
            break

        # Line search
        improved = False
        for t in (0.5, 0.35, 0.25, 0.15, 0.10, 0.05):
            cand = (1 - t) * a + t * g
            cand = np.maximum(cand, 0)
            sc = _fscore(cand)
            if math.isfinite(sc) and sc < best_sc - 1e-10:
                a = _norm(cand)
                best_sc = sc
                best_a = a.copy()
                improved = True
                if consider:
                    try: consider(best_a)
                    except Exception: pass
                break

        if not improved:
            # Try smaller steps
            for t in (0.03, 0.02, 0.01, 0.005):
                cand = (1 - t) * a + t * g
                cand = np.maximum(cand, 0)
                sc = _fscore(cand)
                if math.isfinite(sc) and sc < best_sc - 1e-12:
                    a = _norm(cand)
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    if consider:

                        except Exception: pass
                    break
```

```python
def _subgrad_dir(a, cross_size, cross_len):
    """Compute subgradient of max_k c[k] w.r.t. a, averaged over active lags."""
    n = a.size
    c = _fconv(a)
    cmax = float(c.max())
    thresh = cmax * 0.995
    w = np.where(c >= thresh, 1.0, 0.0)
    w /= w.sum()
    
    F = rfft(a[::-1], cross_size)
    G = rfft(w, cross_size)
    q = irfft(F * G, cross_size)[:cross_len]
    ngrad = 2.0 * q[n-1:2*n-1]
    return ngrad, cmax
```

```python
def _fconv(a):

    if n == 0:

    F = rfft(a, sz)
    c = irfft(F * F, sz)[:L]

def _fscore(a):
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)

    c = _fconv(a)

def _softmax(u):
    u = np.asarray(u, dtype=np.float64).ravel()
    if u.size == 0:
        return u
    u = np.where(np.isfinite(u), u, 0.0)
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _p2u(p):
    p = np.maximum(p, 1e-15)
    u = np.log(p)
    u -= u.mean()

def _lp_dir(f, M, tl=20.0):
    """Solve primal LP: max 1'g s.t. A(f) g <= M, g >= 0."""
    if linprog is None or sp is None:
        return None
    n = f.size
    if n < 32 or n > 5000:
        return None
    if not math.isfinite(M) or M <= 0:
        return None
    try:
        # Build W matrix A[f]
        R = np.repeat(np.arange(n, dtype=np.int32), n)
        C = np.tile(np.arange(n, dtype=np.int32), n)
        off = R - C
        D = np.tile(f, n)
        A = sp.coo_matrix(
            (D, (R, C)), shape=(2 * n - 1, n)).tocsr()
        A.eliminate_zeros()
        cobj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        t0 = time.monotonic()
        res = linprog(cobj, A_ub=A, b_ub=b_ub,
                      bounds=(0, None), method='highs',
                      options={'time_limit': tl})
        dt = time.monotonic() - t0
        if dt > tl + 10:
            return None
        if (res is not None and getattr(res, 'success', False)
                and res.x is not None):
            g = np.asarray(res.x, dtype=np.float64)
            g = np.maximum(g, 0.0)
            s = float(g.sum())
            if s > 1e-30:
                return g / s
    except Exception:
        pass
    return None

        else:
            break
    return best_a, best_sc
```
