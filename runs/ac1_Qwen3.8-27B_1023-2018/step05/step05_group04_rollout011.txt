                    cur = max(float(aff_seg.max()), mc)
                    score = 2.0 * n * cur
                    improved = True

        # Renormalize
        S_new = float(p.sum())
        if S_new > 1e-300 and math.isfinite(S_new):
            p /= S_new
            c = _conv_self(p)
            nsc = 2.0 * n * float(c.max())
            if math.isfinite(nsc) and nsc < best_score - 1e-14:
                best_score = nsc
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass
            cur = float(c.max())
            score = 2.0 * n * cur

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 15.0:
                    ks = 0.015 + 0.04 * min(1.0, flat / 15.0)
                    p = np.maximum(p * np.exp(rng.normal(0, ks, n)), 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur = float(c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    return best_p, best_score

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 35.0
    if hard < t0 + 10.0:
        hard = t0 + budget - 5.0

    parent = _get_parent()
    best_arr = parent.copy().astype(np.float64)
    best_score = float(_score_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            sc = float(_score_arr(a))
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_arr = a.copy()
        except Exception:
            pass

    consider(parent)
    n_work = int(best_arr.size)

    # Phase 1: LSE optimization (up to 120s)
    lse_budget = min(120.0, 0.12 * budget)
    lse_end = t0 + lse_budget
    p_norm = np.maximum(best_arr.copy(), 0.0)
    p_norm /= float(p_norm.sum())
    if lse_end > t0 + 20.0:
        try:
            p2 = _lse_optimize(p_norm, lse_end, consider=consider)
            if p2 is not None and p2.size == n_work and p2.sum() > 1e-100:
                p_norm = p2 / float(p2.sum())
                consider(p_norm)
        except Exception:
            pass

    # Phase 2: Main coordinate descent (most of remaining time)
    cd_end = hard - 8.0
    if cd_end > time.monotonic() + 40.0:
        try:
            p_cd, _ = _coord_descent(p_norm, cd_end, consider=consider,
                                     seed=int(seed) + 1000)
            if p_cd is not None and p_cd.sum() > 1e-100:
                consider(p_cd)
        except Exception:
            pass

    # Phase 3: If time remains, try a perturbed restart
    rem = hard - 6.0 - time.monotonic()
    if rem > 80.0:
        rng2 = np.random.default_rng(int(seed) + 99999)
        p_kick = np.maximum(p_norm * np.exp(rng2.normal(0, 0.05, n_work)), 0.0)
        p_kick /= float(p_kick.sum())
        kick_end = min(hard - 6.0, time.monotonic() + rem * 0.7)
        if kick_end > time.monotonic() + 25.0:
            try:
                p_k, _ = _coord_descent(p_kick, kick_end, consider=consider,
                                        seed=int(seed) + 20000)
                if p_k is not None and p_k.sum() > 1e-100:
                    consider(p_k)
            except Exception:
                pass

    # Phase 4: Final squeeze on best
    rem = hard - 5.0 - time.monotonic()
    if rem > 30.0:
        p_final = np.maximum(best_arr.copy(), 0.0)
        p_final /= float(p_final.sum())
        try:
            p_f, _ = _coord_descent(p_final, hard - 5.0, consider=consider,
                                    seed=int(seed) + 30000)
            if p_f is not None and p_f.sum() > 1e-100:
                consider(p_f)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.ones(64, dtype=np.float64)
        ls = [float(x) for x in out]
        fs = float(evaluate_sequence(ls))
        if not math.isfinite(fs):
            ls = [float(x) for x in parent]
            fs = float(evaluate_sequence(ls))
            if not math.isfinite(fs):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
