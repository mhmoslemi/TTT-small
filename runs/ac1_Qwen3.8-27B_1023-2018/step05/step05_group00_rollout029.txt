                "ftol": 1e-16,
                "maxls": 30,
                "maxcor": 40,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(
                    lse_func, best_u, args=(T,),
                    method="L-BFGS-B", jac=True, options=opts,
                )
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == uu.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 4:
                            break
            if getattr(res, "status", 1) == 0:
                if time.monotonic() > end_stage - 3.0:
                    break
        return best_u

    if fine_only:
        stages = ((0.001, 0.30), (0.0004, 0.40), (0.0001, 0.30))
    else:
        stages = (
            (0.12, 0.08),
            (0.04, 0.10),
            (0.012, 0.12),
            (0.003, 0.14),
            (0.0008, 0.16),
            (0.0002, 0.16),
            (0.00005, 0.14),
            (0.00001, 0.20),
        )

    start = time.monotonic()
    avail = float(stop_time - start - 10.0)
    if avail <= 5.0:
        return p

    best_p = p.copy()
    best_sc = _score_fft(p)
    total_w = float(sum(w for _, w in stages))
    if total_w <= 0.0:
        total_w = 1.0

    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 4.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        sc_cur = _score_fft(p_cur)
        if not math.isfinite(sc_cur):
            p = best_p.copy()
            u = p_to_u(p)
            continue
        if sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif sc_cur > best_sc + max(0.001, 1e-4 * best_sc):
            p = best_p.copy()
        else:
            p = p_cur.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_descent(p, stop_time, max_sweeps=100000, seed=12345, consider=None,
                   use_pairs=True, pair_ratio=0.5):
    """Coordinate descent with optional pair updates."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p, math.inf
    p = p / s

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_max = float(c.max())
    best_score = float(2.0 * n * cur_max)
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    G_single = 48
    G_pair = 40

    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)

        # Determine which indices to do as pairs
        if use_pairs and pair_ratio > 0:
            n_pairs = int(np.floor(n * pair_ratio))
            pair_indices = list(order[:n_pairs])
            single_indices = list(order[n_pairs:])
        else:
            pair_indices = []
            single_indices = list(order)

        # --- Single coordinate updates ---
        for idx_i in single_indices:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue
            base = c[start_i:end_i].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start_i > 0:
                ml = float(c[:start_i].max())
                if ml > max_const:
                    max_const = ml
            if end_i < Ltot:
                mr = float(c[end_i:].max())
                if mr > max_const:
                    max_const = mr

            avg = S / float(n)
            x_hi = max(20.0 * v, 60.0 * avg, 0.03 * S, 1e-14)
            x_hi = min(x_hi, 12.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, avg)

            pts = np.linspace(0.0, x_hi, G_single + 1, dtype=np.float64)
            if 0.0 <= v <= x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag = i - start_i

            w = pts - v
            denom = S + w
            aff = base[None, :] + w[:, None] * infl[None, :]
            maff = np.max(aff, axis=1)
            dval = base[diag] + w * infl[diag] + w * w
            maff = np.maximum(maff, dval)
            maff = np.maximum(maff, max_const)
            dv = denom
            vals = np.where(dv > 1e-15, (2.0 * n * maff) / (dv * dv), np.inf)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Refine
            dw = x_hi / float(G_single)
            lo = max(0.0, best_x - 2.0 * dw)
            hi = min(x_hi, best_x + 2.0 * dw)
            if hi <= lo:
                hi = lo + 1e-12
            fine = np.linspace(lo, hi, 32, dtype=np.float64)
            if 0.0 <= best_x <= x_hi:
                fine = np.concatenate((fine, [best_x]))
                fine = np.unique(fine)
            w2 = fine - v
            denom2 = S + w2
            aff2 = base[None, :] + w2[:, None] * infl[None, :]
            maff2 = np.max(aff2, axis=1)
            dval2 = base[diag] + w2 * infl[diag] + w2 * w2
            maff2 = np.maximum(maff2, dval2)
            maff2 = np.maximum(maff2, max_const)
            dv2 = denom2
            vals2 = np.where(dv2 > 1e-15, (2.0 * n * maff2) / (dv2 * dv2), np.inf)
            if np.any(np.isfinite(vals2)):
                imin2 = int(np.argmin(vals2))
                if float(vals2[imin2]) < best_val:
                    best_x = float(fine[imin2])
                    best_val = float(vals2[imin2])

            if math.isfinite(best_val) and best_val < cur_max * 2.0 * n / (S * S) - 1e-14 and 0.0 <= best_x <= 12.0 * S:
                w_best = best_x - v
                if abs(w_best) > 1e-15:
                    p[i] = best_x
                    S_total += w_best
                    c[start_i:end_i] += w_best * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w_best * w_best
                    cur_max = float(c.max())
                    improved = True

        # --- Pair coordinate updates ---
        if use_pairs and pair_indices:
            # Group into pairs
            for k in range(0, len(pair_indices) - 1, 2):
                if time.monotonic() > stop_time - 0.5:
                    break
                i = int(pair_indices[k])
                j = int(pair_indices[k + 1])
                if i == j:
                    continue
                if S_total <= 1e-300:
                    break
                if S_total > 1.3 or S_total < 0.7:
                    inv = 1.0 / S_total
                    p *= inv
                    c *= inv * inv
                    S_total = 1.0
                S = float(S_total)
                vi = float(p[i])
                vj = float(p[j])

                # Transfer δ from j to i: p[i] += δ, p[j] -= δ
                # δ range: [-vj, vi]
                d_lo = -vj
                d_hi = vi

                if d_hi - d_lo < 1e-15:
                    continue

                # Compute l[k] = a[k-i] - a[k-j] for all k
                l_vec = np.zeros(Ltot, dtype=np.float64)
                # a[k-i] for i <= k < i+n
                l_vec[i:i + n] += p
                # a[k-j] for j <= k < j+n
                l_vec[j:j + n] -= p

                # q[k] = 1_{k=2i} + 1_{k=2j} - 2*1_{k=i+j}
                q_vec = np.zeros(Ltot, dtype=np.float64)
                if 2 * i < Ltot:
                    q_vec[2 * i] += 1.0
                if 2 * j < Ltot:
                    q_vec[2 * j] += 1.0
                if (i + j) < Ltot:
                    q_vec[i + j] -= 2.0

                # Non-affected max (for lags where l_vec and q_vec are both 0)
                mask_affected = (l_vec != 0.0) | (q_vec != 0.0)
                if np.any(~mask_affected):
                    const_max = float(c[~mask_affected].max())
                else:
                    const_max = 0.0

                # Grid search over δ
                G = G_pair
                dnts = np.linspace(d_lo, d_hi, G, dtype=np.float64)
                # c_new[g, k] = c[k] + 2*dnts[g]*l_vec[k] + dnts[g]^2*q_vec[k]
                # Max over k for each g
                c_new_max = np.empty(G, dtype=np.float64)
                for g_idx in range(G):
                    d = dnts[g_idx]
                    c_trial = c + 2.0 * d * l_vec + d * d * q_vec
                    mn = float(c_trial.max())
                    if const_max > mn:
                        mn = const_max
                    c_new_max[g_idx] = mn

                # Score is 2*n*c_new_max[g] (since S=1 after normalization)
                # We want to minimize c_new_max
                imin = int(np.argmin(c_new_max))
                best_d = float(dnts[imin])
                best_cmax = float(c_new_max[imin])

                # Refine
                dd = (d_hi - d_lo) / float(G)
                rlo = max(d_lo, best_d - 2.5 * dd)
                rhi = min(d_hi, best_d + 2.5 * dd)
                if rhi <= rlo:
                    rhi = rlo + 1e-12
                fine_d = np.linspace(rlo, rhi, 24, dtype=np.float64)
                fine_cmax = np.empty(fine_d.size, dtype=np.float64)
                for g_idx in range(fine_d.size):
                    d = fine_d[g_idx]
                    c_trial = c + 2.0 * d * l_vec + d * d * q_vec
                    mn = float(c_trial.max())
                    if const_max > mn:
                        mn = const_max
                    fine_cmax[g_idx] = mn
                imin_f = int(np.argmin(fine_cmax))
                if fine_cmax[imin_f] < best_cmax:
                    best_d = float(fine_d[imin_f])
                    best_cmax = float(fine_cmax[imin_f])

                # Apply if improved
                cur_score = 2.0 * n * cur_max  # S=1
                new_score = 2.0 * n * best_cmax
                if new_score < cur_score - 1e-14:
                    p[i] = vi + best_d
                    p[j] = vj - best_d
                    # Update convolution
                    c += 2.0 * best_d * l_vec
                    c += best_d * best_d * q_vec
                    np.maximum(c, 0.0, out=c)
                    cur_max = float(c.max())
                    improved = True

        # End of sweep: renormalize
        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        # Recompute c exactly to avoid drift
        c = _conv_self(p)
        np.maximum(c, 0.0, out=c)
        cur_max = float(c.max())
        new_score = 2.0 * n * cur_max
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 8.0:
                    # Perturbation to escape local minimum
                    noise_scale = 0.02
                    p = p * np.exp(rng.normal(0.0, noise_scale, n, dtype=np.float64))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0.0:
                        p /= ss
                    c = _conv_self(p)
                    np.maximum(c, 0.0, out=c)
                    S_total = 1.0
                    cur_max = float(c.max())
                    new_score = 2.0 * n * cur_max
                    if math.isfinite(new_score) and new_score < best_score - 1e-14:
                        best_score = new_score
                        best_p = p.copy()
                        if consider is not None:
                            consider(best_p)
                    # Try with different noise scale
                    flat = 0
                    if flat > 12 and time.monotonic() < stop_time - 8.0:
                        noise_scale = 0.05
                        p = best_p.copy() * np.exp(rng.normal(0.0, noise_scale, n, dtype=np.float64))
                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss > 0.0:
                            p /= ss
                        c = _conv_self(p)
                        np.maximum(c, 0.0, out=c)
                        S_total = 1.0
                        cur_max = float(c.max())
                        flat = 0
                else:
                    break

        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 3.0:
        hard = t0 + max(2.0, budget - 1.0)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64)
        best_arr /= 64.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 4e-14:
            best_arr = a.copy()
            best_score = sc

    n_par = int(best_arr.size)

    # === Phase 1: LSE optimization (coarse) ===
    phase1_end = min(hard - 300.0, t0 + 0.18 * budget)
    if phase1_end > t0 + 20.0 and n_par >= 128:
        p1 = best_arr.copy()
        p1 = p1 / float(p1.sum())
        try:
            p1 = _lse_optimize(p1, phase1_end, consider=consider, fine_only=False)
            if p1 is not None and p1.size == n_par and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    work = best_arr.copy() / float(best_arr.sum())

    # === Phase 2: Coordinate descent (single) ===
    phase2_end = min(hard - 400.0, t0 + 0.35 * budget)
    if phase2_end > time.monotonic() + 15.0:
        try:
            work, _ = _coord_descent(work, phase2_end, max_sweeps=200000,
                                     seed=int(seed + 111), consider=consider,
                                     use_pairs=False)
        except Exception:
            pass

    work = best_arr.copy() / float(best_arr.sum())

    # === Phase 3: Coordinate descent with pairs ===
    phase3_end = min(hard - 300.0, t0 + 0.60 * budget)
    if phase3_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord_descent(work, phase3_end, max_sweeps=200000,
                                     seed=int(seed + 222), consider=consider,
                                     use_pairs=True, pair_ratio=0.6)
        except Exception:
            pass

    work = best_arr.copy() / float(best_arr.sum())

    # === Phase 4: Fine LSE ===
    phase4_end = min(hard - 200.0, t0 + 0.72 * budget)
    if phase4_end > time.monotonic() + 15.0:
        try:
            p4 = _lse_optimize(work, phase4_end, consider=consider, fine_only=True)
            if p4 is not None and p4.size == n_par and np.all(np.isfinite(p4)) and p4.sum() > 0:
                p4 = p4 / float(p4.sum())
                consider(p4)
        except Exception:
            pass

    work = best_arr.copy() / float(best_arr.sum())

    # === Phase 5: Final coordinate polish with pairs ===
    phase5_end = min(hard - 30.0, t0 + budget - 40.0)
    if phase5_end > time.monotonic() + 15.0:
        try:
            work, _ = _coord_descent(work, phase5_end, max_sweeps=200000,
                                     seed=int(seed + 333), consider=consider,
                                     use_pairs=True, pair_ratio=0.5)
        except Exception:
            pass

    # === Final squeeze: pure single-coordinate on best ===
    rem = hard - 20.0 - time.monotonic()
    if rem > 20.0:
        fin_end = min(hard - 15.0, t0 + budget - 10.0)
        if fin_end > time.monotonic() + 10.0:
            work = best_arr.copy() / float(best_arr.sum())
            try:
                work, _ = _coord_descent(work, fin_end, max_sweeps=100000,
                                         seed=int(seed + 444), consider=consider,
                                         use_pairs=False)
            except Exception:
                pass

    # Final validation and return
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
