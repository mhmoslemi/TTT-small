    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(a, w):
        F = rfft(a[::-1], cross_size)
        G = rfft(w, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(np.clip(scaled, -700, 700))
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return f, grad

    stages = [
        (0.08, 0.10),
        (0.02, 0.15),
        (0.005, 0.20),
        (0.001, 0.25),
        (0.0003, 0.30),
    ]

    best_p = p.copy()
    best_sc = _score_arr(p)
    start = time.monotonic()
    avail = max(0.0, stop_time - start - 10.0)
    total_w = sum(w for _, w in stages)
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 6.0:
            break
        cum += wt
        end_stage = min(stop_time - 5.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)
        best_u = u.copy()
        try:
            f0, _ = lse_func(u, T)
        except Exception:
            f0 = 1e100
        best_fstage = f0
        stale = 0
        while time.monotonic() < end_stage - 1.5:
            maxit = 35 if n > 2000 else 50
            opts = {"maxiter": maxit, "gtol": 1e-12, "ftol": 1e-15, "maxls": 20, "maxdisp": 0}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_fstage - 1e-14:
                    best_fstage = fval
                    best_u = res.x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if getattr(res, "status", 1) == 0:
                break
        p_cur = _softmax(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        sc = _score_arr(p_cur)
        if math.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best_p = p_cur.copy()
            p = p_cur.copy()
            u = np.log(np.maximum(p, 1e-30))
            u = u - float(u.mean())
        else:
            p = best_p.copy()
            u = np.log(np.maximum(p, 1e-30))
            u = u - float(u.mean())
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p


def _coord_descent(p, stop_time, consider=None, seed=12345):
    n = int(p.size)
    if n == 0:
        return p, float('inf')
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p, float('inf')
    p = p / s
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    Ltot = 2 * n - 1
    flat = 0
    sweep = 0
    GR = (math.sqrt(5.0) - 1.0) / 2.0

    def eval_score_delta(w, base, infl, diag_offset, max_const, S, n):
        s_new = S + w
        if s_new <= 1e-15:
            return float('inf')
        aff = base + w * infl
        if 0 <= diag_offset < len(base):
            aff = aff.copy()
            aff[diag_offset] += w * w
        m = max(float(aff.max()), max_const)
        return 2.0 * n * m / (s_new * s_new)

    G_GRID = 40
    max_sweeps = 100000
    while sweep < max_sweeps and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            v = float(p[i])
            start_k = i
            end_k = min(i + n, Ltot)
            Ls = end_k - start_k
            if Ls <= 0:
                continue
            base = c[start_k:end_k]
            infl = 2.0 * p[:Ls]
            diag = i - start_k  # = 0, since start_k = i, diag index = 2i - start_k = i... 

            # diag: the position where k = 2i, relative to start_k
            # k = 2i is in [start_k, end_k) = [i, min(i+n, Ltot))
            # relative index = 2i - i = i
            diag = i  # relative index in base/infl arrays

            max_const = 0.0
            if start_k > 0:
                mc = float(c[:start_k].max())
                if mc > max_const:
                    max_const = mc
            if end_k < Ltot:
                mc = float(c[end_k:].max())
                if mc > max_const:
                    max_const = mc

            S = 1.0
            x_hi = max(30.0 * v, 20.0 / n, 0.01)
            x_hi = min(x_hi, 0.5)
            x_hi = max(x_hi, 1e-8)

            pts = np.linspace(0.0, x_hi, G_GRID + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                ins = np.searchsorted(pts, v)
                pts = np.insert(pts, ins, v)

            ws = pts - v
            s_new = S + ws
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag < Ls:
                aff[:, diag] = aff[:, diag] + ws * ws
            maff = np.max(aff, axis=1)
            maff = np.maximum(maff, max_const)
            valid = s_new > 1e-12
            scores = np.full(pts.shape, float('inf'), dtype=np.float64)
            if np.any(valid):
                scores[valid] = 2.0 * n * maff[valid] / (s_new[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            # Golden section refinement
            if 0 < imin < len(pts) - 1:
                lo = float(pts[max(0, imin - 2)])
                hi = float(pts[min(len(pts) - 1, imin + 2)])
            else:
                lo = 0.0
                hi = x_hi
            if hi - lo < 1e-14:
                lo = max(0.0, best_x - 1e-6)
                hi = best_x + 1e-6

            a_gs = lo
            b_gs = hi
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)
            f_c = eval_score_delta(c_gs - v, base, infl, diag, max_const, S, n)
            f_d = eval_score_delta(d_gs - v, base, infl, diag, max_const, S, n)
            for _ in range(25):
                if b_gs - a_gs < 1e-13 * max(1.0, abs(best_x)):
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = eval_score_delta(c_gs - v, base, infl, diag, max_const, S, n)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_d = eval_score_delta(d_gs - v, base, infl, diag, max_const, S, n)
            x_mid = (a_gs + b_gs) / 2.0
            f_mid = eval_score_delta(x_mid - v, base, infl, diag, max_const, S, n)
            if f_mid < best_val:
                best_x = x_mid
                best_val = f_mid
            for xp in (0.0, x_hi):
                fp = eval_score_delta(xp - v, base, infl, diag, max_const, S, n)
                if fp < best_val:
                    best_x = xp
                    best_val = fp

            if math.isfinite(best_val) and best_val < cur_score - 1e-15 and 0.0 <= best_x <= x_hi:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    aff_seg = base + w * infl
                    if 0 <= diag < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag] += w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    c[start_k:end_k] = aff_seg
                    cur_score = 2.0 * n * new_max
                    improved = True

        s_new = float(p.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            inv = 1.0 / s_new
            p *= inv
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-15:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 10.0:
                    kick_size = 0.01 + 0.05 * min(1.0, flat / 15.0)
                    p = p * np.exp(rng.normal(0.0, kick_size, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                    # Try an even larger kick after more stalling
                    if flat == 0 and sweep % 200 == 0:
                        pass
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _lp_direction(f, stop_time):
    """Try to get an LP-based direction. Returns normalized g or None."""
    n = f.size
    if n < 16 or n > 4096:
        return None
    c = _conv_self(f)
    M = float(c.max())
    if M <= 0 or not math.isfinite(M):
        return None

    # Build sparse convolution matrix A: A[k,j] = f[j] if 0 <= k-j < n
    # Shape: (2n-1, n)
    rows_list = []
    cols_list = []
    vals_list = []
    for j in range(n):
        r_lo = j
        r_hi = min(j + n, 2 * n - 1)
        cnt = r_hi - r_lo
        if cnt <= 0:
            continue
        rows_list.append(np.arange(r_lo, r_hi))
        cols_list.append(np.full(cnt, j, dtype=np.int32))
        vals_list.append(np.full(cnt, f[j], dtype=np.float64))

    if not rows_list:
        return None

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)

    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    from scipy.optimize import linprog
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)

    try:
        if time.monotonic() > stop_time - 15.0:
            return None
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={"time_limit": max(1.0, stop_time - time.monotonic() - 10.0)})
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    g = np.maximum(res.x, 0.0)
    s = float(g.sum())
    if s < 1e-15:
        return None
    g = g / s
    # Check if g is meaningfully different from f
    diff = np.linalg.norm(g - f) / max(1e-30, np.linalg.norm(f))
    if diff < 1e-5:
        return None
    return g


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    deadline = t0 + budget - 25.0

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float('inf')

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(evaluate_sequence([float(x) for x in a]))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_score = sc
            best_arr = a.copy()

    consider(parent)
    n_work = int(best_arr.size)

    # Normalize working copy
    f = np.maximum(parent, 0.0).astype(np.float64).copy()
    fs = float(f.sum())
    if fs > 0:
        f = f / fs

    # Phase 1: Brief LSE to polish the parent
    lse_time = min(80.0, 0.08 * budget)
    lse_end = t0 + lse_time
    if lse_end > t0 + 10.0:
        try:
            p2 = _lse_optimize(f, lse_end, consider=consider)
            if p2 is not None and p2.size == n_work and np.all(np.isfinite(p2)) and p2.sum() > 0:
                f = p2 / float(p2.sum())
                consider(f)
        except Exception:
            pass

    # Phase 2: Try LP direction if time allows
    if time.monotonic() < deadline - 300.0 and n_work <= 2048:
        lp_deadline = min(deadline - 250.0, time.monotonic() + 30.0)
        try:
            g = _lp_direction(f, lp_deadline)
            if g is not None:
                # Line search along f -> g
                best_t = 0.0
                best_sc_t = _score_arr(f)
                for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    blend = (1 - t) * f + t * g
                    blend = np.maximum(blend, 0.0)
                    bs = float(blend.sum())
                    if bs > 0:
                        blend = blend / bs
                    sc_t = _score_arr(blend)
                    if math.isfinite(sc_t) and sc_t < best_sc_t:
                        best_sc_t = sc_t
                        best_t = t
                if best_t > 0:
                    f_new = (1 - best_t) * f + best_t * g
                    f_new = np.maximum(f_new, 0.0)
                    f_new = f_new / float(f_new.sum())
                    consider(f_new)
                    f = f_new
        except Exception:
            pass

    # Phase 3: Main coordinate descent
    cd_end = deadline - 30.0
    if cd_end > time.monotonic() + 20.0:
        try:
            p_cd, _ = _coord_descent(f, cd_end, consider=consider, seed=int(seed + 7777))
            if p_cd is not None and p_cd.size > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                consider(p_cd)
        except Exception:
            pass

    # Phase 4: Final LSE squeeze if time remains
    rem = deadline - 20.0 - time.monotonic()
    if rem > 40.0:
        try:
            p_start = best_arr.copy()
            p_start = np.maximum(p_start, 0.0)
            ps = float(p_start.sum())
            if ps > 0:
                p_start = p_start / ps
            p2 = _lse_optimize(p_start, deadline - 20.0, consider=consider)
            if p2 is not None and p2.size > 0:
                consider(p2)
        except Exception:
            pass

    # Phase 5: Final CD squeeze
    rem = deadline - 15.0 - time.monotonic()
    if rem > 20.0:
        try:
            p_start = best_arr.copy()
            p_start = np.maximum(p_start, 0.0)
            ps = float(p_start.sum())
            if ps > 0:
                p_start = p_start / ps
            p_cd, _ = _coord_descent(p_start, deadline - 15.0, consider=consider, seed=int(seed + 9999))
            if p_cd is not None and p_cd.size > 0:
                consider(p_cd)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
