2. Do a fine search with G=32 in a small window around the best

This reduces the per-coordinate cost significantly.

Let me think about this. Suppose the max is at lag k*. Then:
g_{k*}(p) = sum_j p[j]*p[k*-j]

The contribution of p[i] to g_{k*} is:
- If k*-i is in [0, n-1]: p[i]*p[k*-i] + p[k*-i]*p[i] = 2*p[i]*p[k*-i] (unless i = k*-i, in which case it's p[i]^2)

So g_{k*}(p) as a function of p[i] (holding other p[j] fixed) is:
- If i ≠ k*-i: g = p[i]^2 (if i=k*-i) + 2*p[i]*p[k*-i] + const = quadratic in p[i]
- If i = k*-i: g = p[i]^2 + const

The constraint is sum(p) = 1, so if we change p[i] by delta, we need to rescale. The score becomes:
S(p[i]) = 2*n * max_k (p*p)[k] / (sum p)^2

This is a ratio of a (possibly) quadratic to a squared linear function. The minimum of a ratio of two quadratics can be found analytically... but it's complicated when the max involves multiple lags.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0)
    if n <= 512:
        c = np.convolve(a, a, mode="full")
    else:
        L = 2 * n - 1
        size = next_fast_len(L)
        F = rfft(a, size)
        c = irfft(F * F, size)[:L]
    return np.maximum(c, 0.0)


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _score_exact_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _softmax_fn(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        return src.copy()
    m = src.size
    xp = (np.arange(m) + 0.5) / m
    fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
    out = np.maximum(fp, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def _lse_optimize(p, stop_time, consider=None, seed=42):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax_fn(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    stages = (
        (0.20, 0.07),
        (0.08, 0.08),
        (0.03, 0.10),
        (0.010, 0.12),
        (0.003, 0.13),
        (0.001, 0.14),
        (0.0003, 0.13),
        (0.0001, 0.12),
        (0.00003, 0.10),
        (0.00001, 0.08),
    )

    start = time.monotonic()
    avail = max(0.0, float(stop_time) - start - 6.0)
    if avail <= 5.0:
        return p

    best_p = p.copy()
    best_sc = _score_fft(p)
    total_w = float(sum(w for _, w in stages))
    rng = np.random.default_rng(seed)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 1.5:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)

        best_u = u.copy()
        f0, _ = lse_func(best_u, T)
        if not math.isfinite(f0):
            continue
        best_f = f0
        stale = 0

        while time.monotonic() < end_stage - 0.5:
            maxit = 30
            opts = {"maxiter": maxit, "gtol": 1e-11, "ftol": 1e-15,
                    "maxls": 25, "maxcor": 25, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-15:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            best_u = best_u + rng.normal(0, 0.004, best_u.size)
                            continue
            if getattr(res, "status", 1) == 0:
                if time.monotonic() < end_stage - 1.0 and stale < 3:
                    best_u = best_u + rng.normal(0, 0.002, best_u.size)
                    f_chk, _ = lse_func(best_u, T)
                    if math.isfinite(f_chk):
                        if f_chk < best_f:
                            best_f = f_chk
                        stale += 1
                else:
                    break

        p_cur = _softmax_fn(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif math.isfinite(sc_cur) and sc_cur < best_sc + 0.003:
            p = p_cur.copy()
        else:
            p = best_p.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, G=None, seed=12345, consider=None):

    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    if G is None:
        if n <= 500:
            G = 56
        elif n <= 1000:
            G = 48
        elif n <= 2500:
            G = 44
        elif n <= 5000:
            G = 40
        else:
            G = 32
    G = int(max(24, G))

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < 300000 and time.monotonic() < stop_time - 0.8:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue
            base = c[start_i:end_i].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start_i > 0:
                ml = float(c[:start_i].max())
                if ml > max_const:
                    max_const = ml
            if end_i < Ltot:
                mr = float(c[end_i:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(25.0 * v, 50.0 * S / float(n), 0.02 * S, 1e-14)
            x_hi = min(x_hi, 10.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, S / float(n))

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag_i = i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[valid, None] * infl[None, :]
                dval = base[diag_i] + w[valid] * infl[diag_i] + w[valid] * w[valid]
                aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                dv = denom[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if G > 6 and best_x > 0 and best_x < x_hi:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.5 * dw)
                hi = min(x_hi, best_x + 2.5 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 32, dtype=np.float64)
                fine = np.concatenate((fine, [best_x]))
                fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val - 1e-15:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 10.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start_i:end_i] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:

                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 6.0:
                    sigma = 0.015 + 0.025 * min(flat, 15) / 15.0
                    mask = rng.random(n) < 0.7
                    pert = np.exp(rng.normal(0.0, sigma, n))
                    pert[~mask] = 1.0
                    p = p * pert
                    p = np.maximum(p, 0.0)
                    ss2 = float(p.sum())
                    if ss2 > 0:
                        p /= ss2
                    else:
                        break
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-14:

                        if consider is not None:

                                pass
                    flat = 0
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        sc = float(_score_fft(p))
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _grad_guided_cd(p, stop_time, seed=777, consider=None):
    """Gradient-guided 1D line search coordinate descent."""
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s
    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < 200000 and time.monotonic() < stop_time - 0.5:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            S = float(p.sum())
            v = float(p[i])
            # Gradient direction: d/dp_i max_k (p*p)[k]
            # For active lag k*, grad = 2*p[k*-i] (if valid)
            k_star = int(np.argmax(c))
            # Multiple possible lags contribute
            # Use the value at k_star as a proxy
            if 0 <= k_star - i <= n - 1:
                grad_dir = -2.0 * float(p[k_star - i])
            elif 0 <= i - k_star <= n - 1:
                grad_dir = -2.0 * float(p[i - k_star])
            else:
                grad_dir = 0.0
            if abs(grad_dir) < 1e-20:
                # Try small random step
                direction = rng.choice([-1, 1])
                step_size = 0.01 * S / n
            else:
                direction = 1.0 if grad_dir < 0 else -1.0
                step_size = 0.05 * S / n

            # Try a few step sizes
            best_step = 0.0
            best_s = cur_score
            for mult in [0.1, 0.3, 0.5, 1.0, 1.5]:
                delta = direction * mult * step_size
                new_v = v + delta
                if new_v < 0 or new_v > 5.0 * S:
                    continue
                # Quick approximate: just adjust the affected region
                new_p = p.copy()
                new_p[i] = new_v
                new_S = S + delta
                if new_S < 1e-15:
                    continue
                # Compute new max in affected region
                start_i = i
                end_i = min(i + n, Ltot)
                base = c[start_i:end_i]
                infl = 2.0 * p[:end_i - start_i]
                w = delta
                new_vals = base + w * infl
                di = 2 * i
                if 0 <= di < Ltot:
                    new_vals[di - start_i] += w * w
                new_max_local = float(np.max(new_vals))
                max_const = 0.0
                if start_i > 0:
                    max_const = max(max_const, float(c[:start_i].max()))
                if end_i < Ltot:
                    max_const = max(max_const, float(c[end_i:].max()))
                new_max = max(new_max_local, max_const)
                new_score = 2.0 * n * new_max / (new_S * new_S)
                if new_score < best_s - 1e-14:
                    best_s = new_score
                    best_step = delta

            if best_step != 0.0:
                p[i] += best_step
                S = float(p.sum())
                start_i = i
                end_i = min(i + n, Ltot)
                infl = 2.0 * p[:end_i - start_i]
                # Recompute affected region
                p[i] -= best_step  # temporarily put back
                c[start_i:end_i] += best_step * 2.0 * np.array([p[j] for j in range(end_i - start_i)])
                p[i] += best_step
                di = 2 * i
                if 0 <= di < Ltot:
                    c[di] += best_step * best_step
                cur_score = best_s
                improved = True

        # Renormalize
        S = float(p.sum())
        if S > 1e-300:
            p /= S
            c = _conv_self(p)
            cur_score = float(2.0 * n * c.max())
        else:
            break

        if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                break
        sweep += 1

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 5.0

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = float(_score_exact_arr(best_arr))

    n_parent = int(best_arr.size)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    # Phase 1: LSE optimization at parent length (~15% of budget)
    lse_time = 0.15 * (hard_deadline - t0)
    lse_stop = t0 + lse_time
    p = best_arr.copy()
    if lse_stop > t0 + 15.0:
        try:
            p = _lse_optimize(p, lse_stop, consider=consider, seed=int(seed + 100))
        except Exception:
            p = best_arr.copy()

    # Phase 2: Primary coordinate descent (~50% of budget)
    coord1_time = 0.50 * (hard_deadline - t0)
    coord1_stop = t0 + coord1_time
    if coord1_stop > time.monotonic() + 15.0:
        try:
            p2, _ = _coord_polish(p, coord1_stop, G=48, seed=int(seed + 200), consider=consider)
            if p2 is not None and p2.size == n_parent and float(p2.sum()) > 0:
                p2 = p2 / float(p2.sum())
                consider(p2)
        except Exception:
            pass

    # Phase 3: Additional LSE rounds with perturbation (~12% of budget)
    if time.monotonic() < hard_deadline - 0.15 * (hard_deadline - t0):
        phase3_stop = min(hard_deadline - 0.20 * (hard_deadline - t0),
                         time.monotonic() + 0.12 * (hard_deadline - t0))
        if phase3_stop > time.monotonic() + 10.0:
            rng3 = np.random.default_rng(int(seed + 5000))
            pert = best_arr * np.exp(rng3.normal(0, 0.025, best_arr.size))
            pert = np.maximum(pert, 0.0)
            ps = float(pert.sum())
            if ps > 0:
                pert /= ps
                p3 = pert.copy()
                # Quick LSE
                lse3_stop = min(phase3_stop - 10.0, time.monotonic() + 0.4 * (phase3_stop - time.monotonic()))
                if lse3_stop > time.monotonic() + 8.0:
                    try:
                        p3 = _lse_optimize(p3, lse3_stop, consider=consider, seed=int(seed + 6000))
                    except Exception:
                        pass
                # Quick coord
                coord3_stop = min(phase3_stop - 5.0, time.monotonic() + 0.5 * (phase3_stop - time.monotonic()))
                if coord3_stop > time.monotonic() + 8.0:
                    try:
                        p4, _ = _coord_polish(p3, coord3_stop, G=40, seed=int(seed + 7000), consider=consider)
                        if p4 is not None and float(p4.sum()) > 0:
                            p4 = p4 / float(p4.sum())
                            consider(p4)
                    except Exception:
                        pass

    # Phase 4: Final long coordinate descent on the best (~18% of budget)
    final_coord_stop = hard_deadline - 8.0
    n_best = int(best_arr.size)
    if final_coord_stop > time.monotonic() + 20.0:
        try:
            p5, _ = _coord_polish(best_arr, final_coord_stop, G=52, seed=int(seed + 99999), consider=consider)
            if p5 is not None and p5.size == n_best and float(p5.sum()) > 0:
                p5 = p5 / float(p5.sum())
                consider(p5)
        except Exception:
            pass

    # Final validation and return
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
