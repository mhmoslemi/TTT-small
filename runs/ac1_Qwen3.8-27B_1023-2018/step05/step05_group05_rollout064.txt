def _cross_conv(a, b):

def _fw_lp_step(f, n, row_arr, col_arr, offset_arr, deadline):
    """Solve LP direction for Frank-Wolfe. Returns normalized direction or None."""
    M = float(_conv_self(f).max())
    if M <= 0 or not math.isfinite(M):
        return None, M

    vals = f[offset_arr]
    A = csc_matrix((vals, (row_arr, col_arr)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    remaining = deadline - time.monotonic()
    if remaining < 3.0:
        return None, M
    lp_time = max(3.0, min(90.0, remaining * 0.6))

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': lp_time, 'presolve': True})
    except Exception:
        return None, M

    if res.status == 0 and res.x is not None:
        b = np.maximum(res.x, 0.0)
        sb = float(b.sum())
        if sb > 1e-10:
            return b / sb, M
    return None, M
