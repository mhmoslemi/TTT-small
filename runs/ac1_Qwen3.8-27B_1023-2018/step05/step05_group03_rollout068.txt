        s = e.sum()
        if not math.isfinite(s) or s <= 0:
            pp = np.full(n, 1.0 / n)
        else:
            pp = e / s
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(T) or T <= 1e-30:
            T = max(1e-14, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(conv_len)
            w[c.argmax()] = 1.0
            f = cmax
        else:
            w = exps / sum_exp
            f = cmax + T * math.log(sum_exp / conv_len)
            if not math.isfinite(f):
                f = cmax
        q = xfast(pp, w)
        gid = 2.0 * q
        gd = float(pp.dot(gid))
        grad = pp * (gid - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score_exact(p)
    consider(best_p)

    c0 = cfast(p)
    cmax0 = float(c0.max())
    stages = [
        (0.012, 0.20),
        (0.004, 0.25),
        (0.0012, 0.25),
        (0.00035, 0.20),
        (0.00009, 0.25),
    ]
    total_w = sum(s for _, s in stages)
    start = time.monotonic()
    avail = max(0.0, stop_time - start - 8.0)
    if avail < 30:
        return best_p
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 10:
            break
        cum += wt
        end_stage = min(stop_time - 6, start + avail * (cum / total_w))
        if end_stage <= now + 10:
            continue
        c = cfast(p)
        cmax = float(c.max())
        T = max(1e-14, frac * cmax)
        stale = 0
        while time.monotonic() < end_stage - 2:
            try:
                res = _min(lse_func, u, args=(T,), method='L-BFGS-B', jac=True,
                           options={'maxiter': 25, 'gtol': 1e-10, 'ftol': 1e-14,
                                    'maxls': 20, 'maxcor': 18, 'disp': False})
            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                x = res.x
                pp = np.asarray(__import__('scipy').special.expit(0)) # dummy
                # convert back
                mx = x.max()
                ee = np.exp(x - mx)
                pp = ee / ee.sum()
                sc = _score_exact(pp)
                if math.isfinite(sc) and sc < best_sc - 1e-12:
                    best_sc = sc
                    best_p = pp.copy()
                    p = pp.copy()
                    u = p_to_u(p)
                    consider(p)
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if getattr(res, 'status', 1) == 0:
                break
        p = best_p.copy()
        u = p_to_u(p)
    return best_p


def _coord_descent(a, stop_time, consider, seed=123, G=48):
    n = a.size
    a = _normalize(a)
    c = _fft_conv(a)
    S = float(a.sum())
    best_p = a.copy()
    best_sc = 2.0 * n * c.max() / (S * S)
    consider(best_p)

    rng = np.random.default_rng(seed)
    flat = 0
    sweep = 0

    while time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)
        L = 2 * n - 1

        for k in range(n):
            if time.monotonic() > stop_time - 1.5:
                break
            # Re-normalize occasionally
            if S > 1.05 or S < 0.95:
                inv = 1.0 / S
                a *= inv
                c *= inv * inv
                S = 1.0
                c = np.maximum(c, 0.0)

            j = int(order[k])
            v = float(a[j])
            # Affected range in c: [j, j+n-1]
            end = j + n
            if end > L:
                end = L
            start_idx = j
            length = end - start_idx
            if length <= 0:
                continue

            base = c[start_idx:end]  # shape (length,)
            # Max of unaffected portion
            max_unaff = 0.0
            if start_idx > 0:
                mu = float(c[:start_idx].max())
                if mu > max_unaff:
                    max_unaff = mu
            if end < L:
                mu = float(c[end:].max())
                if mu > max_unaff:
                    max_unaff = mu

            # Influence vector: a[0:length] corresponds to a[k-j] for k in [j, j+n-1]
            infl = a[:length]  # shape (length,)
            di = j  # position in base where quadratic correction applies (k=2j → pos j)
            valid_di = (0 <= di < length)

            # Candidate x values
            xhi = max(10.0 * v, 0.05 * S / n, 1e-6)
            xhi = min(xhi, 0.5 * S)
            if xhi <= 0:
                xhi = max(S / n, 1e-6)

            pts = np.array([0.0, v], dtype=np.float64)
            m1 = max(8, G // 3)
            lo_pt = max(1e-12, 1e-4 * v)
            if lo_pt < xhi:
                pts = np.concatenate([pts, np.geomspace(lo_pt, xhi, m1)])
            m2 = max(8, G // 4)
            if v > 1e-14:
                fac = np.geomspace(0.3, 4.0, m2)
                pts = np.concatenate([pts, v * fac])
            m3 = max(8, G // 4)
            linhi = min(xhi, max(3.0 * S / n, 0.01))
            pts = np.concatenate([pts, np.linspace(0.0, linhi, m3)])
            pts = np.unique(np.clip(pts, 0.0, xhi))
            if pts.size < 2:
                pts = np.array([0.0, v, xhi])

            # Vectorized evaluation
            d = pts - v  # (M,)
            # new_c[m, p] = base[p] + 2*d[m]*infl[p], for p in [0, length-1]
            # Then new_c[m, di] += d[m]^2
            new_c = base[None, :] + 2.0 * d[:, None] * infl[None, :]  # (M, length)
            if valid_di:
                new_c[:, di] += d * d
            new_max = np.maximum(max_unaff, new_c.max(axis=1))  # (M,)
            new_S = S + d  # (M,)
            valid = new_S > 1e-14
            scores = np.full(pts.size, math.inf)
            scores[valid] = 2.0 * n * new_max[valid] / (new_S[valid] ** 2)
            scores = np.where(np.isfinite(scores), scores, math.inf)

            if not np.any(np.isfinite(scores)):
                continue
            im = int(np.argmin(scores))
            best_x = float(pts[im])
            best_val = float(scores[im])

            # Refine around best
            cur_sc = 2.0 * n * c.max() / (S * S)
            if best_val < cur_sc - 1e-13:
                # Fine search
                step = max(v, S / n, 1e-10)
                lo = max(0.0, best_x - 3.0 * step)
                hi = min(xhi, best_x + 3.0 * step)
                if hi <= lo:
                    hi = lo + 1e-8
                fine = np.linspace(lo, hi, 32)
                if lo > 0:
                    fine = np.concatenate([fine, np.geomspace(lo, hi, 16)])
                fine = np.unique(np.clip(fine, 0.0, xhi))
                fd = fine - v
                fnc = base[None, :] + 2.0 * fd[:, None] * infl[None, :]
                if valid_di:
                    fnc[:, di] += fd * fd
                fm = np.maximum(max_unaff, fnc.max(axis=1))
                fS = S + fd
                fv = np.full(fine.size, math.inf)
                fvalid = fS > 1e-14
                fv[fvalid] = 2.0 * n * fm[fvalid] / (fS[fvalid] ** 2)
                fv = np.where(np.isfinite(fv), fv, math.inf)
                if np.any(np.isfinite(fv)):
                    fim = int(np.argmin(fv))
                    if fv[fim] < best_val:
                        best_x = float(fine[fim])
                        best_val = float(fv[fim])

            if math.isfinite(best_val) and best_val < cur_sc - 1e-12 and 0 <= best_x <= xhi:
                d = best_x - v
                if abs(d) > 1e-15:
                    c[start_idx:end] += d * (2.0 * infl)
                    if valid_di:
                        c[j + di] += d * d  # c[2j] extra
                    a[j] = best_x
                    S += d
                    improved = True

        # End of sweep
        if S > 1e-30:
            inv = 1.0 / S
            a *= inv
            c *= inv * inv
            S = 1.0
        c = _fft_conv(a)
        c = np.maximum(c, 0.0)
        sc = 2.0 * n * c.max()
        if math.isfinite(sc) and sc < best_sc - 1e-13:
            best_sc = sc
            best_p = a.copy()
            consider(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                # Perturbation to escape
                if time.monotonic() < stop_time - 30:
                    sig = min(0.15, 0.03 * (flat - 3))
                    noise = rng.normal(0, sig, n)
                    q = a * np.exp(noise)
                    q = np.maximum(q, 0)
                    qs = q.sum()
                    if qs > 1e-10:
                        a = q / qs
                        c = _fft_conv(a)
                        c = np.maximum(c, 0.0)
                        S = 1.0
                        sc = 2.0 * n * c.max()
                        if math.isfinite(sc) and sc < best_sc - 1e-13:
                            best_sc = sc
                            best_p = a.copy()
                            consider(best_p)
                else:
                    break
            else:
                # Small perturbation
                idx = rng.integers(0, n, size=max(1, n // 50))
                a[idx] *= np.exp(rng.normal(0, 0.05, size=len(idx)))
                a = np.maximum(a, 0)
                S = a.sum()
                if S < 1e-10:
                    a = best_p.copy()
                    c = _fft_conv(a)
                    S = 1.0
                else:
                    c = _fft_conv(a)
        sweep += 1

    return best_p, best_sc


def _lp_small(parent, n_small, stop_time):
    """Try LP-based direction for small n."""
    from scipy.optimize import linprog
    from scipy import sparse as sp

    a = _normalize(parent[:n_small] if parent.size >= n_small else np.resize(_normalize(parent), n_small))
    n = n_small
    c = _fft_conv(a)
    M = float(c.max())
    if not math.isfinite(M) or M <= 0:
        return None

    # Build sparse constraint matrix A where A*(b) = convolve(a, b)
    # Shape: (2n-1, n), A[k, j] = a[k-j] if 0<=k-j<n
    idx = np.arange(n)
    rows = np.repeat(idx, n)
    cols = np.tile(idx, n)
    data = a[idx]
    # Actually need: A[k,j] = a[j] if k-j is in [0,n-1], i.e., j in [k-n+1, k]
    # Let me use a different construction
    # c_full = convolve(a, b) has length 2n-1
    # c_full[k] = sum_j a[j]*b[k-j]
    # As a matrix: A[k,j] = a[k-j] if 0<=k-j<=n-1, else 0
    # This is a Toeplitz-like (actually convolution) matrix
    try:
        # Use the fact that it's a product of two Toeplitz-like structures
        # Build A row by row would be O(n^2) memory needed: (2n-1)*n
        if n > 800:
            return None
        A_data = []
        A_rows = []
        A_cols = []
        for k in range(2 * n - 1):
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                A_rows.append(k)
                A_cols.append(j)
                A_data.append(a[k - j])
        A = sp.coo_matrix((A_data, (A_rows, A_cols)), shape=(2 * n - 1, n)).tocsr()
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': max(5, stop_time - time.monotonic() - 10)})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            bs = b.sum()
            if bs > 1e-10:
                return b / bs
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(20.0, budget)
    reserve = max(5.0, 0.03 * budget)
    hard_deadline = t0 + budget - reserve

    parent = _get_parent()
    best_arr = _normalize(parent.copy())
    n = best_arr.size

    try:
        best_score = float(evaluate_sequence([float(x) for x in best_arr]))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score):
        best_arr = np.ones(64, dtype=np.float64) / 64.0
        n = 64
        best_score = float(evaluate_sequence([1.0] * n))

    def consider(a):
        nonlocal best_arr, best_score, n
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
            if arr.size == 0:
                return
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            s = arr.sum()
            if s < 0.01:
                return
            arr = arr / s
            sc = float(evaluate_sequence([float(x) for x in arr]))
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_score = sc
                best_arr = arr.copy()
                n = arr.size
        except Exception:
            pass

    # Phase 1: Quick exploration with shaped perturbations
    phase1_end = min(hard_deadline - 100, t0 + min(60, 0.06 * budget))
    rng = np.random.default_rng(seed)
    base = best_arr.copy()

    for q in (0.6, 0.8, 1.2, 1.5, 2.0):
        if time.monotonic() > phase1_end - 30:
            break
        cand = np.power(np.maximum(base, 1e-15), q)
        cand = _normalize(cand)
        consider(cand)

    for _ in range(4):
        if time.monotonic() > phase1_end - 20:
            break
        x = (np.arange(n) + 0.5) / n
        amp = 0.3 + 0.7 * rng.random()
        side = 1.0 if rng.random() < 0.5 else -1.0
        cand = base * np.exp(side * amp * (2 * x - 1))
        cand = _normalize(np.maximum(cand, 0))
        consider(cand)

        # Trial wave
        f = 1 + int(rng.integers(0, 5))
        ph = 2 * math.pi * rng.random()
        cand2 = base * np.exp(0.2 * np.sin(2 * math.pi * f * x + ph))
        cand2 = _normalize(np.maximum(cand2, 0))
        consider(cand2)

    # Phase 2: L-BFGS with log-sum-exp smoothing
    lse_stop = min(hard_deadline - 200, time.monotonic() + min(200, 0.22 * budget))
    if lse_stop > time.monotonic() + 30:
        try:
            best_arr = _lbfgs_phase(best_arr, lse_stop, consider)
        except Exception:
            pass
    best_arr = _normalize(best_arr)
    consider(best_arr)

    # Phase 3: LP-based direction for small subproblem (optional, time permitting)
    lp_stop = min(hard_deadline - 300, time.monotonic() + 80)
    if lp_stop > time.monotonic() + 20 and n > 128:
        for n_small in (256, 384):
            if time.monotonic() > lp_stop:
                break
            if n_small < n:
                try:
                    b = _lp_small(best_arr, n_small, lp_stop)
                    if b is not None:
                        # Upsample to full length
                        xp = (np.arange(n_small) + 0.5) / n_small
                        fp = np.interp((np.arange(n) + 0.5) / n, xp, b)
                        cand = _normalize(np.maximum(fp, 0))
                        consider(cand)
                        # Blend
                        for t in (0.3, 0.5, 0.7):
                            blend = (1 - t) * best_arr + t * cand
                            blend = _normalize(np.maximum(blend, 0))
                            consider(blend)
                except Exception:
                    pass
            if n_small >= n:
                try:
                    b = _lp_small(best_arr, n, lp_stop)
                    if b is not None:
                        for t in (0.3, 0.5, 0.7):
                            blend = (1 - t) * best_arr + t * b
                            blend = _normalize(np.maximum(blend, 0))
                            consider(blend)
                except Exception:
                    pass

    # Phase 4: Coordinate descent (main refinement)
    coord_stop = min(hard_deadline - 10, time.monotonic() + (hard_deadline - 15 - time.monotonic()))
    if coord_stop > time.monotonic() + 20:
        try:
            best_arr, _ = _coord_descent(best_arr, coord_stop, consider, seed=seed + 777)
        except Exception:
            pass

    # Phase 5: Final squeeze - another coordinate descent with different seed
    final_stop = min(hard_deadline - 5, time.monotonic() + (hard_deadline - 10 - time.monotonic()))
    if final_stop > time.monotonic() + 15:
        try:
            best_arr, _ = _coord_descent(best_arr, final_stop, consider, seed=seed + 42424)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = _normalize(parent)
            s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = np.ones(64) / 64.0
        out = out / out.sum()
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls2 = [1.0]
            return ls2
        return ls
    except Exception:
        return [1.0]
