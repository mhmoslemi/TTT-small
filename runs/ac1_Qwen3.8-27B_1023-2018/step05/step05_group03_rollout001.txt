def _coord_descent(p, stop_time, G=40, seed=12345, consider=None, verbose=False):

    p = _normalize(p)
    n = p.size
    if n <= 1:
        return p, math.inf

    c = _fft_conv(p)
    c = np.maximum(c, 0.0)
    S = 1.0
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()

        try:
            consider(best_p)
        except Exception:
            pass

    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    L = 2 * n - 1

        improved = False
        order = rng.permutation(n)

        for pos in range(n):
            if time.monotonic() > stop_time - 0.3:
                break
            if S <= 1e-300:
                break
            if S > 1.15 or S < 0.85:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0
                c = np.maximum(c, 0.0)
                cur_score = 2.0 * n * float(c.max())

            i = int(order[pos])
            v = float(p[i])
            start = i
            end = min(i + n, L)
            Ls = end - start
            if Ls <= 0:
                continue

            infl = np.full(Ls, 2.0 * v, dtype=np.float64)
            # influence vector: 2 * a[k-i] for k in [start, end)
            # k-i ranges from 0 to n-1, so infl = 2 * a[0:n] = 2 * p
            infl = 2.0 * p[0:Ls]

            # max outside affected range
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max()) if start > 0 else 0.0
                if mc > max_const:
                    max_const = mc
            if end < L:
                mc = float(c[end:].max()) if (L - end) > 0 else 0.0
                if mc > max_const:
                    max_const = mc

            # Trial values
            mean_v = S / n
            xhi = max(2.5 * v, 3.0 * mean_v, 0.002)

            # Linear around v

            # Above v

                        f1 = 1 + int(rng.integers(0, 6))

    best_p = p.copy()
    best_sc = 2.0 * n * cmax

    for _ in range(n_trials):
        if len(active_idx) == 0:
            break
        k = int(rng.choice(active_idx))
        # Find coordinates contributing to c[k]
        lo_i = max(0, k - n + 1)
        hi_i = min(n - 1, k)
        if hi_i < lo_i:
            continue
        # The pair (i, k-i) with largest product
        ia = np.arange(lo_i, hi_i + 1)
        ib = k - ia
        products = p[ia] * p[ib]
        im = int(np.argmax(products))
        i_top = int(ia[im])
        j_top = int(ib[im])

        # Reduce the larger of the two
        p_new = p.copy()
        if p[i_top] >= p[j_top]:
            p_new[i_top] *= 0.75
        else:
            p_new[j_top] *= 0.75
        p_new = _normalize(p_new)
        sc = _score_fast(p_new)
        if math.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_p = p_new.copy()
            if consider:
                try:
                    consider(best_p)
                except Exception:
                    pass

    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(10.0, budget)
    reserve = 20.0 if budget > 200 else 5.0
    hard_end = t0 + budget - reserve

    # Get parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel().copy()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    p = _normalize(p)
    n = p.size

    best_arr = p.copy()
    best_score = _score_fast(best_arr)

    def consider(a):
        nonlocal best_arr, best_score
        try:
            a = _normalize(a)
            sc = _score_fast(a)
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_arr = a.copy()
                best_score = sc
        except Exception:
            pass

    # Initial power-law and shape perturbations
    rng_top = np.random.default_rng(seed)
    base = best_arr.copy()
    for q in (0.6, 0.8, 1.2, 1.5, 2.0):
        if time.monotonic() > hard_end - 200:
            break
        cand = np.power(np.maximum(base, 1e-30), float(q))
        cand = _normalize(cand)
        consider(cand)

    for _ in range(4):
        if time.monotonic() > hard_end - 250:
            break
        side = 1.0 if rng_top.random() < 0.5 else -1.0
        amp = 0.3 + 0.8 * float(rng_top.random())
        x_ax = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
        cand = base * np.exp(side * amp * (2.0 * x_ax - 1.0))
        cand = _normalize(cand)
        consider(cand)

    current = best_arr.copy()

    # Phase 1: LSE optimization
    if time.monotonic() < hard_end - 300:
        lse_stop = min(t0 + 180.0, hard_end - 200.0)
        if lse_stop > time.monotonic() + 30:
            current = _lse_optimize(current, lse_stop, consider=consider, seed=seed)
            current = _normalize(current)

    # Phase 2: Main coordinate descent
    if time.monotonic() < hard_end - 50:
        coord_stop = min(hard_end - 40.0, time.monotonic() + 500.0)
        if coord_stop > time.monotonic() + 20:
            current, _ = _coord_descent(current, coord_stop, G=36, seed=seed + 7, consider=consider)
            current = _normalize(current)

    # Phase 3: Targeted perturbations + more coordinate descent
    if time.monotonic() < hard_end - 60:
        rng_t = np.random.default_rng(seed + 999)
        current = _targeted_perturb(current, consider, rng_t, n_trials=12)

        if time.monotonic() < hard_end - 40:
            coord_stop2 = min(hard_end - 30.0, time.monotonic() + 250.0)
            if coord_stop2 > time.monotonic() + 15:
                current, _ = _coord_descent(current, coord_stop2, G=32, seed=seed + 42, consider=consider)
                current = _normalize(current)

    # Phase 4: Final LSE polish + coordinate
    if time.monotonic() < hard_end - 30:
        lse2_stop = min(hard_end - 25.0, time.monotonic() + 80.0)
        if lse2_stop > time.monotonic() + 15:
            current = _lse_optimize(current, lse2_stop, consider=consider, seed=seed + 200)
            current = _normalize(current)

        if time.monotonic() < hard_end - 10:
            coord_stop3 = min(hard_end - 5.0, time.monotonic() + 100.0)
            if coord_stop3 > time.monotonic() + 10:
                current, _ = _coord_descent(current, coord_stop3, G=28, seed=seed + 888, consider=consider)
                current = _normalize(current)

    # Final consideration
    consider(best_arr)
    consider(current)

    # Return best
    try:
        out = best_arr.copy()
        out = _normalize(out)
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            fallback = _normalize(p)
            ls2 = [float(x) for x in fallback]
            sc2 = evaluate_sequence(ls2)
            if math.isfinite(sc2):
                return ls2
            return [1.0]
        return ls
    except Exception:
        return [1.0]
