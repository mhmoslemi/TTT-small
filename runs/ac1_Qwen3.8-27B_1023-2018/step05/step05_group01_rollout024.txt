import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:
        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s < 1e-12:
        p = np.ones(3072, dtype=np.float64)
        s = float(p.sum())
    return p / s


def _conv_fft(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_fft(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _softmax_fn(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _p_to_u(p):
    u = np.log(np.maximum(p, 1e-300))
    return u - float(np.mean(u))


def _lse_optimize(p, stop_time, consider, seed=42):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 1e-300).copy()
    p = p / float(p.sum())
    n = int(p.size)
    if n == 0:
        return p
    u = _p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax_fn(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        T = max(1e-300, T)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * math.log(sum_exp / c.size)
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    stages = [
        (0.05, 1.0),
        (0.015, 1.0),
        (0.005, 1.2),
        (0.0015, 1.5),
        (0.0005, 2.0),
        (0.00015, 2.0),
        (0.00005, 2.5),
        (0.00002, 2.0),
    ]

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    avail = max(0.0, float(stop_time) - start - 4.0)
    if avail < 20.0:
        return p
    total_w = float(sum(w for _, w in stages))
    cum = 0.0
    best_u = u.copy()

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 2.0, start + avail * (cum / total_w))
        if end_stage <= now + 5.0:
            continue

        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)
        stale = 0

        while time.monotonic() < end_stage - 1.0 and stale < 4:
            opts = {
                "maxiter": 60, "gtol": 1e-12, "ftol": 1e-16,
                "maxls": 30, "maxcor": 30, "maxdisp": 0, "disp": False
            }
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    fval = float(res.fun)
                    if math.isfinite(fval):
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                else:
                    stale += 1
            if getattr(res, "status", 1) == 0:
                if time.monotonic() < end_stage - 3.0:
                    best_u = best_u + rng.normal(0, 0.003, n)
                    stale += 1
                else:
                    break

        p_cur = _softmax_fn(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
            consider(p_cur)
        p = p_cur.copy()
        best_u = _p_to_u(p)

    return p


def _coord_descent(p, stop_time, consider, seed=999):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    S = float(p.sum())
    if S < 1e-300:
        return p, math.inf
    p = p / S
    S = 1.0

    Ltot = 2 * n - 1
    c = _conv_fft(p)
    cur_max_b = float(c.max())
    cur_score = 2.0 * n * cur_max_b  # S=1

    best_p = p.copy()
    best_score = cur_score
    rng = np.random.default_rng(seed)

    def prefix_suffix_max(arr):
        pm = np.empty(arr.size, dtype=np.float64)
        sm = np.empty(arr.size, dtype=np.float64)
        pm[0] = arr[0]
        sm[-1] = arr[-1]
        for i in range(1, arr.size):
            pm[i] = max(pm[i - 1], arr[i])
        for i in range(arr.size - 2, -1, -1):
            sm[i] = max(sm[i + 1], arr[i])
        return pm, sm

    pm, sm = prefix_suffix_max(c)

    def get_const_max(start, end):
        mx = 0.0
        if start > 0:
            mx = max(mx, pm[start - 1])
        if end < Ltot:
            mx = max(mx, sm[end])
        return mx

    G = 32
    sweep = 0
    no_improve = 0

    while time.monotonic() < stop_time - 2.0:
        improved_any = False
        order = rng.permutation(n)

        for idx_i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            i = int(idx_i)
            v = float(p[i])

            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            max_const = get_const_max(start, end)
            base = c[start:end]
            infl = 2.0 * p[:Ls]

            x_hi = max(15.0 * v, 30.0 / n, 0.01, 1e-14)
            x_hi = min(x_hi, 6.0)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = 1.0 / n

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            diag_i = i
            best_x = v
            best_val = cur_score

            # Vectorized evaluation
            ws = pts - v
            denoms = S + ws
            valid = denoms > 1e-15
            if np.any(valid):
                wv = ws[valid]
                # aff[k] for each candidate w: base[k] + w*infl[k]
                # shape: (num_valid, Ls)
                aff = base[None, :] + wv[:, None] * infl[None, :]
                # diagonal correction
                dvals = base[diag_i] + wv * infl[diag_i] + wv * wv
                aff = np.maximum(aff, dvals[:, None])
                if max_const > 0:
                    aff = np.maximum(aff, max_const)
                mvs = aff.max(axis=1)
                dv = denoms[valid]
                vals = 2.0 * n * mvs / (dv * dv)
                local_best = int(np.argmin(vals))
                bv = float(vals[local_best])
                bx = float(pts[valid][local_best])
                if bv < best_val - 1e-14:
                    best_val = bv
                    best_x = bx

            # Golden section refinement if interior
            if best_x > 0.0 and best_x < x_hi:
                dx = x_hi / G * 1.5
                lo = max(0.0, best_x - dx)
                hi = min(x_hi, best_x + dx)
                if hi > lo + 1e-15:
                    phi = 0.3819660112501051

                    def eval_at(xi):
                        w = xi - v
                        d = S + w
                        if d <= 1e-15:
                            return 1e100
                        aff = base + w * infl
                        dv2 = base[diag_i] + w * infl[diag_i] + w * w
                        aff = np.maximum(aff, dv2)
                        if max_const > 0:
                            aff = np.maximum(aff, max_const)
                        mv = float(aff.max())
                        return 2.0 * n * mv / (d * d)

                    c1 = lo + phi * (hi - lo)
                    c2 = lo + (1 - phi) * (hi - lo)
                    f1 = eval_at(c1)
                    f2 = eval_at(c2)
                    for _ in range(25):
                        if f1 < f2:
                            hi = c2
                            c2 = c1
                            f2 = f1
                            c1 = lo + phi * (hi - lo)
                            f1 = eval_at(c1)
                        else:
                            lo = c1
                            c1 = c2
                            f1 = f2
                            c2 = lo + (1 - phi) * (hi - lo)
                            f2 = eval_at(c2)
                    xg = (lo + hi) / 2.0
                    fg = eval_at(xg)
                    if fg < best_val - 1e-14:
                        best_val = fg
                        best_x = xg

            if best_val < cur_score - 1e-14 and 0.0 <= best_x <= 10.0:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    # Rebuild prefix/suffix (expensive but needed for correctness)
                    pm, sm = prefix_suffix_max(c)
                    cur_max_b = float(c.max())
                    cur_score = 2.0 * n * cur_max_b / (S * S)
                    improved_any = True

        # Normalize if drifted
        if S > 1.05 or S < 0.95:
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
            pm, sm = prefix_suffix_max(c)
            cur_max_b = float(c.max())
            cur_score = 2.0 * n * cur_max_b

        # Update best
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = p.copy()
            consider(p)

        if improved_any:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4 and time.monotonic() < stop_time - 15.0:
                sigma = 0.015 + 0.005 * min(no_improve // 4, 6)
                p = p * np.exp(rng.normal(0, sigma, n))
                p = np.maximum(p, 0.0)
                ss = float(p.sum())
                if ss > 1e-300:
                    p /= ss
                    S = 1.0
                else:
                    p = best_p.copy()
                    S = 1.0
                c = _conv_fft(p)
                pm, sm = prefix_suffix_max(c)
                cur_max_b = float(c.max())
                cur_score = 2.0 * n * cur_max_b
                no_improve = 0

        sweep += 1

    # Final check
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
        sc = _score_arr(p)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p.copy()
    return best_p, best_score


def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        return src.copy()
    m = src.size
    xp = (np.arange(m) + 0.5) / m
    fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
    out = np.maximum(fp, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s

    out = np.where(np.isfinite(out), out, 0.0)
