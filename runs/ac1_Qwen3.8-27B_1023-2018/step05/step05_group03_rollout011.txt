import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize

def _fft_conv(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    sz = next_fast_len(L)
    F = rfft(a, sz)
    c = irfft(F * F, sz)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score_arr(a):
    a = np.maximum(a, 0.0)
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _fft_conv(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / s

def _softmax(u):
    u = np.asarray(u, dtype=np.float64).ravel()
    u = np.where(np.isfinite(u), u, 0.0)
    m = float(u.max())
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s

def _lag_sum_exp_optimize(p, stop_time, consider=None, fine_only=False):
    """L-BFGS on log-sum-exp smoothed objective in log-space."""
    p = _normalize(p)
    n = int(p.size)
    if n <= 1 or time.monotonic() > stop_time:
        return p
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2
    conv_sz = next_fast_len(conv_len)
    cross_sz = next_fast_len(cross_len)

    def cfast(a):
        F = rfft(a, conv_sz)
        c = irfft(F * F, conv_sz)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_sz)
        G = rfft(ww, cross_sz)
        c = irfft(F * G, cross_sz)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_obj(u, T):
        u = np.asarray(u, dtype=np.float64).ravel()
        if u.size != n:
            return 1e100, np.zeros(n)
        u = np.where(np.isfinite(u), u, 0.0)
        m0 = float(u.max())
        e = np.exp(u - m0)
        s = float(e.sum())
        if s <= 0 or not math.isfinite(s):
            pp = np.full(n, 1.0 / n)
        else:
            pp = e / s
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax):
            return 1e100, np.zeros(n)
        T = max(1e-15, T)
        scaled = (c - cmax) / T
        exps = np.exp(np.clip(scaled, -500, 500))
        se = float(exps.sum())
        if se <= 0 or not math.isfinite(se):
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            w = exps / se
            f = cmax + T * math.log(se / conv_len)
        q = xfast(pp, w)
        gid = 2.0 * q
        gd = float(pp @ gid)
        grad = pp * (gid - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    def p_to_u(pp):
        pp = np.maximum(pp, 1e-15)
        uu = np.log(pp, dtype=np.float64)
        uu -= uu.mean()
        return uu

    u = p_to_u(p)
    best_p = p.copy()
    best_f = _score_arr(best_p)
    if consider:
        try: consider(best_p)
        except: pass

    if fine_only:
        stages = [(0.0008, 0.20), (0.0002, 0.30), (0.00005, 0.50)]
    else:
        stages = [(0.015, 0.15), (0.005, 0.15), (0.0012, 0.20), (0.0003, 0.25), (0.00008, 0.25)]

    start_t = time.monotonic()
    avail = max(0.0, stop_time - start_t - 4.0)
    if avail < 10:
        return best_p
    total_w = sum(wt for _, wt in stages)
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += wt
        stage_end = min(stop_time - 4.0, start_t + avail * (cum / total_w))
        if stage_end < now + 8.0:
            continue
        c = cfast(p)
        cmax = float(c.max())
        T = max(1e-15, frac * cmax)
        best_u = u.copy()
        best_lse = lse_obj(u, T)[0]
        stale = 0
        while time.monotonic() < stage_end - 2.0:
            try:
                res = minimize(lse_obj, best_u, args=(T,), method='L-BFGS-B',
                               jac=True, options={'maxiter': 20, 'gtol': 1e-10,
                                                   'ftol': 1e-14, 'maxls': 20, 'maxcor': 15,
                                                   'disp': False, 'maxdisp': 0})
            except:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                fv = float(res.fun)
                if math.isfinite(fv) and fv < best_lse - 1e-12:
                    best_lse = fv
                    best_u = res.x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if getattr(res, 'status', 1) == 0:
                break
        p_new = _softmax(best_u)
        p_new = _normalize(p_new)
        sc = _score_arr(p_new)
        if math.isfinite(sc) and sc < best_f - 1e-13:
            best_f = sc
            best_p = p_new.copy()
            p = p_new.copy()
            if consider:
                try: consider(best_p)
                except: pass
        u = p_to_u(p)
    return best_p

def _coord_polish(p, stop_time, consider=None, G=80, seed=999, n_sweeps_target=200):
    """High-precision coordinate descent with equalization-aware grid."""
    p = _normalize(p).copy()
    n = int(p.size)
    if n <= 1:
        return p
    c = _fft_conv(p)
    S = 1.0
    infl = 2.0 * p
    cur = 2.0 * n * float(c.max())
    best_sc = cur
    best_p = p.copy()
    if consider:
        try: consider(best_p)
        except: pass
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    L = 2 * n - 1

    while sweep < n_sweeps_target and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for pos in range(n):
            if time.monotonic() > stop_time - 0.5:
                break
            if S <= 1e-25:
                break
            if S > 1.15 or S < 0.85:
                inv = 1.0 / S
                p *= inv; c *= inv * inv; infl *= inv; S = 1.0
                np.maximum(c, 0.0, out=c)
                cur = 2.0 * n * float(c.max())
            i = int(order[pos])
            v = float(p[i])
            start = i
            end = min(i + n, L)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                max_const = max(max_const, mc)
            if end < L:
                mc = float(c[end:].max())
                max_const = max(max_const, mc)
            Sloc = S

            # Build influence vector for affected lags
            infl_seg = np.zeros(Ls, dtype=np.float64)
            for k in range(Ls):
                ki = start + k
                val = 0.0
                j1 = ki - i
                if 0 <= j1 < n:
                    val += p[j1]
                j2 = i - ki
                if 0 <= j2 < n:
                    val += p[j2]
                infl_seg[k] = val
            # Diagonal correction
            di = 2 * i - start
            if 0 <= di < Ls:
                pass  # handled in eval

            # Find top K active lags for equalization points
            K = min(6, Ls)
            top_k = np.argpartition(base, -K)[-K:]
            top_k = top_k[np.argsort(base[top_k])[::-1]]

            # Build grid: 0, boundary, equalization points, uniform
            pts = [0.0, -v]
            for ka in range(min(4, K)):
                for kb in range(ka + 1, min(5, K)):
                    ia = int(top_k[ka])
                    ib = int(top_k[kb])
                    d_infl = infl_seg[ia] - infl_seg[ib]
                    if abs(d_infl) > 1e-18:
                        w_eq = (base[ib] - base[ia]) / d_infl
                        if -v * 1.01 < w_eq < v * 30 + Sloc / n:
                            pts.append(float(w_eq))
            pts.append(float(-v / 2.0))
            pts.append(float(v / 2.0))
            # Uniform spread
            xlo = max(-v, -Sloc / n)
            xhi = max(Sloc / n, v * 5)
            mu = max(8, int(G * 0.3))
            pts.extend(np.linspace(xlo, xhi, mu, dtype=np.float64).tolist())
            pts = np.clip(np.asarray(pts, dtype=np.float64), -v, xhi)
            pts = np.unique(pts)
            if pts.size == 0:
                continue

            # Evaluate
            best_w = 0.0
            best_val = cur
            for w in pts:
                w = float(w)
                if w == 0.0:
                    continue
                new_base = base + w * infl_seg
                if 0 <= di < Ls:
                    new_base[di] += w * w
                nm = max(float(new_base.max()), max_const)
                den = Sloc + w
                if den <= 1e-15:
                    continue
                val = 2.0 * n * nm / (den * den)
                if math.isfinite(val) and val < best_val:
                    best_val = val
                    best_w = w

            if best_val < cur - 1e-13:
                w = best_w
                if abs(w) > 1e-16:
                    c[start:end] += w * infl_seg
                    diag = 2 * i
                    if 0 <= diag < L:
                        c[diag] += w * w
                    p[i] = v + w
                    infl[i] = 2.0 * p[i]
                    S += w
                    cur = best_val
                    improved = True

        # End of sweep
        if S > 1e-25:
            inv = 1.0 / S
            p *= inv; c *= inv * inv; infl *= inv; S = 1.0
        else:
            break
        c = _fft_conv(p)
        np.maximum(c, 0.0, out=c)
        ds = 2.0 * n * float(c.max())
        if math.isfinite(ds) and ds < best_sc - 1e-14:
            best_sc = ds
            best_p = p.copy()
            if consider:
                try: consider(best_p)
                except: pass
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 30.0:
                    sigma = min(0.15, 0.02 + 0.02 * flat)
                    noise = rng.normal(0.0, sigma, n)
                    q = p * np.exp(noise)
                    q = np.maximum(q, 0.0)
                    qs = float(q.sum())
                    if qs > 1e-25:
                        p = q / qs
                        c = _fft_conv(p)
                        np.maximum(c, 0.0, out=c)
                        S = 1.0
                        infl = 2.0 * p
                        cur = 2.0 * n * float(c.max())
                else:
                    break
                flat = 0
        sweep += 1
    if consider:
        try: consider(best_p)
        except: pass
    return best_p

def _lp_small(src, n_small, stop_time):
    """LP-based direction for small subproblem."""
    try:
        from scipy.optimize import linprog
        from scipy import sparse as sp_lib
    except:
        return None
    if n_small < 32 or n_small > 512:
        return None
    a = _normalize(src)
    a_small = np.fft.irfft(np.fft.rfft(a, 4 * n_small))[:n_small]
    a_small = np.maximum(a_small, 0.0)
    ss = a_small.sum()
    if ss < 1e-25:
        return None
    a_small /= ss
    c = _fft_conv(a_small)
    M = float(c.max())
    if M <= 0:
        return None
    try:
        idx = np.arange(n_small, dtype=np.int32)
        cols = np.repeat(idx, n_small)
        off = np.tile(idx, n_small)
        rows = cols + off
        data = a_small[off].astype(np.float64)
        A = sp_lib.coo_matrix((data, (rows, cols)), shape=(2 * n_small - 1, n_small)).tocsr()
        cobj = -np.ones(n_small)
        b_ub = np.full(2 * n_small - 1, M * 1.001)
        res = linprog(cobj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                      options={'time_limit': max(1.0, stop_time - time.monotonic() - 5.0)})
        if res is not None and getattr(res, 'success', False) and res.x is not None:
            b = np.asarray(res.x, dtype=np.float64)
            b = np.maximum(b, 0.0)
            if b.sum() > 1e-10:
                return b / b.sum()
    except:
        pass
    return None

    # Final validation and return
    try:
        out = np.asarray(best, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = _normalize(parent)
            s = float(out.sum())
        if s < 0.01:
            out = np.ones(64) / 64.0
        out = out / float(out.sum())
        ls = [float(x) for x in out]
        fs = evaluate_sequence(ls)
        if not math.isfinite(fs):
            ls2 = [float(x) for x in _normalize(parent)]
            if math.isfinite(evaluate_sequence(ls2)):
                return ls2
            return [1.0 / 64.0] * 64
        return ls
    except:
        return [1.0 / 64.0] * 64
