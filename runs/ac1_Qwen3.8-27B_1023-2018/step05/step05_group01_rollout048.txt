def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(c, 0.0)


def _score_norm(a):
    """Score with sum(a)=1: 2n*max(conv)"""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-300:
        return math.inf
    a = a / s
    c = _conv_self(a)
    m = float(c.max())
    return 2.0 * n * m


def _score_exact_arr(a):

def _eval_coord_change(c, a, S, i, w, n):
    """Evaluate score change when a[i] changes by w."""
    v = a[i]
    S_new = S + w
    if S_new < 1e-300:
        return math.inf
    # Active range
    start = i
    end = min(i + n, 2*n - 1)
    Ls = end - start
    if Ls <= 0:
        # No change in convolution
        return 2.0 * n * float(c.max()) / (S_new * S_new)
    
    # delta_c at active positions
    infl = 2.0 * a[0:Ls]  # 2*a[k] for k in 0..Ls-1 (these are a[k-i] for k in i..i+Ls-1)
    aff = c[start:end] + w * infl
    # Add w^2 at position 2i if in range
    diag = 2*i
    if start <= diag < end:
        aff[diag - start] += w * w
    
    # Max over active
    m_active = float(aff.max())
    # Max over inactive
    m_inactive = float(c.max())  # This is always >= max over active old values
    # Actually m_inactive should exclude the active range... but it's an upper bound
    # For exact: we need max of c outside [start, end)
    m_out = 0.0
    if start > 0:
        m_out = max(m_out, float(c[:start].max()))
    if end < 2*n - 1:
        m_out = max(m_out, float(c[end:].max()))
    
    m_new = max(m_active, m_out)
    return 2.0 * n * m_new / (S_new * S_new)

def _get_parent():

def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(c, 0.0)


def _score_exact_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _softmax_fn(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        return src.copy()
    m = src.size
    xp = (np.arange(m) + 0.5) / m
    fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
    out = np.maximum(fp, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def _lse_phase(p, stop_time, consider=None, seed=42):
    """LSE smoothing optimization with fine temperature schedule."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):

        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-5 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            k = int(np.argmax(c))
            w[k] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    # Extended temperature schedule: 14 stages
    stages = (
        (0.10, 0.06),
        (0.04, 0.08),
        (0.015, 0.10),
        (0.005, 0.12),
        (0.0015, 0.14),
        (0.0005, 0.14),
        (0.00015, 0.16),
        (0.00005, 0.16),
        (0.000015, 0.14),
        (0.000005, 0.12),
        (0.000001, 0.10),
        (0.0000003, 0.10),
    )

    start = time.monotonic()

    if avail <= 8.0:
        return p

    best_p = p.copy()
    best_sc = _score_exact_arr(p)
    total_w = float(sum(w for _, w in stages))
    rng = np.random.default_rng(seed)
    cum = 0.0

    for si, (frac, wt) in enumerate(stages):
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 4.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)

        best_u = u.copy()
        f0, _ = lse_func(best_u, T)
        if not math.isfinite(f0):
            continue
        best_f = f0
        stale = 0

        while time.monotonic() < end_stage - 0.5:
            maxit = 30 if si >= 6 else 20
            opts = {"maxiter": maxit, "gtol": 1e-12, "ftol": 1e-16,
                    "maxls": 25, "maxcor": 25, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-15:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            best_u = best_u + rng.normal(0, 0.003, best_u.size)
                            continue
            if getattr(res, "status", 1) == 0:
                if time.monotonic() < end_stage - 1.5 and stale < 3:
                    best_u = best_u + rng.normal(0, 0.002, best_u.size)
                    f_chk, _ = lse_func(best_u, T)
                    if math.isfinite(f_chk):
                        if f_chk < best_f:
                            best_f = f_chk
                        stale += 1
                else:
                    break

        p_cur = _softmax_fn(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_exact_arr(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif math.isfinite(sc_cur) and sc_cur < best_sc + 0.003:
            p = p_cur.copy()
        else:
            p = best_p.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_descent(p, stop_time, G=40, seed=99, consider=None, max_sweeps=500):
    """Coordinate descent with vectorized grid + Brent refinement."""
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0

    best_score = cur_score
    best_p = p.copy()

    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.5:
        improved = False
        # Impact-ordered: process high-value coordinates first
        order = np.argsort(-p)

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            infl = 2.0 * p[:Ls]

            # Outside max
            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:

                if mr > max_const:
                    max_const = mr

                x_hi = max(1e-8, S / float(n))

            # Grid evaluation (vectorized)
            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)

            w_grid = pts - v
            denom = S + w_grid
            valid = denom > 1e-15
            vals = np.full(pts.shape, math.inf, dtype=np.float64)
            if np.any(valid):
                aff = c[start:end][None, :] + w_grid[valid, None] * infl[None, :]
                diag_i = 2 * i
                if start <= diag_i < end:
                    aff[valid, diag_i - start] += w_grid[valid] * w_grid[valid]
                aff = np.maximum(aff, max_const)

            if not np.any(np.isfinite(vals)):

                            return math.inf

                        affv = np.maximum(affv, max_const)
                        mv2 = float(affv.max())

                            best_val = vr
                    except Exception:
                        pass

                w = best_x - v

                    S_total += w
                    c[start:end] = c[start:end] + w * infl
                    di = 2 * i

                    cur_score = best_val
                    improved = True

def _structured_perturb(p, rng, mode=0):
    """Apply structured perturbation to escape local minima."""
    n = p.size
    q = p.copy()
    if mode == 0:
        # Smooth wave
        freq = 1.0 + 3.0 * rng.random()
        phase = 2.0 * math.pi * rng.random()
        x = np.arange(n, dtype=np.float64) / n
        q = q * (1.0 + 0.05 * np.sin(2.0 * math.pi * freq * x + phase))
    elif mode == 1:
        # Two-segment adjustment
        split = int(n * (0.3 + 0.4 * rng.random()))
        amp = 0.03 * rng.random()
        q[:split] *= (1.0 + amp)
        q[split:] *= (1.0 - amp)
    elif mode == 2:
        # Local bump
        ctr = int(n * rng.random())
        width = max(5, int(n * 0.05))
        idx = np.arange(n)
        q *= (1.0 + 0.04 * np.exp(-0.5 * ((idx - ctr) / width) ** 2))
    else:
        # Random Gaussian
        q = q * np.exp(rng.normal(0, 0.05, n))
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if s > 0:
        q /= s
    return q
