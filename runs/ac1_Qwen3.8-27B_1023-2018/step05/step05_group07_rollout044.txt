import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft

def _conv_max(a):
    """Compute max of convolution of a with itself."""
    n = len(a)
    # Use FFT for efficiency
    fft_len = 2 * n - 1
    # Pad to next power of 2 for efficiency
    size = 1
    while size < fft_len:
        size *= 2
    fa = fft(a, size)
    conv = np.real(ifft(fa * fa, size))[:fft_len]
    return float(np.max(conv)), conv

def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    M, _ = _conv_max(a)
    return 2.0 * len(a) * M / (s * s)

f(t) = max_k ((1-t)a + tg) * ((1-t)a + tg))_k

Actually, it's (c(t))^T P_k (c(t)) where c(t) = (1-t)a + tg, which is a quadratic in t. The Hessian with respect to t is 2 * g^T P_k g, which could be negative (as discussed earlier, P_k is not PSD).
