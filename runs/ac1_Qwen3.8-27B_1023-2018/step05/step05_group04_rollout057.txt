        out = src.copy()
    else:
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        xp_new = (np.arange(n, dtype=np.float64) + 0.5) / n
        out = np.interp(xp_new, xp, src)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _lse_optimize(p, stop_time, consider=None):
    n = p.size
    p = np.maximum(np.asarray(p, dtype=np.float64), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    u = np.log(np.maximum(p, 1e-300), dtype=np.float64)
    u = u - float(u.mean())

    conv_len = 2 * n - 1
    conv_size = next_fast_len(conv_len)
    cross_size = next_fast_len(3 * n - 1)

    def lse_func(uu, T):
        pp = _softmax(uu)
        F = rfft(pp, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        T_eff = max(T, 1e-300)
        scaled = (c - cmax) / T_eff
        max_s = float(scaled.max())
        ex = np.exp(scaled - max_s, dtype=np.float64)
        se = float(ex.sum())
        if not math.isfinite(se) or se <= 0:
            w = np.zeros(conv_len, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            w = ex / se
            f = cmax + T_eff * (max_s + math.log(se / conv_len))

        F_wr = rfft(w[::-1], cross_size)
        F_p = rfft(pp, cross_size)
        ccorr = irfft(F_wr * F_p, cross_size)
        g = ccorr[n - 1:2 * n - 1][::-1]
        grad_pp = 2.0 * g
        tdp = float(np.dot(pp, grad_pp))
        grad_u = pp * (grad_pp - tdp)
        grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
        return f, grad_u

    c_init = _conv_self(p, conv_size, conv_len)
    cmax_init = float(c_init.max())
    if not math.isfinite(cmax_init) or cmax_init <= 0:
        return p

    stages = [
        (0.08, 0.08),
        (0.04, 0.10),
        (0.02, 0.12),
        (0.01, 0.13),
        (0.005, 0.14),
        (0.002, 0.15),
        (0.0008, 0.16),
        (0.0003, 0.12),
    ]

    best_p_out = p.copy()
    best_sc = _score_arr(p)
    total_w = sum(wt for _, wt in stages)
    start = time.monotonic()
    avail = max(0.0, stop_time - start - 8.0)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        cum += wt
        end_stage = min(stop_time - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue
        T = frac * cmax_init
        best_u = u.copy()
        stale = 0
        while time.monotonic() < end_stage - 1.5:
            maxit = 35
            opts = {"maxiter": maxit, "gtol": 1e-12, "ftol": 1e-16, "maxls": 20, "maxdisp": 0}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                pp_check = _softmax(res.x)
                sc = _score_arr(pp_check)
                if math.isfinite(sc) and sc < best_sc - 1e-13:
                    best_sc = sc
                    best_p_out = pp_check.copy()
                    p = pp_check.copy()
                    u = np.log(np.maximum(p, 1e-300), dtype=np.float64)
                    u = u - float(u.mean())
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
            else:
                break
            if getattr(res, "status", 1) == 0 and stale > 0:
                break
        if consider:
            try:
                consider(best_p_out)
            except Exception:
                pass

    sc_final = _score_arr(best_p_out)
    if math.isfinite(sc_final) and sc_final < best_sc:
        best_sc = sc_final
    return best_p_out


def _coord_descent(p, stop_time, consider=None, seed=42, G=64):
    n = int(p.size)
    if n == 0:
        return p, float('inf')
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p, float('inf')
    p = p / s
    conv_len = 2 * n - 1
    conv_size = next_fast_len(conv_len)
    c = _conv_self(p, conv_size, conv_len)
    cur_max = float(c.max())
    cur_score = 2.0 * n * cur_max
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    GR = (math.sqrt(5.0) - 1.0) / 2.0

    def eval_1d(w, base, infl, diag, max_const):
        s_new = 1.0 + w
        if s_new <= 1e-15:
            return float('inf')
        aff_max = float(np.max(base + w * infl)) if base.size > 0 else 0.0
        if 0 <= diag < len(base):
            dval = base[diag] + w * infl[diag] + w * w
            if dval > aff_max:
                aff_max = dval
        m = max(aff_max, max_const)
        return 2.0 * n * m / (s_new * s_new)

    max_sweeps = 50000
    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 1.0:
                break
            v = float(p[i])
            start_idx = i
            end_idx = min(i + n, conv_len)
            Ls = end_idx - start_idx
            if Ls <= 0:
                continue
            base = c[start_idx:end_idx].copy()
            infl = 2.0 * p[:Ls]
            diag = i

            max_const = 0.0
            if start_idx > 0:
                mc = float(c[:start_idx].max())
                if mc > max_const:
                    max_const = mc
            if end_idx < conv_len:
                mc = float(c[end_idx:].max())
                if mc > max_const:
                    max_const = mc

            x_hi = max(20.0 * v, 10.0 / n, 0.005)
            x_hi = min(x_hi, 0.5)
            x_hi = max(x_hi, 1e-8)

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            ins = np.searchsorted(pts, v)
            if ins > 0 and ins < G + 1:
                pts = np.concatenate((pts[:ins], [v], pts[ins:]))

            ws = pts - v
            s_new = 1.0 + ws
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag < Ls:
                aff[:, diag] += ws * ws
            maff = np.max(aff, axis=1)
            maff = np.maximum(maff, max_const)
            valid = s_new > 1e-12
            scores = np.full(pts.shape, float('inf'), dtype=np.float64)
            if np.any(valid):
                scores[valid] = 2.0 * n * maff[valid] / (s_new[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            if 0 < imin < len(pts) - 1:
                lo = float(pts[max(0, imin - 2)])
                hi = float(pts[min(len(pts) - 1, imin + 2)])
            else:
                lo = 0.0
                hi = x_hi

            a_gs = lo
            b_gs = hi
            if b_gs - a_gs < 1e-14:
                a_gs = max(0.0, best_x - 1e-5)
                b_gs = best_x + 1e-5
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)
            f_c = eval_1d(c_gs - v, base, infl, diag, max_const)
            f_d = eval_1d(d_gs - v, base, infl, diag, max_const)
            for _ in range(20):
                if b_gs - a_gs < 1e-12 * max(1.0, abs(best_x)):
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = eval_1d(c_gs - v, base, infl, diag, max_const)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_d = eval_1d(d_gs - v, base, infl, diag, max_const)
            x_mid = (a_gs + b_gs) / 2.0
            f_mid = eval_1d(x_mid - v, base, infl, diag, max_const)
            if f_mid < best_val:
                best_x = x_mid
                best_val = f_mid

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= x_hi:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    aff_seg = base + w * infl
                    if 0 <= diag < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag] += w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    c[start_idx:end_idx] = aff_seg
                    cur_score = 2.0 * n * new_max
                    improved = True

        s_new = float(p.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            p = p / s_new
            c = _conv_self(p, conv_size, conv_len)
            new_score = 2.0 * n * float(c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-14:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 12.0:
                    kick_size = 0.02 + 0.03 * min(1.0, flat / 15.0)
                    p = p * np.exp(rng.normal(0.0, kick_size, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p, conv_size, conv_len)
                    cur_score = 2.0 * n * float(c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 5.0:
        hard = t0 + max(3.0, budget - 2.0)

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float('inf')

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_score = sc
            best_arr = a.copy()

    consider(parent)

    n_work = 4096

    # Phase 1: LSE on resampled parent at n_work
    p_work = _resample(best_arr, n_work)
    consider(p_work)
    lse_time = min(180.0, 0.20 * budget)
    lse_end = t0 + lse_time
    if lse_end > time.monotonic() + 20.0:
        try:
            p2 = _lse_optimize(p_work, lse_end, consider=consider)
            if p2 is not None and p2.size == n_work and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p_work = p2 / float(p2.sum())
                consider(p_work)
        except Exception:
            pass

    # Phase 2: Coordinate descent
    cd_end = hard - min(60.0, 0.25 * budget)
    p_start = _resample(best_arr, n_work)
    if cd_end > time.monotonic() + 30.0:
        try:
            p_cd, _ = _coord_descent(p_start, cd_end, consider=consider, seed=int(seed + 777), G=64)
            if p_cd is not None and p_cd.size > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                consider(p_cd)
        except Exception:
            pass

    # Phase 3: Alternative initialization (raised cosine) + CD
    rem = hard - 4.0 - time.monotonic()
    if rem > 40.0:
        x = np.arange(n_work, dtype=np.float64)
        mid = (n_work - 1) / 2.0
        rc = 0.5 * (1.0 + np.cos(np.pi * x / (n_work - 1)))
        s_rc = float(rc.sum())
        if s_rc > 0:
            rc /= s_rc
        p_blend = 0.3 * rc + 0.7 * _resample(best_arr, n_work)
        p_blend = np.maximum(p_blend, 0.0)
        p_blend /= float(p_blend.sum())
        consider(p_blend)
        blend_end = min(hard - 4.0, time.monotonic() + rem * 0.5)
        if blend_end > time.monotonic() + 15.0:
            try:
                p_b, _ = _coord_descent(p_blend, blend_end, consider=consider, seed=int(seed + 999), G=48)
                if p_b is not None and p_b.size > 0 and np.all(np.isfinite(p_b)) and p_b.sum() > 0:
                    consider(p_b)
            except Exception:
                pass

    # Phase 4: Final LSE polish on overall best at low temperature
    rem = hard - 4.0 - time.monotonic()
    if rem > 30.0:
        p_final = _resample(best_arr, n_work)
        polish_end = min(hard - 3.0, time.monotonic() + rem * 0.7)
        if polish_end > time.monotonic() + 15.0:
            try:
                p_pol = _lse_optimize(p_final, polish_end, consider=consider)
                if p_pol is not None and p_pol.size == n_work and np.all(np.isfinite(p_pol)) and p_pol.sum() > 0:
                    consider(p_pol)
            except Exception:
                pass

    # Phase 5: Final CD squeeze
    rem = hard - 3.0 - time.monotonic()
    if rem > 20.0:
        p_sq = _resample(best_arr, n_work)
        try:
            p_f, _ = _coord_descent(p_sq, hard - 2.0, consider=consider, seed=int(seed + 1111), G=80)
            if p_f is not None and p_f.size > 0 and np.all(np.isfinite(p_f)) and p_f.sum() > 0:
                consider(p_f)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in np.clip(parent, 0.0, 1000.0)]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
