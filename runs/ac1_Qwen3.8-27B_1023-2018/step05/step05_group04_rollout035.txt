    return np.asarray(c, dtype=np.float64)

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float('inf')
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return float('inf')
    return float(2.0 * n * m / (s * s))

def _softmax(u):
    u = u - np.max(u)
    e = np.exp(u)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.shape, 1.0 / u.size)
    return e / s

def _adam_lse_optimize(p0, stop_time, n_iter_max=50000):
    """Adam-based LSE optimization."""
    n = p0.size
    p = np.maximum(np.asarray(p0, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p0
    p = p / s
    u = np.log(np.maximum(p, 1e-30))
    u -= float(u.mean())
    
    conv_size = next_fast_len(2*n - 1)
    conv_len = 2*n - 1
    cross_N = next_fast_len(3*n - 2)
    
    def compute_b(a):
        F = rfft(a, conv_size)
        return irfft(F * F, conv_size)[:conv_len]
    
    def lse_func(uu, T):
        a = _softmax(uu)
        b = compute_b(a)
        bmax = float(b.max())
        if not math.isfinite(bmax) or bmax <= 0:
            return bmax if math.isfinite(bmax) else 1e100, np.zeros_like(uu)
        scaled = (b - bmax) / max(T, 1e-300)
        exps = np.exp(scaled)
        se = float(exps.sum())
        if se <= 0 or not math.isfinite(se):
            w = np.zeros(conv_len)
            w[int(np.argmax(b))] = 1.0
            f = bmax
        else:
            f = bmax + T * float(np.log(se / conv_len))
            w = exps / se
        # Gradient
        W = np.fft.rfft(w, cross_N)
        A = np.fft.rfft(a, cross_N)
        s_cross = np.fft.irfft(W * np.conj(A), cross_N)[:n].real
        # Correction
        idx2 = 2 * np.arange(n)
        corr = np.zeros(n)
        valid = idx2 < conv_len
        corr[valid] = w[idx2[valid]] * a[valid]
        grad_a = 2.0 * s_cross - corr
        tdp = float(np.dot(a, grad_a))
        grad_u = a * (grad_a - tdp)
        grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
        return f, grad_u
    
    # Temperature schedule (geometric)
    b0 = compute_b(p)
    bmax0 = float(b0.max())
    if not math.isfinite(bmax0) or bmax0 <= 0:
        return p0
    
    T_start = 0.05 * bmax0
    T_end = 0.0005 * bmax0
    n_phases = 8
    phase_ratio = (T_end / T_start) ** (1.0 / n_phases)
    
    best_f = float('inf')
    best_u = u.copy()
    best_score = _score(p)
    best_p = p.copy()
    
    # Adam state
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    lr = 0.005
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    
    start = time.monotonic()
    iters_done = 0
    
    for phase in range(n_phases):
        T = T_start * (phase_ratio ** phase)
        steps_this_phase = n_iter_max // n_phases
        for step in range(steps_this_phase):
            if time.monotonic() > stop_time - 5.0:
                break
            if iters_done >= n_iter_max:
                break
            
            f_val, g = lse_func(u, T)
            if not math.isfinite(f_val):
                break
            iters_done += 1
            
            t_adam += 1
            m_adam = beta1 * m_adam + (1 - beta1) * g
            v_adam = beta2 * v_adam + (1 - beta2) * g * g
            m_hat = m_adam / (1 - beta1 ** t_adam)
            v_hat = v_adam / (1 - beta2 ** t_adam)
            
            step_size = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            u_new = u - step_size
            u_new = np.where(np.isfinite(u_new), u_new, u)
            
            # Check if this is better
            f_new = f_val  # approximate (we already computed it)
            if math.isfinite(f_new) and f_new < best_f:
                best_f = f_new
                best_u = u_new.copy()
            
            u = u_new
        
        # At phase boundary, evaluate score
        p_check = _softmax(best_u)
        p_check = np.maximum(p_check, 0.0)
        sc = float(p_check.sum())
        if sc > 0:
            p_check /= sc
        sc_val = _score(p_check)
        if math.isfinite(sc_val) and sc_val < best_score:
            best_score = sc_val
            best_p = p_check.copy()
    
    return best_p
