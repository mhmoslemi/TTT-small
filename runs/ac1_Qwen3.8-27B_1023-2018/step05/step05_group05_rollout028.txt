import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:
        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(3072, dtype=np.float64)
    return p


def _conv_self(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return c


def _score_fft(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _build_conv_matrix(f, n, active_k=None):
    """Build constraint submatrix for lags in active_k (or all lags if None)."""
    if active_k is None:
        active_k = np.arange(2 * n - 1)
    ka = np.asarray(active_k)
    cols = np.arange(n)[None, :]
    diff = ka[:, None] - cols
    mask = (diff >= 0) & (diff < n)
    idx = np.clip(diff, 0, n - 1)
    A = f[idx]
    A[~mask] = 0.0
    return A


def _lp_phase(f, stop_time, max_iters=20, consider=None):
    """LP-based direction search: find g maximizing sum(g) s.t. (f*g)[k] <= M for active k."""
    n = f.size
    f = np.maximum(f, 0).copy()
    s = float(f.sum())
    if s < 1e-300:
        return f
    f = f / s
    best_f = f.copy()
    best_sc = _score_fft(f)

    for it in range(max_iters):
        if time.monotonic() > stop_time - 30.0:
            break
        c = _conv_self(f)
        M = float(c.max())
        if not math.isfinite(M) or M <= 0:
            break

        # Use active constraints (top 60 lags by value)
        thresh = 0.92 * M
        active_k = np.where(c >= thresh)[0]
        if active_k.size > 80:
            top_idx = np.argsort(c)[::-1][:80]
            active_k = top_idx
        if active_k.size < 1:
            active_k = np.array([int(np.argmax(c))])

        try:
            A = _build_conv_matrix(f, n, active_k)
            c_obj = -np.ones(n)
            b_ub = np.full(active_k.size, M)
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': 60.0})
            if res.success and res.x is not None and np.all(np.isfinite(res.x)):
                g = np.maximum(res.x, 0)
                gs = float(g.sum())
                if gs < 1e-300:
                    g = np.ones(n) / n
                else:
                    g = g / gs
            else:
                g = np.ones(n) / n
        except Exception:
            g = np.ones(n) / n

        # Adaptive step size with line search
        improved = False
        for t in [0.6, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02]:

            f_new = np.maximum(f_new, 0)
            ss = float(f_new.sum())
            if ss < 1e-300:
                continue
            f_new = f_new / ss
            sc = _score_fft(f_new)
            if math.isfinite(sc) and sc < best_sc - 1e-14:
                f = f_new
                best_f = f.copy()
                best_sc = sc
                improved = True
                if consider is not None:
                    try:
                        consider(f)
                    except Exception:
                        pass
                break

        if not improved:
            # Try perturbation
            rng = np.random.default_rng(1000 + it)
            f_new = f * np.exp(rng.normal(0, 0.05, n))
            f_new = np.maximum(f_new, 0)
            ss = float(f_new.sum())
            if ss > 0:
                f_new /= ss
                sc = _score_fft(f_new)
                if math.isfinite(sc) and sc < best_sc:
                    f = f_new
                    best_f = f.copy()
                    best_sc = sc

    return best_f


def _lse_optimize(p, stop_time, consider=None, fine_only=False):
    """LSE-based optimization using softmax parameterization."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p
    p = p / s

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(np.max(c))
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = 1e-4 * cmax
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            w = exps / sum_exp
        q = xfast(pp, w)
        g = 2.0 * T * q
        tdp = float(np.dot(pp, g))
        grad = pp * (g - tdp)
        return float(f), np.where(np.isfinite(grad), grad, 0.0)

    def p_to_u(pp):
        ll = np.log(np.maximum(pp, 1e-300))
        return ll - float(np.mean(ll))

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score_fft(p)

    if fine_only:
        stages = ((0.005, 0.25), (0.0015, 0.30), (0.0004, 0.30), (0.0001, 0.30))
    else:
        stages = (
            (0.2, 0.08),
            (0.06, 0.12),
            (0.02, 0.12),
            (0.006, 0.14),
            (0.002, 0.15),
            (0.0006, 0.16),
            (0.00015, 0.23),
        )

    total_w = sum(w for _, w in stages)
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        remaining = stop_time - now - 5.0
        time_alloc = max(3.0, (wt / total_w) * remaining)
        end_stage = time.monotonic() + time_alloc
        cmax = float(_conv_self(p).max())
        T = max(1e-15, frac * cmax)
        stale = 0
        while time.monotonic() < end_stage - 1.0:
            opts = {"maxiter": 60, "gtol": 1e-13, "ftol": 1e-16,
                    "maxls": 50, "maxcor": 30, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                p_cur = _softmax(res.x)
                sc_cur = _score_fft(p_cur)
                if math.isfinite(sc_cur) and sc_cur < best_sc - 1e-15:
                    best_sc = sc_cur
                    best_p = p_cur.copy()
                    p = p_cur.copy()
                    u = p_to_u(p)
                    stale = 0
                    if consider is not None:
                        try:
                            consider(p)
                        except Exception:
                            pass
                else:
                    stale += 1
                    if stale >= 4:
                        break
            if getattr(res, "status", 1) == 0:
                break

    p_out = best_p.copy()
    ss = float(p_out.sum())
    if ss > 1e-300:
        p_out = p_out / ss
    return p_out


def _coord_descent(p, stop_time, consider=None, seed=42, max_sweeps=100000):
    """Coordinate descent with FFT-based convolution tracking."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p, math.inf
    p = p / s

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S = 1.0
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    flat = 0
    sweep = 0

    # Adaptive grid size
    G_base = max(24, min(64, int(80000 / max(n, 1)) + 24))

    while sweep < max_sweeps and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S <= 1e-300:
                break

            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start > 0:
                ml = float(np.max(c[:start]))
                max_const = max(max_const, ml)
            if end < Ltot:
                mr = float(np.max(c[end:]))
                max_const = max(max_const, mr)

            # Adaptive upper bound for x
            x_hi = max(50.0 * v, 100.0 / n, 0.01)
            x_hi = min(x_hi, 10.0 / max(S, 1e-12))
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = 1.0 / n

            G = G_base
            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 < v < x_hi:
                pts = np.unique(np.concatenate([pts, [v]]))

            w = pts - v
            denom = S + w
            valid = denom > 1e-14
            if not np.any(valid):
                continue
            aff = base[None, :] + w[:, None] * infl[None, :]
            dvals = base[i - start] + w * infl[i - start] + w * w
            aff = np.maximum(aff, dvals[:, None])
            if max_const > 0:
                aff = np.maximum(aff, max_const)
            mv = np.max(aff, axis=1)
            vals = np.full(pts.shape, math.inf)
            vals[valid] = (2.0 * n * mv[valid]) / (denom[valid] * denom[valid])

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            bx = float(pts[imin])
            bv = float(vals[imin])

            # Fine search
            dw = x_hi / G
            lo = max(0.0, bx - 3.0 * dw)
            hi = min(x_hi, bx + 3.0 * dw)
            if hi > lo:
                fine = np.linspace(lo, hi, 50)
                if lo < v < hi:
                    fine = np.unique(np.concatenate([fine, [v]]))
                w2 = fine - v
                denom2 = S + w2
                valid2 = denom2 > 1e-14
                if np.any(valid2):
                    aff2 = base[None, :] + w2[:, None] * infl[None, :]
                    dvals2 = base[i - start] + w2 * infl[i - start] + w2 * w2
                    aff2 = np.maximum(aff2, dvals2[:, None])
                    if max_const > 0:
                        aff2 = np.maximum(aff2, max_const)
                    mv2 = np.max(aff2, axis=1)
                    vals2 = np.full(fine.shape, math.inf)
                    vals2[valid2] = (2.0 * n * mv2[valid2]) / (denom2[valid2] ** 2)
                    if np.any(np.isfinite(vals2)):
                        im2 = int(np.argmin(vals2))
                        if float(vals2[im2]) < bv:
                            bv = float(vals2[im2])
                            bx = float(fine[im2])

            if math.isfinite(bv) and bv < cur_score - 1e-13 and 0.0 <= bx <= 20.0 * S:
                w_new = bx - v
                if abs(w_new) > 1e-15:
                    p[i] = bx
                    S += w_new
                    c[start:end] += w_new * infl
                    c[start + (i - start)] += w_new * w_new
                    cur_score = bv
                    improved = True

        # Periodic full reconv

            c = _conv_self(p)
        new_sc = 2.0 * n * float(c.max())
