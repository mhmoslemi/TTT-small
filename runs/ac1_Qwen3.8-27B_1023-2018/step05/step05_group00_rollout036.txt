def _lse_optimize(a, deadline, consider):
    """Optimize using log-sum-exp approximation with L-BFGS."""
    n = a.size
    conv_size = next_fast_len(2*n - 1)
    conv_len = 2*n - 1
    
    # Normalize a to sum = 1
    a = a / float(a.sum())
    
    def softmax(u):
        m = u.max()
        e = np.exp(u - m)
        s = e.sum()
        if s <= 0 or not np.isfinite(s):
            return np.full(n, 1.0/n)
        return e / s
    
    u0 = np.log(np.maximum(a, 1e-300))
    u0 = u0 - u0.mean()
    
    def lse_obj(uu, T):
        aa = softmax(uu)
        c = conv_self(aa)
        cmax = c.max()
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = exps.sum()
        if sum_exp <= 0 or not np.isfinite(sum_exp):
            return cmax, np.zeros(n)
        f = cmax + T * math.log(sum_exp / conv_len)
        w = exps / sum_exp  # softmax weights on c
        
        # Gradient: df/da_i = sum_k w[k] * (a[k-i] + a[i-k])
        # = cross-correlation terms
        # Compute using FFT
        F_w = rfft(w, conv_size)
        F_a = rfft(aa, conv_size)
        # conv(w, a) gives H[i] = sum_k w[k] * a[i-k] (the H term)
        H = irfft(F_w * F_a, conv_size)[:n]
        # For G[i] = sum_k w[k] * a[k-i]:
        # G[i] = (aa * w_rev)[n-1-i] where w_rev = w[::-1]
        F_wr = rfft(w[::-1], conv_size)
        G_rev = irfft(F_a * F_wr, conv_size)[:conv_len]
        G = G_rev[n-1::-1][:n]
        
        grad_a = G + H
        mean_g = np.dot(grad_a, aa)
        grad_u = aa * (grad_a - mean_g)
        
        return f, grad_u
    
    best_u = u0.copy()
    best_score_local = score(softmax(u0))
    
    # Multi-stage temperature
    c0 = conv_self(softmax(u0))
    cmax0 = c0.max()
    
    stages = [
        (0.05, 0.15),  # (T fraction, time fraction)
        (0.02, 0.15),
        (0.008, 0.15),
        (0.003, 0.15),
        (0.001, 0.15),
        (0.0004, 0.15),
        (0.0001, 0.10),
    ]
    
    total_w = sum(w for _, w in stages)
    cum = 0.0
    for frac, wt in stages:
        if time.monotonic() > deadline - 10:
            break
        cum += wt
        stage_end = min(deadline - 8, t0 + (deadline - t0) * cum / total_w)
        T = max(1e-10, frac * cmax0)
        
        opts = {
            "maxiter": 50,
            "gtol": 1e-14,
            "maxls": 30,
            "maxcor": 30,
            "disp": False,
        }
        
        stalled = 0
        while time.monotonic() < stage_end - 2:
            try:
                res = minimize(lse_obj, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            
            if res.x is not None and np.all(np.isfinite(res.x)):
                x = res.x
                fval = float(res.fun)
                aa = softmax(x)
                sc = score(aa)
                if math.isfinite(fval) and fval < 0:  # fval can be negative due to LSE shift
                    best_u = x.copy()
                    stalled = 0
                elif math.isfinite(sc) and sc < best_score_local:
                    best_u = x.copy()
                    best_score_local = sc
                    stalled = 0
                else:
                    stalled += 1
                    if stalled >= 4:
                        break
            if getattr(res, "status", 1) == 0:
                break
        
        # Update cmax0 for next stage
        c_now = conv_self(softmax(best_u))
        cmax0 = c_now.max()
    
    result = softmax(best_u)
    consider(result)
    return result

    n = a.size

    c = conv_self(a)
    cur_score = 2.0 * n * c.max()
    best_score = cur_score
    best_a = a.copy()
    
    rng = np.random.default_rng(seed)

        improved = False
        order = rng.permutation(n)
        
        for idx in order:
            i = int(idx)
            if time.monotonic() > deadline - 2:
                break
            
            v = a[i]
            # Determine the affected range of c
            lo = max(0, i - n + 1)
            hi = min(conv_len - 1, i + n - 1)
            L = hi - lo + 1
            
            if L <= 0:
                continue
            
            # Pre-compute the max outside the affected range
            max_out = 0.0
            if lo > 0:
                max_out = max(max_out, c[:lo].max())
            if hi < conv_len - 1:
                max_out = max(max_out, c[hi+1:].max())

            # where base = c before the change
            base = c[lo:hi+1].copy()
            # infl[k] = a[k-i] + a[i-k] for k in [lo, hi]
            infl = np.zeros(L)
            for k in range(lo, hi+1):
                ki = k - i  # index for a[k-i]
                ik = i - k  # index for a[i-k]
                val = 0.0
                if 0 <= ki < n:
                    val += a[ki]
                if 0 <= ik < n:
                    val += a[ik]
                infl[k - lo] = val
            
            # Find good range for delta = new_a[i] - v
            avg = 1.0 / n  # since a sums to 1
            x_lo = 0.0
            x_hi = max(20.0 * v, 50.0 * avg, 0.02)
            
            # Grid search
            pts = np.linspace(x_lo, x_hi, G + 1)
            if 0 <= v <= x_hi:
                # Insert v into the grid
                pos = np.searchsorted(pts, v)
                pts = np.insert(pts, pos, v)
            
            # Evaluate each point
            di = 2 * i - lo  # index of k=2i in the affected range (if valid)
            best_x = v
            best_val = cur_score
            
            for x in pts:
                delta = x - v
                if delta == 0:
                    new_max = c.max()
                else:
                    new_c_affected = base + delta * infl
                    if 0 <= di < L:
                        new_c_affected[di] += delta * delta
