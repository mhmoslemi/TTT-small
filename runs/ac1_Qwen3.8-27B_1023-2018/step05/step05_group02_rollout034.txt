def _score_exact_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * a.size * m / (s * s))


def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == 1:
        return np.full(n, float(src[0]) / n)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m) + 0.5) / m
        fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _lbfgs_optimize(p, n, stop_time, consider=None, fine_only=False):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0.0:
            return 1e100, np.zeros_like(uu)
        if T <= 0.0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -400.0, 400.0)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    def run_chunk(uu, T, end_stage):
        try:
            f0, _ = lse_func(uu, T)
        except Exception:
            return uu
        if not math.isfinite(f0):
            return uu
        best_u = np.array(uu, dtype=np.float64, copy=True)
        best_f = float(f0)
        stale = 0
        while time.monotonic() < end_stage - 1.5:
            maxit = 40 if n > 2000 else 60
            opts = {
                "maxiter": maxit,
                "gtol": 1e-14,
                "ftol": 1e-16,
                "maxls": 30,
                "maxcor": 40,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == uu.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-16:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 4:
                            break
            if getattr(res, "status", 1) == 0:
                break
        return best_u

    if fine_only:
        stages = ((0.001, 0.35), (0.0002, 0.45), (0.00005, 0.20))
    else:
        stages = (
            (0.12, 0.08),
            (0.04, 0.12),
            (0.012, 0.14),
            (0.003, 0.18),
            (0.0008, 0.24),
            (0.0002, 0.24),
        )

    start = time.monotonic()
    avail = float(stop_time - start - 8.0)
    if avail <= 4.0:
        return p

    best_p = p.copy()
    best_sc = _score_fft(p)
    total_w = float(sum(w for _, w in stages))
    if total_w <= 0.0:
        total_w = 1.0

    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 4.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        sc_cur = _score_fft(p_cur)
        if not math.isfinite(sc_cur):
            p = best_p.copy()
            u = p_to_u(p)
            continue
        if sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif sc_cur > best_sc + max(0.002, 2e-4 * best_sc):
            p = best_p.copy()
        else:
            p = p_cur.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, max_sweeps=200000, G=None, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    if G is None:
        if n <= 500:
            G = 50
        elif n <= 1000:
            G = 40
        elif n <= 2000:
            G = 32
        else:
            G = 24
    G = int(max(12, G))

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    max_flat = 8

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.25 or S_total < 0.75:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls].copy()

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(20.0 * v, 60.0 * S / n, 0.05 * S, 1e-12)
            x_hi = min(x_hi, 10.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, S / n)

            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 <= v <= x_hi:
                pts = np.unique(np.concatenate((pts, [v])))

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                di = 0
                if di < Ls:
                    dval = base[di] + w * infl[di] + w * w
                    maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if G > 1:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.5 * dw)
                hi = min(x_hi, best_x + 2.5 * dw)
                if hi <= lo:
                    hi = lo + 1e-14
                fine_pts = np.linspace(lo, hi, 50)
                if 0.0 <= best_x <= hi:
                    fine_pts = np.unique(np.concatenate((fine_pts, [best_x])))
                vals2 = eval_xs(fine_pts)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine_pts[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2

                gr = (math.sqrt(5.0) - 1.0) / 2.0
                g_lo = max(0.0, best_x - dw)
                g_hi = min(x_hi, best_x + dw)
                if g_hi - g_lo > 1e-16:
                    a_gs = g_hi - gr * (g_hi - g_lo)
                    b_gs = g_lo + gr * (g_hi - g_lo)
                    fa = float(eval_xs(np.array([a_gs]))[0])
                    fb = float(eval_xs(np.array([b_gs]))[0])
                    for _ in range(16):
                        if not math.isfinite(fa):
                            fa = 1e100
                        if not math.isfinite(fb):
                            fb = 1e100
                        if fa < fb:
                            g_hi = b_gs
                            b_gs = a_gs
                            fb = fa
                            a_gs = g_hi - gr * (g_hi - g_lo)
                            fa = float(eval_xs(np.array([a_gs]))[0])
                        else:
                            g_lo = a_gs
                            a_gs = b_gs
                            fa = fb
                            b_gs = g_lo + gr * (g_hi - g_lo)
                            fb = float(eval_xs(np.array([b_gs]))[0])
                        if g_hi - g_lo < 1e-16:
                            break
                    gs_x = (g_lo + g_hi) / 2.0
                    gv = float(eval_xs(np.array([gs_x]))[0])
                    if math.isfinite(gv) and gv < best_val:
                        best_x = gs_x
                        best_val = gv

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= 10.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        np.maximum(c, 0.0, out=c)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-16:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= max_flat:
                if time.monotonic() < stop_time - 6.0:
                    try:
                        noise_amt = 0.03 + 0.02 * math.exp(-flat * 0.05)
                        p = p * np.exp(rng.normal(0.0, noise_amt, n))
                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss > 0.0:
                            p /= ss
                        c = _conv_self(p)
                        S_total = 1.0
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider is not None:
                                consider(best_p)
                        flat = 0
                    except Exception:
                        break
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _lp_refine(p, stop_time, consider=None, max_rounds=5):
    """LP-based refinement: find g >= 0 maximizing sum(g) s.t. (p*g)_k <= M."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0)
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    best_p = p.copy()
    best_score = float(2.0 * n * _conv_self(best_p).max())

    for rnd in range(max_rounds):
        if time.monotonic() > stop_time - 3.0:
            break

        c = _conv_self(p)
        M = float(c.max())
        if not math.isfinite(M) or M <= 0:
            break

        # Build sparse matrix A: A[k, j] = p[k-j] if 0 <= k-j < n
        # Rows: k = 0..2n-2, Cols: j = 0..n-1
        rows_list = []
        cols_list = []
        vals_list = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            js = np.arange(j_min, j_max + 1)
            rows_list.extend([k] * len(js))
            cols_list.extend(js.tolist())
            vals_list.extend(p[k - js].tolist())

        A = csr_matrix((vals_list, (rows_list, cols_list)), shape=(2 * n - 1, n))

        try:
            g_var = cp.Variable(n, nonneg=True)
            constraints = [A @ g_var <= M * np.ones(2 * n - 1)]
            obj = cp.Maximize(cp.sum(g_var))
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.GLPK, msg=False)
            if g_var.value is None:
                prob.solve(solver=cp.CBC, msg=False)
            if g_var.value is None:
                prob.solve(solver=cp.ECOS, msg=False)
        except Exception:
            break

        g_val = np.asarray(g_var.value, dtype=np.float64).ravel()
        if g_val.size != n or not np.all(np.isfinite(g_val)):
            break
        g_val = np.maximum(g_val, 0.0)
        gs = float(g_val.sum())
        if gs < 1e-15:
            break
        g_val = g_val / gs

        # Try multiple mixing ratios
        found_improve = False
        for t in (0.3, 0.5, 0.7, 1.0, 0.15, 0.85):
            p_cand = (1.0 - t) * p + t * g_val
            p_cand = np.maximum(p_cand, 0.0)
            ps = float(p_cand.sum())
            if ps < 1e-15:
                continue
            p_cand = p_cand / ps
            sc = float(2.0 * n * _conv_self(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                found_improve = True
                break

        if not found_improve:
            # Try larger mixing
            for t in (0.9, 0.95, 1.0):
                p_cand = (1.0 - t) * p + t * g_val
                p_cand = np.maximum(p_cand, 0.0)
                ps = float(p_cand.sum())
                if ps < 1e-15:
                    continue
                p_cand = p_cand / ps
                sc = float(2.0 * n * _conv_self(p_cand).max())
                if math.isfinite(sc) and sc < best_score - 1e-14:
                    best_score = sc
                    best_p = p_cand.copy()
                    p = p_cand.copy()
                    found_improve = True
                    break

        if not found_improve:
            break

        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _subgradient_polish(p, stop_time, consider=None, max_iters=5000):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    best_p = p.copy()
    best_score = float(2.0 * n * _conv_self(p).max())
    base_step = 0.003
    min_step = 1e-9

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break

        c = _conv_self(p)
        cmax = float(c.max())
        threshold = cmax * 0.9995
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        if len(active) > 50:
            # Take top 50
            idx_sort = np.argsort(c[active])[::-1][:50]
            active = active[idx_sort]

        grad = np.zeros(n)
        for k in active:
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                contrib = p[kj[valid]] + p[valid]
                grad[valid] += contrib
        grad /= len(active)

        # Normalize gradient
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        grad = grad / gn

        step = base_step / (1.0 + 0.01 * it)
        if step < min_step:
            break

        improved = False
        for scale in (1.0, 0.5, 0.25, 0.1, 2.0):
            p_cand = p - scale * step * grad
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = float(2.0 * n * _conv_self(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                improved = True
                break

        if not improved:
            base_step *= 0.92
            if base_step < 1e-12:
                break

        if it % 100 == 0 and consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    usable = max(1.0, hard - t0)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64) / 64.0
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s_val = float(a.sum())
        if s_val < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s_val
            best_arr = a.copy()
            best_score = sc

    def make_skew(src, n, sd):
        p = _resample_mid(src, n)
        rng = np.random.default_rng(sd)
        x = (np.arange(n) + 0.5) / n
        side = 1.0 if rng.random() < 0.5 else -1.0
        amp = 0.3 + 0.6 * float(rng.random())
        q = p * np.exp(side * amp * (2.0 * x - 1.0))
        q = np.maximum(q, 0.0)
        s_val = float(q.sum())
        if s_val > 0.0:
            q /= s_val
        else:
            q = p
        return q

    def make_bump(src, n, sd):
        p = _resample_mid(src, n)
        rng = np.random.default_rng(sd)
        x = (np.arange(n) + 0.5) / n
        center = 0.3 + 0.4 * float(rng.random())
        width = 0.1 + 0.3 * float(rng.random())
        q = p * (1.0 + 0.5 * np.exp(-0.5 * ((x - center) / width) ** 2))
        q = np.maximum(q, 0.0)
        s_val = float(q.sum())
        if s_val > 0.0:
            q /= s_val
        else:
            q = p
        return q

    def run_attempt(n, src_arr, end_time, sd, do_lp=True):
        nonlocal best_arr, best_score
        n = int(max(1, n))
        p = _resample_mid(src_arr, n)
        if p.size != n or not np.all(np.isfinite(p)) or float(p.sum()) <= 0.0:
            p = np.full(n, 1.0 / n)
        p = p / float(p.sum())
        consider(p)

        phase_len = max(0.0, float(end_time) - time.monotonic())
        if phase_len <= 15.0:
            return

        # L-BFGS phase
        lbfgs_stop = min(float(end_time) - 20.0, time.monotonic() + 0.45 * phase_len)
        if lbfgs_stop > time.monotonic() + 15.0:
            try:
                p2 = _lbfgs_optimize(p, n, lbfgs_stop, consider=consider, fine_only=False)
                if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0.0:
                    p = p2 / float(p2.sum())
                    consider(p)
            except Exception:
                pass

        # Coordinate descent phase
        rem = float(end_time) - time.monotonic()
        if rem > 20.0:
            coord_stop = min(float(end_time) - 5.0, time.monotonic() + max(20.0, 0.35 * phase_len))
            try:
                p3, _ = _coord_polish(p, coord_stop, max_sweeps=200000, seed=sd, consider=consider)
                if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and p3.sum() > 0.0:
                    p = p3 / float(p3.sum())
                    consider(p)
            except Exception:
                pass

        # LP refinement
        rem = float(end_time) - time.monotonic()
        if do_lp and rem > 15.0 and n <= 4096:
            lp_stop = min(float(end_time) - 5.0, time.monotonic() + max(10.0, 0.15 * phase_len))
            if lp_stop > time.monotonic() + 8.0:
                try:
                    p4 = _lp_refine(p, lp_stop, consider=consider, max_rounds=8)
                    if p4 is not None and p4.size == n and np.all(np.isfinite(p4)) and p4.sum() > 0.0:
                        p = p4 / float(p4.sum())
                        consider(p)
                except Exception:
                    pass

        # More coordinate descent after LP
        rem = float(end_time) - time.monotonic()
        if rem > 15.0:
            coord2_stop = min(float(end_time) - 3.0, time.monotonic() + max(10.0, 0.12 * phase_len))
            if coord2_stop > time.monotonic() + 8.0:
                try:
                    p5, _ = _coord_polish(p, coord2_stop, max_sweeps=100000, seed=sd + 7777, consider=consider)
                    if p5 is not None and p5.size == n and np.all(np.isfinite(p5)) and p5.sum() > 0.0:
                        p = p5 / float(p5.sum())
                        consider(p)
                except Exception:
                    pass

        if p is not None and p.size == n and np.all(np.isfinite(p)) and float(p.sum()) > 0.0:
            consider(p)

    # Determine lengths to try
    parent_len = int(best_arr.size) if best_arr.size > 0 else 3072
    lengths = [parent_len]
    if parent_len > 1024:
        lengths.append(parent_len * 2 // 3)
    if parent_len > 2048:
        lengths.append(parent_len // 2)
    lengths = sorted(set(lengths))

    # Weight allocation: main gets 70%, others split the rest
    n_lengths = len(lengths)
    if n_lengths == 1:
        weights = [1.0]
    else:
        weights = [0.70] + [0.30 / (n_lengths - 1)] * (n_lengths - 1)

    rng_top = np.random.default_rng(seed)
    phase_limit = min(hard - 15.0, t0 + budget - 30.0)

    for idx, L in enumerate(lengths):
        now = time.monotonic()
        if now >= phase_limit - 5.0:
            break
        rem_to_end = max(0.0, phase_limit - now)
        if rem_to_end <= 15.0:
            break
        w_rem = float(sum(weights[idx:]))
        if w_rem <= 0:
            w_rem = 1.0
        end_phase = min(phase_limit, now + (weights[idx] / w_rem) * rem_to_end)
        if end_phase <= now + 12.0:
            break

        src = best_arr.copy() if (best_arr.size == L or best_arr.size > 0) else parent.copy()

        if idx == 0 and (end_phase - now) > 150.0:
            # Main length: try two initializations
            v0 = src
            v1 = make_skew(src, L, int(seed + 777))
            # Also try a bump
            v2 = make_bump(src, L, int(seed + 888))

            # Split time: 40% v0, 30% v1, 30% v2 (if time allows)
            t1 = min(end_phase, now + 0.40 * (end_phase - now))
            run_attempt(L, v0, t1, int(seed + 100), do_lp=True)

            if time.monotonic() < end_phase - 15.0:
                t2 = min(end_phase, now + 0.70 * (end_phase - now))
                run_attempt(L, v1, t2, int(seed + 200), do_lp=True)

            if time.monotonic() < end_phase - 15.0:
                run_attempt(L, v2, end_phase, int(seed + 300), do_lp=True)
        else:
            src_resized = _resample_mid(src, L) if src.size != L else src
            run_attempt(L, src_resized, end_phase, int(seed + 500 + L), do_lp=True)

    # Final refinement on the best solution
    final_end = min(hard - 10.0, t0 + budget - 8.0)
    now = time.monotonic()
    if final_end > now + 20.0 and best_arr.size > 0:
        nbest = int(best_arr.size)
        rem = final_end - now

        # Fine L-BFGS
        if rem > 50.0 and nbest <= 5000:
            lstop = min(final_end - 30.0, now + 0.35 * rem)
            if lstop > now + 15.0:
                try:
                    p2 = _lbfgs_optimize(best_arr, nbest, lstop, consider=consider, fine_only=True)
                    if p2 is not None and p2.size == nbest and np.all(np.isfinite(p2)) and p2.sum() > 0.0:
                        p2 = p2 / float(p2.sum())
                        consider(p2)
                except Exception:
                    pass

        # Subgradient polish
        now = time.monotonic()
        sg_stop = min(final_end - 15.0, now + max(10.0, 0.20 * (final_end - now)))
        if sg_stop > now + 8.0:
            try:
                p_sg = _subgradient_polish(best_arr, sg_stop, consider=consider, max_iters=5000)
                if p_sg is not None and p_sg.size == nbest and np.all(np.isfinite(p_sg)) and p_sg.sum() > 0.0:
                    p_sg = p_sg / float(p_sg.sum())
                    consider(p_sg)
            except Exception:
                pass

        # LP refinement on best
        now = time.monotonic()
        lp_stop = min(final_end - 10.0, now + max(10.0, 0.15 * (final_end - now)))
        if lp_stop > now + 8.0 and nbest <= 4096:
            try:
                p_lp = _lp_refine(best_arr, lp_stop, consider=consider, max_rounds=10)
                if p_lp is not None and p_lp.size == nbest and np.all(np.isfinite(p_lp)) and p_lp.sum() > 0.0:
                    p_lp = p_lp / float(p_lp.sum())
                    consider(p_lp)
            except Exception:
                pass

        # Final coordinate polish
        now = time.monotonic()
        cstop = min(final_end - 3.0, now + max(10.0, 0.25 * (final_end - now)))
        if cstop > now + 8.0:
            try:
                p3, _ = _coord_polish(best_arr, cstop, max_sweeps=200000, seed=int(seed + 999999), consider=consider)
                if p3 is not None and p3.size == nbest and np.all(np.isfinite(p3)) and p3.sum() > 0.0:
                    p3 = p3 / float(p3.sum())
                    consider(p3)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
