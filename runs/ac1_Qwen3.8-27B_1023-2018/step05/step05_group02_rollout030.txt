import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    if c.size < L:
        c = np.pad(c, (0, L - c.size))
    np.maximum(c, 0.0, out=c)
    return c


def _score_from_conv(c, n):
    m = float(c.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * n * m)


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _lse_optimize(p, stop_time, consider, seed=42):
    """LSE relaxation + L-BFGS for coarse optimization."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        if c.size < conv_len:
            c = np.pad(c, (0, conv_len - c.size))
        np.maximum(c, 0.0, out=c)
        return c

    def cross_fft(pp, ww):
        cross_size = next_fast_len(3 * n - 2)
        cross_len = 3 * n - 2
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0.0:
            return 1e100, np.zeros_like(uu)
        if T <= 0.0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -400.0, 400.0)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = cross_fft(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    # Temperature schedule
    stages = (
        (0.10, 0.10),
        (0.03, 0.15),
        (0.01, 0.20),
        (0.003, 0.25),
        (0.001, 0.30),
        (0.0003, 0.30),
    )

    best_p = p.copy()
    best_sc = _score_from_conv(cfast(p), n)
    total_w = sum(wt for _, wt in stages)
    if total_w <= 0:
        total_w = 1.0
    avail = max(1.0, float(stop_time) - time.monotonic() - 3.0)

    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 2.0:
            break
        cum += wt
        end_stage = min(float(stop_time) - 1.0, time.monotonic() - avail + avail * (cum / total_w))
        if end_stage <= now + 1.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)

        # Run multiple L-BFGS chunks
        stale = 0
        max_chunks = 50
        for _ in range(max_chunks):
            if time.monotonic() > end_stage - 0.5:
                break
            maxit = 40 if n > 1500 else 60
            opts = {
                "maxiter": maxit,
                "gtol": 1e-14,
                "ftol": 1e-17,
                "maxls": 30,
                "maxcor": 40,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    fval = float(res.fun)
                    pp = _softmax(x)
                    sc = _score_from_conv(cfast(pp), n)
                    if math.isfinite(sc) and sc < best_sc - 1e-15:
                        best_sc = sc
                        best_p = pp.copy()
                        u = x.copy()
                        p = pp.copy()
                        stale = 0
                        if consider is not None:
                            try:
                                consider(pp)
                            except Exception:
                                pass
                    else:
                        stale += 1
                        if stale >= 2:
                            break
            if getattr(res, "status", 1) == 0:
                stale += 1
                if stale >= 2:
                    break

    return best_p if np.all(np.isfinite(best_p)) else p


def _coord_descent(p, stop_time, consider, seed=12345, G=None):
    """Fast coordinate descent with incremental convolution updates."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    if G is None:
        if n <= 500:
            G = 40
        elif n <= 1200:
            G = 32
        elif n <= 2500:
            G = 28
        else:
            G = 24
    G = int(max(16, G))

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    flat = 0
    sweep = 0
    max_flat = 5

    while time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for ii in range(n):
            if time.monotonic() > stop_time - 0.3:
                break
            i = int(order[ii])
            if S_total <= 1e-300:
                break
            # Periodic renormalization
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0

            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls].copy()

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(15.0 * v, 50.0 / n, 0.03, 1e-14)
            x_hi = min(x_hi, 5.0)

            # Coarse grid
            pts = np.linspace(0.0, x_hi, G + 1)

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[valid, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                if 0 < Ls:
                    dval = base[0] + w[valid] * infl[0] + w[valid] * w[valid]
                    maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Golden section refinement
            if G > 2:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)
                if hi - lo > 1e-16:
                    gr = (math.sqrt(5.0) - 1.0) / 2.0
                    a_gs = hi - gr * (hi - lo)
                    b_gs = lo + gr * (hi - lo)
                    fa = float(eval_xs(np.array([a_gs]))[0])
                    fb = float(eval_xs(np.array([b_gs]))[0])
                    for _ in range(16):
                        if not math.isfinite(fa):
                            fa = 1e100
                        if not math.isfinite(fb):
                            fb = 1e100
                        if fa < fb:
                            hi = b_gs
                            b_gs = a_gs
                            fb = fa
                            a_gs = hi - gr * (hi - lo)
                            fa = float(eval_xs(np.array([a_gs]))[0])
                        else:
                            lo = a_gs
                            a_gs = b_gs
                            fa = fb
                            b_gs = lo + gr * (hi - lo)
                            fb = float(eval_xs(np.array([b_gs]))[0])
                        if hi - lo < 1e-16:
                            break
                    gs_x = (lo + hi) / 2.0
                    gv = float(eval_xs(np.array([gs_x]))[0])
                    if math.isfinite(gv) and gv < best_val:
                        best_x = gs_x
                        best_val = gv

            if math.isfinite(best_val) and best_val < cur_score - 1e-14:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        # End of sweep: renormalize
        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        # Recompute convolution to avoid drift
        c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= max_flat:
                # Inject noise to escape local minimum
                if time.monotonic() < stop_time - 5.0:
                    noise_amt = 0.02 + 0.02 * math.exp(-0.05 * flat)
                    p = p * np.exp(rng.normal(0.0, noise_amt, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0.0:
                        p /= ss
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                    if math.isfinite(cur_score) and cur_score < best_score:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(p)
                            except Exception:
                                pass
                else:
                    break

        sweep += 1

    return best_p, best_score


def _subgradient_polish(p, stop_time, consider, max_iters=2000):
    """Subgradient descent on the active set."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    best_p = p.copy()
    best_score = float(2.0 * n * _conv_self(p).max())
    base_step = 0.003
    min_step = 1e-9

    for it in range(max_iters):
        if time.monotonic() > stop_time - 1.0:
            break

        c = _conv_self(p)
        cmax = float(c.max())
        threshold = cmax * 0.998
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        if len(active) > 200:
            # Keep only the top 200
            top_idx = np.argsort(c[active])[-200:]
            active = active[top_idx]

        # Subgradient: average of gradients of active constraints
        grad = np.zeros(n)
        for k in active:
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                grad[valid] += p[kj[valid]] + p[valid]
        grad /= len(active)

        # Project gradient onto sum=0 subspace
        grad -= float(np.mean(grad))

        step = base_step / (1.0 + 0.01 * it)
        if step < min_step:
            break

        accepted = False
        for scale in (1.0, 0.5, 0.25, 0.1, 0.05):
            p_cand = p - scale * step * grad
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = float(2.0 * n * _conv_self(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                accepted = True
                break
        if not accepted:
            base_step *= 0.95
            if base_step < 1e-12:
                break

        if it % 100 == 0 and consider is not None:
            try:
                consider(p)
            except Exception:
                pass

    if consider is not None:
        try:
            consider(p)
        except Exception:
            pass
    return best_p

    rem = hard - now
    if rem > 20.0:
        cd2_end = min(hard - 5.0, now + rem * 0.9)
        if cd2_end > now + 10.0:
            try:
                p_final, _ = _coord_descent(best_arr, cd2_end, consider, seed=int(seed + 99999), G=30)
                if p_final is not None and p_final.size == n and p_final.sum() > 0:
                    consider(p_final)
            except Exception:
                pass

    # === Final validation ===
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            # Fallback to parent
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        try:
            ls = [float(x) for x in parent]
            if evaluate_sequence(ls) < math.inf:
                return ls
        except Exception:
            pass
        return [1.0]
