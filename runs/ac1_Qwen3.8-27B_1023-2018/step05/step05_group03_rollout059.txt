    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / s


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:
        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if not math.isfinite(s) or s <= 1e-12:
        p = np.ones(3072, dtype=np.float64)
    return p


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _fft_cross(p, w):
    p = np.asarray(p, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 3 * n - 2
    size = next_fast_len(L)
    F = rfft(p[::-1], size)
    G = rfft(w, size)
    c = irfft(F * G, size)[:L]
    q = c[n - 1:2 * n - 1]
    q = np.asarray(q, dtype=np.float64)
    np.maximum(q, 0.0, out=q)
    return q


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if not math.isfinite(s) or s < 0.01:
        return math.inf
    a = a / s
    c = _fft_conv(a)
    if c.size == 0:
        return math.inf
    m = float(c.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * n * m)


def _score_exact_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if not math.isfinite(s) or s < 0.01:
        return math.inf
    a = a / s
    b = np.convolve(a, a)
    if b.size == 0:
        return math.inf
    m = float(b.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * n * m)


def _softmax(u):
    u = np.asarray(u, dtype=np.float64).ravel()
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        u = np.where(np.isfinite(u), u, 0.0)
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    return e / s


def _lbfgs_optimize(p, stop_time, consider=None, fine_only=False, max_frac=0.25):
    p = _normalize_nonneg(p)
    n = int(p.size)
    if n <= 1 or time.monotonic() > stop_time:
        return p

    conv_len = 2 * n - 1
    cross_len = 3 * n - 2
    conv_size = next_fast_len(conv_len)
    cross_size = next_fast_len(cross_len)

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def p_to_u(pp):
        pp = np.maximum(pp, 1e-15)
        uu = np.log(pp, dtype=np.float64)
        uu -= float(np.mean(uu))
        return uu

    def lse_func(uu, T):
        uu = np.asarray(uu, dtype=np.float64).ravel()
        if uu.size != n:
            return 1e100, np.zeros(n, dtype=np.float64)
        if not np.all(np.isfinite(uu)):
            uu = np.where(np.isfinite(uu), uu, 0.0)
        m = float(np.max(uu))
        e = np.exp(uu - m, dtype=np.float64)
        s = float(e.sum())
        if not math.isfinite(s) or s <= 0.0:
            pp = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            pp = e / s
        pp = np.where(np.isfinite(pp), pp, 1.0 / n)
        ss = float(pp.sum())
        if not math.isfinite(ss) or ss <= 0.0:
            pp = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            pp /= ss

        c = cfast(pp)
        if c.size != conv_len:
            return 1e100, np.zeros(n, dtype=np.float64)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0.0:
            return 1e100, np.zeros(n, dtype=np.float64)

        if not math.isfinite(T) or T <= 0.0:
            T = max(1e-300, 1e-5 * cmax)
        T = max(1e-16, T)

        scaled = (c - cmax) / T
        # Clip to avoid overflow
        scaled = np.clip(scaled, -500.0, 500.0)
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(conv_len, dtype=np.float64)
            im = int(np.argmax(c))
            w[im] = 1.0
            f = cmax
        else:
            w = exps / sum_exp
            ratio = sum_exp / float(c.size)
            if ratio <= 0.0:
                f = cmax
            else:
                f = cmax + T * float(math.log(ratio))
                if not math.isfinite(f):
                    f = cmax

        q = xfast(pp, w)
        gid = 2.0 * q
        gd = float(np.dot(pp, gid))
        grad = pp * (gid - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    def run_chunk(uu, T, end):
        try:
            f0, _ = lse_func(uu, T)
        except Exception:
            return uu
        if not math.isfinite(f0):
            return uu
        best_u = np.array(uu, dtype=np.float64, copy=True)
        best_f = float(f0)
        stale = 0

        while time.monotonic() < end - 2.0:
            maxit = 18 if n > 2000 else 25
            opts = {
                "maxiter": maxit,
                "gtol": 5e-10,
                "ftol": 1e-15,
                "maxls": 18,
                "maxcor": 18,
                "disp": False,
                "maxdisp": 0,
            }
            try:
                res = minimize(
                    lse_func,
                    best_u,
                    args=(T,),
                    method="L-BFGS-B",
                    jac=True,
                    options=opts,
                )
            except Exception:
                break

            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 5e-14:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, "status", 1) == 0:
                break
        return best_u

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score_fft(p)
    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass

    if fine_only:
        stages = ((0.0004, 0.25), (0.0001, 0.35), (0.00003, 0.40))
    else:
        stages = (
            (0.012, 0.12),
            (0.004, 0.16),
            (0.001, 0.20),
            (0.0003, 0.22),
            (0.00008, 0.30),
        )

    start = time.monotonic()
    avail = float(stop_time - start - 4.0)
    if avail <= 4.0:
        return best_p

    total_w = float(sum(w for _, w in stages))
    if total_w <= 0.0:
        total_w = 1.0
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 4.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue

        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-16, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_new = _softmax(u)
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if math.isfinite(ss) and ss > 1e-300:
            p_new /= ss
        else:
            p_new = best_p.copy()

        sc_new = _score_fft(p_new)
        if math.isfinite(sc_new) and sc_new < best_sc - 5e-14:
            best_sc = sc_new
            best_p = p_new.copy()
            p = p_new.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass
        else:
            p = best_p.copy()
        u = p_to_u(p)

    return best_p


def _coord_eval(xs, base, infl_vec, max_const, v, Sloc, n, di):
    xs = np.asarray(xs, dtype=np.float64).ravel()
    if xs.size == 0:
        return np.zeros(0, dtype=np.float64), xs
    w = xs - v
    den = Sloc + w
    valid = den > 1e-14
    aff = base[None, :] + w[:, None] * infl_vec[None, :]
    Ls = base.size
    if 0 <= di < Ls:
        aff[:, di] = base[di] + w * infl_vec[di] + w * w
    np.maximum(aff, max_const, out=aff)
    mv = np.max(aff, axis=1)
    vals = np.full(xs.size, math.inf, dtype=np.float64)
    if np.any(valid):
        vals[valid] = (2.0 * n * mv[valid]) / (den[valid] * den[valid])
    vals = np.where(np.isfinite(vals), vals, math.inf)
    return vals, xs


def _coord_polish(p, stop_time, G=96, seed=12345, consider=None, max_sweeps=500000, flat_limit=6):
    p = _normalize_nonneg(p)
    n = int(p.size)
    if n <= 1:
        return p, math.inf

    c = _fft_conv(p)
    c = np.maximum(c, 0.0)
    S = 1.0
    infl = 2.0 * p
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass

    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 2.0:
        improved = False
        improved_count = 0
        order = rng.permutation(n)

        for pos in range(n):
            if time.monotonic() > stop_time - 0.5:
                break
            if S <= 1e-300:
                break
            if S > 1.25 or S < 0.80:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                infl *= inv
                S = 1.0
                c = np.maximum(c, 0.0)
                cur_score = float(2.0 * n * c.max())

            i = int(order[pos])
            v = float(p[i])
            L = 2 * n - 1
            start = i
            end = min(i + n, L)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end].copy()
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < L:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc
            if not math.isfinite(max_const) or max_const < 0.0:
                max_const = 0.0

            Sloc = float(S)
            # Adaptive upper bound
            xhi = max(25.0 * v, 25.0 * Sloc / float(n), 0.006)
            xhi = min(xhi, max(0.1, 3.0 * v))
            if not math.isfinite(xhi) or xhi <= 0.0:
                xhi = max(1e-8, 10.0 * Sloc / float(n))

            pts = [0.0, v, xhi]

            m1 = max(16, int(G * 0.5))
            minx = max(1e-10, 5e-6 * v, Sloc / (5000.0 * float(n)))
            if minx < xhi:
                pts.extend(np.geomspace(minx, xhi, m1, dtype=np.float64).tolist())

            if v > 1e-12:
                m2 = max(12, int(G * 0.3))
                factors = np.geomspace(0.15, 6.0, m2, dtype=np.float64)
                pts.extend((v * factors).tolist())

            m3 = max(12, int(G * 0.2))
            linhi = min(xhi, max(10.0 * Sloc / float(n), 0.002))
            if linhi > 0.0:
                pts.extend(np.linspace(0.0, linhi, m3, dtype=np.float64).tolist())

            pts = np.unique(np.clip(np.asarray(pts, dtype=np.float64), 0.0, xhi))
            if pts.size == 0:
                pts = np.array([v], dtype=np.float64)

            di = 2 * i - start
            vals, pts = _coord_eval(pts, base, infl, max_const, v, Sloc, n, di)
            if not np.any(np.isfinite(vals)):
                continue
            im = int(np.argmin(vals))
            best_x = float(pts[im])
            best_val = float(vals[im])

            # Fine local search around best
            if best_val < cur_score + max(1e-10, 1e-5 * cur_score):
                if best_x <= 1e-14:
                    lo = 0.0
                    hi = max(5.0 * Sloc / float(n), 5.0 * v, 1e-7)
                else:
                    step = max(v * 0.1, Sloc / float(n) * 0.05, 1e-10)
                    lo = max(0.0, best_x - 3.0 * step, best_x * 0.3)
                    hi = min(xhi, best_x + 3.0 * step, best_x * 3.0)
                if hi <= lo:
                    hi = lo + 1e-9
                fine = [lo, hi, best_x]
                fine.extend(np.linspace(lo, hi, 48, dtype=np.float64).tolist())
                if lo > 0.0:
                    fine.extend(np.geomspace(lo, hi, 24, dtype=np.float64).tolist())
                else:
                    fine.extend(np.linspace(0.0, hi, 24, dtype=np.float64).tolist())
                fine = np.unique(np.clip(np.asarray(fine, dtype=np.float64), 0.0, xhi))
                vals2, fine = _coord_eval(fine, base, infl, max_const, v, Sloc, n, di)
                if np.any(np.isfinite(vals2)):
                    im2 = int(np.argmin(vals2))
                    vx2 = float(fine[im2])
                    vv2 = float(vals2[im2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2

            if (
                math.isfinite(best_val)
                and best_val < cur_score - 5e-12
                and 0.0 <= best_x <= xhi
            ):
                d = best_x - v
                if abs(d) > 1e-17:
                    c[start:end] += d * infl
                    diag = 2 * i
                    if 0 <= diag < L:
                        c[diag] += d * d
                    p[i] = best_x
                    infl[i] = 2.0 * best_x
                    S += d
                    cur_score = best_val
                    improved = True
                    improved_count += 1

        # Renormalize after sweep
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            infl *= inv
            S = 1.0
        else:
            break
        # Recompute convolution to prevent drift
        c = _fft_conv(p)
        c = np.maximum(c, 0.0)
        desc = float(2.0 * n * c.max())
        if math.isfinite(desc) and desc < best_score - 5e-14:
            best_score = desc
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= flat_limit:
                # Perturbation to escape
                if time.monotonic() < stop_time - 30.0:
                    sigmas = (0.025, 0.05, 0.08, 0.12, 0.18)
                    sigma = sigmas[min(flat - flat_limit, len(sigmas) - 1)]
                    noise = rng.normal(0.0, sigma, n, dtype=np.float64)
                    q = p * np.exp(noise)
                    if sigma >= 0.08:
                        x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
                        amp = 0.15 + 0.35 * float(rng.random())
                        f1 = 1 + int(rng.integers(0, 5))
                        ph1 = 2.0 * math.pi * float(rng.random())
                        q *= np.exp(amp * math.sin(2.0 * math.pi * f1 * x + ph1))
                    q = np.maximum(q, 0.0)
                    qs = float(q.sum())
                    if math.isfinite(qs) and qs > 0.0:
                        p = q / qs
                        c = _fft_conv(p)
                        c = np.maximum(c, 0.0)
                        S = 1.0
                        infl = 2.0 * p
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 5e-14:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider is not None:
                                try:
                                    consider(best_p)
                                except Exception:
                                    pass
                    flat = 0
                else:
                    break
        sweep += 1

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p, best_score


def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == 1:
        return np.full(n, float(src[0]) / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / float(n), xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _lp_improve_small(src, n_small, stop_time, max_iter=4):
    if linprog is None or sp is None:
        return None
    n_small = int(n_small)
    if n_small <= 16 or n_small > 576:
        return None
    if time.monotonic() > stop_time:
        return None

    a = _resample_mid(src, n_small)
    a = _normalize_nonneg(a)
    best_a = a.copy()
    best_sc = _score_fft(a)
    if not math.isfinite(best_sc):
        return None

    for _ in range(max_iter):
        if time.monotonic() > stop_time - 6.0:
            break
        c = _fft_conv(a)
        M = float(c.max()) if c.size else 0.0
        if not math.isfinite(M) or M <= 0.0:
            break
        try:
            idx = np.arange(n_small, dtype=np.int32)
            cols = np.repeat(idx, n_small)
            off = np.tile(idx, n_small)
            rows = cols + off
            data = a[off].astype(np.float64)
            A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n_small - 1, n_small)).tocsr()
            cobj = -np.ones(n_small, dtype=np.float64)
            b_ub = np.full(2 * n_small - 1, M, dtype=np.float64)
            try:
                res = linprog(cobj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method="highs")
            except Exception:
                res = linprog(cobj, A_ub=A, b_ub=b_ub, bounds=(0.0, None))
            if res is not None and getattr(res, "success", False):
                b = np.asarray(res.x, dtype=np.float64).ravel()
                b = np.where(np.isfinite(b), b, 0.0)
                b = np.maximum(b, 0.0)
                if float(b.sum()) > 0.0:
                    g = b / b.sum()
                    improved = False
                    for t in (0.5, 0.35, 0.20, 0.10):
                        cand = (1.0 - t) * a + t * g
                        cand = np.maximum(cand, 0.0)
                        ss = float(cand.sum())
                        if not math.isfinite(ss) or ss <= 0.0:
                            continue
                        cand /= ss
                        sc = _score_fft(cand)
                        if math.isfinite(sc) and sc < best_sc - 1e-10:
                            a = cand
                            best_sc = sc
                            best_a = cand.copy()
                            improved = True
                            break
                    if not improved:
                        break
        except Exception:
            break

    if math.isfinite(_score_fft(best_a)):
        return best_a
    return None


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(10.0, budget)
    if budget > 300.0:
        reserve = 20.0
    elif budget > 60.0:
        reserve = 10.0
    else:
        reserve = 2.5
    hard = t0 + budget - reserve
    if hard <= t0 + 2.0:
        hard = t0 + max(1.5, budget - 0.5)

    parent = _get_parent()
    n_target = int(parent.size) if parent.size > 0 else 3072

    best_arr = _normalize_nonneg(parent.copy())
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.full(64, 1.0 / 64.0, dtype=np.float64)
        try:
            best_score = float(_score_exact_arr(best_arr))
        except Exception:
            best_score = math.inf

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return
        if arr.size == 0:
            return
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if not math.isfinite(s) or s < 0.01:
            return
        arr = arr / s
        try:
            sc = float(_score_fft(arr))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 5e-14:
            # Double check with exact
            try:
                sc_exact = float(_score_exact_arr(arr))
                if math.isfinite(sc_exact) and sc_exact <= sc + 1e-10:
                    best_arr = arr.copy()
                    best_score = sc_exact
            except Exception:
                best_arr = arr.copy()
                best_score = sc

    rng_top = np.random.default_rng(seed)

    # Quick initial exploration: shaped perturbations
    base = best_arr.copy()
    power_ts = (0.50, 0.75, 1.30, 1.80)
    for q in power_ts:
        if time.monotonic() > hard - 300.0:
            break
        cand = np.power(np.maximum(base, 1e-30), float(q))
        cand = _normalize_nonneg(cand)
        consider(cand)

    for _ in range(4):
        if time.monotonic() > hard - 280.0:
            break
        side = 1.0 if rng_top.random() < 0.5 else -1.0
        amp = 0.25 + 0.85 * float(rng_top.random())
        x = (np.arange(n_target, dtype=np.float64) + 0.5) / float(n_target)
        cand = base * np.exp(side * amp * (2.0 * x - 1.0))
        cand = np.maximum(cand, 0.0)
        cand = _normalize_nonneg(cand)
        consider(cand)

    # Small LP subproblem for shape guidance
    if budget > 360.0 and time.monotonic() < hard - 200.0:
        for ns in [256, 384]:
            if time.monotonic() > hard - 180.0:
                break
            stop = min(hard - 160.0, time.monotonic() + (25.0 if ns <= 256 else 45.0))
            try:
                small = _lp_improve_small(best_arr, ns, stop, max_iter=3)
            except Exception:
                small = None
            if small is not None:
                cand = _resample_mid(small, n_target)
                cand = _normalize_nonneg(cand)
                consider(cand)

    # ===== Phase 1: Medium-temperature LSE =====
    now = time.monotonic()
    phase1_stop = min(hard - 120.0, now + max(30.0, 0.18 * (hard - now)))
    if phase1_stop > now + 20.0:
        try:
            p1 = best_arr.copy()
            p1 = _lbfgs_optimize(p1, phase1_stop, consider=consider, fine_only=False)
            consider(p1)
        except Exception:
            pass

    # ===== Phase 2: Coordinate descent (medium) =====
    now = time.monotonic()
    phase2_stop = min(hard - 80.0, now + max(30.0, 0.28 * (hard - now)))
    if phase2_stop > now + 20.0:
        try:
            p2, _ = _coord_polish(best_arr, phase2_stop, G=96, seed=seed + 1000, consider=consider)
            consider(p2)
        except Exception:
            pass

    # ===== Phase 3: Fine LSE =====
    now = time.monotonic()
    phase3_stop = min(hard - 60.0, now + max(20.0, 0.15 * (hard - now)))
    if phase3_stop > now + 15.0:
        try:
            p3 = best_arr.copy()
            p3 = _lbfgs_optimize(p3, phase3_stop, consider=consider, fine_only=True)
            consider(p3)
        except Exception:
            pass

    # ===== Phase 4: Fine coordinate descent (main squeeze) =====
    now = time.monotonic()
    phase4_stop = min(hard - 30.0, now + max(20.0, 0.35 * (hard - now)))
    if phase4_stop > now + 15.0:
        try:
            p4, _ = _coord_polish(best_arr, phase4_stop, G=120, seed=seed + 5000, consider=consider,
                                   max_sweeps=800000, flat_limit=7)
            consider(p4)
        except Exception:
            pass

    # ===== Phase 5: Final ultra-fine coordinate descent =====
    now = time.monotonic()
    phase5_stop = min(hard - 5.0, now + max(10.0, 0.12 * (hard - now)))
    if phase5_stop > now + 10.0:
        try:
            p5, _ = _coord_polish(best_arr, phase5_stop, G=160, seed=seed + 99999,
                                   consider=consider, max_sweeps=1000000, flat_limit=8)
            consider(p5)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = _normalize_nonneg(parent)
            s = float(out.sum())
        if out.size == 0 or not math.isfinite(s) or s < 0.01:
            out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        out = out / float(out.sum())
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            try:
                fallback = _normalize_nonneg(parent)
                ls2 = [float(x) for x in fallback]
                sc2 = float(evaluate_sequence(ls2))
                if math.isfinite(sc2):
                    return ls2
            except Exception:
                pass
            return [1.0]
        return ls
    except Exception:
        return [1.0]
