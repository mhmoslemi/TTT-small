import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize
from scipy.linalg import hankel


def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * a.size * m / (s * s))


def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    return e / s


def _coord_descent(p, stop_time, consider=None, seed=42, max_sweeps=200000):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p, math.inf
    p = p / s

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S = 1.0
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    G = 28
    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.5:
        # Build prefix and suffix max arrays
        prefix_max = np.zeros(Ltot + 1, dtype=np.float64)
        prefix_max[0] = 0.0
        for k in range(1, Ltot + 1):
            prefix_max[k] = max(prefix_max[k - 1], float(c[k - 1]))
        suffix_max = np.zeros(Ltot + 1, dtype=np.float64)
        suffix_max[Ltot] = 0.0
        for k in range(Ltot - 1, -1, -1):
            suffix_max[k] = max(suffix_max[k + 1], float(c[k]))

        # Peak-focused ordering
        k_peak = int(np.argmax(c))
        i_lo = max(0, k_peak - n + 1)
        i_hi = min(k_peak, n - 1)
        peak_coords = list(range(i_lo, i_hi + 1))
        other_coords = [i for i in range(n) if i < i_lo or i > i_hi]
        rng.shuffle(peak_coords)
        rng.shuffle(other_coords)
        order = peak_coords + other_coords

        improved = False

        for i in order:
            if time.monotonic() > stop_time - 0.5:
                break
            if S <= 1e-300:
                break

            if S > 1.08 or S < 0.92:
                p /= S
                c /= (S * S)
                S = 1.0

            v = float(p[i])
            start_k = i
            end_k = i + n
            if end_k > Ltot:
                end_k = Ltot
            Ls = end_k - start_k
            if Ls <= 0:
                continue

            max_const = max(prefix_max[start_k], suffix_max[end_k])

            avg = S / float(n)
            x_hi = max(25.0 * v, 50.0 * avg, 0.015 * S, 1e-10)
            x_hi = min(x_hi, 8.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, avg)

            pts = np.linspace(0.0, x_hi, G, dtype=np.float64)

            w = pts - v
            denom = S + w
            valid = denom > 1e-15
            if not np.any(valid):
                continue

            base = c[start_k:end_k].copy()
            infl = 2.0 * p[:Ls]

            aff = base[None, :] + w[:, None] * infl[None, :]
            diag_pos = (2 * i) - start_k
            if 0 <= diag_pos < Ls:
                aff[:, diag_pos] += w * w

            maff = np.maximum(aff, max_const)
            maff = np.max(maff, axis=1)
            vals = np.full(pts.shape, math.inf, dtype=np.float64)
            vals[valid] = (2.0 * n * maff[valid]) / (denom[valid] * denom[valid])

            if not np.any(np.isfinite(vals)):
                continue

            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Refine
            dw = x_hi / float(G)
            lo = max(0.0, best_x - 1.5 * dw)
            hi = min(x_hi, best_x + 1.5 * dw)
            if hi > lo:
                fine = np.linspace(lo, hi, 20, dtype=np.float64)
                w2 = fine - v
                denom2 = S + w2
                valid2 = denom2 > 1e-15
                if np.any(valid2):
                    aff2 = base[None, :] + w2[:, None] * infl[None, :]
                    if 0 <= diag_pos < Ls:
                        aff2[:, diag_pos] += w2 * w2
                    maff2 = np.maximum(aff2, max_const)
                    maff2 = np.max(maff2, axis=1)
                    vals2 = np.full(fine.shape, math.inf, dtype=np.float64)
                    vals2[valid2] = (2.0 * n * maff2[valid2]) / (denom2[valid2] * denom2[valid2])
                    if np.any(np.isfinite(vals2)):
                        imin2 = int(np.argmin(vals2))
                        if vals2[imin2] < best_val:
                            best_x = float(fine[imin2])
                            best_val = float(vals2[imin2])

            if math.isfinite(best_val) and best_val < cur_score * (1.0 - 1e-12) and best_x >= 0.0:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S += w
                    c[start_k:end_k] += w * infl
                    if 0 <= diag_pos < Ls:
                        c[2 * i] += w * w
                    cur_score = best_val
                    improved = True

        # Renormalize
        if S > 1e-300 and math.isfinite(S):
            p /= S
            c /= (S * S)
            S = 1.0

        # Periodic FFT recompute
        if sweep % 40 == 0:
            c = _conv_self(p)

        new_score = 2.0 * n * float(c.max())
        if math.isfinite(new_score) and new_score < best_score * (1.0 - 1e-13):
            best_score = new_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 5.0:
                    sigma = 0.015 + 0.01 * min(flat // 4, 8)
                    p *= np.exp(rng.normal(0.0, sigma, n, dtype=np.float64))
                    p = np.maximum(p, 0.0)
                    ps = float(p.sum())
                    if ps > 0.0:
                        p /= ps
                        S = 1.0
                    c = _conv_self(p)
                    cur_score = 2.0 * n * float(c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    return best_p, best_score


def _lse_optimize(p, stop_time, consider=None):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        pp = np.maximum(pp, 1e-300)
        uu = np.log(pp, dtype=np.float64)
        return uu - float(np.mean(uu))

    u = p_to_u(p)

    fft_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1

    def cfast(aa):
        F = rfft(aa, fft_size)
        cc = irfft(F * F, fft_size)[:conv_len]
        np.maximum(cc, 0.0, out=cc)
        return cc

    def lse_obj(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        scaled = (c - cmax) / T
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            obj = cmax
        else:
            w = exps / sum_exp
            obj = cmax + T * math.log(sum_exp / float(c.size))
        if not math.isfinite(obj):
            obj = cmax
        # Gradient: g_i = 2 * sum_{j=0}^{n-1} w[i+j] * pp[j]
        # Using hankel matrix
        W = hankel(w[:n], w[n - 1:2 * n - 1])
        grad_raw = W @ pp
        grad = 2.0 * grad_raw
        # Project gradient (for softmax param, we subtract mean)
        grad = grad - float(np.mean(grad))
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return obj, grad

    best_u = u.copy()
    best_obj = lse_obj(best_u, 0.1 * float(cfast(p).max()))[0]

    stages = [
        (0.08, 0.15),
        (0.03, 0.18),
        (0.01, 0.20),
        (0.004, 0.20),
        (0.0015, 0.18),
        (0.0005, 0.09),
    ]

    start = time.monotonic()
    avail = max(0.0, float(stop_time) - start - 5.0)
    if avail < 10.0:
        return p
    total_w = sum(w for _, w in stages)

    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += wt
        end_stage = min(stop_time - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:
            continue

        T = frac * float(cfast(_softmax(best_u)).max())
        T = max(T, 1e-6)

        max_chunks = 8
        for chunk in range(max_chunks):
            if time.monotonic() > end_stage - 2.0:
                break
            opts = {
                "maxiter": 35,
                "gtol": 1e-10,
                "ftol": 1e-14,
                "maxls": 25,
                "maxcor": 25,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(lse_obj, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_obj:
                    best_obj = fval
                    best_u = res.x.copy()
            if getattr(res, "status", 1) == 0:
                break

        pp = _softmax(best_u)
        pp = np.maximum(pp, 0.0)
        ss = float(pp.sum())
        if ss > 1e-300:
            pp /= ss
        sc = _score_fft(pp)
        if math.isfinite(sc) and consider is not None:
            try:
                consider(pp)
            except Exception:
                pass
        if not math.isfinite(sc):
            break

    return _softmax(best_u)

                work, _ = _coord_descent(work, coordin_end, consider=consider, seed=int(seed + 77777))

        out = np.clip(out, 0.0, 1000.0)
