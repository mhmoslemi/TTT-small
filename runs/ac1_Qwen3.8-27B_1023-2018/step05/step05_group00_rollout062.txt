        return np.zeros(0)
    L = next_fast_len(2 * n - 1)
    F = rfft(a, L)
    c = irfft(F * F, L)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_norm(a):
    """Score assuming sum(a)=1. Returns 2n*max(c)."""
    c = _conv_full(a)
    return float(2.0 * a.size * c.max())


def _softmax_safe(u):
    u = u - u.max()
    e = np.exp(u, dtype=np.float64)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    return e / s


def _lse_logsoftmax(obj, T, L, clen, n):
    """LSE objective and gradient for log-softmax parameterization."""
    def func(uu):
        a = _softmax_safe(uu)
        F = rfft(a, L)
        c = irfft(F * F, L)[:clen]
        np.maximum(c, 0.0, out=c)
        cmax = c.max()
        if not np.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        T_eff = max(T, cmax * 1e-12)
        scaled = (c - cmax) / T_eff
        exps = np.exp(scaled)
        s_exp = exps.sum()
        if s_exp <= 0 or not np.isfinite(s_exp):
            w = np.zeros(clen)
            w[int(np.argmax(c))] = 1.0
            val = cmax
        else:
            w = exps / s_exp
            val = cmax + T_eff * math.log(s_exp / clen)
        # Gradient: g[i] = 2 * sum_k w[k]*a[k-i]
        corr = 2.0 * np.correlate(w, a, mode='valid')
        # Ensure length n
        if corr.size != n:
            corr = np.resize(corr, n)
        tdp = float(a @ corr)
        grad = a * (corr - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(val), grad
    return func


def _lbfgs_optimize(p0, end_time, consider=None):
    """Multi-stage LSE optimization via L-BFGS-B."""
    p = np.maximum(np.asarray(p0, dtype=np.float64).ravel(), 0.0).copy()
    n = p.size
    s = float(p.sum())
    p = p / s
    
    u = np.log(np.maximum(p, 1e-300), dtype=np.float64)
    u -= u.mean()
    
    L = next_fast_len(2 * n - 1)
    clen = 2 * n - 1
    
    def obj(uu, T):
        a = _softmax_safe(uu)
        F = rfft(a, L)
        c = irfft(F * F, L)[:clen]
        np.maximum(c, 0.0, out=c)
        cmax = float(c.max())
        if not np.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        Te = max(T, cmax * 1e-12)
        scaled = (c - cmax) / Te
        exps = np.exp(scaled, dtype=np.float64)
        s_exp = float(exps.sum())
        if s_exp <= 0 or not np.isfinite(s_exp):
            w = np.zeros(clen, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            val = cmax
        else:
            w = exps / s_exp
            val = cmax + Te * math.log(s_exp / clen)
        corr = 2.0 * np.correlate(w, a, mode='valid').astype(np.float64)
        if corr.size < n:
            corr = np.concatenate([corr, np.zeros(n - corr.size)])
        elif corr.size > n:
            corr = corr[:n]
        tdp = float(np.dot(a, corr))
        grad = a * (corr - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(val), grad
    
    avail = end_time - time.monotonic() - 5.0
    if avail < 10.0:
        p_out = _softmax_safe(u)
        p_out = np.maximum(p_out, 0.0)
        p_out /= max(p_out.sum(), 1e-300)
        return p_out
    
    c0 = _conv_full(p)
    c0max = float(c0.max()) if c0.size else 1.0
    if c0max <= 0 or not np.isfinite(c0max):
        c0max = 1.0
    
    stages = [
        (0.10, 0.10), (0.04, 0.12), (0.01, 0.14), (0.003, 0.16),
        (0.001, 0.18), (0.0003, 0.16), (0.0001, 0.14), (0.00003, 0.14),
        (0.00001, 0.16),
    ]
    total_w = sum(w for _, w in stages)
    
    best_u = u.copy()
    best_f = float(obj(u, stages[0][0] * c0max)[0])
    
    cum = 0.0
    for frac, wt in stages:
        if time.monotonic() >= end_time - 3.0:
            break
        Te = max(frac * c0max, c0max * 1e-14)
        stage_budget = avail * (wt / total_w)
        stage_end = time.monotonic() + stage_budget
        if stage_end < time.monotonic() + 2.0:
            continue
        
        f_cur, _ = obj(best_u, Te)
        if not np.isfinite(f_cur):
            continue
        local_best_f = f_cur
        stale = 0
        
        while time.monotonic() < stage_end - 1.0:
            maxit = min(60, max(20, int(10000 / max(n, 1))))
            try:
                res = minimize(
                    obj, best_u, args=(Te,),
                    method='L-BFGS-B', jac=True,
                    options={'maxiter': maxit, 'gtol': 1e-14,
                             'ftol': 1e-16, 'maxls': 30,
                             'maxcor': 50, 'disp': False}
                )
            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                try:
                    fv = float(res.fun)
                except Exception:
                    fv = math.inf
                if np.isfinite(fv) and fv < local_best_f - 1e-16:
                    local_best_f = fv
                    best_u = np.array(res.x, dtype=np.float64)
                    stale = 0
                else:
                    stale += 1
                    if stale >= 5:
                        break
            if getattr(res, 'status', 1) == 0:
                break
        
        p_cur = _softmax_safe(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ps = float(p_cur.sum())
        if ps > 1e-300:
            p_cur /= ps
        if consider:
            try:
                consider(p_cur)
            except Exception:
                pass
        c0max = max(_conv_full(p_cur).max(), c0max * 0.5) if p_cur.size else c0max
    
    p_out = _softmax_safe(best_u)
    p_out = np.maximum(p_out, 0.0)
    p_out /= max(p_out.sum(), 1e-300)
    return p_out
