def _get_parent():

def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(np.asarray(c, dtype=np.float64), 0.0)


def _cross_conv(a, b):
    na, nb = a.size, b.size
    L = na + nb - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    G = rfft(b, size)
    c = irfft(F * G, size)[:L]
    return np.maximum(np.asarray(c, dtype=np.float64), 0.0)


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())

    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0 / u.size)
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0 / u.size)
    return e / s

        cmax = float(cfast(p).max())
        T = max(1e-15, frac * cmax)
        end_stage = min(stop_time - 5.0, time.monotonic() + max(5.0, wt * 30.0))
        stale = 0
        while time.monotonic() < end_stage - 2.0:
            opts = {"maxiter": 35, "gtol": 1e-12, "ftol": 1e-16,
                    "maxls": 30, "maxcor": 25, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval):
                    p_cur = _softmax(res.x)
                    sc_cur = _score(p_cur)
                    if math.isfinite(sc_cur) and sc_cur < best_sc:
                        best_sc = sc_cur
                        best_p = p_cur.copy()
                        p = p_cur.copy()
                        u = p_to_u(p)
                        stale = 0
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, "status", 1) == 0:
                break

    p_out = best_p.copy()
    ss = float(p_out.sum())
    if ss > 1e-300:
        p_out = p_out / ss
    return p_out


def _coord_descent(p, stop_time, consider=None, seed=42, max_sweeps=200000):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p, math.inf
    p = p / s

    G = max(16, min(50, int(150000 / max(n, 1)) + 16))
    Ltot = 2 * n - 1
    c = _conv_self(p)
    S = 1.0
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    flat = 0
    sweep = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S <= 1e-300:
                break
            if S > 1.3 or S < 0.7:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0

            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]
            diag_pos = i  # offset in base where k=2i

            max_const = 0.0
            if start > 0:
                max_const = max(max_const, float(np.max(c[:start])))
            if end < Ltot:
                max_const = max(max_const, float(np.max(c[end:])))

            x_hi = max(30.0 * v, 80.0 / n, 0.02)
            x_hi = min(x_hi, 5.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = 1.0 / n

            n_coarse = G
            pts = np.linspace(0.0, x_hi, n_coarse + 1)

            def eval_cand(xs):
                w = xs - v
                denom = S + w
                valid = denom > 1e-14
                aff = base[None, :] + w[:, None] * infl[None, :]
                if diag_pos < Ls:
                    aff[:, diag_pos] += w * w
                aff = np.maximum(aff, max_const)
                mv = np.max(aff, axis=1)
                out = np.full(xs.shape, math.inf)
                dv = denom[valid]
                out[valid] = (2.0 * n * mv[valid]) / (dv * dv)
                return out

            vals = eval_cand(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            bx = float(pts[imin])
            bv = float(vals[imin])

            dw = x_hi / n_coarse
            lo = max(0.0, bx - 3.0 * dw)
            hi = min(x_hi, bx + 3.0 * dw)
            if hi > lo:
                fine = np.linspace(lo, hi, 40)
                vals2 = eval_cand(fine)
                if np.any(np.isfinite(vals2)):
                    im2 = int(np.argmin(vals2))
                    if float(vals2[im2]) < bv:
                        bv = float(vals2[im2])
                        bx = float(fine[im2])

            if math.isfinite(bv) and bv < cur_score - 1e-13 and 0.0 <= bx <= 10.0 * S:
                w = bx - v
                if abs(w) > 1e-15:
                    p[i] = bx
                    S += w
                    c[start:end] += w * infl
                    if diag_pos < Ls:
                        c[start + diag_pos] += w * w
                    cur_score = bv
                    improved = True

        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        if sweep % 10 == 0:
            c = _conv_self(p)
        new_sc = 2.0 * n * float(c.max())
        if math.isfinite(new_sc) and new_sc < best_score - 1e-14:
            best_score = new_sc
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 15.0:
                    p = p * np.exp(rng.normal(0, 0.02, n))
                    p = np.maximum(p, 0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    S = 1.0
                    cur_score = 2.0 * n * float(c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_self(p)
    sc = 2.0 * n * float(c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score
