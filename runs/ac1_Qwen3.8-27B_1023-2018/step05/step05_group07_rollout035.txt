```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.signal import fftconvolve

def _score_normalized(a):
    """Assume a is already clipped to [0, 1000]. Compute score."""
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    return 2.0 * len(a) * b.max() / (S * S)

def _score_unit(a):
    """Assume sum(a) = 1. Return 2n * max(b)."""
    b = np.convolve(a, a)
    return 2.0 * len(a) * float(b.max())

def _build_conv_mat(n, x):
    """Build (2n-1) x n matrix A where A[k,j] = x[k-j] if 0<=k-j<n."""
    # Use the fact that it's a banded matrix
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        for j in range(jlo, jhi):
            A[k, j] = x[k - j]
    return A

def _build_conv_mat_fast(n, x):
    """Faster using broadcasting."""
    k = np.arange(2*n-1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    A[mask] = x[d[mask]]
    return A
```

```python
                      options={'time_limit': tl,
                               'primal_feasibility_tolerance': 1e-12,
                               'dual_feasibility_tolerance': 1e-12})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def _line_search(x, y):
    """Find best t minimizing max-conv of (1-t)*x + t*y (normalized to sum 1)."""
    n = x.size
    best_t, best_M = 0.0, float(np.convolve(x, x).max())
    for t in [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]:
        cand = (1 - t) * x + t * y
        cand = np.maximum(cand, 0)
        cs = cand.sum()
        if cs < 1e-15:
            continue
        cand = cand / cs
        M = float(np.convolve(cand, cand).max())
        if M < best_M:
            best_M, best_t = M, t
    if best_t == 0.0:
        return None, best_M
    lo = max(0.0, best_t - 0.06)
    hi = min(1.0, best_t + 0.06)
    for _ in range(60):
        if hi - lo < 1e-10:
            break
        mid = 0.5 * (lo + hi)
        cand = (1 - mid) * x + mid * y
        cand = np.maximum(cand, 0)
        cs = cand.sum()
        if cs < 1e-15:
            lo = mid
            continue
        cand = cand / cs
        M = float(np.convolve(cand, cand).max())
        if M < best_M:
            best_M, best_t = M, mid
            hi = mid
        else:
            lo = mid
    cand = (1 - best_t) * x + best_t * y
    cand = np.maximum(cand, 0)
    cs = cand.sum()
    if cs < 1e-15:
        return None, best_M
    return cand / cs, best_M


def _subgrad_step(x, b, M):
    """Subgradient descent step on max autocorrelation over simplex."""
    n = x.size
    active = np.where(b >= M * 0.9999)[0]
    if len(active) == 0:
        return None
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        for j in range(jlo, jhi):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += 2.0 * x[idx]
    grad = grad - grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None
    for alpha in [0.04, 0.02, 0.01, 0.005, 0.002]:
        cand = x - alpha * grad / gn
        cand = np.maximum(cand, 0)
        cs = cand.sum()
        if cs < 1e-15:
            continue
        cand = cand / cs
        v = float(np.convolve(cand, cand).max())
        if v < M * (1 - 1e-9):
            return cand
    return None


def _optimize(x0, deadline, seed=42, max_stuck=80):
    """LP best-response + subgradient optimization loop."""
    x = np.asarray(x0, dtype=np.float64).copy()
    x = np.maximum(x, 0)
    S = x.sum()
    if S < 1e-15:
        x = np.ones(x.size) / x.size
    x = x / S
    n = x.size
    best_x = x.copy()
    best_M = float(np.convolve(x, x).max())
    rng = np.random.default_rng(seed)
    stuck = 0
    perturb_count = 0

    while time.monotonic() < deadline:
        b = np.convolve(x, x)
        M = float(b.max())

        if M < best_M * (1 - 1e-12):
            best_M = M
            best_x = x.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > max_stuck:
            sigma = min(0.015 + 0.004 * (stuck - max_stuck), 0.35)
            mode = perturb_count % 5
            xx = np.arange(n, dtype=np.float64)
            if mode == 0:
                x *= np.exp(rng.normal(0, sigma, n))
            elif mode == 1:
                f1 = rng.uniform(0.5, 4.0)
                f2 = rng.uniform(1.0, 6.0)
                x *= (1.0 + sigma * (0.6 * np.sin(2 * np.pi * f1 * xx / n + rng.uniform(0, 6.28))
                                      + 0.4 * np.cos(2 * np.pi * f2 * xx / n + rng.uniform(0, 6.28))))
            elif mode == 2:
                blk = max(n // 6, 2)
                start = int(rng.integers(0, n - blk))
                x[start:start + blk] *= np.exp(rng.normal(0, sigma * 1.5, blk))
            elif mode == 3:
                x *= np.exp(rng.normal(0, sigma, n))
                x = (x + x[::-1]) / 2  # symmetrize
            else:
                half = n // 2
                x[:half] = x[half:half + half] if 2 * half <= n else x[:half]
                x = np.maximum(x, 1e-15)
            x = np.maximum(x, 1e-15)
            x = x / x.sum()
            stuck = 0
            perturb_count += 1
            if sigma > 0.3:
                break
            continue

        moved = False

        # Primary LP
        y = _solve_lp(x, M * (1 + 1e-10), tl=2.5)
        if y is not None and y.sum() > 1e-15:
            y = y / y.sum()
            if np.max(np.abs(y - x)) > 1e-10:
                cand, val = _line_search(x, y)
                if cand is not None and val < M * (1 - 1e-11):
                    x = cand
                    moved = True

        # Tighter LP
        if not moved:
            y2 = _solve_lp(x, M * 0.9995, tl=2.0)
            if y2 is not None and y2.sum() > 1e-15:
                y2 = y2 / y2.sum()
                cand, val = _line_search(x, y2)
                if cand is not None and val < M * (1 - 1e-11):
                    x = cand
                    moved = True

        # Averaged LP
        if not moved and y is not None and y.sum() > 1e-15:
            y_avg = 0.5 * y / y.sum() + 0.5 * x
            y_avg = np.maximum(y_avg, 0)
            ys = y_avg.sum()
            if ys > 1e-15:
                y_avg = y_avg / ys
                cand, val = _line_search(x, y_avg)
                if cand is not None and val < M * (1 - 1e-11):
                    x = cand
                    moved = True

        # Subgradient
        if not moved:
            cand = _subgrad_step(x, b, M)
            if cand is not None:
                x = cand
                moved = True

        # Coordinate descent on most expensive lag
        if not moved:
            k_max = int(np.argmax(b))
            # Find the pair (i,j) with i+j=k_max maximizing x[i]*x[j], i != j
            jlo = max(0, k_max - n + 1)
            jhi = min(n, k_max + 1)
            best_i, best_j, best_p = -1, -1, 0.0
            for i in range(jlo, jhi):
                j = k_max - i
                if 0 <= j < n and i != j:
                    p = x[i] * x[j]
                    if p > best_p:
                        best_p = p
                        best_i, best_j = i, j
            if best_i >= 0:
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    d = frac * 0.5 * abs(x[best_i] - x[best_j])
                    if d < 1e-15:
                        continue
                    for sign in [1, -1]:
                        cand = x.copy()
                        hi = max(best_i, best_j)
                        lo = min(best_i, best_j)
                        cand[hi] += sign * d
                        cand[lo] -= sign * d
                        cand = np.maximum(cand, 0)
                        cs = cand.sum()
                        if cs < 1e-15:
                            continue
                        cand = cand / cs
                        v = float(np.convolve(cand, cand).max())
                        if v < M * (1 - 1e-11):
                            x = cand
                            moved = True
                            break
                    if moved:
                        break

        if not moved:
            x *= np.exp(rng.normal(0, 0.006, n))
            x = np.maximum(x, 1e-15)
            x = x / x.sum()

    return best_x


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 20.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except (NameError, Exception):
        parent = np.ones(200, dtype=np.float64)
    parent = np.maximum(parent, 0)
    ps = parent.sum()
    if ps < 1e-15 or not np.all(np.isfinite(parent)):
        parent = np.ones(200, dtype=np.float64)
    parent = parent / parent.sum()
    n_par = parent.size

    best_x = parent.copy()
    best_score = _score_raw(best_x)

    def try_update(cand):
        nonlocal best_x, best_score
        try:
            cand = np.asarray(cand, dtype=np.float64)
            cand = np.clip(cand, 0, 1000)
            S = cand.sum()
            if S < 0.01:
                return
            sc = float(2.0 * len(cand) * np.convolve(cand, cand).max() / (S * S))
            if sc < best_score:
                best_score = sc
                best_x = cand.copy()
        except Exception:
            pass

    # === Phase 1: Main optimization on parent ===
    t_phase1 = t0 + min(450.0, budget * 0.32)
    try:
        rx = _optimize(parent, t_phase1, seed=1, max_stuck=100)
        try_update(rx)
    except Exception:
        pass

    # === Phase 2: Diverse initializations at parent length ===
    t_phase2 = t0 + min(850.0, budget * 0.62)
    n = n_par
    xx = np.linspace(0, 1, n)
    inits = []

    # Gaussian bumps of various widths
    for sig in [0.05, 0.08, 0.12, 0.15, 0.2, 0.25, 0.35, 0.5]:
        v = np.exp(-0.5 * ((xx - 0.5) / sig) ** 2)
        inits.append(v / v.sum())

    # Bimodal
    for w in [0.1, 0.15, 0.2, 0.25, 0.35]:
        v = np.exp(-0.5 * ((xx - w) / 0.07) ** 2) + np.exp(-0.5 * ((xx - (1 - w)) / 0.07) ** 2)
        inits.append(v / v.sum())

    # Power law from center
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        v = (1.0 - np.abs(xx - 0.5) * 2.0) ** p + 0.005
        inits.append(np.maximum(v, 0) + 0.001)
        inits[-1] = inits[-1] / inits[-1].sum()

    # Cosine raised
    for m in [2, 3, 4, 5]:
        v = np.abs(np.cos(m * np.pi * (xx - 0.5))) ** 2 + 0.01
        inits.append(v / v.sum())

    # Blends with parent
    pn = parent
    for frac in [0.3, 0.5, 0.7]:
        for init in inits[:6]:
            v = frac * pn + (1 - frac) * init
            inits.append(v / v.sum())

    # Random (seeded)
    rng_init = np.random.default_rng(4242)
    for _ in range(6):
        v = np.abs(rng_init.normal(1.0, 0.6, n)) + 0.01
        inits.append(v / v.sum())

    # Deduplicate (remove near-identical inits)
    unique_inits = [inits[0]]
    for ini in inits[1:]:
        if all(np.max(np.abs(ini - u)) > 0.05 for u in unique_inits):
            unique_inits.append(ini)
    inits = unique_inits

    remaining_2 = t_phase2 - time.monotonic()
    per_init = max(4.0, remaining_2 / max(1, len(inits)))

    for i, init in enumerate(inits):
        if time.monotonic() > t_phase2 - 10:
            break
        alloc = min(time.monotonic() + per_init * 0.9, t_phase2 - 5)
        if alloc - time.monotonic() < 3:
            break
        try:
            rx = _optimize(init, alloc, seed=2000 + i, max_stuck=40)
            try_update(rx)
        except Exception:
            pass

    # === Phase 3: Length exploration ===
    t_phase3 = deadline - 150.0
    if time.monotonic() < t_phase3:
        for dn in [-40, -20, -10, 10, 20, 40, 60, -60, 80]:
            if time.monotonic() > t_phase3 - 20:
                break
            nt = n_par + dn
            if nt < 80 or nt > 700:
                continue
            x_old = np.linspace(0, 1, len(best_x))
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, best_x)
            init_new = np.maximum(init_new, 0)
            sn = init_new.sum()
            if sn < 1e-10:
                continue
            init_new = init_new / sn
            alloc = min(time.monotonic() + 35, t_phase3 - 10)
            try:
                rx = _optimize(init_new, alloc, seed=6000 + nt, max_stuck=50)
                try_update(rx)
            except Exception:
                pass

    # === Phase 4: Final polish on best ===
    if time.monotonic() < deadline - 5:
        af = best_x.copy()
        af = np.maximum(af, 0)
        af = af / af.sum()
        try:
            rx = _optimize(af, deadline, seed=9999, max_stuck=150)
            try_update(rx)
        except Exception:
            pass

    # === Final validation ===
    try:
        final_list = [float(v) for v in best_x]
        fs = float(evaluate_sequence(final_list))
        if not np.isfinite(fs) or fs < 0:
            raise ValueError("invalid score")
    except Exception:
        return [1.0] * 100

    return [float(v) for v in best_x]
```
