def _lp_best_response(f, stop_time):
    """Solve the LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k"""
    n = f.size
    M = float(_conv_self(f).max())
    if not math.isfinite(M) or M <= 0:
        return f.copy()
    
    # Build sparse constraint matrix A (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: max 1^T g s.t. A g <= M, g >= 0
    g_var = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g_var))
    constraints = [A @ g_var <= M]
    prob = cp.Problem(objective, constraints)
    
    # Solve with a solver
    try:
        prob.solve(solver=cp.SCS, max_iters=10000)
    except:
        try:
            prob.solve(solver=cp.ECOS, max_iters=10000)
        except:
            return f.copy()
    
    if g_var.value is None:
        return f.copy()
    
    g = np.asarray(g_var.value, dtype=np.float64).ravel()
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s < 1e-10:
        return f.copy()
    g = g / s
    return g

def _lse_optimize(p, stop_time, consider=None):
    n = p.size
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    u = np.log(np.maximum(p, 1e-30))
    u -= float(u.mean())

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(a, w):
        F = rfft(a[::-1], cross_size)
        G = rfft(w, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
        else:
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return 2.0 * n * (cmax + T * math.log(sum_exp / c.size)), grad

    stages = [
        (0.10, 0.10),
        (0.03, 0.12),
        (0.008, 0.14),
        (0.002, 0.16),
        (0.0005, 0.18),
        (0.0001, 0.20),
        (0.00003, 0.20),
    ]

    best_p = p.copy()
    best_sc = _score_arr(p)
    start = time.monotonic()
    avail = max(0.0, stop_time - start - 10.0)
    total_w = sum(w for _, w in stages)
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        cum += wt
        end_stage = min(stop_time - 4.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)
        best_u = u.copy()
        stale = 0
        while time.monotonic() < end_stage - 1.5:
            maxit = 50 if n > 2000 else 60
            opts = {"maxiter": maxit, "gtol": 1e-12, "ftol": 1e-16, "maxls": 20, "maxdisp": 0}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < float(best_u.size) * 0 + 1e100:
                    pp = _softmax(res.x)
                    sc = _score_arr(pp)
                    if math.isfinite(sc) and sc < best_sc - 1e-13:
                        best_sc = sc
                        best_p = pp.copy()
                        p = pp.copy()
                        best_u = res.x.copy()
                        u = np.log(np.maximum(p, 1e-30))
                        u -= float(u.mean())
                        stale = 0
                        if consider:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    else:
                        stale += 1
                        if stale >= 3:
                            break
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if getattr(res, "status", 1) == 0:
                break

    p_out = _softmax(best_u) if 'best_u' in dir() else best_p
    p_out = np.maximum(p_out, 0.0)
    sc = _score_arr(p_out)
    if math.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_p = p_out.copy()
    return best_p


def _coord_descent(p, stop_time, consider=None, seed=12345):
    n = int(p.size)
    if n == 0:
        return p, float('inf')
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    ss = float(p.sum())
    if ss < 1e-300:
        return p, float('inf')
    p = p / ss
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    Ltot = 2 * n - 1
    G = 56
    flat = 0
    sweep = 0
    GR = (math.sqrt(5.0) - 1.0) / 2.0

    if consider:
        try:
            consider(best_p)
        except Exception:
            pass

    max_sweeps = 100000
    while sweep < max_sweeps and time.monotonic() < stop_time - 2.0:
        improved = False
        if sweep % 3 == 0:
            order = rng.permutation(n)
        elif sweep % 3 == 1:
            order = np.arange(n)
        else:
            order = np.arange(n - 1, -1, -1)

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 1.0:
                break
            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue
            base = c[start_i:end_i].copy()
            infl = 2.0 * p[:Ls]
            diag = i

            max_const = 0.0
            if start_i > 0:
                mc = float(c[:start_i].max())
                if mc > max_const:
                    max_const = mc
            if end_i < Ltot:
                mc = float(c[end_i:].max())
                if mc > max_const:
                    max_const = mc

            S = 1.0
            x_hi = max(50.0 * v, 40.0 / n, 0.02)
            x_hi = min(x_hi, 0.4)
            x_hi = max(x_hi, 1e-8)

            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 < v < x_hi:
                ins = np.searchsorted(pts, v)
                pts = np.insert(pts, ins, v)

            ws = pts - v
            s_new = S + ws
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag < Ls:
                aff[:, diag] += ws * ws
            maff = np.max(aff, axis=1)
            maff = np.maximum(maff, max_const)
            valid = s_new > 1e-12
            scores = np.full(pts.shape, float('inf'))
            if np.any(valid):
                scores[valid] = 2.0 * n * maff[valid] / (s_new[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            if 0 < imin < len(pts) - 1:
                lo = float(pts[max(0, imin - 2)])
                hi = float(pts[min(len(pts) - 1, imin + 2)])
            else:
                lo = 0.0
                hi = x_hi
            if hi - lo < 1e-14:
                lo = max(0.0, best_x - 1e-6)
                hi = best_x + 1e-6

            a_gs = lo
            b_gs = hi
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)

            def eval_1d(xp):
                w = xp - v
                s_nw = S + w
                if s_nw <= 1e-15:
                    return float('inf')
                aff_m = float(np.max(base + w * infl))
                if 0 <= diag < Ls:
                    dval = base[diag] + w * infl[diag] + w * w
                    if dval > aff_m:
                        aff_m = dval
                m = max(aff_m, max_const)
                return 2.0 * n * m / (s_nw * s_nw)

            f_c = eval_1d(c_gs)
            f_d = eval_1d(d_gs)
            for _ in range(28):
                if b_gs - a_gs < 1e-13 * max(1.0, abs(best_x)):
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = eval_1d(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_d = eval_1d(d_gs)
            x_mid = (a_gs + b_gs) / 2.0
            f_mid = eval_1d(x_mid)
            if f_mid < best_val:
                best_x = x_mid
                best_val = f_mid
            for xp in (0.0, x_hi):
                fp = eval_1d(xp)
                if fp < best_val:
                    best_x = xp
                    best_val = fp

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= x_hi:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    aff_seg = base + w * infl
                    if 0 <= diag < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag] += w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    c[start_i:end_i] = aff_seg
                    cur_score = 2.0 * n * new_max
                    improved = True

        ss_new = float(p.sum())
        if ss_new > 1e-300 and math.isfinite(ss_new):
            inv = 1.0 / ss_new
            p *= inv
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-15:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 20.0:
                    kick_size = 0.015 + 0.05 * min(1.0, flat / 15.0)
                    p = p * np.exp(rng.normal(0.0, kick_size, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m) + 0.5) / m
        fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n)
    return out / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 35.0
    if hard <= t0 + 5.0:
        hard = t0 + max(3.0, budget - 2.0)

    parent = _get_parent()
    best_arr = parent.astype(np.float64).copy()
    best_score = _score_arr(best_arr)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        ss = float(a.sum())
        if ss < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()

    consider(parent)
    n_work = int(best_arr.size)
    if n_work < 128:
        n_work = 384
    if n_work > 4096:
        n_work = 3072

    p_work = _resample(best_arr, n_work)

    # Phase 1: LSE optimization
    lse_time = min(180.0, 0.14 * budget)
    lse_end = t0 + lse_time
    if lse_end > t0 + 20.0:
        try:

                p_work = p2 / float(p2.sum())
                consider(p_work)
        except Exception:
            pass

    # Phase 2: Coordinate descent (main)
    cd_end = hard - 20.0

        p_start = _resample(best_arr, n_work)
        try:
            p_cd, _ = _coord_descent(p_start, cd_end, consider=consider, seed=int(seed + 7777))
            if p_cd is not None and p_cd.size > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                consider(p_cd)
        except Exception:
            pass

    # Phase 3: LSE refinement on best
    rem = hard - 15.0 - time.monotonic()
    if rem > 40.0:
        p_ref = _resample(best_arr, n_work)
        try:
            p_r = _lse_optimize(p_ref, t0 + time.monotonic() - t0 + rem * 0.7, consider=consider)
            if p_r is not None and p_r.size == n_work and np.all(np.isfinite(p_r)) and p_r.sum() > 0:
                consider(p_r)
        except Exception:
            pass

    # Phase 4: Final CD squeeze
    rem = hard - 10.0 - time.monotonic()
    if rem > 20.0:
        p_final = _resample(best_arr, n_work)
        try:
            p_f, _ = _coord_descent(p_final, hard - 10.0, consider=consider, seed=int(seed + 9999))
            if p_f is not None and p_f.size > 0 and np.all(np.isfinite(p_f)) and p_f.sum() > 0:
                consider(p_f)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
