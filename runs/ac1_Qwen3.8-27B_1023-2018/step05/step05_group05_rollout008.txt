import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():

def _conv(a):
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    sz = next_fast_len(L)
    F = rfft(a, sz)
    c = irfft(F * F, sz)[:L]
    return np.asarray(c, dtype=np.float64)


def _score(a):
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _lse_stage(p, T_frac, stop_t, max_rounds=12):

    p = p / s

    cm = float(c.max())

def _cd_round(p, stop_t, seed=42, G=60, fine_pts=35):
    """One coordinate descent sweep with incremental convolution."""
    n = p.size
    p = np.maximum(p.astype(np.float64).copy(), 0.0)
    S = float(p.sum())
    if S < 1e-300:
        return p, math.inf
    c = _conv(p)
    Ltot = 2 * n - 1
    cur_sc = 2.0 * n * float(c.max()) / (S * S)
    best_sc = cur_sc
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    flat = 0
    max_sweeps = 20000
    sweep = 0

    while sweep < max_sweeps and time.monotonic() < stop_t - 2.0:
        improved = False
        order = rng.permutation(n)
        for i_idx in order:
            i = int(i_idx)
            if time.monotonic() > stop_t - 0.3:
                break
            v = float(p[i])
            start = i
            end = i + n  # affected: c[start:end]
            if end > Ltot:
                end = Ltot
                infl_len = end - start
            else:
                infl_len = n

            c_aff = c[start:end].copy()
            infl = 2.0 * p[0:infl_len]

            max_out = 0.0
            if start > 0:
                max_out = max(max_out, float(c[:start].max()))
            if end < Ltot:
                max_out = max(max_out, float(c[end:].max()))

            # Range for new value x
            x_hi = max(30.0 * v, 15.0 / n, 0.005)
            x_hi = min(x_hi, 2.0 * S)
            if x_hi <= 1e-18:
                x_hi = 2.0 / n
            x_lo = 0.0

            # Build grid (non-uniform: denser near 0 and near v)
            Gc = G
            pts_lin = np.linspace(x_lo, x_hi, Gc)
            # Add extra points near v and near 0
            extra = np.concatenate([
                np.linspace(0, min(v * 0.5, x_hi / 10), min(10, Gc // 6)),
                [v] if 0 < v < x_hi else [],
                np.linspace(max(x_hi * 0.7, v - x_hi / Gc), x_hi, 8),
            ])
            pts = np.unique(np.concatenate([pts_lin, extra]))
            pts = pts[pts <= x_hi]
            Gc = len(pts)
            if Gc < 5:
                continue

            w = pts - v
            new_S = S + w
            valid = (new_S > 1e-14) & (pts >= 0)
            if not np.any(valid):
                continue

            c_new = c_aff[None, :] + w[:, None] * infl[None, :]
            diag_pos = 2 * i - start

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            bx = float(pts[imin])
            bv = float(scores[imin])

            # Fine search
            if Gc > 1:

                lo = max(x_lo, bx - 3.0 * dw)
                hi = min(x_hi, bx + 3.0 * dw)
                if hi - lo > 1e-16:
                    fine = np.linspace(lo, hi, fine_pts)
                    wf = fine - v

                    vf = (nSf > 1e-14) & (fine >= 0)
                    if np.any(vf):

                if abs(w_new) > 1e-18:

                    c[start:end] = c_aff + w_new * infl
                    if 0 <= diag_pos < infl_len:

                    improved = True

            p *= inv
            c *= inv * inv
            S = 1.0
            cur_sc = 2.0 * n * float(c.max())

            S = float(p.sum())
            if S > 1e-10:

                p *= inv
                c *= inv * inv
                S = 1.0
            cur_sc = 2.0 * n * float(c.max())

        if cur_sc < best_sc - 1e-16:
            best_sc = cur_sc
            best_p = p.copy()

        if improved:
            flat = 0
        else:

            if flat >= 4:

                    noise = rng.normal(0, 0.025, n)
                    p = p * np.exp(noise)
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 1e-10:
                        p /= ss
                    c = _conv(p)
                    S = 1.0
                    cur_sc = 2.0 * n * float(c.max())
                    flat = 0
                else:

        sweep += 1

    return best_p, best_sc
