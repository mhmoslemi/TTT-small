def _lp_solve_direction(a, M):
    if linprog is None or sp is None:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()

        return None

    try:
        idx = np.arange(n, dtype=np.int32)
        cols = np.repeat(idx, n)
        off = np.tile(idx, n)
        rows = cols + off
        data = a[off].astype(np.float64)
        A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        cobj = -np.ones(n, dtype=np.float64)

        try:
            res = linprog(
                cobj,

            )

            res = linprog(cobj, A_ub=A, b_ub=b_ub, bounds=(0.0, None))
