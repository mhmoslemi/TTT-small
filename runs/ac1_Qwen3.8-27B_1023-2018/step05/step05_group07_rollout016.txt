def _score_fast(a):

def _build_A(n, a):
    """Convolution matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    d = k_idx - j_idx
    mask = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[d[mask]]
    return A


def _lp_solve(a, M, tl=5.0):

    A = _build_A(n, a)

                               'dual_feasibility_tolerance': 1e-12})

    except Exception:
        pass
    return None

    for i, ini in enumerate(top_inits):
        if time.monotonic() > t2 - 15:
            break
        alloc = min(time.monotonic() + per_init, t2 - 10)
        if alloc < 10:
            break
        try:
            r = _optimize(ini, alloc, seed=3000 + i)
            try_update(r)
        except Exception:
            pass

    # === Phase 3: Different sequence lengths ===
    if time.monotonic() < deadline - 120:
        for nt in [150, 200, 220, 250, 300, 320, 350, 400, 450, 500, 600]:
            if time.monotonic() > deadline - 50:
                break
            x_new = np.linspace(0, 1, nt)
            init_n = np.interp(x_new, np.linspace(0, 1, len(best_seq)), best_seq)
            init_n = np.maximum(init_n, 0)
            if init_n.sum() < 1e-10:
                continue
            init_n = init_n / init_n.sum()
            alloc = min(100.0, deadline - time.monotonic() - 40)
            if alloc < 20:
                break
            try:
                r = _optimize(init_n, time.monotonic() + alloc, seed=9000 + nt)
                try_update(r)
            except Exception:
                pass

    # === Phase 4: Final deep polish on best ===
    if time.monotonic() < deadline - 20:
        af = best_seq.copy()
        af = af / af.sum()
        try:
            r = _optimize(af, deadline, seed=31415)
            try_update(r)
        except Exception:
            pass

    # === Final validation ===
    try:
        fs = float(evaluate_sequence([float(v) for v in best_seq]))
        if not np.isfinite(fs):
            return [1.0] * 100
    except Exception:
        return [1.0] * 100

    return [float(v) for v in best_seq]
