def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size)
    return e / s

def _p2u(p):
    return np.log(np.maximum(p, 1e-300))

def _lse_optimize(p, stop_time, consider):
    """LSE soft-max optimization with L-BFGS-B"""
    n = p.size
    s = float(p.sum())
    p = p / s
    
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c
    
    # Cross-correlation for gradient
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2
    
    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n-1:2*n-1]
        np.maximum(q, 0.0, out=q)
        return q
    
    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -400, 400)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * math.log(sum_exp / c.size)
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad
    
    u = _p2u(p)
    best_u = u.copy()
    
    # Multi-stage temperature
    c0 = cfast(p)
    cmax0 = float(c0.max())
    
    stages = [
        (0.05, 0.15),  # (frac of cmax, time fraction)
        (0.02, 0.15),
        (0.005, 0.20),
        (0.001, 0.25),
        (0.0002, 0.25),
    ]
    
    start = time.monotonic()
    total_w = sum(w for _, w in stages)
    cum = 0.0
    best_f = float('inf')
    
    for frac, wt in stages:
        cum += wt
        end_stage = min(stop_time - 2.0, start + (stop_time - start - 5.0) * (cum / total_w))
        if time.monotonic() > end_stage - 3.0:
            break
        
        T = max(1e-300, frac * cmax0)
        f0, _ = lse_func(u, T)
        if not math.isfinite(f0):
            continue
        if f0 < best_f:
            best_f = f0
            best_u = u.copy()
        
        stale = 0
        while time.monotonic() < end_stage - 1.0:
            try:
                res = minimize(lse_func, u, args=(T,), method='L-BFGS-B', jac=True,
                              options={'maxiter': 50, 'gtol': 1e-14, 'ftol': 1e-16,
                                       'maxls': 30, 'maxcor': 30, 'maxdisp': 0, 'disp': False})
            except Exception:
                break
            
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_f - 1e-15:
                    best_f = fval
                    best_u = res.x.copy()
                    u = res.x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 4:
                        break
            if getattr(res, 'status', 1) == 0:
                break
    
    p_out = _softmax(best_u)
    p_out = np.maximum(p_out, 0.0)
    ss = float(p_out.sum())
    if ss > 1e-300:
        p_out /= ss
    consider(p_out)
    return p_out

    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _p2u(p):
    return np.log(np.maximum(p, 1e-300))


def _lse_optimize(p, stop_time, consider):
    n = p.size
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -400.0, 400.0)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * math.log(sum_exp / c.size)
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    u = _p2u(p)
    best_u = u.copy()
    best_f = float('inf')

    c0 = cfast(p)
    cmax0 = float(c0.max()) if c0.size else 1.0
    if not math.isfinite(cmax0) or cmax0 <= 0:
        return p

    stages = [
        (0.04, 0.14),
        (0.015, 0.16),
        (0.004, 0.20),
        (0.001, 0.25),
        (0.0002, 0.25),
    ]

    start = time.monotonic()
    total_w = sum(w for _, w in stages)
    cum = 0.0

    for frac, wt in stages:
        cum += wt
        end_stage = min(stop_time - 2.0, start + (stop_time - start - 5.0) * (cum / total_w))
        if time.monotonic() > end_stage - 2.0:
            break

        T = max(1e-300, frac * cmax0)
        try:
            f0, _ = lse_func(u, T)
        except Exception:
            continue
        if not math.isfinite(f0):
            continue
        if f0 < best_f:
            best_f = f0
            best_u = u.copy()

        stale = 0
        while time.monotonic() < end_stage - 1.0:
            try:
                res = minimize(lse_func, u, args=(T,), method='L-BFGS-B', jac=True,
                               options={'maxiter': 50, 'gtol': 1e-14, 'ftol': 1e-16,
                                        'maxls': 30, 'maxcor': 30, 'maxdisp': 0, 'disp': False})
            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                try:
                    fval = float(res.fun)
                except Exception:
                    fval = float('inf')
                if math.isfinite(fval) and fval < best_f - 1e-15:
                    best_f = fval
                    best_u = res.x.copy()
                    u = res.x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 4:
                        break
            if getattr(res, 'status', 1) == 0:
                break

    p_out = _softmax(best_u)
    p_out = np.maximum(p_out, 0.0)
    ss = float(p_out.sum())
    if ss > 1e-300:
        p_out /= ss
    consider(p_out)
    return p_out


def _coord_descent(p, stop_time, consider, seed=42):
    n = p.size
    p = np.maximum(p.astype(np.float64), 0.0).copy()
    S = float(p.sum())
    if S < 1e-300:
        return p, float('inf')
    p /= S
    S = 1.0

    L = 2 * n - 1
    c = _conv_fft(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    consider(p)

    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while time.monotonic() < stop_time - 0.5:
        if sweep > 200000:
            break

        if sweep < 30:
            G = 28
        elif sweep < 80:
            G = 20
        else:
            G = 14

        improved = False
        order = rng.permutation(n)

        for idx in order:
            i = int(idx)
            if time.monotonic() > stop_time - 0.3:
                break

            v = float(p[i])
            start_k = i
            end_k = min(i + n, L)
            Ls = end_k - start_k
            if Ls <= 0:
                continue

            diag_pos = i  # position in base where k = 2i
            has_diag = diag_pos < Ls

            base = c[start_k:end_k].copy()
            base -= 2.0 * v * p[:Ls]
            if has_diag:
                base[diag_pos] += 2.0 * v * p[diag_pos] - v * v

            max_const = 0.0
            if start_k > 0:
                ml = float(c[:start_k].max())
                if ml > max_const:
                    max_const = ml
            if end_k < L:
                mr = float(c[end_k:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(20.0 * v, 50.0 / n, 0.02, 1e-10)
            x_hi = min(x_hi, 1.0)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = 1.0 / n

            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 < v < x_hi:
                pts = np.unique(np.concatenate((pts, [v])))

            xs = pts
            denom = 1.0 + (xs - v)
            valid = denom > 1e-14

            if not np.any(valid):
                continue

            xsv = xs[valid]
            denomv = denom[valid]

            aff = base[None, :] + 2.0 * xsv[:, None] * p[:Ls][None, :]
            if has_diag:
                aff[:, diag_pos] -= 2.0 * xsv * p[diag_pos]
                aff[:, diag_pos] += xsv * xsv

            maff = np.maximum(aff.max(axis=1), max_const)
            scores_v = (2.0 * n * maff) / (denomv * denomv)

            imin = int(np.argmin(scores_v))
            best_x = float(xsv[imin])
            best_val = float(scores_v[imin])

            if G > 6 and math.isfinite(best_val):
                dw = x_hi / G
                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)
                if hi - lo > 1e-15:
                    fine_pts = np.linspace(lo, hi, 16)
                    denom_f = 1.0 + (fine_pts - v)
                    valid_f = denom_f > 1e-14
                    if np.any(valid_f):
                        xf = fine_pts[valid_f]
                        denomf = denom_f[valid_f]
                        aff_f = base[None, :] + 2.0 * xf[:, None] * p[:Ls][None, :]
                        if has_diag:
                            aff_f[:, diag_pos] -= 2.0 * xf * p[diag_pos]
                            aff_f[:, diag_pos] += xf * xf
                        maff_f = np.maximum(aff_f.max(axis=1), max_const)
                        scores_f = (2.0 * n * maff_f) / (denomf * denomf)
                        imin_f = int(np.argmin(scores_f))
                        if float(scores_f[imin_f]) < best_val - 1e-16:
                            best_x = float(xf[imin_f])
                            best_val = float(scores_f[imin_f])

            if math.isfinite(best_val) and best_val < cur_score - 1e-14:
                dx = best_x - v
                if abs(dx) > 1e-14:
                    p[i] = best_x
                    S += dx
                    c[start_k:end_k] = base + 2.0 * best_x * p[:Ls]
                    if has_diag:
                        c[start_k + diag_pos] = base[diag_pos] + best_x * best_x
                    cur_score = best_val
                    improved = True

        if S > 1e-300:
            inv = 1.0 / S
            if abs(inv - 1.0) > 1e-10:
                p *= inv
                c *= inv * inv
            S = 1.0

        if sweep % 10 == 0:
            c = _conv_fft(p)

        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
            consider(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 6:
                if time.monotonic() < stop_time - 5.0:
                    noise_amt = 0.018 + 0.012 * math.exp(-flat * 0.12)
                    p = p * np.exp(rng.normal(0.0, noise_amt, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                        S = 1.0
                    c = _conv_fft(p)
                    cur_score = float(2.0 * n * c.max())
                    if cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_p = p.copy()
                        consider(best_p)
                    flat = 0
                else:
                    break

        sweep += 1

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 22.0

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float('inf')

    s = float(best_arr.sum())
    if s > 0.01 and best_arr.size > 0:
        c = _conv_fft(best_arr)
        best_score = float(2.0 * best_arr.size * c.max() / (s * s))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            sa = float(a.sum())
            if sa < 0.01:
                return
            ca = _conv_fft(a)
            sc = float(2.0 * a.size * ca.max() / (sa * sa))
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_arr = a.copy()
                best_score = sc
        except Exception:
            pass

    n = best_arr.size
    p = best_arr.copy()
    ps = float(p.sum())
    if ps > 0:
        p /= ps

    # Phase 1: LSE optimization
    lse_end = min(hard - 80.0, t0 + 0.35 * (budget - 30.0))
    if lse_end > time.monotonic() + 30.0:
        try:
            p = _lse_optimize(p, lse_end, consider)
        except Exception:
            pass

    # Phase 2: Main coordinate descent
    coord_end = min(hard - 25.0, t0 + 0.88 * (budget - 30.0))
    now = time.monotonic()
    if coord_end > now + 20.0:
        try:
            p2, _ = _coord_descent(p, coord_end, consider, seed=int(seed))
            if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p = p2
        except Exception:
            pass

    # Phase 3: Final coordinate polish on best found
    now = time.monotonic()
    final_end = hard - 5.0
    if final_end > now + 20.0 and best_arr.size > 0:
        try:
            pb = best_arr / float(best_arr.sum())
            p3, _ = _coord_descent(pb, final_end, consider, seed=int(seed + 77777))
            if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and p3.sum() > 0:
                consider(p3)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        if parent.size > 0:
            return [float(x) for x in parent[:10]]
        return [1.0]
