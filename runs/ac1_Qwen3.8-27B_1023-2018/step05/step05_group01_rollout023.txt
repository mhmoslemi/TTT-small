import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m) / (s * s)


def _score_exact_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m) / (s * s)


def _softmax_fn(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        return src.copy()
    m = src.size
    xp = (np.arange(m) + 0.5) / m
    fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
    out = np.maximum(fp, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def _lse_optimize(p, stop_time, consider=None, seed=42):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    u0 = np.log(np.maximum(p, 1e-300))
    u0 = u0 - float(np.mean(u0))

    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax_fn(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    # Adaptive geometric cooling: more stages, finer control
    c_init = cfast(p)
    c0 = float(c_init.max()) if c_init.size else 1.0
    T_start = max(1e-300, 0.10 * c0)
    T_end = max(1e-300, 2e-5 * c0)
    n_stages = 14
    ratios = (T_end / T_start) ** (1.0 / n_stages)

    start = time.monotonic()
    best_p = p.copy()
    best_sc = _score_fft(p)
    rng = np.random.default_rng(seed)

    for si in range(n_stages):
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        remaining = stop_time - now
        if remaining < 8.0:
            break

        T = T_start * (ratios ** si)
        frac_of_time = 0.12 if si < 3 else 0.14
        stage_budget = min(remaining * 0.25, frac_of_time * (stop_time - start))
        stage_end = now + max(5.0, stage_budget)

        f0, _ = lse_func(u0, T)
        if not math.isfinite(f0):
            continue
        best_f = f0
        x_cur = u0.copy()
        stale = 0
        max_iter_per_stage = 35

        iter_count = 0
        while time.monotonic() < stage_end - 0.5 and iter_count < max_iter_per_stage:
            iter_count += 1
            opts = {"maxiter": 30, "gtol": 1e-11, "ftol": 1e-16,
                    "maxls": 25, "maxcor": 25, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, x_cur, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u0.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-16:
                        best_f = fval
                        x_cur = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 2:
                            sigma = 0.003 * (1.0 + 0.1 * stale)
                            x_cur = x_cur + rng.normal(0, sigma, x_cur.size)
                            continue
            if getattr(res, "status", 1) == 0:
                if time.monotonic() < stage_end - 1.0 and stale < 2 and iter_count < max_iter_per_stage - 2:
                    x_cur = x_cur + rng.normal(0, 0.002, x_cur.size)
                    f_chk, _ = lse_func(x_cur, T)
                    if math.isfinite(f_chk) and f_chk < best_f:
                        best_f = f_chk
                    stale += 1
                else:
                    break

        p_cur = _softmax_fn(x_cur)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            u0 = np.log(np.maximum(best_p, 1e-300))
            u0 = u0 - float(np.mean(u0))
        else:
            if math.isfinite(sc_cur) and sc_cur < best_sc + 0.003:
                u0 = np.log(np.maximum(p_cur, 1e-300))
                u0 = u0 - float(np.mean(u0))
            else:
                p_cur = best_p.copy()

        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, G=None, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()

    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()

    flat = 0
    sweep = 0
    max_sweeps = 50000

        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])

            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            # Quick skip: if v is 0 and the box max is well below M_off, skip
            if v < 1e-15:
                box_max = float(c[start:end].max()) if Ls > 0 else 0.0
                off_max = 0.0
                if start > 0:
                    off_max = max(off_max, float(c[:start].max()))
                if end < Ltot:
                    off_max = max(off_max, float(c[end:].max()))

                    continue

            base = c[start:end].copy()
            # Influence: when we add delta to p[i], c[start:end] += delta * p[cyclic]
            # p[cyclic] for positions start..end: p[j-start] for j in [start,end)
            infl = p[:Ls].copy()

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:

                    max_const = mr

            # Determine search range

            diag_i = i - start if 0 <= i < Ltot else -1

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[valid, None] * infl[None, :]
                if 0 <= diag_i < Ls:

                    aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                dv = denom[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)

            vals = eval_xs(pts)

            # Fine refinement around best

                hi = min(x_hi, best_x + 2.0 * dw)

        if consider is not None:

            except Exception:

        else:

                    sigma = 0.015 + 0.01 * min(flat, 10) / 10.0

                    p = np.maximum(p, 0.0)

                        p /= ss
                    else:

                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            try:

                            except Exception:
                                pass
                    flat = 0
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:

    except Exception:

    def run_opt(n, init_arr, end_time, sd, do_lse=True, do_coord=True):
        nonlocal best_arr, best_score
        p = _resample_mid(init_arr, n)
        if p.size != n or float(p.sum()) <= 0:
            p = np.full(n, 1.0 / n)
        p = p / float(p.sum())
        consider(p)

        if time.monotonic() > end_time - 10.0:
            return

        phase_len = max(0.0, float(end_time) - time.monotonic())

        if do_lse and phase_len > 25.0:
            lse_stop = min(float(end_time) - max(15.0, 0.3 * phase_len),
                           time.monotonic() + 0.6 * phase_len)
            if lse_stop > time.monotonic() + 8.0:
                try:
                    p2 = _lse_optimize(p, lse_stop, consider=consider, seed=sd)
                    if p2 is not None and p2.size == n and float(p2.sum()) > 0:
                        p = p2 / float(p2.sum())
                        consider(p)
                except Exception:
                    pass

        rem = float(end_time) - time.monotonic()
        if do_coord and rem > 12.0:
            cstop = min(float(end_time) - 1.0, time.monotonic() + rem - 2.0)
            if cstop > time.monotonic() + 8.0:
                try:
                    p3, _ = _coord_polish(p, cstop, seed=sd, consider=consider)
                    if p3 is not None and p3.size == n and float(p3.sum()) > 0:
                        p = p3 / float(p3.sum())
                        consider(p)
                except Exception:
                    pass
        return p

    # Make alternative initializations
    def make_perturbed(src, n, sd, sigma=0.03):
        p = _resample_mid(src, n)
        rng = np.random.default_rng(sd)
        q = p * np.exp(rng.normal(0, sigma, n))
        q = np.maximum(q, 0.0)
        s = float(q.sum())
        if s > 0:
            q /= s
        else:
            q = p
        return q

    def make_asym_init(src, n, sd):
        p = _resample_mid(src, n)
        rng = np.random.default_rng(sd)
        x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
        side = 1.0 if rng.random() < 0.5 else -1.0
        amp = 0.2 + 1.0 * float(rng.random())
        curv = 0.3 + 1.5 * float(rng.random())
        q = p * np.exp(side * amp * (2.0 * x - 1.0) + curv * (2.0 * x - 1.0) ** 2)
        q = np.maximum(q, 0.0)
        s = float(q.sum())
        if s > 0:
            q /= s
        else:
            q = p
        return q

    # Phase 1: Main optimization at parent length (40% of budget)
    p1_end = t0 + 0.40 * (hard - t0)
    run_opt(n_parent, best_arr, p1_end, int(seed + 100), do_lse=True, do_coord=True)

    # Phase 2: Second pass with perturbation (20%)
    if time.monotonic() < hard - 60.0:
        rng2 = np.random.default_rng(seed + 777)
        pert = make_perturbed(best_arr, n_parent, int(seed + 2000), sigma=0.04)
        sub_end = t0 + 0.60 * (hard - t0)
        run_opt(n_parent, pert, sub_end, int(seed + 2100), do_lse=True, do_coord=True)

    # Phase 3: Asymmetric initialization (15%)
    if time.monotonic() < hard - 50.0:
        init = make_asym_init(best_arr, n_parent, int(seed + 5000))
        sub_end = t0 + 0.75 * (hard - t0)
        run_opt(n_parent, init, sub_end, int(seed + 5100), do_lse=True, do_coord=True)

    # Phase 4: Try different length if time allows (10%)
    if time.monotonic() < hard - 80.0:
        n_alt = min(n_parent * 2, 4096) if n_parent <= 2048 else max(n_parent // 2, 1024)
        if n_alt != n_parent:
            sub_end = t0 + 0.88 * (hard - t0)
            if sub_end > time.monotonic() + 30.0:
                run_opt(n_alt, best_arr, sub_end, int(seed + 8000), do_lse=True, do_coord=True)

    # Phase 5: Final intensive polishing (remaining time)
    final_end = min(hard - 10.0, t0 + budget - 10.0)
    if final_end > time.monotonic() + 20.0 and best_arr.size > 0:
        nbest = int(best_arr.size)
        p_final = best_arr.copy()
        s_final = float(p_final.sum())
        if s_final > 1e-300:
            p_final /= s_final

        # Coordinate polish on the best
        cstop = min(final_end - 3.0, time.monotonic() + max(15.0, 0.7 * (final_end - time.monotonic())))
        if cstop > time.monotonic() + 10.0:
            try:
                p3, _ = _coord_polish(p_final, cstop, seed=int(seed + 123456789), consider=consider)
                if p3 is not None and p3.size == nbest and float(p3.sum()) > 0:
                    p_final = p3 / float(p3.sum())
                    consider(p_final)
            except Exception:
                pass

        # Quick LSE if time remains
        rem = final_end - time.monotonic()
        if rem > 20.0 and nbest <= 5000:
            lstop = min(final_end - 5.0, time.monotonic() + 0.5 * rem)
            if lstop > time.monotonic() + 10.0:
                try:
                    p2 = _lse_optimize(p_final, lstop, consider=consider, seed=int(seed + 99999))
                    if p2 is not None and p2.size == nbest and float(p2.sum()) > 0:
                        p_final = p2 / float(p2.sum())
                        consider(p_final)
                except Exception:
                    pass

        # One more coord polish
        cstop2 = min(final_end - 1.0, time.monotonic() + max(5.0, (final_end - time.monotonic()) * 0.8))
        if cstop2 > time.monotonic() + 5.0:
            try:
                p3, _ = _coord_polish(best_arr, cstop2, seed=int(seed + 987654321), consider=consider)
                if p3 is not None and p3.size > 0 and float(p3.sum()) > 0:
                    p_final = p3 / float(p3.sum())
                    consider(p_final)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in np.clip(np.where(np.isfinite(parent), parent, 0.0), 0, 1000)]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
