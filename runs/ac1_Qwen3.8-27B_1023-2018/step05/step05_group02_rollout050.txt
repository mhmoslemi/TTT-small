        # k-j ranges from k-j_lo to k-j_hi
        # = min(k, n-1) to max(0, k-n+1)

        sb = float(b.sum())
        if sb > 1e-15:
            g = b / sb
            return g

        return best_p_fw

    # ===== Coordinate Descent =====
    def coord_descent(p_in, stop_time, max_sweeps=2000, G=40, seed_cd=12345):
        n_local = n
        p_in = np.maximum(np.asarray(p_in, dtype=np.float64).ravel(), 0.0).copy()
        ss = float(p_in.sum())
        if ss < 1e-300:
            return p_in
        p_in = p_in / ss
        Ltot = 2 * n_local - 1

        c = conv_self(p_in)
        S = 1.0
        cur_score = float(2.0 * n_local * c.max())
        best_p_cd = p_in.copy()
        best_score_cd = cur_score
        rng = np.random.default_rng(seed_cd)
        sweep = 0
        flat = 0

        while sweep < max_sweeps and time.monotonic() < stop_time - 1.5:
            improved = False
            order = rng.permutation(n_local)
            for idx_i in order:
                i = int(idx_i)
                if time.monotonic() > stop_time - 0.5:
                    break
                if S <= 1e-300:
                    break
                if S > 1.3 or S < 0.7:
                    inv = 1.0 / S
                    p_in *= inv
                    c *= inv * inv
                    S = 1.0
                v = float(p_in[i])
                start = i
                end = min(i + n_local, Ltot)
                Lseg = end - start
                if Lseg <= 0:
                    continue
                base = c[start:end]
                infl = 2.0 * p_in[:Lseg]
                # Max outside affected region
                max_out = 0.0
                if start > 0:
                    mo = float(c[:start].max())
                    if mo > max_out:
                        max_out = mo
                if end < Ltot:
                    mo = float(c[end:].max())
                    if mo > max_out:
                        max_out = mo

                x_hi = max(20.0 * v, 50.0 / n_local, 0.03 / n_local, 1e-14)
                x_hi = min(x_hi, 5.0 / n_local)
                x_hi = max(x_hi, 1e-14)

                pts = np.linspace(0.0, x_hi, G + 1)
                if 0.0 <= v <= x_hi:
                    pts = np.unique(np.concatenate((pts, [v])))

                def eval_xs(xs):
                    xs = np.asarray(xs, dtype=np.float64).ravel()
                    w = xs - v
                    denom = S + w
                    vals = np.full(xs.shape, 1e100)
                    valid = denom > 1e-15
                    if not np.any(valid):
                        return vals
                    aff = base[None, :] + w[:, None] * infl[None, :]
                    maff = np.max(aff, axis=1)
                    di = 2 * i - start  # position of k=2i within [start, end)
                    if 0 <= di < Lseg:
                        dval = base[di] + w * infl[di] + w * w
                        maff = np.maximum(maff, dval)
                    maff = np.maximum(maff, max_out)
                    dv = denom[valid]
                    mv = maff[valid]
                    vals[valid] = (2.0 * n_local * mv) / (dv * dv)
                    return vals

                vals = eval_xs(pts)
                if not np.any(np.isfinite(vals)):
                    continue
                imin = int(np.argmin(vals))
                best_x = float(pts[imin])
                best_val = float(vals[imin])

                # Fine search
                dw = x_hi / float(G)
                lo_f = max(0.0, best_x - 2.0 * dw)
                hi_f = min(x_hi, best_x + 2.0 * dw)
                if hi_f > lo_f:
                    fine_pts = np.linspace(lo_f, hi_f, 50)
                    if lo_f <= v <= hi_f:
                        fine_pts = np.unique(np.concatenate((fine_pts, [v])))
                    vals2 = eval_xs(fine_pts)
                    if np.any(np.isfinite(vals2)):
                        imin2 = int(np.argmin(vals2))
                        vx2 = float(fine_pts[imin2])
                        vv2 = float(vals2[imin2])
                        if vv2 < best_val:
                            best_x = vx2
                            best_val = vv2

                # Golden section
                gr = (math.sqrt(5.0) - 1.0) / 2.0
                g_lo = max(0.0, best_x - dw)
                g_hi = min(x_hi, best_x + dw)
                if g_hi - g_lo > 1e-18:
                    a_gs = g_hi - gr * (g_hi - g_lo)
                    b_gs = g_lo + gr * (g_hi - g_lo)
                    fa = float(eval_xs(np.array([a_gs]))[0])
                    fb = float(eval_xs(np.array([b_gs]))[0])
                    for _ in range(12):
                        if not math.isfinite(fa):
                            fa = 1e100
                        if not math.isfinite(fb):
                            fb = 1e100
                        if fa < fb:
                            g_hi = b_gs
                            b_gs = a_gs
                            fb = fa
                            a_gs = g_hi - gr * (g_hi - g_lo)
                            fa = float(eval_xs(np.array([a_gs]))[0])
                        else:
                            g_lo = a_gs
                            a_gs = b_gs
                            fa = fb
                            b_gs = g_lo + gr * (g_hi - g_lo)
                            fb = float(eval_xs(np.array([b_gs]))[0])
                        if g_hi - g_lo < 1e-17:
                            break
                    gs_x = (g_lo + g_hi) / 2.0
                    gv = float(eval_xs(np.array([gs_x]))[0])
                    if math.isfinite(gv) and gv < best_val:
                        best_x = gs_x
                        best_val = gv

                if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= 10.0:
                    w = best_x - v
                    if abs(w) > 1e-16:
                        p_in[i] = best_x
                        S += w
                        c[start:end] += w * infl
                        di = 2 * i
                        if 0 <= di < Ltot:
                            c[di] += w * w
                        cur_score = best_val
                        improved = True

            # Normalize at end of sweep
            if S > 1e-300 and math.isfinite(S):
                inv = 1.0 / S
                p_in *= inv
                c *= inv * inv
                S = 1.0
            else:
                break
            c = conv_self(p_in)
            np.maximum(c, 0.0, out=c)
            new_score = float(2.0 * n_local * c.max())
            if math.isfinite(new_score) and new_score < best_score_cd - 1e-16:
                best_score_cd = new_score
                best_p_cd = p_in.copy()
                consider(p_in)

            if improved:
                flat = 0
            else:
                flat += 1
                if flat >= 5:
                    if time.monotonic() < stop_time - 8.0:
                        try:
                            noise_amt = 0.02 + 0.01 * math.exp(-flat * 0.15)
                            p_in = p_in * np.exp(rng.normal(0.0, noise_amt, n_local))
                            p_in = np.maximum(p_in, 0.0)
                            ss2 = float(p_in.sum())
                            if ss2 > 1e-15:
                                p_in /= ss2
                            else:
                                p_in = best_p_cd.copy()
                            c = conv_self(p_in)
                            S = 1.0
                            cur_score = float(2.0 * n_local * c.max())
                            if math.isfinite(cur_score) and cur_score < best_score_cd - 1e-15:
                                best_score_cd = cur_score
                                best_p_cd = p_in.copy()
                                consider(p_in)
                            flat = 0
                        except Exception:
                            break
                    else:
                        break
            sweep += 1

        p_in = np.maximum(p_in, 0.0)
        ss = float(p_in.sum())
        if ss > 1e-300:
            p_in /= ss
        return p_in

    # ===== Main optimization loop =====
    now = time.monotonic()
    rem = deadline - now

    # Phase 1: LSE (use 25% of time)
    p1_stop = now + 0.22 * rem
    p = lse_optimize(p, p1_stop, fine_only=False)
    consider(p)

    now = time.monotonic()
    rem = deadline - now
    if rem < 10:
        p = best_p.copy()

    # Phase 2: LP Frank-Wolfe (use 25% of remaining)
    p2_stop = now + 0.22 * rem
    if rem > 20:
        p = lp_fw_step(p, p2_stop, n_iter=80)
        consider(p)

    now = time.monotonic()
    rem = deadline - now
    if rem < 10:
        p = best_p.copy()

    # Phase 3: Main coordinate descent (use 35% of remaining)
    p3_stop = now + 0.35 * rem
    if rem > 15:
        p = coord_descent(p, p3_stop, max_sweeps=5000, G=40, seed_cd=seed + 111)
        consider(p)

    now = time.monotonic()
    rem = deadline - now
    if rem < 10:
        p = best_p.copy()

    # Phase 4: LSE fine-tuning (use 15% of remaining)
    p4_stop = now + 0.15 * rem
    if rem > 20:
        p = lse_optimize(p, p4_stop, fine_only=True)
        consider(p)

    now = time.monotonic()
    rem = deadline - now
    if rem < 10:
        p = best_p.copy()

    # Phase 5: Final coordinate descent (use remaining)
    p5_stop = deadline - 3.0
    if rem > 15:
        p = coord_descent(p, p5_stop, max_sweeps=5000, G=50, seed_cd=seed + 999)
        consider(p)

    # Final validation
    try:
        out = np.maximum(best_p, 0.0).copy()
        out = np.clip(out, 0.0, 1000.0)
        out = np.where(np.isfinite(out), out, 0.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.ones(64, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in np.clip(parent, 0, 1000)]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
