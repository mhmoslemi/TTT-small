def _get_parent():

def _conv_self(a):
    """Self-convolution via FFT."""
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2*n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(c, 0.0)

def _score(a):

        return math.inf
    c = _conv_self(a)
    m = c.max()
    return 2.0 * n * m / (s * s)

def _lse_optimize(p, stop_time, consider, T_fracs=(0.1, 0.03, 0.01, 0.003, 0.001)):
    """LSE optimization with L-BFGS-B."""
    n = p.size
    p = np.maximum(p, 0.0).astype(np.float64)
    s = p.sum()
    if s < 1e-300:
        return p
    p = p / s
    
    conv_size = next_fast_len(2*n-1)
    cross_size = next_fast_len(3*n-2)
    conv_len = 2*n-1
    cross_len = 3*n-2
    
    def cfast(a):
        F = rfft(a, conv_size)
        return np.maximum(irfft(F*F, conv_size)[:conv_len], 0.0)
    
    def xfast(pp, ww):
        # Compute (pp * ww)[n-1:2n-1] (the "effective" part for gradient)
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F*G, cross_size)[:cross_len]
        return np.maximum(c[n-1:2*n-1], 0.0)
    
    def lse_func(u, T):
        # Softmax parametrization
        u_shift = u - u.max()
        e = np.exp(u_shift)
        pp = e / e.sum()
        
        c = cfast(pp)
        cmax = c.max()
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = exps.sum()
        f = cmax + T * math.log(sum_exp / conv_len)
        w = exps / sum_exp
        
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = pp @ gdp
        grad = pp * (gdp - tdp)
        return f, np.where(np.isfinite(grad), grad, 0.0)
    
    def p_to_u(p):
        ll = np.log(np.maximum(p, 1e-300))
        return ll - ll.mean()
    
    u = p_to_u(p)

                    p = p_cur.copy()

def _lp_direction(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns optimal b or None on failure."""
    n = a.size
    # Build constraint matrix A (2n-1 x n): A[k,j] = a[j] if 0<=k-j<n
    # This is a Toeplitz-like matrix
    # A[k, j] = a[j] * (0 <= k-j < n)
    
    # Build sparse rows
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            if a[j] > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(a[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: max sum(b) → min -sum(b)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 60})
        if res.status == 0 and res.x is not None:
            b = np.maximum(res.x, 0)
            if b.sum() > 0:
                return b
    except:
        pass
    return None

    rng = np.random.default_rng(seed)
    
    G = max(8, min(32, int(50000 / max(n,1))))  # grid points per coordinate
    
    sweep = 0
    flat = 0
    while time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            
            v = p[i]
            c_start = max(0, 0)  # c indices affected: [i, i+n-1] in the conv
            # Changing a[i] affects c[k] for k in [i, i+n-1] (since c[k] involves a[i]*a[k-i])
            ci_lo = i
            ci_hi = min(i + n - 1, 2*n-2)
            Ls = ci_hi - ci_lo + 1
            
            # Linear part: Δc[k] = w * (a[k-i] + a[i-k]) for typical k
            # Actually: ∂c[k]/∂a[i] = 2*a[k-i] if 0 ≤ k-i < n
            # So Δc[k] ≈ w * 2*a[k-i] + w^2 (if k=2i, the w^2 term)
            
            # For the grid search, we need: new_c[k] = c[k] + w*a[k-i] + w*a[i-k] + w^2*(1 if k=2i)
            # Wait, from our analysis: c[k] has two terms involving a[i]: a[i]*a[k-i] (from j=i) and a[k-i]*a[i] (from j=k-i)
            # When we change a[i] to a[i]+w:
            # new term from j=i: (a[i]+w)*a[k-i] = a[i]*a[k-i] + w*a[k-i]
            # new term from j=k-i: a[k-i]*(a[i]+w) = a[k-i]*a[i] + w*a[k-i]
            # If k≠2i: total change = 2w*a[k-i]
            # If k=2i: the single term changes from a[i]^2 to (a[i]+w)^2, change = 2w*a[i] + w^2
            
            # So: new_c[k] = c[k] + 2w*a[k-i] for k≠2i (and 0≤k-i<n)
            #       new_c[k] = c[k] + 2w*a[i] + w^2 for k=2i
            
            # Affected c indices: k such that 0≤k-i<n, i.e., i≤k≤i+n-1
            # Plus the boundary k where k-i might be negative or ≥n... 
            # Actually, the terms a[i]*a[k-i] exist for 0≤k-i<n, i.e., i≤k≤i+n-1
            # And a[k-i]*a[i] exist for 0≤k-i<n, same range.
            # But also, for k<i: the term a[k-i]... hmm, k-i < 0, so a[k-i] doesn't exist. 
            # Wait, I think I need to be more careful.
            
            # c[k] = sum_j a[j]*a[k-j] for valid j,k-j
            # a[i] appears in c[k] when:
            #   j=i: need 0≤i<n (yes) and 0≤k-i<n → i≤k≤i+n-1
            #   k-j=i: j=k-i, need 0≤k-i<n and 0≤i<n → i≤k≤i+n-1 (same)
            # So a[i] affects c[k] for k in [i, i+n-1].
            
            if Ls <= 0:
                continue
            
            # The "influence" vector: infl[k-i] = 2*a[k-i] for k in [i, i+n-1]
            # But for k=2i (if 2i < 2n-1), the coefficient is 2*a[i] (same as 2*a[k-i] since k-i=i)
            # So in all cases: coefficient is 2*a[k-i]
            infl = 2.0 * a_slice  # a_slice = a[k-i] for k in [i, i+n-1]
            # k-i ranges from 0 to n-1, so infl = 2*a[0:n] = 2*a
            
            # Wait: for k in [i, i+n-1], k-i ranges from 0 to n-1.
            # So the influence on c[k] is 2*a[k-i] = 2*a[0] to 2*a[n-1], i.e., 2*a!
            
            # Hmm, that's for the linear part. The quadratic part (w^2) only exists at k=2i.
            
            # So: new_c[k] = c[k] + 2w*a[k-i] + w^2 * δ(k=2i)  for k in [i, i+n-1]
            #     new_c[k] = c[k]  for k not in [i, i+n-1]
            
            base = c[ci_lo:ci_hi+1].copy()
            infl_vec = 2.0 * a  # since k-i ranges 0 to n-1
            
            # Check: is i+n-1 < 2n-1? i < n, so i+n-1 < 2n-1. Yes.
            # Is i >= 0? Yes.
            # So the affected range is always [i, i+n-1] with length n.
            Ls_actual = n  # always n elements affected
            
            # const_max = max of c outside the affected range
            const_max = 0.0
            if ci_lo > 0:
                const_max = max(const_max, c[:ci_lo].max())
            if ci_hi + 1 < 2*n:
                const_max = max(const_max, c[ci_hi+1:].max())
            
            # Grid search for w (change from current v)
            x_hi = max(20*v, 30.0/n, 0.01)
            x_hi = min(x_hi, 5.0)  # since S=1, a[i] can't be much > 1
            
            n_grid = G
            xs = np.linspace(0.0, x_hi, n_grid)
            # Also include current value
            xs = np.unique(np.concatenate([xs, [v]]))
            
            best_w = 0.0
            best_val = cur_score
            for x in xs:
                w = x - v
                if w == 0:
                    continue
                # new c in affected range
                new_c_aff = base + w * infl_vec + (w*w if abs(2*i - ci_lo) < n else 0)
                # Wait, the w^2 term: it's at position k=2i in the full c.
                # In the affected range [ci_lo, ci_hi], position 2i corresponds to index (2i - ci_lo) = (2i - i) = i
                diag_pos = 2*i - ci_lo  # = i
                if 0 <= diag_pos < Ls_actual:
                    new_c_aff[diag_pos] += w * w  # the extra w^2
                # Also need: new_c_aff[diag_pos] already has 2w*a[i] from the linear part
                # Actually: at k=2i, the change is (a[i]+w)^2 - a[i]^2 = 2w*a[i] + w^2
                # The linear part gives 2w*a[2i-i] = 2w*a[i]. So we add w^2 extra. ✓
                
                new_max = max(new_c_aff.max(), const_max)
                # Score (S=1): 2*n*new_max
                val = 2.0 * n * new_max
                if val < best_val:
                    best_val = val
                    best_w = w
            
            if best_w != 0 and best_val < cur_score - 1e-14:
                p[i] += best_w
                c[ci_lo:ci_hi+1] += best_w * infl_vec
                diag_pos = 2*i - ci_lo
                if 0 <= diag_pos < Ls_actual:
                    c[ci_lo + diag_pos] += best_w * best_w
                cur_score = best_val
                improved = True
        
        # Renormalize
        s_new = p.sum()
        if s_new > 0:
            p /= s_new
            c /= (s_new * s_new)
            cur_score = 2.0 * n * c.max()
        
        if not improved:
            flat += 1
            if flat >= 5:
                # Perturb
                if time.monotonic() < stop_time - 10.0:
                    p *= np.exp(rng.normal(0, 0.05, n))
                    p = np.maximum(p, 0)
                    p /= p.sum()
                    c = _conv_self(p)
                    cur_score = 2.0 * n * c.max()
                    flat = 0
                else:
                    break
        
        if time.monotonic() % 1 < 0:  # every iteration (just for structure)
            if math.isfinite(cur_score) and cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()
                if consider:
                    try: consider(best_p)
                    except: pass
        
        sweep += 1
    
    return best_p, best_score
