        m = u.max()
        e = np.exp(u - m)
        p_val = e / e.sum()

        c = irfft(F*F, conv_size)[:L]
        c = np.maximum(c, 0.0)

            val = cmax + T * np.log(np.exp((c - cmax)/T).sum() / L)

            w /= w.sum()

            w[np.argmax(c)] = 1.0
