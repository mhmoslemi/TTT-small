def _lp_step(f, stop_time):
    """LP-based direction: maximize sum(g) s.t. f*g <= M, g >= 0."""
    n = int(f.size)
    if n < 10 or time.monotonic() > stop_time - 5:
        return None
    c = _conv_self(f)
    M = float(c.max())
    if M <= 0 or not math.isfinite(M):
        return None

    # Sparse construction: for each k, the constraint is sum_j f[j]*g[k-j] <= M
    # For the LP, we need A[k,j] = f[j] if 0 <= k-j < n, else 0
    # This is a banded-like matrix. Use dense for n up to ~4096.
    if n > 5000:
        return None

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        # Build constraints using matrix-free approach for memory
        # A @ g where A[k,j] = f[j] if k-j in [0,n-1]
        # For efficiency, use the precomputed A
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f[j]

        b = np.full(2 * n - 1, M, dtype=np.float64)
        constraints = [A @ g <= b]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        solved = False
        for slv in [cp.SCS, cp.ECOS, cp.CLARABEL]:
            if time.monotonic() > stop_time - 5:
                break
            try:
                prob.solve(solver=slv, verbose=False)
                solved = True
                break
            except Exception:
                continue
        if not solved or g.value is None:
            return None
        g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        gv = float(g_val.sum())
        if gv < 1e-10 or not math.isfinite(gv):
            return None
        return g_val / gv
    except Exception:
        return None


def _block_coord(p, stop_time, seed=999, consider=None):
    """Block coordinate descent: update small blocks of 2-4 coordinates."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p, math.inf
    p /= s
    rng = np.random.default_rng(seed)
    cur_score = _score_fft(p)
    best_score = cur_score
    best_p = p.copy()
    Ltot = 2 * n - 1
    c = _conv_self(p)

    for sweep in range(200):
        if time.monotonic() > stop_time - 1.0:
            break
        improved = False
        # Random blocks of size 2 or 3
        for _ in range(min(n // 2, 5000)):
            if time.monotonic() > stop_time - 0.5:
                break
            i = int(rng.integers(0, max(1, n - 2)))
            blk_size = int(rng.integers(2, 4))
            if i + blk_size > n:
                i = n - blk_size
            idxs = list(range(i, i + blk_size))
            old_vals = p[idxs].copy()
            # Try perturbing the block
            for _trial in range(3):
                pert = rng.uniform(-0.3, 0.3, blk_size) * (0.5 + np.abs(old_vals))
                new_vals = np.maximum(old_vals + pert, 0.0)
                if np.all(new_vals <= old_vals):
                    continue
                p_test = p.copy()
                p_test[idxs] = new_vals
                st = float(p_test.sum())
                if st < 0.01:
                    continue
                p_test /= st
                sc = _score_fft(p_test)
                if math.isfinite(sc) and sc < cur_score - 1e-13:
                    p = p_test
                    cur_score = sc
                    c = _conv_self(p)
                    improved = True
                    break
        # Renormalize
        ss = float(p.sum())
        if ss > 0 and math.isfinite(ss):
            p /= ss
            c = _conv_self(p)
            cur_score = float(2.0 * n * c.max())
        if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass
        if not improved and sweep > 20:
            break

    return best_p, best_score

            if p1 is not None and p1.size == n_par and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 2: Coordinate descent (main) ===
    phase2_end = min(hard - 100.0, t0 + 0.58 * budget)
    if phase2_end > time.monotonic() + 15.0:
        try:
            work, _ = _coord_polish(work, phase2_end, max_sweeps=500000,
                                    seed=int(seed + 7777), consider=consider)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 3: LP direction + blend ===
    lp_time = min(60.0, hard - time.monotonic() - 80.0)
    if lp_time > 15.0 and np.all(np.isfinite(work)):
        g_dir = _lp_step(work, time.monotonic() + lp_time)
        if g_dir is not None and g_dir.size == n_par:
            for t_alpha in [0.3, 0.2, 0.12, 0.06]:
                blended = (1.0 - t_alpha) * work + t_alpha * g_dir
                blended = np.maximum(blended, 0.0)
                bs = float(blended.sum())
                if bs > 0:
                    blended /= bs
                    consider(blended)
            # Polish the best blend
            work = best_arr.copy()
            work = work / float(work.sum())

    # === Phase 4: Block coordinate + more 1D coordinate ===
    phase4_end = min(hard - 60.0, t0 + 0.78 * budget)
    if phase4_end > time.monotonic() + 15.0:
        # Block coordinate first
        blk_end = min(phase4_end - 30.0, time.monotonic() + 0.3 * (phase4_end - time.monotonic()))
        if blk_end > time.monotonic() + 8.0:
            try:
                work, _ = _block_coord(work, blk_end, seed=int(seed + 31415), consider=consider)
            except Exception:
                pass
        work = best_arr.copy()
        work = work / float(work.sum())
        # Then 1D coordinate
        if phase4_end > time.monotonic() + 10.0:
            try:
                work, _ = _coord_polish(work, phase4_end, max_sweeps=300000,
                                        seed=int(seed + 27182), consider=consider)
            except Exception:
                pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # === Phase 5: Fine LSE ===
    phase5_end = min(hard - 30.0, t0 + budget - 40.0)
    if phase5_end > time.monotonic() + 20.0:
        lse5_end = min(phase5_end - 25.0, time.monotonic() + 0.4 * (phase5_end - time.monotonic()))
        if lse5_end > time.monotonic() + 10.0:
            try:
                p5 = _lbfgs_optimize(work, lse5_end, consider=consider, fine_only=True)
                if p5 is not None and p5.size == n_par and np.all(np.isfinite(p5)) and p5.sum() > 0:
                    p5 = p5 / float(p5.sum())
                    consider(p5)
            except Exception:
                pass
        work = best_arr.copy()
        work = work / float(work.sum())
        if phase5_end > time.monotonic() + 10.0:
            try:
                work, _ = _coord_polish(work, phase5_end, max_sweeps=200000,
                                        seed=int(seed + 16180), consider=consider)
            except Exception:
                pass

    # === Phase 6: Try alternate length if time permits ===
    rem = hard - 20.0 - time.monotonic()
    if rem > 150.0:
        for n2 in [n_par - 512, n_par + 512]:
            if rem < 80.0:
                break
            n2 = max(256, n2)
            # Interpolate to new length
            xp = (np.arange(n_par, dtype=np.float64) + 0.5) / float(n_par)
            fp = np.interp((np.arange(n2, dtype=np.float64) + 0.5) / float(n2), xp, best_arr)
            p2 = np.maximum(fp, 0.0)
            p2s = float(p2.sum())
            if p2s <= 0:
                continue
            p2 /= p2s
            consider(p2)
            # Coordinate polish on new length
            p2_end = min(hard - 25.0, time.monotonic() + rem * 0.4)
            if p2_end > time.monotonic() + 10.0:
                try:
                    p2, _ = _coord_polish(p2, p2_end, max_sweeps=100000,
                                          seed=int(seed + 4242), consider=consider)
                except Exception:
                    pass
            rem = hard - 20.0 - time.monotonic()

    # === Phase 7: Final squeeze ===
    fin_end = min(hard - 10.0, t0 + budget - 8.0)
    if fin_end > time.monotonic() + 10.0:
        work = best_arr.copy()
        work = work / float(work.sum())
        try:
            work, _ = _coord_polish(work, fin_end, max_sweeps=150000,
                                    seed=int(seed + 5555), consider=consider)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
