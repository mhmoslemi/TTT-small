    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)

def _get_parent():
    try:
        h = globals()["height_sequence_1"]

    except Exception:

def _conv_fft(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _eval_fast(a):
    n = a.size
    s = float(a.sum())

        return math.inf
    c = _conv_fft(a)
    m = float(c.max())
    if not math.isfinite(m):

                    improved = True

            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0

        # Actually, let's just trust the incremental updates

            best_score = new_score
            best_p = p.copy()

            flat = 0
        else:

                        S = 1.0
                    c = _conv_fft(p)

                else:

def propose_candidate(seed=42, budget_s=1300, **kwargs):

        best_score = _eval_exact(best_arr)
