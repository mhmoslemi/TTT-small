def _score_fast(a):
    """Fast score using FFT. For nonnegative a, max_b = sum(a_i^2)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    q = float(np.dot(a, a))
    return float(2.0 * n * q / (s * s))


def _uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_uniform(n):
    """Create uniform sequence of length n."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _proj_toward_uniform(p, alpha):
    """Blend p towards uniform: (1-alpha)*p_normalized + alpha*(1/n)."""
    n = p.size
    sn = float(p.sum())
    if sn < 1e-300:
        return _uniform(n)
    pn = p / sn
    u = np.full(n, 1.0 / n, dtype=np.float64)
    return (1.0 - alpha) * pn + alpha * u


def _coord_optimize_uniform(p, stop_time, n_iter_max=5000):
    """
    Coordinate descent to minimize 2n*sum(a^2)/sum(a)^2.
    For nonnegative sequences this is minimized at uniform.
    We use this to verify and potentially refine.
    """
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    best_p = p.copy()
    best_sc = _score_fast(p)

    rng = np.random.default_rng(12345)

    for it in range(n_iter_max):
        if time.monotonic() > stop_time - 2.0:
            break
        # The optimal is uniform. Move towards it.
        # A simple approach: replace the max element with the mean of the rest
        # to reduce variance.
        p_sorted_idx = np.argsort(p)
        # Reduce the largest, increase the smallest
        i_max = int(p_sorted_idx[-1])
        i_min = int(p_sorted_idx[0])
        if p[i_max] - p[i_min] < 1e-15:
            break
        # Move epsilon from max to min
        delta = (p[i_max] - p[i_min]) * 0.5
        p[i_max] -= delta
        p[i_min] += delta
        sc = _score_fast(p)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_p = p.copy()

    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))

    parent = _get_parent()
    n_par = int(parent.size)

    # Compute parent's exact score
    parent_score = _score_exact(parent)

    # The theoretical minimum of the evaluator score for nonnegative sequences
    # is 2.0, achieved by uniform sequences. Let's verify and use this.
    best_arr = parent.copy()
    best_score = parent_score

    # Try uniform sequence of same length
    unif = _uniform(n_par)
    unif_score = _score_exact(unif)
    if unif_score < best_score:
        best_arr = unif.copy()
        best_score = unif_score

    # Try a few other lengths
    for n_try in [256, 512, 1024, 2048, 4096, 8192]:
        if time.monotonic() > hard_deadline - 5.0:
            break
        u = _uniform(n_try)
        sc = _score_exact(u)
        if sc < best_score:
            best_arr = u.copy()
            best_score = sc

    # Also try: maybe the parent has a better score than uniform due to
    # some numerical effect. Keep the best.
    if parent_score < best_score:
        best_arr = parent.copy()
        best_score = parent_score

    # If the parent is very close to uniform (score close to 2), it might
    # have better theoretical properties. Check if parent score is within
    # epsilon of 2.
    if parent_score < 2.0 + 1e-9:
        # Parent is essentially uniform, use it (preserves any structure)
        best_arr = parent.copy()
        best_score = parent_score

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:

            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)

        return ls

        return [1.0]
