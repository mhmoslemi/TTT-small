import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return c


def _score_fft(a):
    a = np.maximum(a, 0.0)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _score_exact_local(a):
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    m = src.size
    if m == 0:
        return np.full(n, 1.0 / n)
    if m == n:
        return src.copy()
    xp = (np.arange(m) + 0.5) / m
    fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
    out = np.maximum(fp, 0.0)
    s = float(out.sum())
    return out / s if s > 1e-300 else np.full(n, 1.0 / n)


def _lse_optimize(p_init, stop_time, seed=42, consider=None):
    p = np.maximum(np.asarray(p_init, dtype=np.float64).ravel(), 1e-300).copy()
    n = p.size
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return c

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if cmax < 1e-300:
            return cmax + 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            f_val = cmax
        else:
            f_val = cmax + T * math.log(sum_exp / conv_len)
            if not math.isfinite(f_val):
                f_val = cmax
            w = exps / sum_exp

        # Gradient via FFT cross-correlation
        W = rfft(w, cross_size)
        P = rfft(pp, cross_size)
        cross = irfft(W * np.conj(P), cross_size)[:n]
        av = float(np.dot(w, c))  # = dot(p, cross)
        grad = 2.0 * pp * (cross - av)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f_val), grad

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    start = time.monotonic()
    avail = max(0.0, stop_time - start - 5.0)
    if avail < 10.0:
        return p

    best_p = p.copy()
    best_sc = _score_fft(p)
    rng = np.random.default_rng(seed)

    # Temperature stages: (frac_of_cmax, time_weight)
    stages = [
        (0.12, 0.12),
        (0.06, 0.12),
        (0.03, 0.14),
        (0.012, 0.16),
        (0.005, 0.16),
        (0.002, 0.14),
        (0.0008, 0.16),
    ]

    total_w = sum(w for _, w in stages)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += wt
        end_stage = min(stop_time - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 3.0:
            continue

        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)

        best_u = u.copy()
        f0, _ = lse_func(best_u, T)
        if not math.isfinite(f0):
            continue
        best_f = f0
        stale = 0

        while time.monotonic() < end_stage - 1.0:
            opts = {"maxiter": 30, "gtol": 1e-10, "ftol": 1e-14,
                    "maxls": 25, "maxcor": 20, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-12:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            best_u = best_u + rng.normal(0, 0.004, best_u.size)
                            continue
            if getattr(res, "status", 1) == 0:
                if time.monotonic() < end_stage - 2.0 and stale < 3:
                    best_u = best_u + rng.normal(0, 0.002, best_u.size)
                    stale += 1
                else:
                    break

        p_cur = _softmax(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass
        u = p_to_u(p)

    return best_p


def _coord_descent(p_init, stop_time, seed=12345, consider=None):

    p = np.maximum(np.asarray(p_init, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    G = 32  # grid points
    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()

    sweep = 0
    flat = 0

    while sweep < 100000 and time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)

            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.4 or S_total < 0.65:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])

            start_k = i
            end_k = min(i + n, Ltot)
            Ls = end_k - start_k
            if Ls <= 0:
                continue

            base = c[start_k:end_k].copy()
            infl = 2.0 * p[:Ls].copy()

            max_const = 0.0
            if start_k > 0:
                ml = float(c[:start_k].max())
                if ml > max_const:
                    max_const = ml
            if end_k < Ltot:
                mr = float(c[end_k:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(10.0 * v, 20.0 * S / float(n), 0.01 * S, 1e-14)
            x_hi = min(x_hi, 6.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, S / float(n))

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag_i = i  # position within the slice where k=2i

            def eval_xs(xs):
                xs_a = np.asarray(xs, dtype=np.float64).ravel()
                w = xs_a - v
                denom = S + w
                vals = np.full(xs_a.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[valid, None] * infl[None, :]
                dval = base[diag_i] + w[valid] * infl[diag_i] + w[valid] * w[valid]
                aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                dv = denom[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Refine around the best
            if G > 4 and best_x > 0 and best_x < x_hi:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 20, dtype=np.float64)
                fine = np.concatenate((fine, [best_x]))
                fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val - 1e-14:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-12 and 0.0 <= best_x <= 8.0 * S:
                w_val = best_x - v
                if abs(w_val) > 1e-15:
                    p[i] = best_x
                    S_total += w_val
                    c[start_k:end_k] += w_val * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w_val * w_val
                    cur_score = best_val
                    improved = True

        # Renormalize
        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 6.0:
                    sigma = 0.015 + 0.01 * min(flat, 12) / 12.0
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 1e-300:
                        p = p / ss
                        c = _conv_self(p)
                        S_total = 1.0
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider is not None:
                                try:

                                except Exception:
                                    pass
                        flat = 0
                    else:
                        break
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = _score_fft(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score

    # Phase 2: Coordinate descent (25%)

    # Phase 3: Perturbation + re-optimization (12%)

                except Exception:
