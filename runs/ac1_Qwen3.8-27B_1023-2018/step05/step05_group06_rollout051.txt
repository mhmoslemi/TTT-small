import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _conv_fft(a):
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fft(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    b = _conv_fft(p)
    m = float(b.max())
    return float(2.0 * n * m / (s * s))


def _lbfgs_phase(p_in, n, stop_time, consider):
    p = np.maximum(p_in, 1e-30).copy()
    s = p.sum()
    p = p / s
    v = np.log(p)
    v -= v.mean()
    Ltot = 2 * n - 1
    fft_sz = next_fast_len(Ltot)

    def softmax(vv):
        mm = vv.max()
        e = np.exp(vv - mm)
        ss = e.sum()
        if ss <= 0 or not np.isfinite(ss):
            return np.full(n, 1.0 / n)
        return e / ss

    best_v = v.copy()
    best_exact = _score_fft(softmax(v))

    elapsed_avail = max(1.0, stop_time - time.monotonic())
    t_stages = [
        (0.06, time.monotonic() + elapsed_avail * 0.45),
        (0.025, time.monotonic() + elapsed_avail * 0.75),
        (0.008, stop_time),
    ]

    for T_frac, t_end in t_stages:
        def fg(vv, Tf=T_frac):
            a = softmax(vv)
            F = rfft(a, fft_sz)
            b = irfft(F * F, fft_sz)[:Ltot]
            np.maximum(b, 0.0, out=b)
            bmax = max(float(b.max()), 1e-30)
            T = Tf * bmax
            eb = np.exp((b - bmax) / T)
            S = float(eb.sum())
            if S <= 0 or not np.isfinite(S):
                return 1e10, np.zeros(n)
            w = eb / S
            fval = float(2.0 * n * (bmax + T * math.log(S / Ltot)))
            c = np.correlate(w, a, 'valid')
            dot_ca = float(np.dot(c, a))
            grad = 4.0 * n * a * (c - dot_ca)
            if not np.all(np.isfinite(grad)):
                grad = np.zeros(n)
            return fval, grad

        while time.monotonic() < t_end - 2.0:
            try:
                res = minimize(fg, best_v, method='L-BFGS-B', jac=True,
                               options={'maxiter': 40, 'gtol': 1e-11,
                                        'ftol': 1e-15, 'maxls': 30,
                                        'maxcor': 25, 'disp': False})
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                best_v = res.x.copy()
            else:
                break
            a_out = softmax(best_v)
            sc = _score_fft(a_out)
            if math.isfinite(sc) and sc < best_exact:
                best_exact = sc
                consider(a_out)
            if getattr(res, 'status', 1) == 0:
                break

    return softmax(best_v)


def _coord_descent(p_in, n, stop_time, consider, seed):
    p = np.maximum(p_in, 0.0).copy()
    S = float(p.sum())
    if S < 1e-30:
        return p_in
    p /= S
    S = 1.0
    Ltot = 2 * n - 1
    b = _conv_fft(p)
    cur_score = 2.0 * n * float(b.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    G = 32
    flat = 0
    sweep = 0

    while time.monotonic() < stop_time - 2.0:
        sweep += 1
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            if S <= 1e-30:
                break
            if S > 1.3 or S < 0.7:
                inv = 1.0 / S
                p *= inv
                b *= inv * inv
                S = 1.0

            v_i = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            infl = 2.0 * p[:Ls]
            diag_off = 2 * i - start

            max_out = 0.0
            if start > 0:
                mo = float(b[:start].max())
                if mo > max_out:
                    max_out = mo
            if end < Ltot:
                mo = float(b[end:].max())
                if mo > max_out:
                    max_out = mo

            base = b[start:end]

            x_hi = max(12.0 * v_i, 35.0 / n, 0.002)
            x_hi = min(x_hi, 0.06)
            if not np.isfinite(x_hi) or x_hi < 1e-8:
                x_hi = max(v_i * 2.0, 1e-6)
            x_lo = 0.0

            if x_hi <= x_lo:
                continue

            pts = np.linspace(x_lo, x_hi, G + 1)
            ws = pts - v_i
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag_off < Ls:
                aff[:, diag_off] += ws * ws
            maff = aff.max(axis=1)
            maff = np.maximum(maff, max_out)
            denoms = S + ws
            vals = np.where(denoms > 1e-15,
                            2.0 * n * np.maximum(maff, 0.0) / (denoms * denoms),
                            math.inf)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            bx = float(pts[imin])
            bv = float(vals[imin])

            if G > 4:
                dw = x_hi / G
                lo_gs = max(x_lo, bx - 2.5 * dw)
                hi_gs = min(x_hi, bx + 2.5 * dw)
                if hi_gs - lo_gs > 1e-13:
                    gr = (math.sqrt(5.0) - 1.0) / 2.0
                    a_gs, b_gs = lo_gs, hi_gs
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    d_gs = a_gs + gr * (b_gs - a_gs)

                    def ev(x):
                        w_val = x - v_i
                        dn = S + w_val
                        if dn <= 1e-15:
                            return math.inf
                        mm = float((base + w_val * infl).max())
                        if 0 <= diag_off < Ls:
                            cand = base[diag_off] + w_val * infl[diag_off] + w_val * w_val
                            if cand > mm:
                                mm = cand
                        mm = max(mm, max_out, 0.0)
                        return 2.0 * n * mm / (dn * dn)

                    f_c, f_d = ev(c_gs), ev(d_gs)
                    for _ in range(18):
                        if f_c < f_d:
                            b_gs, d_gs, f_d = d_gs, c_gs, f_c
                            c_gs = b_gs - gr * (b_gs - a_gs)
                            f_c = ev(c_gs)
                        else:
                            a_gs, c_gs, f_c = c_gs, d_gs, f_d
                            d_gs = a_gs + gr * (b_gs - a_gs)
                            f_d = ev(d_gs)
                    x_opt = 0.5 * (a_gs + b_gs)
                    f_opt = ev(x_opt)
                    if f_opt < bv:
                        bx, bv = x_opt, f_opt

            if np.isfinite(bv) and bv < cur_score - 1e-14 and bx >= 0.0:
                w_val = bx - v_i
                if abs(w_val) > 1e-17:
                    p[i] = bx
                    S += w_val
                    b[start:end] += w_val * infl
                    if 0 <= diag_off < Ls:
                        b[start + diag_off] += w_val * w_val
                    cur_score = bv
                    improved = True

        if S > 1e-30 and np.isfinite(S):
            p /= S
            b /= (S * S)
            S = 1.0

        if sweep % 4 == 0:
            b = _conv_fft(p)
            cur_score = 2.0 * n * float(b.max())

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            consider(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4 and time.monotonic() < stop_time - 60.0:
                sigma = 0.015 + 0.008 * min(flat, 12)
                noise = rng.normal(0.0, sigma, n)
                p *= np.exp(noise)
                np.maximum(p, 0.0, out=p)
                ss = float(p.sum())
                if ss > 1e-30:
                    p /= ss
                    S = 1.0
                    b = _conv_fft(p)
                    cur_score = 2.0 * n * float(b.max())
                    if cur_score < best_score:
                        best_score = cur_score
                        best_p = p.copy()
                        consider(best_p)
                flat = 0
            if flat >= 35:
                break

    np.maximum(p, 0.0, out=p)
    return best_p


def _focus_polish(p_in, n, stop_time, consider, seed):
    p = np.maximum(p_in, 0.0).copy()
    S = float(p.sum())
    p /= S
    Ltot = 2 * n - 1
    b = _conv_fft(p)
    cur_score = 2.0 * n * float(b.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed + 99)

    cmax = float(b.max())
    thresh = cmax * 0.99
    active = np.where(b >= thresh)[0]
    important = np.zeros(n, dtype=bool)
    for lag in active:
        lo = max(0, int(lag) - n + 1)
        hi = min(n - 1, int(lag))
        important[lo:hi + 1] = True
    imp_idx = np.where(important)[0]
    oth_idx = np.where(~important)[0]
    if len(imp_idx) > 400:
        imp_idx = rng.choice(imp_idx, 400, replace=False)

    G_f = 40
    sweep = 0
    flat = 0

    while time.monotonic() < stop_time - 2.0:
        sweep += 1
        improved = False
        if sweep % 4 < 3 and len(imp_idx) > 0:
            order = rng.permutation(imp_idx)
        elif len(oth_idx) > 0:
            order = rng.permutation(oth_idx)
        else:
            order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            if S <= 1e-30:
                break
            if S > 1.2 or S < 0.83:
                p /= S

                S = 1.0

            v_i = float(p[i])

            Ls = end - start

            infl = 2.0 * p[:Ls]

            max_out = 0.0

                if mo > max_out:

            if end < Ltot:

                if mo > max_out:

            base = b[start:end]

            x_hi = max(6.0 * v_i, 15.0 / n, 0.001)
            x_hi = min(x_hi, v_i + 0.008, 0.03)
            x_lo = max(0.0, v_i - 0.004)
            if x_hi <= x_lo:
                x_lo = 0.0
                x_hi = max(v_i, 1e-4) + 0.002

            pts = np.linspace(x_lo, x_hi, G_f + 1)
            ws = pts - v_i
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag_off < Ls:

            maff = np.maximum(aff.max(axis=1), max_out)
            denoms = S + ws
            vals = np.where(denoms > 1e-15,
                            2.0 * n * np.maximum(maff, 0.0) / (denoms * denoms),
                            math.inf)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            bx = float(pts[imin])
            bv = float(vals[imin])

            if G_f > 4:
                dw = (x_hi - x_lo) / G_f
                lo_gs = max(x_lo, bx - 1.5 * dw)
                hi_gs = min(x_hi, bx + 1.5 * dw)
                if hi_gs - lo_gs > 1e-13:
                    gr = (math.sqrt(5.0) - 1.0) / 2.0
                    a_gs, b_gs = lo_gs, hi_gs
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    d_gs = a_gs + gr * (b_gs - a_gs)

                    def ev(x):
                        w_val = x - v_i
                        dn = S + w_val
                        if dn <= 1e-15:
                            return math.inf
                        mm = float((base + w_val * infl).max())
                        if 0 <= diag_off < Ls:

                            if cand > mm:
                                mm = cand
                        mm = max(mm, max_out, 0.0)
                        return 2.0 * n * mm / (dn * dn)

                    f_c, f_d = ev(c_gs), ev(d_gs)
                    for _ in range(14):
                        if f_c < f_d:
                            b_gs, d_gs, f_d = d_gs, c_gs, f_c

                            a_gs, c_gs, f_c = c_gs, d_gs, f_d

                    x_opt = 0.5 * (a_gs + b_gs)

def propose_candidate(seed=42, budget_s=1300, **kwargs):

    except Exception:
        p = np.ones(3072, dtype=np.float64)

    n = len(p)
    s = float(p.sum())
    if s < 1e-12:
        p = np.ones(n)
        s = float(n)
    p = p / s

    best_p = p.copy()
    best_score = _score_fft(p)

    def consider(arr):
        nonlocal best_p, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.clip(a, 0.0, 1000.0)
            a = np.where(np.isfinite(a), a, 0.0)
            sa = float(a.sum())
            if sa < 0.01:
                return
            a = a / sa
            sc = _score_fft(a)
            if np.isfinite(sc) and sc < best_score - 1e-14:
                best_p = a.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 1: L-BFGS warm-up (~150s)
    lbfgs_stop = min(deadline - 1050.0, t0 + 150.0)
    if lbfgs_stop > time.monotonic() + 40.0:
        try:
            p2 = _lbfgs_phase(p, n, lbfgs_stop, consider)
            if p2 is not None and len(p2) == n and np.all(np.isfinite(p2)) and float(p2.sum()) > 0.01:
                p2 = p2 / float(p2.sum())
                consider(p2)
        except Exception:
            pass
        if _score_fft(p) > best_score + 0.005:
            p = best_p.copy()
        else:
            p = p.copy()

    # Phase 2: Main coordinate descent
    coord_stop = min(deadline - 120.0, t0 + 1150.0)
    if coord_stop > time.monotonic() + 40.0:
        try:
            p3 = _coord_descent(p, n, coord_stop, consider, seed + 111)

        except Exception:

    # Phase 3: Focus polish

    if polish_stop > time.monotonic() + 30.0:
        try:
            p4 = _focus_polish(best_p, n, polish_stop, consider, seed + 222)
            if p4 is not None and len(p4) == n and np.all(np.isfinite(p4)):

    # Phase 4: One more coord pass if time remains
    if time.monotonic() < deadline - 40.0:
        try:
            p5 = _coord_descent(best_p, n, deadline - 20.0, consider, seed + 555)
            if p5 is not None and len(p5) == n and np.all(np.isfinite(p5)):

        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        out = np.asarray(best_p, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(128, 1.0 / 128.0)
        ls = [float(x) for x in out]
        fs = evaluate_sequence(ls)
        if not np.isfinite(fs):
            try:
                pp = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()

                fs2 = evaluate_sequence(ls2)
                if np.isfinite(fs2):
                    return ls2
            except Exception:
                pass
            return [1.0]
        return ls
    except Exception:
        return [1.0]
