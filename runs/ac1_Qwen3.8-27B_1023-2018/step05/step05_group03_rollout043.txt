For the LSE optimization, the key idea is:
- We want to minimize `max_k b_k(a)` where `b = conv(a, a)`
- We approximate this with `F_T(a) = T * log(sum_k exp(b_k / T))` (log-sum-exp)
- The gradient of `F_T` is: `∇F_T = (1/sum_k exp(b_k/T)) * sum_k exp(b_k/T) * ∇b_k`

    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _softmax(u):
    u = np.asarray(u, dtype=np.float64).ravel()
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        u = np.where(np.isfinite(u), u, 0.0)
    m = float(u.max())
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    return e / s


def _lse_optimize(p, stop_time, consider=None):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n <= 1 or time.monotonic() > stop_time:
        return p

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss <= 1e-300:
        p = np.full(n, 1.0 / n)
    else:
        p = p / ss

    conv_len = 2 * n - 1
    conv_size = next_fast_len(conv_len)
    cross_len = 3 * n - 2
    cross_size = next_fast_len(cross_len)

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        return c[n - 1:2 * n - 1]

    def p_to_u(pp):
        pp = np.maximum(pp, 1e-15)
        uu = np.log(pp, dtype=np.float64)
        uu -= float(uu.mean())
        return uu

    def lse_func(uu, T):
        uu = np.asarray(uu, dtype=np.float64).ravel()
        if uu.size != n:
            return 1e100, np.zeros(n)
        if not np.all(np.isfinite(uu)):
            uu = np.where(np.isfinite(uu), uu, 0.0)
        m = float(uu.max())
        e = np.exp(uu - m, dtype=np.float64)
        s = float(e.sum())
        if not math.isfinite(s) or s <= 0:
            pp = np.full(n, 1.0 / n)
        else:
            pp = e / s
        pp = np.maximum(pp, 0.0)
        ss = float(pp.sum())
        if ss <= 1e-300:
            pp = np.full(n, 1.0 / n)
        else:
            pp = pp / ss

        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros(n)

        if not math.isfinite(T) or T <= 0:
            T = max(1e-14, 1e-4 * cmax)
        T = max(1e-14, T)

        scaled = (c - cmax) / T
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            w = exps / sum_exp
            f = cmax + T * math.log(sum_exp / conv_len)
            if not math.isfinite(f):
                f = cmax

        q = xfast(pp, w)
        gid = 2.0 * q
        gd = float(np.dot(pp, gid))
        grad = pp * (gid - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score_fft(p)
    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass

    stages = (
        (0.015, 0.25),
        (0.005, 0.25),
        (0.0015, 0.25),
        (0.0004, 0.25),
    )

    start = time.monotonic()
    avail = max(0.0, stop_time - start - 3.0)
    if avail <= 5.0:
        return best_p

    total_w = sum(wt for _, wt in stages)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        cum += wt
        end_stage = min(stop_time - 2.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue

        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-14, frac * cmax)

        maxit = 20 if n > 2000 else 30
        opts = {
            "maxiter": maxit,
            "gtol": 1e-10,
            "ftol": 1e-15,
            "maxls": 10,
            "maxcor": 12,
            "disp": False,
            "maxdisp": 0,
        }
        try:
            res = minimize(lse_func, u, args=(T,), method="L-BFGS-B",
                           jac=True, options=opts)
            if res.x is not None and np.all(np.isfinite(res.x)):
                p_new = _softmax(res.x)
                p_new = np.maximum(p_new, 0.0)
                sn = float(p_new.sum())
                if sn > 1e-300:
                    p_new /= sn
                sc_new = _score_fft(p_new)
                if math.isfinite(sc_new) and sc_new < best_sc - 1e-14:
                    best_sc = sc_new
                    best_p = p_new.copy()
                    p = p_new.copy()
                    if consider is not None:
                        try:
                            consider(best_p)
                        except Exception:
                            pass
        except Exception:
            pass

        u = p_to_u(p)

    if best_p.size == n:
        return best_p
    return p


def _coord_polish(p, stop_time, G=48, seed=12345, consider=None):
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    if n <= 1:
        return p, math.inf

    p = np.maximum(p, 0.0)
    S = float(p.sum())
    if S <= 1e-300:
        p = np.full(n, 1.0 / n)
        S = 1.0
    else:
        p = p / S
        S = 1.0

    c = _fft_conv(p)
    c = np.maximum(c, 0.0)
    infl = 2.0 * p
    full_len = 2 * n - 1
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass

    rng = np.random.default_rng(seed)
    flat = 0
    sweep = 0

    while time.monotonic() < stop_time - 0.5:
        improved = False
        order = rng.permutation(n)

        for pos in range(n):
            if time.monotonic() > stop_time - 0.3:
                break
            if S <= 1e-300:
                break

            i = int(order[pos])
            v = float(p[i])
            start = i
            end = min(i + n, full_len)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end]
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < full_len:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

            di = 2 * i - start  # = i

            # Determine if the max is in the affected range
            base_max = float(base.max())
            max_in_range = base_max >= max_const * 0.95

            # Build candidate points
            if not max_in_range:
                # Max is outside; just try a few values
                pts = np.array([0.0, v, 0.5 * v, 2.0 * v, S / n], dtype=np.float64)
                pts = np.unique(np.clip(pts, 0.0, max(10.0 * S / n, v * 10)))
            else:
                # Max is in range; use adaptive grid
                npts = G if v > 1e-10 else max(12, G // 2)
                lo = max(0.0, v - 3.0 * max(v, S / n))
                hi = max(5.0 * v, 5.0 * S / n, 0.001)
                hi = min(hi, 50.0 * S / n)
                if lo < hi:
                    n_geo = max(8, npts * 2 // 3)
                    n_lin = max(4, npts // 3)
                    pts_geo = np.geomspace(max(lo, 1e-15), hi, n_geo) if lo > 0 else np.linspace(0, hi, n_geo)
                    pts_lin = np.linspace(lo, hi, n_lin)
                    pts = np.unique(np.concatenate([pts_geo, pts_lin]))
                else:
                    pts = np.array([v])

            if pts.size == 0:
                pts = np.array([v])

            # Evaluate all candidates
            w = pts - v
            den = S + w
            valid = den > 1e-14
            if not np.any(valid):
                continue

            aff = base[None, :] + w[:, None] * infl[None, :]
            if 0 <= di < Ls:
                aff[:, di] = base[di] + w * infl[di] + w * w
            np.maximum(aff, max_const, out=aff)
            mv = aff.max(axis=1)

            vals = np.full(pts.size, math.inf, dtype=np.float64)
            vals[valid] = (2.0 * n * mv[valid]) / (den[valid] * den[valid])
            vals = np.where(np.isfinite(vals), vals, math.inf)

            if not np.any(np.isfinite(vals)):
                continue

            im = int(np.argmin(vals))
            best_x = float(pts[im])
            best_val = float(vals[im])

            # Fine refinement around best
            if best_val < cur_score - 1e-13:
                step = max(v, S / n, 1e-10)
                flo = max(0.0, best_x - 2.0 * step)
                fhi = min(hi, best_x + 2.0 * step)
                if fhi > flo:
                    fine = np.linspace(flo, fhi, 24)
                    if flo > 0:
                        fine = np.unique(np.concatenate([fine, np.geomspace(flo, fhi, 10)]))
                    wf = fine - v
                    denf = S + wf
                    vf = denf > 1e-14
                    if np.any(vf):
                        afff = base[None, :] + wf[:, None] * infl[None, :]
                        if 0 <= di < Ls:
                            afff[:, di] = base[di] + wf * infl[di] + wf * wf
                        np.maximum(afff, max_const, out=afff)
                        mvf = afff.max(axis=1)
                        valsf = np.full(fine.size, math.inf)
                        valsf[vf] = (2.0 * n * mvf[vf]) / (denf[vf] * denf[vf])
                        if np.any(np.isfinite(valsf)):
                            imf = int(np.argmin(valsf))
                            if valsf[imf] < best_val:
                                best_x = float(fine[imf])
                                best_val = float(valsf[imf])

            if best_val < cur_score - 1e-13 and best_x >= 0.0:
                d = best_x - v
                if abs(d) > 1e-16:
                    c[start:end] += d * infl
                    diag_pos = 2 * i
                    if 0 <= diag_pos < full_len:
                        c[diag_pos] += d * d
                    p[i] = best_x
                    infl[i] = 2.0 * best_x
                    S += d
                    cur_score = best_val
                    improved = True

        # Periodic renormalization
        if S > 1.15 or S < 0.85:
            inv = 1.0 / S
            p *= inv
            c = c * (inv * inv)
            infl *= inv
            S = 1.0

        # Recompute exact convolution periodically to avoid drift
        if sweep % 50 == 0 and sweep > 0:
            c = _fft_conv(p)
            c = np.maximum(c, 0.0)
            infl = 2.0 * p
            S = float(p.sum())
            cur_score = float(2.0 * n * c.max())

        desc = float(2.0 * n * c.max())
        if desc < best_score - 1e-14:
            best_score = desc
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                # Perturbation to escape local minimum
                if time.monotonic() < stop_time - 30.0:
                    sigma = 0.05 + 0.05 * min(flat - 3, 4)
                    noise = rng.normal(0.0, sigma, n)
                    q = p * np.exp(noise)
                    q = np.maximum(q, 0.0)
                    qs = float(q.sum())
                    if math.isfinite(qs) and qs > 0:
                        p = q / qs
                        c = _fft_conv(p)
                        c = np.maximum(c, 0.0)
                        infl = 2.0 * p
                        S = 1.0
                        cur_score = float(2.0 * n * c.max())
                        flat = 0
                else:
                    break
        sweep += 1

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(20.0, budget)
    reserve = 20.0 if budget > 200 else 5.0
    hard = t0 + budget - reserve

    parent = _get_parent()
    n = int(parent.size) if parent.size > 0 else 3072

    # Initialize from parent
    best_arr = parent.copy()
    best_arr = np.maximum(best_arr, 0.0)
    ss = float(best_arr.sum())
    if ss > 1e-12:
        best_arr = best_arr / ss
    best_score = _score_fft(best_arr)

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return
        if arr.size != n:
            return
        arr = np.maximum(arr, 0.0)
        s = float(arr.sum())
        if not math.isfinite(s) or s < 0.01:
            return
        arr = arr / s
        try:
            sc = _score_fft(arr)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_arr = arr.copy()
            best_score = sc

    rng = np.random.default_rng(seed)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n

    # Phase 1: Quick exploration (~80s max)
    explore_end = min(hard - 100.0, t0 + 80.0)

    # Power perturbations
    for q in [0.7, 0.85, 1.0, 1.15, 1.4, 1.8]:
        if time.monotonic() > explore_end - 20:
            break
        cand = np.power(np.maximum(best_arr, 1e-15), q)
        cs = float(cand.sum())
        if cs > 0:
            cand /= cs
            consider(cand)

    # Linear and quadratic shape perturbations
    for amp in [0.3, 0.6, 1.0, 1.5]:
        if time.monotonic() > explore_end - 20:
            break
        for f in [1, 2, 3]:
            for ph in [0.0, math.pi / 2, math.pi]:
                cand = best_arr * np.exp(amp * np.sin(2 * math.pi * f * x + ph))
                cand = np.maximum(cand, 0.0)
                cs = float(cand.sum())
                if cs > 0:
                    cand /= cs
                    consider(cand)

    # Asymmetric perturbations
    for amp in [0.4, 0.8, 1.2, 2.0]:
        if time.monotonic() > explore_end - 10:
            break
        for side in [1, -1]:
            cand = best_arr * np.exp(side * amp * (2 * x - 1))
            cand = np.maximum(cand, 0.0)
            cs = float(cand.sum())
            if cs > 0:
                cand /= cs
                consider(cand)

    # Phase 2: LSE optimization
    lse_stop = min(hard - 200.0, time.monotonic() + max(30.0, 0.25 * (hard - time.monotonic())))
    if lse_stop > time.monotonic() + 15.0:
        try:
            best_arr = _lse_optimize(best_arr, lse_stop, consider=consider)
        except Exception:
            pass

    # Phase 3: Coordinate descent (bulk of remaining time)
    coord_stop = min(hard - 5.0, time.monotonic() + max(30.0, hard - time.monotonic() - 5.0))
    if coord_stop > time.monotonic() + 10.0:
        try:
            best_arr, _ = _coord_polish(best_arr, coord_stop, G=56, seed=seed + 7, consider=consider)
        except Exception:
            pass

    # Phase 4: Final LSE squeeze if time remains
    if time.monotonic() < hard - 60.0:
        lse2_stop = min(hard - 30.0, time.monotonic() + 40.0)
        if lse2_stop > time.monotonic() + 10.0:
            try:
                best_arr = _lse_optimize(best_arr, lse2_stop, consider=consider)
            except Exception:
                pass

    # Phase 5: Final coordinate polish
    if time.monotonic() < hard - 10.0:
        final_stop = min(hard - 3.0, time.monotonic() + max(10.0, hard - time.monotonic() - 3.0))
        if final_stop > time.monotonic() + 5.0:
            try:
                best_arr, _ = _coord_polish(best_arr, final_stop, G=48, seed=seed + 999, consider=consider)
            except Exception:
                pass

    # Final validation and return
    out = np.asarray(best_arr, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if out.size == 0 or s < 0.01:
        out = np.full(n, 1.0 / n, dtype=np.float64)
        s = float(out.sum())
    if s >= 0.01:
        out = out / s
    ls = [float(v) for v in out]

    try:
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            fb = np.full(n, 1.0 / n, dtype=np.float64)
            return [float(v) for v in fb]
    except Exception:
        pass

    return ls
