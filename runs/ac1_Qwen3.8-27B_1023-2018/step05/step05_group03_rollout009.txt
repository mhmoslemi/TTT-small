            pass

    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    max_flat = 6

    while time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)

        for pos in range(n):
            if time.monotonic() > stop_time - 0.3:
                break
            if S <= 1e-300:
                break
            if S > 1.15 or S < 0.85:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                infl *= inv
                S = 1.0
                c = np.maximum(c, 0.0)
                cur_score = float(2.0 * n * c.max())

            i = int(order[pos])
            v = float(p[i])
            start = i
            end = min(i + n, L)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end]
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < L:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc
            if not math.isfinite(max_const):
                max_const = 0.0

            Sloc = float(S)
            xhi = max(25.0 * v, 25.0 * Sloc / float(n), 0.006)
            xhi = min(xhi, max(0.08, 1.5 * v))
            if not math.isfinite(xhi) or xhi <= 0:
                xhi = max(1e-8, 15.0 * Sloc / float(n))

            pts = [0.0, v, xhi]

            m1 = max(14, int(G * 0.4))
            minx = max(1e-9, 5e-6 * v, Sloc / (3000.0 * float(n)))
            if minx < xhi:
                pts.extend(np.geomspace(minx, xhi, m1).tolist())

            if v > 1e-12:
                m2 = max(10, int(G * 0.3))
                factors = np.geomspace(0.15, 6.5, m2)
                pts.extend((v * factors).tolist())

            m3 = max(12, int(G * 0.3))
            linhi = min(xhi, max(10.0 * Sloc / float(n), 0.002))
            if linhi > 0:
                pts.extend(np.linspace(0.0, linhi, m3).tolist())

            pts = np.unique(np.clip(np.asarray(pts, dtype=np.float64), 0.0, xhi))
            if pts.size == 0:
                pts = np.array([v])

            di = 2 * i - start
            w = pts - v
            den = Sloc + w
            valid = den > 1e-14
            aff = base[None, :] + w[:, None] * infl[None, :start:end]
            if 0 <= di < aff.shape[1]:
                aff[:, di] = base[di] + w * infl[i] + w * w
            np.maximum(aff, max_const, out=aff)
            mv = np.max(aff, axis=1)
            vals = np.full(pts.size, math.inf)
            if np.any(valid):
                vals[valid] = (2.0 * n * mv[valid]) / (den[valid] * den[valid])
            vals = np.where(np.isfinite(vals), vals, math.inf)

            if not np.any(np.isfinite(vals)):
                continue
            im = int(np.argmin(vals))
            best_x = float(pts[im])
            best_val = float(vals[im])

            if best_val < cur_score + max(1e-10, 1e-5 * cur_score):
                if best_x <= 0.0:
                    lo = 0.0
                    hi = max(6.0 * Sloc / float(n), 6.0 * v, 1e-7)
                else:
                    step = max(v, Sloc / float(n), 1e-9)
                    lo = max(0.0, best_x * 0.4, best_x - 3.0 * step)
                    hi = min(xhi, max(best_x * 1.6, best_x + 3.0 * step))
                if hi <= lo:
                    hi = lo + 1e-9
                fine = [best_x]
                fine.extend(np.linspace(lo, hi, 44).tolist())
                if lo > 0:
                    fine.extend(np.geomspace(lo, hi, 18).tolist())
                else:
                    fine.extend(np.linspace(0.0, hi, 18).tolist())
                fine = np.unique(np.clip(np.asarray(fine, dtype=np.float64), 0.0, xhi))
                w2 = fine - v
                den2 = Sloc + w2
                aff2 = base[None, :] + w2[:, None] * infl[None, :start:end]
                if 0 <= di < aff2.shape[1]:
                    aff2[:, di] = base[di] + w2 * infl[i] + w2 * w2
                np.maximum(aff2, max_const, out=aff2)
                mv2 = np.max(aff2, axis=1)
                vals2 = np.full(fine.size, math.inf)
                valid2 = den2 > 1e-14
                if np.any(valid2):
                    vals2[valid2] = (2.0 * n * mv2[valid2]) / (den2[valid2] * den2[valid2])
                vals2 = np.where(np.isfinite(vals2), vals2, math.inf)
                if np.any(np.isfinite(vals2)):
                    im2 = int(np.argmin(vals2))
                    if vals2[im2] < best_val:
                        best_x = float(fine[im2])
                        best_val = float(vals2[im2])

            if (math.isfinite(best_val) and best_val < cur_score - 1e-11
                    and 0.0 <= best_x <= xhi):
                d = best_x - v
                if abs(d) > 1e-16:
                    c[start:end] += d * infl[:n]
                    diag = 2 * i
                    if 0 <= diag < L:
                        c[diag] += d * d
                    p[i] = best_x
                    infl[i] = 2.0 * best_x
                    S += d
                    cur_score = best_val
                    improved = True

        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            infl *= inv
            S = 1.0
        else:
            break
        c = np.maximum(c, 0.0)
        desc = float(2.0 * n * c.max())
        if math.isfinite(desc) and desc < best_score - 1e-14:
            best_score = desc
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= max_flat:
                if time.monotonic() < stop_time - 30.0:
                    sigmas = (0.03, 0.06, 0.10, 0.15, 0.22)
                    sigma = sigmas[min(flat - max_flat, len(sigmas) - 1)]
                    noise = rng.normal(0.0, sigma, n)
                    q = p * np.exp(noise)
                    if sigma >= 0.10:
                        x = (np.arange(n) + 0.5) / float(n)
                        amp = 0.15 + 0.4 * float(rng.random())
                        f1 = 1 + int(rng.integers(0, 6))
                        ph1 = 2.0 * math.pi * float(rng.random())
                        q *= np.exp(amp * np.sin(2.0 * math.pi * f1 * x + ph1))
                    q = np.maximum(q, 0.0)
                    qs = float(q.sum())
                    if math.isfinite(qs) and qs > 0:
                        p = q / qs
                        c = _fft_conv(p)
                        S = 1.0
                        infl = 2.0 * p
                        cur_score = float(2.0 * n * c.max())
                        if cur_score < best_score - 1e-14:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider is not None:
                                try:
                                    consider(best_p)
                                except Exception:
                                    pass
                    flat = 0
                else:
                    break
        sweep += 1

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p, best_score


def _top_lag_refine(p, stop_time, consider=None, top_k=5):
    """Target the top-k convolution lags for reduction."""
    p = _normalize_nonneg(p)
    n = int(p.size)
    if n <= 1 or time.monotonic() > stop_time:
        return p

    c = _fft_conv(p)
    L = c.size
    best_p = p.copy()
    best_score = float(2.0 * n * c.max())
    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass

    # Find top-k lags and try reducing them via coordinate adjustments
    rng = np.random.default_rng(999)
    for inner_iter in range(200):
        if time.monotonic() > stop_time - 1.0:
            break
        c = _fft_conv(p)
        cmax = float(c.max())
        # Find lags near the max
        threshold = cmax * 0.995
        active_lags = np.where(c >= threshold)[0]
        if active_lags.size == 0:
            active_lags = np.array([int(np.argmax(c))])

        # For each active lag, find which coordinates most affect it
        # and try to reduce them
        improved = False
        for lag in active_lags[:top_k]:
            lag_i = int(lag)
            # c[lag_i] involves p[j]*p[lag_i-j] for valid j
            # The "influence" of coordinate i on c[lag_i] is p[lag_i-i] (if valid)
            # To reduce c[lag_i], decrease p[i] where p[lag_i-i] is large
            j_vals = np.arange(max(0, lag_i - n + 1), min(n, lag_i + 1))
            j_partners = lag_i - j_vals
            for jj in range(len(j_vals)):
                i = int(j_vals[jj])
                partner = int(j_partners[jj])
                if partner < 0 or partner >= n:
                    continue
                # Current contribution to l
                contrib = p[i] * p[partner]
                if contrib < 1e-12:
                    continue
                # Try reducing p[i] slightly
                for frac in (0.85, 0.7, 0.5):
                    new_val = p[i] * frac
                    p[i] = new_val
                    p_sum = float(p.sum())
                    if p_sum < 0.01:
                        p[i] = contrib / p[partner] if p[partner] > 0 else 0
                        break
                    p_norm = p / p_sum
                    c_new = _fft_conv(p_norm)
                    sc_new = float(2.0 * n * c_new.max())
                    if math.isfinite(sc_new) and sc_new < best_score - 1e-13:
                        best_score = sc_new
                        best_p = p_norm.copy()
                        improved = True
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    else:
                        break
                p = p_norm if improved else p / p_sum

        if not improved:
            # Small random perturbation
            eps = 0.02
            noise = rng.normal(0, eps, n)
            q = np.maximum(p * np.exp(noise), 0)
            qs = float(q.sum())
            if qs > 0:
                p = q / qs
            if inner_iter > 30:
                break

    p = best_p
    if consider is not None:
        try:
            consider(p)
        except Exception:
            pass
    return p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(10.0, budget)
    reserve = 22.0
    hard = t0 + budget - reserve
    if hard <= t0 + 2.0:
        hard = t0 + max(1.5, budget - 0.5)

    parent = _get_parent()
    n_target = int(parent.size) if parent.size > 0 else 3072

    best_arr = _normalize_nonneg(parent.copy())
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.full(64, 1.0 / 64.0)
        best_score = float(_score_exact_arr(best_arr))

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return
        if arr.size == 0:
            return
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if not math.isfinite(s) or s < 0.01:
            return
        arr = arr / s
        try:
            sc = float(_score_exact_arr(arr))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-13:
            best_arr = arr.copy()
            best_score = sc

    # Quick structured perturbations
    rng_top = np.random.default_rng(seed)
    base = best_arr.copy()
    explored = [base.copy()]

    # Power transforms
    for q in (0.5, 0.7, 1.3, 1.6):
        if time.monotonic() > hard - 600:
            break
        cand = np.power(np.maximum(base, 1e-15), float(q))
        cand = _normalize_nonneg(cand)
        consider(cand)

    # Sinusoidal modulations
    for _ in range(8):
        if time.monotonic() > hard - 500:
            break
        amp = 0.2 + 0.6 * float(rng_top.random())
        f = 1 + int(rng_top.integers(0, 8))
        ph = 2.0 * math.pi * float(rng_top.random())
        x = (np.arange(n_target) + 0.5) / float(n_target)
        cand = base * np.exp(amp * np.sin(2.0 * math.pi * f * x + ph))
        cand = _normalize_nonneg(cand)
        consider(cand)
        if cand.size == n_target:
            explored.append(cand.copy())

    # Exponential tilts
    for _ in range(4):
        if time.monotonic() > hard - 450:
            break
        side = 1.0 if rng_top.random() < 0.5 else -1.0
        amp = 0.3 + 1.2 * float(rng_top.random())
        x = (np.arange(n_target) + 0.5) / float(n_target)
        cand = base * np.exp(side * amp * (2.0 * x - 1.0))
        cand = _normalize_nonneg(cand)
        consider(cand)

    # Main L-BFGS phase
    now = time.monotonic()
    lse_stop = min(hard - 300.0, now + 120.0)
    if lse_stop > now + 15.0:
        try:
            p_lse = _lbfgs_optimize(best_arr.copy(), lse_stop, consider=consider)
            consider(p_lse)
        except Exception:
            pass

    # First coordinate polish phase
    now = time.monotonic()
    cp1_stop = min(hard - 200.0, now + 400.0)
    if cp1_stop > now + 20.0:
        try:
            p_cp1, _ = _coord_polish(best_arr.copy(), cp1_stop, G=64,
                                     seed=seed + 111, consider=consider)
            consider(p_cp1)
        except Exception:
            pass

    # Second L-BFGS phase (after coordinate polish)
    now = time.monotonic()
    lse2_stop = min(hard - 150.0, now + 80.0)
    if lse2_stop > now + 10.0:
        try:
            p_lse2 = _lbfgs_optimize(best_arr.copy(), lse2_stop, consider=consider)
            consider(p_lse2)
        except Exception:
            pass

    # Top-lag refinement
    now = time.monotonic()
    tl_stop = min(hard - 100.0, now + 120.0)
    if tl_stop > now + 10.0:
        try:
            p_tl = _top_lag_refine(best_arr.copy(), tl_stop, consider=consider, top_k=6)
            consider(p_tl)
        except Exception:
            pass

    # Long coordinate polish (main workhorse)
    now = time.monotonic()
    cp2_stop = min(hard - 40.0, now + max(60.0, hard - 50.0 - now))
    if cp2_stop > now + 15.0:
        try:
            p_cp2, _ = _coord_polish(best_arr.copy(), cp2_stop, G=88,
                                     seed=seed + 4242, consider=consider)
            consider(p_cp2)
        except Exception:
            pass

    # Final L-BFGS squeeze
    now = time.monotonic()
    lse3_stop = min(hard - 15.0, now + 40.0)
    if lse3_stop > now + 8.0:
        try:
            p_lse3 = _lbfgs_optimize(best_arr.copy(), lse3_stop, consider=consider)
            consider(p_lse3)
        except Exception:
            pass

    # Final coord polish squeeze
    now = time.monotonic()
    cp3_stop = min(hard - 5.0, now + 30.0)
    if cp3_stop > now + 5.0:
        try:
            p_cp3, _ = _coord_polish(best_arr.copy(), cp3_stop, G=96,
                                     seed=seed + 7777, consider=consider)
            consider(p_cp3)
        except Exception:
            pass

    # Final return
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = _normalize_nonneg(parent)
            s = float(out.sum())
        if out.size == 0 or not math.isfinite(s) or s < 0.01:
            out = np.full(64, 1.0 / 64.0)
        out = out / float(out.sum())
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            fallback = _normalize_nonneg(parent)
            ls2 = [float(x) for x in fallback]
            sc2 = float(evaluate_sequence(ls2))
            if math.isfinite(sc2):
                return ls2
            return [1.0]
        return ls
    except Exception:
        return [1.0]
