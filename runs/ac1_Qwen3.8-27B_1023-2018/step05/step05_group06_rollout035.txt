```python
def _exact_score(p):
    """Exact score matching evaluator."""
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)

def _lp_average_step(f, M, solver='CLARABEL'):
    """
    Solve the LP:
      max sum(b)
      s.t. b >= 0
           (f*b)[k] <= M for all k
    Returns b (normalized to sum = sum(f)).
    """

    b_var = cp.Variable(n, nonneg=True)

    # (f*b)[k] = sum_j f[k-j] * b[j] for valid j

    # For memory efficiency, build as sparse
    import scipy.sparse as sp
    
    # Build row by row
    rows = []
    cols = []
    vals = []

        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo > j_hi:
            continue

        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

        except:
            try:
                prob.solve(solver='CVXOPT')
            except:

    b = np.asarray(b_var.value, dtype=np.float64).ravel()
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s < 1e-15:
        return None
    return b

    new_f = (1.0 - t) * f + t * g
```

```python
    # That's: row indices = j + [0, 1, ..., n-1] for j = 0, 1, ..., n-1
    # = all pairs (j, j+k) for j in [0,n-1], k in [0,n-1]
    # = j + k for j in [0,n-1], k in [0,n-1]
    # So rows = j+k for all j,k in [0,n-1]
    # = [0, 1, ..., n-1, 1, 2, ..., n, ..., n-1, n, ..., 2n-2]
    j_idx = np.arange(n)[:, None]  # n x 1
    k_idx = np.arange(n)[None, :]  # 1 x n
    rows = (j_idx + k_idx).ravel()  # n*n entries
    cols = np.tile(np.arange(n), n)  # [0,0,...,0, 1,1,...,1, ..., n-1,...,n-1]
    vals = np.tile(f, n)  # [f[0],...,f[n-1], f[0],...,f[n-1], ...]
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy import sparse as sp

t0_global = 0.0

def _conv_self(a):
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _fft_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(p)
    return float(2.0 * n * c.max() / (s * s))

def _build_A(f):
    """Build constraint matrix for LP."""
    n = len(f)
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(n)[None, :]
    rows = (j_idx + k_idx).ravel()
    cols = np.tile(np.arange(n), n)
    vals = np.tile(f, n)
    return sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def _solve_lp(f, M, solver_list):
    """Solve the LP: max sum(b) s.t. b>=0, A@b <= M."""
    import cvxpy as cp
    n = len(f)
    A = _build_A(f)
    b_var = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(b_var))
    constraints = [A @ b_var <= M]
    prob = cp.Problem(objective, constraints)
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if b_var.value is not None:
                b = np.asarray(b_var.value, dtype=np.float64).ravel()
                b = np.maximum(b, 0.0)
                if float(b.sum()) > 1e-15 and np.all(np.isfinite(b)):
                    return b
        except Exception:
            continue
    return None

def _lp_step(f, t=0.4, solver_list=None):
    """One LP averaging step."""
    if solver_list is None:
        solver_list = ['GUROBI', 'MOSEK', 'ECOS', 'CVXOPT', 'SCS', 'CLARABEL']
    n = len(f)
    c = _conv_self(f)
    M = float(c.max())
    if not math.isfinite(M) or M <= 0:
        return f.copy()
    
    b = _solve_lp(f, M, solver_list)
    if b is None:
        return f.copy()
    
    S_f = float(f.sum())
    S_b = float(b.sum())
    g = S_f * b / S_b
    new_f = (1.0 - t) * f + t * g
    return np.maximum(new_f, 0.0)

def _coord_descent(p, stop_time, consider_callback, seed=42, G=40):
    """Coordinate descent with vectorized grid + golden section."""
    n = len(p)
    p = np.maximum(p, 0.0).copy()
    S = float(p.sum())
    if S < 1e-300:
        return p, math.inf
    p = p / S
    S = 1.0
    
    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    
    flat = 0
    sweep_count = 0
    
    while time.monotonic() < stop_time - 2.0:
        sweep_count += 1
        improved = False
        order = rng.permutation(n)
        
        for ii, i in enumerate(order):
            if time.monotonic() > stop_time - 1.0:
                break
            if S <= 1e-300:
                break
            if S > 1.3 or S < 0.7:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0
            
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            
            # Influence: c[start:end] += 2*w*p_old[j] + w^2*delta for j in window
            # c[j] for j in [i, i+Ls-1]: c[j] += 2*w*p[j-i] + w^2*(j-i==i)
            # So influence on c[start:end] is 2*w*p[k] for k=0,...,Ls-1, plus w^2 at position i-start+... 
            # Actually: for j in [i, i+n-1], the change in c[j] is:
            #   2*(w)*p[j-i] + w^2 * 1(j==2i) but only if 2i is in [i, i+n-1]
            #   = 2*w*p[j-i] + w^2 * delta_{j,2i}
            # In window coordinates: k = j-i, so k in [0, Ls-1]
            #   change = 2*w*p[k] + w^2 * delta_{k, i} (if i < Ls, i.e., 2i < i+n, i.e., i < n)
            
            infl = 2.0 * p[:Ls].copy()  # 2*p[0], 2*p[1], ..., 2*p[Ls-1]
            
            # Outside max
            max_const = 0.0
            if start > 0:
                max_const = max(max_const, float(c[:start].max()))
            if end < Ltot:
                max_const = max(max_const, float(c[end:].max()))
            
            base = c[start:end].copy()
            diag_offset = 2 * i - start  # position of 2i in window, if valid
            
            # Search range for new value x (absolute, not delta)
            x_hi = max(10.0 * v, 30.0 / n, 0.02)
            x_hi = min(x_hi, 3.0)
            x_lo = max(0.0, v - 0.5 * v) if v > 1e-10 else 0.0
            
            def vec_eval(pts):
                """Vectorized evaluation of score for multiple candidate values."""
                ws = pts - v  # deltas
                if len(ws) == 0:
                    return np.array([])
                # New c values in window: base + w*infl + w^2*delta
                # For each candidate w_t:
                #   new_c[k] = base[k] + w_t * infl[k] + w_t^2 * (k==diag_offset)
                ws_col = ws[:, None]  # (T, 1)
                aff = base[None, :] + ws_col * infl[None, :]  # (T, Ls)
                if 0 <= diag_offset < Ls:
                    aff[:, diag_offset] += ws_col.ravel()**2
                maff = aff.max(axis=1)
                maff = np.maximum(maff, max_const)
                denoms = S + ws
                vals = np.where(denoms > 1e-15, 2.0 * n * maff / (denoms * denoms), 1e100)
                return vals
            
            # Grid
            pts = np.linspace(x_lo, x_hi, G + 1)
            if x_lo < v <= x_hi:
                pts = np.sort(np.unique(np.concatenate((pts, [v]))))
            
            vals = vec_eval(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])
            
            # Golden section refinement
            if G > 4 and math.isfinite(best_val):
                dw = (x_hi - x_lo) / G
                lo_gs = max(0.0, best_x - 2.0 * dw)
                hi_gs = min(x_hi, best_x + 2.0 * dw)
                if hi_gs - lo_gs > 1e-12:
                    gr = (math.sqrt(5.0) - 1.0) / 2.0
                    a_gs, b_gs = lo_gs, hi_gs
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    f_c = vec_eval(np.array([c_gs]))[0]
                    f_d = vec_eval(np.array([d_gs]))[0]
                    for _ in range(15):
                        if f_c < f_d:
                            b_gs, d_gs, f_d = d_gs, c_gs, f_c
                            c_gs = b_gs - gr * (b_gs - a_gs)
                            f_c = vec_eval(np.array([c_gs]))[0]
                        else:
                            a_gs, c_gs, f_c = c_gs, d_gs, f_d
                            d_gs = a_gs + gr * (b_gs - a_gs)
                            f_d = vec_eval(np.array([d_gs]))[0]
                    x_opt = (a_gs + b_gs) / 2.0
                    f_opt = vec_eval(np.array([x_opt]))[0]
                    if f_opt < best_val:
                        best_x, best_val = x_opt, f_opt
            
            # Apply update if improved
            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and best_x >= 0:
                w = best_x - v
                if abs(w) > 1e-16:
                    # Update c
                    c[start:end] = base + w * infl
                    if 0 <= diag_offset < Ls:
                        c[start + diag_offset] += w * w
                    p[i] = best_x
                    S += w
                    cur_score = best_val
                    improved = True
        
        # Renormalize
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
            cur_score = float(2.0 * n * c.max())
        
        # Periodic exact recompute
        if sweep_count % 5 == 0:
            c = _conv_self(p)
            cur_score = float(2.0 * n * c.max())
        
        if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = p.copy()
            consider_callback(best_p)
        
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3 and time.monotonic() < stop_time - 60.0:
                # Perturb
                sigma = 0.03 + 0.01 * min(flat, 10)
                noise = rng.normal(0.0, sigma, n)
                p *= np.exp(noise)
                p = np.maximum(p, 0.0)
                ss = float(p.sum())
                if ss > 1e-300:
                    p /= ss
                    S = 1.0
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    if cur_score < best_score:
                        best_score = cur_score
                        best_p = p.copy()
                        consider_callback(best_p)
                flat = 0
            
            if flat >= 15:
                break
    
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    return best_p, best_score
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy import sparse as sp

t0_global = 0.0


def _conv_self(a):
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _fft_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(p)
    return float(2.0 * n * c.max() / (s * s))


def _build_A(f):
    n = len(f)
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(n)[None, :]
    rows = (j_idx + k_idx).ravel()
    cols = np.tile(np.arange(n), n)
    vals = np.tile(f, n)
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp(f, M, solver_list):
    import cvxpy as cp
    n = len(f)
    A = _build_A(f)
    b_var = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(b_var))
    constraints = [A @ b_var <= M]
    prob = cp.Problem(objective, constraints)
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if b_var.value is not None:
                b = np.asarray(b_var.value, dtype=np.float64).ravel()
                b = np.maximum(b, 0.0)
                if float(b.sum()) > 1e-15 and np.all(np.isfinite(b)):
                    return b
        except Exception:
            continue
    return None


def _lp_step(f, t=0.4, solver_list=None, time_limit=30.0):
    if solver_list is None:
        solver_list = ['GUROBI', 'MOSEK', 'CLARABEL', 'ECOS', 'SCS', 'CVXOPT', 'GLPK', 'CBC']
    n = len(f)
    c = _conv_self(f)
    M = float(c.max())
    if not math.isfinite(M) or M <= 0:
        return f.copy()
    t_start = time.monotonic()
    b = _solve_lp(f, M, solver_list)
    if b is None or time.monotonic() - t_start > time_limit:
        return f.copy()
    S_f = float(f.sum())
    S_b = float(b.sum())
    g = S_f * b / S_b
    new_f = (1.0 - t) * f + t * g
    return np.maximum(new_f, 0.0)

    c = _conv_self(p)

                pts = np.linspace(x_lo, x_hi, G + 1)
                if x_lo < v <= x_hi:
                    pts = np.sort(np.unique(np.concatenate((pts, [v]))))

                ws = pts - v
                aff = base[None, :] + ws[:, None] * infl[None, :]
                if 0 <= diag_offset < Ls:
                    aff[:, diag_offset] += ws ** 2
                maff = aff.max(axis=1)
                if max_const > 0:
                    maff = np.maximum(maff, max_const)
                denoms = S + ws
                vals = np.where(denoms > 1e-15, 2.0 * n * maff / (denoms * denoms), 1e100)

                if not np.any(np.isfinite(vals)):
                    continue
                imin = int(np.argmin(vals))
                best_x = float(pts[imin])
                best_val = float(vals[imin])

                # Quick golden section
                if G > 4 and math.isfinite(best_val):
                    dw = (x_hi - x_lo) / G
                    lo_gs = max(0.0, best_x - 1.5 * dw)
                    hi_gs = min(x_hi, best_x + 1.5 * dw)
                    if hi_gs - lo_gs > 1e-12:
                        a_gs, b_gs = lo_gs, hi_gs
                        c_gs = b_gs - gr_phi * (b_gs - a_gs)
                        d_gs = a_gs + gr_phi * (b_gs - a_gs)

                        def ev(x):
                            w = x - v
                            nb = base + w * infl
                            if 0 <= diag_offset < Ls:
                                nb = nb.copy()
                                nb[diag_offset] += w * w
                            mv = max(float(nb.max()), max_const)
                            denom = S + w
                            if denom <= 1e-15:
                                return 1e100
                            return 2.0 * n * mv / (denom * denom)

                        f_c, f_d = ev(c_gs), ev(d_gs)
                        for _ in range(10):
                            if f_c < f_d:
                                b_gs, d_gs, f_d = d_gs, c_gs, f_c
                                c_gs = b_gs - gr_phi * (b_gs - a_gs)
                                f_c = ev(c_gs)
                            else:
                                a_gs, c_gs, f_c = c_gs, d_gs, f_d
                                d_gs = a_gs + gr_phi * (b_gs - a_gs)
                                f_d = ev(d_gs)
                        x_opt = (a_gs + b_gs) / 2.0
                        f_opt = ev(x_opt)
                        if f_opt < best_val:
                            best_x, best_val = x_opt, f_opt

                if math.isfinite(best_val) and best_val < cur_score - 1e-14 and best_x >= 0:
                    w = best_x - v
                    if abs(w) > 1e-16:
                        c[start:end] = base + w * infl
                        if 0 <= diag_offset < Ls:
                            c[start + diag_offset] += w * w
                        p[i] = best_x
                        S += w
                        cur_score = best_val
                        improved = True

        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
            cur_score = float(2.0 * n * c.max())

        if sweep % 3 == 0:
            c = _conv_self(p)
            cur_score = float(2.0 * n * c.max())

        if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = p.copy()
            consider_callback(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5 and time.monotonic() < stop_time - 60.0:
                sigma = 0.02 + 0.02 * min(flat, 8)
                noise = rng.normal(0.0, sigma, n)
                p *= np.exp(noise)
                p = np.maximum(p, 0.0)
                ss = float(p.sum())
                if ss > 1e-300:
                    p /= ss
                    S = 1.0
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    if cur_score < best_score:
                        best_score = cur_score
                        best_p = p.copy()
                        consider_callback(best_p)
                flat = 0
            if flat >= 15:
                break

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    global t0_global
    t0_global = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0_global + budget - 35.0

    # Load parent
    try:
        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(512, dtype=np.float64)
    if p.size == 0:
        p = np.ones(512, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(512, dtype=np.float64)
    n = len(p)
    p = p / float(p.sum())

    # Track best
    best_p = p.copy()
    best_score = _fft_score(p)

    def consider(arr):
        nonlocal best_p, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.clip(a, 0.0, 1000.0)
            a = np.where(np.isfinite(a), a, 0.0)
            if a.size == 0 or float(a.sum()) < 0.01:
                return
            sc = _fft_score(a)
            if math.isfinite(sc) and sc < best_score - 1e-13:
                a = a / float(a.sum())
                best_p = a.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 1: LP averaging
    lp_stop = min(hard_deadline - 700.0, t0_global + 120.0)
    lp_ok = False
    if lp_stop > time.monotonic() + 20.0 and n <= 4096:
        solver_list = ['GUROBI', 'MOSEK', 'CLARABEL', 'ECOS', 'SCS', 'CVXOPT', 'GLPK', 'CBC']
        for lp_iter in range(15):
            if time.monotonic() > lp_stop:
                break
            t_lp = 0.35 if lp_iter < 8 else 0.25
            try:
                new_p = _lp_step(p, t=t_lp, solver_list=solver_list, time_limit=25.0)
                if new_p is not None and np.all(np.isfinite(new_p)) and float(new_p.sum()) > 1e-10:
                    new_p = new_p / float(new_p.sum())
                    consider(new_p)
                    lp_ok = True
                    if _fft_score(best_p) <= _fft_score(new_p) + 1e-10:
                        p = best_p.copy()
                    else:
                        p = new_p.copy()
            except Exception:
                if not lp_ok:
                    break
                else:
                    break

    # Phase 2: Main coordinate descent
    cd_stop = min(hard_deadline - 120.0, t0_global + budget - 120.0)
    if cd_stop > time.monotonic() + 30.0:
        try:
            p2, sc2 = _coord_descent(p, cd_stop, consider, seed=seed + 1000, G=44)
            if p2 is not None and len(p2) == n and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p = p2 / float(p2.sum())
                consider(p)
        except Exception:
            pass

    # Phase 3: Block CD polish with important coordinates
    blk_stop = min(hard_deadline - 40.0, t0_global + budget - 40.0)
    if blk_stop > time.monotonic() + 20.0:
        try:
            p3, sc3 = _block_cd(best_p.copy(), blk_stop, consider, seed=seed + 9999,
                                block_size=3, G=36)
            if p3 is not None and len(p3) == n and np.all(np.isfinite(p3)) and p3.sum() > 0:
                p = p3 / float(p3.sum())
                consider(p)
        except Exception:
            pass

    # Phase 4: One more LP + CD cycle if time remains
    rem = hard_deadline - 30.0
    if rem > time.monotonic() + 40.0 and lp_ok:
        lp_stop2 = min(rem - 20.0, time.monotonic() + 40.0)
        if lp_stop2 > time.monotonic() + 10.0:
            try:
                new_p = _lp_step(best_p.copy(), t=0.2,
                                 solver_list=['GUROBI', 'MOSEK', 'ECOS', 'SCS'],
                                 time_limit=15.0)
                if new_p is not None and np.all(np.isfinite(new_p)):
                    new_p = new_p / float(new_p.sum())
                    consider(new_p)
            except Exception:
                pass
            cd_stop2 = rem
            if cd_stop2 > time.monotonic() + 10.0:
                try:
                    p4, _ = _coord_descent(best_p.copy(), cd_stop2, consider,
                                           seed=seed + 55555, G=36)
                    if p4 is not None:
                        consider(p4)
                except Exception:
                    pass

    # Final validation
    try:
        out = np.asarray(best_p, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(128, 1.0 / 128.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            try:
                pp = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
                pp = np.clip(pp, 0.0, 1000.0)
                ls = [float(x) for x in pp]
                final_sc = float(evaluate_sequence(ls))
            except Exception:
                return [1.0]
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
```
