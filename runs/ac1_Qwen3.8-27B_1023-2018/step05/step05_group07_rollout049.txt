    n = a.size
    best_t = 0.0
    best_v = M

def _subgradient_step(a, c_full, M, deadline, rng):
    """Subgradient step reducing contribution at active lags."""
    n = a.size
    active_lags = np.where(c_full >= M * 0.999)[0]
    if len(active_lags) == 0:
        return None
    grad = np.zeros(n)
    for k_star in active_lags:
        ks = int(k_star)
        lo = max(0, ks - n + 1)
        hi = min(n, ks + 1)
        for j in range(lo, hi):
            idx = ks - j
            if 0 <= idx < n:
                grad[j] += 2.0 * a[idx]
    # Project to zero-mean (to preserve sum)
    grad = grad - grad.mean()
    gn = float(np.linalg.norm(grad))
    if gn < 1e-14:
        return None
    for alpha in [0.05, 0.03, 0.01, 0.005, 0.002]:
        cand = a - alpha * grad / gn
        cand = np.maximum(cand, 0.0)
        cs = cand.sum()
        if cs < 1e-12:
            continue
        cand = cand / cs
        v = float(np.convolve(cand, cand).max())
        if v < M * (1 - 1e-12):
            return cand
    return None

def _coordinate_descent(a, c_full, M, deadline, rng):
    """Try to reduce the max convolution by adjusting pairs."""
    n = a.size
    active = np.where(c_full >= M * 0.998)[0]
    for k_star in active[:8]:
        ks = int(k_star)
        lo = max(0, ks - n + 1)
        hi = min(n, ks + 1)
        # Find the pair (i, k-i) with largest product
        best_i, best_j, best_prod = -1, -1, 0
        for i in range(lo, hi):
            j = ks - i
            if 0 <= j < n:
                prod = a[i] * a[j]
                if prod > best_prod:
                    best_prod = prod
                    best_i, best_j = i, j
        if best_i >= 0 and best_i != best_j:
            delta = 0.05 * min(a[best_i], a[best_j])
