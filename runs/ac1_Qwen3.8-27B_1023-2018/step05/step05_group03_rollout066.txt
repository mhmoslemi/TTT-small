                fine.extend(np.linspace(lo, hi, 40, dtype=np.float64).tolist())
                if lo > 0.0:
                    fine.extend(np.geomspace(lo, hi, 16, dtype=np.float64).tolist())
                else:
                    fine.extend(np.linspace(0.0, hi, 16, dtype=np.float64).tolist())
                fine = np.unique(np.clip(np.asarray(fine, dtype=np.float64), 0.0, xhi))
                vals2, fine = _coord_eval(fine, base, infl, max_const, v, Sloc, n, di)
                if np.any(np.isfinite(vals2)):
                    im2 = int(np.argmin(vals2))
                    vx2 = float(fine[im2])
                    vv2 = float(vals2[im2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2
            if (math.isfinite(best_val) and best_val < cur_score - 1e-11
                    and 0.0 <= best_x <= xhi):
                d = best_x - v
                if abs(d) > 1e-16:
                    c[start:end] += d * infl
                    diag = 2 * i
                    if 0 <= diag < L:
                        c[diag] += d * d
                    p[i] = best_x
                    infl[i] = 2.0 * best_x
                    S += d
                    cur_score = best_val
                    improved = True
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            infl *= inv
            S = 1.0
            c = np.maximum(c, 0.0)
            desc = float(2.0 * n * c.max())
            if math.isfinite(desc) and desc < best_score - 1e-13:
                best_score = desc
                best_p = p.copy()
                if consider is not None:
                    try:
                        consider(best_p)
                    except Exception:
                        pass
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 25.0:
                    sigmas = (0.035, 0.07, 0.11, 0.16)
                    sigma = sigmas[min(flat - 5, len(sigmas) - 1)]
                    noise = rng.normal(0.0, sigma, n)
                    q = p * np.exp(noise)
                    if sigma >= 0.11:
                        x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
                        amp = 0.20 + 0.35 * float(rng.random())
                        f1 = 1 + int(rng.integers(0, 5))
                        ph1 = 2.0 * math.pi * float(rng.random())
                        q *= np.exp(amp * math.sin(2.0 * math.pi * f1 * x + ph1))
                    q = _normalize(np.maximum(q, 0.0))
                    p = q
                    c = _fft_conv(p)
                    c = np.maximum(c, 0.0)
                    S = 1.0
                    infl = 2.0 * p
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    flat = 0
                else:
                    break
        sweep += 1
    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(10.0, budget)
    reserve = 28.0 if budget > 300.0 else (14.0 if budget > 60.0 else 2.5)
    hard = t0 + budget - reserve
    if hard <= t0 + 2.0:
        hard = t0 + max(1.5, budget - 0.5)

    parent = _get_parent()
    n_target = int(parent.size) if parent.size > 0 else 3072
    best_arr = _normalize(parent.copy())
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf
    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.full(64, 1.0 / 64.0, dtype=np.float64)
        try:
            best_score = float(_score_exact_arr(best_arr))
        except Exception:
            best_score = math.inf

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return
        if arr.size == 0:
            return
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if not math.isfinite(s) or s < 0.01:
            return
        arr = arr / s
        try:
            sc = float(_score_exact_arr(arr))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-13:
            best_arr = arr.copy()
            best_score = sc
            return sc
    consider(best_arr)

    rng_top = np.random.default_rng(seed)

    # --- Phase A: deflate the peak (reweighting) ---
    if time.monotonic() < hard - 160.0:
        a_defl = best_arr.copy()
        a_defl = _deflate_max(a_defl, min(hard - 150.0, time.monotonic() + 110.0),
                              consider, rng_top)
        consider(a_defl)
        if _score_fft(a_defl) < _score_fft(best_arr):
            best_arr = a_defl.copy()

    # --- cheap structured perturbations for diversification ---
    base = best_arr.copy()
    for q in (0.7, 1.3):
        if time.monotonic() > hard - 120.0:
            break
        cand = _normalize(np.power(np.maximum(base, 0.0), float(q)))
        consider(cand)
    for _ in range(4):
        if time.monotonic() > hard - 110.0:
            break
        side = 1.0 if rng_top.random() < 0.5 else -1.0
        amp = 0.30 + 0.95 * float(rng_top.random())
        x = (np.arange(n_target, dtype=np.float64) + 0.5) / float(n_target)
        cand = _normalize(np.maximum(base * np.exp(side * amp * (2.0 * x - 1.0)), 0.0))
        consider(cand)

    # --- Phase B: L-BFGS log-sum-exp ---
    if time.monotonic() < hard - 120.0:
        try:
            lse_stop = min(hard - 100.0, time.monotonic() + min(150.0, 0.30 * (budget)))
            p_b = _lbfgs_optimize(best_arr.copy(), lse_stop, consider=consider,
                                  fine_only=False)
            consider(p_b)
        except Exception:
            pass

    # --- Phase C: coordinate polish ---
    if time.monotonic() < hard - 20.0:
        try:
            cstop = min(hard - 8.0, time.monotonic() + max(30.0, 0.55 * max(0.0, hard - 8.0 - time.monotonic())))
            if cstop > time.monotonic() + 15.0:
                p_c, _ = _coord_polish(best_arr.copy(), cstop, G=72,
                                       seed=seed + 1729, consider=consider)
                consider(p_c)
        except Exception:
            pass

    # --- final fine squeeze ---
    if time.monotonic() < hard - 70.0:
        try:
            lse_stop = min(hard - 40.0, time.monotonic() + min(70.0, 0.18 * max(0.0, hard - 40.0 - time.monotonic())))
            if lse_stop > time.monotonic() + 12.0:
                p_f = _lbfgs_optimize(best_arr.copy(), lse_stop, consider=consider,
                                      fine_only=True)
                consider(p_f)
        except Exception:
            pass
    if time.monotonic() < hard - 12.0:
        try:
            cstop = min(hard - 4.0, time.monotonic() + max(10.0, 0.6 * max(0.0, hard - 4.0 - time.monotonic())))
            if cstop > time.monotonic() + 8.0:
                p_f, _ = _coord_polish(best_arr.copy(), cstop, G=84,
                                       seed=seed + 424242, consider=consider)
                consider(p_f)
        except Exception:
            pass

    # --- final validation ---
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        out = out / float(out.sum())
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            fb = _normalize(parent)
            ls2 = [float(x) for x in fb]
            sc2 = float(evaluate_sequence(ls2))
            return ls2 if math.isfinite(sc2) else [1.0]
        return ls
    except Exception:
        return [1.0]
