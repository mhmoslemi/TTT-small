                                      consider=consider, max_sweeps=800000)
                if p3 is not None and p3.size == nbest and np.all(np.isfinite(p3)) and float(p3.sum()) > 0.0:
                    p3 = p3 / float(p3.sum())
                    consider(p3)
            except Exception:
                pass

        # One more short coordinate pass with different seed for different random order
        now = time.monotonic()
        if now < final_end - 8.0:
            try:
                p4, _ = _coord_polish(best_arr, final_end - 2.0, G=G_fine,
                                      seed=int(seed + 555555), consider=consider, max_sweeps=200000)
                if p4 is not None and p4.size == nbest and np.all(np.isfinite(p4)) and float(p4.sum()) > 0.0:
                    p4 = p4 / float(p4.sum())
                    consider(p4)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
