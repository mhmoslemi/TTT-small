            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                cand_a = _softmax(res.x)
                cand_a = np.maximum(cand_a, 0.0)
                s = float(cand_a.sum())
                if s > 1e-300:
                    cand_a /= s
                    sc = _score_arr(cand_a)
                    if math.isfinite(sc) and sc < local_best_sc - 1e-15:
                        local_best_sc = sc
                        local_best = res.x.copy()
                        if sc < best_sc:
                            best_sc = sc
                            best_a = cand_a.copy()
                            best_u = res.x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, "status", 1) == 0:
                break
    return local_best if np.all(np.isfinite(local_best)) else best_u, best_a, best_sc


def _coord_descent(a, stop_time, seed=12345, max_sweeps=5000):
    n = a.size
    a = np.maximum(a, 0.0).copy()
    S = float(a.sum())
    if S < 1e-300:
        return a, math.inf
    a /= S
    S = 1.0
    c = _conv_fft(a)
    L = 2 * n - 1
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    GR = (math.sqrt(5.0) - 1.0) / 2.0
    flat = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > stop_time - 1.5:
            break
        improved = False
        order = rng.permutation(n)
        for ii in order:
            i = int(ii)
            if time.monotonic() > stop_time - 0.3:
                break
            v = float(a[i])
            start_k = i
            end_k = min(i + n, L)
            Ls = end_k - start_k
            if Ls <= 0:
                continue
            base = c[start_k:end_k].copy()
            infl = 2.0 * a[:Ls].copy()
            max_const = 0.0
            if start_k > 0:
                max_const = max(max_const, float(c[:start_k].max()))
            if end_k < L:
                max_const = max(max_const, float(c[end_k:].max()))

            def eval_d(delta):
                denom = 1.0 + delta
                if denom < 1e-12:
                    return 1e100
                aff = base + delta * infl
                d2i = 2 * i
                if start_k <= d2i < end_k:
                    idx = d2i - start_k
                    aff[idx] = base[idx] + delta * infl[idx] + delta * delta
                m = max(float(aff.max()), max_const)
                return 2.0 * n * m / (denom * denom)

            x_hi = max(8.0 * v, 40.0 / n, 0.02 / n)
            x_hi = min(x_hi, 0.5)
            lo = -v + 1e-15
            hi = x_hi
            a_gs = hi - GR * (hi - lo)
            b_gs = lo + GR * (hi - lo)
            fa = eval_d(a_gs)
            fb = eval_d(b_gs)
            for _ in range(30):
                if hi - lo < 1e-14:
                    break
                if fa < fb:
                    hi = b_gs
                    b_gs = a_gs
                    fb = fa
                    a_gs = hi - GR * (hi - lo)
                    fa = eval_d(a_gs)
                else:
                    lo = a_gs
                    a_gs = b_gs
                    fa = fb
                    b_gs = lo + GR * (hi - lo)
                    fb = eval_d(b_gs)
            best_d = (lo + hi) / 2.0
            bv = eval_d(best_d)
            bv0 = eval_d(0.0)
            if math.isfinite(bv) and bv < bv0 - 1e-13:
                a[i] = v + best_d
                if a[i] < 0:
                    a[i] = 0.0
                delta = best_d
                c[start_k:end_k] += delta * infl
                d2i = 2 * i
                if start_k <= d2i < end_k:
                    idx = d2i - start_k
                    c[d2i] += delta * delta
                cur_score = bv
                improved = True

        S = float(a.sum())
        if S < 1e-300:
            break
        a /= S
        S = 1.0
        c = _conv_fft(a)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_a = a.copy()
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 5.0:
                    noise = 0.02 + 0.01 * math.exp(-0.05 * flat)
                    a = a * np.exp(rng.normal(0.0, noise, n))
                    a = np.maximum(a, 0.0)
                    ss = float(a.sum())
                    if ss > 1e-300:
                        a /= ss
                        c = _conv_fft(a)
                        cur_score = float(2.0 * n * c.max())
                        if cur_score < best_score:
                            best_score = cur_score
                            best_a = a.copy()
                    flat = 0
                else:
                    break
    return best_a, best_score


def _subgradient_step(a, stop_time, max_iters=2000, consider=None):
    n = a.size
    a = np.maximum(a, 0.0).copy()
    S = float(a.sum())
    if S < 1e-300:
        return a
    a /= S
    best_a = a.copy()
    best_score = _score_arr(a)
    base_lr = 0.003
    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break
        c = _conv_fft(a)
        cmax = float(c.max())
        thresh = cmax * (1.0 - 0.001 * math.exp(-0.01 * it))
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        if len(active) > 200:
            idx = np.argsort(c[active])[::-1][:100]
            active = active[idx]
        grad = np.zeros(n)
        for k in active:
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += a[k - j]
                if k - j != j:
                    grad[k - j] += a[j]
        grad /= float(len(active))
        for scale in (1.0, 0.5, 0.2):
            step = scale * base_lr / (1.0 + 0.005 * it)
            p_cand = a - step * grad
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = _score_arr(p_cand)
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_a = p_cand.copy()
                a = p_cand.copy()
                break
        else:
            base_lr *= 0.92
            if base_lr < 1e-10:
                break
    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    deadline = t0 + budget - 18.0

    parent = _get_parent()
    n = parent.size
    parent = np.maximum(parent, 0.0)
    ps = float(parent.sum())
    if ps < 0.01:
        parent = np.ones(n) / n
        ps = 1.0
    else:
        parent /= ps

    best_a = parent.copy()
    best_score = _score_arr(best_a)

    def try_update(a):
        nonlocal best_a, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0)
        s = float(a.sum())
        if s < 0.01 or a.size == 0:
            return
        a = a / s
        sc = _score_arr(a)
        if math.isfinite(sc) and sc < best_score:
            best_a = a.copy()
            best_score = sc

    # Phase 1: LSE optimization with annealing
    p1_end = min(deadline - 200.0, t0 + 0.30 * (budget - 30))
    if p1_end > time.monotonic() + 15:
        stages = [
            (0.08, 60), (0.03, 60), (0.01, 50),
            (0.004, 50), (0.001, 40), (0.0003, 30),
        ]
        _, p1_a, p1_sc = _lse_optimize(parent, p1_end, stages)
        try_update(p1_a)

    # Phase 2: Coordinate descent
    p2_end = min(deadline - 150.0, t0 + 0.50 * (budget - 30))
    now = time.monotonic()
    if p2_end > now + 20:
        p2_a, p2_sc = _coord_descent(best_a, p2_end, seed=int(seed + 100), max_sweeps=8000)
        try_update(p2_a)

    # Phase 3: LSE fine-tune from Phase 2
    p3_end = min(deadline - 100.0, t0 + 0.70 * (budget - 30))
    now = time.monotonic()
    if p3_end > now + 15:
        fine_stages = [
            (0.001, 40), (0.0003, 35), (0.0001, 30), (0.00003, 25),
        ]
        _, p3_a, p3_sc = _lse_optimize(best_a, p3_end, fine_stages)
        try_update(p3_a)

    # Phase 4: More coordinate descent
    p4_end = min(deadline - 60.0, t0 + 0.88 * (budget - 30))
    now = time.monotonic()
    if p4_end > now + 15:
        p4_a, p4_sc = _coord_descent(best_a, p4_end, seed=int(seed + 500), max_sweeps=10000)
        try_update(p4_a)

    # Phase 5: Subgradient refinement
    p5_end = deadline - 20.0
    now = time.monotonic()
    if p5_end > now + 10:
        p5_a = _subgradient_step(best_a, p5_end, max_iters=2000)
        try_update(p5_a)

    # Phase 6: Final coordinate polish
    now = time.monotonic()
    fin_end = deadline - 5.0
    if fin_end > now + 8:
        p6_a, _ = _coord_descent(best_a, fin_end, seed=int(seed + 999), max_sweeps=5000)
        try_update(p6_a)

    # Final validation using exact evaluator
    try:
        out_arr = np.asarray(best_a, dtype=np.float64).ravel()
        out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
        out_arr = np.clip(out_arr, 0.0, 1000.0)
        if out_arr.size == 0 or float(out_arr.sum()) < 0.01:
            out_arr = parent.copy()
        out_list = [float(x) for x in out_arr]
        final_sc = float(evaluate_sequence(out_list))
        if not math.isfinite(final_sc):
            # Try parent
            out_list = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(out_list))
            if not math.isfinite(final_sc):
                return [1.0]
        return out_list
    except Exception:
        return [1.0]
