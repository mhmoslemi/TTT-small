def _toeplitz_conv(a):
    """Build the (2n-1) x n matrix for a*g convolution."""
    n = a.size
    # A[k, j] = a[k-j] if 0 <= k-j < n else 0

    k = np.arange(2*n-1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    A = np.zeros((2*n-1, n), dtype=np.float64)
    mask = (d >= 0) & (d < n)
    A[mask] = a[d[mask]]
    return A

def _build_conv_mat(a):
    """Build (2n-1) x n matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    n = a.size
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    mask = (d >= 0) & (d < n)
    A[mask] = a[d[mask]]
    return A


def _line_search(a, g):

            return np.inf
        return float(np.convolve(cand, cand).max()) / (s * s)

    """Solve: max sum(g) s.t. A@G <= M, g>=0. Returns g or None."""
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': tl,
                               'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0, None)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None
