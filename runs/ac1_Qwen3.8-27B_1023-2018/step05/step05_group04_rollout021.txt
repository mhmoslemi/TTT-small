import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.asarray(c, dtype=np.float64)


def _score_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float('inf')
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return float('inf')
    return float(2.0 * n * m / (s * s))


def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    s = float(a.sum())
    if s < 1e-300:
        return None
    return a / s


def _coord_descent_fast(p, stop_time, consider=None, seed=12345):
    n = int(p.size)
    if n == 0:
        return p, float('inf')
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p, float('inf')
    p = p / s
    c = _conv_self(p)
    L = 2 * n - 1
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    GR = (math.sqrt(5.0) - 1.0) / 2.0
    sweep = 0
    flat = 0
    recompute_every = 8

    while time.monotonic() < stop_time - 4.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 2.0:
                break
            v = float(p[i])
            k_lo = i
            k_hi = min(i + n - 1, L - 1)
            length = k_hi - k_lo + 1
            if length <= 0:
                continue

            # Influence: 2*a[k-i] for affected k
            infl = 2.0 * p[k_lo - i:k_hi - i + 1]
            quad_pos = 2 * i - k_lo
            is_quad = np.zeros(length, dtype=np.float64)
            if 0 <= quad_pos < length:
                is_quad[quad_pos] = 1.0

            # Inactive max
            if k_lo > 0:
                inactive_max = float(c[:k_lo].max())
            else:
                inactive_max = 0.0
            if k_hi < L - 1:
                im2 = float(c[k_hi + 1:].max())
                if im2 > inactive_max:
                    inactive_max = im2

            # 1D search: x in [0, x_hi], delta = x - v
            x_hi = min(0.5, max(4.0 * v, 3.0 / n, 0.005))
            G = 40
            eps = 1e-14
            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                ins = np.searchsorted(pts, v)
                pts = np.insert(pts, ins, v)

            base = c[k_lo:k_hi + 1].copy()
            deltas = pts - v
            s_new = 1.0 + deltas
            # Vectorized: new peaks
            cn = base[None, :] + deltas[:, None] * infl[None, :] + deltas[:, None] ** 2 * is_quad[None, :]
            cn = np.maximum(cn, 0.0)
            peak_max = np.maximum(cn, inactive_max)
            pk_m = peak_max.max(axis=1)
            valid = s_new > 1e-12
            scores = np.full(pts.shape, np.inf, dtype=np.float64)
            scores[valid] = 2.0 * n * pk_m[valid] / (s_new[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            # Golden section refinement
            if 0 < imin < len(pts) - 1:
                lo = float(pts[max(0, imin - 1)])
                hi = float(pts[min(len(pts) - 1, imin + 1)])
            else:
                lo, hi = 0.0, x_hi
            if hi - lo > 1e-14:
                aGs, bGs = lo, hi
                cGs = bGs - GR * (bGs - aGs)
                dGs = aGs + GR * (bGs - aGs)

                def f1d(x):
                    d = x - v
                    sv = 1.0 + d
                    if sv <= 1e-12:
                        return np.inf
                    cn2 = base + d * infl + d * d * is_quad
                    cn2 = np.maximum(cn2, 0.0)
                    pk = max(float(cn2.max()), inactive_max)
                    return 2.0 * n * pk / (sv * sv)

                fc = f1d(cGs)
                fd = f1d(dGs)
                for _ in range(25):
                    if bGs - aGs < 1e-13 * max(1.0, abs(best_x)):
                        break
                    if fc < fd:
                        bGs = dGs; dGs = cGs; fd = fc
                        cGs = bGs - GR * (bGs - aGs); fc = f1d(cGs)
                    else:
                        aGs = cGs; cGs = dGs; fc = fd
                        dGs = aGs + GR * (bGs - aGs); fd = f1d(dGs)
                xm = 0.5 * (aGs + bGs)
                fm = f1d(xm)
                if fm < best_val:
                    best_x, best_val = xm, fm
                for xp in (0.0, x_hi):
                    fp = f1d(xp)
                    if fp < best_val:
                        best_x, best_val = xp, fp

            if math.isfinite(best_val) and best_val < cur_score - 1e-15 and best_x >= 0.0:
                delta = best_x - v
                if abs(delta) > 1e-16:
                    p[i] = best_x
                    c[k_lo:k_hi + 1] += delta * infl
                    if 0 <= quad_pos < length:
                        c[k_lo + quad_pos] += delta * delta
                    new_peak = max(float(c[k_lo:k_hi + 1].max()), inactive_max)
                    cur_score = 2.0 * n * new_peak
                    improved = True

        # Renormalize periodically
        if sweep % 2 == 0:
            sn = float(p.sum())
            if sn > 1e-300 and math.isfinite(sn):
                p /= sn
                c = _conv_self(p)
                new_sc = float(2.0 * n * c.max())
                if math.isfinite(new_sc) and new_sc < best_score - 1e-16:
                    best_score = new_sc
                    best_p = p.copy()
                    if consider:
                        try:
                            consider(best_p)
                        except Exception:
                            pass
                cur_score = new_sc

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 6:
                if time.monotonic() < stop_time - 15.0:
                    nk = 0.015 + 0.035 * min(1.0, flat / 20.0)
                    p = p * np.exp(rng.normal(0.0, nk, n))
                    p = np.maximum(p, 0.0)
                    ps = float(p.sum())
                    if ps > 0:
                        p /= ps
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    # Final
    sn = float(p.sum())
    if sn > 1e-300:
        p /= sn
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _lse_optimize(p, stop_time, consider=None):
    n = p.size
    p = np.maximum(np.asarray(p, dtype=np.float64), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    u = np.log(np.maximum(p, 1e-30))
    u = u - float(u.mean())

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2

    def softmax(uu):
        m = float(np.max(uu))
        e = np.exp(uu - m)
        ss = float(e.sum())
        if not math.isfinite(ss) or ss <= 0:
            return np.full(u.shape, 1.0 / n)
        return e / ss

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(a, w):
        F = rfft(a[::-1], cross_size)
        G = rfft(w, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return f, grad

    from scipy.optimize import minimize

    stages = [(0.08, 0.12), (0.025, 0.18), (0.006, 0.22), (0.0015, 0.25)]
    best_p = p.copy()
    best_sc = _score_arr(p)
    start = time.monotonic()
    avail = max(0.0, stop_time - start - 10.0)
    total_w = sum(w for _, w in stages)
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 6.0:
            break
        cum += wt
        end_stage = min(stop_time - 5.0, start + avail * (cum / total_w))
        if end_stage <= now + 5.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)
        best_u = u.copy()
        best_fstage = lse_func(u, T)[0]
        stale = 0
        while time.monotonic() < end_stage - 2.0:
            opts = {"maxiter": 30, "gtol": 1e-11, "ftol": 1e-16, "maxls": 12, "maxdisp": 0}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_fstage - 1e-15:
                    best_fstage = fval
                    best_u = res.x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 2:
                        break
            if getattr(res, "status", 1) == 0:
                break
        p_cur = softmax(best_u)
        sc = _score_arr(p_cur)
        if math.isfinite(sc) and sc < best_sc - 1e-13:
            best_sc = sc
            best_p = p_cur.copy()
            p = p_cur.copy()
            u = np.log(np.maximum(p, 1e-30))
            u = u - float(u.mean())
    if consider:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p
