                c = _conv_self(p)
                cur_score = float(2.0 * n * c.max())
                flat = 0
            elif flat >= 30:
                break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _subgradient_phase(p, stop_time, consider=None):
    """Subgradient method on the exact max for fine polish."""
    n = p.size
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    conv_len = 2 * n - 1
    conv_size = next_fast_len(conv_len)

    def conv_fast(a):
        F = rfft(a, conv_size)
        return irfft(F * F, conv_size)[:conv_len]

    c = conv_fast(p)
    cur_max = float(c.max())
    best_score = 2.0 * n * cur_max
    best_p = p.copy()

    lr = 0.5 / n
    lr_decay = 0.999
    lr_min = 1e-12
    rng = np.random.default_rng(777)

    for it in range(200000):
        if time.monotonic() > stop_time - 2.0:
            break
        threshold = cur_max * 0.9995
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        q = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_hi >= i_lo:
                ii = np.arange(i_lo, i_hi + 1)
                jj = k - ii
                q[ii] += p[jj]
        q /= len(active)
        grad = 2.0 * q

        p_new = p - lr * grad
        # Project onto simplex
        if p_new.min() < 0 or abs(p_new.sum() - 1) > 1e-10:
            sorted_p = np.sort(p_new)[::-1]
            nn = sorted_p.size
            cssv = np.cumsum(sorted_p)
            rho = np.where(1.0 - sorted_p / (cssv / np.arange(nn, 0, -1)) > 0)[0]
            if len(rho) > 0:
                rho = int(rho[-1])
            else:
                rho = 0
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            p_new = np.maximum(p_new - theta, 0.0)
            sn = float(p_new.sum())
            if sn > 0:
                p_new /= sn

        c_new = conv_fast(p_new)
        max_new = float(c_new.max())

        if max_new < cur_max * (1.0 - 1e-12):
            p = p_new
            c = c_new
            cur_max = max_new
            score = 2.0 * n * cur_max
            if score < best_score:
                best_score = score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass
        else:
            lr *= lr_decay
            if lr < lr_min:
                break

        if it % 200 == 0 and it > 100:
            kick = 0.005 * rng.normal(0, 1, n)
            p_kick = p * np.exp(kick)
            p_kick = np.maximum(p_kick, 0)
            sk = float(p_kick.sum())
            if sk > 0:
                p_kick /= sk
            ck = conv_fast(p_kick)
            mk = float(ck.max())
            if mk < cur_max * 1.02:
                p = p_kick
                c = ck
                cur_max = mk

    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 35.0
    if hard <= t0 + 5.0:
        hard = t0 + max(3.0, budget - 3.0)

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float('inf')

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()

    consider(parent)
    n_work = int(best_arr.size)

    # Phase 1: LSE optimization
    lse_time = min(90.0, 0.08 * budget)
    p_work = best_arr.astype(np.float64).copy()
    p_work = np.maximum(p_work, 0)
    sw = float(p_work.sum())
    if sw > 0:
        p_work /= sw
    lse_end = t0 + lse_time
    if lse_end > time.monotonic() + 10.0:
        try:
            p2 = _lse_phase(p_work, lse_end, consider=consider)
            if p2 is not None and p2.size == n_work and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p_work = p2 / float(p2.sum())
                consider(p_work)
        except Exception:
            pass

    # Phase 2: Subgradient polish
    sg_time = min(60.0, 0.05 * budget)
    sg_end = time.monotonic() + sg_time
    if sg_end < hard - 100.0:
        try:
            p_sg = _subgradient_phase(p_work, sg_end, consider=consider)
            if p_sg is not None and np.all(np.isfinite(p_sg)) and p_sg.sum() > 0:
                p_work = p_sg / float(p_sg.sum())
                consider(p_work)
        except Exception:
            pass

    # Phase 3: Coordinate descent (main workhorse)
    cd_end = hard - 10.0
    if cd_end > time.monotonic() + 30.0:
        p_start = best_arr.astype(np.float64).copy()
        p_start = np.maximum(p_start, 0)
        ss = float(p_start.sum())
        if ss > 0:
            p_start /= ss
        try:
            p_cd, _ = _coord_descent(p_start, cd_end, consider=consider, seed=int(seed + 7777))
            if p_cd is not None and p_cd.size > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                consider(p_cd)
        except Exception:
            pass

    # Phase 4: If time remains, do another CD pass from the best
    rem = hard - 10.0 - time.monotonic()
    if rem > 60.0:
        p_final = best_arr.astype(np.float64).copy()
        p_final = np.maximum(p_final, 0)
        ss = float(p_final.sum())
        if ss > 0:
            p_final /= ss
        try:
            p_f, _ = _coord_descent(p_final, hard - 10.0, consider=consider, seed=int(seed + 9999))
            if p_f is not None and p_f.size > 0 and np.all(np.isfinite(p_f)) and p_f.sum() > 0:
                consider(p_f)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.astype(np.float64).copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc) or final_sc > 10.0:
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
