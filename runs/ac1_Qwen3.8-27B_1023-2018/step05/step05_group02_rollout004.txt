import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _score_exact_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * a.size * m / (s * s))


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == 1:
        return np.full(n, float(src[0]) / n)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m) + 0.5) / m
        fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _lse_optimize(p, stop_time, consider=None):
    """LSE-approximated L-BFGS optimization"""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    
    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))
    
    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c
    
    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q
    
    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0.0:
            return 1e100, np.zeros_like(uu)
        if T <= 0.0:
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -400.0, 400.0)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad
    
    # Temperature schedule
    c = cfast(p)
    cmax = float(c.max())
    
    start = time.monotonic()
    avail = float(stop_time - start - 4.0)
    if avail <= 5.0:
        return p
    
    best_p = p.copy()
    best_sc = _score_fast(p)
    
    stages = [
        (0.08, 0.10),
        (0.03, 0.12),
        (0.01, 0.15),
        (0.003, 0.18),
        (0.001, 0.20),
        (0.0002, 0.12),
        (0.00005, 0.13),
    ]
    
    total_w = sum(w for _, w in stages)
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        cum += wt
        end_stage = min(stop_time - 2.0, start + avail * (cum / total_w))
        if end_stage <= now + 1.0:
            continue
        
        T = max(1e-300, frac * cmax)
        stale = 0
        max_rounds = 8 if n > 2000 else 12
        
        for _ in range(max_rounds):
            if time.monotonic() > end_stage - 0.5:
                break
            opts = {
                "maxiter": 30 if n > 2000 else 50,
                "gtol": 1e-13,
                "ftol": 1e-16,
                "maxls": 25,
                "maxcor": 25,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                u = res.x
                p_cur = _softmax(u)
                p_cur = np.maximum(p_cur, 0.0)
                ss = float(p_cur.sum())
                if ss > 1e-300:
                    p_cur /= ss
                sc_cur = _score_fast(p_cur)
                if math.isfinite(sc_cur):
                    if sc_cur < best_sc:
                        best_sc = sc_cur
                        best_p = p_cur.copy()
                        p = p_cur.copy()
                        u = p_to_u(p)
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 2:
                            break
            if getattr(res, "status", 1) == 0:
                break
        
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass
    
    return best_p


def _coord_descend(p, stop_time, seed=12345, consider=None, G_base=None):

    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s0 = float(p.sum())
    if s0 < 1e-300:
        return p, math.inf
    p = p / s0
    
    if G_base is None:
        if n <= 500:
            G_base = 50
        elif n <= 1000:
            G_base = 40
        elif n <= 2500:
            G_base = 32
        else:
            G_base = 24
    G = int(max(16, G_base))
    
    Ltot = 2 * n - 1
    c = _conv_self(p)
    S = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    
    flat = 0
    sweep = 0
    golden_gr = (math.sqrt(5.0) - 1.0) / 2.0
    
    while time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            if S <= 1e-300:
                break
            
            # Rescale if needed
            if S > 1.5 or S < 0.67:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0
            
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls].copy()
            diag_offset = 2 * i - start  # index in base where the diagonal is
            
            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr
            
            # Determine search range
            x_hi = max(15.0 * v, 40.0 / n, 0.03 / n, 1e-12)
            x_hi = min(x_hi, 5.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, S / n)
            
            # Grid evaluation
            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 < v < x_hi:
                pts = np.unique(np.concatenate((pts, [v])))
            
            w = pts - v
            denom = S + w
            valid = denom > 1e-15
            
            if np.any(valid):
                # Affine part: base + w * infl
                aff = base[None, :] + w[None, :-1, None] * infl[None, :, None] if False else None
                # Vectorized: for each candidate x, compute max of base + w*infl
                # Shape: (G+1, Ls)
                vals_aff = base[None, :] + w[:, None] * infl[None, :]
                # Diagonal term
                if 0 <= diag_offset < Ls:
                    vals_aff[:, diag_offset] += w * w  # w^2 at diagonal
                vals_aff = np.maximum(vals_aff, max_const)
                
                dv = denom[None, :]
                # We need max over axis=1, then divide by dv^2
                maff = vals_aff.max(axis=1)
                candidate_vals = np.full(pts.shape, math.inf)
                candidate_vals[valid] = (2.0 * n * maff[valid]) / (denom[valid] ** 2)
            else:
                candidate_vals = np.full(pts.shape, math.inf)
            
            if not np.any(np.isfinite(candidate_vals)):
                continue
            
            imin = int(np.argmin(candidate_vals))
            best_x = float(pts[imin])
            best_val = float(candidate_vals[imin])
            
            # Refine with golden section
            if G > 4 and math.isfinite(best_val):
                dw = x_hi / G
                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)
                if hi - lo > 1e-15:
                    # Fine grid in refinement region
                    fine_n = min(30, max(10, int(dw * 10)))
                    fpts = np.linspace(lo, hi, fine_n + 1)
                    if lo < v < hi:

                    fvalid = fdenom > 1e-15

                        fcand = np.full(fpts.shape, math.inf)

                        if np.any(np.isfinite(fcand)):
                            fim = int(np.argmin(fcand))
                            if fcand[fim] < best_val:
                                best_x = float(fpts[fim])
                                best_val = float(fcand[fim])
                    
                    # Golden section
                    if hi - lo > 1e-14:
                        a_gs = hi - golden_gr * (hi - lo)
                        b_gs = lo + golden_gr * (hi - lo)
                        
                        def eval_one(x_single):
                            w_s = x_single - v
                            d_s = S + w_s
                            if d_s <= 1e-15:
                                return math.inf
                            aff_s = base + w_s * infl
                            if 0 <= diag_offset < Ls:
                                aff_s[diag_offset] += w_s * w_s
                            aff_s = np.maximum(aff_s, max_const)
                            maff_s = float(aff_s.max())
                            return (2.0 * n * maff_s) / (d_s * d_s)
                        
                        fa = eval_one(a_gs)
                        fb = eval_one(b_gs)

                                fa = 1e100
                            if not math.isfinite(fb):
                                fb = 1e100

                                b_gs = a_gs
                                fb = fa

                                fa = eval_one(a_gs)
                            else:
                                lo = a_gs
                                a_gs = b_gs
                                fa = fb
                                b_gs = lo + golden_gr * (hi - lo)
                                fb = eval_one(b_gs)
                            if hi - lo < 1e-15:
                                break
                        
                        gs_x = (lo + hi) / 2.0
                        gv = eval_one(gs_x)
                        if math.isfinite(gv) and gv < best_val:
                            best_x = gs_x
                            best_val = gv

            if math.isfinite(best_val) and best_val < cur_score * (1 - 1e-12) and best_val < cur_score - 1e-14:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S += w
                    c[start:end] += w * infl
                    diag = 2 * i
                    if 0 <= diag < Ltot:
                        c[diag] += w * w
                    cur_score = min(cur_score, best_val)
                    improved = True

            p *= inv
            c *= inv * inv
        else:
            break
        S = 1.0

        c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score * (1 - 1e-10):
            best_score = new_score
            best_p = p.copy()
        
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass
        
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:

                    p = p * np.exp(rng.normal(0.0, noise_amt, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0.0:
                        p /= ss
                        S = 1.0
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score * (1 - 1e-10):
                        best_score = cur_score
                        best_p = p.copy()
                    flat = 0
                elif flat >= 6:
                    break
    
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    c = _conv_self(p)
    sc = float(2.0 * n * c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _pair_coord_descend(p, stop_time, seed=777, consider=None):
    """Pair coordinate descent: optimize pairs of coordinates jointly"""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p
    s0 = float(p.sum())
    if s0 < 1e-300:
        return p
    p = p / s0
    
    rng = np.random.default_rng(seed)
    Ltot = 2 * n - 1
    c = _conv_self(p)
    S = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    
    G = 12  # Grid size per dimension in 2D
    
    while time.monotonic() < stop_time - 1.0:
        improved = False
        # Random pairs
        for _ in range(n):
            if time.monotonic() > stop_time - 0.3:
                break
            
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i == j:
                continue
            
            vi = float(p[i])
            vj = float(p[j])
            
            # Affected region: union of [i, i+n-1] and [j, j+n-1] intersected with [0, Ltot-1]
            a_start = min(i, j)
            a_end = max(i + n, j + n)
            a_end = min(a_end, Ltot)
            if a_end <= a_start:
                continue
            La = a_end - a_start
            
            max_const = 0.0
            if a_start > 0:
                max_const = max(max_const, float(c[:a_start].max()))
