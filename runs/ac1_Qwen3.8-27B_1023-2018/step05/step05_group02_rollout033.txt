The terms involving $a[i]$ are those where $j = i$ or $k-j = i$ (i.e., $j = k-i$).

Case 1: $j = i$ is in $J_k$ (i.e., $0 \leq k-i < n$, so $i \leq k \leq i+n-1$).
Term: $a[i] \cdot a[k-i]$. Change: $w \cdot a[k-i]$.

Case 2: $j = k-i$ is in $J_k$ (i.e., $0 \leq k-i < n$ and $0 \leq i < n$, so $i \leq k \leq i+n-1$).
Term: $a[k-i] \cdot a[i]$. Change: $a[k-i] \cdot w$.

If both cases apply (which they do simultaneously when $0 \leq k-i < n$), and if $j = i$ and $j = k-i$ are the same index (i.e., $i = k-i$, so $k = 2i$), then we have a single term $a[i]^2$ that changes to $(a[i]+w)^2$, giving $\Delta = 2wa[i] + w^2$.

If $k \neq 2i$ and $0 \leq k-i < n$, then we have two distinct terms, each changing by $w \cdot a[k-i]$, so total $\Delta = 2w \cdot a[k-i]$.

If $k < i$ or $k > i+n-1$, then no terms involve $a[i]$, so $\Delta = 0$.

So the change in $c_k$ when $a[i]$ changes by $w$ is:
- $0$ if $k < i$ or $k \geq i+n$
- $2w \cdot a[k-i]$ if $i \leq k \leq i+n-1$ and $k \neq 2i$
- $2w \cdot a[i] + w^2$ if $k = 2i$

        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -400.0, 400.0)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f_val = cmax
        else:
            f_val = cmax + T * float(np.log(sum_exp / c.size))
            if not math.isfinite(f_val):
                f_val = cmax
            w = exps / sum_exp
        # Gradient via cross-correlation
        Fp = rfft(pp[::-1], conv_size)
        Gw = rfft(w, conv_size)
        cross = irfft(Fp * Gw, conv_size)[:conv_len]
        q = cross[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f_val), grad

    if fine_only:
        stages = ((0.001, 0.35), (0.0003, 0.40), (0.00008, 0.25))
    else:
        stages = (
            (0.10, 0.10),
            (0.04, 0.12),
            (0.01, 0.15),
            (0.003, 0.20),
            (0.0008, 0.25),
            (0.0002, 0.18),
        )

    start = time.monotonic()
    avail = float(stop_time - start - 5.0)
    if avail <= 3.0:
        return p

    best_p = p.copy()
    best_sc = _score_arr(p)
    total_w = float(sum(w for _, w in stages))
    if total_w <= 0:
        total_w = 1.0

    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 1.5:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)
        stale = 0
        while time.monotonic() < end_stage - 1.0:
            maxit = 30 if n > 1500 else 45
            opts = {
                "maxiter": maxit,
                "gtol": 1e-13,
                "ftol": 1e-16,
                "maxls": 25,
                "maxcor": 25,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    u = x
                    pp = np.exp(uu - uu.max()) if False else np.exp(x - x.max())
                    ss = float(pp.sum())
                    if ss > 1e-300:
                        pp /= ss
                    sc_cur = _score_arr(pp)
                    if math.isfinite(sc_cur) and sc_cur < best_sc - 1e-15:
                        best_sc = sc_cur
                        best_p = pp.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
                    p = pp
                    u = p_to_u(p)
            if getattr(res, "status", 1) == 0:
                break
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _subgradient_polish(p, stop_time, consider=None, max_iters=2000):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    best_p = p.copy()
    best_score = float(2.0 * n * _conv_self(p).max())
    base_step = 0.008
    min_step = 1e-9

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break
        c = _conv_self(p)
        cmax = float(c.max())
        threshold = cmax * 0.998
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        grad = np.zeros(n)
        for k in active:
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                grad[valid] += p[kj[valid]] + p[valid]
        grad /= len(active)

        step = base_step / (1.0 + 0.03 * it)
        if step < min_step:
            break

        for scale in (1.0, 0.5, 0.25, 0.1):
            p_cand = p - scale * step * grad
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = float(2.0 * n * _conv_self(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                break
        else:
            base_step *= 0.85
            if base_step < 1e-13:
                break

        if it % 30 == 0 and consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 18.0

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = math.inf
    try:
        best_score = _score_exact_list(best_arr.tolist())
    except Exception:
        pass

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(128, dtype=np.float64) / 128.0
        best_score = _score_exact_list(best_arr.tolist())

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            sc = _score_exact_list(a.tolist())
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_arr = a.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 1: Try parent length with coordinate descent (main phase)
    n_parent = int(best_arr.size)
    p = best_arr.copy()
    p = p / float(p.sum())
    consider(p)

    # Give most time to coordinate descent on the parent
    # Phase 1a: Brief LBFGS to improve shape
    if n_parent <= 6000 and time.monotonic() < hard - 200.0:
        lbfgs_end = min(hard - 180.0, t0 + 0.20 * budget)
        if lbfgs_end > time.monotonic() + 20.0:
            try:
                p2 = _lbfgs_optimize(p, lbfgs_end, consider=consider, fine_only=False)
                if p2 is not None and p2.size == n_parent and np.all(np.isfinite(p2)) and p2.sum() > 0:
                    p2 = p2 / float(p2.sum())
                    consider(p2)
                    p = p2.copy()
            except Exception:
                pass

    # Phase 1b: Main coordinate descent
    cd_end = min(hard - 60.0, t0 + 0.65 * budget)
    if cd_end > time.monotonic() + 10.0:
        try:
            p3, _ = _coord_descent(p, cd_end, seed=int(seed + 1), consider=consider, max_flat=5)
            if p3 is not None and p3.size == n_parent:
                p = p3.copy()
                consider(p)
        except Exception:
            pass

    # Phase 1c: Additional coordinate descent rounds with different seeds
    now = time.monotonic()
    if now < hard - 40.0:
        cd_end2 = min(hard - 30.0, now + 0.15 * budget)
        if cd_end2 > now + 10.0:
            try:
                p4, _ = _coord_descent(p, cd_end2, seed=int(seed + 777), consider=consider, max_flat=4)
                if p4 is not None and p4.size == n_parent:
                    p = p4.copy()
                    consider(p)
            except Exception:
                pass

    # Phase 2: Try a different length (shorter for potentially better optimization)
    now = time.monotonic()
    if now < hard - 80.0:
        n_alt = max(1024, n_parent // 2)
        p_alt = _resample(best_arr, n_alt)
        consider(p_alt)
        alt_end = min(hard - 40.0, now + 0.15 * budget)
        if alt_end > now + 15.0:
            try:
                p5, _ = _coord_descent(p_alt, alt_end, seed=int(seed + 999), consider=consider, max_flat=4)
                if p5 is not None:
                    consider(p5)
            except Exception:
                pass

    # Phase 3: Final polish on the best
    now = time.monotonic()
    best_arr = best_arr.copy()
    nbest = int(best_arr.size)
    nb = best_arr.copy()
    nb = nb / float(nb.sum())

    if now < hard - 20.0:
        # Subgradient
        sg_end = min(hard - 15.0, now + max(10.0, 0.08 * budget))
        if sg_end > now + 8.0:
            try:
                p_sg = _subgradient_polish(nb, sg_end, consider=consider, max_iters=1500)
                if p_sg is not None and p_sg.size == nbest:
                    consider(p_sg)
                    nb = p_sg.copy()
                    nb = nb / float(nb.sum())
            except Exception:
                pass

    # Final coordinate descent squeeze
    now = time.monotonic()
    if now < hard - 5.0:
        c_end = hard - 3.0
        if c_end > now + 5.0:
            try:
                p_fin, _ = _coord_descent(nb, c_end, seed=int(seed + 55555), consider=consider, max_flat=3, G=20)
                if p_fin is not None and p_fin.size == nbest:
                    consider(p_fin)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
