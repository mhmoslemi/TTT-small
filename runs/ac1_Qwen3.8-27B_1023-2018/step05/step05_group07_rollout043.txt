def _simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _grad_convolve(a, m):
    """Gradient of c[m] = (a*a)[m] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)

    k_range = np.arange(n)
    idx = m - k_range
    valid = (idx >= 0) & (idx < n)
    g[valid] = 2.0 * a[idx[valid]]
    return g

    n = len(a)

    n = len(a)

def _soft_max_convolve(a, beta=100):
    """Soft approximation of max(convolve(a,a))."""
    c = np.convolve(a, a)
    c_max = c.max()
    return c_max + np.log(np.exp(beta * (c - c_max)).sum()) / beta

def _soft_max_grad(a, beta=100):
    """Gradient of soft max of convolve(a,a)."""
    c = np.convolve(a, a)
    c_max = c.max()
    weights = np.exp(beta * (c - c_max)) / np.exp(beta * (c - c_max)).sum()
    g = np.zeros_like(a)
    for m in range(len(c)):
        if weights[m] > 1e-15:
            k_range = np.arange(len(a))
            idx = m - k_range
            valid = (idx >= 0) & (idx < len(a))
            g[valid] += weights[m] * 2.0 * a[idx[valid]]
    return g, _soft_max_convolve(a, beta)

The LP-based method is essentially a variant of the "best response" or "fictitious play" algorithm for a zero-sum game. The convergence rate depends on the "payoff matrix" and can be slow in the worst case.

Let me think about what's limiting the convergence. At each iteration:

2. Move from a towards g with step size t

**Improvement: Use the LP dual to get a better direction**

The LP is: max sum(g) s.t. A g <= M, g >= 0.
The dual is: min M * 1^T y s.t. A^T y >= 1, y >= 0.

The dual variables y give the "shadow prices" of the constraints. They tell us which lags are most "active" or "binding."

We can use y to construct a better direction: instead of just moving towards g, we can use y to weight the subgradient.

Actually, the LP dual is exactly the subgradient information! The y* from the dual gives a convex combination of the active constraint gradients, which is a valid subgradient of max_m c[m].

So we can do:
1. Solve LP (primal) to get g, and extract dual variables y
2. Use y to form the subgradient: grad = sum_m y[m] * ∇c[m]
3. Do a projected subgradient step in the -grad direction

This combines the global information from the LP with the local information from the subgradient.

Let me also consider: can we solve the LP more efficiently? The constraint matrix A has structure: A[m, j] = a[m-j]. This is a "convolution matrix" or "Toeplitz-like" matrix. There might be faster algorithms, but for n=281, the dense LP is already fast enough.

            stuck = 0
        else:
            stuck += 1

        if stuck > 100:
            a *= np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            stuck = 0
            continue

        # For the peak lag m*, reduce products a[j]*a[m*-j]
        jlo = max(0, peak_m - n + 1)
        jhi = min(n, peak_m + 1)
        moved = False

        # Try pair swaps: move mass from high-product pairs
        pairs = []
        for j in range(jlo, jhi):
            k = peak_m - j
            if 0 <= k < n:
                pairs.append((j, k, a[j] * a[k]))
        pairs.sort(key=lambda x: -x[2])

        for j, k, prod in pairs[:20]:
            if j == k:
                continue
            for frac in [0.1, 0.2, 0.3, 0.5]:
                d = frac * min(a[j], a[k])
                for (si, sk) in [(1, -1), (-1, 1), (-1, -1), (1, 1)]:
                    cand = a.copy()
                    cand[j] += si * d
                    cand[k] += sk * d
                    cand = np.maximum(cand, 0)
                    s = cand.sum()
                    if s < 1e-12:
                        continue
                    cand = cand / s
                    v = float(np.convolve(cand, cand).max())
                    if v < M * (1.0 - 1e-12):
                        a = cand
                        moved = True
                        break
                if moved:
                    break
            if moved:
                break

        if not moved:
            # Small random move
            j = rng.integers(0, n)
            k = rng.integers(0, n)
            d = 0.01 * a.tolist()[j]
            a[j] -= d
            a[k] += d
            a = np.maximum(a, 1e-12)
            a = a / a.sum()

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    n_parent = parent.size

    best_seq = parent.copy().astype(np.float64)
    best_score = _score(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(100)
        best_score = _score(best_seq)

    def try_update(cand):
        nonlocal best_seq, best_score
        try:
            cand = np.asarray(cand, dtype=np.float64).ravel()
            cand = np.clip(cand, 0, 1000)
            if cand.size == 0 or not np.all(np.isfinite(cand)):
                return
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand.copy()
        except Exception:
            pass

    # Phase 1: LP optimization on parent (first 450s)
    t1 = t0 + min(450.0, budget * 0.35)
    try:
        r = _optimize_lp(parent, t1, seed=1, lp_tl=1.0)
        try_update(r)
    except Exception:
        pass

    # Phase 2: Soft-max gradient on best (next 150s)
    t2 = t1 + min(150.0, budget * 0.12)
    if time.monotonic() < t2 - 5:
        ab = best_seq.copy()
        ab = ab / ab.sum()
        try:
            r = _optimize_softmax(ab, t2, seed=77, beta=30)
            try_update(r)
        except Exception:
            pass

    # Phase 3: LP again on best (next 250s)
    t3 = t2 + min(250.0, budget * 0.20)
    if time.monotonic() < t3 - 5:
        ab = best_seq.copy()
        ab = ab / ab.sum()
        try:
            r = _optimize_lp(ab, t3, seed=200, lp_tl=1.0)
            try_update(r)
        except Exception:
            pass

    # Phase 4: Multi-start with diverse initializations (next 200s)
    t4 = t3 + min(200.0, budget * 0.15)
    if time.monotonic() < t4 - 10:
        pn = parent / max(parent.sum(), 1e-15)
        x = np.linspace(0, 1, n_parent)
        ctr = (n_parent - 1) / 2.0

        inits = []
        # Gaussian bumps
        for sigma in [0.08, 0.12, 0.2, 0.3, 0.4, 0.6]:
            v = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
            inits.append(v / v.sum())
        # Bimodal
        for w in [0.15, 0.25, 0.3]:
            v = np.exp(-0.5 * ((x - w) / 0.08) ** 2) + np.exp(-0.5 * ((x - (1 - w)) / 0.08) ** 2)
            inits.append(v / v.sum())
        # Polynomial
        for p in [2, 3, 4, 6]:
            v = (1 - 4 * (x - 0.5) ** 2) ** p + 0.01
            v = np.maximum(v, 0)
            inits.append(v / v.sum())
        # Blends with parent
        for frac in [0.3, 0.5, 0.7]:
            inits.append((frac * pn + (1 - frac) * inits[0]) / (frac * pn + (1 - frac) * inits[0]).sum())

        remaining = t4 - time.monotonic() - 5
        per_init = max(5.0, remaining / max(1, len(inits)))

        for i, init in enumerate(inits):
            if time.monotonic() > t4 - 5:
                break
            alloc = min(time.monotonic() + per_init, t4 - 3)
            try:
                r = _optimize_lp(init, alloc, seed=500 + i, lp_tl=0.8)
                try_update(r)
            except Exception:
                pass

    # Phase 5: Coordinate descent polish (next 100s)
    t5 = t4 + min(100.0, budget * 0.08)
    if time.monotonic() < t5 - 5:
        ab = best_seq.copy()
        ab = ab / ab.sum()
        try:
            r = _optimize_coord(ab, t5, seed=999)
            try_update(r)
        except Exception:
            pass

    # Phase 6: Final LP polish on best (remaining time)
    if time.monotonic() < deadline - 10:
        ab = best_seq.copy()
        ab = ab / ab.sum()
        try:
            r = _optimize_lp(ab, deadline, seed=7777, lp_tl=0.8)
            try_update(r)
        except Exception:
            pass

    # Phase 7: Try a couple different lengths
    if time.monotonic() < deadline - 30:
        for dn in [50, -30, 80]:
            if time.monotonic() > deadline - 15:
                break
            nt = n_parent + dn
            if nt < 50 or nt > 600:
                continue
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, nt)
            init_new = np.interp(x_new, x_old, best_seq)
            init_new = np.clip(init_new, 0, 1000)
            sn = init_new.sum()
            if sn < 1e-10:
                continue
            init_new = init_new / sn
            alloc = min(time.monotonic() + 25, deadline - 10)
            try:
                r = _optimize_lp(init_new, alloc, seed=3000 + nt, lp_tl=0.7)
                try_update(r)
            except Exception:
                pass

    # Final validation
    try:
        fs = float(evaluate_sequence([float(v) for v in best_seq]))
        if not np.isfinite(fs):
            raise ValueError("invalid final")
    except Exception:
        fallback = [1.0] * 100
        return fallback

    return [float(v) for v in best_seq]
