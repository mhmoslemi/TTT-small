        # Update a[i], c, S

def _fft_conv(a):

                improved = True

        # Renormalize after sweep
        if S > 1e-300 and math.isfinite(S) and abs(S - 1.0) > 1e-8:
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
            np.maximum(c, 0.0, out=c)
            cur_sc = float(2.0 * n * c.max())

        if improved:
            if math.isfinite(cur_sc) and cur_sc < best_sc - 1e-13:
                best_sc = cur_sc
                best_p = p.copy()
                consider(p)
            flat_count = 0
        else:
            flat_count += 1
            if flat_count >= 4 and time.monotonic() < stop_time - 15.0:
                # Perturb
                p *= np.exp(rng.normal(0.0, 0.02, n, dtype=np.float64))
                p = np.maximum(p, 0.0)
                ss = float(p.sum())
                if ss > 1e-300:
                    p /= ss
                    S = 1.0
                c = _fft_conv(p)
                cur_sc = float(2.0 * n * c.max())
                if math.isfinite(cur_sc) and cur_sc < best_sc - 1e-13:
                    best_sc = cur_sc
                    best_p = p.copy()
                    consider(p)
                flat_count = 0
            elif flat_count >= 10:
                break

        sweep += 1

    # Final
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    sc_f = _score_fast(p)
    if math.isfinite(sc_f) and sc_f < best_sc:
        best_sc = sc_f
        best_p = p.copy()
        consider(p)
    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 25.0

    parent = _get_parent()
    n = int(parent.size)
    best_arr = parent.copy()
    try:
        best_score = float(_score_fast(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score):
        best_arr = np.ones(n, dtype=np.float64) / n
        best_score = float(_score_fast(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0 or float(a.sum()) < 0.01:
                return
            sc = float(_score_fast(a))
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_arr = a.copy()
                best_score = sc
        except Exception:
            pass

    work = best_arr.copy()
    work /= float(work.sum())

    # === Phase 1: Coarse L-BFGS ===
    p1_end = min(hard_deadline - 600.0, t0 + 0.10 * budget)
    if p1_end > time.monotonic() + 15.0:
        try:
            work = _lse_lbfgs(work, p1_end, consider, coarse=True)
            if work is not None and work.size == n and work.sum() > 0:
                work = np.maximum(work, 0.0)
                work /= float(work.sum())
                consider(work)
        except Exception:
            pass

    work = best_arr.copy()
    work /= float(work.sum())

    # === Phase 2: Coordinate descent (main) ===
    p2_end = min(hard_deadline - 200.0, t0 + 0.55 * budget)
    if p2_end > time.monotonic() + 30.0:
        try:
            work = _coord_descent_v2(work, p2_end, consider,
                                     max_sweeps=300000, grid_pts=32,
                                     seed=int(seed))
            if work is not None and work.size == n and work.sum() > 0:
                work = np.maximum(work, 0.0)
                work /= float(work.sum())
                consider(work)
        except Exception:
            pass

    work = best_arr.copy()
    work /= float(work.sum())

    # === Phase 3: Fine L-BFGS ===
    p3_end = min(hard_deadline - 300.0, time.monotonic() + 0.10 * budget)
    if p3_end > time.monotonic() + 15.0:
        try:
            work = _lse_lbfgs(work, p3_end, consider, coarse=False)
            if work is not None and work.size == n and work.sum() > 0:
                work = np.maximum(work, 0.0)
                work /= float(work.sum())
                consider(work)
        except Exception:
            pass

    work = best_arr.copy()
    work /= float(work.sum())

    # === Phase 4: More coordinate descent ===
    p4_end = min(hard_deadline - 100.0, t0 + 0.82 * budget)
    if p4_end > time.monotonic() + 20.0:
        try:
            work = _coord_descent_v2(work, p4_end, consider,
                                     max_sweeps=200000, grid_pts=36,
                                     seed=int(seed + 99999))
            if work is not None and work.size == n and work.sum() > 0:
                work = np.maximum(work, 0.0)
                work /= float(work.sum())
                consider(work)
        except Exception:
            pass

    work = best_arr.copy()
    work /= float(work.sum())

    # === Phase 5: Another short L-BFGS + coordinate ===
    rem = hard_deadline - time.monotonic()
    if rem > 60.0:
        p5_lse = min(hard_deadline - 40.0, time.monotonic() + 0.25 * rem)
        if p5_lse > time.monotonic() + 10.0:
            try:
                work = _lse_lbfgs(work, p5_lse, consider, coarse=False)
                if work is not None and work.size == n and work.sum() > 0:
                    work = np.maximum(work, 0.0)
                    work /= float(work.sum())
                    consider(work)
            except Exception:
                pass

        work = best_arr.copy()
        work /= float(work.sum())
        p5_cd = min(hard_deadline - 20.0, time.monotonic() + 0.5 * rem)
        if p5_cd > time.monotonic() + 15.0:
            try:
                work = _coord_descent_v2(work, p5_cd, consider,
                                         max_sweeps=100000, grid_pts=40,

                if work is not None and work.size == n and work.sum() > 0:
                    work = np.maximum(work, 0.0)
                    work /= float(work.sum())
                    consider(work)
            except Exception:

    # === Phase 6: Final fine coordinate polish ===
    work = best_arr.copy()

    if fin_end > time.monotonic() + 10.0:
        try:

    try:

        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)

            out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
