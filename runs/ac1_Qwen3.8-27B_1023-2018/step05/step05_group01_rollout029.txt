def _lp_phase(a, stop_time, max_iters=300, seed=42):
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0).copy()
    n = a.size
    s = float(a.sum())
    if s < 1e-12:
        return a
    a /= s

    c = _conv_full(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M

    rows, cols, nz = _build_conv_matrix_struct(n)

    t = 0.5
    rng = np.random.default_rng(seed)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 3.0:
            break

        A = _make_conv_matrix(a, rows, cols, nz)

        g = cp.Variable(n, non_negative=True)
        obj = cp.Maximize(cp.sum(g))
        constraints = [A @ g <= M]
        prob = cp.Problem(obj, constraints)

        solved = False
        for solver_name in ['GLOP', 'MOSEK', 'CBC', 'ECOS', 'CLARABEL']:
            try:
                solver = getattr(cp, solver_name, None)
                if solver is None:
                    continue
                prob.solve(solver=solver, verbose=False, time_limit=15)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            try:
                prob.solve(verbose=False, time_limit=15)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
            except Exception:
                pass

        if not solved or g.value is None:
            break

        g_val = np.asarray(g.value, dtype=np.float64).ravel()
        g_val = np.maximum(g_val, 0.0)
        sg = float(g_val.sum())
        if sg < 1e-12:
            break
        g_val /= sg

        # Try with various step sizes
        accepted = False
        for step in [t, 0.5 * t, 0.25 * t, 0.1 * t]:
            a_try = (1.0 - step) * a + step * g_val
            a_try = np.maximum(a_try, 0.0)
            st = float(a_try.sum())
            if st < 1e-12:
                continue
            a_try /= st
            c_try = _conv_full(a_try)
            M_try = float(c_try.max())
            if M_try <= M + 1e-14:
                a = a_try
                M = M_try
                accepted = True
                if M_try < best_M - 1e-15:
                    best_M = M_try
                    best_a = a_try.copy()
                    stall = 0
                else:
                    stall += 1
                break

        if not accepted:
            stall += 1
            if stall >= 15:
                # Perturb and restart

                a = np.maximum(a, 0.0)
                sa = float(a.sum())

                M = float(c.max())
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                stall = 0
                t = 0.3
            else:
                t = max(0.05, t * 0.8)

def _lse_optimize(p, stop_time, consider=None, seed=42):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = p.size
    s = float(p.sum())
    if s < 1e-12:
        return p
    p /= s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    def softmax_fn(uu):
        m = float(np.max(uu))
        e = np.exp(uu - m)
        ss = float(e.sum())
        if ss <= 0 or not math.isfinite(ss):
            return np.full(uu.size, 1.0 / uu.size)
        return e / ss

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(pa):
        F = rfft(pa, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        cc = irfft(F * G, cross_size)[:cross_len]
        q = cc[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = softmax_fn(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f_val = cmax
        else:
            f_val = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f_val):
                f_val = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f_val), grad

    stages = (
        (0.1, 0.10),
        (0.03, 0.12),
        (0.01, 0.13),
        (0.003, 0.13),
        (0.001, 0.14),
        (0.0003, 0.13),
        (0.00008, 0.12),
        (0.00002, 0.13),
    )

    start = time.monotonic()
    avail = max(0.0, float(stop_time) - start - 5.0)
    if avail < 10.0:
        return p

    best_p = p.copy()
    total_w = float(sum(w for _, w in stages))
    rng = np.random.default_rng(seed)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 2.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)
        best_u = u.copy()
        f0, _ = lse_func(best_u, T)
        if not math.isfinite(f0):
            continue
        best_f = f0
        stale = 0

        while time.monotonic() < end_stage - 0.5:
            opts = {"maxiter": 30, "gtol": 1e-12, "ftol": 1e-15,
                    "maxls": 20, "maxcor": 20, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-15:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            best_u = best_u + rng.normal(0, 0.005, best_u.size)
                            continue
            if time.monotonic() < end_stage - 1.0 and stale < 3:
                best_u = best_u + rng.normal(0, 0.002, best_u.size)
            else:
                break

        p_cur = softmax_fn(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score(p_cur)
        if math.isfinite(sc_cur) and sc_cur < _score(best_p):
            best_p = p_cur.copy()
            p = p_cur.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, G=None, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p /= s

    if G is None:
        if n <= 500:
            G = 48
        elif n <= 1000:
            G = 40
        elif n <= 2000:
            G = 32
        else:
            G = 24
    G = int(max(16, G))

    Ltot = 2 * n - 1
    c = _conv_full(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < 500000 and time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue
            base = c[start_i:end_i].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start_i > 0:
                ml = float(c[:start_i].max())
                if ml > max_const:
                    max_const = ml
            if end_i < Ltot:
                mr = float(c[end_i:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(20.0 * v, 40.0 * S / float(n), 0.02 * S, 1e-14)
            x_hi = min(x_hi, 8.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, S / float(n))

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag_i = i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[valid, None] * infl[None, :]
                dval = base[diag_i] + w[valid] * infl[diag_i] + w[valid] * w[valid]
                aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                dv = denom[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if G > 4 and best_x > 0 and best_x < x_hi:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 20, dtype=np.float64)
                if 0.0 <= best_x <= x_hi:
                    fine = np.concatenate((fine, [best_x]))
                    fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val - 1e-15:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 10.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start_i:end_i] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_full(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 8.0:
                    sigma = 0.02 + 0.02 * min(flat, 10) / 10.0
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    else:
                        break
                    c = _conv_full(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    flat = 0
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    return p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 25.0

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    best_score = _score(best_arr)

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = _score(best_arr)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        sa = float(a.sum())
        if sa < 0.01:
            return
        sc = _score(a)
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_arr = a.copy()
            best_score = sc

    n = int(best_arr.size)

    # Phase 1: LP iteration (main driver)
    lp_stop = min(hard_deadline - 300.0, t0 + 0.4 * (hard_deadline - t0))
    if lp_stop > t0 + 30.0:
        try:
            p_lp = _lp_phase(best_arr, lp_stop, max_iters=200, seed=int(seed))
            if p_lp is not None and p_lp.size == n and float(p_lp.sum()) > 0:
                p_lp = p_lp / float(p_lp.sum())
                consider(p_lp)
                best_arr = p_lp.copy()
        except Exception:
            pass

    # Phase 2: LSE optimization
    lse_stop = min(hard_deadline - 150.0, t0 + 0.65 * (hard_deadline - t0))
    if time.monotonic() < lse_stop - 20.0:
        try:
            p_lse = _lse_optimize(best_arr, lse_stop, consider=consider, seed=int(seed + 500))
            if p_lse is not None and p_lse.size == n and float(p_lse.sum()) > 0:
                p_lse = p_lse / float(p_lse.sum())
                consider(p_lse)
                best_arr = p_lse.copy()
        except Exception:
            pass

    # Phase 3: Second LP pass (refined)
    lp2_stop = min(hard_deadline - 100.0, t0 + 0.8 * (hard_deadline - t0))
    if time.monotonic() < lp2_stop - 15.0 and n <= 4096:
        try:
            p_lp2 = _lp_phase(best_arr, lp2_stop, max_iters=100, seed=int(seed + 1000))
            if p_lp2 is not None and p_lp2.size == n and float(p_lp2.sum()) > 0:
                p_lp2 = p_lp2 / float(p_lp2.sum())
                consider(p_lp2)
        except Exception:
            pass

    # Phase 4: Coordinate descent (final refinement)
    coord_stop = min(hard_deadline - 20.0, t0 + budget - 40.0)
    if time.monotonic() < coord_stop - 10.0:
        try:
            p_cd, _ = _coord_polish(best_arr, coord_stop, seed=int(seed + 7777),
                                    consider=consider)
            if p_cd is not None and p_cd.size == n and float(p_cd.sum()) > 0:
                p_cd = p_cd / float(p_cd.sum())
                consider(p_cd)
        except Exception:
            pass

    # Final validation using exact evaluator
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.asarray(parent, dtype=np.float64).ravel()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
