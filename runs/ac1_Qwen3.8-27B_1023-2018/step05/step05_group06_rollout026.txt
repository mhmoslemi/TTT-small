```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len

def _conv_self(a):
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _exact_score(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(p, p)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))
```

```python
import math
```
