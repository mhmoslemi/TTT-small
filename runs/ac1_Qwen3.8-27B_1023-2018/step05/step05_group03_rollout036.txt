import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize

def _fft_conv(a):
    """Compute full convolution a*a using FFT."""
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    """Compute the score 2*n*max(conv)/sum^2."""
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return math.inf
    c = _fft_conv(a)
    m = c.max()
    return 2.0 * n * m / (s * s)

def _normalize(a):
    """Normalize to sum=1, clip to [0, 1000]."""
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, 1.0/a.size)
    return a / s

                continue

            base = c[start:end].copy()
            # Influence vector: 2*a[k-i] for k in [start, end)
            infl = 2.0 * p[:n] if start == 0 else 2.0 * p[0:end - 0 if False else n]
            # Actually: for k in [i, i+n-1], infl[k-i] = 2*p[k-i]... wait
            # conv[k] affected = conv[k] + 2*(x-v)*p[k-i] for k != 2i
            # k ranges from i to i+n-1, so k-i ranges from 0 to n-1
            infl_vec = 2.0 * p[0:n]  # length n
            # But we only need the portion from start to end
            # k-i for k in [start, end) is [start-i, end-i) = [0, min(n, 2n-1-i))
            # Since start=i and end=min(i+n, 2n-1), k-i goes from 0 to min(n, 2n-1-i)-1
            # infl_vec relevant: p[0:min(n, end-start)]
            infl_slice = 2.0 * p[:Ls]  # p[k-i] for k-i = 0 to Ls-1

            # Max of unaffected positions
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < L:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

            di = 2 * i - start  # position of k=2i within [start, end)
            if di < 0 or di >= Ls:
                di = -1

            # Trial points
            xhi = max(20.0 * v, 5.0 * S / n, 0.01)
            xhi = min(xhi, max(0.5, 2.0 * v + 0.5 * S / n))
            xhi = min(xhi, 0.5)  # cap: no single element > 50% of sum

            pts = [0.0, v]
            if v > 1e-12:
                for fac in (0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 3.0):
                    xv = v * fac
                    if 0.0 <= xv <= xhi:
                        pts.append(xv)
            # Linear spacing
            linhi = min(xhi, max(2.0 * S / n, 0.005))
            if linhi > 0:
                pts.extend(np.linspace(0.0, linhi, 20, dtype=np.float64).tolist())
            # Geometric spacing
            if v > 1e-12:
                lo_g = max(1e-12, v * 0.05)
                hi_g = min(xhi, v * 20.0)
                if hi_g > lo_g:
                    pts.extend(np.geomspace(lo_g, hi_g, 20, dtype=np.float64).tolist())
            # Peak-equalizing: find x that makes the affected max = max_const
            if max_const > 0 and Ls > 0:
                # Approximate: at the currently-maximizing affected position
                im_aff = int(np.argmax(base))
                bmax = float(base[im_aff])
                if bmax > max_const:
                    # We want base[im] + d*infl[im] = max_const => d = (max_const - bmax)/infl[im]
                    if abs(infl_slice[im_aff]) > 1e-15:
                        d_eq = (max_const - bmax) / infl_slice[im_aff]
                        x_eq = v + d_eq
                        if 0.0 < x_eq < xhi:
                            pts.append(x_eq)

            pts = np.unique(np.clip(np.asarray(pts, dtype=np.float64), 0.0, xhi))
            if pts.size == 0:
                pts = np.array([v], dtype=np.float64)

            # Evaluate
            w = pts - v
            den = S + w
            valid = den > 1e-14
            aff = base[None, :] + w[:, None] * infl_slice[None, :]
            if 0 <= di < Ls:
                aff[:, di] = base[di] + w * infl_slice[di] + w * w
            np.maximum(aff, max_const, out=aff)
            mv = np.max(aff, axis=1)
            vals = np.full(pts.size, math.inf, dtype=np.float64)
            if np.any(valid):
                vals[valid] = (2.0 * n * mv[valid]) / (den[valid] * den[valid])
            vals = np.where(np.isfinite(vals), vals, math.inf)

            if not np.any(np.isfinite(vals)):
                continue
            im = int(np.argmin(vals))
            best_x = float(pts[im])
            best_val = float(vals[im])

            # Local refinement around best
            if best_val < cur_score - 1e-14:
                step = max(v * 0.2, S / (10.0 * n), 1e-10)
                lo_r = max(0.0, best_x - 2.0 * step)
                hi_r = min(xhi, best_x + 2.0 * step)
                if hi_r > lo_r:
                    fine = np.linspace(lo_r, hi_r, 24, dtype=np.float64)
                    if lo_r > 0:
                        fine = np.concatenate([fine, np.geomspace(max(lo_r,1e-15), hi_r, 12)])
                    w2 = fine - v
                    den2 = S + w2
                    valid2 = den2 > 1e-14
                    aff2 = base[None, :] + w2[:, None] * infl_slice[None, :]
                    if 0 <= di < Ls:
                        aff2[:, di] = base[di] + w2 * infl_slice[di] + w2 * w2
                    np.maximum(aff2, max_const, out=aff2)
                    mv2 = np.max(aff2, axis=1)
                    vals2 = np.full(fine.size, math.inf, dtype=np.float64)
                    if np.any(valid2):
                        vals2[valid2] = (2.0 * n * mv2[valid2]) / (den2[valid2] * den2[valid2])
                    vals2 = np.where(np.isfinite(vals2), vals2, math.inf)
                    if np.any(np.isfinite(vals2)):
                        im2 = int(np.argmin(vals2))
                        if float(vals2[im2]) < best_val:
                            best_x = float(fine[im2])
                            best_val = float(vals2[im2])

            if (math.isfinite(best_val) and best_val < cur_score - 1e-12
                    and 0.0 <= best_x <= xhi):
                d = best_x - v
                if abs(d) > 1e-16:
                    c[start:end] += d * infl_slice
                    diag = 2 * i
                    if 0 <= diag < L:
                        c[diag] += d * d
                    p[i] = best_x
                    S += d
                    cur_score = best_val
                    improved = True

        # Renormalize
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        c = np.maximum(c, 0.0)

        desc = float(2.0 * n * c.max())
        if math.isfinite(desc) and desc < best_score - 1e-13:
            best_score = desc
            best_p = p.copy()
            if consider is not None:
                consider(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                # Perturbation strategies
                if time.monotonic() < stop_time - 20.0:
                    level = min(flat - 3, 4)
                    if level == 0:
                        # Small multiplicative noise
                        sigma = 0.02
                        noise = rng.normal(0.0, sigma, n)
                        q = p * np.exp(noise)
                    elif level == 1:
                        # Medium noise
                        sigma = 0.06
                        noise = rng.normal(0.0, sigma, n)
                        q = p * np.exp(noise)
                    elif level == 2:
                        # Targeted mass transfer from peak contributors
                        peak_pos = int(np.argmax(c))
                        # Find top contributors to the peak
                        # Peak at position k*: contributions from pairs (i, k*-i)
                        # Transfer from high-a[i] at positions contributing to peak
                        indices = np.arange(n)
                        if 0 <= peak_pos <= 2*n-2:
                            # Positions i such that i or peak_pos-i contributes
                            contrib = np.zeros(n)
                            for i in range(n):
                                j = peak_pos - i
                                if 0 <= j < n:
                                    contrib[i] = p[i] * p[j]
                            top_mask = contrib > 0.5 * contrib.max()
                            if np.any(top_mask):
                                # Move some mass from top to bottom
                                bottom_mask = contrib < 0.1 * (contrib.max() + 1e-30)
                                q = p.copy()
                                total_move = 0.01 * float(p[top_mask].sum())
                                if np.any(bottom_mask):
                                    n_bottom = int(bottom_mask.sum())
                                    q[top_mask] -= total_move * p[top_mask] / max(1e-30, p[top_mask].sum())
                                    q[bottom_mask] += total_move * (p[bottom_mask] + 1e-10) / max(1e-30, (p[bottom_mask] + 1e-10).sum())
                        else:
                            q = p * np.exp(rng.normal(0, 0.1, n))
                    elif level == 3:
                        # Local window reshaping
                        wlen = max(8, n // 32)
                        wstart = int(rng.integers(0, n - wlen))
                        window = p[wstart:wstart+wlen].copy()
                        # Apply a smooth perturbation
                        t = np.linspace(0, 1, wlen)
                        amp = 0.1 + 0.15 * float(rng.random())
                        shape = 1.0 + amp * np.sin(np.pi * t) ** int(rng.integers(1, 5))
                        q = p.copy()
                        q[wstart:wstart+wlen] = window * shape
                    else:
                        # Large structural perturbation
                        x = (np.arange(n) + 0.5) / n
                        f1 = int(rng.integers(1, 6))
                        f2 = int(rng.integers(1, 6))
                        ph1 = 2.0 * math.pi * float(rng.random())
                        ph2 = 2.0 * math.pi * float(rng.random())
                        amp = 0.15 + 0.25 * float(rng.random())
                        mod = 1.0 + amp * math.sin(2*math.pi*f1*x + ph1) ** 2 - amp * math.sin(2*math.pi*f2*x + ph2) ** 2
                        q = p * mod

                    q = np.maximum(q, 0.0)
                    qs = float(q.sum())
                    if math.isfinite(qs) and qs > 1e-15:
                        q /= qs
                        c = _fft_conv(q)
                        c = np.maximum(c, 0.0)
                        S = 1.0
                        p = q
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider is not None:
                                consider(best_p)
                    flat = 0
                else:
                    break
        sweep += 1

    if consider is not None:
        consider(best_p)
    return best_p, best_score

def _subproblem_roundtrip(n_target, src, stop_time, n_small=128):
    """Solve at lower resolution, upsample, and return the result."""
    if time.monotonic() > stop_time - 30:
        return None
    # Downsample
    m = src.size
    xp = (np.arange(m) + 0.5) / m
    xp_new = (np.arange(n_small) + 0.5) / n_small
    small = np.interp(xp_new, xp, src)
    small = _normalize(small)

    # Optimize at small scale (fast)
    sub_stop = min(stop_time - 25.0, time.monotonic() + 30.0)
    small, _ = _coord_polish(small, sub_stop, consider=None, seed=999, G=48)

    # Upsample back
    xp2 = (np.arange(n_small) + 0.5) / n_small
    xp2_new = (np.arange(n_target) + 0.5) / n_target
    upsampled = np.interp(xp2_new, xp2, small)
    upsampled = _normalize(upsampled)
    return upsampled

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(10.0, budget)
    reserve = 20.0 if budget > 200 else 8.0
    hard = t0 + budget - reserve

    parent = _get_parent()
    n_target = int(parent.size)

    best_arr = _normalize(parent.copy())
    best_score = _score_norm(best_arr)

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
            if arr.size == 0:
                return
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            s = float(arr.sum())
            if not math.isfinite(s) or s < 0.01:
                return
            arr = arr / s
            sc = _score_norm(arr)
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_arr = arr.copy()
                best_score = sc
        except Exception:
            pass

    now = time.monotonic()
    remaining = max(0.0, hard - now)

    # Phase 1: Quick LSE to improve starting point
    lse_time = min(remaining * 0.15, 120.0)
    if lse_time > 20.0:
        lse_stop = now + lse_time
        try:
            p1 = _lse_optimize(best_arr, lse_stop, consider=consider)
            best_arr = p1
        except Exception:
            pass

    # Phase 2: Subproblem roundtrip for a better initialization
    now = time.monotonic()
    remaining = max(0.0, hard - now)
    if remaining > 100.0:
        for ns in (96, 128):
            if time.monotonic() > hard - 80.0:
                break
            try:
                sub = _subproblem_roundtrip(n_target, best_arr, hard - 60.0, ns)
                if sub is not None:
                    consider(sub)
                    # Blend with current best
                    for alpha in (0.3, 0.5, 0.7):
                        blend = (1 - alpha) * best_arr + alpha * sub
                        blend = _normalize(blend)
                        consider(blend)
            except Exception:
                pass

    # Phase 3: Main coordinate descent
    now = time.monotonic()
    remaining = max(0.0, hard - now)
    if remaining > 30.0:
        coord_stop = hard - 10.0
        try:
            p2, _ = _coord_polish(best_arr, coord_stop, consider=consider,
                                  seed=seed, G=72)
            consider(p2)
        except Exception:
            pass

    # Phase 4: Final LSE squeeze
    now = time.monotonic()
    remaining = max(0.0, hard - now)
    if remaining > 40.0:
        lse_stop = min(hard - 5.0, now + min(60.0, remaining * 0.5))
        try:
            p3 = _lse_optimize(best_arr, lse_stop, consider=consider)
            consider(p3)
        except Exception:
            pass

    # Phase 5: Final coordinate polish
    now = time.monotonic()
    remaining = max(0.0, hard - now)
    if remaining > 20.0:
        try:
            p4, _ = _coord_polish(best_arr, hard - 3.0, consider=consider,
                                  seed=seed + 7777, G=80)
            consider(p4)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = np.full(64, 1.0 / 64.0, dtype=np.float64)
            s = float(out.sum())
        out = out / s
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls2 = [float(x) for x in _normalize(parent)]
            sc2 = float(evaluate_sequence(ls2))
            if math.isfinite(sc2):
                return ls2
            return [1.0]
        return ls
    except Exception:
        return [1.0]
