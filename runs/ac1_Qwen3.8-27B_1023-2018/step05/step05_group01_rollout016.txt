    n = a.size

def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))

    out = np.maximum(fp, 0.0)
    s = float(out.sum())

    return out / s

                    break
            else:
                stale += 1
                if stale >= 3:
                    break

        p_cur = _softmax_fn(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        else:
            # Try keeping current if it's close
            if math.isfinite(sc_cur) and sc_cur < best_sc + 0.003:
                p = p_cur.copy()
            else:
                p = best_p.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, G=None, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    if G is None:
        if n <= 500:
            G = 56
        elif n <= 1000:
            G = 48
        elif n <= 2500:
            G = 40
        elif n <= 5000:
            G = 34
        else:
            G = 28
    G = int(max(16, G))

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    n_evals = 0
    max_sweeps = 300000

    while sweep < max_sweeps and time.monotonic() < stop_time - 0.8:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(25.0 * v, 50.0 * S / float(n), 0.03 * S)
            x_hi = min(x_hi, 10.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, S / float(n))

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag_i = i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[valid, None] * infl[None, :]
                dval = base[diag_i] + w[valid] * infl[diag_i] + w[valid] * w[valid]
                aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                dv = denom[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            nonlocal n_evals
            vals = eval_xs(pts)
            n_evals += len(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Fine refinement
            if G > 6 and 0 < best_x < x_hi:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.5 * dw)
                hi = min(x_hi, best_x + 2.5 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 32, dtype=np.float64)
                if 0.0 <= best_x <= x_hi:
                    fine = np.concatenate((fine, [best_x]))
                    fine = np.unique(fine)
                vals2 = eval_xs(fine)
                n_evals += len(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val - 1e-14:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-12 and 0.0 <= best_x <= 10.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        # End of sweep
        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0

        # Recompute convolution from scratch every sweep for numerical accuracy
        c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 5.0:
                    sigma = 0.015 + 0.01 * min(flat, 8) / 8.0
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    else:
                        p = best_p.copy()
                        c = _conv_self(p)
                    c = _conv_self(p)
                    S_total = float(p.sum())
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    flat = 0
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        sc = float(_score_fft(p))
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 2.0:
        hard = t0 + max(2.0, budget * 0.9)
    final_reserve = max(10.0, 0.05 * (hard - t0))
    phase_limit = hard - final_reserve

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    def run_full_opt(n, init_arr, end_time, sd, do_lse=True, do_coord=True, lse_frac=0.55):
        nonlocal best_arr, best_score
        p = _resample_mid(init_arr, n)
        if p.size != n or float(p.sum()) <= 0:
            p = np.full(n, 1.0 / n)
        p = p / float(p.sum())
        consider(p)

        if time.monotonic() > end_time - 8.0:
            return
        phase_len = max(0.0, float(end_time) - time.monotonic())

        if do_lse and phase_len > 25.0:
            lse_stop = min(float(end_time) - max(15.0, (1.0 - lse_frac) * phase_len),
                           time.monotonic() + lse_frac * phase_len)
            if lse_stop > time.monotonic() + 8.0:
                try:
                    p2 = _lse_optimize(p, lse_stop, consider=consider, seed=sd)
                    if p2 is not None and p2.size == n and float(p2.sum()) > 0:
                        p = p2 / float(p2.sum())
                        consider(p)
                except Exception:
                    pass

        rem = float(end_time) - time.monotonic()
        if do_coord and rem > 12.0:
            cstop = min(float(end_time) - 1.0, time.monotonic() + max(5.0, rem - 1.0))
            if cstop > time.monotonic() + 8.0:
                try:
                    p3, _ = _coord_polish(p, cstop, seed=sd, consider=consider)
                    if p3 is not None and p3.size == n and float(p3.sum()) > 0:
                        p = p3 / float(p3.sum())
                        consider(p)
                except Exception:
                    pass
        return p

    n_parent = int(best_arr.size)

    # Phase 1: Main optimization at parent length
    p1_end = min(phase_limit, t0 + 0.50 * (phase_limit - t0))
    run_full_opt(n_parent, best_arr, p1_end, int(seed + 100), do_lse=True, do_coord=True)

    # Phase 2: Perturb and re-optimize
    if time.monotonic() < phase_limit - 50.0:
        rng2 = np.random.default_rng(seed + 777)
        for pr in range(3):
            if time.monotonic() > phase_limit - 30.0:
                break
            sigma_perturb = 0.03 + 0.01 * pr
            pert = best_arr * np.exp(rng2.normal(0, sigma_perturb, best_arr.size))
            pert = np.maximum(pert, 0.0)
            ps = float(pert.sum())
            if ps > 0:
                pert /= ps
            sub_end = min(phase_limit - 25.0, time.monotonic() + 0.30 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 20.0:
                run_full_opt(n_parent, pert, sub_end, int(seed + 2000 + pr * 100),
                             do_lse=True, do_coord=True, lse_frac=0.50)

    # Phase 3: Asymmetric initialization
    if time.monotonic() < phase_limit - 40.0:
        for init_seed in range(2):
            if time.monotonic() > phase_limit - 25.0:
                break
            rng3 = np.random.default_rng(seed + 5000 + init_seed * 777)
            x = (np.arange(n_parent, dtype=np.float64) + 0.5) / float(n_parent)
            side = 1.0 if rng3.random() < 0.5 else -1.0
            amp = 0.2 + 0.8 * float(rng3.random())
            curv = 0.3 + 1.5 * float(rng3.random())
            init = best_arr * np.exp(side * amp * (2.0 * x - 1.0) + curv * (2.0 * x - 1.0) ** 2)
            init = np.maximum(init, 0.0)
            is_sum = float(init.sum())
            if is_sum > 0:
                init /= is_sum
            else:
                init = best_arr.copy()
            sub_end = min(phase_limit - 20.0, time.monotonic() + 0.20 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 15.0:
                run_full_opt(n_parent, init, sub_end, int(seed + 6000 + init_seed * 333),
                             do_lse=True, do_coord=True, lse_frac=0.45)

    # Phase 4: Try a slightly different length
    if time.monotonic() < phase_limit - 60.0:
        n_alt = n_parent // 2 if n_parent > 1024 else n_parent * 2
        n_alt = int(max(256, min(n_alt, 6144)))
        if n_alt != n_parent:
            sub_end = min(phase_limit - 15.0, time.monotonic() + 0.25 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 25.0:
                run_full_opt(n_alt, best_arr, sub_end, int(seed + 8000),
                             do_lse=True, do_coord=True, lse_frac=0.50)

    # Phase 5: Final coordinated squeeze on the best
    final_end = min(hard - 10.0, t0 + budget - 10.0)
    if final_end > time.monotonic() + 20.0 and best_arr.size > 0:
        nbest = int(best_arr.size)
        rem = final_end - time.monotonic()

        # Final LSE push at very low temperature
        if rem > 50.0 and nbest <= 5000:
            lstop = min(final_end - 25.0, time.monotonic() + 0.35 * rem)
            if lstop > time.monotonic() + 10.0:
                try:
                    p2 = _lse_optimize(best_arr, lstop, consider=consider, seed=int(seed + 99999))
                    if p2 is not None and p2.size == nbest and float(p2.sum()) > 0:
                        p2 = p2 / float(p2.sum())
                        consider(p2)
                except Exception:
                    pass

        # Final coordinate descent
        cstop = min(final_end - 3.0, time.monotonic() + max(10.0, 0.5 * (final_end - time.monotonic())))
        if cstop > time.monotonic() + 8.0:
            try:
                p3, _ = _coord_polish(best_arr, cstop, seed=int(seed + 123456789), consider=consider)
                if p3 is not None and p3.size == nbest and float(p3.sum()) > 0:
                    p3 = p3 / float(p3.sum())
                    consider(p3)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
