```python
def _build_lp_mat(f, n):
    # Column i has f[0..n-1] at rows i..i+n-1 (clipped to 2n-2)
    i_idx = np.arange(n)
    # For column i, the rows are i, i+1, ..., min(i+n-1, 2n-2)
    # Length of column i: min(n, 2n-1-i)
    lens = np.minimum(n, 2*n-1-i_idx)
    # Total entries
    total = int(lens.sum())
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    data = np.empty(total, dtype=np.float64)
    offset = 0
    for i in range(n):
        l = int(lens[i])
        rows[offset:offset+l] = np.arange(i, i+l)
        cols[offset:offset+l] = i
        data[offset:offset+l] = f[:l]
        offset += l
    return sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
```

This is O(n^2) in Python but with small constants. For n=3072, ~3072 iterations with vectorized inner ops. Should take < 1 second.
