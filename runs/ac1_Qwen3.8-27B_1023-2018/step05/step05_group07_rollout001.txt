                         options={'maxiter': max_iter, 'gtol': 1e-8,
                                  'maxls': 50, 'maxfun': max_iter * 10})
        
        if result.success or hasattr(result, 'x'):
            z_new = result.x
            a_new = np.exp(z_new - z_new.max())
            a_new = a_new / a_new.sum()
            a_new = np.clip(a_new, 0, 1000)
            sc = _score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = a_new
            z = np.log(a + 1e-30)
    
    return best_a, best_score

def _lp_game(a, deadline, max_iter=200, tol=1e-10):
    """LP game iteration."""
    n = len(a)
    a = np.clip(a, 0, 1000).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(a, a)
        M = c.max()
        S = a.sum()
        score = 2 * n * M / (S * S)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Build convolution matrix: (2n-1) x n
        # Row k, col j: a[k-j] if 0<=k-j<n else 0
        k_idx = np.arange(2*n-1)
        j_idx = np.arange(n)
        d = k_idx[:, None] - j_idx[None, :]
        mask = (d >= 0) & (d < n)
        A = np.zeros((2*n-1, n))
        A[mask] = a[d[mask]]
        
        # LP: max sum(g) s.t. A @ g <= M, g >= 0
        c_lp = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        
        try:
            res = linprog(c_lp, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                         method='highs', options={'time_limit': 3.0})
            if res.status == 0 and res.x is not None:
                g = np.clip(res.x, 0, None)
                g_sum = g.sum()
                if g_sum > 1e-12:
                    g = g / g_sum
                    if np.max(np.abs(g - a)) > 1e-10:
                        # Line search
                        best_t, best_v = 0.0, M
                        for t in np.linspace(0.1, 1.0, 20):
                            cand = (1-t)*a + t*g
                            cand = np.maximum(cand, 0)
                            cs = cand.sum()
                            if cs < 1e-12:
                                continue
                            cand = cand / cs
                            v = np.convolve(cand, cand).max()
                            if v < best_v:
                                best_v = v
                                best_t = t
                        if best_t > 0 and best_v < M * (1 - 1e-8):
                            a = (1-best_t)*a + best_t*g
                            a = np.maximum(a, 0)
                            a = a / a.sum()
                    else:
                        break
                else:
                    break
            else:
                break
        except Exception:
            break
    
    return best_a, best_score

def _local_search(a, deadline, max_iter=5000):
    """Coordinate descent / local search."""
    n = len(a)
    a = np.clip(a, 0, 1000).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng(123)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(a, a)
        M = c.max()
        score = 2 * n * M / a.sum()**2
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Pick a random pair of coordinates to modify
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Try to transfer mass from i to j (or vice versa)
        for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
            for direction in [1, -1]:
                d = frac * direction * min(a[i], a[j])
                cand = a.copy()
                cand[i] -= d
                cand[j] += d
                cand = np.maximum(cand, 0)
                cs = cand.sum()
                if cs < 1e-12:
                    continue
                cand = cand / cs
                v = np.convolve(cand, cand).max()
                if v < M * (1 - 1e-10):
                    a = cand
                    break
            if a is not cand:  # if we moved
                break
    
    return best_a, best_score
