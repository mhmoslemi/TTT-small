def _get_parent():
    try:
        h = globals()["height_sequence_1"]

def _conv_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fft(a):
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_fft(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _softmax_fn(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _p_to_u(p):
    return np.log(np.maximum(p, 1e-300)) - float(np.mean(np.log(np.maximum(p, 1e-300))))


def _lse_optimize(p, stop_time, consider=None, seed=42):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    u = _p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax_fn(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    stages = (
        (0.10, 0.10),
        (0.04, 0.12),
        (0.012, 0.13),
        (0.004, 0.14),
        (0.001, 0.15),
        (0.0003, 0.14),
        (0.00008, 0.12),
        (0.00002, 0.10),
    )

    start = time.monotonic()
    avail = max(0.0, float(stop_time) - start - 5.0)
    if avail <= 5.0:
        return p

    best_p = p.copy()
    best_sc = _score_fft(p)
    total_w = float(sum(w for _, w in stages))
    rng = np.random.default_rng(seed)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 2.5, start + avail * (cum / total_w))
        if end_stage <= now + 1.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)

        best_u = u.copy()
        f0, _ = lse_func(best_u, T)
        if not math.isfinite(f0):
            continue
        best_f = f0
        stale = 0

        while time.monotonic() < end_stage - 0.3:
            opts = {"maxiter": 30, "gtol": 1e-12, "ftol": 1e-16,
                    "maxls": 25, "maxcor": 25, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-16:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, "status", 1) == 0 and stale < 3:
                best_u = best_u + rng.normal(0, 0.002, best_u.size)

        p_cur = _softmax_fn(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif math.isfinite(sc_cur) and sc_cur < best_sc + 0.01:
            p = p_cur.copy()
        else:
            p = best_p.copy()
        u = _p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _lp_direction(f, M, timeout_s=30.0):
    """Solve LP: max sum(b) s.t. C @ b <= M*1, b >= 0. Returns b or None."""
    n = f.size
    rows = 2 * n - 1
    # Build sparse constraint matrix C[k,j] = f[k-j] if 0<=k-j<n else 0
    row_idx = []
    col_idx = []
    vals = []
    for j in range(n):
        for k in range(j, min(j + n, rows)):
            row_idx.append(k)
            col_idx.append(j)
            vals.append(f[k - j])
    C = csr_matrix((vals, (row_idx, col_idx)), shape=(rows, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(rows)
    bounds = [(0, None)] * n
    t_start = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': max(1.0, timeout_s)})
    except Exception:
        return None
    if res.status == 0 and res.x is not None:
        b = np.maximum(res.x, 0.0)
        if float(b.sum()) > 1e-12:
            return b
    return None


def _lp_update_iter(p, n_lp_iters, stop_time, consider=None, seed=0):
    """Apply LP-based direction updates."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = p.size
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    best_p = p.copy()
    best_sc = _score_fft(p)
    rng = np.random.default_rng(seed)

    for it in range(n_lp_iters):
        if time.monotonic() > stop_time - 5.0:
            break
        c = _conv_fft(p)
        M = float(c.max())
        if M <= 0:
            break
        # Solve LP
        t0 = time.monotonic()
        b = _lp_direction(p, M, timeout_s=min(25.0, stop_time - time.monotonic() - 10.0))
        lp_time = time.monotonic() - t0
        if b is None:
            break
        b = b / float(b.sum())
        # Line search over t
        best_t = 0.0
        best_trial_sc = best_sc
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            trial = (1 - t) * p + t * b
            trial = np.maximum(trial, 0.0)
            ts = float(trial.sum())
            if ts < 1e-12:
                continue
            trial = trial / ts
            sc = _score_fft(trial)
            if math.isfinite(sc) and sc < best_trial_sc:
                best_trial_sc = sc
                best_t = t
        if best_t > 0:
            p_new = (1 - best_t) * p + best_t * b
            p_new = np.maximum(p_new, 0.0)
            ps = float(p_new.sum())
            if ps > 1e-12:
                p_new = p_new / ps
            p = p_new
            sc = _score_fft(p)
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_p = p.copy()
                if consider is not None:
                    try:
                        consider(best_p)
                    except Exception:
                        pass
        elif it < n_lp_iters - 1:
            # If LP direction didn't help, perturb
            p = p * np.exp(rng.normal(0, 0.01, n))
            p = np.maximum(p, 0.0)
            p = p / float(p.sum())

    return best_p


def _coord_polish(p, stop_time, G=None, seed=12345, consider=None):

    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()

            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start > 0:

            diag_i = i

            def eval_xs(xs):

                w = xs - v

                    return vals

                    if ss > 0:

def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        return src.copy()
    m = src.size
    xp = (np.arange(m) + 0.5) / m
    fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
    out = np.maximum(fp, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):

        best_arr = np.ones(64) / 64.0
        best_score = float(_score_fft(best_arr))

    n_best = int(best_arr.size)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)

                return
            s = float(a.sum())
            if s < 0.01:
                return

            if math.isfinite(sc) and sc < best_score - 1e-14:
                a = a / s
                best_arr = a.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 1: LSE optimization at parent length
    p = best_arr.copy()
    lse_end = min(phase_limit * 0.25, t0 + 0.25 * (phase_limit - t0))
    if lse_end > t0 + 15.0:
        try:
            p2 = _lse_optimize(p, lse_end, consider=consider, seed=int(seed + 100))
            if p2 is not None and p2.size == n_best and float(p2.sum()) > 0:
                p = p2 / float(p2.sum())
