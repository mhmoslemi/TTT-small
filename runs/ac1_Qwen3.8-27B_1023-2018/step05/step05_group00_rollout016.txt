Actually wait - the flat spectrum approach has an issue. When we do F/|F| * T, we're keeping the phase of F and setting the magnitude to T. But T should be an array of the right size. The magnitude of F has (2n-1)/2 + 1 real FFT bins. We want |F[k]|^2 to be as flat as possible, so we set |F'[k]| = sqrt(mean(|F|^2)) for all k.

Let me code this:

```python
def flat_spectrum_step(a, n_iter=100):
    n = len(a)
    size = next_fast_len(2*n-1)
    for _ in range(n_iter):
        F = rfft(a, size)
        # Keep DC component, flatten the rest
        mag = np.abs(F)
        # Target magnitude: constant for all bins except DC
        # Parseval: sum(a^2) = (1/size) * sum(|F|^2)... 
        # Actually for rfft: sum(a^2) = (1/size) * (|F[0]|^2 + 2*sum(|F[1:]|^2))
        # We want to flatten |F[1:]|
        target_mag = np.mean(mag[1:])
        F_new = F.copy()
        for k in range(1, len(F)):
            if mag[k] > 0:
                F_new[k] = F[k] / mag[k] * target_mag
        F_new[0] = F[0]  # Keep DC (sum)
        a_new = irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 0:
            a_new /= s
        a = a_new
    return a
```

This is O(n log n) per iteration, so for n=3072 and 100 iterations, it's about 100 * 3072 * log(3072) ≈ 100 * 3072 * 12 ≈ 3.7M operations. Very fast.

def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()

        return math.inf

    return float(2.0 * a.size * m / (s * s))


def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    return e / s


def _p_to_u(p):

def _lbfgs_optimize(p, stop_time, consider=None, fine_only=False):

    conv_len = 2 * n - 1

        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], conv_size)
        G = rfft(ww, conv_size)

        np.maximum(q, 0.0, out=q)

            w = np.zeros(c.size, dtype=np.float64)

        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    def run_chunk(uu, T, end_stage):
