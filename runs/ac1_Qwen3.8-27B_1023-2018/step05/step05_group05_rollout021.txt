    # c: current convolution values (length 2n-1)
    # a: current sequence (length n)

    k_max = min(L-1, i + n - 1)

    if k_min > 0 or k_max < L-1:

        M_inactive = 0.0

    offsets = ks - i  # this is k-i, ranges from k_min-i to k_max-i
