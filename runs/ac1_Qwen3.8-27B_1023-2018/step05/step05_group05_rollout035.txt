def _lse_optimize(p, stop_time, consider=None):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p
    p = p / s

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2

    def _softmax(uu):
        uu = np.asarray(uu, dtype=np.float64)
        if not np.all(np.isfinite(uu)):
            return np.full(uu.size, 1.0 / uu.size)
        m = float(np.max(uu))
        e = np.exp(uu - m)
        ss = float(e.sum())
        if not math.isfinite(ss) or ss <= 0:
            return np.full(uu.size, 1.0 / uu.size)
        return e / ss

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return np.maximum(c, 0.0)

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        return np.maximum(q, 0.0)

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = 1e-4 * cmax
        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -60, 60)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        return float(f), np.where(np.isfinite(grad), grad, 0.0)

    def p_to_u(pp):
        ll = np.log(np.maximum(pp, 1e-300))
        return ll - float(np.mean(ll))

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = 2.0 * n * float(cfast(p).max())

    stages = [
        (0.15, 25.0), (0.07, 25.0), (0.03, 30.0),
        (0.012, 30.0), (0.005, 25.0), (0.002, 25.0), (0.0008, 20.0),
    ]

    for frac, stage_budget in stages:
        if time.monotonic() > stop_time - 15.0:
            break
        stage_end = min(stop_time - 12.0, time.monotonic() + stage_budget)
        cmax = float(cfast(p).max())
        T = max(1e-15, frac * cmax)
        stale = 0
        while time.monotonic() < stage_end and stale < 5:
            opts = {"maxiter": 50, "gtol": 1e-11, "ftol": 1e-16,
                    "maxls": 50, "maxcor": 30, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                p_cur = _softmax(res.x)
                c_cur = cfast(p_cur)
                sc_cur = 2.0 * n * float(c_cur.max())
                if math.isfinite(sc_cur) and sc_cur < best_sc - 1e-11:
                    best_sc = sc_cur
                    best_p = p_cur.copy()
                    p = p_cur.copy()
                    u = p_to_u(p)
                    stale = 0
                    if consider:
                        try:
                            consider(p_cur.copy())
                        except Exception:
                            pass
                else:
                    stale += 1
            if getattr(res, "status", 1) == 0:
                break

    return best_p


def _coord_descent(p, stop_time, consider=None, seed=42, verbose=False):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p, math.inf
    p = p / s

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S = 1.0
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    flat = 0
    sweep = 0
    max_sweeps = 50000

    while sweep < max_sweeps and time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)
        sweep_start = time.monotonic()

        for ii in range(n):
            i = int(order[ii])
            if time.monotonic() > stop_time - 1.0:
                break
            if S <= 1e-300:
                break

            v = float(p[i])
            if v < 1e-12:
                # Try setting to a small positive value
                continue

            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end]
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start > 0:
                max_const = max(max_const, float(np.max(c[:start])))
            if end < Ltot:
                max_const = max(max_const, float(np.max(c[end:])))

            diag_idx = i  # index within base where k=2i

            # Evaluate function at several candidate w values
            def eval_w(w):
                if S + w < 1e-12:
                    return math.inf
                vals = base + w * infl
                vals = vals.copy()
                vals[diag_idx] += w * w
                mv = max(float(vals.max()), max_const)
                ns = S + w
                return 2.0 * n * mv / (ns * ns)

            # Coarse candidates
            candidates = [0.0, -0.999 * v, -0.7 * v, -0.4 * v, -0.1 * v,
                          0.2 * v, 0.5 * v, 1.0 * v]
            best_w = 0.0
            best_val = cur_score
            for wc in candidates:
                if abs(wc) < 1e-15:
                    val = cur_score
                else:
                    val = eval_w(wc)
                if val < best_val - 1e-14:
                    best_val = val
                    best_w = wc

            if best_val >= cur_score - 1e-13:
                continue

            # Refine: golden section around best_w
            if abs(best_w) > 1e-12:
                # Determine bracket
                if best_w < 0:
                    lo = -0.999 * v
                    hi = max(0.0, best_w + abs(best_w) * 2.0)
                else:
                    lo = min(0.0, best_w - best_w * 2.0)
                    hi = best_w + max(abs(best_w), 0.1 * v) * 3.0
                    hi = min(hi, 5.0 * S / n * 10)  # sanity bound

                lo = max(lo, -0.999 * v)
                # Ensure S+lo > 0
                lo = max(lo, -S + 1e-10)

                # Golden section search
                gr = (math.sqrt(5.0) - 1.0) / 2.0
                for _ in range(25):
                    if hi - lo < 1e-13 * max(1.0, v):
                        break
                    m1 = hi - gr * (hi - lo)
                    m2 = lo + gr * (hi - lo)
                    f1 = eval_w(m1)
                    f2 = eval_w(m2)
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                best_w = (lo + hi) * 0.5
                best_val = eval_w(best_w)
                if best_val >= cur_score - 1e-13:
                    continue

            # Apply update
            if best_val < cur_score - 1e-14:
                w = best_w
                if S + w < 1e-10:
                    break
                p[i] = max(0.0, v + w)
                S += w
                # Update convolution
                c[start:end] += w * infl
                c[start + diag_idx] += w * w
                cur_score = best_val
                improved = True

        # Rescale to sum=1
        if S > 1e-300 and math.isfinite(S) and abs(S - 1.0) > 1e-6:
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
            cur_score = 2.0 * n * float(c.max())

        # Verify periodically
        if sweep % 20 == 0:
            c = _conv_self(p)
            new_sc = 2.0 * n * float(c.max())
            if math.isfinite(new_sc) and new_sc < best_score - 1e-14:
                best_score = new_sc
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p.copy())
                    except Exception:
                        pass
            elif math.isfinite(new_sc) and new_sc < best_score:
                best_score = new_sc
                best_p = p.copy()

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 20.0:
                    # Perturb to escape
                    noise_scale = 0.03 + 0.02 * (flat / 10.0)
                    p = p * np.exp(rng.normal(0, noise_scale, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                        S = 1.0
                    c = _conv_self(p)
                    cur_score = 2.0 * n * float(c.max())
                    flat = 0
                else:
                    break

        sweep += 1

    # Final
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c_final = _conv_self(p)
    sc_final = 2.0 * n * float(c_final.max())
    if math.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_p = p.copy()
        if consider:
            try:
                consider(best_p.copy())
            except Exception:
                pass
    return best_p, best_score


def _block_descent(p, stop_time, consider=None, seed=123, block_size=2):
    """Update pairs of coordinates using alternating 1D optimization."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    ss = float(p.sum())
    if ss < 1e-300 or n < 2:

    S = 1.0
    Ltot = 2 * n - 1

    best_p = p.copy()

    flat = 0
    iter_count = 0

    max_iters = 20000
    while iter_count < max_iters and time.monotonic() < stop_time - 3.0:
        # Pick a random pair
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            j = (j + 1) % n

        # Alternate: optimize i with j fixed, then j with i fixed
        for coord in (i, j):
            v = float(p[coord])
            start = coord
            end = min(coord + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end]
            infl = 2.0 * p[:Ls]
            max_const = 0.0
            if start > 0:
                max_const = max(max_const, float(np.max(c[:start])))
            if end < Ltot:
                max_const = max(max_const, float(np.max(c[end:])))
            diag_idx = coord

            def eval_w(w):
                if S + w < 1e-12:
                    return math.inf
                vals = base + w * infl
                vals = vals.copy()
                vals[diag_idx] += w * w
                mv = max(float(vals.max()), max_const)
                ns = S + w
                return 2.0 * n * mv / (ns * ns)

            # Quick 3-point check
            w_cands = [-0.5 * v, 0.0, 0.5 * v]
            best_w = 0.0
            best_val = cur_score
            for wc in w_cands:
                val = eval_w(wc)
                if val < best_val - 1e-14:
                    best_val = val
                    best_w = wc

            if best_val < cur_score - 1e-14:

                if abs(best_w) > 1e-12:
                    lo = max(-0.999 * v, -S + 1e-10)
                    hi = max(v, abs(best_w) * 3.0)

                        else:
                            lo = m1
                    best_w = 0.5 * (lo + hi)

                if best_val < cur_score - 1e-14 and S + best_w > 1e-10:

                    S += best_w
                    c[start:end] += best_w * infl

                    cur_score = best_val
                    flat = 0
                else:

            else:

        # Rescale
        if abs(S - 1.0) > 1e-5:

            p *= inv
            c *= inv * inv
            S = 1.0
            cur_score = 2.0 * n * float(c.max())

            S = float(p.sum())

                S = 1.0
            c = _conv_self(p)

                c = _conv_self(p)

    if ss > 1e-300:

                return

            if s < 0.01:
                return
            sc = _score_exact_list([float(x) for x in a])

                best_arr = a.copy()

            pass

        return ls
    except Exception:
        return [1.0]
