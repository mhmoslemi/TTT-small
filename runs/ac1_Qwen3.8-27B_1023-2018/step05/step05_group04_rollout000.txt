            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            # Golden section refinement (fewer iterations for speed)
            if 0 < imin < G:
                lo = float(pts[max(0, imin - 2)])
                hi = float(pts[min(G, imin + 2)])
            else:
                lo, hi = 0.0, x_hi
            if hi - lo < 1e-14:
                lo = max(0.0, best_x - 1e-7)
                hi = best_x + 1e-7

            a_gs, b_gs = lo, hi
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)
            # Quick eval function
            def eval_scalar(x_new, base, infl, diag_pos, Ls, max_const, n, S):
                w = x_new - v  # Actually we need to recompute... 
                # Simpler: just evaluate directly
                s_n = S + (x_new - v)
                if s_n <= 1e-15:
                    return float('inf')
                aff_m = float(np.max(base + (x_new - v) * infl))
                if 0 <= diag_pos < Ls:
                    dval = base[diag_pos] + (x_new - v) * infl[diag_pos] + (x_new - v) ** 2
                    if dval > aff_m:
                        aff_m = dval
                m = max(aff_m, max_const)
                return 2.0 * n * m / (s_n * s_n)

            f_c = eval_scalar(c_gs, base, infl, diag_pos, Ls, max_const, n, S)
            f_d = eval_scalar(d_gs, base, infl, diag_pos, Ls, max_const, n, S)
            for _ in range(18):
                if b_gs - a_gs < 1e-12 * max(1.0, abs(best_x)):
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = eval_scalar(c_gs, base, infl, diag_pos, Ls, max_const, n, S)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_d = eval_scalar(d_gs, base, infl, diag_pos, Ls, max_const, n, S)
            x_mid = (a_gs + b_gs) / 2.0
            f_mid = eval_scalar(x_mid, base, infl, diag_pos, Ls, max_const, n, S)
            if f_mid < best_val:
                best_x, best_val = x_mid, f_mid

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= x_hi + 1e-9:
                w = best_x - v
                if abs(w) > 1e-14:
                    p[i] = best_x
                    aff_seg = base + w * infl
                    if 0 <= diag_pos < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag_pos] += w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    c[start_idx:end_idx] = aff_seg
                    cur_score = 2.0 * n * new_max
                    improved = True

        # Renormalize
        s_new = float(p.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            inv = 1.0 / s_new
            p *= inv
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-15:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 20.0:
                    kick_size = 0.02 + 0.06 * min(1.0, flat / 15.0)
                    p = p * np.exp(rng.normal(0.0, kick_size, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p, best_score


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n)
    return out / s


def _tent_init(n):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    t = np.maximum(0.0, 1.0 - 2.0 * np.abs(x - 0.5))
    s = float(t.sum())
    if s > 0:
        t /= s
    return t


def _gauss_init(n):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    sigma = 0.8 / n  # Wide Gaussian
    g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    s = float(g.sum())
    if s > 0:
        g /= s
    return g
