            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-15:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            best_u = best_u + rng.normal(0, 0.004, best_u.size)
                            continue
            if getattr(res, "status", 1) == 0:
                if time.monotonic() < end_stage - 0.8 and stale < 3:
                    best_u = best_u + rng.normal(0, 0.002, best_u.size)
                else:
                    break

        p_cur = _softmax_fn(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif math.isfinite(sc_cur) and sc_cur < best_sc + 0.01:
            p = p_cur.copy()
        else:
            p = best_p.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, G=None, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    if G is None:
        if n <= 500:
            G = 48
        elif n <= 1000:
            G = 40
        elif n <= 3000:
            G = 32
        elif n <= 6000:
            G = 28
        else:
            G = 24
    G = int(max(16, G))

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    best_lag = int(np.argmax(c))

    while sweep < 300000 and time.monotonic() < stop_time - 0.8:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.2:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(20.0 * v, 40.0 * S / float(n), 0.02 * S, 1e-14)
            x_hi = min(x_hi, 10.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, S / float(n))

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag_i = i

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[valid, None] * infl[None, :]
                dval = base[diag_i] + w[valid] * infl[diag_i] + w[valid] * w[valid]
                aff = np.maximum(aff, dval[:, None])
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                dv = denom[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if G > 4 and best_x > 0 and best_x < x_hi:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 32, dtype=np.float64)
                fine = np.concatenate((fine, [best_x]))
                fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val - 1e-15:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= 10.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0

        if sweep % 8 == 7:
            c = _conv_self(p)

        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 6.0:
                    sigma = 0.015 + 0.02 * min(flat, 12) / 12.0
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    else:
                        break
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    flat = 0
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 2.0:
        hard = t0 + max(2.0, budget * 0.9)
    final_reserve = max(10.0, 0.06 * (hard - t0))
    phase_limit = hard - final_reserve

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    n_parent = int(best_arr.size)

    # Generate diverse initializations
    inits = []
    inits.append(("parent", best_arr.copy()))

    # Analytical shapes at parent length
    x = (np.arange(n_parent) + 0.5) / float(n_parent)

    # Clip ramp: 1 to 1000 pattern
    ramp = np.minimum(np.linspace(1.0, n_parent, n_parent), 1000.0)
    ramp = ramp / ramp.sum()
    inits.append(("ramp", ramp.copy()))

    # Smooth ramp (logistic)
    for sh in [3.0, 5.0, 8.0]:
        sm = 1.0 / (1.0 + np.exp(-sh * (x - 0.5)))
        sm = sm / sm.sum()
        inits.append((f"logistic_{sh}", sm.copy()))

    # Geometric
    for r in [0.97, 0.94, 0.90]:
        geo = np.exp(np.arange(n_parent, dtype=np.float64) * math.log(r))
        geo = geo / geo.sum()
        inits.append((f"geo_{r}", geo.copy()))

    # Asymmetric perturbations of parent
    rng_init = np.random.default_rng(seed + 111)
    for asd in range(3):
        a_init = _make_asym_init(best_arr, n_parent, int(seed + 3000 + asd * 500))
        inits.append((f"asym_{asd}", a_init))

    # Score all initializations
    for name, arr in inits:
        consider(arr)

    # Pick top candidates for full optimization
    scored_inits = []
    for name, arr in inits:
        sc = _score_fft(arr)
        if math.isfinite(sc):
            scored_inits.append((sc, name, arr))
    scored_inits.sort(key=lambda t: t[0])

    # Run LSE + coordinate descent on top candidates
    top_n = 3
    if time.monotonic() < phase_limit:
        for ti in range(min(top_n, len(scored_inits))):
            if time.monotonic() > phase_limit - 30:
                break
            _, name, init_arr = scored_inits[ti]
            p_norm = init_arr / init_arr.sum()

            # LSE phase
            lse_end = time.monotonic() + max(30.0, 0.4 * (phase_limit - time.monotonic()) / max(1, top_n - ti))
            lse_end = min(lse_end, phase_limit - 40.0)
            if lse_end > time.monotonic() + 15:
                try:
                    p2 = _lse_optimize(p_norm, lse_end, consider=consider,
                                       seed=int(seed + 7000 + ti * 31))
                    if p2 is not None and p2.size == n_parent and p2.sum() > 0:
                        p2 = p2 / p2.sum()
                        consider(p2)
                except Exception:
                    pass

            # Coordinate descent
            if time.monotonic() < phase_limit - 20:
                cend = min(phase_limit - 15, time.monotonic() + 0.45 * (phase_limit - time.monotonic()) / max(1, top_n - ti))
                if cend > time.monotonic() + 10:
                    try:
                        p3, _ = _coord_polish(p2 if 'p2' in dir() and p2.size == n_parent else p_norm,
                                              cend, seed=int(seed + 9000 + ti * 47),
                                              consider=consider)
                        if p3 is not None and p3.size == n_parent and p3.sum() > 0:
                            p3 = p3 / p3.sum()
                            consider(p3)
                    except Exception:
                        pass

    # Multi-restart: perturb best and re-optimize
    restarts_done = 0
    max_restarts = 4
    while time.monotonic() < phase_limit - 50 and restarts_done < max_restarts:
        rng_r = np.random.default_rng(seed + 20000 + restarts_done * 733)
        sigma = 0.03 + 0.02 * restarts_done
        pert = best_arr * np.exp(rng_r.normal(0, sigma, best_arr.size))
        pert = np.maximum(pert, 0.0)
        ps = float(pert.sum())
        if ps > 0:
            pert /= ps
        else:
            break

        sub_lse_end = time.monotonic() + max(20.0, 0.25 * (phase_limit - time.monotonic()))
        sub_lse_end = min(sub_lse_end, phase_limit - 30.0)
        if sub_lse_end > time.monotonic() + 10:
            try:
                p2 = _lse_optimize(pert, sub_lse_end, consider=consider,
                                   seed=int(seed + 30000 + restarts_done * 101))
                if p2 is not None and p2.size == n_parent and p2.sum() > 0:
                    p2 = p2 / p2.sum()
                    consider(p2)
            except Exception:
                p2 = pert

            if time.monotonic() < phase_limit - 15:
                cend = min(phase_limit - 10, time.monotonic() + 0.3 * (phase_limit - time.monotonic()))
                if cend > time.monotonic() + 8:
                    try:
                        p3, _ = _coord_polish(p2, cend, seed=int(seed + 40000 + restarts_done * 89),
                                              consider=consider)
                        if p3 is not None and p3.size == n_parent and p3.sum() > 0:
                            p3 = p3 / p3.sum()
                            consider(p3)
                    except Exception:
                        pass
        restarts_done += 1

    # Final squeeze on the best
    final_end = min(hard - 10.0, t0 + budget - 10.0)
    if final_end > time.monotonic() + 20 and best_arr.size > 0:
        nbest = int(best_arr.size)

        # One more LSE pass
        rem = final_end - time.monotonic()
        if rem > 50 and nbest <= 6000:
            lstop = min(final_end - 30.0, time.monotonic() + 0.4 * rem)
            if lstop > time.monotonic() + 12:
                try:
                    p2 = _lse_optimize(best_arr, lstop, consider=consider,
                                       seed=int(seed + 99999))
                    if p2 is not None and p2.size == nbest and p2.sum() > 0:
                        p2 = p2 / p2.sum()
                        consider(p2)
                except Exception:
                    pass

        # Final coordinate polish
        cstop = min(final_end - 4.0, time.monotonic() + max(15.0, 0.5 * (final_end - time.monotonic())))
        if cstop > time.monotonic() + 8:
            try:
                p3, _ = _coord_polish(best_arr, cstop, seed=int(seed + 123456789),
                                      consider=consider)
                if p3 is not None and p3.size == nbest and p3.sum() > 0:
                    p3 = p3 / p3.sum()
                    consider(p3)
            except Exception:
                pass

    # Final validation with exact evaluator
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
