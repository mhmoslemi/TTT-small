        return np.zeros(0, dtype=np.float64)

        _csz = next_fast_len(_cln)
    else:
        _cln = _cln if _cln else 2 * n - 1
    F = rfft(a, _csz)
    c = irfft(F * F, _csz)[:_cln]
    np.maximum(c, 0.0, out=c)
    return c

        return p, math.inf
    p = p / s

    Ltot = 2 * n - 1
    _csz = next_fast_len(Ltot)
    c = _conv_self(p, _csz, Ltot)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    # Grid size adapts to n
    if n <= 512:
        G = 40
    elif n <= 2048:
        G = 28
    else:
        G = 20

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break

            S = float(p.sum())
            if S < 1e-300:
                break
            v = float(p[i])

            start_k = i
            end_k = min(i + n, Ltot)
            if start_k >= end_k:
                continue

            Ls = end_k - start_k
            base = c[start_k:end_k]
            infl = p[:Ls]  # p[k-i] for k in [start_k, end_k]

            # Precompute max outside affected range
            max_const = 0.0
            if start_k > 0:
                ml = float(c[:start_k].max())
                if ml > max_const: max_const = ml
            if end_k < Ltot:
                mr = float(c[end_k:].max())
                if mr > max_const: max_const = mr

            # Analytical critical points
            S_prime = S - v
            avg = S / n
            x_hi = max(15.0 * v, 40.0 * avg, 0.01)
            x_hi = min(x_hi, 8.0 * S)

            # Build candidate deltas
            cands = [-v, 0.0, x_hi - v]
            # Top K c entries
            K = min(8, Ls)
            if K > 1:
                top = np.argpartition(base, -K)[-K:]
                for ti in top:
                    kv = int(start_k + ti)
                    ajv = float(infl[ti])
                    ckv = float(base[ti])
                    if kv == 2 * i:
                        den = S_prime - v
                        if abs(den) > 1e-15:
                            cands.append((ckv - v * S_prime) / den)
                    else:
                        if ajv > 1e-15:
                            cands.append(S_prime - ckv / ajv)

            # Grid
            pts = np.linspace(0.0, x_hi, G, dtype=np.float64)
            for pt in pts:
                cands.append(pt - v)

            cands = np.array(cands, dtype=np.float64)
            cands = cands[(cands >= -v - 1e-12) & (cands <= x_hi - v)]
            cands = np.unique(cands)

            if cands.size == 0:
                cands = np.array([0.0])

            # Evaluate
            best_d = 0.0
            best_cscore = cur_score
            di = 2 * i
            di_in = (start_k <= di < end_k)
            di_off = di - start_k if di_in else -1

            for d in cands:
                d = float(d)
                x = v + d
                if x < 0: x = 0.0; d = -v
                if x > 1000: continue
                S2 = S + d
                if S2 < 1e-15: continue

                aff = base + 2.0 * d * infl
                if di_in:
                    aff[di_off] += d * d
                maff = float(aff.max()) if aff.size else 0.0
                mtot = max(maff, max_const)
                sc = 2.0 * n * mtot / (S2 * S2)
                if sc < best_cscore:
                    best_cscore = sc
                    best_d = d

            if best_d != 0.0 and best_cscore < cur_score - 1e-14:
                p[i] = v + best_d
                c[start_k:end_k] += 2.0 * best_d * infl
                if di_in:
                    c[di] += best_d * best_d
                cur_score = best_cscore
                improved = True

        # Renormalize
        S = float(p.sum())
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            # Recompute c from scratch every 20 sweeps for accuracy
            if sweep % 20 == 0:
                c = _conv_self(p, _csz, Ltot)
            else:
                c *= inv * inv
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-14:
                best_score = new_score
                best_p = p.copy()
            if consider and sweep % 50 == 0:
                try: consider(best_p)
                except: pass
            cur_score = new_score
        else:
            break

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 5:
                    p *= np.exp(rng.normal(0, 0.02, n))
                    p = np.maximum(p, 0)
                    ss = float(p.sum())
                    if ss > 0: p /= ss
                    c = _conv_self(p, _csz, Ltot)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break

        sweep += 1

    p = np.maximum(p, 0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_self(p, _csz, Ltot)
    sc = float(2.0 * n * c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score
