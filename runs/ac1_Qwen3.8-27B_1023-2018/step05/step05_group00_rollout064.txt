def _structured_perturb(p, rng, level=0.02):
    n = p.size
    x = np.arange(n, dtype=np.float64)
    perturbations = []
    for freq in [2, 3, 5, 8]:
        phase = rng.uniform(0, 2 * math.pi)
        perturbations.append(np.sin(2 * math.pi * freq * x / n + phase))
    comb = np.zeros(n, dtype=np.float64)
    for k, pt in enumerate(perturbations):
        comb += pt / (k + 1)
    comb /= (comb.max() + 1e-30)
    new_p = p * np.exp(rng.normal(0.0, level, n, dtype=np.float64) + level * 0.5 * comb)
    new_p = np.maximum(new_p, 0.0)
    s = float(new_p.sum())
    if s > 1e-300:
        new_p /= s
    return new_p
