        else:
            # Standard single-coordinate descent
            order = rng.permutation(n)
            for idx_i in order:
                i = int(idx_i)
                if time.monotonic() > stop_time - 1.0:
                    break
                if S > 1.15 or S < 0.85:
                    inv = 1.0 / S
                    p *= inv
                    c *= inv * inv
                    S = 1.0

                v = float(p[i])
                start_i = i
                end_i = min(i + n, Ltot)
                Ls = end_i - start_i
                if Ls <= 0:
                    continue

                base = c[start_i:end_i]
                infl = 2.0 * p[:Ls]

                max_const = 0.0
                if start_i > 0:
                    ml = float(c[:start_i].max())
                    if ml > max_const:
                        max_const = ml
                if end_i < Ltot:
                    mr = float(c[end_i:].max())
                    if mr > max_const:
                        max_const = mr

                avg = 1.0 / n
                x_hi = max(25.0 * v, 80.0 * avg, 0.02)
                x_hi = min(x_hi, 18.0)
                if not math.isfinite(x_hi) or x_hi <= 0:
                    x_hi = max(1e-8, avg)

                # Adaptive grid
                pts = np.linspace(0.0, x_hi, G_coarse + 1, dtype=np.float64)
                if 0.0 <= v <= x_hi:
                    local_w = max(x_hi / G_coarse * 3.0, 1e-6)
                    fp = np.linspace(max(0, v - local_w), min(x_hi, v + local_w), G_fine, dtype=np.float64)
                    pts = np.unique(np.concatenate([pts, fp, [v]]))

                diag = i
                w = pts - v
                denom = S + w
                valid = denom > 1e-15

                if not np.any(valid):
                    continue

                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                dval = base[diag] + w * infl[diag] + w * w
                maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)

                vals = np.full(pts.shape, math.inf, dtype=np.float64)
                vals[valid] = (2.0 * n * maff[valid]) / (denom[valid] ** 2)

                if not np.any(np.isfinite(vals)):
                    continue

                imin = int(np.argmin(vals))
                best_x = float(pts[imin])
                best_val = float(vals[imin])

                # Fine refinement
                if best_val < cur_score - 1e-13:
                    dw_grid = x_hi / G_coarse
                    lo = max(0.0, best_x - 2.5 * dw_grid)
                    hi = min(x_hi, best_x + 2.5 * dw_grid)
                    if hi > lo + 1e-12:
                        fine2 = np.linspace(lo, hi, 80, dtype=np.float64)
                        fine2 = np.unique(np.concatenate([fine2, [best_x]]))
                        w2 = fine2 - v
                        denom2 = S + w2
                        valid2 = denom2 > 1e-15
                        if np.any(valid2):
                            aff2 = base[None, :] + w2[:, None] * infl[None, :]
                            maff2 = np.max(aff2, axis=1)
                            dval2 = base[diag] + w2 * infl[diag] + w2 * w2
                            maff2 = np.maximum(maff2, dval2)
                            maff2 = np.maximum(maff2, max_const)
                            vals2 = np.full(fine2.shape, math.inf)
                            vals2[valid2] = (2.0 * n * maff2[valid2]) / (denom2[valid2] ** 2)
                            if np.any(np.isfinite(vals2)):
                                imin2 = int(np.argmin(vals2))
                                if vals2[imin2] < best_val:
                                    best_x = float(fine2[imin2])
                                    best_val = float(vals2[imin2])

                if math.isfinite(best_val) and best_val < cur_score - 1e-13:
                    w_new = best_x - v
                    if abs(w_new) > 1e-15:
                        p[i] = best_x
                        S += w_new
                        c[start_i:end_i] += w_new * infl
                        di = 2 * i
                        if 0 <= di < Ltot:
                            c[di] += w_new * w_new
                        cur_score = best_val
                        improved = True

        # End of sweep
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0

        if sweep % 20 == 0:
            c = _conv_self(p)
            np.maximum(c, 0.0, out=c)

        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat_count = 0
        else:
            flat_count += 1
            if flat_count >= 5:
                if time.monotonic() < stop_time - 8.0:
                    sigma = 0.025 / (1.0 + 0.05 * flat_count)
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 1e-15:
                        p /= ss
                        S = 1.0
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat_count = 0
                else:
                    break

        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    c = _conv_self(p)
    sc = float(2.0 * n * c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _symmetrize(p):
    n = p.size
    sym = 0.5 * (p + p[::-1].copy())
    return sym


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 15.0

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64) / 64.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_arr = a.copy()
            best_score = sc

    n_par = int(best_arr.size)

    # === Phase 1: LSE optimization ===
    phase1_end = min(hard_deadline - 60.0, t0 + 0.18 * budget)
    if phase1_end > time.monotonic() + 15.0 and n_par >= 128:
        try:
            p1 = best_arr.copy()
            p1 /= float(p1.sum())
            p1 = _lse_optimize(p1, phase1_end, consider=consider)
            if p1 is not None and p1.size == n_par and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 /= float(p1.sum())
                consider(p1)
        except Exception:
            pass

    # === Phase 2: Main coordinate descent ===
    work = best_arr.copy()
    work /= float(work.sum())
    phase2_end = min(hard_deadline - 40.0, t0 + 0.48 * budget)
    if phase2_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord_descent(work, phase2_end, seed=int(seed + 1111),
                                     consider=consider, max_sweeps=2000000, paired=False)
        except Exception:
            pass

    # === Phase 3: Paired coordinate descent ===
    work = best_arr.copy()
    work /= float(work.sum())
    phase3_end = min(hard_deadline - 40.0, t0 + 0.65 * budget)
    if phase3_end > time.monotonic() + 15.0:
        try:
            work, _ = _coord_descent(work, phase3_end, seed=int(seed + 2222),
                                     consider=consider, max_sweeps=1000000, paired=True)
        except Exception:
            pass

    # === Phase 4: Symmetrization check + coordinate descent ===
    work = best_arr.copy()
    work /= float(work.sum())
    sym = _symmetrize(work)
    ss = float(sym.sum())
    if ss > 0:
        sym /= ss
        consider(sym)

    work = best_arr.copy()
    work /= float(work.sum())
    phase4_end = min(hard_deadline - 30.0, t0 + 0.82 * budget)
    if phase4_end > time.monotonic() + 15.0:
        try:
            work, _ = _coord_descent(work, phase4_end, seed=int(seed + 3333),
                                     consider=consider, max_sweeps=2000000, paired=False)
        except Exception:
            pass

    # === Phase 5: Final LSE + coordinate polish ===
    work = best_arr.copy()
    work /= float(work.sum())
    phase5_end = min(hard_deadline - 15.0, t0 + budget - 25.0)
    if phase5_end > time.monotonic() + 25.0:
        lse_end = min(phase5_end - 15.0, time.monotonic() + 25.0)
        if lse_end > time.monotonic() + 10.0:
            try:
                p5 = _lse_optimize(work, lse_end, consider=consider)
                if p5 is not None and p5.size == n_par and p5.sum() > 0:
                    p5 /= float(p5.sum())
                    consider(p5)
            except Exception:
                pass

        work = best_arr.copy()
        work /= float(work.sum())
        if phase5_end > time.monotonic() + 10.0:
            try:
                work, _ = _coord_descent(work, phase5_end, seed=int(seed + 4444),
                                         consider=consider, max_sweeps=1000000, paired=False)
            except Exception:
                pass

    # === Final validation ===
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
