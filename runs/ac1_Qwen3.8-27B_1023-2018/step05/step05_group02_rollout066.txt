def _subgradient_polish(p, stop_time, consider=None, max_iters=5000):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    best_p = p.copy()
    best_score = float(2.0 * n * _conv_self(p).max())
    base_step = 0.003
    min_step = 1e-9

    for it in range(max_iters):
        if time.monotonic() > stop_time - 2.0:
            break
        c = _conv_self(p)
        cmax = float(c.max())
        threshold = cmax * 0.998
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        if len(active) > 200:
            idx = np.argsort(c)[::-1][:100]
            active = idx

        grad = np.zeros(n)
        for k in active:
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                contrib = p[kj[valid]] + p[valid]
                grad[valid] += contrib
        grad /= max(1, len(active))

        step = base_step / (1.0 + 0.03 * it)
        if step < min_step:
            break

        found = False
        for scale in (1.0, 0.7, 0.4, 0.2, 0.1):
            p_cand = p - scale * step * grad
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = float(2.0 * n * _conv_self(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                found = True
                break
        if not found:
            base_step *= 0.92
            if base_step < 1e-13:
                break

        if it % 30 == 0 and consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider is not None:

        except Exception:

def _lp_direction_step(p, stop_time, consider=None, max_lp_iters=3):
    """Use LP to find optimal perturbation direction (state-of-the-art method)."""
    from scipy.sparse import csr_matrix
    from scipy.optimize import linprog

    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    c = _conv_self(p)
    M = float(c.max())
    if M <= 0:
        return p

    # Build sparse constraint matrix A: (2n-1) x n
    # A[k, m] = p[k-m] if 0 <= k-m < n else 0
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            data.append(p[k - j])

    if len(rows) == 0:
        return p

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    best_p = p.copy()
    for lp_it in range(max_lp_iters):
        if time.monotonic() > stop_time - 5.0:
            break
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True,
                                                    'time_limit': max(1.0, stop_time - time.monotonic() - 10.0)})
        except Exception:
            break

        if res.status != 0 or res.x is None:
            break
        g = np.maximum(res.x, 0.0)
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break

        # Mix: h = (1-t)*p + t*g (both sum to 1)
        for t in (0.3, 0.5, 0.7, 0.9):
            h = (1.0 - t) * p + t * g
            h = np.maximum(h, 0.0)
            sh = float(h.sum())
            if sh < 1e-15:
                continue
            h /= sh
            sc = _score_norm(h)
            sc_best = _score_norm(best_p)
            if math.isfinite(sc) and sc < sc_best - 1e-10:
                best_p = h.copy()
                p = h.copy()
                c = _conv_self(p)
                M = float(c.max())
                b_ub = np.full(2 * n - 1, M)
                break

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p


def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m) + 0.5) / m
        fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s

    # Phase 5: LSE fine-tune + subgradient + final coord
    phase5_end = t0 + budget - 35.0
    now = time.monotonic()
    rem = phase5_end - now

    if rem > 30.0:
        lse_stop = now + 0.35 * rem
        try:
            p5 = _lse_optimize(best_arr, lse_stop, consider=consider)
            if p5 is not None and p5.size == n and np.all(np.isfinite(p5)) and p5.sum() > 0:
                p5 = p5 / float(p5.sum())
                consider(p5)
        except Exception:
            pass

    now = time.monotonic()
    rem = phase5_end - now
    if rem > 20.0:
        sg_stop = now + 0.25 * rem
        try:
            p6 = _subgradient_polish(best_arr, sg_stop, consider=consider,
                                     max_iters=8000)
            if p6 is not None and p6.size == n and np.all(np.isfinite(p6)) and p6.sum() > 0:
                p6 = p6 / float(p6.sum())
                consider(p6)
        except Exception:
            pass

    # Final coordinate polish
    now = time.monotonic()
    rem = phase5_end - now
    if rem > 15.0:
        cf_stop = now + max(10.0, 0.6 * rem)
        try:
            p7, _ = _coord_polish(best_arr, cf_stop, max_sweeps=200000,
                                  G=80, seed=int(seed) + 99999, consider=consider)
            if p7 is not None and p7.size == n and np.all(np.isfinite(p7)) and p7.sum() > 0:
                p7 = p7 / float(p7.sum())
                consider(p7)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.ones(64) / 64.0
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
