                    if 0 <= di < Ls:
                        dval2 = base[di] + w2 * infl[di] + w2 * w2
                        maff2 = np.maximum(maff2, dval2)
                    maff2 = np.maximum(maff2, max_const)
                    v2 = np.full(fine_pts.shape, math.inf)
                    valid2 = den2 > 1e-15
                    if np.any(valid2):
                        v2[valid2] = (2.0 * n * maff2[valid2]) / (den2[valid2] * den2[valid2])
                    if np.any(np.isfinite(v2)):
                        im2 = int(np.argmin(v2))
                        if v2[im2] < best_val:
                            best_val = float(v2[im2])
                            best_x = float(fine_pts[im2])

                    gr = (math.sqrt(5.0) - 1.0) / 2.0
                    g_lo = max(0.0, best_x - (hi - lo) / 8.0)
                    g_hi = min(x_hi, best_x + (hi - lo) / 8.0)
                    if g_hi - g_lo > 1e-14:
                        a_gs = g_lo + (1.0 - gr) * (g_hi - g_lo)
                        b_gs = g_lo + gr * (g_hi - g_lo)

                        def eval1(x_val):
                            wv = x_val - v
                            dv = S + wv
                            if dv <= 1e-15:
                            return math.inf
                            av = base + wv * infl
                            if 0 <= di < Ls:
                                av[di] += wv * wv
                            mv = float(np.max(av))
                            mv = max(mv, max_const)
                            return float(2.0 * n * mv / (dv * dv))

                        fa = eval1(a_gs)
                        fb = eval1(b_gs)
                        for _ in range(12):
                            if g_hi - g_lo < 1e-14:
                                break
                            if fa < fb:
                                g_hi = b_gs
                                b_gs = a_gs
                                fb = fa
                                a_gs = g_lo + (1.0 - gr) * (g_hi - g_lo)
                                fa = eval1(a_gs)
                            else:
                                g_lo = a_gs
                                a_gs = b_gs
                                fa = fb
                                b_gs = g_lo + gr * (g_hi - g_lo)
                                fb = eval1(b_gs)
                        gs_x = (g_lo + g_hi) / 2.0
                        gv = eval1(gs_x)
                        if math.isfinite(gv) and gv < best_val:
                            best_val = gv
                            best_x = gs_x

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 10.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start_k:end_k] += w * infl
                    di_abs = 2 * i
                    if 0 <= di_abs < Ltot:
                        c[di_abs] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        np.maximum(c, 0.0, out=c)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-16:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 8.0:
                    try:
                        noise_amt = 0.02 + 0.01 * math.exp(-flat * 0.05)
                        p = p * np.exp(rng.normal(0.0, noise_amt, n))
                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss > 0.0:
                            p /= ss
                        c = _conv_self(p)
                        S_total = 1.0
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
                            best_score = cur_score
                            best_p = p.copy()
                        flat = 0
                    except Exception:
                        break
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        sc = float(2.0 * n * _conv_self(p).max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def imax(a, b):
    if a > b:
        return a
    return b
