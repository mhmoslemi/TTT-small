def solve_dual_lp(a, M):
    n = len(a)
    # Build sparse matrix A: (2n-1) x n
    # A[k,j] = a[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    
    A = scipy.sparse.csc_array((vals, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)  # minimize -1^T b = max 1^T b
    d = np.full(2*n-1, M)
    
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=d, bounds=[(0, None)]*n, method='highs')
    
    return res.x if res.success else None
