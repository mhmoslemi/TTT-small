        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0.0:
            return 1e100, np.zeros_like(uu)
        if T <= 1e-300:
            T = 1e-4 * cmax
        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -400.0, 400.0)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            val = cmax
        else:
            val = cmax + T * math.log(sum_exp / c.size)
            if not math.isfinite(val):
                val = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(val), grad

    best_p = p.copy()
    best_sc = _score_fft(p)

    stages = (
        (0.1, 0.08),
        (0.04, 0.12),
        (0.01, 0.15),
        (0.003, 0.18),
        (0.0008, 0.20),
        (0.0002, 0.20),
        (0.00005, 0.07),
    )

    start = time.monotonic()
    avail = max(1.0, float(stop_time) - start - 6.0)
    if avail < 10.0:
        return best_p

    total_w = sum(w for _, w in stages)
    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += wt
        end_stage = min(float(stop_time) - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue

        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            cmax = 1.0
        T = max(1e-300, frac * cmax)

        maxit = 40 if n > 2000 else 60
        opts = {"maxiter": maxit, "gtol": 1e-14, "ftol": 1e-18,
                "maxls": 30, "maxcor": 40, "maxdisp": 0, "disp": False}
        stale = 0
        while time.monotonic() < end_stage - 1.0 and stale < 4:
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    u = x
                    p_cur = _softmax(u)
                    p_cur = np.maximum(p_cur, 0.0)
                    ss = float(p_cur.sum())
                    if ss > 1e-300:
                        p_cur /= ss
                    sc_cur = _score_fft(p_cur)
                    if math.isfinite(sc_cur) and sc_cur < best_sc - 1e-16:
                        best_sc = sc_cur
                        best_p = p_cur.copy()
                        p = p_cur.copy()
                        stale = 0
                    else:
                        stale += 1
            if getattr(res, "status", 1) == 0:
                break

    return best_p


def _coord_polish(p, stop_time, consider=None, seed=12345, G=40):
    """Coordinate descent with incremental convolution updates."""
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    Ltot = 2 * n - 1
    c = _conv_fft(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    # Identify peak lags for prioritization
    peak_lags = np.argsort(c)[::-1][:min(n, 64)]

    sweep = 0
    flat = 0
    max_sweeps = 999999

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        S = 1.0

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break

            # Rescale check
            if S > 1.25 or S < 0.75:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0

            v = float(p[i])
            start_k = i
            end_k = min(i + n, Ltot)
            Ls = end_k - start_k
            if Ls <= 0:
                continue

            # Influence: 2*a[k-i] for k in [i, i+n-1]
            influence = 2.0 * p[:Ls].copy()

            # Constant max outside affected range
            max_const = 0.0
            if start_k > 0:
                max_const = max(max_const, float(c[:start_k].max()))
            if end_k < Ltot:
                max_const = max(max_const, float(c[end_k:].max()))

            base = c[start_k:end_k]

            # Search range for new value
            x_hi = max(20.0 * v, 50.0 / n, 0.02, 1e-12)
            x_hi = min(x_hi, 5.0 / n * 10.0)  # generous upper bound
            x_hi = min(x_hi, 8.0 / n)

            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 < v < x_hi:
                merge_pt = v
                idx_insert = int(np.searchsorted(pts, merge_pt))
                pts = np.insert(pts, idx_insert, merge_pt)

            # Vectorized evaluation
            w = pts - v  # delta
            denom_sq = (1.0 + w) ** 2
            valid = (1.0 + w) > 1e-12
            if not np.any(valid):
                continue

            # affected[k] = base[k] + w * influence[k]
            aff = base[None, :] + w[:, None] * influence[None, :]
            # diagonal correction at k = 2i - i = i in the affected range
            diag_idx = 2 * i - start_k
            if 0 <= diag_idx < Ls:
                aff[:, diag_idx] += w * w  # wait, it's w_i^2 = delta^2
                # Actually the diagonal term: delta^2 where delta = w
                # But we already added w*influence which includes 2*v at diagonal
                # The correction is: the true change is 2*delta*v + delta^2 at diagonal
                # influence[diag_idx] = 2*a[i] = 2*v, so w*2v + w^2 is correct
                # Actually influence already has 2*p[k-i], at k=2i that's 2*p[i]=2*v
                # So aff at diagonal = base + w*2v + w^2. That's correct!

            maff = np.max(aff, axis=1)
            maff = np.maximum(maff, max_const)

            vals = np.full(pts.shape, math.inf)
            vals[valid] = 2.0 * n * maff[valid] / denom_sq[valid]

            if not np.any(np.isfinite(vals)):
                continue

            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Golden section refinement
            if G > 2:
                dw = x_hi / float(G)
                g_lo = max(0.0, best_x - 2.0 * dw)
                g_hi = min(x_hi, best_x + 2.0 * dw)
                if g_hi - g_lo > 1e-14:
                    gr = (math.sqrt(5.0) - 1.0) / 2.0
                    a_gs = g_hi - gr * (g_hi - g_lo)
                    b_gs = g_lo + gr * (g_hi - g_lo)

                    def fast_eval(xv):
                        wv = xv - v
                        dm = (1.0 + wv) ** 2
                        if dm < 1e-24:
                            return math.inf
                        av = base + wv * influence
                        if 0 <= diag_idx < Ls:
                            av[diag_idx] += wv * wv
                        mvm = max(float(av.max()), max_const)
                        return 2.0 * n * mvm / dm

                    fa = fast_eval(a_gs)
                    fb = fast_eval(b_gs)
                    for _ in range(12):
                        if not math.isfinite(fa):
                            fa = 1e100
                        if not math.isfinite(fb):
                            fb = 1e100
                        if fa < fb:
                            g_hi = b_gs
                            b_gs = a_gs
                            fb = fa
                            a_gs = g_hi - gr * (g_hi - g_lo)
                            fa = fast_eval(a_gs)
                        else:
                            g_lo = a_gs
                            a_gs = b_gs
                            fa = fb
                            b_gs = g_lo + gr * (g_hi - g_lo)
                            fb = fast_eval(b_gs)
                        if g_hi - g_lo < 1e-15:
                            break
                    gs_x = (g_lo + g_hi) / 2.0
                    gv = fast_eval(gs_x)
                    if math.isfinite(gv) and gv < best_val:
                        best_x = gs_x
                        best_val = gv

            if math.isfinite(best_val) and best_val < cur_score - 1e-14:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S += w
                    c[start_k:end_k] += w * influence
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        # Rescale
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        # Verify with full convolution periodically
        if sweep % 5 == 0:
            c = _conv_fft(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-15:
                best_score = new_score
                best_p = p.copy()
                if consider is not None:
                    try:
                        consider(best_p)
                    except Exception:
                        pass
            cur_score = new_score

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 5.0:
                    noise = 0.02 + 0.01 * math.exp(-flat * 0.15)
                    p = p * np.exp(rng.normal(0.0, noise, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_fft(p)
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
                        best_score = cur_score
                        best_p = p.copy()
                        if consider is not None:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    flat = 0
                else:
                    break

        sweep += 1

    # Final verification
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_fft(p)
    sc = float(2.0 * n * c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p, best_score


def _subgradient_step(p, stop_time, consider=None, max_iters=2000):
    """Subgradient descent on the max autocorrelation."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    best_p = p.copy()
    best_score = float(2.0 * n * _conv_fft(p).max())
    base_lr = 0.003
    min_lr = 1e-9
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 1.5:
            break

        c = _conv_fft(p)
        cmax = float(c.max())
        threshold = cmax * 0.9995
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Subgradient: average of d(a*a)/da over active lags
        grad = np.zeros(n)
        for k in active:
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                contrib = p[kj[valid]] + p[valid]
                grad[valid] += contrib
        grad /= float(len(active))

        # Normalize sum
        grad -= float(np.mean(grad))

        lr = base_lr / (1.0 + 0.01 * it)
        if lr < min_lr:
            break

        found = False
        for scale in (1.0, 0.5, 0.25, 0.125, 0.0625):
            p_cand = p - scale * lr * grad
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = float(2.0 * n * _conv_fft(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-16:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                found = True
                no_improve = 0
                break
        if not found:
            no_improve += 1
            base_lr *= 0.92
            if no_improve > 200:
                break

        if it % 100 == 0 and consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p


def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == 1:
        return np.full(n, float(src[0]) / n)
    if src.size == n:
        out = src.copy()
    else:
        xp = (np.arange(src.size) + 0.5) / src.size
        out = np.interp((np.arange(n) + 0.5) / n, xp, src)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 20.0  # leave 20s for final validation

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float(_score_exact(best_arr))

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(384, dtype=np.float64) / 384.0
        best_score = float(_score_exact(best_arr))

    n = int(best_arr.size)
    best_arr = best_arr / float(best_arr.sum())

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0:
                return
            s = float(a.sum())
            if s < 0.01:
                return
            sc = float(_score_exact(a))
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_arr = (a / s).copy()
                best_score = sc
        except Exception:
            pass

    # Phase 1: LSE optimization
    phase1_end = min(hard, t0 + 0.28 * (hard - t0))
    now = time.monotonic()
    if phase1_end > now + 20.0:
        try:
            p1 = _lse_optimize(best_arr, phase1_end, consider=consider)
            if p1 is not None and p1.size == n and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
                best_arr = p1.copy()
        except Exception:
            pass

    # Phase 2: Main coordinate descent
    phase2_end = min(hard, t0 + 0.72 * (hard - t0))
    now = time.monotonic()
    if phase2_end > now + 15.0:
        try:
            p2, sc2 = _coord_polish(best_arr, phase2_end, consider=consider,
                                    seed=int(seed + 11111), G=48)
            if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p2 = p2 / float(p2.sum())
                consider(p2)
                best_arr = p2.copy()
        except Exception:
            pass

    # Phase 3: Subgradient + more coordinate descent
    now = time.monotonic()
    if hard - now > 30.0:
        # Subgradient
        sg_end = min(hard - 15.0, now + 0.3 * (hard - now))
        if sg_end > now + 10.0:
            try:
                p3 = _subgradient_step(best_arr, sg_end, consider=consider, max_iters=1500)
                if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and p3.sum() > 0:
                    p3 = p3 / float(p3.sum())
                    consider(p3)
                    best_arr = p3.copy()
            except Exception:
                pass

    # Phase 4: Final coordinate descent polish
    now = time.monotonic()
    cstop = min(hard - 5.0, t0 + budget - 6.0)
    if cstop > now + 8.0:
        try:
            p4, _ = _coord_polish(best_arr, cstop, consider=consider,
                                  seed=int(seed + 99999), G=56)
            if p4 is not None and p4.size == n and np.all(np.isfinite(p4)) and p4.sum() > 0:
                p4 = p4 / float(p4.sum())
                consider(p4)
                best_arr = p4.copy()
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
