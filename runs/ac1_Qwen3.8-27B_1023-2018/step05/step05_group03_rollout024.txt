            (0.00080, 0.20),
            (0.00030, 0.25),
            (0.00010, 0.30),
            (0.00003, 0.25),
        )
    else:
        stages = (
            (0.020, 0.10),
            (0.008, 0.12),
            (0.0030, 0.14),
            (0.0010, 0.16),
            (0.00035, 0.18),
            (0.00012, 0.20),
            (0.00004, 0.20),
        )

    start = time.monotonic()
    avail = float(stop_time - start - 4.0)
    if avail <= 4.0:
        return best_p

    total_w = float(sum(w for _, w in stages))
    if total_w <= 0.0:
        total_w = 1.0
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue

        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-14, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_new = _softmax(u)
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if math.isfinite(ss) and ss > 1e-300:
            p_new /= ss
        else:
            p_new = best_p.copy()

        sc_new = _score(p_new)
        if math.isfinite(sc_new) and sc_new < best_sc - 1e-13:
            best_sc = sc_new
            best_p = p_new.copy()
            p = p_new.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass
        else:
            p = best_p.copy()
        u = p_to_u(p)

    return best_p


def _lp_direction(a, M):
    if linprog is None or sp is None:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n < 16 or n > 768:
        return None
    if not math.isfinite(M) or M <= 0.0:
        return None
    try:
        idx = np.arange(n, dtype=np.int32)
        cols = np.repeat(idx, n)
        off = np.tile(idx, n)
        rows = cols + off
        valid = (rows >= 0) & (rows < 2 * n - 1)
        rows = rows[valid]
        cols = cols[valid]
        data = a[off[valid]].astype(np.float64)
        A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        cobj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        res = linprog(cobj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method="highs")
        if res is not None and getattr(res, "success", False):
            b = np.asarray(res.x, dtype=np.float64).ravel()
            b = np.where(np.isfinite(b), b, 0.0)
            b = np.maximum(b, 0.0)
            if float(b.sum()) > 0.0:
                return b
    except Exception:
        pass
    return None


def _lp_improve_small(src, n_small, stop_time, max_iter=6):
    n_small = int(n_small)
    if n_small < 16 or n_small > 700:
        return None
    if time.monotonic() > stop_time:
        return None
    a = _resample(src, n_small)
    best_a = a.copy()
    best_sc = _score(a)
    if not math.isfinite(best_sc):
        return None
    for _ in range(max_iter):
        if time.monotonic() > stop_time - 5.0:
            break
        c = _fft_conv(a)
        M = float(c.max()) if c.size else 0.0
        if not math.isfinite(M) or M <= 0.0:
            break
        b = _lp_direction(a, M)
        if b is None:
            break
        s = float(b.sum())
        if not math.isfinite(s) or s <= 0.0:
            break
        g = b / s
        improved = False
        for t in (0.60, 0.45, 0.30, 0.20, 0.10, 0.05):
            cand = (1.0 - t) * a + t * g
            cand = np.maximum(cand, 0.0)
            ss = float(cand.sum())
            if not math.isfinite(ss) or ss <= 0.0:
                continue
            cand /= ss
            sc = _score(cand)
            if math.isfinite(sc) and sc < best_sc - 1e-10:
                a = cand
                best_sc = sc
                best_a = cand.copy()
                improved = True
                break
        if not improved:
            break
    if math.isfinite(_score(best_a)):
        return best_a
    return None


def _coord_polish(p, stop_time, G=72, seed=12345, consider=None):
    p = _normalize(p)
    n = int(p.size)
    if n <= 1:
        return p, math.inf

    c = _fft_conv(p)
    c = np.maximum(c, 0.0)
    S = 1.0
    infl = 2.0 * p
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass

    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < 2000000 and time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)

        for pos in range(n):
            if time.monotonic() > stop_time - 0.3:
                break
            if S <= 1e-300:
                break
            if S > 1.25 or S < 0.75:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                infl *= inv
                S = 1.0
                c = np.maximum(c, 0.0)
                cur_score = float(2.0 * n * c.max())

            i = int(order[pos])
            v = float(p[i])
            L = 2 * n - 1
            start = i
            end = min(i + n, L)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end].copy()
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                max_const = max(max_const, mc)
            if end < L:
                mc = float(c[end:].max())
                max_const = max(max_const, mc)
            if not math.isfinite(max_const):
                max_const = 0.0

            Sloc = float(S)
            di = 2 * i - start
            xhi = max(1.0, 12.0 * v, 12.0 * Sloc / n, 0.002)
            xhi = min(xhi, max(0.03, v))
            if not math.isfinite(xhi) or xhi <= 0.0:
                xhi = max(1e-8, 8.0 * Sloc / n)

            pts = [0.0, v, xhi]
            m1 = max(10, int(G * 0.40))
            minx = max(1e-10, 1e-6 * v, Sloc / (3000.0 * n))
            if minx < xhi:
                pts.extend(np.geomspace(minx, xhi, m1).tolist())
            if v > 1e-12:
                m2 = max(8, int(G * 0.30))
                factors = np.geomspace(0.15, 6.0, m2)
                pts.extend((v * factors).tolist())
            m3 = max(8, int(G * 0.30))
            linhi = min(xhi, max(6.0 * Sloc / n, 0.001))
            if linhi > 0.0:
                pts.extend(np.linspace(0.0, linhi, m3).tolist())

            pts = np.unique(np.clip(np.asarray(pts, dtype=np.float64), 0.0, xhi))
            if pts.size == 0:
                pts = np.array([v])

            # Vectorized evaluation
            w = pts - v
            den = Sloc + w
            valid = den > 1e-14
            aff = base[None, :] + w[:, None] * infl[None, :]
            if 0 <= di < Ls:
                aff[:, di] = base[di] + w * infl[di] + w * w
            np.maximum(aff, max_const, out=aff)
            mv = np.max(aff, axis=1)
            vals = np.full(pts.size, math.inf)
            if np.any(valid):
                vals[valid] = (2.0 * n * mv[valid]) / (den[valid] ** 2)
            vals = np.where(np.isfinite(vals), vals, math.inf)
            if not np.any(np.isfinite(vals)):
                continue
            im = int(np.argmin(vals))
            best_x = float(pts[im])
            best_val = float(vals[im])

            # Refine around best
            if best_val < cur_score + max(1e-10, 1e-6 * cur_score):
                if best_x <= 0.0:
                    lo = 0.0
                    hi = max(4.0 * Sloc / n, 4.0 * v, 1e-7)
                else:
                    step = max(v, Sloc / n, 1e-9)
                    lo = max(0.0, best_x * 0.6, best_x - 2.0 * step)
                    hi = min(xhi, max(best_x * 1.4, best_x + 2.0 * step))
                if hi <= lo:
                    hi = lo + 1e-10
                fine_n = 40
                fine = np.linspace(lo, hi, fine_n)
                if lo > 0:
                    fine = np.concatenate([fine, np.geomspace(lo, hi, 12)])
                else:
                    fine = np.concatenate([np.array([0.0]), fine])
                fine = np.unique(np.clip(fine, 0.0, xhi))

                w2 = fine - v
                den2 = Sloc + w2
                valid2 = den2 > 1e-14
                aff2 = base[None, :] + w2[:, None] * infl[None, :]
                if 0 <= di < Ls:
                    aff2[:, di] = base[di] + w2 * infl[di] + w2 * w2
                np.maximum(aff2, max_const, out=aff2)
                mv2 = np.max(aff2, axis=1)
                vals2 = np.full(fine.size, math.inf)
                if np.any(valid2):
                    vals2[valid2] = (2.0 * n * mv2[valid2]) / (den2[valid2] ** 2)
                vals2 = np.where(np.isfinite(vals2), vals2, math.inf)
                if np.any(np.isfinite(vals2)):
                    im2 = int(np.argmin(vals2))
                    vx2 = float(fine[im2])
                    vv2 = float(vals2[im2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2

            if (math.isfinite(best_val) and best_val < cur_score - 1e-12
                    and 0.0 <= best_x <= xhi):
                d = best_x - v
                if abs(d) > 1e-16:
                    c[start:end] += d * infl
                    diag = 2 * i
                    if 0 <= diag < L:
                        c[diag] += d * d
                    p[i] = best_x
                    infl[i] = 2.0 * best_x
                    S += d
                    cur_score = best_val
                    improved = True

        # Periodic recompute
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            infl *= inv
            S = 1.0
        else:
            break
        c = _fft_conv(p)
        c = np.maximum(c, 0.0)
        desc = float(2.0 * n * c.max())
        if math.isfinite(desc) and desc < best_score - 1e-14:
            best_score = desc
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 20.0:
                    sigmas = (0.025, 0.05, 0.08, 0.12, 0.18)
                    sigma = sigmas[min(flat - 3, len(sigmas) - 1)]
                    noise = rng.normal(0.0, sigma, n)
                    q = p * np.exp(noise)
                    if sigma >= 0.08:
                        x = (np.arange(n, dtype=np.float64) + 0.5) / n
                        amp = 0.15 + 0.30 * rng.random()
                        f1 = 1 + int(rng.integers(0, 5))
                        ph1 = 2.0 * math.pi * rng.random()
                        q *= np.exp(amp * np.sin(2.0 * math.pi * f1 * x + ph1))
                    q = np.maximum(q, 0.0)
                    qs = float(q.sum())
                    if math.isfinite(qs) and qs > 0.0:
                        p = q / qs
                        c = _fft_conv(p)
                        c = np.maximum(c, 0.0)
                        S = 1.0
                        infl = 2.0 * p
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider is not None:
                                try:
                                    consider(best_p)
                                except Exception:
                                    pass
                    flat = 0
                else:
                    break
        sweep += 1

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(10.0, budget)
    reserve = 20.0 if budget > 200.0 else 8.0
    hard = t0 + budget - reserve
    if hard <= t0 + 2.0:
        hard = t0 + max(1.5, budget - 0.5)

    parent = _get_parent()
    n_target = int(parent.size) if parent.size > 0 else 3072

    best_arr = _normalize(parent.copy())
    best_score = float(_score(best_arr))
    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.full(64, 1.0 / 64.0)
        best_score = float(_score(best_arr))

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return
        if arr.size == 0:
            return
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if not math.isfinite(s) or s < 0.01:
            return
        arr = arr / s
        sc = _score(arr)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_arr = arr.copy()
            best_score = sc

    rng_top = np.random.default_rng(seed)
    explored = [best_arr.copy()]

    def add_explore(cand):
        cand = _normalize(cand)
        if cand.size != n_target:
            cand = _resample(cand, n_target)
        cand = _normalize(cand)
        sc = _score(cand)
        base_sc = _score(best_arr)
        if math.isfinite(sc) and math.isfinite(base_sc) and sc < base_sc + 0.04:
            explored.append(cand)

    # Phase 1: Cheap shaped perturbations
    phase1_limit = min(hard - 4.0, t0 + max(2.0, 0.04 * budget))
    base = best_arr.copy()
    for q in (0.50, 0.70, 1.0, 1.4, 1.8, 2.5):
        if time.monotonic() > phase1_limit:
            break
        cand = np.power(np.maximum(base, 1e-15), q)
        cand = _normalize(cand)
        consider(cand)
        add_explore(cand)

    for _ in range(8):
        if time.monotonic() > phase1_limit:
            break
        side = 1.0 if rng_top.random() < 0.5 else -1.0
        amp = 0.25 + 0.80 * rng_top.random()
        x = (np.arange(n_target, dtype=np.float64) + 0.5) / n_target
        cand = base * np.exp(side * amp * (2.0 * x - 1.0))
        cand = np.maximum(cand, 0.0)
        cand = _normalize(cand)
        consider(cand)
        add_explore(cand)

    # Also try Gaussian-like shapes
    for _ in range(4):
        if time.monotonic() > phase1_limit:
            break
        xc = 0.3 + 0.4 * rng_top.random()
        sig = 0.05 + 0.15 * rng_top.random()
        x = (np.arange(n_target, dtype=np.float64) + 0.5) / n_target
        cand = np.exp(-0.5 * ((x - xc) / sig) ** 2)
        cand = np.maximum(cand, 1e-8)
        cand = _normalize(cand)
        consider(cand)
        add_explore(cand)

    # Phase 2: LP-based exploration at small sizes
    if budget > 200.0 and time.monotonic() < hard - 100.0:
        small_lens = [128, 192, 256, 384, 512]
        if n_target <= 512:
            small_lens = [x for x in small_lens if x < n_target]
        else:
            small_lens = [x for x in small_lens if x < n_target]
        small_lens = sorted(set(small_lens))
        for ns in small_lens:
            if time.monotonic() > hard - 60.0:
                break
            stop = min(hard - 5.0, time.monotonic() + max(15.0, min(60.0, 0.03 * budget * (ns / 128.0))))
            try:
                small = _lp_improve_small(best_arr, ns, stop, max_iter=5)
            except Exception:
                small = None
            if small is not None:
                cand = _resample(small, n_target)
                cand = _normalize(cand)
                consider(cand)
                add_explore(cand)

    # Phase 3: LSE optimization on best
    now = time.monotonic()
    phase3_stop = min(hard - 10.0, time.monotonic() + max(30.0, 0.30 * (hard - now)))
    if phase3_stop > now + 20.0:
        p_main = best_arr.copy()
        p_main = _lbfgs_optimize(p_main, phase3_stop, consider=consider, fine_only=False)
        consider(p_main)

    # Phase 4: Coordinate descent polish
    now = time.monotonic()
    phase4_stop = min(hard - 5.0, now + max(30.0, 0.40 * (hard - now)))
    if phase4_stop > now + 15.0:
        p_main = best_arr.copy()
        p_main, _ = _coord_polish(p_main, phase4_stop, G=72, seed=seed + 1000, consider=consider)
        consider(p_main)

    # Phase 5: Fine LSE + coordinate if time remains
    now = time.monotonic()
    if now < hard - 40.0:
        p_fine = best_arr.copy()
        lse_stop = min(hard - 20.0, now + min(80.0, max(15.0, 0.25 * (hard - now))))
        if lse_stop > now + 10.0:
            p_fine = _lbfgs_optimize(p_fine, lse_stop, consider=consider, fine_only=True)
            consider(p_fine)
        now = time.monotonic()
        cstop = min(hard - 3.0, now + max(15.0, 0.50 * max(0.0, hard - now)))
        if cstop > now + 10.0:
            p_fine, _ = _coord_polish(p_fine, cstop, G=80, seed=seed + 99999, consider=consider)
            consider(p_fine)

    # Phase 6: If time remains, try another diverse start
    now = time.monotonic()
    if now < hard - 80.0 and len(explored) > 1:
        alt = explored[1]
        sc_alt = _score(alt)
        if math.isfinite(sc_alt) and sc_alt < best_score + 0.02:
            alt_stop = min(hard - 10.0, now + 60.0)
            p_alt = alt.copy()
            p_alt = _lbfgs_optimize(p_alt, alt_stop, consider=consider, fine_only=False)
            cstop = min(hard - 5.0, time.monotonic() + max(10.0, 0.3 * max(0.0, hard - time.monotonic())))
            if cstop > time.monotonic() + 8.0:
                p_alt, _ = _coord_polish(p_alt, cstop, G=64, seed=seed + 55555, consider=consider)
            consider(p_alt)

    # Final: one more coordinate squeeze
    now = time.monotonic()
    if now < hard - 8.0:
        p_final = best_arr.copy()
        cstop = min(hard - 2.0, now + max(5.0, 0.6 * max(0.0, hard - 2.0 - now)))
        if cstop > now + 5.0:
            p_final, _ = _coord_polish(p_final, cstop, G=80, seed=seed + 777777, consider=consider)
            consider(p_final)

    # Return best
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = _normalize(parent)
            s = float(out.sum())
        if out.size == 0 or not math.isfinite(s) or s < 0.01:
            out = np.full(64, 1.0 / 64.0)
        out = out / float(out.sum())
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            fallback = _normalize(parent)
            ls2 = [float(x) for x in fallback]
            sc2 = float(evaluate_sequence(ls2))
            if math.isfinite(sc2):
                return ls2
            return [1.0]
        return ls
    except Exception:
        return [1.0]
