```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, fft, ifft, next_fast_len
from scipy.optimize import minimize


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 30.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(3072, dtype=np.float64)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    parent = np.maximum(parent, 0.0)
    n = parent.size
    if n == 0 or parent.sum() < 1e-10:
        parent = np.ones(3072, dtype=np.float64)
        n = 3072
    parent = parent / float(parent.sum())

    best_arr = parent.copy()
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_arr]))
    except Exception:
        best_score = math.inf

    # FFT setup
    fft_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)

    def conv_self(a):
        F = rfft(a, fft_size)
        c = irfft(F * F, fft_size)[:conv_len]
        return np.maximum(c, 0.0)

    def score_fast(a):
        c = conv_self(a)
        m = float(c.max())
        s = float(a.sum())
        if s < 0.01:
            return math.inf
        return 2.0 * n * m / (s * s)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.maximum(a, 0.0)
            a = np.minimum(a, 1000.0)
            if a.size == 0 or float(a.sum()) < 0.01:
                return
            sc = float(evaluate_sequence([float(x) for x in a]))
            if math.isfinite(sc) and sc < best_score:
                best_score = sc
                best_arr = a.copy()
        except Exception:
            pass

    def softmax(u):
        u = u - u.max()
        e = np.exp(u)
        s = e.sum()
        if s <= 0 or not math.isfinite(s):
            return np.ones_like(u) / u.size
        return e / s

    def p_to_u(p):
        return np.log(np.maximum(p, 1e-300))

    # === LSE Optimization ===
    def lse_objective(u, T):
        p = softmax(u)
        c = conv_self(p)
        cmax = float(c.max())
        if T <= 0 or not math.isfinite(T):
            T = 1e-5
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            f_val = cmax
        else:
            f_val = cmax + T * math.log(sum_exp / conv_len)
            w = exps / sum_exp

        # Gradient: df/du[i] = p[i] * (2*sum_k w[k]*p[k-i] - sum_j p[j]*2*sum_k w[k]*p[k-j])
        # = 2*p[i] * (r[i] - dot(p, r)) where r[i] = sum_k w[k]*p[k-i]
        # Compute r via FFT cross-correlation
        W_fft = fft(w, cross_size)
        P_fft = fft(p, cross_size)
        R = ifft(W_fft * np.conj(P_fft))[:n].real
        r = np.maximum(R, 0.0)  # should be nonnegative

        tdp = float(np.dot(p, r))
        grad = 2.0 * p * (r - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return 2.0 * n * f_val, grad

    # Phase 1: LSE with L-BFGS-B
    def lse_phase(p_init, stop_time):
        p = p_init.copy()
        u = p_to_u(p)
        local_best = p.copy()
        local_best_sc = score_fast(p)

        cmax0 = float(conv_self(p).max())
        stages = [
            (0.2, 0.0), (0.08, 0.2), (0.03, 0.3), (0.01, 0.35), (0.004, 0.4), (0.001, 0.45), (0.0003, 0.5)
        ]
        total_wt = sum(wt for _, wt in stages)
        elapsed_total = 0
        remaining_frac = 1.0

        for frac, cum_frac in stages:
            now = time.monotonic()
            if now > stop_time - 10.0:
                break
            # Time for this stage
            stage_time = (cum_frac - (stages.index((frac, cum_frac)) and sum(w for _, w in stages[:stages.index((frac, cum_frac))]) or 0)) / total_wt * (stop_time - now - 10.0)
            stage_end = min(stop_time - 8.0, now + max(10.0, stage_time))

            cmax_cur = float(conv_self(p).max())
            T = max(1e-6, frac * cmax_cur)

            stale = 0
            while time.monotonic() < stage_end - 2.0:
                opts = {"maxiter": 50, "gtol": 1e-14, "ftol": 1e-18,
                        "maxls": 40, "maxcor": 30, "maxdisp": 0, "disp": False}
                try:
                    res = minimize(lse_objective, u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
                except Exception:
                    break
                if res.x is not None and np.all(np.isfinite(res.x)):
                    p_cur = softmax(res.x)
                    sc_cur = score_fast(p_cur)
                    if math.isfinite(sc_cur) and sc_cur < local_best_sc - 1e-12:
                        local_best_sc = sc_cur
                        local_best = p_cur.copy()
                        p = p_cur.copy()
                        u = p_to_u(p)
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 4:
                            break
                if getattr(res, "status", 1) == 0:
                    break

        return local_best

    # Phase 2: Mirror descent with subgradient
    def mirror_descent(p_init, stop_time, seed_md=42):
        rng = np.random.default_rng(seed_md)
        p = p_init.copy()
        local_best = p.copy()
        local_best_sc = score_fast(p)
        c = conv_self(p)

        eta = 0.1 / n  # initial step size
        eta_min = 1e-10
        eta_max = 1.0
        patience = 0
        max_patience = 200

        for it in range(1000000):
            now = time.monotonic()
            if now > stop_time - 5.0:
                break

            # Every 500 iterations, verify with full conv
            if it % 500 == 0:
                c = conv_self(p)

            cmax = float(c.max())
            # Find top-K peaks for subgradient
            k_peaks = min(5, conv_len)
            top_k = np.argpartition(c, -k_peaks)[-k_peaks:]
            top_k = top_k[np.argsort(c[top_k])[::-1]]
            peak_val = c[top_k[0]]
            # Weight by how close to max
            weights = np.exp((c[top_k] - peak_val) / (0.01 * cmax + 1e-30))
            weights /= weights.sum()

            # Subgradient: g[i] = 2 * sum_w w_m * a[p_m - i]
            # For efficiency, if top peak dominates, just use that one
            if weights[0] > 0.95:
                p_peak = int(top_k[0])
                # g[i] = 2*a[p_peak - i] for 0 <= p_peak - i < n
                lo = max(0, p_peak - (n - 1))
                hi = min(n - 1, p_peak)
                g = np.zeros(n)
                for i in range(lo, hi + 1):
                    g[i] = 2.0 * p[p_peak - i]
            else:
                g = np.zeros(n)
                for w_m, p_m in zip(weights, top_k):
                    p_m = int(p_m)
                    lo = max(0, p_m - (n - 1))
                    hi = min(n - 1, p_m)
                    for i in range(lo, hi + 1):
                        g[i] += 2.0 * w_m * p[p_m - i]

            # Multiplicative update
            # a_new[i] = a[i] * exp(-eta * g[i])
            # Normalize after
            log_update = -eta * g
            # Clip to avoid underflow
            log_update = np.clip(log_update, -50, 50)
            p_new = p * np.exp(log_update)
            p_new = np.maximum(p_new, 0.0)
            s_new = float(p_new.sum())
            if s_new < 1e-300:
                break
            p_new = p_new / s_new

            # Score improvement check (use subgradient score as proxy)
            # Full evaluation every 200 iters
            if it % 200 == 0:
                sc_new = score_fast(p_new)
                if math.isfinite(sc_new) and sc_new < local_best_sc - 1e-14:
                    local_best_sc = sc_new
                    local_best = p_new.copy()
                    c = conv_self(p_new)
                    patience = 0
                    # Increase step size slightly
                    eta = min(eta * 1.05, eta_max)
                else:
                    patience += 1
                    eta = max(eta * 0.8, eta_min)
            else:
                # Skip full eval, just update
                pass

            p = p_new

            if patience > max_patience:
                # Add small perturbation
                p = p * (1.0 + rng.normal(0, 0.01, n))
                p = np.maximum(p, 0.0)
                s = p.sum()
                if s > 0:
                    p = p / s
                patience = 0
                eta = min(eta * 1.5, eta_max)

        # Final check
        sc_final = score_fast(p)
        if math.isfinite(sc_final) and sc_final < local_best_sc:
            local_best_sc = sc_final
            local_best = p.copy()
        return local_best

    # Phase 3: Localized peak reduction
    def peak_reduction(p_init, stop_time, seed_pr=99):
        rng = np.random.default_rng(seed_pr)
        p = p_init.copy()
        local_best = p.copy()
        local_best_sc = score_fast(p)
        c = conv_self(p)

        it = 0
        while time.monotonic() < stop_time - 5.0:
            it += 1
            c = conv_self(p)
            cmax = float(c.max())
            cmin_nontriv = float(c.mean())

            # Find the peak
            p_peak = int(np.argmax(c))

            # Try to reduce the peak by adjusting a in the neighborhood
            # The peak c[p_peak] = sum_i a[i]*a[p_peak-i]
            # We want to reduce this while not increasing other peaks too much

            # Strategy: select a random coordinate i in the "peak support"
            # and try to reduce it (or increase a non-peak-contributing coordinate)
            lo = max(0, p_peak - (n - 1))
            hi = min(n - 1, p_peak)

            # Try a random move: reduce a[i] for i in [lo, hi] and increase a[j] for j not in peak support
            if hi > lo:
                i = rng.integers(lo, hi + 1)
            else:
                i = rng.integers(0, n)

            # Try reducing a[i] by a small amount and distributing to a random other coordinate
            delta = rng.uniform(0.001, 0.01) * float(p[i])
            if delta < 1e-15:
                continue

            # Pick a target coordinate (prefer ones not contributing to the peak)
            # Coordinates j where c is small
            c_flat = c.copy()
            c_flat[max(0, i - (n - 1)):min(conv_len, i + n)] = -1  # mask peak-contributing
            # Actually, let's just pick randomly
            j = rng.integers(0, n)
            if j == i:
                j = (j + 1) % n

            p_trial = p.copy()
            p_trial[i] -= delta
            p_trial[j] += delta
            p_trial = np.maximum(p_trial, 0.0)
            s_trial = float(p_trial.sum())
            if s_trial < 1e-10:
                continue
            p_trial = p_trial / s_trial

            sc_trial = score_fast(p_trial)
            if math.isfinite(sc_trial) and sc_trial < local_best_sc - 1e-15:
                local_best_sc = sc_trial
                local_best = p_trial.copy()
                p = p_trial.copy()

            # Also try a "swap" move: move mass from a peak contributor to a non-contributor
            if it % 10 == 0:
                # Identify top contributors to the peak
                contribs = np.zeros(hi - lo + 1)
                for idx in range(lo, hi + 1):
                    k = p_peak - idx
                    if 0 <= k < n:
                        contribs[idx - lo] = p[idx] * p[k]
                top_idx = np.argmax(contribs) + lo
                # Move mass from top_idx to a random low-contribution coordinate
                delta2 = 0.01 * float(p[top_idx])
                if delta2 > 1e-15:
                    j2 = rng.integers(0, n)
                    if j2 != top_idx:
                        p_trial2 = p.copy()
                        p_trial2[top_idx] -= delta2
                        p_trial2[j2] += delta2
                        p_trial2 = np.maximum(p_trial2, 0.0)
                        s2 = float(p_trial2.sum())
                        if s2 > 1e-10:
                            p_trial2 = p_trial2 / s2
                            sc2 = score_fast(p_trial2)
                            if math.isfinite(sc2) and sc2 < local_best_sc - 1e-15:
                                local_best_sc = sc2
                                local_best = p_trial2.copy()
                                p = p_trial2.copy()

        return local_best

    # === Execute phases ===
    # Phase 1: LSE
    lse_stop = t0 + min(350.0, 0.30 * budget)
    if lse_stop > time.monotonic() + 30.0:
        try:
            p1 = lse_phase(parent, lse_stop)
            if p1 is not None and p1.size == n and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    # Phase 2: Mirror descent
    md_stop = min(hard - 120.0, time.monotonic() + 500.0)
    if md_stop > time.monotonic() + 60.0:
        try:
            p2 = mirror_descent(best_arr, md_stop, seed_md=int(seed + 1))
            if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p2 = p2 / float(p2.sum())
                consider(p2)
        except Exception:
            pass

    # Phase 3: Second LSE pass (fine)
    lse2_stop = min(hard - 60.0, time.monotonic() + 200.0)
    if lse2_stop > time.monotonic() + 30.0:
        try:
            p3 = lse_phase(best_arr, lse2_stop)
            if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and p3.sum() > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)
        except Exception:
            pass

    # Phase 4: Peak reduction (local search)
    pr_stop = hard - 30.0
    if pr_stop > time.monotonic() + 30.0:
        try:
            p4 = peak_reduction(best_arr, pr_stop, seed_pr=int(seed + 77))
            if p4 is not None and p4.size == n and np.all(np.isfinite(p4)) and p4.sum() > 0:
                p4 = p4 / float(p4.sum())
                consider(p4)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.maximum(out, 0.0)
        out = np.minimum(out, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
```
