from scipy.fft import rfft, irfft, next_fast_len
import time
import math

def _conv_self(a):
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = _conv_self(a)
    return float(2.0 * n * c.max() / (s * s))

def _subgrad(a, c, rel_tol=0.005):
    """Compute subgradient direction using active set."""
    n = len(a)
    L = len(c)
    c_max = float(c.max())
    if c_max <= 0:
        return np.ones(n) / n
    thresh = c_max * (1.0 - rel_tol)
    active = c >= thresh
    k = int(active.sum())
    if k == 0:
        active[int(c.argmax())] = True
        k = 1
    s = active.astype(np.float64) / k
    # Cross-correlation: b_i = sum_j s_j * a_{j-i}
    size = next_fast_len(L + n - 1)  # 3n-2
    S_c = np.fft.fft(s, size)
