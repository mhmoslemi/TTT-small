import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    n = a.size
    if n <= 512:
        c = np.convolve(a, a, mode="full")
    else:
        L = 2 * n - 1
        size = next_fast_len(L)
        F = rfft(a, size)
        c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _score_fft(a):
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _score_exact_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * a.size * m / (s * s))


def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == 1:
        return np.full(n, float(src[0]) / n)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m) + 0.5) / m
        fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _lse_optimize(p, stop_time, consider=None, fine_only=False):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0.0:
            return 1e100, np.zeros_like(uu)
        if T <= 0.0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -400.0, 400.0)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    def run_chunk(uu, T, end_stage):
        try:
            f0, _ = lse_func(uu, T)
        except Exception:
            return uu
        if not math.isfinite(f0):
            return uu
        best_u = np.array(uu, dtype=np.float64, copy=True)
        best_f = float(f0)
        stale = 0
        while time.monotonic() < end_stage - 1.5:
            maxit = 40 if n > 2000 else 60
            opts = {
                "maxiter": maxit,
                "gtol": 1e-14,
                "ftol": 1e-16,
                "maxls": 30,
                "maxcor": 30,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == uu.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-16:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, "status", 1) == 0:
                break
        return best_u

    if fine_only:
        stages = ((0.001, 0.30), (0.0002, 0.40), (0.00005, 0.30))
    else:
        stages = (
            (0.08, 0.06),
            (0.03, 0.10),
            (0.01, 0.14),
            (0.003, 0.16),
            (0.001, 0.20),
            (0.0003, 0.20),
            (0.0001, 0.14),
        )

    start = time.monotonic()
    avail = float(stop_time - start - 6.0)
    if avail <= 4.0:
        return p

    best_p = p.copy()
    best_sc = _score_fft(p)
    total_w = float(sum(w for _, w in stages))
    if total_w <= 0.0:
        total_w = 1.0

    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 1.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        sc_cur = _score_fft(p_cur)
        if not math.isfinite(sc_cur):
            p = best_p.copy()
            u = p_to_u(p)
            continue
        if sc_cur < best_sc:
            best_sc = sc_cur
            best_p = p_cur.copy()
            p = p_cur.copy()
        elif sc_cur > best_sc + max(0.003, 3e-4 * best_sc):
            p = best_p.copy()
        else:
            p = p_cur.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, max_sweeps=50000, G=None, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    if G is None:
        if n <= 450:
            G = 50
        elif n <= 900:
            G = 40
        elif n <= 2000:
            G = 32
        else:
            G = 26
    G = int(max(14, G))

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.25 or S_total < 0.75:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls].copy()

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(20.0 * v, 60.0 * S / n, 0.05 * S, 1e-12)
            x_hi = min(x_hi, 10.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, S / n)

            di = 2 * i
            has_di = 0 <= di < Ltot

            def eval_xs(xs, base, infl, max_const, di_local, has_di_local):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                if has_di_local:
                    dval = base[di_local] + w * infl[di_local] + w * w
                    maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 <= v <= x_hi:
                pts = np.unique(np.concatenate((pts, [v])))
            vals = eval_xs(pts, base, infl, max_const, di, has_di)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            dw = x_hi / float(G)
            lo = max(0.0, best_x - 2.5 * dw)
            hi = min(x_hi, best_x + 2.5 * dw)
            if hi <= lo:
                hi = lo + 1e-14
            fine_pts = np.linspace(lo, hi, 30)
            fine_pts = np.unique(np.concatenate((fine_pts, [best_x])))
            vals2 = eval_xs(fine_pts, base, infl, max_const, di, has_di)
            if np.any(np.isfinite(vals2)):
                imin2 = int(np.argmin(vals2))
                vx2 = float(fine_pts[imin2])
                vv2 = float(vals2[imin2])
                if vv2 < best_val:
                    best_x = vx2
                    best_val = vv2

            gr = (math.sqrt(5.0) - 1.0) / 2.0
            g_lo = max(0.0, best_x - dw)
            g_hi = min(x_hi, best_x + dw)
            if g_hi - g_lo > 1e-16:
                a_gs = g_hi - gr * (g_hi - g_lo)
                b_gs = g_lo + gr * (g_hi - g_lo)
                fa = float(eval_xs(np.array([a_gs]), base, infl, max_const, di, has_di)[0])
                fb = float(eval_xs(np.array([b_gs]), base, infl, max_const, di, has_di)[0])
                for _ in range(14):
                    if not math.isfinite(fa):
                        fa = 1e100
                    if not math.isfinite(fb):
                        fb = 1e100
                    if fa < fb:
                        g_hi = b_gs
                        b_gs = a_gs
                        fb = fa
                        a_gs = g_hi - gr * (g_hi - g_lo)
                        fa = float(eval_xs(np.array([a_gs]), base, infl, max_const, di, has_di)[0])
                    else:
                        g_lo = a_gs
                        a_gs = b_gs
                        fa = fb
                        b_gs = g_lo + gr * (g_hi - g_lo)
                        fb = float(eval_xs(np.array([b_gs]), base, infl, max_const, di, has_di)[0])
                    if g_hi - g_lo < 1e-16:
                        break
                gs_x = (g_lo + g_hi) / 2.0
                gv = float(eval_xs(np.array([gs_x]), base, infl, max_const, di, has_di)[0])
                if math.isfinite(gv) and gv < best_val:
                    best_x = gs_x
                    best_val = gv

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 10.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    if has_di:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        np.maximum(c, 0.0, out=c)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-16:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 6.0:
                    try:
                        noise_amt = 0.03 + 0.02 * math.exp(-flat * 0.1)
                        p = p * np.exp(rng.normal(0.0, noise_amt, n))
                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss > 0.0:
                            p /= ss
                        c = _conv_self(p)
                        S_total = 1.0
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider is not None:
                                consider(best_p)
                        flat = 0
                    except Exception:
                        break
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 20.0
    usable = max(1.0, hard - t0)
    final_reserve = max(12.0, 0.12 * usable) if usable > 80.0 else 0.0
    phase_limit = hard - final_reserve

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64)
        best_arr /= 64.0
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    def make_cosine(n, phase, amp, sd):
        rng = np.random.default_rng(sd)
        x = np.arange(n, dtype=np.float64) / n
        base = 1.0 + amp * np.cos(2.0 * np.pi * (x + phase))
        base = np.maximum(base, 0.01)
        base = base / base.sum()
        noise = 1.0 + 0.05 * rng.standard_normal(n)
        base = base * noise
        base = np.maximum(base, 0.0)
        s = base.sum()
        if s > 0:
            base /= s
        return base

    def make_two_seg(n, frac, amp, sd):
        rng = np.random.default_rng(sd)
        m = int(n * frac)
        m = max(1, min(n - 1, m))
        p = np.ones(n, dtype=np.float64)
        p[:m] = amp
        p = p * np.exp(0.03 * rng.standard_normal(n))
        p = np.maximum(p, 0.0)
        s = p.sum()
        if s > 0:
            p /= s
        return p

    def run_attempt(n, src_arr, end_time, sd, do_coord=True):
        nonlocal best_arr, best_score
        n = int(max(1, n))
        p = _resample_mid(src_arr, n)
        if p.size != n or not np.all(np.isfinite(p)) or p.sum() <= 0.0:
            p = np.full(n, 1.0 / n)
        p = p / float(p.sum())
        consider(p)

        phase_len = max(0.0, float(end_time) - time.monotonic())
        if phase_len <= 8.0:
            return

        lbfgs_stop = min(float(end_time) - 8.0, time.monotonic() + 0.70 * phase_len)
        if lbfgs_stop > time.monotonic() + 8.0:
            try:
                p2 = _lse_optimize(p, lbfgs_stop, consider=consider, fine_only=False)
                if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0.0:
                    p = p2 / float(p2.sum())
            except Exception:
                pass

        rem = float(end_time) - time.monotonic()
        if do_coord and rem > 12.0:
            coord_stop = min(
                float(end_time) - 1.0,
                time.monotonic() + max(12.0, min(150.0, 0.35 * phase_len)),
            )
            try:
                p3, _ = _coord_polish(p, coord_stop, max_sweeps=50000, seed=sd, consider=consider)
                if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and p3.sum() > 0.0:
                    p = p3
            except Exception:
                pass

        if p is not None and p.size == n and np.all(np.isfinite(p)) and float(p.sum()) > 0.0:
            consider(p)

    # Determine lengths to try
    parent_len = int(best_arr.size) if best_arr.size > 0 else 3072
    lengths = [parent_len]
    # Try a few other lengths for diversity
    if parent_len > 1024:
        lengths.append(max(512, parent_len // 2))
    if parent_len >= 2048:
        lengths.append(min(4096, parent_len + 512))
    lengths = sorted(set(lengths))

    # Allocate time based on number of lengths
    if len(lengths) == 1:
        weights = [1.0]
    elif len(lengths) == 2:
        weights = [0.80, 0.20]
    else:
        weights = [0.65, 0.20, 0.15]

    rng_top = np.random.default_rng(seed)
    phase_limit = min(phase_limit, hard - 10.0)

    for idx, L in enumerate(lengths):
        now = time.monotonic()
        if now >= phase_limit - 5.0:
            break
        rem_to_end = max(0.0, phase_limit - now)
        if rem_to_end <= 10.0:
            break
        w_rem = float(sum(weights[idx:]))
        if w_rem <= 0:
            w_rem = 1.0
        end_phase = min(phase_limit, now + (weights[idx] / w_rem) * rem_to_end)
        if end_phase <= now + 8.0:
            break

        src = best_arr.copy() if best_arr.size > 0 else parent.copy()
        L_use = int(L)
        src_L = _resample_mid(src, L_use)

        if idx == 0 and (end_phase - now) > 150.0:
            # Main length: try parent and a cosine init
            v0 = src_L
            mid = min(end_phase, now + 0.50 * (end_phase - now))
            run_attempt(L_use, v0, mid, int(seed + 100), do_coord=True)
            if time.monotonic() < end_phase - 10.0:
                v1 = make_cosine(L_use, 0.0, 0.3, int(seed + 300))
                run_attempt(L_use, v1, end_phase, int(seed + 400), do_coord=True)
        elif idx == 1:
            # Secondary: try a different init
            v1 = make_two_seg(L_use, 0.5, 1.5, int(seed + 600))
            run_attempt(L_use, v1, end_phase, int(seed + 700), do_coord=True)
        else:
            # Tertiary
            v1 = make_cosine(L_use, 0.25, 0.5, int(seed + 800))
            run_attempt(L_use, v1, end_phase, int(seed + 900), do_coord=True)

    # Final squeeze on the best solution found
    final_end = min(hard - 6.0, t0 + budget - 4.0)
    now = time.monotonic()
    if final_end > now + 25.0 and best_arr.size > 0:
        nbest = int(best_arr.size)
        rem = final_end - now

        # LSE fine-tune
        if rem > 50.0 and nbest <= 5000:
            lstop = min(final_end - 30.0, now + 0.40 * rem)
            try:
                p2 = _lse_optimize(best_arr, lstop, consider=consider, fine_only=True)
                if p2 is not None and p2.size == nbest and np.all(np.isfinite(p2)) and p2.sum() > 0.0:
                    p2 = p2 / float(p2.sum())
                    consider(p2)
            except Exception:
                pass

        # Coordinate polish
        now = time.monotonic()
        cstop = min(final_end - 5.0, now + max(15.0, 0.40 * (final_end - now)))
        if cstop > now + 10.0:
            try:
                p3, _ = _coord_polish(best_arr, cstop, max_sweeps=50000, seed=int(seed + 999999), consider=consider)
                if p3 is not None and p3.size == nbest and np.all(np.isfinite(p3)) and p3.sum() > 0.0:
                    p3 = p3 / float(p3.sum())
                    consider(p3)
            except Exception:
                pass

        # Another short coordinate pass
        now = time.monotonic()
        cstop2 = min(final_end - 2.0, now + max(5.0, 0.20 * (final_end - now)))
        if cstop2 > now + 5.0:
            try:
                p4, _ = _coord_polish(best_arr, cstop2, max_sweeps=50000, seed=int(seed + 777777), consider=consider)
                if p4 is not None and p4.size == nbest and np.all(np.isfinite(p4)) and p4.sum() > 0.0:
                    p4 = p4 / float(p4.sum())
                    consider(p4)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
