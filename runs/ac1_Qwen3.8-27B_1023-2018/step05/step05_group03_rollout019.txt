        cv = irfft(Fr * G, cross_size)[:3*n-2]
        q = cv[n-1:2*n-1]
        gid = 2.0 * q
        gd = float(np.dot(pp, gid))
        grad = pp * (gid - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    p_safe = np.maximum(p, 1e-30)
    u = np.log(p_safe)
    u -= float(np.mean(u))
    c0 = _fft_conv(p)
    cmax0 = float(c0.max())
    T = max(1e-16, T_frac * cmax0)

    best_f, _ = lse_func(u, T)
    best_u = u.copy()

    for _ in range(max_chunks):
        if time.monotonic() > stop_time - 1:
            break
        opts = {'maxiter': 25, 'gtol': 1e-11, 'ftol': 1e-15,
                'maxls': 20, 'maxcor': 18, 'disp': False, 'maxdisp': 0}
        try:
            res = minimize(lse_func, best_u, args=(T,),
                           method='L-BFGS-B', jac=True, options=opts)
        except Exception:
            break
        if res.x is not None:
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if x.size == n and np.all(np.isfinite(x)):
                fv = float(res.fun)
                if math.isfinite(fv) and fv < best_f - 1e-13:
                    best_f = fv
                    best_u = x.copy()
                else:
                    break
        else:
            break
        if getattr(res, 'status', 1) == 0:
            break

    p_out = _softmax(best_u)
    p_out = np.maximum(p_out, 0.0)
    ss = float(p_out.sum())
    if math.isfinite(ss) and ss > 0:
        p_out /= ss
    else:
        p_out = p.copy()
    return p_out

def _coord_descent(p, stop_time, G=56, max_sweeps=500, seed=42, consider=None):
    n = p.size
    L = 2 * n - 1
    p = _softmax(np.log(np.maximum(p, 1e-30)))
    c = _fft_conv(p)
    S = 1.0
    infl = 2.0 * p
    cur_m = float(c.max())
    cur_score = 2.0 * n * cur_m
    best_score = cur_score
    best_p = p.copy()

    if consider:
        try: consider(best_p)
        except: pass

    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)

        for pos in range(n):
            if time.monotonic() > stop_time - 0.3:
                break
            i = int(order[pos])
            v = float(p[i])
            start = i
            end = min(i + n, L)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end]
            # precompute max of unaffected part
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < L:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

            di = 2 * i - start
            xhi = max(25.0 * v, 25.0 / float(n), 0.008)
            xhi = min(xhi, 10.0 / float(n) + v)

            # Build evaluation points
            m1 = max(16, G)
            minx = max(1e-12, 1e-6)
            pts = np.geomspace(minx, xhi, m1)
            if v > 1e-14:
                m2 = max(6, G // 3)
                pts2 = np.geomspace(max(1e-12, v * 0.15), min(xhi, v * 5.0), m2)
                pts = np.concatenate([pts, pts2])
            pts = np.unique(np.clip(pts, 0.0, xhi))
            pts = np.insert(pts, 0, 0.0)
            if v > 0:
                pts = np.append(pts, v)

            # Evaluate: score = 2*n*max(R')/(S+d)^2
            d_vals = pts - v
            den = S + d_vals
            valid = den > 1e-14
            if not np.any(valid):
                continue
            aff = base[None, :] + d_vals[:, None] * infl[None, start:end]
            if 0 <= di < Ls:
                aff[:, di] = base[di] + d_vals * infl[di] + d_vals * d_vals
            np.maximum(aff, max_const, out=aff)
            mv = np.max(aff, axis=1)
            vals = np.full(pts.size, math.inf)
            if np.any(valid):
                vals[valid] = (2.0 * n * mv[valid]) / (den[valid] ** 2)
            vals = np.where(np.isfinite(vals), vals, math.inf)

            if not np.any(np.isfinite(vals)):
                continue
            im = int(np.argmin(vals))
            bx = float(pts[im])
            bv = float(vals[im])

            # Fine search around best
            if math.isfinite(bv) and bv < cur_score * (1 - 1e-8):
                step = max(v, 1.0 / float(n), 1e-10)
                lo = max(0.0, bx - 3.0 * step)
                hi = min(xhi, bx + 3.0 * step)
                if hi > lo:
                    fine = np.linspace(lo, hi, 40)
                    if lo > 1e-14:
                        fine = np.concatenate([fine, np.geomspace(lo, hi, 20)])
                    fine = np.unique(np.clip(fine, 0.0, xhi))
                    d_f = fine - v
                    den_f = S + d_f
                    vld_f = den_f > 1e-14
                    if np.any(vld_f):
                        aff_f = base[None, :] + d_f[:, None] * infl[None, start:end]
                        if 0 <= di < Ls:
                            aff_f[:, di] = base[di] + d_f * infl[di] + d_f * d_f
                        np.maximum(aff_f, max_const, out=aff_f)
                        mv_f = np.max(aff_f, axis=1)
                        vals_f = np.full(fine.size, math.inf)
                        vals_f[vld_f] = (2.0 * n * mv_f[vld_f]) / (den_f[vld_f] ** 2)
                        vals_f = np.where(np.isfinite(vals_f), vals_f, math.inf)
                        if np.any(np.isfinite(vals_f)):
                            im2 = int(np.argmin(vals_f))
                            if vals_f[im2] < bv:
                                bx = float(fine[im2])
                                bv = float(vals_f[im2])

            if math.isfinite(bv) and bv < cur_score - 1e-12 and 0.0 <= bx <= xhi:
                d = bx - v
                if abs(d) > 1e-15:
                    c[start:end] += d * infl[start:end]
                    diag = 2 * i
                    if 0 <= diag < L:
                        c[diag] += d * d
                    p[i] = bx
                    infl[i] = 2.0 * bx
                    S += d
                    cur_score = bv
                    cur_m = c[start:end + 1].max() if start + 1 < L else float(c.max())
                    improved = True

        # Renormalize
        if S > 1e-14:
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            infl *= inv
            S = 1.0
        else:
            break
        c = _fft_conv(p)
        desc = 2.0 * n * float(c.max())
        if math.isfinite(desc) and desc < best_score - 1e-13:
            best_score = desc
            best_p = p.copy()
            if consider:
                try: consider(best_p)
                except: pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                # Perturbation to escape local minimum
                if time.monotonic() < stop_time - 30:
                    sigma = 0.04 + 0.04 * min(flat - 3, 4)
                    noise = rng.normal(0, sigma, n)
                    q = p * np.exp(noise)
                    qs = float(q.sum())
                    if qs > 0 and math.isfinite(qs):
                        p = q / qs
                        c = _fft_conv(p)
                        infl = 2.0 * p
                        S = 1.0
                        cur_score = 2.0 * n * float(c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    return best_p, best_score

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except:
        budget = 1300.0
    budget = max(20.0, budget)
    hard = t0 + budget - 15.0
    if hard <= t0 + 5:
        hard = t0 + max(3.0, budget - 1.0)

    parent = _get_parent()
    n = int(parent.size)
    best_arr = parent.copy()
    best_score = _score_exact(best_arr)

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            s = float(arr.sum())
            if s < 0.01 or not math.isfinite(s):
                return
            arr = arr / s
            sc = _score_exact(arr)
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_arr = arr.copy()
                best_score = sc
        except:
            pass

    rng_top = np.random.default_rng(seed)

    # Phase 1: Quick exploration (up to 40s)
    p1_stop = min(hard - 400, t0 + 40)
    base = best_arr.copy()
    for q_exp in (0.5, 0.7, 1.3, 1.6, 2.0, 2.5):
        if time.monotonic() > p1_stop:
            break
        cand = np.power(np.maximum(base, 1e-30), q_exp)
        cs = float(cand.sum())
        if cs > 0:
            cand /= cs
            consider(cand)

    for _ in range(8):
        if time.monotonic() > p1_stop:
            break
        side = 1.0 if rng_top.random() < 0.5 else -1.0
        amp = 0.4 + 1.2 * rng_top.random()
        x = (np.arange(n) + 0.5) / n
        cand = base * np.exp(side * amp * (2.0 * x - 1.0))
        cand = np.maximum(cand, 0)
        cs = float(cand.sum())
        if cs > 0:
            cand /= cs
            consider(cand)

    # sinusoidal modulation
    for _ in range(6):
        if time.monotonic() > p1_stop:
            break
        f1 = 1 + int(rng_top.integers(0, 5))
        ph = 2 * math.pi * rng_top.random()
        amp = 0.2 + 0.5 * rng_top.random()
        x = (np.arange(n) + 0.5) / n
        cand = best_arr * (1.0 + amp * np.sin(2 * math.pi * f1 * x + ph))
        cand = np.maximum(cand, 0)
        cs = float(cand.sum())
        if cs > 0:
            cand /= cs
            consider(cand)

    # Phase 2: L-BFGS coarse-to-fine (up to ~250s)
    p2_stop = min(hard - 350, time.monotonic() + 250)
    lse_stages = [(0.04, 5), (0.015, 6), (0.005, 6), (0.0015, 7), (0.0005, 8)]
    chunk_time = max(10.0, (p2_stop - time.monotonic()) / max(1, sum(mx for _, mx in lse_stages)))
    for frac, mx in lse_stages:
        if time.monotonic() > p2_stop - 2:
            break
        st = time.monotonic() + chunk_time * mx
        st = min(st, p2_stop - 1)
        if st <= time.monotonic() + 5:
            continue
        try:
            p_try = best_arr.copy()
            p_try = _lbfgs_stage(p_try, st, frac, max_chunks=mx, consider=consider)
            sc = _score_fft(p_try)
            if math.isfinite(sc) and sc < best_score - 1e-10:
                best_arr = p_try
                best_score = sc
        except:
            pass

    # Phase 3: Coordinate descent (up to ~350s)
    p3_stop = min(hard - 200, time.monotonic() + 350)
    if p3_stop > time.monotonic() + 20:
        try:
            best_arr, _ = _coord_descent(
                best_arr, p3_stop, G=52, max_sweeps=400,
                seed=seed + 101, consider=consider)
        except:
            pass

    # Phase 4: Fine L-BFGS (up to ~120s)
    p4_stop = min(hard - 100, time.monotonic() + 120)
    if p4_stop > time.monotonic() + 15:
        lse_fine = [(0.0003, 6), (0.0001, 7), (0.00004, 8)]
        ct = max(8.0, (p4_stop - time.monotonic()) / max(1, sum(mx for _, mx in lse_fine)))
        for frac, mx in lse_fine:
            if time.monotonic() > p4_stop - 2:
                break
            st = min(time.monotonic() + ct * mx, p4_stop - 1)
            if st <= time.monotonic() + 5:
                continue
            try:
                p_try = best_arr.copy()
                p_try = _lbfgs_stage(p_try, st, frac, max_chunks=mx, consider=consider)
                sc = _score_fft(p_try)
                if math.isfinite(sc) and sc < best_score - 1e-10:
                    best_arr = p_try
                    best_score = sc
            except:
                pass

    # Phase 5: Final coordinate polish (remaining time)
    p5_stop = hard - 5
    if p5_stop > time.monotonic() + 10:
        try:
            best_arr, _ = _coord_descent(
                best_arr, p5_stop, G=60, max_sweeps=1000,
                seed=seed + 999, consider=consider)
        except:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = parent.copy()
            s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = np.full(64, 1.0 / 64.0)
        out = out / float(out.sum())
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            ls = [float(x) / float(np.sum(parent)) for x in parent]
            final_sc = evaluate_sequence(ls)
            if not math.isfinite(final_sc):
                ls = [1.0]
        return ls
    except:
        return [1.0]
