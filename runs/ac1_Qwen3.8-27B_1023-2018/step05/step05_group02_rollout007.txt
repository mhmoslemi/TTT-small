def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * a.size * m / (s * s))


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _p_to_u(p):
    uu = np.log(np.maximum(p, 1e-300))
    return uu - float(np.mean(uu))


def _lbfgs_lse(p, stop_time, temperature_frac, consider=None, max_chunks=20):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    u = _p_to_u(p)

    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    c = cfast(p)
    cmax_init = float(c.max())
    T = max(1e-300, temperature_frac * cmax_init)

    def lse_func(uu, Tval):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0.0:
            return 1e100, np.zeros_like(uu)
        scaled = (c - cmax) / Tval
        scaled = np.clip(scaled, -500.0, 500.0)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
        else:
            w = exps / sum_exp
        f = cmax + Tval * float(np.log(sum_exp / c.size))
        if not math.isfinite(f):
            f = cmax
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    best_u = np.array(u, dtype=np.float64, copy=True)
    best_f = float('inf')
    try:
        f0, _ = lse_func(best_u, T)
        best_f = f0
    except Exception:
        return p

    chunk_count = 0
    stale = 0
    while time.monotonic() < stop_time - 3.0 and chunk_count < max_chunks:
        maxit = 40 if n > 1500 else 60
        opts = {
            "maxiter": maxit,
            "gtol": 1e-14,
            "ftol": 1e-18,
            "maxls": 25,
            "maxcor": 40,
            "maxdisp": 0,
            "disp": False,
        }
        try:
            res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
        except Exception:
            break
        chunk_count += 1
        if res.x is not None:
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if x.size == n and np.all(np.isfinite(x)):
                try:
                    fval = float(res.fun)
                except Exception:
                    fval = math.inf
                if math.isfinite(fval) and fval < best_f - 1e-15:
                    best_f = fval
                    best_u = x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 4:
                        break
        if getattr(res, "status", 1) == 0:
            break

    p_out = _softmax(best_u)
    p_out = np.maximum(p_out, 0.0)
    ss = float(p_out.sum())
    if ss > 1e-300:
        p_out = p_out / ss
    if consider is not None:
        try:
            consider(p_out)
        except Exception:
            pass
    return p_out


def _coord_polish(p, stop_time, max_sweeps=200000, consider=None, seed=12345, focus_active=True):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    G = 40 if n <= 2000 else 28
    G = int(max(16, G))

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    inv_sqrt5 = (math.sqrt(5.0) - 1.0)

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.0:
        improved = False
        if focus_active:
            cmax = float(c.max())
            thresh = cmax * 0.998
            active_lags = np.where(c >= thresh)[0]
            if len(active_lags) > 0:
                lo_idx = max(0, int(active_lags.min()) - n + 1)
                hi_idx = min(n - 1, int(active_lags.max()) + n - 1)
                base_order = np.arange(lo_idx, hi_idx + 1)
                outer_order = np.concatenate([np.arange(0, lo_idx), np.arange(hi_idx + 1, n)])
                order = np.concatenate([base_order, rng.permutation(outer_order)])
            else:
                order = rng.permutation(n)
        else:
            order = rng.permutation(n)

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.25 or S_total < 0.75:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])

            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls].copy()

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(15.0 * v, 40.0 * S / n, 0.03 * S, 1e-12)
            x_hi = min(x_hi, 8.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, S / n)

            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 <= v <= x_hi:
                pts = np.unique(np.concatenate((pts, [v])))

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                di = 2 * i - start
                if 0 <= di < Ls:
                    dval = base[di] + w * infl[di] + w * w
                    aff = np.maximum(aff, dval[:, None] if aff.ndim > 1 else dval)
                    aff = np.maximum(aff, dval[:, None])
                maff = np.maximum(aff, max_const)
                mv = maff[valid]
                dv = denom[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if G > 2:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)
                if hi <= lo:
                    hi = lo + 1e-14
                fine_pts = np.linspace(lo, hi, 30)
                if 0.0 <= best_x <= hi:
                    fine_pts = np.unique(np.concatenate((fine_pts, [best_x])))
                vals2 = eval_xs(fine_pts)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine_pts[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2

                gr = inv_sqrt5
                g_lo = max(0.0, best_x - dw)
                g_hi = min(x_hi, best_x + dw)
                if g_hi - g_lo > 1e-16:
                    a_gs = g_hi - gr * (g_hi - g_lo)
                    b_gs = g_lo + gr * (g_hi - g_lo)
                    fa = float(eval_xs(np.array([a_gs]))[0])
                    fb = float(eval_xs(np.array([b_gs]))[0])
                    for _ in range(16):
                        if not math.isfinite(fa):
                            fa = 1e100
                        if not math.isfinite(fb):
                            fb = 1e100
                        if fa < fb:
                            g_hi = b_gs
                            b_gs = a_gs
                            fb = fa
                            a_gs = g_hi - gr * (g_hi - g_lo)
                            fa = float(eval_xs(np.array([a_gs]))[0])
                        else:
                            g_lo = a_gs
                            a_gs = b_gs
                            fa = fb
                            b_gs = g_lo + gr * (g_hi - g_lo)
                            fb = float(eval_xs(np.array([b_gs]))[0])
                        if g_hi - g_lo < 1e-16:
                            break
                    gs_x = (g_lo + g_hi) / 2.0
                    gv = float(eval_xs(np.array([gs_x]))[0])
                    if math.isfinite(gv) and gv < best_val:
                        best_x = gs_x
                        best_val = gv

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= 8.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        if S_total > 1e-300 and math.isfinite(S_total):
            inv = 1.0 / S_total
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        np.maximum(c, 0.0, out=c)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
        if consider is not None and sweep % 5 == 0:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 5.0:
                    noise_amt = 0.02 + 0.012 * math.exp(-flat * 0.15)
                    p = p * np.exp(rng.normal(0.0, noise_amt, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_p = p.copy()
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard_deadline = t0 + budget - 20.0

    parent = _get_parent()
    best_arr = parent.copy()
    best_arr = np.where(np.isfinite(best_arr), best_arr, 0.0)
    best_arr = np.clip(best_arr, 0.0, 1000.0)
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64) / 64.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    consider(best_arr)

    n = int(best_arr.size)
    p = best_arr.copy()
    s = float(p.sum())
    if s > 0:
        p = p / s

    # Phase 1: LSE warmup - moderate temperature
    now = time.monotonic()
    phase1_end = min(hard_deadline - 200.0, now + 0.12 * (hard_deadline - now))
    if phase1_end > now + 15.0:
        p = _lbfgs_lse(p, phase1_end, 0.04, consider=consider, max_chunks=8)
        consider(p)

    # Phase 2: LSE - lower temperature
    now = time.monotonic()
    rem_to_end = hard_deadline - now
    if rem_to_end > 100.0:
        phase2_end = min(hard_deadline - 100.0, now + 0.18 * rem_to_end)
        if phase2_end > now + 15.0:
            p = _lbfgs_lse(p, phase2_end, 0.01, consider=consider, max_chunks=10)
            consider(p)

    # Phase 3: Coordinate descent
    now = time.monotonic()
    rem_to_end = hard_deadline - now
    if rem_to_end > 80.0:
        cd3_end = min(hard_deadline - 40.0, now + 0.25 * rem_to_end)
        if cd3_end > now + 20.0:
            p, _ = _coord_polish(p, cd3_end, max_sweeps=200000, consider=consider, seed=100)
            consider(p)

    # Phase 4: LSE - low temperature
    now = time.monotonic()
    rem_to_end = hard_deadline - now
    if rem_to_end > 60.0:
        phase4_end = min(hard_deadline - 30.0, now + 0.15 * rem_to_end)
        if phase4_end > now + 15.0:
            p = _lbfgs_lse(p, phase4_end, 0.003, consider=consider, max_chunks=10)
            consider(p)

    # Phase 5: Coordinate descent (main polish)
    now = time.monotonic()
    rem_to_end = hard_deadline - now
    if rem_to_end > 50.0:
        cd5_end = min(hard_deadline - 20.0, now + 0.30 * rem_to_end)
        if cd5_end > now + 15.0:
            p, _ = _coord_polish(p, cd5_end, max_sweeps=200000, consider=consider, seed=200)
            consider(p)

    # Phase 6: Very low temperature LSE
    now = time.monotonic()
    rem_to_end = hard_deadline - now
    if rem_to_end > 40.0:
        phase6_end = min(hard_deadline - 15.0, now + 0.20 * rem_to_end)
        if phase6_end > now + 12.0:
            p = _lbfgs_lse(p, phase6_end, 0.0008, consider=consider, max_chunks=15)
            consider(p)

    # Phase 7: Final coordinate descent
    now = time.monotonic()
    rem_to_end = hard_deadline - now
    if rem_to_end > 20.0:
        cd7_end = min(hard_deadline - 8.0, now + rem_to_end - 8.0)
        if cd7_end > now + 8.0:
            p, _ = _coord_polish(p, cd7_end, max_sweeps=200000, consider=consider, seed=300)
            consider(p)

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.ones(64, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
