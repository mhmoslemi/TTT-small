import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _fft_conv_self(a):
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return c


def _score_fft(p):
    n = len(p)
    s = float(p.sum())
    if s < 1e-15:
        return math.inf
    c = _fft_conv_self(p)
    return float(2.0 * n * c.max() / (s * s))


def _score_exact(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(p, p)
    return float(2.0 * n * b.max() / (s * s))


def _coord_descent(p_init, n, stop_time, rng_seed, grid_pts=48, max_sweeps=None):

    rng = np.random.default_rng(rng_seed)
    p = np.maximum(p_init, 0.0).copy()
    S = float(p.sum())
    if S < 1e-15:
        return p_init, math.inf
    p /= S
    S = 1.0

    Ltot = 2 * n - 1
    c = _fft_conv_self(p)
    np.maximum(c, 0.0, out=c)
    cur_score = float(2.0 * n * c.max())

    best_p = p.copy()
    best_local = cur_score
    sweep = 0
    stagnant = 0

    G = grid_pts
    gr_golden = (math.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() < stop_time - 3.0:
        if max_sweeps and sweep >= max_sweeps:
            break
        sweep += 1
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            if S < 1e-15:
                break

            # Renormalize if drifted
            if S > 1.08 or S < 0.92:

                p *= inv
                c *= inv * inv
                S = 1.0
                cur_score = float(2.0 * n * c.max())

            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            infl = 2.0 * p[:Ls]  # influence vector (uses OLD p)

            # Outside max
            max_const = 0.0
            if start > 0:
                max_const = max(max_const, float(c[:start].max()))
            if end < Ltot:

            v_hi = max(v * 2.5, 80.0 / n, 0.05 / max(n, 1))
            v_hi = min(v_hi, 0.5)  # cap

                vmin = 0.0

            ws = pts - v

                        return 2.0 * n * m / (d * d)

                    gc_pt = gb - gr_golden * (gb - ga)
                    gd_pt = ga + gr_golden * (gb - ga)
                    fc = _ev(gc_pt)
                    fd = _ev(gd_pt)
                    for _ in range(22):

                    x_opt = (ga + gb) * 0.5

            # Apply

        if sweep % 4 == 0:

            np.maximum(c, 0.0, out=c)
            cur_score = float(2.0 * n * c.max())

        # Track local best
        if math.isfinite(cur_score) and cur_score < best_local - 1e-14:
            best_local = cur_score
            best_p = p.copy()

            active_lags = np.where(c >= cmax_val * 0.97)[0]

                    if prod_j < prod_top * 0.3:

            p_new = np.maximum(p_new, 0.0)
            s2 = float(p_new.sum())
            if s2 > 1e-15:
                p_new /= s2
                p = p_new
                S = 1.0
                c = _fft_conv_self(p)
                np.maximum(c, 0.0, out=c)
                cur_score = float(2.0 * n * c.max())
            stagnant = 0

            break
