def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.asarray(c, dtype=np.float64)


def _score_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size

        return float('inf')
    a = np.where(np.isfinite(a), a, 0.0)

def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.shape, 1.0 / u.size, dtype=np.float64)
    return e / s

        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)
        best_u = u.copy()
        try:
            f0, _ = lse_func(u, T)
        except Exception:
            f0 = 1e100
        best_fstage = f0
        stale = 0
        while time.monotonic() < end_stage - 1.0:
            maxit = 40
            opts = {"maxiter": maxit, "gtol": 1e-12, "ftol": 1e-16, "maxls": 20, "maxdisp": 0}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval) and fval < best_fstage - 1e-14:
                    best_fstage = fval
                    best_u = res.x.copy()
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if getattr(res, "status", 1) == 0 and stale >= 2:
                break
        p_cur = _softmax(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        sc = _score_arr(p_cur)
        if math.isfinite(sc) and sc < best_sc - 1e-13:
            best_sc = sc
            best_p = p_cur.copy()
            p = p_cur.copy()
            u = np.log(np.maximum(p, 1e-30), dtype=np.float64)
            u = u - float(u.mean())
        if consider:
            try:
                consider(best_p)
            except Exception:
                pass
    return best_p


def _coord_descent(p, stop_time, consider=None, seed=0, G=80):
    n = int(p.size)
    if n == 0:
        return p, float('inf')
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p, float('inf')
    p = p / s
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    Ltot = 2 * n - 1
    flat = 0
    sweep = 0
    GR = (math.sqrt(5.0) - 1.0) / 2.0

    # Pre-allocate buffer for affine evaluation
    max_Ls = n
    aff_buf = np.empty((G + 4, max_Ls + 4), dtype=np.float64)

    def eval_1d(w, base, infl, diag_idx, max_const):
        s_new = 1.0 + w
        if s_new <= 1e-15:
            return float('inf')
        Ls = len(base)
        af = base + w * infl
        if diag_idx >= 0 and diag_idx < Ls:
            af = af.copy()
            af[diag_idx] += w * w
        m = float(af.max())
        if max_const > m:
            m = max_const
        if not math.isfinite(m):
            return float('inf')
        return 2.0 * n * m / (s_new * s_new)

    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 1.0:
                break
            v = float(p[i])
            start_k = i
            end_k = min(i + n, Ltot)
            Ls = end_k - start_k
            if Ls <= 0:
                continue
            base = c[start_k:end_k]
            infl = 2.0 * p[:Ls]
            diag = i

            max_const = 0.0
            if start_k > 0:
                mc = float(c[:start_k].max())
                if mc > max_const:
                    max_const = mc
            if end_k < Ltot:
                mc = float(c[end_k:].max())
                if mc > max_const:
                    max_const = mc

            x_hi = max(50.0 * v, 40.0 / n, 0.005)
            x_hi = min(x_hi, 0.5)
            x_hi = max(x_hi, 1e-8)

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                ins = np.searchsorted(pts, v)
                pts = np.insert(pts, ins, v)

            ws = pts - v
            s_new = 1.0 + ws
            # Vectorized affine evaluation
            aff = base[None, :] + ws[:, None] * infl[None, :]
            if 0 <= diag < Ls:
                aff[:, diag] += ws * ws
            maff = np.max(aff, axis=1)
            maff = np.maximum(maff, max_const)
            valid = s_new > 1e-12
            scores = np.full(pts.shape, float('inf'), dtype=np.float64)
            if np.any(valid):
                scores[valid] = 2.0 * n * maff[valid] / (s_new[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            # Golden section
            if 0 < imin < len(pts) - 1:
                lo = float(pts[max(0, imin - 3)])
                hi = float(pts[min(len(pts) - 1, imin + 3)])
            else:
                lo = 0.0
                hi = x_hi
            if hi - lo < 1e-14:
                lo = max(0.0, best_x - 1e-6)
                hi = best_x + 1e-6

            a_gs = lo
            b_gs = hi
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)
            f_c = eval_1d(c_gs - v, base, infl, diag, max_const)
            f_d = eval_1d(d_gs - v, base, infl, diag, max_const)
            for _ in range(32):
                if b_gs - a_gs < 1e-13 * max(1.0, abs(best_x)):
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = eval_1d(c_gs - v, base, infl, diag, max_const)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_d = eval_1d(d_gs - v, base, infl, diag, max_const)
            x_mid = (a_gs + b_gs) / 2.0
            f_mid = eval_1d(x_mid - v, base, infl, diag, max_const)
            if f_mid < best_val:
                best_x = x_mid
                best_val = f_mid
            for xp in (0.0, x_hi):
                fp = eval_1d(xp - v, base, infl, diag, max_const)
                if fp < best_val:
                    best_x = xp
                    best_val = fp

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= x_hi + 1e-15:
                w = best_x - v
                if abs(w) > 1e-16:
                    p[i] = max(0.0, best_x)
                    aff_seg = base + w * infl
                    if 0 <= diag < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag] += w * w
                    new_max = max(float(aff_seg.max()), max_const)
                    c[start_k:end_k] = aff_seg
                    cur_score = 2.0 * n * new_max
                    improved = True

        # Renormalize
        s_new = float(p.sum())
        if s_new > 1e-300 and math.isfinite(s_new):
            inv = 1.0 / s_new
            p *= inv
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-16:
                best_score = new_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 8.0:
                    kick_size = 0.012 + 0.04 * min(1.0, flat / 12.0)
                    x = (np.arange(n, dtype=np.float64) + 0.5) / n
                    k1 = rng.integers(1, 25)
                    k2 = rng.integers(1, 25)
                    phase = rng.uniform(0, 2 * np.pi)
                    pert = (rng.normal(0.0, kick_size * 0.6, n)
                            + 0.4 * kick_size * np.sin(2 * np.pi * k1 * x + phase)
                            + 0.3 * kick_size * np.cos(2 * np.pi * k2 * x))
                    p = p * np.exp(pert)
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 1e-300:
                        p /= ss
                    else:
                        p = np.full(n, 1.0 / n)
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 35.0
    if hard <= t0 + 5.0:
        hard = t0 + max(3.0, budget - 2.0)

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float('inf')

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()

    consider(parent)

    n_work = int(best_arr.size)
    if n_work < 128:
        n_work = 384

    # Phase 1: LSE optimization (15% of budget)
    lse_time = min(150.0, 0.15 * budget)
    p_work = _resample(best_arr, n_work)
    lse_end = t0 + lse_time
    if lse_end > t0 + 15.0:
        try:
            p2 = _lse_optimize(p_work, lse_end, consider=consider)
            if p2 is not None and p2.size == n_work and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p_work = p2 / float(p2.sum())
                consider(p_work)
        except Exception:
            pass

    # Phase 2: Main coordinate descent (55% of budget)
    cd2_end = t0 + 0.60 * budget
    if cd2_end > time.monotonic() + 30.0:
        p_start = _resample(best_arr, n_work)
        try:
            p_cd, _ = _coord_descent(p_start, min(cd2_end, hard - 40.0),
                                     consider=consider, seed=int(seed + 7777), G=80)
            if p_cd is not None and p_cd.size > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
                consider(p_cd)
        except Exception:
            pass

    # Phase 3: Short LSE (8%)
    rem3 = hard - 20.0 - time.monotonic()
    lse3_time = min(80.0, 0.08 * budget)
    if rem3 > lse3_time + 20.0:
        p3 = _resample(best_arr, n_work)
        lse3_end = time.monotonic() + lse3_time
        try:
            p2b = _lse_optimize(p3, lse3_end, consider=consider)
            if p2b is not None and p2b.size == n_work and np.all(np.isfinite(p2b)) and p2b.sum() > 0:
                consider(p2b)
        except Exception:
            pass

    # Phase 4: Another CD pass with different seed (20%)
    cd4_end = min(hard - 15.0, time.monotonic() + 0.20 * budget)
    if cd4_end > time.monotonic() + 25.0:
        p_start4 = _resample(best_arr, n_work)
        try:
            p_cd4, _ = _coord_descent(p_start4, cd4_end,
                                      consider=consider, seed=int(seed + 11111), G=96)
            if p_cd4 is not None and p_cd4.size > 0 and np.all(np.isfinite(p_cd4)) and p_cd4.sum() > 0:
                consider(p_cd4)
        except Exception:
            pass

    # Phase 5: Try cosine-tapered initialization if time permits
    rem5 = hard - 12.0 - time.monotonic()
    if rem5 > 80.0:
        x = (np.arange(n_work, dtype=np.float64) + 0.5) / n_work
        tent = np.sin(np.pi * x) ** 2
        s_t = float(tent.sum())
        if s_t > 0:
            tent /= s_t
        p_blend = 0.3 * tent + 0.7 * _resample(best_arr, n_work)
        p_blend = np.maximum(p_blend, 0.0)
        sb = float(p_blend.sum())
        if sb > 0:
            p_blend /= sb
        consider(p_blend)

        blend_end = min(hard - 12.0, time.monotonic() + rem5 * 0.55)
        if blend_end > time.monotonic() + 20.0:
            try:
                p_b, _ = _coord_descent(p_blend, blend_end,
                                        consider=consider, seed=int(seed + 22222), G=80)
                if p_b is not None and p_b.size > 0 and np.all(np.isfinite(p_b)) and p_b.sum() > 0:
                    consider(p_b)
            except Exception:
                pass

    # Phase 6: Final CD squeeze on the overall best
    rem6 = hard - 10.0 - time.monotonic()
    if rem6 > 30.0:
        p_final = _resample(best_arr, n_work)
        try:
            p_f, _ = _coord_descent(p_final, hard - 10.0,
                                    consider=consider, seed=int(seed + 33333), G=100)
            if p_f is not None and p_f.size > 0 and np.all(np.isfinite(p_f)) and p_f.sum() > 0:
                consider(p_f)
        except Exception:
            pass

    # Final validation with the authoritative evaluator
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
