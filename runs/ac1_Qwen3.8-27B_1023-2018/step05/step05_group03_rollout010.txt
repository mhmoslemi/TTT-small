            w = exps / Z
            f = cmax + T * math.log(Z / float(conv_len))
        q = _fft_cross_for_grad(pp, w, n, cross_size)
        grad = 2.0 * q
        gd = float(pp @ grad)
        grad = pp * (grad - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return f, grad

    c = _fft_conv(a)
    cmax = float(c.max()) if c.size else 1.0
    if not math.isfinite(cmax) or cmax <= 0:
        cmax = 1.0

    stages = (
        (0.025, 0.12),
        (0.010, 0.14),
        (0.004, 0.16),
        (0.0015, 0.18),
        (0.0005, 0.20),
        (0.00015, 0.22),
    )
    best_u = u0.copy()
    best_f = float('inf')
    total_w = sum(w for _, w in stages)
    cum = 0.0
    start = time.monotonic()
    avail = max(0.0, stop_time - start - 5.0)

    for T_frac, sw in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += sw
        stage_end = min(stop_time - 3.0, start + avail * (cum / total_w))
        if stage_end <= now + 4.0:
            continue
        ctx.T = max(1e-14, T_frac * cmax)
        while time.monotonic() < stage_end - 2.0:
            maxit = 18 if n > 2000 else 25
            try:
                res = minimize(obj_grad, best_u, method='L-BFGS-B', jac=True,
                               options={'maxiter': maxit, 'gtol': 1e-10, 'ftol': 1e-15,
                                        'maxls': 20, 'maxcor': 12, 'disp': False})
            except Exception:
                break
            if res.fun < best_f and np.all(np.isfinite(res.x)):
                best_f = float(res.fun)
                best_u = np.asarray(res.x, dtype=np.float64).copy()
            if getattr(res, 'status', 1) == 0:
                break
        m = float(best_u.max())
        e = np.exp(best_u - m)
        a = e / e.sum()
        a = np.maximum(a, 0.0)
        c = _fft_conv(a)
        cmax = float(c.max()) if c.size else 1.0
        consider(a)

    m = float(best_u.max())
    e = np.exp(best_u - m)
    a = e / e.sum()
    return np.maximum(a, 0.0)


def _coord_descent(a, stop_time, consider, G=80, max_flat=6):
    n = len(a)
    a = np.maximum(a, 0.0)
    p = a / a.sum()
    b = _fft_conv(p)
    S = 1.0
    cur_sc = float(2.0 * n * b.max())
    best_p = p.copy()
    best_sc = cur_sc
    rng = np.random.default_rng(12345)
    prev_best = np.zeros(n, dtype=np.float64)
    sweep = 0
    flat = 0

    while sweep < 999999 and time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for pos in range(n):
            if time.monotonic() > stop_time - 0.5:
                break
            i = int(order[pos])
            v = float(p[i])
            lo = i
            hi = min(i + n - 1, 2 * n - 2)
            Ls = hi - lo + 1
            if Ls <= 0:
                continue
            bs = b[lo:hi + 1]
            other_idx = np.arange(lo, hi + 1) - i
            infl = 2.0 * p[other_idx]
            quad_idx = i - lo
            if lo > 0:
                uam = float(b[:lo].max())
            else:
                uam = 0.0
            if hi + 1 < 2 * n - 1:
                r = float(b[hi + 1:].max())
                if r > uam:
                    uam = r
            # Build candidate d values
            scale = max(v, S / n, 1e-8)
            d_list = [0.0]
            # Wide geometric
            lo_d = -v * (1.0 + 1e-12)
            hi_d = max(4.0 * S / n, 4.0 * v, 0.01)
            if hi_d > lo_d + 1e-12:
                m1 = max(20, G // 4)
                d_list.extend(np.geomspace(max(lo_d + abs(lo_d) * 0.01 + 1e-15, 1e-15), hi_d, m1).tolist())
            # Around current
            if prev_best[i] != 0.0:
                pb = prev_best[i]
                d_list.extend(np.linspace(pb * 0.5, pb * 1.5, 10).tolist())
            # Fine around v
            fine_half = 0.15 * scale
            d_list.extend(np.linspace(-fine_half, fine_half, 16).tolist())
            # Very fine
            d_list.extend(np.linspace(-0.02 * scale, 0.02 * scale, 8).tolist())
            # Zero
            d_list.append(-v)
            d_arr = np.array(d_list, dtype=np.float64)
            d_arr = d_arr[np.isfinite(d_arr)]
            # Filter: new value = v + d must be >= 0 and < 1000
            d_arr = d_arr[(v + d_arr >= -1e-12) & (v + d_arr < 1000.0)]
            if d_arr.size == 0:
                d_arr = np.array([0.0])
            d_arr = np.unique(d_arr)
            # Vectorized eval
            new_bs = bs[None, :] + d_arr[:, None] * infl[None, :]
            if 0 <= quad_idx < Ls:
                new_bs[:, quad_idx] += d_arr ** 2
            new_maxs = np.maximum(new_bs.max(axis=1), uam)
            new_S = S + d_arr
            new_S = np.maximum(new_S, 1e-14)
            new_scores = 2.0 * n * new_maxs / (new_S * new_S)
            new_scores = np.where(np.isfinite(new_scores), new_scores, 1e30)
            im = int(np.argmin(new_scores))
            best_d = float(d_arr[im])
            best_val = float(new_scores[im])
            if best_val < cur_sc - 1e-12:
                nd = best_d
                p[i] = v + nd
                b[lo:hi + 1] = bs + nd * infl
                if 0 <= quad_idx < Ls:
                    b[lo + quad_idx] += nd * nd
                diag = 2 * i
                # Already handled via quad_idx
                S += nd
                cur_sc = best_val
                prev_best[i] = nd
                improved = True
        # Renormalize
        if S > 1e-15 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            b *= inv * inv
            S = 1.0
        else:
            break
        b = _fft_conv(p)
        b = np.maximum(b, 0.0)
        desc = float(2.0 * n * b.max())
        if math.isfinite(desc) and desc < best_sc - 1e-13:
            best_sc = desc
            best_p = p.copy()
            consider(best_p)
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= max_flat:
                if time.monotonic() < stop_time - 30.0:
                    sig = 0.04 + 0.04 * min(flat - max_flat, 4)
                    noise = rng.normal(0, sig, n)
                    q = p * np.exp(noise)
                    q = np.maximum(q, 0.0)
                    qs = float(q.sum())
                    if math.isfinite(qs) and qs > 0:
                        p = q / qs
                        b = _fft_conv(p)
                        b = np.maximum(b, 0.0)
                        S = 1.0
                        cur_sc = float(2.0 * n * b.max())
                        prev_best[:] = 0.0
                    flat = 0
                else:
                    break
        sweep += 1

    consider(best_p)
    return best_p


def _fine_coord(a, stop_time, consider):
    n = len(a)
    a = np.maximum(a, 0.0)
    p = a / a.sum()
    b = _fft_conv(p)
    b = np.maximum(b, 0.0)
    S = 1.0
    cur_sc = float(2.0 * n * b.max())
    best_p = p.copy()
    best_sc = cur_sc
    rng = np.random.default_rng(99999)
    sweep = 0
    flat = 0

    while sweep < 999999 and time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for pos in range(n):
            if time.monotonic() > stop_time - 0.5:
                break
            i = int(order[pos])
            v = float(p[i])
            lo = i
            hi = min(i + n - 1, 2 * n - 2)
            Ls = hi - lo + 1
            if Ls <= 0:
                continue
            bs = b[lo:hi + 1]
            other_idx = np.arange(lo, hi + 1) - i
            infl = 2.0 * p[other_idx]
            quad_idx = i - lo
            if lo > 0:
                uam = float(b[:lo].max())
            else:
                uam = 0.0
            if hi + 1 < 2 * n - 1:
                r = float(b[hi + 1:].max())
                if r > uam:
                    uam = r
            scale = max(v, 1e-6)
            # Very fine grid
            d_arr = np.concatenate([
                np.linspace(-0.5 * scale, 0.5 * scale, 40),
                np.linspace(-0.1 * scale, 0.1 * scale, 30),
                np.linspace(-0.02 * scale, 0.02 * scale, 20),
                [0.0, -v],
            ])
            d_arr = np.unique(d_arr[np.isfinite(d_arr)])
            d_arr = d_arr[(v + d_arr >= -1e-12) & (v + d_arr < 1000.0)]
            if d_arr.size < 2:
                continue
            new_bs = bs[None, :] + d_arr[:, None] * infl[None, :]
            if 0 <= quad_idx < Ls:
                new_bs[:, quad_idx] += d_arr ** 2
            new_maxs = np.maximum(new_bs.max(axis=1), uam)
            new_S = S + d_arr
            new_S = np.maximum(new_S, 1e-14)
            new_scores = 2.0 * n * new_maxs / (new_S * new_S)
            new_scores = np.where(np.isfinite(new_scores), new_scores, 1e30)
            im = int(np.argmin(new_scores))
            best_d = float(d_arr[im])
            best_val = float(new_scores[im])
            if best_val < cur_sc - 1e-14:
                nd = best_d
                p[i] = v + nd
                b[lo:hi + 1] = bs + nd * infl
                if 0 <= quad_idx < Ls:
                    b[lo + quad_idx] += nd * nd
                S += nd
                cur_sc = best_val
                improved = True
        if S > 1e-15 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            b *= inv * inv
            S = 1.0
        else:
            break
        b = _fft_conv(p)
        b = np.maximum(b, 0.0)
        desc = float(2.0 * n * b.max())
        if math.isfinite(desc) and desc < best_sc - 1e-14:
            best_sc = desc
            best_p = p.copy()
            consider(best_p)
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                break
        sweep += 1

    consider(best_p)
    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(10.0, min(budget, 1300.0))
    hard = t0 + budget - 20.0
    if hard <= t0 + 5.0:
        hard = t0 + max(1.0, budget - 0.5)

    # Get parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size < 2 or not np.all(np.isfinite(p)):
        p = np.ones(3072, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if not math.isfinite(s) or s < 0.01:
        p = np.ones(3072)
        s = 3072.0
    p = p / s
    n = len(p)

    best_a = p.copy()
    best_sc = _score_fft(best_a)

    def consider(a):
        nonlocal best_a, best_sc
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
            a = np.clip(np.where(np.isfinite(a), a, 0.0), 0.0, 1000.0)
            if a.sum() < 0.01:
                return
            sc = _score_fft(a)
            if math.isfinite(sc) and sc < best_sc - 1e-15:
                best_a = a.copy()
                best_sc = sc
        except Exception:
            pass

    # Phase 1: Quick exploration (40s)
    p1_end = min(t0 + 40.0, hard - 300.0)
    rng = np.random.default_rng(seed)
    base = best_a.copy()
    for q in (0.3, 0.5, 0.7, 1.3, 1.7, 2.5):
        if time.monotonic() > p1_end:
            break
        c = np.power(np.maximum(base, 1e-15), float(q))
        c = c / c.sum()
        consider(c)
    for _ in range(6):
        if time.monotonic() > p1_end:
            break
        x = np.linspace(0, 1, n)
        f = 1 + int(rng.integers(0, 6))
        ph = 2.0 * math.pi * float(rng.random())
        amp = 0.3 + 1.2 * float(rng.random())
        c = base * np.exp(amp * np.sin(2.0 * math.pi * f * x + ph))
        c = c / c.sum()
        consider(c)
    # Side ramp
    x = np.linspace(-1, 1, n)
    for amp in (0.5, 1.0, 2.0):
        c = base * np.exp(amp * x)
        c = c / c.sum()
        consider(c)

    # Phase 2: LS + L-BFGS
    p2_end = min(hard - 300.0, time.monotonic() + min(350.0, 0.28 * max(0.0, hard - time.monotonic())))
    if p2_end > time.monotonic() + 20.0:
        try:
            a2 = _lse_lbfgs(best_a.copy(), p2_end, consider)
            if time.monotonic() < p2_end - 10 and a2 is not None:
                a2 = np.maximum(a2, 0.0)
                a2 = a2 / a2.sum()
                consider(a2)
                # A second short LS pass with finer temperature
                p2b_end = min(hard - 250.0, time.monotonic() + 80.0)
                if p2b_end > time.monotonic() + 15:
                    a2b = _lse_lbfgs(a2.copy(), p2b_end, consider)
                    a2b = np.maximum(a2b, 0.0)
                    a2b = a2b / a2b.sum()
                    consider(a2b)
        except Exception:
            pass

    # Phase 3: Coordinate descent (main)
    p3_end = min(hard - 100.0, time.monotonic() + max(100.0, 0.50 * max(0.0, hard - time.monotonic())))
    if p3_end > time.monotonic() + 20.0:
        try:
            a3 = _coord_descent(best_a.copy(), p3_end, consider, G=90, max_flat=5)
        except Exception:
            pass

    # Phase 4: Fine coordinate descent
    p4_end = min(hard - 5.0, time.monotonic() + max(30.0, 0.70 * max(0.0, hard - time.monotonic())))
    if p4_end > time.monotonic() + 10.0:
        try:
            a4 = _fine_coord(best_a.copy(), p4_end, consider)
        except Exception:
            pass

    # Final validation and return
    try:
        out = best_a.copy()
        out = np.clip(np.where(np.isfinite(out), out, 0.0), 0.0, 1000.0)
        s_out = float(out.sum())
        if out.size == 0 or not math.isfinite(s_out) or s_out < 0.01:
            out = p / p.sum()
            s_out = float(out.sum())
        out = out / s_out
        result = [float(x) for x in out]
        sc = evaluate_sequence(result)
        if not math.isfinite(sc):
            fallback = [float(x) for x in (best_a / best_a.sum())]
            sc2 = evaluate_sequence(fallback)
            if math.isfinite(sc2):
                return fallback
            return [1.0]
        return result
    except Exception:
        return [1.0]
