def _lbfgs_optimize(p, stop_time, consider=None):
    """LSE-smoothed L-BFGS optimization."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0 or n < 32:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    
    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))
    
    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c
    
    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        # Gradient: df/dpp_i = w^T (dw/dpp_i) ... 
        # Simpler: use the subgradient of max approximated by w
        # d(f)/d(pp_i) = 2 * (w * conv(pp, e_i))_... 
        
        # Actually f = logsumexp((c - cmax)/T) * T + cmax
        # df/dc_k = w[k]
        # dc_k/dpp_i involves the convolution

grad_u = pp * (grad_pp - dot(pp, grad_pp))

And dot(pp, grad_pp) = 2 * sum_i pp[i] * sum_m pp[m] * w[m+i] = 2 * (pp ⊙ (pp ⋆ w)) dotted with 1...

- dot = sum(pp * grad_pp)

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, fft, ifft, next_fast_len
from scipy.optimize import minimize


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _softmax(u):
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    return e / s


def _coord_descent(p, stop_time, seed=42, consider=None, max_sweeps=200000):
    """Vectorized coordinate descent for min 2n*max(a*a) s.t. sum(a)=1, a>=0."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n < 8:
        return p, math.inf
    p = p / s

    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)

        for i in order:
            i = int(i)
            if time.monotonic() > stop_time - 0.3:
                break

            v = float(p[i])
            k_start = i
            k_end = min(i + n - 1, Ltot - 1)
            Ls = k_end - k_start + 1
            if Ls <= 0:
                continue

            infl = 2.0 * p[:Ls]
            base = c[k_start:k_end + 1]

            max_const = 0.0
            if k_start > 0:
                ml = float(c[:k_start].max())
                if ml > max_const:
                    max_const = ml
            if k_end + 1 < Ltot:
                mr = float(c[k_end + 1:].max())
                if mr > max_const:
                    max_const = mr

            local_i = i  # position of 2i in local coords: 2i - k_start = i

            avg = 1.0 / n
            x_hi = max(40.0 * v, 120.0 * avg, 0.04 / n, 1e-12)
            x_hi = min(x_hi, 20.0 / n)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = avg

            G = 32
            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                idx = int(round(v / x_hi * G))
                pts = np.insert(pts, min(idx, G + 1), v)

            # Vectorized evaluation
            d = pts - v
            S_new = 1.0 + d
            valid = S_new > 1e-12
            if not np.any(valid):
                continue

            # c_new[point, j] = base[j] + d[point] * infl[j]
            c_new = base[None, :] + d[None, :] * infl[None, :]
            if 0 <= local_i < Ls:
                c_new[:, local_i] += d * d

            max_vals = np.maximum(np.max(c_new, axis=1), max_const)
            scores = np.full(pts.shape, math.inf, dtype=np.float64)
            scores[valid] = 2.0 * n * np.maximum(max_vals[valid], 0.0) / (S_new[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue

            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])

            # Fine search
            if x_hi > 0 and G > 2:
                dw = x_hi / G
                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)
                if hi > lo:
                    fine_pts = np.linspace(lo, hi, 32, dtype=np.float64)
                    if 0.0 < v < hi:
                        fine_pts = np.insert(fine_pts, int(round((v - lo) / (hi - lo) * 31)), v)
                    d_f = fine_pts - v
                    S_f = 1.0 + d_f
                    valid_f = S_f > 1e-12
                    if np.any(valid_f):
                        c_f = base[None, :] + d_f[None, :] * infl[None, :]
                        if 0 <= local_i < Ls:
                            c_f[:, local_i] += d_f * d_f
                        max_f = np.maximum(np.max(c_f, axis=1), max_const)
                        scores_f = np.full(fine_pts.shape, math.inf, dtype=np.float64)
                        scores_f[valid_f] = 2.0 * n * np.maximum(max_f[valid_f], 0.0) / (S_f[valid_f] ** 2)
                        if np.any(np.isfinite(scores_f)):
                            imin_f = int(np.argmin(scores_f))
                            if scores_f[imin_f] < best_val:
                                best_x = float(fine_pts[imin_f])
                                best_val = float(scores_f[imin_f])

            if math.isfinite(best_val) and best_val < cur_score - 1e-15 and 0.0 <= best_x <= 20.0 / n:
                dd = best_x - v
                if abs(dd) > 1e-15:
                    p[i] = best_x
                    c[k_start:k_end + 1] = base + dd * infl
                    if 0 <= local_i < Ls:
                        c[k_start + local_i] += dd * dd
                    cur_score = best_val
                    improved = True

        # End of sweep
        S = float(p.sum())
        if not math.isfinite(S) or S < 1e-300:
            break
        if S > 1.002 or S < 0.998:
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            cur_score = float(2.0 * n * c.max())
            if math.isfinite(cur_score) and cur_score < best_score - 1e-12:
                best_score = cur_score
                best_p = p.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass
            continue

        # Periodic recompute
        if sweep % 8 == 0:
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score):
                cur_score = new_score
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_p = p.copy()
                    if consider:
                        try:
                            consider(best_p)
                        except Exception:
                            pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 8.0:
                    strategy = int(rng.integers(0, 5))
                    if strategy == 0:
                        p = best_p * np.exp(rng.normal(0.0, 0.04, n))
                    elif strategy == 1:
                        freq = int(rng.integers(1, 6))
                        ph = rng.uniform(0, 2 * math.pi)
                        p = best_p * (1.0 + 0.12 * np.sin(2 * math.pi * freq * np.arange(n) / n + ph))
                    elif strategy == 2:
                        bs = max(4, n // 10)
                        nblk = (n + bs - 1) // bs
                        sc = rng.uniform(0.7, 1.3, nblk)
                        p = best_p * np.repeat(sc, bs)[:n]
                    elif strategy == 3:
                        p = best_p * np.exp(rng.normal(0.0, 0.02, n))
                    else:
                        # Structural perturbation: swap random blocks
                        p = best_p.copy()
                        nb = int(rng.integers(1, max(2, n // 20)))
                        blen = int(rng.integers(max(4, n // 50), max(5, n // 10)))
                        for _ in range(nb):
                            s1 = int(rng.integers(0, max(1, n - blen)))
                            s2 = int(rng.integers(0, max(1, n - blen)))
                            if abs(s1 - s2) > blen:
                                p[s1:s1 + blen], p[s2:s2 + blen] = p[s2:s2 + blen].copy(), p[s1:s1 + blen].copy()

                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 1e-10:
                        p = p / ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    if not math.isfinite(cur_score):
                        p = best_p.copy()
                        c = _conv_self(p)
                        cur_score = best_score
                else:
                    break
                flat = 0
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    c = _conv_self(p)
    sc = float(2.0 * n * c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _lbfgs_optimize(p, stop_time, consider=None):
    """LSE-smoothed L-BFGS for global exploration."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n < 64:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300), dtype=np.float64)
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    corr_size = next_fast_len(3 * n - 2)

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        scaled = (c - cmax) / T
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp

        # Cross-correlation: grad_pp[i] = 2 * sum_m pp[m] * w[m+i]
        F_pp = fft(pp, corr_size)
        F_w = fft(w, corr_size)
        corr_full = ifft(F_pp * np.conj(F_w)).real[:n]
        grad_pp = 2.0 * corr_full
        dot = float(np.dot(pp, grad_pp))
        grad_u = pp * (grad_pp - dot)
        grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
        return float(f), grad_u

    start = time.monotonic()
    avail = stop_time - start - 3.0
    if avail < 8.0:
        return p

    stages = (
        (0.04, 0.12), (0.015, 0.13), (0.006, 0.15),
        (0.002, 0.16), (0.0008, 0.18), (0.00025, 0.18), (0.00008, 0.15),
    )
    total_w = sum(w for _, w in stages)
    cum = 0.0
    best_u = u.copy()

    try:
        best_f, _ = lse_func(best_u, 0.01)
    except Exception:
        best_f = 1e100

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 2.0:
            break
        cum += wt
        end_stage = min(float(stop_time) - 1.5, start + avail * (cum / total_w))
        if end_stage <= now + 1.0:
            continue

        c = cfast(_softmax(best_u))
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)

        stale = 0
        while time.monotonic() < end_stage - 0.5:
            opts = {"maxiter": 25, "gtol": 1e-13, "ftol": 1e-16,
                    "maxls": 15, "maxcor": 25, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,),
                               method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                try:
                    fval = float(res.fun)
                except Exception:
                    fval = 1e100
                if math.isfinite(fval) and fval < best_f:
                    best_f = fval
                    best_u = np.array(res.x, dtype=np.float64, copy=True)
                    stale = 0
                else:
                    stale += 1
                    if stale >= 3:
                        break
            if getattr(res, "status", 1) == 0:
                break

        p_cur = _softmax(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        if consider:
            try:
                consider(p_cur)
            except Exception:
                pass

    p_out = _softmax(best_u)
    p_out = np.maximum(p_out, 0.0)
    ss = float(p_out.sum())
    if ss > 1e-300:
        p_out /= ss
    return p_out


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 12.0

    parent = _get_parent()

    best_score = math.inf

        best_score = float(evaluate_sequence([float(x) for x in best_arr]))

        pass

            return

            return

                consider(p1)
        except Exception:
            pass

    # Phase 2: Coordinate descent (main workhorse)
    work = best_arr.copy()

        p2 = np.maximum(fp, 0.0)

        if p2s > 0:

            p2_end = min(hard - 30.0, time.monotonic() + rem4 * 0.6)

        if p3s > 0:

            consider(p3)
            p3_end = min(hard - 25.0, time.monotonic() + rem5 * 0.5)
            if p3_end > time.monotonic() + 10.0:
