def _score_fft(a):

def _resample_mid(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n)
    if src.size == n:
        return src.copy()
    m = src.size
    xp = (np.arange(m) + 0.5) / m
    fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
    out = np.maximum(fp, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def _flat_conv_init(n, sd=0):
    """Initialize using the flat-convolution DFT approach."""
    rng = np.random.default_rng(sd)
    L = next_fast_len(2 * n - 1)
    # Target: make FFT(a_padded)^2 as flat as possible
    # |A[k]|^2 should be ~ constant
    # A[0] = sum(a) = 1 (normalized)
    # For k=1..L/2, target magnitude ~ 1/sqrt(L)
    # This gives a "white" convolution in Fourier domain
    freqs = np.arange(L)
    target_mag = np.ones(L) / math.sqrt(L)
    target_mag[0] = 1.0 / math.sqrt(L) * n  # DC component
    # Random phase
    phases = rng.uniform(0, 2 * math.pi, L)
    # Ensure conjugate symmetry
    A = target_mag * np.exp(1j * phases)
    # Enforce conjugate symmetry
    A[0] = target_mag[0].real
    for k in range(1, L // 2 + 1):
        if k < L - k:
            A[L - k] = np.conj(A[k])
    a_padded = np.real(irfft(A, L))
    a = a_padded[:n]
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def _power_law_init(n, alpha, sd=0):

def _cosine_shape_init(n, A, sd=0):
    """Cosine-modulated uniform initialization."""
    rng = np.random.default_rng(sd)

    side = 1.0 if rng.random() < 0.5 else -1.0

            break
        cur_score = float(2.0 * n * c.max())

        if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 8.0:
                    sigma = 0.015 + 0.01 * min(flat, 8)
                    p = p * np.exp(rng.normal(0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    return best_p, best_score

    if ss > 1e-300:
        best_arr = best_arr / ss

    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = float(_score_exact_arr(best_arr))

    n_parent = int(best_arr.size)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

        p = _resample_mid(init_arr, n)
        if p.size != n or float(p.sum()) <= 0:
            p = np.full(n, 1.0 / n)

                           time.monotonic() + 0.55 * phase_len)
            if lse_stop > time.monotonic() + 10.0:
                try:

                    if p2 is not None and p2.size == n and float(p2.sum()) > 0:
                        p = p2 / float(p2.sum())
                        consider(p)
                except Exception:
                    pass

        if do_active and rem > 12.0:

            if astop > time.monotonic() + 8.0:
                try:

                    if p2 is not None and p2.size == n and float(p2.sum()) > 0:
                        p = p2 / float(p2.sum())
                        consider(p)
                except Exception:
                    pass

        rem = float(end_time) - time.monotonic()
        if do_fine and rem > 10.0:
            fstop = min(float(end_time) - 1.0, time.monotonic() + rem - 1.0)
            if fstop > time.monotonic() + 5.0:
                try:

                    if p3 is not None and p3.size == n and float(p3.sum()) > 0:

                        consider(p)
                except Exception:
                    pass
        return p
