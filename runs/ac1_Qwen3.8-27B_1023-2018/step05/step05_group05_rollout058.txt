def _lp_direction(f, M):
    n = f.size
    # Construct A_ub: (2n-1) x n, sparse
    # A_ub[k, i] = f[k-i] if 0 <= k-i < n
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for j in range(n):
            k = i + j
            rows.append(k)
            cols.append(i)
            vals.append(f[j])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)  # minimize -sum(b)
    b_ub = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        b = np.maximum(res.x, 0)
        s = b.sum()
        if s > 1e-10:
            return b / s
    return None

def _lp_direction(f, M):
    n = f.size
    # A[k, i] = f[k-i] if 0 <= k-i < n
    # For column i: rows i+j for j=0..n-1, values f[0], f[1], ..., f[n-1]

    col_idx = np.repeat(np.arange(n), n)  # each column appears n times
    row_idx = col_idx + np.tile(np.arange(n), n)  # row = col + j
    vals = np.tile(f, n)  # values repeat for each column
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))

        s = b.sum()
        if s > 1e-10:

    return None

def _conv_self(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(c, 0.0)

def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))

def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s

def _p_to_u(p):
    return np.log(np.maximum(p, 1e-300)) - np.mean(np.log(np.maximum(p, 1e-300)))

def _line_search(f, g, max_t=1.0, deadline=None):

            continue

        if math.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_t = t

    if best_t > 0:
        lo = max(0, best_t * 0.5)
        hi = min(max_t, best_t * 2.0)
        # Golden section
        gr = (math.sqrt(5) - 1) / 2
        a = lo
        b = hi
        c_ = b - gr * (b - a)
        d_ = a + gr * (b - a)
