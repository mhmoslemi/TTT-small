def _try_lp_direction(f, end_time):

        work /= ws

    p2_end = min(hard - 100.0, t0 + 0.65 * budget)
    if p2_end > time.monotonic() + 15.0:
        try:
            work, _ = _coord_desc(work, p2_end, consider=consider,
                                  seed=int(seed) + 111, G=36, max_flat=7)
        except Exception:
            pass

    work = best_arr.copy()
    ws = float(work.sum())
    if ws > 1e-300:
        work /= ws

    # Phase 3: Fine LSE (15% of remaining)
    rem = hard - time.monotonic() - 60.0
    if rem > 30.0:
        p3_end = time.monotonic() + 0.4 * rem
        try:
            p3 = _lbfgs_optimize(work, p3_end, consider=consider)
            if p3 is not None and p3.size == n_par and np.all(np.isfinite(p3)):
                consider(p3)
        except Exception:
            pass

    work = best_arr.copy()
    ws = float(work.sum())
    if ws > 1e-300:
        work /= ws

    # Phase 4: Fine coordinate descent
    p4_end = min(hard - 50.0, time.monotonic() + 0.35 * (hard - time.monotonic() - 50.0))
    if p4_end > time.monotonic() + 10.0:
        try:
            work, _ = _coord_desc(work, p4_end, consider=consider,
                                  seed=int(seed) + 222, G=44, max_flat=8)
        except Exception:
            pass

    work = best_arr.copy()
    ws = float(work.sum())
    if ws > 1e-300:
        work /= ws

    # Phase 5: Final coordinate polish
    p5_end = min(hard - 20.0, time.monotonic() + max(10.0, 0.25 * (hard - time.monotonic() - 20.0)))
    if p5_end > time.monotonic() + 8.0:
        try:
            work, _ = _coord_desc(work, p5_end, consider=consider,
                                  seed=int(seed) + 333, G=48, max_flat=10)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.ones(64, dtype=np.float64)
            out /= 64.0
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
