        order = rng.permutation(n)

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 1.5:
                break

            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end]
            infl = 2.0 * p[:Ls]
            diag_pos = i  # 2i - start = 2i - i = i

            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < Ltot:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc

            w_opt, f_opt = find_1d_opt(v, base, infl, diag_pos, max_const, Ls)

            if f_opt < cur_score - 1e-15 and abs(w_opt) > 1e-15:
                p[i] = v + w_opt
                # Update convolution
                new_seg = base + w_opt * infl
                if 0 <= diag_pos < Ls:
                    new_seg = new_seg + w_opt * w_opt
                c[start:end] = new_seg
                new_peak = max(float(new_seg.max()), max_const)
                cur_score = 2.0 * n * new_peak
                improved = True

        # Renormalize periodically
        if sweep % 10 == 0:
            s_new = float(p.sum())
            if s_new > 1e-300 and math.isfinite(s_new):
                p /= s_new
                c = _conv_self(p)
                cur_score = 2.0 * n * float(c.max())
                if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_p = p.copy()
                    if consider:
                        try:
                            consider(p)
                        except Exception:
                            pass

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 8:
                if time.monotonic() < stop_time - 15.0:
                    # Small kick
                    kick_std = 0.01 + 0.02 * min(1.0, no_improve / 30.0)
                    p = p * np.exp(rng.normal(0.0, kick_std, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    cur_score = 2.0 * n * float(c.max())
                    no_improve = 0

        sweep += 1

    # Final check
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    sc = _score_arr(p)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
        if consider:
            try:
                consider(p)
            except Exception:
                pass
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard_deadline = t0 + budget - 30.0

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = float('inf')

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s_val = float(a.sum())
        if s_val < 0.01:
            return
        try:
            sc = float(_score_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = a.copy()

    consider(parent)
    n = int(best_arr.size)

    # Normalize parent
    p = parent.copy().astype(np.float64)
    ss = float(p.sum())
    if ss > 0:
        p = p / ss

    # Phase 1: LSE optimization
    lse_time = min(80.0, 0.12 * budget)
    lse_end = t0 + lse_time
    if lse_end > t0 + 25.0:
        try:
            p2 = _lse_optimize(p, lse_end, consider=consider)
            if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p = p2 / float(p2.sum())
                consider(p)
        except Exception:
            pass

    # Phase 2: Main coordinate descent on best found
    p_start = _resample(best_arr, n)
    cd_end = hard_deadline - 10.0
    if cd_end > time.monotonic() + 30.0:
        try:
            p_cd, _ = _coord_descent_fast(p_start, cd_end, consider=consider, seed=int(seed + 1111))
            if p_cd is not None and p_cd.size > 0:
                consider(p_cd)
        except Exception:
            pass

    # Phase 3: If time remains, try a different initialization
    rem = hard_deadline - 10.0 - time.monotonic()
    if rem > 60.0:
        # Smooth initialization
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        # Cosine bump
        bump = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
        bl = 0.4 * bump + 0.6 * _resample(best_arr, n)
        bl = np.maximum(bl, 0.0)
        sb = float(bl.sum())
        if sb > 0:
            bl /= sb
        consider(bl)

        blend_end = min(hard_deadline - 10.0, time.monotonic() + rem * 0.7)
        if blend_end > time.monotonic() + 20.0:
            try:
                p_b, _ = _coord_descent_fast(bl, blend_end, consider=consider, seed=int(seed + 2222))
                if p_b is not None and p_b.size > 0:
                    consider(p_b)
            except Exception:
                pass

    # Phase 4: Final squeeze on overall best
    rem2 = hard_deadline - 10.0 - time.monotonic()
    if rem2 > 20.0:
        p_final = _resample(best_arr, n)
        try:
            p_f, _ = _coord_descent_fast(p_final, hard_deadline - 10.0, consider=consider, seed=int(seed + 3333))
            if p_f is not None and p_f.size > 0:
                consider(p_f)
        except Exception:
            pass

    # Final validation and return
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [1.0]
        return ls
    except Exception:
        return [1.0]
