    c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _score_exact(a):
    n = a.size
    if n == 0:
        return math.inf
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s < 0.01:
        return math.inf
    c = _fft_conv(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.ones(a.size) / a.size
    return a / s


def _lse_optimize(p, stop_time, consider=None):
    """Log-sum-exp based optimization using L-BFGS-B."""
    p = _normalize(p)
    n = int(p.size)
    if n <= 1:
        return p

    conv_len = 2 * n - 1
    cross_len = 3 * n - 2
    conv_size = next_fast_len(conv_len)
    cross_size = next_fast_len(cross_len)

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def p_to_u(pp):
        pp = np.maximum(pp, 1e-15)
        uu = np.log(pp, dtype=np.float64)
        uu -= float(np.mean(uu))
        return uu

    def softmax_u(uu):
        m = float(np.max(uu))
        e = np.exp(uu - m)
        s = float(e.sum())
        if not math.isfinite(s) or s <= 0:
            return np.ones(n) / n
        pp = e / s
        return pp

    def lse_obj(uu, T):
        uu = np.asarray(uu, dtype=np.float64).ravel()
        if not np.all(np.isfinite(uu)):
            uu = np.where(np.isfinite(uu), uu, 0.0)
        pp = softmax_u(uu)
        c = cfast(pp)
        cmax = float(c.max())
        T = max(T, 1e-14)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            w = exps / sum_exp
            f = cmax + T * math.log(sum_exp / conv_len)
        q = xfast(pp, w)
        gid = 2.0 * q
        gd = float(np.dot(pp, gid))
        grad = pp * (gid - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    u = p_to_u(p)
    best_p = p.copy()
    best_f = float(cfast(p).max())

    stages = ((0.012, 0.15), (0.005, 0.20), (0.0015, 0.25), (0.0004, 0.30), (0.0001, 0.28))
    start = time.monotonic()
    total_w = sum(w for _, w in stages)
    cum = 0.0
    avail = max(0.0, stop_time - start - 3.0)

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 4.0:
            break
        cum += wt
        end_stage = min(stop_time - 3.0, start + avail * (cum / total_w))
        if end_stage <= now + 5.0:
            continue

        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-14, frac * cmax)

        stale = 0
        while time.monotonic() < end_stage - 1.0 and stale < 3:
            try:
                res = minimize(
                    lse_obj, u, args=(T,), method="L-BFGS-B", jac=True,
                    options={"maxiter": 20, "gtol": 1e-9, "ftol": 1e-14, "maxls": 15,
                             "maxcor": 15, "disp": False, "maxdisp": 0}
                )
            except Exception:
                break
            if res.x is not None and res.x.size == n:
                x = np.asarray(res.x, dtype=np.float64)
                if np.all(np.isfinite(x)):
                    fval = float(res.fun)
                    if math.isfinite(fval) and fval < best_f - 1e-13:
                        best_f = fval
                        u = x.copy()
                        p = softmax_u(u)
                        best_p = p.copy()
                        if consider:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                        stale = 0
                    else:
                        stale += 1
            if getattr(res, "status", 1) == 0:
                break

    return best_p


def _coord_descent(a, stop_time, seed=123, consider=None):
    """Fast coordinate descent minimizing 2n*max(a*a)/S^2."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S <= 1e-300:
        a[:] = 1.0 / a.size
        S = 1.0
    else:
        a /= S
        S = 1.0
    n = a.size
    L = 2 * n - 1
    c = _fft_conv(a)
    cur_score = 2.0 * n * float(c.max()) / (S * S)
    best_score = cur_score
    best_a = a.copy()
    if consider:
        try:
            consider(best_a)
        except Exception:
            pass

    rng = np.random.default_rng(seed)
    sweep = 0
    no_improve = 0

    # Precompute grid templates
    n_fine = 24
    n_coarse = 16
    pos_fine = np.geomspace(0.01, 1.0, n_fine)
    pos_coarse = np.geomspace(0.1, 1.0, n_coarse)
    neg_fine = -np.geomspace(0.01, 1.0, n_fine)
    neg_coarse = -np.geomspace(0.1, 0.99, n_coarse)

    while time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)

        for idx in order:
            if time.monotonic() > stop_time - 0.3:
                break
            i = int(idx)
            v = float(a[i])
            if v < 1e-15:
                # Try setting to small positive
                small = max(1e-6, 2.0 / float(n))
                start = i
                end = min(i + n, L)
                length = end - start
                if length <= 0:
                    continue
                infl = 2.0 * a[:length]
                di = 2 * i - start
                um = 0.0
                if start > 0:
                    um = float(c[:start].max())
                if end < L:
                    m2 = float(c[end:].max())
                    if m2 > um:
                        um = m2
                deltas = np.array([small, 2 * small, 4 * small])
                c_aff = c[start:end]
                e_di = np.zeros(length, dtype=np.float64)
                if 0 <= di < length:
                    e_di[di] = 1.0
                new_c = c_aff[None, :] + deltas[:, None] * infl[None, :] + (deltas * deltas)[:, None] * e_di[None, :]
                new_max = np.maximum(new_c.max(axis=1), um)
                new_S = S + deltas
                new_scores = 2.0 * n * new_max / (new_S * new_S)
                im = int(np.argmin(new_scores))
                if new_scores[im] < cur_score - 1e-13:
                    a[i] = small
                    actual_d = small
                    c[start:end] += actual_d * infl
                    c[2 * i] += actual_d * actual_d
                    c = np.maximum(c, 0.0)
                    S += actual_d
                    cur_score = 2.0 * n * float(c.max()) / (S * S)
                    improved = True
                    if cur_score < best_score - 1e-13:
                        best_score = cur_score
                        best_a = a.copy()
                continue

            # Normal coordinate with v > 0
            start = i
            end = min(i + n, L)
            length = end - start
            if length <= 0:
                continue

            infl = 2.0 * a[:length]
            di = 2 * i - start

            # Unaffected max
            um = 0.0
            if start > 0:
                um = float(c[:start].max())
            if end < L:
                m2 = float(c[end:].max())
                if m2 > um:
                    um = m2

            # Determine upper bound for positive delta
            max_c = float(c.max())
            c2i = float(c[2 * i]) if 2 * i < L else 0.0
            upper_pos = max(5.0 * v, 10.0 / float(n), 0.002)
            if max_c > c2i + 1e-15:
                u2 = -v + math.sqrt(v * v + (max_c - c2i))
                if u2 > 0:
                    upper_pos = min(upper_pos, 2.5 * u2)

            # Build candidate deltas - first try fine grid around 0
            step_base = max(v / float(n) * 2, 1e-6, v * 0.05)
            if step_base < upper_pos:
                # Fine grid
                pos_pts = np.minimum(pos_fine * upper_pos, upper_pos)
                neg_pts = np.maximum(neg_fine * v, -v * 0.99)
                pts = np.concatenate([neg_pts, pos_pts])
                pts = np.unique(np.clip(pts, -v + 1e-14, upper_pos))
            else:
                pts = np.array([-0.5 * v, -0.1 * v, 0.5 * v, upper_pos])

            if pts.size == 0:
                continue

            c_aff = c[start:end]
            e_di = np.zeros(length, dtype=np.float64)
            if 0 <= di < length:
                e_di[di] = 1.0

            new_c = c_aff[None, :] + pts[:, None] * infl[None, :] + (pts * pts)[:, None] * e_di[None, :]
            new_max = np.maximum(new_c.max(axis=1), um)
            new_S = S + pts
            new_S_safe = np.where(new_S > 1e-10, new_S, 1e-10)
            new_scores = 2.0 * n * new_max / (new_S_safe * new_S_safe)

            valid = np.isfinite(new_scores)
            if not np.any(valid):
                continue

            im = int(np.argmin(np.where(valid, new_scores, np.inf)))
            best_delta = float(pts[im])
            best_new_score = float(new_scores[im])

            # If best is at boundary, try a coarser wider grid
            if (im == 0 and best_delta < -0.5 * v) or (im == pts.size - 1 and best_delta > 0.5 * upper_pos):
                pos_pts2 = np.minimum(pos_coarse * upper_pos * 2, upper_pos * 2)
                neg_pts2 = np.array([-v * x for x in [0.5, 0.8, 0.95, 0.99]])
                pts2 = np.concatenate([neg_pts2, pos_pts2])
                pts2 = np.unique(np.clip(pts2, -v + 1e-14, upper_pos * 2))
                new_c2 = c_aff[None, :] + pts2[:, None] * infl[None, :] + (pts2 * pts2)[:, None] * e_di[None, :]
                new_max2 = np.maximum(new_c2.max(axis=1), um)
                new_S2 = S + pts2
                new_S2_safe = np.where(new_S2 > 1e-10, new_S2, 1e-10)
                new_scores2 = 2.0 * n * new_max2 / (new_S2_safe * new_S2_safe)
                valid2 = np.isfinite(new_scores2)
                if np.any(valid2):
                    im2 = int(np.argmin(np.where(valid2, new_scores2, np.inf)))
                    if new_scores2[im2] < best_new_score:
                        best_delta = float(pts2[im2])
                        best_new_score = float(new_scores2[im2])

            if best_new_score < cur_score - 1e-13:
                new_val = v + best_delta
                new_val = max(0.0, new_val)
                actual_d = new_val - v
                if abs(actual_d) > 1e-15:
                    a[i] = new_val
                    c[start:end] += actual_d * infl
                    c[2 * i] += actual_d * actual_d
                    c = np.maximum(c, 0.0)
                    S += actual_d
                    cur_score = 2.0 * n * float(c.max()) / (S * S)
                    improved = True
                    if cur_score < best_score - 1e-13:
                        best_score = cur_score
                        best_a = a.copy()
                        if consider:
                            try:
                                consider(best_a)
                            except Exception:
                                pass

        # Periodic FFT recompute to correct drift
        if sweep % 80 == 79:
            c = _fft_conv(a)
            cur_score = 2.0 * n * float(c.max()) / (S * S)
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_a = a.copy()

        if not improved:
            no_improve += 1
            if no_improve >= 4:
                if time.monotonic() < stop_time - 40.0:
                    sigma = 0.015 + 0.01 * min(no_improve - 4, 8)
                    noise = rng.normal(0, sigma, n)
                    a = a * np.exp(noise)
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    S = 1.0
                    c = _fft_conv(a)
                    cur_score = 2.0 * n * float(c.max()) / (S * S)
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0

        sweep += 1

    best_a = np.maximum(best_a, 0.0)
    s = float(best_a.sum())
    if s > 1e-300:
        best_a /= s
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(10.0, budget)
    reserve = 15.0
    hard = t0 + budget - reserve
    if hard <= t0 + 5.0:
        hard = t0 + max(3.0, budget - 1.0)

    # Get parent
    parent = _get_parent()
    n = int(parent.size)
    best_arr = _normalize(parent.copy())
    best_score = math.inf
    try:
        best_score = _score_exact(best_arr)
    except Exception:
        pass
    if not math.isfinite(best_score):
        best_arr = np.ones(64) / 64
        n = 64
        best_score = _score_exact(best_arr)

    def consider(a):
        nonlocal best_arr, best_score
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return
        if arr.size == 0:
            return
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if not math.isfinite(s) or s < 0.01:
            return
        arr = arr / s
        try:
            sc = _score_exact(arr)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_arr = arr.copy()

    consider(best_arr)

    # Phase 1: Quick LSE optimization
    lse_budget = min(100.0, 0.12 * budget)
    lse_stop = t0 + lse_budget
    if lse_stop > t0 + 20.0 and n > 100:
        try:
            p_lse = _lse_optimize(best_arr, lse_stop, consider=consider)
            consider(p_lse)
        except Exception:
            pass

    # Phase 2: Coordinate descent (main workhorse)
    cd_stop = min(hard - 5.0, t0 + budget - reserve - 5.0)
    if time.monotonic() < cd_stop - 30.0:
        try:
            p_cd, sc_cd = _coord_descent(best_arr, cd_stop, seed=seed + 7, consider=consider)
            consider(p_cd)
        except Exception:
            pass

    # Phase 3: Final fine coordinate descent with tighter tolerance
    final_stop = hard - 3.0
    if time.monotonic() < final_stop - 10.0 and best_arr.size == n:
        try:
            p_final, _ = _coord_descent(best_arr, final_stop, seed=seed + 99999, consider=consider)
            consider(p_final)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = _normalize(parent)
            s = float(out.sum())
        if out.size == 0 or not math.isfinite(s) or s < 0.01:
            out = np.ones(64) / 64
        out = out / float(out.sum())
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            fallback = _normalize(parent)
            ls2 = [float(x) for x in fallback]
            sc2 = evaluate_sequence(ls2)
            if math.isfinite(sc2):
                return ls2
            return [1.0]
        return ls
    except Exception:
        return [1.0]
