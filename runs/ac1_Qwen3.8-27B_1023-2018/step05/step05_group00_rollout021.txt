    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * a.size * m / (s * s))

def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    return e / s

def _lbfgs_optimize(p, stop_time, consider=None, fine_only=False):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)

    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        # Cross-correlation: (pp * ww)[k] for k in [0, conv_len)
        # We need sum_j pp[j] * ww[k-j] for valid j
        # This is convolution of pp with reverse(ww)... 
        # Actually, we need the cross-correlation of pp and ww
        # cross[k] = sum_j pp[j] * ww[j + k - (n-1)] ... 
        # Let me think. w is the weight on the convolution c, and we need:
        # grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k, k-i)
        # This is 2 * (w * a_rev)[i + n - 1] where a_rev is reversed a
        # Actually, let's just do:
        # For each i, sum over k of w[k] * a[k-i] where 0<=k<conv_len and 0<=k-i<n
        # Let j = k-i, so k = j+i, and we need 0<=j<n and 0<=j+i<conv_len
        # sum_j w[j+i] * a[j] for j in [max(0, i-(conv_len-1)), min(n-1, conv_len-1-i)]
        # This is a cross-correlation
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:conv_len]
        q = c[n-1:2*n-1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0.0:
            return 1e100, np.zeros_like(uu)
        if T <= 0.0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    def run_chunk(uu, T, end_stage):
        try:
            f0, _ = lse_func(uu, T)
        except Exception:
            return uu
        if not math.isfinite(f0):
            return uu
        best_u = np.array(uu, copy=True)
        best_f = float(f0)
        stale = 0

        while time.monotonic() < end_stage - 1.5:
            maxit = 30 if n > 2000 else 40
            opts = {
                "maxiter": maxit,
                "gtol": 1e-12,
                "ftol": 1e-15,
                "maxls": 20,
                "maxcor": 30,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(
                    lse_func, best_u, args=(T,),
                    method="L-BFGS-B", jac=True, options=opts,
                )
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x).ravel()
                if x.size == uu.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, "status", 1) == 0:
                break
        return best_u

    if fine_only:
        stages = ((0.002, 0.30), (0.0006, 0.40), (0.00015, 0.30))
    else:
        stages = (
            (0.15, 0.10), (0.05, 0.12), (0.015, 0.15),
            (0.004, 0.18), (0.001, 0.22), (0.0003, 0.18),
            (0.00008, 0.12), (0.00002, 0.13),
        )

    start = time.monotonic()
    avail = float(stop_time - start - 10.0)
    if avail <= 5.0:
        return p

    best_p = p.copy()
    best_sc = _score_fft(p)
    total_w = float(sum(w for _, w in stages))
    if total_w <= 0.0:
        total_w = 1.0

    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 4.0, start + avail * (cum / total_w))
        if end_stage <= now + 2.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        sc_cur = _score_fft(p_cur)
        if not math.isfinite(sc_cur):
            p = best_p.copy()
            u = p_to_u(p)
            continue
        if sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif sc_cur > best_sc + max(0.001, 1e-4 * best_sc):
            p = best_p.copy()
        else:
            p = p_cur.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, max_sweeps=200000, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p, math.inf
    p = p / s

    if n <= 450:
        G = 48
    elif n <= 1200:
        G = 36
    elif n <= 4096:
        G = 28
    else:
        G = 22

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    # Track per-coordinate windows
    win_lo = np.zeros(n)
    win_hi = np.zeros(n)
    for i in range(n):
        win_hi[i] = max(25.0 * p[i], 0.04)

    while sweep < max_sweeps and time.monotonic() < stop_time - 1.5:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.3 or S_total < 0.7:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue
            base = c[start_i:end_i].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start_i > 0:
                ml = float(c[:start_i].max())
                if ml > max_const:
                    max_const = ml
            if end_i < Ltot:
                mr = float(c[end_i:].max())
                if mr > max_const:
                    max_const = mr

            avg = S / float(n)
            # Adaptive window
            x_lo = max(0.0, win_lo[i])
            x_hi = max(win_hi[i], v + 1e-8)
            x_hi = min(x_hi, 15.0 * S)
            x_lo = min(x_lo, x_hi)
            if x_hi <= x_lo:
                x_hi = x_lo + max(1e-8, 0.01 * avg)

            pts = np.linspace(x_lo, x_hi, G + 1)
            if 0.0 <= v <= x_hi:
                pts = np.concatenate((pts, [v]))
                pts = np.unique(pts)

            diag = i - start_i

            def eval_xs(xs):
                xs = np.asarray(xs).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                dval = base[diag] + w * infl[diag] + w * w
                maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if G > 4:
                dw = (x_hi - x_lo) / float(G)
                lo = max(0.0, best_x - 2.5 * dw)
                hi = min(x_hi, best_x + 2.5 * dw)
                if hi <= lo:
                    hi = lo + 1e-12
                fine = np.linspace(lo, hi, 40)
                if 0.0 <= best_x <= x_hi:
                    fine = np.concatenate((fine, [best_x]))
                    fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 15.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start_i:end_i] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True
                    # Shrink window around new value
                    win_lo[i] = max(0.0, best_x * 0.5)
                    win_hi[i] = best_x * 1.5 + 0.01 * avg
                    win_hi[i] = min(win_hi[i], 15.0 * S)
                else:
                    # No improvement, shrink window
                    win_lo[i] = max(0.0, v * 0.7)
                    win_hi[i] = v * 1.3 + 0.005 * avg
                    win_hi[i] = min(win_hi[i], 15.0 * S)
            else:
                # No improvement found in window
                win_lo[i] = max(0.0, v * 0.3)
                win_hi[i] = v * 2.0 + 0.02 * avg
                win_hi[i] = min(win_hi[i], 15.0 * S)

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        np.maximum(c, 0.0, out=c)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 8.0:
                    try:
                        sigma = 0.05 if sweep < 100 else 0.025
                        p = p * np.exp(rng.normal(0.0, sigma, n))
                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss > 0.0:
                            p /= ss
                        c = _conv_self(p)
                        S_total = 1.0
                        cur_score = float(2.0 * n * c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
                            best_score = cur_score
                            best_p = p.copy()
                            if consider is not None:
                                consider(best_p)
                        # Reset windows
                        for i in range(n):
                            win_lo[i] = max(0.0, p[i] * 0.5)
                            win_hi[i] = min(p[i] * 2.0 + 0.05, 15.0)
                        flat = 0
                    except Exception:
                        break
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score

            if not np.any(np.isfinite(vals)):
                continue
            im = int(np.argmin(vals))
            bx = float(pts[im])
            bv = float(vals[im])

            # Fine refinement
            dw = (x1 - x0) / float(G)
            fl = max(0.0, bx - 2.5 * dw)
            fh = min(x1, bx + 2.5 * dw)
            if fh <= fl:
                fh = fl + 1e-12
            fine = np.linspace(fl, fh, 48)
            if 0.0 <= bx <= x1:
                fine = np.concatenate((fine, [bx]))
                fine = np.unique(fine)
            v2 = ev(fine)
            if np.any(np.isfinite(v2)):
                im2 = int(np.argmin(v2))
                if float(v2[im2]) < bv:
                    bx = float(fine[im2])
                    bv = float(v2[im2])

            if math.isfinite(bv) and bv < cur - 1e-14 and 0.0 <= bx <= 15.0:
                w = bx - v
                if abs(w) > 1e-15:
                    p[i] = bx
                    Scur += w
                    c[si:ei] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur = bv
                    improved = True
                    wlo[i] = max(0.0, bx * 0.5)
                    whi[i] = min(bx * 1.5 + 0.01 * avg, 15.0)
                else:
                    wlo[i] = max(0.0, v * 0.7)
                    whi[i] = min(v * 1.3 + 0.005 * avg, 15.0)
            else:
                wlo[i] = max(0.0, v * 0.3)
                whi[i] = min(v * 2.0 + 0.02 * avg, 15.0)

        # End of sweep: renormalize
        S = float(Scur)
        if S > 1e-300 and math.isfinite(S):
            p *= 1.0 / S
            c *= 1.0 / (S * S)
        else:
            break
        Scur = 1.0
        c = _conv_fft(p)
        ns = float(2.0 * n * c.max())
        if math.isfinite(ns) and ns < best_sc - 1e-15:
            best_sc = ns
            best_p = p.copy()
            if consider:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
            pert_level = 0
        else:
            flat += 1
            if flat >= 4:
                if time.monotonic() < stop_time - 8.0:
                    sigmas = [0.05, 0.10, 0.20, 0.40]
                    smax = sigmas[min(pert_level, len(sigmas) - 1)]
                    p = p * np.exp(rng.normal(0.0, smax, n))
                    p = np.maximum(p, 0.0)
                    ps = float(p.sum())
                    if ps > 0:
                        p /= ps
                    c = _conv_fft(p)
                    Scur = 1.0
                    cur = float(2.0 * n * c.max())
                    if math.isfinite(cur) and cur < best_sc - 1e-14:
                        best_sc = cur
                        best_p = p.copy()
                        if consider:
                            try:
                                consider(best_p)
                            except Exception:
                                pass
                    # Reset windows after perturbation
                    wlo = np.clip(p * 0.3, 0.0, None)
                    whi = np.clip(p * 2.5 + 0.05 / n, 1e-8, 15.0)
                    pert_level = min(pert_level + 1, len(sigmas) - 1)
                    flat = 0
                else:
                    break
        swp += 1

    p = np.maximum(p, 0.0)
    ps = float(p.sum())
    if ps > 1e-300:
        p /= ps
    try:
        cc = _conv_fft(p)
        sc = float(2.0 * n * cc.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_p = p.copy()
    return best_p, best_sc

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0
    if hard <= t0 + 3.0:
        hard = t0 + max(2.0, budget - 1.0)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 5e-15:
            best_arr = a.copy()
            best_score = sc

    n_par = int(best_arr.size)

    # --- Phase 1: L-BFGS coarse ---
    p1_end = min(hard - 120.0, t0 + 0.20 * budget)
    if p1_end > t0 + 20.0 and n_par >= 128:
        w1 = best_arr.copy() / max(float(best_arr.sum()), 1e-300)
        try:
            r1 = _lbfgs_opt(w1, p1_end, consider=consider)
            if r1 is not None and r1.size == n_par and np.all(np.isfinite(r1)) and r1.sum() > 0:
                r1 /= r1.sum()
                consider(r1)
        except Exception:
            pass

    work = best_arr.copy() / max(float(best_arr.sum()), 1e-300)

    # --- Phase 2: Coordinate descent (main) ---
    p2_end = min(hard - 180.0, t0 + 0.48 * budget)
    if p2_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord(work, p2_end, mx_sw=500000, seed=int(seed + 7777), consider=consider)
        except Exception:
            pass

    work = best_arr.copy() / max(float(best_arr.sum()), 1e-300)

    # --- Phase 3: L-BFGS fine ---
    p3_end = min(hard - 120.0, time.monotonic() + 0.15 * (hard - time.monotonic()))
    if p3_end > time.monotonic() + 15.0 and n_par >= 128:
        try:
            r3 = _lbfgs_opt(work, p3_end, consider=consider, fine_only=True)
            if r3 is not None and r3.size == n_par and np.all(np.isfinite(r3)) and r3.sum() > 0:
                r3 /= r3.sum()
                consider(r3)
        except Exception:
            pass

    work = best_arr.copy() / max(float(best_arr.sum()), 1e-300)

    # --- Phase 4: Coordinate descent (long) ---
    p4_end = min(hard - 60.0, t0 + 0.88 * budget)
    if p4_end > time.monotonic() + 20.0:
        try:
            work, _ = _coord(work, p4_end, mx_sw=500000, seed=int(seed + 123456), consider=consider)
        except Exception:
            pass

    # --- Phase 5: Final coordinate squeeze ---
    work = best_arr.copy() / max(float(best_arr.sum()), 1e-300)
    fin_end = min(hard - 12.0, t0 + budget - 8.0)
    if fin_end > time.monotonic() + 10.0:
        try:
            work, _ = _coord(work, fin_end, mx_sw=200000, seed=int(seed + 555555), consider=consider)
        except Exception:
            pass

    # --- Final validation ---
    try:
        if time.monotonic() < hard:
            out = np.asarray(best_arr, dtype=np.float64).ravel()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = parent.copy()
                out = np.where(np.isfinite(out), out, 0.0)
                out = np.clip(out, 0.0, 1000.0)
                if out.size == 0 or float(out.sum()) < 0.01:
                    out = np.full(64, 1.0 / 64.0)
            ls = [float(x) for x in out]
            fsc = float(evaluate_sequence(ls))
            if not math.isfinite(fsc):
                ls = [float(x) for x in parent]
                fsc = float(evaluate_sequence(ls))
                if not math.isfinite(fsc):
                    return [1.0]
            return ls
    except Exception:
        pass
    return [1.0]
