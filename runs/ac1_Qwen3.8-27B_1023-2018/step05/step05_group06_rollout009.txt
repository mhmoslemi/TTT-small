def _smooth_lbfgs(p, n, fft_size, L, deadline, consider):
    """L-BFGS with log-sum-exp smoothing in log-softmax parameterization."""
    p = p.copy()
    s = p.sum()
    if s < 1e-15:
        return p
    p = p / s

    # Parameterize: a = softmax(u), start with u = log(p) - mean(log(p))
    with np.errstate(divide='ignore', invalid='ignore'):
        u = np.log(np.maximum(p, 1e-300))
    u = u - u.mean()

    cross_size = next_fast_len(3 * n - 1)

    def softmax(uu):
        m = uu.max()
        e = np.exp(uu - m)
        ss = e.sum()
        if ss <= 0 or not math.isfinite(ss):
            return np.ones(n) / n
        return e / ss

    def obj_grad(uu, T):
        a = softmax(uu)
        F = rfft(a, fft_size)
        c = irfft(F * F, fft_size)[:L]
        np.maximum(c, 0.0, out=c)
        cmax = c.max()
        if not math.isfinite(cmax):
            cmax = 1.0
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_e = exps.sum()
        if sum_e <= 0 or not math.isfinite(sum_e):
            w = np.zeros(L)
            w[int(np.argmax(c))] = 1.0
            fval = 2.0 * n * cmax
        else:
            w = exps / sum_e
            fval = 2.0 * n * (cmax + T * math.log(sum_e / L))

        # Gradient: g[i] = 4n * sum_k w[k]*a[k-i]
        # Use sparse w for efficiency
        nz_w = w[w > 1e-12]
        nz_idx = np.where(w > 1e-12)[0]
        g = np.zeros(n)
        for k, wk in zip(nz_idx, nz_w):
            i_lo = max(0, int(k) - n + 1)
            i_hi = min(n - 1, int(k))
            if i_lo <= i_hi:
                # g[i] += wk * a[k-i] for i in [i_lo, i_hi]
                # a[k-i] for i=i_lo..i_hi: a[k-i_lo]..a[k-i_hi]
                a_slice = a[k - i_hi:k - i_lo + 1][::-1].copy()
                g[i_lo:i_hi + 1] += wk * a_slice

        g *= 4.0 * n
        # Transform to u-space: grad_u[i] = a[i]*(g[i] - sum_j g[j]*a[j])
        mean_g = float(np.dot(g, a))
        grad_u = a * (g - mean_g)
        grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
        return float(fval), grad_u

    best_u = u.copy()
    best_f = obj_grad(u, 0.3)[0]

    # Multi-stage temperature
    stages = [(0.30, 0.35), (0.08, 0.35), (0.02, 0.30)]
    cum = 0.0
    total_w = sum(wt for _, wt in stages)
    elapsed0 = time.monotonic() - t
    total_time = deadline - elapsed0

    for T, wt in stages:
        cum += wt
        stage_end = min(deadline - 5.0, t0_global + total_time * (cum / total_w))
        stage_end = min(stage_end, deadline - 5.0)
        if time.monotonic() > stage_end - 3.0:
            break
        while time.monotonic() < stage_end - 2.0:
            opts = {"maxiter": 50, "gtol": 1e-14, "ftol": 1e-18, "maxls": 30,
                    "maxcor": 25, "disp": False}
            try:
                res = minimize(obj_grad, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fv = float(res.fun)
                if math.isfinite(fv) and fv < best_f:
                    best_f = fv
                    best_u = res.x.copy()
                if getattr(res, "status", 1) == 0:
                    break
            else:
                break

    out = softmax(best_u)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss > 0:
        out /= ss
    else:
        out = p.copy()
    consider(out)
    return out


def _coord_descent(p, n, fft_size, L, deadline, consider, seed=0):
    """Efficient coordinate descent with exact 1D optimization per coordinate."""
    p = np.maximum(p, 0.0).copy()
    S = p.sum()
    if S < 1e-15:
        return
    p /= S

    c = conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)

    flat = 0
    sweep = 0
    G = 40  # grid points

    while time.monotonic() < deadline - 3.0:
        sweep += 1
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > deadline - 2.0:
                break

            v = p[i]
            start = i
            end = min(i + n, L)
            Ls = end - start
            if Ls <= 0:
                continue

            infl = 2.0 * p[:Ls]  # influence on c[start:end]
            max_const = 0.0
            if start > 0:
                mc = c[:start].max()
                if mc > max_const:
                    max_const = float(mc)
            if end < L:
                mc = c[end:].max()
                if mc > max_const:
                    max_const = float(mc)

            base = c[start:end]
            diag_off = 2 * i - start  # position of w^2 term in window

            # Search range for new value x = p[i]
            x_hi = max(12.0 * v, 30.0 / n, 0.02)
            x_hi = min(x_hi, 3.0)
            x_lo = max(0.0, v - 0.5 * x_hi)
            if not math.isfinite(x_hi) or x_hi <= x_lo:
                x_hi = v + 1.0 / n
                x_lo = 0.0

            # Vectorized grid evaluation
            pts = np.linspace(x_lo, x_hi, G + 1)
            ws = pts - v  # (G+1,)
            # aff[k] = base[k] + ws*w * infl[k] for each grid point w
            aff = base[None, :] + ws[:, None] * infl[None, :]  # (G+1, Ls)
            if 0 <= diag_off < Ls:
                aff[:, diag_off] += ws[:, None] * ws[:, None]
            maff = aff.max(axis=1)
            np.maximum(maff, max_const, out=maff)
            denoms = 1.0 + ws  # sum changes by w
            vals = np.where(denoms > 1e-15, 2.0 * n * maff / (denoms * denoms), math.inf)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Golden section refinement
            dw = (x_hi - x_lo) / G
            lo_gs = max(0.0, best_x - 2.5 * dw)
            hi_gs = min(x_hi, best_x + 2.5 * dw)
            if hi_gs - lo_gs > 1e-14:
                phi = (math.sqrt(5.0) - 1.0) / 2.0
                a_gs, b_gs = lo_gs, hi_gs

                def eval_x(xx):
                    ww = xx - v
                    nb = base + ww * infl
                    if 0 <= diag_off < Ls:
                        nb = nb.copy()
                        nb[diag_off] += ww * ww
                    mv = float(nb.max())
                    if max_const > mv:
                        mv = max_const
                    den = 1.0 + ww
                    if den < 1e-15:
                        return math.inf
                    return 2.0 * n * mv / (den * den)

                c_gs = b_gs - phi * (b_gs - a_gs)
                d_gs = a_gs + phi * (b_gs - a_gs)
                f_c = eval_x(c_gs)
                f_d = eval_x(d_gs)
                for _ in range(20):
                    if f_c < f_d:
                        b_gs, d_gs, f_d = d_gs, c_gs, f_c
                        c_gs = b_gs - phi * (b_gs - a_gs)
                        f_c = eval_x(c_gs)
                    else:
                        a_gs, c_gs, f_c = c_gs, d_gs, f_d
                        d_gs = a_gs + phi * (b_gs - a_gs)
                        f_d = eval_x(d_gs)
                x_opt = (a_gs + b_gs) * 0.5
                f_opt = eval_x(x_opt)
                if f_opt < best_val:
                    best_x, best_val = x_opt, f_opt

            # Apply update if improved
            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and best_x >= 0:
                w = best_x - v
                if abs(w) > 1e-16:
                    # Save old influence (before p[i] changes)
                    old_infl = infl.copy()  # 2*p[:Ls] with old p[i]
                    p[i] = best_x
                    c[start:end] += w * old_infl
                    if 0 <= diag_off < Ls:
                        c[start + diag_off] += w * w
                    cur_score = best_val
                    improved = True

        # Renormalize periodically
        if sweep % 5 == 0:
            ss = p.sum()
            if ss > 1e-15 and math.isfinite(ss):
                inv = 1.0 / ss
                p *= inv
                c *= inv * inv
                cur_score = float(2.0 * n * c.max())

        # Periodic FFT recompute for numerical accuracy
        if sweep % 4 == 0:
            c = conv_self(p)
            cur_score = float(2.0 * n * c.max())
            if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
                best_score = cur_score
                best_p = p.copy()
                consider(best_p)
            elif not math.isfinite(cur_score):
                p = best_p.copy()
                c = conv_self(p)
                cur_score = best_score

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3 and time.monotonic() < deadline - 60.0:
                # Perturb to escape local minimum
                sigma = 0.015 + 0.01 * flat
                noise = rng.normal(0.0, sigma, n)
                p *= np.exp(noise)
                p = np.maximum(p, 0.0)
                ss = p.sum()
                if ss > 1e-15:
                    p /= ss
                    c = conv_self(p)
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score:
                        best_score = cur_score
                        best_p = p.copy()
                        consider(best_p)
                flat = 0

        if flat >= 15:
            break

    # Return best
    p_out = np.maximum(best_p, 0.0)
    ss = p_out.sum()
    if ss > 1e-15:
        p_out /= ss
    else:
        p_out = p.copy()
    return p_out


def _active_polish(p, n, fft_size, L, deadline, consider):

    return p_out
