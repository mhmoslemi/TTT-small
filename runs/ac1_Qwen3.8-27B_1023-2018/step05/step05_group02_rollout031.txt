            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == n and np.all(np.isfinite(x)):
                    fval = float(res.fun)
                    if math.isfinite(fval) and fval < best_f - 1e-15:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
            if getattr(res, "status", 1) == 0:
                break

        u = best_u
        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fast(p_cur)
        if not math.isfinite(sc_cur):
            p = best_p.copy()
            u = p_to_u(p)
            continue
        if sc_cur < best_sc:
            best_sc = sc_cur
            best_p = p_cur.copy()
            p = p_cur.copy()
        else:
            p = p_cur.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_desc(p, stop_time, seed=42, consider=None, G=28):
    """Coordinate descent with exact score evaluation."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    while time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.25 or S_total < 0.75:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls].copy()

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(30.0 * v, 80.0 * S / n, 0.1 * S, 1e-14)
            x_hi = min(x_hi, 15.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, S / n)

            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 <= v <= x_hi:
                pts = np.unique(np.concatenate((pts, [v])))

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[valid, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                dval = base[0] + w[valid] * infl[0] + w[valid] * w[valid]
                maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Golden section refinement
            dw = x_hi / float(G)
            lo = max(0.0, best_x - 2.0 * dw)
            hi = min(x_hi, best_x + 2.0 * dw)
            if hi <= lo:
                hi = lo + 1e-14
            if hi > 0:
                gr = (math.sqrt(5.0) - 1.0) / 2.0
                g_lo = lo
                g_hi = hi
                a_gs = g_hi - gr * (g_hi - g_lo)
                b_gs = g_lo + gr * (g_hi - g_lo)
                fa = float(eval_xs(np.array([a_gs]))[0])
                fb = float(eval_xs(np.array([b_gs]))[0])
                for _ in range(12):
                    if not math.isfinite(fa):
                        fa = 1e100
                    if not math.isfinite(fb):
                        fb = 1e100
                    if fa < fb:
                        g_hi = b_gs
                        b_gs = a_gs
                        fb = fa
                        a_gs = g_hi - gr * (g_hi - g_lo)
                        fa = float(eval_xs(np.array([a_gs]))[0])
                    else:
                        g_lo = a_gs
                        a_gs = b_gs
                        fa = fb
                        b_gs = g_lo + gr * (g_hi - g_lo)
                        fb = float(eval_xs(np.array([b_gs]))[0])
                    if g_hi - g_lo < 1e-16:
                        break
                gs_x = (g_lo + g_hi) / 2.0
                gv = float(eval_xs(np.array([gs_x]))[0])
                if math.isfinite(gv) and gv < best_val:
                    best_x = gs_x
                    best_val = gv

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= 15.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 5.0:
                    noise_mu = 0.02 + 0.01 * math.exp(-flat * 0.08)
                    p = p * np.exp(rng.normal(0.0, noise_mu, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0.0:
                        p /= ss
                    c = _conv_self(p)
                    S_total = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if math.isfinite(cur_score) and cur_score < best_score - 1e-15:
                        best_score = cur_score
                        best_p = p.copy()
                    flat = 0
                else:
                    break
        sweep += 1

    return best_p, best_score


def _subgrad_polish(p, stop_time, consider=None, max_iters=2000):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    best_p = p.copy()
    best_score = float(2.0 * n * _conv_self(p).max())
    base_step = 0.003
    min_step = 1e-10

    for it in range(max_iters):
        if time.monotonic() > stop_time - 1.5:
            break
        c = _conv_self(p)
        cmax = float(c.max())
        threshold = cmax * 0.998
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        elif len(active) > 50:
            idx = np.argsort(c[active])[-50:]
            active = active[idx]

        grad = np.zeros(n)
        for k in active:
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                grad[valid] += p[kj[valid]] + p[valid]
        grad /= len(active)

        step = base_step / (1.0 + 0.01 * it)
        if step < min_step:
            break

        for scale in (1.0, 0.5, 0.25, 0.1, 0.05):
            p_cand = p - scale * step * grad
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = float(2.0 * n * _conv_self(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                break
        else:
            base_step *= 0.92
            if base_step < 1e-13:
                break

        if it % 100 == 0 and consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 22.0

    parent = _get_parent()
    n = parent.size

    best_arr = parent.copy()
    try:
        best_score = float(_score_exact_list(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64) / 64.0
        best_score = float(_score_exact_list(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_list(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_arr = a.copy()
            best_score = sc

    # Phase 1: Step-function exploration (50s)
    phase1_end = min(hard - 200.0, t0 + 50.0)
    if phase1_end > t0 + 5.0 and n >= 512:
        try:
            sf, sf_sc = _step2_find(n, phase1_end)
            if sf is not None:
                consider(sf)
        except Exception:
            pass

    # Create diverse initializations
    candidates = [parent.copy()]
    # Skewed version
    rng = np.random.default_rng(seed)
    for sk in range(3):
        if time.monotonic() > hard - 300.0:
            break
        x = (np.arange(n) + 0.5) / n
        side = 1.0 if sk % 2 == 0 else -1.0
        amp = 0.3 + 0.4 * rng.random()
        q = parent * np.exp(side * amp * (2.0 * x - 1.0))
        q = np.maximum(q, 0.0)
        sq = float(q.sum())
        if sq > 0.0:
            q /= sq
            consider(q)
            candidates.append(q.copy())

    # Phase 2: L-BFGS optimization (up to 300s)
    p_best = best_arr.copy()
    if p_best.size != n:
        p_best = _resample(p_best, n)

    lbfgs_end = min(hard - 200.0, t0 + 350.0)
    if lbfgs_end > time.monotonic() + 30.0:
        try:
            p2 = _lbfgs_lse(p_best, lbfgs_end, consider=consider)
            if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0.0:
                p2 /= p2.sum()
                consider(p2)
        except Exception:
            pass

    # Phase 3: Coordinate descent (majority of remaining time)
    p_best = best_arr.copy()
    if p_best.size != n:
        p_best = _resample(p_best, n)
    p_best = p_best / p_best.sum()

    now = time.monotonic()
    coord_end = min(hard - 30.0, t0 + budget - 35.0)
    if coord_end > now + 20.0:
        try:
            p3, _ = _coord_desc(p_best, coord_end, seed=seed + 12345,
                                consider=consider, G=24)
            if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and p3.sum() > 0.0:
                p3 /= p3.sum()
                consider(p3)
        except Exception:
            pass

    # Phase 4: Subgradient polish
    p_best = best_arr.copy()
    if p_best.size != n:
        p_best = _resample(p_best, n)
    now = time.monotonic()
    sg_end = min(hard - 25.0, now + max(10.0, min(120.0, 0.15 * (hard - now))))
    if sg_end > now + 8.0:
        try:
            p4 = _subgrad_polish(p_best, sg_end, consider=consider, max_iters=2000)
            if p4 is not None and p4.size == n and np.all(np.isfinite(p4)) and p4.sum() > 0.0:
                p4 /= p4.sum()
                consider(p4)
        except Exception:
            pass

    # Phase 5: Final coordinate descent squeeze
    p_best = best_arr.copy()
    if p_best.size != n:
        p_best = _resample(p_best, n)
    now = time.monotonic()
    final_cd_end = min(hard - 15.0, now + max(8.0, (hard - now) * 0.5))
    if final_cd_end > now + 5.0 and n <= 5000:
        try:
            p5, _ = _coord_desc(p_best, final_cd_end, seed=seed + 99999,
                                consider=consider, G=20)
            if p5 is not None and p5.size == n and np.all(np.isfinite(p5)) and p5.sum() > 0.0:
                p5 /= p5.sum()
                consider(p5)
        except Exception:
            pass

    # Final validation and return
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            if not np.all(np.isfinite(ls)):
                ls = [1.0]
        return ls
    except Exception:
        return [1.0]
