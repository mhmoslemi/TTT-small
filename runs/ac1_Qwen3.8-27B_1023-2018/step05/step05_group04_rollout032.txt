    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    Ltot = 2 * n - 1
    flat = 0
    sweep = 0
    GR = (math.sqrt(5.0) - 1.0) / 2.0
    G = 32  # grid points for 1D search
    
    def eval_1d(w, base, infl, diag_idx, max_const):
        """Evaluate score when coordinate changes by w from current value."""
        s_new = 1.0 + w  # normalized sum is 1
        if s_new <= 1e-15:
            return float('inf')
        val = base + w * infl
        if 0 <= diag_idx < len(base):
            val[diag_idx] += w * w
        m = float(np.max(val))
        if m < max_const:
            m = max_const
        return 2.0 * n * m / (s_new * s_new)
    
    max_sweeps = 100000
    while sweep < max_sweeps and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 1.0:
                break
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]
            diag = i  # position of the quadratic term: 2i - start = i
            
            max_const = 0.0
            if start > 0:
                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < Ltot:
                mc = float(c[end:].max())
                if mc > max_const:
                    max_const = mc
            
            # Upper bound for search
            x_hi = max(5.0 * v, 5.0 / n, 0.005)
            x_hi = min(x_hi, 0.3)
            x_hi = max(x_hi, 1e-8)
            
            # Coarse grid
            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)
            if 0.0 < v < x_hi:
                ins = np.searchsorted(pts, v)
                pts = np.insert(pts, ins, v)
            
            scores = np.full(pts.shape, float('inf'), dtype=np.float64)
            for j_pt in range(len(pts)):
                w = float(pts[j_pt]) - v
                sc = eval_1d(w, base, infl, diag, max_const)
                scores[j_pt] = sc
            
            if not np.any(np.isfinite(scores)):
                continue
            imin = int(np.argmin(scores))
            best_x = float(pts[imin])
            best_val = float(scores[imin])
            
            # Golden section refinement
            if 0 < imin < len(pts) - 1:
                lo = float(pts[max(0, imin - 1)])
                hi = float(pts[min(len(pts) - 1, imin + 1)])
            else:
                lo = 0.0
                hi = x_hi
            
            a_gs, b_gs = lo, hi
            c_gs = b_gs - GR * (b_gs - a_gs)
            d_gs = a_gs + GR * (b_gs - a_gs)
            f_c = eval_1d(c_gs - v, base, infl, diag, max_const)
            f_d = eval_1d(d_gs - v, base, infl, diag, max_const)
            for _ in range(16):
                if b_gs - a_gs < 1e-13 * max(1.0, abs(best_x)):
                    break
                if f_c < f_d:
                    b_gs = d_gs
                    d_gs = c_gs
                    f_d = f_c
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    f_c = eval_1d(c_gs - v, base, infl, diag, max_const)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    f_c = f_d
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_d = eval_1d(d_gs - v, base, infl, diag, max_const)
            x_mid = (a_gs + b_gs) / 2.0
            f_mid = eval_1d(x_mid - v, base, infl, diag, max_const)
            if f_mid < best_val:
                best_x, best_val = x_mid, f_mid
            
            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= x_hi:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    aff_seg = base + w * infl
                    if 0 <= diag < Ls:
                        aff_seg = aff_seg.copy()
                        aff_seg[diag] += w * w
