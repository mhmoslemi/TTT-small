    """Score assuming sum(a) = 1."""
    c = np.convolve(a, a)
    return 2.0 * n * float(c.max())

def _build_conv_mat(n, a):
    """Build (2n-1) x n matrix for convolution with a."""
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        jlo = max(0, k-n+1)
        jhi = min(n, k+1)
        for j in range(jlo, jhi):
            A[k, j] = a[k-j]
    return A

def _proj_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if v.sum() <= 1 + 1e-15:
        v = np.maximum(v, 0)
        s = v.sum()
        if s < 1e-15:
            return np.ones(n)/n
        return v/s
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]

        return np.ones(n)/n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
