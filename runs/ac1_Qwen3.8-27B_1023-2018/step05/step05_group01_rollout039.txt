def _anti_peak_step(p, n_steps=200, stop_time=None, consider=None, seed=0):
    """Targeted moves to reduce the maximum autoconvolution lag."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = p.size
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    rng = np.random.default_rng(seed)

    c = _conv_self(p)
    for step in range(n_steps):
        if stop_time is not None and time.monotonic() > stop_time - 1.0:
            break
        k_star = int(np.argmax(c))
        # Gradient of b[k_star] w.r.t. a[i] is a[k_star - i] (valid indices)
        # Indices contributing to b[k_star]: i in [max(0, k_star-n+1), min(n-1, k_star)]
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        contributors = np.arange(i_lo, i_hi + 1)
        # Contribution of each a[i] to b[k_star] is a[i] * a[k_star - i]
        contrib = p[contributors[:]] * p[k_star - contributors]
        total_contrib = float(contrib.sum())
        if total_contrib < 1e-30:
            break
        # Move mass from high-contributing indices to low-contributing ones
        # Select top contributors and bottom non-contributors
        n_move = max(1, n // 50)
        top_idx = contributors[np.argsort(contrib)[-n_move:]]
        # Non-contributors (don't affect b[k_star])
        non_contrib_mask = np.ones(n, dtype=bool)
        non_contrib_mask[i_lo:i_hi+1] = False
        non_contrib_idx = np.where(non_contrib_mask)[0]
        if len(non_contrib_idx) == 0:
            break
        # Pick random non-contributors to receive mass
        n_recv = min(n_move, len(non_contrib_idx))
        recv_idx = rng.choice(non_contrib_idx, size=n_recv, replace=False)
        # Move a small fraction
        for i_out in top_idx:
            frac = 0.02 + 0.03 * rng.random()
            delta = frac * p[i_out]
            if delta < 1e-15:
                continue
            j_in = int(recv_idx[rng.integers(n_recv)])
            p[i_out] -= delta
            p[j_in] += delta
            p = np.maximum(p, 0.0)
            ss = float(p.sum())
            if ss > 0:
                p /= ss
        # Update convolution
        c = _conv_self(p)
        if consider is not None:
            try:
                consider(p)
            except Exception:
                pass
    return p

        best_arr = np.ones(64) / 64.0
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    n_parent = int(best_arr.size)

    # Phase 0: Try parametric initializations and score them
    if time.monotonic() < phase_limit - 60.0:
        inits = _parametric_inits(n_parent, seed=int(seed))
        for idx, init in enumerate(inits):
            if time.monotonic() > phase_limit - 50.0:
                break
            sc = _score_fft(init)
            if math.isfinite(sc) and sc < best_score:
                consider(init)

    # Phase 1: LSE + Coord on the current best
    p1_end = min(phase_limit, t0 + 0.40 * (phase_limit - t0))
    if p1_end > time.monotonic() + 30.0:
        # LSE
        lstop = min(p1_end - 15.0, time.monotonic() + 0.6 * (p1_end - time.monotonic()))
        if lstop > time.monotonic() + 10.0:
            try:
                p2 = _lse_optimize(best_arr, lstop, consider=consider, seed=int(seed + 1))
                if p2 is not None and p2.size == n_parent and float(p2.sum()) > 0:
                    p2 = p2 / float(p2.sum())
                    consider(p2)
            except Exception:
                pass
        # Coord
        cstop = min(p1_end - 1.0, time.monotonic() + max(10.0, 0.4 * (p1_end - time.monotonic())))
        if cstop > time.monotonic() + 10.0:
            try:
                p3, _ = _coord_polish(best_arr, cstop, seed=int(seed + 2), consider=consider)
                if p3 is not None and p3.size == n_parent and float(p3.sum()) > 0:
                    p3 = p3 / float(p3.sum())
                    consider(p3)
            except Exception:
                pass

    # Phase 2: Anti-peak targeted moves
    if time.monotonic() < phase_limit - 40.0:
        ap_end = min(phase_limit - 30.0, time.monotonic() + 0.2 * (phase_limit - time.monotonic()))
        if ap_end > time.monotonic() + 5.0:
            try:
                p4 = _anti_peak_step(best_arr, n_steps=500, stop_time=ap_end,
                                     consider=consider, seed=int(seed + 100))
                if p4 is not None and p4.size == n_parent and float(p4.sum()) > 0:
                    p4 = p4 / float(p4.sum())
                    consider(p4)
            except Exception:
                pass
            # Coord after anti-peak
            cstop2 = min(phase_limit - 25.0, time.monotonic() + 0.15 * (phase_limit - time.monotonic()))
            if cstop2 > time.monotonic() + 8.0:
                try:
                    p5, _ = _coord_polish(best_arr, cstop2, seed=int(seed + 200), consider=consider)
                    if p5 is not None and p5.size == n_parent and float(p5.sum()) > 0:
                        p5 = p5 / float(p5.sum())
                        consider(p5)
                except Exception:
                    pass

    # Phase 3: Perturb and re-optimize
    if time.monotonic() < phase_limit - 50.0:
        rng3 = np.random.default_rng(seed + 555)
        for pr in range(3):
            if time.monotonic() > phase_limit - 35.0:
                break
            sigma = 0.03 + 0.02 * pr
            pert = best_arr * np.exp(rng3.normal(0, sigma, best_arr.size))
            pert = np.maximum(pert, 0.0)
            ps = float(pert.sum())
            if ps > 0:
                pert /= ps
            consider(pert)
            sub_end = min(phase_limit - 25.0, time.monotonic() + 0.25 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 20.0:
                # LSE on perturbed
                lstop = min(sub_end - 10.0, time.monotonic() + 0.5 * (sub_end - time.monotonic()))
                if lstop > time.monotonic() + 8.0:
                    try:
                        p6 = _lse_optimize(pert, lstop, consider=consider, seed=int(seed + 300 + pr*10))
                        if p6 is not None and p6.size == n_parent and float(p6.sum()) > 0:
                            p6 = p6 / float(p6.sum())
                            consider(p6)
                    except Exception:
                        pass
                # Coord
                cstop3 = min(sub_end - 1.0, time.monotonic() + max(8.0, 0.4 * (sub_end - time.monotonic())))
                if cstop3 > time.monotonic() + 8.0:
                    try:
                        p7, _ = _coord_polish(best_arr, cstop3, seed=int(seed + 400 + pr*10), consider=consider)
                        if p7 is not None and p7.size == n_parent and float(p7.sum()) > 0:
                            p7 = p7 / float(p7.sum())
                            consider(p7)
                    except Exception:
                        pass

    # Phase 4: Try a different length
    if time.monotonic() < phase_limit - 60.0:
        for n_alt in [n_parent * 2, n_parent // 2]:
            if time.monotonic() > phase_limit - 40.0:
                break
            if n_alt < 128 or n_alt > 8192 or n_alt == n_parent:
                continue
            init_alt = _resample_mid(best_arr, n_alt)
            consider(init_alt)
            sub_end = min(phase_limit - 30.0, time.monotonic() + 0.3 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 25.0:
                # LSE
                lstop = min(sub_end - 12.0, time.monotonic() + 0.5 * (sub_end - time.monotonic()))
                if lstop > time.monotonic() + 10.0:
                    try:
                        p8 = _lse_optimize(init_alt, lstop, consider=consider, seed=int(seed + 700))
                        if p8 is not None and float(p8.sum()) > 0:
                            p8 = p8 / float(p8.sum())
                            consider(p8)
                    except Exception:
                        pass
                # Coord
                cstop4 = min(sub_end - 1.0, time.monotonic() + max(8.0, 0.4 * (sub_end - time.monotonic())))
                if cstop4 > time.monotonic() + 8.0:
                    try:
                        if float(best_arr.size) != n_alt:
                            cur = _resample_mid(best_arr, n_alt)
                        else:
                            cur = best_arr.copy()
                        p9, _ = _coord_polish(cur, cstop4, seed=int(seed + 800), consider=consider)
                        if p9 is not None and float(p9.sum()) > 0:
                            p9 = p9 / float(p9.sum())
                            consider(p9)
                    except Exception:
                        pass

    # Phase 5: Final squeeze
    final_end = min(hard - 10.0, t0 + budget - 6.0)
    if final_end > time.monotonic() + 15.0 and best_arr.size > 0:
        nbest = int(best_arr.size)
        rem = final_end - time.monotonic()
        if rem > 30.0 and nbest <= 6000:
            # One more LSE pass
            lstop = min(final_end - 25.0, time.monotonic() + 0.35 * rem)
            if lstop > time.monotonic() + 10.0:
                try:
                    p10 = _lse_optimize(best_arr, lstop, consider=consider, seed=int(seed + 9999))
                    if p10 is not None and p10.size == nbest and float(p10.sum()) > 0:
                        p10 = p10 / float(p10.sum())
                        consider(p10)
                except Exception:
                    pass
        # Final coordinate descent
        cstop_final = min(final_end - 3.0, time.monotonic() + max(10.0, 0.5 * max(0, final_end - time.monotonic())))
        if cstop_final > time.monotonic() + 8.0:
            try:
                p11, _ = _coord_polish(best_arr, cstop_final, seed=int(seed + 123456789), consider=consider)
                if p11 is not None and p11.size == nbest and float(p11.sum()) > 0:
                    p11 = p11 / float(p11.sum())
                    consider(p11)
            except Exception:
                pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
