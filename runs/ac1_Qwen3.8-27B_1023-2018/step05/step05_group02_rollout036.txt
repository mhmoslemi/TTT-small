    """Self-convolution via FFT."""
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score_norm(a):
    """Score for normalized (sum=1) array. = 2n * max(conv)."""
    c = _fft_conv(a)
    return float(2.0 * a.size * c.max())

def _score_exact_list(seq):
    """Exact score using the evaluator's logic."""
    a = np.array([float(max(0, min(1000.0, x))) for x in seq])
    n = len(a)
    if n == 0 or a.sum() < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (a.sum()**2))

def _power_init(n, p):
    x = (np.arange(n) + 1.0) / n
    a = x ** p
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _exp_init(n, alpha):
    x = np.arange(n) / max(1, n-1)
    a = np.exp(alpha * x)
    return a / a.sum()
