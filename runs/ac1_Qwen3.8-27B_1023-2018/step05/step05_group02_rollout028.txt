```python
    p = np.clip(p, 0.0, 1000.0)

def _conv_self(a):
    """Self-convolution using FFT for efficiency."""
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return c

def _conv_cross(a, b):
    """Cross-convolution a * b using FFT."""
    n = a.size
    m = b.size
    L = n + m - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    G = rfft(b, size)
    return irfft(F * G, size)[:L]

def _score(a):
    """Compute score = 2n * max(a*a) / (sum a)^2."""
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(np.max(c))
    return float(2.0 * n * m / (s * s))

    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g, S) or (None, 0) if failed.
    """
    n = a.size
    # Construct convolution matrix
    first_col = np.concatenate([a, np.zeros(n - 1)])
    first_row = np.concatenate([a[:1], np.zeros(n - 1)])
    C = toeplitz(first_col, first_row)
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

        S = float(np.sum(g))
```

```python
def _lp_update(a, M, time_limit=5.0):
    """Perform one LP-based update. Returns new a or None."""
    g, S = _lp_solve(a, M, time_limit)
    if g is None or S <= 1.0 + 1e-8:
        return None
    
    # Try full step first
    g_norm = g / S
    sc_full = _score(g_norm)
    sc_cur = _score(a)
    
    if sc_full < sc_cur:
        return g_norm
    
    # Line search: try smaller steps
    for t in [0.5, 0.25, 0.1, 0.05, 0.01]:
        a_try = (1 - t) * a + t * g_norm
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try > 0:
            a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < sc_cur:
            return a_try
    
    return None
```
