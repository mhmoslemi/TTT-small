                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(20.0 * v, 60.0 * S / n, 0.05 * S, 1e-12)
            x_hi = min(x_hi, 10.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0.0:
                x_hi = max(1e-8, S / n)

            pts = np.linspace(0.0, x_hi, G + 1)
            if 0.0 <= v <= x_hi:
                pts = np.unique(np.concatenate((pts, [v])))

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf)
                valid = denom > 1e-15
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[:, None] * infl[None, :]
                maff = np.max(aff, axis=1)
                di = 0
                if di < Ls:
                    dval = base[di] + w * infl[di] + w * w
                    maff = np.maximum(maff, dval)
                maff = np.maximum(maff, max_const)
                dv = denom[valid]
                mv = maff[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            if G > 2:
                dw = x_hi / float(G)
                lo = max(0.0, best_x - 2.5 * dw)
                hi = min(x_hi, best_x + 2.5 * dw)
                if hi <= lo:
                    hi = lo + 1e-14
                fine_pts = np.linspace(lo, hi, 50)
                if 0.0 <= best_x <= hi:
                    fine_pts = np.unique(np.concatenate((fine_pts, [v])))
                vals2 = eval_xs(fine_pts)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine_pts[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val:
                        best_x = vx2
                        best_val = vv2

                gr = (math.sqrt(5.0) - 1.0) / 2.0
                g_lo = max(0.0, best_x - dw)
                g_hi = min(x_hi, best_x + dw)
                if g_hi - g_lo > 1e-16:
                    a_gs = g_hi - gr * (g_hi - g_lo)
                    b_gs = g_lo + gr * (g_hi - g_lo)
                    fa = float(eval_xs(np.array([a_gs]))[0])
                    fb = float(eval_xs(np.array([b_gs]))[0])
                    for _ in range(16):
                        if not math.isfinite(fa):
                            fa = 1e100
                        if not math.isfinite(fb):
                            fb = 1e100
                        if fa < fb:
                            g_hi = b_gs
                            b_gs = a_gs
                            fb = fa
                            a_gs = g_hi - gr * (g_hi - g_lo)
                            fa = float(eval_xs(np.array([a_gs]))[0])
                        else:
                            g_lo = a_gs
                            a_gs = b_gs
                            fa = fb
                            b_gs = g_lo + gr * (g_hi - g_lo)
                            fb = float(eval_xs(np.array([b_gs]))[0])
                        if g_hi - g_lo < 1e-16:
                            break
                    gs_x = (g_lo + g_hi) / 2.0
                    gv = float(eval_xs(np.array([gs_x]))[0])
                    if math.isfinite(gv) and gv < best_val:
                        best_x = gs_x
                        best_val = gv

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and 0.0 <= best_x <= 10.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_fft(p)
        new_score = float(2.0 * n * c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-16:
            best_score = new_score
            best_p = p.copy()
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 5:
                if time.monotonic() < stop_time - 5.0:
                    try:
                        ni = min(noise_idx, len(noise_schedule) - 1)
                        noise_amt = noise_schedule[ni]
                        p = p * np.exp(rng.normal(0.0, noise_amt, n))
                        p = np.maximum(p, 0.0)
                        ss = float(p.sum())
                        if ss > 0.0:
                            p /= ss
                        c = _conv_fft(p)
                        S_total = 1.0
                        new_sc = float(2.0 * n * c.max())
                        if math.isfinite(new_sc) and new_sc < best_score - 1e-15:
                            best_score = new_sc
                            best_p = p.copy()
                            if consider is not None:
                                consider(best_p)
                        if new_sc >= best_score - 1e-15:
                            noise_idx += 1
                        flat = 0
                    except Exception:
                        break
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_fft(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _subgradient_polish(p, stop_time, consider=None, max_iters=5000):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    best_p = p.copy()
    best_score = float(2.0 * n * _conv_fft(p).max())
    base_step = 0.008
    min_step = 1e-9
    prev_score = best_score
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time - 1.5:
            break

        c = _conv_fft(p)
        cmax = float(c.max())
        eps = max(1e-6, 0.001 * cmax)
        active = np.where(c >= cmax - eps)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        grad = np.zeros(n)
        for k in active:
            kj = k - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                grad[valid] += p[kj[valid]] + p[valid]
        grad /= len(active)

        step = base_step / (1.0 + 0.01 * it)
        if step < min_step:
            break

        improved_this_iter = False
        for scale in (1.0, 0.5, 0.25, 0.125, 0.06):
            p_cand = p - scale * step * grad
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = float(2.0 * n * _conv_fft(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                improved_this_iter = True
                break

        if improved_this_iter:
            stall = 0
        else:
            stall += 1
            base_step *= 0.92
            if stall > 200:
                base_step *= 0.5
            if base_step < 1e-12:
                break

        if it % 100 == 0 and consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    if consider is not None:
        try:
            consider(best_p)
        except Exception:
            pass
    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(60.0, float(budget_s))
    hard = t0 + budget - 25.0
    final_reserve = max(12.0, 0.08 * budget)

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact_arr(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64, dtype=np.float64)
        best_arr /= 64.0
        best_score = float(_score_exact_arr(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact_arr(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    phase_limit = min(hard, t0 + budget - final_reserve)

    # Phase 1: L-BFGS main optimization on parent
    n = int(best_arr.size)
    p0 = best_arr / float(best_arr.sum())
    consider(p0)

    p1_end = min(phase_limit, t0 + 0.35 * (phase_limit - t0))
    if p1_end > time.monotonic() + 15.0:
        try:
            p1 = _lbfgs_optimize(p0, p1_end, consider=consider, fine_only=False)
            if p1 is not None and p1.size == n and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    # Phase 2: Coordinate descent
    p2_end = min(phase_limit, t0 + 0.60 * (phase_limit - t0))
    now = time.monotonic()
    if p2_end > now + 10.0:
        try:
            p_cur = best_arr.copy() / float(best_arr.sum())
            p2, _ = _coord_polish(p_cur, p2_end, max_sweeps=300000, seed=int(seed + 77),
                                  consider=consider)
            if p2 is not None and p2.size == n and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p2 = p2 / float(p2.sum())
                consider(p2)
        except Exception:
            pass

    # Phase 3: Fine L-BFGS
    p3_end = min(phase_limit, t0 + 0.80 * (phase_limit - t0))
    now = time.monotonic()
    if p3_end > now + 15.0:
        try:
            p_cur = best_arr.copy() / float(best_arr.sum())
            p3 = _lbfgs_optimize(p_cur, p3_end, consider=consider, fine_only=True)
            if p3 is not None and p3.size == n and np.all(np.isfinite(p3)) and p3.sum() > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)
        except Exception:
            pass

    # Phase 4: More coordinate descent
    p4_end = min(phase_limit, t0 + 0.93 * (phase_limit - t0))
    now = time.monotonic()
    if p4_end > now + 10.0:
        try:
            p_cur = best_arr.copy() / float(best_arr.sum())
            p4, _ = _coord_polish(p_cur, p4_end, max_sweeps=300000,
                                  seed=int(seed + 555), consider=consider)
            if p4 is not None and p4.size == n and np.all(np.isfinite(p4)) and p4.sum() > 0:
                p4 = p4 / float(p4.sum())
                consider(p4)
        except Exception:
            pass

    # Phase 5: Final subgradient + micro coordinate polish
    now = time.monotonic()
    sg_end = min(hard - 5.0, now + max(8.0, 0.5 * (hard - 5.0 - now) if hard - 5.0 > now else 0))
    if sg_end > now + 5.0:
        try:
            p_cur = best_arr.copy() / float(best_arr.sum())
            p5 = _subgradient_polish(p_cur, sg_end, consider=consider, max_iters=5000)
            if p5 is not None and p5.size == n and np.all(np.isfinite(p5)) and p5.sum() > 0:
                p5 = p5 / float(p5.sum())
                consider(p5)
        except Exception:
            pass

    # Final micro-polish: very short coordinate descent
    now = time.monotonic()
    mc_end = min(hard - 3.0, now + max(5.0, 0.5 * (hard - 3.0 - now) if hard - 3.0 > now else 0))
    if mc_end > now + 3.0:
        try:
            p_cur = best_arr.copy() / float(best_arr.sum())
            p6, _ = _coord_polish(p_cur, mc_end, max_sweeps=50000,
                                  seed=int(seed + 99999), consider=consider)
            if p6 is not None and p6.size == n and np.all(np.isfinite(p6)) and p6.sum() > 0:
                p6 = p6 / float(p6.sum())
                consider(p6)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
