import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.sparse import csr_matrix

def _conv_self(a):
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score(p):

    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(p)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))

def _build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A[k,i] = f[i] if 0 <= k-i < n."""
    n = len(f)
    L = 2 * n - 1

    cols = []
    vals = []

        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(f[idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(L, n))
    return A

```python
def _lp_iteration(f, M, build_matrix_fn, solve_fn, t=0.15):
    """One LP iteration: find g to maximize sum(g) s.t. f*g <= M, g >= 0."""
    g = solve_fn(f, M)
    if g is None:
        return f.copy()
    g = np.maximum(g, 0)
    sg = g.sum()
    if sg <= 0:
        return f.copy()
    sf = f.sum()
    # Rescale g to have same sum as f
    g = g * (sf / sg)
    # Blend
    new_f = (1 - t) * f + t * g
    new_f = np.maximum(new_f, 0)
    # Renormalize
    s = new_f.sum()
    if s > 1e-15:
        new_f /= s
    return new_f
```

    p /= S

    sweep = 0

        sweep += 1
        improved = False
        order = rng.permutation(n)

            if time.monotonic() > stop_time - 1.0:

                c *= inv*inv

            # i.e., c[start:end] += w * 2*p[0:Ls]

            max_out = 0.0
            if start > 0:
                max_out = c[:start].max()
            if end < Ltot:
                m2 = c[end:].max()
                if m2 > max_out:
                    max_out = m2

            diag_off = 2*i - start  # position where w^2 contributes

            x_lo = max(0.0, v - max(v*0.5, 3.0/n))
            x_hi = v + max(v*2.0, 8.0/n)
            x_hi = min(x_hi, 5.0)

            pts = np.linspace(x_lo, x_hi, G+1)
            ws = pts - v
            # new c values: base + w*infl + w^2 * delta_diag
            maff = (base[None,:] + ws[:,None] * infl[None,:]).max(axis=1)
            if 0 <= diag_off < Ls:

            denoms = S + ws

                continue
            
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            dw = (x_hi - x_lo) / G
            lo_g = max(0.0, best_x - 2*dw)
            hi_g = min(x_hi, best_x + 2*dw)
            if hi_g - lo_g > 1e-12:

                def ev(x):
                    w = x - v

                    if d <= 1e-15:
                        return math.inf

                    if 0 <= diag_off < Ls:
                        nb[diag_off] += w*w
                    mv = max(float(nb.max()), max_out)
                    return 2.0*n*mv/(d*d)

                f_c, f_d = ev(c_g), ev(d_g)
                for _ in range(14):

                    else:

                        d_g = a_g + gr*(b_g - a_g)
                        f_d = ev(d_g)
                x_opt = (a_g + b_g)/2.0
                f_opt = ev(x_opt)
                if f_opt < best_val:
                    best_x, best_val = x_opt, f_opt

                w = best_x - v
                if abs(w) > 1e-16:

                    old_p_slice = p[:Ls].copy()
                    if i < Ls:
                        old_p_slice[i] = v  # ensure old value

                    if 0 <= diag_off < Ls:

                    cur_score = best_val
                    improved = True
        
        # End of sweep

            inv = 1.0/S
            p *= inv
            c *= inv*inv
            S = 1.0
        else:
            break

        new_score = 2.0*n*float(c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()

            flat = 0
        else:

                        best_p = p.copy()
                        consider(best_p)
                flat = 0

                break

                        b_g = a_g + gr * (b_g - a_g)
                        f_d = ev(b_g - gr * (b_g - a_g))
                    else:
                        a_g = b_g - gr * (b_g - a_g)
                        f_c = ev(a_g + gr * (b_g - a_g))
                x_opt = 0.5 * (a_g + b_g)
                f_opt = ev(x_opt)
                if f_opt < best_val:
                    best_x, best_val = x_opt, f_opt

            if math.isfinite(best_val) and best_val < cur_score - 1e-14 and best_x >= 0:
                w = best_x - v
                if abs(w) > 1e-17:
                    old_slice = p[:Ls].copy()
                    if i < Ls:
                        old_slice[i] = v
                    p[i] = best_x
                    S += w
                    c[start:end] += 2.0 * w * old_slice
                    if 0 <= diag_off < Ls:
                        c[start + diag_off] += w * w
                    cur_score = best_val
                    improved = True

        # End sweep
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        if sweep % 4 == 0:
            c = _conv_self(p)

        new_score = 2.0 * n * float(c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-14:
            best_score = new_score
            best_p = p.copy()
            consider(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 6 and time.monotonic() < stop_time - 40:
                sig = 0.008 + 0.006 * flat
                noise = rng.normal(0, sig, n)
                p *= np.exp(noise)
                p = np.maximum(p, 0)
                ss = float(p.sum())
                if ss > 1e-300:
                    p /= ss
                    S = 1.0
                    c = _conv_self(p)
                    cur_score = 2.0 * n * float(c.max())
                    if math.isfinite(cur_score) and cur_score < best_score:
                        best_score = cur_score
                        best_p = p.copy()
                        consider(best_p)
                flat = 0
            if flat >= 30:
                break

    return best_p, best_score


def _focus_polish(p, stop_time, consider, top_k=400, G=40):
    n = len(p)
    p = np.maximum(p, 0.0).copy()
    S = float(p.sum())
    p /= S
    S = 1.0
    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(77777)

    cmax = float(c.max())
    active = np.where(c >= cmax * 0.998)[0]

    important = np.zeros(n, dtype=bool)
    for lag in active:
        lo = max(0, int(lag) - n + 1)
        hi = min(n - 1, int(lag))
        important[lo:hi + 1] = True
    imp_idx = np.where(important)[0]
    oth_idx = np.where(~important)[0]
    if len(imp_idx) > top_k:
        imp_idx = imp_idx[:top_k]

    sweep = 0
    flat = 0

    while time.monotonic() < stop_time - 2.0:
        sweep += 1
        improved = False
        if sweep % 4 < 3:
            order = rng.permutation(imp_idx)
        else:
            order = rng.permutation(oth_idx)

        for i in order:
            if time.monotonic() > stop_time - 1.0:
                break
            if S > 1.2 or S < 0.85:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0

            i = int(i)
            v = p[i]
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            infl = 2.0 * p[:Ls]
            max_out = 0.0
            if start > 0:
                mo = float(c[:start].max())
                if mo > max_out:
                    max_out = mo
            if end < Ltot:
                mo = float(c[end:].max())
                if mo > max_out:
                    max_out = mo

            base = c[start:end]
            diag_off = 2 * i - start

            x_lo = max(0.0, v - max(v * 0.3, 2.0 / n))
            x_hi = v + max(v * 0.8, 4.0 / n)
            x_hi = min(x_hi, 6.0)
            if x_lo >= x_hi:
                x_lo = max(0.0, v - 1e-7)
                x_hi = v + 1e-4

            pts = np.linspace(x_lo, x_hi, G + 1)
            ws = pts - v
            maff = (base[None, :] + ws[:, None] * infl[None, :]).max(axis=1)
            if 0 <= diag_off < Ls:
                diag_cand = base[diag_off] + ws * infl[diag_off] + ws * ws
                maff = np.maximum(maff, diag_cand)
            maff = np.maximum(maff, max_out)
            denoms = S + ws
            vals = np.where(denoms > 1e-15, 2.0 * n * maff / (denoms * denoms), np.inf)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Fine golden
            dw = (x_hi - x_lo) / max(G, 1)
            lo_g = max(0.0, best_x - 1.0 * dw)
            hi_g = min(x_hi, best_x + 1.0 * dw)
            if hi_g - lo_g > 1e-14:
                gr = (math.sqrt(5.0) - 1.0) / 2.0

                def ev(x):
                    w = x - v
                    d = S + w
                    if d <= 1e-15:
                        return math.inf
                    nb = base + w * infl
                    if 0 <= diag_off < Ls:
                        nb = nb.copy()
                        nb[diag_off] += w * w
                    mv = float(nb.max())
                    if mv < max_out:
                        mv = max_out
                    return 2.0 * n * mv / (d * d)

                a_g, b_g = lo_g, hi_g
                f_c = ev(b_g - gr * (b_g - a_g))
                f_d = ev(a_g + gr * (b_g - a_g))
                for _ in range(10):
                    if f_c < f_d:
                        b_g = a_g + gr * (b_g - a_g)
                        f_d = ev(b_g - gr * (b_g - a_g))
                    else:
                        a_g = b_g - gr * (b_g - a_g)
                        f_c = ev(a_g + gr * (b_g - a_g))
                x_opt = 0.5 * (a_g + b_g)
                f_opt = ev(x_opt)
                if f_opt < best_val:
                    best_x, best_val = x_opt, f_opt

            if math.isfinite(best_val) and best_val < cur_score - 1e-15 and best_x >= 0:
                w = best_x - v
                if abs(w) > 1e-18:
                    old_slice = p[:Ls].copy()
                    if i < Ls:
                        old_slice[i] = v
                    p[i] = best_x
                    S += w
                    c[start:end] += 2.0 * w * old_slice
                    if 0 <= diag_off < Ls:
                        c[start + diag_off] += w * w
                    cur_score = best_val
                    improved = True

        if S > 1e-300:
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        if sweep % 2 == 0:
            c = _conv_self(p)
            new_score = 2.0 * n * float(c.max())
            if math.isfinite(new_score) and new_score < best_score - 1e-15:
                best_score = new_score
                best_p = p.copy()
                consider(best_p)

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 10:
                break

    return best_p, best_score


def _lp_phase(p, stop_time, consider, max_iters=4, lp_timeout=25):
    """Try a few LP-based iterations to improve shape."""
    try:
        import cvxpy as cp
        from scipy.sparse import csr_matrix
    except ImportError:
        return p

    n = len(p)
    p = p.copy()
    S = float(p.sum())
    p /= S

    elapsed = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() > stop_time - 120:
            break
        c = _conv_self(p)
        M = float(c.max())
        if M <= 0 or not math.isfinite(M):
            break

        # Build constraint matrix A: (2n-1) x n
        L = 2 * n - 1
        # Build efficiently
        rows_list = []
        cols_list = []
        vals_list = []
        for k in range(L):
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo <= i_hi:
                rows_list.extend([k] * (i_hi - i_lo + 1))
                cols_list.extend(range(i_lo, i_hi + 1))
                vals_list.extend(p[i_lo:i_hi + 1].tolist())

        if len(vals_list) == 0:
            break
        A = csr_matrix((vals_list, (rows_list, cols_list)), shape=(L, n))

        g_var = cp.Variable(n)
        constraints = [A @ g_var <= M]
        constraints += [g_var >= 0]
        try:
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
            solvers = [cp.GLPK, cp.CBC, cp.SCS]
            solved = False
            for solver in solvers:
                try:
                    prob.solve(solver=solver, time_limit=lp_timeout)
                    if g_var.value is not None:
                        gv = np.asarray(g_var.value, dtype=np.float64)
                        if np.all(np.isfinite(gv)) and gv.sum() > 0:
                            g_var.value = gv
                            solved = True
                            break
                except Exception:
                    continue
            if not solved or g_var.value is None:
                break
        except Exception:
            break

        gv = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        sg = float(gv.sum())
        if sg <= 1e-15:
            break
        gv = gv / sg  # normalize sum to 1

        # Deterministic blend
        t = 0.12 + 0.03 * it
        p = (1 - t) * p + t * gv
        p = np.maximum(p, 0.0)
        ss = float(p.sum())
        if ss > 1e-15:
            p /= ss
        else:
            break
        consider(p)

    return p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard = t0 + budget - 35.0  # 35s buffer

    # Get initialization
    try:
        p = np.asarray(globals()['height_sequence_1'], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(2048, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 0.01:
        p = np.ones(2048, dtype=np.float64)
    n = len(p)
    p = p / float(p.sum())

    best_p = p.copy()
    best_score = _score(p)

    def consider(arr):
        nonlocal best_p, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.size == 0 or float(a.sum()) < 0.01:
                return
            sc = _score(a)
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_p = (a / float(a.sum())).copy()
        except Exception:
            pass

    # Phase 1: LP-based shape improvement (~100s)
    lp_stop = min(hard - 1050.0, t0 + 100.0)
    if lp_stop > time.monotonic() + 30.0:
        try:
            p2 = _lp_phase(p, lp_stop, consider, max_iters=3, lp_timeout=20)
            if p2 is not None and len(p2) == n and np.all(np.isfinite(p2)) and p2.sum() > 0.01:
                consider(p2)
        except Exception:
            pass
        p = best_p.copy()

    # Phase 2: Main coordinate descent (~850s)
    cd_stop = min(hard - 200.0, t0 + 1050.0)
    if cd_stop > time.monotonic() + 30.0:
        try:
            p3, _ = _coord_descent(p, cd_stop, consider, seed=seed + 111, G=22)
            if p3 is not None and len(p3) == n:
                consider(p3)
        except Exception:
            pass

    # Phase 3: Focus polish (~150s)
    pol_stop = min(hard - 40.0, t0 + 1200.0)
    if pol_stop > time.monotonic() + 25.0:
        try:
            p4, _ = _focus_polish(best_p, pol_stop, consider, top_k=500, G=44)
            if p4 is not None and len(p4) == n:
                consider(p4)
        except Exception:
            pass

    # Phase 4: Second coordinate descent pass if time remains
    cd2_stop = hard - 40.0
    if cd2_stop > time.monotonic() + 30.0:
        try:
            p5, _ = _coord_descent(best_p, cd2_stop, consider, seed=seed + 999, G=18)
            if p5 is not None and len(p5) == n:
                consider(p5)
        except Exception:
            pass

    # Final validation with exact evaluator
    try:
        out = np.asarray(best_p, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(128, 1.0 / 128.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            try:
                pp = np.asarray(globals()['height_sequence_1'], dtype=np.float64).ravel()
                pp = np.clip(pp, 0.0, 1000.0)
                pp = np.where(np.isfinite(pp), pp, 0.0)
                ls = [float(x) for x in pp]
                final_sc = float(evaluate_sequence(ls))
            except Exception:
                return [1.0]
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
