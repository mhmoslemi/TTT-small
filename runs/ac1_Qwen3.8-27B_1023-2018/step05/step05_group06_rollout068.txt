def _conv_self_fft(a):
    n = len(a)

    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return c

    c = _conv_self_fft(p)
    return float(2.0 * n * c.max() / (s * s))
