    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]

    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(p, p)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))

                    if 0 <= diag_offset < Ls:
                        aff[:, diag_offset] += ws[:, None] * ws[:, None]
                    maff = aff.max(axis=1)
                    maff = np.maximum(maff, max_const)
                    denoms = S + ws
                    vals = np.where(denoms > 1e-15,
                                    2.0 * n * maff / (denoms * denoms),
                                    math.inf * np.ones_like(vals))

                    if not np.any(np.isfinite(vals)):
                        continue
                    imin = int(np.argmin(vals))
                    best_x = float(pts[imin])
                    best_val = float(vals[imin])

                    # Tiny golden section
                    dw = (x_hi - x_lo) / G_fine
                    lo_gs = max(0.0, best_x - 1.0 * dw)
                    hi_gs = min(x_hi, best_x + 1.0 * dw)
                    if hi_gs - lo_gs > 1e-15:
                        gr = (math.sqrt(5.0) - 1.0) / 2.0
                        a_gs, b_gs = lo_gs, hi_gs
                        c_gs = b_gs - gr * (b_gs - a_gs)
                        d_gs = a_gs + gr * (b_gs - a_gs)

                        def eval_x(x):
                            w = x - v
                            denom = S + w
                            if denom <= 1e-15:
                                return math.inf
                            nb = base + w * infl
                            if 0 <= diag_offset < Ls:
                                nb = nb.copy()
                                nb[diag_offset] += w * w
                            mv = float(nb.max())
                            if max_const > mv:
                                mv = max_const
                            return 2.0 * n * mv / (denom * denom)

                        f_c = eval_x(c_gs)
                        f_d = eval_x(d_gs)
                        for _ in range(14):
                            if f_c < f_d:
                                b_gs, d_gs, f_d = d_gs, c_gs, f_c
                                c_gs = b_gs - gr * (b_gs - a_gs)
                                f_c = eval_x(c_gs)
                            else:
                                a_gs, c_gs, f_c = c_gs, d_gs, f_d
                                d_gs = a_gs + gr * (b_gs - a_gs)
                                f_d = eval_x(d_gs)
                        x_opt = (a_gs + b_gs) / 2.0
                        f_opt = eval_x(x_opt)
                        if f_opt < best_val:
                            best_x, best_val = x_opt, f_opt

                    if math.isfinite(best_val) and best_val < cur_score - 1e-15 and best_x >= 0:
                        w = best_x - v
                        if abs(w) > 1e-18:
                            a_old = cur_p[:Ls].copy()
                            if i < Ls:
                                a_old[i] = v
                            cur_p[i] = best_x
                            S += w
                            c[start:end] += 2.0 * w * a_old
                            if 0 <= diag_offset < Ls:
                                c[start + diag_offset] += w * w
                            cur_score = best_val
                            improved_p = True

                # Renormalize
                if S > 1.02 or S < 0.98:
                    inv = 1.0 / S
                    cur_p *= inv
                    c *= inv * inv
                    S = 1.0
                    cur_score = 2.0 * n * float(c.max())

                # Check improvement
                if not improved_p:
                    break
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_p = cur_p.copy()
        except Exception:
            pass

    # Final validation and return
    try:
        out = np.asarray(best_p, dtype=np.float64).ravel()
        out = np.clip(out, 0.0, 1000.0)
        out = np.where(np.isfinite(out), out, 0.0)
        if out.size == 0 or out.sum() < 0.01:
            out = np.full(128, 1.0 / 128.0)
        result = [float(x) for x in out]
        final_sc = float(evaluate_sequence(result))
        if not math.isfinite(final_sc):
            # Fallback
            try:
                pp = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
                pp = np.clip(pp, 0.0, 1000.0)
                pp = np.where(np.isfinite(pp), pp, 0.0)
                result = [float(x) for x in pp]
                final_sc = float(evaluate_sequence(result))
            except Exception:
                return [1.0]
            if not math.isfinite(final_sc):
                return [1.0]
        return result
    except Exception:
        return [1.0]
