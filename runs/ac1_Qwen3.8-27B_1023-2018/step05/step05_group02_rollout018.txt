                    if rm > rest_max:
                        rest_max = rm

                x_max = max(8.0 * old, 0.008 / nn, 0.0003)
                x_max = min(x_max, 0.04)

                # Quick reject: if old is at boundary and can only worsen
                if old <= 0 and rest_max >= float(conv[lo:hi].max()):
                    continue

                xs = np.linspace(0.0, x_max, G + 1)
                best_x = old
                best_val = cur_score

                for x in xs:
                    d = x - old
                    if abs(d) < 1e-16:
                        continue
                    new_aff = conv[lo:hi] + 2.0 * d * a_slice
                    qi = 2 * i - lo
                    if 0 <= qi < Ls:
                        new_aff[qi] += d * d
                    new_peak = max(float(new_aff.max()), rest_max)
                    new_S = 1.0 + d
                    if new_S < 1e-12:
                        continue
                    val = 2.0 * nn * new_peak / (new_S * new_S)
                    if val < best_val:
                        best_val = val
                        best_x = x

                # Golden section refinement
                if best_x != old and G > 2:
                    step = x_max / float(G)
                    gl = max(0.0, best_x - 2.0 * step)
                    gh = min(x_max, best_x + 2.0 * step)
                    if gh - gl > 1e-14:
                        gr = (math.sqrt(5.0) - 1.0) / 2.0

                        def ev_x(xv):
                            dv = xv - old
                            na = conv[lo:hi] + 2.0 * dv * a_slice
                            qv = 2 * i - lo
                            if 0 <= qv < Ls:
                                na[qv] += dv * dv
                            npk = max(float(na.max()), rest_max)
                            nv = 1.0 + dv
                            if nv < 1e-12:
                                return 1e10
                            return 2.0 * nn * npk / (nv * nv)

                        a1 = gl + (1.0 - gr) * (gh - gl)
                        b1 = gl + gr * (gh - gl)
                        fa = ev_x(a1)
                        fb = ev_x(b1)
                        for _ in range(18):
                            if gh - gl < 1e-14:
                                break
                            if fa < fb:
                                gh = b1
                                b1 = a1
                                fb = fa
                                a1 = gl + (1.0 - gr) * (gh - gl)
                                fa = ev_x(a1)
                            else:
                                gl = a1
                                a1 = b1
                                fa = fb
                                b1 = gl + gr * (gh - gl)
                                fb = ev_x(b1)
                        xm = 0.5 * (gl + gh)
                        fm = ev_x(xm)
                        if fm < best_val:
                            best_val = fm
                            best_x = xm

                if math.isfinite(best_val) and best_val < cur_score - 1e-14:
                    d = best_x - old
                    conv[lo:hi] += 2.0 * d * a_slice
                    qi = 2 * i - lo
                    if 0 <= qi < Ls:
                        conv[2 * i] += d * d
                    p[i] = best_x
                    cur_score = best_val
                    improved = True

            if improved:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 3:
                    break

            # Periodic renormalization and conv recompute
            ps2 = float(p.sum())
            if ps2 < 1e-12:
                return p
            p /= ps2
            if sweep % 4 == 3:
                conv = conv_full(p)
                cur_score = 2.0 * nn * float(conv.max())

        return p
