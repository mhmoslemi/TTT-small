```python
def _coord_descend(p, stop_time, max_sweeps=200000, seed=12345, consider=None):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p, math.inf
    p = p / s  # normalize to sum=1
    
    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    
    # Number of grid points for coarse search
    G = 32 if n <= 4096 else 24
    
    while sweep < max_sweeps and time.monotonic() < stop_time - 1.0:
        improved = False
        order = rng.permutation(n)
        
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            
            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue
            
            S = 1.0  # normalized
            # Affected convolution values
            base = c[start_i:end_i].copy()
            infl = 2.0 * p[:Ls]
            diag = i - start_i  # = i
            
            # Max of unaffected convolution
            max_const = 0.0
            if start_i > 0:
                ml = float(c[:start_i].max())
                if ml > max_const:

            if end_i < Ltot:

                if mr > max_const:

                new_S = S + w
```

```python
    maff = aff.max(axis=1)
    maff = np.maximum(maff, max_const)
    dv = new_S[valid]
    mv = maff[valid]
    vals[valid] = (2.0 * n * mv) / (dv * dv)
    return vals
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p

def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c

def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    if c.size == 0:
        return math.inf
    m = float(c.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * n * m / (s * s))

def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    if b.size == 0:
        return math.inf
    m = float(b.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * a.size * m / (s * s))

def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(u.size, 1.0 / u.size, dtype=np.float64)
    return e / s

def _lbfgs_optimize(p, stop_time, consider=None, fine_only=False):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300), dtype=np.float64)
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:conv_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu, dtype=np.float64)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0.0:
            return 1e100, np.zeros_like(uu, dtype=np.float64)
        if T <= 0.0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled, dtype=np.float64)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0.0:
            w = np.zeros(c.size, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    def run_chunk(uu, T, end_stage):
        try:
            f0, _ = lse_func(uu, T)
        except Exception:
            return uu
        if not math.isfinite(f0):
            return uu
        best_u = np.array(uu, dtype=np.float64, copy=True)
        best_f = float(f0)
        stale = 0
        while time.monotonic() < end_stage - 1.0:
            maxit = 35 if n > 2000 else 50
            opts = {
                "maxiter": maxit,
                "gtol": 1e-13,
                "ftol": 1e-16,
                "maxls": 30,
                "maxcor": 40,
                "maxdisp": 0,
                "disp": False,
            }
            try:
                res = minimize(lse_func, best_u, args=(T,),
                               method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == uu.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-16:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 4:
                            break
            if getattr(res, "status", 1) == 0:
                break
        return best_u

    if fine_only:
        stages = ((0.003, 0.30), (0.0008, 0.40), (0.0002, 0.30))
    else:
        stages = (
            (0.20, 0.08),
            (0.08, 0.10),
            (0.03, 0.12),
            (0.01, 0.14),
            (0.003, 0.16),
            (0.0008, 0.16),
            (0.0002, 0.14),
            (0.00005, 0.14),
            (0.00001, 0.16),
        )

    start = time.monotonic()
    avail = float(stop_time - start - 8.0)
    if avail <= 5.0:
        return p

    best_p = p.copy()
    best_sc = _score_fft(p)
    total_w = float(sum(w for _, w in stages))
    if total_w <= 0.0:
        total_w = 1.0

    cum = 0.0
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 2.0, start + avail * (cum / total_w))
        if end_stage <= now + 1.5:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            break
        T = max(1e-300, frac * cmax)
        u = run_chunk(u, T, end_stage)
        p_cur = _softmax(u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur = p_cur / ss
        sc_cur = _score_fft(p_cur)
        if not math.isfinite(sc_cur):
            p = best_p.copy()
            u = p_to_u(p)
            continue
        if sc_cur < best_sc:
            best_p = p_cur.copy()
            best_sc = sc_cur
            p = p_cur.copy()
        elif sc_cur > best_sc + max(0.002, 2e-4 * best_sc):
            p = best_p.copy()
        else:
            p = p_cur.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p

def _coord_polish(p, stop_time, max_sweeps=500000, seed=12345, consider=None):
    n = int(np.asarray(p).size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300 or not math.isfinite(s):
        return p, math.inf
    p = p / s

    G = 28 if n <= 5000 else 20
    Ltot = 2 * n - 1
    c = _conv_self(p)
    S_total = 1.0
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0

    def eval_w_scalar(w, base, infl, diag, max_const, S, n_local):
        new_S = S + w
        if new_S <= 1e-15:
            return math.inf
        a_aff = base + w * infl
        a_aff[diag] += w * w
        m = float(a_aff.max())
        if m < max_const:
            m = max_const
        if m < 0.0:
            m = 0.0
        return (2.0 * n_local * m) / (new_S * new_S)

    def eval_w_batch(ws, base, infl, diag, max_const, S, n_local):
        ws = np.asarray(ws, dtype=np.float64)
        new_S = S + ws
        vals = np.full(ws.shape, math.inf, dtype=np.float64)
        valid = new_S > 1e-15
        if not np.any(valid):
            return vals
        aff = base[None, :] + ws[:, None] * infl[None, :]
        aff[:, diag] += ws[valid, None] * 0  # placeholder
        # apply diagonal correction
        aff = aff.copy()
        aff[:, diag] += (ws ** 2)
        maff = aff.max(axis=1)
        maff = np.maximum(maff, max_const)
        maff = np.maximum(maff, 0.0)
        dv = new_S
        vals = (2.0 * n_local * maff) / (dv * dv)
        vals[~valid] = math.inf
        return vals

    while sweep < max_sweeps and time.monotonic() < stop_time - 0.8:
        improved = False
        order = rng.permutation(n)

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.15 or S_total < 0.85:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0

            S = float(S_total)
            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue

            base = c[start_i:end_i]
            infl = 2.0 * p[:Ls]
            diag = i - start_i

            max_const = 0.0
            if start_i > 0:
                ml = float(c[:start_i].max())
                if ml > max_const:
                    max_const = ml
            if end_i < Ltot:
                mr = float(c[end_i:].max())
                if mr > max_const:
                    max_const = mr

            avg = S / float(n)
            x_hi = max(30.0 * v, 100.0 * avg, 0.03 * S, 1e-12)
            x_hi = min(x_hi, 12.0 * S)
            if not math.isfinite(x_hi) or x_hi <= v + 1e-14:
                x_hi = v + max(1e-8, avg * 0.5)
            x_lo = -v * 0.999

            # Coarse grid
            n_pts = G
            wlo, whi = x_lo, x_hi
            pts = np.linspace(wlo, whi, n_pts + 1)
            # Ensure current w=0 is included
            vals = eval_w_batch(pts, base, infl, diag, max_const, S, n)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))

            # Golden section refinement on best interval
            if 0 < imin < n_pts:
                ga = float(pts[imin - 1])
                gb = float(pts[imin + 1])
                gr = (math.sqrt(5.0) - 1.0) / 2.0
                gc = gb - gr * (gb - ga)
                gd = ga + gr * (gb - ga)
                fc = eval_w_scalar(gc, base, infl, diag, max_const, S, n)
                fd = eval_w_scalar(gd, base, infl, diag, max_const, S, n)
                for _ in range(25):
                    if fc < fd:
                        gb = gd
                        gd = gc
                        fd = fc
                        gc = gb - gr * (gb - ga)
                        fc = eval_w_scalar(gc, base, infl, diag, max_const, S, n)
                    else:
                        ga = gc
                        gc = gd
                        fc = fd
                        gd = ga + gr * (gb - ga)
                        fd = eval_w_scalar(gd, base, infl, diag, max_const, S, n)
                best_w = 0.5 * (ga + gb)
                best_val = eval_w_scalar(best_w, base, infl, diag, max_const, S, n)
            else:
                best_w = float(pts[imin])
                best_val = float(vals[imin])

            if math.isfinite(best_val) and best_val < cur_score - 1e-14:
                if x_lo - 1e-15 <= best_w <= x_hi + 1e-15:
                    w = best_w
                    p[i] = v + w
                    c[start_i:end_i] += w * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w * w
                    S_total += w
                    cur_score = best_val
                    improved = True

        # Rescale
        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0

        # Periodic recomputation
        if sweep % 4 == 0:
            c = _conv_self(p)
            new_score = float(2.0 * n * c.max())
            if math.isfinite(new_score):
                cur_score = new_score
                if new_score < best_score - 1e-15:
                    best_score = new_score
                    best_p = p.copy()

        if consider is not None and sweep % 10 == 0:
            try:
                consider(best_p)
            except Exception:
                pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 6:
                if time.monotonic() < stop_time - 4.0:
                    try:
                        perturb_scale = 0.02 + 0.02 * (flat // 6)
                        p_pert = p * np.exp(rng.normal(0.0, min(perturb_scale, 0.15), n))
                        p_pert = np.maximum(p_pert, 0.0)
                        ss = float(p_pert.sum())
                        if ss > 0:
                            p_pert /= ss
                        c_pert = _conv_self(p_pert)
                        sc_pert = float(2.0 * n * c_pert.max())
                        if math.isfinite(sc_pert) and sc_pert < best_score - 1e-14:
                            p = p_pert.copy()
                            c = c_pert
                            S_total = 1.0
                            cur_score = sc_pert
                            best_score = sc_pert
                            best_p = p.copy()
                            if consider is not None:
                                consider(best_p)
                        flat = 0
                    except Exception:
                        break
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _conv_self(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score

def _lp_step(f, stop_time):
    n = int(f.size)
    if n < 20:
        return None
    M = float(_conv_self(f).max())
    if M <= 0 or not math.isfinite(M):
        return None
    # Use a simpler iterative approach instead of full LP for n > 1024
    if n > 1024:
        # Use power-iteration-like approach
        g = np.ones(n, dtype=np.float64) / n
        for _ in range(50):
            if time.monotonic() > stop_time - 3:
                break
            fg = np.convolve(f, g)
            # Scale g down where f*g is near M
            mask = fg > M * 0.95
            g = np.where(mask, g * 0.9, g)
            g = np.maximum(g, 0.0)
            gv = float(g.sum())
            if gv < 1e-10:
                return None
            g /= gv
        return g
    else:
        try:
            import cvxpy as cp
            A = np.zeros((2 * n - 1, n), dtype=np.float64)
            for j in range(n):
                A[j:j + n, j] = f[j]
            g = cp.Variable(n, nonneg=True)
            b = np.full(2 * n - 1, M, dtype=np.float64)
            constraints = [A @ g <= b]
            obj = cp.Maximize(cp.sum(g))
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.SCS, verbose=False)
            if g.value is None:
                return None
            g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
            gv = float(g_val.sum())
            if gv < 1e-10:
                return None
            return g_val / gv
        except Exception:
            return None

def propose_candidate(seed=42, budget_s=1300, **kwargs):

            for t_alpha in [0.3, 0.15, 0.08, 0.04]:
                blended = (1.0 - t_alpha) * work + t_alpha * g_dir
                blended = np.maximum(blended, 0.0)
                bs = float(blended.sum())
                if bs > 0:
                    blended /= bs
                    consider(blended)

    work = best_arr.copy()
    work /= float(work.sum())

    # === Phase 4: Second coordinate descent pass ===
    phase4_end = min(hard - 60.0, t0 + 0.78 * budget)
    if phase4_end > time.monotonic() + 25.0:
        try:
            work, _ = _coord_polish(work, phase4_end, max_sweeps=500000,
                                    seed=int(seed + 123456), consider=consider)
        except Exception:
            pass

    work = best_arr.copy()
    work /= float(work.sum())

    # === Phase 5: Fine LSE + final coordinate polish ===
    phase5_end = min(hard - 30.0, t0 + budget - 35.0)
    if phase5_end > time.monotonic() + 25.0:
        lse5_end = min(phase5_end - 25.0, time.monotonic() + 0.30 * (phase5_end - time.monotonic()))
        if lse5_end > time.monotonic() + 12.0:
            try:
                p5 = _lbfgs_optimize(work, lse5_end, consider=consider, fine_only=True)
                if p5 is not None and p5.size == n_par and np.all(np.isfinite(p5)) and p5.sum() > 0:
                    p5 /= float(p5.sum())
                    consider(p5)
                    work = best_arr.copy()
                    work /= float(work.sum())
            except Exception:
                pass

        coord5_end = phase5_end
        if coord5_end > time.monotonic() + 12.0:
            try:
                work, _ = _coord_polish(work, coord5_end, max_sweeps=300000,
                                        seed=int(seed + 987654), consider=consider)
            except Exception:
                pass

    # === Phase 6: Multi-start exploration if time permits ===
    rem = hard - 20.0 - time.monotonic()
    if rem > 80.0:
        rng_ex = np.random.default_rng(int(seed + 314159))
        for trial in range(3):
            if time.monotonic() > hard - 25.0:
                break
            trial_end = min(hard - 20.0, time.monotonic() + rem * 0.3)
            if trial_end <= time.monotonic() + 15.0:
                break
            # Random perturbation of best
            p_try = best_arr.copy()
            p_try *= np.exp(rng_ex.normal(0.0, 0.05, n_par))
            p_try = np.maximum(p_try, 0.0)
            ts = float(p_try.sum())
            if ts > 0:
                p_try /= ts
                try:
                    p_try, _ = _coord_polish(p_try, trial_end, max_sweeps=100000,
                                             seed=int(seed + 1000 * trial), consider=consider)
                except Exception:
                    pass

    # === Phase 7: Final squeeze ===
    fin_end = min(hard - 10.0, t0 + budget - 8.0)
    if fin_end > time.monotonic() + 10.0:
        work = best_arr.copy()
        work /= float(work.sum())
        try:
            work, _ = _coord_polish(work, fin_end, max_sweeps=100000,
                                    seed=int(seed + 555555), consider=consider)
        except Exception:
            pass

    # Final validation and return
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
```
