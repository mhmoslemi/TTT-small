    if not math.isfinite(m):
        return math.inf
    return float(2.0 * a.size * m / (s * s))


def _lbfgs_lse(p0, stop_time, n_stages=6, consider=None):
    """LSE-smoothed minimization via L-BFGS-B on log-softmax params."""
    p = np.maximum(np.asarray(p0, dtype=np.float64).ravel(), 1e-300)
    n = p.size
    p = p / float(p.sum())
    conv_len = 2 * n - 1
    conv_sz = next_fast_len(conv_len)
    cross_sz = next_fast_len(conv_len)

    def cfast(a):
        F = rfft(a, conv_sz)
        c = irfft(F * F, conv_sz)[:conv_len]
        return np.maximum(c, 0.0)

    def xfast(pp, w):
        F = rfft(pp[::-1], cross_sz)
        G = rfft(w, cross_sz)
        c = irfft(F * G, cross_sz)[:conv_len]
        q = c[n - 1:2 * n - 1]
        return np.maximum(q, 0.0)

    def lse_eval(uu, T):
        uu = np.asarray(uu, dtype=np.float64)
        m = float(np.max(uu))
        e = np.exp(uu - m)
        s = float(e.sum())
        if s < 1e-300 or not math.isfinite(s):
            pp = np.full(n, 1.0 / n)
        else:
            pp = e / s
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros(n)
        T = max(T, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        se = float(exps.sum())
        if not math.isfinite(se) or se <= 0:
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            f_val = cmax
        else:
            f_val = cmax + T * math.log(se / conv_len)
            w = exps / se
            if not math.isfinite(f_val):
                f_val = cmax
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f_val), grad

    # Log-softmax parameterization
    p_init = np.maximum(p, 1e-15)
    u = np.log(p_init) - float(np.mean(np.log(p_init)))

    c_cur = cfast(p)
    cmax_cur = float(c_cur.max())

    total_time = stop_time - time.monotonic()
    if total_time < 10:
        return p

    fracs = [0.08, 0.03, 0.01, 0.003, 0.0008, 0.0002]
    if n_stages < 6:
        fracs = fracs[:n_stages]
    weights = [max(1, int(n_stages - i * 0.5)) for i in range(n_stages)]
    tw = sum(weights)

    best_p = p.copy()
    best_sc = _score_norm(p)
    cum = 0.0

    for i in range(n_stages):
        if time.monotonic() > stop_time - 15:
            break
        cum += weights[i]
        stage_end = min(stop_time - 12.0, t0_global + total_time * 0.85 * (cum / tw))
        if stage_end < time.monotonic() + 5:
            continue
        T = fracs[i] * cmax_cur
        maxit = 50 if n <= 2048 else 35

        for _iter in range(4):
            if time.monotonic() > stage_end - 2:
                break
            opts = {"maxiter": maxit, "gtol": 1e-10, "ftol": 1e-14,
                    "maxls": 20, "maxcor": 30, "disp": False}
            try:
                res = minimize(lse_eval, u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                u = res.x
                pp = np.exp(u - np.max(u))
                pp = pp / pp.sum()
                pp = np.maximum(pp, 0.0)
                sc = _score_norm(pp)
                if math.isfinite(sc) and sc < best_sc - 1e-13:
                    best_sc = sc
                    best_p = pp.copy()
                cmax_cur = float(cfast(pp).max())
                T *= 0.5  # cool within stage
            if getattr(res, "status", 1) == 0:
                break

        if consider:
            try:
                consider(best_p / best_p.sum() * best_p.sum())  # already normalized
            except Exception:
                pass

    return best_p


def _coord_descent(p0, stop_time, rng, consider=None, max_sweeps=200000):
    """Fast coordinate descent with incremental convolution updates."""
    p = np.maximum(np.asarray(p0, dtype=np.float64).ravel(), 0.0).copy()
    n = p.size
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    Ltot = 2 * n - 1
    c = _conv_fft(p)
    S = 1.0  # maintain sum=1 via rescaling
    cur_score = float(2.0 * n * c.max())
    best_score = cur_score
    best_p = p.copy()

    G = 40  # grid points
    sweep = 0
    flat = 0
    t_start = time.monotonic()

    while sweep < max_sweeps and time.monotonic() < stop_time - 2.0:
        improved = False
        order = rng.permutation(n)

        for idx in order:
            i = int(idx)
            if time.monotonic() > stop_time - 0.5:
                break
            if S <= 1e-300:
                break
            # Rescale if drift
            if S > 1.05 or S < 0.95:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0
                cur_score = float(2.0 * n * c.max())

            v = float(p[i])
            start_i = i
            end_i = min(i + n, Ltot)
            Ls = end_i - start_i
            if Ls <= 0:
                continue

            # Unaffected max
            max_const = 0.0
            if start_i > 0:
                mc = float(c[:start_i].max())
                if mc > max_const:
                    max_const = mc
            if end_i < Ltot:
                mc = float(c[end_i:].max())
                if mc > max_const:
                    max_const = mc

            # Affected base and influence
            base = c[start_i:end_i].copy()
            infl = 2.0 * p[:Ls]
            diag = i - start_i  # index within base where w² term applies

            # Grid range
            avg = 1.0 / n
            x_hi = max(30.0 * v, 60.0 * avg, 0.03, 1e-10)
            x_hi = min(x_hi, 10.0)
            lo = 0.0

            pts = np.linspace(lo, x_hi, G, dtype=np.float64)
            # Add current value
            if lo < v < x_hi:
                pts = np.sort(np.concatenate([pts, [v]]))

            # Vectorized evaluation
            w = pts - v
            denom = S + w
            valid = denom > 1e-14
            if not np.any(valid):
                continue

            aff = base[None, :] + w[:, None] * infl[None, :]
            dval = base[diag] + w * infl[diag] + w * w
            aff = np.maximum(aff, dval[:, None])
            aff = np.maximum(aff, max_const)
            mv = aff[valid]
            dv = denom[valid]
            vals = np.where(valid, (2.0 * n * mv) / (dv * dv), np.inf)

            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Refine
            if G > 8:
                dw = x_hi / G
                rlo = max(lo, best_x - 3.0 * dw)
                rhi = min(x_hi, best_x + 3.0 * dw)
                if rhi > rlo:
                    fine = np.linspace(rlo, rhi, 24, dtype=np.float64)
                    wf = fine - v
                    denf = S + wf
                    vf_mask = denf > 1e-14
                    if np.any(vf_mask):
                        afff = base[None, :] + wf[:, None] * infl[None, :]
                        dvf = base[diag] + wf * infl[diag] + wf * wf
                        afff = np.maximum(afff, dvf[:, None])
                        afff = np.maximum(afff, max_const)
                        mfv = afff[vf_mask]
                        dfv = denf[vf_mask]
                        vvals = (2.0 * n * mfv) / (dfv * dfv)
                        if np.any(np.isfinite(vvals)):
                            im2 = int(np.argmin(np.where(np.isfinite(vvals), vvals, np.inf)))
                            ix2 = int(np.where(vf_mask)[0][im2])
                            xv2 = float(fine[ix2])
                            vv2 = float(vvals[im2])
                            if math.isfinite(vv2) and vv2 < best_val:
                                best_x = xv2
                                best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-13:
                w_ch = best_x - v
                if abs(w_ch) > 1e-15:
                    p[i] = best_x
                    S += w_ch
                    c[start_i:end_i] += w_ch * infl
                    di = 2 * i
                    if 0 <= di < Ltot:
                        c[di] += w_ch * w_ch
                    cur_score = best_val
                    improved = True

        # End of sweep: rescale and recompute
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        # Periodic FFT recompute to avoid drift
        if (sweep + 1) % 25 == 0 or not improved:
            c = _conv_fft(p)
            cur_score = float(2.0 * n * c.max())

        if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            if consider:
                try:
                    consider(p.copy())
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                # Perturb to escape
                if time.monotonic() < stop_time - 10:
                    p = p * np.exp(rng.normal(0.0, 0.03, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_fft(p)
                    S = 1.0
                    cur_score = float(2.0 * n * c.max())
                    if cur_score < best_score - 1e-13:
                        best_score = cur_score
                        best_p = p.copy()
                    flat = 0
                else:
                    break
        sweep += 1

    # Final
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_fft(p)
    sc = float(2.0 * n * c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    global t0_global
    t0 = time.monotonic()
    t0_global = t0
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0

    parent = _get_parent()
    n = parent.size
    p = parent.copy()
    ps = float(p.sum())
    if ps < 1e-12:
        p = np.ones(3072)
        n = 3072
    else:
        p = p / ps

    best_arr = p.copy()
    best_score = _score_norm(p)

    def consider(arr):
        nonlocal best_arr, best_score
        a = np.asarray(arr, dtype=np.float64).ravel()
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s < 0.01 or a.size == 0:
            return
        a = a / s
        try:
            sc = _score_norm(a)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_arr = a.copy()
            best_score = sc

    rng = np.random.default_rng(int(seed))

    # === Phase 1: LSE Optimization (45% of time) ===
    p1_end = min(hard - 200.0, t0 + 0.45 * budget)
    if p1_end > t0 + 30.0:
        try:
            p1 = _lbfgs_lse(p, p1_end, n_stages=6, consider=consider)
            if p1 is not None and p1.size == n and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / p1.sum()
                consider(p1)
        except Exception:
            pass

    work = best_arr.copy()

    # === Phase 2: Coordinate Descent (35% of time) ===
    p2_end = min(hard - 120.0, t0 + 0.80 * budget)
    if p2_end > time.monotonic() + 30.0:
        try:
            work, _ = _coord_descent(work, p2_end, rng, consider=consider,
                                     max_sweeps=200000)
        except Exception:
            pass

    work = best_arr.copy()

    # === Phase 3: Perturbation + coordinate descent (10%) ===
    p3_end = min(hard - 60.0, t0 + 0.90 * budget)
    if p3_end > time.monotonic() + 20.0:
        for attempt in range(4):
            if time.monotonic() > p3_end - 15:
                break
            rng2 = np.random.default_rng(int(seed + 1000 * (attempt + 1)))
            perturbed = work * np.exp(rng2.normal(0.0, 0.05, n))
            perturbed = np.maximum(perturbed, 0.0)
            ps2 = float(perturbed.sum())
            if ps2 > 0:
                perturbed /= ps2
            consider(perturbed)
            sub_end = min(p3_end - 10.0, time.monotonic() + (p3_end - p3_start) / 4) \
                if 'p3_start' in dir() else min(p3_end - 10.0, time.monotonic() + 60)
            if sub_end > time.monotonic() + 10:
                try:
                    _, _ = _coord_descent(best_arr.copy(), sub_end, rng2,
                                          consider=consider, max_sweeps=50000)
                except Exception:
                    pass
        # Fix: p3_start variable
        pass

    work = best_arr.copy()

    # === Phase 4: Fine LSE on best (10%) ===
    p4_end = min(hard - 30.0, t0 + 0.95 * budget)
    if p4_end > time.monotonic() + 20.0:
        try:
            p4 = _lbfgs_lse(work, p4_end, n_stages=3, consider=consider)
            if p4 is not None and p4.size == n and np.all(np.isfinite(p4)) and p4.sum() > 0:
                p4 = p4 / p4.sum()
                consider(p4)
        except Exception:
            pass

    # === Phase 5: Final coordinate polish ===
    p5_end = min(hard - 20.0, t0 + budget - 15.0)
    work = best_arr.copy()
    if p5_end > time.monotonic() + 10.0:
        rng_final = np.random.default_rng(int(seed + 777777))
        try:
            _, _ = _coord_descent(work, p5_end, rng_final,
                                  consider=consider, max_sweeps=100000)
        except Exception:
            pass

    # === Final validation and return ===
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = evaluate_sequence(ls)
        if not math.isfinite(final_sc):
            ls = [float(x) for x in out]
            for x in ls:
                if x < 0 or x > 1000:
                    ls = [1.0 / len(out)] * len(out)
                    break
            final_sc = evaluate_sequence(ls)
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
