    n = a.size

    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

    m = float(c.max()) if c.size > 0 else math.inf
    if not math.isfinite(m) or m < 0:

    return float(2.0 * n * m)

    """Compute Q_j = sum_k w_k*a[k-j] for j=0..n-1 using FFT."""
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 3 * n - 2
    size = next_fast_len(L)
    F = rfft(a[::-1], size)
    G = rfft(w, size)
    c = irfft(F * G, size)[:L]
    q = c[n - 1:2 * n - 1]
    np.maximum(q, 0.0, out=q)
    return q
