def _lp_dual_step(f, M, n, rows, cols, f_idx, b_vec, time_limit=10):
    """
    LP: maximize sum(g) s.t. A@g <= M*1, g >= 0
    where A[k,j] = f[k-j] (convolution matrix)
    Returns g/sum(g) or None.
    """
    data = f[f_idx].astype(np.float64)
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n, dtype=np.float64)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_vec, 
                      bounds=[(0, None)]*n,
                      method='highs',
                      options={'time_limit': time_limit, 'output_flag': 0})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = float(g.sum())
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

def _lse_optimize(p, stop_time, consider, seed, conv_size, cross_size, conv_len, cross_len):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = p.size
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    
    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))
    
    u = p_to_u(p)
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c
    
    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n-1:2*n-1]
        np.maximum(q, 0.0, out=q)
        return q
    
    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(np.clip(scaled, -500, 500))
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f_val = cmax
        else:
            f_val = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f_val):
                f_val = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f_val), grad

        (0.003, 0.14),

        (0.0003, 0.12),
        (0.0001, 0.10),
        (0.00003, 0.08),
        (0.00001, 0.06),
    )
