```python
def solve_lp_dual(a, M, n):
    g = cp.Variable(n)
    constraints = [g >= 0]
    # (a*g)[k] <= M for k = 0, ..., 2n-2
    # a is length n, g is length n
    # (a*g)[k] = sum_i a[i]*g[k-i] for valid k-i
```

```python
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def conv_self(a):
    """Compute a * a (autocorrelation) using FFT."""
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
```

```python
def lp_iteration(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Return normalized g."""
    n = len(a)
    # Build A
    k = np.arange(2*n-1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    A = np.where((idx >= 0) & (idx < n), a[idx], 0.0)
    
    # Solve
    c_obj = -np.ones(n)
    b = M * np.ones(2*n-1)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        s = g.sum()
        if s > 1e-10:
            return g / s
    return None
```

For the subgradient method:
- Compute c = a*a (FFT, O(n log n))
- Find active set A = {k : c[k] >= (1-ε) * max(c)}
- Compute subgradient: g = (1/|A|) * sum_{k in A} 2 * shifted_reversed(a, k)
- This is O(|A| * n) for the sum, or O(n log n) with FFT if we use a specific structure
- Update: a_new = a - η * g (projected onto simplex)

For the projected update onto the simplex {x >= 0, sum(x) = 1}, we can use the standard algorithm (sort-based, O(n log n)).

This is much cheaper than the LP approach! Each iteration is O(n log n).

def _conv(a):
    n = len(a)
    L = 2*n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F*F, size)[:L]
    np.maximum(c, 0, out=c)
    return c

def _score(p):

    c = _conv(p)
    return 2.0 * n * c.max() / (s*s)

def _simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)

    theta = (cssv[rho] - 1) / (rho + 1.0)
