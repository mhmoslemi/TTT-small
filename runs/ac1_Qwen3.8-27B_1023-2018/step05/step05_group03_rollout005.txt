import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _fft_convolve(a, n, conv_size):
    F = rfft(a, conv_size)
    c = irfft(F * F, conv_size)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_from_conv(n, c, s):
    if s < 0.01:
        return math.inf
    return float(2.0 * n * c.max() / (s * s))


def _normalize1(a):
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.full_like(a, 1.0 / a.size)
    return a / s, s

def _lbfgs_soft(a, stop_time, n, conv_size, consider=None, T_frac_start=0.003, T_frac_end=0.00008):
    """L-BFGS optimization of log-sum-exp smoothed objective."""
    a = _normalize1(a)[0]
    n = int(a.size)
    if n <= 1 or time.monotonic() > stop_time:
        return a

    def cfast(aa):
        F = rfft(aa, conv_size)
        c = irfft(F * F, conv_size)[:2 * n - 1]
        np.maximum(c, 0.0, out=c)
        return c

    def pp_to_u(pp):
        pp = np.maximum(pp, 1e-300)
        uu = np.log(pp)
        uu -= float(np.mean(uu))
        return uu

    def u_to_pp(uu):
        m = float(np.max(uu))
        e = np.exp(uu - m)
        s = float(e.sum())
        if not math.isfinite(s) or s <= 0:
            return np.full(n, 1.0 / n)
        return e / s

    conv_len = 2 * n - 1

    def lse_func(uu, T):
        uu = np.asarray(uu, dtype=np.float64).ravel()
        if uu.size != n or not np.all(np.isfinite(uu)):
            return 1e100, np.zeros(n)
        pp = u_to_pp(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros(n)
        T = max(1e-15, T)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        se = float(exps.sum())
        if not math.isfinite(se) or se <= 0:
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            w = exps / se
            f = cmax + T * math.log(se / conv_len)

        # Gradient: d/dpp [log sum exp(c/T)] = (1/T) * w * (dc/dpp)
        # dc/dpp[i] = 2*conv(pp, vector with 1 at i) -> the i-th column of 2*toeplitz
        # For the cross-correlation: (pp * w)_k = sum_j pp[j] * w[k-j]
        # Using FFT:
        cross_size = next_fast_len(3 * n - 2)
        F = rfft(pp[::-1], cross_size)
        G = rfft(w, cross_size)
        cross = irfft(F * G, cross_size)[:3 * n - 2]
        gid = 2.0 * cross[n - 1:2 * n - 1]

        gd = float(np.dot(pp, gid))
        grad = pp * (gid - gd) / T
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    u = pp_to_u(a)
    best_a_local = a.copy()
    best_f = 1e100

    # Multi-stage temperature
    c = cfast(a)
    cmax = float(c.max()) if c.size > 0 else 1.0
    if not math.isfinite(cmax) or cmax <= 0:
        return a

    n_stages = 4
    fracs = np.geomspace(T_frac_start, T_frac_end, n_stages)
    total_time = max(1.0, stop_time - time.monotonic())
    time_per_stage = total_time / n_stages

    for si, frac in enumerate(fracs):
        stage_end = min(stop_time - 1.0, time.monotonic() + time_per_stage)
        if stage_end <= time.monotonic() + 5.0:
            break
        T = frac * cmax
        opts = {
            "maxiter": 25,
            "gtol": 1e-10,
            "ftol": 1e-15,
            "maxls": 12,
            "maxcor": 12,
            "disp": False,
        }
        for _ in range(3):
            if time.monotonic() > stage_end - 1.0:
                break
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)) and math.isfinite(res.fun):
                if res.fun < best_f - 1e-12:
                    best_f = float(res.fun)
                    u = res.x.copy()
                    a_new = u_to_pp(u)
                    sc = _score_from_conv(n, cfast(a_new), a_new.sum())
                    if math.isfinite(sc):
                        best_a_local = a_new
                    if consider is not None:
                        try:
                            consider(a_new)
                        except Exception:
                            pass

    return best_a_local


def _lp_direction(a, M):
    """LP-based direction finding for small-to-medium n."""
    try:
        from scipy.optimize import linprog
        from scipy import sparse as sp
    except Exception:
        return None

    n = int(a.size)
    if n < 64 or n > 800:
        return None
    if not math.isfinite(M) or M <= 0:
        return None

    try:
        idx = np.arange(n, dtype=np.int32)
        cols = np.repeat(idx, n)
        off = np.tile(idx, n)
        rows = cols + off
        data = a[off].astype(np.float64)
        A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        cobj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        res = linprog(cobj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method="highs")
        if res is not None and getattr(res, "success", False):
            b = np.asarray(res.x, dtype=np.float64).ravel()
            b = np.where(np.isfinite(b), b, 0.0)
            b = np.maximum(b, 0.0)
            if float(b.sum()) > 1e-10:
                return b / float(b.sum())
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(10.0, budget)

    # Reserve time for final evaluation
    reserve = 12.0
    hard_stop = t0 + budget - reserve

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(3072, dtype=np.float64)

    parent = np.where(np.isfinite(parent), parent, 0.0)
    parent = np.clip(parent, 0.0, 1000.0)
    ps = float(parent.sum())
    if not math.isfinite(ps) or ps < 0.01:
        parent = np.ones(3072, dtype=np.float64)
        ps = 3072.0
    a = parent / ps  # Normalize to sum=1
    n = a.size

    conv_size = next_fast_len(2 * n - 1)

    def _eval_exact(arr):
        try:
            ls = [float(x) for x in np.asarray(arr, dtype=np.float64).ravel()]
            return float(evaluate_sequence(ls))
        except Exception:
            return math.inf

    def _score_arr(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        s = float(arr.sum())
        if s < 0.01:
            return math.inf
        c = _fft_convolve(arr, arr.size, next_fast_len(2 * arr.size - 1))
        return _score_from_conv(arr.size, c, s)

    best_a = a.copy()
    best_score = _score_arr(best_a)

    def consider(cand):
        nonlocal best_a, best_score
        try:
            arr = np.asarray(cand, dtype=np.float64).ravel()
            if arr.size != n:
                return
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            s = float(arr.sum())
            if s < 0.01:
                return
            sc = _score_arr(arr)
            if math.isfinite(sc) and sc < best_score - 1e-14:
                best_score = sc
                best_a = arr.copy()
        except Exception:
            pass

    rng = np.random.default_rng(seed)

    # Phase 1: Quick perturbations (40s)
    p1_end = min(hard_stop - 60.0, t0 + 40.0)
    base = best_a.copy()
    power_ts = (0.6, 0.75, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5)
    for q in power_ts:
        if time.monotonic() > p1_end - 10.0:
            break
        cand = np.power(np.maximum(base, 1e-300), float(q))
        cand = _normalize1(cand)[0]
        consider(cand)

    for _ in range(8):
        if time.monotonic() > p1_end - 5.0:
            break
        side = 1.0 if rng.random() < 0.5 else -1.0
        amp = 0.2 + 0.8 * float(rng.random())
        x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
        f1 = 1 + int(rng.integers(0, 3))
        ph = 2.0 * math.pi * float(rng.random())
        cand = base * np.exp(side * amp * 0.5 * np.sin(2.0 * math.pi * f1 * x + ph))
        cand = _normalize1(np.maximum(cand, 0.0))[0]
        consider(cand)

    # Small LP direction if n allows
    if n <= 800 and time.monotonic() < p1_end:
        c = _fft_convolve(best_a, n, conv_size)
        M = float(c.max())
        g = _lp_direction(best_a, M)
        if g is not None:
            for t in (0.3, 0.2, 0.1, 0.05):
                cand = (1.0 - t) * best_a + t * g
                cand = _normalize1(cand)[0]
                consider(cand)

    # Phase 2: L-BFGS with log-sum-exp (250s)
    if time.monotonic() < hard_stop - 400.0:
        p2_end = min(hard_stop - 350.0, time.monotonic() + 250.0)
        a_work = best_a.copy()
        try:
            a_work = _lbfgs_soft(a_work, p2_end, n, conv_size, consider=consider,
                                  T_frac_start=0.002, T_frac_end=0.0001)
        except Exception:
            pass
        consider(a_work)

    # Phase 3: Main intensive coordinate descent (large chunk)
    p3_end = min(hard_stop - 120.0, time.monotonic() + max(300.0, (hard_stop - time.monotonic()) * 0.70))
    if p3_end > time.monotonic() + 30.0:
        a_work = best_a.copy()
        best_a, _ = _coord_descent_intensive(
            a_work, p3_end, n, conv_size,
            seed=seed + 7777, consider=consider, initial_flat_thresh=4
        )
        consider(best_a)

    # Phase 4: Fine L-BFGS polish
    if time.monotonic() < hard_stop - 80.0:
        p4_end = min(hard_stop - 60.0, time.monotonic() + 80.0)
        a_work = best_a.copy()
        try:
            a_work = _lbfgs_soft(a_work, p4_end, n, conv_size, consider=consider,
                                  T_frac_start=0.0005, T_frac_end=0.00003)
        except Exception:
            pass
        consider(a_work)

    # Phase 5: Final coordinate polish
    if time.monotonic() < hard_stop - 15.0:
        a_work = best_a.copy()
        best_a, _ = _coord_descent_intensive(
            a_work, hard_stop - 5.0, n, conv_size,
            seed=seed + 424242, consider=consider, initial_flat_thresh=6
        )
        consider(best_a)

    # Final: validate with exact evaluator
    try:
        out = np.asarray(best_a, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = _normalize1(parent)[0]
            s = float(out.sum())
        if out.size == 0 or s < 0.01:
            out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = _eval_exact(ls)
        if math.isfinite(final_sc):
            return ls
    except Exception:
        pass

    # Fallback
    try:
        ls = [float(x) for x in _normalize1(parent)[0]]
        return ls
    except Exception:
        return [1.0]
