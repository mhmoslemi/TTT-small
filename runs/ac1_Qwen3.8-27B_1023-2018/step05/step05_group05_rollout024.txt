    G = max(24, min(50, int(120000 / max(n, 1)) + 24))
    G_fine = 25
    
    # Track coordinate activity
    activity = np.zeros(n)
    
    while sweep < 100000 and time.monotonic() < stop_time - 2.0:
        improved = False
        n_improved = 0
        
        # Use activity-weighted random order
        w = 1.0 + 5.0 * activity / (activity.sum() + 1e-10)
        w = w / w.sum()
        order = rng.choice(n, size=n, replace=False, p=w)
        
        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S <= 1e-300:
                break
            
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]
            
            max_const = 0.0
            if start > 0:
                max_const = max(max_const, float(np.max(c[:start])))
            if end < Ltot:
                max_const = max(max_const, float(c[end:]))
            
            x_hi = max(25.0 * v, 60.0 / n, 0.005)
            x_hi = min(x_hi, 5.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = 1.0 / n
            
            pts = np.linspace(0.0, x_hi, G + 1)
            
            w_ = pts - v
            denom = S + w_
            valid = denom > 1e-14
            if not np.any(valid):
                continue
            
            aff = base[None, :] + w_[valid, None] * infl[None, :]
            diag_idx = i - start
            if 0 <= diag_idx < Ls:
                dvals = base[diag_idx] + w_[valid] * infl[diag_idx] + w_[valid] ** 2
                aff = np.maximum(aff, dvals[:, None])
            aff = np.maximum(aff, max_const)
            mv = np.max(aff, axis=1)
            
            out = np.full(pts.shape, math.inf)
            dv = denom[valid]
            out[valid] = (2.0 * n * mv) / (dv * dv)
            
            if not np.any(np.isfinite(out)):
                continue
            imin = int(np.argmin(out))
            bx = float(pts[imin])
            bv = float(out[imin])
            
            # Fine refinement
            dw = x_hi / G
            lo = max(0.0, bx - 2.5 * dw)
            hi = min(x_hi, bx + 2.5 * dw)
            if hi - lo > 1e-12:
                fine = np.linspace(lo, hi, G_fine)
                w_f = fine - v
                denom_f = S + w_f
                valid_f = denom_f > 1e-14
                if np.any(valid_f):
                    aff_f = base[None, :] + w_f[valid_f, None] * infl[None, :]
                    if 0 <= diag_idx < Ls:
                        dvals_f = base[diag_idx] + w_f[valid_f] * infl[diag_idx] + w_f[valid_f] ** 2
                        aff_f = np.maximum(aff_f, dvals_f[:, None])
                    aff_f = np.maximum(aff_f, max_const)
                    mv_f = np.max(aff_f, axis=1)
                    out_f = np.full(fine.shape, math.inf)
                    out_f[valid_f] = (2.0 * n * mv_f) / (denom_f[valid_f] ** 2)
                    if np.any(np.isfinite(out_f)):
                        im2 = int(np.argmin(out_f))
                        if out_f[im2] < bv:
                            bv = float(out_f[im2])
                            bx = float(fine[im2])
            
            if math.isfinite(bv) and bv < cur_score - 1e-14 and 0.0 <= bx:
                w_ = bx - v
                if abs(w_) > 1e-15:
                    p[i] = bx
                    S += w_
                    c[start:end] += w_ * infl
                    if 0 <= diag_idx < Ls:
                        c[start + diag_idx] += w_ * w_
                    cur_score = bv
                    improved = True
                    n_improved += 1
                    activity[i] += 1.0
                    activity[i // max(1, n // 100)] += 0.5  # also boost block
        
        # Rescale
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break
        
        # Verify
        if sweep % 3 == 0:
            c = _conv_self(p)
        new_sc = 2.0 * n * float(c.max())
        if math.isfinite(new_sc) and new_sc < best_score - 1e-15:
            best_score = new_sc
            best_p = p.copy()
            if consider:
                consider(best_p)
        
        # Decay activity
        activity *= 0.95
        
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 30.0:
                    # Adaptive perturbation
                    sigma = 0.02 * (1.0 + 0.3 * min(flat, 20) / 20.0)
                    # Perturb with decorrelation
                    noise = rng.normal(0, 1, n)
                    # Smooth the noise to create correlated perturbations
                    kernel = np.ones(7) / 7.0
                    noise = np.convolve(noise, kernel, mode='same')
                    noise = noise / (np.std(noise) + 1e-10)
                    p = p * np.exp(sigma * noise)
                    p = np.maximum(p, 0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                    c = _conv_self(p)
                    S = 1.0
                    cur_score = 2.0 * n * float(c.max())
                    flat = 0
                    activity *= 0.5
                else:
                    break
        sweep += 1
    
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_self(p)
    sc = 2.0 * n * float(c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score
