import time
import math
from scipy.fft import rfft, irfft, next_fast_len

def _conv_self(a):
    """FFT-based autocorrelation."""
    n = len(a)
    L = 2*n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fft(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    return 2.0 * n * c.max() / (s*s)

def _conv_self(a):
    n = len(a)

    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    # small negative values from FFT rounding
    np.maximum(c, 0.0, out=c)
    return c
