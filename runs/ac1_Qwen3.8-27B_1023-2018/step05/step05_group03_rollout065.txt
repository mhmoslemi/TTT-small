                best_a = a.copy()
            if not improved or nimp < 2:
                stale5 += 1
                if stale5 >= 3:
                    break
            else:
                stale5 = 0

    # Final validation
    try:
        out = np.asarray(best_a, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        ss = float(out.sum())
        if out.size == 0 or ss < 0.01:
            out = np.full(n, 1.0 / n, dtype=np.float64)
            ss = 1.0
        out = out / ss
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            # Fallback: parent
            out = np.asarray(p, dtype=np.float64).ravel()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            ss = float(out.sum())
            if ss < 0.01:
                out = np.full(n, 1.0 / n, dtype=np.float64)
                    ss = 1.0
            out = out / ss
            ls = [float(x) for x in out]
        return ls
    except Exception:
        return [1.0]
