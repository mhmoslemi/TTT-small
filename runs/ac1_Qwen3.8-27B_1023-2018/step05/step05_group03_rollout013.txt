def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)

    s = float(p.sum())
    if not math.isfinite(s) or s < 1e-12:
        p = np.ones(3072, dtype=np.float64)
    return p

def _fft_conv(a):
    n = a.size
    L = 2*n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F*F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s*s))

def _lse_smooth(p, stop_time, consider):
    """Log-sum-exp based smooth optimization (L-BFGS in log-space)."""
    n = p.size
    if n < 4:
        return p
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2
    conv_size = next_fast_len(conv_len)
    cross_size = next_fast_len(cross_len)
    F_p = rfft(p[::-1], cross_size)

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(w):
        G = rfft(w, cross_size)
        c = irfft(F_p * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def obj_grad(u_arr, T):
        m = float(np.max(u_arr))
        e = np.exp(u_arr - m)
        s = float(e.sum())
        if s < 1e-300 or not math.isfinite(s):
            pp = np.full(n, 1.0/n)
        else:
            pp = e / s
        c = cfast(pp)
        cmax = float(c.max())
        T = max(T, 1e-14)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_e = float(exps.sum())
        if sum_e < 1e-300 or not math.isfinite(sum_e):
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            w = exps / sum_e
            f = cmax + T * math.log(sum_e / conv_len)
            if not math.isfinite(f):
                f = cmax
        q = xfast(w)
        gid = 2.0 * q
        gd = float(np.dot(pp, gid))
        grad = pp * (gid - gd)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return f, grad

    from scipy.optimize import minimize

    def p_to_u(pp):
        pp = np.maximum(pp, 1e-15)
        uu = np.log(pp)
        uu -= float(np.mean(uu))
        return uu

    def u_to_pp(uu):
        uu = np.where(np.isfinite(uu), uu, 0.0)
        m = float(np.max(uu))
        e = np.exp(uu - m)
        s = float(e.sum())
        if s <= 1e-300:
            return np.full(n, 1.0/n)
        return e / s

    u = p_to_u(p)
    best_u = u.copy()
    best_p_out = p.copy()
    best_sc = _score_f(p)

    stages = [(0.005, 0.4), (0.001, 0.3), (0.0002, 0.3)]
    for frac, _ in stages:

            break

                lambda uu: (obj_grad(uu, T)),
                best_u,
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': 25, 'gtol': 1e-8, 'ftol': 1e-14,

            )

                x = np.asarray(res.x, dtype=np.float64).ravel()

                        best_p_out = pp.copy()
                        best_u = u_to_pp(pp)  # update u
                        if consider is not None:
                            consider(pp.copy())
        except Exception:
            pass

    return best_p_out
