    """Find exact minimizer of v(t) = max_k ((1-t)a + tg) * ((1-t)a + tg)_k for t in [0,1]."""
    n = a.size
    aa = np.convolve(a, a)
    ag = np.convolve(a, g)
    gg = np.convolve(g, g)

    # At the minimizer, the derivative at the active lag is 0.
    # t_k = (aa_k - ag_k) / (aa_k - 2*ag_k + gg_k)
    # if denominator > 0
    
    # For each lag k, compute the t where that lag's quadratic is minimized:
    # q_k(t) = (1-t)^2 * aa_k + 2t(1-t) * ag_k + t^2 * gg_k
    # q_k'(t) = -2(1-t)*aa_k + 2(1-2t)*ag_k + 2t*gg_k = 0
    # -aa_k + t*aa_k + ag_k - 2t*ag_k + t*gg_k = 0
    # t*(aa_k - 2*ag_k + gg_k) = aa_k - ag_k
    # t_k = (aa_k - ag_k) / (aa_k - 2*ag_k + gg_k)
    
    denom = aa - 2*ag + gg
    numer = aa - ag
    
    # Find the lag that gives the minimum of v(t)
    # v(t) is the upper envelope of the quadratics q_k(t)
    # The minimum of v(t) occurs where v'(t) = 0
    
    # Strategy: binary search on t using the subgradient
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        # Compute v'(mid) using the active lag
        v_at_mid = (1-mid)**2 * aa + 2*mid*(1-mid) * ag + mid**2 * gg
        k_star = np.argmax(v_at_mid)
        # Derivative at k_star
        dp = -2*(1-mid)*aa[k_star] + 2*(1-2*mid)*ag[k_star] + 2*mid*gg[k_star]
        if dp < 0:
            lo = mid
        else:
            hi = mid

    result = (1-t_opt)*a + t_opt*g
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-15:
        return None, float(aa.max())
    result = result / s
    val = float(np.convolve(result, result).max())

def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)

def _build_A(n, a):
    """Build (2n-1) x n convolution matrix: A[k,j] = a[k-j] if valid."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)

        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        A[k, jlo:jhi] = a[k - jhi + 1:k - jlo + 1][::-1]

                          options={'time_limit': lp_time,
                                   'dual_feasibility_tolerance': 1e-14,
                                   'primal_feasibility_tolerance': 1e-14})
            if res.status == 0 and res.x is not None:
                g = np.maximum(res.x, 0.0)
                gs = g.sum()
                if gs > 1e-12:
                    g = g / gs
                    # Check if g differs from a
                    if float(np.max(np.abs(g - a))) > 1e-12:
                        # Exact convex line search
                        aa = np.convolve(a, a)
                        ag = np.convolve(a, g)
                        gg = np.convolve(g, g)
                        cand, val = _exact_line_search(aa, ag, gg, a, g)
                        if cand is not None and val < M * (1 - 1e-12):
                            a = cand
                            moved = True
                            no_lp_stuck = 0
        except Exception:
            pass

        if not moved:
            no_lp_stuck += 1
            # Try subgradient
            cand = _subgradient_step(a, conv, M)
            if cand is not None:
                a = cand
                moved = True
                no_lp_stuck = 0

        # If LP and subgradient both failed, try averaging direction
        if not moved and no_lp_stuck < 10:
            try:
                c_obj = -np.ones(n)
                b_ub = np.full(A.shape[0], M * (1 + 1e-4))
                res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                              bounds=[(0, None)] * n, method='highs',
                              options={'time_limit': lp_time * 0.5})
                if res.status == 0 and res.x is not None:
                    g2 = np.maximum(res.x, 0.0)
                    g2s = g2.sum()
                    if g2s > 1e-12:
                        g2 = g2 / g2s
                        g_avg = 0.5 * g2 + 0.5 * a
                        g_avg = np.maximum(g_avg, 0)
                        g_avg_s = g_avg.sum()
                        if g_avg_s > 1e-12:
                            g_avg = g_avg / g_avg_s
                            if float(np.max(np.abs(g_avg - a))) > 1e-12:
                                aa = np.convolve(a, a)
                                ag = np.convolve(a, g_avg)
                                gg = np.convolve(g_avg, g_avg)
                                cand, val = _exact_line_search(aa, ag, gg, a, g_avg)
                                if cand is not None and val < M * (1 - 1e-12):
                                    a = cand
                                    moved = True
                                    no_lp_stuck = 0
            except Exception:
                pass

        if not moved:
            # Small random walk
            a *= np.exp(rng.normal(0, 0.015, n))
            a = np.maximum(a, 1e-12)
            a = a / a.sum()

    return best_a


def _optimize_gradient(a0, deadline, seed=42):
    """Smoothed gradient descent as an alternative optimizer."""
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_v = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)

    # Use soft-max smoothing with annealing
    tau = 200.0
    lr = 0.1
    pat = 0

    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        M = float(conv.max())
        if M < best_v * (1 - 1e-13):
            best_v = M
            best_a = a.copy()
            pat = 0
        else:
            pat += 1

        if pat > 500:
            break

        # Soft-max weights
        shift = conv.max()
        w = np.exp(tau * (conv - shift))
        w_sum = w.sum()
        if w_sum < 1e-15:
            break
        w = w / w_sum

        # Gradient of soft-logsumexp
        dLSE = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            jlo = max(0, k - n + 1)
            jhi = min(n, k + 1)
            for j in range(jlo, jhi):
                idx = k - j
                if 0 <= idx < n:
                    dLSE[j] += 2.0 * wk * a[idx]

        # Full gradient including 1/s^2 factor (s=1 so just dLSE)
        grad = dLSE - dLSE.sum() / n  # project to tangent of simplex
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        # Gradient step with backtracking line search
        improved = False
        for _ in range(20):
            step = lr * grad / grad_norm
            cand = a - step
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-12:
                break
            cand = cand / cs
            v = float(np.convolve(cand, cand).max())
            if v < M * (1 - 1e-10):
                a = cand
                improved = True
                break
            lr *= 0.5

        if not improved:
            lr = min(lr * 2, 0.3)
            # Anneal tau (make less smooth)
            tau = max(tau * 0.95, 20.0)
            # Small perturbation
            if pat > 100:
                a *= np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                pat = 0

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0

    # Get parent
    try:
        parent = np.asarray(globals()['height_sequence_1'], dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(200)
    parent = np.clip(parent, 0, 1000)
    if parent.size == 0 or not np.all(np.isfinite(parent)):
        parent = np.ones(200)
    n_parent = parent.size

    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(100)
        best_score = _score(best_seq)

    def try_update(seq):
        nonlocal best_seq, best_score
        try:
            seq = np.clip(np.asarray(seq, dtype=np.float64), 0, 1000)
            sc = _score(seq)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
        except Exception:
            pass

    # ===== Phase 1: LP optimization on parent (40% of budget) =====
    t1 = t0 + min(budget_s * 0.35, 450.0)
    try:
        r = _optimize(parent, t1, seed=1, lp_time=4.0)
        try_update(r)
    except Exception:
        pass

    # ===== Phase 2: Gradient-based refinement (15%) =====
    if time.monotonic() < t1 + budget_s * 0.15:
        t2 = min(t0 + budget_s * 0.50, t1 + budget_s * 0.15)
        try:
            r = _optimize_gradient(best_seq, t2, seed=55)
            try_update(r)
        except Exception:
            pass
        # More LP polishing
        t2b = min(t0 + budget_s * 0.55, t2 + 60)
        if time.monotonic() < t2b:
            try:
                r = _optimize(best_seq, t2b, seed=222, lp_time=3.0)
                try_update(r)
            except Exception:
                pass

    # ===== Phase 3: Different lengths (20%) =====
    t3 = t0 + min(budget_s * 0.72, 900.0)
    length_offsets = [-40, -20, 20, 40, 80, 120]
    tried_lengths = set()

    for dn in length_offsets:
        if time.monotonic() > t3 - 30:
            break
        nl = n_parent + dn
        if nl < 50 or nl > 600 or nl in tried_lengths:
            continue
        tried_lengths.add(nl)

        # Interpolate
        x_old = np.linspace(0, 1, n_parent)
        x_new = np.linspace(0, 1, nl)
        init = np.interp(x_new, x_old, best_seq)
        init = np.clip(init, 0, 1000)
        init_s = init.sum()
        if init_s < 1e-10:
            continue
        init = init / init_s

        alloc = min(time.monotonic() + 50, t3 - 20)
        try:
            r = _optimize(init, alloc, seed=1000 + nl, lp_time=3.0)
            try_update(r)
        except Exception:
            pass

    # ===== Phase 4: Diverse initializations (15%) =====
    t4 = t0 + min(budget_s * 0.85, 1100.0)
    pn = best_seq / max(best_seq.sum(), 1e-15)
    n_use = pn.size
    x = np.linspace(0, 1, n_use)
    rng_init = np.random.default_rng(2024)

    inits = []
    # Gaussian
    for sig in [0.04, 0.08, 0.12, 0.2, 0.35, 0.6]:
        v = np.exp(-0.5 * ((x - 0.5) / sig) ** 2)
        inits.append(v / v.sum())
    # Cosine
    for m in [1, 2, 3, 5, 8, 12]:
        v = np.maximum(np.cos(m * np.pi * (x - 0.5)) ** 2 + 0.001, 0)
        inits.append(v / v.sum())
    # Power law from center
    for p in [0.3, 0.7, 1.5, 3.0]:
        d = np.abs(x - 0.5)
        v = (1 - d * (n_use - 1)) ** p
        v = np.maximum(v, 0)
        if v.sum() > 0:
            inits.append(v / v.sum())
    # Two-peaked
    for w in [0.1, 0.2, 0.3]:
        v = np.exp(-0.5 * ((x - w) / 0.06) ** 2) + np.exp(-0.5 * ((x - (1-w)) / 0.06) ** 2)
        inits.append(v / v.sum())
    # Blends
    for frac in [0.3, 0.6]:
        for init in inits[:4]:
            v = frac * pn + (1 - frac) * init
            inits.append(v / v.sum())
    # Random
    for _ in range(8):
        v = np.abs(rng_init.normal(1, 0.6, n_use)) + 0.01
        inits.append(v / v.sum())

    per_init = max(4.0, (t4 - time.monotonic()) / max(1, len(inits)))
    for i, init in enumerate(inits):
        if time.monotonic() > t4 - 5:
            break
        alloc = min(time.monotonic() + per_init, t4 - 5)
        try:
            r = _optimize(init, alloc, seed=3000 + i, lp_time=2.0)
            try_update(r)
        except Exception:
            pass

    # ===== Phase 5: Final deep polish (10%) =====
    if time.monotonic() < deadline - 10:
        ab = best_seq / max(best_seq.sum(), 1e-15)
        try:
            r = _optimize(ab, deadline, seed=9999, lp_time=3.0)
            try_update(r)
        except Exception:
            pass

    # Final validation
    try:
        fs = float(evaluate_sequence([float(v) for v in best_seq]))
        if not np.isfinite(fs):
            return [1.0] * 100
    except Exception:
        return [1.0] * 100

    return [float(v) for v in best_seq]
