from scipy.optimize import minimize

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)

def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c

def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(a, a)
    if b.size == 0:
        return math.inf
    m = float(b.max())
    if not math.isfinite(m) or m < 0.0:
        return math.inf
    return float(2.0 * a.size * m / (s * s))

def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    m = float(np.max(u))
    e = np.exp(u - m, dtype=np.float64)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0/u.size)
    return e / s

def _lse_optimize(p, stop_time, consider=None, fine_only=False):
    """Log-sum-exp smooth optimization using L-BFGS-B."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = p.size
    if n == 0: return p
    S = float(p.sum())
    if S < 1e-300: return p
    p = p / S
    
    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))
    
    u = p_to_u(p)
    conv_size = next_fast_len(2*n-1)
    conv_len = 2*n-1
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F*F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c
    
    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        scaled = (c - cmax) / max(T, 1e-300)
        exps = np.exp(scaled)
        se = float(exps.sum())
        if se <= 0 or not math.isfinite(se):
            w = np.zeros(c.size); w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * math.log(se / c.size)
            w = exps / se
        
        # Gradient: cross-correlation of w with pp
        Fp = rfft(pp[::-1], conv_size)
        Fw = rfft(w, conv_size)
        q = irfft(Fp * Fw, conv_size)[:conv_len]
        q2 = q[n-1:2*n-1]
        np.maximum(q2, 0.0, out=q2)
        grad = pp * (2.0*q2 - float(np.dot(pp, 2.0*q2)))
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return f, grad
    
    if fine_only:
        stages = ((0.002, 0.3), (0.0006, 0.4), (0.00015, 0.3))
    else:
        stages = ((0.1, 0.1), (0.03, 0.15), (0.008, 0.15),
                  (0.002, 0.2), (0.0005, 0.18), (0.0001, 0.15), (0.00003, 0.17))
    
    start = time.monotonic()
    avail = stop_time - start - 5.0
    if avail <= 3: return p
    
    best_p = p.copy()
    best_sc = 2.0*n*float(cfast(p).max())
    total_w = sum(wt for _, wt in stages)
    cum = 0.0
    
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3: break
        cum += wt
        end_stage = min(stop_time - 2.0, start + avail*(cum/total_w))
        if end_stage <= now + 1: continue
        c = cfast(p)
        cmax = float(c.max())
        if cmax <= 0: break
        T = max(1e-300, frac*cmax)
        
        # Run L-BFGS for a limited time
        f0, _ = lse_func(u, T)
        best_u = u.copy()
        best_f = f0
        
        while time.monotonic() < end_stage - 1.0:
            try:
                res = minimize(lse_func, best_u, args=(T,), method='L-BFGS-B',
                             jac=True, options={'maxiter': 30, 'gtol': 1e-12,
                                                'ftol': 1e-15, 'maxls': 15, 'maxcor': 25, 'disp': False})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    fv = float(res.fun)
                    if math.isfinite(fv) and fv < best_f:
                        best_f = fv
                        best_u = res.x.copy()
                    if getattr(res, 'status', 1) == 0:
                        break
                else:
                    break
            except Exception:
                break
        
        p_cur = _softmax(best_u)
        p_cur = np.maximum(p_cur, 0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
            sc = 2*n*float(cfast(p_cur).max())
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_p = p_cur.copy()
                p = p_cur.copy()
                if consider: consider(best_p)
            u = p_to_u(p)
    
    return best_p

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p

def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    sz = next_fast_len(L)
    F = rfft(a, sz)
    c = irfft(F * F, sz)[:L]
    c = np.asarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c

def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0: return math.inf
    s = float(a.sum())
    if s < 0.01: return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m) or m < 0: return math.inf
    return 2.0 * n * m / (s*s)

def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0: return math.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01: return math.inf
    b = np.convolve(a, a)
    if b.size == 0: return math.inf
    m = float(b.max())
    if not math.isfinite(m) or m < 0: return math.inf
    return 2.0 * a.size * m / (s*s)

def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0: return u
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0/u.size)
    return e / s

def _coord_desc(p, stop_time, consider=None, seed=42, G=24, Gf=14):
    """Fast coordinate descent for AC1 minimization."""
    n = len(p)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    S = float(p.sum())
    if S < 1e-300:
        return p, math.inf
    p /= S
    
    Lt = 2*n - 1
    c = _conv_self(p)
    cur_sc = 2.0*n*float(c.max())
    best_sc = cur_sc
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    two_p = 2.0 * p
    flat_count = 0
    
    # Preallocate work arrays
    max_grid = G + Gf
    w_arr = np.empty(max_grid)
    vals_arr = np.empty(G+1)
    
    sweep = 0
    while time.monotonic() < stop_time - 2.0:
        # Precompute prefix and suffix max
        pref = np.maximum.accumulate(c)
        suf = np.maximum.accumulate(c[::-1])[::-1]
        
        order = rng.permutation(n)
        improved = False
        
        for idx in order:
            i = int(idx)
            v = p[i]
            start = i
            end = min(i + n, Lt)
            Ls = end - start
            if Ls <= 0: continue
            
            # Outside max
            mc = 0.0
            if start > 0:
                pv = pref[start-1]
                if pv > mc: mc = pv
            if end < Lt:
                sv = suf[end]
                if sv > mc: mc = sv
            
            base = c[start:end]  # view
            infl = two_p[:Ls]   # view
            diag = i            # diagonal position within affected range
            
            if diag < 0 or diag >= Ls:
                continue
            
            # Search range
            x_hi = min(20.0, max(30.0*v, 100.0/n, 0.005))
            if x_hi <= 0: continue
            
            # Coarse grid
            pts = np.linspace(0.0, x_hi, G+1)
            w = pts - v
            aff = w[:,None] * infl[None,:] + base[None,:]
            aff[:,diag] += w*w
            maff = np.maximum(aff.max(axis=1), mc)
            vals = 2.0*n*maff / ((1.0+w)**2)
            
            imin = int(np.argmin(vals))
            bx = float(pts[imin])
            bv = float(vals[imin])
            
            # Fine grid around best
            if math.isfinite(bv):
                dw = x_hi / G
                lo = max(0.0, bx - 2.0*dw)
                hi = min(x_hi, bx + 2.0*dw)
                if hi - lo > 1e-12:
                    fine = np.linspace(lo, hi, Gf)
                    wf = fine - v
                    afff = wf[:,None] * infl[None,:] + base[None,:]
                    afff[:,diag] += wf*wf
                    mafff = np.maximum(afff.max(axis=1), mc)
                    valsf = 2.0*n*mafff / ((1.0+wf)**2)
                    imf = int(np.argmin(valsf))
                    if valsf[imf] < bv - 1e-15:
                        bx = float(fine[imf])
                        bv = float(valsf[imf])
            
            # Apply update
            if bv < cur_sc - 1e-13 and 0.0 <= bx <= x_hi:
                wb = bx - v
                if abs(wb) > 1e-15:
                    p[i] = bx
                    c[start:end] += wb * infl
                    di = 2*i
                    if 0 <= di < Lt:
                        c[di] += wb*wb
                    two_p[i] = 2.0*bx
                    cur_sc = bv
                    improved = True
        
        sweep += 1
        
        # Periodic recompute
        if sweep % 30 == 0:
            c = _conv_self(p)
            cur_sc = 2.0*n*float(c.max())
            two_p = 2.0*p
        
        # Track best
        if cur_sc < best_sc - 1e-14:
            best_sc = cur_sc
            best_p = p.copy()
            if consider:
                try: consider(best_p)
                except: pass
            flat_count = 0
        elif not improved:
            flat_count += 1
        else:
            flat_count = 0
        
        # Diversification
        if flat_count >= 8 and flat_count % 6 == 0:
            sigma = min(0.15, 0.02 * (flat_count // 6 + 1))
            p = p * np.exp(rng.normal(0, sigma, n))
            p = np.maximum(p, 0)
            ss = float(p.sum())
            if ss > 0: p /= ss
            c = _conv_self(p)
            cur_sc = 2.0*n*float(c.max())
            two_p = 2.0*p
            flat_count = 0
    
    return best_p, best_sc

def _lse_opt(p, stop_time, consider=None, fine_only=False):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = p.size
    if n == 0: return p
    S = float(p.sum())
    if S < 1e-300: return p
    p = p / S
    
    def p2u(pp):
        uu = np.log(np.clip(pp, 1e-300, None))
        return uu - float(np.mean(uu))
    
    u = p2u(p)
    sz = next_fast_len(2*n-1)
    clen = 2*n-1
    
    def cfast(a):
        F = rfft(a, sz)
        c = irfft(F*F, sz)[:clen]
        np.maximum(c, 0.0, out=c)
        return c
    
    def lse_fn(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cm = float(c.max())
        sc_ = (c - cm) / max(T, 1e-300)
        ex = np.exp(sc_)
        se = float(ex.sum())
        if se <= 0 or not math.isfinite(se):
            w = np.zeros(c.size); w[int(np.argmax(c))] = 1.0
            f = cm
        else:
            f = cm + T*math.log(se/c.size)
            w = ex / se
        
        # Cross-correlation for gradient
        Fp = rfft(pp[::-1], sz)
        Fw = rfft(w, sz)
        q = irfft(Fp*Fw, sz)[:clen]
        q2 = q[n-1:2*n-1].copy()
        np.maximum(q2, 0.0, out=q2)
        g2 = 2.0*q2
        grad = pp*(g2 - float(np.dot(pp, g2)))
        return f, np.where(np.isfinite(grad), grad, 0.0)
    
    if fine_only:
        stages = [(0.002, 0.3), (0.0006, 0.4), (0.00015, 0.3)]
    else:
        stages = [(0.08, 0.12), (0.025, 0.15), (0.007, 0.15),
                  (0.002, 0.18), (0.0005, 0.18), (0.00012, 0.16), (0.00003, 0.06)]
    
    t0 = time.monotonic()
    avail = max(0.0, stop_time - t0 - 5.0)
    if avail < 3: return p
    
    best_p = p.copy()
    best_sc = 2.0*n*float(cfast(p).max())
    tw = sum(wt for _, wt in stages)
    cum = 0.0
    
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3: break
        cum += wt
        end_st = min(stop_time - 2.0, t0 + avail*(cum/tw))
        if end_st <= now + 1: continue
        
        c = cfast(p)
        cm = float(c.max())
        if cm <= 0: break
        T = max(1e-300, frac*cm)
        
        f0, _ = lse_fn(u, T)
        bu = u.copy()
        bf = f0
        
        while time.monotonic() < end_st - 1.0:
            try:
                res = minimize(lse_fn, bu, args=(T,), method='L-BFGS-B',
                             jac=True, options={'maxiter':25, 'gtol':1e-12,
                                                'ftol':1e-15, 'maxls':12,
                                                'maxcor':20, 'disp':False})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    fv = float(res.fun)
                    if math.isfinite(fv) and fv < bf:
                        bf = fv; bu = res.x.copy()
                    if getattr(res, 'status', 1) == 0: break
                else: break
            except: break
        
        pc = _softmax(bu)
        pc = np.maximum(pc, 0)
        ss = float(pc.sum())
        if ss > 1e-300:
            pc /= ss
            sc = 2*n*float(cfast(pc).max())
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc; best_p = pc.copy(); p = pc.copy()
                if consider: consider(best_p)
            u = p2u(p)
    
    return best_p

        except: return
