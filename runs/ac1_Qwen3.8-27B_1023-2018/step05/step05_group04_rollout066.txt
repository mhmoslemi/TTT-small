def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = int(max(1, n))
    if src.size == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if src.size == n:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m, dtype=np.float64) + 0.5) / m
        fp = np.interp((np.arange(n, dtype=np.float64) + 0.5) / n, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300 or not math.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s

    if lse_end > time.monotonic() + 20.0:
        p_work = _resample(best_arr, n_work)
        try:
            p2 = _lse_optimize(p_work, lse_end, consider=consider)
            if p2 is not None and p2.size == n_work and np.all(np.isfinite(p2)) and p2.sum() > 0:
                p_work = p2 / float(p2.sum())
                consider(p_work)
        except Exception:
            pass

    # Phase 2: main coordinate descent on the best
    cd_end = max(time.monotonic() + 30.0, hard - 90.0)
    p_start = _resample(best_arr, n_work)
    try:
        p_cd, _ = _coord_descent(p_start, cd_end, consider=consider,
                                 seed=int(seed + 7777))
        if p_cd is not None and p_cd.size > 0 and np.all(np.isfinite(p_cd)) and p_cd.sum() > 0:
            consider(p_cd)
    except Exception:
        pass

    # Phase 3: secondary restarts from perturbed best if there is time left
    rng = np.random.default_rng(int(seed + 1234))
    restart_budget = hard - 60.0 - time.monotonic()
    while restart_budget > 90.0:
        cur = _resample(best_arr, n_work)
        kick = 0.04 + 0.04 * (1.0 - min(1.0, restart_budget / 400.0))
        cand = cur * np.exp(rng.normal(0.0, kick, n_work))
        cand = np.maximum(cand, 0.0)
        ss = float(cand.sum())
        if ss > 0:
            cand = cand / ss
        consider(cand)
        r_end = min(hard - 55.0, time.monotonic() + restart_budget * 0.7)
        r_end = max(r_end, time.monotonic() + 30.0)
        try:
            p_r, _ = _coord_descent(cand, r_end, consider=consider,
                                    seed=int(seed + 5555 + restart_budget))
            if p_r is not None and p_r.size > 0 and np.all(np.isfinite(p_r)) and p_r.sum() > 0:
                consider(p_r)
        except Exception:
            break
        restart_budget = hard - 60.0 - time.monotonic()

    # Phase 4: final polish on the overall best
    rem = hard - 12.0 - time.monotonic()
    if rem > 40.0:
        p_final = _resample(best_arr, n_work)
        try:
            p_f, _ = _coord_descent(p_final, hard - 10.0, consider=consider,
                                    seed=int(seed + 9999))
            if p_f is not None and p_f.size > 0 and np.all(np.isfinite(p_f)) and p_f.sum() > 0:
                consider(p_f)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
