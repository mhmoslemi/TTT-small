Now let me flesh out the LSE optimization. The key is the efficient gradient computation.

For the LSE objective with softmax parameterization:
- $a = \text{softmax}(u)$
- $c = a * a$ (convolution, computed via FFT)
- $F = \max(c) + T \log(\frac{1}{m} \sum_k \exp((c_k - \max(c))/T))$
- Gradient w.r.t. $u$: $a \odot (g - \langle g, a \rangle)$ where $g_j = 2 \sum_i w_{i+j} a_i$

The computation of $g$ is a "cross-correlation" of $a$ and $w$:
$g_j = 2 \sum_{i=0}^{n-1} a_i w_{i+j}$ for $j = 0, ..., n-1$

This is the cross-correlation $(a \star w)_j = \sum_i a_i w_{i+j}$, which can be computed as the convolution of $a$ with $w$ (reversed):
$(a \star w) = \text{convolve}(a[::-1], w)[n-1:n+n-1]$

Wait, let me be more careful. The cross-correlation is:
$(a \star w)(j) = \sum_i a(i) w(i+j)$

This equals the convolution $a \cdot w_{\text{rev}}$ evaluated at a specific index:
$\text{convolve}(a, w_{\text{rev}})(n-1+j) = \sum_i a(i) w_{\text{rev}}(n-1+j-i) = \sum_i a(i) w(n-1-(-1-n+j... ))$

Hmm, let me just use the definition directly. I'll compute it as:
$g_j = 2 \sum_{i=0}^{n-1} a_i w_{i+j}$

Note that $w$ has length $2n-1$ (same as the convolution). For $j = 0, ..., n-1$, we need $w_j, w_{j+1}, ..., w_{j+n-1}$, which are all valid indices (since $j+n-1 \leq 2n-2 = m-1$).

I can compute this as a convolution: define $a_{\text{rev}} = a[::-1]$ (length $n$). Then:
$\text{convolve}(a_{\text{rev}}, w)(k) = \sum_i a_{\text{rev}}(i) w(k-i) = \sum_i a(n-1-i) w(k-i)$

Let $j' = n-1-i$, so $i = n-1-j'$:
$= \sum_{j'} a(j') w(k-n+1+j')$

We want $g_j = 2 \sum_i a_i w_{i+j}$. Let $i = j'$:
$g_j = 2 \sum_{j'} a(j') w(j+j')$

Comparing with the convolution: $\text{convolve}(a_{\text{rev}}, w)(k) = \sum_{j'} a(j') w(k-n+1+j')$

So we need $k-n+1 = j$, i.e., $k = n-1+j$.

Therefore: $g_j = 2 \cdot \text{convolve}(a_{\text{rev}}, w)(n-1+j)$

And this can be computed via FFT!

$a_{\text{rev}} = (a_2, a_1, a_0)$.
$w = (w_0, w_1, w_2, w_3, w_4)$.

    cross_size = next_fast_len(cross_result_len)

    def lseObj(u, T):
        u_max = float(np.max(u))
        e = np.exp(u - u_max)
        e_sum = float(e.sum())
        if e_sum <= 0 or not math.isfinite(e_sum):
            return 1e100, np.zeros_like(u)
        a = e / e_sum

        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]

        c_max = float(c.max())
        if not math.isfinite(c_max):
            return 1e100, np.zeros_like(u)

        scaled = (c - c_max) / T
        exps = np.exp(scaled)
        exps_sum = float(exps.sum())
        if exps_sum <= 0 or not math.isfinite(exps_sum):
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
            obj = c_max
        else:
            w = exps / exps_sum
            obj = c_max + T * math.log(exps_sum / conv_len)

        # Gradient: g_j = 2 * conv(a_rev, w)[n-1 : 2n-1]
        a_rev = a[::-1]
        FA = rfft(a_rev, cross_size)
        FW = rfft(w, cross_size)
        conv_g = irfft(FA * FW, cross_size)[:cross_result_len]
        g = 2.0 * conv_g[n - 1:2 * n - 1]

        ag = float(np.dot(a, g))
        grad = a * (g - ag)
        grad = np.where(np.isfinite(grad), grad, 0.0)

        return obj, grad

    def coord_descent(a, deadline, seed_cd=12345):
        n_cd = a.size
        a = np.asarray(a, dtype=np.float64).copy()
        s = float(a.sum())
        a = a / s
        c = _fft_conv(a)
        cur_score = float(2.0 * n_cd * c.max())
        rng = np.random.default_rng(seed_cd)
        GR = (math.sqrt(5.0) - 1.0) / 2.0
        Ltot = 2 * n_cd - 1
        flat = 0
        sweep = 0
        max_sweeps = 5000

        while time.monotonic() < deadline - 2.0 and flat < 4 and sweep < max_sweeps:
            improved = False
            order = rng.permutation(n_cd)
            for i in order:
                if time.monotonic() > deadline - 1.0:
                    break
                v = float(a[i])
                start = i
                end = min(i + n_cd, Ltot)
                Ls = end - start
                if Ls <= 0:
                    continue

                base = c[start:end]
                infl = 2.0 * a[:Ls]
                diag_idx = i

                max_const = 0.0
                if start > 0:
                    mc = float(c[:start].max())
                    if mc > max_const:
                        max_const = mc
                if end < Ltot:
                    mc = float(c[end:].max())
                    if mc > max_const:
                        max_const = mc

                lo_w = -v
                hi_w = max(20.0 * v, 5.0 / n_cd)
                hi_w = min(hi_w, 0.5)
                hi_w = max(hi_w, 1e-10)

                def f1d(w):
                    aff = base + w * infl
                    if 0 <= diag_idx < Ls:
                        m_aff = max(float((aff[:diag_idx] + (w * w) * 0).max()) if diag_idx > 0 else 0.0,
                                    float(aff[diag_idx:].max()))
                        # Actually need to handle the diagonal properly
                        # Let me just compute directly
                        pass
                    # Simpler: compute max including diagonal
                    aff_max = float(aff.max())
                    if 0 <= diag_idx < Ls:
                        dval = base[diag_idx] + w * infl[diag_idx] + w * w
                        if dval > aff_max:
                            aff_max = dval
                    m = max(aff_max, max_const)
                    s_new = 1.0 + w
                    if s_new <= 1e-15:
                        return float('inf')
                    return 2.0 * n_cd * m / (s_new * s_new)

                G = 32
                pts = np.linspace(lo_w, hi_w, G + 1)
                if lo_w < 0 < hi_w:
                    ins = np.searchsorted(pts, 0.0)
                    pts = np.insert(pts, ins, 0.0)

                vals = np.array([f1d(float(p)) for p in pts])
                valid = np.isfinite(vals)
                if not np.any(valid):
                    continue
                imin = int(np.argmin(np.where(valid, vals, np.inf)))
                best_w = float(pts[imin])
                best_val = float(vals[imin])

                if not math.isfinite(best_val):
                    continue

                if 0 < imin < len(pts) - 1:
                    a_gs = float(pts[max(0, imin - 1)])
                    b_gs = float(pts[min(len(pts) - 1, imin + 1)])
                else:
                    a_gs, b_gs = lo_w, hi_w

                if b_gs - a_gs > 1e-14:
                    c_gs = b_gs - GR * (b_gs - a_gs)
                    d_gs = a_gs + GR * (b_gs - a_gs)
                    f_c = f1d(c_gs)
                    f_d = f1d(d_gs)
                    for _ in range(18):
                        if b_gs - a_gs < 1e-13 * max(1.0, abs(best_w)):
                            break
                        if f_c < f_d:
                            b_gs = d_gs
                            d_gs = c_gs
                            f_d = f_c
                            c_gs = b_gs - GR * (b_gs - a_gs)
                            f_c = f1d(c_gs)
                        else:
                            a_gs = c_gs
                            c_gs = d_gs
                            f_c = f_d
                            d_gs = a_gs + GR * (b_gs - a_gs)
                            f_d = f1d(d_gs)
                    mid = (a_gs + b_gs) / 2.0
                    f_mid = f1d(mid)
                    if f_mid < best_val:
                        best_w = mid
                        best_val = f_mid

                if math.isfinite(best_val) and best_val < cur_score - 1e-14 and best_w > lo_w + 1e-15:
                    a[i] += best_w
                    c[start:end] = base + best_w * infl
                    if 0 <= diag_idx < Ls:
                        c_tmp = c[start:end].copy()
                        c_tmp[diag_idx] += best_w * best_w
                        c[start:end] = c_tmp
                    new_max = max(float(c[start:end].max()), max_const)
                    cur_score = 2.0 * n_cd * new_max
                    improved = True

            s_new = float(a.sum())
            if s_new > 1e-300 and math.isfinite(s_new):
                a /= s_new
                c = _fft_conv(a)
                new_score = float(2.0 * n_cd * c.max())
                if math.isfinite(new_score) and new_score < cur_score:
                    cur_score = new_score
                consider(a)

            if improved:
                flat = 0
            else:
                flat += 1
                if flat == 1 and time.monotonic() < deadline - 40.0:
                    kick = 0.015 + 0.03 * min(1.0, flat / 15.0)
                    a = a * np.exp(rng.normal(0.0, kick, n_cd))
                    a = np.maximum(a, 0.0)
                    ss = float(a.sum())
                    if ss > 1e-300:
                        a /= ss
                    c = _fft_conv(a)
                    cur_score = float(2.0 * n_cd * c.max())
                    flat = 0
                elif flat >= 3:
                    break
            sweep += 1

        return a, cur_score

    # === Phase 1: LSE Optimization ===
    a = best_arr.copy()
    S = float(a.sum())
    a = a / S

    c0 = _fft_conv(a)
    c_range = max(float(c0.max() - c0.min()), 1e-6)

    u = np.log(np.maximum(a, 1e-15))
    u = u - float(u.mean())

    lse_budget = min(350.0, 0.30 * budget)
    lse_end = t0 + lse_budget

    temps = [0.25, 0.10, 0.04, 0.015, 0.006, 0.002, 0.0008, 0.0003]
    temps = [t * c_range for t in temps]

    n_temps = len(temps)
    for ti, T in enumerate(temps):
        if time.monotonic() > lse_end:
            break
        # Allocate time: more for later (lower T) stages
        frac = (ti + 1) / n_temps
        prev_frac = ti / n_temps
        stage_start = t0 + lse_budget * prev_frac
        stage_end = min(t0 + lse_budget * frac, lse_end - 2.0)
        if stage_end <= time.monotonic() + 5.0:
            continue

        maxiter = 150
        try:
            res = minimize(lseObj, u, args=(T,), method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter, 'gtol': 1e-10,
                                   'ftol': 1e-15, 'maxls': 30,
                                   'maxfun': maxiter * 30, 'maxdisp': 0})
            if res.x is not None and res.x.size == n and np.all(np.isfinite(res.x)):
                u = res.x.copy()
                a_new = np.exp(u - float(u.max()))
                a_new = a_new / float(a_new.sum())
                consider(a_new)
        except Exception:
            pass

        a = np.exp(u - float(u.max()))
        a = a / float(a.sum())

    # === Phase 2: Coordinate Descent ===
    cd_deadline = hard - 15.0
    if time.monotonic() < cd_deadline - 60.0:
        a_start = best_arr.copy()
        a_start = a_start / float(a_start.sum())
        try:
            a_cd, _ = coord_descent(a_start, cd_deadline, seed_cd=int(seed))
            consider(a_cd)
        except Exception:
            pass

    # === Phase 3: Try different initialization + CD ===
    rem = hard - 15.0 - time.monotonic()
    if rem > 150.0:
        # Tent-like blend
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        tent = np.maximum(0.0, 1.0 - 2.0 * np.abs(x - 0.5))
        ts = float(tent.sum())
        if ts > 0:
            tent /= ts
        blend = 0.3 * tent + 0.7 * (best_arr / float(best_arr.sum()))
        blend = blend / float(blend.sum())
        phase3_end = time.monotonic() + min(rem * 0.6, 300.0)
        try:
            a_cd2, _ = coord_descent(blend, phase3_end, seed_cd=int(seed + 777))
            consider(a_cd2)
        except Exception:
            pass

    # === Phase 4: Final LSE polish ===
    rem = hard - 15.0 - time.monotonic()
    if rem > 50.0:
        a_best = best_arr / float(best_arr.sum())
        u = np.log(np.maximum(a_best, 1e-15))
        u = u - float(u.mean())
        c_best = _fft_conv(a_best)
        c_rng = max(float(c_best.max() - c_best.min()), 1e-10)
        T_fin = 0.001 * c_rng
        final_end = hard - 20.0
        try:
            res = minimize(lseObj, u, args=(T_fin,), method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 80, 'gtol': 1e-12,
                                   'ftol': 1e-16, 'maxls': 20, 'maxdisp': 0})
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_fin = np.exp(res.x - float(res.x.max()))
                a_fin = a_fin / float(a_fin.sum())
                consider(a_fin)
        except Exception:
            pass

    # === Phase 5: Final CD squeeze ===
    rem = hard - 10.0 - time.monotonic()
    if rem > 40.0:
        a_fin = best_arr / float(best_arr.sum())
        try:
            a_s, _ = coord_descent(a_fin, hard - 12.0, seed_cd=int(seed + 4242))
            consider(a_s)
        except Exception:
            pass

    # Final validation with exact evaluator
    out = np.asarray(best_arr, dtype=np.float64).ravel()
    out = np.clip(out, 0.0, 1000.0)
    out = np.where(np.isfinite(out), out, 0.0)
    if out.size == 0 or float(out.sum()) < 0.01:
        out = parent.copy()
        out = np.clip(out, 0.0, 1000.0)
        out = np.where(np.isfinite(out), out, 0.0)

    result = [float(x) for x in out]
    final_sc = evaluate_sequence(result)
    if not math.isfinite(final_sc):
        result = [float(x) for x in parent]
        final_sc = evaluate_sequence(result)
        if not math.isfinite(final_sc):
            return [1.0, 0.0, 1.0]

    return result
