def _build_A(n, a):
    """Build (2n-1) x n matrix: A[k,m] = a[k-m] if 0<=k-m<n else 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)

            mode = pert_count % 5
            xx = np.arange(n, dtype=np.float64)
            if mode == 0:
                a = a * np.exp(rng.normal(0, sigma, n))
            elif mode == 1:
                freq = rng.uniform(1.0, 6.0)
                phase = rng.uniform(0, 2 * np.pi)
                a = a * (1.0 + sigma * np.sin(2.0 * np.pi * freq * xx / n + phase))
            elif mode == 2:
                a = a * (1.0 + sigma * rng.uniform(-1, 1, n))
            elif mode == 3:
                center = rng.integers(n // 4, 3 * n // 4)
                width = max(5, n // 8)
                mask = np.zeros(n)
                mask[max(0, center - width):min(n, center + width)] = 1.0
                a = a * (1.0 + sigma * rng.normal(0, 1, n) * mask)
            else:
                bsz = max(3, n // 12)
                i1 = rng.integers(0, max(1, n - 2 * bsz))
                i2 = rng.integers(0, max(1, n - 2 * bsz))
                if abs(int(i1) - int(i2)) > bsz:
                    tmp = a[int(i1):int(i1) + bsz].copy()
                    a[int(i1):int(i1) + bsz] = a[int(i2):int(i2) + bsz]
                    a[int(i2):int(i2) + bsz] = tmp
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck = 0
            pert_count += 1
            if pert_count > 120:
                break
            continue

        remaining = deadline - time.monotonic()
        if remaining < 1.0:
            break

        A = _build_A(n, a)
        moved = False
        tl = min(lp_time, max(0.5, remaining * 0.08))

        g = _solve_lp(A, M * (1 + 1e-9), n, time_limit=tl)

        if g is not None:
            g = g / g.sum()
            if float(np.max(np.abs(g - a))) > 1e-10:
                cand, val = _line_search(a, g, M)
                if cand is not None and val < M * (1.0 - 1e-11):
                    a = cand
                    moved = True

        if not moved and g is not None:
            g_avg = 0.5 * (a + g)
            g_avg = np.maximum(g_avg, 0)
            gs = g_avg.sum()
            if gs > 1e-14:
                g_avg = g_avg / gs
                cand, val = _line_search(a, g_avg, M)
                if cand is not None and val < M * (1.0 - 1e-11):
                    a = cand
                    moved = True

        if not moved:
            cand = _subgradient_step(a, c, M, n)
            if cand is not None:
                a = cand
                moved = True

        if not moved:
            a = a * np.exp(rng.normal(0, 0.008, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 12.0

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(281)
    parent = np.clip(parent, 0, 1000)
    if parent.size == 0 or not np.all(np.isfinite(parent)):
        parent = np.ones(281)
    n_parent = parent.size
    pn = parent / parent.sum()

    best_seq = parent.copy()
    best_sc = _score_full(best_seq)
    if not np.isfinite(best_sc):
        best_seq = np.ones(100)
        best_sc = _score_full(best_seq)

    def try_update(cand):
        nonlocal best_seq, best_sc
        try:
            cand = np.clip(np.asarray(cand, dtype=np.float64), 0, 1000)
            if cand.size == 0:
                return
            s = float(cand.sum())
            if s < 0.01:
                return
            sc = 2.0 * len(cand) * float(np.convolve(cand, cand).max()) / (s * s)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = cand.copy()
        except Exception:
            pass

    # Phase 1: Deep optimization on parent
    t_phase1 = t0 + budget * 0.52
    try:
        r = _optimize_single(pn, n_parent, t_phase1, seed=1, lp_time=5.0, max_stuck=100)
        try_update(r)
    except Exception:
        pass

    # Phase 2: Diverse initializations
    t_phase2 = t0 + budget * 0.78
    n_init = n_parent
    x = np.linspace(0, 1, n_init)
    inits = []
    for sigma in [0.08, 0.13, 0.2, 0.3, 0.45]:
        v = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(v / v.sum())
    for w in [0.15, 0.25, 0.35]:
        v = np.exp(-0.5 * ((x - w) / 0.08) ** 2) + np.exp(-0.5 * ((x - (1 - w)) / 0.08) ** 2)
        inits.append(v / v.sum())
    for m in [1, 2, 3]:
        v = np.maximum(np.cos(m * np.pi * (x - 0.5)) ** 2 + 0.005, 0.001)
        inits.append(v / v.sum())
    rng_bl = np.random.default_rng(4242)
    for frac in [0.35, 0.55, 0.75]:
        for init in inits[:3]:
            v = frac * pn + (1 - frac) * init
            inits.append(v / v.sum())
    for _ in range(5):
        v = np.abs(rng_bl.normal(1, 0.5, n_init)) + 0.01
        inits.append(v / v.sum())

    remaining2 = max(2.0, t_phase2 - time.monotonic())
    per_init = max(2.0, remaining2 / max(1, len(inits)))

    for i, init in enumerate(inits):
        if time.monotonic() > t_phase2 - 5.0:
            break
        alloc = min(time.monotonic() + per_init, t_phase2 - 3.0)
        try:
            r = _optimize_single(init, n_init, alloc, seed=500 + i, lp_time=2.5, max_stuck=60)
            try_update(r)
        except Exception:
            pass

    # Phase 3: Nearby lengths
    t_phase3 = t0 + budget * 0.94
    ab = np.clip(best_seq, 0, 1000).astype(np.float64)
    sb = ab.sum()
    ab_norm = ab / sb if sb > 1e-10 else pn

    for dn in [0, -10, 10, -20, 20, -40, 40, -60, 60]:
        if time.monotonic() > t_phase3 - 5.0:
            break
        nt = n_parent + dn
        if nt < 50 or nt > 500:
            continue
        x_old = np.linspace(0, 1, len(ab_norm))
        x_new = np.linspace(0, 1, nt)
        init_new = np.interp(x_new, x_old, ab_norm)
        init_new = np.clip(init_new, 0, None)
        sn = init_new.sum()
        if sn < 1e-10:
            continue
        init_new = init_new / sn
        remaining3 = t_phase3 - time.monotonic()
        per_len = max(3.0, remaining3 / 6.0)
        alloc = min(time.monotonic() + per_len, t_phase3 - 3.0)
        try:
            r = _optimize_single(init_new, nt, alloc, seed=3000 + nt, lp_time=3.0, max_stuck=80)
            try_update(r)
        except Exception:
            pass

    # Phase 4: Final polish
    if time.monotonic() < deadline - 8:
        af = np.clip(best_seq, 0, 1000).astype(np.float64)
        sf = af.sum()
        if sf > 1e-10:
            af = af / sf
        nf = len(af)
        try:
            r = _optimize_single(af, nf, deadline, seed=8888, lp_time=5.0, max_stuck=200)
            try_update(r)
        except Exception:
            pass

    # Final validation
    final = np.clip(best_seq, 0, 1000)
    final = np.where(np.isfinite(final), final, 0.0)
    if final.size == 0 or final.sum() < 0.01:
        final = np.ones(100)
    final = np.where(np.isfinite(final), final, 0.0)

    try:
        test = [float(v) for v in final]
        ev = evaluate_sequence(test)
        if not np.isfinite(ev):
            raise ValueError
    except Exception:
        final = np.ones(100)

    return [float(v) for v in final]
