```python
        T = max(1e-8, T_frac * c_max_init)
        stage_time = time_frac * (stop_time - start - 5.0) / total_time
        stage_end = time.monotonic() + stage_time
        
        while time.monotonic() < stage_end - 1.0:
            if time.monotonic() > stop_time - 10.0:
                break
            try:
                res = minimize(lse_obj, u, args=(T,), method='L-BFGS-B',
                              jac=True, options={'maxiter': 50, 'gtol': 1e-14,
                                                'ftol': 1e-18, 'maxls': 30,
                                                'maxcor': 20, 'disp': False})
            except Exception:
                break
            
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_cur = np.exp(res.x - np.max(res.x))
                a_sum = float(a_cur.sum())
                if a_sum > 0 and math.isfinite(a_sum):

                        best_sc = sc
                        best_a = a_cur.copy()
                        u = p_to_u(a_cur)
                        if consider:
                            try: consider(a_cur)
                            except: pass
            if time.monotonic() > stop_time - 10.0:
                break
    
    return best_a
```

Hmm wait, I realize the LSE function I defined uses `f = c_max + T * log(sum_exps / (2*n-1))`. The division by `(2*n-1)` is a normalization that doesn't affect the gradient (it just shifts the value by a constant). Actually it does affect the gradient in the sense that the LSE is an approximation to `c_max + T*log(average exp)`, which approximates `c_max + T*log(1 + (n-1)) = c_max + T*log(n)` as T→0... no, that's not right.

The standard LSE approximation to max is: `max(c) ≈ T * log(sum(exp(c_k / T)))` as T→0.
More precisely: `T * log(sum(exp(c_k/T))) = c_max + T * log(sum(exp((c_k-c_max)/T))) = c_max + T * log(N + sum_{k≠argmax}(exp((c_k-c_max)/T)))`
≈ `c_max + T * log(N)` as T→0 (since all other terms → 0).

So the LSE overestimates the max by `T * log(N)` where N is the number of terms. To make it converge to the max, we should use:
`f = T * log(sum(exp(c_k/T))) - T * log(N) = c_max + T * log(sum(exp((c_k-c_max)/T)) / N)`

This is exactly what I have: `f = c_max + T * log(sum_exps / (2*n-1))`.

As T→0, `sum_exps → 1` (only the max term survives), so `f → c_max + T*log(1/(2n-1)) → c_max`. Good.

Actually, let me think about this differently. The score as a function of `a[i] = v + δ` (with all other coordinates fixed) is:

`score(δ) = 2n * max_k (c[k] + δ*(a[k-i] + a[i]) + δ^2*[k==2i]) / (S + δ)^2`

where `S = sum(a)` (before the change) and the max is over all k.

For k in the affected range `[i, i+n-1]`:
`c_new[k] = c[k] + δ*(a[k-i] + a[i]) + δ^2*[k==2i]`

For k outside: `c_new[k] = c[k]`.

The max of `c_new` is: `max(max_affected(δ), max_const)` where `max_const = max of c outside the affected range`.

`max_affected(δ) = max_{k∈[i,i+n-1]} (c[k] + δ*(a[k-i]+a[i]) + δ^2*[k==2i])`

For a given δ, this is the max of a set of linear functions (plus one quadratic at k=2i). The max of linear functions is convex, and adding a convex quadratic keeps it convex. So `max_affected(δ)` is convex in δ.

The score is `2n * max(max_affected(δ), max_const) / (S+δ)^2`. The numerator is convex (max of convex functions) and the denominator is a convex function of δ (squared). The ratio of a convex function to a convex function is not necessarily convex, but in practice it's well-behaved.

For practical purposes, I'll use a grid of candidate values for δ, but make the grid adaptive (finer near the current value and near the optimum).

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 20.0

    # Get parent
    try:
        h = globals()["height_sequence_1"]
        a0 = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        a0 = np.ones(1024, dtype=np.float64)
    a0 = np.where(np.isfinite(a0), a0, 0.0)
    a0 = np.clip(a0, 0.0, 1000.0)
    s0 = float(a0.sum())
    if s0 < 1e-6:
        a0 = np.ones(1024, dtype=np.float64)
        s0 = 1024.0
    a = a0 / s0
    n = int(a.size)

    best_a = a.copy()
    best_score = float('inf')

    def consider(arr):
        nonlocal best_a, best_score
        try:
            x = np.asarray(arr, dtype=np.float64).ravel()
            x = np.where(np.isfinite(x), x, 0.0)
            x = np.clip(x, 0.0, 1000.0)
            if x.size == 0:
                return
            s = float(x.sum())
            if s < 0.01:
                return
            sc = float(evaluate_sequence([float(v) for v in x]))
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_a = x.copy()
        except Exception:
            pass

    consider(a)

    # FFT helpers
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1

    def conv_self(x):
        F = rfft(x, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return np.maximum(c, 0.0)

    def cross_corr_w(w, a_vec):
        # q[j] = sum_k w[k] * a_vec[k-j] for j=0..n-1
        a_flip = a_vec[::-1]
        F = rfft(w, cross_size)
        G = rfft(a_flip, cross_size)
        c = irfft(F * G, cross_size)
        return c[n - 1:2 * n - 1]

    def score_fft(a_vec):
        s = float(a_vec.sum())
        if s < 0.01:
            return math.inf
        c = conv_self(a_vec)
        m = float(c.max())
        return float(2.0 * n * m / (s * s))

    def p_to_u(p):
        ps = np.maximum(p, 1e-300)
        ll = np.log(ps)
        return ll - float(np.mean(ll))

    def u_to_p(u):
        us = u - np.max(u)
        e = np.exp(us)
        se = float(np.sum(e))
        if se <= 0 or not math.isfinite(se):
            return np.full(n, 1.0 / n)
        return e / se

    # Phase 1: LSE optimization
    lse_stop = t0 + min(600.0, 0.55 * budget)

    def lse_obj(u, T):
        a_cur = u_to_p(u)
        c = conv_self(a_cur)
        c_max = float(c.max())
        scaled = (c - c_max) / T
        exps = np.exp(scaled)
        sum_exps = float(np.sum(exps))
        if sum_exps <= 0 or not math.isfinite(sum_exps):
            f = c_max
            w = np.zeros(conv_len)
            w[int(np.argmax(c))] = 1.0
        else:
            f = c_max + T * math.log(sum_exps / conv_len)
            w = exps / sum_exps
        q = cross_corr_w(w, a_cur)
        q = 2.0 * q
        mu = float(np.dot(a_cur, q))
        grad = a_cur * (q - mu)
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)
        return float(f), grad

    # Multi-stage LSE
    c_init = conv_self(a)
    cmax_init = float(c_init.max())
    u = p_to_u(a)

    stages = [
        (0.08, 0.08), (0.03, 0.10), (0.01, 0.12),
        (0.004, 0.15), (0.0015, 0.18), (0.0005, 0.18),
        (0.00015, 0.15), (0.00004, 0.14),
    ]

    for idx, (T_frac, _) in enumerate(stages):
        if time.monotonic() > lse_stop - 5.0:
            break
        T = max(1e-9, T_frac * cmax_init)
        max_iter = 80 if idx < 3 else 150

        stagnant = 0
        while time.monotonic() < lse_stop - 5.0 and stagnant < 4:
            try:
                res = minimize(lse_obj, u, args=(T,), method='L-BFGS-B',
                               jac=True, options={'maxiter': max_iter, 'gtol': 1e-14,
                                                  'ftol': 1e-18, 'maxls': 30,
                                                  'maxcor': 20, 'disp': False})
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = u_to_p(res.x)
                sc_new = score_fft(a_new)
                if math.isfinite(sc_new) and sc_new < best_score - 1e-12:
                    best_score = sc_new
                    best_a = a_new.copy()
                    a = a_new.copy()
                    u = p_to_u(a)
                    stagnant = 0
                    consider(a)
                else:
                    stagnant += 1
            if time.monotonic() > lse_stop - 5.0:
                break

    # Phase 2: Coordinate descent
    cd_stop = hard_deadline - 60.0
    if time.monotonic() < cd_stop - 30.0:
        a_cd = best_a.copy()
        a_cd = np.maximum(a_cd, 0.0)
        ss = float(a_cd.sum())
        if ss > 1e-300:
            a_cd = a_cd / ss

        c_cd = conv_self(a_cd)
        S = 1.0  # sum after normalization
        cur_sc = float(2.0 * n * c_cd.max())
        rng = np.random.default_rng(int(seed) + 7)
        flat = 0
        sweep = 0
        G = max(24, min(60, int(120000 / max(n, 1))))

        while sweep < 5000 and time.monotonic() < cd_stop - 2.0:
            improved = False
            order = rng.permutation(n)
            for ii in order:
                if time.monotonic() > cd_stop - 1.0:
                    break
                i = int(ii)
                v = float(a_cd[i])
                start_k = i
                end_k = min(i + n, conv_len)
                Ls = end_k - start_k
                if Ls <= 0:
                    continue

                base = c_cd[start_k:end_k].copy()
                infl = 2.0 * a_cd[:Ls]

                max_const = 0.0
                if start_k > 0:
                    ml = float(np.max(c_cd[:start_k]))
                    max_const = max(max_const, ml)
                if end_k < conv_len:
                    mr = float(np.max(c_cd[end_k:]))
                    max_const = max(max_const, mr)

                # Candidate range
                x_hi = max(20.0 * v, 40.0 / n, 0.01)
                x_hi = min(x_hi, 4.0 * S)
                if x_hi <= 0:
                    x_hi = 1.0 / n

                pts = np.linspace(0.0, x_hi, G + 1)
                if 0.0 <= v <= x_hi:
                    pts = np.unique(np.concatenate([pts, [v]]))

                # Vectorized evaluation
                w_vals = pts - v
                denom = S + w_vals
                valid = denom > 1e-14
                if not np.any(valid):
                    continue

                aff = base[None, :] + w_vals[None, :] * infl[None, :]
                diag = i - start_k
                dvals = base[diag] + w_vals * infl[diag] + w_vals * w_vals
                aff = np.maximum(aff, dvals[:, None])
                aff = np.maximum(aff, max_const)
                mv = np.max(aff, axis=1)
                out = np.full(pts.shape, math.inf)
                dv = denom[valid]
                out[valid] = (2.0 * n * mv[valid]) / (dv * dv)

                if not np.any(np.isfinite(out)):
                    continue
                imin = int(np.argmin(out))
                bx = float(pts[imin])
                bv = float(out[imin])

                # Fine search
                dw = x_hi / G
                lo = max(0.0, bx - 3.0 * dw)
                hi = min(x_hi, bx + 3.0 * dw)
                if hi > lo:
                    fine = np.linspace(lo, hi, 32)
                    fine = np.unique(np.concatenate([fine, [bx, v]]))
                    w_f = fine - v
                    den_f = S + w_f
                    val_f = den_f > 1e-14
                    if np.any(val_f):
                        aff_f = base[None, :] + w_f[None, :] * infl[None, :]
                        dv_f = base[diag] + w_f * infl[diag] + w_f * w_f
                        aff_f = np.maximum(aff_f, dv_f[:, None])
                        aff_f = np.maximum(aff_f, max_const)
                        mv_f = np.max(aff_f, axis=1)
                        out_f = np.full(fine.shape, math.inf)
                        out_f[val_f] = (2.0 * n * mv_f[val_f]) / (den_f[val_f] ** 2)
                        if np.any(np.isfinite(out_f)):
                            im2 = int(np.argmin(out_f))
                            if float(out_f[im2]) < bv:
                                bv = float(out_f[im2])
                                bx = float(fine[im2])

                if math.isfinite(bv) and bv < cur_sc - 1e-13 and 0.0 <= bx <= 10.0 * S:
                    w_new = bx - v
                    if abs(w_new) > 1e-15:
                        a_cd[i] = bx
                        S += w_new
                        c_cd[start_k:end_k] += w_new * infl
                        c_cd[start_k + diag] += w_new * w_new
                        cur_sc = bv
                        improved = True

            # Rescale
            if S > 1e-300 and math.isfinite(S):
                inv = 1.0 / S
                a_cd *= inv
                c_cd *= inv * inv
                S = 1.0

            # Verify periodically
            if sweep % 20 == 0:
                c_check = conv_self(a_cd)
                sc_check = float(2.0 * n * c_check.max())
                if math.isfinite(sc_check) and sc_check < best_score - 1e-14:
                    best_score = sc_check
                    best_a = a_cd.copy()
                    consider(a_cd)

            if improved:
                flat = 0
            else:
                flat += 1
                if flat >= 5:
                    if time.monotonic() < cd_stop - 30.0:
                        a_cd = a_cd * np.exp(rng.normal(0, 0.03, n))
                        a_cd = np.maximum(a_cd, 0)
                        ss2 = float(a_cd.sum())
                        if ss2 > 0:
                            a_cd /= ss2
                        c_cd = conv_self(a_cd)
                        S = 1.0
                        cur_sc = float(2.0 * n * c_cd.max())
                        flat = 0
                    else:
                        break
            sweep += 1

        # Final check from CD
        a_cd = np.maximum(a_cd, 0.0)
        ss3 = float(a_cd.sum())
        if ss3 > 1e-300:
            a_cd /= ss3
        sc_final_cd = score_fft(a_cd)
        if math.isfinite(sc_final_cd):
            consider(a_cd)

    # Phase 3: Final LSE polish
    rem = hard_deadline - 15.0 - time.monotonic()
    if rem > 30.0:
        u3 = p_to_u(best_a)
        T3 = max(1e-10, 0.0001 * cmax_init)
        final_lse_stop = time.monotonic() + max(15.0, 0.6 * rem)
        while time.monotonic() < final_lse_stop - 2.0:
            try:
                res = minimize(lse_obj, u3, args=(T3,), method='L-BFGS-B',
                               jac=True, options={'maxiter': 60, 'gtol': 1e-15,
                                                  'ftol': 1e-19, 'maxls': 20,
                                                  'maxcor': 15, 'disp': False})
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = u_to_p(res.x)
                sc_new = score_fft(a_new)
                if math.isfinite(sc_new) and sc_new < best_score - 1e-12:
                    best_score = sc_new
                    best_a = a_new.copy()
                    u3 = p_to_u(a_new)
                    consider(a_new)
                else:
                    break
            if time.monotonic() > final_lse_stop - 2.0:
                break

    # Final validation
    try:
        out = np.asarray(best_a, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            # Fallback
            a0 = np.where(np.isfinite(a0), a0, 0.0)
            a0 = np.clip(a0, 0.0, 1000.0)
            s0 = float(a0.sum())
            if s0 < 0.01:
                return [1.0]
            ls = [float(x) for x in a0]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
```
