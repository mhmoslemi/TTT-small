                    c_gs = b_gs - gr * (b_gs - a_gs)
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    f_c = eval_local(c_gs - v, base, infl, diag_off, Ls, max_const)
                    f_d = eval_local(d_gs - v, base, infl, diag_off, Ls, max_const)
                    for _ in range(22):
                        if hi_gs - lo_gs < 1e-15:
                            break
                        if f_c < f_d:
                            b_gs = d_gs
                            d_gs = c_gs
                            f_d = f_c
                            c_gs = b_gs - gr * (b_gs - a_gs)
                            f_c = eval_local(c_gs - v, base, infl, diag_off, Ls, max_const)
                        else:
                            a_gs = c_gs
                            c_gs = d_gs
                            f_c = f_d
                            d_gs = a_gs + gr * (b_gs - a_gs)
                            f_d = eval_local(d_gs - v, base, infl, diag_off, Ls, max_const)
                    x_opt = (a_gs + b_gs) * 0.5
                    f_opt = eval_local(x_opt - v, base, infl, diag_off, Ls, max_const)
                    if f_opt < best_val:
                        best_x = x_opt
                        best_val = f_opt

            # Apply update
            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and best_x >= 0.0:
                w = best_x - v
                if abs(w) > 1e-17:
                    old_v = v
                    p[i] = best_x
                    S += w

                    # c[j] += 2*w*a_old[j-i] + w^2*delta(j,2i)
                    a_old = p[0:Ls].copy()
                    if i < Ls:
                        a_old[i] = old_v
                    c[start:end] += 2.0 * w * a_old
                    if 0 <= diag_off < Ls:
                        c[start + diag_off] += w * w

                    cur_score = best_val
                    improved = True

        # End of sweep
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 6 and time.monotonic() < hard_stop - 30.0:
                # Perturbation to escape local min
                sigma = 0.02 + 0.008 * min(flat, 20)
                mask = rng.random(n) < 0.15
                p[mask] = np.maximum(p[mask] * np.exp(rng.standard_normal(mask.sum()) * sigma), 0.0)
                ss = float(p.sum())
                if ss < 1e-300:
                    p[:] = 1.0 / n
                    ss = 1.0
                p /= ss
                S = 1.0
                c = _fft_conv_self(p)
                c = np.maximum(c, 0.0)
                cur_score = float(2.0 * n * c.max())
                if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
                    best_p = p.copy()
                    best_score = cur_score
                    last_improve = sweep
                flat = 0

        # Periodic full recompute
        if sweep % 8 == 0:
            c = _fft_conv_self(p)
            c = np.maximum(c, 0.0)
            new_score = float(2.0 * n * c.max())
            cur_score = new_score
            if math.isfinite(new_score) and new_score < best_score - 1e-13:
                best_p = p.copy()
                best_score = new_score
                last_improve = sweep

        # Track improvement
        if math.isfinite(cur_score) and cur_score < best_score - 1e-13:
            best_p = p.copy()
            best_score = cur_score
            last_improve = sweep

        # Early convergence check
        if sweep > 10 and sweep - last_improve > 50:
            # Stronger perturbation
            mask = rng.random(n) < 0.25
            p[mask] = p[mask] * (1.0 + 0.1 * rng.standard_normal(mask.sum()))
            p = np.maximum(p, 0.0)
            ss = float(p.sum())
            if ss > 1e-300:
                p /= ss
                S = 1.0
                c = _fft_conv_self(p)
                c = np.maximum(c, 0.0)
                cur_score = float(2.0 * n * c.max())
            last_improve = sweep

    # Final: verify and return best
    out = np.asarray(best_p, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    if out.size == 0 or float(out.sum()) < 0.01:
        return [1.0]

    # Normalize to ensure sum >= 0.01
    ss = float(out.sum())
    if ss < 0.01:
        out = out / ss * 0.01

    ls = [float(x) for x in out]

    # Validate with exact evaluator
    try:
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            # Fallback to parent
            try:
                pp = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
                pp = np.clip(pp, 0.0, 1000.0)
                pp = np.where(np.isfinite(pp), pp, 0.0)
                ls = [float(x) for x in pp]
                final_sc = float(evaluate_sequence(ls))
            except Exception:
                return [1.0]
            if not math.isfinite(final_sc):
                return [1.0]
    except Exception:
        return [1.0]

    return ls
