The optimal solution will have sum(g) = S >= 1 (since g = a is feasible: (a*a)_k <= M = max(a*a), and sum(a) = 1, so S >= 1).

If S = 1, then g = a is the unique optimum (or one of them), meaning a is a "fixed point" of the LP. At this point, the LP can't improve, and we need a different strategy.

import time
from scipy.optimize import linprog

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + budget_s - 10.0
    
    # Get parent
    try:
        p = np.asarray(globals()['height_sequence_1'], dtype=np.float64).ravel()
    except:
        p = np.ones(200)
    p = np.clip(p, 0, 1000)
    if p.size < 10 or not np.all(np.isfinite(p)) or p.sum() < 0.01:
        p = np.ones(200)
    
    def eval_score(a):
        a = np.clip(np.asarray(a, float), 0, 1000)
        if a.size == 0 or a.sum() < 0.01:
            return float('inf')
        c = np.convolve(a, a)
        return 2.0 * a.size * float(c.max()) / (a.sum()**2)
    
    best_a = p / p.sum()
    best_score = eval_score(best_a)
    
    def build_A(a):
        n = len(a)
        A = np.zeros((2*n-1, n))
        for m in range(n):
            klo = m
            khi = min(2*n-1, m+n)
            A[klo:khi, m] = a[:khi-klo]
        return A
    
    def solve_lp(A, M):
        n = A.shape[1]
        try:
            res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(A.shape[0], M),
                         bounds=[(0, None)]*n, method='highs',
                         options={'time_limit': 5, 'presolve': True})
            if res.status == 0 and res.x is not None:
                g = np.clip(res.x, 0, None)
                if g.sum() > 1e-15:
                    return g
        except:
            pass
        return None
    
    def line_search(a, g, M, n):
        best_t, best_M = 0.0, M
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            c = (1-t)*a + t*g
            sc = c.sum()
            if sc < 1e-15: continue
            c /= sc
            M_new = float(np.convolve(c, c).max())
            if M_new < best_M:
                best_M, best_t = M_new, t
        if best_t == 0:
            return None, M
        # Refine
        lo, hi = max(0, best_t-0.1), min(1, best_t+0.1)
        for _ in range(40):
            mid = (lo+hi)/2
            c = (1-mid)*a + mid*g
            sc = c.sum()
            if sc < 1e-15: lo = mid; continue
            c /= sc
            M_new = float(np.convolve(c, c).max())
            if M_new < best_M:
                best_M, best_t = M_new, mid
                hi = mid
            else:
                lo = mid
            if hi-lo < 1e-10: break
        c = (1-best_t)*a + best_t*g
        c = np.maximum(c, 0)
        c /= c.sum()
        return c, best_M

            M = float(c.max())

                local_best_M = M
                local_best = a.copy()
                stuck = 0
            else:
                stuck += 1
            
            if stuck > 100:

                a *= np.exp(rng.normal(0, sigma, n))

                a /= a.sum()

            A = build_A(a)

    rng = np.random.default_rng(seed)

        inits.append(v/v.sum())

        inits.append(v/v.sum())
    
    t2 = t0 + min(500, budget_s * 0.40)
    per_init = max(5.0, (t2 - time.monotonic()) / max(1, len(inits)))
    
    for i, init in enumerate(inits):

            break
        alloc = min(time.monotonic() + per_init, t2 - 3)

        except:
            pass
