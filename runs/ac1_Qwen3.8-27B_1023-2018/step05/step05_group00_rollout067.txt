        try:
            p3 = _lse_optimize(work, p3_end, consider=consider, fine_only=True)
            if p3 is not None and p3.size == n_par and np.all(np.isfinite(p3)) and p3.sum() > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # Phase 4: Final coordinate descent
    p4_end = min(hard - 20.0, t0 + budget - 15.0)
    if p4_end > time.monotonic() + 15.0:
        try:
            work, _ = _coord_descent(work, p4_end, max_sweeps=500000,
                                     seed=int(seed + 123456), consider=consider)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0, dtype=np.float64)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        try:
            ls = [float(x) for x in parent]
            if float(evaluate_sequence(ls)) < math.inf:
                return ls
        except Exception:
            pass
        return [1.0]
