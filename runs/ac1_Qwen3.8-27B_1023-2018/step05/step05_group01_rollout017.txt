        (0.0000007, 0.11),
    )

    start = time.monotonic()
    avail = max(0.0, float(stop_time) - start - 4.0)
    if avail <= 5.0:
        return p

    best_p = p.copy()
    best_sc = _score_fft(p)
    total_w = float(sum(w for _, w in stages))
    rng = np.random.default_rng(seed)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 2.5:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 2.0, start + avail * (cum / total_w))
        if end_stage <= now + 1.0:
            continue
        cc = cfast(p)
        cmax = float(cc.max()) if cc.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)

        best_u = u.copy()
        f0, _ = lse_func(best_u, T)
        if not math.isfinite(f0):
            continue
        best_f = f0
        stale = 0

        while time.monotonic() < end_stage - 0.3:
            opts = {"maxiter": 30, "gtol": 1e-12, "ftol": 1e-16,
                    "maxls": 20, "maxcor": 25, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-16:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            best_u = best_u + rng.normal(0, 0.003, best_u.size)
                            continue
            if getattr(res, "status", 1) == 0 and stale < 3:
                if time.monotonic() < end_stage - 0.8:
                    best_u = best_u + rng.normal(0, 0.002, best_u.size)
                    f_chk, _ = lse_func(best_u, T)
                    if math.isfinite(f_chk) and f_chk < best_f:
                        best_f = f_chk
                        stale += 1
                    else:
                        stale += 1
                else:
                    break

        p_cur = _softmax_fn(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_fft(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_sc = sc_cur
            best_p = p_cur.copy()
            p = p_cur.copy()
        elif math.isfinite(sc_cur) and sc_cur < best_sc + 0.003:
            p = p_cur.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass

    return best_p


def _coord_polish(p, stop_time, seed=12345, consider=None):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p, math.inf
    s = float(p.sum())
    if s < 1e-300:
        return p, math.inf
    p = p / s

    if n <= 500:
        G = 56
    elif n <= 4096:
        G = 48
    else:
        G = 36
    G_local = G * 2

    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    sweep = 0
    flat = 0
    S_total = 1.0

    while time.monotonic() < stop_time - 0.5:
        improved = False
        order = rng.permutation(n)

        for idx_i in order:
            if time.monotonic() > stop_time - 0.2:
                break
            if S_total <= 1e-300:
                break
            if S_total > 1.5 or S_total < 0.67:
                inv = 1.0 / S_total
                p *= inv
                c *= inv * inv
                S_total = 1.0
            S = float(S_total)
            i = int(idx_i)
            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue
            base = c[start:end].copy()
            infl = 2.0 * p[:Ls]

            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                if ml > max_const:
                    max_const = ml
            if end < Ltot:
                mr = float(c[end:].max())
                if mr > max_const:
                    max_const = mr

            x_hi = max(10.0 * v, 20.0 * S / float(n), 0.01 * S, 1e-15)
            x_hi = min(x_hi, 6.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = max(1e-8, S / float(n))

            pts = np.linspace(0.0, x_hi, G + 1, dtype=np.float64)

            def eval_xs(xs):
                xs = np.asarray(xs, dtype=np.float64).ravel()
                w = xs - v
                denom = S + w
                vals = np.full(xs.shape, math.inf, dtype=np.float64)
                valid = (denom > 1e-15) & (xs >= -1e-12)
                if not np.any(valid):
                    return vals
                aff = base[None, :] + w[valid, None] * infl[None, :]
                dval = base[i] + w[valid] * infl[i] + w[valid] * w[valid]
                aff[valid, i] = np.maximum(aff[valid, i], dval)
                aff = np.maximum(aff, max_const)
                mv = aff.max(axis=1)
                dv = denom[valid]
                vals[valid] = (2.0 * n * mv) / (dv * dv)
                return vals

            vals = eval_xs(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

            # Local refinement
            dw = x_hi / float(G)
            lo = max(0.0, best_x - 2.0 * dw)
            hi = min(x_hi, best_x + 2.0 * dw)
            if hi > lo:
                fine = np.linspace(lo, hi, G_local + 1, dtype=np.float64)
                if 0.0 <= best_x <= x_hi:
                    fine = np.concatenate((fine, [best_x]))
                    fine = np.unique(fine)
                vals2 = eval_xs(fine)
                if np.any(np.isfinite(vals2)):
                    imin2 = int(np.argmin(vals2))
                    vx2 = float(fine[imin2])
                    vv2 = float(vals2[imin2])
                    if vv2 < best_val - 1e-15:
                        best_x = vx2
                        best_val = vv2

            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and 0.0 <= best_x <= 8.0 * S:
                w = best_x - v
                if abs(w) > 1e-15:
                    p[i] = best_x
                    S_total += w
                    c[start:end] += w * infl
                    c[start + i] += w * w
                    cur_score = best_val
                    improved = True

        S = float(S_total)
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
        else:
            break
        S_total = 1.0
        c = _conv_self(p)
        new_score = 2.0 * n * float(c.max())
        if math.isfinite(new_score) and new_score < best_score - 1e-15:
            best_score = new_score
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 4.0:
                    sigma = 0.015 + 0.025 * min(flat, 12) / 12.0
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 1e-300:
                        p /= ss
                        c = _conv_self(p)
                        S_total = 1.0
                        cur_score = 2.0 * n * float(c.max())
                        if math.isfinite(cur_score) and cur_score < best_score - 1e-14:
                            best_score = cur_score
                            best_p = p.copy()
                        flat = 0
                    else:
                        break
                else:
                    break
        sweep += 1

    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        sc = float(_score_fft(p))
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _subgradient_step(p, stop_time, n_steps=50, lr_decay=0.999, seed=999, consider=None):
    """Projected subgradient on the exact max."""
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    c = _conv_self(p)
    best_score = 2.0 * n * float(c.max())
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    lr = 0.01 / float(n)

    for step in range(n_steps):
        if time.monotonic() > stop_time - 1.0:
            break
        c_max = float(c.max())
        tol = 1e-10 * c_max
        active = np.where(c >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if i == j:
                    g[i] += 2.0 * p[i]
                else:
                    g[i] += p[j]

        g /= float(len(active))
        g = g - float(np.sum(g * p))  # center for normalization constraint

        p_new = p - lr * g
        p_new = np.maximum(p_new, 0.0)
        ss = float(p_new.sum())
        if ss > 1e-300:
            p_new /= ss
        else:
            break

        c = _conv_self(p_new)
        sc = 2.0 * n * float(c.max())
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_score = sc
            best_p = p_new.copy()
            if consider:
                try:
                    consider(best_p)
                except Exception:
                    pass
        p = p_new
        lr *= lr_decay

    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 20.0
    if hard <= t0 + 2.0:
        hard = t0 + max(2.0, budget * 0.9)
    final_reserve = max(8.0, 0.05 * (hard - t0))
    phase_limit = hard - final_reserve

    parent = _get_parent()
    best_arr = parent.copy()
    ss = float(best_arr.sum())
    if ss > 1e-300:
        best_arr = best_arr / ss
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(64) / 64.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            a = a / s
            best_arr = a.copy()
            best_score = sc

    n_parent = int(best_arr.size)

    # Phase 1: LSE refinement at parent length
    lse_end = min(phase_limit, t0 + 0.30 * (phase_limit - t0))
    if lse_end > t0 + 15.0:
        try:
            p1 = _lse_optimize(best_arr, lse_end, seed=int(seed + 100), consider=consider)
            if p1 is not None and p1.size == n_parent and float(p1.sum()) > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    # Phase 2: Coordinate descent polish (main workhorse)
    rem = phase_limit - time.monotonic()
    if rem > 20.0:
        cd_end = min(phase_limit, time.monotonic() + 0.55 * rem)
        if cd_end > time.monotonic() + 10.0:
            try:
                p2, _ = _coord_polish(best_arr, cd_end, seed=int(seed + 200), consider=consider)
                if p2 is not None and p2.size == best_arr.size and float(p2.sum()) > 0:
                    p2 = p2 / float(p2.sum())
                    consider(p2)
            except Exception:
                pass

    # Phase 3: Perturb + re-optimize
    if time.monotonic() < phase_limit - 30.0:
        rng2 = np.random.default_rng(seed + 777)
        for pr in range(3):
            if time.monotonic() > phase_limit - 20.0:
                break
            sigma = 0.03 + 0.02 * pr
            pert = best_arr * np.exp(rng2.normal(0, sigma, best_arr.size))
            pert = np.maximum(pert, 0.0)
            ps = float(pert.sum())
            if ps > 0:
                pert /= ps
            sub_end = min(phase_limit - 15.0, time.monotonic() + 0.35 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 15.0:
                # Short LSE + coord
                lse_sub = min(sub_end - 10.0, time.monotonic() + 0.4 * (sub_end - time.monotonic()))
                if lse_sub > time.monotonic() + 8.0:
                    try:
                        pp = _lse_optimize(pert, lse_sub, seed=int(seed + 3000 + pr * 100), consider=consider)
                        if pp is not None and pp.size == best_arr.size and float(pp.sum()) > 0:
                            pp = pp / float(pp.sum())
                            consider(pp)
                    except Exception:
                        pass
                cd_sub = min(sub_end - 1.0, time.monotonic() + (sub_end - time.monotonic()) - 2.0)
                if cd_sub > time.monotonic() + 10.0:
                    try:
                        pp2, _ = _coord_polish(best_arr, cd_sub, seed=int(seed + 4000 + pr * 200), consider=consider)
                        if pp2 is not None and pp2.size == best_arr.size and float(pp2.sum()) > 0:
                            pp2 = pp2 / float(pp2.sum())
                            consider(pp2)
                    except Exception:
                        pass

    # Phase 4: Try different length
    if time.monotonic() < phase_limit - 60.0:
        n_alt = 4096 if n_parent <= 4096 else 2048
        if n_alt != n_parent:
            alt_init = _resample_mid(best_arr, n_alt)
            sub_end = min(phase_limit - 15.0, time.monotonic() + 0.25 * (phase_limit - time.monotonic()))
            if sub_end > time.monotonic() + 30.0:
                lse_a = min(sub_end - 15.0, time.monotonic() + 0.45 * (sub_end - time.monotonic()))
                if lse_a > time.monotonic() + 10.0:
                    try:
                        pp = _lse_optimize(alt_init, lse_a, seed=int(seed + 8000), consider=consider)
                        if pp is not None and pp.size == n_alt and float(pp.sum()) > 0:
                            pp = pp / float(pp.sum())
                            consider(pp)
                    except Exception:
                        pass
                cd_a = min(sub_end - 1.0, time.monotonic() + (sub_end - time.monotonic()) - 2.0)
                if cd_a > time.monotonic() + 10.0:
                    try:
                        pp2, _ = _coord_polish(pp if pp is not None and float(pp.sum()) > 0 else alt_init,
                                               cd_a, seed=int(seed + 9000), consider=consider)
                        if pp2 is not None and float(pp2.sum()) > 0:
                            pp2 = pp2 / float(pp2.sum())
                            consider(pp2)
                    except Exception:
                        pass

    # Phase 5: Final polish on the best
    final_end = min(hard - 15.0, t0 + budget - 8.0)
    if final_end > time.monotonic() + 20.0 and best_arr.size > 0:
        nbest = int(best_arr.size)
        rem = final_end - time.monotonic()

        # Subgradient pass
        if rem > 60.0 and nbest <= 5000:
            sg_end = min(final_end - 40.0, time.monotonic() + 0.25 * rem)
            if sg_end > time.monotonic() + 10.0:
                try:
                    pp = _subgradient_step(best_arr, sg_end, n_steps=200, seed=int(seed + 99999), consider=consider)
                    if pp is not None and pp.size == nbest and float(pp.sum()) > 0:
                        pp = pp / float(pp.sum())
                        consider(pp)
                except Exception:
                    pass

        # Final LSE squeeze
        if time.monotonic() < final_end - 40.0:
            lstop = min(final_end - 35.0, time.monotonic() + 0.3 * (final_end - time.monotonic()))
            if lstop > time.monotonic() + 10.0:
                try:
                    pp = _lse_optimize(best_arr, lstop, consider=consider, seed=int(seed + 999999))
                    if pp is not None and pp.size == nbest and float(pp.sum()) > 0:
                        pp = pp / float(pp.sum())
                        consider(pp)
                except Exception:
                    pass

        # Final coordinate polish
        cstop = min(final_end - 3.0, time.monotonic() + max(15.0, 0.5 * (final_end - time.monotonic())))
        if cstop > time.monotonic() + 8.0:
            try:
                pp2, _ = _coord_polish(best_arr, cstop, seed=int(seed + 123456789), consider=consider)
                if pp2 is not None and pp2.size == nbest and float(pp2.sum()) > 0:
                    pp2 = pp2 / float(pp2.sum())
                    consider(pp2)
            except Exception:
                pass

    # Final validation with exact evaluator
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
