def _optimize_lp(a0, deadline, seed=42):

                    a = cand
                    moved = True

            grad = np.zeros(n)
            for k in active:
                for j in range(max(0, k-n+1), min(n, k+1)):

    return p


def _conv_mat(n, a):
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[d[mask]]
    return A


def _lp_dir(a, M, n, tl=5.0):
    A = _conv_mat(n, a)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': tl,
                               'dual_feasibility_tolerance': 1e-13,
                               'primal_feasibility_tolerance': 1e-13})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                dual = None
                try:
                    dual = res.ineqlin.marginals
                except Exception:
                    pass
                return g / s, dual
    except Exception:
        pass
    return None, None


def _linesearch(a, g):
    n = a.size
    best_t, best_v = 0.0, float(np.convolve(a, a).max())
    for t in np.linspace(0.05, 1.0, 16):
        h = (1 - t) * a + t * g
        h = np.maximum(h, 0)
        s = h.sum()
        if s < 1e-15:
            continue
        h /= s
        v = float(np.convolve(h, h).max())
        if v < best_v:
            best_v, best_t = v, t
    if best_t > 0:
        lo, hi = max(0.0, best_t - 0.06), min(1.0, best_t + 0.06)
        for _ in range(40):
            mid = 0.5 * (lo + hi)
            h = (1 - mid) * a + mid * g
            h = np.maximum(h, 0)
            s = h.sum()
            if s < 1e-15:
                lo = mid
                continue
            h /= s
            v = float(np.convolve(h, h).max())
            if v < best_v:
                best_v, hi = v, mid
            else:
                lo = mid
        best_t = 0.5 * (lo + hi)
    h = (1 - best_t) * a + best_t * g
    h = np.maximum(h, 0)
    s = h.sum()
    if s < 1e-15:
        return None, best_v
    return h / s, best_v


def _subgrad_step(a, conv, M, n, dual=None):
    active = np.where(conv >= M * 0.9995)[0]
    if len(active) == 0:
        return None
    if dual is not None and np.any(dual > 1e-14):
        w = np.zeros(len(active))
        for i, k in enumerate(active):
            w[i] = max(dual[k], 0)
        tw = w.sum()
        if tw < 1e-14:
            w = np.ones(len(active)) / len(active)
        else:
            w /= tw
    else:
        w = np.ones(len(active)) / len(active)
    grad = np.zeros(n)
    for i, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += w[i] * 2.0 * a[idx]
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-14:
        return None
    grad /= gn
    best_cand, best_v = None, M
    for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
        cand = a - alpha * grad
        cand = np.maximum(cand, 0)
        cs = cand.sum()
        if cs < 1e-12:
            continue
        cand /= cs
        v = float(np.convolve(cand, cand).max())
        if v < best_v:
            best_v = v
            best_cand = cand
    return best_cand


def _coord_descent(a, M, n, deadline, max_sweeps=3):
    n = a.size
    improved = True
    sweeps = 0
    while improved and sweeps < max_sweeps and time.monotonic() < deadline:
        improved = False
        sweeps += 1
        conv = np.convolve(a, a)
        M = float(conv.max())
        for j in range(n):
            if time.monotonic() > deadline:
                return a
            for frac in [0.05, 0.1, 0.2, -0.05, -0.1, -0.2]:
                delta = frac * a[j]
                cand = a.copy()
                cand[j] += delta
                cand = np.maximum(cand, 0)
                cs = cand.sum()
                if cs < 1e-12:
                    continue
                cand /= cs
                v = float(np.convolve(cand, cand).max())
                if v < M - 1e-14:
                    a = cand
                    M = v
                    improved = True
                    break
    return a


def _optimize(a0, deadline, seed=42):
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(a.size)
        s = 1.0
    a /= a.sum()
    n = a.size
    best_a = a.copy()
    best_v = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0
    coord_count = 0

    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        M = float(conv.max())

        if M < best_v - 1e-14:
            best_v = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 30:
            sigma = min(0.015 + 0.008 * (stuck - 30), 0.3)
            xx = np.arange(n, dtype=np.float64)
            freq = rng.uniform(0.5, 6.0)
            phase = rng.uniform(0, 2 * np.pi)
            a *= (1.0 + sigma * (rng.normal(0, 1, n) +
                                  0.4 * np.sin(2 * np.pi * freq * xx / n + phase) +
                                  0.3 * np.cos(2 * np.pi * (freq * 0.7) * xx / n + phase * 1.3)))
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            stuck = 0
            if sigma > 0.28:
                break
            continue

        g, dual = _lp_dir(a, M * (1 + 1e-12), n, tl=5.0)
        moved = False

        if g is not None:
            diff = float(np.max(np.abs(g - a)))
            if diff > 1e-11:
                cand, val = _linesearch(a, g)
                if cand is not None and val < M - 1e-14:
                    a = cand
                    moved = True

        if not moved:
            g2, dual2 = _lp_dir(a, M * (1 + 1e-5), n, tl=3.0)
            if g2 is not None:
                cand, val = _linesearch(a, g2)
                if cand is not None and val < M - 1e-14:
                    a = cand
                    moved = True
            if dual2 is not None:
                cand_sc = _subgrad_step(a, conv, M, n, dual2)
                if cand_sc is not None:
                    a = cand_sc
                    moved = True

        if not moved:
            cand_sc = _subgrad_step(a, conv, M, n, dual)
            if cand_sc is not None:
                a = cand_sc
                moved = True

        if not moved:
            coord_count += 1
            if coord_count % 15 == 0 and time.monotonic() < deadline - 5:
                a = _coord_descent(a, M, n, deadline - 3, max_sweeps=1)
                cv = float(np.convolve(a, a).max())
                if cv < best_v - 1e-14:
                    best_v = cv
                    best_a = a.copy()
                continue

        if not moved:
            a *= np.exp(rng.normal(0, 0.008, n))
            a = np.maximum(a, 1e-12)
            a /= a.sum()

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 20.0

    parent = _get_parent()
    best_seq = parent.copy()
    best_score = _score(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(100)
        best_score = _score(best_seq)

    def try_update(cand):
        nonlocal best_seq, best_score
        try:
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.clip(np.asarray(cand, dtype=np.float64), 0, 1000).copy()
        except Exception:
            pass

    n = parent.size

    # Phase 1: Deep LP optimization of parent (45% of budget)
    t1 = t0 + budget * 0.45
    try:
        try_update(_optimize(parent, t1, seed=1))
    except Exception:
        pass

    # Phase 2: Continue polishing best (25% of budget)
    t2 = t0 + budget * 0.70
    try:
        try_update(_optimize(best_seq, t2, seed=555))
    except Exception:
        pass

    # Phase 3: Asymmetric initializations
    if time.monotonic() < deadline - 80:
        inits = []
        x = np.arange(n, dtype=np.float64) / n
        pn = parent / max(parent.sum(), 1e-15)

        # Exponential decay (key: asymmetric shapes beat bound 2)
        for rate in [1.0, 2.0, 3.0, 4.0, 6.0, 8.0]:
            inits.append(np.exp(-rate * x))
            inits.append(np.exp(-rate * (1.0 - x)))
        # Power law
        for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
            inits.append((x + 0.02) ** (-p))
            inits.append(((1.0 - x) + 0.02) ** (-p))
        # Avoid endpoints
        for eps in [0.05, 0.1, 0.2]:
            v = np.ones(n)
            fade = int(eps * n)
            if fade > 0:
                v[:fade] *= np.linspace(0, 1, fade)
                v[-fade:] *= np.linspace(1, 0, fade)
            inits.append(v)
        # Blends with parent
        rng_b = np.random.default_rng(42)
        for frac in [0.7, 0.85]:
            for init in inits[:12:2]:
                vv = init / init.sum()
                inits.append(frac * pn + (1 - frac) * vv)

        remaining = max(deadline - time.monotonic(), 20.0)
        per_init = max(4.0, remaining / (len(inits) + 2))

        for i, init in enumerate(inits):
            if time.monotonic() > deadline - 40:
                break
            alloc = min(time.monotonic() + per_init, deadline - 30)
            try:
                r = _optimize(init, alloc, seed=2000 + i)
                try_update(r)
            except Exception:
                pass

    # Phase 4: Final deep polish
    if time.monotonic() < deadline - 15:
        try:
            try_update(_optimize(best_seq, deadline, seed=77777))
        except Exception:
            pass

    # Final validation
    try:
        fs = float(evaluate_sequence([float(v) for v in best_seq]))
        if not np.isfinite(fs):
            raise ValueError("invalid")
    except Exception:
        return [1.0] * 100

    return [float(v) for v in best_seq]
