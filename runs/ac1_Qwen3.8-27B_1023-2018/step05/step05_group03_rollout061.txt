from scipy.fft import rfft, irfft, next_fast_len

def _fft_conv(a, size=None):
    n = a.size
    L = 2 * n - 1
    if size is None:
        size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return c

    c = _fft_conv(p)
    m = c.max()
