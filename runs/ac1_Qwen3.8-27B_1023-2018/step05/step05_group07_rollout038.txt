For the linear part: `max_k (base_k + 2*a[k-j]*x)` is a piecewise-linear convex function. Its minimum is where the two highest-slope lines cross (or at a boundary).

For the full function including the quadratic: we need to find `x >= 0` minimizing `max(linear_part(x), base_{2j} + x^2)`.

This can be solved by:
1. Find `x_lin` that minimizes the linear part (subject to `x >= 0`).
2. Find `x_quad` that minimizes `base_{2j} + x^2` (which is `x = 0`).
3. The minimum of the max is at the intersection or at one of these.

Actually, since the quadratic `x^2` increases for `x > 0` and the linear part is convex, the minimum of `max(linear, quadratic)` is at a point where either:
- The linear part is minimized (and `x^2` is below it), or
- They're equal (intersection point).

This is solvable in O(n log n) or O(n) per coordinate.

def _cd_step(a, j):
    """Coordinate descent: optimize a[j] to minimize max(convolve(a,a))."""
    n = len(a)
    x = a[j]
    # Compute base_k and coefficients
    base = np.zeros(2*n-1)
    lin_coeff = np.zeros(2*n-1)  # coefficient of x in linear part
    quad_coeff = np.zeros(2*n-1)  # coefficient of x^2 (1 for k=2j, else 0)
    
    # base_k = convolve(a,a)[k] - contribution of a[j]
    # Original: convolve(a,a)[k] = sum_i a[i]*a[k-i]
    # Remove terms involving a[j]:
    #   i=j: a[j]*a[k-j] (valid if 0<=k-j<n)
    #   k-i=j: a[k-j]*a[j] (valid if 0<=k-j<n and k-j != j)
    
    conv_full = np.convolve(a, a)  # current full convolution
    
    # Contribution of a[j] to convolve(a,a)[k]:
    # If 0 <= k-j < n: contribution is a[j]*a[k-j] (from i=j)
    # If 0 <= k-j < n and k-j != j: additional a[k-j]*a[j] (from k-i=j)
    # Total: if 0 <= k-j < n:
    #   if k != 2j: 2*a[j]*a[k-j]
    #   if k == 2j: a[j]^2
    
    # So base_k = conv_full[k] - contribution
    contribution = np.zeros(2*n-1)
    # k ranges from j to j+n-1 (where k-j ranges from 0 to n-1)
    k_valid = np.arange(j, j+n)
    for k in k_valid:
        idx = k - j  # = a[k-j]
        if k == 2*j:
            contribution[k] = x*x
        else:
            contribution[k] = 2*x*a[idx]
    
    base = conv_full - contribution
    
    # Now f(x) = max_k (base[k] + lin_coeff[k]*x + quad_coeff[k]*x^2)
    # lin_coeff[k] = 2*a[k-j] if 0<=k-j<n and k != 2j, else 0
    # quad_coeff[k] = 1 if k==2j, else 0
    lin_coeff = np.zeros(2*n-1)
    for k in k_valid:
        idx = k - j
        if k != 2*j:
            lin_coeff[k] = 2*a[idx]
    quad_coeff = np.zeros(2*n-1)
    if 2*j < 2*n-1:
        quad_coeff[2*j] = 1.0
    
    # Minimize f(x) = max_k (base[k] + lin_coeff[k]*x + quad_coeff[k]*x^2) over x >= 0
    # This is a convex optimization in 1D
    
    # Use ternary search or bisection on the derivative
    # f(x) is convex (max of convex functions), so we can use bisection on f'(x)
    
    # f'(x) = lin_coeff[k*] * 1 + 2*x*quad_coeff[k*] where k* is an argmax
    # More precisely, f'(x) is in the subdifferential.
    
    # Simpler: use golden section search on [0, x_max]
    # x_max: we need a reasonable upper bound. Since a[j] was x, try [0, max(x*2, 2*x_mean)]
    x_mean = a.sum() / n
    x_max = max(x * 2, 4 * x_mean, 0.1)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    lo, hi = 0.0, x_max
    
    def f_val(xx):
        vals = base + lin_coeff * xx + quad_coeff * xx * xx
        return float(vals.max())
    
    # First check if the minimum is at x=0 or x=x_max
    f_lo = f_val(0)
    f_hi = f_val(x_max)
    
    if f_lo <= f_hi:
        hi = x_max
        lo = 0.0
    else:
        # minimum might be beyond x_max, expand
        for _ in range(20):
            x_max *= 2
            if f_val(x_max) > f_hi:
                break
        hi = x_max
        lo = 0.0
    
    # Golden section
    for _ in range(50):
        if hi - lo < 1e-12:
            break
        m1 = lo + gr * (hi - lo)
        m2 = hi - gr * (hi - lo)
        f1 = f_val(m1)
        f2 = f_val(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    x_opt = 0.5 * (lo + hi)
    return x_opt

    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.full(A.shape[0], M),
                      bounds=[(0, None)]*n, method='highs',
                      options={'time_limit': tl, 'primal_feasibility_tolerance': 1e-10,
                               'dual_feasibility_tolerance': 1e-10})
        if res.status == 0:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None

def _line_search(a, g):
    """Find t in [0,1] minimizing max(convolve((1-t)a+tg, itself))."""
    sa, sg = a.sum(), g.sum()
    if sa < 1e-15 or sg < 1e-15:
        return a, float(np.convolve(a, a).max())
    a_n, g_n = a/sa, g/sg
    best_t, best_v = 0.0, float(np.convolve(a_n, a_n).max())
    ts = np.linspace(0, 1, 21)
    for t in ts[1:]:
        cand = (1-t)*a_n + t*g_n

import numpy as np
import time
from scipy.optimize import linprog


def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)


def _get_parent():
    try:
        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(200)
    p = np.clip(p, 0, 1000)
    if p.size == 0 or not np.all(np.isfinite(p)):
        p = np.ones(200)
    return p


def _build_A(n, a):
    k = np.arange(2 * n - 1)[:, None].astype(np.intp)
    j = np.arange(n)[None, :].astype(np.intp)
    d = k - j
    mask = (d >= 0) & (d < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = a[d[mask]]
    return A


def _solve_lp(A, M, tl=3.0):
    n = A.shape[1]
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(A.shape[0], M),
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': tl,
                               'primal_feasibility_tolerance': 1e-11,
                               'dual_feasibility_tolerance': 1e-11,
                               'dual_staleness_limit': 10})
        if res.status == 0 and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None


def _line_search(a, g):
    sa, sg = a.sum(), g.sum()
    if sa < 1e-15 or sg < 1e-15:
        return a.copy(), float(np.convolve(a, a).max())
    a_n, g_n = a / sa, g / sg
    best_t, best_v = 0.0, float(np.convolve(a_n, a_n).max())
    for t in np.linspace(0.05, 1.0, 20):
        cand = (1 - t) * a_n + t * g_n
        v = float(np.convolve(cand, cand).max())
        if v < best_v:
            best_v, best_t = v, t
    if 0 < best_t < 1:
        lo, hi = max(0.0, best_t - 0.05), min(1.0, best_t + 0.05)
        for _ in range(45):
            mid = 0.5 * (lo + hi)
            cand = (1 - mid) * a_n + mid * g_n
            v = float(np.convolve(cand, cand).max())
            if v < best_v:
                best_v, best_t = v, mid
                hi = mid
            else:
                lo = mid
            if hi - lo < 1e-11:
                break
    cand = np.maximum((1 - best_t) * a_n + best_t * g_n, 0.0)
    sc = cand.sum()
    if sc < 1e-15:
        return a_n, best_v
    return cand / sc, best_v


def _lp_optimize(a0, deadline, seed=42):
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stuck = 0

    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        M = float(conv.max())
        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 80:
            sig = min(0.015 + 0.008 * ((stuck - 80) // 80), 0.25)
            a *= np.exp(rng.normal(0, sig, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck = 0
            if sig > 0.24:
                break
            continue

        A = _build_A(n, a)
        g = _solve_lp(A, M * (1 + 1e-9), tl=2.5)
        moved = False

        if g is not None:
            gs = g.sum()
            if gs > 1.0 + 1e-7:
                g_n = g / gs
                cand, val = _line_search(a, g_n)
                if val < best_M * (1 - 1e-10) or val < M * (1 - 1e-9):
                    a = cand
                    moved = True

        if not moved:
            # Subgradient step
            active = np.where(conv >= M * (1 - 1e-10))[0]
            if len(active) > 0:
                grad = np.zeros(n)
                for k in active:
                    ki = int(k)
                    jlo = max(0, ki - n + 1)
                    jhi = min(n, ki + 1)
                    idx = ki - np.arange(jlo, jhi)
                    grad[jlo:jhi] += 2.0 * a[idx]
                grad /= len(active)
                for alpha in [0.02, 0.01, 0.005, 0.002, 0.05]:
                    cand = a - alpha * grad
                    cand = np.maximum(cand, 0)
                    sc = cand.sum()
                    if sc < 1e-12:
                        continue
                    cand = cand / sc
                    v = float(np.convolve(cand, cand).max())
                    if v < M * (1 - 1e-9):
                        a = cand
                        moved = True
                        break

        if not moved:
            a *= np.exp(rng.normal(0, 0.008, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

    return best_a


def _cd_optimize(a0, deadline, seed=42):
    a = np.clip(np.asarray(a0, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_M = float(np.convolve(a, a).max())
    rng = np.random.default_rng(seed)
    stagnant = 0

    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() > deadline:
                break
            x = a[j]
            if x < 1e-15:
                continue

            # f(xt) = max_k conv_new[k] where conv_new[k] = conv[k] + delta_k
            # delta_k = 2*(xt-x)*a[k-j] if k!=2j and 0<=k-j<n
            #         = xt^2 - x^2 if k==2j
            #         = 0 otherwise

            k_lo = max(0, j)
            k_hi = min(2 * n - 1, j + n)
            if k_hi <= k_lo:
                continue

            outside_max = float(conv[:k_lo].max() if k_lo > 0 else 0)
            if k_hi < 2 * n - 1:
                outside_max = max(outside_max, float(conv[k_hi:].max()))

            def f_val(xt):
                ks = np.arange(k_lo, k_hi)
                idx = ks - j
                deltas = 2.0 * (xt - x) * a[idx]
                if k_lo <= 2 * j < k_hi:
                    deltas[2 * j - k_lo] = xt * xt - x * x
                return max(float((conv[ks] + deltas).max()), outside_max)

            # Find optimum via golden section on [0, x_hi]
            x_hi = max(x * 2.5, 5.0 * a.sum() / n)
            f0 = f_val(0.0)
            fx = f_val(x)
            fhi = f_val(x_hi)

            if f0 <= fx and f0 < fhi:
                x_opt = 0.0
            else:
                # Expand range if needed
                for _e in range(8):
                    x_hi *= 2
                    fhi = f_val(x_hi)
                    if fhi > fx:
                        break
                lo, hi = 0.0, x_hi
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                for _g in range(40):
                    if hi - lo < 1e-13 * max(1.0, x):
                        break
                    m1 = lo + gr * (hi - lo)
                    m2 = hi - gr * (hi - lo)
                    if f_val(m1) < f_val(m2):
                        hi = m2
                    else:
                        lo = m1
                x_opt = max(0.0, 0.5 * (lo + hi))

            if abs(x_opt - x) > 1e-14 * max(1.0, x):
                # Update convolution
                dx = x_opt - x
                if 2 * j < 2 * n - 1:
                    conv[2 * j] += x_opt * x_opt - x * x
                for k in range(k_lo, k_hi):
                    if k == 2 * j:
                        continue
                    idx = k - j
                    if 0 <= idx < n:
                        conv[k] += 2.0 * dx * a[idx]
                a[j] = x_opt
                improved = True

        if not improved:
            stagnant += 1
            if stagnant > 15:
                sig = min(0.01 + 0.005 * (stagnant - 15), 0.15)
                a *= np.exp(rng.normal(0, sig, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                conv = np.convolve(a, a)
                stagnant = 0
                if sig > 0.14:
                    break
                continue
        else:
            stagnant = 0

        M = float(conv.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()

    return best_a


def _generate_inits(n, parent, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    pn = parent / parent.sum()
    inits = []

    # Gaussian bumps at different positions
    for w in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sig in [0.05, 0.1, 0.15, 0.25]:
            v = np.exp(-0.5 * ((x - w) / sig) ** 2)
            inits.append(v / v.sum())

    # Symmetric pairs
    for w in [0.15, 0.25, 0.35]:
        for sig in [0.05, 0.1, 0.2]:
            v = np.exp(-0.5 * ((x - w) / sig) ** 2) + np.exp(-0.5 * ((x - (1 - w)) / sig) ** 2)
            inits.append(v / v.sum())

    # Cosine windows
    for m in [1, 2, 3, 4, 5]:
        v = np.maximum(np.cos(m * np.pi * (x - 0.5)) ** 2 + 0.001, 0)
        inits.append(v / v.sum())

    # Power law
    for p in [1, 2, 3, 5]:
        v = (x + 0.01) ** p + (1 - x + 0.01) ** p
        inits.append(v / v.sum())

    # Blends with parent
    for frac in [0.3, 0.5, 0.7, 0.9]:
        for init in inits[:6:2]:
            v = frac * pn + (1 - frac) * init
            inits.append(v / v.sum())

    # Random smooth
    for _ in range(6):
        noise = rng.normal(0, 0.5, n)
        # Smooth
        k = np.ones(7) / 7
        noise = np.convolve(noise, k, mode='same')
        v = np.abs(noise) + 0.01
        inits.append(v / v.sum())

    return inits
