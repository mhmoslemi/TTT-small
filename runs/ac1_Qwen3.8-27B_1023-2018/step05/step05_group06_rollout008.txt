```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len

def _conv_self(a):
    """Self-convolution using FFT, returns nonnegative array of length 2n-1."""
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score_fft(p):
    """Score using FFT convolution (fast, slightly approximate)."""
    n = len(p)
    S = float(p.sum())
    if S < 0.01:
        return math.inf
    c = _conv_self(p)
    m = float(c.max())
    return float(2.0 * n * m / (S * S))

def _score_exact(p):
    """Score using exact np.convolve (matches evaluator)."""
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    n = len(p)
    S = float(p.sum())
    if S < 0.01:
        return math.inf
    b = np.convolve(p, p)
    m = float(b.max())
    return float(2.0 * n * m / (S * S))
```

def _optimized_coord_descent(p, stop_time, consider, seed=123):
    n = len(p)
    p = np.maximum(p, 0.0).astype(np.float64).copy()
    S = float(p.sum())
    p /= S
    S = 1.0
    Ltot = 2 * n - 1
    c = _conv_self(p)
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    
    G = 20  # grid points
    
    flat = 0
    sweep = 0
    
    while time.monotonic() < stop_time - 3.0:
        sweep += 1
        improved = False
        order = rng.permutation(n)
        
        if S < 0.7 or S > 1.3:
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
            cur_score = 2.0 * n * float(c.max())
        
        # Precompute prefix/suffix max of c
        # pref_max[k] = max(c[0:k]) for k=0..2n-1 (pref_max[0]=0)
        # suff_max[k] = max(c[k:2n-1]) for k=0..2n-1 (suff_max[2n-1]=0)
        pref_max = np.zeros(Ltot + 1)
        pref_max[1:] = np.maximum.accumulate(c)
        suff_max = np.zeros(Ltot + 1)
        suff_max[:Ltot-1] = np.maximum.accumulate(c[::-1])[::-1]
        # suff_max[k] = max(c[k:])
        # Actually: suff_max[k] should be max(c[k:Ltot])

        c_rev_max = np.maximum.accumulate(c[::-1])
        suff_max = c_rev_max[::-1]  # suff_max[k] = max(c[k:])
        # Check: suff_max[Ltot-1] = c[Ltot-1], suff_max[0] = max(c)
        
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        
        for i in order:
            if time.monotonic() > stop_time - 2.0:
                break
            
            v = float(p[i])
            start = i
            end = i + n  # always <= Ltot since i <= n-1, end <= 2n-1 = Ltot
            # Lw = n always
            Lw = n
            
            base = c[start:end]  # length n
            # out_max: max of c[:start] and c[end:]
            out_max = 0.0
            if start > 0:
                out_max = max(out_max, float(pref_max[start]))
            if end < Ltot:
                out_max = max(out_max, float(suff_max[end]))
            
            infl = 2.0 * p  # length n, same for all i
            diag_pos = i  # 0 <= i < n = Lw, always valid
            
            # Determine search range
            if v > 1e-12:
                x_lo = max(0.0, 0.3 * v)
                x_hi = min(4.0 * v, 20.0 / n)
                if x_hi < x_lo:
                    x_lo, x_hi = x_lo * 0.5, x_lo * 2.0
            else:
                x_lo = 0.0
                x_hi = 10.0 / n
            
            # Coarse grid
            pts = np.linspace(x_lo, x_hi, G + 1)
            ws = pts - v  # (G+1,)
            
            # Vectorized: (G+1, n)
            new_b = base[None, :] + ws[:, None] * infl[None, :]
            new_b[:, diag_pos] += ws * ws
            maff = new_b.max(axis=1)
            maff = np.maximum(maff, out_max)
            denoms = S + ws
            vals = 2.0 * n * maff / (denoms * denoms)
            vals = np.where(denoms > 1e-15, vals, np.inf)
            
            if not np.any(np.isfinite(vals)):
                continue
            
            imin = int(np.argmin(vals))
            best_x = float(pts[imin])
            best_val = float(vals[imin])

                    d_g = a_g + gr * (b_g - a_g)

                            c_g = b_g - gr * (b_g - a_g)
                            w_c = c_g - v
                            nb = base + w_c * infl
                            nb = nb.copy(); nb[diag_pos] += w_c * w_c
                            f_c = 2.0 * n * max(float(nb.max()), out_max) / (S + w_c)**2
                        else:
                            a_g = c_g; c_g = d_g; f_c = f_d
                            d_g = a_g + gr * (b_g - a_g)
                            w_d = d_g - v
                            nb = base + w_d * infl
                            nb = nb.copy(); nb[diag_pos] += w_d * w_d

                    w_opt = x_opt - v
                    nb = base + w_opt * infl
                    nb = nb.copy(); nb[diag_pos] += w_opt * w_opt
                    f_opt = 2.0 * n * max(float(nb.max()), out_max) / (S + w_opt)**2
                    if f_opt < best_val:
                        best_x, best_val = x_opt, f_opt
            
            # Apply
            if math.isfinite(best_val) and best_val < cur_score - 1e-13 and best_x >= 0:
                w = best_x - v
                if abs(w) > 1e-15:
                    # Update c using OLD p
                    # c[start:end] += 2*w*p_old[k] + w^2*delta
                    # p_old[k] = p[k] for k != i, p[i] = v
                    # But p[i] is at position i in the infl vector
                    # So: c[start:end] += w * infl (where infl uses old p)
                    #   then c[start+diag_pos] += w*w
                    c[start:end] += w * infl  # infl = 2*p (old)
                    c[start + diag_pos] += w * w
                    p[i] = best_x
                    S += w
                    cur_score = best_val
                    improved = True
        
        # Renormalize
        if S > 1e-100 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
            cur_score = 2.0 * n * float(c.max())
        
        # Periodic FFT recompute
        if sweep % 10 == 0:
            c = _conv_self(p)
            cur_score = 2.0 * n * float(c.max())
        
        if math.isfinite(cur_score) and cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            consider(p.copy())
        
        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 8 and time.monotonic() < stop_time - 120.0:
                sigma = 0.02 + 0.01 * min(flat, 10)
                p *= np.exp(rng.normal(0, sigma, n))
                p = np.maximum(p, 0)
                ss = float(p.sum())
                if ss > 1e-100:
                    p /= ss; S = 1.0
                    c = _conv_self(p)
                    cur_score = 2.0 * n * float(c.max())
                flat = 0
            if flat >= 30:
                break
    
    p = np.maximum(p, 0)
    ss = float(p.sum())
    if ss > 1e-100:
        p /= ss
    return best_p, best_score
