            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 30:
                if time.monotonic() < stop_time - 5.0:
                    sigma = 0.02 + 0.01 * (flat // 60)
                    p = p * np.exp(rng.normal(0.0, sigma, n))
                    p = np.maximum(p, 0.0)
                    ss = float(p.sum())
                    if ss > 0:
                        p /= ss
                        s = 1.0
                    c = _fft_conv(p)
                    cur_score = float(2.0 * n * c.max())
                    flat = 0
                else:
                    break

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p = p / ss
    try:
        c = _fft_conv(p)
        sc = float(2.0 * n * c.max())
    except Exception:
        sc = math.inf
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard = t0 + budget - 25.0

    parent = _get_parent()
    best_arr = parent.copy()
    try:
        best_score = float(_score_exact(best_arr))
    except Exception:
        best_score = math.inf

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(256) / 256.0
        best_score = float(_score_exact(best_arr))

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s_val = float(a.sum())
        if s_val < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_arr = a.copy()
            best_score = sc

    n_par = int(best_arr.size)

    # Phase 1: LSE-LBFGS (18% of budget)
    p1_end = min(hard - 60.0, t0 + 0.18 * budget)
    if p1_end > t0 + 20.0:
        p1 = best_arr.copy()
        p1 = p1 / float(p1.sum())
        try:
            p1 = _lbfgs_optimize(p1, p1_end, consider=consider, fine_only=False)
            if p1 is not None and p1.size == n_par and np.all(np.isfinite(p1)) and p1.sum() > 0:
                p1 = p1 / float(p1.sum())
                consider(p1)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # Phase 2: Main coordinate descent (55% of budget)
    p2_end = min(hard - 80.0, t0 + 0.73 * budget)
    if p2_end > time.monotonic() + 30.0:
        try:
            work, _ = _coord_descent(work, p2_end, seed=int(seed + 1111),
                                     consider=consider, G=48)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # Phase 3: Second LSE pass (10%)
    p3_end = min(hard - 50.0, time.monotonic() + 0.10 * budget)
    if p3_end > time.monotonic() + 20.0:
        try:
            p3 = _lbfgs_optimize(work, p3_end, consider=consider, fine_only=True)
            if p3 is not None and p3.size == n_par and np.all(np.isfinite(p3)) and p3.sum() > 0:
                p3 = p3 / float(p3.sum())
                consider(p3)
        except Exception:
            pass

    work = best_arr.copy()
    work = work / float(work.sum())

    # Phase 4: Fine coordinate descent (15%)
    p4_end = min(hard - 30.0, t0 + 0.92 * budget)
    if p4_end > time.monotonic() + 25.0:
        try:
            work, _ = _coord_descent(work, p4_end, seed=int(seed + 99999),
                                     consider=consider, G=64)
        except Exception:
            pass

    # Phase 5: Final squeeze
    fin_end = min(hard - 10.0, t0 + budget - 12.0)
    if fin_end > time.monotonic() + 15.0:
        work = best_arr.copy()
        work = work / float(work.sum())
        try:
            work, _ = _coord_descent(work, fin_end, seed=int(seed + 314159),
                                     consider=consider, G=80, fine_pts=80)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent.copy()
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.full(64, 1.0 / 64.0)
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
