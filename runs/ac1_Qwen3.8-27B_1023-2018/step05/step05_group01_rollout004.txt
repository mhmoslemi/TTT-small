Let me think about the coordinate descent more carefully.

For coordinate i with current value v:
- Current convolution: c (length 2n-1)
- Current sum: S
- Current score: 2n * max(c) / S^2

If we change a[i] to x (delta = x - v):
- Affected c entries: c[k] for k in [i, i+n-1]
  - c[k] += 2*(x-v)*a[k-i] for k != 2i
  - c[2i] += 2*(x-v)*a[i] + (x-v)^2
- New sum: S' = S + (x-v)
- New score: 2n * max(updated c) / (S')^2

The max of updated c is:
max(
  max(c[k] + 2*(x-v)*a[k-i]) for k in [i, i+n-1], k != 2i,
  c[2i] + 2*(x-v)*a[i] + (x-v)^2,
  max(c[k]) for k not in [i, i+n-1]
)

Let M_const = max(c[k]) for k not in [i, i+n-1].
Let infl[k] = 2*a[k-i] for k in [i, i+n-1], k != 2i (these are the "influence" coefficients).

Then the max is:
max(M_const, max(base[k] + delta*infl[k] for k in affected, k != 2i), c[2i] + delta*a[i]*2 + delta^2)

where base[k] = c[k] for affected k.

The score as a function of x (or delta) is:
f(x) = 2n * max(M_const, max_k (base_k + (x-v)*infl_k), c_2i + (x-v)*2a_i + (x-v)^2) / (S + x - v)^2

This is a ratio of a convex, piecewise-affine (or piecewise-quadratic) function to a convex quadratic. The minimum can be found by:
1. Grid search (coarse)
2. Refine with bisection or Newton's method in the best interval

For the grid search, we need to evaluate f at G points. The cost per evaluation is O(n) (to find the max of the affected entries). For G points, that's O(G*n) per coordinate. We can vectorize this.

The vectorized evaluation:
- Create a (G+1) x L matrix where L = n (length of affected region)
- Row j corresponds to x_j = pts[j]
- Entry (j, k) = base[k] + (pts[j] - v) * infl[k]
- Take max along axis 1 to get the max of affected entries for each x_j
- Also compute c_2i + (pts[j]-v)*2*a_i + (pts[j]-v)^2 for each x_j
- Take element-wise max with M_const
- Compute score = 2n * that / (S + pts[j] - v)^2

This is what the previous code does. The main cost is the (G+1) x n matrix, which for G=48, n=3072 is about 1.2 MB. That's fine.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:
        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(3072, dtype=np.float64)
    return p

def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0)
    if n <= 512:
        c = np.convolve(a, a, mode='full')
    else:
        L = 2*n - 1
        size = next_fast_len(L)
        F = rfft(a, size)
        c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _score_arr(a):
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)

def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0/u.size)
    return e / s

def _resample(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    n = max(1, int(n))
    if src.size == 0:
        return np.full(n, 1.0/n)
    if src.size == n:
        return src.copy()
    m = src.size
    xp = (np.arange(m) + 0.5) / m
    fp = np.interp((np.arange(n) + 0.5) / n, xp, src)
    out = np.maximum(fp, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n, 1.0/n)
    return out / s

def _lse_optimize(p, stop_time, seed=42, consider=None):
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    n = int(p.size)
    if n == 0:
        return p
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2*n - 1)
    cross_size = next_fast_len(3*n - 2)
    conv_len = 2*n - 1
    cross_len = 3*n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n-1:2*n-1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        if c.size == 0:
            return 1e100, np.zeros_like(uu)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / float(c.size)))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    stages = (
        (0.15, 1.0),
        (0.08, 1.0),
        (0.04, 1.0),
        (0.02, 1.0),
        (0.01, 1.0),
        (0.005, 1.0),
        (0.002, 1.0),
        (0.001, 1.0),
        (0.0004, 1.0),
        (0.0001, 1.0),
    )

    start = time.monotonic()
    avail = max(0.0, float(stop_time) - start - 5.0)
    if avail <= 5.0:
        return p

    best_p = p.copy()
    best_sc = _score_arr(p)
    total_w = float(sum(w for _, w in stages))
    rng = np.random.default_rng(seed)
    cum = 0.0

    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 3.0:
            break
        cum += float(wt)
        end_stage = min(float(stop_time) - 2.0, start + avail * (cum / total_w))
        if end_stage <= now + 1.0:
            continue
        c = cfast(p)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0:
            break
        T = max(1e-300, frac * cmax)

        best_u = u.copy()
        f0, _ = lse_func(best_u, T)
        if not math.isfinite(f0):
            continue
        best_f = f0
        stale = 0

        while time.monotonic() < end_stage - 0.5:
            opts = {"maxiter": 30, "gtol": 1e-11, "ftol": 1e-14,
                    "maxls": 20, "maxcor": 20, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if x.size == u.size and np.all(np.isfinite(x)):
                    try:
                        fval = float(res.fun)
                    except Exception:
                        fval = math.inf
                    if math.isfinite(fval) and fval < best_f - 1e-15:
                        best_f = fval
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 2:
                            best_u = best_u + rng.normal(0, 0.005, best_u.size)
                            continue
            if getattr(res, "status", 1) == 0:
                if time.monotonic() < end_stage - 1.0 and stale < 2:
                    best_u = best_u + rng.normal(0, 0.003, best_u.size)
                else:
                    break

        p_cur = _softmax(best_u)
        p_cur = np.maximum(p_cur, 0.0)
        ss = float(p_cur.sum())
        if ss > 1e-300:
            p_cur /= ss
        sc_cur = _score_arr(p_cur)
        if math.isfinite(sc_cur) and sc_cur < best_sc:
            best_sc = sc_cur
            best_p = p_cur.copy()
            p = p_cur.copy()
        elif math.isfinite(sc_cur) and sc_cur < best_sc + 0.003:
            p = p_cur.copy()
        else:
            p = best_p.copy()
        u = p_to_u(p)
        if consider is not None:
            try:
                consider(best_p)
            except Exception:
                pass
