def _lp_dual_weights(a, M, active_lags):
    """Solve dual LP to get subgradient weights for active lags."""
    n = len(a)
    K = list(active_lags)
    m = len(K)
    
    # Dual: min sum(y_k) s.t. sum_k y_k * a[k-j] >= 1 for all j, y_k >= 0
    # (scaled: the constraint is sum_k y_k * a[k-j] >= (sum a)/(2M) ... )
    
    # Actually, let me use the primal LP dual directly.
    # Primal: max sum(b) s.t. (a*b)_k <= M, b >= 0
    # Dual:   min M * sum(y_k) s.t. sum_k y_k * a[k-j] >= 1 for all j, y_k >= 0
    
    # But we only care about y_k for k in K (active set).
    # The dual constraint for coordinate j: sum_{k in K} y_k * a[k-j] >= 1 (approximately)
    
    # This is: min 1^T y s.t. B^T y >= 1, y >= 0
    # where B[j, idx] = a[K[idx] - j] if 0 <= K[idx]-j < n else 0
    
    c_obj = np.ones(m)
    B = np.zeros((n, m))
    for idx, k in enumerate(K):
        k = int(k)
        for j in range(n):
            idx2 = k - j
            if 0 <= idx2 < n:
                B[j, idx] = a[idx2]
    
    # min c_obj^T y s.t. B^T y >= 1  =>  -B^T y <= -1
    A_ub = -B.T
    b_ub = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*m,
                      method='highs', options={'time_limit': 1.0})
        if res.status == 0:
            y = np.maximum(res.x, 0)
            y /= y.sum()  # normalize to sum 1
            return y
    except:
        pass
    return np.ones(m) / m

But actually, the dual LP as I wrote it might be infeasible if the active set K doesn't "cover" all coordinates j. Let me be more careful.

The dual of:
max sum(b) s.t. (a*b)_k <= M for k=0,...,2n-2; b >= 0

is:
min M * sum_{k=0}^{2n-2} y_k
s.t. sum_{k} y_k * a_{k-j} >= 1 for j=0,...,n-1
     y_k >= 0

where a_{k-j} = 0 if k-j is out of [0, n-1].

If I restrict to y_k = 0 for k not in K, the dual becomes:
min M * sum_{k in K} y_k
s.t. sum_{k in K} y_k * a_{k-j} >= 1 for all j
     y_k >= 0 for k in K

This might be infeasible if K is too small (the weights can't "reach" all coordinates j). In that case, I should use the full active set or add more lags.

For a robust implementation, I'll:
1. Use the full set of "nearly active" lags (c[k] >= M * 0.99)
2. Solve the restricted dual LP
3. If infeasible, fall back to uniform weights

import numpy as np
import time
from scipy.optimize import linprog
import math

def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * float(c.max()) / (s * s)

def _get_parent():
    try:
        p = np.asarray(globals()["height_sequence_1"], dtype=np.float64).ravel()
    except Exception:
        p = np.ones(200)
    p = np.clip(p, 0, 1000)
    if p.size == 0 or not np.all(np.isfinite(p)):
        p = np.ones(200)
    return p

def _conv_matrix(n, a):
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        jlo, jhi = max(0, k-n+1), min(n, k+1)
        A[k, jlo:jhi] = a[k - np.arange(jlo, jhi)]
    return A

# LP best response: max sum(b) s.t. (a*b)_k <= M, b >= 0
def _lp_br(a, M, tl=2.0):
    n = len(a)
    A = _conv_matrix(n, a)
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=[(0,None)]*n, method='highs',
                      options={'time_limit': tl})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except:
        pass
    return None

# Min-t LP: min t s.t. (a'*a)_k <= t, sum(a')=1, a'>=0
def _lp_mint(a, tl=2.0):
    n = len(a)
    A = _conv_matrix(n, a)
    c = np.zeros(n+1); c[-1] = 1
    A_ub = np.hstack([A, -np.ones((2*n-1,1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.hstack([np.ones(n), 0.0])
    b_eq = np.array([1.0])
    bounds = [(0,None)]*n + [(0,None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': tl})
        if res.status == 0:
            return np.maximum(res.x[:n], 0), float(res.x[-1])
    except:
        pass
    return None, None

def _line_search(a, d, tol=1e-10):
    n = len(a)
    v0 = _score(a)
    best_t, best_v = 0.0, v0
    for t in [0.05,0.1,0.2,0.3,0.5,0.7,1.0]:
        cand = np.maximum((1-t)*a + t*d, 0)
        s = cand.sum()
        if s < 1e-12: continue
        cand /= s
        v = _score(cand)
        if v < best_v: best_v, best_t = v, t
    if best_t == 0: return None, best_v
    lo, hi = max(0, best_t-0.05), min(1, best_t+0.05)
    for _ in range(18):
        mid = (lo+hi)/2
        cand = np.maximum((1-mid)*a + mid*d, 0)
        s = cand.sum()
        if s < 1e-12: lo = mid; continue
        cand /= s
        v = _score(cand)
        if v < best_v: best_v, best_t, hi = v, mid, mid
        else: lo = mid
    cand = np.maximum((1-best_t)*a + best_t*d, 0)
    s = cand.sum()
    if s < 1e-12: return None, best_v
    return cand/s, best_v

def _subgrad_dir(a, c, M, eps=0.9999):
    """Compute subgradient direction. Returns gradient vector or None."""
    n = len(a)
    active = np.where(c >= M*eps)[0]
    if len(active) == 0:
        active = np.where(c >= M*0.99)[0]
    if len(active) == 0:
        return None
    
    # Try LP dual weights
    m = len(active)
    if m <= 100:
        B = np.zeros((n, m))
        for idx, k in enumerate(active):
            k = int(k)
            for j in range(n):
                d = k - j
                if 0 <= d < n:
                    B[j, idx] = a[d]
        try:
            res = linprog(np.ones(m), A_ub=-B.T, b_ub=-np.ones(n),
                          bounds=[(0,None)]*m, method='highs',
                          options={'time_limit': 1.0})
            if res.status == 0:
                y = np.maximum(res.x, 0)
                ys = y.sum()
                if ys > 1e-12:
                    y /= ys
                    grad = np.zeros(n)
                    for idx, k in enumerate(active):
                        k = int(k)
                        for j in range(n):
                            d = k - j
                            if 0 <= d < n:
                                grad[j] += 2*y[idx]*a[d]
                    return grad
        except:
            pass
    
    # Fallback: uniform weights
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        for j in range(n):
            d = k - j
            if 0 <= d < n:
                grad[j] += 2*a[d]
    grad /= len(active)
    return grad

            mode = stuck % 4
            if mode == 0:

            elif mode == 1:

            elif mode == 2:

            else:

                st = int(rng.integers(0, max(1, n-blk)))
                a[st:st+blk] *= (1 + sig*rng.uniform(-1, 1, blk))
            a = np.maximum(a, 1e-14)
            a /= a.sum()
            stuck = 0

            continue
        
        moved = False
