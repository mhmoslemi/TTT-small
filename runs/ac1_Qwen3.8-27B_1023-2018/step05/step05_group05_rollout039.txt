import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(384, dtype=np.float64)
    if p.size == 0:
        p = np.ones(384, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(384, dtype=np.float64)
    return p


def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(np.asarray(c, dtype=np.float64), 0.0)


def _score_fft(a):
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(a)
    m = float(c.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return float(2.0 * n * m / (s * s))


def _softmax(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        return np.full(u.size, 1.0 / u.size)
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _lp_phase(f, stop_time, consider, rng, n):
    """LP-based direction finding to escape local minima."""
    f = np.maximum(np.asarray(f, dtype=np.float64).ravel(), 0.0).copy()
    s = float(f.sum())
    if s < 1e-300:
        return
    f = f / s

    def solve_lp(f_cur, M):
        try:
            cols = np.repeat(np.arange(n), n, dtype=np.int64)
            rows = np.empty(n * n, dtype=np.int64)
            for j in range(n):
                rows[j * n:(j + 1) * n] = np.arange(j, j + n, dtype=np.int64)
            vals = np.repeat(f_cur, n)
            A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
            c_obj = -np.ones(n)
            b_ub = np.full(2 * n - 1, M)
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': max(1.0, stop_time - time.monotonic() - 10.0)})
            if res.success and res.x is not None and np.all(np.isfinite(res.x)):
                return np.maximum(res.x, 0.0)
        except Exception:
            pass
        return None

    max_iters = 30
    for it in range(max_iters):
        if time.monotonic() > stop_time - 5.0:
            break
        c = _conv_self(f)
        M = float(c.max())
        if not math.isfinite(M) or M <= 0:
            break

        t_lp_start = time.monotonic()
        g0 = solve_lp(f, M)
        if g0 is None or float(g0.sum()) <= 1e-15:
            break

        g0 = g0 / float(g0.sum())

        cur_sc = _score_fft(f)
        best_t = 0.0
        best_sc = cur_sc
        for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            h = (1.0 - t) * f + t * g0
            h = np.maximum(h, 0.0)
            sh = float(h.sum())
            if sh < 1e-300:
                continue
            sc = _score_fft(h)
            if math.isfinite(sc) and sc < best_sc - 1e-14:
                best_sc = sc
                best_t = t

        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * g0
            f = np.maximum(f, 0.0)
            sw = float(f.sum())
            if sw > 1e-300:
                f = f / sw
            consider(f)
        else:
            break

    return f


def _lse_optimize(p, stop_time, consider=None, fine_only=False):
    """Log-sum-exp optimization with L-BFGS-B."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p
    p = p / s

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return np.maximum(c, 0.0)

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n - 1:2 * n - 1]
        return np.maximum(q, 0.0)

    def lse_func(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = 1e-4 * cmax
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f_val = cmax
        else:
            f_val = cmax + T * float(np.log(sum_exp / c.size))
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        return float(f_val), np.where(np.isfinite(grad), grad, 0.0)

    def p_to_u(pp):
        ll = np.log(np.maximum(pp, 1e-300))
        return ll - float(np.mean(ll))

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score_fft(p)

    if fine_only:
        stages = ((0.003, 0.30), (0.001, 0.35), (0.0003, 0.35))
    else:
        stages = (
            (0.2, 0.08),
            (0.07, 0.12),
            (0.02, 0.12),
            (0.006, 0.15),
            (0.002, 0.18),
            (0.0005, 0.18),
            (0.0001, 0.17),
        )

    total_w = sum(w for _, w in stages)

    frac_to_time = (stop_time - 8.0 - time.monotonic())
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 6.0:
            break
        time_for_stage = max(5.0, (wt / total_w) * frac_to_time)
        end_stage = min(stop_time - 4.0, now + time_for_stage)
        cmax = float(cfast(p).max())
        T = max(1e-15, frac * cmax)
        stale = 0
        while time.monotonic() < end_stage - 2.0:
            maxit = 50
            opts = {"maxiter": maxit, "gtol": 1e-14, "ftol": 1e-18,
                    "maxls": 40, "maxcor": 30, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, u, args=(T,), method="L-BFGS-B",
                               jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                fval = float(res.fun)
                if math.isfinite(fval):
                    p_cur = _softmax(res.x)
                    sc_cur = _score_fft(p_cur)
                    if math.isfinite(sc_cur) and sc_cur < best_sc - 1e-14:
                        best_sc = sc_cur
                        best_p = p_cur.copy()
                        p = p_cur.copy()
                        u = p_to_u(p)
                        stale = 0
                        if consider is not None:
                            try:
                                consider(p)
                            except Exception:
                                pass
                    else:
                        stale += 1
                        if stale >= 4:
                            break
            if getattr(res, "status", 1) == 0:
                break

    p_out = best_p.copy()
    ss = float(p_out.sum())
    if ss > 1e-300:
        p_out = p_out / ss
    return p_out


def _coord_descent(p, stop_time, consider=None, seed=42, max_sweeps=50000):
    """Efficient coordinate descent with incremental convolution updates."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p
    p = p / s

    Ltot = 2 * n - 1
    c = _conv_self(p)
    S = 1.0
    cur_score = 2.0 * n * float(c.max())
    best_score = cur_score
    best_p = p.copy()
    rng = np.random.default_rng(seed)
    flat = 0
    sweep = 0

    # Grid size adaptive: fewer for large n
    G = max(32, min(64, int(200000 / max(n, 1))))

    while sweep < max_sweeps and time.monotonic() < stop_time - 3.0:
        improved = False
        order = rng.permutation(n)

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.5:
                break
            if S <= 1e-300:
                break
            if S > 1.5 or S < 0.67:
                inv = 1.0 / S
                p *= inv
                c *= inv * inv
                S = 1.0

            v = float(p[i])
            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            # Influence vector: a[k-i] for k in [start, end-1]
            infl = p[:Ls]  # a[0], a[1], ..., a[Ls-1]
            base = c[start:end].copy()

            # Constant part (unaffected positions)
            max_const = 0.0
            if start > 0:
                ml = float(c[:start].max())
                max_const = max(max_const, ml)
            if end < Ltot:
                mr = float(c[end:].max())
                max_const = max(max_const, mr)

            # Diagonal: position 2i within [start, end-1]?
            diag_pos = 2 * i - start
            has_diag = 0 <= diag_pos < Ls

            # Candidates for new value
            x_hi = max(30.0 * v + 1e-10, 80.0 / n, 0.01)
            x_hi = min(x_hi, 5.0 * S)
            if not math.isfinite(x_hi) or x_hi <= 0:
                x_hi = 1.0 / n

            # Smart grid: uniform + extra near 0 and near v
            pts = np.linspace(0.0, x_hi, G)
            if v > 0 and v < x_hi:
                pts = np.unique(np.concatenate([pts, [v * 0.5, v, v * 1.5]]))
            pts = np.unique(pts)

            # Evaluate all candidates vectorized
            ws = pts - v  # changes
            denom = S + ws
            valid = denom > 1e-14

            if not np.any(valid):
                continue

            wv = ws[valid]
            # New conv = base + 2*w*infl + w^2 (at diag)
            aff = base[None, :] + 2.0 * wv[:, None] * infl[None, :]
            if has_diag:
                aff[:, diag_pos] += wv * wv
            aff = np.maximum(aff, 0.0)
            mv = np.maximum(np.max(aff, axis=1), max_const)
            scores = np.full(pts.shape, math.inf)
            scores[valid] = (2.0 * n * mv) / (denom[valid] ** 2)

            if not np.any(np.isfinite(scores)):
                continue

            imin = int(np.argmin(scores))
            bx = float(pts[imin])
            bv = float(scores[imin])

            # Fine search around best
            dw = x_hi / max(G - 1, 1)
            lo = max(0.0, bx - 3.0 * dw)
            hi = min(x_hi, bx + 3.0 * dw)
            if hi > lo + 1e-15:
                fine = np.linspace(lo, hi, 30)
                if 0.0 <= bx <= hi:
                    fine = np.unique(np.concatenate([fine, [bx, v]]))
                wf = fine - v
                denf = S + wf
                vf_mask = denf > 1e-14
                if np.any(vf_mask):
                    wfv = wf[vf_mask]
                    afff = base[None, :] + 2.0 * wfv[:, None] * infl[None, :]
                    if has_diag:
                        afff[:, diag_pos] += wfv * wfv
                    afff = np.maximum(afff, 0.0)
                    mvf = np.maximum(np.max(afff, axis=1), max_const)
                    scores_f = np.full(fine.shape, math.inf)
                    scores_f[vf_mask] = (2.0 * n * mvf) / (denf[vf_mask] ** 2)
                    if np.any(np.isfinite(scores_f)):
                        imf = int(np.argmin(scores_f))
                        if float(scores_f[imf]) < bv:
                            bv = float(scores_f[imf])
                            bx = float(fine[imf])

            if math.isfinite(bv) and bv < cur_score - 1e-13 and 0.0 <= bx <= 10.0 * S:
                w = bx - v
                if abs(w) > 1e-15:
                    p[i] = bx
                    S += w
                    c[start:end] += 2.0 * w * infl
                    if has_diag:
                        c[start + diag_pos] += w * w
                    cur_score = bv
                    improved = True

        # Rescale
        if S > 1e-300 and math.isfinite(S):
            inv = 1.0 / S
            p *= inv
            c *= inv * inv
            S = 1.0
        else:
            break

        # Verify periodically
        if sweep % 8 == 0:
            c = _conv_self(p)
        new_sc = 2.0 * n * float(c.max())
        if math.isfinite(new_sc) and new_sc < best_score - 1e-14:
            best_score = new_sc
            best_p = p.copy()
            if consider is not None:
                try:
                    consider(best_p)
                except Exception:
                    pass

        if improved:
            flat = 0
        else:
            flat += 1
            if flat >= 3:
                if time.monotonic() < stop_time - 10.0:
                    # Perturb: multiply by random factors clipped to [0.8, 1.2]
                    p = p * np.exp(rng.normal(0, 0.03, n))
                    p = np.maximum(p, 0)
                    ss = float(p.sum())
                    if ss > 1e-300:
                        p /= ss
                        c = _conv_self(p)
                        S = 1.0
                        cur_score = 2.0 * n * float(c.max())
                    flat = 0
                else:
                    break
        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    c = _conv_self(p)
    sc = 2.0 * n * float(c.max())
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p
