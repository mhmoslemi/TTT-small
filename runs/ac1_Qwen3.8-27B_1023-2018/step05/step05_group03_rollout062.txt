            else:

            valid = (S + d) > 1e-10

            aff = base[None,:] + d[:,None] * infl[None,:]
            if has_diag:

            amax = aff.max(axis=1)
            tmax = np.maximum(amax, mu)
            scores = 2.0*n*tmax / (S+d)**2

            # At the end of the sweep, I renormalize.
            
            # The score for the current state is 2*n*max(c)/S^2.
            # After changing a[i] to bx, the new score would be 2*n*max(new_c)/(S+d)^2.

                S += db
                improved += 1

        if S > 1e-300:
            a /= S
            c /= (S*S)
            S = 1.0

        if sc_now < best_sc - 1e-14:
            best_sc = sc_now
            best_a = a.copy()
            consider(a)
            flat = 0
        else:
            flat += 1

            sigma = 0.03 * min(flat - 2, 5)
            noise = rng.normal(0, sigma, n)
            q = a * np.exp(noise)
            q = np.maximum(q, 0)
            qs = q.sum()

                a = q / qs
            c = _fft_conv(a)
            S = a.sum()

            S = 1.0

            start = int(rng.integers(0, n - block))
            q[start:start+block] = S / n  # reset to uniform
            q = _norm1(q)
            c = _fft_conv(q)
            a = q
            S = 1.0
            flat = 0
    
    return best_a, best_sc
