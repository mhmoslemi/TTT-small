def _conv_self(a):
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c

def _fft_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(p)
    return float(2.0 * n * c.max() / (s * s))

def _exact_score(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(p, p)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))

- If k < n: l ranges from 0 to k, A[k,l] = a[k-l]
- If k >= n: l ranges from k-n+1 to n-1, A[k,l] = a[k-l]

So A is a "banded" matrix where each row has at most min(k+1, 2n-1-k, n) nonzeros. For k near n/2, it has about n/2 nonzeros (wait, no, for k < n, it has k+1 nonzeros; for k >= n, it has 2n-1-k nonzeros).

Total nonzeros = sum_k min(k+1, 2n-1-k) for k = 0 to 2n-2
= sum_{k=0}^{n-1} (k+1) + sum_{k=n}^{2n-2} (2n-1-k)
= n(n+1)/2 + (n-1)n/2
= n^2

So A has exactly n^2 = 9.4M nonzeros. Dense would be 6143*3072 = 18.9M. So it's about 50% dense.

I'll build A as a dense matrix for simplicity.

        if g.sum() > 0:
            return g / g.sum()
    return None

For improvement 5: Process coordinates in order of their "impact" on the peak. A coordinate i affects lags in [i, i+n-1]. The impact is how much it contributes to the current peak. Coordinates that contribute more to the peak should be updated first.

def _lp_direction(a, M, timeout=30.0):

    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(a)

    # A[k, j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((L, n))

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi+1] = a[k-hi:k-lo+1][::-1]

    rhs = M * np.ones(L)
    constraints = [A @ b <= rhs]
    prob = cvxpy.Problem(cvxpy.Maximize(cvxpy.sum(b)), constraints)
    
    t_start = time.monotonic()

                gs = float(g.sum())
                if gs > 0:
                    return g / gs
            break  # Don't try next solver if first one ran (even if subtle failure)
        except Exception:
            continue
    
    del A, b, prob
    return None

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _conv_self(a):
    n = len(a)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _fft_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(p)
    return float(2.0 * n * c.max() / (s * s))


def _exact_score(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    b = np.convolve(p, p)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _eval_single(x, v, S, base, infl, max_const, diag_off, Ls, n):
    w = x - v
    denom = S + w
    if denom <= 1e-15:
        return math.inf
    nv = base + w * infl
    if 0 <= diag_off < Ls:
        nv = nv.copy()
        nv[diag_off] += w * w
    m = float(nv.max())
    if max_const > m:
        m = max_const
    if m < 0.0:
        m = 0.0
    return 2.0 * n * m / (denom * denom)

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _conv_self(a):
    n = len(a)
    L = 2 * n - 1
    sz = next_fast_len(L)
    F = rfft(a, sz)
    c = irfft(F * F, sz)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _fft_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math.inf
    c = _conv_self(p)
    return float(2.0 * n * c.max() / (s * s))


def _exact_score(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    n = len(p)
    s = float(p.sum())
    if s < 0.01:
        return math_inf_placeholder()
    b = np.convolve(p, p)
    m = float(b.max())
    if not math.isfinite(m):
        return math_inf_placeholder()
    return float(2.0 * n * m / (s * s))


def math_inf_placeholder():
    return float('inf')


def _eval_single(x, v, S, base, infl, mc, doff, Ls, n):
    w = x - v
    d = S + w
    if d <= 1e-15:
        return float('inf')
    nv = base + w * infl
    if 0 <= doff < Ls:
        nv = nv.copy()
        nv[doff] += w * w
    m = float(nv.max())
    if mc > m:
        m = mc
    if m < 0.0:
        m = 0.0
    return 2.0 * n * m / (d * d)

def _lp_direction(a, M, timeout=20.0):
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(a)
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for k in range(L):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    try:
        b = cp.Variable(n, nonneg=True)
        rhs = M * np.ones(L)
        constraints = [A @ b <= rhs]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

        t_s = time.monotonic()
        solved = False
        for solver in [cp.GLPK, cp.CBC]:
            if time.monotonic() - t_s > timeout:
                break
            try:
                prob.solve(solver=solver, verbose=False)
                solved = True
                break
            except Exception:
                continue

        if solved and b.value is not None and np.all(np.isfinite(b.value)):
            g = np.maximum(b.value, 0.0)
            gs = float(g.sum())
            if gs > 1e-20:
                return g / gs
    except Exception:
        pass
    finally:
        try:
            del A
        except Exception:
            pass
    return None
