    fb = _conv_cross(f, b_current)
    
    # Select active constraints: those where fb(k) >= M * tolerance
    active = np.where(fb >= M * tolerance)[0]
    
    if len(active) == 0:
        active = np.where(fb >= M * 0.95)[0]
    if len(active) == 0:
        return None
    
    # Build LP with only active constraints
    A_active = np.zeros((len(active), n), dtype=np.float64)
    for i, k in enumerate(active):
        k_min = max(0, k - n + 1)
        k_max = min(n - 1, k)
        for j in range(k_min, k_max + 1):
            A_active[i, j] = f[k - j]
    
    # Solve reduced LP
    c_obj = -np.ones(n)
    b_ub = np.full(len(active), M)
    
    res = linprog(c_obj, A_ub=A_active, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        b = res.x
        s = b.sum()
        if s > 1e-10:
            return b / s
    return None

def _lp_direction(f, M, n, max_constr=2000):
    """Solve LP to find direction that reduces peak autocorrelation."""
    # Build sparse constraint matrix for (f*b)(k) <= M
    # A[k,j] = f[k-j] if 0 <= k-j < n
    # Use active set: start with all, but if too many, subsample

    L = 2 * n - 1
    if L <= max_constr:
        all_ks = np.arange(L)
    else:
        # Subsample: take every other, plus the peaks
        # First compute f*self to find peaks
        fc = _conv_self(f)
        peak_threshold = M * 0.8
        peak_ks = np.where(fc >= peak_threshold)[0]
        regular_ks = np.arange(0, L, max(1, L // max_constr))
        all_ks = np.union1d(peak_ks, regular_ks)
        if len(all_ks) > max_constr:
            all_ks = all_ks[:max_constr]

    # Build dense matrix for active constraints (small enough)
    n_active = len(all_ks)
    A = np.zeros((n_active, n), dtype=np.float64)
    for ii, k in enumerate(all_ks):
        k_min = max(0, k - n + 1)
        k_max = min(n - 1, k)
        if k_min <= k_max:
            A[ii, k_min:k_max + 1] = f[k - k_min:k - k_min + 1]

    c_obj = -np.ones(n)
    b_ub = np.full(n_active, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})

    if res.success:
        b = res.x
        s = b.sum()
        if s > 1e-10:
            return b / s
    return None


def _line_search(f, g, c, n, max_iter=40):
    """Find best t in (0,1) to minimize max of ((1-t)*f + t*g)^2."""
    fg = _conv_cross(f, g)
    gg = _conv_self(g)
    L = 2 * n - 1
    fg = fg[:L]
    gg = gg[:L]

    def score_at_t(t):
        c_t = (1 - t) ** 2 * c + 2 * t * (1 - t) * fg + t ** 2 * gg
        return float(c_t.max())

    GR = (math.sqrt(5.0) - 1.0) / 2
    a, b = 0.0, 1.0
    c_gs = b - GR * (b - a)
    d_gs = a + GR * (b - a)
    f_c = score_at_t(c_gs)
    f_d = score_at_t(d_gs)

    for _ in range(max_iter):
        if b - a < 1e-10:
            break
        if f_c < f_d:
            b = d_gs
            d_gs = c_gs
            f_d = f_c
            c_gs = b - GR * (b - a)
            f_c = score_at_t(c_gs)
        else:
            a = c_gs
            c_gs = d_gs
            f_c = f_d
            d_gs = a + GR * (b - a)
            f_d = score_at_t(d_gs)

    t_opt = (a + b) / 2.0
    # Also check small t values
    for t_test in [0.01, 0.02, 0.05, 0.1]:
        if score_at_t(t_test) < score_at_t(t_opt):
            t_opt = t_test
    return t_opt

    block_sizes = [8, 16, 32]

    def make_obj(fixed_indices, free_indices):
        fixed_vals = p[fixed_indices].copy()

        def obj_and_grad(x):
            pp = p.copy()
            pp[free_indices] = np.maximum(x, 0.0)
            s = pp.sum()
            if s < 1e-10:
                return 1e100, np.zeros_like(x)
            c = _conv_self(pp)
            cmax = float(c.max())
            val = 2.0 * n * cmax / (s * s)
            # Gradient via subgradient of max
            peak_idx = int(np.argmax(c))
            # dc/dp_i for the peak entry
            k = peak_idx
            grad = np.zeros_like(pp)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for j in range(i_min, i_max + 1):
                grad[j] += pp[k - j]
                if j != k - j:
                    grad[k - j] += pp[j]
                else:
                    grad[j] += pp[j]  # double count for the diagonal
            # Actually grad[k-j] gets pp[j] and grad[j] gets pp[k-j]
            # Let me redo: c[k] = sum_j pp[j]*pp[k-j]
            # dc[k]/dp[i] = pp[k-i] (if 0<=k-i<n) + pp[i-(k-i)] = pp[2i-k] (if 0<=2i-k<n)
            # More carefully:
            # c[k] = sum_{j: 0<=j<n, 0<=k-j<n} pp[j] * pp[k-j]
            # d(c[k])/d(pp[i]) = pp[k-i] (if 0<=k-i<n) + pp[2i-k] (if 0<=2i-k<n)
            # But if k-i == 2i-k, i.e., k==3i... no.
            # If j=i: term is pp[i]*pp[k-i], derivative is pp[k-i]
            # If k-j=i, i.e., j=k-i: term is pp[k-i]*pp[i], derivative is pp[i]
            # If i == k-i (k=2i): same term twice, derivative is 2*pp[i]
            grad2 = np.zeros_like(pp)
            if 0 <= k - i_min and k - i_min < n:
                pass
            # Simpler: for peak index k,
            # d c[k] / d p[i] = p[k-i] + p[2i-k] (with boundary checks)
            for i in range(n):
                g = 0.0
                if 0 <= k - i < n:
                    g += pp[k - i]
                if 0 <= 2 * i - k < n and 2 * i - k != k - i:
                    g += pp[2 * i - k]
                elif 0 <= 2 * i - k < n and 2 * i - k == k - i:
                    g += pp[2 * i - k]
                grad2[i] = g
            # Full gradient of 2*n*cmax/s^2:
            # = 2*n * (1/s^2 * grad2 - 2*cmax/s^3 * ones)
            full_grad = (2.0 * n / (s * s)) * grad2 - (4.0 * n * cmax / (s * s * s)) * 1.0
            gx = full_grad[free_indices].copy()
            gx = np.where(np.isfinite(gx), gx, 0.0)
            return val, gx

        return obj_and_grad

    attempts = 0
    while time.monotonic() < stop_time - 3.0 and attempts < 20:
        bs = block_sizes[attempts % len(block_sizes)]
        indices = rng.choice(n, size=bs, replace=False)
        fixed = np.setdiff1d(np.arange(n), indices)
        bf = make_obj(fixed, indices)
        x0 = p[indices].astype(np.float64).copy()
        x0 = np.maximum(x0, 0.0)

        try:
            res = minimize(bf, x0, method='L-BFGS-B', jac=True,
                           options={'maxiter': 30, 'maxdisp': 0})
        except Exception:
            attempts += 1
            continue

        if res.x is not None and np.all(np.isfinite(res.x)):
            p_new = p.copy()
            p_new[indices] = np.maximum(res.x, 0.0)
            p_new /= p_new.sum()
            sc = 2.0 * n * _conv_self(p_new).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_p = p_new.copy()
                p = p_new.copy()
                if consider:
                    try:
                        consider(best_p)
                    except Exception:
                        pass
        attempts += 1

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(20.0, float(budget_s))
    hard_deadline = t0 + budget - 25.0

    parent = _get_parent()
    n = parent.size

    p = np.maximum(parent, 0.0).copy()
    s = float(p.sum())
    if s < 1e-10:
        p = np.ones(n) / n
    else:
        p = p / s

    c = _conv_self(p)
    best_score = float(2.0 * n * c.max())
    best_arr = p.copy()

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        if a.size != n:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.maximum(a, 0.0)
        ss = float(a.sum())
        if ss < 1e-10:
            return
        a = a / ss
        cc = _conv_self(a)
        sc = float(2.0 * n * cc.max())
        if math.isfinite(sc) and sc < best_score - 1e-13:
            best_score = sc
            best_arr = a.copy()

    # Phase 1: LSE optimization (45s max)
    lse_time = min(45.0, 0.06 * budget)
    if lse_time > 15.0:
        lse_end = t0 + lse_time
        try:
            p_lse = _lse_optimize(p, lse_end, consider=consider)
            if p_lse is not None and p_lse.size == n and np.all(np.isfinite(p_lse)):
                p = p_lse / p_lse.sum()
                c = _conv_self(p)
        except Exception:
            pass

    # Phase 2: Main loop - CD with periodic LP kicks
    cd_seed = int(seed) + 1000
    lp_interval = 60.0  # seconds between LP attempts
    last_lp = time.monotonic()
    lp_count = 0
    max_lp = 15

    while time.monotonic() < hard_deadline - 30.0:
        # Determine how long to run CD before next LP
        time_to_lp = last_lp + lp_interval - time.monotonic()
        if lp_count >= max_lp or time_to_lp < 5.0:
            cd_stop = hard_deadline - 30.0
        else:
            cd_stop = min(last_lp + lp_interval, hard_deadline - 30.0)

        if cd_stop > time.monotonic() + 10.0:
            p, c, _ = _cd_sweeps(p, c, n, cd_stop, consider=consider, seed=cd_seed, max_sweeps=999999)
            cd_seed += 7919

        # LP kick
        if lp_count < max_lp and time.monotonic() < hard_deadline - 80.0:
            now = time.monotonic()
            M = float(c.max())
            if math.isfinite(M) and M > 0:
                try:
                    g = _lp_direction(p, M, n, max_constr=2000)
                    if g is not None:
                        t_opt = _line_search(p, g, c, n)
                        if t_opt > 0.001:
                            p_new = (1.0 - t_opt) * p + t_opt * g
                            p_new = np.maximum(p_new, 0.0)
                            ss = float(p_new.sum())
                            if ss > 1e-10:
                                p_new = p_new / ss
                                c_new = _conv_self(p_new)
                                new_sc = float(2.0 * n * c_new.max())
                                if new_sc < float(2.0 * n * c.max()) - 1e-12:
                                    p = p_new
                                    c = c_new
                                    consider(p)
                except Exception:
                    pass
            lp_count += 1
            last_lp = time.monotonic()
            # Shrink LP interval slightly as we go
            lp_interval = max(30.0, lp_interval * 0.85)

    # Phase 3: Final CD squeeze
    rem = hard_deadline - 10.0 - time.monotonic()
    if rem > 20.0:
        try:
            p, c, _ = _cd_sweeps(p, c, n, hard_deadline - 10.0, consider=consider,
                                  seed=cd_seed + 55555, max_sweeps=999999)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = np.asarray(parent, dtype=np.float64)
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                return [1.0]
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in np.asarray(parent, dtype=np.float64)]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
