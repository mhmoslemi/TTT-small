import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(512, dtype=np.float64)
    if p.size == 0:
        p = np.ones(512, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if float(p.sum()) < 1e-12:
        p = np.ones(512, dtype=np.float64)
    return p

def _fft_conv_self(a):
    n = a.size
    size = next_fast_len(2 * n - 1)
    F = rfft(a, size)
    c = irfft(F * F, size)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _fft_cross(p, w):
    """Cross-correlation: q_j = sum_k w_k * p_{k-j} (valid part)"""
    n = p.size
    L = 2 * n - 1  # length of p*p
    size = next_fast_len(L)
    F = rfft(p[::-1], size)
    G = rfft(w, size)
    c = irfft(F * G, size)[:L]
    q = c[n-1:2*n-1]
    np.maximum(q, 0.0, out=q)
    return q

def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0/u.size)
    return e / s

def _score_fft(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _fft_conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))

def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)

def _lse_optimize(p, stop_time, consider, seed=42):
    """LSE-smoothed max optimization via L-BFGS-B on log-softmax parameters."""
    n = p.size
    p = np.maximum(p, 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    
    u = np.log(np.maximum(p, 1e-300))
    u -= float(np.mean(u))
    
    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)
    conv_len = 2 * n - 1
    cross_len = 3 * n - 2
    
    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c
    
    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)
        c = irfft(F * G, cross_size)[:cross_len]
        q = c[n-1:2*n-1]
        np.maximum(q, 0.0, out=q)
        return q
    
    def lse_obj(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax < 0:
            return 1e100, np.zeros_like(uu)
        if T <= 0 or not math.isfinite(T):
            T = max(1e-300, 1e-4 * cmax)
        scaled = (c - cmax) / T
        scaled = np.clip(scaled, -400.0, 400.0)
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * float(np.log(sum_exp / c.size))
            if not math.isfinite(f):
                f = cmax
            w = exps / sum_exp
        q = xfast(pp, w)
        gdp = 2.0 * q
        tdp = float(np.dot(pp, gdp))
        grad = pp * (gdp - tdp)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad
    
    best_u = u.copy()
    # Check time budget
    avail = stop_time - time.monotonic() - 3.0
    if avail <= 5:
        return p
    
    # Multi-stage temperature annealing
    c0 = cfast(p)
    cmax0 = float(c0.max())

        (0.02, 0.15),
        (0.008, 0.15),

        (0.001, 0.15),
        (0.0003, 0.15),
        (0.0001, 0.15),

                        a_gs = b_gs

def _get_parent():

        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:

def _fft_conv_self(a):
    n = a.size
    size = next_fast_len(2 * n - 1)
    F = rfft(a, size)
    c = irfft(F * F, size)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _fft_cross(p, w):
    n = p.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(p[::-1], size)
    G = rfft(w, size)
    c = irfft(F * G, size)[:L]
    q = c[n - 1:2 * n - 1]
    np.maximum(q, 0.0, out=q)
    return q


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0:
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _score_fft(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _fft_conv_self(a)
    m = float(c.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))

    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)

        return math.inf
    b = np.convolve(a, a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))

    conv_size = next_fast_len(2 * n - 1)
    cross_size = next_fast_len(3 * n - 2)

    cross_len = 3 * n - 2

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        F = rfft(pp[::-1], cross_size)
        G = rfft(ww, cross_size)

def _coord_descent(p, stop_time, consider, seed=123, G=20):

        return p

    s_total = 1.0

    Ltot = 2 * n - 1
    c = _fft_conv_self(p)
    cur_score = float(2.0 * n * c.max())

    flat_count = 0
    sweep_num = 0

    while time.monotonic() < stop_time - 1.0:

        for idx_i in order:
            i = int(idx_i)
            if time.monotonic() > stop_time - 0.3:
                break

            S = float(s_total)
            v = float(p[i])

            start = i
            end = min(i + n, Ltot)
            Ls = end - start
            if Ls <= 0:
                continue

            base = c[start:end].copy()
            infl = 2.0 * p[:Ls].copy()

            max_const = 0.0

                mc = float(c[:start].max())
                if mc > max_const:
                    max_const = mc
            if end < Ltot:

            x_hi = max(30.0 * v, 80.0 / n, 1e-10)

            pts = np.linspace(0.0, x_hi, G + 1)

            ws = pts - v
            denoms = S + ws
            valid = denoms > 1e-14
            if not np.any(valid):

            di_mod = i % Ls

            aff = np.maximum(aff, max_const)
            maxes = aff.max(axis=1)
            vals = np.full(pts.shape, math.inf)

            imin = int(np.argmin(vals))
            best_x = float(pts[imin])

            if best_val < cur_score - 1e-13:

                lo = max(0.0, best_x - 2.0 * dw)
                hi = min(x_hi, best_x + 2.0 * dw)

                def eval_x(xv):

                    dv = base[di_mod] + w * infl[di_mod] + w * w
                    aff_l = np.maximum(aff_l, dv)
                    aff_l = np.maximum(aff_l, max_const)
                    mv = float(aff_l.max())
                    return (2.0 * n * mv) / (denom * denom)

                gr = (math.sqrt(5.0) - 1.0) / 2.0
                a_gs = hi - gr * (hi - lo)
                b_gs = lo + gr * (hi - lo)
                fa = eval_x(a_gs)
                fb = eval_x(b_gs)
                for _ in range(12):
                    if hi - lo < 1e-15:

                    if not math.isfinite(fa):

                    if not math.isfinite(fb):
                        fb = 1e100
                    if fa < fb:

                        b_gs = a_gs
                        fb = fa
                        a_gs = hi - gr * (hi - lo)
                        fa = eval_x(a_gs)
                    else:
                        lo = a_gs
                        a_gs = b_gs
                        fa = fb
                        b_gs = lo + gr * (hi - lo)

                    c[start:end] += w * infl

    return p

        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        grad = np.zeros(n)
        for k in active[:50]:
            kj = int(k) - np.arange(n)
            valid = (kj >= 0) & (kj < n)
            if np.any(valid):
                contrib = p[kj[valid]] + p[valid]
                grad[valid] += contrib
        grad /= float(len(active))

        step = base_step / (1.0 + 0.05 * it)
        if step < min_step:
            break

        improved_step = False
        for scale in (1.0, 0.5, 0.25, 0.1, 0.05):
            p_cand = p - scale * step * grad
            p_cand = np.maximum(p_cand, 0.0)
            ss = float(p_cand.sum())
            if ss < 1e-15:
                continue
            p_cand /= ss
            sc = float(2.0 * n * _fft_conv_self(p_cand).max())
            if math.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_p = p_cand.copy()
                p = p_cand.copy()
                consider(p)
                improved_step = True
                break
        if not improved_step:
            base_step *= 0.85
            if base_step < 1e-12:
                break

    consider(best_p)
    return best_p


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(10.0, float(budget_s))
    hard_deadline = t0 + budget - 18.0

    parent = _get_parent()
    best_arr = parent.copy()
    best_score = _score_exact(best_arr)

    if not math.isfinite(best_score) or best_arr.size == 0:
        best_arr = np.ones(256, dtype=np.float64) / 256.0
        best_score = _score_exact(best_arr)

    def consider(arr):
        nonlocal best_arr, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).ravel()
        except Exception:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = float(_score_exact(a))
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_arr = (a / s).copy()
            best_score = sc

    now = time.monotonic()
    total_avail = max(1.0, hard_deadline - now)

    # Phase 1: LSE optimization (fast for n=3072)
    p1_end = min(hard_deadline, now + 0.12 * total_avail)
    if p1_end > now + 15.0:
        try:
            p1 = _lse_optimize(best_arr.copy(), p1_end, consider, seed=seed)
            if p1 is not None and p1.size == best_arr.size:
                consider(p1)
        except Exception:
            pass

    # Phase 2: Coordinate descent (main workhorse)
    now = time.monotonic()
    p2_end = min(hard_deadline, now + 0.38 * total_avail)
    if p2_end > now + 20.0:
        try:
            p2 = _coord_descent(best_arr.copy(), p2_end, consider, seed=seed + 7, G=20)
            if p2 is not None:
                consider(p2)
        except Exception:
            pass

    # Phase 3: LSE fine-tune
    now = time.monotonic()
    p3_end = min(hard_deadline, now + 0.12 * total_avail)
    if p3_end > now + 10.0:
        try:
            p3 = _lse_optimize(best_arr.copy(), p3_end, consider, seed=seed + 13)
            if p3 is not None and p3.size == best_arr.size:
                consider(p3)
        except Exception:
            pass

    # Phase 4: Subgradient polish
    now = time.monotonic()
    p4_end = min(hard_deadline, now + 0.10 * total_avail)
    if p4_end > now + 10.0:
        try:
            p4 = _subgradient_polish(best_arr.copy(), p4_end, consider, max_iters=5000)
            if p4 is not None:
                consider(p4)
        except Exception:
            pass

    # Phase 5: Final coordinate descent
    now = time.monotonic()
    p5_end = min(hard_deadline, now + 0.25 * total_avail)
    if p5_end > now + 10.0:
        try:
            p5 = _coord_descent(best_arr.copy(), p5_end, consider, seed=seed + 99, G=24)
            if p5 is not None:
                consider(p5)
        except Exception:
            pass

    # Final validation
    try:
        out = np.asarray(best_arr, dtype=np.float64).ravel()
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        if out.size == 0 or float(out.sum()) < 0.01:
            out = parent
            out = np.where(np.isfinite(out), out, 0.0)
            out = np.clip(out, 0.0, 1000.0)
            if out.size == 0 or float(out.sum()) < 0.01:
                out = np.ones(64) / 64.0
        ls = [float(x) for x in out]
        final_sc = float(evaluate_sequence(ls))
        if not math.isfinite(final_sc):
            ls = [float(x) for x in parent]
            final_sc = float(evaluate_sequence(ls))
            if not math.isfinite(final_sc):
                return [1.0]
        return ls
    except Exception:
        return [1.0]
