def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(3072, dtype=np.float64)
    if p.size == 0:
        p = np.ones(3072, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s < 1e-12:
        p = np.ones(3072, dtype=np.float64)
    else:

    return p

def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(c, 0.0)

def _score(p):
    n = p.size
    c = _conv_self(p)
    m = float(c.max())
    return 2.0 * n * m

def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0/u.size)
    return e / s

def _lse_optimize(p, stop_time, fine_only=False):
    n = int(p.size)
    p = np.maximum(p, 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s
    
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)
    
    def cfast(pp):
        F = rfft(pp, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return np.maximum(c, 0.0)
    
    def lse_obj(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if cmax <= 0 or not math.isfinite(cmax):
            return 1e100, np.zeros_like(uu)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[np.argmax(c)] = 1.0
            f = cmax
        else:
            f = cmax + T * math.log(sum_exp / c.size)
            w = exps / sum_exp
        
        # Gradient: 2 * cross_correlation(w, pp)[:n]
        W = rfft(w, cross_size)
        P = rfft(pp, cross_size)
        cross = irfft(np.conj(W) * P, cross_size)
        g_p = 2.0 * cross[:n]
        
        tdp = float(np.dot(pp, g_p))
        grad = pp * (g_p - tdp)
        return float(f), np.where(np.isfinite(grad), grad, 0.0)
    
    def p_to_u(pp):
        ll = np.log(np.maximum(pp, 1e-300))
        return ll - float(np.mean(ll))
    
    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score(p)
    
    if fine_only:
        stages = [(0.003, 0.20), (0.001, 0.25), (0.0003, 0.25), (0.0001, 0.30)]
    else:
        stages = [
            (0.15, 0.08),
            (0.05, 0.12),
            (0.02, 0.12),
            (0.008, 0.15),
            (0.003, 0.15),
            (0.001, 0.15),
            (0.0003, 0.15),
            (0.0001, 0.18),
        ]
    
    total_w = sum(w for _, w in stages)
    
    for frac, wt in stages:
        now = time.monotonic()
        if now > stop_time - 5.0:
            break
        cmax_cur = float(cfast(p).max())
        T = max(1e-18, frac * cmax_cur)
        stage_time = (wt / total_w) * (stop_time - 5.0 - now)
        end_stage = time.monotonic() + max(3.0, stage_time)
        
        stale = 0
        while time.monotonic() < end_stage - 1.0 and stale < 4:
            try:
                res = minimize(lse_obj, u, args=(T,), method='L-BFGS-B', jac=True,
                             options={'maxiter': 60, 'gtol': 1e-14, 'ftol': 1e-16,
                                     'maxdisp': 0, 'disp': False, 'maxls': 40, 'maxcor': 30})
            except Exception:
                break
            if res.x is not None and np.all(np.isfinite(res.x)):
                p_cur = _softmax(res.x)
                sc_cur = _score(p_cur)
                if math.isfinite(sc_cur) and sc_cur < best_sc - 1e-15:
                    best_sc = sc_cur
                    best_p = p_cur.copy()
                    p = p_cur.copy()
                    u = p_to_u(p)
                    stale = 0
                else:
                    stale += 1
                    if stale >= 4:
                        break
            if getattr(res, 'status', 1) == 0:
                break
    
    ss = float(best_p.sum())
    return best_p / ss if ss > 1e-300 else best_p

                w = xs - v
                denom = S + w
                valid = denom > 1e-14
                aff = base[None, :] + w[:, None] * infl[None, :]
                dvals = base[i - start] + w * infl[i - start] + w * w
                aff = np.maximum(aff, dvals[:, None])
                aff = np.maximum(aff, max_const)
                mv = np.max(aff, axis=1)
                out = np.full(xs.shape, math.inf)
                dv = denom[valid]

                return out
            
            vals = eval_cand(pts)
            if not np.any(np.isfinite(vals)):
                continue
            imin = int(np.argmin(vals))
            bx = float(pts[imin])
            bv = float(vals[imin])
            
            # Fine search: parabolic interpolation + golden section
            im = imin
            if 0 < im < G:
                x1, x2, x3 = pts[im-1], pts[im], pts[im+1]
                f1, f2, f3 = float(vals[im-1]), float(vals[im]), float(vals[im+1])
                if f1 != f2 and f2 != f3 and math.isfinite(f1) and math.isfinite(f3):
                    # Parabolic vertex
                    A = (f1 - f3) / (x1 - x3) if x1 != x3 else 0
                    B = (f3 - f2) / (x3 - x2) if x3 != x2 else 0
                    denom = A - B
                    if abs(denom) > 1e-30:
                        xv = 0.5 * (x1 + x3 + (f1 - f3) / denom)
                        if 0 <= xv <= x_hi:
                            fv = float(eval_cand(np.array([xv]))[0])

            # Local golden section refinement
            if math.isfinite(bv):

                lo = max(0.0, bx - 2.0 * dw)
                hi = min(x_hi, bx + 2.0 * dw)

                        f2 = float(eval_cand(np.array([x2]))[0])
                        if f1 < f2:
                            hi = x2
                        else:
                            lo = x1
                        xm = (lo + hi) / 2
                        fm = float(eval_cand(np.array([xm]))[0])
                        if fm < bv:
                            bx, bv = xm, fm
                
            if math.isfinite(bv) and bv < cur_score - 1e-14 and 0.0 <= bx:
                w = bx - v

                    c[start:end] += w * infl

                    cur_score = bv
                    improved = True

            c *= inv * inv

        new_sc = 2.0 * n * float(c.max())

import math

from scipy.optimize import minimize


def _get_parent():

        h = globals()["height_sequence_1"]

def _conv_self(a):
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return np.maximum(np.asarray(c, dtype=np.float64), 0.0)


def _score(p):
    n = p.size
    c = _conv_self(p)
    m = float(c.max())
    return 2.0 * n * m


def _softmax(u):
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.full(u.size, 1.0 / u.size)
    return e / s


def _lse_optimize(p, stop_time, fine_only=False):
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0)
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return p.copy()
    p = p / s

    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1
    cross_size = next_fast_len(3 * n - 2)

    def cfast(pp):
        F = rfft(pp, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        return np.maximum(c, 0.0)

    def lse_obj(uu, T):
        pp = _softmax(uu)
        c = cfast(pp)
        cmax = float(c.max())
        if cmax <= 0 or not math.isfinite(cmax):
            return 1e100, np.zeros_like(uu)
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
            f = cmax
        else:
            f = cmax + T * math.log(sum_exp / c.size)
            w = exps / sum_exp

        W = rfft(w, cross_size)
        P = rfft(pp, cross_size)
        cross = irfft(np.conj(W) * P, cross_size)
        g_p = 2.0 * cross[:n]

        tdp = float(np.dot(pp, g_p))
        grad = pp * (g_p - tdp)
        return float(f), np.where(np.isfinite(grad), grad, 0.0)

    def p_to_u(pp):
        ll = np.log(np.maximum(pp, 1e-300))
        return ll - float(np.mean(ll))

    u = p_to_u(p)
    best_p = p.copy()
    best_sc = _score(p)

    if fine_only:
        stages = [(0.002, 0.20), (0.0008, 0.25), (0.0003, 0.25), (0.0001, 0.30)]
    else:

            (0.05, 0.12),
            (0.02, 0.12),
            (0.008, 0.14),
            (0.003, 0.14),
            (0.001, 0.14),
            (0.0004, 0.14),
            (0.00015, 0.16),

            break
        cmax_cur = float(cfast(p).max())

                                        'maxls': 50, 'maxcor': 30})

                break

                sc_cur = _score(p_cur)

                    best_sc = sc_cur
                    best_p = p_cur.copy()
                    p = p_cur.copy()
                    u = p_to_u(p)
                    stale = 0
                else:
                    stale += 1
            if getattr(res, 'status', 1) == 0:
                break

    ss = float(best_p.sum())

def _try_alternative_init(n, seed=42):
    """Try a few structured initializations and return the best one."""
    candidates = []

    # Raised cosine
    j = np.arange(n, dtype=np.float64)
    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * j / (n - 1)))
    rc = rc / rc.sum()
    candidates.append(rc)

    # Quadratic (parabolic)
    t = 2.0 * j / (n - 1) - 1.0
    quad = 1.0 - t * t
    quad = np.maximum(quad, 0)
    if quad.sum() > 0:
        quad = quad / quad.sum()
        candidates.append(quad)

    # One-sided exponential
    tau = n / 3.0
    exp1 = np.exp(-j / tau)
    exp1 = exp1 / exp1.sum()
    candidates.append(exp1)

    # Symmetric two-sided (U-shaped)
    mid = (n - 1) / 2.0
    u_shape = 0.3 + 0.7 * ((j - mid) / mid) ** 2
    u_shape = u_shape / u_shape.sum()
    candidates.append(u_shape)

    best = None
    best_sc = math.inf
    for cand in candidates:
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best = cand.copy()
    return best
