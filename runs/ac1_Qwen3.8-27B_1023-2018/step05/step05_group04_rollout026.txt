import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _score(p):
    """Score assuming sum(p) = 1. Returns 2*n*max(conv(p,p))."""
    n = p.size
    if n == 0:
        return float('inf')
    s = p.sum()
    if s < 1e-30:
        return float('inf')
    p = p / s
    c = _conv(p)
    return 2.0 * n * c.max()

def _conv(p):
    n = p.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(p, size)
    c = irfft(F * F, size)[:L]
    return c

import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _conv_fast(a):
    n = a.size
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    return c


def _score_norm(p):
    """Score for a normalized (sum=1) sequence: 2*n*max(conv)."""
    n = p.size
    if n == 0:
        return float('inf')
    s = float(p.sum())
    if s < 1e-30:
        return float('inf')
    if abs(s - 1.0) > 1e-10:
        p = p / s
    c = _conv_fast(p)
    m = float(c.max())
    if not math.isfinite(m):
        return float('inf')
    return 2.0 * n * m

def _resample_to(src, n_target):
    src = np.asarray(src, dtype=np.float64).ravel()
    n_target = int(max(1, n_target))
    if src.size == 0:
        return np.full(n_target, 1.0 / n_target)
    if src.size == n_target:
        out = src.copy()
    else:
        m = src.size
        xp = (np.arange(m) + 0.5) / m
        fp = np.interp((np.arange(n_target) + 0.5) / n_target, xp, src)
        out = fp
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s < 1e-300:
        return np.full(n_target, 1.0 / n_target)
    return out / s


def _spectral_perturb(p, rng, strength=0.3):
    """Perturb via spectral amplitude compression."""
    n = p.size
    size = next_fast_len(n)
    F = rfft(p, size)
    amp = np.abs(F)
    phase = np.angle(F)
    # Compress amplitude: new_amp = amp^alpha / sum(amp^alpha) * sum(amp)
    alpha = 1.0 - strength  # strength in [0,1), lower alpha = more compression
    alpha = max(0.1, min(0.95, alpha))
    amp_new = np.power(np.maximum(amp, 1e-30), alpha)
    # Blend
    amp_blend = (1.0 - strength) * amp + strength * amp_new * (amp.sum() / (amp_new.sum() + 1e-30))
    F_new = amp_blend * np.exp(1j * phase)
    p_new = irfft(F_new, size)[:n]
    p_new = np.maximum(p_new, 0.0)
    s = float(p_new.sum())
    if s > 1e-300:
        p_new = p_new / s
    else:
        p_new = p.copy()
    return p_new


def _smooth_perturb(p, rng, sigma=0.03):
    """Add a smooth random perturbation."""
    n = p.size
    # Generate using a few random sinusoids at low frequencies
    n_modes = 8
    freqs = rng.uniform(0.0, 1.5, size=n_modes)
    phases = rng.uniform(0.0, 2.0 * np.pi, size=n_modes)
    amps = rng.normal(0.0, 1.0, size=n_modes)
    x = np.arange(n, dtype=np.float64)
    sig = np.zeros(n)
    for f, ph, a in zip(freqs, phases, amps):
        sig += a * np.cos(2.0 * np.pi * f * x / n + ph)
    std_sig = float(np.std(sig))
    if std_sig > 1e-10:

    p_new = p + sigma * sig
    p_new = np.maximum(p_new, 0.0)
    s = float(p_new.sum())
    if s > 1e-300:
        p_new = p_new / s
    else:
        p_new = p.copy()
    return p_new

                return [1.0]
        return ls
    except Exception:
        return [1.0]
