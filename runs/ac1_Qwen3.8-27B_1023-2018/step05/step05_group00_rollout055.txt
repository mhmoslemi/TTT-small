import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def _conv_fft(a):
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_fast_len(L)
    F = rfft(a, size)
    c = irfft(F * F, size)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return math.inf
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    c = _conv_fft(a)
    m = float(c.max())
    if not math.isfinite(m) or m < 0:
        return math.inf
    return float(2.0 * n * m / (s * s))

                    p /= ss
                    s = 1.0
                conv = _conv_fft(p)
                cur_score = float(2.0 * n * conv.max())
                if math.isfinite(cur_score) and cur_score < best_score:
                    best_score = cur_score
                    best_p = p.copy()
                flat_count = 0
                if time.monotonic() > stop_time - 10:
                    break

        sweep += 1

    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss > 1e-300:
        p /= ss
    return best_p


def _lse_refine(p, stop_time, consider=None):
    """LSE-based gradient refinement using L-BFGS-B."""
    n = int(p.size)
    p = np.maximum(np.asarray(p, dtype=np.float64).ravel(), 0.0).copy()
    s = float(p.sum())
    if s < 1e-300:
        return p
    p = p / s

    def p_to_u(pp):
        uu = np.log(np.maximum(pp, 1e-300))
        return uu - float(np.mean(uu))

    u = p_to_u(p)
    conv_size = next_fast_len(2 * n - 1)
    conv_len = 2 * n - 1

    def cfast(a):
        F = rfft(a, conv_size)
        c = irfft(F * F, conv_size)[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def xfast(pp, ww):
        # Cross-convolution of reversed pp with ww, take the relevant part
        F = rfft(pp[::-1], conv_size)
        G = rfft(ww, conv_size)
        c = irfft(F * G, conv_size)[:conv_len]
        q = c[n - 1:2 * n - 1]
        np.maximum(q, 0.0, out=q)
        return q

    def lse_func(uu, T):
        pp = np.exp(uu - float(np.max(uu)))
        ss = float(pp.sum())
        pp = pp / ss
        c = cfast(pp)
        cmax = float(c.max())
        if not math.isfinite(cmax) or cmax <= 0:
            return 1e100, np.zeros_like(uu)
        if T < 1e-300:
            T = 1e-6
        scaled = (c - cmax) / T
        exps = np.exp(scaled)
        sum_exp = float(exps.sum())
        if not math.isfinite(sum_exp) or sum_exp <= 0:
            w_vec = np.zeros(c.size)
            w_vec[int(np.argmax(c))] = 1.0
        else:
            f = cmax + T * math.log(sum_exp / c.size)
            w_vec = exps / sum_exp
        q = xfast(pp, w_vec)
        grad = 2.0 * pp * q
        grad = grad - float(np.dot(pp, grad)) * pp
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(f), grad

    best_u = u.copy()
    stages = [
        (0.01, 0.12),
        (0.003, 0.18),
        (0.0008, 0.25),
        (0.0002, 0.3),
        (0.00005, 0.15),
    ]

    total_w = sum(w for _, w in stages)
    start_t = time.monotonic()
    avail = stop_time - start_t - 8.0
    if avail < 10:
        return p

    cum = 0.0
    for frac, wt in stages:
        if time.monotonic() > stop_time - 5:
            break
        cum += wt
        end_stage = min(stop_time - 4.0, start_t + avail * (cum / total_w))
        if end_stage <= time.monotonic() + 5:
            continue

        c = cfast(p)
        cmax = float(c.max()) if c.size else 1.0
        T = max(1e-300, frac * cmax)

        stale = 0
        while time.monotonic() < end_stage - 1.0:
            opts = {"maxiter": 25, "gtol": 1e-12, "ftol": 1e-15,
                    "maxls": 20, "maxcor": 30, "maxdisp": 0, "disp": False}
            try:
                res = minimize(lse_func, best_u, args=(T,),
                               method="L-BFGS-B", jac=True, options=opts)
            except Exception:
                break
            if res.x is not None and res.x.size == n:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                if np.all(np.isfinite(x)):
                    fval, _ = lse_func(x, T)
                    f0, _ = lse_func(best_u, T)
                    if math.isfinite(fval) and fval < f0 - 1e-12:
                        best_u = x.copy()
                        stale = 0
                    else:
                        stale += 1
                        if stale >= 3:
                            break
                if getattr(res, "status", 1) == 0:
                    break
            else:
                break

        p = np.exp(best_u - float(np.max(best_u)))
        ps = float(p.sum())
        if ps > 0:
            p /= ps

    p = np.maximum(p, 0.0)
    ps = float(p.sum())
    if ps > 0:
        p /= ps
    return p
