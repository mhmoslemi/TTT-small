def _fft_convolve(a, fsize=None):
    n = len(a)
    if fsize is None:
        fsize = next_fast_len(2*n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    c = np.maximum(c, 0)
    return c

def _score_fast(a):
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _fft_convolve(a)
    return 2.0 * n * float(b.max()) / (s * s)

def _lp_direction(a, M, verbose=False):
    """Find b >= 0 maximizing sum(b) subject to (a*b)_k <= M for all k."""
    n = len(a)
    # Build constraint matrix
    C = np.zeros((2*n-1, n))
    for i in range(n):
        k_lo = max(0, 0)  # k-i >= 0 => k >= i
        k_hi = min(2*n-2, i+n-1)  # k-i <= n-1 => k <= i+n-1
        for k in range(i, k_hi+1):
            C[k, i] = a[k-i]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def _lp_refine(a, stop_time, max_lp_iters=100, t_start=0.1):
    """Iteratively apply LP-based refinement."""
    n = len(a)
    a = np.clip(a, 0, 1000).astype(np.float64).copy()
    best = a.copy()
    best_score = _score_fast(a)
    t = t_start
    
    for it in range(max_lp_iters):
        if time.monotonic() > stop_time:
            break
        
        b_conv = _fft_convolve(a)
        M = float(b_conv.max())
        S = float(a.sum())
        
        b = _lp_direction(a, M)
        if b is None:
            break
        
        b = np.clip(b, 0, 1000)
        b_sum = float(b.sum())
        if b_sum < 1e-10:
            break
        
        # Normalize b to have same sum as a
        b_norm = b * (S / b_sum)
        
        # Try different step sizes
        improved = False
        for t_candidate in [t, 0.5*t, 0.25*t]:
            a_new = (1-t_candidate)*a + t_candidate*b_norm
            a_new = np.clip(a_new, 0, 1000)
            s_new = float(a_new.sum())
            if s_new < 0.01:
                continue
            score_new = _score_fast(a_new)
            if score_new < best_score - 1e-12:
                a = a_new
                best = a.copy()
                best_score = score_new
                t = min(t_candidate * 1.3, 0.4)
                improved = True
                break
            elif score_new < best_score:
                # Accept even small improvements
                a = a_new
                best = a.copy()
                best_score = score_new
                t = t_candidate
                improved = True
                break
        
        if not improved:
            t *= 0.5
            if t < 0.001:
                break
    
    return best, best_score

    if not np.isfinite(m) or m < 0:
        return math.inf
    return 2.0 * n * m / (s * s)

def _grad_z(a, b, n):
    """Exact subgradient of 2n*max(b) w.r.t. softmax parameter z."""
    M = float(b.max())
    tol = max(1e-10, M * 1e-5)
    K = np.where(b >= M - tol)[0]
    if K.size == 0:
        K = np.array([int(np.argmax(b))])
    if K.size > 64:
        K = np.argpartition(b, -64)[-64:]
    
    s = np.zeros(n)
    for k in K:
        k = int(k)
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if hi1 >= lo1:
            s[lo1:hi1 + 1] += a[k - hi1:k - lo1 + 1][::-1]
        lo2 = max(k, 0)
        hi2 = min(n - 1, k + n - 1)
        if hi2 >= lo2:
            s[lo2:hi2 + 1] += a[:hi2 - lo2 + 1]
    s /= float(K.size)
    
    s_proj = float(np.dot(a, s))
    return 2.0 * n * a * (s - s_proj)

def _lp_solve(a, M, n):
    """LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0."""
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_lo = i
        k_hi = min(2 * n - 2, i + n - 1)
        C[k_lo:k_hi + 1, i] = a[:k_hi + 1 - i]
    try:
        res = linprog(-np.ones(n), A_ub=C, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.status == 0 and np.isfinite(res.x).all():
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None
