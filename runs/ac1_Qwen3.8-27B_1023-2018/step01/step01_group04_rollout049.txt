import numpy as np
import time
import math
from scipy.optimize import linprog
import scipy.sparse as sps

def _score(a):
    """Fast score computation."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, None)
    n = a.size
    S = a.sum()
    if S < 1e-15:
        return 1e10
    B = np.convolve(a, a)
    M = B.max()
    return 2.0 * n * M / (S * S)

def _lp_step(a, M, n):
    """Solve LP to find best response g."""
    # conv(a, g)[k] = sum_j a[k-j] * g[j] (for valid k-j)
    # Matrix: A[k, j] = a[k-j] if 0<=k-j<n else 0
    # Shape: (2n-1, n)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            rows.append(k)
            cols.append(j)
            data.append(a[k-j])
    A = sps.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
