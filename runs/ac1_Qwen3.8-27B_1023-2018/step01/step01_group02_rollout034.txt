∂c[2]/∂a[2] = 2*a[2-2] = 2*a[0]. ✓

Good. So for the sharp subgradient at lag k*:
g[i] = 2 * a[k*-i] if 0 <= k*-i <= n-1, else 0.

This is 2 times the "reflected" version of a. If k* = n-1, it's 2*a[::-1].
If k* = 0, it's 2*[a[0], 0, 0, ..., 0].
If k* = n, it's 2*[0, a[0], a[1], ..., a[n-1]] but truncated to [0, n-1], so 2*[0, a[0], ..., a[n-2]].

For the smoothed version with weights p[k]:
g[i] = 2 * sum_k p[k] * a[k-i] (when 0 <= k-i <= n-1)
= 2 * (p_conv[i]) where p_conv is the cross-correlation.

Now, for the FFT-based computation:
sum_k p[k] * a[k-i] for k=0,...,2n-2, and we want this for i=0,...,n-1.

Let's define the cross-correlation: for i = 0, ..., n-1:
corr[i] = sum_{k=max(0,i)}^{min(2n-2, i+n-1)} p[k] * a[k-i]

This is the cross-correlation of p (length 2n-1) and a (length n) at "lag" i, but with a specific convention.

Alternative: corr[i] = sum_k p[k] * a[k-i] = (p * a_flip)[i + n - 1] where a_flip = a[::-1].

Let me verify: (p * a_flip)[j] = sum_k p[k] * a_flip[j-k] = sum_k p[k] * a[n-1-(j-k)] = sum_k p[k] * a[n-1-j+k].
Setting j = i + n - 1: sum_k p[k] * a[n-1-(i+n-1)+k] = sum_k p[k] * a[k-i]. ✓

So g[i] = 2 * (p * a_flip)[i + n - 1] for i = 0, ..., n-1.

a_flip = a[::-1]
L = next_power_of_2(len(p) + n - 1)
conv = ifft(rfft(p, L) * rfft(a_flip, L))
g = 2.0 * conv[n-1 : n-1+n]

import numpy as np
import time


def _score_fast(a):
    """Compute score for a nonnegative array (assumes sum ~ 1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    c = np.convolve(a, a)
    mx = float(c.max())
    return float(2.0 * n * mx / (s * s))


def _proj_simplex(x):
    """Project x onto the probability simplex."""
    x = np.asarray(x, dtype=np.float64).copy()
    n = x.size
    if n == 1:
        return np.array([max(0.0, float(x[0]))])
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(x - theta, 0.0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return w / s


def _subgradient_fft(a, p):
    """Compute 2 * sum_k p[k] * a[k-i] for i=0..n-1 using FFT."""
    n = a.size
    a_flip = a[::-1]
    L = 1 << (len(p) + n - 1 - 1).bit_length()
    P = np.fft.rfft(p, L)
    A = np.fft.rfft(a_flip, L)
    conv = np.fft.irfft(P * A, L)
    return 2.0 * conv[n - 1:n - 1 + n]


def _soft_subgradient(a, beta):
    """Compute smoothed subgradient of max(a*a)."""
    n = a.size
    c = np.convolve(a, a)
    cmax = c.max()
    z = beta * (c - cmax)
    z = np.minimum(z, 0.0)
    p = np.exp(z)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p /= ps
    g = _subgradient_fft(a, p)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(a)
    return g


def _hard_subgradient(a):
    """Compute hard subgradient (single argmax lag)."""
    n = a.size
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    # g[i] = 2 * a[k_star - i] if 0 <= k_star - i <= n-1, else 0
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[k_star - idx]
    return g


def _local_polish(a, rng, n_trials=50, time_limit=5.0):
    """Local search: random perturbations on the simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    best = a.copy()
    best_sc = _score_fast(best)
    t_start = time.monotonic()

    for _ in range(n_trials):
        if time.monotonic() - t_start > time_limit:
            break
        pert = rng.standard_normal(a.size) * 0.08
        cand = _proj_simplex(best + pert)
        sc = _score_fast(cand)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best = cand.copy()
    return best, best_sc


def _optimize_subgrad(a_init, start_time, end_time, rng):
    """Main subgradient optimization loop with projection."""
    a = _proj_simplex(a_init.copy())
    best = a.copy()
    best_sc = _score_fast(best)

    n = a.size
    momentum = np.zeros(n)
    beta = 30.0
    lr = 0.02
    beta_growth = 0.003
    total_time = max(0.1, end_time - start_time)
    elapsed_frac = 0.0

    while time.monotonic() < end_time - 0.5:
        elapsed_frac = (time.monotonic() - start_time) / total_time
        beta = 30.0 + 200.0 * min(1.0, elapsed_frac * 3.0)

        g = _soft_subgradient(a, beta)

        # Polyak-like step size
        lum = float(np.linalg.norm(g))
        if lum < 1e-10:
            lum = 1e-10
        lr = 0.03 / (1.0 + elapsed_frac * 2.0)

        momentum = 0.85 * momentum + 0.15 * g
        a_new = a - lr * momentum
        a = _proj_simplex(a_new)

        sc = _score_fast(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()

        # Periodic restart
        if np.random.random() < 0.02:
            a = _proj_simplex(best + 0.06 * rng.standard_normal(n))
            momentum = np.zeros(n)
            best_p, best_sc_p = _local_polish(best, rng, n_trials=10, time_limit=1.0)
            if best_sc_p < best_sc:
                best = best_p
                best_sc = best_sc_p
                a = best_p.copy()

    # Final polish
    if time.monotonic() < end_time - 2.0:
        polish_time = min(10.0, end_time - 2.0 - time.monotonic())
        if polish_time > 0.5:
            best, best_sc = _local_polish(best, rng, n_trials=200, time_limit=polish_time)

    return best, best_sc


def _make_inits(m, rng, parent=None):
    """Generate diverse initializations of length m."""
    inits = []

    def add(x):
        try:
            a = np.asarray(x, dtype=np.float64)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-12:
                a = a / s
                if a.size == m and np.all(np.isfinite(a)):
                    inits.append(a.copy())
        except Exception:
            pass

    # Parent (up/downsampled)
    if parent is not None and parent.size > 0:
        if parent.size == m:
            add(parent)
            add(parent + 0.05 * rng.standard_normal(m))
            add(parent[::-1])
        else:
            x0 = np.linspace(0, 1, parent.size)
            x1 = np.linspace(0, 1, m)
            up = np.interp(x1, x0, parent)
            add(up)
            add(up + 0.05 * rng.standard_normal(m))

    # Uniform
    add(np.ones(m) / m)

    # Exponential decay
    for rate in (0.5, 1.0, 1.5, 2.0):
        x = np.exp(-rate * np.linspace(0, 1, m))
        add(x)

    # Power law
    for p in (1.5, 2.0, 2.5, 3.0):
        x = (1.0 - np.linspace(0, 1, m) + 0.01) ** p
        add(x)

    # Linear ramp
    add(np.linspace(0.1, 1.0, m))
    add(np.linspace(1.0, 0.1, m))

    # Parabolic
    x = np.linspace(0, 1, m)
    add(1.0 - (2 * x - 1) ** 2)
    add(1.0 + (2 * x - 1) ** 2)

    # Sine-based
    add(1.0 + 0.5 * np.sin(np.pi * x))
    add(1.0 + 0.5 * np.sin(2 * np.pi * x + 0.5))

    # Random
    for _ in range(3):
        add(rng.standard_normal(m).clip(min=0.01) + 0.1)

    # Gaussian bump
    for center in (0.25, 0.5, 0.75):
        for width in (0.1, 0.2, 0.3):
            add(np.exp(-0.5 * ((x - center) / width) ** 2) + 0.05)

    if not inits:
        inits.append(np.ones(m) / m)
    return inits
