    deadline = t0 + max(5.0, float(budget_s)) - 25.0

    best_list = None
    best_score = float('inf')

    def consider(arr):
        nonlocal best_list, best_score
        try:
            a = np.asarray(arr, dtype=np.float64).flatten()
            if a.size == 0:
                return
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() < 0.01:
                return
            lst = [float(x) for x in a]
            v = float(evaluate_sequence(lst))
            if np.isfinite(v) and v < best_score:
                best_score = v
                best_list = lst
        except Exception:
            pass

    # Try parent and fallbacks
    consider([0.2, 0.4, 0.2, 0.2])
    hs = globals().get('height_sequence_1', None)
    if hs is not None:
        consider(np.asarray(hs, dtype=np.float64))

    # Determine working length and initial vector
    if hs is not None:
        n = len(hs)
        q0 = np.asarray(hs, dtype=np.float64).copy()
    else:
        n = 512
        base = np.array([0.2, 0.4, 0.2, 0.2])
        q0 = np.tile(base, n // 4 + 1)[:n].astype(np.float64)

    q0 = np.clip(q0, 0.0, 1000.0)
    if q0.sum() < 0.01:

    def norm1(v):
        v = np.asarray(v, dtype=np.float64)
        v = np.clip(v, 0.0, 1000.0)
        s = v.sum()
        if s < 1e-12:
            return np.ones(len(v)) / len(v)
        return v / s

    def conv_full(v, w=None):
        if w is None:
            w = v
        return np.convolve(v, w, mode='full')

    def quick_maxb(v):
        """Return max of conv(v,v) assuming sum(v)=1."""
        return float(np.convolve(v, v).max())

    def build_A(qv):
        """(2n-1) x n matrix A where A @ x = conv(qv, x)."""
        nn = len(qv)
        A = np.zeros((2 * nn - 1, nn), dtype=np.float64)
        for j in range(nn):
            A[j:j + nn, j] = qv
        return A

    def lp_dual(qv, tl=12.0):
        """Solve LP: max sum(g) s.t. conv(qv,g) <= M, g >= 0."""
        nn = len(qv)
        b_self = conv_full(qv)
        M = float(b_self.max())
        if M <= 0 or not np.isfinite(M):
            return None
        A = build_A(qv)
        c = -np.ones(nn)
        b_ub = np.full(2 * nn - 1, M)
        bounds = [(0.0, 1e6)] * nn
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': tl, 'presolve': True})
            if res.success and res.x is not None:
                g = np.clip(res.x, 0.0, 1e6)
                gs = g.sum()
                if gs > 0.01 and np.isfinite(gs):
                    return g / gs
        except Exception:
            pass
        return None

    def subgrad_step(qv, alpha, n_active_max=40):
        """One step of projected subgradient descent on max_b."""
        nn = len(qv)
        b = conv_full(qv)
        m = float(b.max())
        if m <= 0 or not np.isfinite(m):
            return qv
        # Select active lags (top values)
        thresh = 0.75 * m
        active = np.where(b >= thresh)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(b))])
        if active.size > n_active_max:
            idx = np.argsort(b[active])[::-1][:n_active_max]
            active = active[idx]
        w = b[active].copy()
        w /= w.sum()

        # Gradient: grad[a] = 2 * sum_k w_k * qv[k-a] for valid (k-a)
        grad = np.zeros(nn)
        for k_val, wv in zip(active, w):
            k = int(k_val)
            lo = max(0, k - nn + 1)
            hi = min(k, nn - 1)
            if lo <= hi:
                # For a in [lo, hi]: contribution = 2*wv*qv[k-a]
                # k-a ranges from k-lo down to k-hi
                grad[lo:hi + 1] += 2.0 * wv * qv[k - hi:k - lo + 1][::-1]

        # Project out uniform direction (since we renormalize)
        grad -= grad.mean()
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:

        cand = qv - (alpha / gn) * grad
        return norm1(cand)

                    q = cand.copy()
                    alpha = min(0.6, alpha * 1.08)

                        break
                    cand = random_kick(q, s)
                    mb = quick_maxb(cand)
                    if mb < best_mb:

                        best_mb = mb
                        best_q = cand.copy()
                stagn = 1

                mb = quick_maxb(cand)
                if mb < best_mb - 1e-13:
                    best_mb = mb
                    best_q = cand.copy()
                    q = cand.copy()
                else:
                    alpha *= 0.85
                    if alpha < 0.01:
                        break

        return best_q, best_mb

    # Phase 1: Optimize from the parent
    if time.monotonic() < deadline - 15:
        phase1_end = min(deadline - 15, t0 + (deadline - t0) * 0.55)
        q_opt, mb_opt = optimize_from(q0, phase1_end)
        consider(q_opt)

    if time.monotonic() < deadline - 12:
        starts = []
        # Gaussian-like
        ctr = n // 2
        t_arr = np.arange(n, dtype=np.float64) - ctr
        sig = n / 6.0

        per_start_time = max(5.0, (deadline - 12 - time.monotonic()) / len(starts))
