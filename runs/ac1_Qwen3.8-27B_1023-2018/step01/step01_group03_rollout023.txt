        q0 = np.ones(n, dtype=np.float64) / float(n)
    q0 = _norm(q0)
    consider(q0)

    # --- Phase 1: LP + polish on parent ---
    if time.monotonic() < deadline - 60.0:
        q1, s1 = _lp_optimize(q0, deadline, max_iters=35, t_limit=6.0, polish_iters=300)
        consider(q1)

    # --- Phase 2: Alternative initializations ---
    if time.monotonic() < deadline - 40.0:
        alts = []

        # Periodic patterns
        for period in (3, 5, 7):
            pat = np.ones(n, dtype=np.float64)
            pat[::period] = 0.2
            pat = _norm(pat)
            alts.append(pat)

        # Smooth random
        for _ in range(2):
            t = np.arange(n, dtype=np.float64)
            sig = np.ones(n, dtype=np.float64)
            for _ in range(6):
                f = rng.integers(1, 25)
                ph = rng.uniform(0, 2 * np.pi)
                am = rng.uniform(0.05, 0.25)
                sig += am * np.cos(2.0 * np.pi * f * t / n + ph)
            sig = np.maximum(sig, 1e-3)
            alts.append(_norm(sig))

        # Concentrated center
        center = np.exp(-0.5 * ((np.arange(n) - n / 2) / (n / 6)) ** 2)
        alts.append(_norm(center + 0.05))

        for i, q_alt in enumerate(alts):
            if time.monotonic() > deadline - 20.0:
                break
            remaining = len(alts) - i
            alloc = min(40.0, max(10.0, (deadline - time.monotonic() - 25.0) / remaining))
            sub_deadline = min(deadline - 20.0, time.monotonic() + alloc)
            q_opt, s_opt = _lp_optimize(q_alt, sub_deadline, max_iters=15, t_limit=4.0, polish_iters=100)
            consider(q_opt)

    # --- Phase 3: Final polish on best ---
    if best_list is not None and time.monotonic() < deadline - 8.0:
        barr = np.asarray(best_list, dtype=np.float64)
        if barr.size == n:
            qf, sf = _polish(barr, deadline, max_iters=300, seed=99)
            consider(qf)
        elif barr.size > 0:
            # Try to polish at its own length
            qf, sf = _polish(barr, deadline, max_iters=200, seed=77)
            consider(qf)

    # --- Fallback ---
    if best_list is None:
        qfb = np.array([0.2, 0.4, 0.2, 0.2], dtype=np.float64)
        consider(qfb)
    if best_list is None:
        best_list = [1.0]

    # --- Final validation ---
    try:
        v = float(evaluate_sequence(best_list))
        if not np.isfinite(v):
            fb = [0.2, 0.4, 0.2, 0.2]
            if np.isfinite(float(evaluate_sequence(fb))):
                best_list = fb
            else:
                best_list = [1.0]
    except Exception:
        best_list = [1.0]

    return [float(x) for x in best_list]
