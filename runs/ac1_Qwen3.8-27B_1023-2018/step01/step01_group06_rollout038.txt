                a = a + 0.1 * a[::-1]
            else:
                # Smooth perturbation
                kernel = np.ones(max(3, n // 500)) / max(3, n // 500)
                a = np.convolve(a, kernel, mode='same')
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

        # Anneal parameters
        if iter_count % 500 == 0:
            K_active = min(1200, K_active + 40)
            lp_every = max(2, lp_every if K_active > 800 else 3)

    # Final validation
    result = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(result))
    if not np.isfinite(final_score):
        result = [1.0] * n
    return result
