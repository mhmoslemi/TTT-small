def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.full(z.size, 1.0 / z.size)
    return e / s


def _fft_conv(a, fsize):
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: 2 * len(a) - 1]
    c = np.maximum(c, 0.0)
    if not np.isfinite(c).all():

def _soft_lse(b, tau):
    m = float(b.max())
    x = (b - m) / tau
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-300:
        U = m
        idx = np.where(b == m)[0]
        p = np.zeros_like(b)
        p[idx] = 1.0 / idx.size
        return U, p
    U = m + tau * math.log(s)
    p = ex / s
    return U, p

def _polish(best_a, best_score, stop_time, seed):
