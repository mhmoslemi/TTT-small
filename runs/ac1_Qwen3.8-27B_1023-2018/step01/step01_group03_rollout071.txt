            h = (1.0 - t) * a + t * g
            h = norm1(h)
            o = quick_obj(h)
            if o < best_o - 1e-15:
                best_o = o
                best_t = t
        return best_t, best_o

    def grad_polish(a, max_iters, end_time):
        a = norm1(a)
        best = a.copy()
        best_o = quick_obj(a)
        lr = 0.06
        stagnant = 0

        for _ in range(max(1, int(max_iters))):
            if time.monotonic() > end_time - 3.0:
                break

            b = np.convolve(a, a)
            peak = float(b.max())
            if peak <= 0 or not np.isfinite(peak):
                break

            threshold = 0.90 * peak
            active = np.where(b >= threshold)[0]
            if active.size == 0:
                break
            if active.size > 25:
                top = active[np.argsort(b[active])[::-1]][:25]
            else:
                top = active

            grad = np.zeros(n, dtype=np.float64)
            for k in top:
                kk = int(k)
                w = (b[kk] - 0.85 * peak) / max(0.15 * peak, 1e-30)
                w = max(0.1, w)
                lo = max(0, kk - n + 1)
                hi = min(n - 1, kk)
                for j in range(lo, hi + 1):
                    if kk == 2 * j:
                        grad[j] += w * a[kk - j]
                    else:
                        grad[j] += 2.0 * w * a[kk - j]

            ga = float(np.dot(grad, a))
            aa = float(np.dot(a, a))
            if aa > 1e-30:
                grad = grad - (ga / aa) * a

            gn = float(np.linalg.norm(grad))
            if gn < 1e-14:
                break

            improved = False
            for scale in [lr, lr * 0.6, lr * 0.3, lr * 0.1]:
                h = a - scale * grad / gn
                h = norm1(h)
                o = quick_obj(h)
                if o < best_o - 1e-14:
                    best = h.copy()
                    best_o = o
                    a = h
                    stagnant = 0
                    improved = True
                    break

            if improved:
                lr = min(0.25, lr * 1.06)
            else:
                lr *= 0.6
                stagnant += 1
                if stagnant >= 12:
                    break

        return best, best_o

    # Main optimization loop
    n_lp = 60
    stagnant_lp = 0

    for it in range(n_lp):
        elapsed = time.monotonic() - start_time
        remaining = deadline - elapsed
        if remaining < 15.0:
            break

        half = n_lp // 2
        if it < half:
            t_alloc = min(35.0, (remaining - 15.0) / max(1.0, float(n_lp - it)))
        else:
            t_alloc = min(25.0, (remaining - 15.0) / max(1.0, float(n_lp - it)))
        t_alloc = max(3.0, t_alloc)

        g = lp_min_peak(q, t_alloc)
        if g is None:
            g = lp_max_sum(q, t_alloc)

        if g is None:
            stagnant_lp += 1
            if stagnant_lp >= 4:
                break
            continue

        stagnant_lp = 0
        t_best, o_best = line_search(q, g)

        if t_best > 0:
            q_new = (1.0 - t_best) * q + t_best * g
            q_new = norm1(q_new)
            o_new = quick_obj(q_new)
            if o_new < best_score:
                best_score = o_new
                best_q = q_new.copy()
            q = q_new
        else:
            # Tiny random nudge to escape stagnation
            q = norm1(q + 0.01 * rng.normal(size=n))

        # Periodic gradient polish
        if it % 5 == 4 or it == n_lp - 1:
            rem = deadline - (time.monotonic() - start_time) - 20.0
            if rem > 8.0:
                n_pol = min(30, int(rem / 2.0))
                q_p, o_p = grad_polish(q, n_pol, deadline - 20.0)
                if o_p < best_score:
                    best_score = o_p
                    best_q = q_p.copy()
                q = q_p

        # Verify with exact evaluator every 10 iterations
        if (it + 1) % 10 == 0:
            vs = get_score([float(x) for x in q])
            if np.isfinite(vs) and vs < best_score:
                best_score = vs
                best_q = q.copy()

    # Final intensive polish
    rem = deadline - (time.monotonic() - start_time) - 20.0
    if rem > 5.0:
        n_pol = min(100, int(rem * 0.8))
        q_p, o_p = grad_polish(q, n_pol, deadline - 20.0)
        if o_p < best_score:
            best_score = o_p
            best_q = q_p.copy()

    # Prepare result
    result_q = best_q.copy()
    result_q = np.clip(result_q, 0.0, 1000.0)
    s = float(result_q.sum())
    if s < 0.01:
        result_q = np.ones(n) / n

    result = [float(x) for x in result_q]

    # Final validation
    fs = get_score(result)
    if not np.isfinite(fs):
        fallback = [1.0 / n] * n
        fs2 = get_score(fallback)
        if np.isfinite(fs2):
            result = fallback

    return result
