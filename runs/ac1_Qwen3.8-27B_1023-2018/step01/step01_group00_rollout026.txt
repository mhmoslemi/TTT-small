                s_i = int(rng_p.integers(0, n_p - w + 1))
                cand_prov[s_i:s_i + w] *= float(np.exp(rng_p.normal(0, 0.15)))
            elif r < 0.80:
                cand_prov = a_p.copy()
                for _ in range(2):
                    j = int(rng_p.integers(0, n_p))
                    cand_prov[j] *= float(rng_p.uniform(0.6, 1.6))
            else:
                cand_prov = 0.5 * (a_p + a_p[::-1])

            cand_prov = np.maximum(cand_prov, 0.0)
            cs = _score_fast(cand_prov)
            if np.isfinite(cs) and cs < best_score - 1e-10:
                best_seq = cand_prov.copy()
                best_score = cs
                a_p = cand_prov.copy()
                stuck = 0
            else:
                stuck += 1
                if stuck % 40 == 0:
                    sigma *= 1.4
                    sigma = min(sigma, 0.8)
                if stuck > 300:
                    break

    # --- Final exact validation ---
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_score = float('inf')
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        pass

    if not np.isfinite(final_score):
        try:
            final_score = float(evaluate_sequence([float(x) for x in np.clip(parent, 0.0, 1000.0)]))
            if np.isfinite(final_score):
                best_seq = np.clip(parent, 0.0, 1000.0).astype(np.float64)
        except Exception:
            best_seq = np.ones(192, dtype=np.float64)
            final_score = 2.0

    # Ensure non-empty finite list
    if len(best_seq) == 0:
        return [1.0]
    out = [float(x) for x in best_seq]
    if not all(np.isfinite(x) for x in out):
        out = [float(x) for x in np.clip(parent, 0.0, 1000.0)]
        if not all(np.isfinite(x) for x in out):
            out = [1.0]
    return out
