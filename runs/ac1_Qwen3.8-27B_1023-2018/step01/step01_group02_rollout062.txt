```python
def lp_step(f, M):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0"""
    import cvxpy as cp
    n = len(f)
    g = cp.Variable(n)
    # Constraints: (f * g)[k] <= M for k = 0, ..., 2n-2
    # (f * g)[k] = sum_i f[i] * g[k-i] (valid indices)
    # This can be computed as a matrix-vector product
    # Build the convolution matrix
    # For each output index k, the constraint is sum_i f[i]*g[k-i] <= M
    # Let's build the matrix A where A[k, j] = f[j] if 0 <= k-j <= n-1, else 0
    # Actually, (f*g)[k] = sum_j f[j] * g[k-j] where j ranges over valid indices
    
    # More precisely, for np.convolve(f, g):
    # result[k] = sum_j f[j] * g[k-j] for valid j
    # k ranges from 0 to 2n-2
    # For a given k, j ranges from max(0, k-n+1) to min(n-1, k)
    
    constraints = [g >= 0]
    for k in range(2*n - 1):
        # (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j]
        idx = []
        for j in range(max(0, k-n+1), min(n-1, k) + 1):
            idx.append(j)
        if idx:
            g_idx = [k - j for j in idx]
            constraints.append(sum(f[j] * g[g_idx[i]] for i, j in enumerate(idx)) <= M)
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve()
    return g.value
```

If we allowed signed sequences, we could use Barker-like sequences or Zadoff-Chu sequences with much lower peak correlation. But with nonnegativity, the problem is much harder.

The optimal nonnegative sequence should have:
- High values where the "overlap" contributes less to the peak
- The peak of $x*x$ occurs at $k=n-1$ (for symmetric $x$), where $(x*x)_{n-1} = \sum_i x_i x_{n-1-i}$
- To minimize this, we want $x_i$ and $x_{n-1-i}$ to be "anti-correlated" in some sense
- But since both are nonnegative, we can't make one large and the other small (that would help the center but hurt the edges)

Actually, for a symmetric sequence $x_i = x_{n-1-i}$:
$(x*x)_{n-1} = \sum x_i^2$

And the score is $2n \sum x_i^2 / (\sum x_i)^2$.

To minimize this, we want to minimize $\sum x_i^2 / (\sum x_i)^2$, which by Cauchy-Schwarz is minimized when all $x_i$ are equal (giving $1/n$). So for symmetric sequences, the uniform sequence is optimal, giving score 2.

This means the optimal (non-uniform) sequence must be ASYMMETRIC! The asymmetry allows the "cross terms" $x_i x_{n-1-i}$ to be smaller than $x_i^2 + x_{n-1-i}^2$ in the sum.

Wait, but $\sum x_i x_{n-1-i} \leq \sum x_i^2$ by the rearrangement inequality (for nonnegative sequences). So for symmetric sequences, $(x*x)_{n-1} = \sum x_i^2$ is the maximum possible overlap. For asymmetric sequences, $(x*x)_{n-1} = \sum x_i x_{n-1-i} < \sum x_i^2$.

But the maximum of $x*x$ might not be at $k=n-1$ for asymmetric sequences. It could be at a different lag where the overlap is even larger.

So the optimal asymmetric sequence should:
1. Have $\sum x_i x_{n-1-i} < \sum x_i^2$ (asymmetry helps the center)
2. Ensure that no other lag has a larger overlap

This is a complex constraint. The LP approach should handle it naturally since it checks all lags.

```python
import numpy as np
import time
from scipy.optimize import linprog


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * float(np.max(c)) / (s * s))


def _conv_matrix(f, n):
    """Build (2n-1) x n matrix A such that A @ g = np.convolve(f, g)."""
    k = np.arange(2 * n - 1, dtype=np.int64)[:, None]
    j = np.arange(n, dtype=np.int64)[None, :]
    mask = (j >= k - n + 1) & (j <= k)
    return np.where(mask, f[None, :], 0.0)


def _lp_optimize(f0, n, end_t, rng, max_stagnation=120):
    """LP-based iterative optimization. Returns (best_f, best_score)."""
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s
    n = int(n)

    best_f = f.copy()
    best_sc = _score(f)
    t = 0.12
    no_improve = 0
    stagnation = max_stagnation

    while time.monotonic() < end_t - 1.0:
        f = np.maximum(f, 0.0)
        fs = f.sum()
        if fs < 1e-12:
            break
        f = f / fs

        M = float(np.max(np.convolve(f, f)))
        A_ub = _conv_matrix(f, n)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True,
                                                   'dual_feasibility_tolerance': 1e-9})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g = np.maximum(res.x, 0.0)
        sg = float(g.sum())
        if sg < 1e-12:
            break
        g /= sg

        # Line search
        candidates = [max(0.01, t * 0.3), max(0.01, t * 0.6), t,
                      min(0.45, t * 1.6), min(0.5, t * 2.2)]
        best_tt = t
        best_sc_try = np.inf
        for tt in candidates:
            ft = (1.0 - tt) * f + tt * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-12:
                continue
            sc_try = _score(ft)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_tt = tt

        if not np.isfinite(best_sc_try):
            best_tt = t
            best_sc_try = _score(f)

        f = (1.0 - best_tt) * f + best_tt * g
        f = np.maximum(f, 0.0)
        f /= f.sum()

        if best_sc_try < best_sc - 1e-13:
            best_sc = best_sc_try
            best_f = f.copy()
            no_improve = 0
            t = min(0.35, t * 1.03)
        else:
            no_improve += 1
            if no_improve > 8:
                t = max(0.01, t * 0.92)

            if fs > 1e-12:

            t = max(0.04, t * 0.6)
            no_improve = 0
            stagnation = int(stagnation * 1.4)
            if t < 0.008:

    return best_f, best_sc

    # Power-law edges
    for pw in [0.3, 0.7, 1.3]:
        add(np.maximum(x, 0.02) ** pw * np.maximum(1.0 - x, 0.02) ** pw)

    # Reverse of best parent
    if parent is not None and parent.size > 1:
        xp = np.linspace(0, 1, n)
        pp = np.linspace(0, 1, parent.size)
        add(np.interp(xp, pp, parent[::-1]))

    return inits[:14]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 12.0  # safety margin

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.maximum(p, 0.0)
            if p.sum() > 1e-12:
                parent = p / p.sum()
    except Exception:
        parent = None

    best_score = np.inf
    best_seq = np.array([0.5, 0.1, 0.4], dtype=np.float64)

    # Try multiple lengths
    lengths = set()
    if parent is not None:
        lengths.add(parent.size)
    for L in [127, 200, 300, 500, 750, 1000, 1500, 2000, 3000]:
        lengths.add(L)
    lengths = sorted(lengths)

    # Allocate time: weight by 1/sqrt(n) (faster iterations for small n)
    # But cap the time per length
    remaining = deadline - time.monotonic()
    weights = [1.0 / np.sqrt(L) for L in lengths]
    tw = sum(weights)
    time_per = [remaining * w / tw * 0.7 for w in weights]  # 70% to LP
    min_t = [3.0, 5.0, 8.0, 12.0, 15.0, 20.0, 30.0, 40.0, 50.0]

    for idx, n in enumerate(lengths):
        if time.monotonic() > deadline - 25:
            break

        alloc = min(time_per[idx] if idx < len(time_per) else 20.0,
                    deadline - time.monotonic() - 20.0)
        alloc = max(alloc, min_t[idx] if idx < len(min_t) else 5.0)
        alloc = min(alloc, max(3.0, deadline - time.monotonic() - 15.0))

        end_lp = time.monotonic() + alloc * 0.75
        end_polish = time.monotonic() + alloc

        inits = _make_inits(n, rng, parent)

        for ii, init in enumerate(inits):
            if time.monotonic() > end_polish - 1:
                break
            try:
                # LP phase
                lp_end = min(end_lp, end_polish - 2)
                if lp_end > time.monotonic() + 1.0:
                    f_lp, sc_lp = _lp_optimize(init, n, lp_end, rng)
                else:
                    f_lp, sc_lp = init.copy(), _score(init)

                # Polish phase
                pol_end = min(end_polish - 1, deadline - 20)
                if pol_end > time.monotonic() + 2.0 and sc_lp < best_score:
                    f_pol, sc_pol = _gradient_polish(f_lp, n, pol_end, rng)
                    if sc_pol < sc_lp:
                        f_lp, sc_lp = f_pol, sc_pol

                if sc_lp < best_score - 1e-12:
                    best_score = sc_lp
                    best_seq = f_lp.copy()

            except Exception:
                pass

    # Final polish on the best
    if time.monotonic() < deadline - 15:
        n_best = int(best_seq.size)
        try:
            f2, sc2 = _lp_optimize(best_seq, n_best, deadline - 10, rng)
            if sc2 < best_score:
                best_score = sc2
                best_seq = f2
        except Exception:
            pass

    # Final validation
    lst = [float(max(0.0, min(1000.0, x))) for x in np.asarray(best_seq, dtype=np.float64)]
    final_sc = evaluate_sequence(lst)
    if not np.isfinite(final_sc):
        lst = [0.5, 0.1, 0.4]
        final_sc = evaluate_sequence(lst)

    return lst
```
