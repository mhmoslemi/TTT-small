        # Gradient: grad[i] = 2 * sum_j a[j] * W[i+j]
        # = 2 * np.correlate(W, a, mode='valid')
        grad = 2.0 * np.correlate(W, a, mode='valid')

        # Mirror descent (multiplicative) step
        gmax = grad.max()
        if gmax > 1e-15:
            a = a * np.exp(-eta * grad / gmax)

        a = np.maximum(a, 1e-18)
        a_sum = a.sum()
        if a_sum < 1e-15:
            a = np.ones(n) / n
        else:
            a /= a_sum

        # Periodic score check (cheap: sum=1 so score = 2*n*max(conv))
        eval_count += 1
        if eval_count % 10 == 0:
            score = 2.0 * n * max_conv
            # Also compute fresh max after update
            conv_check = np.convolve(a, a)
            fresh_score = 2.0 * n * conv_check.max()
            if fresh_score < best_score:
                best_score = fresh_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # === Adaptive scheduling ===
        # Gradually increase K to handle more lags
        if it % 400 == 0:
            K = min(K + 8, 300)

        # Adjust learning rate
        if it % 150 == 0:
            if no_improve > 8:
                eta = max(0.01, eta * 0.7)
            elif no_improve < 3:
                eta = min(0.25, eta * 1.08)

        # Restart strategies when stuck
        if no_improve > 12:
            phase = no_improve % 4
            if phase == 0:
                # Multiplicative noise
                sigma = 0.08 + 0.04 * min(no_improve / 12.0, 3.0)
                a *= np.exp(rng.normal(0.0, sigma, size=n))
                a = np.maximum(a, 1e-18)
            elif phase == 1:
                # Symmetry blend
                blend = rng.uniform(0.15, 0.45)
                a = (1.0 - blend) * a + blend * a[::-1]
                a = np.maximum(a, 1e-18)
            elif phase == 2:
                # First-order blend with shifted version
                shift = rng.integers(1, max(2, n // 50))
                shifted = np.roll(a, shift)
                blend = rng.uniform(0.1, 0.3)
                a = (1.0 - blend) * a + blend * shifted
                a = np.maximum(a, 1e-18)
            else:
                # Blend with broader basis
                blend = rng.uniform(0.05, 0.15)
                a = (1.0 - blend) * a + blend * np.sin(np.pi * (j + 0.5) / n) ** 2
                a = np.maximum(a, 1e-18)

            a /= a.sum()
            eta = max(eta, 0.04)
            K = max(10, K // 2)
            no_improve = 0

        # Time check
        if time.monotonic() > deadline - 5.0:
            break

    # === Final: try a few more evals to confirm best ===
    # Do a final precise check on best_a
    final_score = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(final_score) or final_score > 2.0:
        best_a = np.ones(n, dtype=np.float64)
        final_score = float(evaluate_sequence(best_a.tolist()))

    return [float(x) for x in best_a]
