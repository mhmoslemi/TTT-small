```python
        cols.extend([j] * (k_hi - k_lo + 1))
        data.extend([a[j]] * (k_hi - k_lo + 1))
    
    A = sps.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A
```

This is O(n^2) to build, which is fine for n up to a few hundred.

Now for the main algorithm:

```python
def _optimize_lp(a0, max_iters, time_limit):
    """Iterate LP-based refinement."""
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s  # normalize to sum 1
    
    n = a.size
    c = -np.ones(n)
    best_local = _fast_score(a)
    
    for it in range(max_iters):
        if time.monotonic() > time_limit:
            break
        
        b = np.convolve(a, a)
        M = float(b.max())
        if M < 1e-30:
            break
        
        A = _build_A(a)
        b_ub = np.full(2*n-1, M * (1 + 1e-9))
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                         method='highs', options={'presolve': True})
        except:
            break
        
        if not res.success or res.x is None:
            break
        
        g = np.clip(res.x, 0, None)
        sg = g.sum()
        if sg < 1e-15:
            break
        g = g / sg
        
        # Line search: blend a and g
        best_t = a.copy()
        best_sc = best_local
        for t in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
            cand = (1-t)*a + t*g
            cand = np.clip(cand, 0, None)
            sc = _fast_score(cand)
            if sc < best_sc:
                best_sc = sc
                best_t = cand.copy()
        
        if best_sc < best_local * (1 - 1e-10):
            a = best_t
            a = a / a.sum()
            best_local = best_sc
        else:
            break
    
    return a, best_local
```

Now let me think about initializations. Good starting points:
1. Uniform: a = ones(n)/n (score = 2)
2. Linear ramp: a = (1, 2, ..., n) normalized
3. Truncated: a[i] = max(0, 1 - |2i/(n-1) - 1|) (triangular)
4. Parent sequence (interpolated)
5. Random Dirichlet
6. From smaller n solutions (upsampled)

For the gradient-based refinement, using the analytical gradient:

Given a, compute b = conv(a, a), find the set S = {k : b[k] >= 0.999 * max(b)} (the "active" set).
Then the gradient of max(b) with respect to a[i] is:
(1/|S|) * sum_{k in S} 2*a[k-i] (for 0 <= k-i < n)

This is like a "blurred" version of a.

For the objective f(a) = 2n * max(b) (with sum(a) = 1), the gradient (restricted to the simplex) is:
grad[i] = 2n/|S| * sum_{k in S} 2*a[k-i] - lambda (for the sum constraint)

We project: grad_projected[i] = grad[i] - mean(grad), then normalize and do a step.

Actually, since we're on the simplex and the constraint is sum(a) = 1, the projected gradient is:
g[i] = raw_grad[i] - sum(raw_grad)/n

Then we do: a_new = a - step * g, clip to [0, inf], renormalize.

Let me also think about whether we can use FFT to speed up the convolution. For n up to a few hundred, direct convolution is O(n^2) which is fine. For n up to a few thousand, FFT would help but the LP would be too slow. So I'll stick with direct convolution for n <= 300.

def _fast_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if n == 0:
        return float('inf')

        return float('inf')

    mb = float(b.max())
    if not math.isfinite(mb):
        return float('inf')
    return float(2.0 * n * mb / (s * s))

def _build_A(a):
    n = a.size
    rows = np.empty(n*n, dtype=np.int32)
    cols = np.empty(n*n, dtype=np.int32)
    data = np.empty(n*n, dtype=np.float64)
    idx = 0
    for j in range(n):
        k_lo = j
        k_hi = min(j + n - 1, 2*n - 2)
        cnt = k_hi - k_lo + 1
        rows[idx:idx+cnt] = k_lo + np.arange(cnt)
        cols[idx:idx+cnt] = j
        data[idx:idx+cnt] = a[j]
        idx += cnt
    return sps.csr_matrix((data[:idx], (rows[:idx], cols[:idx])), shape=(2*n-1, n))

def _lp_refine(a, stops, max_iters=200):
    """LP-based refinement."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = a.size
    c_vec = -np.ones(n)
    best_local = _fast_score(a)
    
    for it in range(max_iters):
        if time.monotonic() > stops:
            break
        b = np.convolve(a, a)
        M = float(b.max())
        if M < 1e-30:
            break
        A = _build_A(a)
        b_ub = np.full(2*n-1, M * 1.0000001)
        try:
            res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                         bounds=[(0.0, None)]*n,
                         method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None:
            break
        g = np.clip(res.x, 0, None)
        sg = float(g.sum())
        if sg < 1e-15:
            break
        g /= sg
        # Line search
        best_t = None
        best_sc = best_local
        for t in (0.15, 0.3, 0.5, 0.7, 0.9, 1.0):
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0, None)
            sc = _fast_score(cand)
            if sc < best_sc:
                best_sc = sc
                best_t = cand

            best_local = best_sc
        else:

                ap = np.clip(ap, 0, None)
                if ap.sum() > 1e-15:
                    ap /= ap.sum()
                    if _fast_score(ap) < best_local:
                        a = ap
                        best_local = _fast_score(a)
                        continue
            break
    return a, best_local

def _gradient_refine(a, stops, max_iters=500):
    """Gradient-based refinement using analytical subgradient."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = a.size
    best_local = _fast_score(a)
    step = 0.1
    stale = 0
    
    for it in range(max_iters):
        if time.monotonic() > stops:
            break
        b = np.convolve(a, a)
        bmax = float(b.max())
        if bmax < 1e-30:
            break
        # Active set: indices where b[k] >= 0.99 * bmax
        active = np.where(b >= 0.99 * bmax)[0]
        # Subgradient of max(b) wrt a[i] = mean over active k of 2*a[k-i] (if valid)
        grad = np.zeros(n)
        for k in active:
            # 2*a[k-i] for 0<=k-i<n, i.e., i in [k-n+1, k] intersect [0, n-1]
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                # a[k-i] for i in [i_lo, i_hi]
                # a[k-i_lo], a[k-i_lo-1], ..., a[k-i_hi]
                # = a[k - i_lo : k - i_hi + 1][::-1]
                seg = a[k - i_hi : k - i_lo + 1][::-1]
                grad[i_lo:i_hi+1] += 2.0 * seg
        grad /= len(active)
        # Project gradient onto simplex (subtract mean)
        grad -= grad.mean()
        # Adaptive step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        # Try multiple step sizes
        best_sc = best_local
        best_a = a
        for alpha in (step * 0.1, step * 0.3, step, step * 3, step * 10):
            cand = a - alpha * grad
            cand = np.clip(cand, 0, None)
            if cand.sum() < 1e-15:
                continue
            cand /= cand.sum()
            sc = _fast_score(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand
        if best_sc < best_local * (1 - 1e-10):
            a = best_a
            best_local = best_sc
            stale = 0
            step = min(step * 1.2, 1.0)
        else:
            stale += 1
            step *= 0.8
            if stale > 30:
                break
    return a, best_local
