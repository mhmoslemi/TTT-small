import time

def _proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # Project onto simplex with sum = s
    u = v / s  # normalize to sum = 1 problem
    # Project u onto {x >= 0, sum(x) = 1}
    mu = np.sort(u)[::-1]
    cv = np.cumsum(mu)
    rho = np.nonzero(mu * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    w = np.maximum(u - theta, 0)
    return w * s

    b = np.convolve(q, q)

        w = np.ones(len(k_set)) / len(k_set)
    # Gradient: for each k in k_set, grad[j] = 2*q[k-j] if 0<=k-j<n

import numpy as np
import time

def _proj_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho_candidates = np.nonzero(mu * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.full(n, 1.0 / n)
    r = rho_candidates[-1]
    theta = (cv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def _score_q(q):
    """Score assuming sum(q) is already 1. Returns 2*n*max(conv(q,q))."""
    n = q.size
    b = np.convolve(q, q)
    return 2.0 * n * float(np.max(b))


def _build_conv_matrix(q, n):
    """Build A (2n-1 x n) such that A @ g = n * convolve(q, g)."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for kk in range(2 * n - 1):
        lo = max(0, kk - n + 1)
        hi = min(n, kk + 1)
        if lo < hi:
            # A[kk, lo:hi] = n * q[kk-hi+1 : kk-lo+1][::-1]
            idx_lo = kk - (hi - 1)
            idx_hi = kk - lo
            A[kk, lo:hi] = n * q[idx_lo:idx_hi + 1][::-1]
    return A


def _lp_response(q, n):
    """LP: max sum(g) s.t. A@g <= M, g>=0. Returns normalized g or None."""
    from scipy.optimize import linprog
    b = np.convolve(q, q)
    M = n * float(b.max())
    if M <= 0 or not np.isfinite(M):
        return None
    A = _build_conv_matrix(q, n)
    c = -np.ones(n, dtype=np.float64) - 1e-7 * (n * q)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0.0, 1e4),
                      method='highs', options={'presolve': True, 'time_limit': 15})
        if res.status in (0, 1) and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            g = np.maximum(g, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None


def _lsm_grad(q, eps, n):
    """Compute log-sum-exp smoothed objective and gradient."""
    b = np.convolve(q, q)
    b_max = b.max()
    shift = b - b_max
    exp_b = np.exp(shift / eps)
    S = exp_b.sum()
    if S < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w = exp_b / S
    # Gradient: grad[j] = 2 * sum_k w[k] * q[k-j] for 0<=k-j<n
    grad = np.zeros(n)
    for kk in range(2 * n - 1):
        wk = w[kk]
        if wk < 1e-15:
            continue
        lo = max(0, kk - n + 1)
        hi = min(n, kk + 1)
        if lo < hi:
            idx_lo = kk - (hi - 1)
            idx_hi = kk - lo
            grad[lo:hi] += wk * 2.0 * q[idx_lo:idx_hi + 1][::-1]
    obj = eps * (np.log(S) + shift.max()) + b_max
    return obj, grad


def _subgrad_step(q, n, momentum=0.0):
    """One projected subgradient step on max conv(q,q)."""
    b = np.convolve(q, q)
    b_max_val = float(b.max())
    tol = 1e-4 * b_max_val
    active = np.where(b >= b_max_val - tol)[0]
    if active.size == 0:
        active = np.array([np.argmax(b)])
    # Weight by how close to max
    raw_w = b[active] - (b_max_val - tol)
    raw_w = np.maximum(raw_w, 0.0)
    sw = raw_w.sum()
    if sw < 1e-30:
        w = np.ones(active.size) / active.size
    else:
        w = raw_w / sw
    grad = np.zeros(n)
    for idx, kk in enumerate(active):
        lo = max(0, int(kk) - n + 1)
        hi = min(n, int(kk) + 1)
        if lo < hi:
            idx_lo = int(kk) - (hi - 1)
            idx_hi = int(kk) - lo
            grad[lo:hi] += w[idx] * 2.0 * q[idx_lo:idx_hi + 1][::-1]
    return grad, b_max_val

    n = q0.size

    lag = np.zeros(n)  # momentum accumulator

            sc = _score_q(q)

                stagnation = 0

            if arr.ndim != 1 or arr.size == 0:
                return
            arr = np.maximum(arr, 0.0)
            arr = np.minimum(arr, 1000.0)
            s = arr.sum()
            if s <= 0.01 or not np.isfinite(s):
                return
            arr = arr / s  # normalize to sum=1
            lst = [float(x) for x in arr]
            val = float(evaluate_sequence(lst))
            if np.isfinite(val) and val < best_score:
                best_score = val
                best_list = lst
        except Exception:
            pass

    # Try the parent construction
    try:
        h1 = globals().get('height_sequence_1', None)
        if h1 is not None:
            consider(h1)
    except Exception:
        pass

    # Quick baseline
    n_default = 512
    q_base = _proj_simplex(np.ones(n_default))
    consider(q_base)

    # Schedule of (n, num_starts)
    schedule = [(512, 4), (256, 2), (1024, 2)]

    for n, num_starts in schedule:
        if time.monotonic() > deadline - 50.0:
            break

        # Build starts
        starts = []
        if n == 512:
            try:
                h1 = globals().get('height_sequence_1', None)
                if h1 is not None:
                    starts.append(_proj_simplex(np.asarray(h1, dtype=np.float64)))
            except Exception:
                pass
        starts.append(_make_wave_start(n, rng))
        starts.append(_make_random_start(n, rng))
        while len(starts) < num_starts:
            starts.append(_make_wave_start(n, rng))

        n_best_q = None
        n_best_sc = float('inf')

        for i, q0 in enumerate(starts):
            if time.monotonic() > deadline - 30.0:
                break
            remaining = len(starts) - i
            alloc = (deadline - time.monotonic()) / max(1, remaining)
            if alloc < 15.0:
                break
            end = min(deadline, time.monotonic() + alloc)

            # Scale iterations with time available
            max_lp = min(60, max(10, int(alloc / 3.0)))
            lsm_it = min(20000, max(2000, int(alloc * 80)))
            sg_it = min(10000, max(1000, int(alloc * 30)))

            q_opt, sc_opt = _optimize_sequence(q0, end, max_lp, lsm_it, sg_it)
            if sc_opt < n_best_sc:
                n_best_sc = sc_opt
                n_best_q = q_opt
            consider(q_opt)

        if n_best_q is not None:
            consider(n_best_q)

    # Final safety: simple 4-element asymmetric
    q4 = np.array([0.2, 0.4, 0.2, 0.2])
    consider(q4)

    if best_list is None:
        best_list = [1.0] / 1.0

    # Final validation
    try:
        val = float(evaluate_sequence(best_list))
        if not np.isfinite(val):
            best_list = [0.2 / 1.0, 0.4 / 1.0, 0.2 / 1.0, 0.2 / 1.0]
    except Exception:
        best_list = [1.0]

    return [float(x) for x in best_list]
