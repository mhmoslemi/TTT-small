            if strat == 0:
                a = a + rng.normal(0, 0.04, n)
            elif strat == 1:
                idx = np.arange(n, dtype=np.float64)
                for f in [1, 2, 3, 4, 5]:
                    if rng.random() < 0.5:
                        a += rng.normal(0, 0.03) * np.sin(2 * np.pi * f * idx / n)
                    else:
                        a += rng.normal(0, 0.03) * np.cos(2 * np.pi * f * idx / n)
            elif strat == 2:
                A = np.fft.rfft(a, conv_fft)
                nl = min(100, A.size)
                A[1:nl] += rng.normal(0, 0.08, nl - 1) + 1j * rng.normal(0, 0.08, nl - 1)
                a = np.fft.irfft(A, conv_fft)[:n]
            elif strat == 3:
                m = rng.random(n) < 0.2
                a[m] += rng.normal(0, 0.06, m.sum())
                a[~m] *= 1.0 + rng.normal(0, 0.01, (~m).sum())
            else:
                half = n // 2
                pert = rng.normal(0, 0.05, half)
                a[:half] += pert
                a[half:] -= pert
            a = np.maximum(a, 0.0)
            aa = a.sum()
            if aa > 0.01:
                a = a * n / aa
            no_improve = 0
            lr_mult = 1.0
            beta = max(30.0, beta * 0.5)
            continue

        # Adaptive beta schedule
        if elapsed_frac < 0.3:
            beta = 30.0 + 80.0 * (iter_count / max(1, 0.3 * 50000))
            beta = min(beta, 150.0)
        elif elapsed_frac < 0.7:
            beta = 150.0 + 0.02 * no_improve
            beta = min(beta, 800.0)
        else:
            beta = 500.0 + 0.05 * no_improve
            beta = min(beta, 5000.0)

        # LSE weights
        Bs = B - M
        clip_val = -12.0 / beta
        Bs = np.maximum(Bs, clip_val)
        eB = np.exp(beta * Bs)
        es = eB.sum()

        if es < 1e-200 or not np.isfinite(es):
            w = np.zeros(2 * n - 1)
            w[np.argmax(B)] = 1.0
            LSE = M
        else:
            w = eB / es
            LSE = M + np.log(es) / beta

        # Gradient of 2n*LSE/S^2
        c = get_c(a, w)
        grad = 2.0 * n * (S * c - 2.0 * LSE) / (S * S * S)
        grad -= grad.mean()

        # Step size
        gn = np.sqrt(np.dot(grad, grad))
        if gn < 1e-14:
            break

        # Scale step with problem size and schedule
        base_lr = 8.0 / (gn * np.sqrt(n))
        step = base_lr * lr_mult

        # Decay step in later phase
        if elapsed_frac > 0.8:
            step *= max(0.1, 1.0 - (elapsed_frac - 0.8) / 0.2)

        # Nonnegativity: ensure we don't go below 0
        pos = grad > 1e-12
        if pos.any():
            max_step = 0.85 * np.min(a[pos] / grad[pos])
            step = min(step, max(1e-20, max_step))

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn > 0.01:
            a = a_new * n / sn
        else:
            a = np.maximum(a * (1.0 - 0.002), 0.0)
            a = a * n / max(a.sum(), 0.01)

        # LR adaptation
        if no_improve % 200 == 0:
            if no_improve > 500:
                lr_mult *= 0.93
            elif no_improve < 100:
                lr_mult = min(lr_mult * 1.03, 3.0)
        lr_mult = max(lr_mult, 0.02)

        # Periodic exact check
        if iter_count % 2000 == 0 and time.monotonic() < deadline - 10:
            exact_sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(exact_sc) and exact_sc < best_score:
                best = a.copy()
                best_score = exact_sc

    # Final polish: try tiny perturbations of best
    for _ in range(20):
        if time.monotonic() > deadline - 3:
            break
        a_try = best + rng.normal(0, 0.015, n)
        a_try = np.maximum(a_try, 0.0)
        st = a_try.sum()
        if st > 0.01:
            a_try = a_try * n / st
            sc = float(evaluate_sequence(a_try.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = a_try.copy()
                best_score = sc

    # Validate final answer
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = np.ones(n, dtype=np.float64)
        final_sc = float(evaluate_sequence(best.tolist()))

    # Ensure valid output
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.ones(n, dtype=np.float64)

    return [float(x) for x in best]
