    u=np.sort(v)[::-1]
    css=np.cumsum(u)
    rho=np.nonzero(u > (css - 1)/np.arange(1,len(v)+1))[0][-1]

def _project_simplex(v):
    v=np.asarray(v,float)
    if v.size==1: return np.ones(1) if v.sum()>0 else np.ones(1)
    u=np.sort(v)[::-1]
    css=np.cumsum(u)
    rho=np.arange(1,v.size+1)
    cond = u > (css-1)/rho
    if not cond.any(): rho_val=0
    else: rho_val = int(np.where(cond)[0][-1])
    theta = (css[rho_val]-1)/(rho_val+1)
    w = np.maximum(v-theta,0)
    s=w.sum()
    if s<=0 or not finite: return np.full(v.size,1/v.size)
    return w/s

import numpy as np
import time


def _norm1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _upsample(a, m):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = int(m)
    if m <= 0:
        return np.ones(1, dtype=np.float64)
    if a.size == m:
        return _norm1(a)
    if a.size == 0:
        return np.ones(m, dtype=np.float64) / float(m)
    x0 = np.linspace(0.0, 1.0, a.size)
    x1 = np.linspace(0.0, 1.0, m)
    out = np.interp(x1, x0, a)
    return _norm1(out)


def _score_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    c = np.convolve(a, a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * n * mx / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.ones(1, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    a = e / s
    a = np.where(np.isfinite(a), a, 0.0)
    s2 = float(a.sum())
    if not np.isfinite(s2) or s2 <= 0.0:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return a / s2


def _z_of(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-14)
    z = np.log(a)
    z = np.where(np.isfinite(z), z, 0.0)
    z -= float(np.mean(z))
    return z


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = int(v.size)
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (css - 1.0) / rho
    idx = np.nonzero(cond)[0]
    if idx.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    r = int(idx[-1])
    theta = (css[r] - 1.0) / float(r + 1)
    w = np.maximum(v - theta, 0.0)
    w = np.where(np.isfinite(w), w, 0.0)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return w / s


def _softmax_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    M = float(np.max(c))
    x = float(beta) * (c - M)
    x = np.where(np.isfinite(x), x, -745.0)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(c.size, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return p
    return e / s


def _active_weights(a, c, M, frac):
    n = int(a.size)
    L = int(c.size)
    frac = float(frac)
    tol = float(M) * (0.005 + 0.05 * max(0.0, 1.0 - frac))
    tol = max(1e-18, tol)
    active = np.where(c >= float(M) - tol)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(c))], dtype=np.int64)
    else:
        if active.size > 24:
            active = active[np.argsort(c[active])[::-1][:24]]

    pfull = np.zeros(L, dtype=np.float64)

    ar = np.arange(n, dtype=np.int64)

    if active.size == 1:
        k = int(active[0])
        pfull[k] = 1.0
        r = k - ar
        valid = (r >= 0) & (r < n)
        g = np.zeros(n, dtype=np.float64)
        g[valid] = 2.0 * a[r[valid]]
        t = g - float(np.mean(g))
        return pfull, float(np.linalg.norm(t))

    S = int(active.size)
    gT = np.zeros((n, S), dtype=np.float64)
    for j, k in enumerate(active):
        r = int(k) - ar
        valid = (r >= 0) & (r < n)
        gT[valid, j] = 2.0 * a[r[valid]]

    colmean = gT.mean(axis=0)
    tmat = gT - colmean[None, :]
    A = tmat.T @ tmat
    A = A + 1e-12 * np.eye(S, dtype=np.float64)

    K = np.zeros((S + 1, S + 1), dtype=np.float64)
    K[:S, :S] = A
    K[:S, S] = 1.0
    K[S, :S] = 1.0

    rhs = np.zeros(S + 1, dtype=np.float64)
    rhs[S] = 1.0

    try:
        sol = np.linalg.solve(K, rhs)
    except np.linalg.LinAlgError:
        sol = np.linalg.lstsq(K, rhs, rcond=None)[0]

    p = sol[:S]
    p = _project_simplex(p)

    maxnorm = float(np.linalg.norm(A, ord=np.inf))
    step = 2.0 / (maxnorm + 1e-12)
    for _ in range(24):
        grad = 2.0 * (A @ p)
        pnew = _project_simplex(p - step * grad)
        if np.linalg.norm(pnew - p) < 1e-10:
            p = pnew
            break
        p = pnew

    pfull[active] = p
    ss = float(pfull.sum())
    if not np.isfinite(ss) or ss <= 0.0:
        pfull = np.zeros(L, dtype=np.float64)
        pfull[int(active[0])] = 1.0
    else:
        pfull = pfull / ss

    nt = float(np.linalg.norm(tmat @ p))
    return pfull, nt


def _grad_z_from_p(a, p):
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if p.size != 2 * n - 1:
        p = _softmax_weights(np.convolve(a, a), 1000.0)
    h = np.correlate(p, a, mode="valid")
    if h.size != n:
        h = np.zeros(n, dtype=np.float64)
    ga = 2.0 * h
    dot = float(np.dot(ga, a))
    gz = a * (ga - dot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz -= np.mean(gz)
    return gz


def _ga_from_p(a, p):
    n = int(a.size)
    if p.size != 2 * n - 1:
        p = _softmax_weights(np.convolve(a, a), 1000.0)
    h = np.correlate(p, a, mode="valid")
    if h.size != n:
        h = np.zeros(n, dtype=np.float64)
    ga = 2.0 * h
    ga = np.where(np.isfinite(ga), ga, 0.0)
    return ga


def _z_optimize(n, a0, t_start, t_end, rng, best_a, best_s):
    n = int(n)
    if n <= 1:
        a = _norm1(a0)
        if a.size != n:
            a = np.ones(n, dtype=np.float64) / float(max(1, n))
        s = _score_arr(a)
        if np.isfinite(s) and s < best_s:
            return a, s
        return best_a, best_s

    t_now = time.monotonic()
    t_start = max(float(t_start), t_now)
    t_end = max(float(t_end), t_start)
    if t_end <= t_start + 0.5:
        return best_a, best_s

    a = _norm1(a0)
    if a.size != n:
        a = _upsample(a, n)
    z = _z_of(a)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    b1p = 1.0
    b2p = 1.0

    local_a = a.copy()
    local_s = _score_arr(a)
    if np.isfinite(local_s) and local_s < best_s:
        best_a = local_a.copy()
        best_s = local_s
    else:
        local_s = best_s

    lr = 0.05
    no_soft = 0
    no_local = 0
    it = 0

    while time.monotonic() < t_end - 0.05:
        it += 1
        a = _softmax(z)
        c = np.convolve(a, a)
        M = float(np.max(c))
        score = float(2.0 * n * M)

        if not np.isfinite(score):
            z = _z_of(local_a)
            no_soft += 50
            continue

        if score < local_s - 1e-14:
            local_s = score
            local_a = a.copy()
            no_local = 0
        else:
            no_local += 1

        if score < best_s:
            best_a = a.copy()
            best_s = score

        frac = (time.monotonic() - t_start) / max(1e-12, t_end - t_start)
        frac = min(1.0, max(0.0, frac))

        beta = 500.0
        if no_local > 150:
            beta = 3000.0
        if no_local > 500:
            beta = 15000.0
        if no_local > 1500:
            beta = 60000.0
        if frac > 0.70:
            beta = max(beta, 120000.0)

        use_active = (it > 30) and (frac > 0.15 or no_soft > 50)
        stat = False
        if use_active:
            p, statval = _active_weights(a, c, M, frac)
            stat = bool(statval < 1e-8)
        else:
            p = _softmax_weights(c, beta)

        gz = _grad_z_from_p(a, p)
        if not np.all(np.isfinite(gz)):
            gz = np.zeros(n, dtype=np.float64)

        b1p *= b1
        b2p *= b2
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mhat = m / max(1e-12, 1.0 - b1p)
        vhat = v / max(1e-12, 1.0 - b2p)
        step = -mhat / (np.sqrt(vhat) + eps)
        step = np.where(np.isfinite(step), step, 0.0)
        rms = float(np.sqrt(np.mean(step * step)))
        if rms < 1e-14:
            step = rng.standard_normal(n)
            step -= step.mean()
            rms = float(np.sqrt(np.mean(step * step))) + 1e-300

        step = step / rms * lr
        step -= step.mean()
        step = np.clip(step, -2.0, 2.0)
        step -= step.mean()

        accepted = False
        cur = score
        for alpha in (1.0, 0.5, 0.25, 0.10, 0.05):
            zt = z - alpha * step
            zt -= zt.mean()
            at = _softmax(zt)
            st = _score_arr(at)
            if np.isfinite(st) and st < cur - 1e-14:
                z = zt
                a = at
                score = st
                accepted = True
                break

        if accepted:
            no_soft = 0
            if score < cur - 1e-14:
                lr = min(0.25, lr * 1.03)
        else:
            zt = z - 0.04 * step
            zt -= zt.mean()
            at = _softmax(zt)
            st = _score_arr(at)
            if np.isfinite(st) and st < cur + 1e-6:
                z = zt
                a = at
                score = st
                no_soft = 0
            else:
                no_soft += 1
            lr = max(0.01, lr * 0.85)

        if no_soft > 1200 or no_local > 7000 or (stat and no_soft > 100):
            base = np.asarray(best_a, dtype=np.float64)
            if rng.random() < 0.30:
                base = base[::-1]
            ns = 0.04 + 0.20 * min(1.0, float(no_local) / 7000.0)
            cand = base * (1.0 + ns * rng.standard_normal(n))
            if rng.random() < 0.18:
                cand = rng.dirichlet(np.full(n, 0.5, dtype=np.float64))
            a = _norm1(cand)
            z = _z_of(a)
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            b1p = 1.0
            b2p = 1.0
            lr = 0.05
            no_soft = 0
            no_local = 0
            local_s = _score_arr(a)
            local_a = a.copy()

    a = _softmax(z)
    s = _score_arr(a)
    if np.isfinite(s) and s < best_s:
        return a, s
    return best_a, best_s


def _a_optimize(n, a0, t_start, t_end, rng, best_a, best_s):
    n = int(n)
    if n <= 1:
        a = _norm1(a0)
        if a.size != n:
            a = np.ones(n, dtype=np.float64) / float(max(1, n))
        s = _score_arr(a)
        if np.isfinite(s) and s < best_s:
            return a, s
        return best_a, best_s

    t_now = time.monotonic()
    t_start = max(float(t_start), t_now)
    t_end = max(float(t_end), t_start)
    if t_end <= t_start + 0.5:
        return best_a, best_s

    a = _norm1(a0)
    if a.size != n:
        a = _upsample(a, n)

    local_a = a.copy()
    local_s = _score_arr(a)
    if np.isfinite(local_s) and local_s < best_s:
        best_a = local_a.copy()
        best_s = local_s
    else:
        local_s = best_s

    v = np.zeros(n, dtype=np.float64)
    lr = 0.004
    no_soft = 0
    no_local = 0
    it = 0

    while time.monotonic() < t_end - 0.05:
        it += 1
        a = _project_simplex(a)
        if a.size != n:
            a = _upsample(a, n)

        c = np.convolve(a, a)
        M = float(np.max(c))
        score = float(2.0 * n * M)

        if not np.isfinite(score):
            a = local_a.copy()
            no_soft += 50
            continue

        if score < local_s - 1e-14:
            local_s = score
            local_a = a.copy()
            no_local = 0
        else:
            no_local += 1

        if score < best_s:
            best_a = a.copy()
            best_s = score

        frac = (time.monotonic() - t_start) / max(1e-12, t_end - t_start)
        frac = min(1.0, max(0.0, frac))

        beta = 500.0
        if no_local > 150:
            beta = 3000.0
        if no_local > 500:
            beta = 15000.0
        if no_local > 1500:
            beta = 60000.0
        if frac > 0.70:
            beta = max(beta, 120000.0)

        use_active = (it > 30) and (frac > 0.15 or no_soft > 50)
        stat = False
        if use_active:
            p, statval = _active_weights(a, c, M, frac)
            stat = bool(statval < 1e-8)
        else:
            p = _softmax_weights(c, beta)

        ga = _ga_from_p(a, p)
        g = ga - float(np.mean(ga))
        g = np.where(np.isfinite(g), g, 0.0)

        v = 0.99 * v + 0.01 * (g * g)
        step = -g / (np.sqrt(v) + 1e-10)
        step = np.where(np.isfinite(step), step, 0.0)
        rms = float(np.sqrt(np.mean(step * step)))
        if rms < 1e-14:
            step = rng.standard_normal(n)
            step -= step.mean()
            rms = float(np.sqrt(np.mean(step * step))) + 1e-300

        step = step / rms * lr
        step -= step.mean()

        accepted = False
        cur = score
        for alpha in (1.0, 0.5, 0.25, 0.10, 0.05):
            at = _project_simplex(a + alpha * step)
            if at.size != n:
                at = _upsample(at, n)
            st = _score_arr(at)
            if np.isfinite(st) and st < cur - 1e-14:
                a = at
                score = st
                accepted = True
                break

        if accepted:
            no_soft = 0
            lr = min(0.035, lr * 1.04)
        else:
            at = _project_simplex(a + 0.03 * step)
            if at.size != n:
                at = _upsample(at, n)
            st = _score_arr(at)
            if np.isfinite(st) and st < cur + 1e-6:
                a = at
                score = st
                no_soft = 0
            else:
                no_soft += 1
            lr = max(0.0005, lr * 0.85)

        if no_soft > 1200 or no_local > 8000 or (stat and no_soft > 100):
            base = np.asarray(best_a, dtype=np.float64)
            if rng.random() < 0.30:
                base = base[::-1]
            ns = 0.04 + 0.25 * min(1.0, float(no_local) / 8000.0)
            cand = base + ns * rng.standard_normal(n)
            if rng.random() < 0.18:
                cand = rng.dirichlet(np.full(n, 0.5, dtype=np.float64))
            a = _project_simplex(cand)
            if a.size != n:
                a = _upsample(a, n)
            v = np.zeros(n, dtype=np.float64)
            lr = 0.004
            no_soft = 0
            no_local = 0
            local_s = _score_arr(a)
            local_a = a.copy()

    a = _project_simplex(a)
    if a.size != n:
        a = _upsample(a, n)
    s = _score_arr(a)
    if np.isfinite(s) and s < best_s:
        return a, s
    return best_a, best_s


def _polish(a, rng, time_limit=4.0):
    a = _norm1(a)
    n = int(a.size)
    if n <= 0:
        return np.ones(1, dtype=np.float64), np.inf
    best = a.copy()
    bs = _score_arr(best)
    end = time.monotonic() + max(0.0, float(time_limit))
    trials = 0

    while time.monotonic() < end - 0.02:
        trials += 1
        cand = best.copy()
        mode = int(rng.integers(0, 5))

        if mode == 0:
            nb = int(rng.integers(1, 4))
            for _ in range(nb):
                lo = max(1, n // 500)
                hi = max(lo + 1, min(n, n // 10))
                width = int(rng.integers(max(1, lo), max(2, hi + 1)))
                width = min(max(1, width), n)
                if n > width:
                    start = int(rng.integers(0, n - width + 1))
                else:
                    start = 0
                cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
        elif mode == 1:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                amt = best[i] * float(rng.uniform(0.05, 0.50))
                cand[i] -= amt
                cand[j] += amt
        elif mode == 2:
            off = int(rng.integers(-3, 4))
            cand = 0.5 * (cand + np.roll(cand, off))
        elif mode == 3:
            pw = float(rng.uniform(0.6, 2.2))
            cand = np.power(np.maximum(cand, 0.0), pw)
        else:
            delta = rng.standard_normal(n) * 0.01
            delta -= delta.mean()
            cand = best + delta

        cand = _norm1(cand)
        sc = _score_arr(cand)
        if np.isfinite(sc) and sc < bs - 1e-15:
            best = cand
            bs = sc

        if trials % 7 == 0:
            cand2 = best.copy()
            cand2[cand2 < 1e-8] = 0.0
            cand2 = _norm1(cand2)
            sc2 = _score_arr(cand2)
            if np.isfinite(sc2) and sc2 < bs - 1e-15:
                best = cand2
                bs = sc2

    return best, bs


def _trim_best(a, max_trim=18):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = int(a.size)
    if n == 0:
        return np.inf, np.ones(1, dtype=np.float64)

    best_s = _score_arr(a)
    best_a = a
    tmax = min(int(max_trim), max(0, n - 1))

    for l in range(0, tmax + 1):
        for r in range(0, tmax + 1):
            if l == 0 and r == 0:
                continue
            m = n - l - r
            if m < 1:
                continue
            cand = a[l:n - r]
            try:
                sc = _score_arr(cand)
            except Exception:
                sc = np.inf
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_a = cand

    return best_s, best_a


def _add_candidate(cands, arr):
    try:
        a = np.asarray(arr, dtype=np.float64).ravel()
        if a.size == 0:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if not np.isfinite(s) or s < 1e-12:
            return
        a = a / s
        cands.append(a)
    except Exception:
        pass


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    try:
        total_budget = max(1.0, float(budget_s))
    except Exception:
        total_budget = 1300.0
    safety = 18.0
    deadline = t0 + max(1.0, total_budget - safety)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size > 0:
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.maximum(p, 0.0)
            if float(p.sum()) > 1e-12:
                parent = p / float(p.sum())
    except Exception:
        parent = None

    if parent is not None:
        n = int(parent.size)
        best_a = parent.copy()
    else:
        n = 127
        best_a = np.ones(n, dtype=np.float64) / float(n)

    best_s = _score_arr(best_a)
    if not np.isfinite(best_s):
        best_a = np.ones(n, dtype=np.float64) / float(max(1, n))
        best_s = _score_arr(best_a)

    if n <= 2:
        cands = []
        _add_candidate(cands, best_a)
        _add_candidate(cands, np.array([2.0, 1.0], dtype=np.float64) / 3.0)
        cands.append(np.ones(max(1, n), dtype=np.float64) / float(max(1, n)))
    else:
        total = max(1e-3, deadline - t0)

        if total > 25.0:
            quick_end = t0 + min(35.0, 0.08 * total)
            if time.monotonic() < quick_end - 2.0:
                best_a, best_s = _z_optimize(
                    n, best_a, time.monotonic(), quick_end, rng, best_a, best_s
                )

        z2_end = t0 + 0.56 * total
        if time.monotonic() < z2_end - 2.0:
            best_a, best_s = _z_optimize(
                n, best_a, time.monotonic(), z2_end, rng, best_a, best_s
            )

        a_end = t0 + 0.90 * total
        if time.monotonic() < a_end - 2.0:
            best_a, best_s = _a_optimize(
                n, best_a, time.monotonic(), a_end, rng, best_a, best_s
            )

        final_z_end = deadline - 4.0
        if time.monotonic() < final_z_end - 2.0:
            best_a, best_s = _z_optimize(
                n, best_a, time.monotonic(), final_z_end, rng, best_a, best_s
            )

        remain = deadline - time.monotonic()
        if remain > 5.0:
            best_a, best_s = _polish(best_a, rng, min(5.0, remain - 1.0))

    cands = []
    _add_candidate(cands, best_a)
    _add_candidate(cands, best_a[::-1])
    if parent is not None:
        _add_candidate(cands, parent)

    _add_candidate(cands, np.ones(n, dtype=np.float64) / float(max(1, n)))
    _add_candidate(cands, np.ones(max(1, n), dtype=np.float64) / float(max(1, n)))
    _add_candidate(cands, np.array([2.0, 1.0], dtype=np.float64) / 3.0)
    _add_candidate(cands, np.array([1.0, 0.0, 1.0, 1.0], dtype=np.float64) / 3.0)

    for pw in (0.65, 0.85, 1.25, 1.60):
        try:
            arr = np.power(np.maximum(best_a, 0.0), pw)
            _add_candidate(cands, arr)
        except Exception:
            pass

    for k in (1, 5, 10, 20, 40):
        if 0 < k < n:
            try:
                idx = np.argsort(best_a)[:k]
                arr = best_a.copy()
                arr[idx] = 0.0
                _add_candidate(cands, arr)
            except Exception:
                pass

    try:
        ts, ta = _trim_best(best_a)
        _add_candidate(cands, ta)
    except Exception:
        pass
    try:
        ts, ta = _trim_best(best_a[::-1])
        _add_candidate(cands, ta)
    except Exception:
        pass

    best_list = None
    best_eval = np.inf
    for cand in cands:
        try:
            lst = [float(x) for x in cand]
            if not lst:
                continue
            if any(not np.isfinite(x) for x in lst):
                continue
            ev = float(evaluate_sequence(lst))
            if np.isfinite(ev) and ev < best_eval:
                best_eval = ev
                best_list = lst
        except Exception:
            continue

    if best_list is None:
        best_list = [float(x) for x in np.array([2.0, 1.0], dtype=np.float64) / 3.0]

    return best_list
