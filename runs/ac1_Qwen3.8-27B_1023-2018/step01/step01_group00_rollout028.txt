def _fft_conv(a, n, fsize):
    F = rfft(a[:n], fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    c = np.maximum(c, 0.0)
    return c

def _score(a, fsize):
    n = a.size
    b = _fft_conv(a, n, fsize)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return math.inf
    return 2.0 * n * max_b / (s * s)

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / z.size
    return e / s

def _gradient_z(a, b, tau, n):
    """Compute gradient of soft-logsumexp(b, tau) w.r.t. z (log a)."""
    m = b.max()
    x = (b - m) / tau
    ex = np.exp(x)
    p = ex / ex.sum()
    
    # g[i] = 2 * sum_{m=0}^{n-1} p[i+m] * a[m]
    # p has length 2n-1, a has length n
    g = np.zeros(n)
    for i in range(n):
        g[i] = 2.0 * np.dot(p[i:i+n], a)
    
    # In log-space: grad_z[i] = a[i] * (g[i] - sum(a*g))
    mean_g = np.dot(a, g)
    grad_z = a * (g - mean_g)
    
    return grad_z

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
