def _sc(a):
    """Fast score computation."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, None)
    n = a.size
    if n == 0:
        return float('inf')
    s = float(a.sum())
    if s <= 1e-15:
        return float('inf')
    b = np.convolve(a, a)
    mb = float(b.max())
    if not math.isfinite(mb):
        return float('inf')
    return 2.0 * n * mb / (s * s)


def _proj1(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    idx = np.nonzero(u * np.arange(1, n + 1) > cs - 1.0)[0]
    if idx.size == 0:
        return np.ones(n, dtype=np.float64) / n
    k = idx[-1]
    theta = (cs[k] - 1.0) / (k + 1.0)
    return np.maximum(v - theta, 0.0)

    return np.convolve(a, a)

    rows = j + k
    cols = j
    data = np.tile(a.astype(np.float64), n)
    return sps.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

def _upsample(parent, n):
    """Upsample parent to length n using interpolation."""
    p = np.asarray(parent, dtype=np.float64).ravel()
    p = np.clip(p, 0, None)
    if p.size < 2 or p.sum() <= 0:
        return np.ones(n, dtype=np.float64) / n
    x_old = np.linspace(0, 1, p.size)
    x_new = np.linspace(0, 1, n)
    return np.interp(x_new, x_old, p)
