```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft

def _fft_convolve(a):
    """Compute full convolution using FFT."""
    n = len(a)
    size = 2 * (2 * n - 1)  # or just 4n
    f = fft(a, n=size)
    result = ifft(f * f, n=size).real[:2 * n - 1]
    return result

def _score_fast(a):
    n = len(a)
    b = _fft_convolve(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def _build_A(a):
    """Build the convolution matrix A such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je+1] = a[k - np.arange(js, je + 1)]
    return A

def _lp_direction(a, time_limit=5.0):
    """Solve LP to find direction g."""
    start = time.monotonic()
    n = len(a)
    s = a.sum()
    if s <= 0:
        return None
    a_n = a / s
    
    b = _fft_convolve(a_n)
    M = b.max()
    if M <= 0 or not np.isfinite(M):
        return None
    
    A = _build_A(a_n)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, 
                     method='highs', options={'time_limit': time_limit})
    except:
        return None
    
    if res.success:
        g = np.maximum(res.x, 0)
        gs = g.sum()
        if gs > 0:
            return g / gs
    return None

def _mirror_descent_step(a, b, step_size):
    """One mirror descent step."""
    n = len(a)
    s = a.sum()
    if s <= 0:
        return a.copy()
    a_n = a / s
    
    # Find active lags (top few)
    m = b.max()
    if m <= 0:
        return a_n
    
    # Softmax weights over top lags
    top_k = np.argsort(b)[::-1][:min(20, 2*n-1)]
    w = np.exp(8.0 * (b[top_k] - m))
    w = w / w.sum()
    
    # Subgradient
    sg = np.zeros(n)
    for k_idx, weight in zip(top_k, w):
        k = int(k_idx)
        # sg[j] += weight * a_n[k - j] for valid indices
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je+1] += weight * a_n[k - np.arange(js, je + 1)]
    
    # Mirror descent update: a_new[j] = a[j] * exp(-step * 2*sg[j])
    # Then normalize to sum = 1
    log_weights = -2.0 * step_size * sg
    # For numerical stability
    log_weights -= log_weights.max()
    a_new = a_n * np.exp(log_weights)
    s_new = a_new.sum()
    if s_new > 0 and np.isfinite(s_new):
        a_new = a_new / s_new
    else:
        a_new = a_n
    return a_new

def _simplex_project(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    # Find tau such that sum(max(v - tau, 0)) = 1
    # v_sorted in decreasing order
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0][-1]
    tau = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - tau, 0)
```

Given current `a` (normalized to sum=1), we want to find `g` (also to be normalized) that, when blended with `a`, reduces the score.

Score of blend `(1-t)a + tg` (normalized):
- sum = (1-t) + t = 1 (already normalized)
- convolve = (1-t)²conv(a) + 2t(1-t)conv(a,g) + t²conv(g)
- We want to minimize max of this.

The LP finds `g` maximizing sum(g), s.t. conv(a,g) ≤ max(conv(a)), g ≥ 0.

After normalizing g (dividing by sum), we have a unit-sum g. The constraint becomes conv(a, g/sum_g) ≤ max(conv(a)), i.e., conv(a,g) ≤ max(conv(a)) * sum_g.

Hmm, actually the LP as stated (max sum(g) s.t. conv(a,g) ≤ M, g ≥ 0) finds the "largest" g in L1 norm that doesn't exceed the current bound. After normalization, this gives a direction that "spreads out" the energy.

```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def _fft_conv(a):
    n = len(a)
    m = 1
    while m < 2 * n - 1:
        m *= 2
    f = fft(a, m)
    return ifft(f * f, m).real[:2 * n - 1]


def _score_fast(a):
    n = len(a)
    b = _fft_conv(a)
    mb = b.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * mb / (s * s)


def _simplex_project(v):
    n = len(v)
    if n == 0:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho = np.where(u * idx > cv - 1.0)[0]
    if len(rho) == 0:
        return np.full(n, 1.0 / n)
    tau = (cv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(v - tau, 0.0)


def _build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je + 1] = a[k - np.arange(js, je + 1)]
    return A


def _lp_direction(a, deadline):
    n = len(a)
    s = a.sum()
    if s <= 0:
        return None
    a_n = a / s
    b = _fft_conv(a_n)
    M = b.max()
    if M <= 0 or not np.isfinite(M):
        return None
    if time.monotonic() > deadline - 10.0:
        return None
    A = _build_A(a_n)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    time_limit = max(1.0, deadline - time.monotonic() - 15.0)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': time_limit,
                                                'presolve': True})
    except Exception:
        return None
    if res is not None and res.success:
        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs > 1e-10 and np.isfinite(gs):
            return g / gs
    return None


def _mirror_step(a, b, step, n_top=16):
    n = len(a)
    s = a.sum()
    if s <= 0:
        return a
    a_n = a / s
    m = b.max()
    if m <= 0:
        return a_n
    top_k = np.argsort(b)[::-1][:min(n_top, 2 * n - 1)]
    w = np.exp(10.0 * (b[top_k] - m))
    ts = w.sum()
    if ts <= 0:
        w = np.ones(len(top_k)) / len(top_k)
    else:
        w = w / ts
    sg = np.zeros(n)
    for k_idx, weight in zip(top_k, w):
        k = int(k_idx)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += weight * a_n[k - np.arange(js, je + 1)]
    log_w = -2.0 * step * sg
    log_w -= log_w.max()
    a_new = a_n * np.exp(log_w)
    sn = a_new.sum()
    if sn > 0 and np.isfinite(sn):
        return a_new / sn
    return a_n


def _subgrad_step(a, b, step, n_top=10):
    n = len(a)
    s = a.sum()
    if s <= 0:
        return a
    a_n = a / s
    m = b.max()
    if m <= 0:
        return a_n
    top_k = np.argsort(b)[::-1][:min(n_top, 2 * n - 1)]
    w = np.exp(6.0 * (b[top_k] - m))
    ts = w.sum()
    if ts <= 0:
        w = np.ones(len(top_k)) / len(top_k)
    else:
        w = w / ts
    sg = np.zeros(n)
    for k_idx, weight in zip(top_k, w):
        k = int(k_idx)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += 2.0 * weight * a_n[k - np.arange(js, je + 1)]
    a_new = a_n - step * sg
    return _simplex_project(a_new)


def _optimize(a_init, deadline, max_lp_iters=3000):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s <= 0:
        return a_init
    a = a / s
    best = a.copy()
    best_sc = _score_fast(a)
    stag = 0

    # Phase 1: LP iterations
    for it in range(max_lp_iters):
        if time.monotonic() > deadline - 60.0:
            break
        g = _lp_direction(a, deadline)
        if g is None:
            break
        # Line search over t
        best_t = 0.0
        best_cand_sc = best_sc
        best_cand = best
        for t in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.65, 0.8, 1.0):
            cand = (1.0 - t) * a + t * g
            cand = _simplex_project(cand)
            sc = _score_fast(cand)
            if sc < best_cand_sc - 1e-12:
                best_cand_sc = sc
                best_t = t
                best_cand = cand
        if best_t > 0:
            a = best_cand
            if best_cand_sc < best_sc - 1e-12:
                best = a.copy()
                best_sc = best_cand_sc
            stag = 0
        else:
            stag += 1
            if stag >= 8:
                break

    # Phase 2: Mirror descent refinement
    remaining = deadline - time.monotonic() - 10.0
    if remaining > 5.0:
        step = 0.3
        md_stag = 0
        while time.monotonic() < deadline - 10.0 and md_stag < 30:
            b = _fft_conv(a)
            a_new_m = _mirror_step(a, b, step)
            a_new_s = _subgrad_step(a, b, step * 0.1)
            sc_m = _score_fast(a_new_m)
            sc_s = _score_fast(a_new_s)
            if sc_m < best_sc - 1e-12:
                a = a_new_m
                best = a.copy()
                best_sc = sc_m
                md_stag = 0
                step = min(2.0, step * 1.05)
            elif sc_s < best_sc - 1e-12:
                a = a_new_s
                best = a.copy()
                best_sc = sc_s
                md_stag = 0
                step = min(2.0, step * 1.05)
            else:
                step = max(0.01, step * 0.85)
                md_stag += 1

    # Phase 3: Fine subgradient polish
    if time.monotonic() < deadline - 5.0:
        step = 0.01
        fine_stag = 0
        while time.monotonic() < deadline - 5.0 and fine_stag < 50:
            b = _fft_conv(a)
            a_new = _subgrad_step(a, b, step)
            sc = _score_fast(a_new)
            if sc < best_sc - 1e-14:
                a = a_new
                best = a.copy()
                best_sc = sc
                fine_stag = 0
            else:

                fine_stag += 1

    return best, best_sc
```
