        sx = ex.sum()
        if sx > 0 and np.isfinite(sx):
            p = ex / sx
        else:
            p = np.zeros(2 * n - 1)
            p[int(np.argmax(b))] = 1.0

        h = _grad_h(p, a, fsize)
        c_val = float(np.dot(p, b))
        grad = 2.0 * a * (h - c_val)
        grad -= grad.mean()

            stuck += 2
            if stuck > 40:

            z += rng.normal(0, 0.3, n)
            z -= z.mean()
            continue

            if stuck > 40:

            continue

        m_ad = 0.9 * m_ad + 0.1 * grad
        v_ad = 0.999 * v_ad + 0.001 * grad * grad

        v_hat = v_ad / (1.0 - 0.999 ** t_ad)
