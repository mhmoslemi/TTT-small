def _normalize_prob(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return np.ones(1, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)
    return x / s


def _get_parent():

        p = np.asarray(p, dtype=np.float64).ravel()
    except Exception:

def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.ones(1, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z - np.max(z), -50.0, 50.0)
    e = np.exp(z)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0 / z.size, dtype=np.float64)
    return e / s


def _dense_max_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size <= 1:
        return float(a[0] * a[0]) if a.size else 0.0
    c = np.convolve(a, a)
    c = np.where(np.isfinite(c), c, 0.0)
    return float(np.max(c)) if c.size else 0.0


def _fft_conv(a, L=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    if L is None:
        L = next_fast_len(2 * n - 1)
    F = rfft(a, L)
    c = irfft(F * F, L)
    c = c[: (2 * n - 1)]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return c


def _fft_max_conv(a, L=None):
    c = _fft_conv(a, L)
    return float(np.max(c)) if c.size else 0.0

def _corr_p_a(a, p, L=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    p = np.asarray(p, dtype=np.float64).ravel()
    need = 2 * n - 1
    if p.size != need:
        tmp = np.zeros(need, dtype=np.float64)
        m = min(p.size, need)
        if m:
            tmp[:m] = p[:m]
        s = float(tmp.sum())
        if s > 0:
            tmp /= s
        else:
            tmp[:] = 1.0 / need
        p = tmp

    if L is None:
        L = next_fast_len(3 * n - 2)
    Fa = rfft(a, L)
    Fp = rfft(p[::-1], L)
    c = irfft(Fa * Fp, L)
    c = np.where(np.isfinite(c), c, 0.0)
    idx = np.arange(2 * n - 2, n - 2, -1)
    q = c[idx]
    q = np.where(np.isfinite(q), q, 0.0)
    q = np.maximum(q, 0.0)
    return q


def _lp_direction(f, rng=None):
    f = _normalize_prob(f)
    n = f.size
    if n == 0:
        return f
    if n == 1:
        return f.copy(), 1.0

    scale = float(n)
    fw = f * scale
    m = 2 * n - 1

    A_ub = np.zeros((m, n + 1), dtype=np.float64)
    for j in range(n):
        A_ub[j : j + n, j] = fw
    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([scale], dtype=np.float64)
    b_ub = np.zeros(m, dtype=np.float64)

    c = np.zeros(n + 1, dtype=np.float64)
    if rng is not None:
        try:
            c[:n] = 1e-10 * rng.standard_normal(n)
        except Exception:
            pass
    c[-1] = 1.0

    bounds = [(0.0, None)] * n + [(0.0, None)]

    res = None
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"time_limit": 20.0},
        )
    except Exception:
        try:
            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
            )
        except Exception:
            try:
                res = linprog(
                    c,
                    A_ub=A_ub,
                    b_ub=b_ub,
                    A_eq=A_eq,
                    b_eq=b_eq,
                    bounds=bounds,
                    method="interior-point",
                )
            except Exception:
                res = None

    if (
        res is not None
        and getattr(res, "success", False)
        and getattr(res, "x", None) is not None

        g = np.asarray(res.x[:n], dtype=np.float64)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        gs = float(g.sum())
        if gs > 1e-300:

    return f.copy(), _dense_max_conv(f)

            gdir = grad / gmax

            steps = []
            for base in (eta, 0.60, 0.42, 0.30, 0.22, 0.15, 0.10, 0.06, 0.03, 0.015):
                for mult in (1.0, 0.5):
                    st = float(base) * float(mult)
                    if 1e-8 < st < 1.5:
                        if all(abs(st - u) > 1e-9 for u in steps):
                            steps.append(st)
            steps.sort(reverse=True)
            if not steps:
                steps = [0.3, 0.2, 0.1, 0.05, 0.025, 0.01]

            accepted = False
            best_step = None
            best_z = None
            best_a = None
            best_step_type = 0
            best_md = None
            best_wc = None

            for st in steps:
                zc = z - st * gdir
                zc = np.where(np.isfinite(zc), zc, 0.0)
                zc = np.clip(zc, -50.0, 50.0)
                ac = _softmax(zc)
                ac = _normalize_prob(ac)

                cc = _fft_conv(ac, Lconv)
                Mcff = float(cc.max()) if cc.size else 0.0
                Wc = float(np.dot(p, cc))

                if Mcff < Mfft * (1.0 - 1e-12) or (
                    math.isfinite(Wc)
                    and Wc < W * (1.0 - 1e-8)
                    and Mcff < Mfft * 1.05
                ):
                    md = _dense_max_conv(ac)
                    if md < M_dense_cur - 1e-13:
                        best_step = st
                        best_z = zc
                        best_a = ac
                        best_md = md
                        best_wc = Wc
                        best_step_type = 1
                        break
                    if (
                        surrogate_streak < 60
                        and mat...
                        and md < M_dense_cur * 1.015
                    ):
                        if best_step is None or Wc < best_wc:
                            best_step = st
                            best_z = zc
                            best_a = ac
                            best_md = md
                            best_wc = Wc
                            best_step_type = 2

            if best_step is not None and best_a is not None:
                z = best_z
                a = best_a
                M_dense_cur = float(best_md)
                accepted = True

                if best_step_type == 1:
                    surrogate_streak = 0
                    eta = min(0.85, eta * 1.05)
                    tau_frac = max(0.003, tau_frac * 0.996)
                    noise = min(noise, 0.10)
                    stall = 0
                else:
                    surrogate_streak += 1
                    eta = min(0.85, eta * 1.01)

                if M_dense_cur < best_M - 1e-12:
                    best_M = M_dense_cur
                    best = a.copy()

            if not accepted:
                stall += 1
                eta = max(0.02, eta * 0.86)
                tau_frac = min(0.45, tau_frac * 1.25)
                noise = min(0.9, noise * 1.18)
                if stall > 30:
                    base = best.copy()
                    z = np.log(base + 1e-300)
                    z += noise * rng.standard_normal(n)
                    z -= z.mean()
                    a = _softmax(z)
                    a = _normalize_prob(a)
                    M_dense_cur = _dense_max_conv(a)
                    stall = 10
                    eta = max(eta, 0.18)
                    tau_frac = min(0.25, max(tau_frac, 0.12))
                    surrogate_streak = 0
                if stall > 160:
                    break

    return best, float(2.0 * n * best_M)


def _random_polish(f0, stop_time, seed=3):
    a = _normalize_prob(f0)
    n = a.size
    if n <= 1:
        return a, (2.0 if n == 1 else math.inf)

    rng = np.random.default_rng(seed)
    best = a.copy()
    best_M = _dense_max_conv(best)
    cur_M = best_M
    Lconv = next_fast_len(2 * n - 1)

    sigma = 0.045
    stuck = 0

    while time.monotonic() < stop_time:
        if time.monotonic() > stop_time:
            break

        r = rng.random()
        if r < 0.38:
            z = np.log(a + 1e-300)
            z += sigma * rng.standard_normal(n)
            z -= z.mean()
            cand = _softmax(z)
        elif r < 0.60:
            w = max(1, n // 40)
            w = min(w, max(1, n // 4))
            s = int(rng.integers(0, max(1, n - w)))
            cand = a.astype(np.float64).copy()
            nd = min(w, n - s)
            if nd > 0:
                cand[s : s + nd] *= float(np.exp(rng.normal(0.0, 0.22)))
        elif r < 0.72:
            t = float(rng.uniform(0.0, 0.06))
            cand = (1.0 - t) * a + t * a[::-1]
        elif r < 0.86:
            f = int(rng.integers(1, max(2, n // 3)))
            ph = float(rng.uniform(0.0, 2.0 * math.pi))
            xx = np.arange(n, dtype=np.float64)
            mod = 1.0 + float(rng.uniform(0.03, 0.12)) * np.sin(
                2.0 * math.pi * f * xx / float(n) + ph
            )
            cand = a * mod
        else:
            cand = a.astype(np.float64).copy()
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            cand[i] *= float(rng.uniform(0.55, 1.7))
            cand[j] *= float(rng.uniform(0.55, 1.7))

        cand = _normalize_prob(cand)
        cf = cand
        Mc_flash = _fft_max_conv(cf, Lconv)
        if Mc_flash < cur_M - 1e-12:
            Mc = _dense_max_conv(cand)
            if Mc < cur_M - 1e-13:
                a = cand
                cur_M = Mc
                stuck = 0
                sigma = max(0.018, sigma * 0.965)
                if Mc < best_M - 1e-13:
                    best_M = Mc
                    best = a.copy()
            else:
                stuck += 1
        else:
            stuck += 1
            if stuck % 60 == 0:
                sigma = min(0.18, sigma * 1.35)
            if stuck > 700:
                a = best.copy()
                cur_M = _dense_max_conv(a)
                stuck = 0
                sigma = 0.045

    return best, float(2.0 * n * best_M)


def _resample_to_length(x, n_new):
    x = np.asarray(x, dtype=np.float64).ravel()
    n_old = x.size
    if n_old == 0:
        return np.ones(max(1, int(n_new)), dtype=np.float64)
    n_new = int(max(1, n_new))
    if n_old == n_new:
        return x.copy()
    xo = np.linspace(0.0, 1.0, n_old)
    xn = np.linspace(0.0, 1.0, n_new)
    y = np.interp(xn, xo, x)
    return _normalize_prob(y)


def _make_stop(deadline, frac, cap, min_time=5.0):
    rem = max(0.0, float(deadline) - time.monotonic())
    if rem <= min_time + 1.0:
        return time.monotonic()
    alloc = min(float(cap), frac * rem, rem - 10.0)
    if alloc < min_time:
        return time.monotonic()
    return time.monotonic() + alloc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1300.0
    deadline = t0 + max(20.0, budget - 25.0)

    parent = _get_parent()
    fallback = _normalize_prob(parent)
    fallback_score = _exact_score(fallback)
    if not math.isfinite(fallback_score):
        fallback = np.ones(192, dtype=np.float64) / 192.0
        fallback_score = _exact_score(fallback)
        if not math.isfinite(fallback_score):
            return [1.0]

    best = fallback.copy()
    best_score = fallback_score

    rng = np.random.default_rng(seed)

    # A few very cheap smooth restarts around the supplied parent.
    init_candidates = [best]
    try:
        n0 = best.size
        if n0 > 1:
            init_candidates.append(0.55 * best + 0.45 * best[::-1])
            init_candidates.append(best * (1.0 + 0.05 * rng.standard_normal(n0)))
            init_candidates.append(best * (1.0 + 0.08 * rng.standard_normal(n0)))
            init_candidates.append(
                0.85 * best + 0.15 * (1.0 / n0 + 0.5 * (1.0 + np.cos(np.pi * np.arange(n0) / max(1, n0))))
            )
    except Exception:
        pass

    for cand in init_candidates:
        cand = _normalize_prob(cand)
        fast = 2.0 * cand.size * _fft_max_conv(cand) / (float(cand.sum()) ** 2)
        if math.isfinite(fast) and fast < best_score - 1e-9:
            best, best_score = _update_if_better(best, best_score, cand)

    n = best.size

    # Main LP projection iterations on the current best length.
    if 2 <= n <= 512:
        stop = _make_stop(deadline, 0.36, 700.0)
        if stop > time.monotonic() + 6.0:
            try:
                cand, _ = _lp_refine(best, stop, int(seed) + 101)
            except Exception:
                cand = best
            best, best_score = _update_if_better(best, best_score, cand)

    # Gradient / soft-peak descent.
    stop = _make_stop(deadline, 0.30, 650.0)
    if stop > time.monotonic() + 6.0:
        try:
            cand, _ = _gradient_refine(best, stop, int(seed) + 202)
        except Exception:
            cand = best
        best, best_score = _update_if_better(best, best_score, cand)

    # Another LP pass after gradient movement.
    n = best.size
    if 2 <= n <= 512:
        stop = _make_stop(deadline, 0.12, 260.0)
        if stop > time.monotonic() + 5.0:
            try:
                cand, _ = _lp_refine(best, stop, int(seed) + 303)
            except Exception:
                cand = best
            best, best_score = _update_if_better(best, best_score, cand)

    # Final directed gradient pass.
    stop = _make_stop(deadline, 0.18, 420.0)
    if stop > time.monotonic() + 5.0:
        try:
            cand, _ = _gradient_refine(best, stop, int(seed) + 404)
        except Exception:
            cand = best
        best, best_score = _update_if_better(best, best_score, cand)

    # If still above target, briefly explore nearby compact lengths.
    if best_score > 1.5028:
        n_cur = best.size
        tried = set()
        for N in (int(round(n_cur * 4.0 / 3.0)), int(round(n_cur * 5.0 / 4.0)), max(16, n_cur - 16), n_cur + 16):
            if time.monotonic() > deadline - 60.0:
                break
            if N in tried or N < 16 or N > 1024 or N == n_cur:
                continue
            tried.add(N)

            cand = _resample_to_length(best, N)
            # A short LP pass if size allows.
            if 2 <= N <= 768:
                stopN = _make_stop(deadline, 0.18, 120.0)
                if stopN > time.monotonic() + 5.0:
                    try:
                        c2, _ = _lp_refine(cand, stopN, int(seed) + 500 + N)
                    except Exception:
                        c2 = cand
                    cand = c2
            stopN = _make_stop(deadline, 0.12, 90.0)
            if stopN > time.monotonic() + 4.0:
                try:
                    c3, _ = _gradient_refine(cand, stopN, int(seed) + 600 + N)
                except Exception:
                    c3 = cand
                cand = c3

            best, best_score = _update_if_better(best, best_score, cand)

    # Random local polish on the retained best sequence.
    stop = _make_stop(deadline, 0.25, 650.0, min_time=3.0)
    if time.monotonic() < stop:
        try:
            cand, _ = _random_polish(best, stop, int(seed) + 707)
        except Exception:
            cand = best
        best, best_score = _update_if_better(best, best_score, cand)

    # Final exact validation and safe return.
    best = _normalize_prob(best)
    best_score_final = _exact_score(best)
    fallback_score_final = _exact_score(fallback)

    if (
        not math.isfinite(best_score_final)
        or (math.isfinite(fallback_score_final) and best_score_final > fallback_score_final + 1e-12)
    ):
        out = fallback
    else:
        out = best

    out = np.asarray(out, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if out.size == 0 or s <= 1e-8:
        out = np.ones(192, dtype=np.float64) / 192.0
    else:
        out = out / s

    if out.size == 0:
        return [1.0]

    return [float(v) for v in out]
