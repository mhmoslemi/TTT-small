    return (e / s).astype(np.float64, copy=False)


def _log_z(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-12)
    z = np.log(a)
    z = z - float(z.mean())
    z = np.where(np.isfinite(z), z, 0.0)
    return np.clip(z, -25.0, 25.0).astype(np.float64, copy=False)

def _corr_valid(p, a):
    n = int(a.size)
    m = int(p.size)
    if n == 0 or m == 0:
        return np.zeros(0, dtype=np.float64)
    if n == 1:
        return np.array([float(p.sum()) * float(a[0])], dtype=np.float64)
    if n <= 384:
        val = np.convolve(p, a[::-1], mode="valid")
    else:
        L = next_fast_len(m + n - 1)
        fp = rfft(p, L)
        fa = rfft(a[::-1], L)
        val = irfft(fp * fa, L)
        val = val[n - 1 : n - 1 + n]
    val = np.asarray(val, dtype=np.float64)
    val = np.nan_to_num(val, nan=0.0, posinf=0.0, neginf=0.0)
    if val.size != n:
        out = np.empty(n, dtype=np.float64)
        for i in range(n):
            out[i] = float(np.dot(p[i : i + n], a))
        val = out
    return val


def _soft_weights_U(c, M, tau):
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size == 0:
        return np.zeros(0, dtype=np.float64), 0.0
    M = float(M)
    if not np.isfinite(M) or M <= 0.0:
        p = np.zeros(c.size, dtype=np.float64)
        if c.size:
            p[int(np.argmax(c))] = 1.0
        return p, 0.0
    if not np.isfinite(tau) or tau <= 1e-15:
        tol = 1e-12 * max(1.0, abs(M))
        idx = np.where(np.abs(c - M) <= tol)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(c))], dtype=int)
        p = np.zeros(c.size, dtype=np.float64)
        p[idx] = 1.0 / idx.size
        return p, M
    x = (c - M) / tau
    x = np.minimum(0.0, x)
    x = np.maximum(x, -745.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        idx = np.where(np.abs(c - M) <= 1e-12 * max(1.0, abs(M)))[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(c))], dtype=int)
        p = np.zeros(c.size, dtype=np.float64)
        p[idx] = 1.0 / idx.size
        return p, M
    p = ex / s
    U = M + tau * math.log(s)
    if not np.isfinite(U):
        U = M
    return p, float(U)


def _grad_soft(a, tau):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64), 0.0, 0.0
    c = _conv(a)
    if c.size == 0:
        return np.zeros(n, dtype=np.float64), 0.0, 0.0
    M = float(c.max())
    if not np.isfinite(M) or M <= 0.0:
        return np.zeros(n, dtype=np.float64), 0.0, 0.0
    p, U = _soft_weights_U(c, M, tau)
    corr = _corr_valid(p, a)
    hg = 2.0 * corr
    if hg.size != n:
        hg = np.nan_to_num(hg, nan=0.0, posinf=0.0, neginf=0.0)
        if hg.size > n:
            hg = hg[:n]
        else:
            out = np.zeros(n, dtype=np.float64)
            out[: hg.size] = hg
            hg = out
    S = float(np.dot(a, hg))
    if not np.isfinite(S):
        S = 0.0
    gz = a * (hg - S)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = gz - float(gz.mean())
    gz = np.where(np.isfinite(gz), gz, 0.0)
    return gz.astype(np.float64, copy=False), float(U), M


def _lbfgs_phase(a, tau, maxiter, deadline):
    a = _normalize(a)
    n = a.size
    if not _HAVE_SCIPY_OPT or n == 0:
        return a
    z0 = _log_z(a)

    def fg(z):
        if time.monotonic() > deadline:
            raise _Deadline
        z = np.asarray(z, dtype=np.float64).ravel()
        z = np.where(np.isfinite(z), z, 0.0)
        zc = z - float(z.mean())
        zc = np.clip(zc, -25.0, 25.0)
        aa = _softmax(zc)
        _, U, M = _grad_soft(aa, tau)
        if not np.isfinite(U):
            U = 1.0e12
        if not np.isfinite(M):
            gz = np.zeros(n, dtype=np.float64)
        else:
            gz, U2, _ = _grad_soft(aa, tau)
            gz = np.where(np.isfinite(gz), gz, 0.0)
            U = U2
        return float(U), gz

    try:
        res = _sp_minimize(
            fg,
            z0,
            jac=True,
            method="L-BFGS-B",
            options={
                "maxiter": int(max(1, maxiter)),
                "ftol": 1e-12,
                "gtol": 1e-6,
                "maxcor": 10,
                "maxls": 30,
            },
        )
    except _Deadline:
        return a
    except Exception:
        return a

    best = a.copy()
    bestM = _peak(best)
    if hasattr(res, "x") and res.x is not None:
        try:
            x = np.asarray(res.x, dtype=np.float64).ravel()
            x = np.where(np.isfinite(x), x, 0.0)
            aa = _softmax(x - float(x.mean()))
            M2 = _peak(aa)
            if np.isfinite(M2) and M2 < bestM - 1e-14:
                best = aa
        except Exception:
            pass
    return best


def _lbfgs_loop(a, stop):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return a
    current = a.copy()
    currentM = _peak(current)
    frac = 0.30
    min_frac = 1.0e-3
    maxiter = 120 if n <= 256 else (70 if n <= 512 else 40)
    phase_cap = 60.0 if n <= 256 else (30.0 if n <= 512 else 15.0)

    while time.monotonic() < stop - 0.4 and frac > min_frac * 0.5:
        tau = frac * max(currentM, 1.0e-12)
        now = time.monotonic()
        phase_stop = min(stop, now + max(2.0, min(phase_cap, (stop - now) / 3.0)))
        a2 = _lbfgs_phase(current, tau, maxiter, phase_stop)
        M2 = _peak(a2)
        if np.isfinite(M2) and M2 < currentM - 1e-14:
            current = a2
            currentM = M2
            frac = max(min_frac, frac * 0.60)
        else:
            frac = max(min_frac, frac * 0.70)
        if time.monotonic() > stop - 1.5:
            break
    return current


def _adam_search(a_init, stop, seed=0):
    a0 = _normalize(a_init)
    n = a0.size
    if n == 0:
        return a0, math.inf
    best = a0.copy()
    bestM = _peak(best)

    z = _log_z(best)
    a_cur = best.copy()
    M_cur = bestM

    rng = np.random.default_rng(seed)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    tcount = 0
    base_lr = 0.35 if n <= 256 else 0.22
    lr = base_lr
    tau_frac = 0.18
    perturb = 0.7
    stuck = 0

    while time.monotonic() < stop - 0.2:
        if M_cur > bestM * (1.0 + 1e-5) and stuck > 6:
            z = _log_z(best)
            a_cur = best.copy()
            M_cur = bestM
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            tcount = 0
            lr = base_lr
            tau_frac = 0.18
            stuck = 0

        tau = max(1.0e-18, tau_frac * max(M_cur, 1.0e-12))
        gz, _, _ = _grad_soft(a_cur, tau)
        gmax = float(np.max(np.abs(gz))) if gz.size else 0.0

        if not np.isfinite(gmax) or gmax < 1e-14:
            z = z + rng.normal(0.0, perturb, n)
            z = z - float(z.mean())
            z = np.clip(z, -25.0, 25.0)
            a_cur = _softmax(z)
            M_cur = _peak(a_cur)
            if M_cur < bestM - 1e-14:
                best = a_cur.copy()
                bestM = M_cur
                stuck = 0
            else:
                stuck += 1
                if stuck > 12:
                    z = _log_z(best)
                    a_cur = best.copy()
                    M_cur = bestM
                    m = np.zeros(n, dtype=np.float64)
                    v = np.zeros(n, dtype=np.float64)
                    tcount = 0
                    lr = base_lr
                    tau_frac = min(0.40, tau_frac * 1.5 + 0.01)
                    stuck = 0
            continue

        tcount += 1
        m = 0.90 * m + 0.10 * gz
        v = 0.999 * v + 0.001 * (gz * gz)
        mhat = m / (1.0 - 0.90 ** tcount)
        vhat = v / (1.0 - 0.999 ** tcount)
        direction = mhat / (np.sqrt(vhat) + 1e-8)
        direction = np.where(np.isfinite(direction), direction, 0.0)
        dmax = float(np.max(np.abs(direction))) if direction.size else 0.0

        if not np.isfinite(dmax) or dmax < 1e-14:
            z = z + rng.normal(0.0, perturb, n)
            z = z - float(z.mean())
            z = np.clip(z, -25.0, 25.0)
            a_cur = _softmax(z)
            M_cur = _peak(a_cur)
            stuck += 1
            if stuck > 10:
                z = _log_z(best)
                a_cur = best.copy()
                M_cur = bestM
                m = np.zeros(n, dtype=np.float64)
                v = np.zeros(n, dtype=np.float64)
                tcount = 0
                lr = base_lr
                tau_frac = min(0.40, tau_frac * 1.4 + 0.02)
                stuck = 0
            continue

        direction = direction / dmax
        accepted = False

        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
            zn = z - lr * factor * direction
            zn = zn - float(zn.mean())
            zn = np.where(np.isfinite(zn), zn, 0.0)
            zn = np.clip(zn, -25.0, 25.0)
            an = _softmax(zn)
            Mn = _peak(an)
            if np.isfinite(Mn) and Mn < M_cur - 1e-14:
                z = zn
                a_cur = an
                M_cur = Mn
                accepted = True
                break

        if accepted:
            if M_cur < bestM - 1e-14:
                best = a_cur.copy()
                bestM = M_cur
            stuck = 0
            if tcount % 25 == 0:
                tau_frac = max(0.003, tau_frac * 0.85)

            sym = 0.5 * (a_cur + a_cur[::-1])
            ss = float(sym.sum())
            if ss > 1e-15:
                sym = sym / ss
                Ms = _peak(sym)
                if np.isfinite(Ms) and Ms < M_cur - 1e-14:
                    a_cur = sym
                    M_cur = Ms
                    if M_cur < bestM - 1e-14:
                        best = sym.copy()
                        bestM = M_cur
        else:
            stuck += 1
            lr = max(0.02, lr * 0.86)
            if stuck >= 12:
                z = _log_z(best) + rng.normal(0.0, perturb, n)
                z = z - float(z.mean())
                z = np.clip(z, -25.0, 25.0)
                a_cur = _softmax(z)
                M_cur = _peak(a_cur)
                if not np.isfinite(M_cur) or M_cur > bestM * 1.25:
                    z = _log_z(best)
                    a_cur = best.copy()
                    M_cur = bestM
                m = np.zeros(n, dtype=np.float64)
                v = np.zeros(n, dtype=np.float64)
                tcount = 0
                lr = base_lr
                tau_frac = min(0.40, tau_frac * 1.4 + 0.02)
                stuck = 0

    return best, bestM


def _lp_polish(a, stop, max_steps=40):
    a = _normalize(a)
    n = a.size
    if n == 0 or n > 512 or not _HAVE_SCIPY_OPT:
        return a
    best = a.copy()
    bestM = _peak(best)
    steps = 0

    while time.monotonic() < stop - 1.2 and steps < max_steps:
        C = np.zeros((2 * n - 1, n), dtype=np.float64)
        try:
            for i in range(n):
                C[i : i + n, i] = a
        except Exception:
            break

        b1 = np.ones(2 * n - 1, dtype=np.float64)
        h = None
        rem = max(0.1, stop - time.monotonic() - 2.0)

        try:
            res = _sp_linprog(
                -np.ones(n, dtype=np.float64),
                A_ub=C,
                b_ub=b1,
                bounds=(0.0, 1000.0),
                method="highs",
                options={"time_limit": max(0.2, min(30.0, rem))},
            )
            if getattr(res, "success", False) and res.x is not None:
                h = res.x
        except Exception:
            h = None

        if h is None:
            try:
                res = _sp_linprog(
                    -np.ones(n, dtype=np.float64),
                    A_ub=C,
                    b_ub=b1,
                    bounds=(0.0, 1000.0),
                    method="revised simplex",
                )
                if getattr(res, "success", False) and res.x is not None:
                    h = res.x
            except Exception:
                h = None

        if h is None:
            break

        h = np.asarray(h, dtype=np.float64)
        h = np.where(np.isfinite(h), h, 0.0)
        h = np.maximum(h, 0.0)
        hs = float(h.sum())
        if hs <= 1e-15:
            break
        h = h / hs
        hs2 = 0.5 * (h + h[::-1])
        hs2s = float(hs2.sum())
        if hs2s > 1e-15:
            hs2 = hs2 / hs2s
        else:
            hs2 = h

        best_cand = None
        best_candM = bestM
        dirs = (h, hs2)
        for d in dirs:
            for t in (0.1, 0.2, 0.35, 0.5, 0.7, 1.0):
                cand = (1.0 - t) * best + t * d
                cand = _normalize(cand)
                Mc = _peak(cand)
                if np.isfinite(Mc) and Mc < best_candM - 1e-14:
                    best_cand = cand
                    best_candM = Mc

        if best_cand is None:
            break
        best = best_cand
        bestM = best_candM
        steps += 1

    return best


def _analytic_seed(n, shift=True):
    try:
        n = int(max(2, n))
        L = next_fast_len(max(16 * n, 256))
        k = np.arange(L, dtype=np.float64)
        omega = 2.0 * math.pi * k / float(L)
        omega = np.where(omega > math.pi, omega - 2.0 * math.pi, omega)
        aw = np.abs(omega)
        val = np.ones(L, dtype=np.float64)
        mask = aw > 1e-12
        val[mask] = np.sin(0.5 * aw[mask]) / aw[mask]
        val = np.clip(val, 0.0, None)
        G = np.sqrt(val)
        idx = np.arange(L)
        ev = np.minimum(idx, L - idx)
        G = G[ev]

        f = np.fft.ifft(G).real
        if shift:
            t0 = 0.5
        else:
            t0 = 0.0
        idxn = np.rint((np.arange(n, dtype=np.float64) / float(n) + t0) * float(L)).astype(int) % L
        seg = f[idxn]
        seg = np.where(np.isfinite(seg), seg, 0.0)
        seg = np.maximum(seg, 0.0)
        s = float(seg.sum())
        if s <= 1e-15:
            seg = np.ones(n, dtype=np.float64)
            s = float(seg.sum())
        seg = seg / s
        seg = 0.5 * (seg + seg[::-1])
        ss = float(seg.sum())
        if ss <= 1e-15:
            return np.full(n, 1.0 / n, dtype=np.float64)
        return (seg / ss).astype(np.float64, copy=False)
    except Exception:
        return np.full(int(max(2, n)), 1.0 / int(max(2, n)), dtype=np.float64)


def _flat_iter_seed(n, iterations=20):
    try:
        n = int(max(2, n))
        L = next_fast_len(2 * n + 128)
        b = np.zeros(L, dtype=np.float64)
        b[: (2 * n - 1)] = 1.0 / float(2 * n - 1)
        B = rfft(b, L)
        target_mag = np.sqrt(np.abs(B) + 1.0e-30)

        prior = _analytic_seed(n, shift=True)
        P = rfft(prior, L)
        p = prior.astype(np.float64, copy=True)

        for _ in range(int(max(0, iterations))):
            phase = np.exp(1.0j * np.angle(P))
            Pnew = target_mag * phase
            Pnew[0] = abs(Pnew[0])
            if L % 2 == 0:
                Pnew[-1] = abs(Pnew[-1])
            pn = irfft(Pnew, L)[:n]
            pn = np.where(np.isfinite(pn), pn, 0.0)
            pn = np.maximum(pn, 0.0)
            s = float(pn.sum())
            if s <= 1e-15:
                break
            pn = pn / s
            p = pn
            P = rfft(p, L)

        p = np.where(np.isfinite(p), p, 1.0 / n)
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if s <= 1e-15:
            return np.full(n, 1.0 / n, dtype=np.float64)
        return (p / s).astype(np.float64, copy=False)
    except Exception:
        return np.full(int(max(2, n)), 1.0 / int(max(2, n)), dtype=np.float64)


def _random_seed(n, rng, vid):
    try:
        n = int(max(2, n))
        t = np.linspace(0.0, 1.0, n)
        v = int(vid) % 7
        if v == 0:
            a = np.ones(n, dtype=np.float64)
        elif v == 1:
            a = 0.15 + 0.85 * np.sin(math.pi * t) + 0.08 * rng.normal(0.0, 1.0, n)
        elif v == 2:
            a = 0.25 + 0.75 * (2.0 * t - 1.0) ** 2 + 0.10 * rng.normal(0.0, 1.0, n)
        elif v == 3:
            a = np.exp(-2.5 * (t - 0.5) ** 2 + 0.12 * rng.normal(0.0, 1.0, n))
        elif v == 4:
            a = 0.35 + 0.55 * np.cos(2.0 * math.pi * t) + 0.10 * rng.normal(0.0, 1.0, n)
        else:
            a = np.maximum(0.1, 0.5 + 0.5 * rng.normal(0.0, 0.5, n))
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.maximum(a, 0.0)
        if v % 2 == 0:
            a = 0.5 * (a + a[::-1])
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s <= 1e-15:
            return np.full(n, 1.0 / n, dtype=np.float64)
        return (a / s).astype(np.float64, copy=False)
    except Exception:
        return np.full(int(max(2, n)), 1.0 / int(max(2, n)), dtype=np.float64)


def _optimize_one(a, stop, seed=0):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return a
    best = a.copy()
    bestM = _peak(best)
    now = time.monotonic()
    total = max(0.0, stop - now)
    if total < 2.5:
        return best

    if total > 12 and n <= 1024:
        lb_budget = min(total * 0.40, 300.0 if n <= 256 else (120.0 if n <= 512 else 60.0))
        lb_stop = min(stop, now + max(2.0, lb_budget))
        if lb_stop > now + 3.0:
            a2 = _lbfgs_loop(best, lb_stop)
            M2 = _peak(a2)
            if np.isfinite(M2) and M2 < bestM - 1e-14:
                best = a2
                bestM = M2

    if time.monotonic() < stop - 4.0:
        rem = stop - time.monotonic()
        ad_budget = min(rem * 0.60, 700.0)
        ad_stop = min(stop, time.monotonic() + max(2.0, ad_budget))
        if ad_stop > time.monotonic() + 3.0:
            a2, M2 = _adam_search(best, ad_stop, seed=seed)
            if np.isfinite(M2) and M2 < bestM - 1e-14:
                best = a2
                bestM = M2

    if n <= 512 and time.monotonic() < stop - 4.0:
        rem = stop - time.monotonic()
        lp_budget = min(90.0, 0.25 * rem)
        lp_stop = min(stop, time.monotonic() + max(2.0, lp_budget))
        if lp_stop > time.monotonic() + 3.0:
            a2 = _lp_polish(best, lp_stop, max_steps=40)
            M2 = _peak(a2)
            if np.isfinite(M2) and M2 < bestM - 1e-14:
                best = a2
                bestM = M2

    if time.monotonic() < stop - 3.0:
        a2, M2 = _adam_search(best, max(time.monotonic() + 1.0, stop - 1.0), seed=seed + 991)
        if np.isfinite(M2) and M2 < bestM - 1e-14:
            best = a2
            bestM = M2

    return best


def _search_length(n, alloc, parent, global_best, seed, deadline):
    stop = min(deadline, time.monotonic() + max(0.0, alloc))
    n = int(max(2, n))
    if stop - time.monotonic() < 6.0:
        return None

    seeds = []
    if parent is not None and parent.size > 0:
        p = _resize(parent, n)
        seeds.append(p)
        ps = 0.5 * (p + p[::-1])
        seeds.append(_normalize(ps))
    if global_best is not None and global_best.size > 0:
        seeds.append(_resize(global_best, n))
    seeds.append(_analytic_seed(n, shift=True))
    seeds.append(_flat_iter_seed(n, iterations=20))
    rng = np.random.default_rng(seed + 9176)
    for j in range(3):
        seeds.append(_random_seed(n, rng, j))

    candidates = []
    for s in seeds:
        try:
            s2 = _normalize(s)
            if s2.size != n:
                s2 = _resize(s2, n)
            sc = _fast_score(s2)
            if np.isfinite(sc):
                candidates.append((sc, s2))
        except Exception:
            continue

    if not candidates:
        return None

    candidates.sort(key=lambda x: x[0])
    top = candidates[:4]
    local_best = top[0][1].copy()
    local_score = top[0][0]

    for idx, (sc, s) in enumerate(top):
        if time.monotonic() > stop - 8.0:
            break
        remaining_terms = len(top) - idx
        per = (stop - time.monotonic() - 5.0) / max(1, remaining_terms)
        if per < 8.0:
            break
        seg_stop = min(stop, time.monotonic() + per)
        try:
            a2 = _optimize_one(s, seg_stop, seed=seed + 1000 * n + 31 * idx)
        except Exception:
            continue
        if a2 is None or a2.size != n:
            continue
        a2 = _normalize(a2)
        sc2 = _fast_score(a2)
        if np.isfinite(sc2) and sc2 < local_score - 1e-12:
            local_score = sc2
            local_best = a2.copy()

    return local_best


def _candidate_lengths(primary):
    primary = int(primary)
    standard = [
        8, 12, 16, 24, 32, 48, 64, 96, 128,
        160, 192, 224, 256, 320, 384, 512, 640,
        768, 896, 1024,
    ]
    others = []
    for m in standard:
        if 8 <= m <= 1024 and m != primary:
            others.append(m)
    others.sort(key=lambda m: abs(math.log(max(1, m) / max(2, primary))))
    out = []
    seen = {primary}
    for m in others:
        if m not in seen:
            seen.add(m)
            out.append(m)
        if len(out) >= 5:
            break
    return out


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1300.0
    try:
        seed_i = int(seed)
    except Exception:
        seed_i = 42

    if budget > 120.0:
        deadline = t0 + budget - 45.0
    elif budget > 60.0:
        deadline = t0 + budget - 15.0
    else:
        deadline = t0 + max(1.0, budget - 3.0)
    deadline = max(t0 + 1.0, min(deadline, t0 + budget - 1.0))

    parent = _get_parent()
    if parent is not None:
        parent = np.where(np.isfinite(parent), parent, 0.0)
        parent = np.clip(parent, 0.0, 1000.0)
    if parent is None or parent.size == 0:
        parent = np.ones(192, dtype=np.float64) / 192.0

    initial = _normalize(parent)
    initial_score = _exact_seq(initial)
    if not np.isfinite(initial_score):
        initial = np.ones(192, dtype=np.float64) / 192.0
        initial_score = 2.0

    best_seq = initial.astype(np.float64, copy=True)
    best_score = float(initial_score)

    parent_n = int(initial.size)
    if parent_n in (1, 2, 3) or parent_n == 0:
        primary_n = 192
    elif parent_n > 1024:
        primary_n = 1024
    else:
        primary_n = parent_n

    primary_alloc = max(20.0, (deadline - time.monotonic()) * 0.60)
    primary_stop = min(deadline, time.monotonic() + primary_alloc)

    local = _search_length(
        primary_n,
        primary_alloc,
        parent,
        best_seq,
        seed_i,

    if local is not None and local.size == primary_n:
        local = _normalize(local)
        es = _exact_seq(local)
        if np.isfinite(es) and es < best_score - 1e-12:
            best_score = float(es)
            best_seq = local.astype(np.float64, copy=True)

    other_lengths = _candidate_lengths(primary_n)
    other_budget = max(0.0, (deadline - time.monotonic()) * 0.30)
    if other_lengths and other_budget > 20.0:
        per_other = other_budget / float(len(other_lengths))
        for L in other_lengths:
            if time.monotonic() > deadline - 30.0:
                break
            if per_other < 18.0:
                continue
            alloc = min(per_other * 1.25, max(20.0, (deadline - time.monotonic()) * 0.18))
            if alloc < 12.0:
                continue
            seg_stop = min(deadline, time.monotonic() + alloc)
            try:
                local = _search_length(L, alloc, parent, best_seq, seed_i + 777 * L, seg_stop)
            except Exception:
                continue
            if local is None or local.size != L:
                continue
            local = _normalize(local)
            es = _exact_seq(local)
            if np.isfinite(es) and es < best_score - 1e-12:
                best_score = float(es)
                best_seq = local.astype(np.float64, copy=True)

    if time.monotonic() < deadline - 8.0 and best_seq.size > 0:
        a0 = _normalize(best_seq)
        if a0.size <= 2048:
            final_stop = min(deadline - 3.0, time.monotonic() + max(10.0, (deadline - time.monotonic()) * 0.35))
            try:
                a2 = _optimize_one(a0, final_stop, seed=seed_i + 424242)
            except Exception:
                a2 = a0
            a2 = _normalize(a2)
            es2 = _exact_seq(a2)
            if np.isfinite(es2) and es2 < best_score - 1e-12:
                best_score = float(es2)
                best_seq = a2.astype(np.float64, copy=True)

    best_seq = np.asarray(best_seq, dtype=np.float64).ravel()
    best_seq = np.where(np.isfinite(best_seq), best_seq, 0.0)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    if best_seq.size == 0:
        best_seq = np.ones(192, dtype=np.float64) / 192.0

    if best_seq.size > 2048:
        cand = _resize(best_seq, 1024)
        fc = _fast_score(cand)
        ec = _exact_seq(cand)
        if np.isfinite(ec):
            best_seq = cand
            best_score = float(ec)
        elif np.isfinite(fc):
            best_seq = cand

    final = _normalize(best_seq)
    fe = _exact_seq(final)
    if not np.isfinite(fe):
        fallback = np.ones(192, dtype=np.float64) / 192.0
        fe2 = _exact_seq(fallback)
        if np.isfinite(fe2):
            final = fallback
        else:
            final = fallback
            fe = 2.0

    try:
        out = [float(x) for x in final]
    except Exception:
        out = [1.0]

    if not out or not all(np.isfinite(x) for x in out):
        out = [1.0]

    return out
