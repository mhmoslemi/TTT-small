    if L is None:
        L = 2 * n - 1
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(A * A, N)
    R = R[:L]
    m = float(R.max())
    return float(2.0 * n * m / (s * s))


def _interp_to_n(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == 0:
        return None
    if m == n:
        return a
    if m == 1:
        return np.full(n, float(a[0]), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a)


def _make_inits(n, rng, start_seq=None, parent=None, max_inits=None):
    if n <= 1:
        return [np.ones(n, dtype=np.float64)]
    if max_inits is None:
        max_inits = 42 if n > 2000 else 72
    inits = []

    def add(arr):
        if len(inits) >= max_inits or arr is None:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size == 0:
            return
        if arr.size != n:
            if arr.size == 1:
                arr = np.full(n, float(arr[0]), dtype=np.float64)
            else:
                arr = _interp_to_n(arr, n)
                if arr is None:
                    return
        arr = _norm(arr, n)
        inits.append(arr)

    add(np.ones(n, dtype=np.float64))
    if parent is not None:
        add(parent)
    if start_seq is not None:
        add(start_seq)

    t = np.linspace(0.0, 1.0, n)
    for amp in (0.2, 0.4, 0.7, 0.9):
        add(1.0 + amp * (t - 0.5))
        add(1.0 - amp * (t - 0.5))
        add(1.0 + amp * (2.0 * t - 1.0) ** 2)

    for mode in (1, 2, 3, 4, 6):
        c1 = np.cos(np.pi * mode * t)
        c2 = np.cos(2.0 * np.pi * mode * t)
        s1 = np.sin(np.pi * mode * t)
        for amp in (0.3, 0.6, 0.9):
            add(1.0 + amp * c1)
            add(1.0 + amp * c2)
            add(1.0 + amp * s1)

    for c in (-3.0, -2.0, -1.0, -0.5, 0.5, 1.0, 2.0, 3.0):
        add(np.exp(c * t))

    for amp in (0.5, 1.0):
        add(1.0 + amp * np.cos(np.pi * t) ** 2)

    for center in (0.25, 0.4, 0.6, 0.75):
        arg = (t - center) ** 2 / (0.05 * 0.05)
        add(1.0 + 0.7 * np.exp(-arg))

    for _ in range(3):
        add(rng.normal(size=n))
        add(rng.exponential(scale=1.0, size=n))

    return inits[:max_inits]


def _gradient_descent(a, N, L, n, time_limit, stop_deadline, seed):
    a = _norm(a, n)
    best = a.copy()
    best_sc = _fft_score(a, N, L)
    if not np.isfinite(best_sc):
        return best, best_sc

    t0 = time.monotonic()
    hard_stop = min(stop_deadline, t0 + float(time_limit))
    rng = np.random.default_rng(seed)
    rfft = np.fft.rfft
    irfft = np.fft.irfft

    T0 = max(2.0, n / 25.0)
    T1 = 1.0
    T = T0
    alpha = 0.75
    no_imp = 0
    stall = 0

    while time.monotonic() < hard_stop:
        if time.monotonic() > stop_deadline - 4.0:
            break

        A = rfft(a, N)
        R = irfft(A * A, N)[:L]
        if not np.all(np.isfinite(R)):
            break
        R = np.maximum(R, 0.0)
        rm = float(R.max())
        cur_sc = float(2.0 * n * rm / (n * n))
        if np.isfinite(cur_sc) and cur_sc < best_sc:
            best = a.copy()
            best_sc = cur_sc
            no_imp = 0
        else:
            no_imp += 1

        elapsed = time.monotonic() - t0
        frac = 0.0 if time_limit <= 0 else min(1.0, elapsed / float(time_limit))
        T_sched = T0 * (T1 / T0) ** frac
        if no_imp > 350:
            T_sched *= 0.5
        T = max(T1, min(T0, T_sched))

        sh = (R - rm) / T
        sh = np.clip(sh, -700.0, 0.0)
        e = np.exp(sh)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 0.0:
            p = np.zeros(L, dtype=np.float64)
            p[int(np.argmax(R))] = 1.0
            f_old = float(rm / T)
        else:
            p = e / s
            f_old = float(rm / T + np.log(s) / T)

        W = rfft(p, N)
        C = irfft(W * np.conj(A), N)[:n]
        grad = C - C.mean()
        if not np.all(np.isfinite(grad)):
            break
        gnorm = float(np.sqrt(np.dot(grad, grad)))
        if gnorm < 1e-13:
            break

        dirn = -grad / gnorm
        pos = dirn > 0.0
        if pos.any():
            bound = float(np.min((a[pos] + 1e-12) / dirn[pos]))
            amax = 0.9 * bound
        else:
            amax = 20.0
        alpha = min(alpha, max(1e-6, amax), 20.0)
        if alpha < 1e-6:
            alpha = 1e-4

        cur_alpha = alpha
        best_a2 = None
        best_f2 = float("inf")
        best_R2 = None
        accepted = False

        for _ in range(4):
            if time.monotonic() > stop_deadline - 5.0:
                break
            a2 = a - cur_alpha * dirn
            if a2.min() < 0.0:
                a2 = np.maximum(a2, 0.0)
                s2 = float(a2.sum())
                if s2 <= 1e-12:
                    cur_alpha *= 0.5
                    continue
                a2 = a2 * (n / s2)

            A2 = rfft(a2, N)
            R2 = irfft(A2 * A2, N)[:L]
            if not np.all(np.isfinite(R2)):
                cur_alpha *= 0.5
                continue
            R2 = np.maximum(R2, 0.0)
            rm2 = float(R2.max())
            sh2 = (R2 - rm2) / T
            sh2 = np.clip(sh2, -700.0, 0.0)
            e2 = np.exp(sh2)
            s2 = float(e2.sum())
            if np.isfinite(s2) and s2 > 0.0:
                f2 = float(rm2 / T + np.log(s2) / T)
            else:
                f2 = float("inf")

            if f2 < best_f2:
                best_f2 = f2
                best_a2 = a2.copy()
                best_R2 = R2.copy()

            need = max(1e-9, 1e-4 * cur_alpha * (2.0 * gnorm))
            if f2 <= f_old - need:
                accepted = True
                break
            cur_alpha *= 0.5
            if cur_alpha < 1e-6:
                break

        if best_a2 is not None and (accepted or best_f2 < f_old - 1e-10):
            a = best_a2
            sc2 = float(2.0 * n * best_R2.max() / (n * n))
            if np.isfinite(sc2) and sc2 < best_sc:
                best = a.copy()
                best_sc = sc2
                no_imp = 0

            if cur_alpha > 0.01:
                alpha = min(20.0, cur_alpha * 1.1)
            else:
                alpha = max(1e-4, cur_alpha)

            if no_imp > 1200:
                T = max(T1, T * 0.7)
        else:
            no_imp += 1
            stall += 1
            alpha *= 0.5
            if alpha < 1e-5 or stall > 250:
                sig = 0.12 if best_sc > 1.65 else 0.045
                a2 = a + rng.normal(0.0, sig, n)
                a2 = _norm(a2, n)
                A2 = rfft(a2, N)
                R2 = irfft(A2 * A2, N)[:L]
                R2 = np.maximum(R2, 0.0)
                sc2 = float(2.0 * n * R2.max() / (n * n))
                if np.isfinite(sc2) and sc2 < best_sc:
                    best = a2.copy()
                    best_sc = sc2
                a = a2
                alpha = 0.5
                no_imp = 0
                stall = 0
                T = max(T1, min(T0, T * 0.65))
            continue

    return best, best_sc


def _optimize_n(n, time_limit, seed, start_seq, parent, stop_deadline):
    n = int(n)
    if n < 2 or time_limit <= 0.0:
        return None, float("inf")

    time_limit = min(float(time_limit), max(0.0, stop_deadline - time.monotonic() - 2.0))
    if time_limit < 2.0:
        return None, float("inf")

    t0 = time.monotonic()
    N = _pow2(3 * n - 2)
    L = 2 * n - 1
    rng = np.random.default_rng(seed)

    inits = _make_inits(n, rng, start_seq=start_seq, parent=parent)
    eval_deadline = min(stop_deadline - 8.0, t0 + min(10.0, 0.22 * time_limit))
    evaluated = []
    for init in inits:
        if time.monotonic() >= eval_deadline:
            break
        sc = _fft_score(init, N, L)
        if np.isfinite(sc):
            evaluated.append((sc, init))

    if not evaluated:
        return None, float("inf")

    evaluated.sort(key=lambda x: x[0])
    local_seq = evaluated[0][1].copy()
    local_sc = evaluated[0][0]

    runs = [evaluated[0][1]]
    if len(evaluated) > 1:
        sc2, init2 = evaluated[1]
        if sc2 < local_sc + 0.08 and stop_deadline - time.monotonic() > 12.0:
            runs.append(init2)

    run_horizon = min(stop_deadline, t0 + time_limit)
    for i, init in enumerate(runs):
        if time.monotonic() >= run_horizon - 5.0:
            break
        remaining = max(0.0, run_horizon - time.monotonic())
        limit = remaining * (0.72 if i == 0 else 0.28)
        if limit < 4.0 and i > 0:
            break
        if limit < 1.0:
            break
        seq, sc = _gradient_descent(init, N, L, n, limit, run_horizon, seed + 1009 * (i + 1))
        if seq is None:
            continue
        if np.isfinite(sc) and sc < local_sc:
            local_seq = seq
            local_sc = sc

    return local_seq, local_sc


def _final_normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    n = a.size
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    for _ in range(20):
        if a.max() > 1000.0:
            a = np.minimum(a, 1000.0)
        s = float(a.sum())
        if s < 0.01:
            a = np.ones(n, dtype=np.float64)
            break
        a = a * (float(n) / s)
        if a.max() <= 1000.0 + 1e-9:
            break
    if a.max() > 1000.0:
        a = np.ones(n, dtype=np.float64)
    return a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    start = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1300.0
    reserve = max(2.0, min(25.0, 0.10 * budget))
    deadline = start + max(1.0, budget - reserve)

    rng = np.random.default_rng(int(seed))

    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size >= 2 and arr.sum() >= 0.01:
            parent = arr
    except Exception:
        parent = None

    candidates = []
    best_seq = np.ones(2, dtype=np.float64)
    best_fft_score = float("inf")

    def add_candidate(seq, sc):
        nonlocal best_seq, best_fft_score
        if seq is None:
            return
        arr = np.asarray(seq, dtype=np.float64).ravel()
        if arr.size == 0:
            return
        if not np.all(np.isfinite(arr)):
            arr_arr = np.where(np.isfinite(arr), arr, 0.0)
        else:
            arr_arr = arr
        arr_arr = _norm(arr_arr, arr_arr.size)
        try:
            scf = float(sc)
        except Exception:
            scf = float("inf")
        if not np.isfinite(scf):
            return
        candidates.append((scf, arr_arr.copy()))
        if scf < best_fft_score:
            best_fft_score = scf
            best_seq = arr_arr.copy()
        if len(candidates) > 36:
            candidates.sort(key=lambda x: x[0])
            candidates = candidates[:36]

    if parent is not None:
        seq = _norm(parent, parent.size)
        sc = _fft_score(seq)
        add_candidate(seq, sc)
    else:
        add_candidate(np.ones(2, dtype=np.float64), 2.0)

    small_budget = min(60.0, max(4.0, 0.04 * budget))
    if parent is not None and best_fft_score < 1.5027:
        small_budget = min(small_budget, 15.0)

    small_end = min(deadline, start + small_budget)
    small_ns = [4, 8, 16, 32, 64, 128]
    for i, n in enumerate(small_ns):
        now = time.monotonic()
        if now >= small_end - 4.0 or now > deadline - 60.0:
            break
        limit = min(small_end - now, max(2.0, small_budget / max(1.0, float(len(small_ns) - i))))
        if limit < 2.0:
            continue
        seq, sc = _optimize_n(n, limit, seed + 17 * i, best_seq, parent, deadline)
        add_candidate(seq, sc)

    now = time.monotonic()
    remaining = max(0.0, deadline - now)

    if parent is not None and parent.size >= 2:
        n_parent = int(parent.size)

        if remaining > 80.0:
            limit = min(remaining * 0.45, 620.0)
            seq, sc = _optimize_n(n_parent, limit, seed + 777, parent, parent, deadline)
            add_candidate(seq, sc)

        if remaining = max(0.0, deadline - time.monotonic()) > 80.0:
            n_large = int(min(10900, max(8000, int(round(1.35 * n_parent)))))
            limit = min(remaining * 0.45, 560.0)
            seq, sc = _optimize_n(n_large, limit, seed + 1234, best_seq, parent, deadline)
            add_candidate(seq, sc)

        remaining = max(0.0, deadline - time.monotonic())
        if remaining > 45.0:
            n_cur = int(best_seq.size)
            limit = remaining - 18.0
            seq, sc = _optimize_n(n_cur, limit, seed + 3456, best_seq, parent, deadline)
            add_candidate(seq, sc)
    else:
        phases = [
            (128, 1),
            (256, 1),
            (512, 2),
            (1024, 2),
            (2048, 3),
            (4096, 4),
            (8192, 6),
        ]
        total_w = sum(w for _, w in phases)
        remaining = max(0.0, deadline - time.monotonic())
        for n, w in phases:
            if time.monotonic() > deadline - 35.0:
                break
            limit = (remaining * float(w) / total_w) * 0.92
            limit = min(limit, 320.0)
            if limit < 12.0:
                continue
            seq, sc = _optimize_n(n, limit, seed + 10000 + n, best_seq, None, deadline)
            add_candidate(seq, sc)

    if best_seq is not None:
        add_candidate(best_seq, best_fft_score)

    candidates.sort(key=lambda x: x[0])

    final_seq = None
    final_score = float("inf")

    for idx, (sc, seq) in enumerate(candidates):
        if time.monotonic() > deadline - 5.0:
            break
        if idx > 24:
            break
        seq = _final_normalize(seq)
        try:
            acc = evaluate_sequence(seq.tolist())
        except Exception:
            acc = float("inf")
        if np.isfinite(acc) and acc < final_score:
            final_score = float(acc)
            final_seq = seq

    if final_seq is None and best_seq is not None:
        seq = _final_normalize(best_seq)
        try:
            acc = evaluate_sequence(seq.tolist())
        except Exception:
            acc = float("inf")
        if np.isfinite(acc):
            final_seq = seq
        else:
            final_seq = seq

    if final_seq is None:
        final_seq = np.ones(2, dtype=np.float64)

    final_seq = _final_normalize(final_seq)
    return [float(max(0.0, min(1000.0, x))) for x in final_seq]
