                    a_ex = a * np.exp(-step_ex)
                    a_ex = np.maximum(a_ex, 0.0)
                    s_ex = a_ex.sum()
                    if s_ex > 1e-15:
                        a_ex /= s_ex
                        b_ex = conv(a_ex)
                        if b_ex.max() < max_b * (1 - 1e-8):
                            a = a_ex

        # --- Periodic: try additive update for diversity ---
        if iter_count % 5000 == 0 and stagnant > 10:
            grad_add = grad_c / max(np.max(np.abs(grad_c)), 1e-18)
            a_add = a - 0.02 * grad_add
            a_add = np.maximum(a_add, 0.0)
            s_add = a_add.sum()
            if s_add > 1e-15:
                a_add /= s_add
                b_add = conv(a_add)
                if b_add.max() < max_b:
                    a = a_add

        # Time guard
        if iter_count % 500 == 0 and time.monotonic() > deadline - 5.0:
            break

    # Final validation
    try:
        fs = float(evaluate_sequence([float(x) for x in best_a]))
        if not np.isfinite(fs):
            best_a = np.ones(n, dtype=np.float64) / n
    except Exception:
        best_a = np.ones(n, dtype=np.float64) / n

    # Ensure return is valid
    result = [float(x) for x in best_a]
    if len(result) == 0 or any(not np.isfinite(x) or x < 0 for x in result):
        result = [1.0 / n] * n

    return result
