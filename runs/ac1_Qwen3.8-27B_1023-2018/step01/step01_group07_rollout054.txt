def _lp_refine(a, M, n, fft_len):
    # Solve: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0
    # conv(a,b)_k = sum_i a[i]*b[k-i]
    # For k=0: a[0]*b[0] <= M
    # For k=1: a[0]*b[1] + a[1]*b[0] <= M
    # ...
    # This is a sparse LP.
    
    # Build constraint matrix A (2n-1 rows, n cols)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # (since conv(a,b)_k = sum_j a[k-j]*b[j] for valid k-j)
    
    # This matrix is structured (Toeplitz-like). For n=4806, it's 9611 x 4806 which is ~46M entries.
    # Too large for dense. Need to exploit structure.
    
    # Alternative: use scipy's sparse LP or solve iteratively.
    # Actually, for the LP, we can use the fact that the constraint matrix is a (reversed) Toeplitz matrix.
    
    # Let me use cvxpy or scipy.optimize.linprog with the structured form.
    
    # Actually, for n=4806, building the dense A matrix would be 9611*4806*8 bytes ≈ 369 MB. That might be too much.
    # Let's use scipy.sparse.
    
    import scipy.sparse as sp
    
    # Build A as sparse
    # A[k, j] = a[k-j] if 0<=k-j<n
    # For row k, the non-zero entries are at columns j where 0<=k-j<n, i.e., max(0,k-n+1) <= j <= min(k, n-1)
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n-1)
        for j in range(j_start, j_end+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
    
    if res.success:
        b = res.x
        b = np.maximum(b, 0)
        sb = b.sum()
        if sb > 1e-15:
            b = b / sb
            # Check if this actually helps
            c_b = fftconvolve(b, b)
            if c_b.max() < M:
                # Blend
                t = min(0.3, (M - c_b.max()) / (M + 1e-30))
                result = (1-t)*a + t*b
                result = np.maximum(result, 0)
                result = result / result.sum()
                return result
    
    return None
