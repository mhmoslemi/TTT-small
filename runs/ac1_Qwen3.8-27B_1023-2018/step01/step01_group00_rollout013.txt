    F = rfft(a, fsize)
    c = irfft(F*F, fsize)[:2*n-1]

    s = a.sum()

    b = _convolve(a, n)
    return 2*n*b.max()/(s*s)

def _lp_step(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0"""
    n = len(a)
    # (a*g)_k = sum_{i} a[i] * g[k-i]
    # Build constraint matrix A such that A @ g <= M * ones(2n-1)
    # A[k, j] = coefficient of g[j] in (a*g)_k
    # (a*g)_k = sum_{j} a[k-j] * g[j] where 0<=k-j<n and 0<=j<n
    # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    # This is a (2n-1) x n matrix
    # For efficiency, build it as a sparse-like structure
    
    A_rows = []
    A_cols = []
    A_vals = []
    
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            # A[k, j] = a[k-j]
            A_rows.append(k)
            A_cols.append(j)
            A_vals.append(a[k-j])

    A = csr_matrix((A_vals, (A_rows, A_cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)  # minimize -sum(g)
    b = np.full(2*n-1, M)

    # (a*a)_k without a[i]: remove contributions of a[i]

    # Plus a[k-i]*a[i] if k-i in [0,n) and k-i != i (i.e., k != 2i)

    # Now minimize max(b[k] + 2*a_{k-i}*a_i for k≠2i, b[2i] + a_i^2) over a_i >= 0
    # This is a 1D convex optimization
    
    # Slopes for each k (k≠2i): d_k = 2*a_{k-i} if 0<=k-i<n else 0
    # For k=2i: quadratic term
    
    def obj(a_i):
        val = b[2*i] + a_i*a_i if 2*i < 2*n-1 else b[2*i]
        for k in range(2*n-1):
            if k == 2*i:
                continue
            ki = k - i
            if 0 <= ki < n:
                val = max(val, b[k] + 2*a_i*a[ki])
        return val
    
    # Use golden section search on [0, upper]
    # Upper bound: if a_i is too large, a_i^2 dominates
    upper = max(10*a[i], b.max()**0.5 + 1) if a[i] > 0 else max(b.max()**0.5 + 1, 1)
    
    # Actually, let's use scipy
    from scipy.optimize import minimize_scalar
    res = minimize_scalar(obj, bounds=(0, upper), method='bounded', options={'xatol': 1e-10})
    return res.x

import numpy as np
from scipy.fft import rfft, irfft, next_fast_len
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize_scalar
import time, math

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size == 0 or not np.isfinite(p).all():
            return np.ones(192, dtype=np.float64)
        return p
    except Exception:
        return np.ones(192, dtype=np.float64)

def _conv(a):
    n = len(a)
    fs = next_fast_len(2*n-1)
    F = rfft(a, fs)
    c = irfft(F*F, fs)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    return float(2.0*n*_conv(a).max()/(s*s))

def _lp_step(a, M, timeout=10.0):
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2*n-1):
        js = max(0, k-n+1)
        je = min(n-1, k)
        for j in range(js, je+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0,None)]*n, method='highs',
                  options={'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0)
    return None

def _lp_iterate(a_init, n_iters, t_stop, step=0.2):
    n = len(a_init)
    tgt = math.sqrt(2.0*n)
    a = np.clip(a_init, 0, None).astype(np.float64).copy()
    s = a.sum()
    a = (a*tgt/s) if s > 0 else np.full(n, tgt/n)
    
    best_a = a.copy()
    best_M = float(_conv(a).max())
    ticks = 0
    
    for it in range(n_iters):
        if time.monotonic() > t_stop:
            break
        M = float(_conv(a).max())
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            ticks = 0
        else:
            ticks += 1
        
        g = _lp_step(a, M)
        if g is None:
            if ticks > 20:
                break
            continue
        
        gs = g.sum()
        if gs < 1e-15:
            g = a.copy()
        g = g * tgt / gs
        
        new_a = (1-step)*a + step*g
        new_a = np.maximum(new_a, 0)
        
        new_M = float(_conv(new_a).max())
        if new_M > M * (1+1e-9):
            for s2 in [step*0.5, step*0.25, step*0.1]:
                t_a = np.maximum((1-s2)*a + s2*g, 0)
                if float(_conv(t_a).max()) <= M * (1+1e-9):
                    step = s2
                    break
            else:
                step = max(0.01, step*0.8)
                if ticks > 50:
                    break
                continue
        else:
            a = new_a
    
    return best_a, best_M

def _coord_descent(a_init, n_sweeps, t_stop):
    n = len(a_init)
    tgt = math.sqrt(2.0*n)
    a = np.clip(a_init, 0, None).astype(np.float64).copy()
    s = a.sum()
    a = (a*tgt/s) if s > 0 else np.full(n, tgt/n)
    
    best_a = a.copy()
    best_M = float(_conv(a).max())
    
    for sweep in range(n_sweeps):
        if time.monotonic() > t_stop:
            break
        for i in range(n):
            if time.monotonic() > t_stop:
                break
            
            # Compute current convolution
            b = _conv(a)
            ai = a[i]
            
            # Remove a[i] contributions
            for k in range(2*n-1):
                ki = k - i
                if 0 <= ki < n:
                    b[k] -= (2*ai*a[ki]) if k != 2*i else (ai*ai)
            
            # 1D optimization: minimize max(b[k] + 2*a[k-i]*a_i for k≠2i, b[2i]+a_i^2)
            def obj(x):
                mx = b[2*i] + x*x if 2*i < 2*n-1 else 0
                for k in range(2*n-1):
                    if k == 2*i:
                        continue
                    ki = k - i
                    if 0 <= ki < n:
                        v = b[k] + 2*x*a[ki]
                        if v > mx:
                            mx = v
                return mx
            
            res = minimize_scalar(obj, bounds=(0, 10*max(ai, tgt/n, 0.01)),
                                  method='bounded', options={'xatol': 1e-8})
            a[i] = max(0, res.x)
        
        s = a.sum()
        if s > 0:
            a = a * tgt / s
        M = float(_conv(a).max())
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
    
    return best_a, best_M
