import time

    t0 = time.monotonic()

        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s < 0.01: return np.inf
        c = np.convolve(a, a)
        return float(2.0 * len(a) * float(c.max()) / (s*s))

        a = np.maximum(np.asarray(a, np.float64), 0.0)

            lo = max(0, k-n+1)
            hi = min(n-1, k)
            if lo <= hi:
                A[k, lo:hi+1] = a[k - np.arange(lo, hi+1)]
        return A

        A = _make_lp_matrix(a)
        try:
            r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                       bounds=[(0,None)]*n, method='highs')
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                gs = g.sum()
                if gs > 1e-10:

def _sa(a0, t_end, temp0=0.001):
    a = _n1(a0.copy())
    best = a.copy()
    best_sc = _sc(a)
    cur_sc = best_sc
    temp = temp0
    
    while time.monotonic() < t_end - 1.0:
        # Perturb
        k = int(_rng.integers(3, n//3 + 1))
        idx = _rng.choice(n, k, replace=False)
        scale = 0.05 + np.sqrt(temp) * 0.3
        a_p = a.copy()
        a_p[idx] *= (1.0 + scale * _rng.standard_normal(k))
        a_p = _n1(a_p)
        s_p = _sc(a_p)
        
        # Accept
        if s_p < cur_sc or (np.isfinite(s_p) and _rng.random() < np.exp((cur_sc - s_p) / temp)):
            a, cur_sc = a_p, s_p
            if cur_sc < best_sc:
                best, best_sc = a.copy(), cur_sc
                _upd(best)
        
        # Cool
        temp *= 0.9995
    
    return best
