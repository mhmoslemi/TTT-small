def obj_and_grad(a):
    conv = np.convolve(a, a)
    mc = conv.max()
    f = mc  # we minimize this (since sum=1, score = 2n*mc)
    
    # Gradient: gradient of the max lag
    k_star = np.argmax(conv)
    grad = np.zeros(n)
    if k_star < n:
        grad[:k_star+1] = a[k_star::-1]
    else:
        lo = k_star - n + 1
        grad[lo:] = a[n-1:lo-1:-1]
    
    return f, grad

        if s < 1e-10:
            return np.ones(n) / n
        return a / s
    
    def proxy(a):
        """Exact score when sum(a)=1 and 0<=a[i]<=1000."""
        return 2.0 * n * np.convolve(a, a).max()
