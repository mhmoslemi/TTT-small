        try:
            s = evaluate_sequence([float(x) for x in a_vec])
            return s if np.isfinite(s) else np.inf
        except:
            return np.inf

    best_a = a.copy()
    best_score = safe_eval(best_a)
    
    def smooth_obj_grad(a_vec, tau):
        """Return (f, grad) for F(a) = 2n * smooth_max(conv(a,a)) / (sum(a))^2"""
        S = a_vec.sum()
        if S < 1e-12:
            return 1e10, np.zeros_like(a_vec)
        
        # Compute convolution
        fa = np.fft.rfft(a_vec, sizeC)
        C = np.fft.irfft(fa * fa, sizeC)[:2*n-1]
        
        # Smooth max
        Cmax = C.max()
        z = (C - Cmax) / tau
        z_max = 0.0  # since Cmax = max(C), max(z) = 0
        exp_z = np.exp(z)
        sum_exp = exp_z.sum()
        if sum_exp <= 0 or not np.isfinite(sum_exp):
            # Fallback: hard max
            w = np.zeros(2*n-1)
            w[C == Cmax] = 1.0 / np.sum(C == Cmax)
            M = Cmax
        else:
            w = exp_z / sum_exp
            M = Cmax + np.log(sum_exp) / tau
        
        f = 2.0 * n * M / (S * S)
        
        # Gradient
        wfl = w[::-1].copy()
        fa_g = np.fft.rfft(a_vec, sizeG)
        fw = np.fft.rfft(wfl, sizeG)
        conv2 = np.fft.irfft(fa_g * fw, sizeG)[:3*n-2]
        dM = 2.0 * conv2[n-1:2*n-1][::-1]
        
        invS2 = 1.0 / (S * S)
        grad = (2.0 * n) * invS2 * dM - (4.0 * n * M) * invS2 / S
        # grad = (2n/S^2) * dM - (4nM/S^3)
        
        if not np.isfinite(f):
            return 1e10, np.zeros_like(a_vec)
        
        # Clip gradient for stability
        gmax = np.abs(grad).max()
        if gmax > 1e6:
            grad = grad * (1e6 / gmax)
        
        return f, grad
    
    # L-BFGS-B phases with decreasing temperature
    phases = [
        (200.0, 30.0),   # (tau, time_budget_seconds)
        (60.0, 40.0),
        (20.0, 40.0),

    current_a = a.copy()

        phase_start = time.monotonic()

        def obj_grad(x):

                    'max_ls': 50,
                    'maxcor': 20,

        sc = safe_eval(current_a)
        if sc < best_score:
            best_score = sc
            best_a = current_a.copy()
        
        # Small perturbation to escape local minima
        if phase_idx < len(phases) - 2:
            noise = rng.standard_normal(n) * 0.01 * current_a.mean()
            current_a = np.clip(current_a + noise, 0.0, 1000.0)
        
        elapsed = time.monotonic() - phase_start
        # If the phase finished early, that's fine
    
    # Final gradient descent polish with very low temperature
    remain = deadline - time.monotonic()
    if remain > 30:
        tau_fine = 0.1
        a_gd = best_a.copy()
        m = np.zeros(n)
        v = np.zeros(n)
        step = 0
        lr0 = 0.001 * a_gd.mean()
        gd_start = time.monotonic()
        gd_budget = min(remain - 30, 300.0)
        
        while time.monotonic() - gd_start < gd_budget:
            f, g = smooth_obj_grad(a_gd, tau_fine)
            step += 1
            m = 0.9 * m + 0.1 * g
            v = 0.999 * v + 0.001 * (g * g)

            lr = lr0 * (0.99 ** (step / 500.0))
            a_gd = a_gd - lr * mb / (np.sqrt(vb) + 1e-10)
            a_gd = np.clip(a_gd, 0.0, 1000.0)

                if sc < best_score:
                    best_score = sc
                    best_a = a_gd.copy()
    
    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    final_score = safe_eval(final)
    if not np.isfinite(final_score) or final_score > best_score:
        final = best_a.copy()
        final_score = safe_eval(final)

        final = np.ones(n, dtype=np.float64)

    if not out:

    if any(not np.isfinite(x) or x < 0 for x in out):
