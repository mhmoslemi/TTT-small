import numpy as np
from scipy.sparse import diags, hstack as sp_hstack, csc_matrix, csr_matrix
from scipy.optimize import linprog
import time

def _build_conv_matrix(f, n):
    """Return csc (2n-1) x n Toeplitz: row k has f[j] for j in range where f*g lag k uses it."""
    # offset d (superdiag d): positions (d+c, c), c=0..n-1 all have value f[c]
    arrs = [f.copy() for _ in range(n)]
    return diags(arrs, list(range(n)), shape=(2 * n - 1, n), format='csc')

def _score_fast(a):
    b = np.convolve(a, a)
    return float(2.0 * len(a) * b.max() / (a.sum() ** 2))

def _lp_step(f, n, max_time=25.0):
    """Solve min t s.t. (f*g)[k] <= t, sum g = 1, g >= 0. Return g or None."""
    try:
        A_sparse = _build_conv_matrix(f, n)
        A_aug = sp_hstack([A_sparse, csc_matrix(-np.ones((2 * n - 1, 1)))], format='csc')
        c = np.zeros(n + 1); c[-1] = 1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * (n + 1)
        res = linprog(c, A_ub=A_aug, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': max_time, 'presolve': True})
        if res.success:
            g = np.clip(res.x[:n], 0, None); s = g.sum()
            if s > 1e-9:
                return g / s
        return None
    except Exception:
        return None
