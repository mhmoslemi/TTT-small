def _resize_to(n, arr):

        a = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None

def _softmax_clip(z, zlim=40.0):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([1.0], dtype=np.float64)
    z = z - float(np.mean(z))
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -zlim, zlim)
    m = float(np.max(z))
    e = np.exp(z - m)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0 / z.size, dtype=np.float64)
    return e / s


def _softmax_noclip(z):

        return np.array([1.0], dtype=np.float64)

    m = float(np.max(z))
    e = np.exp(z - m)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:

    n = int(max(2, n))
    i = np.arange(n, dtype=np.float64)
    t = i / float(n - 1) if n > 1 else np.zeros(n)
    v = int(variant) % 12
    p = None

    if v == 0:
        pa = _resize_to(n, parent)
        pc = _resize_to(n, coarse) if coarse is not None else None
        if pa is None and pc is None:
            p = np.ones(n, dtype=np.float64)
        elif pc is None:
            p = pa.copy()
        elif pa is None:
            p = pc.copy()
        else:
            p = 0.6 * pa + 0.4 * pc
        p = p * (1.0 + 0.02 * rng.normal(size=n))

    elif v == 1:
        if coarse is not None:
            p = _resize_to(n, coarse)
        if p is None:
            p = np.ones(n, dtype=np.float64)
        p = p * (1.0 + 0.08 * rng.normal(size=n))

    elif v == 2:
        amp = float(rng.uniform(0.35, 1.1) / math.sqrt(n))
        sig = float(rng.uniform(1.5, max(3.0, n * 0.05)))
        p = np.ones(n, dtype=np.float64) / n
        p += amp * np.exp(-i / sig)
        p += amp * float(rng.uniform(0.6, 1.4)) * np.exp(-i[::-1] / sig)
        p = p * n
        p += rng.normal(0.0, 0.08, n)

    elif v == 3:
        gamma = float(rng.uniform(0.2, 1.4))
        p = np.power(i + 1.0, -gamma) + np.power(n - i, -gamma)
        p += float(rng.uniform(0.0, 0.3))

    elif v == 4:
        a = float(rng.uniform(0.5, 4.0))
        k = int(rng.integers(1, 5))
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        amp = float(rng.uniform(0.1, 0.6))
        p = 1.0 + a * np.square(2.0 * t - 1.0) + amp * np.cos(2.0 * math.pi * k * t + ph)
        p += rng.normal(0.0, 0.05, n)

    elif v == 5:
        lam = float(rng.uniform(-2.5, 2.5))
        p = np.exp(lam * t) + 0.3 * np.exp(-lam * t)
        p += float(rng.uniform(0.05, 0.25))

    elif v == 6:
        z = float(rng.uniform(-0.3, 0.3))
        for k in range(1, 5):
            amp = float(rng.uniform(0.05, 0.55) / k)
            ph = float(rng.uniform(0.0, 2.0 * math.pi))
            z = z + amp * np.cos(2.0 * math.pi * k * t + ph)
        p = np.exp(z)
        p += float(rng.uniform(0.01, 0.06))

    elif v == 7:
        nb = int(rng.integers(2, 5))
        bounds = np.sort(rng.uniform(0.0, 1.0, nb - 1))
        cuts = np.concatenate(([0.0], bounds, [1.0]))
        weights = rng.gamma(1.0, 1.0, nb) + 0.05
        idx = np.searchsorted(cuts, t, side="right") - 1
        idx = np.clip(idx, 0, nb - 1)
        p = weights[idx].astype(np.float64)
        p += float(rng.uniform(0.01, 0.05))

    elif v == 8:
        a = float(rng.uniform(0.4, 3.0))
        p = np.ones(n, dtype=np.float64)
        p[: n // 2] *= a
        p[n // 2 :] *= 1.0 / a
        p += 0.2

    elif v == 9:
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        p = 1.0 + float(rng.uniform(0.1, 0.45)) * np.sin(2.0 * math.pi * 2.0 * t + ph)
        p += float(rng.uniform(0.1, 0.25)) * np.cos(2.0 * math.pi * 5.0 * t - ph)

    elif v == 10:
        p = np.ones(n, dtype=np.float64)

    else:
        pa = _resize_to(n, parent)
        if pa is None:
            p = np.ones(n, dtype=np.float64)
        else:
            p = 0.5 * pa + 0.5 / n
        p = p * (1.0 + 0.05 * rng.normal(size=n))

    p = _normalize(p)
    if float(rng.random()) < 0.75:
        p = _enhance_endpoints(p, rng)
    return _normalize(p)


def _perturb_p(p, rng, sigma):
    sigma = float(max(1e-4, sigma))
    p = _normalize(p)
    n = p.size
    z = np.log(np.maximum(p, 1e-15))
    z += rng.normal(0.0, sigma, n)
    z -= float(np.mean(z))
    z = np.clip(z, -40.0, 40.0)
    return _softmax_clip(z)


def _soft_lse(b, tau):
    b = np.asarray(b, dtype=np.float64).ravel()
    if b.size == 0:
        return 0.0, np.array([], dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m) or m <= 0.0:
        q = np.full(b.size, 1.0 / b.size, dtype=np.float64)
        return 0.0, q

    tau = float(tau)
    if not np.isfinite(tau) or tau <= 1e-18:
        tol = max(1e-12, 1e-8 * m)
        idx = np.where(b >= m - tol)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(b))], dtype=int)
        q = np.zeros(b.size, dtype=np.float64)
        q[idx] = 1.0 / idx.size
        return m, q

    x = (b - m) / tau
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        idx = np.where(b == m)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(b))], dtype=int)
        q = np.zeros(b.size, dtype=np.float64)
        q[idx] = 1.0 / idx.size
        return m, q
    q = ex / s
    U = m + tau * math.log(s)
    if not np.isfinite(U):
        U = m
    return float(U), q


def _grad_z(p, q, fsize=None):
    p = np.asarray(p, dtype=np.float64).ravel()
    q = np.asarray(q, dtype=np.float64).ravel()
    n = p.size
    L = 2 * n - 1
    if q.size != L:
        q = np.zeros(L, dtype=np.float64)
    qs = float(np.sum(q))
    if np.isfinite(qs) and qs > 0.0:
        q = q / qs
    r = q[::-1]
    if fsize is None:
        fsize = next_fast_len(3 * n - 2)
    conv = irfft(rfft(p, fsize) * rfft(r, fsize), n=fsize)
    seg = conv[n - 1 : 2 * n - 1]
    g = 2.0 * seg[::-1]
    g = np.where(np.isfinite(g), g, 0.0)
    dot = float(np.dot(p, g))
    if not np.isfinite(dot):
        dot = 0.0
    gz = p * (g - dot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    return gz, g


def _lp_update(p, fsize_b, stop_time=None):
    try:
        p = _normalize(p)
    except Exception:
        return p
    n = p.size
    if n < 4 or n > 512:
        return p
    if stop_time is not None and time.monotonic() > stop_time - 2.0:
        return p
    b = _convolution(p, fsize_b)
    M = float(np.max(b)) if b.size else 0.0
    if not np.isfinite(M) or M <= 0.0:
        return p
    try:
        from scipy.optimize import linprog
    except Exception:
        return p

    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        end = min(L, j + n)
        if end > j:
            A[j:end, j] = p[: end - j]
    A = A / M

    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = None
    last_exc = None
    for meth in ("highs", "revised simplex", "interior-point"):
        try:
            res = linprog(c, A_ub=A, b_ub=np.ones(L, dtype=np.float64),
                          bounds=bounds, method=meth)
            if res is not None:
                break
        except Exception as exc:
            last_exc = exc

    if res is None or not getattr(res, "success", False) or res.x is None:
        return p

    g = np.asarray(res.x, dtype=np.float64).ravel()
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(np.sum(g))
    if not np.isfinite(s) or s <= 0.0:
        return p
    g = g / s

    cur_p, _, cur_sc = _eval_p(p, n, fsize_b)
    best = (cur_p, cur_sc)
    for mix in (0.10, 0.20, 0.35, 0.50, 0.70):
        cand = (1.0 - mix) * p + mix * g
        cp, _, cs = _eval_p(cand, n, fsize_b)
        if np.isfinite(cs) and cs < best[1]:
            best = (cp, cs)
    return best[0]


def _choose_start(n, parent, coarse):
    n = int(max(2, n))
    options = []
    pa = _resize_to(n, parent)
    if pa is not None:
        options.append(pa)
    if coarse is not None:
        pc = _resize_to(n, coarse)
        if pc is not None:
            options.append(pc)
    if not options:
        return np.ones(n, dtype=np.float64) / n
    fsize = next_fast_len(2 * n - 1)
    best = options[0]
    best_sc = math.inf
    for p in options:
        sc, _ = _fast_score_normalized(p, fsize)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = p
    return _normalize(best)


def _optimize_n(n, seed, stop_time, parent, coarse=None):
    if time.monotonic() >= stop_time - 1.0:
        return None, math.inf

    n = int(max(2, n))
    rng = np.random.default_rng(int(seed))
    fsize_b = next_fast_len(2 * n - 1)
    fsize_g = next_fast_len(3 * n - 2)

    best_p = None
    best_score = math.inf
    base_restart = 8.0 if n <= 128 else (14.0 if n <= 512 else 22.0)
    base_lr = 0.50 if n <= 64 else (0.40 if n <= 192 else (0.30 if n <= 1024 else 0.20))
    lp_calls = 0
    restart = 0
    variant = 1

    while time.monotonic() < stop_time - 3.0:
        remaining = stop_time - time.monotonic()
        if remaining < 4.0:
            break

        if restart == 0:
            p0 = _choose_start(n, parent, coarse)
            alloc = min(0.60 * remaining, max(12.0, 2.0 * base_restart))
        else:
            if best_p is not None and variant % 5 == 0:
                p0 = _perturb_p(best_p, rng, 0.18 + 0.02 * min(5.0, float(variant)))
            else:
                p0 = _init_p(n, variant, rng, parent, coarse)
                variant += 1
            alloc = min(0.18 * remaining, base_restart)

        if alloc < 2.0:
            break
        local_dead = time.monotonic() + alloc

        z, p = _to_z_p(p0)
        b = _convolution(p, fsize_b)
        score = float(2.0 * n * np.max(b)) if b.size else math.inf
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_p = p.copy()

        if n <= 512 and time.monotonic() < local_dead - 4.0 and lp_calls < 4:
            try:
                lp = _lp_update(p, fsize_b, local_dead)
                lp_calls += 1
                if lp is not None:
                    lp_p, lp_b, lp_sc = _eval_p(lp, n, fsize_b)
                    if np.isfinite(lp_sc) and lp_sc < score:
                        p, b, score = lp_p, lp_b, lp_sc
                        z, _ = _to_z_p(p)
                        if score < best_score:
                            best_score = score
                            best_p = p.copy()
            except Exception:
                pass

        m = np.zeros(n, dtype=np.float64)
        v = np.zeros(n, dtype=np.float64)
        tcount = 0
        lr = base_lr
        tau_rel = 0.12
        stuck = 0
        last_improve = time.monotonic()
        best_local_p = p.copy()
        best_local_score = score

        while time.monotonic() < local_dead:
            now = time.monotonic()
            if now >= local_dead:
                break

            b = _convolution(p, fsize_b)
            score = float(2.0 * n * np.max(b)) if b.size else math.inf
            if not np.isfinite(score):
                p = _normalize(best_local_p)
                z, p = _to_z_p(p)
                break

            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
            if score < best_local_score - 1e-12:
                best_local_score = score
                best_local_p = p.copy()
                last_improve = now
                tau_rel = max(0.008, 0.94 * tau_rel)

            if now - last_improve > max(2.0, 0.10 * alloc):
                p = _normalize(best_local_p)
                z, p = _to_z_p(p)
                b = _convolution(p, fsize_b)
                score = float(2.0 * n * np.max(b)) if b.size else math.inf
                m = np.zeros(n, dtype=np.float64)
                v = np.zeros(n, dtype=np.float64)
                tcount = 0
                lr = base_lr
                tau_rel = max(0.06, 1.4 * tau_rel)
                stuck = 0
                continue

            bmax = float(np.max(b)) if b.size else 0.0
            tau = max(1e-18, tau_rel * max(bmax, 1e-18))
            U_b, q = _soft_lse(b, tau)
            U = 2.0 * n * U_b
            gz, g = _grad_z(p, q, fsize_g)
            if not np.all(np.isfinite(gz)):
                stuck += 1
                if stuck > 5:
                    break
                z += rng.normal(0.0, 0.20, n)
                z -= z.mean()
                p = _softmax_clip(z)
                continue

            gz = gz * (2.0 * n)
            maxgz = float(np.max(np.abs(gz))) if gz.size else 0.0
            if not np.isfinite(maxgz) or maxgz < 1e-13:
                stuck += 1
                if stuck > 5:
                    break
                z += rng.normal(0.0, 0.25, n)
                z -= z.mean()
                p = _softmax_clip(z)
                m = np.zeros(n, dtype=np.float64)
                v = np.zeros(n, dtype=np.float64)
                tcount = 0
                continue

            tcount += 1
            m = 0.90 * m + 0.10 * gz
            v = 0.999 * v + 0.001 * (gz * gz)
            mhat = m / (1.0 - 0.90 ** tcount)
            vhat = v / (1.0 - 0.999 ** tcount)
            d = mhat / (np.sqrt(vhat) + 1e-9)
            d = np.where(np.isfinite(d), d, 0.0)
            dmax = float(np.max(np.abs(d))) if d.size else 0.0
            if not np.isfinite(dmax) or dmax <= 0.0:
                break
            d = d / dmax

            accepted = False
            trial = lr
            for _ in range(3):
                cand_z = z - trial * d
                cand_z -= cand_z.mean()
                cand_z = np.where(np.isfinite(cand_z), cand_z, 0.0)
                np.clip(cand_z, -40.0, 40.0, out=cand_z)
                cand_p = _softmax_clip(cand_z)
                cand_b = _convolution(cand_p, fsize_b)
                cand_score = float(2.0 * n * np.max(cand_b)) if cand_b.size else math.inf

                ok = False
                if np.isfinite(cand_score) and cand_score < score * (1.0 - 1e-10):
                    ok = True
                elif np.isfinite(cand_score):
                    cand_U_b, _ = _soft_lse(cand_b, tau)
                    cand_U = 2.0 * n * cand_U_b
                    limit = min(score * 1.002, best_local_score * 1.02 + 1e-9)
                    if np.isfinite(cand_U) and cand_U < U * (1.0 - 1e-9) and cand_score < limit:
                        ok = True
                if ok:
                    accepted = True
                    z = cand_z
                    p = cand_p
                    b = cand_b
                    score = cand_score
                    break
                trial *= 0.5

            if accepted:
                lr = min(1.0, lr * 1.06)
                stuck = 0
                if score < best_score - 1e-12:
                    best_score = score
                    best_p = p.copy()
                if score < best_local_score - 1e-12:
                    best_local_score = score
                    best_local_p = p.copy()
                    last_improve = time.monotonic()
            else:
                lr = max(1e-4, lr * 0.80)
                stuck += 1

                if stuck % 8 == 0:
                    if np.all(np.isfinite(g)):
                        k = min(5, n)
                        idx = np.argpartition(g, k - 1)[:k] if k < n else np.arange(n)
                        found = False
                        for j in idx:
                            for gamma in (0.25, 0.10, 0.05):
                                cand = (1.0 - gamma) * p
                                cand[j] += gamma
                                cand = np.maximum(cand, 0.0)
                                s = float(np.sum(cand))
                                if s <= 0.0:
                                    continue
                                cand = cand / s
                                cb = _convolution(cand, fsize_b)
                                cs = float(2.0 * n * np.max(cb)) if cb.size else math.inf
                                if np.isfinite(cs) and cs < score - 1e-12:
                                    z, p = _to_z_p(cand)
                                    b, score = cb, cs
                                    found = True
                                    break
                            if found:
                                break
                        if found:
                            stuck = 0
                            if score < best_score - 1e-12:
                                best_score = score
                                best_p = p.copy()
                            if score < best_local_score - 1e-12:
                                best_local_score = score
                                best_local_p = p.copy()
                                last_improve = time.monotonic()
                            continue

                if stuck % 12 == 0:
                    cand = 0.5 * (p + p[::-1])
                    pb, bb, sb = _eval_p(cand, n, fsize_b)
                    if np.isfinite(sb) and sb < score - 1e-12:
                        z, p = _to_z_p(pb)
                        b, score = bb, sb
                        if score < best_score - 1e-12:
                            best_score = score
                            best_p = p.copy()
                        if score < best_local_score - 1e-12:
                            best_local_score = score
                            best_local_p = p.copy()
                            last_improve = time.monotonic()

                if stuck % 20 == 0 and n <= 512 and lp_calls < 4 and time.monotonic() < local_dead - 3.0:
                    try:
                        lp = _lp_update(p, fsize_b, local_dead)
                        lp_calls += 1
                        if lp is not None:
                            lp_p, lp_b, lp_sc = _eval_p(lp, n, fsize_b)
                            if np.isfinite(lp_sc) and lp_sc < score - 1e-12:
                                z, p = _to_z_p(lp_p)
                                b, score = lp_b, lp_sc
                                if score < best_score - 1e-12:
                                    best_score = score
                                    best_p = p.copy()
                                if score < best_local_score - 1e-12:
                                    best_local_score = score
                                    best_local_p = p.copy()
                                    last_improve = time.monotonic()
                                stuck = 0
                    except Exception:
                        pass

                if stuck > 28:
                    z = z + rng.normal(0.0, 0.40, n)
                    z -= z.mean()
                    p = _softmax_clip(z)
                    z, p = _to_z_p(p)
                    m = np.zeros(n, dtype=np.float64)
                    v = np.zeros(n, dtype=np.float64)
                    tcount = 0
                    lr = base_lr
                    tau_rel = 0.10
                    stuck = 0
                    if score > best_local_score * 1.25:
                        break

        restart += 1
        if restart > 80:
            break

    if best_p is None:
        return None, math.inf
    return best_p, best_score


def _lbfgs_polish(n, p0, stop_time):
    if _scipy_minimize is None:
        return p0
    n = int(p0.size)
    if n < 2 or time.monotonic() > stop_time - 15.0:
        return p0

    p0 = _normalize(p0)
    fsize_b = next_fast_len(2 * n - 1)
    fsize_g = next_fast_len(3 * n - 2)

    b0 = _convolution(p0, fsize_b)
    sc0 = float(2.0 * n * np.max(b0)) if b0.size else math.inf
    if not np.isfinite(sc0):
        return p0

    best_p = p0.copy()
    best_sc = sc0
    z = np.log(np.maximum(p0, 1e-15))
    z -= float(np.mean(z))
    z = np.clip(z, -40.0, 40.0)

    for tau_rel in (0.08, 0.04, 0.02, 0.01):
        if time.monotonic() > stop_time - 10.0:
            break

        pz = _softmax_noclip(z)
        bz = _convolution(pz, fsize_b)
        scz = float(2.0 * n * np.max(bz)) if bz.size else math.inf
        if np.isfinite(scz) and scz < best_sc:
            best_sc = scz
            best_p = pz.copy()
        bmax = float(np.max(bz)) if bz.size else 0.0
        tau = max(1e-18, tau_rel * max(bmax, 1e-18))

        state = {
            "evals": 0,
            "stopped": False,
            "last_val": 0.0,
            "best_p": best_p,
            "best_sc": best_sc,
            "tau": tau,
            "stop_time": stop_time,
            "max_evals": 4000 if n <= 512 else (1500 if n <= 1024 else 500),
            "max_iter": 3000 if n <= 512 else (1200 if n <= 1024 else 400),
        }
        cache = {"z": None, "val": None, "grad": None}

        def compute(zv):
            if state["stopped"]:
                return state["last_val"], np.zeros(n, dtype=np.float64)
            zarr = np.asarray(zv, dtype=np.float64).ravel()
            if cache["z"] is not None and np.array_equal(zarr, cache["z"]):
                return cache["val"], cache["grad"]

            state["evals"] += 1
            if state["evals"] > state["max_evals"] or time.monotonic() > state["stop_time"] - 4.0:
                state["stopped"] = True
                return state["last_val"], np.zeros(n, dtype=np.float64)

            if not np.all(np.isfinite(zarr)):
                p = np.ones(n, dtype=np.float64) / n
            else:
                p = _softmax_noclip(zarr)

            b = _convolution(p, fsize_b)
            sc = float(2.0 * n * np.max(b)) if b.size else math.inf
            if np.isfinite(sc) and sc < state["best_sc"]:
                state["best_sc"] = sc
                state["best_p"] = p.copy()

            targa = state["tau"]
            Ub, q = _soft_lse(b, targa)
            val = 2.0 * n * Ub
            if not np.isfinite(val):
                val = 0.0

            gz, _ = _grad_z(p, q, fsize_g)
            grad = gz * (2.0 * n)
            grad = np.where(np.isfinite(grad), grad, 0.0)

            cache["z"] = zarr.copy()
            cache["val"] = val
            cache["grad"] = grad
            state["last_val"] = val
            return val, grad

        def fun(x):
            val, _ = compute(x)
            return float(val)

        def jac(x):
            _, grad = compute(x)
            return np.asarray(grad, dtype=np.float64)

        def cb(x):
            if state["stopped"] or state["evals"] > state["max_evals"] or time.monotonic() > state["stop_time"] - 4.0:
                state["stopped"] = True
                return True
            return False

        try:
            res = _scipy_minimize(
                fun,
                z,
                method="L-BFGS-B",
                jac=jac,
                callback=cb,
                options={
                    "maxiter": int(state["max_iter"]),
                    "ftol": 1e-12,
                    "gtol": 1e-6,
                    "maxls": 20,
                },
            )
        except Exception:
            res = None

        if np.isfinite(state["best_sc"]) and state["best_sc"] < best_sc:
            best_sc = state["best_sc"]
            best_p = np.asarray(state["best_p"], dtype=np.float64).copy()

        if res is not None and getattr(res, "x", None) is not None:
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if x.size == n and np.all(np.isfinite(x)):
                x = x - float(np.mean(x))
                x = np.clip(x, -50.0, 50.0)
                z = x

    psym = 0.5 * (best_p + best_p[::-1])
    psym = _normalize(psym)
    scsym, _, _ = _eval_p(psym, n, fsize_b)
    if np.isfinite(scsym) and scsym < best_sc:
        best_p = psym
        best_sc = scsym

    return _normalize(best_p)


def _exact_score(p):
    try:
        return float(evaluate_sequence([float(x) for x in np.asarray(p).ravel()]))
    except Exception:
        return math.inf


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    total_bud = max(1.0, float(budget_s))
    reserve = min(45.0, max(5.0, 0.05 * total_bud))
    deadline = t0 + max(0.5, total_bud - reserve)

    parent = _get_parent()
    parent_p = _normalize(parent)
    fb = np.ones(32, dtype=np.float64) / 32.0

    best_p = parent_p
    best_sc = _exact_score(best_p)
    if not np.isfinite(best_sc):
        best_p = fb
        best_sc = _exact_score(fb)
    if not np.isfinite(best_sc):
        return [1.0]

    if time.monotonic() > deadline - 5.0:
        return [float(x) for x in best_p]

    coarse = None
    if total_bud > 420.0 and time.monotonic() < deadline - 140.0:
        n_coarse = 64
        coarse_stop = min(deadline - 60.0, time.monotonic() + 28.0)
        try:
            cp, cs = _optimize_n(n_coarse, int(seed) + 11, coarse_stop, parent_p, coarse=None)
            if cp is not None:
                cs = _exact_score(cp)
                if np.isfinite(cs) and cs < best_sc:
                    best_sc = cs
                    best_p = cp
                cp2, _ = _eval_p(cp, n_coarse, next_fast_len(2 * n_coarse - 1))
                coarse = cp2
        except Exception:
            coarse = None

    if time.monotonic() > deadline - 10.0:
        return [float(x) for x in best_p]

    P = int(best_p.size)
    if P < 2:
        P = 192

    candidates = []
    raw = [
        P,
        2 * P,
        max(8, P // 2),
        max(8, 2 * P),
        max(8, P // 4),
        max(8, P // 8),
        32, 64, 96, 128, 192, 256, 384, 512, 768, 1024, 1536, 2048, 3072, 4096,
    ]
    seen = set()
    for L in raw:
        L = int(L)
        if L >= 8 and L <= 4096 and L not in seen:
            seen.add(L)
            candidates.append(L)

    def _prio(L):
        lp = math.log(max(2, L))
        lP = math.log(max(2, P))
        return abs(lp - lP) + 0.25 * lp

    candidates.sort(key=_prio)

    fracs = [
        0.32, 0.10, 0.08, 0.08, 0.06, 0.05, 0.04, 0.04,
        0.03, 0.03, 0.03, 0.02, 0.02, 0.02, 0.01, 0.01, 0.01, 0.01, 0.01, 0.01,
    ]
    initial_budget = max(0.0, deadline - time.monotonic() - 5.0)

    for idx, L in enumerate(candidates):
        if time.monotonic() > deadline - 15.0:
            break
        if best_sc < 1.48:
            break
        frac = fracs[idx] if idx < len(fracs) else 0.005
        alloc = max(3.0, initial_budget * frac)
        alloc = min(alloc, deadline - time.monotonic() - 10.0)
        if alloc < 4.0:
            continue

        try:
            cand, fast_sc = _optimize_n(L, int(seed) + 100003 + 17 * L,
                                        time.monotonic() + alloc, parent_p, coarse)
        except Exception:
            cand = None

        if cand is None:
            continue
        cand = _normalize(cand)
        if cand.size != L:
            continue
        if not np.isfinite(fast_sc):
            fast_sc, _ = _fast_score_normalized(cand)
        if not np.isfinite(fast_sc):
            continue

        if fast_sc < best_sc + 0.05:
            exact_sc = _exact_score(cand)
            if np.isfinite(exact_sc) and exact_sc < best_sc - 1e-12:
                best_sc = exact_sc
                best_p = cand

    if time.monotonic() < deadline - 12.0:
        try:
            polished = _lbfgs_polish(best_p.size, best_p, deadline)
            if polished is not None:
                polished = _normalize(polished)
                exact_sc = _exact_score(polished)
                if np.isfinite(exact_sc) and exact_sc < best_sc - 1e-12:
                    best_sc = exact_sc
                    best_p = polished
        except Exception:
            pass

    psym = _normalize(0.5 * (best_p + best_p[::-1]))
    sym_sc = _exact_score(psym)
    if np.isfinite(sym_sc) and sym_sc < best_sc - 1e-12:
        best_p = psym
        best_sc = sym_sc

    final_sc = _exact_score(best_p)
    if not np.isfinite(final_sc):
        final_sc = _exact_score(fb)
        if np.isfinite(final_sc):
            best_p = fb
        else:
            return [1.0]

    return [float(x) for x in _normalize(best_p)]
