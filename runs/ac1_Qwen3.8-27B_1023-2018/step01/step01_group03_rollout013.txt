    s = a.sum()

    """Build (2n-1) x n matrix for convolution with f."""
    n = len(f)

        if f[j] > 0:
            A[j:j+n, j] = f[j]
    return A

    """Find g maximizing sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = _build_A(f)

        if res.success:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)

        seg = a[kk-hi:kk-lo+1][::-1]
        grad[lo:hi+1] += 2.0 * ww * seg
    
    # Normalize the subgradient relative to the objective
    gn = np.linalg.norm(grad)
    if gn < 1e-14:

    a_new = a - step * grad / gn
    a_new = np.maximum(a_new, 0)
