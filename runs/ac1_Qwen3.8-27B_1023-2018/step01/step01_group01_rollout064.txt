def _p2(l):
    return 1 << max(0, (int(l) - 1).bit_length())


def _full_conv_aa(a, size):
    fa = np.fft.rfft(a, size)
    out = np.fft.irfft(fa * fa, size)
    return out[: 2 * len(a) - 1]


def _grad_soft(a, sizeC, sizeG, temp):
    """Return (C, grad, M_est) for objective 2n*max(C)/S^2 with C=a*a (full)."""
    n = len(a)
    C = _full_conv_aa(a, sizeC)
    Cmax = float(C.max())
    denom = Cmax if Cmax > 1e-300 else 1.0
    z = temp * (C - Cmax) / denom
    z = z - z.max()
    w = np.exp(z)
    wsum = float(w.sum())
    if wsum <= 0.0 or not np.isfinite(wsum):
        w = np.full_like(C, 1.0 / len(C))
        wsum = 1.0
    wn = w / wsum
    M = Cmax + denom * np.log(wsum) / temp
    wfl = wn[::-1]
    fa = np.fft.rfft(a, sizeG)

    S = float(a.sum())
    invS2 = 1.0 / (S * S)
    grad = (2.0 * n) * invS2 * dM - (4.0 * n * M) * invS2 / S
    return C, grad, M
