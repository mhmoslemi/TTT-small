    n = v.size
    u = np.sort(v)[::-1]
    cu = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rhos = np.nonzero(u * idx > (cu - 1))[0]

    theta = (cu[r] - 1) / (r + 1.0)
