import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _norm(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _upsample(a, m):
    a = _norm(a)
    m = int(m)
    if a.size == m:
        return a.copy()
    if a.size == 0:
        return np.ones(m, dtype=np.float64) / max(1, m)
    x0 = np.linspace(0.0, 1.0, a.size)
    x1 = np.linspace(0.0, 1.0, m)
    out = np.interp(x1, x0, a)
    return _norm(out)


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.inf
    c = np.convolve(a, a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * a.size * mx / (s * s))


def _convolve(x, y):
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    if x.size == 0 or y.size == 0:
        return np.array([], dtype=np.float64)
    if x.size * y.size < 200000:
        out = np.convolve(x, y)
    else:
        need = x.size + y.size - 1
        L = 1
        while L < need:
            L <<= 1
        X = np.fft.rfft(x, L)
        Y = np.fft.rfft(y, L)
        out = np.fft.irfft(X * Y, L)[:need]
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    return out


def _score_internal(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:

    c = _convolve(a, a)
    mx = float(c.max())
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * a.size * mx / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    e = np.exp(np.clip(z, -745.0, 745.0))
    s = float(e.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.full(z.shape, 1.0 / float(z.size), dtype=np.float64)
    return e / s


def _logit(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-12)
    z = np.log(a)
    z = z - float(np.mean(z))
    return np.where(np.isfinite(z), z, 0.0)


def _weights_from_c(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return np.array([], dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    cm = float(np.max(c))
    z = beta * (c - cm)
    z = np.where(np.isfinite(z), z, -745.0)
    q = np.exp(np.clip(z, -745.0, 0.0))
    s = float(q.sum())
    if not np.isfinite(s) or s <= 0.0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
    else:
        q /= s
    return q


def _matvec_q(a, q):
    n = int(a.size)
    if n == 1:
        return np.array([2.0 * float(q[0]) * float(a[0])], dtype=np.float64)

    K = np.empty(2 * n - 1, dtype=np.float64)
    K[:n] = q[n - 1::-1]
    K[n:] = q[1:]

    T = _convolve(a, K)
    T = T[n - 1:2 * n - 1]

    b = a[::-1]
    H = _convolve(q, b)
    H = H[n - 1:2 * n - 1]

    g = T + H
    g = np.where(np.isfinite(g), g, 0.0)
    return np.maximum(g, 0.0)


def _solve_lp_direction(a):

        return None
    n = int(a.size)
    if n > 640:
        return None
    c = np.convolve(a, a)

    if not np.isfinite(M) or M <= 0.0:

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):

        hi = min(n - 1, k)

    b = np.where(np.isfinite(b), b, 0.0)

    if float(b.sum()) <= 1e-15:
        return None
    return b

            stagnant += 1
            if stagnant > 60:
                cur_a = best_a * (1.0 + 0.06 * rng.standard_normal(n))
                cur_a = np.maximum(cur_a, 0.0)
                cur_a = _norm(cur_a)
                tau = 0.12
                stagnant = 0

    exact = _score_exact(cur_a)
    if np.isfinite(exact) and exact < best_sc:
        best_sc = exact
        best_a = cur_a.copy()
    return best_a, best_sc


def _adam_phase(inits, start, end, rng):
    if not inits:
        return None, np.inf

    scored = []
    for a in inits:
        a = _norm(a)
        if a.size == 0:
            continue
        sc = _score_exact(a)
        if np.isfinite(sc):
            scored.append((sc, a))
    if not scored:
        n = int(inits[0].size) if inits[0].size > 0 else 1
        a = np.ones(n, dtype=np.float64) / float(n)
        return a, _score_exact(a)

    scored.sort(key=lambda x: x[0])
    best_sc, best_a = scored[0]
    queue = [a.copy() for _, a in scored[1:][:6]]

    n = int(best_a.size)
    a = best_a.copy()
    z = _logit(a)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    b1_t = 1.0
    b2_t = 1.0
    t = 0

    lr = 0.65
    cap = 1.1
    check_every = max(5, min(60, n // 20 + 5))
    patience = 550 if n <= 256 else 1100
    since = 0
    restarts = 0

    while time.monotonic() < end:
        elapsed_frac = 0.0
        total = max(1e-6, float(end) - float(start))
        if end > start:
            elapsed_frac = min(1.0, max(0.0, (time.monotonic() - start) / total))

        a = _softmax(z)
        c = _convolve(a, a)
        cm = float(c.max())
        if not np.isfinite(cm):
            z = _logit(best_a)
            m.fill(0.0)
            v.fill(0.0)
            b1_t = b2_t = 1.0
            t = 0
            since = 0
            continue

        if t == 0 or t % check_every == 0:
            sc_ex = _score_exact(a)
            if np.isfinite(sc_ex) and sc_ex < best_sc - 1e-12:
                best_sc = sc_ex
                best_a = a.copy()
                since = 0
                lr = min(0.95, lr * 1.01)

        since += 1
        if since >= patience:
            if queue and restarts % 3 == 0:
                a = queue.pop(0)
                a = a * (1.0 + 0.08 * rng.standard_normal(n))
                a = _norm(np.maximum(a, 0.0))
            else:
                a = best_a.copy()
                if rng.random() < 0.25:
                    a = a[::-1]
                a = a * (1.0 + (0.05 if best_sc < 1.55 else 0.12) * rng.standard_normal(n))
                a = _norm(np.maximum(a, 0.0))
            z = _logit(a)
            m.fill(0.0)
            v.fill(0.0)
            b1_t = b2_t = 1.0
            t = 0
            since = 0
            restarts += 1
            lr = 0.50 if best_sc < 1.53 else 0.70
            patience = max(350, int(patience * 1.12))
            continue

        beta = min(220000.0, max(50.0, (30.0 + 140.0 * elapsed_frac) * n))
        q = _weights_from_c(c, beta)
        g = _matvec_q(a, q)
        if not np.all(np.isfinite(g)):
            g = np.zeros_like(a)

        dot = float(np.dot(g, a))
        grad = a * (g - dot)
        if not np.all(np.isfinite(grad)):
            z = _logit(best_a)
            m.fill(0.0)
            v.fill(0.0)
            b1_t = b2_t = 1.0
            t = 0
            since = 0
            continue

        t += 1
        b1_t *= b1
        b2_t *= b2
        den1 = max(1e-12, 1.0 - b1_t)
        den2 = max(1e-12, 1.0 - b2_t)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mhat = m / den1
        vhat = v / den2
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -cap, cap)
        step = np.where(np.isfinite(step), step, 0.0)

        z = z - step
        z = np.where(np.isfinite(z), z, 0.0)
        z -= float(np.mean(z))
        if not np.all(np.isfinite(z)):
            z = _logit(best_a)
            m.fill(0.0)
            v.fill(0.0)
            b1_t = b2_t = 1.0
            t = 0
            since = 0

    return best_a, best_sc


def _optimize_length(n, inits, start, end, rng, use_lp=True):
    n = int(n)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64)

    scored = []
    for a in inits:
        a = _norm(a)
        if a.size == n:
            sc = _score_exact(a)
            if np.isfinite(sc):
                scored.append((sc, a))

    if not scored:
        a = np.ones(n, dtype=np.float64) / float(n)
        return a

    scored.sort(key=lambda x: x[0])
    best_a = scored[0][1].copy()
    best_sc = scored[0][0]

    dur = max(0.0, float(end) - float(start))
    if use_lp and n <= 640 and dur > 12.0:
        lp_end = min(float(end) - 2.0, float(start) + 0.32 * dur)
        if lp_end - float(start) > 2.0:
            la, ls = _lp_phase(best_a, float(start), lp_end, rng)
            if np.isfinite(ls) and ls < best_sc:
                best_a, best_sc = la, ls

    adam_inits = [best_a.copy()]
    for _, aa in scored[1:5]:
        adam_inits.append(aa.copy())

    now = time.monotonic()
    if end - now > 2.0:
        ra, rs = _adam_phase(adam_inits, now, end - 1.0, rng)
        if ra is not None and np.isfinite(rs) and rs < best_sc:
            best_a, best_sc = ra, rs

    return best_a


def _make_inits(n, rng, parent=None, best=None, limit=10):
    n = int(n)
    out = []

    def add(arr):
        try:
            arr = np.asarray(arr, dtype=np.float64)
            if arr.size != n:
                if arr.size == 0:
                    arr = np.ones(n, dtype=np.float64) / float(n)
                else:
                    arr = _upsample(arr, n)
            arr = _norm(arr)
            if arr.size == n and np.all(np.isfinite(arr)):
                out.append(arr)
        except Exception:
            pass
        if len(out) > max(20, 3 * limit):
            return

    if parent is not None:
        p = _norm(parent)
        if p.size > 0:
            add(_upsample(p, n))
            add(_upsample(p[::-1], n))
            add(_norm(np.maximum(_upsample(p, n) * (1.0 + 0.08 * rng.standard_normal(n)), 0.0)))
            add(_norm(np.maximum(_upsample(p, n) * (1.0 + 0.20 * rng.standard_normal(n)), 0.0)))

    if best is not None:
        b = _norm(best)
        if b.size > 0:
            add(_upsample(b, n))
            add(_upsample(b[::-1], n))
            add(_norm(np.maximum(_upsample(b, n) * (1.0 + 0.10 * rng.standard_normal(n)), 0.0)))

    add(np.ones(n, dtype=np.float64) / float(n))

    if n > 2:
        x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
        for pp in (-0.7, -0.3, 0.2, 0.8, 1.6, 2.5, 4.0):
            add(np.maximum(x ** pp, 1e-9))
            add(np.maximum((1.0 - x) ** pp, 1e-9))

        sx = np.maximum(np.sin(np.pi * x), 1e-6)
        add(sx ** 0.7)
        add(sx ** 1.5)

        add(1.0 + 0.35 * np.cos(2.0 * np.pi * x))
        add(1.0 + 0.50 * np.cos(4.0 * np.pi * x))
        add(1.0 + 0.70 * np.sin(2.0 * np.pi * x + 0.7))

        for alpha, beta in [(0.3, 0.3), (0.7, -0.3), (-0.3, 0.7), (1.0, 0.2), (0.2, 1.0)]:
            add(np.maximum(x ** alpha * (1.0 - x) ** beta, 1e-9))

    if len(out) > limit:
        out = out[:limit]
    if not out:
        out.append(np.ones(n, dtype=np.float64) / float(n))
    return out


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 18.0)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.maximum(p, 0.0)
            s = float(p.sum())
            if s > 1e-12:
                parent = p / s
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(127, dtype=np.float64) / 127.0

    def consider(arr):
        nonlocal best_a, best_sc
        try:
            a = np.asarray(arr, dtype=np.float64)
            if a.size == 0:
                return
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.maximum(a, 0.0)
            s = float(a.sum())
            if s < 0.01:
                return
            vals = [float(x) for x in a]
            sc = float(evaluate_sequence(vals))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        except Exception:
            pass

    best_a = parent.copy()
    best_sc = np.inf
    consider(best_a)

    fallback = np.asarray([0.5, 0.1, 0.4], dtype=np.float64)
    if not np.isfinite(best_sc):
        consider(fallback)
    if not np.isfinite(best_sc):
        return [0.5, 0.1, 0.4]

    remaining = deadline - time.monotonic()
    if remaining < 25.0:
        lst = [float(max(0.0, min(1000.0, x))) for x in best_a]
        if lst and all(np.isfinite(x) for x in lst):
            return [float(x) for x in lst]
        return [0.5, 0.1, 0.4]

    p_len = int(parent.size)
    lengths = [p_len] if p_len > 0 else [127]
    for L in (255, 511, 1023):
        if L not in lengths and (p_len is None or L > p_len):
            lengths.append(L)

    weights = []
    for L in lengths:
        if L == (p_len if p_len > 0 else 127):
            weights.append(0.25 if L <= 255 else 0.50)
        elif L < 350:
            weights.append(0.12)
        elif L < 700:
            weights.append(0.30)
        else:
            weights.append(0.30)

    total_w = float(sum(weights))
    if total_w <= 0.0:
        weights = [1.0]
        total_w = 1.0
        lengths = [lengths[0]]

    final_reserve = min(80.0, 0.05 * max(1.0, deadline - t0))
    avail = max(1.0, deadline - t0 - final_reserve)

    for L, w in zip(lengths, weights):
        now = time.monotonic()
        if now >= deadline - final_reserve - 5.0:
            break
        alloc = max(12.0, avail * (float(w) / total_w))
        stage_end = min(now + alloc, deadline - final_reserve)
        if stage_end - now < 4.0:
            continue

        inits = _make_inits(int(L), rng, parent=parent, best=best_a,
                            limit=7 if int(L) < 600 else 9)
        res = _optimize_length(int(L), inits, now, stage_end, rng,
                               use_lp=(int(L) <= 640))
        if res is not None and res.size > 0:
            consider(res)

    now = time.monotonic()
    final_end = deadline - 2.0
    if final_end - now > 10.0:
        final_inits = [best_a.copy(), _norm(np.maximum(best_a, 0.0))[::-1]]
        res = _optimize_length(int(best_a.size), final_inits, now, final_end, rng, use_lp=False)
        if res is not None and res.size > 0:
            consider(res)

    a = _norm(best_a)
    vals = [float(max(0.0, min(1000.0, x))) for x in a]
    if not vals or any(not np.isfinite(x) for x in vals):
        vals = [0.5, 0.1, 0.4]

    try:
        sc = float(evaluate_sequence(vals))
        if not np.isfinite(sc):
            vals = [0.5, 0.1, 0.4]
    except Exception:
        vals = [0.5, 0.1, 0.4]

    return vals
