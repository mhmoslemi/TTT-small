        a_rev = a[::-1]
        FW = np.fft.rfft(weights, N_grad)
        FAr = np.fft.rfft(a_rev, N_grad)
        cross = np.fft.irfft(FW * np.conj(FAr), N_grad)
        grad = 2.0 * cross[:n]

        g_max = grad.max()
        if g_max < 1e-18:
            break

        # Multiplicative (mirror descent) update
        a *= np.exp(-eta * grad / g_max)
        a = np.maximum(a, 1e-20)
        a /= a.sum()

        # Periodic scoring
        if iter_count % 10 == 0:
            est = eval_fast(a)
            if est < best_score * 0.999:
                exact = float(evaluate_sequence(a.tolist()))
                if np.isfinite(exact) and exact < best_score:
                    best_score = exact
                    best_a = a.copy()
                    patience = 0
                    phase_patience = 0
                else:
                    patience += 3
            else:
                patience += 1
        elif iter_count % 3 == 0:
            patience += 1

        # --- Perturbation strategies ---
        if patience >= 60:
            # Small perturbation
            npert = max(3, n // 200)
            idx = rng.integers(0, n, size=npert)
            a[idx] *= np.exp(rng.normal(0.0, 0.10, npert))
            a = np.clip(a, 1e-20, 1000.0)
            a /= a.sum()
            patience = 0
            eta = min(0.7, eta * 1.03)

        if patience >= 200:
            # Medium perturbation + possible symmetrize
            npert = max(5, n // 50)
            idx = rng.integers(0, n, size=npert)
            a[idx] *= np.exp(rng.normal(0.0, 0.30, npert))
            a = np.clip(a, 1e-20, 1000.0)
            if rng.random() < 0.35:
                bl = rng.uniform(0.1, 0.35)
                a = (1 - bl) * a + bl * a[::-1]
            a = np.maximum(a, 1e-20)
            a /= a.sum()
            patience = 0
            tau_frac = min(0.15, tau_frac * 1.2)
            phase_patience += 1

        if patience >= 500:
            # Large perturbation: partial restart
            npert = max(10, n // 15)
            idx = rng.integers(0, n, size=npert, replace=False)
            a[idx] *= np.exp(rng.normal(0.0, 0.6, npert))
            a = np.clip(a, 1e-20, 1000.0)
            # Blend with a fresh random smooth init
            fr = np.exp(-((j - rng.uniform(0.3, 0.7) * n) / (rng.uniform(0.1, 0.3) * n)) ** 2)
            fr = np.maximum(fr, 0.05)
            blend = rng.uniform(0.1, 0.3)
            a = (1 - blend) * a + blend * fr / fr.sum()
            a = np.maximum(a, 1e-20)
            a /= a.sum()
            patience = 0
            phase_patience = 0
            tau_frac = 0.04
            k_top = max(20, int(0.12 * conv_len))

        # Anneal every 2000 iterations
        if iter_count % 2000 == 0:
            eta = max(0.08, eta * 0.93)
            k_top = max(10, int(k_top * 0.93))
            tau_frac = max(0.01, tau_frac * 0.9)

        # Time management: hard stop with margin
        if time.monotonic() > deadline - 5.0:
            break

    # --- Final validation ---
    result = [float(x) for x in best_a]
    fs = float(evaluate_sequence(result))
    if not np.isfinite(fs) or fs > 2.0:
        # Fallback
        result = [1.0] * n
        fs = float(evaluate_sequence(result))

    return result
