            A[k, j] = f[k - j]
    return A


def _lp_solve(f, M, n):
    """Solve max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    if linprog is None:
        return None
    c = -np.ones(n, dtype=np.float64)
    A = _make_A_dense(f, n)
    b_ub = np.full(2 * n - 1, M * (1.0 + 1e-10) + 1e-12, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30.0})
        if res is not None and getattr(res, 'x', None) is not None and res.status == 0:
            g = np.asarray(res.x, dtype=np.float64)
            g = np.clip(g, 0.0, None)
            sg = float(g.sum())
            if sg > 1e-12:
                return g / sg
    except Exception:
        pass
    return None


def _subgrad_direction(f, n, topk=5):
    """Compute subgradient of max(f*f) w.r.t. f."""
    b = np.convolve(f, f)
    # Get top-k indices
    k = min(topk, len(b))
    idx = np.argsort(b)[-k:][::-1]
    # Build gradient as average over active set
    grad = np.zeros(n, dtype=np.float64)
    for k_i in idx:
        # d(f*f)_{k_i}/d f_i = 2*f[k_i - j] for valid j
        for j in range(n):
            if 0 <= k_i - j < n:
                grad[j] += 2.0 * f[k_i - j]
    grad /= len(idx)
    return grad, float(b.max())


def _optimize_sequence(f, n, time_limit, deadl, best, rng):
    """Main optimization loop for a given sequence."""
    f = np.asarray(f, dtype=np.float64).copy()
    f = np.clip(f, 0.0, None)
    s = f.sum()
    if s < 1e-15:
        return
    f = f / s
    local = _score(f)
    if math.isfinite(local):
        _update_best(f, best)
    if local >= target[0]:
        return

    stop = time.monotonic() + time_limit
    stagnant = 0

    while time.monotonic() < stop - 3.0:
        if time.monotonic() >= deadl - 15.0:
            break
        b = np.convolve(f, f)
        M = float(b.max())
        if not math.isfinite(M) or M <= 0:
            break

        # LP step
        g = _lp_solve(f, M, n)
        if g is None:
            stagnant += 1
            if stagnant > 8:
                break
            # Try perturbation
            fp = f * np.exp(0.08 * rng.standard_normal(n))
            fp = np.clip(fp, 0.0, None)
            sp_ = fp.sum()
            if sp_ > 1e-12:
                fp = fp / sp_
                scp = _score(fp)
                if math.isfinite(scp) and scp < local * (1 - 1e-10):
                    f = fp
                    local = scp
                    stagnant = 0
                    _update_best(f, best)
                else:
                    stagnant += 1
            continue

        # Line search: mix f and g
        best_sc = local
        best_cand = None
        for t in (0.15, 0.3, 0.5, 0.7, 1.0):
            cand = (1.0 - t) * f + t * g
            cand = np.clip(cand, 0.0, None)
            sc = _score(cand)
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_cand = cand

        if best_cand is not None and best_sc < local * (1.0 - 1e-8):
            f = best_cand / best_cand.sum()
            local = best_sc
            stagnant = 0
            _update_best(f, best)
            if local < target[0] and stagnant <= 0:
                # Keep going to squeeze more
                pass
        else:
            stagnant += 1
            if stagnant % 5 == 0:
                # Perturb to escape
                fp = f * np.exp(0.1 * rng.standard_normal(n))
                fp = np.clip(fp, 0.0, None)
                sp_ = fp.sum()
                if sp_ > 1e-12:
                    fp = fp / sp_
                    scp = _score(fp)
                    if math.isfinite(scp) and scp < local * (1 - 1e-10):
                        f = fp
                        local = scp
                        stagnant = 0
                        _update_best(f, best)
            if stagnant > 20:
                break

    # Subgradient polishing
    if time.monotonic() < stop - 5.0:
        for it in range(200):
            if time.monotonic() >= stop - 3.0 or time.monotonic() >= deadl - 10.0:
                break
            grad, M = _subgrad_direction(f, n, topk=3)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-14:
                break
            grad = grad / gnorm
            best_sc = local
            best_cand = None
            for alpha in (0.01, 0.05, 0.1, 0.3, 0.5):
                cand = f - alpha * grad
                cand = np.clip(cand, 0.0, None)
                sc_ = cand.sum()
                if sc_ > 1e-12:
                    cand = cand / sc_
                    sc = _score(cand)
                    if math.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_cand = cand
            if best_cand is not None and best_sc < local * (1 - 1e-9):
                f = best_cand
                local = best_sc
                _update_best(f, best)
            else:
                if it > 30:
                    break


target = [2.0]


def _gen_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0.0, None)
        if a.size != n:
            return
        s = a.sum()
        if s > 1e-12:
            inits.append(a / s)

    # Uniform
    add(np.ones(n))

    # Two-level steps: first m are a, rest are b
    for m in range(1, n):
        ratio = m / (n - m) if m < n else 1
        f = np.ones(n)
        f[:m] = 1.0
        f[m:] = 0.3
        add(f)
        f2 = np.ones(n)
        f2[:m] = 0.3
        f2[m:] = 1.0
        add(f2)

    # Ramps
    for slope in (0.5, 1.0, 2.0, 5.0):
        add(1.0 + slope * x)
        add(1.0 + slope * (1.0 - x))

    # Power law
    for p in (0.5, 1.0, 2.0, 3.0):
        add(x ** p + 0.01)
        add((1 - x) ** p + 0.01)

    # Sine/cosine modulations
    for k in (1, 2, 3, 5):
        for amp in (0.3, 0.6, 0.9):
            add(1.0 + amp * np.sin(2 * np.pi * k * x))
            add(1.0 + amp * np.cos(2 * np.pi * k * x))

    # Dirichlet random
    for alpha in (0.5, 1.0, 2.0, 5.0):
        add(rng.dirichlet(np.full(n, alpha)))

    # Two-segment with random split
    for _ in range(4):
        m = rng.integers(1, n)
        a1 = rng.uniform(0.5, 2.0)
        a2 = rng.uniform(0.5, 2.0)
        f = np.ones(n)
        f[:m] = a1
        f[m:] = a2
        add(f)

    # Parent interpolation
    if parent is not None and parent.size > 1:
        p = np.asarray(parent, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, None)
        if p.sum() > 0:
            old_x = np.linspace(0, 1, p.size)
            new_x = np.linspace(0, 1, n)
            g = np.interp(new_x, old_x, p)
            add(g)
            add(g * np.exp(0.1 * rng.standard_normal(n)))

    if not inits:
        inits.append(np.ones(n) / n)
    return inits


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    hard_start = time.monotonic()
    try:
        remaining = max(10.0, float(budget_s))
    except Exception:
        remaining = 1300.0
    deadline = hard_start + remaining - 10.0

    best = [2.0, np.array([1.0, 1.0], dtype=np.float64) / 2.0]
    _update_best(best[1], best)
    target[0] = 2.0

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 0:
                sc = evaluate_sequence([float(x) for x in parent])
                if math.isfinite(sc) and sc < best[0]:
                    best[0] = float(sc)
                    best[1] = parent.copy()
                target[0] = best[0]
    except Exception:
        parent = None

    # Try key lengths with LP optimization
    lengths = [32, 48, 64, 96, 128, 160, 192, 256]

    for n in lengths:
        if time.monotonic() >= deadline - 60.0:
            break
        if n <= 64:
            max_starts = 8
            time_per_start = 8.0
        elif n <= 128:
            max_starts = 6
            time_per_start = 10.0
        elif n <= 192:
            max_starts = 5
            time_per_start = 12.0
        else:
            max_starts = 4
            time_per_start = 15.0

        inits = _gen_initializations(n, rng, parent)
        # Score and sort
        scored = []
        for a in inits:
            sc = _score(a)
            if math.isfinite(sc):
                scored.append((sc, a))
        scored.sort(key=lambda z: z[0])

        for sc0, a0 in scored[:max_starts]:
            if time.monotonic() >= deadline - 30.0:
                break
            if sc0 >= best[0] * 1.05 and len(scored) > 3:
                continue
            _optimize_sequence(a0, n, time_per_start, deadline, best, rng)
            target[0] = best[0]

    # Final subgradient polishing on best
    if time.monotonic() < deadline - 20.0:
        f_best = np.asarray(best[1], dtype=np.float64).copy()
        f_best = np.clip(f_best, 0, None)
        s = f_best.sum()
        if s > 1e-12:
            f_best = f_best / s
        n = f_best.size
        # Subgradient iterations
        for it in range(500):
            if time.monotonic() >= deadline - 5.0:
                break
            b = np.convolve(f_best, f_best)
            M = float(b.max())
            # Subgradient
            k_active = np.where(b > M * 0.999)[0]
            if len(k_active) == 0:
                break
            grad = np.zeros(n)
            for k_i in k_active:
                for j in range(n):
                    if 0 <= k_i - j < n:
                        grad[j] += 2.0 * f_best[k_i - j]
            grad /= len(k_active)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            improved = False
            for alpha in (0.005, 0.01, 0.02, 0.05, 0.1):
                cand = f_best - alpha * grad
                cand = np.clip(cand, 0, None)
                sc_ = cand.sum()
                if sc_ > 1e-12:
                    cand = cand / sc_
                    sc = _score(cand)
                    if math.isfinite(sc) and sc < _score(f_best) * (1 - 1e-10):
                        f_best = cand
                        improved = True
                        _update_best(f_best, best)
                        break
            if not improved and it > 50:
                break

    # Final validation
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.array([1.0])
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(final.size + 1, dtype=np.float64)
        s = final.sum()
    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc) or fin_sc > best[0] + 1e-12:
        # Fallback
        final = np.ones(4, dtype=np.float64)
        fin_sc = evaluate_sequence([float(x) for x in final])

    if time.monotonic() >= deadline - 2.0:
        pass  # Just return

    return [float(x) for x in final]
