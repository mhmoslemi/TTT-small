import numpy as np
import time

def _p2(l):
    return 1 << max(0, (int(l) - 1).bit_length())

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    dl = t0 + budget - 30.0

    # Load parent
    try:
        bp = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        assert bp.size > 0 and np.all(np.isfinite(bp))
    except Exception:
        bp = np.ones(6962, dtype=np.float64)
    bp = np.clip(bp, 0.0, 1000.0)
    n = int(bp.shape[0])

    L2 = 2 * n - 1
    L3 = 3 * n - 2
    SF = _p2(L3)

    def ev(a):
        try:
            v = float(evaluate_sequence([float(x) for x in a]))
            return v if np.isfinite(v) else np.inf
        except Exception:
            return np.inf

    def nrm(x):
        s = float(x.sum())
        return x / s if s > 1e-15 else x.copy()

    def og(x, T):
        """Smooth-max objective value and gradient."""
        S = float(x.sum())
        F = np.fft.rfft(x, SF)
        C = np.fft.irfft(F * F, SF)[:L2]
        cm = float(C.max())

        z = T * (C - cm)
        zm = float(z.max())
        e = np.exp(z - zm)
        es = float(e.sum())
        w = e / es
        M = cm + (zm + np.log(es)) / T

        W = np.fft.rfft(w[::-1], SF)
        cv = np.fft.irfft(F * W, SF)[:L3]
        dM = 2.0 * cv[n - 1: 2 * n - 1][::-1]

        S2 = S * S
        f = 2.0 * n * M / S2
        g = 2.0 * n * dM / S2 - 4.0 * n * M / (S2 * S)
        # Project gradient onto simplex tangent (remove mean)
        g -= g.mean()
        return f, g

    def run_adam(x, T, lr0, tstop):
        m1 = np.zeros_like(x)
        m2 = np.zeros_like(x)
        t = 0
        while time.monotonic() < tstop and time.monotonic() < dl - 15:
            t += 1
            lr = lr0 * (0.99995 ** t)
            _, g = og(x, T)
            m1 = 0.9 * m1 + 0.1 * g
            m2 = 0.999 * m2 + 0.001 * g * g
            x = x - lr * (m1 / (1.0 - 0.9 ** t)) / (np.sqrt(m2 / (1.0 - 0.999 ** t)) + 1e-8)
            x = np.clip(x, 0.0, 1000.0)
            x = nrm(x)
        return x

    # Initialize from parent, normalized to sum=1
    a = nrm(bp.copy())
    best_a = a.copy()
    best_sc = ev(np.clip(a * n, 0.0, 1000.0))
    print(f"Initial: {best_sc:.8f}")

    # Multi-temperature Adam phases
    avail = dl - t0
    phases = [
        (15.0,   0.020, 0.25),
        (50.0,   0.010, 0.22),
        (150.0,  0.005, 0.18),
        (500.0,  0.002, 0.15),
        (2000.0, 0.001, 0.12),
        (10000.0,0.0005,0.08),
    ]

    cum = 0.0
    for i, (T, lr, fr) in enumerate(phases):
        pe = t0 + (cum + fr) * avail
        a = run_adam(a, T, lr, pe)
        sc = ev(np.clip(a * n, 0.0, 1000.0))
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Guard: if score blew up, reset
        if not np.isfinite(sc) or sc > 4.0:
            a = best_a.copy()
        cum += fr
        print(f"Phase {i+1} (T={T:.0f}): {best_sc:.8f}")

        fs = ev(out)
    if not np.isfinite(fs):
        out = np.ones(n, dtype=np.float64)

    return [float(x) for x in out]
