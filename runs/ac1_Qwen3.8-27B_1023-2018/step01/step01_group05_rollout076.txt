def _next_pow2(m: int) -> int:
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def _autocorr_full(a: np.ndarray, fft_n: int) -> np.ndarray:
    n = len(a)
    A = np.fft.rfft(a, fft_n)
    return np.fft.irfft(A * A, fft_n)[: 2 * n - 1]


def _cross_weights(a: np.ndarray, w: np.ndarray, fft_n: int) -> np.ndarray:
    n = len(a)
    W = np.fft.rfft(w[::-1], fft_n)
    A = np.fft.rfft(a, fft_n)
    conv = np.fft.irfft(W * A, fft_n)
    return conv[n - 1: 2 * n - 1][::-1]

    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size != n:
        a = a[:n] if a.size > n else np.pad(a, (0, max(0, n - a.size)), mode="edge")

        return None
    a = a * (float(n) / s)
