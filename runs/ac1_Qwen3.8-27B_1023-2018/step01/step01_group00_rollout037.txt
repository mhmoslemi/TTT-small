import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(256, dtype=np.float64)
    try:
        p = np.clip(p, 0.0, 1000.0)
    except Exception:
        p = np.ones(256, dtype=np.float64)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(256, dtype=np.float64)
    return p


def _convolve_fft(a, fsize):
    """Full linear convolution of a with itself via FFT."""
    n = a.size
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: 2 * n - 1]
    c = np.maximum(c, 0.0)
    if not np.isfinite(c).all():
        c = np.where(np.isfinite(c), c, 0.0)
    return c


def _score_and_b(a, fsize):
    """Return (score, convolution) for probability vector a."""
    n = a.size
    b = _convolve_fft(a, fsize)
    s = 2.0 * n * float(b.max())
    return s, b


def _subgradient(a, half_b, n, top_k=12, tau_frac=0.0):
    """
    Compute subgradient of max(half_b) w.r.t. a.
    half_b = b[0:n] (lags 0 to n-1).
    Returns (subgrad_vector, max_value).
    """
    bmax = float(half_b.max())
    if bmax <= 0:
        return np.zeros(n), 0.0

    # Select active lags
    k_use = min(top_k, n)
    top_idx = np.argpartition(half_b, -k_use)[-k_use:]
    vals = half_b[top_idx]

    if tau_frac > 0:
        tau = tau_frac * bmax
        w = np.exp((vals - bmax) / tau)
        ws = w.sum()
        if ws > 0 and np.isfinite(ws):
            w = w / ws
        else:
            w = np.zeros_like(w)
            w[np.argmax(vals)] = 1.0
    else:
        # Hard top-1 with tie-breaking
        w = np.zeros_like(vals)
        mx = vals.max()
        ties = np.where(vals == mx)[0]
        w[ties] = 1.0 / len(ties)

    subgrad = np.zeros(n, dtype=np.float64)
    for wi, ki in zip(w, top_idx):
        ki = int(ki)
        wi = float(wi)
        if wi <= 0.0:
            continue
        # (grad b_ki)[j] = 2*a[ki-j] for j=0..ki, 0 for j>ki
        subgrad[: ki + 1] += 2.0 * wi * a[ki :: -1]

    return subgrad, bmax


def _grad_softmax_z(a, subgrad, n):
    """
    Convert subgradient w.r.t. a to gradient w.r.t. z (softmax input).
    grad_z = 2n * a * (subgrad - <subgrad, a>)
    """
    dot = float(np.dot(subgrad, a))
    return 2.0 * n * a * (subgrad - dot)

    s = e.sum()
    if s <= 0 or not np.isfinite(s):

    return e / s

                z += rng.normal(0, noise_scale, n)
                z -= z.mean()
                current_a = _softmax_raw(z)
                current_b = _convolve_fft(current_a, fsize)
                current_score = 2.0 * n * float(current_b.max())
                if not np.isfinite(current_score):
                    current_a = local_best_a.copy()
                    z = np.log(current_a + 1e-30)
                    z -= z.mean()
                    current_b = _convolve_fft(current_a, fsize)
                    current_score = local_best_score
                m_adam = np.zeros(n)
                v_adam = np.zeros(n)
                t_count = 0
                lr = 0.4
                tau_ann = max(0.3, tau_ann * 1.3)
                tau_frac = 0.015 * tau_ann
                stuck = 0

            if stuck > 80:
                # Try a fresh perturbation from the best
                z = np.log(local_best_a + 1e-30)
                z += rng.normal(0, 0.8, n)
                z -= z.mean()
                current_a = _softmax_raw(z)
                current_b = _convolve_fft(current_a, fsize)
                current_score = 2.0 * n * float(current_b.max())
                m_adam = np.zeros(n)
                v_adam = np.zeros(n)
                t_count = 0
                lr = 0.5
                tau_ann = 1.0
                tau_frac = 0.015
                stuck = 0

    # Polish with direct coordinate methods
    if time.monotonic() < stop_time - 2.0:
        local_best_a, local_best_score = _polish_direct(
            local_best_a, local_best_score, n, fsize, stop_time - 1.5, rng
        )

    return local_best_a, local_best_score


def _polish_direct(a, score, n, fsize, stop_time, rng):
    """
    Direct coordinate/element-wise polish on the probability vector.
    """
    a = np.clip(a.copy(), 1e-15, None)
    a = a / a.sum()
    best = a.copy()
    best_s = score
    b = _convolve_fft(a, fsize)
    bmax = float(b.max())
    current_s = 2.0 * n * bmax

    # Element-wise optimization: for each element, find the best value
    # that minimizes max(b), while maintaining sum = 1.
    # Strategy: for pair (i, j), transfer mass from one to the other.

    stuck_count = 0
    sigma = 0.05

    while time.monotonic() < stop_time:
        # Pick a random pair of indices to perturb
        strategy = rng.random()
        if strategy < 0.4:
            # Single element change (compensated by splitting to all others)
            i = int(rng.integers(0, n))
            delta = rng.normal(0, sigma)
            a_new = a.copy()
            a_new[i] = max(1e-15, a[i] + delta)
            # Renormalize
            a_new = a_new / a_new.sum()
        elif strategy < 0.7:
            # Pair transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            delta = rng.normal(0, sigma)
            a_new = a.copy()
            a_new[i] = max(1e-15, a[i] + delta)
            a_new[j] = max(1e-15, a[j] - delta)
            a_new = a_new / a_new.sum()
        elif strategy < 0.85:
            # Block perturbation
            w = int(rng.integers(1, max(2, n // 20)))
            s = int(rng.integers(0, n - w))
            a_new = a.copy()
            a_new[s:s+w] *= np.exp(rng.normal(0, 0.1, w))
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
        else:
            # Gaussian noise
            a_new = a * np.exp(rng.normal(0, sigma * 0.5, n))
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()

        b_new = _convolve_fft(a_new, fsize)
        s_new = 2.0 * n * float(b_new.max())

        if np.isfinite(s_new) and s_new < best_s * (1 - 1e-11):
            a = a_new
            b = b_new
            bmax = float(b.max())
            current_s = s_new
            if s_new < best_s:
                best = a.copy()
                best_s = s_new
                stuck_count = 0
                sigma = max(0.01, sigma * 0.95)
        else:
            stuck_count += 1
            if stuck_count % 30 == 0:
                sigma = min(0.5, sigma * 1.3)
            if stuck_count > 500:
                # Restart from best with larger perturbation
                a = best.copy()
                a = a * np.exp(rng.normal(0, 0.3, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                sigma = 0.1
                stuck_count = 0

    return best, best_s


def _try_chirp_init(n, rng):
    """Try a chirp-like initialization."""
    i = np.arange(n, dtype=np.float64)
    f = rng.uniform(0.5, 3.0)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.3, 0.9)
    a = 1.0 + amp * np.cos(2 * np.pi * f * i / n + phase)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def _try_multi_chirp_init(n, rng):
    """Multi-frequency cosine initialization."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(rng.integers(2, 5)):
        f = rng.uniform(1, max(2, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.2, 0.7)
        a += amp * np.cos(2 * np.pi * f * i / n + phase)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def _run_optimizer(n, seed, stop_time, parent, best_score):
    """Run optimizer with multiple initializations, return best."""
    rng = np.random.default_rng(seed)
    fsize = next_fast_len(2 * n - 1)

    best_a = parent[:n].copy() if parent.size >= n else np.ones(n) / n
    best_a = np.maximum(best_a, 1e-12)
    best_a = best_a / best_a.sum()
    b = _convolve_fft(best_a, fsize)
    best_s = 2.0 * n * float(b.max())

    if not np.isfinite(best_s):
        best_a = np.ones(n) / n
        best_s = 2.0

    # Multiple restarts with different initializations
    n_restarts = 3
    time_per = max(5.0, (stop_time - time.monotonic() - 3.0) / n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > stop_time - 3.0:
            break
        sub_stop = min(stop_time - 2.0, time.monotonic() + time_per)

        if r == 0:
            # Use parent
            init_a = best_a.copy()
        elif r == 1 and parent is not None and parent.size > 0:
            # Perturbed parent
            init_a = parent[:n].copy() if parent.size >= n else np.ones(n) / n
            init_a = np.maximum(init_a, 1e-12)
            init_a = init_a * np.exp(rng.normal(0, 0.3, n))
            init_a = np.maximum(init_a, 1e-12)
            init_a = init_a / init_a.sum()
        else:
            # Random structured init
            if rng.random() < 0.5:
                init_a = _try_chirp_init(n, rng)
            else:
                init_a = _try_multi_chirp_init(n, rng)

        # Convert to z-space and optimize
        z = np.log(init_a + 1e-30)
        z -= z.mean()

        current_a = init_a.copy()
        current_b = _convolve_fft(current_a, fsize)
        current_score = 2.0 * n * float(current_b.max())

        if not np.isfinite(current_score):
            current_a = np.ones(n) / n
            z = np.zeros(n)
            current_b = _convolve_fft(current_a, fsize)
            current_score = 2.0

        m_adam = np.zeros(n)
        v_adam = np.zeros(n)
        t_count = 0
        lr = 0.45
        stuck = 0
        tau_ann = 1.0
        top_k = 10
        tau_frac = 0.012 * tau_ann

        while time.monotonic() < sub_stop:
            half_b = current_b[:n]
            subgrad, bmax = _subgradient(current_a, half_b, n, top_k, tau_frac)
            grad_z = _grad_softmax_z(current_a, subgrad, n)

            gnorm = float(np.linalg.norm(grad_z))
            if not np.isfinite(gnorm) or gnorm < 1e-16:
                stuck += 3
                if stuck > 40:
                    break
                z += rng.normal(0, 0.4, n)
                z -= z.mean()
                current_a = _softmax_raw(z)
                current_b = _convolve_fft(current_a, fsize)
                current_score = 2.0 * n * float(current_b.max())
                if not np.isfinite(current_score):
                    z = np.log(best_a + 1e-30)
                    z -= z.mean()
                    current_a = _softmax_raw(z)
                    current_b = _convolve_fft(current_a, fsize)
                    current_score = 2.0 * n * float(current_b.max())
                continue

            t_count += 1
            b1, b2 = 0.9, 0.999
            m_adam = b1 * m_adam + (1 - b1) * grad_z
            v_adam = b2 * v_adam + (1 - b2) * grad_z ** 2
            m_hat = m_adam / (1 - b1 ** t_count)
            v_hat = v_adam / (1 - b2 ** t_count)
            direction = m_hat / (np.sqrt(v_hat) + 1e-8)
            direction[~np.isfinite(direction)] = 0.0

            improved = False
            for factor in [1.0, 0.5, 0.25, 0.125]:
                z_new = z - lr * factor * direction
                z_new -= z_new.mean()
                z_new = np.where(np.isfinite(z_new), z_new, 0.0)
                a_new = _softmax_raw(z_new)
                b_new = _convolve_fft(a_new, fsize)
                s_new = 2.0 * n * float(b_new.max())

                if np.isfinite(s_new) and s_new < current_score * (1 - 1e-10):
                    z = z_new
                    current_a = a_new
                    current_b = b_new
                    current_score = s_new
                    improved = True
                    if s_new < best_s:
                        best_s = s_new
                        best_a = a_new.copy()
                    break

            if improved:
                stuck = 0
                if t_count % 25 == 0:
                    tau_ann = max(0.08, tau_ann * 0.88)
                    tau_frac = 0.012 * tau_ann
            else:
                stuck += 1
                lr = max(0.02, lr * 0.88)
                if stuck % 12 == 0:
                    z += rng.normal(0, 0.35, n)
                    z -= z.mean()
                    current_a = _softmax_raw(z)
                    current_b = _convolve_fft(current_a, fsize)
                    current_score = 2.0 * n * float(current_b.max())
                    if not np.isfinite(current_score):
                        z = np.log(best_a + 1e-30)
                        z -= z.mean()
                        current_a = _softmax_raw(z)
                        current_b = _convolve_fft(current_a, fsize)
                        current_score = 2.0 * n * float(current_b.max())
                    m_adam = np.zeros(n)
                    v_adam = np.zeros(n)
                    t_count = 0
                    lr = 0.45
                    tau_ann = max(0.5, tau_ann * 1.2)
                    tau_frac = 0.012 * tau_ann
                    stuck = 0

                if stuck > 60:
                    z = np.log(best_a + 1e-30)
                    z += rng.normal(0, 0.6, n)
                    z -= z.mean()
                    current_a = _softmax_raw(z)
                    current_b = _convolve_fft(current_a, fsize)
                    current_score = 2.0 * n * float(current_b.max())
                    m_adam = np.zeros(n)
                    v_adam = np.zeros(n)
                    t_count = 0
                    lr = 0.5
                    tau_ann = 1.0
                    tau_frac = 0.012
                    stuck = 0

        # Final polish for this restart
        if time.monotonic() < stop_time - 2.0 and best_s < current_best_score_or_inf:
            pass

    # Final direct polish on the overall best
    if time.monotonic() < stop_time - 1.5:
        best_a, best_s = _polish_direct(
            best_a, best_s, n, fsize, stop_time - 1.0, rng
        )

    return best_a, best_s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(10.0, float(budget_s)) - 5.0

    parent = _get_parent()
    parent = np.clip(parent, 0.0, 1000.0)
    n_parent = parent.size

    # Validate parent
    try:
        ps = float(evaluate_sequence([float(x) for x in parent]))
    except Exception:
        ps = math.inf

    if not np.isfinite(ps) or parent.size < 2:
        parent = np.ones(192) / 192
        n_parent = 192
        ps = 2.0

    best_seq = parent.copy()
    best_score = ps

    # Primary optimization at parent length
    if time.monotonic() < deadline - 60.0:
        primary_stop = min(deadline - 30.0, t0 + budget_s * 0.55)
        try:
            opt_a, opt_s = _run_optimizer(
                n_parent, int(seed) + 7, primary_stop, parent, best_score
            )
            # Validate with exact evaluator
            try:
                exact_s = float(evaluate_sequence([float(x) for x in opt_a]))
                if np.isfinite(exact_s) and exact_s < best_score:
                    best_seq = opt_a.copy()
                    best_score = exact_s
            except Exception:
                pass
        except Exception:
            pass

    # Try other lengths
    other_lengths = [256, 512, 384, 128, 448, 640]
    for L in other_lengths:
        if time.monotonic() > deadline - 40.0:
            break
        if L == n_parent:
            continue
        if L < 4:
            continue
        alloc = min(50.0, max(15.0, (deadline - time.monotonic() - 20.0) / max(1, len([x for x in other_lengths if x != n_parent]))))
        sub_stop = min(deadline - 15.0, time.monotonic() + alloc)
        if sub_stop < time.monotonic() + 5.0:
            break
        try:
            opt_a, opt_s = _run_optimizer(
                L, int(seed) + 10000 + L, sub_stop, parent, best_score
            )
            try:
                exact_s = float(evaluate_sequence([float(x) for x in opt_a]))
                if np.isfinite(exact_s) and exact_s < best_score:
                    best_seq = opt_a.copy()
                    best_score = exact_s
            except Exception:
                pass
        except Exception:
            pass

    # Final polish on best
    if time.monotonic() < deadline - 5.0:
        try:
            n_best = best_seq.size
            fsize = next_fast_len(2 * n_best - 1)
            rng = np.random.default_rng(int(seed) + 99999)
            stopped = min(deadline - 3.0, time.monotonic() + max(0, deadline - time.monotonic() - 5.0))
            best_seq, best_score = _polish_direct(
                best_seq, best_score, n_best, fsize, stopped, rng
            )
        except Exception:
            pass

    # Final exact validation
    try:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if not np.isfinite(fs):
            raise ValueError
    except Exception:
        # Fall back to parent
        try:
            fs = float(evaluate_sequence([float(x) for x in parent]))
            if np.isfinite(fs):
                return [float(x) for x in parent]
        except Exception:
            pass
        return [1.0]

    return [float(x) for x in best_seq]
