```python
def build_conv_matrix(a, lags):
    """Build the sparse constraint matrix for the LP.
    Row k corresponds to autocorrelation lag lags[k].
    Column i corresponds to variable b[i].
    Entry (k, i) = a[i] if 0 <= lags[k] - i < n, else 0.
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    
    for ki, lag in enumerate(lags):
        # b[i] appears in (a*b)[lag] when 0 <= lag - i < n
        # i.e., lag - n + 1 <= i <= lag
        i_min = max(0, lag - n + 1)
        i_max = min(lag, n - 1)
        if i_max >= i_min:
            idx = np.arange(i_min, i_max + 1)
            rows.append(np.full(len(idx), ki))
            cols.append(idx)
            vals.append(a[idx])
    
    return csr_matrix((np.concatenate(vals), (np.concatenate(rows), np.concatenate(cols))),
                      shape=(len(lags), n))
```
