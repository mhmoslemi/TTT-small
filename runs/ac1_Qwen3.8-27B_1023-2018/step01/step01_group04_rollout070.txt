  Total: 2*a[k-i]. But only if both j=i and j=k-i are valid (i.e., 0<=i<n and 0<=k-i<n).

If k-i = i (k=2i):
  There's only one term: j=i=k-i, term = a[i]^2. Derivative = 2*a[i] = 2*a[k-i]. Same formula.

If k-i is not valid (k-i < 0 or k-i >= n):
  Only j=i contributes (if valid): derivative = a[k-i]... but k-i is out of range, so this term doesn't exist.
  Only j=k-i contributes (if valid): but k-i is out of range, so j is out of range too.

  - j=k-i is out of range (since k-i out of [0,n-1]), so this term doesn't exist either.
  
  So ∂c[k]/∂a[i] = 0 if k-i is not in [0,n-1].

Similarly, if i is not in [0,n-1], the derivative is 0 (but we only care about i in [0,n-1]).

So: ∂c[k]/∂a[i] = 2*a[k-i] if 0 <= k-i < n, else 0.

This is simple! The gradient of c[k] w.r.t. a is: 2 * shift(a, k) (where shift(a,k)[i] = a[k-i] if valid).

Now, the score is S = 2n * max_k c[k] / (sum a)^2. The gradient is complicated because of the max. But if we know which k* is the argmax, then:
∂S/∂a[i] = 2n * ∂c[k*]/∂a[i] / (sum a)^2 - 2n * c[k*] * 2 * sum a / (sum a)^4 * (sum of ∂sum_a/∂a[i])
= 2n * 2*a[k*-i]*I(0<=k*-i<n) / (sum a)^2 - 4n * c[k*] / (sum a)^3

(since ∂sum_a/∂a[i] = 1 for all i).

So: ∂S/∂a[i] = 4n*a[k*-i]*I(0<=k*-i<n)/(sum a)^2 - 4n*c[k*]/(sum a)^3
= (4n/(sum a)^2) * [a[k*-i]*I(0<=k*-i<n) - c[k*]/sum_a]

To minimize S, we want to decrease a[i] where ∂S/∂a[i] > 0 and increase where < 0. But we must keep a[i] >= 0.

def _score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()

    n = a.size

def _lp_iteration(a, M, n):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    # Build convolution matrix A: A @ g = conv(a, g)
    # conv(a,g)[k] = sum_i a[i] * g[k-i] for valid indices
    # For each k in 0..2n-2: sum_i a[i]*g[k-i] <= M
    # Coefficient of g[j] in constraint k: a[k-j] if 0<=k-j<n else 0
    # So A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                val = a[idx]
                if val > 0:
                    rows.append(k)
                    cols.append(j)
                    data.append(val)
    
    A = sps.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.status == 0 and res.x is not None:
        g = np.asarray(res.x, dtype=np.float64)
        g = np.clip(g, 0.0, None)
        return g
    return None

def _gradient_polish(a0, time_limit, rng, best):
    """Gradient-based polish."""
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 0.0, None)
    n = a.size
    s = float(np.sum(a))
    if s < 0.01:
        return

        if time.monotonic() > stop:
            break
        
        b = np.convolve(a, a)
        k_star = int(np.argmax(b))
        max_b = float(b[k_star])
        S = float(np.sum(a))
        
        # Gradient: dS/da[i] = (4n/S^2) * [a[k_star-i]*I(0<=k_star-i<n) - max_b/S]

                grad[i] += a[idx]

        for alpha in (lr, lr*0.5, lr*0.25, lr*0.1):
            cand = a - alpha * grad
            cand = np.clip(cand, 0.0, None)
            cs = float(np.sum(cand))

def _make_starts(n, rng, parent):
    """Generate diverse starting points."""
    starts = []

            return

            starts.append(a / s)

    add(np.ones(n))
    
    # Ramps (various powers)

        add(np.arange(1, n+1)**p)
        add((np.arange(1, n+1)**p)[::-1])
    
    # Exponentials
    for c in (0.5, 1.0, 2.0, 3.0, -0.5, -1.0, -2.0, -3.0):
        add(np.exp(c * (x - 0.5)))
    
    # Sine-based asymmetric
    for k in (1, 2, 3):
        for amp in (0.3, 0.6, 0.9):
            add(1.0 + amp * np.sin(2.0*np.pi*k*x + np.pi/4))
            add(1.0 + amp * np.cos(2.0*np.pi*k*x))
    
    # Random
    for _ in range(6):
        add(rng.dirichlet(np.full(n, 1.0)))
    for _ in range(4):
        v = np.abs(rng.standard_normal(n))
        add(v + 0.1)
    
    # Interpolated parent
    if parent is not None and len(parent) >= 2:
        p = np.asarray(parent, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, None)
        if p.sum() > 0.01:
            tp = np.linspace(0, 1, len(p))
            gp = np.interp(x, tp, p)
            add(gp)
            # Slightly perturbed
            add(gp * (1 + 0.1 * np.sin(4*np.pi*x)))
            add(gp * (1 + 0.1 * np.cos(2*np.pi*x)))
    
    if not starts:
        starts.append(np.ones(n) / n)
    
    # Sort by score
    scored = []
    for a in starts:
        sc = _score_fast(a)
        if math.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda z: z[0])
    return [a for _, a in scored]
