import time
import math

def _clean(v):
    try:
        arr = np.asarray(v, dtype=np.float64).ravel()
    except Exception:
        return np.array([1.0], dtype=np.float64)
    if arr.size == 0:
        return np.array([1.0], dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    return arr


def _norm0(a):
    a = _clean(a)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    s = float(np.sum(a))
    if not math.isfinite(s) or s <= 0.0:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s


def _conv(a):
    a = _clean(a)
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    if n <= 2048:
        return np.convolve(a, a, mode="full")
    N = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
    return b


def _score(a):
    a = _clean(a)
    n = a.size
    if n == 0:
        return float("inf")
    s = float(np.sum(a))
    if not math.isfinite(s) or s < 1e-15:
        return float("inf")
    b = _conv(a)
    if b.size == 0:
        return float("inf")
    mb = float(np.max(b))
    if not math.isfinite(mb):
        return float("inf")
    if mb < 0.0:
        mb = 0.0
    return float(2.0 * n * mb / (s * s))


def _eval_exact(seq):
    ev = globals().get("evaluate_sequence")
    if ev is None:
        return _score(seq)
    try:
        arr = _clean(seq)
        sc = ev([float(x) for x in arr])
    except Exception:
        return float("inf")
    try:
        sc = float(sc)
    except Exception:
        return float("inf")
    if not math.isfinite(sc):
        return float("inf")
    return sc


def _update_best(seq, best):

        arr = _clean(seq)
        if arr.size == 0:
            return
        s = float(np.sum(arr))
        if not math.isfinite(s) or s < 0.01:
            if s > 0.0:

            else:
                arr = np.full(arr.size, 0.01 / max(1, arr.size), dtype=np.float64)
        arr = np.clip(arr, 0.0, 1000.0)
        sc = _eval_exact(arr)
        if math.isfinite(sc) and sc < float(best[0]):
            best[0] = float(sc)
            best[1] = arr.copy()
    except Exception:
        pass


def _resample(parent, n):
    p = _clean(parent)
    if p.size == 0:
        p = np.array([1.0], dtype=np.float64)
    if p.size <= 1:
        return np.full(max(1, int(n)), float(p[0]), dtype=np.float64)
    n = int(max(1, int(n)))
    xo = np.linspace(0.0, 1.0, p.size)
    xn = np.linspace(0.0, 1.0, n)
    y = np.interp(xn, xo, p)
    y = np.clip(y, 0.0, 1000.0)
    return _norm0(y)


def _make_starts(n, parent, rng, num=5):
    n = int(max(2, n))
    starts = []

    def add(a):
        try:
            arr = _clean(a)
            if arr.size != n:
                arr = _resample(arr, n)
            arr = _norm0(arr)
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size == n and float(np.sum(arr)) > 0.0:
                starts.append(arr.copy())
        except Exception:
            pass

    if parent is not None:
        p = _resample(parent, n)
        add(p)
        add(p * np.exp(0.05 * rng.standard_normal(n)))
        add(p * np.exp(0.15 * rng.standard_normal(n)))

    add(np.ones(n, dtype=np.float64))

    x = np.linspace(1e-9, 1.0 - 1e-9, n)

    for _ in range(4):
        p = float(rng.uniform(0.1, 4.0))
        q = float(rng.uniform(0.1, 4.0))
        add((x ** p) * ((1.0 - x) ** q))

    for _ in range(3):
        c = float(rng.uniform(0.20, 0.80))
        s = float(rng.uniform(0.05, 0.45))
        add(np.exp(-((x - c) ** 2) / (2.0 * s * s)))

    for _ in range(3):
        K = max(1, min(12, n // 6))
        f = np.zeros(n, dtype=np.float64)
        for k in range(1, K + 1):
            f += (rng.normal(0.0, 1.0 / math.sqrt(max(1.0, float(k)))) *
                  np.cos(2.0 * math.pi * k * (x - 0.5) + rng.uniform(0.0, 2.0 * math.pi)))
        add(np.exp(f))

    max_k = max(1, min(4, n // 8))
    for k in range(1, max_k + 1):
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        for amp in (0.3, 0.7):
            add(1.0 + amp * np.sin(2.0 * math.pi * k * (x - 0.5) + ph))
            add(1.0 - amp * np.sin(2.0 * math.pi * k * (x - 0.5) + ph))

    for alpha in (1.0, 0.5, 2.0):
        try:
            add(rng.dirichlet(np.full(n, alpha)))
        except Exception:
            pass

    if not starts:
        add(np.ones(n, dtype=np.float64))

    scored = []
    for a in starts:
        a = _norm0(a)
        sc = _score(a)
        if math.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda z: z[0])
    return [a for _, a in scored[: max(1, int(num))]]


def _lp_polish(a0, time_limit, best, rng=None):
    a = _norm0(a0)
    n = a.size
    if _scipy_linprog is None or n < 3 or n > 160 or time_limit <= 2.0:
        _update_best(a, best)
        return a

    stop = time.monotonic() + float(time_limit)
    current = _score(a)
    local_best_a = a.copy()
    local_best_sc = current
    _update_best(a, best)
    stagnant = 0
    it = 0

    while time.monotonic() < stop - 4.0 and it < 2000:
        it += 1
        b = _conv(a)
        M = float(np.max(b))
        if not math.isfinite(M) or M <= 1e-15:
            break

        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j : j + n, j] = a

        b_ub = M * np.ones(2 * n - 1, dtype=np.float64) + (1e-9 * M + 1e-12)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = None
        try:
            res = _scipy_linprog(
                c, A_ub=A, b_ub=b_ub, bounds=bounds,
                method="highs", options={"presolve": True}
            )
        except Exception:
            try:
                res = _scipy_linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds)
            except Exception:
                res = None

        if res is None or getattr(res, "x", None) is None:
            break

        g = _clean(res.x)
        if g.size != n:
            break
        sg = float(np.sum(g))
        if not math.isfinite(sg) or sg <= 1e-12:
            stagnant += 1
            if stagnant > 25:
                break
            continue
        g = g / sg

        best_t = 0.0
        best_sc = current
        best_cand = a
        for t in np.linspace(0.0, 1.0, 25, dtype=np.float64):
            if t <= 0.0:
                continue
            cand = (1.0 - t) * a + t * g
            cand = _norm0(cand)
            sc = _score(cand)
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_t = float(t)
                best_cand = cand

        if math.isfinite(best_sc) and best_t > 1e-6 and best_sc < current * (1.0 - 1e-9):
            a = best_cand
            current = best_sc
            stagnant = 0
            if current < local_best_sc:
                local_best_sc = current
                local_best_a = a.copy()
            _update_best(a, best)
        else:
            stagnant += 1
            if stagnant in (6, 12, 20):
                improved = False
                if rng is not None:
                    for ns in (0.03, 0.08, 0.18):
                        cand = a * np.exp(ns * rng.standard_normal(n))
                        cand = _norm0(cand)
                        sc = _score(cand)
                        if math.isfinite(sc) and sc < current * (1.0 - 1e-9):
                            a = cand
                            current = sc
                            improved = True
                            break
                        cand = np.abs(a + ns * rng.standard_normal(n))
                        cand = _norm0(cand)
                        sc = _score(cand)
                        if math.isfinite(sc) and sc < current * (1.0 - 1e-9):
                            a = cand
                            current = sc
                            improved = True
                            break
                if improved:
                    stagnant = 0
                    if current < local_best_sc:
                        local_best_sc = current
                        local_best_a = a.copy()
                    _update_best(a, best)
            if stagnant > 40:
                break

    _update_best(local_best_a, best)
    return local_best_a


def _zfree_from_a(a, fix=None):
    a = _norm0(a)
    n = a.size
    if n <= 1:
        return a, 0, np.array([], dtype=np.int64), np.array([], dtype=np.float64)
    if fix is None:
        fix = int(np.argmax(a))
    z = np.log(np.clip(a, 1e-300, None))
    z -= z[fix]
    free = np.array([i for i in range(n) if i != fix], dtype=np.int64)
    return a, fix, free, z[free].copy()


def _softmax_from_free(zf, free, fix):
    n = int(len(free) + 1)
    z = np.zeros(n, dtype=np.float64)
    if len(free):
        zf = np.asarray(zf, dtype=np.float64).ravel()
        if zf.size != free.size:
            tmp = np.zeros(free.size, dtype=np.float64)
            m = min(free.size, zf.size)
            if m > 0:
                tmp[:m] = zf[:m]
            zf = tmp
        z[free] = np.nan_to_num(zf, nan=0.0, posinf=0.0, neginf=0.0)
    zmax = float(np.max(z)) if z.size else 0.0
    if not math.isfinite(zmax):
        return np.ones(n, dtype=np.float64) / max(1, n)
    z -= zmax
    z = np.clip(z, -745.0, 709.0)
    e = np.exp(z)
    s = float(np.sum(e))
    if not math.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / max(1, n)
    return e / s


def _active_polish(a0, time_limit, best, rng=None):
    a = _norm0(a0)
    n = a.size
    if n < 2 or time_limit <= 2.0:
        _update_best(a, best)
        return a

    stop = time.monotonic() + float(time_limit)

    local_best_sc = _score(a)
    _update_best(a, best)

    a, fix, free, zfree = _zfree_from_a(a)
    stale = 0
    it = 0

    while time.monotonic() < stop - 2.0 and it < 5000:

        zfree = np.nan_to_num(zfree, nan=0.0, posinf=0.0, neginf=0.0)
        a = _softmax_from_free(zfree, free, fix)

        if math.isfinite(sc) and sc < local_best_sc:

            _update_best(a, best)

        b = _conv(a)
        M = float(np.max(b))
        if not math.isfinite(M) or M <= 0.0:
            break

        tol = max(1e-12, 1e-7 * M)
        active = b >= (M - tol)
        cnt = int(np.sum(active))
        if cnt == 0:
            active = np.zeros(b.size, dtype=bool)
            active[int(np.argmax(b))] = True
            cnt = 1
        p = np.zeros(b.size, dtype=np.float64)
        p[active] = 1.0 / cnt

        gc = np.convolve(p[::-1], a, mode="full")
        grad_a = 2.0 * gc[n - 1 : 2 * n - 1][::-1]
        grad_a = np.nan_to_num(grad_a, nan=0.0, posinf=0.0, neginf=0.0)
        full = grad_a * a - float(np.dot(grad_a, a)) * a
        g = (2.0 * n) * full[free]
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
        gnorm = float(np.linalg.norm(g))

        if not math.isfinite(gnorm) or gnorm <= 1e-14:
            stale += 1
            if rng is not None:
                zfree = zfree + 0.02 * rng.standard_normal(free.size)
            if stale > 25:
                break
            continue

        zstd = float(np.std(zfree)) if zfree.size else 0.0
        if not math.isfinite(zstd) or zstd <= 1e-8:
            zstd = 0.5
        step_scale = max(0.02, min(2.0, zstd))
        d = g / gnorm
        improved = False

        for mult in (0.02, 0.05, 0.10, 0.25, 0.50, 1.00, 2.00, 4.00):
            nz = zfree - mult * step_scale * d
            na = _softmax_from_free(nz, free, fix)
            nasc = _score(na)
            if math.isfinite(nasc) and nasc < sc - 1e-12:
                zfree = nz
                a = na
                sc = nasc
                improved = True
                break

        if improved:
            stale = 0
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = a.copy()
            _update_best(a, best)
        else:
            stale += 1
            if rng is not None:
                zfree = zfree + 0.03 * rng.standard_normal(free.size)
            if stale > 30:
                break

        if it % 50 == 0:
            a, fix, free, zfree = _zfree_from_a(a)

    _update_best(local_best_a, best)
    return local_best_a


def _lbfgs_refine(a0, time_limit, best, rng=None):
    a = _norm0(a0)
    n = a.size
    if n < 2 or time_limit <= 2.0:
        _update_best(a, best)
        return a

    stop = time.monotonic() + float(time_limit)
    local_best_a = a.copy()
    local_best_sc = _score(a)
    _update_best(a, best)

    if _scipy_minimize is None:
        if time_limit > 5.0:
            return _active_polish(local_best_a, max(0.0, time_limit - 2.0), best, rng)
        return local_best_a

    if n <= 96:
        hard_nf = [4000, 2000, 1000, 500]
    elif n <= 192:
        hard_nf = [2500, 1200, 600, 300]
    elif n <= 384:
        hard_nf = [1200, 700, 350, 180]
    elif n <= 768:
        hard_nf = [700, 400, 220, 120]
    else:
        hard_nf = [400, 250, 150, 80]

    multipliers = [20.0, 80.0, 320.0, 1000.0]

    for phase, mult in enumerate(multipliers):
        remaining = stop - time.monotonic()
        if remaining < 7.0:
            break

        a, fix, free, zfree = _zfree_from_a(a)
        if free.size == 0:
            break

        b_start = _conv(a)
        scale = float(np.max(b_start)) if b_start.size else (1.0 / n)
        if not math.isfinite(scale) or scale <= 1e-15:
            scale = 1.0 / n
        tau = float(mult) / scale
        if not math.isfinite(tau) or tau <= 0.0:
            tau = 1000.0 / scale

        def fun(zf, free=free, n=n, tau=tau):
            z = np.zeros(n, dtype=np.float64)
            zf_arr = np.asarray(zf, dtype=np.float64).ravel()
            if zf_arr.size != free.size:
                tmp = np.zeros(free.size, dtype=np.float64)
                m = min(free.size, zf_arr.size)
                if m > 0:
                    tmp[:m] = zf_arr[:m]
                zf_arr = tmp
            z[free] = np.nan_to_num(zf_arr, nan=0.0, posinf=0.0, neginf=0.0)
            zmax = float(np.max(z)) if z.size else 0.0
            if not math.isfinite(zmax):
                a_tmp = np.ones(n, dtype=np.float64) / n
                return 1e100, np.zeros(free.size, dtype=np.float64)
            z -= zmax
            z = np.clip(z, -745.0, 709.0)
            e = np.exp(z)
            s = float(np.sum(e))
            if not math.isfinite(s) or s <= 1e-300:
                a_tmp = np.ones(n, dtype=np.float64) / n
                return 1e100, np.zeros(free.size, dtype=np.float64)
            a_tmp = e / s

            b = _conv(a_tmp)
            m = float(np.max(b)) if b.size else 0.0
            if not math.isfinite(m):
                m = 0.0
            shifted = b - m
            with np.errstate(over="ignore", under="ignore", invalid="ignore"):
                ex = tau * shifted
            ex = np.nan_to_num(ex, nan=0.0, posinf=0.0, neginf=0.0)
            ex = np.minimum(ex, 0.0)
            ex = np.maximum(ex, -745.0)
            e2 = np.exp(ex)
            S = float(np.sum(e2))
            if not math.isfinite(S) or S <= 1e-300:
                p = np.zeros(b.size, dtype=np.float64)
                if b.size:
                    p[int(np.argmax(b))] = 1.0
                L = m
            else:
                p = e2 / S
                L = m + np.log(S) / tau
            if not math.isfinite(L):
                L = m

            gc = np.convolve(p[::-1], a_tmp, mode="full")
            start_idx = len(p) - n
            if start_idx < 0 or start_idx + n > gc.size:
                grad_a = np.zeros(n, dtype=np.float64)
            else:
                grad_a = 2.0 * gc[start_idx : start_idx + n][::-1]
            grad_a = np.nan_to_num(grad_a, nan=0.0, posinf=0.0, neginf=0.0)
            dot = float(np.dot(grad_a, a_tmp))
            grad_full = grad_a * a_tmp - dot * a_tmp
            gf = (2.0 * n) * grad_full[free]
            gf = np.nan_to_num(gf, nan=0.0, posinf=0.0, neginf=0.0)

            val = 2.0 * n * L
            if not math.isfinite(val):
                val = 1e100
                gf = np.zeros(free.size, dtype=np.float64)
            return float(val), gf.astype(np.float64, copy=False)

        try:
            t0 = time.monotonic()
            fun0 = fun(zfree)
            t_eval = max(1e-6, time.monotonic() - t0)
        except Exception:
            break

        if math.isfinite(local_best_sc) and not math.isfinite(fun0[0]):
            continue

        nf = int(max(30, min(hard_nf[phase], (remaining - 5.0) / max(1e-6, 12.0 * t_eval))))
        nf = int(min(nf, 6000))
        if nf < 30:
            nf = 30

        try:
            res = _scipy_minimize(
                fun,
                zfree,
                method="L-BFGS-B",
                jac=True,
                options={
                    "max_nfev": nf,
                    "maxiter": nf,
                    "ftol": 1e-12,
                    "gtol": 1e-8,
                    "maxls": 20,
                    "disp": False,
                },
            )
            if res is not None and getattr(res, "x", None) is not None:
                nz = np.asarray(res.x, dtype=np.float64).ravel()
                if nz.size == free.size and np.all(np.isfinite(nz)):
                    zfree = nz
        except Exception:
            pass

        a = _softmax_from_free(zfree, free, fix)
        sc = _score(a)
        if math.isfinite(sc) and sc <= local_best_sc * 1.05:
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = a.copy()
        else:
            a = local_best_a.copy()
        _update_best(a, best)

    remaining = stop - time.monotonic()
    if remaining > 12.0:
        local_best_a = _active_polish(local_best_a, min(45.0, remaining), best, rng)

    _update_best(local_best_a, best)
    return local_best_a


def _process_length(n, alloc, rng, best, parent):
    n = int(n)
    end = time.monotonic() + float(max(0.0, alloc))
    if time.monotonic() >= end - 12.0:
        return

    num_starts = 5 if n <= 256 else 3
    if n <= 128:
        num_starts = 6
    if end - time.monotonic() < 80.0:
        num_starts = min(num_starts, 2)
    if end - time.monotonic() < 40.0:
        num_starts = 1

    starts = _make_starts(n, parent, rng, num_start=num_starts)
    if not starts:
        return

    for idx, a0 in enumerate(starts):
        if time.monotonic() >= end - 12.0:
            break

        if idx == 0 and n <= 128:
            lp_time = min(75.0, max(0.0, 0.22 * (end - time.monotonic())))
            if lp_time > 18.0:
                a0 = _lp_polish(a0, lp_time, best, rng)

        remaining = max(0.0, end - time.monotonic() - 6.0)
        rem_starts = max(1, len(starts) - idx)
        a = _lbfgs_refine(a0, remaining / float(rem_starts), best, rng)
        _update_best(a, best)

    if best[1].size == n and n <= 128:
        short = max(0.0, end - time.monotonic() - 6.0)
        if short > 25.0:
            a = _lp_polish(np.asarray(best[1], dtype=np.float64), min(40.0, short), best, rng)
            _update_best(a, best)

        parent = None

    if parent is None:
        x = np.linspace(1e-6, 1.0 - 1e-6, 64)
        parent = _norm0((x ** 1.0) * ((1.0 - x) ** 1.0) + 0.25)
        _update_best(parent, best)

    tasks = []
    seen = set()

    def add_len(n, alloc):
        try:
            n = int(round(float(n)))
        except Exception:
            return
        if 2 <= n <= 1024 and n not in seen:
            tasks.append((n, float(alloc)))
            seen.add(n)

    pn = parent.size if parent is not None else 0
    if pn > 0:
        add_len(pn, 340.0 if pn in (64, 96, 128, 192) else 290.0)
    add_len(96, 320.0)
    add_len(192, 200.0)
    add_len(256, 190.0)
    add_len(384, 160.0)
    add_len(512, 170.0)
    add_len(128, 150.0)
    add_len(64, 110.0)
    add_len(768, 150.0)
    add_len(1024, 120.0)

    for n, alloc in tasks:
        now = time.monotonic()
        if now >= deadline - 25.0:
            break
        avail = deadline - now - 15.0
        if avail < 25.0:
            break
        _process_length(n, min(alloc, avail), rng, best, parent)

    end_final = deadline - 6.0
    if time.monotonic() < end_final:
        cur = np.asarray(best[1], dtype=np.float64)
        if cur.size <= 128:
            tp = end_final - time.monotonic() - 12.0
            if tp > 25.0:
                cur = _lp_polish(cur, min(45.0, tp), best, rng)
        cur = np.asarray(best[1], dtype=np.float64)
        if cur.size <= 512:
            lp = max(0.0, end_final - time.monotonic() - 12.0)
            if lp > 25.0:
                cur = _lbfgs_refine(cur, min(70.0, lp), best, rng)

    last_good = float(best[0])
    no_improve = 0
    while time.monotonic() < end_final - 6.0:
        arr = np.asarray(best[1], dtype=np.float64).copy()
        n = arr.size
        if n < 2:
            break
        tried = 0
        for scale in (0.005, 0.02, 0.06, 0.12):
            if time.monotonic() >= end_final - 6.0:
                break
            tried += 1
            cand = arr * np.exp(scale * rng.standard_normal(n))
            cand = _norm0(cand)
            _update_best(cand, best)
            cand = np.clip(arr + scale * rng.standard_normal(n), 0.0, 1000.0)
            cand = _norm0(cand)
            _update_best(cand, best)
        if tried % 3 == 0:
            try:
                r = rng.dirichlet(np.ones(n, dtype=np.float64))
            except Exception:
                r = np.ones(n, dtype=np.float64) / n
            mix = 0.85 * arr + 0.15 * r
            _update_best(_norm0(mix), best)

        if float(best[0]) < last_good - 1e-12:
            last_good = float(best[0])
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 60:
                break

    try:
        final = _clean(best[1])
        if final.size == 0:
            final = np.array([1.0], dtype=np.float64)
        s = float(np.sum(final))
        if not math.isfinite(s) or s < 0.01:
            if s > 0.0:
                final = final * (0.01 / s)
            else:
                final = np.ones(max(1, final.size), dtype=np.float64) * (0.01 / max(1, final.size))
        final = np.clip(final, 0.0, 1000.0)
        sc = _eval_exact(final)
        if not math.isfinite(sc):
            fallback = np.asarray(parent, dtype=np.float64) if parent is not None else np.array([1.0], dtype=np.float64)
            fallback = _clean(fallback)
            if fallback.size == 0:
                fallback = np.array([1.0], dtype=np.float64)
            if fallback.size > 1024:
                fallback = _resample(fallback, 1024)
            final = _clean(fallback)
            s = float(np.sum(final))
            if not math.isfinite(s) or s < 0.01:
                final = np.ones(1, dtype=np.float64)
            final = np.clip(final, 0.0, 1000.0)
    except Exception:
        final = np.array([1.0], dtype=np.float64)

    return [float(x) for x in final]
