    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(5.0, float(budget_s)) - 12.0

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        ps_sum = parent.sum()
        if ps_sum > 1e-12:
            parent = parent / ps_sum
        else:
            parent = np.ones(127) / 127
    except Exception:
        parent = np.ones(127) / 127

    def exact_score(a):
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s < 1e-12:
            return np.inf
        c = np.convolve(a, a)
        mx = float(c.max())
        return 2.0 * len(a) * mx / (s * s)

    def softmax_fn(z):
        z = z - z.max()
        e = np.exp(np.clip(z, -700, 700))
        s = e.sum()
        if s < 1e-300:
            return np.full(len(z), 1.0 / len(z))
        return e / s

    def optimize_grad(a0, n_iter_max, end_time, beta_lo=10.0, beta_hi=800.0, lr_init=0.05):
        """Gradient-based optimization in softmax parameterization."""
        n = len(a0)
        z = np.log(np.maximum(a0, 1e-15)).copy()
        z = z - z.mean()

        # Adam
        m_adam = np.zeros(n)
        v_adam = np.zeros(n)
        steps = 0
        best_a = a0.copy()
        best_s = 2.0 * n * np.max(np.convolve(a0, a0))  # sum=1

        lr = lr_init
        trail = 0
        trail_lim = 300

        # Precompute L for FFT
        L = 1
        while L < 2 * n:
            L <<= 1

        for it in range(n_iter_max):
            if time.monotonic() > end_time:
                break

            steps += 1
            # Anneal beta
            frac = min(1.0, it / max(1, n_iter_max * 0.8))
            beta = beta_lo + (beta_hi - beta_lo) * (frac ** 1.5)

            z = z - z.mean()
            a = softmax_fn(z)
            c = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(a, L), L)[:2 * n - 1]
            c = np.maximum(c, 0.0)

            cur_s = 2.0 * n * float(c.max())
            if cur_s < best_s - 1e-12:
                best_s = cur_s
                best_a = a.copy()
                trail = 0
            else:
                trail += 1

            # Compute p = softmax(beta * c)
            c_shifted = c - c.max()
            p = np.exp(np.clip(beta * c_shifted, -700, 0.0))
            p_sum = p.sum()
            if p_sum < 1e-300:
                p = np.zeros_like(c)
                p[int(c.argmax())] = 1.0
            else:
                p /= p_sum

            C_beta = float(np.dot(p, c))

            # r[i] = sum_k p[k]*a[k-i] via FFT cross-correlation
            P = np.fft.rfft(p, L)
            A = np.fft.rfft(a, L)
            r = np.fft.irfft(P * np.conj(A), L)[:n]
            r = np.where(np.isfinite(r), r, 0.0)

            # Gradient in z-space: gz_i = 4n * a_i * (r[i] - C_beta)
            gz = 4.0 * n * a * (r - C_beta)
            gz = np.where(np.isfinite(gz), gz, 0.0)

            # Adam update
            b1, b2, eps = 0.9, 0.999, 1e-6
            m_adam = b1 * m_adam + (1 - b1) * gz
            v_adam = b2 * v_adam + (1 - b2) * gz * gz
            m_hat = m_adam / (1 - b1 ** steps)
            v_hat = v_adam / (1 - b2 ** steps)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z = np.where(np.isfinite(z), z, 0.0)

            # Restart if stuck
            if trail > trail_lim:
                warm = 0.15 + 0.25 * (trail / trail_lim % 1.0)
                z = np.log(np.maximum(best_a, 1e-15)) + warm * rng.standard_normal(n)
                z = z - z.mean()
                m_adam = np.zeros(n)
                v_adam = np.zeros(n)
                steps = 0
                lr = max(0.01, 0.6 * lr)
                trail = 0
                trail_lim = max(200, int(trail_lim * 1.3))

        return best_a, best_s

    def upsample(src, target_n):
        src = np.asarray(src, dtype=np.float64)
        if len(src) == target_n:
            return src.copy()
        x1 = np.linspace(0.0, 1.0, target_n)
        x0 = np.linspace(0.0, 1.0, len(src))
        out = np.interp(x1, x0, src)
        out = np.maximum(out, 0.0)
        s = out.sum()
        if s < 1e-12:
            out = np.ones(target_n) / target_n
        else:
            out = out / s
        return out

    # Track global best
    global_best = parent.copy()
    global_best_score = exact_score(parent)

    fallback = [0.5, 0.1, 0.4]
    fb_score = exact_score(np.array(fallback))
    if fb_score < global_best_score:
        global_best_score = fb_score
        global_best = np.array(fallback)

    def try_update(a):
        nonlocal global_best, global_best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        if a.size == 0:
            return
        s = a.sum()
        if s < 1e-12:
            return
        a = a / s
        sc = exact_score(a)
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best = a.copy()

    # Phase 1: Optimize parent length
    n_parent = len(parent)
    phase1_end = t0 + min(300.0, 0.35 * (deadline - t0))
    a1, s1 = optimize_grad(parent, 200000, phase1_end, beta_lo=8.0, beta_hi=600.0, lr_init=0.04)
    try_update(a1)

    # Phase 2: Restart with perturbation
    if time.monotonic() < deadline - 60:
        phase2_end = t0 + min(450.0, 0.5 * (deadline - t0))
        a_pert = global_best * (1.0 + 0.1 * rng.standard_normal(len(global_best)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a2, s2 = optimize_grad(a_pert, 200000, phase2_end, beta_lo=30.0, beta_hi=1000.0, lr_init=0.02)
        try_update(a2)

    # Phase 3: Try longer sequences
    if time.monotonic() < deadline - 40:
        for target_n in (255, 511, 1023):
            if time.monotonic() > deadline - 30:
                break
            phase_end = min(deadline - 20, time.monotonic() + 80.0)
            init = upsample(global_best, target_n)
            init_noisy = init * (1.0 + 0.05 * rng.standard_normal(target_n))
            init_noisy = np.maximum(init_noisy, 0)
            init_noisy = init_noisy / init_noisy.sum()
            a_long, s_long = optimize_grad(init_noisy, 50000, phase_end, beta_lo=15.0, beta_hi=500.0, lr_init=0.03)
            try_update(a_long)

    # Phase 4: Final polish on best
    if time.monotonic() < deadline - 10:
        phase4_end = deadline - 5.0
        a_fin, s_fin = optimize_grad(global_best, 100000, phase4_end, beta_lo=50.0, beta_hi=2000.0, lr_init=0.01)
        try_update(a_fin)

    # Final validation with exact evaluator
    candidates = [global_best]
    # Add a few more candidates
    for trial in range(3):
        pert = global_best.copy()
        idx = rng.choice(len(pert), size=max(1, len(pert) // 5), replace=False)
        pert[idx] *= np.exp(rng.normal(0, 0.05, size=len(idx)))
        pert = np.maximum(pert, 0)
        if pert.sum() > 1e-12:
            pert = pert / pert.sum()
            sc = exact_score(pert)
            if np.isfinite(sc) and sc < global_best_score:
                global_best = pert
                global_best_score = sc
                candidates = [pert]
                break

    # Validate with evaluate_sequence
    final_list = None
    final_score = np.inf
    for cand in candidates:
        try:
            lst = [float(x) for x in np.asarray(cand, dtype=np.float64)]
            if not lst or any(not np.isfinite(x) or x < -1e-15 for x in lst):
                continue
            lst = [max(0.0, x) for x in lst]
            sc = float(evaluate_sequence(lst))
            if np.isfinite(sc) and sc < final_score:
                final_score = sc
                final_list = lst
        except Exception:
            pass

    if final_list is None:
        lst = [max(0.0, float(x)) for x in global_best]
        sc = float(evaluate_sequence(lst))
        if np.isfinite(sc):
            final_list = lst
            final_score = sc

    if final_list is None:
        final_list = fallback

    return final_list
