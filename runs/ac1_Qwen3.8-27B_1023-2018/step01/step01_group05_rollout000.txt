def project_simplex(v, s=1.0):
    v = np.asarray(v,float)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, len(u)+1) > (cssv - s))[0][-1]

    w = np.maximum(v - theta, 0.0)
    # due numeric, ensure sum s
    w *= s / w.sum()

def _choose_beta(B, frac, focus=3.0, beta_max=1e7, beta_min=1e3):
    m=len(B)

                if a2.sum() > 1e-6:
                    consider_candidate(a2)

    consider_candidate(best_a)

    final_a = np.clip(np.asarray(best_a, dtype=np.float64), 0.0, 1000.0)
    if not np.all(np.isfinite(final_a)) or float(final_a.sum()) < 0.01:
        final_a = np.ones(n, dtype=np.float64)
    final_a = _normalize(final_a, n)
    final_score = exact_score(final_a)
    if not np.isfinite(final_score):
        final_a = np.ones(n, dtype=np.float64) / n

    return [float(v) for v in final_a]
