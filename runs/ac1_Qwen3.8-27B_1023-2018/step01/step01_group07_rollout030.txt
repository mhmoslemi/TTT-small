from scipy.signal import fftconvolve
import time

def _simplex_proj(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        return fftconvolve(x, x)[:conv_len]

    def compute_grad_subgrad(b):

        max_b = b.max()
