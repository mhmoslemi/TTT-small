def lp_direction(a, K, time_budget=60):
    c_peak = conv_a(a); max_c = c_peak.max()
    # choose lags: top K by c_peak and maybe endpoints? top = np.argpartition(c_peak, -K)[-K:]
    lags = np.sort(top)
    # Build A rows
    n = len(a)
    rows=[]; cols=[]; vals=[]
    for r,k in enumerate(lags):
       i_min=max(0,k-n+1); i_max=min(k,n-1)
       if i_max>=i_min:
          idx=np.arange(i_min,i_max+1)
          rows.append(np.full(idx.shape, r, dtype=np.int32))
          cols.append(idx.astype(np.int32))
          vals.append(a[k - idx]) # coeff a[k-i]
    if no vals: return None
    A = csr_matrix((np.concatenate(vals),(np.concatenate(rows),np.concatenate(cols)),shape=(len(lags),n)))
    # Build A_ub for variables [b (n), t]
    e = np.ones((len(lags),1))
    A_ub = scipy.sparse.hstack([A, -e], format='csr')
    b_ub = np.zeros(len(lags))
    A_eq = np.hstack([np.ones((1,n)), np.zeros((1,1))])
    b_eq = np.array([1.0])
    c_obj = np.zeros(n+1); c_obj[-1]=1.0
    bounds = [(0,None)]*n + [(None,None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
       b = res.x[:n]; b=np.maximum(b,0); b/=b.sum(); return b, res.x[-1]
    return None

def lp_direction(a, limit_s=45, K=512):
    c,_=fft_self(a)
    if c.size <= K: lags=np.arange(c.size)
    else: lags=np.argpartition(c, -K)[-K:]; lags=np.sort(lags)
    rows=[]; cols=[]; vals=[]
    for r,k in enumerate(lags):
       imin=max(0,k-n+1); imax=min(k,n-1)
       if imax>=imin:
          idx=np.arange(imin,imax+1)
          rows.append(np.full(idx.size,r,dtype=np.int32))
          cols.append(idx.astype(np.int32))
          vals.append(a[k-idx])
    if not rows: return None
    A=csr_matrix((np.concatenate(vals),(np.concatenate(rows),np.concatenate(cols)),shape=(len(lags),n)))
    one_col = csr_matrix((np.ones(len(lags)),(np.arange(len(lags)),np.zeros(len(lags),dtype=int))),(len(lags),1))
    A_ub = hstack([A, -one_col], format='csr')
    b_ub = np.zeros(len(lags))
    A_eq = np.hstack([np.ones((1,n)), [[0.0]]])
    b_eq = np.array([1.0])
    c_obj = np.zeros(n+1); c_obj[-1]=1
    bounds=[(0,None)]*n + [(None,None)]
    try:
       res=linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs', options={'time_limit':60})
    except Exception:
       try: res=linprog(... method='highs')
       except: return None
    if res.success:
       b=res.x[:n]; b=np.maximum(b,0); s=b.sum(); if s>0: b/=s; return b
    return None

        K = max(16, min(K, 2 * n - 1))

        if K >= c.size:
            lags = np.arange(c.size, dtype=np.int64)
        else:
            lags = np.argpartition(c, -K)[-K:]
            lags = np.sort(lags)

        rows = []
        cols = []
        vals = []
        for r, k in enumerate(lags):
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            if i_max >= i_min:
                idx = np.arange(i_min, i_max + 1, dtype=np.int32)
                rows.append(np.full(idx.size, r, dtype=np.int32))
                cols.append(idx)
                vals.append(a[k - idx])

        if not rows:
            return None

        A = csr_matrix(
            (np.concatenate(vals), (np.concatenate(rows), np.concatenate(cols))),
            shape=(len(lags), n),
        )
        ar = np.arange(len(lags), dtype=np.int32)
        neg_col = csr_matrix((-np.ones(len(lags)), (ar, np.zeros(len(lags), dtype=np.int32))), shape=(len(lags), 1))
        A_ub = hstack([A, neg_col], format="csr")
        b_ub = np.zeros(len(lags), dtype=np.float64)
        A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
        b_eq = np.array([1.0], dtype=np.float64)
        c_obj = np.zeros(n + 1, dtype=np.float64)
        c_obj[-1] = 1.0
        bounds = [(0.0, None)] * n + [(None, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method="highs")
        except Exception:
            return None
        if getattr(res, "success", False) is True and res.x is not None:
            bx = np.asarray(res.x[:n], dtype=np.float64)
            bx = np.where(np.isfinite(bx), bx, 0.0)
            bx = np.maximum(bx, 0.0)
            s = bx.sum()
            if s > 1e-12:
                return bx / s
        return None

    def try_accept_direction(dirx, cur_peak):
        if dirx is None:
            return False
        dirx = _normalize(dirx)
        cur = _normalize(current_a)
        best_x = cur
        best_f = cur_peak
        best_lam = 0.0
        for lam in (1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001):
            if time.monotonic() > final_deadline - 2.0:
                break
            x = (1.0 - lam) * cur + lam * dirx
            x = np.maximum(x, 0.0)
            s = x.sum()
            if s <= 1e-300:
                continue
            x = x / s
            f = internal_peak(x)
            if f < best_f:
                best_f = f
                best_x = x
                best_lam = lam
                if best_f < cur_peak * (1.0 - 1e-4):
                    break
        if best_lam > 0.0 and best_f < cur_peak * (1.0 - 1e-7):
            set_current_from_x(best_x)
            consider(best_x, require=False)
            return True
        return False

    def mixture_search(allow_current_update=True):
        if len(good) < 2 or time.monotonic() > final_deadline - 10.0:
            return
        top = sorted(good, key=lambda t: t[0])[:min(8, len(good))]
        # Average of the best stored candidates.
        x = np.mean([arr for _, arr in top], axis=0)
        x = _normalize(x)
        sc = 2.0 * float(n) * internal_peak(x)
        if sc < best_int:
            consider(x, require=False)
            if allow_current_update:
                set_current_from_x(x)
        # A few random Dirichlet mixtures.
        for _ in range(3):
            if time.monotonic() > final_deadline - 10.0:
                break
            idx = rng.choice(len(good), size=min(3, len(good)), replace=False)
            ws = rng.dirichlet(np.ones(idx.size, dtype=np.float64))
            x = np.zeros(n, dtype=np.float64)
            for i, j in enumerate(idx):
                x += float(ws[i]) * good[j][1]
            x = _normalize(x)
            sc = 2.0 * float(n) * internal_peak(x)
            if sc < best_int:
                consider(x, require=False)
                if allow_current_update:
                    set_current_from_x(x)
        # Mix current with the exact best.
        if best_a is not None:
            for lam in (0.2, 0.35):
                x = (1.0 - lam) * current_a + lam * best_a
                x = _normalize(x)
                sc = 2.0 * float(n) * internal_peak(x)
                if sc < best_int:
                    consider(x, require=False)
                    if allow_current_update:
                        set_current_from_x(x)

    def permute_restart():
        src = best_a if best_a is not None else current_a
        src = _normalize(src)
        base_pk = internal_peak(src)
        best_x = None
        best_pk = base_pk
        for tr in range(6):
            if time.monotonic() > final_deadline - 30.0:
                break
            x = src[rng.permutation(n)].copy()
            if tr % 2 == 1:
                x = np.maximum(x * np.exp(rng.normal(0.0, 0.08, n)), 0.0)
            x = _normalize(x)
            pk = internal_peak(x)
            if pk < best_pk:
                best_pk = pk
                best_x = x
        if best_x is not None and best_pk < base_pk * 0.999:
            set_current_from_x(best_x)
            consider(best_x, require=False)
        else:
            z = np.log(np.maximum(src, 1e-300)) + rng.normal(0.0, 0.08, n)
            set_current_from_x(_softmax(z))

    # Main optimization loop.
    while time.monotonic() < deadline:
        now = time.monotonic()

        if now - seg_start >= seg_len:
            seg_start = now
            consider(current_a, require=False)
            if len(good) >= 2 and now - last_mix > 30.0:
                mixture_search(True)
                last_mix = now
            if no_improve > 1200 and now - last_perm > 60.0 and now < deadline - 200.0:
                permute_restart()
                last_perm = now
            if best_exact < 1.505:
                seg_len = 180.0
            elif best_exact < 1.52:
                seg_len = 120.0
            else:
                seg_len = 90.0
            continue

        batch_break = False
        for _ in range(6):
            if time.monotonic() > deadline - 1.0:
                break

            cur = current_a
            c, Fa = fft_self(cur)
            cmax = float(c.max())
            if not np.isfinite(cmax) or cmax <= 0.0:
                batch_break = True
                break

            sc = 2.0 * float(n) * cmax
            if sc < best_int - 1e-12:
                best_int = sc
                best_int_a = cur.copy()
                if time.monotonic() - last_exact > 20.0 and sc < best_exact * 0.999:
                    exact_eval(cur)

            mean_c = 1.0 / float(2 * n - 1)
            excess = max(cmax - mean_c, 1e-12)
            if no_improve < 300:
                factor = 0.18
            elif no_improve < 1500:
                factor = 0.10
            else:
                factor = 0.05
            tau = factor * excess

            sh = (c - cmax) / tau
            e = np.exp(np.clip(sh, -40.0, 0.0))
            swe = e.sum()
            if swe <= 0.0 or not np.isfinite(swe):
                w = np.zeros(2 * n - 1, dtype=np.float64)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / swe

            g = 2.0 * cross_corr(w, cur, Fa)
            gd = float(np.dot(cur, g))
            gz = cur * (g - gd)

            # Mild entropy/barrier regularization only when the iterate has become too concentrated.
            if cur.max() > 1.0 / (3.0 * float(n)):
                loga = np.log(np.maximum(cur, 1e-300))
                gz = gz + 0.02 * cur * (loga - float(np.dot(cur, loga)))

            ad_t += 1
            m = b1 * m + (1.0 - b1) * gz
            v = b2 * v + (1.0 - b2) * (gz * gz)
            bc1 = 1.0 - b1 ** ad_t
            bc2 = 1.0 - b2 ** ad_t
            new_z = current_z - lr * (m / bc1) / (np.sqrt(v / bc2) + eps)

            try_a = _softmax(new_z)
            try_c, _ = fft_self(try_a)
            try_max = float(try_c.max())

            allow = (1e-9 + (1e-5 if no_improve > 1000 else 0.0)) * cmax
            if try_max <= cmax + allow:
                current_z = new_z
                current_a = try_a
                if try_max < cmax - 1e-13:
                    no_improve = 0
                else:
                    no_improve += 1

                if no_improve > 700 and now - last_dir > 15.0 and time.monotonic() < deadline - 120.0:
                    last_dir = time.monotonic()
                    dirx = fw_direction(current_a, limit_s=20.0)
                    if dirx is None and no_improve > 1800 and time.monotonic() < deadline - 180.0:
                        dirx = lp_direction(current_a)
                    if dirx is not None:
                        ok = try_accept_direction(dirx, cmax)
                        if not ok:
                            current_z += rng.normal(0.0, 0.05, n)
                            current_a = _softmax(current_z)
                            m *= 0.7
                            v *= 0.7
                            no_improve = 300
                    batch_break = True

                if no_improve > 2500:
                    current_z += rng.normal(0.0, 0.08, n)
                    current_a = _softmax(current_z)
                    m *= 0.5
                    v *= 0.5
                    lr = min(lr * 1.3, 0.25)
                    no_improve = 800
                    batch_break = True
            else:
                no_improve += 1
                lr = max(lr * 0.93, 1e-5)
                if no_improve > 500:
                    batch_break = True

        if batch_break:
            # Nudge slowly if a direction was accepted; otherwise ready for segment boundary.
            if time.monotonic() > deadline - 1.0:
                break

    # Final polishing phase: try mixtures and exact-evaluate the strongest candidates.
    while time.monotonic() < final_deadline - 2.0:
        if len(good) >= 2:
            top = sorted(good, key=lambda t: t[0])[:min(8, len(good))]
            for _ in range(5):
                if time.monotonic() > final_deadline - 2.0:
                    break
                idx = rng.choice(len(good), size=min(4, len(good)), replace=False)
                ws = rng.dirichlet(np.ones(idx.size, dtype=np.float64))
                x = np.zeros(n, dtype=np.float64)
                for i, j in enumerate(idx):
                    x += float(ws[i]) * good[j][1]
                x = _normalize(x)
                sc = 2.0 * float(n) * internal_peak(x)
                if sc < best_exact * 1.005 or sc < best_int:
                    exact_eval(x)
        exact_eval(current_a)
        if best_int_a is not None:
            exact_eval(best_int_a)

    # Choose the best exact-scoring finite candidate.
    final_candidates = []
    if best_a is not None:
        final_candidates.append(best_a)
    if best_int_a is not None:
        final_candidates.append(best_int_a)
    if current_a is not None:
        final_candidates.append(current_a)
    if base is not None:
        final_candidates.append(base)

    chosen = None
    chosen_e = np.inf
    for x in final_candidates:
        if x is None:
            continue
        x = _normalize(x)
        try:
            e = evaluate_sequence(x.tolist())
        except Exception:
            e = np.inf
        if np.isfinite(e) and e < chosen_e:
            chosen_e = e
            chosen = x

    if chosen is None:
        chosen = np.ones(n, dtype=np.float64) / float(n)

    return [float(x) for x in chosen]
