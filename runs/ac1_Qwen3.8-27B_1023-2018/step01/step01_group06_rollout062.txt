def build_conv_matrix_fast(a):
    n = len(a)
    conv_len = 2*n - 1
    # For each k, the nonzeros are at j from max(0, k-n+1) to min(n-1, k)
    # Total nonzeros = n^2
    # Use scipy's diags or manual construction
    
    # Actually, let's use the structure: row k has a[j] at column j for valid j
    # This is like a stacked set of shifted copies of a
    
    # Method: for each j, a[j] appears in rows k = j, j+1, ..., j+n-1
    # So the matrix is: for column j, nonzeros at rows max(0,j) to min(2n-2, j+n-1)
    
    rows = np.zeros(n * n, dtype=np.int32)  # approximate size
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    
    idx = 0
    for j in range(n):
        k_lo = j  # minimum k where j is valid (k-j = j >= 0)
        k_hi = min(conv_len - 1, j + n - 1)  # maximum k (k-j <= n-1, i.e., k <= j+n-1)
        # Wait, k-j must be in [0, n-1], so k in [j, j+n-1]
        # And k in [0, 2n-2]
        k_lo = max(0, j)  # = j since j >= 0
        k_hi = min(2*n-2, j+n-1)
        ks = np.arange(k_lo, k_hi+1)
        count = len(ks)
        rows[idx:idx+count] = ks
        cols[idx:idx+count] = j
        vals[idx:idx+count] = a[j]
        idx += count
    
    A = sparse.csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(conv_len, n))
    return A
