import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len

def _fft_convolve(a):
    n = len(a)
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    c = irfft(F*F, fsize)[:2*n-1]
    c[c < 0] = 0  # remove numerical negatives
    return c

def _score(a):
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = len(a)
    b = np.convolve(a, a)
    return 2*n*b.max()/(s*s)

def _lp_best_response(a, M):
    """Find g maximizing sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    # Build constraint matrix A of shape (2n-1, n)
    # A[k,i] = a[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        for i in range(lo, hi+1):
            A[k, i] = a[k-i]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.status != 0:
        return None
    return res.x

def _lp_iteration(a, max_iters=200, stop_time=None):
    """Iterate the LP best-response method."""
    n = len(a)
    a = np.clip(a, 0, 1000).copy().astype(float)
    s = a.sum()
    if s <= 0:
        return a
    a = a / s  # normalize to sum=1
    
    best_a = a.copy()
    best_M = a @ a  # wait, this isn't right for convolution
    b = _fft_convolve(a)
    best_M = b.max()
    
    for it in range(max_iters):
        if stop_time and time.monotonic() > stop_time:
            break
        
        b = _fft_convolve(a)
        M = b.max()
        
        if M < best_M:
            best_M = M
            best_a = a.copy()
        
        g = _lp_best_response(a, M)
        if g is None:
            break
        
        gs = g.sum()
        if gs <= 0:
            break
        g = g / gs
        
        # Line search on step size
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.1]:
            cand = (1-t)*a + t*g
            cand = np.clip(cand, 0, 1000)
            cs = cand.sum()
            if cs <= 0:
                continue
            cand = cand / cs
            cb = _fft_convolve(cand)
            cM = cb.max()
            if cM < M * (1 - 1e-10):
                a = cand
                improved = True
                break
        
        if not improved:

                cand = (1-t)*a + t*g
                cand = np.clip(cand, 0, 1000)
                cs = cand.sum()

                cM = cb.max()

                    a = cand
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_a

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(192, dtype=np.float64) / 192.0
    p = np.clip(p, 0.0, 1000.0)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(192, dtype=np.float64) / 192.0
    return p


def _convolve_square(a):
    """Compute a * a (full convolution of a with itself)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([0.0])
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_normalized(a):
    """Score assuming sum(a)=1. Returns 2*n*max(a*a)."""
    b = _convolve_square(a)
    return 2.0 * a.size * float(b.max())


def _exact_score(a):
    """Compute the exact evaluator score."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    n = a.size
    b = _convolve_square(a)
    mb = float(b.max())
    return 2.0 * n * mb / (s * s)


def _build_A(a):
    """Build constraint matrix A of shape (2n-1, n) where A[k,i] = a[k-i] if valid."""
    n = a.size
    m = 2 * n - 1
    k_idx = np.arange(m, dtype=np.int64)[:, None]
    i_idx = np.arange(n, dtype=np.int64)[None, :]
    j = k_idx - i_idx
    mask = (j >= 0) & (j < n)
    safe_j = np.clip(j, 0, n - 1)
    A = np.where(mask, a[safe_j], 0.0)
    return A


def _lp_best_response(a, M, timeout_s=2.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = a.size
    A = _build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    t_start = time.monotonic()
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': timeout_s, 'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if not np.isfinite(g).all() or g.sum() <= 0:
        return None
    return g


def _lp_optimize(a0, stop_time, max_iters=500):

    best_M = float(_convolve_square(a).max())

    stuck = 0

        b = _convolve_square(a)

            best_a = a.copy()
            stuck = 0
        else:

        g = g / gs

def _initializations(parent, seed, n):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    inits = []

    # 1. Parent normalized
    p = np.clip(parent.astype(np.float64), 0.0, 1000.0)
    ps = p.sum()
    if ps > 0 and p.size == n:
        inits.append(p / ps)

    # 2. Linear ramp (increasing)
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())

    # 3. Linear ramp (decreasing)
    inits.append(ramp[::-1] / ramp.sum())

    # 4. Exponential decay
    alpha = rng.uniform(0.5, 2.0)
    exps = np.exp(-alpha * np.arange(n) / n)
    inits.append(exps / exps.sum())

    # 5. "Lopsided" - concentrated on one side
    center = n * rng.uniform(0.2, 0.8)
    width = rng.uniform(0.1, 0.5) * n
    x = np.arange(n, dtype=np.float64)
    gauss = np.exp(-0.5 * ((x - center) / width) ** 2)
    gauss = np.maximum(gauss, 1e-10)
    inits.append(gauss / gauss.sum())

    # 6. Quadratic (parabolic)
    x = np.linspace(0, 1, n)
    quad = x ** 2
    inits.append(quad / quad.sum())
    quad2 = (1 - x) ** 2
    inits.append(quad2 / quad2.sum())

    # 7. Sinusoidal modulation
    freq = rng.uniform(0.3, 1.5)
    sines = 0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n) / n)
    sines = np.maximum(sines, 0.01)
    inits.append(sines / sines.sum())

    # 8. Random smooth
    r = rng.normal(0, 1, n)
    # Smooth with a small kernel
    k = np.array([0.1, 0.8, 1.0, 0.8, 0.1])
    k = k / k.sum()
    smooth = np.convolve(r, k, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    # 9. Parent with noise
    if p.size == n and ps > 0:
        pn = p / ps
        noisy = pn * np.exp(rng.normal(0, 0.3, n))
        noisy = np.maximum(noisy, 1e-6)

    # 10. "Two-block" - high on one side, low on other
    split = int(n * rng.uniform(0.4, 0.6))

    two_block[split:] = rng.uniform(0.1, 0.5)
    inits.append(two_block / two_block.sum())

            # Gaussian perturbation in log space
            z = np.log(np.maximum(a, 1e-8))
            z += rng.normal(0, 0.2, n)
            cand = np.exp(z - z.max())
        elif r < 0.75:
            # Block scaling
            cand = a.copy()
            width = rng.integers(3, max(4, n // 10))
            start = rng.integers(0, n - width)
            factor = np.exp(rng.normal(0, 0.3))
            cand[start:start + width] *= factor
        elif r < 0.85:
            # Coordinate pair swap
            i, j = rng.integers(0, n, 2)
            cand = a.copy()
            delta = rng.uniform(-0.05, 0.05)
            cand[i] += delta
            cand[j] -= delta
        else:
            # Mirror blend
            cand = (1 - 0.5) * a + 0.5 * a[::-1]

        cand = np.clip(cand, 0.0, 1000.0)
        cs = cand.sum()
        if cs <= 0:
            continue
        cand = cand / cs
        cM = float(_convolve_square(cand).max())

        if cM < best_M * (1 - 1e-11):
            best_M = cM
            best_a = cand.copy()
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                # Reset with larger perturbation
                a = a * np.exp(rng.normal(0, 0.5, n))
                a = np.maximum(a, 1e-8)
                a = a / a.sum()
                no_improve = 0

    return best_a, best_M


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + max(0.0, budget - 8.0)

    parent = _get_parent()
    n_parent = parent.size

    # Validate parent
    best_seq = parent.copy()
    try:
        best_score = float(evaluate_sequence([float(x) for x in parent]))
    except Exception:
        best_score = float('inf')
    if not np.isfinite(best_score):
        best_seq = np.ones(32, dtype=np.float64) / 32.0
        best_score = 2.0

    rng = np.random.default_rng(seed)

    # Phase 1: LP optimization from multiple initializations
    n = n_parent  # Use parent length (192)
    inits = _initializations(parent, seed, n)

    # Allocate time for each initialization
    phase1_end = t0 + min(budget * 0.35, n * 2.0 + 60)
    per_init_time = max(5.0, (phase1_end - t0) / len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_end - 5:
            break
        alloc_end = min(t0 + idx * per_init_time + per_init_time + 10, phase1_end)
        try:
            result = _lp_optimize(init, stop_time=alloc_end)
            sc = _exact_score(result)
            if np.isfinite(sc) and sc < best_score:
                best_seq = result.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 2: Extended LP polishing from best
    phase2_end = t0 + min(budget * 0.55, 300)
    if time.monotonic() < phase2_end - 5:
        try:
            polished = _polish_lp(best_seq, stop_time=phase2_end)
            sc = _exact_score(polished)
            if np.isfinite(sc) and sc < best_score:
                best_seq = polished.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 3: Local search
    phase3_end = t0 + min(budget * 0.75, 500)
    if time.monotonic() < phase3_end - 5:
        try:
            result, _ = _local_search(best_seq, stop_time=phase3_end)
            sc = _exact_score(result)
            if np.isfinite(sc) and sc < best_score:
                best_seq = result.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 4: More LP polishing on the result
    phase4_end = t0 + min(budget * 0.88, 650)
    if time.monotonic() < phase4_end - 10:
        try:
            polished = _polish_lp(best_seq, stop_time=phase4_end)
            sc = _exact_score(polished)
            if np.isfinite(sc) and sc < best_score:
                best_seq = polished.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 5: Try nearby lengths
    if time.monotonic() < deadline - 30:
        for n_try in [n - 2, n - 1, n + 1, n + 2]:
            if time.monotonic() > deadline - 15:
                break
            if n_try < 4:
                continue
            # Resample parent to new length
            idxs = (np.arange(n_try) * n_parent // n_try).astype(int)
            init_new = np.clip(parent[idxs], 0.0, 1000.0).astype(np.float64)
            init_new = init_new / init_new.sum()
            try:
                result = _lp_optimize(init_new, stop_time=time.monotonic() + 15)
                sc = _exact_score(result)
                if np.isfinite(sc) and sc < best_score:
                    best_seq = result.copy()
                    best_score = sc
            except Exception:
                pass

    # Final validation
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_score = float('inf')

    if not np.isfinite(final_score):
        # Fallback
        return [1.0] * 32

    return [float(x) for x in best_seq]
