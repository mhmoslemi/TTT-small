from scipy.optimize import linprog

def _norm(q):
    q = np.asarray(q, dtype=np.float64)
    q = np.where(np.isfinite(q) & (q > 0.0), q, 0.0)
    s = q.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.full(q.shape, 1.0 / max(1, q.shape[0]))
    q /= s
    return np.clip(q, 0.0, 1000.0)

def _sc(q):
    n = q.size
    if n == 0:
        return float('inf')
    b = np.convolve(q, q)
    v = 2.0 * n * float(b.max())
    return v if np.isfinite(v) else float('inf')

def _build_A(q, k):
    A = np.zeros((2 * k - 1, k))
    for r in range(2 * k - 1):
        lo = max(0, r - k + 1)
        hi = min(r, k - 1)
        if lo <= hi:
            A[r, lo:hi+1] = q[r - hi : r - lo + 1][::-1]
    return A

    M = mscale * float(b.max())
