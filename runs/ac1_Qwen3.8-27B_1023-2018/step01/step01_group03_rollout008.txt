import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def _convolve_score(q):
    """Compute 2n*max(conv(q,q))/(sum(q))^2"""
    n = q.size
    b = np.convolve(q, q)
    s = q.sum()
    return 2.0 * n * b.max() / (s * s)

def _build_A(q, n):
    """Build convolution matrix: A@g = convolve(q, g) (full, length 2n-1)"""
    m = 2 * n - 1
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    
    idx = 0
    for j in range(n):
        r0 = j
        r1 = min(j + n, m)
        L = r1 - r0
        rows[idx:idx+L] = range(r0, r1)
        cols[idx:idx+L] = j
        vals[idx:idx+L] = q[:L]
        idx += L
    
    # Actual number of nonzeros
    nnz = idx
    return coo_matrix((vals[:nnz], (rows[:nnz], cols[:nnz])), shape=(m, n)).tocsr()

def _lp_find_g(q, n, M_mult=1.0):
    """LP: max sum(g) s.t. conv(q,g)_k <= M, g >= 0. Returns normalized g or None."""
    b = np.convolve(q, q)
    M = M_mult * b.max()
    if M <= 0 or not np.isfinite(M):
        return None
    
    A = _build_A(q, n)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    except Exception:
        return None
    
    if res is None or not res.success:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0, None)
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s

def _gradient_step(q, n, step_size=0.02):
    """Subgradient step reducing max(conv(q,q))."""
    b = np.convolve(q, q)
    # Use top few peaks for a smoother gradient
    m = b.max()
    threshold = 0.8 * m
    active = np.where(b >= threshold)[0]
    
    grad = np.zeros(n, dtype=np.float64)
    wts = b[active] - threshold
    wts = wts / (wts.sum() + 1e-30)
    
    for kk, ww in zip(active, wts):
        lo = max(0, kk - n + 1)
        hi = min(kk, n - 1)
        # grad[j] += 2 * ww * q[kk - j] for j in [lo, hi]
        # q[kk-j] for j in [lo, hi] is q[kk-hi : kk-lo+1][::-1]
        seg = q[kk-hi : kk-lo+1][::-1]
        grad[lo:hi+1] += 2.0 * ww * seg
    
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return q.copy()
    
    for step in [step_size, step_size*0.5, step_size*2]:
        cand = q - step * grad / gn
        cand = np.clip(cand, 0, None)
        s = cand.sum()
        if s < 1e-15:
            continue
        cand = cand / s
        sc = _convolve_score(cand)
        if np.isfinite(sc) and sc < _convolve_score(q):
            return cand
    return q.copy()

            s = arr.sum()
            if s < 0.01:

            lst = [float(x) for x in arr]
            sc = float(evaluate_sequence(lst))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_list = lst
        except Exception:
            pass
    
    # --- Initialization ---
    # Use parent if available
    n = 512
    if 'height_sequence_1' in dir() or 'height_sequence_1' in globals():
        try:

            if hs1.size == n:

            else:

    else:

    if s < 1e-15:

    else:

    consider(q)
    
    cur_score = _convolve_score(q)

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def _score(q):
    """Compute 2*n*max(conv(q,q))/(sum(q))^2 for normalized q (sum=1)."""
    n = q.size
    b = np.convolve(q, q)
    s = q.sum()
    if s < 1e-15:
        return float('inf')
    val = 2.0 * n * b.max() / (s * s)
    return val if np.isfinite(val) else float('inf')


def _build_A(q, n):
    """Build sparse matrix A (2n-1 x n) such that A@g = full_convolve(q, g)."""
    m = 2 * n - 1
    # Column j: entries q[0], q[1], ..., q[n-1] in rows j, j+1, ..., j+n-1 (clamped)
    total_nnz = n * (n + 1)  # upper bound, will trim
    rows = np.concatenate([np.arange(j, min(j + n, m)) for j in range(n)])
    cols = np.concatenate([np.full(min(j + n, m) - j, j) for j in range(n)])
    vals = np.concatenate([q[:min(n, m - j)] for j in range(n)])
    return coo_matrix((vals, (rows, cols)), shape=(m, n)).tocsr()


def _lp_step(q, n, M_mult=1.0):
    """LP: max sum(g) s.t. (q*g)_k <= M, g >= 0. Returns normalized g or None."""
    b = np.convolve(q, q)
    M = M_mult * float(b.max())
    if M <= 0 or not np.isfinite(M):
        return None
    A = _build_A(q, n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    except Exception:
        return None
    if res is None or not getattr(res, 'success', False):
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0, None)
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s


def _grad_step(q, n, step=0.02, top_k=8):

        return q.copy()
    # Pick top active lags

    if active.size < 2:
        active = np.argsort(b)[::-1][:top_k]
    else:
        active = active[np.argsort(b[active])[::-1]][:top_k]
    wts = b[active] - 0.7 * m
    wts = wts / (wts.sum() + 1e-30)
    grad = np.zeros(n, dtype=np.float64)
    for kk, ww in zip(active, wts):
        lo = max(0, kk - n + 1)
        hi = min(kk, n - 1)
        seg = q[kk - hi : kk - lo + 1][::-1]
        grad[lo : hi + 1] += 2.0 * float(ww) * seg
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return q.copy()
    base_sc = _score(q)
    for st in [step, step * 0.5, step * 2.0, step * 0.25]:
        cand = q - st * grad / gn
        cand = np.clip(cand, 0, None)
        s = cand.sum()
        if s < 1e-15:
            continue
        cand = cand / s
        sc = _score(cand)
        if sc < base_sc - 1e-12:
            return cand
    return q.copy()
