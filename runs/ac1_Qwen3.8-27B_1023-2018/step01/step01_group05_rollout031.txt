    """Find g maximizing sum(g) s.t. f*g <= M (elementwise), g >= 0."""
    n = len(f)

    # A_ub[k, j] = f[k-j] if 0 <= k-j < n
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_lo = max(0, k-n+1)
        j_hi = min(n-1, k)
