import time
from scipy.optimize import linprog

def _norm_sum1(a):
    """Normalize to sum 1, nonnegative."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return np.full(a.shape, 1.0/a.size)
    return a / s

def _score_fast(a):
    """Fast score computation (scale-invariant)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    b = np.convolve(a, a)
    return 2.0 * n * b.max() / (a.sum()**2)

def _score_norm1(a):
    """Score assuming sum(a)=1."""
    n = a.size
    b = np.convolve(a, a)
    return 2.0 * n * b.max()

def _build_conv_matrix(a, n):
    """Build the (2n-1) x n matrix A where row k has A[k,j] = a[k-j] if valid."""
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # A[k, j] = a[k-j] for j in [j_start, j_end]
            vals = a[k - j_end : k - j_start + 1][::-1]
            A[k, j_start:j_end+1] = vals
    return A

def _lp_step(a, M):
    """Solve LP: max sum(g) s.t. A(a)*g <= M, g >= 0."""
    n = a.size
    A = _build_conv_matrix(a, n)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                      bounds=[(0, 1e6)]*n, method='highs',
                      options={'time_limit': 5.0})
        if res.success:
            g = np.clip(res.x, 0, 1e6)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except:
        pass
    return None

def _subgrad_step(a, t, K=5, beta=10.0):
    """One subgradient step on the simplex."""
    n = a.size
    b = np.convolve(a, a)
    max_b = b.max()
    if max_b < 1e-15:
        return a
    
    # Top-K active lags
    top_k = np.argsort(b)[::-1][:K]
    w = np.exp(beta * (b[top_k] - max_b))
    w /= w.sum()
    
    # Compute weighted subgradient
    grad = np.zeros(n)
    for kk, ww in zip(top_k, w):
        kk = int(kk)
        for i in range(n):
            v = 0.0
            if 0 <= kk - i < n:
                v += a[kk - i]
            if 0 <= i - kk < n:
                v += a[i - kk]
            grad[i] += ww * v
    
    # Projected subgradient step
    grad_p = grad - grad.mean()
    a_new = a - t * grad_p
    # Project onto simplex
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new /= s
    return a_new

So the Hessian H has H_{ij} = 2 if i+j = k, 0 otherwise. This is a Hankel matrix. Is it PSD?

For a vector v, v^T H v = 2 * sum_{i+j=k} v_i v_j = 2 * (v * v)_k (the k-th auto-correlation of v). This is NOT always non-negative! For example, if v has alternating signs, (v*v)_k can be negative.

But we're restricted to a >= 0, so v >= 0, and (v*v)_k >= 0. So on the nonnegative orthant, each b_k is convex. And max of convex functions is convex. And log-sum-exp of convex functions is convex.

So f_ε(a) = (1/ε) * log(sum_k exp(ε * b_k)) is convex on a >= 0! And as ε → ∞, f_ε → max(b) = max_k b_k.

This means we can use convex optimization (e.g., gradient descent, Newton's method) on f_ε!

The gradient: ∇f_ε = (1/sum_k exp(ε*b_k)) * sum_k exp(ε*b_k) * ∇b_k = sum_k p_k * ∇b_k where p_k = exp(ε*b_k) / sum_j exp(ε*b_j).

This is exactly the softmax-weighted subgradient I was considering before! With ε large, it concentrates on the argmax.

And the Hessian: H_f_ε = ε * (sum_k p_k * H_{b_k} - sum_k (dp_k * ...)) 

Actually, since f_ε is convex, the Hessian is PSD. We could use Newton's method, but computing the Hessian is expensive.

I think the gradient-based approach (softmax-weighted subgradient) is the right one, combined with the LP for making larger jumps.

But it's not strictly convex, and the max function is not differentiable at points where multiple b_k achieve the maximum. Still, it's convex.

For convex optimization over the simplex, the Frank-Wolfe (conditional gradient) method works well:
1. Compute subgradient g at current point a
2. Solve min g^T s s.t. s in simplex → s = e_{argmin g}
3. a <- a + t * (s - a) for t in [0, 1]

Or the inertia Bregman Frank-Wolfe for faster convergence.

But the LP approach is essentially a richer version of Frank-Wolfe: instead of just moving towards one coordinate, we move towards the LP-optimal g.

I think the best approach is:
1. LP iterations for large moves (find the "best" direction)
2. Softmax subgradient for fine polishing
3. Frank-Wolfe as a fallback when LP stalls

import numpy as np
import time
from scipy.optimize import linprog


def _simproj(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    r = np.arange(1, n + 1, dtype=np.float64)
    cond = mu > (cv - s) / r
    if not np.any(cond):
        return np.full(n, s / n, dtype=np.float64)
    rho = int(np.max(np.where(cond)[0])) + 1
    theta = (cv[rho - 1] - s) / rho
    return np.clip(v - theta, 0.0, None)


def _score(a):
    """2*n*max(conv(a,a))/(sum(a))^2"""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 1e18
    b = np.convolve(a, a)
    sb = float(b.max())
    sa = float(a.sum())
    if sa < 1e-12:
        return 1e18
    return 2.0 * n * sb / (sa * sa)


def _build_A(a):
    n = a.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je + 1] = a[k - je:k - js + 1][::-1]
    return A


def _lp_step(a, M, tl=3.0):
    n = a.size
    A = _build_A(a)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, 1e6)] * n,
                      method='highs',
                      options={'time_limit': tl})
        if res is not None and getattr(res, 'success', False):
            g = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, 1e6)
            sg = g.sum()
            if sg > 1e-10:
                return g / sg
    except Exception:
        pass
    return None


def _sg_step(a, t, K=8, beta=50.0):
    """One softmax-weighted subgradient step on the simplex."""
    n = a.size
    b = np.convolve(a, a)
    mb = float(b.max())
    if mb < 1e-15:
        return a
    idx = np.argsort(b)[::-1][:min(K, b.size)]
    w = np.exp(beta * (b[idx] - mb))
    w /= w.sum()
    grad = np.zeros(n, dtype=np.float64)
    for kk, ww in zip(idx, w):
        kk = int(kk)
        istart = max(0, kk - n + 1)
        iend = min(n - 1, kk)
        if istart <= iend:
            jstart = kk - iend
            jend = kk - istart
            grad[istart:iend + 1] += 2.0 * ww * a[jstart:jend + 1][::-1]
    a_new = a - t * grad
    return _simproj(a_new, 1.0)


def _optimize(a0, deadline, n_lp=200, n_sg=2000):
    """Combined LP + subgradient optimization."""
    a = _simproj(a0, 1.0)
    best_a = a.copy()
    best_sc = _score(a)
    lp_stall = 0
    sg_stall = 0

            else:

def _upscale(a, k):
    a = np.asarray(a, dtype=np.float64)
    if a.size == k:
        return _simproj(a, 1.0)
    if a.size == 0:
        return np.full(k, 1.0 / k)
    if k % a.size == 0:
        out = np.repeat(a, k // a.size)
    else:
        x = np.linspace(0.0, 1.0, a.size, dtype=np.float64)
        y = np.linspace(0.0, 1.0, k, dtype=np.float64)
        out = np.interp(y, x, a)
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s < 1e-12:
        return np.full(k, 1.0 / k)
    return out / s

import numpy as np
import time
from scipy.optimize import linprog


def _simproj(v, s=1.0):
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    r = np.arange(1, n + 1, dtype=np.float64)
    cond = mu > (cv - s) / r
    if not np.any(cond):
        return np.full(n, s / n, dtype=np.float64)
    rho = int(np.max(np.where(cond)[0])) + 1
    theta = (cv[rho - 1] - s) / rho
    return np.clip(v - theta, 0.0, None)


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 1e18
    b = np.convolve(a, a)
    sb = float(b.max())
    sa = float(a.sum())
    if sa < 1e-12:
        return 1e18
    return 2.0 * n * sb / (sa * sa)


def _build_A(a):
    n = a.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je + 1] = a[k - je:k - js + 1][::-1]
    return A


def _lp_step(a, M, tl=3.0):
    n = a.size
    if n == 0:
        return None
    A = _build_A(a)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, 1e6)] * n,
                      method='highs',
                      options={'time_limit': max(0.5, tl), 'presolve': True})
        if res is not None and getattr(res, 'success', False):
            g = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, 1e6)
            sg = float(g.sum())
            if sg > 1e-10:
                return g / sg
    except Exception:
        pass
    return None


def _softmax_grad(a, K=8, beta=50.0):
    n = a.size
    b = np.convolve(a, a)
    mb = float(b.max())
    if mb < 1e-15:
        return np.zeros(n), b
    idx = np.argsort(b)[::-1][:min(K, b.size)]
    w = np.exp(beta * (b[idx] - mb))
    ws = float(w.sum())
    if ws < 1e-30:
        return np.zeros(n), b
    w /= ws
    grad = np.zeros(n, dtype=np.float64)
    for kk, ww in zip(idx, w):
        kk = int(kk)
        istart = max(0, kk - n + 1)
        iend = min(n - 1, kk)
        if istart <= iend:
            jstart = kk - iend
            jend = kk - istart
            grad[istart:iend + 1] += 2.0 * ww * a[jstart:jend + 1][::-1]
    return grad, b


def _lp_optimize(a0, deadline, max_lp=300):
    a = _simproj(a0, 1.0)
    best_a = a.copy()
    best_sc = _score(a)
    stall = 0
    t_start = 0.15

    for it in range(max_lp):
        if time.monotonic() > deadline - 5.0:
            break
        b = np.convolve(a, a)
        M = float(b.max())
        if M < 1e-15:

        remaining = max(0.5, deadline - time.monotonic() - 5.0)
        g = _lp_step(a, M, tl=min(4.0, remaining))
        if g is None:

            if stall > 8:

            continue
        g = _simproj(g, 1.0)
        gdiff = float(np.linalg.norm(g - a))
        if gdiff < 1e-6:
            stall += 2
            if stall > 12:

        best_t = 0.0
        best_sc_t = best_sc
        best_cand = best_a.copy()

            sc = best_sc_t

def _upscale(a, k):
    a = np.asarray(a, dtype=np.float64)
    if a.size == k:
        return _simproj(a, 1.0)
    if a.size == 0:
        return np.full(k, 1.0 / k, dtype=np.float64)
    if k % a.size == 0:
        out = np.repeat(a, k // a.size)
    else:
        x = np.linspace(0.0, 1.0, a.size, dtype=np.float64)
        y = np.linspace(0.0, 1.0, k, dtype=np.float64)
        out = np.interp(y, x, a)
    out = np.clip(out, 0.0, None)
    s = float(out.sum())
    if s < 1e-12:
        return np.full(k, 1.0 / k, dtype=np.float64)
    return out / s
