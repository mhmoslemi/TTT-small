∂f/∂z_j = a_j * (g_j - a·g) where g = ∇_a f

1. a is automatically in the simplex (no projection needed)
2. The step naturally respects the simplex geometry
3. It's equivalent to a multiplicative update (like the "Hedge" algorithm)

For the step: z ← z - lr * ∂f/∂z

And the corresponding update to a is:
a_new[j] ∝ a[j] * exp(-lr * (g_j - a·g))

This is a multiplicative weight update!

```python
def _softmax_step(z, a, g, lr):
    """Take one softmax gradient step."""
    adv = g - np.dot(a, g)  # advantage
    z_new = z - lr * adv
    # Renormalize for numerical stability
    z_new -= z_new.max()
    a_new = np.exp(z_new)
    a_new /= a_new.sum()
    return z_new, a_new
```

This is clean and efficient. The "advantage" g - a·g is the gradient projected to the tangent space, and the softmax step is the natural gradient on the simplex.

```python
import numpy as np
import time

def _proj_simplex(v):
    n = v.size
    if n == 0:
        return v
    if n == 1:
        return np.ones(1)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find rho
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def _score(a):
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _grad_score_soft(a, beta):
    """Compute (score, gradient) with soft-max approximation."""
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return np.inf, np.zeros(n)
    c = np.convolve(a, a)
    
    if beta <= 1e-6:
        # Exact subgradient
        k_max = int(c.argmax())
        M = c[k_max]
        # grad_M[j] = 2*a[k_max - j] for valid j
        grad_M = np.zeros(n)
        if k_max < n:
            grad_M[:k_max+1] = 2.0 * a[k_max::-1]
        else:
            m = k_max - n + 1
            grad_M[m:] = 2.0 * a[n-1:m-1:-1]
    else:
        # Soft gradient
        z = beta * c
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        p = exp_z / exp_z.sum()
        M = c.max() + np.log(exp_z.sum()) / beta
        
        # grad_M = 2 * cross_correlate(p, a)
        # Using the formula: grad_M[j] = 2 * np.correlate(p, a, 'full')[j + n - 1]
        cross = np.correlate(p, a, 'full')
        grad_M = 2.0 * cross[n-1:n-1+n]
    
    score = 2.0 * n * M / (s * s)
    # d(score)/da = 2n/s^3 * (s * grad_M - 2M * 1)
    grad = 2.0 * n / (s**3) * (s * grad_M - 2.0 * M)
    return score, grad

def _softmax(a):
    """Log of a, with floor."""
    return np.log(np.maximum(a, 1e-30))

def _from_logit(z):
    """Softmax from logit."""
    z = z - z.max()
    a = np.exp(z)
    s = a.sum()
    if s < 1e-30:
        return np.ones(a.size) / a.size
    return a / s

def _main_optimize(a0, n, rng, t_start, t_end):
    """Main optimization using softmax parameterization with Adam."""
    z = _softmax(a0.copy())
    a = a0.copy()
    best_a = a0.copy()
    best_score = _score(a0)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 1.0 / n * 5  # initial learning rate
    beta_start = 100.0
    beta_end = 50000.0
    
    no_improve = 0
    patience = 500
    
    while time.monotonic() < t_end:
        t += 1
        # Adaptive beta
        frac = min(1.0, (time.monotonic() - t_start) / max(1e-3, t_end - t_start))
        beta = beta_start * (beta_end / beta_start) ** frac  # exponential increase
        
        # Compute score and gradient
        score, grad = _grad_score_soft(a, beta)
        
        if not np.isfinite(score):
            z = _softmax(best_a)
            a = _from_logit(z)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        
        # Convert to z-gradient
        # d(score)/dz_j = a_j * (grad_j - sum(a * grad))
        adv = grad - np.dot(a, grad)
        zgrad = a * adv
        
        # Adam update
        t += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1 - b1) * zgrad
        v = b2 * v + (1 - b2) * zgrad**2
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Clip step
        max_step = 2.0
        step = np.clip(step, -max_step, max_step)
        
        z = z - step
        z = np.where(np.isfinite(z), z, 0.0)
        a = _from_logit(z)
        
        # Check improvement
        cur_score = _score(a)
        if np.isfinite(cur_score) and cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > patience:
            lr *= 0.7
            no_improve = 0
            if rng.random() < 0.3:
                # Perturb
                z = _softmax(best_a) + 0.1 * rng.standard_normal(n)
                a = _from_logit(z)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
        
        # Periodic: try random polish
        if t % 2000 == 0:

                a = _from_logit(z)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
    
    return best_a, best_score
```
