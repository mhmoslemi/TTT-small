def _conv(a):
    """Full auto-convolution."""

        c = np.convolve(a, a, mode='full')
    else:
        fsize = next_fast_len(2*n-1)
        F = rfft(a, fsize)
        c = irfft(F*F, fsize)[:2*n-1]
    return c

def _score(a):
    """Compute the AC1 score."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv(a)
    m = float(b.max())
    return 2.0 * n * m / (s*s)

def _lp_direction(f, n, M_scale=1.0):
    """
    Solve the LP to find the best direction.
    Returns (g_normalized, M_used) or (None, M).
    """
    M = M_scale * float(np.max(_conv(f)))
    if M <= 1e-15:
        return None, M
    
    # Build A: (2n-1) x n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    
    # LP: maximize sum(g) s.t. A@g <= M, g >= 0
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)]*n, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
    except Exception:
        return None, M
    
    if res.status != 0 or res.x is None:
        return None, M
    
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum <= 1e-15:
        return None, M
    
    return g / g_sum, M

            f = np.clip(f, 0, None)

                f = f / fs
            continue

            f_new = np.clip(f_new, 0, None)

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(192, dtype=np.float64)
    if p.size == 0:
        p = np.ones(192, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    p = np.where(np.isfinite(p), p, 0.0)
    if p.size == 0:
        p = np.ones(192, dtype=np.float64)
    return p


def _conv_full(a):
    """Full auto-convolution of a."""
    n = len(a)
    if n <= 256:
        return np.convolve(a, a, mode='full')
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    c[c < 0] = 0
    return c


def _score(a):
    """Compute AC1 score."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _conv_full(a)
    m = float(b.max())
    if not math.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    """Build the (2n-1) x n constraint matrix for the LP."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, n):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns (g_normalized_to_sum_1, M) or (None, M).
    """
    b = _conv_full(f)
    M = float(b.max())
    if M < 1e-15:
        return None, M
    
    A = _build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
    except Exception:
        return None, M
    
    if res.status != 0 or res.x is None:
        return None, M
    
    g = np.asarray(res.x, dtype=np.float64)
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return None, M
    
    return g / g_sum, M


def _lp_optimize(f0, n, stop_time, seed):
