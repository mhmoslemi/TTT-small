    if linprog is None:
        return None
    q = np.asarray(q, dtype=np.float64)
    b = np.convolve(q, q)
    M = float(k) * float(b.max())
    if not np.isfinite(M) or M <= 0:
        return None

    # A[j, i] = q[j-i] if 0 <= j-i < k
    A = np.zeros((2*k-1, k), dtype=np.float64)

        lo = max(0, j-k+1)
        hi = min(k-1, j)
        for i in range(lo, hi+1):

    """Subgradient descent step on max(conv(q,q)) with sum-normalization."""
    q = _normalize(q)
    b = np.convolve(q, q)
    m = float(b.max())
    if m <= 0 or not np.isfinite(m):
        return q.copy()

    # Select top peaks (weighted)
    threshold = 0.7 * m
    act = np.where(b >= threshold)[0]
    if act.size == 0:
        act = np.array([int(np.argmax(b))])
    if act.size > n_peaks:
        idx = np.argsort(b[act])[::-1][:n_peaks]
        act = act[idx]

    # Weight peaks
    w = np.exp((b[act] - m) * 4.0)
    ws = float(w.sum())
    if ws <= 0 or not np.isfinite(ws):
        w = np.ones(act.size)
    else:
        w = w / ws

    # Compute subgradient
    grad = np.zeros(k, dtype=np.float64)
    for j, ww in zip(act, w):
        j = int(j)
        lo = max(0, j - k + 1)
        hi = min(k - 1, j)
        # grad[lo:hi+1] += 2*ww * q[j-hi : j-lo+1][::-1]
        seg = q[lo:hi + 1][::-1]  # q[j-lo], q[j-lo-1], ..., q[j-hi]
        # Wait: for i in [lo, hi], grad[i] += 2*ww*q[j-i]
        # j-i ranges from j-lo down to j-hi
        # So seg = q[j-hi : j-lo+1][::-1]
        seg = q[j - hi : j - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * float(ww) * seg

    # Project gradient: subtract mean (for sum constraint)
    grad -= grad.mean()
    gn = float(np.linalg.norm(grad))
    if gn < 1e-15:
        return q.copy()
    grad /= gn

    cand = q - alpha * grad
    return _normalize(cand)


def _local_polish(q, k, rng, max_iters, deadline):
    """Iterative gradient/subgradient polish."""
    q = _normalize(q)
    best = q.copy()
    best_sc = _score_fast(q)
    step = 0.08
    stuck = 0

    for it in range(max(1, int(max_iters))):
        if time.monotonic() > deadline - 5.0:
            break

        # Subgradient step
        cand = _gradient_step(q, k, step, n_peaks=10)
        sc = _score_fast(cand)
        if sc < best_sc - 1e-11:
            best = cand.copy()
            best_sc = sc
        if sc < _score_fast(q) - 1e-11:
            q = cand
            step = min(0.35, step * 1.03)
            stuck = 0
        else:
            step = max(0.005, step * 0.85)
            stuck += 1

        # Random perturbation if stuck
        if stuck >= 3:
            r = rng.normal(size=k)
            r -= r.mean()
            rn = float(np.linalg.norm(r))
            if rn > 1e-15:
                cand2 = _normalize(q + (step * 2.0 / np.sqrt(k)) * (r / rn))
                sc2 = _score_fast(cand2)
                if sc2 < best_sc - 1e-11:
                    best = cand2.copy()
                    best_sc = sc2
                    q = cand2
                stuck = max(0, stuck - 1)

        if step <= 0.006 and stuck >= 10:
            break

    return best, best_sc


def _lp_descent(q0, k, deadline, max_iters=200):
    """LP-based descent with interpolation."""
    q = _normalize(q0)
    sc = _score_fast(q)
    best_q = q.copy()
    best_sc = sc
    stagnation = 0

    for it in range(max(1, int(max_iters))):
        if time.monotonic() > deadline - 10.0:
            break

        g = _lp_response(q, k, time_limit=min(30.0, max(5.0, deadline - time.monotonic() - 15.0)))
        if g is None:
            break
        g = _normalize(g)

        # Try multiple interpolation parameters
        best_cand = q
        best_csc = sc
        for t in (0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.7, 1.0):
            cand = (1.0 - t) * q + t * g
            cand = _normalize(cand)
            cs = _score_fast(cand)
            if cs < best_csc - 1e-12:
                best_csc = cs
                best_cand = cand

        if best_csc < sc - 1e-10:
            q = best_cand
            sc = best_csc
            if sc < best_sc:
                best_sc = sc
                best_q = q.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation >= 8:
                break

    return best_q, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = max(10.0, float(budget_s))
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0  # 15s headroom

    best_list = None
    best_score = float('inf')

    def consider(seq):
        nonlocal best_list, best_score
        try:
            arr = np.asarray(seq, dtype=np.float64)
            if arr.ndim != 1 or arr.size == 0:
                return
            arr = np.where(np.isfinite(arr) & (arr >= 0.0), arr, 0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            s = float(arr.sum())
            if s < 0.01 or not np.isfinite(s):
                return
            lst = [float(x) for x in arr]
            val = float(evaluate_sequence(lst))
            if np.isfinite(val) and val < best_score:
                best_score = val
                best_list = lst
        except Exception:
            pass

    # Use parent if available
    k = 512
    q_parent = None
    try:
        hs = globals().get('height_sequence_1', None)
        if hs is not None:
            q_parent = _normalize(np.asarray(hs, dtype=np.float64))
            consider(q_parent)
    except Exception:
        pass

    # Fallback seed: asymmetric tile
    if q_parent is None:
        q4 = np.array([0.2, 0.4, 0.2, 0.2], dtype=np.float64)
        q_parent = np.repeat(q4, k // 4)[:k]
        consider(_normalize(q_parent))

    # Phase 1: LP descent from parent
    if time.monotonic() < deadline - 60.0:
        lp_deadline = min(deadline - 120.0, t_start + 500.0)
        q_lp, sc_lp = _lp_descent(q_parent, k, lp_deadline, max_iters=100)
        if sc_lp < _score_fast(q_parent):
            q_parent = q_lp
        consider(q_parent)

    # Phase 2: Gradient polish
    if time.monotonic() < deadline - 30.0:
        pol_deadline = min(deadline - 20.0, t_start + 800.0)
        n_pol_iters = max(50, min(5000, int((pol_deadline - time.monotonic()) * 20)))
        q_pol, sc_pol = _local_polish(q_parent, k, rng, n_pol_iters, pol_deadline)
        if sc_pol < _score_fast(q_parent):
            q_parent = q_pol
        consider(q_parent)

    # Phase 3: More LP + polish cycles
    if time.monotonic() < deadline - 40.0:
        for cycle in range(5):
            if time.monotonic() > deadline - 25.0:
                break
            # Small LP steps
            if time.monotonic() < deadline - 30.0:
                q_lp2, sc_lp2 = _lp_descent(q_parent, k, deadline - 40.0, max_iters=15)
                if sc_lp2 < _score_fast(q_parent):
                    q_parent = q_lp2
                    consider(q_parent)
            # Quick polish
            if time.monotonic() < deadline - 15.0:
                q_p2, sc_p2 = _local_polish(q_parent, k, rng, 100, deadline - 10.0)
                if sc_p2 < _score_fast(q_parent):
                    q_parent = q_p2
                    consider(q_parent)

    # Phase 4: Try a random restart if time allows
    if time.monotonic() < deadline - 40.0:
        q_rand = rng.dirichlet(np.full(k, 2.0))
        for _ in range(5):
            if time.monotonic() > deadline - 20.0:
                break
            q_r, sc_r = _local_polish(q_rand, k, rng, 200, deadline - 10.0)
            if sc_r < _score_fast(q_parent):
                q_parent = q_r
            consider(q_parent)
            if q_parent is not q_r:
                break

    # Phase 5: Final fine polish
    if time.monotonic() < deadline - 10.0:
        q_final, sc_final = _local_polish(q_parent, k, rng, 200, deadline - 5.0)
        if sc_final < _score_fast(q_parent):
            q_parent = q_final
        consider(q_parent)

    # Final validation
    if best_list is None:
        q4 = np.array([0.2, 0.4, 0.2, 0.2], dtype=np.float64)
        consider(_normalize(np.repeat(q4, k // 4)[:k]))

    if best_list is None:
        best_list = [1.0 / k] * k

    # Validate with authoritative evaluator
    try:
        val = float(evaluate_sequence(best_list))
        if not np.isfinite(val) or val > 2.0:
            q4 = np.array([0.2, 0.4, 0.2, 0.2], dtype=np.float64)
            fb = [float(x) for x in _normalize(np.repeat(q4, k // 4)[:k])]
            if evaluate_sequence(fb) < float('inf'):
                best_list = fb
    except Exception:
        best_list = [1.0 / k] * k

    return [float(x) for x in best_list]
