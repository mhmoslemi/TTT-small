def _score_arr(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    mx = float(np.max(c))
    return 2.0 * n * mx / (s * s)


def _softmax(z):
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z, dtype=np.float64) / len(z)
    return e / s


def _gradient_smooth(a, c, beta):
    """Compute gradient of smoothed objective 2n*sum_k w[k]*C(k) in z-domain.
    a = softmax(z), sum(a)=1. Returns gradient w.r.t. z."""
    n = len(a)
    M = float(np.max(c))
    zc = beta * (c - M)
    w = np.exp(zc)
    ws = float(np.sum(w))
    if ws < 1e-300:
        idx = int(np.argmax(c))
        w = np.zeros(len(c), dtype=np.float64)
        w[idx] = 1.0
    else:
        w = w / ws

    E_C = float(np.dot(w, c))

    # h[j] = sum_k w[k] * a[k-j] for valid k
    h = np.zeros(n, dtype=np.float64)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(k, n - 1) + 1
        ss = k - je + 1
        se = k - js + 1
        h[js:je] += wk * a[ss:se][::-1]

    g = 4.0 * n * a * (h - E_C)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=np.float64)
    return g

    return g
