```python
import numpy as np
import time
import math
from scipy.optimize import linprog
import scipy.sparse as sps

TARGET = 1.50274349

def _convolve(a, b=None):
    """Compute convolution, using FFT for large n."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    if n * m < 1000000:
        return np.convolve(a, b)
    else:
        N = 1 << (n + m - 1).bit_length()
        A = np.fft.rfft(a, N)
        B = np.fft.rfft(b, N)
        return np.fft.irfft(A * B, N)[:n+m-1]

def _score(a):
    n = len(a)
    b = _convolve(a)
    S = sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * max(b) / (S*S)

def _lp_step(a, M):
    """Solve the LP: max sum(g) s.t. convolve(a,g) <= M, g >= 0."""
    n = len(a)
    # Build the constraint matrix A such that A @ g <= M * ones
    # (A @ g)[k] = sum_i a[i] * g[k-i] = (a * g)[k] (cross-correlation-like)
    # Actually convolve(a, g)[k] = sum_i a[i] * g[k-i]
    # We need this <= M for all k = 0, ..., 2n-2
    
    # The matrix A has shape (2n-1, n)
    # A[k, j] = a[j] if 0 <= k-j < n, else 0
    # Wait, convolve(a, g)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * g[k-j]
    # = sum_{j: max(0,k-n+1) <= j <= min(k, n-1)} a[j] * g[k-j]
    # Let i = k-j, then j = k-i, and we need 0 <= i < n and 0 <= k-i < n
    # = sum_{i: max(0,k-n+1) <= k-i <= min(n-1, k)} a[k-i] * g[i]
    # So A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    
    # For sparse representation:
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                data.append(a[j])
    
    A = sps.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
```

Actually, let me reconsider the step size. The blend `a_new = (1-t)*a + t*g` where `g` is normalized to have the same sum as `a`. The score of `a_new` as a function of `t` is:

`score(t) = 2*n*max_k b_{a_new}[k] / S^2`

where `b_{a_new} = convolve((1-t)a + tg, (1-t)a + tg)` and `S = sum(a) = sum(g)` (same sum).

`b_{a_new}[k] = (1-t)^2 b_a[k] + 2t(1-t) b_{a,g}[k] + t^2 b_g[k]`

where `b_a = convolve(a,a)`, `b_{a,g} = convolve(a,g)`, `b_g = convolve(g,g)`.

So `score(t)` depends on `max_k [(1-t)^2 b_a[k] + 2t(1-t) b_{a,g}[k] + t^2 b_g[k]]`.

This is the max of quadratic functions in `t`, so it's convex in `t` (the max of convex functions is convex). Therefore, `score(t)` is convex in `t`, and the minimum over `t in [0,1]` can be found by ternary search or by checking the vertex of the active quadratic.

Since it's convex, we can use ternary search! Or even simpler: the minimum is either at `t=0`, `t=1`, or at the point where two quadratics cross (the kink point).

Actually, for a convex function on [0,1], the minimum is achieved at the unique point where the subgradient contains 0, or at an endpoint. We can find it by:
1. Evaluating at `t = 0` and `t = 1`
2. If `score(1) > score(0)`, the minimum is somewhere in between
3. Use ternary search to find it

But ternary search requires many evaluations, and each evaluation is O(n) (computing the max of a vector). So it's fast.

Let me implement this:

```python
def _best_blend(a, g, b_a, b_ag, n):

        b = (1-t)**2 * b_a + 2*t*(1-t) * b_ag + t**2 * b_g
```

```python
import numpy as np
import time
import math
from scipy.optimize import linprog
import scipy.sparse as sps

TARGET = 1.50274349

def _conv(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    if n * m <= 200000:
        return np.convolve(a, b)
    N = 1 << (n + m - 1).bit_length()
    return np.real(np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N))[:n+m-1]

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, None)
    n = a.size
    if n == 0:
        return float('inf')
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    b = _conv(a)
    Mb = float(b.max())
    return 2.0 * n * Mb / (S * S)

def _build_A(a):
    n = len(a)
    j = np.arange(n)
    cols = np.repeat(j, n)
    rows = (j[:, None] + np.arange(n)[None, :]).ravel()
    data = np.tile(a, n)
    return sps.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def _lp_solve(a, M):
    n = len(a)
    A = _build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)]*n,
                  method='highs', options={'presolve': True})
    if res.success and res.x is not None:
        g = np.clip(res.x, 0, None)
        S = float(g.sum())
        if S > 1e-15:
            return g / S
    return None

def _best_t(a, g, b_ag, n, S):
    b_a = _conv(a)
    b_g = _conv(g)
    best_sc = float('inf')
    best_t = 0.0
    for t in np.linspace(0, 1, 21):
        b = (1-t)**2 * b_a + 2*t*(1-t) * b_ag + t**2 * b_g
        sc = 2*n * float(b.max()) / (S*S)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    return best_t, best_sc

def _lp_iterate(a0, max_iters, deadline):
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 0, None)
    S = float(a.sum())
    if S < 0.01:
        return None
    a = a / S  # normalize to sum 1
    
    local_sc = _score(a)
    best_local = a.copy()
    best_local_sc = local_sc
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = _conv(a)
        M = float(b.max())
        if not math.isfinite(M) or M <= 0:
            break
        
        g = _lp_solve(a, M)
        if g is None:
            break
        
        # Check if g ≈ a (convergence)
        if np.linalg.norm(g - a) / max(1, np.linalg.norm(a)) < 1e-8:
            break
        
        b_ag = _conv(a, g)
        S = float(a.sum())  # should be 1
        t_best, sc_best = _best_t(a, g, b_ag, len(a), S)
        
        if t_best <= 0 or t_best >= 1:
            break
        
        a_new = (1 - t_best) * a + t_best * g
        a_new = np.clip(a_new, 0, None)
        sc_new = _score(a_new)
        
        if math.isfinite(sc_new) and sc_new < best_local_sc * (1 - 1e-10):
            a = a_new / a_new.sum()  # renormalize
            best_local_sc = sc_new
            best_local = a.copy()
        else:
            # Try smaller step
            for t in (0.1, 0.05, 0.02):
                a_try = (1-t) * a + t * g
                a_try = np.clip(a_try, 0, None)
                sc_try = _score(a_try)
                if math.isfinite(sc_try) and sc_try < best_local_sc * (1 - 1e-10):
                    a = a_try / a_try.sum()
                    best_local_sc = sc_try
                    best_local = a.copy()
                    break
    
    return best_local

def _make_inits(n, parent, rng):
    inits = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Ramps
    for p in [0.5, 1, 2, 3]:
        for dir_ in [1, -1]:
            a = np.abs(dir_ * (np.arange(n) - n/2) + n/4) ** p + 0.1
            a = np.clip(a, 0.01, None)
            inits.append(a / a.sum())
    
    # Exponentials
    for c in [0.5, 1, 2, 3, -0.5, -1, -2, -3]:
        a = np.exp(c * (x - 0.5))
        inits.append(a / a.sum())
    
    # Anti-symmetric perturbations
    half = n // 2
    for _ in range(5):
        v = np.tanh(0.5 * rng.standard_normal(half))
        f = np.ones(n)
        f[:half] = 1 + v
        f[half:] = 1 - v[::-1][:n-half]
        f = np.clip(f, 0.01, None)
        inits.append(f / f.sum())
    
    # Parent resampled
    if parent is not None and parent.size >= 2:
        p = np.interp(x, np.linspace(0, 1, parent.size), parent)
        p = np.clip(p, 0.01, None)
        inits.append(p / p.sum())
    
    # Score and sort
    scored = []
    for a in inits:
        sc = _score(a)
        if math.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda z: z[0])
    return [a for _, a in scored]
```

```python
import numpy as np

from scipy.optimize import linprog
import scipy.sparse as sps

def _conv(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    if n * m <= 200000:
        return np.convolve(a, b)
    N = 1 << (n + m - 1).bit_length()
    return np.real(np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N))[:n + m - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, None)
    n = a.size
    if n == 0:
        return float('inf')
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    b = _conv(a)
    Mb = float(b.max())
    if not math.isfinite(Mb):
        return float('inf')
    return 2.0 * n * Mb / (S * S)


def _build_A(a):
    n = len(a)
    j = np.arange(n)
    cols = np.repeat(j, n)
    rows = (j[:, None] + np.arange(n)[None, :]).ravel()
    data = np.tile(a, n)
    return sps.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_solve(a, M, deadline=None):
    n = len(a)
    if deadline is not None and time.monotonic() > deadline:
        return None
    A = _build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method='highs', options={'presolve': True})
    except Exception:
        return None
    if res is not None and res.success and res.x is not None:
        g = np.clip(res.x, 0, None)
        Sg = float(g.sum())
        if Sg > 1e-15:
            return g / Sg
    return None


def _blend_score(b_a, b_ag, b_g, t, n, S):
    b = (1 - t) ** 2 * b_a + 2 * t * (1 - t) * b_ag + t * t * b_g
    return 2 * n * float(b.max()) / (S * S)


def _find_best_t(b_a, b_ag, b_g, n, S):

def _make_inits(n, parent, rng):
    inits = []
    x = np.linspace(0, 1, n)
    half = n // 2

    def add(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0.01, None)
        if a.size == n:
            s = a.sum()
            if s > 0:
                inits.append(a / s)

    # Uniform
    add(np.ones(n))

    # Monotone ramps (key for asymmetry)
    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        add((np.arange(n, dtype=float) + 1) ** p)
        add((n - np.arange(n, dtype=float)) ** p)

    # Exponentials
    for c in [0.5, 1.0, 2.0, 3.0, 5.0, -0.5, -1.0, -2.0, -3.0]:
        add(np.exp(c * (x - 0.5)))

    # Anti-symmetric perturbations (break left-right symmetry)
    for _ in range(8):
        v = np.tanh(0.8 * rng.standard_normal(half))
        f = np.ones(n)
        f[:half] = 1.0 + v
        if n % 2 == 0:
            f[half:] = 1.0 - v[::-1]
        else:
            f[half:] = 1.0 - v[::-1][:n - half]
            f[half] = 1.0
        add(f)

    # Quadratic shapes
    for c in [0.5, 1.0, 2.0]:
        add((x - c * (x - 0.5)) ** 2 + 0.1)
        add(np.abs(x - 0.3) ** 2 + np.abs(x - 0.7) ** 2 + 0.1)

    # Parent resampled
    if parent is not None and parent.size >= 2:
        p = np.interp(x, np.linspace(0, 1, parent.size), parent)
        add(p)
        p2 = p * (1 + 0.1 * rng.standard_normal(n))
        add(p2)

    # Score and sort
    scored = []
    for a in inits:
        sc = _score(a)
        if math.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda z: z[0])
    return [a for _, a in scored]
```
