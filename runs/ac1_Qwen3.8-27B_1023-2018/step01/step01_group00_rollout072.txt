def _project_simplex(v):
 v=np.asarray(v,float).ravel().copy()
 n=v.size
 u=np.sort(v)[::-1]

 rho=np.where(u*np.arange(1,n+1) > (cssv-1.0))[0]
 if len(rho)==0: rho=np.array([0])
 rho=int(rho[-1])

import numpy as np
import math
import time
from scipy.fft import rfft, irfft, next_fast_len


def _norm(a):
    a = np.array(a, dtype=np.float64, copy=True).ravel()
    a[~np.isfinite(a)] = 0.0
    np.maximum(a, 0.0, out=a)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        if a.size == 0:
            a = np.ones(1, dtype=np.float64)
        else:
            a = np.full(a.size, 1.0 / a.size, dtype=np.float64)
        return a
    return a / s


def _sanitize_z(z):
    z = np.array(z, dtype=np.float64, copy=True).ravel()
    z[~np.isfinite(z)] = 0.0
    if z.size:
        z -= z.mean()
    return np.clip(z, -50.0, 50.0)


def _softmax_z(z):
    z = _sanitize_z(z)
    n = z.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    z = z - z.max()
    e = np.exp(np.clip(z, -700.0, 700.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return e / s


def _autoconv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 512:
        b = np.convolve(a, a, "full")
    else:
        N = next_fast_len(2 * n - 1)
        A = rfft(a, N)
        b = irfft(A * A, N)[:2 * n - 1]
    b = np.array(b, dtype=np.float64, copy=True).ravel()
    b[~np.isfinite(b)] = 0.0
    np.maximum(b, 0.0, out=b)
    return b


def _score_from_b(b, n):
    if b.size == 0:
        return math.inf
    m = float(np.max(b))
    if not np.isfinite(m) or m <= 0.0:
        return math.inf
    return float(2.0 * n * m)


def _grad_a(p, a):
    n = a.size
    if n <= 256:
        c = np.correlate(p, a, "valid")
    else:
        N = next_fast_len(p.size + n - 1)
        P = rfft(p, N)
        A = rfft(a, N)
        c = irfft(P * np.conjugate(A), N)[:n]
    c = np.array(c, dtype=np.float64, copy=True).ravel()
    if c.size != n:
        if c.size > n:
            c = c[:n]
        else:
            c = np.pad(c, (0, n - c.size), mode="constant")
    c[~np.isfinite(c)] = 0.0
    np.maximum(c, 0.0, out=c)
    return 2.0 * c


def _soft_logsumexp(b, tau):
    L = b.size
    b = np.asarray(b, dtype=np.float64).ravel()
    m = float(np.max(b)) if L else 0.0
    if not np.isfinite(m):
        m = 0.0
        b = np.array(b, dtype=np.float64, copy=True)
        b[~np.isfinite(b)] = 0.0
    if not np.isfinite(tau) or tau <= 1e-14:
        p = np.zeros(L, dtype=np.float64)
        tol = 1e-12 * max(1.0, abs(m))
        idx = np.where(np.abs(b - m) <= tol)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(b))])
        p[idx] = 1.0 / idx.size
        return float(m), p
    x = (b - m) / tau
    x = np.where(np.isfinite(x), x, -60.0)
    x = np.clip(x, -60.0, 60.0)
    e = np.exp(x)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(L, dtype=np.float64)
        idx = np.where(np.abs(b - m) <= 1e-12 * max(1.0, abs(m)))[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(b))])
        p[idx] = 1.0 / idx.size
        return float(m), p
    p = e / s
    U = m + tau * math.log(s)
    if not np.isfinite(U):
        U = m
    return float(U), p


def _project_simplex(v):
    v = np.array(v, dtype=np.float64, copy=True).ravel()
    n = v.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ns = np.arange(1, n + 1)
    cond = u * ns > (cssv - 1.0)
    idx = np.where(cond)[0]
    rho = int(idx[-1]) if idx.size else 0
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 1e-300:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w /= s
    return w

def _random_z(L, rng):
    if L <= 1:
        return np.zeros(L, dtype=np.float64)
    t = np.linspace(0.0, 1.0, L)
    z = np.zeros(L, dtype=np.float64)
    nterms = int(rng.integers(1, 5))
    for _ in range(nterms):
        f = int(rng.integers(1, 7))
        amp = float(rng.uniform(0.2, 1.2))
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        if rng.random() < 0.5:
            z += amp * np.sin(2.0 * math.pi * f * t + ph)
        else:
            z += amp * np.cos(2.0 * math.pi * f * t + ph)
    sigma = float(rng.uniform(0.02, 0.20))
    z += sigma * rng.standard_normal(L)
    z = np.convolve(z, np.ones(3, dtype=np.float64) / 3.0, mode="same")
    z = _sanitize_z(z)
    return z


def _make_pool(L, parent, seed, n_rand=4):
    L = int(max(1, L))
    rng = np.random.default_rng(seed)
    pool = []

    def add(a):
        b = _norm(a)
        if b.size == L and np.isfinite(b).all():
            pool.append(b.copy())

    if L == 1:
        add(np.ones(1, dtype=np.float64))
        return pool

    if parent is not None and parent.size > 1:
        xp = np.linspace(0.0, 1.0, parent.size)
        xn = np.linspace(0.0, 1.0, L)
        base = np.interp(xn, xp, parent)
        add(base)
        add(np.convolve(base, np.ones(3, dtype=np.float64) / 3.0, mode="same"))
        add(0.5 * (base + base[::-1]))

    t = np.linspace(0.0, 1.0, L)
    add(0.5 + 0.5 * np.cos(math.pi * t))
    add(0.42 - 0.5 * np.cos(2.0 * math.pi * t) + 0.08 * np.cos(4.0 * math.pi * t))
    add(0.34 - 0.49 * np.cos(2.0 * math.pi * t) + 0.16 * np.cos(4.0 * math.pi * t))
    add(0.5 + 0.5 * np.cos(2.0 * math.pi * t))

    w1 = np.maximum(0.5 + 0.5 * np.cos(math.pi * t), 1e-8)
    for pw in (0.5, 1.5):
        add(w1 ** pw)

    for _ in range(max(0, int(n_rand))):
        z = _random_z(L, rng)
        e = np.exp(np.clip(z, -12.0, 12.0))
        e = np.where(np.isfinite(e), e, 1.0)
        s = float(e.sum())
        if s > 0:
            add(e / s)

    if not pool:
        add(np.ones(L, dtype=np.float64))
    return pool


def _best_pool(pool):
    best = None
    best_sc = math.inf
    for a in pool:
        b = _autoconv(a)
        sc = _score_from_b(b, a.size)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a.copy()
    if best is None:
        best = _norm(np.ones(pool[0].size, dtype=np.float64))
    return best


def _optimize_logspace(n, init_a, deadline, seed, pool=None, global_best=math.inf):
    a = _norm(init_a)
    n = a.size
    rng = np.random.default_rng(seed)

    b = _autoconv(a)
    bmax = float(np.max(b)) if b.size else 0.0
    score = _score_from_b(b, n)
    if not np.isfinite(score):
        a = np.full(n, 1.0 / n, dtype=np.float64)
        b = _autoconv(a)
        score = _score_from_b(b, n)
    if not np.isfinite(score):
        score = 2.0
    bmax = score / (2.0 * n) if score > 0 else 1.0 / n

    best_a = a.copy()
    best_sc = score

    def make_z(arr):
        return _sanitize_z(np.log(np.maximum(np.asarray(arr, dtype=np.float64).ravel(), 1e-300)))

    z = make_z(a)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    base_step = 0.45 if n <= 256 else 0.25
    min_step = 0.02 if n <= 256 else 0.01
    max_step = 1.8 if n <= 256 else 1.0
    step = 0.40 if n <= 256 else 0.20

    tau_frac = 0.30
    fail = 0
    exact_stuck = 0
    iter_count = 0

    while time.monotonic() < deadline - 1.0:
        if not np.isfinite(score) or score <= 0.0:
            a = _norm(np.ones(n, dtype=np.float64) / n)
            b = _autoconv(a)
            bmax = float(np.max(b)) if b.size else 0.0
            score = _score_from_b(b, n)
            z = make_z(a)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            step = base_step
            fail = 0
            if not np.isfinite(score):
                break

        if score < best_sc:
            best_sc = score
            best_a = a.copy()

        tau = max(1e-12, tau_frac * max(bmax, 1e-300))
        U, p = _soft_logsumexp(b, tau)
        if p is None or p.size != b.size:
            p = np.full(max(1, b.size), 1.0 / max(1, b.size), dtype=np.float64)

        g = _grad_a(p, a)
        S = float(np.dot(a, g))
        if not np.isfinite(S):
            S = 0.0
        gz = a * (g - S)
        gz = _sanitize_z(gz)
        gmax = float(np.max(np.abs(gz))) if gz.size else 0.0

        if not np.isfinite(gmax) or gmax < 1e-14:
            z = _sanitize_z(z + float(rng.uniform(0.15, 0.55)) * rng.standard_normal(n))
            a = _softmax_z(z)
            b = _autoconv(a)
            bmax = float(np.max(b)) if b.size else 0.0
            score = _score_from_b(b, n)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            step = base_step
            fail += 1
            exact_stuck += 1
            iter_count += 1
            continue

        t += 1
        m = 0.90 * m + 0.10 * gz
        v = 0.998 * v + 0.002 * (gz * gz)
        mhat = m / (1.0 - 0.90 ** t)
        vhat = v / (1.0 - 0.998 ** t)
        direction = mhat / (np.sqrt(vhat) + 1e-8)
        direction = _sanitize_z(direction)
        dn = float(np.max(np.abs(direction))) if direction.size else 0.0

        if not np.isfinite(dn) or dn < 1e-14:
            direction = gz / gmax
            direction = _sanitize_z(direction)
            dn = float(np.max(np.abs(direction))) if direction.size else 0.0
            if not np.isfinite(dn) or dn < 1e-14:
                z = _sanitize_z(z + float(rng.uniform(0.10, 0.35)) * rng.standard_normal(n))
                a = _softmax_z(z)
                b = _autoconv(a)
                bmax = float(np.max(b)) if b.size else 0.0
                score = _score_from_b(b, n)
                m.fill(0.0)
                v.fill(0.0)
                t = 0
                step = base_step
                fail += 1
                exact_stuck += 1
                iter_count += 1
                continue

        direction /= dn
        zc = _sanitize_z(z - step * direction)
        ac = _softmax_z(zc)
        bc = _autoconv(ac)
        cbmax = float(np.max(bc)) if bc.size else 0.0
        cscore = _score_from_b(bc, n)
        if np.isfinite(cscore):
            Uc, _ = _soft_logsumexp(bc, tau)
        else:
            Uc = math.inf

        exact_ok = np.isfinite(cscore) and cscore < score - 1e-12
        soft_ok = (
            np.isfinite(cscore)
            and np.isfinite(Uc)
            and cscore <= score + 1e-5
            and Uc < U - 1e-10
        )

        if exact_ok or soft_ok:
            z = zc
            a = ac
            b = bc
            bmax = cbmax
            score = cscore
            if exact_ok:
                step = min(max_step, step * 1.12)
                tau_frac = max(0.015, tau_frac * 0.90)
                exact_stuck = 0
                fail = 0
            else:
                step = max(min_step, step * 0.98)
                tau_frac = max(0.015, tau_frac * 0.98)
                exact_stuck += 1
                fail = min(fail + 1, 3)
        else:
            fail += 1
            exact_stuck += 1
            step = max(min_step, step * 0.62)
            tau_frac = min(0.80, tau_frac * 1.18)

        iter_count += 1

        should_reset = (
            fail >= 10
            or exact_stuck >= 120
            or (score > best_sc + 0.01 and exact_stuck > 40)
            or (pool is not None and len(pool) > 1 and np.isfinite(global_best) and score > global_best + 0.03 and iter_count % 200 == 0)
        )

        if should_reset:
            if pool is not None and len(pool) > 0 and rng.random() < 0.75:
                a0 = pool[int(rng.integers(0, len(pool)))]
            else:
                a0 = best_a.copy()
            a0 = _norm(a0 + float(rng.uniform(0.02, 0.12)) * rng.standard_normal(n))
            z = make_z(a0)
            a = a0
            b = _autoconv(a)
            bmax = float(np.max(b)) if b.size else 0.0
            score = _score_from_b(b, n)
            if not np.isfinite(score):
                a = _norm(np.ones(n, dtype=np.float64) / n)
                b = _autoconv(a)
                score = _score_from_b(b, n)
                bmax = score / (2.0 * n) if score > 0 else 1.0 / n
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            step = base_step
            tau_frac = 0.30
            fail = 0
            exact_stuck = max(0, exact_stuck // 2)
            continue

        if exact_stuck > 30 and exact_stuck % 25 == 0:
            tau_frac = min(0.80, tau_frac * 1.6)
            if exact_stuck > 60:
                z = _sanitize_z(z + 0.15 * rng.standard_normal(n))
                a = _softmax_z(z)
                b = _autoconv(a)
                bmax = float(np.max(b)) if b.size else 0.0
                score = _score_from_b(b, n)
                m.fill(0.0)
                v.fill(0.0)
                t = 0
                step = base_step

    return best_a, best_sc


def _pair_polish(a, score, deadline, seed):
    a = _norm(a)
    n = a.size
    if n < 2 or time.monotonic() > deadline - 3.0 or n > 512:
        return a, score

    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_sc = float(score) if np.isfinite(score) else math.inf
    if not np.isfinite(best_sc):
        best_sc = _score_from_b(_autoconv(a), n)

    max_eval = 48 if n <= 224 else 20
    fracs = (0.001, 0.005, 0.02, 0.08, 0.20)
    K = 4
    no_imp = 0

    while time.monotonic() < deadline - 1.0:
        b = _autoconv(a)
        bmax = float(np.max(b)) if b.size else 0.0
        sc = _score_from_b(b, n)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        tau = max(1e-12, 0.03 * max(bmax, 1e-300))
        U, p = _soft_logsumexp(b, tau)
        g = _grad_a(p, a)
        h = g - g.mean()
        hinf = float(np.max(np.abs(h))) if h.size else 0.0
        if not np.isfinite(hinf) or hinf < 1e-14:
            break

        hi = np.argpartition(h, -K)[-K:].astype(int)
        lo = np.argpartition(h, K - 1)[:K].astype(int)

        accepted = False
        evals = 0
        for i in hi:
            ai = float(a[i])
            if not np.isfinite(ai) or ai <= 1e-15:
                continue
            for j in lo:
                if i == j:
                    continue
                for fr in fracs:
                    eps = float(fr * ai)
                    if eps <= 0.0 or eps > ai:
                        continue
                    cand = a.copy()
                    cand[i] -= eps
                    cand[j] += eps
                    np.maximum(cand, 0.0, out=cand)
                    s = float(cand.sum())
                    if s <= 0.0:
                        continue
                    cand /= s
                    bc = _autoconv(cand)
                    csc = _score_from_b(bc, n)
                    evals += 1
                    if np.isfinite(csc) and csc < best_sc - 1e-12:
                        best_a = cand.copy()
                        best_sc = csc
                        accepted = True
                        no_imp = 0
                        break
                    if evals >= max_eval:
                        break
                if accepted or evals >= max_eval:
                    break
            if accepted or evals >= max_eval:
                break

        if accepted:
            a = best_a.copy()
        else:
            no_imp += 1
            sigma = 0.001 + 0.005 * min(1.0, no_imp / 20.0)
            a = _norm(best_a + sigma * rng.standard_normal(n))
            if no_imp > 60:
                break

    return best_a, best_sc


def _threshold_improve(seq, score):
    seq = _norm(seq)
    n = seq.size
    best = seq.copy()
    best_sc = float(score) if np.isfinite(score) else math.inf

    for thr in (1e-8, 1e-7, 1e-6, 1e-5, 5e-5, 1e-4, 5e-4, 1e-3):
        cand = seq.copy()
        cand[cand < thr] = 0.0
        s = float(cand.sum())
        if s <= 0.0:
            continue
        cand /= s
        try:
            sc = float(evaluate_sequence([float(x) for x in cand]))
        except Exception:
            sc = math.inf
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best = cand.copy()
            best_sc = sc
    return best, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s))
    t_end = t0 + max(0.0, budget - 20.0)

    parent = _get_parent_norm()

    fallback = parent.copy()
    try:
        fallback_score = float(evaluate_sequence([float(x) for x in fallback]))
    except Exception:
        fallback_score = math.inf
    if not np.isfinite(fallback_score):
        fallback = np.ones(32, dtype=np.float64) / 32.0
        try:
            fallback_score = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            fallback_score = 2.0
        if not np.isfinite(fallback_score):
            fallback_score = 2.0

    best_seq = fallback.copy()
    best_score = fallback_score

    scale = min(1.0, max(0.15, (budget - 60.0) / 1300.0)) if budget > 60.0 else 0.20
    target = 1.50274349

    phase_specs = [
        (192, 520.0),
        (256, 140.0),
        (160, 90.0),
        (224, 80.0),
        (320, 80.0),
        (128, 70.0),
        (384, 80.0),
    ]

    for L, base_alloc in phase_specs:
        if time.monotonic() > t_end - 45.0:
            break

        alloc = float(base_alloc) * scale
        if np.isfinite(best_score) and best_score < target and L != len(best_seq):
            alloc *= 0.35
        rem = max(0.0, t_end - time.monotonic() - 30.0)
        alloc = min(alloc, rem)
        if alloc < 8.0:
            continue

        try:
            n_rand = 6 if L == 192 else 3
            pool = _make_pool(int(L), parent, int(seed) + 100000 + int(L), n_rand)
            init = _best_pool(pool)
        except Exception:
            pool = [_norm(np.ones(int(L), dtype=np.float64))]
            init = pool[0]

        deadline = min(t_end - 15.0, time.monotonic() + 0.75 * alloc)
        cand = None
        sc_int = math.inf
        try:
            cand, sc_int = _optimize_logspace(
                int(L),
                init,
                deadline,
                int(seed) + 1009 + int(L),
                pool=pool,
                global_best=best_score,
            )
            cand = _norm(cand)
        except Exception:
            cand = None

        if cand is not None and cand.size == int(L):
            try:
                es = float(evaluate_sequence([float(x) for x in cand]))
            except Exception:
                es = math.inf
            if np.isfinite(es) and es < best_score - 1e-12:
                best_score = es
                best_seq = cand.copy()

        deadline2 = min(t_end - 12.0, time.monotonic() + 0.25 * alloc)
        if cand is not None and cand.size == int(L) and time.monotonic() < deadline2 - 5.0:
            local_ok = np.isfinite(sc_int) and (not np.isfinite(best_score) or sc_int < best_score + 0.08)
            if local_ok:
                try:
                    cand2, _ = _optimize_logspace(
                        int(L),
                        cand,
                        deadline2,
                        int(seed) + 777 + int(L),
                        pool=pool,
                        global_best=best_score,
                    )
                    cand2 = _norm(cand2)
                    es2 = float(evaluate_sequence([float(x) for x in cand2]))
                    if np.isfinite(es2) and es2 < best_score - 1e-12:
                        best_score = es2
                        best_seq = cand2.copy()
                except Exception:
                    pass

    seq = _norm(best_seq)
    rem = max(0.0, t_end - 10.0 - time.monotonic())
    if rem > 35.0 and seq.size <= 512:
        pdeadline = min(t_end - 8.0, time.monotonic() + min(rem, 200.0))
        try:
            seq_pair, sc_pair = _pair_polish(seq, best_score, pdeadline, int(seed) + 31337)
            seq_pair = _norm(seq_pair)
            es = float(evaluate_sequence([float(x) for x in seq_pair]))
            if np.isfinite(es) and es < best_score - 1e-12:
                best_score = es
                seq = seq_pair
                best_seq = seq.copy()
        except Exception:
            pass

    rem = max(0.0, t_end - 8.0 - time.monotonic())
    if rem > 25.0:
        L = int(best_seq.size)
        pool = [_norm(best_seq)]
        d2 = min(t_end - 5.0, time.monotonic() + min(rem, 80.0))
        try:
            cand, _ = _optimize_logspace(
                L,
                best_seq,
                d2,
                int(seed) + 999,
                pool=pool,
                global_best=best_score,
            )
            cand = _norm(cand)
            es = float(evaluate_sequence([float(x) for x in cand]))
            if np.isfinite(es) and es < best_score - 1e-12:
                best_score = es
                best_seq = cand.copy()
        except Exception:
            pass

    seq = _norm(best_seq)
    try:
        seq, best_score = _threshold_improve(seq, best_score)
        seq = _norm(seq)
    except Exception:
        pass

    seq = _norm(seq)
    try:
        final_score = float(evaluate_sequence([float(x) for x in seq]))
    except Exception:
        final_score = math.inf

    if not np.isfinite(final_score):
        seq = _norm(fallback)
        try:
            fallback_score = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            fallback_score = 2.0
        if not np.isfinite(fallback_score) or fallback_score > 2.5:
            seq = np.ones(32, dtype=np.float64) / 32.0
            return [float(x) for x in seq]

    return [float(x) for x in seq]
