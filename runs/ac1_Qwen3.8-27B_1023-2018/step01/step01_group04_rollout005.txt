    with np.errstate(over="ignore", invalid="ignore"):
        c = np.convolve(a, a)
    m = float(np.max(c))
    if not math.isfinite(m) or m < 0.0:
        return float("inf")
    return float(2.0 * n * m / (s * s))


def _update_best(a, best):
    try:
        arr = _normalize_sum1(a)
        if arr.size == 0:
            return
        s = float(arr.sum())
        if s < 0.01:
            arr = np.clip(arr + 0.001, 0.0, None)
            s = float(arr.sum())
            if s > 1e-15:
                arr = arr / s
            else:
                arr = np.ones(arr.size, dtype=np.float64) / max(1, arr.size)
        arr = np.clip(arr, 0.0, 1000.0)
        seq = [float(x) for x in arr]
        sc = _exact_score(seq)
        if math.isfinite(sc) and sc < float(best[0]) - 1e-14:
            best[0] = float(sc)
            best[1] = arr.copy()
    except Exception:
        pass


def _softmax(y):
    y = np.asarray(y, dtype=np.float64).ravel()
    if y.size == 0:
        return np.ones(1, dtype=np.float64)
    y = np.nan_to_num(y, nan=0.0, posinf=700.0, neginf=-700.0)
    if y.size > 1:
        y = y - float(np.max(y))
    with np.errstate(under="ignore", over="ignore", invalid="ignore"):
        e = np.exp(np.clip(y, -745.0, 0.0))
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.ones(y.size, dtype=np.float64) / y.size
    return e / s


def _softmax_obj_grad(y, beta, scale=1.0):
    y = np.asarray(y, dtype=np.float64).ravel()
    n = y.size
    if n == 0:
        return 1.0e6, np.array([], dtype=np.float64)
    y = np.nan_to_num(y, nan=0.0, posinf=700.0, neginf=-700.0)
    if n > 1:
        y = y - float(np.mean(y))
        y = y - float(np.max(y))
    with np.errstate(under="ignore", over="ignore", invalid="ignore"):
        e = np.exp(np.clip(y, -745.0, 0.0))
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = e / s

    with np.errstate(over="ignore", invalid="ignore"):
        c = np.convolve(a, a)
    if not np.all(np.isfinite(c)):
        return 1.0e6, np.zeros(n, dtype=np.float64)

    m = float(np.max(c))
    if beta <= 0.0 or not math.isfinite(beta):
        beta = 1.0
    arg = np.clip(float(beta) * (c - m), -745.0, 0.0)
    with np.errstate(under="ignore", over="ignore", invalid="ignore"):
        w = np.exp(arg)
    sw = float(w.sum())
    if not math.isfinite(sw) or sw <= 0.0:
        w = np.exp(np.clip(float(beta) * (c - m), -745.0, 0.0))
        sw = float(w.sum())
        if not math.isfinite(sw) or sw <= 0.0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            sw = 1.0
    w = w / max(1e-300, sw)

    smax = m + math.log(max(1e-300, sw)) / float(beta)

    h = np.convolve(a[::-1], w)[n - 1:2 * n - 1]
    ga = 2.0 * h
    if not np.all(np.isfinite(ga)):
        ga = np.zeros(n, dtype=np.float64)
    dot = float(np.dot(a, ga))
    gy = a * (ga - dot)
    if not np.all(np.isfinite(gy)):
        gy = np.zeros(n, dtype=np.float64)
    return float(scale * smax), (scale * gy)


def _polish(a0, time_limit, rng, best, betas=None, maxiter=None):
    if minimize is None or time_limit is None or float(time_limit) <= 5.0:
        return
    stop = time.monotonic() + max(5.0, float(time_limit))
    a = _normalize_sum1(a0)
    n = a.size
    if n < 2:
        return
    if maxiter is None:
        maxiter = int(max(30, min(150, 9000.0 / max(1, n))))
    if betas is None:
        betas = tuple(int(x * n) for x in (10.0, 50.0, 200.0, 1000.0))
    scale = 2.0 * float(n)

    y = np.log(np.maximum(a, 1e-300))
    y = y - float(np.max(y))
    state = {"y": y.copy(), "a": a.copy(), "score": _score_fast(a), "count": 0}
    last = {"y": y.copy()}
    freq = 2 if n <= 128 else (4 if n <= 256 else 8)

    def consider(x, force=False):
        x = np.asarray(x, dtype=np.float64).ravel()
        if x.size != n:
            return
        if not np.all(np.isfinite(x)):
            return
        state["count"] += 1
        if not force and (state["count"] % freq != 0):
            return
        aa = _softmax(x)
        sc = _score_fast(aa)
        if math.isfinite(sc):
            if sc < state["score"] - 1e-12:
                state["score"] = sc
                state["a"] = aa.copy()
                state["y"] = x - float(np.mean(x))
            if sc < float(best[0]) + 1e-12:
                _update_best(aa, best)

    consider(y, True)

    for beta in betas:
        if time.monotonic() >= stop - 3.0:
            break

        def fun(xx, beta=beta):
            if time.monotonic() > stop:
                raise _Stop()
            xx = np.asarray(xx, dtype=np.float64).ravel()
            last["y"] = xx
            consider(xx, False)
            try:
                val, grad = _softmax_obj_grad(xx, beta, scale)
            except Exception:
                val, grad = 1.0e6, np.zeros(n, dtype=np.float64)
            if not math.isfinite(val):
                val, grad = 1.0e6, np.zeros(n, dtype=np.float64)
            return val, grad

        def callback(xx):
            if time.monotonic() > stop:
                raise _Stop()

        try:
            res = minimize(
                fun,
                state["y"].copy(),
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={
                    "maxiter": int(maxiter),
                    "gtol": 1e-10,
                    "ftol": 1e-13,
                    "maxls": 25,
                },
            )
            if res is not None and getattr(res, "x", None) is not None:
                last["y"] = np.asarray(res.x, dtype=np.float64).ravel()
        except _Stop:
            pass
        except Exception:
            pass

        consider(last["y"], True)

    if state["score"] < float(best[0]) + 1e-12:
        _update_best(state["a"], best)


def _convolution_matrix(a):
    n = len(a)
    m = 2 * n - 1
    if toeplitz is not None and n <= 512:
        c = np.zeros(m, dtype=np.float64)
        c[:n] = a
        r = np.zeros(n, dtype=np.float64)
        r[0] = a[0]
        try:
            return toeplitz(c, r).astype(np.float64, copy=False)
        except Exception:
            pass

    if sps is not None and n > 256:
        ii = np.repeat(np.arange(n, dtype=np.int64), n)
        jj = np.tile(np.arange(n, dtype=np.int64), n)
        rows = ii + jj
        cols = jj
        data = np.tile(a, n)
        try:
            return sps.csr_matrix((data.astype(np.float64), (rows, cols)), shape=(m, n))
        except Exception:
            pass

    A = np.zeros((m, n), dtype=np.float64)
    if n <= 2048:
        ii = np.repeat(np.arange(n, dtype=np.int64), n)
        jj = np.tile(np.arange(n, dtype=np.int64), n)
        A[ii + jj, jj] = np.tile(a, n)
    return A


def _lp_search(a0, time_limit, rng, best):
    if linprog is None or time_limit is None or float(time_limit) <= 8.0:
        return
    stop = time.monotonic() + max(8.0, float(time_limit))
    a = _normalize_sum1(a0)
    n = a.size
    if n < 3:
        return

    old_score = _score_fast(a)
    if math.isfinite(old_score):
        _update_best(a, best)

    stagnant = 0
    fails = 0
    it = 0

    while time.monotonic() < stop - 2.0:
        it += 1
        aw = np.maximum(a, 1e-12).astype(np.float64, copy=True)
        s = float(aw.sum())
        if s <= 1e-15:
            break
        aw = aw / s

        with np.errstate(over="ignore", invalid="ignore"):
            c = np.convolve(aw, aw)
        M = float(np.max(c))
        if not math.isfinite(M) or M <= 0.0:
            break

        if M > TARGET_M:
            goals = [M, 0.5 * (M + TARGET_M), TARGET_M]
        else:
            goals = [M]

        gi = (it // 2) % len(goals)
        linear = (it % 2) == 1
        goal = float(goals[gi])
        if linear:
            level = 0.5 * (c + goal)
        else:
            level = np.full_like(c, float(goal))
        level = np.maximum(level, 1e-12)

        try:
            A = _convolution_matrix(aw)
            if sps is not None and sps.issparse(A):
                one_col = sps.csr_matrix(-np.ones((A.shape[0], 1), dtype=np.float64))
                A_aug = sps.hstack([A, one_col], format="csc")
            else:
                A = np.asarray(A, dtype=np.float64)
                A_aug = np.column_stack((A, -np.ones((A.shape[0], 1), dtype=np.float64)))

            c_obj = np.zeros(n + 1, dtype=np.float64)
            c_obj[n] = 1.0
            A_eq = np.zeros((1, n + 1), dtype=np.float64)
            A_eq[0, :n] = 1.0
            bounds = [(0.0, None)] * n + [(-100.0, 100.0)]

            try:
                res = linprog(
                    c_obj,
                    A_ub=A_aug,
                    b_ub=level,
                    A_eq=A_eq,
                    b_eq=1.0,
                    bounds=bounds,
                    method="highs",
                    options={"presolve": True},
                )
            except Exception:
                res = linprog(
                    c_obj,
                    A_ub=A_aug,
                    b_ub=level,
                    A_eq=A_eq,
                    b_eq=1.0,
                    bounds=bounds,
                    options={"presolve": True},
                )
        except Exception:
            fails += 1
            if fails > 6:
                break
            continue

        if res is None or getattr(res, "x", None) is None:
            fails += 1
            if fails > 6:
                break
            continue

        b = np.asarray(res.x[:n], dtype=np.float64).ravel()
        b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
        b = np.clip(b, 0.0, None)
        bs = float(b.sum())
        if bs <= 1e-12:
            stagnant += 1
            if stagnant > 25:
                break
            continue
        b = b / bs

        sc_best = _score_fast(a)
        best_cand = a
        if goal < M * 0.95:
            ts = (0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.75, 1.0)
        else:
            ts = (0.1, 0.2, 0.35, 0.5, 0.75, 1.0)

        for t in ts:
            cand = (1.0 - float(t)) * a + float(t) * b
            cand = np.clip(cand, 0.0, None)
            cs = float(cand.sum())
            if cs > 1e-15:
                cand = cand / cs
            sc = _score_fast(cand)
            if math.isfinite(sc) and sc < sc_best - 1e-12:
                sc_best = sc
                best_cand = cand

        if math.isfinite(sc_best) and sc_best < _score_fast(a) * (1.0 - 1e-10):
            a = best_cand
            stagnant = 0
            fails = 0
            _update_best(a, best)
        else:
            stagnant += 1
            fails = 0
            if stagnant % 6 == 0:
                y = np.log(np.maximum(a, 1e-12))
                amp = min(1.0, 0.05 + 0.02 * stagnant)
                pa = _softmax(y + amp * rng.standard_normal(n))
                ps = _score_fast(pa)
                if math.isfinite(ps) and ps < sc_best * (1.0 - 1e-9):
                    a = pa
                    stagnant = 0
                    _update_best(a, best)
            if stagnant > 30:
                break

    _update_best(a, best)


def _resample(a, n):
    a = _normalize_sum1(a)
    n = max(1, int(n))
    if n == 1:
        return np.ones(1, dtype=np.float64)
    if a.size == 1:
        return np.ones(n, dtype=np.float64) / n
    xp = np.linspace(0.0, 1.0, a.size, dtype=np.float64)
    xnew = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(xnew, xp, a)
    return _normalize_sum1(b)


def _make_starts(a_parent, n, rng, count=8):
    out = []
    n = max(1, int(n))

    def add(x):
        x = _normalize_sum1(x)
        if x.size == n:
            sc = _score_fast(x)
            if math.isfinite(sc):
                out.append((sc, x))

    if a_parent is not None:
        p = _resample(a_parent, n)
        add(p)
        add(p[::-1])
        y = np.log(np.maximum(p, 1e-300))
        y -= float(np.max(y))
        for amp in (0.03, 0.1, 0.25, 0.5):
            add(_softmax(y + amp * rng.standard_normal(n)))

    add(np.ones(n, dtype=np.float64))

    if n > 1:
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        for c in (0.2, 0.6, 1.2, 2.0, 3.0, -0.2, -0.6, -1.2, -2.0, -3.0):
            add(np.exp(c * (x - 0.5)))
        for p in (0.4, 0.7, 1.0, 1.6, 2.5):
            add(x**p)
            add((1.0 + 1e-12 - x)**p)

        max_modes = min(6, max(1, n // 4))
        for k in range(1, max_modes + 1):
            for amp in (0.25, 0.5, 0.8, 1.1):
                for ph in (0.0, 0.5, math.pi * rng.random()):
                    add(1.0 + amp * np.sin(2.0 * math.pi * k * x + ph))
                    add(1.0 + amp * np.cos(2.0 * math.pi * k * x + ph))

        for _ in range(6):
            yf = np.zeros(n, dtype=np.float64)
            for k in range(1, min(8, max(2, n // 4)) + 1):
                amp = rng.normal(0.0, 0.35 / float(k + 1))
                ph = rng.uniform(0.0, 2.0 * math.pi)
                yf += amp * np.sin(2.0 * math.pi * k * x + ph)
                yf += 0.5 * amp * np.cos(2.0 * math.pi * k * x + rng.uniform(0.0, 2.0 * math.pi))
            add(_softmax(yf))

    out.sort(key=lambda z: z[0])
    res = [arr.copy() for _, arr in out[:count]]
    if not res:
        res = [np.ones(n, dtype=np.float64) / n]
    return res


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        B = float(budget_s)
    except Exception:
        B = 1300.0
    if not math.isfinite(B) or B <= 120.0:
        B = 1300.0
    deadline = t0 + max(120.0, B - 25.0)

    rng = np.random.default_rng(seed)
    best = [2.0, np.array([0.5, 0.5], dtype=np.float64)]
    _update_best(best[1], best)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = _normalize_sum1(h)
            if p.size > 0:
                parent = p
                _update_best(p, best)
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(96, dtype=np.float64)
        _update_best(parent, best)

    parent = _normalize_sum1(parent)
    if parent.size < 8:
        parent = np.ones(96, dtype=np.float64)
    elif parent.size > 256:
        parent = _resample(parent, 192)
    n_base = parent.size

    def rem(reserve=40.0):
        return max(0.0, deadline - time.monotonic() - reserve)

    if minimize is not None and rem(180.0) > 40.0:
        _polish(parent, rem(180.0), rng, best)

    if linprog is not None and rem(180.0) > 40.0:
        a0 = best[1] if np.asarray(best[1]).size == n_base else parent
        _lp_search(a0, rem(160.0), rng, best)

    if minimize is not None and rem(160.0) > 40.0:
        a0 = best[1] if np.asarray(best[1]).size == n_base else parent
        _polish(a0, rem(140.0), rng, best,
                betas=tuple(int(x * n_base) for x in (50.0, 200.0, 800.0, 3000.0)))

    if linprog is not None and rem(140.0) > 30.0:
        a0 = best[1] if np.asarray(best[1]).size == n_base else parent
        _lp_search(a0, rem(120.0), rng, best)

    if rem(120.0) > 60.0:
        starts = _make_starts(parent, n_base, rng, 8)
        for s in starts:
            if rem(90.0) <= 35.0:
                break
            if minimize is not None:
                _polish(s, min(55.0, rem(80.0)), rng, best,
                        betas=tuple(int(x * n_base) for x in (50.0, 250.0, 1000.0)),
                        maxiter=70)
            if linprog is not None and rem(80.0) > 20.0:
                _lp_search(s, min(35.0, rem(65.0)), rng, best)

    other_lengths = []
    for nn in (96, 128, 192, 256):
        if nn != n_base and nn <= 256:
            other_lengths.append(nn)

    for nn in other_lengths:
        if rem(220.0) < 120.0:
            break
        if best[1].size == nn:
            a0 = best[1]
        else:
            a0 = _resample(parent, nn)
        if minimize is not None and rem(180.0) > 45.0:
            _polish(a0, min(180.0, rem(160.0)), rng, best)
        if linprog is not None and rem(160.0) > 40.0:
            a1 = best[1] if best[1].size == nn else a0
            _lp_search(a1, min(120.0, rem(140.0)), rng, best)
        if minimize is not None and rem(150.0) > 35.0:
            a2 = best[1] if best[1].size == nn else a0
            _polish(a2, min(80.0, rem(130.0)), rng, best,
                    betas=tuple(int(x * nn) for x in (100.0, 500.0, 2000.0)),
                    maxiter=60)

    if rem(45.0) > 25.0:
        a0 = best[1]
        if minimize is not None:
            _polish(a0, min(70.0, rem(35.0)), rng, best,
                    betas=tuple(int(x * a0.size) for x in (200.0, 800.0, 3500.0)),
                    maxiter=70)
        if linprog is not None and rem(30.0) > 15.0:
            _lp_search(a0, min(35.0, rem(25.0)), rng, best)

    final = _normalize_sum1(best[1])
    if final.size == 0:
        final = np.array([1.0], dtype=np.float64)
    seq = [float(x) for x in np.clip(final, 0.0, 1000.0)]
    sc = _exact_score(seq)
    if not math.isfinite(sc):
        seq = [0.5, 0.5]
    seq = [float(x) for x in seq if math.isfinite(float(x))]
    if not seq:
        seq = [1.0]
    return seq
