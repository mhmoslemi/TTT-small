def _grad(a, b, tau, n):

    bmax = np.max(b)

def _polish(a, score, stop_time):

    n = len(a)
    fsize = next_fast_len(2*n-1)
    rng = np.random.default_rng(12345)

    best_score = score

        s = cur.sum()
        if s > 0:
            cur = cur / s

        mode = rng.random()
        if mode < 0.4:
            # Gaussian noise
            cand = cur + rng.normal(0, sigma, n)
        elif mode < 0.7:
            # Local segment scaling
            width = max(1, n // int(rng.integers(20, 80)))
            start = rng.integers(0, max(1, n - width))
            end = min(n, start + width)
            cand = cur.copy()
            cand[start:end] *= np.exp(rng.normal(0, 0.3))

            # Symmetrize
            cand = 0.5*(cur + cur[::-1])
        else:

            kernel = np.ones(max(1, n//100)) / max(1, n//100)
            cand = np.convolve(cur, kernel, mode='same')

        cs = cand.sum()
        if cs < 0.01:

        cand = cand / cs

            best_score = cs2
            stuck = 0
            sigma = max(0.05, sigma * 0.95)

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(192, dtype=np.float64)
    p = np.clip(np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0), 0.0, 1000.0)

def _fft_conv(a, fsize):
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)
    n = len(a)
    return np.maximum(c[:2*n-1], 0.0)

def _grad(a, b, tau, n, K_top=100):
    bmax = b.max()
    x = np.clip((b - bmax) / max(tau, 1e-12), -50, 0)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w = ex / s
    
    if K_top < len(b):
        idx = np.argpartition(w, -K_top)[-K_top:]
    else:
        idx = np.where(w > 1e-12)[0]
    
    G = np.zeros(n)
    for k in idx:
        wk = w[k]
        if wk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            G[lo:hi+1] += wk * a[k-hi:k-lo+1][::-1]
    return G  # Factor of 2n (or 4n) absorbed into lr

def _proj_simplex(v):
    n = len(v)
    if n <= 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n+1, dtype=np.float64)
    cond = u * ind > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
