```python
def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

    cssv = np.cumsum(u)
```

```python
def _proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.full(n, 1.0/n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _compute_subgradient(a, active_frac=1.0):
    """
    Compute subgradient of max(conv(a,a)) w.r.t. a.
    active_frac: fraction of max to consider active (1.0 = exact)
    """
    n = len(a)
    b = np.convolve(a, a)

        return np.zeros(n), b

    thresh = M * active_frac if active_frac < 1.0 else M

            a_slice = a[k-n+1:n][::-1]

    return grad, b

                best_score = s
                a = a_new

        else:

            if step <= 0.0015:

    return best, best_score

def _lp_step(a, deadline):
    """
    LP-based step: find g maximizing sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns the LP solution g (normalized to sum 1) or None.
    """

    b = np.convolve(a, a)
    M = float(b.max())
    if M <= 0 or not np.isfinite(M):
        return None

            if s > 0:
```

```python
import numpy as np
import time
from scipy.optimize import linprog


def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score_unit(a):
    """Score assuming sum(a)=1: 2n * max(conv(a,a))."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * float(b.max())


def _compute_subgrad(a, n_active_max=30):
    """
    Compute subgradient of max(conv(a,a)) w.r.t. a.
    Returns (gradient_normalized, b, max_b).
    """
    n = len(a)
    b = np.convolve(a, a)
    M = float(b.max())
    if M <= 0:
        return np.zeros(n), b, M
    # Use top active indices
    idx = np.argsort(b)[::-1][:n_active_max]
    # Only keep those very close to max
    active = idx[b[idx] >= M - 1e-12]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        if k < n:
            j_end = k
            a_seg = a[0:k + 1][::-1]
            grad[0:j_end + 1] += a_seg
        else:
            j_start = k - n + 1
            a_seg = a[k - n + 1:n][::-1]
            grad[j_start:n] += a_seg
    grad /= float(len(active))
    return grad, b, M

def _smoothed_grad(a, T):
    """Smoothed max gradient using softmax weights."""
    n = len(a)
    b = np.convolve(a, a)
    bm = b.max()
    if bm <= 0:
        return np.zeros(n), b, bm
    b_shift = (b - bm) / T
    max_bs = b_shift.max()
    exp_b = np.exp(b_shift - max_bs)
    p = exp_b / exp_b.sum()
    
    # Only use significant weights
    thresh = max(1e-8, 0.01 / n)
    active = np.nonzero(p > thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    wa = p[active]
    wa = wa / wa.sum()
    
    grad = np.zeros(n)
    for k, w in zip(active, wa):
        k = int(k)
        wk = 2.0 * w
        if k < n:
            grad[0:k + 1] += wk * a[0:k + 1][::-1]
        else:
            j_start = k - n + 1
            grad[j_start:n] += wk * a[k - n + 1:n][::-1]
    
    sm = bm + T * np.log(exp_b.sum() * np.exp(max_bs))
    return grad, b, sm

def _build_A(a, n):
    """Build the (2n-1) x n matrix for convolve(a, g)."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        if k < n:
            A[k, 0:k + 1] = a[0:k + 1][::-1]
        else:
            A[k, k - n + 1:n] = a[k - n + 1:n][::-1]
    return A


def _lp_step(a, deadline):
    """LP: max sum(g) s.t. A*g <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    b = np.convolve(a, a)
    M = float(b.max())
    if M <= 0 or not np.isfinite(M):
        return None
    
    time_limit = max(2.0, deadline - time.monotonic() - 10.0)
    if time_limit < 2.0:
        return None
    
    A = _build_A(a, n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if getattr(res, 'success', False):
            g = np.asarray(res.x, dtype=np.float64)
            g = np.where(np.isfinite(g) & (g > 0), g, 0.0)
            s = g.sum()
            if s > 0:
                return g / s
    except Exception:
        pass
    return None


def _lp_iterate(a, deadline, max_iters=30):
    """LP-based iteration with blending."""
    a = _proj_simplex(np.array(a, dtype=np.float64).copy())
    best = a.copy()
    best_sc = _score_unit(a)
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 15.0:
            break
        
        g = _lp_step(a, deadline)
        if g is None:
            break
        g = _proj_simplex(g)
        
        improved = False
        for t in [0.4, 0.6, 0.3, 0.5, 0.7, 0.2, 0.8, 0.1, 1.0]:
            cand = _proj_simplex((1.0 - t) * a + t * g)
            sc = _score_unit(cand)
            if sc < best_sc - 1e-12:
                best = cand
                best_sc = sc
                a = cand
                improved = True
                break
        
        if not improved:
            stall += 1
            if stall >= 3:
                break
    
    return best, best_sc


def _upscale(q, k):
    """Upscale sequence q to length k."""
    q = np.asarray(q, dtype=np.float64)
    if q.size == k:
        return _proj_simplex(q.copy())
    if q.size == 0:
        return _proj_simplex(np.ones(k))
    if k % q.size == 0:
        out = np.repeat(q, k // q.size)
    else:
        x = np.linspace(0.0, 1.0, q.size)
        y = np.linspace(0.0, 1.0, k)
        out = np.interp(y, x, q)
    out = np.where(np.isfinite(out) & (out > 0), out, 0.0)
    s = out.sum()
    if s <= 0:
        return _proj_simplex(np.ones(k))
    return out / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = max(10.0, float(budget_s))
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 35.0  # headroom
    
    best_list = None
    best_score = float('inf')
    
    def consider(seq):
        nonlocal best_list, best_score
        try:
            arr = np.asarray(seq, dtype=np.float64)
            if arr.ndim != 1 or arr.size == 0:
                return
            arr = np.where(np.isfinite(arr) & (arr > 0), arr, 0.0)
            s = float(arr.sum())
            if s <= 0.01:
                return
            arr = arr / s
            lst = [float(x) for x in arr]
            val = float(evaluate_sequence(lst))
            if np.isfinite(val) and val < best_score:
                best_score = val
                best_list = lst
        except Exception:
            pass
    
    # Try parent sequence
    parent_arr = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            parent_arr = np.asarray(p, dtype=np.float64).copy()
            consider(parent_arr)
    except Exception:
        pass
    
    # Basic candidates
    q4 = np.array([0.2, 0.4, 0.2, 0.2], dtype=np.float64)
    consider(q4)
    
    # Build starting points for n=512
    n512 = 512
    starts_512 = []
    if parent_arr is not None:
        starts_512.append(_proj_simplex(_upscale(parent_arr, n512)))
    
    # Structured starts
    t = np.arange(n512, dtype=np.float64)
    # Cosine taper
    starts_512.append(_proj_simplex(0.5 + 0.5 * np.cos(np.pi * t / (n512 - 1))))
    # Raised cosine
    starts_512.append(_proj_simplex(0.5 * (1 + np.cos(2 * np.pi * t / n512))))
    # Asymmetric: bump shifted left
    mu = n512 * 0.35
    sigma = n512 * 0.2
    starts_512.append(_proj_simplex(np.exp(-0.5 * ((t - mu) / sigma) ** 2)))
    # Random Dirichlet
    starts_512.append(_proj_simplex(rng.dirichlet(np.full(n512, 1.0))))
    
    # Optimize each start for n=512
    for i, a0 in enumerate(starts_512):
        if time.monotonic() > deadline - 40.0:
            break
        remaining = len(starts_512) - i
        alloc = (deadline - time.monotonic()) / remaining
        if alloc < 20.0:
            break
        end_time = min(deadline - 10.0, time.monotonic() + alloc)
        
        # Phase 1: Smooth gradient descent (explore)
        a1, sc1 = _smooth_gradient_descent(a0, end_time * 0.5 + start_time, max_iters=2000)
        consider(a1)
        
        # Phase 2: Sharp gradient descent (polish)
        a2, sc2 = _gradient_descent(a1, end_time * 0.8 + start_time, max_iters=3000)
        consider(a2)
        
        # Phase 3: LP refinement
        if time.monotonic() < deadline - 80.0:
            a3, sc3 = _lp_iterate(a2, min(deadline - 30.0, time.monotonic() + 120.0), max_iters=25)
            consider(a3)
            if sc3 < sc2:
                a2 = a3
                sc2 = sc3
        
        # Phase 4: Final gradient polish
        if time.monotonic() < deadline - 20.0:
            a4, sc4 = _gradient_descent(a2, deadline - 10.0, max_iters=2000)
            consider(a4)
            if sc4 < sc2:
                a2 = a4
                sc2 = sc4
    
    # Try n=256 as well (faster LP, possibly different optimum)
    if time.monotonic() < deadline - 80.0:
        n256 = 256
        starts_256 = []
        if parent_arr is not None:
            starts_256.append(_proj_simplex(_upscale(parent_arr, n256)))
        t256 = np.arange(n256, dtype=np.float64)
        starts_256.append(_proj_simplex(0.5 + 0.5 * np.cos(np.pi * t256 / (n256 - 1))))
        starts_256.append(_proj_simplex(rng.dirichlet(np.full(n256, 2.0))))
        
        for a0 in starts_256:
            if time.monotonic() > deadline - 40.0:
                break
            a1, sc1 = _smooth_gradient_descent(a0, min(deadline - 30.0, time.monotonic() + 60.0), max_iters=2000)
            consider(a1)
            a2, sc2 = _gradient_descent(a1, min(deadline - 20.0, time.monotonic() + 60.0), max_iters=2000)
            consider(a2)
    
    # Final: if we still have time, do one more round of LP on the best
    if time.monotonic() < deadline - 60.0 and best_list is not None:
        a_best = _proj_simplex(np.array(best_list, dtype=np.float64))
        n_best = len(a_best)
        a_lp, sc_lp = _lp_iterate(a_best, deadline - 30.0, max_iters=20)
        consider(a_lp)
        if sc_lp < _score_unit(a_best):
            a_fin, sc_fin = _gradient_descent(a_lp, deadline - 10.0, max_iters=1500)
            consider(a_fin)
    
    # Fallback
    if best_list is None:
        consider(q4)
    if best_list is None:
        best_list = [1.0]
    
    # Final validation
    try:
        val = float(evaluate_sequence(best_list))
        if not np.isfinite(val):
            best_list = _proj_simplex(q4).tolist()
    except Exception:
        best_list = [1.0]
    
    # Ensure we return in time
    if time.monotonic() > deadline - 5.0:
        q4_list = _proj_simplex(q4).tolist()
        try:
            if np.isfinite(float(evaluate_sequence(q4_list))):
                if best_list is None:
                    best_list = q4_list
        except Exception:
            best_list = q4_list
    
    return [float(x) for x in best_list]
```
