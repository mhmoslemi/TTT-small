        x = np.array(x, dtype=np.float64, copy=True)
        if x.size != n:
            return None
        x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
        np.maximum(x, 0.0, out=x)

            return c

                best_x = cand.copy()

        if best_m < M - 1e-10:
            return best_x, best_m
        return None

    Kinit = min(64, L)
    Kmax = min(256, L)

    def improve(a0, end_time):
        a = np.array(a0, dtype=np.float64, copy=True)
        B = conv(a)
        M = float(B.max())
        best_local = a.copy()
        best_local_M = M

        K = Kinit
        beta = 4.0
        alpha_m = 0.05
        alpha_a = 0.05
        stall = 0
        it = 0
        last_lp = 0
        zero_frac = float(np.mean(a <= 1e-8 * max(mean_a, 1e-300)))

        while time.monotonic() < end_time:
            if M < best_local_M - 1e-12:
                best_local_M = M
                best_local = a.copy()

            now = time.monotonic()

            if it % 100 == 0:
                zero_frac = float(np.mean(a <= 1e-8 * max(mean_a, 1e-300)))

            # Occasional active-set LP jump.
            if (it - last_lp >= 180 or stall > 400) and (end_time - now) > 90:
                lp_K = max(1, min(160, K))
                lpres = try_active_lp(a, B, M, lp_K)
                last_lp = it
                if lpres is not None:
                    a2, M2 = lpres
                    a = a2
                    B = conv(a)
                    M = float(B.max())
                    stall = 0
                    it += 1
                    continue

            Keff = min(K, L)
            if Keff < L:
                idx = np.argpartition(B, -Keff)[-Keff:]
            else:
                idx = np.arange(L)
            Bidx = B[idx]
            w = make_weights(Bidx, beta)

            nz = w > 1e-14
            if not np.any(nz):
                idx = idx.copy()
                w = np.ones_like(w)
                w /= float(w.size)
                nz = np.ones_like(nz, dtype=bool)

            c = mirror_weighted(a, idx[nz], w[nz])

            mode = "add" if (zero_frac > 0.005 or it % 400 < 60) else "mult"

            if mode == "mult":
                center = c - float(np.dot(a, c) / S)
                alpha = alpha_m
            else:
                center = c - float(c.mean())
                alpha = alpha_a

            rngc = float(center.max() - center.min())
            accepted = False

            if rngc > 1e-14 * max(1.0, mean_a):
                for trial in (alpha, alpha * 0.5, alpha * 0.25):
                    if mode == "mult":
                        rel = -trial * center / rngc
                        np.clip(rel, -5.0, 5.0, out=rel)
                        a2 = a * np.exp(rel)
                    else:
                        step = trial * mean_a / rngc
                        a2 = a - step * center
                        np.maximum(a2, 0.0, out=a2)

                    s2 = float(a2.sum())
                    if not math.isfinite(s2) or s2 <= 1e-300:
                        continue
                    a2 *= S / s2

                    B2 = conv(a2)
                    M2 = float(B2.max())
                    if M2 < M - 1e-11:
                        a = a2
                        B = B2
                        M = M2
                        if mode == "mult":
                            alpha_m = min(0.12, alpha_m * 1.15)
                        else:
                            alpha_a = min(0.12, alpha_a * 1.15)
                        accepted = True
                        stall = 0
                        break

                if not accepted:
                    if mode == "mult":
                        alpha_m = max(0.003, alpha_m * 0.5)
                    else:
                        alpha_a = max(0.003, alpha_a * 0.5)
                    stall += 1

                    if stall == 150:
                        beta = min(40.0, beta * 1.4)
                        K = min(Kmax, max(K + 24, int(K * 1.2)))

                    if stall == 400:
                        last_lp = it - 1000

                    if stall > 2500:
                        pv = perturb(a, 0.12)
                        if pv is not None:
                            a = pv
                            B = conv(a)
                            M = float(B.max())
                            stall = 0
                            beta = 4.0
                            K = min(Kmax, max(Kinit, K // 2))
                            alpha_m = 0.05
                            alpha_a = 0.05

                    if stall > 9000 and (end_time - now) > 400:
                        rv = random_init()
                        if rv is not None:
                            a = rv
                            B = conv(a)
                            M = float(B.max())
                            stall = 0
                            beta = 4.0
                            K = Kinit
                            alpha_m = 0.05
                            alpha_a = 0.05
            else:
                stall += 1
                if stall > 100:
                    pv = perturb(a, 0.2)
                    if pv is not None:
                        a = pv
                        B = conv(a)
                        M = float(B.max())
                        stall = 0
                        beta = 4.0
                        alpha_m = 0.05
                        alpha_a = 0.05

            it += 1

        if M < best_local_M:
            best_local_M = M
            best_local = a.copy()

        return best_local, float(best_local_M)

    optimize_end = hard_end - 3.0
    if time.monotonic() < optimize_end:
        a_im, M_im = improve(best_a, optimize_end)
        if math.isfinite(M_im) and M_im < best_M:
            best_M = M_im
            best_a = a_im.copy()

    def finalize(a):
        a = np.array(a, dtype=np.float64, copy=True)
        if a.size == 0:
            return [1.0, 1.0]
        a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
        np.maximum(a, 0.0, out=a)
        s = float(a.sum())
        if s <= 1e-15:
            return [1.0, 1.0]
        maxv = float(a.max())
        if maxv > 1000.0:
            a *= 1000.0 / maxv
        s = float(a.sum())
        if s < 0.01:
            a *= 0.01 / s
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s < 0.01:
            return [1.0, 1.0]
        return [float(x) for x in a]

    final_list = finalize(best_a)

    try:
        final_score = float(evaluate_sequence(final_list))
    except Exception:
        final_score = np.inf

    if not math.isfinite(final_score):
        final_list = [1.0, 1.0]

    return final_list
