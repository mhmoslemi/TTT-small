def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.reshape(-1)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s

def _score_a(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1: a = a.reshape(-1)
    if a.size == 0: return np.inf
    a = np.where(np.isfinite(a), a, 0.0); a = np.maximum(a,0); 
    s=float(a.sum())
    if s < 1e-12: return np.inf
    n=a.size
    if n == 1:

    L = 1 << (2*n-2).bit_length()
    A=np.fft.rfft(a,n=L)
    q=np.fft.irfft(A*A,n=L)[:2*n-1]
    mx=float(np.max(q))
    if not np.isfinite(mx) or mx < 0: mx=0.0
    return float(2.0*n*mx/(s*s))

import numpy as np
import time


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.reshape(-1)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _score_a(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.reshape(-1)
    if a.size == 0:
        return np.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    n = int(a.size)
    if n == 1:
        return 2.0
    L = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, n=L)
    q = np.fft.irfft(A * A, n=L)[:2 * n - 1]
    mx = float(np.max(q))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * n * mx / (s * s))


def _self_conv_n(a, L, n):
    A = np.fft.rfft(a, n=L)
    q = np.fft.irfft(A * A, n=L)
    return q[:2 * n - 1]


def _corr_n(p, a, L, n):
    P = np.fft.rfft(p, n=L)
    A = np.fft.rfft(a[::-1], n=L)
    c = np.fft.irfft(P * A, n=L)
    return c[n - 1:n + n]


def _exp_norm(loga):
    z = np.where(np.isfinite(loga), loga, -1e300)
    z = z - float(np.max(z))
    a = np.exp(z)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(loga.size, dtype=np.float64) / float(loga.size)
    return a / s


def _upsample(a, m):
    a = _normalize(a)
    m = int(m)
    if m <= 0:
        return np.ones(1, dtype=np.float64)
    if a.size == m:
        return a.copy()
    if a.size == 0:
        return np.ones(m, dtype=np.float64) / float(m)
    x1 = np.linspace(0.0, 1.0, m)
    x0 = np.linspace(0.0, 1.0, a.size)
    out = np.interp(x1, x0, a)
    return _normalize(out)


def _dedupe(inits, max_count=10):
    out = []
    for a in inits:
        b = _normalize(a)
        if b.size == 0 or not np.all(np.isfinite(b)):
            continue
        dup = False
        for c in out:
            if c.size == b.size and float(np.linalg.norm(b - c, ord=1)) < 0.03:
                dup = True
                break
        if not dup:
            out.append(b)
            if len(out) >= max_count:
                break
    return out


def _make_inits(n, parent, rng):
    n = int(n)
    if n <= 0:
        n = 1
    inits = []

    def add(arr):
        try:
            a = _normalize(arr)
            if a.size == n and np.all(np.isfinite(a)):
                inits.append(a)
        except Exception:
            pass

    if parent is not None:

        add(p)
        add(p[::-1])
        if n > 5:
            for sig in (0.8, 1.5, 3.0):
                radius = max(1, int(sig))
                idx = np.arange(-radius, radius + 1)
                k = np.exp(-0.5 * (idx / max(1e-6, sig)) ** 2)
                k = k / max(1e-300, float(k.sum()))
                add(np.convolve(p, k, mode="same"))
        for pw in (0.7, 1.3, 2.0):

        if best_cand_log is not None:
            loga = best_cand_log
            a = _exp_norm(loga)
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 25:
                step = 0.02
                if stagnant > 100:
                    step = 0.08
                if stagnant > 300:
                    step = 0.2
                loga += step * rng.standard_normal(n)
                loga -= float(loga.mean())
                a = _exp_norm(loga)
                stagnant = 0

    return _normalize(best_a), best_score


def _run_for_length(n, inits_supplied, t_start, t_end, rng, record):
    n = int(n)
    if n <= 0 or t_end - time.monotonic() < 3.0:
        return

    inits = []
    for arr in inits_supplied:
        if arr is None:
            continue
        try:
            a = _upsample(arr, n)
            if a is not None and a.size == n:
                inits.append(a)
        except Exception:
            pass

    inits = _dedupe(inits, 8)
    if not inits:
        inits = [np.ones(n, dtype=np.float64) / float(n)]

    scored = []
    for a in inits:
        sc = _score_a(a)
        if np.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda z: z[0])
    if not scored:
        record(np.ones(n, dtype=np.float64) / float(n))
        return

    K = min(5, len(scored))
    weights = np.array([0.45, 0.25, 0.15, 0.10, 0.05][:K], dtype=np.float64)
    weights = weights / float(weights.sum())
    total = max(0.0, t_end - t_start)
    cum = 0.0
    max_etas = 7 if n <= 512 else 5

    for i in range(K):
        now = time.monotonic()
        if now >= t_end - 2.0:
            break
        slice_start = max(now, t_start + total * cum)
        slice_end = t_start + total * (cum + weights[i])
        if slice_end > t_end:
            slice_end = t_end
        if slice_end <= slice_start + 1.5:
            break

        opt_a, opt_sc = _optimize_one(
            scored[i][1],
            slice_start,
            slice_end,
            rng,
            max_etas=max_etas,
        )
        if opt_a is not None:
            record(opt_a)
        cum += weights[i]


def _candidate_lengths(base_n, max_n=4096, limit=2):
    base_n = int(base_n)
    prefers = []
    if base_n < 255:
        prefers.extend([255, 511])
    elif base_n < 511:
        favors = [511, 1023]
        prefers.extend(favors)
    elif base_n < 1023:
        prefers.extend([1023, 2047])
    else:
        prefers.extend([2 * base_n, 4 * base_n])

    out = []
    for L in prefers:
        L = int(min(max_n, max(base_n + 1, L)))
        if base_n < L <= max_n and L not in out:
            out.append(L)
    return out[:limit]


def _try_trim(a):
    a = _normalize(a)
    n = int(a.size)
    if n <= 5:
        return a
    best_a = a.copy()
    best_sc = _score_a(a)
    max_t = 8 if n <= 512 else 4
    for i in range(max_t + 1):
        for j in range(max_t + 1):
            if i + j >= n - 3:
                continue
            cand = a[i:n - j]
            if cand.size < 3:
                continue
            try:
                sc = _score_a(cand)
            except Exception:
                continue
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = sc
                best_a = cand.copy()
    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1300.0
    safety = 25.0 if budget > 100.0 else max(1.0, 0.1 * budget)
    deadline = start + max(1.0, budget - safety)
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    fallback_list = [1.0]
    try:
        parent = None
        n0 = 127
        try:
            p = np.asarray(height_sequence_1, dtype=np.float64)
            if p.ndim != 1:
                p = p.reshape(-1)
            if p.size > 0 and np.all(np.isfinite(p)):
                pp = np.maximum(p, 0.0)
                ss = float(pp.sum())
                if ss > 1e-12:
                    parent = pp / ss
                    n0 = int(p.size)
        except Exception:
            parent = None
            n0 = 127

        if not (5 <= n0 <= 4096):
            main_n = 127
        else:
            main_n = n0
        main_n = int(main_n)

        best_arr = None
        best_score = np.inf

        def record(a):
            nonlocal best_arr, best_score
            try:
                arr = _normalize(a)
                if arr.size == 0 or not np.all(np.isfinite(arr)):
                    return
                sc = _score_a(arr)
                if np.isfinite(sc) and sc < best_score - 1e-13:
                    best_score = sc
                    best_arr = arr.copy()
            except Exception:
                pass

        # Seed with the supplied parent when available, plus diverse analytic starts.
        try:
            inits0 = _make_inits(main_n, parent, rng)
            for a in inits0:
                record(a)
        except Exception:
            pass

        if best_arr is None:
            base = parent if parent is not None else None
            if base is None or base.size == 0:
                base = np.ones(main_n, dtype=np.float64) / float(main_n)
            record(_upsample(base, main_n))
        record(np.array([1.0], dtype=np.float64))

        # Phase 1: focused search at the native/main length.
        rem0 = max(0.0, deadline - start)
        main_end = min(deadline - 5.0, start + 0.60 * rem0)
        if main_end > time.monotonic() + 5.0:
            try:
                inits_main = _make_inits(main_n, parent, rng)
                _run_for_length(main_n, inits_main, time.monotonic(), main_end, rng, record)
            except Exception:
                pass

        # Phase 2: try larger resampled length variants if time remains.
        if best_arr is not None:
            left = deadline - time.monotonic()
            if left > 150.0:
                longs = _candidate_lengths(main_n, max_n=4096, limit=2)
                for L in longs:
                    if time.monotonic() >= deadline - 90.0:
                        break
                    left = deadline - time.monotonic()
                    alloc = min(180.0, 0.35 * max(0.0, left))
                    if left - alloc > 60.0:
                        try:
                            initsL = _make_inits(L, best_arr, rng)
                            _run_for_length(L, initsL, time.monotonic(), time.monotonic() + alloc, rng, record)
                        except Exception:
                            pass

        # Phase 3: final polish of the current best construction at its own length.
        left = deadline - time.monotonic()
        if best_arr is not None and left > 50.0:
            try:
                L = int(best_arr.size)
                base_inits = [best_arr, best_arr[::-1].copy()]
                if L > 1:
                    perturb = best_arr * (1.0 + 0.04 * rng.standard_normal(L))
                    base_inits.append(perturb)
                _run_for_length(L, base_inits, time.monotonic(), max(time.monotonic() + 1.0, deadline - 3.0), rng, record)
            except Exception:
                pass

        # Final exact validation against the supplied evaluator.
        cand_list = []
        if best_arr is not None:
            cand_list.append(best_arr)
            try:
                cand_list.append(_try_trim(best_arr))
            except Exception:
                pass
            if best_arr.size > 12:
                for r in (2, 3, 4):
                    m = best_arr.size // r
                    if m >= 5:
                        try:
                            cand = best_arr[:m * r].reshape(m, r).mean(axis=1)
                            cand_list.append(cand)
                            cand_list.append(_try_trim(cand))
                        except Exception:
                            pass
        if parent is not None:
            cand_list.append(parent)
        cand_list.append(np.array([1.0], dtype=np.float64))

        scored_cands = []
        for arr in cand_list:
            try:
                a = _normalize(arr)
                if a.size == 0:
                    continue
                sc = _score_a(a)
                if np.isfinite(sc):
                    scored_cands.append((sc, a))
            except Exception:
                pass
        scored_cands.sort(key=lambda z: z[0])

        # Avoid repeated expensive exact convolutions; evaluate the top few.
        final_list = None
        final_score = np.inf
        evaluated = 0
        for _, a in scored_cands[:8]:
            if time.monotonic() > deadline - 2.0 or evaluated >= 6:
                break
            try:
                lst = [float(x) for x in a]
                if not lst:
                    continue
                if any(not np.isfinite(x) for x in lst):
                    continue
                sc = float(evaluate_sequence(lst))
                evaluated += 1
                if np.isfinite(sc) and sc < final_score:
                    final_score = sc
                    final_list = lst
            except Exception:
                pass

        if final_list is None:
            final_list = fallback_list
        return final_list
    except Exception:
        return fallback_list
