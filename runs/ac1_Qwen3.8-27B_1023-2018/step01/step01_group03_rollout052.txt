def _norm1(q):
    q = np.asarray(q, dtype=np.float64)
    q = np.maximum(q, 0.0)
    s = q.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(q.size, 1.0 / max(1, q.size))

def _conv_max(q):
    return np.float64(np.convolve(q, q).max())


def _build_conv_A(q):
    n = q.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        idx = np.arange(i_lo, i_hi + 1)
        A[k, idx] = q[k - idx]
    return A


def _lp_step(q, time_limit_s=3.0):
    n = q.size
    b = np.convolve(q, q)
    M = float(b.max())
    if M <= 0 or not np.isfinite(M):
        return None
    A = _build_conv_A(q)
    c = -np.ones(n, dtype=np.float64) - 1e-7 * q
    b_ub = np.full(2 * n - 1, M * (1.0 + 1e-9), dtype=np.float64)
    bounds = [(0.0, 10.0)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_s})
    except Exception:
        return None
    if res is None or not getattr(res, 'success', False):
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = g.sum()
    if s <= 0 or not np.isfinite(s):
        return None
    return g / s


def _grad_step(q, step, n_active=None, alpha=10.0):
    n = q.size
    b = np.convolve(q, q)
    M = float(b.max())
    if M <= 0 or not np.isfinite(M):
        return q
    thresh = 0.80 * M
    active = np.where(b >= thresh)[0]
    na = min(n_active, len(active)) if n_active else len(active)
    if na == 0:
        return q
    # Pick top na by value
    if na < len(active):
        order = np.argsort(-b[active])[:na]
        active = active[order]
    w = np.exp(alpha * (b[active] - M))
    wsum = w.sum()
    if wsum <= 0:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / wsum
    grad = np.zeros(n, dtype=np.float64)
    for kk, ww in zip(active, w):
        kk = int(kk)
        m_lo = max(0, kk - n + 1)
        m_hi = min(n - 1, kk)
        idx = np.arange(m_lo, m_hi + 1)
        grad[idx] += 2.0 * ww * q[kk - idx]
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return q
    cand = q - step * (grad / gn) * np.sqrt(n)
    return _norm1(cand)


def _random_step(q, rng, step=0.1):
    n = q.size
    r = rng.standard_normal(n)
    r -= r.mean()
    rn = np.linalg.norm(r)
    if rn < 1e-14:
        return q
    cand = q + step * (r / rn) * np.sqrt(n)
    return _norm1(cand)

        step = 0.05
        for _ in range(300):
            if time.monotonic() > polish_dl:
                break
            gc = _grad_step(bq, step, n_active=min(20, bn))
            gm = _conv_max(gc)
            if gm < bsc - 1e-12:
                bq = gc
                bsc = gm
                if step < 0.15:
                    step = min(0.15, step * 1.05)
            else:
                step *= 0.85
                if step < 0.003:
                    break
            # Small LP refinement
            if _ % 10 == 5 and bn <= 512:
                g_lp = _lp_step(bq, time_limit_s=2.0)
                if g_lp is not None:
                    for t in (0.2, 0.4, 0.6):
                        cand = _norm1((1 - t) * bq + t * g_lp)
                        cm = _conv_max(cand)
                        if cm < bsc - 1e-12:
                            bq = cand
                            bsc = cm
                            break
        consider([float(v) for v in bq])

    # ── Validate and return ──────────────────────────────────────────────────
    if best_list is None:
        best_list = [float(v) for v in _norm1(q4)]
    try:
        v = float(evaluate_sequence(best_list))
        if not np.isfinite(v):
            best_list = [float(x) for x in _norm1(q4)]
    except Exception:
        best_list = [float(x) for x in _norm1(q4)]

    if time.monotonic() > deadline - 5.0 and len(best_list) > 4:
        fb = [float(x) for x in _norm1(q4)]
        try:
            if np.isfinite(float(evaluate_sequence(fb))):
                best_list = fb
        except Exception:
            best_list = [1.0]

    return [float(x) for x in best_list]
