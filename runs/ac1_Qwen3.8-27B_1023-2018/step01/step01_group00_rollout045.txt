def solve_lp(a, M):
    n = len(a)
    # Build the convolution constraint matrix
    # (a*g)[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[i]*g[k-i]
    # For each k, the coefficient of g[j] is a[k-j] if 0<=k-j<n, else 0
    
    # A_ub is (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum(g)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x

def _subgradient_polish(a, fsize, stop_time, rng):
    """Local polish: subgradient toward the peak lag + random perturbations."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a, fsize)
    if not np.isfinite(best_sc):
        return best_a, best_sc
    stuck = 0
    lr = 0.08

    while time.monotonic() < stop_time:
        a_cur = best_a.copy()
        b = _fft_conv(a_cur, fsize)
        k_star = int(np.argmax(b))

        # Subgradient of (a*a)[k_star] w.r.t. a[j] = 2*a[k_star - j]
        grad = np.zeros(n)
        for j in range(n):
            i = k_star - j
            if 0 <= i < n:
                grad[j] = 2.0 * a_cur[i]

        gmax = float(np.max(np.abs(grad)))
        if gmax > 1e-15:
            a_new = a_cur - lr * grad / gmax
            a_new = np.clip(a_new, 0.0, None)
            sn = float(a_new.sum())
            if sn > 0.01:
                a_new = a_new / sn
                sc = _score(a_new, fsize)
                if np.isfinite(sc) and sc < best_sc - 1e-13:
                    best_a = a_new.copy()
                    best_sc = sc
                    stuck = 0
                    lr = min(lr * 1.1, 0.3)
                    continue
                lr = max(lr * 0.85, 0.01)

        # Random moves
        r = rng.random()
        if r < 0.40:
            z = np.log(np.clip(a_cur, 1e-12, None))
            z += rng.normal(0.0, 0.12, n)
            z -= z.max()
            a_new = np.exp(z)
            a_new = a_new / a_new.sum()
        elif r < 0.65:
            a_new = a_cur.copy()
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            amt = a_cur[i1] * rng.uniform(0.05, 0.40)
            a_new[i1] -= amt
            a_new[i2] += amt
            a_new = np.clip(a_new, 0.0, None)
            sn = a_new.sum()
            if sn > 0.01:
                a_new = a_new / sn
            else:
                a_new = a_cur.copy()
        elif r < 0.80:
            w = int(rng.integers(2, max(3, n // 10)))
            s = int(rng.integers(0, max(1, n - w)))
            a_new = a_cur.copy()
            a_new[s : s + w] *= np.exp(rng.normal(0.0, 0.15, w))
            a_new = np.clip(a_new, 0.0, None)
            sn = a_new.sum()
            if sn > 0.01:
                a_new = a_new / sn
        else:
            a_new = 0.92 * a_cur + 0.08 * (1.0 / n)
            a_new = a_new / a_new.sum()

        sc = _score(a_new, fsize)
        if np.isfinite(sc) and sc < best_sc - 1e-13:
            best_a = a_new.copy()
            best_sc = sc
            stuck = 0
        else:
            stuck += 1
            if stuck > 150:
                break
    return best_a, best_sc


def _try_start(a0, n, fsize, deadline, rng, label=""):
    """Run LP iterations from a starting point, return best (a, score)."""
    a_cur = a0.copy()
    a_cur = np.clip(a_cur, 0.0, None)
    s = a_cur.sum()
    if s < 0.01:
        a_cur = np.ones(n) / n
    else:
        a_cur = a_cur / s

    best_a = a_cur.copy()
    best_sc = _score(a_cur, fsize)
    if not np.isfinite(best_sc):
        return best_a, best_sc

    for it in range(2000):
        if time.monotonic() > deadline:
            break
        a_new, sc = _lp_step(a_cur, fsize)
        if sc < best_sc - 1e-13:
            best_a = a_new.copy()
            best_sc = sc
        a_cur = a_new

        # Periodic perturbation to escape local minima
        if it > 0 and it % 60 == 0:
            a_try = best_a + 0.025 * rng.standard_normal(n)
            a_try = np.clip(a_try, 0.0, None)
            st = a_try.sum()
            if st > 0.01:
                a_try = a_try / st
                stc = _score(a_try, fsize)
                if stc < best_sc + 0.008:
                    a_cur = a_try.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(60.0, float(budget_s))
    deadline = t0 + budget - 6.0

    parent = _get_parent()
    n = max(2, len(parent))
    fsize = next_fast_len(2 * n - 1)

    a0 = np.clip(parent, 0.0, 1000.0).astype(np.float64)
    s0 = a0.sum()
    if s0 < 0.01:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / s0

    best_a = a0.copy()
    best_score = _score(a0, fsize)

    # Validate parent with exact evaluator
    try:
        ps = evaluate_sequence([float(x) for x in a0])
        if np.isfinite(ps) and ps < best_score:
            best_score = ps
    except Exception:
        pass

    rng = np.random.default_rng(seed)

    # Phase 1: LP alternating from parent
    lp_stop = min(deadline - 40.0, t0 + 400.0)
    a1, s1 = _try_start(a0, n, fsize, lp_stop, rng)
    if s1 < best_score:
        best_a, best_score = a1, s1

    # Phase 2: LP from a few perturbed / structured starts
    starts = []
    # Uniform
    starts.append(np.ones(n) / n)
    # Cosine window
    i = np.arange(n, dtype=np.float64)
    cw = 0.5 + 0.5 * np.cos(2.0 * math.pi * i / n)
    starts.append(cw / cw.sum())
    # Raised cosine (sinc-like)
    rc = np.exp(-0.5 * ((i - (n - 1) / 2.0) / (n * 0.25)) ** 2)
    starts.append(rc / rc.sum())
    # Linear ramp
    lr = np.linspace(0.1, 1.0, n)
    starts.append(lr / lr.sum())
    # Random smooth
    for _ in range(3):
        rs = np.abs(rng.normal(0.0, 1.0, n)) + 0.1
        # Smooth
        kernel = np.ones(3) / 3.0
        rs = np.convolve(rs, kernel, "same")
        starts.append(rs / rs.sum())
    # Parent with noise
    for sigma in (0.05, 0.1, 0.2):
        noisy = a0 + sigma * rng.standard_normal(n)
        noisy = np.clip(noisy, 0.0, None)
        sn = noisy.sum()
        if sn > 0.01:
            starts.append(noisy / sn)

    for si, s in enumerate(starts):
        if time.monotonic() > deadline - 20.0:
            break
        alloc = min(60.0, max(5.0, deadline - time.monotonic() - 15.0))
        a_s, sc_s = _try_start(s, n, fsize, time.monotonic() + alloc, rng)
        if sc_s < best_score - 1e-13:
            best_a = a_s.copy()
            best_score = sc_s

    # Phase 3: Polish
    polish_end = deadline - 4.0
    if time.monotonic() < polish_end:
        pa, ps = _subgradient_polish(best_a, fsize, polish_end, rng)
        if np.isfinite(ps) and ps < best_score:
            best_a = pa.copy()
            best_score = ps

    # Final validation with exact evaluator
    try:
        final_sc = evaluate_sequence([float(x) for x in best_a])
    except Exception:
        final_sc = math.inf

    if not np.isfinite(final_sc):
        # Fallback: return a simple valid sequence
        fb = [1.0] / 1.0
        try:
            fs = evaluate_sequence(fb)
            if np.isfinite(fs):
                return fb
        except Exception:
            pass
        return [1.0]

    return [float(x) for x in best_a]
