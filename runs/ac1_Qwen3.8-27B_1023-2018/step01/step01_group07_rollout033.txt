        grad = compute_grad(y, w)

        a_new = _simplex_project(y - lr * grad)
        b_new = compute_b(a_new)
        max_b_new = float(b_new.max())

        if max_b_new <= max_b + 1e-15:
            lr = min(lr * 1.02, 5.0)
            stagnant = 0
        else:
            lr = max(lr * 0.6, 1e-13)
            stagnant += 1

        a_prev = a
        a = a_new
        max_b = max_b_new

        if tau > tau_floor and it % 100 == 0:
            tau = max(tau * 0.95, tau_floor)

        if it % 200 == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

        if stagnant > 300:
            break

    # === Phase 3: Coordinate-descent polishing on active constraints ===
    if time.monotonic() < deadline - 5.0:
        a = best_a.copy()
        n_cd = min(n, 4096)
        step_cd = 0.01
        for idx in range(n_cd):
            if time.monotonic() > deadline - 3.0:
                break
            i = idx % n
            orig = a[i]
            # Try small perturbations
            for delta in [step_cd * orig, -step_cd * orig, 2 * step_cd * orig, -2 * step_cd * orig]:
                a[i] = max(0.0, orig + delta)
                sc = float(evaluate_sequence(a.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                a[i] = orig
            # Restore
            a[i] = orig

    # Final validation
    sc = float(evaluate_sequence(a.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a.copy()

    try:
        bs = float(evaluate_sequence(best_a.tolist()))
        if not np.isfinite(bs):
            best_a = np.ones(n) / n
    except Exception:
        best_a = np.ones(n) / n

    return [float(x) for x in best_a]
