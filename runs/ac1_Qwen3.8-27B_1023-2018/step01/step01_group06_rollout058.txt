    """Compute gradient of max(convolve(a,a)) w.r.t. a."""
    k_star = np.argmax(conv)
    # Gradient: 2*a[k_star - j] for valid j
    # k_star - j must be in [0, n-1], so j in [k_star-(n-1), k_star]
    j_lo = max(0, k_star - (len(a)-1))
    j_hi = min(k_star, len(a)-1)
    grad = np.zeros_like(a)
    idx = np.arange(j_lo, j_hi+1)
    grad[idx] = 2.0 * a[k_star - idx]
    # Fix diagonal: if k_star is even, the middle term should be a[j] not 2*a[j]
    if k_star % 2 == 0 and j_lo <= k_star//2 <= j_hi:
        grad[k_star//2] = a[k_star//2]  # not 2*a
    return grad
