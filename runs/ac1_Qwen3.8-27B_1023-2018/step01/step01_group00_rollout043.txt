import cvxpy as cp

def dual_step(f, M, n):
    """Solve the LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k"""
    g = cp.Variable(n)
    # f*g convolution
    # (f*g)_k = sum_i f_i * g_{k-i} for valid indices
    # This is a linear constraint in g
    constraints = [g >= 0]
    # Add convolution constraints
    for k in range(2*n - 1):
        # (f*g)_k = sum_{i} f_i * g_{k-i} where k-i is in [0, n-1]
        # i ranges from max(0, k-n+1) to min(n-1, k)
        coeffs = []
        for i in range(max(0, k-n+1), min(n-1, k) + 1):
            coeffs.append((f[i], k-i))
        # sum_i f_i * g_{k-i} <= M
        lhs = sum(f[i] * g[k-i] for i in range(max(0, k-n+1), min(n-1, k) + 1))
        constraints.append(lhs <= M)
    
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS)  # or another solver
    return g.value

    """Optimize a[j] while keeping other coordinates fixed."""
    # Compute A_k, B_k, C_k for b_k = A_k * a_j^2 + B_k * a_j + C_k

    # The terms involving a_j:
    # If i = j: a_j * a_{k-j}
    # If k-i = j, i.e., i = k-j: a_{k-j} * a_j
    # If i = j and k-i = j, i.e., k = 2j: a_j * a_j = a_j^2
    
    # So:
    # A_k = 1 if k = 2j and 2j < 2n-1 (i.e., k = 2j)
    #       2 if k = 2j... wait
    # Actually b_k = sum_i a_i * a_{k-i}
    # Terms with a_j:
    #   i=j: a_j * a_{k-j} (valid if 0 <= k-j < n)
    #   i=k-j: a_{k-j} * a_j (valid if 0 <= k-j < n AND k-j != j, i.e., k != 2j)
    #   If k = 2j: both terms give a_j * a_j, so total a_j^2
    #   If k != 2j: terms give a_j * a_{k-j} + a_{k-j} * a_j = 2*a_j*a_{k-j} (if k-j valid and != j)
    #   If k-j not in [0,n-1]: no contribution from a_j

    # b_k = (terms not involving a_j) + (terms involving a_j)

    #   a_j * a_{k-j} (if 0 <= k-j < n and k-j != j)
    #   a_{k-j} * a_j (if 0 <= k-j < n and k-j != j)  [same as above, so 2*a_j*a_{k-j}]
    #   a_j^2 (if k = 2j)

    a_save = a[j]

    """Perform coordinate descent to minimize max_k (a*a)_k."""
    a = a.copy().astype(np.float64)

        return np.ones(n) / n
    a = a / a.sum()
    
    best_bmax = float(_fft_conv(a, fsize).max())
    
    for pass_num in range(max_passes):
        if time.monotonic() > stop_time:
            break
        
        improved = False
        b = _fft_conv(a, fsize)
        bmax = float(b.max())
        
        for j in range(n):
            if time.monotonic() > stop_time:
                break

            a_save = a[j]
            a[j] = 0
            C = _fft_conv(a, fsize)
            a[j] = a_save

            A = np.zeros(2*n-1)
            if 2*j < 2*n-1:
                A[2*j] = 1.0

            k_lo = max(0, k_lo)
            k_hi = min(2*n-2, k_hi)
            for k in range(k_lo, k_hi+1):
                jj = k - j
                if jj != j:
                    B[k] = 2.0 * a[jj]

            # Golden section search

            x_hi = max(a_save * 2.5, 1.0/(3*n))

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if not np.isfinite(s) or s <= 0:
        return np.full(z.shape, 1.0 / len(z))

def _fft_conv(a, fsize):
    F = rfft(a, fsize)

def _grad_descent(a_init, n, fsize, stop_time, max_steps=200000, lr_init=0.5):

    v = np.zeros(n)

    lr = lr_init

            patience += 1

        k_star = int(np.argmax(b))

def _lp_dual_step(f, M, n, stop_time):
    try:
        import cvxpy as cp
    except ImportError:
        return None

    g = cp.Variable(n)
    constraints = [g >= 0]

    C_mat = np.zeros((2 * n - 1, n))
    for i in range(n):
        ks = np.arange(i, min(2 * n - 1, i + n))
        C_mat[ks, i] = f[i]

    constraints.append(C_mat @ g <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iter=50000)
    except Exception:
        return None

    if g.value is None:
        return None
    gv = np.array(g.value, dtype=np.float64).ravel()
    gv = np.maximum(gv, 0.0)
    gs = gv.sum()
    if gs < 1e-10 or not np.isfinite(gs):
        return None
    return gv / gs
