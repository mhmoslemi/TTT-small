from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix

def compute_convolution(a, n):
    return fftconvolve(a, a, mode='full')

def solve_lp(a, M, n):
    # Maximize sum(g) subject to g >= 0, (a*g)[k] <= M for all k
    # Variables: g[0..n-1]
    # c = -1 (minimize -sum(g))
    c = -np.ones(n)
    
    # Build A_ub sparsely
    # A_ub[k, p] = a[p] if max(0, k-n+1) <= p <= min(k, n-1)
    rows = []
    cols = []
    vals = []

        p_lo = max(0, k - n + 1)
        p_hi = min(k, n - 1)
        for p in range(p_lo, p_hi + 1):
            rows.append(k)
            cols.append(p)
            vals.append(a[p])
    
    A_ub = lil_matrix((2*n-1, n))

        p_hi = min(k, n - 1)

    A_ub = A_ub.tocsr()
    b_ub = np.full(2*n-1, M)
