```python
    c = irfft(F * F, fsize)[: (2 * n - 1)]
    c = np.asarray(c, dtype=np.float64)

    b = _fft_convolve(a, n, fsize)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    return float(2.0 * n * b.max() / (s * s))

    """Build the dense convolution matrix A where (A @ g)[k] = (f*g)(k) for k=0..2n-2"""
    # A[k, j] = f(k-j) if 0 <= k-j < n, else 0
    # Equivalently, A[k,j] = f(k-j) if j <= k < j+n
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    # k - j should be in [0, n-1]
    valid = (k_idx - j_idx >= 0) & (k_idx - j_idx < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    # For each valid (k,j), A[k,j] = f[k-j]
    valid_k, valid_j = np.where(valid)
    A[valid_k, valid_j] = f[valid_k - valid_j]
    return A

    """Solve: maximize sum(g) subject to A@g <= M, g >= 0"""
    A = _convolution_matrix(f, n)
    c = -np.ones(n)  # minimize -sum(g)

    M = float(b.max())

    if g is None:

    if sum_g < 1e-12:

    # Rescale g to match sum(f)
    sum_f = float(f.sum())
    g = g * (sum_f / sum_g)
    
    # Update

    f_new = np.clip(f_new, 0, 1000)
```

```python
def _solve_lp(x, M):
    """Solve: max sum(y) s.t. A@y <= M*ones, y >= 0"""
    n = x.size
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = x
    
    c = -np.ones(n)  # minimize -sum(y)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return np.clip(res.x, 0, None)
    return None
```

```python
def _fft_conv(a, fsize):
    n = a.size
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: 2 * n - 1]
    c = np.asarray(c, dtype=np.float64)
    c[~np.isfinite(c)] = 0.0
    c[c < 0.0] = 0.0
    return c


def _maxb(a, fsize):
    return float(_fft_conv(a, fsize).max())


def _exact_score(a):
    n = a.size
    fsize = next_fast_len(2 * n - 1)
    b = _fft_conv(a, fsize)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    return float(2.0 * n * b.max() / (s * s))


def _build_A(x):
    n = x.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = x
    return A


def _solve_lp(x, M):
    n = x.size
    A = _build_A(x)
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0 and res.x is not None:
            y = np.maximum(res.x, 0.0)
            if float(y.sum()) > 1e-15:

    except Exception:
        pass
    return None


def _fw_step(x, M, fsize):
    n = x.size
    y = _solve_lp(x, M)
    if y is None:
        return x, M, False

    sum_x = float(x.sum())
    sum_y = float(y.sum())
    if sum_y < 1e-15 or sum_x < 1e-15:
        return x, M, False

    y = y * (sum_x / sum_y)

    best_t = 0.0
    best_M = M
    for t in (0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
        xn = (1.0 - t) * x + t * y
        mn = _maxb(xn, fsize)

            best_t = t

            xn = (1.0 - t) * x + t * y
            mn = _maxb(xn, fsize)
            if mn < best_M:
                best_M = mn
                best_t = t

        return x, M, False

    x_new = (1.0 - best_t) * x + best_t * y
    x_new = np.maximum(x_new, 0.0)
    return x_new, best_M, True
```
