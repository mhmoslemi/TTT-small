        c = -np.ones(n, dtype=np.float64)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0.0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'time_limit': tl})
            if res.success and res.x is not None:
                g = np.asarray(res.x, dtype=np.float64)
                np.clip(g, 0.0, None, out=g)
                s = float(g.sum())
                if s > 1e-12:
                    return g / s
        except Exception:
            pass
        return None

    def _grad_polish(q, max_it, dl):
        n = q.size
        q = q.copy()
        bq = q.copy()
        bs = _sf(q)
        step = 0.06
        stuck = 0
        for _ in range(max_it):
            if time.monotonic() > dl:
                break
            conv = np.convolve(q, q)
            M = float(conv.max())
            S = float(q.sum())
            sc = 2.0 * n * M / (S * S)
            if sc < bs - 1e-14:
                bs = sc
                bq = q.copy()
                stuck = 0
                if step < 0.12:
                    step = min(0.12, step * 1.02)
            else:
                stuck += 1
                if stuck > 80:
                    break
                step = max(0.001, step * 0.95)

            th = 0.90 * M
            idx = np.where(conv >= th)[0]
            if idx.size == 0:
                idx = np.array([int(np.argmax(conv))])
            if idx.size > 40:
                order = np.argsort(conv[idx])[::-1][:40]
                idx = idx[order]
            w = np.exp((conv[idx] - M) * 12.0)
            ws = float(w.sum())
            if ws < 1e-15:
                break
            w /= ws

            f = np.zeros(n, dtype=np.float64)
            for kk, ww in zip(idx, w):
                kk = int(kk)
                i0 = max(0, kk - n + 1)
                i1 = min(n - 1, kk)
                seg = q[kk - i1: kk - i0 + 1][::-1]
                f[i0: i1 + 1] += ww * seg
            d = M - S * f
            d -= d.mean()
            dn = float(np.linalg.norm(d))
            if dn < 1e-14:
                break
            qn = q + step * d / dn
            np.clip(qn, 0.0, None, out=qn)
            sn = float(qn.sum())
            if sn < 1e-10:
                break
            qn /= sn
            q = qn
        return bq, bs

    def _optimize(q0, dl, nlp, ngrad):
        q = _norm(q0)
        sc = _sf(q)
        for _ in range(nlp):
            if time.monotonic() > dl:
                break
            g = _lp_step(q)
            if g is None:
                break
            ok = False
            for t in (0.3, 0.1, 0.5, 0.2, 0.7, 0.15, 1.0, 0.4):
                qn = _norm((1.0 - t) * q + t * g)
                sn_ = _sf(qn)
                if sn_ < sc - 1e-13:
                    q = qn
                    sc = sn_
                    ok = True
                    break
            if not ok:
                break
        q, sc = _grad_polish(q, ngrad, dl)
        return q, sc

    # ===== MAIN =====
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        consider(parent)
    except Exception:
        pass

    # Phase 1: primary LP+gradient at n=512 from parent
    if time.monotonic() < deadline - 400:
        dl1 = min(deadline - 380, t0 + 400)
        q1 = _resize(parent, 512)
        q1, s1 = _optimize(q1, dl1, nlp=50, ngrad=400)
        consider(q1)

    # Phase 2: n=256 from parent
    if time.monotonic() < deadline - 250:
        dl2 = min(deadline - 230, time.monotonic() + 180)
        q2 = _resize(parent, 256)
        q2, s2 = _optimize(q2, dl2, nlp=60, ngrad=500)
        consider(q2)

    # Phase 3: n=512 diverse starts
    if time.monotonic() < deadline - 180:
        t_arr = np.arange(512, dtype=float)
        starts = []
        for _ in range(2):
            raw = np.ones(512)
            for mm in range(1, 5):
                raw += 0.5 * rng.normal() * np.cos(2 * np.pi * mm * t_arr / 512)
                raw += 0.5 * rng.normal() * np.sin(2 * np.pi * mm * t_arr / 512)
            starts.append(raw)
        starts.append(rng.dirichlet(np.full(512, 0.5)))
        starts.append(rng.dirichlet(np.full(512, 5.0)))
        if parent is not None:
            pz = _resize(parent, 512)
            starts.append(pz + 0.2 * rng.standard_normal(512))

        for s0 in starts:
            if time.monotonic() > deadline - 80:
                break
            dl3 = min(deadline - 70, time.monotonic() + 100)
            q3, _ = _optimize(_norm(s0), dl3, nlp=35, ngrad=250)
            consider(q3)

    # Phase 4: n=1024
    if time.monotonic() < deadline - 80:
        dl4 = min(deadline - 55, time.monotonic() + 100)
        q4 = _resize(parent, 1024)
        q4, _ = _optimize(q4, dl4, nlp=25, ngrad=200)
        consider(q4)

    # Phase 5: final polish
    if best_seq is not None and time.monotonic() < deadline - 3:
        bq = _norm(np.asarray(best_seq, dtype=np.float64))
        bq, _ = _grad_polish(bq, 300, deadline - 3)
        consider(bq)

    # Safety
    if best_seq is None:
        consider([1.0] * 8)
    if best_seq is None:

    try:
        v = float(evaluate_sequence(best_seq))
        if not np.isfinite(v):
            best_seq = [1.0]
    except Exception:
        best_seq = [1.0]

    return [float(x) for x in best_seq]
