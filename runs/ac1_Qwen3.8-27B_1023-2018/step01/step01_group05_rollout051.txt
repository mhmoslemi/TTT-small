def _fft_convolve(a, b, fft_n):
    A = np.fft.rfft(a, fft_n)
    B = np.fft.rfft(b, fft_n)
    return np.fft.irfft(A * B, fft_n)

    n = len(a)
    fft_n = 1 << (2*n - 1).bit_length()
    c = _fft_convolve(a, a, fft_n)[:2*n-1]
    M = c.max()
    S = a.sum()
    score = 2.0 * n * M / (S*S)
    return score, c, M, S

        return np.fft.irfft(A * A, conv_fft_n)[:2 * n - 1]
