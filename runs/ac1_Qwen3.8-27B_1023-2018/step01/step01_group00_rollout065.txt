            ramp = x ** p
            ramp /= ramp.sum()
            mix = float(rng.uniform(0.3, 0.8))
            cand = mix * ramp + (1.0 - mix) * best_a
            s_c = cand.sum()
            if s_c > 0:
                cand /= s_c
        elif r < 0.9:
            x = np.arange(1, n + 1, dtype=np.float64)
            p = float(rng.uniform(0.5, 2.0))
            cand = x ** p
            cand /= cand.sum()
            mix = float(rng.uniform(0.2, 0.6))
            cand = mix * cand + (1.0 - mix) * best_a
            s_c = cand.sum()
            if s_c > 0:
                cand /= s_c
        else:
            cand = best_a.copy()
            perm = rng.permutation(n)
            flip_start = int(rng.integers(0, n - 1))
            cand[flip_start:] = best_a[flip_start:][::-1]
            s_c = cand.sum()
            if s_c > 0:
                cand /= s_c

        cand = np.clip(cand, 0.0, 1000.0)
        s_c = float(cand.sum())
        if s_c < 0.01:
            continue
        b_c = _fft_conv(cand, fsize)
        fs = 2.0 * n * float(b_c.max()) / (s_c * s_c)
        if not np.isfinite(fs) or fs >= best_s - 1e-10:
            stuck += 1
            if stuck % 30 == 0:
                sigma = min(2.0, sigma * 1.3)
            continue
        es = _exact_score(cand)
        if np.isfinite(es) and es < best_s - 1e-10:
            best_a = cand.copy()
            best_s = es
            stuck = 0
            sigma = max(0.2, sigma * 0.85)
        else:
            stuck += 1
    return best_a, best_s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s)) - 8.0
    parent = _get_parent()

    best_seq = None
    best_score = float('inf')

    def _try_update(a, fast_sc):
        nonlocal best_seq, best_score
        if a is None or len(a) == 0:
            return
        a = np.clip(a, 0.0, 1000.0)
        if not np.isfinite(a).all() or a.sum() < 0.01:
            return
        if not np.isfinite(fast_sc) or fast_sc > best_score + 0.1:
            return
        es = _exact_score(a)
        if np.isfinite(es) and es < best_score:
            best_score = es
            best_seq = a.copy()

    # Try raw ramps at various lengths for quick baseline
    lengths = [128, 192, 256, 384, 512, 768, 1024, 1536, 2048, 3072, 4096, 6144, 8192]
    if parent is not None and parent.size > 0 and parent.size not in lengths:
        lengths.append(int(parent.size))
    lengths = sorted(set(l for l in lengths if l >= 16))

    for L in lengths:
        if time.monotonic() > deadline - 15:
            break
        z = _ramp_init(L)
        a = _softmax(z)
        sc = _fast_score(a)
        _try_update(a, sc)
        if best_score < 1.40:
            break

    # Optimized search at selected lengths
    opt_lengths = [256, 384, 512, 768, 1024, 1536, 2048, 3072, 4096]
    if parent is not None and parent.size > 0:
        opt_lengths.insert(0, int(parent.size))
    opt_lengths = sorted(set(l for l in opt_lengths if l >= 64))

    alloc_map = {64: 15, 128: 20, 192: 25, 256: 30, 384: 35, 512: 45,
                 768: 50, 1024: 60, 1536: 70, 2048: 80, 3072: 90, 4096: 100}

    for L in opt_lengths:
        if time.monotonic() > deadline - 30:
            break
        if best_score < 1.38:
            break
        alloc = alloc_map.get(L, 80)
        remaining = max(0.0, deadline - time.monotonic() - 20)
        alloc = min(alloc, remaining)
        if alloc < 5:
            continue
        stop_t = time.monotonic() + alloc
        inits = [
            _ramp_init(L),
            _ramp_init_p(L, 0.7),
            _ramp_init_p(L, 1.5),
            _ramp_init_p(L, 2.0),
            _exp_init(L, 2.0),
            _exp_init(L, 4.0),
            _parent_init(L, parent),
        ]
        rng_local = np.random.default_rng(int(seed) + L)
        inits.append(_ramp_init(L) + rng_local.normal(0, 0.5, L))
        inits.append(_ramp_init(L) + rng_local.normal(0, 1.0, L))

        for ii, z0 in enumerate(inits):
            if time.monotonic() > stop_t - 2:
                break
            sub_stop = min(stop_t - 1, time.monotonic() + max(3.0, (stop_t - time.monotonic()) / max(1, len(inits) - ii)))
            try:
                a_opt, sc_opt = _optimize(L, z0, sub_stop, int(seed) + 1000 * L + 200 * ii, best_score)
                _try_update(a_opt, sc_opt)
            except Exception:
                pass

    # Polish the best
    if best_seq is not None and time.monotonic() < deadline - 5:
        polish_stop = deadline - 3
        try:
            best_seq, best_score = _polish(best_seq, best_score, polish_stop, int(seed) + 99999)
        except Exception:
            pass

    # Final validation
    if best_seq is not None:
        final_sc = _exact_score(best_seq)
        if np.isfinite(final_sc):
            return [float(x) for x in best_seq]

    # Fallback
    a_fallback = _softmax(_ramp_init(192))
    return [float(x) for x in a_fallback]
