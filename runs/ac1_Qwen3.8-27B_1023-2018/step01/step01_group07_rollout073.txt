                hi_pos = np.nonzero(a > 0.0)[0]
            lo_arr = lo[: max(top, 256)]

        try:
            refresh()
        except Exception:
            hi_pos = np.nonzero(a > 1e-10)[0]
            lo_arr = np.arange(n)

        while time.monotonic() < phase_end:
            attempt += 1
            if attempt % 16 == 0 or no > 100:
                try:
                    refresh()
                except Exception:
                    pass

            if len(hi_pos) == 0:
                hi_pos = np.nonzero(a > 1e-12)[0]
            if len(lo_arr) == 0:
                lo_arr = np.arange(n)
            if len(hi_pos) == 0:
                break

            q = int(hi_pos[rng.integers(len(hi_pos))])
            p = int(lo_arr[rng.integers(len(lo_arr))])
            if p == q:
                if len(lo_arr) > 1:
                    p = int(lo_arr[rng.integers(len(lo_arr))])
                if p == q:
                    p = (q + 1) % n
                if p == q:
                    continue

            Ltmp.fill(0.0)
            Ltmp[p:p + n] += a
            Ltmp[q:q + n] -= a
            Ltmp *= 2.0

            best_m = M
            best_delta = 0.0
            for frac in frac_list:
                delta = frac * a[q]
                if delta <= 1e-300:
                    continue
                tmp = b + delta * Ltmp
                d2 = delta * delta
                i1 = 2 * p
                i2 = 2 * q
                ip = p + q
                tmp[i1] += d2
                tmp[i2] += d2
                tmp[ip] -= 2.0 * d2
                mval = float(tmp.max())
                if mval < best_m - 1e-14:
                    best_m = mval
                    best_delta = delta

            if best_delta > 0.0 and best_m < M * (1.0 - 1e-12):
                delta = best_delta
                a[p] += delta
                a[q] -= delta
                if a[q] < 0.0:
                    a[q] = 0.0

                b += delta * Ltmp
                d2 = delta * delta
                b[2 * p] += d2
                b[2 * q] += d2
                b[p + q] -= 2.0 * d2

                M = float(b.max())
                sc = 2.0 * n * M
                if np.isfinite(sc) and sc < best_local_score - 1e-13:
                    best_local_score = float(sc)
                    best_local = a.copy()

                no = 0
                upds += 1
                top = min(int(top * 1.15), 8192)

                if upds % 80 == 0:
                    a = norm_pos(a)
                    b, fa = b_from_a(a)
                    M = float(b.max())
                    sc = 2.0 * n * M
                    if np.isfinite(sc) and sc < best_local_score - 1e-13:
                        best_local_score = float(sc)
                        best_local = a.copy()
            else:
                no += 1
                if no == 50:
                    top = min(top * 2, 16384)
                    beta = min(beta * 1.25, 1e10)
                if no == 250:
                    sigma = 0.01 + 0.01 * (int(no // 250) % 3)
                    a = norm_pos(a * np.exp(rng.normal(0.0, sigma, n)))
                    b, fa = b_from_a(a)
                    M = float(b.max())
                    sc = 2.0 * n * M
                    if np.isfinite(sc) and sc < best_local_score - 1e-13:
                        best_local_score = float(sc)
                        best_local = a.copy()
                    no = 0
                    top = 512
                    beta = float(np.clip(30.0 / max(M, 1e-13), 5e3, 1e10))
                    try:
                        refresh()
                    except Exception:
                        pass
                if no > 2500:
                    p2 = int(rng.integers(n))
                    q2 = int(rng.integers(n))
                    if p2 != q2 and a[q2] > 1e-12:
                        fr = float(rng.uniform(0.02, 0.5))
                        a[p2] += fr * a[q2]
                        a[q2] *= (1.0 - fr)
                    a = norm_pos(a)
                    b, fa = b_from_a(a)
                    M = float(b.max())
                    no = 0

        return best_local, float(best_local_score)

    def lbfgs_phase(a_init, phase_end, seed_phase):
        if minimize is None:
            return a_init, score_of(a_init)

        a = norm_pos(a_init)
        b, _ = b_from_a(a)
        M = float(b.max())
        curr = 2.0 * n * M
        best_local = a.copy()
        best_local_score = curr
        rng = np.random.default_rng(seed_phase)

        y = np.log(np.maximum(a, 1e-300))
        y -= y.mean()
        beta = float(np.clip(50.0 / max(M, 1e-13), 1e4, 1e9))
        loops = 0
        improved = False

        while time.monotonic() < phase_end and loops < 25:
            loops += 1
            beta = min(beta * 1.35, 5e10)

            def fgy(yy):
                aa = softmax(yy)
                b2, fa2 = b_from_a(aa, None)
                M2 = float(b2.max())
                shifted = beta * (b2 - M2)
                shifted = np.minimum(shifted, 0.0)
                e = np.exp(shifted)
                se = e.sum()
                if se <= 1e-300 or not np.isfinite(se):
                    w2 = np.zeros(b_len, dtype=np.float64)
                    w2[int(np.argmax(b2))] = 1.0
                    logse = 0.0
                else:
                    w2 = e / se
                    logse = float(np.log(se))
                val = float(2.0 * n * (M2 + logse / beta))
                c = coeffs(aa, w2, fa2)
                ga = 2.0 * c
                gy = ga - aa * np.dot(ga, aa)
                gy -= gy.mean()
                if not np.all(np.isfinite(gy)):
                    gy = np.zeros(n, dtype=np.float64)
                return val, gy

            maxiter = 35 if (phase_end - time.monotonic()) > 120.0 else 12
            res = None
            try:
                res = minimize(
                    fgy,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter, "maxls": 8, "disp": False},
                )
            except Exception:
                res = None

            if res is not None and hasattr(res, "x") and np.all(np.isfinite(res.x)):
                a2 = softmax(res.x)
                sc = score_of(a2)
                if np.isfinite(sc) and sc < best_local_score - 1e-14:
                    best_local_score = float(sc)
                    best_local = a2.copy()
                    y = np.asarray(res.x, dtype=np.float64).copy()
                    a = a2.copy()
                    curr = float(sc)
                    improved = True
            else:
                improved = False

            if time.monotonic() >= phase_end - 15.0:
                break

            a = norm_pos(best_local)
            y = np.log(np.maximum(a, 1e-300))
            y -= y.mean()
            if not improved or loops % 3 == 0:
                y += rng.normal(0.0, 0.025, n)
                y -= y.mean()

        return best_local, float(best_local_score)

    # ------------------------------------------------------------------ schedule ----
    phases = [
        (pair_phase, 0.07),
        (adam_phase, 0.20),
        (lp_phase, 0.16),
        (spectral_phase, 0.06),
        (pair_phase, 0.32),
        (lbfgs_phase, 0.19),
    ]

    total_work = max(1.0, work_end - start)
    phase_seed = int(seed) + 5000

    for i, (func, frac) in enumerate(phases):
        if time.monotonic() >= work_end:
            break
        rem = max(0.0, work_end - time.monotonic())
        if rem <= 2.0:
            break
        dur = min(frac * total_work, rem)
        if dur < 3.0:
            dur = rem
        phase_end = min(work_end, time.monotonic() + dur)

        try:
            new_a, new_score = func(best_a, phase_end, phase_seed + 1000 * (i + 1))
            new_a = norm_pos(new_a)
            new_score = float(new_score)

            # Final exact verification before accepting a phase candidate.
            try:
                exact = float(evaluate_sequence([float(x) for x in new_a]))
                if np.isfinite(exact):
                    new_score = exact
            except Exception:
                pass

            if np.isfinite(new_score) and new_score < best_score - 1e-14:
                best_score = new_score
                best_a = new_a.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------ finalize ----
    try:
        best_a = norm_pos(best_a)
    except Exception:
        best_a = uniform.copy()

    final_list = [float(x) for x in best_a]

    # If anything went wrong, make sure the returned object is a valid nonempty list.
    try:
        sc = float(evaluate_sequence(final_list))
        if not np.isfinite(sc):
            final_list = [float(x) for x in uniform]
    except Exception:
        if not final_list or any(not np.isfinite(float(x)) for x in final_list) or all(float(x) == 0.0 for x in final_list):
            final_list = [float(x) for x in uniform]

    return final_list
