import numpy as np

def _self_conv(a, fft_n):
    """Full linear self-convolution via FFT."""
    A = np.fft.rfft(a, fft_n)
    return np.fft.irfft(A * A, fft_n)[:2 * len(a) - 1]


def _cross_corr(w, a, fft_n):
    """Compute c[j] = sum_m a[m]*w[j+m] for j=0..n-1 via FFT."""
    n = len(a)
    w_rev = w[::-1]
    W = np.fft.rfft(w_rev, fft_n)
    A = np.fft.rfft(a, fft_n)
    conv = np.fft.irfft(W * A, fft_n)
    return conv[n - 1: 2 * n - 1][::-1]


def _score(a, B):
    """Compute 2*n*max(B)/sum(a)^2."""
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * len(a) * B.max() / (S * S)
