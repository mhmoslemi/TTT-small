        if grad_norm < 1e-14:
            break

        # Direction (unit norm)
        direction = -grad / grad_norm

        # Limit step to preserve nonnegativity
        pos_mask = direction > 1e-15
        if pos_mask.any():
            max_step = 0.85 * np.min(a[pos_mask] / direction[pos_mask])
            alpha = min(step, max_step)
        else:
            alpha = step

        if alpha <= 0:
            alpha = step * 0.01

        # Update
        a_new = a + alpha * direction
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new > 0.01:
            a = a_new * (n / S_new)
        else:
            break

        # Track best (every 50 iterations to save time)
        if iter_count % 50 == 0:
            A2 = np.fft.rfft(a, fft_n)
            B2 = np.fft.irfft(A2 * A2, fft_n)[:2 * n - 1]
            S2 = a.sum()
            sc = 2.0 * n * B2.max() / (S2 * S2)
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                stagnation = 0
            else:
                stagnation += 50

        # Stagnation handling
        if stagnation > 3000 and iter_count % 100 == 0:
            # Perturb and reduce beta to escape
            pert = rng.normal(0.0, 0.05, n)
            a = a + pert
            a = np.maximum(a, 0.0)
            a *= n / a.sum()
            beta = max(beta * 0.5, 10.0)
            stagnation = 0

    # Polish phase: use exact subgradient with very small steps
    if time.monotonic() < deadline - 5.0:
        polish_deadline = time.monotonic() + min(60.0, deadline - time.monotonic() - 3.0)
        a_polish = best.copy()
        beta_polish = 5000.0
        step_polish = 0.01

        for _ in range(50000):
            if time.monotonic() > polish_deadline:
                break

            A = np.fft.rfft(a_polish, fft_n)
            B = np.fft.irfft(A * A, fft_n)[:2 * n - 1]
            S = a_polish.sum()
            M = B.max()

            B_shift = B - M
            B_clip = np.maximum(B_shift, -800.0 / beta_polish)
            exp_B = np.exp(beta_polish * B_clip)
            exp_sum = exp_B.sum()
            if exp_sum < 1e-300:
                w = np.zeros(2 * n - 1, dtype=np.float64)
                w[int(np.argmax(B))] = 1.0
            else:
                w = exp_B / exp_sum

            w_rev = w[::-1]
            W_fft = np.fft.rfft(w_rev, fft_n)
            conv_grad = np.fft.irfft(W_fft * A, fft_n)
            grad = conv_grad[n - 1:2 * n - 1][::-1].copy()
            grad -= grad.mean()

            grad_norm = np.sqrt(np.dot(grad, grad))
            if grad_norm < 1e-14:
                break

            direction = -grad / grad_norm
            pos_mask = direction > 1e-15
            if pos_mask.any():
                max_step = 0.8 * np.min(a_polish[pos_mask] / direction[pos_mask])
                alpha = min(step_polish, max_step)
            else:
                alpha = step_polish

            if alpha <= 0:
                break

            a_polish = a_polish + alpha * direction
            a_polish = np.maximum(a_polish, 0.0)
            S_pol = a_polish.sum()
            if S_pol > 0.01:
                a_polish *= n / S_pol

            if _ % 100 == 0:
                A2 = np.fft.rfft(a_polish, fft_n)
                B2 = np.fft.irfft(A2 * A2, fft_n)[:2 * n - 1]
                S2 = a_polish.sum()
                sc = 2.0 * n * B2.max() / (S2 * S2)
                if sc < best_score:
                    best_score = sc
                    best = a_polish.copy()

            step_polish *= 0.999

    # Final exact evaluation
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = np.ones(n, dtype=np.float64)
        final_sc = float(evaluate_sequence(best.tolist()))
    else:
        best_score = final_sc

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best]
    final_check = float(evaluate_sequence(result))
    if not np.isfinite(final_check):
        result = [1.0] * n

    return result
