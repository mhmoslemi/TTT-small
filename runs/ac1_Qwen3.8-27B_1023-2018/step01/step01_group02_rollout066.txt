            lr = min(0.25, lr * 1.01)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr = max(0.02, lr * 0.7)

        if no_improve > gap_lim:
            best = best + 0.03 * rng.standard_normal(n)
            best = project_simplex(best)
            best_sc = compute_score(best)
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            adam_t = 0
            no_improve = 0
            gap_lim = int(gap_lim * 1.3)
            lr = 0.05

    # Phase 3: Coordinate descent
    p3_end = t_start + 0.87 * (deadline - t_start)
    best = project_simplex(best)
    best_sc = compute_score(best)

    cd_round = 0
    while time.monotonic() < p3_end:
        cd_round += 1
        improved = False

        for k in range(n):
            if time.monotonic() >= p3_end:
                break
            old = best[k]
            cands_vals = [old * 0.4, old * 0.6, old * 0.8, old * 1.2, old * 1.4, old * 1.8, 0.0,
                          old + 0.008 * rng.standard_normal(3)]
            for cv in cands_vals:
                if cv < 0.0:
                    continue
                trial = best.copy()
                trial[k] = cv
                trial = project_simplex(trial)
                sc = compute_score(trial)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best = trial
                    improved = True
                    old = cv

        if not improved and cd_round > 40:
            best = best + 0.004 * rng.standard_normal(n)
            best = project_simplex(best)
            best_sc = compute_score(best)

    # Phase 4: Random local search
    while time.monotonic() < deadline - 5.0:
        trial = best.copy()
        nb = int(rng.integers(1, 4))
        for _i in range(nb):
            idx = int(rng.integers(0, n))
            trial[idx] *= np.exp(rng.normal(0.0, 0.10))
        if n > 6 and rng.random() < 0.25:
            off = int(rng.integers(-2, 3))
            trial = 0.5 * (trial + np.roll(trial, off))
        trial = project_simplex(trial)
        sc = compute_score(trial)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best = trial

    # Final validation
    best = np.where(np.isfinite(best), best, 0.0)
    best = np.maximum(best, 0.0)
    best = np.minimum(best, 1000.0)

    lst = [float(x) for x in best]
    if not lst:
        return [1.0]

    final_sc = evaluate_sequence(lst)
    if not np.isfinite(final_sc):
        lst = [float(max(0.0, min(1000.0, x))) for x in parent]
        if not lst:
            return [1.0]
        final_sc = evaluate_sequence(lst)
        if not np.isfinite(final_sc):
            return [1.0]

    return lst
