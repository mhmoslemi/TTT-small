        ar = np.maximum(ar, 1e-10)
        ar = ar / ar.sum()
        inits.append(ar)

    x = np.linspace(0.0, 1.0, n)
    for pw in (0.3, 0.7, 1.5, 2.5):
        a_s = 1.0 + 0.6 * x ** pw + 0.3 * (1.0 - x) ** pw + 1e-10
        inits.append(a_s / a_s.sum())

    for f in (1, 2, 3):
        ph = rng.uniform(0, 2 * np.pi)
        a_h = 1.0 + 0.4 * np.sin(2 * np.pi * f * x + ph) + 1e-10
        inits.append(a_h / a_h.sum())

    inits.append(np.ones(n) / n * (1.0 + 0.3 * rng.standard_normal(n)))
    inits[-1] = np.maximum(inits[-1], 1e-10)
    inits[-1] = inits[-1] / inits[-1].sum()

    # Main optimization
    for init in inits:
        if time.monotonic() > deadline - 10:
            break

        a = init.copy()
        m_state = np.zeros(n)
        v_state = np.zeros(n)
        t_adam = 0
        lr = 0.4
        no_improve = 0
        it = 0

        while time.monotonic() < deadline - 10:
            it += 1
            t_adam += 1

            c = np.convolve(a, a)
            M = float(np.max(c))
            S = float(np.sum(a))
            if S < 1e-12 or not np.isfinite(S):
                a = init.copy()
                m_state = np.zeros(n)
                v_state = np.zeros(n)
                t_adam = 0
                no_improve = 0
                continue

            score = 2.0 * n * M / (S * S)

            if np.isfinite(score) and score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.003, 2.0)
            else:
                no_improve += 1

            # Smooth max distribution
            beta = 500.0 + 5000.0 * min(1.0, t_adam / 3000.0)
            z = beta * (c - M)
            p = np.exp(np.minimum(z, 700.0))
            psum = p.sum()
            if psum < 1e-300 or not np.isfinite(psum):
                p = np.zeros(len(c))
                p[int(np.argmax(c))] = 1.0
            else:
                p = p / psum

            # Gradient
            r = _compute_r(a, p)
            grad = (4.0 * n / (S * S)) * (r - M / S)
            grad = np.where(np.isfinite(grad), grad, 0.0)

            # Adam step
            a, m_state, v_state = _adam_update(a, grad, m_state, v_state, t_adam, lr)
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new < 1e-12:
                a = init.copy()
                m_state = np.zeros(n)
                v_state = np.zeros(n)
                t_adam = 0
                no_improve = 0
                continue
            a = a / s_new

            # Periodic LP refinement
            if it % 200 == 0 and time.monotonic() < deadline - 30:
                try:
                    c_now = np.convolve(a, a)
                    M_now = float(np.max(c_now))
                    b_star = _lp_solve(a, M_now)
                    if b_star is not None:
                        for t_mix in (0.3, 0.5, 0.8, 1.0):
                            a_mix = (1.0 - t_mix) * a + t_mix * b_star
                            a_mix, best_a, best_score, improved = _try_candidate(
                                a_mix, best_a, best_score, n
                            ) if False else (None, best_a, best_score, False)
                            # Manual check
                            a_mix = np.maximum(a_mix, 0.0)
                            S_mix = a_mix.sum()
                            if S_mix > 1e-12:
                                a_mix = a_mix / S_mix
                                sc_mix = _score_array(a_mix)
                                if np.isfinite(sc_mix) and sc_mix < best_score - 1e-14:
                                    best_score = sc_mix
                                    best_a = a_mix.copy()
                                    a = a_mix.copy()
                                    no_improve = 0
                                    break
                except Exception:
                    pass

            # Restart if stuck
            if no_improve > 600:
                pert_scale = 0.1 + 0.3 * rng.random()
                a = best_a * (1.0 + pert_scale * rng.standard_normal(n))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                m_state = np.zeros(n)
                v_state = np.zeros(n)
                t_adam = 0
                lr = max(0.15, lr * 0.55)
                no_improve = 0
                it = 0

            # Check deadline
            if time.monotonic() > deadline - 10:
                break

    # Final random polishing
    polish_end = min(deadline - 3, t0 + budget_s - safety + 10)
    trials = 0
    while time.monotonic() < polish_end and trials < 200:
        trials += 1
        a_pol = best_a.copy()
        nb = int(rng.integers(1, 5))
        for _ in range(nb):
            lo = max(2, n // 200)
            hi = max(lo + 1, min(n, max(4, n // 6)))
            width = int(rng.integers(lo, hi + 1))
            width = min(width, n)
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            a_pol[start:start + width] *= factor
        if n > 6 and rng.random() < 0.3:
            off = int(rng.integers(-n // 30, n // 30 + 1))
            a_pol = 0.7 * a_pol + 0.3 * np.roll(a_pol, off)
        a_pol = np.maximum(a_pol, 0.0)
        S_pol = a_pol.sum()
        if S_pol > 1e-12:
            a_pol = a_pol / S_pol
            sc_pol = _score_array(a_pol)
            if np.isfinite(sc_pol) and sc_pol < best_score - 1e-14:
                best_score = sc_pol
                best_a = a_pol.copy()

    # Final LP polish on best
    try:
        if time.monotonic() < deadline - 3:
            S_best = float(np.sum(best_a))
            if S_best > 1e-12:
                best_a_norm = best_a / S_best
                c_fin = np.convolve(best_a_norm, best_a_norm)
                M_fin = float(np.max(c_fin))
                b_fin = _lp_solve(best_a_norm, M_fin)
                if b_fin is not None:
                    for t_mix in (0.2, 0.5, 0.8, 1.0):
                        a_f = (1 - t_mix) * best_a_norm + t_mix * b_fin
                        a_f = np.maximum(a_f, 0.0)
                        S_f = a_f.sum()
                        if S_f > 1e-12:
                            a_f = a_f / S_f
                            sc_f = _score_array(a_f)
                            if np.isfinite(sc_f) and sc_f < best_score - 1e-14:
                                best_score = sc_f
                                best_a = a_f.copy()
    except Exception:
        pass

    # Final validation with the exact evaluator
    final_list = [float(max(0.0, x)) for x in best_a]
    sc_final = evaluate_sequence(final_list)
    if not np.isfinite(sc_final):
        final_list = [0.5, 0.1, 0.4]

    return final_list
