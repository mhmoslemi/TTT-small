    except Exception:
        return None
    if res is None or getattr(res, 'x', None) is None:
        return None
    g = np.asarray(res.x, dtype=np.float64).ravel()
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    g = np.clip(g, 0.0, None)
    sg = float(g.sum())
    if sg < 1e-15:
        return None
    return g / sg


def _lp_search(a, deadline, max_iters=2000, verbose=False):
    """Iterative LP-based optimization. Returns best sequence found."""
    a = _normalize(a.copy())
    n = a.size
    local_score = _score(a)
    a = a.copy()
    
    # Precompute A once (it changes when a changes, but we rebuild each iter)
    stagnant = 0
    for it in range(max_iters):
        if time.monotonic() >= deadline - 3:
            break
        
        b = _convolve(a, a)
        M = float(b.max())
        S = float(a.sum())
        if not math.isfinite(M) or M <= 0:
            break
        
        # Solve LP
        g = _lp_iteration(a, M, max_time=2.0)
        if g is None:
            stagnant += 2
            if stagnant > 10:
                break
            # Perturb
            a = _normalize(a * np.exp(0.05 * np.random.default_rng(it).standard_normal(n)))
            continue
        
        # Try various step sizes
        best_sc = local_score
        best_cand = a.copy()
        
        # Geometric step sizes
        for t in (1.0, 0.85, 0.7, 0.55, 0.4, 0.3, 0.2, 0.15, 0.1):
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0.0, None)
            sc = _score(cand)
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_cand = cand.copy()
        
        # Also try the LP solution directly
        sc_direct = _score(g)
        if math.isfinite(sc_direct) and sc_direct < best_sc:
            best_sc = sc_direct
            best_cand = g.copy()
        
        if best_sc < local_score * (1.0 - 1e-10):
            a = _normalize(best_cand)
            local_score = best_sc
            stagnant = 0
        else:
            stagnant += 1
            if stagnant % 5 == 0:
                # Perturb to escape
                rng_p = np.random.default_rng(1000 + it)
                scale = 0.03 + 0.02 * stagnant
                a_new = a * np.exp(scale * rng_p.standard_normal(n))
                a_new = np.clip(a_new, 0.0, None)
                sc_new = _score(a_new)
                if math.isfinite(sc_new) and sc_new < local_score:
                    a = _normalize(a_new)
                    local_score = sc_new
                    stagnant = 0
            if stagnant > 30:
                break
    
    return a, local_score


def _subgradient_polish(a, deadline, max_iters=5000):
    """Subgradient method for final polishing."""
    a = _normalize(a.copy())
    n = a.size
    local_score = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() >= deadline - 3:
            break
        
        S = float(a.sum())
        if S < 1e-15:
            break
        b = _convolve(a, a)
        idx = int(np.argmax(b))
        M = float(b[idx])
        if not math.isfinite(M):
            break
        
        # Subgradient of max(b) w.r.t. a
        # grad[m] = 2*a[idx-m] if 0 <= idx-m < n else 0
        grad = np.zeros(n, dtype=np.float64)
        lo = max(0, idx - n + 1)
        hi = min(n - 1, idx)
        if lo <= hi:
            grad[lo:hi + 1] = 2.0 * a[idx - (np.arange(lo, hi + 1))]
        
        # Full subgradient of score = 2n*max(b)/S^2
        # d(score)/da[m] = 2n*grad[m]/S^2 - 2n*M*2/S^3
        score_grad = (2.0 * n / (S * S)) * grad - (4.0 * n * M / (S * S * S))
        
        # Adaptive step size
        lr = 0.01 / (1.0 + 0.01 * it)
        
        # Try a few step sizes
        best_sc = local_score
        best_cand = a.copy()
        for factor in (1.0, 0.5, 2.0, 0.25, 4.0):
            lr_i = lr * factor
            cand = a - lr_i * score_grad
            cand = np.clip(cand, 0.0, None)
            cs = float(cand.sum())
            if cs < 1e-15:
                continue
            cand = cand / cs
            sc = _score(cand)
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_cand = cand.copy()
        
        if best_sc < local_score * (1.0 - 1e-12):
            a = best_cand
            local_score = best_sc
        else:
            # Decay step size
            lr *= 0.8
    
    return a, local_score


def _make_initializations(n, rng, parent):
    """Generate diverse initializations for length n."""
    inits = []
    
    if parent is not None and len(parent) > 1:
        p = _normalize(parent)
        if n == len(p):
            inits.append(p.copy())
        else:
            old_x = np.linspace(0, 1, len(p))
            new_x = np.linspace(0, 1, n)
            g = np.interp(new_x, old_x, p)
            inits.append(_normalize(g))
    
    # Uniform
    inits.append(np.ones(n, dtype=np.float64) / n)
    
    # Triangular
    x = np.linspace(0, 1, n)
    tri = 1.0 - np.abs(2.0 * x - 1.0)
    inits.append(_normalize(tri))
    
    # Parabolic
    par = 1.0 - (2.0 * x - 1.0) ** 2
    par = np.clip(par, 0.01, None)
    inits.append(_normalize(par))
    
    # Exponential ramps
    for c in (0.5, 1.0, 2.0, 3.0):
        exp_ramp = np.exp(c * (x - 0.5))
        inits.append(_normalize(exp_ramp))
    
    # Sine-modulated
    for k in (1, 2, 3):
        for amp in (0.3, 0.5, 0.7):
            s = 1.0 + amp * np.sin(2.0 * np.pi * k * x)
            s = np.clip(s, 0.01, None)
            inits.append(_normalize(s))
    
    # Random Dirichlet
    for alpha in (0.5, 1.0, 2.0):
        try:
            d = rng.dirichlet(np.full(n, alpha))
            inits.append(d)
        except Exception:
            pass
    
    # Smooth random
    for _ in range(3):
        v = np.cumsum(rng.standard_normal(n))
        v = v - v.min() + 0.1
        inits.append(_normalize(v))
    
    # Sort by score
    scored = []
    for a in inits:
        sc = _score(a)
        if math.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda z: z[0])
    return [a for _, a in scored]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    hard_start = time.monotonic()
    try:
        remaining = max(0.0, float(budget_s))
    except Exception:
        remaining = 1300.0
    deadline = hard_start + remaining - 15.0  # 15s headroom

    best = [2.0, np.array([1.0], dtype=np.float64)]

    # Try parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 0:
                sc = evaluate_sequence([float(x) for x in parent])
                if math.isfinite(sc) and sc < float(best[0]):
                    best[0] = float(sc)
                    best[1] = parent.copy()
    except Exception:
        parent = None

    if parent is not None:
        _update_best(parent, best)

    # Main LP search on parent length
    n_parent = 96 if parent is None else len(parent)
    
    # Phase 1: LP search on parent length
    if time.monotonic() < deadline - 100:
        inits = _make_initializations(n_parent, rng, parent)
        a_best_local = None
        best_local_sc = float('inf')
        
        for i, a0 in enumerate(inits[:8]):
            phase_deadline = min(deadline - 30, hard_start + remaining * 0.4)
            if time.monotonic() >= phase_deadline:
                break
            a_res, sc_res = _lp_search(a0, phase_deadline, max_iters=3000)
            if sc_res < best_local_sc:
                best_local_sc = sc_res
                a_best_local = a_res
            _update_best(a_res, best)
        
        # Continue from best found
        if a_best_local is not None and time.monotonic() < deadline - 80:
            a_res, sc_res = _lp_search(a_best_local, min(deadline - 30, time.monotonic() + 200), max_iters=5000)
            if sc_res < best_local_sc:
                best_local_sc = sc_res
                a_best_local = a_res
            _update_best(a_res, best)

    # Phase 2: Try other lengths
    other_lengths = [n_parent - 16, n_parent + 16, n_parent + 32, n_parent // 2, n_parent * 2]
    other_lengths = [l for l in other_lengths if 8 <= l <= 512 and l != n_parent]
    other_lengths = sorted(set(other_lengths))
    
    for n_try in other_lengths[:4]:
        if time.monotonic() >= deadline - 60:
            break
        inits = _make_initializations(n_try, rng, parent)
        for a0 in inits[:4]:
            if time.monotonic() >= deadline - 50:
                break
            a_res, sc_res = _lp_search(a0, min(deadline - 40, time.monotonic() + 60), max_iters=1500)
            _update_best(a_res, best)

    # Phase 3: Subgradient polishing on the best so far
    if time.monotonic() < deadline - 20:
        a_cur = best[1].copy()
        a_pol, sc_pol = _subgradient_polish(a_cur, deadline, max_iters=10000)
        _update_best(a_pol, best)

    # Phase 4: One more LP round from the polished result
    if time.monotonic() < deadline - 15:
        a_cur = _normalize(best[1].copy())
        a_res, sc_res = _lp_search(a_cur, min(deadline - 5, time.monotonic() + 30), max_iters=2000)
        _update_best(a_res, best)

    # Final validation
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.array([1.0], dtype=np.float64)
    s = float(final.sum())
    if s <= 1e-12:
        final = np.ones_like(final, dtype=np.float64)
        s = float(final.size)
    final = final / s
    final = np.clip(final, 0.0, 1000.0)
    
    # Verify with exact evaluator
    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc):
        final = np.array([1.0, 1.0], dtype=np.float64)
    
    _update_best(final, best)
    final = np.asarray(best[1], dtype=np.float64).ravel()
    
    # Ensure sum >= 0.01 for evaluator
    s = float(final.sum())
    if s < 0.01:
        final = final / s * 0.02 if s > 0 else np.full_like(final, 0.02 / final.size)
    
    return [float(x) for x in final]
