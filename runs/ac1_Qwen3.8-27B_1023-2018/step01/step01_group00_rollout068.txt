import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size == 0 or not np.isfinite(p).all():
            p = np.ones(192)
    except Exception:
        p = np.ones(192)
    s = p.sum()
    if s < 0.01:
        p = np.ones(192)
    return p / p.sum()


def _fft_conv(a, n, fs):
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    np.maximum(c, 0, out=c)
    c[~np.isfinite(c)] = 0.0
    return c


def _eval_fast(a):
    n = a.size
    if n < 1:
        return math.inf
    s = a.sum()
    if s < 0.01:
        return math.inf
    a_n = a / s
    fs = next_fast_len(2 * n - 1)
    b = _fft_conv(a_n, n, fs)
    return 2.0 * n * b.max()


def _proj_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if rho.size == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _soft_lse(b, tau):
    m = b.max()
    x = (b - m) / tau
    ex = np.exp(x)
    s = ex.sum()
    if s <= 0 or not np.isfinite(s):
        p = np.zeros_like(b)
        p[b == m] = 1.0 / np.sum(b == m)
        return m, p
    p = ex / s
    U = m + tau * math.log(s)
    return U, p


def _grad_U(a, p, n, fs_g):
    # dU/da[j] = a[j] * S[j] + sum_m a[m] * p[j+m]
    # S[j] = sum_{k=j}^{j+n-1} p[k]
    pc = np.zeros(2 * n)
    pc[:2 * n - 1] = p
    pc_cum = np.zeros(2 * n + 1)
    np.cumsum(pc, out=pc_cum[1:])
    j = np.arange(n)
    S = pc_cum[j + n] - pc_cum[j]
    
    # cross_corr[j] = sum_m a[m]*p[j+m] = (a_rev * p)[n-1+j]
    a_rev = a[::-1]
    A_r = rfft(a_rev, fs_g)
    P = rfft(p, fs_g)
    full = irfft(A_r * P, fs_g)
    cross = full[n - 1:2 * n - 1].copy()
    
    g = a * S + cross
    return g


def _init_z(n, rng, variant, parent):
    i = np.arange(n, dtype=np.float64)
    eps = 1e-10
    v = int(variant)
    if v % 9 == 0:
        a = np.ones(n) / n
    elif v % 9 == 1:
        a = rng.dirichlet(np.ones(n) * 2.0)
    elif v % 9 == 2:
        a = np.cos(np.pi * i / (2 * n)) ** 2
        a = np.maximum(a, eps)
    elif v % 9 == 3:
        mu = n / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif v % 9 == 4:
        a = np.ones(n)
        for _ in range(3):
            f = rng.integers(1, max(2, n // 2))
            ph = rng.uniform(0, 2 * math.pi)
            amp = rng.uniform(0.2, 0.7)
            a += amp * np.sin(2 * math.pi * f * i / n + ph)
        a = np.maximum(a, eps)
    elif v % 9 == 5 and parent is not None and parent.size > 0:
        P = parent.size
        idx = (np.arange(n) * P // n).astype(int)
        a = np.maximum(parent[idx], 0.0)
        a += rng.uniform(0.0, 0.05 * max(1.0, a.mean()), n)
    elif v % 9 == 6 and parent is not None and parent.size > 0:
        P = parent.size
        starts = np.floor(np.linspace(0, P, n, endpoint=False)).astype(int)
        stops = np.floor(np.linspace(0, P, n, endpoint=True)).astype(int)
        chunks = np.array([float(parent[s:e].mean()) if e > s else float(parent[-1]) for s, e in zip(starts, stops)])
        a = np.maximum(chunks, eps)
    elif v % 9 == 7:
        a = (i + 1.0) / (n + 1.0)  # linear ramp
    else:
        a = 1.0 - np.abs(2.0 * i / n - 1.0)  # triangular
        a = np.maximum(a, eps)
    
    a = np.maximum(a, eps)
    a = a / a.sum()
    return a.copy()

    fs_g = next_fast_len(3 * n - 2)
    
    best_a = None
    best_s = math.inf
    
    n_starts = 8 if n <= 256 else 5 if n <= 512 else 3

            break

        if s < best_s:
            best_s = s
            best_a = a.copy()
        
        fs_c_l = next_fast_len(2 * n - 1)
        
        m_v = np.zeros(n)
        v_v = np.zeros(n)
        t_count = 0
        lr = 0.12
        tau_scale = 0.12
        stuck = 0
        last_improve = time.monotonic()
        
        while time.monotonic() < t_stop - 0.3:
            b = _fft_conv(a, n, fs_c_l)
            bmax = float(b.max())
            s = 2.0 * n * bmax
            if s < best_s:
                best_s = s
                best_a = a.copy()
                last_improve = time.monotonic()
            
            tau = max(1e-14, tau_scale * max(bmax, 1e-14))
            U, p = _soft_lse(b, tau)
            g = _grad_U(a, p, n, fs_g)
            
            if not np.isfinite(g).all():

            if g_norm < 1e-18:
                stuck += 1
                if stuck > 15:

                a = _proj_simplex(a + rng.normal(0, 0.05, n))
                m_v[:] = 0
                v_v[:] = 0
                t_count = 0
                continue
            
            t_count += 1
            b1, b2 = 0.9, 0.999
            m_v = b1 * m_v + (1 - b1) * g
            v_v = b2 * v_v + (1 - b2) * g * g
            m_h = m_v / (1 - b1 ** t_count)
            v_h = v_v / (1 - b2 ** t_count)

            d[~np.isfinite(d)] = 0.0
            d = d / max(np.sqrt((d * d).sum()), 1e-30)
            
            improved = False

def _polish(a, score, t_stop):
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return a, score
    a = a / s
    
    best_a = a.copy()
    best_s = score
    rng = np.random.default_rng(777)
    fs = next_fast_len(2 * n - 1)
    stuck = 0
    sigma = 0.12
    
    while time.monotonic() < t_stop - 1.0:
        r = rng.random()
        if r < 0.3:
            # Gaussian perturbation in log space
            z = np.log(np.maximum(best_a, 1e-15))
            z += rng.normal(0, sigma, n)
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
        elif r < 0.55:
            # Block perturbation
            a_new = best_a.copy()
            w = max(2, n // 20)
            w = min(w, n // 3)
            st = rng.integers(0, n - w + 1)
            a_new[st:st + w] *= np.exp(rng.normal(0, 0.2, 1))
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            a_new = a_new / ss if ss > 0.01 else best_a.copy()
        elif r < 0.7:
            # Symmetrize
            a_new = 0.5 * (best_a + best_a[::-1])
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
        elif r < 0.85:
            # Flatten toward uniform
            eps_f = rng.uniform(0.02, 0.15)
            a_new = (1 - eps_f) * best_a + eps_f / n
            a_new = a_new / a_new.sum()
        else:
            # Pairwise swap
            a_new = best_a.copy()
            i1 = rng.integers(0, n)
            i2 = rng.integers(0, n)
            if i1 != i2:
                amt = rng.uniform(-0.5, 0.5) * min(best_a[i1], best_a[i2])
                a_new[i1] -= amt
                a_new[i2] += amt
                a_new = np.maximum(a_new, 0)
                ss = a_new.sum()
                a_new = a_new / ss if ss > 0.01 else best_a.copy()
        
        fs_e = next_fast_len(2 * n - 1)
        b_e = _fft_conv(a_new, n, fs_e)
        s_e = 2.0 * n * float(b_e.max())
        
        if s_e < best_s * (1.0 - 1e-12):
            best_a = a_new
            best_s = s_e
            stuck = 0
            sigma = max(0.05, sigma * 0.9)
        else:
            stuck += 1
            if stuck % 30 == 0:
                sigma = min(0.4, sigma * 1.5)
            if stuck > 200:
                break
    
    return best_a, best_s

def _softmax(z):

def _fft_conv(a, n, fs):
    F = rfft(a, fs)
    c = irfft(F * F, fs)[:2 * n - 1]
    np.maximum(c, 0, out=c)
    c[~np.isfinite(c)] = 0.0

def _eval_fast(a):

def _soft_lse(b, tau):
    m = float(b.max())
    x = (b - m) / tau
    ex = np.exp(x)
    s = float(ex.sum())

        p = np.zeros_like(b)
        mask = b == m
        p[mask] = 1.0 / mask.sum()
        return m, p
    p = ex / s
    U = m + tau * math.log(s)
    return U, p
