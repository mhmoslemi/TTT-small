import numpy as np
import time
import cvxpy as cp

def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if not np.isfinite(s) or s <= 0:
        return np.full(z.shape, 1.0/len(z))
    return e / s

def _score_array(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    c = np.convolve(a, a)
    mx = float(np.max(c))
    return float(2.0 * n * mx / (s * s))

def _convolve_ff(a, b, L=None):
    """FFT-based linear convolution."""
    n, m = len(a), len(b)
    size = n + m - 1
    if L is None:
        L = 1 << (size - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)[:size]

def _cross_corr_ff(a, p, n_out):
    """
    Cross-correlation: result[j] = sum_m a[m] * p[m+j] for j=0..n_out-1
    a has length n, p has length >= n_out + n - 1
    """
    n_a = len(a)
    L = 1 << (n_a + len(p) - 1).bit_length()
    A = np.fft.rfft(a, L)
    P = np.fft.rfft(p, L)
    corr = np.fft.irfft(A * np.conj(P), L)
    return corr[:n_out]

def _grad_z(a, c, beta):
    """
    Gradient of soft_max_beta(c) w.r.t. z where a = softmax(z).
    Returns (grad, soft_max_value)
    """
    n = len(a)
    cmax = float(np.max(c))
    zc = beta * (c - cmax)
    p = np.exp(np.clip(zc, -700, 700))
    ps = float(p.sum())
    if not np.isfinite(ps) or ps <= 0:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = p / ps
    
    f_val = float(p @ c)
    corr = _cross_corr_ff(a, p, n)
    grad = 2.0 * a * (corr - f_val)
    return grad, f_val

def _adam(params, grad, state, lr, cap=1.0):
    b1, b2, eps = 0.9, 0.999, 1e-8
    state['t'] += 1
    t = state['t']
    state['m'] = b1 * state['m'] + (1 - b1) * grad
    state['v'] = b2 * state['v'] + (1 - b2) * grad * grad
    mhat = state['m'] / (1 - b1**t)
    vhat = state['v'] / (1 - b2**t)
    step = lr * mhat / (np.sqrt(vhat) + eps)
    step = np.clip(step, -cap, cap)
    return params - step

def _adam_state(dim):
    return {'m': np.zeros(dim), 'v': np.zeros(dim), 't': 0}

def _lp_direction(f, M):
    """
    Solve LP: max sum(b) s.t. f*b <= M (elementwise), b >= 0
    Returns the optimal b.
    """
    n = len(f)
    b_var = cp.Variable(n, nonneg=True)
    
    # Constraint matrix A: A[k,j] = f[k-j] if 0<=k-j<n else 0
    # Actually: (f*b)[k] = sum_j f[valid] * b[valid]
    # (f*b)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j] * b[j]
    # Let me build the constraint directly using the convolution.
    
    # For each k in 0..2n-2: sum_j (f shifted) * b[j] <= M
    # (f*b)[k] = sum_j f[k-j]*b[j] where f[k-j] is 0 if k-j out of [0,n-1]
    
    # Build constraint matrix
    A_mat = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_mat[k, j] = f[idx]
    
    constraints = [A_mat @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    if b_var.value is None:
        return f.copy()
    b_opt = np.maximum(b_var.value, 0)
    if b_opt.sum() < 1e-10:
        return f.copy()
    return b_opt
