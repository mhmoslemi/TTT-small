    """Project x onto {y: sum(y) = target_sum, y >= 0}."""
    n = len(x)
    # First, if already feasible
    if x.min() >= 0 and abs(x.sum() - target_sum) < 1e-15:
        return x.copy()
    # Water-filling
    # We need sum(max(x_i - tau, 0)) = target_sum
    # Sort in decreasing order
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    condition = u - (cssv - target_sum) / rho > 0
    if not condition.any():
        return np.full(n, target_sum / n)
    i = np.max(np.where(condition)[0])

import numpy as np
import time


def _next_pow2(x):
    p = 1
    while p < x:
        p <<= 1
    return p


def _project_simplex(x, target=1.0):
    """Project x onto {y >= 0, sum(y) = target}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > cssv - target
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n, dtype=np.float64)
    i = idx[-1]
    tau = (cssv[i] - target) / (i + 1)
    return np.maximum(x - tau, 0.0)
