def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(1.0, float(budget_s))
    except Exception:
        budget = 1300.0

    hard_return = t0 + budget - 8.0

    try:

        if b.size > 0 and np.all(np.isfinite(b)):
            fb = np.clip(b, 0.0, 1000.0)
            s = float(fb.sum())
            if s > 0.0:
                fb *= 1.0 / s
                fallback_a = fb
    except Exception:
        fallback_a = None

    if budget < 30.0:

            return [float(x) for x in fallback_a]

        m = int(m)
        if m <= 1:
            return 1
        return 1 << (m - 1).bit_length()

    # Work at a length near 10922 when the parent is 6962.  This keeps the
    # FFT size 32768 for the main deterministic pipeline.
    if fallback_a is not None:
        parent_len = int(fallback_a.size)
        if parent_len >= 10922:
            n = min(parent_len, 10922)
        else:
            n = 10922
    else:
        n = 10000
    n = int(max(512, n))

    L = 2 * n - 1
    N = _p2(3 * n - 2)
    one_n = 1.0 / float(n)
    factor = 2.0 * float(n)

    def _norm_fast(a):
        a = np.asarray(a, dtype=np.float64)
        s = float(a.sum())
        if s > 0.0 and np.isfinite(s):
            a *= 1.0 / s
        else:
            a = np.full(n, one_n, dtype=np.float64)
        return a

    def _norm_safe(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size != n:
            if a.size == 0:
                a = np.ones(n, dtype=np.float64)
            else:
                a = np.resize(a, n).astype(np.float64)

        s = float(a.sum())

        a = np.asarray(a, dtype=np.float64).ravel()
        m = int(m)
        if a.size == m:
            return a.copy()

        new = np.linspace(0.0, 1.0, m, endpoint=False)
        b = np.interp(new, old, a)
        return b

            if frac > 1.0:

                else:
                    fail = 0
                    if lr_factor < 1.0:
                        lr_factor = min(1.0, lr_factor * 1.05)
            else:
                lp = np.clip(lr * p, -6.0, 6.0)
                a = _norm_fast(a * np.exp(-lp))
                D, fa = _compute_D(a)

        Mcur = float(D.max()) if D is not None else np.inf
        if np.isfinite(Mcur) and Mcur < best_M:
            best_M = Mcur
            best_a = a.copy()

        return _norm_fast(best_a.copy()), float(best_M)

    # Build the working initial vector at the selected length.
    if fallback_a is not None:
        base_opt = _upsample(fallback_a, n)
        base_opt = _norm_safe(base_opt)
    else:
        base_opt = np.full(n, one_n, dtype=np.float64)

    # Seed pool: parent, power laws, exponentials, smooth random fields,
    # and perturbed parent.  The seeds are scored and the best few are used.
    seeds = [base_opt]
    i = np.arange(n, dtype=np.float64)

    pvals = (
        -0.49, -0.47, -0.45, -0.42, -0.40,
        -0.36, -0.30, -0.24, -0.18, -0.12,
        -0.06, 0.0, 0.06, 0.12, 0.24
    )
    for p in pvals:
        a = np.power(i + 1.0, float(p))
        seeds.append(_norm_safe(a))
        seeds.append(_norm_safe(a[::-1].copy()))

    for lam in (0.5, 1.0, 1.8, 3.0, 5.0, 8.0):
        a = np.exp(-float(lam) * i / max(1, n - 1))
        seeds.append(_norm_safe(a))

    kernel = np.array([0.1, 0.8, 0.1], dtype=np.float64)
    for _ in range(3):
        x = np.abs(rng.standard_normal(n)) ** 2
        x = np.convolve(x, kernel, mode="same")
        seeds.append(_norm_safe(x))

    for sig in (0.02, 0.05, 0.10, 0.20):
        a = base_opt * np.exp(float(sig) * rng.standard_normal(n))
        seeds.append(_norm_safe(a))

    scored = []
    for s in seeds:
        s = _norm_safe(s)
        sc = _score_opt(s)
        if np.isfinite(sc):
            scored.append((sc, s))

    scored.sort(key=lambda z: z[0])
    if not scored:
        scored = [(np.inf, base_opt)]

    top = scored[:4]
    best_a = top[0][1].copy()
    best_M = float(top[0][0])

    # Main optimization schedule.
    if time.monotonic() < opt_deadline:
        main_time = max(0.0, opt_deadline - time.monotonic())
        tasks = [
            (0.45, top[0][1],
             dict(tau0=2.0, tau1=70.0, lr0=0.14, lr1=0.02,
                  floor=1e-8, kmax=256, line=True)),
            (0.15, top[1][1] if len(top) > 1 else top[0][1],
             dict(tau0=3.0, tau1=60.0, lr0=0.10, lr1=0.02,
                  floor=1e-8, kmax=256, line=True)),
            (0.10, top[2][1] if len(top) > 2 else top[0][1],
             dict(tau0=5.0, tau1=50.0, lr0=0.08, lr1=0.02,
                  floor=1e-8, kmax=256, line=True)),
            (0.30, None,
             dict(tau0=50.0, tau1=250.0, lr0=0.05, lr1=0.004,
                  floor=1e-6, kmax=0, line=True)),
        ]

        for frac, seed, params in tasks:
            if time.monotonic() >= opt_deadline - 5.0:
                break
            remain = max(0.0, opt_deadline - time.monotonic() - 2.0)
            seconds = min(frac * main_time, remain)
            if seconds < 8.0:
                break
            seed_a = best_a.copy() if seed is None else seed
            a_best, M_best = _run(seed_a, seconds, opt_deadline, **params)
            if np.isfinite(M_best) and M_best < best_M - 1e-12:
                best_M = M_best
                best_a = a_best

    # Candidate pool: optimized vector, parent fallback, and a few modest
    # resamplings of the optimized vector.  Resampling can reduce discretization
    # error without requiring another full optimization.
    candidates = []
    if best_a.size > 0 and np.all(np.isfinite(best_a)):
        candidates.append((best_a, float(best_M)))

    if fallback_a is not None:
        fb = _norm_safe(fallback_a)
        fs = _score_len(fb)
        if np.isfinite(fs):
            candidates.append((fb, float(fs)))

    if time.monotonic() < hard_return - 25.0:
        for mult in (1.10, 1.15, 1.25):
            m2 = int(n * mult)
            if m2 <= n + 50 or m2 > 15000:
                continue
            a2 = _upsample(best_a, m2)
            a2 = np.asarray(a2, dtype=np.float64)
            s = float(a2.sum())
            if s > 0:
                a2 *= 1.0 / s
            s2 = _score_len(a2)
            if np.isfinite(s2):
                candidates.append((a2, float(s2)))

    candidates = [(a, s) for a, s in candidates if a is not None and a.size > 0]
    if not candidates:
        return [1.0]

    candidates.sort(key=lambda z: z[1])

    # Final validation with the authoritative evaluator when there is enough
    # time; otherwise return the internally validated best candidate.
    best_list = None
    best_exact = np.inf

    for a, _ in candidates[:3]:
        if time.monotonic() >= hard_return - 2.0:
            break
        try:
            arr = np.asarray(a, dtype=np.float64).ravel()
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            lst = [float(x) for x in arr]
            if not lst:
                continue
            ex = evaluate_sequence(lst)
            if np.isfinite(ex) and ex < best_exact:
                best_exact = ex
                best_list = lst
        except Exception:
            pass

    if best_list is None:
        arr = np.asarray(candidates[0][0], dtype=np.float64).ravel()
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if s > 0.0:
            arr *= 1.0 / s
        best_list = [float(x) for x in arr]

    if not best_list:
        best_list = [1.0]

    return best_list
