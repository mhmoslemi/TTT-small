            A[i, lo:hi + 1] = f[i - lo : i - hi - 1 : -1] if hi > lo else f[hi:lo+1]
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0.0, None)] * n, method='highs',
                  options={'time_limit': 10})
    if res.success:
        return np.asarray(res.x, dtype=np.float64)
    return None


def _lp_optimize(f_init, seed, stop_time):
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 0.01:
        f = np.ones(f.size)
        s = f.size
    n = len(f)
    f = f / s * n  # normalize sum to n

    best_f = f.copy()
    best_sc = _sc(f)
    rng = np.random.default_rng(seed)

    # Adaptive blend factor
    t_blend = 0.08

    for it in range(500):
        if time.monotonic() > stop_time - 1.0:
            break

        b = _conv(f)
        M = float(b.max())
        if not math.isfinite(M) or M <= 0:
            break

        t0_lp = time.monotonic()
        g = _lp_solve(f, M)
        lp_time = time.monotonic() - t0_lp

        if g is None:
            break
        g = np.maximum(g, 0.0)
        sg = g.sum()
        sf = f.sum()
        if sg < 1e-15:
            break
        g = g * (sf / sg)

        # Check similarity
        denom = math.sqrt(float(np.dot(f, f)) * float(np.dot(g, g)))
        corr = float(np.dot(f, g)) / (denom + 1e-30)

        if corr > 0.9999:
            break

        # Adapt blend factor
        if corr > 0.99:
            t_blend = 0.02
        elif corr > 0.95:
            t_blend = 0.05
        else:
            t_blend = 0.1

        f_new = (1.0 - t_blend) * f + t_blend * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 0.01:
            break
        f_new = f_new / s_new * n

        sc_new = _sc(f_new)
        if math.isfinite(sc_new) and sc_new < best_sc:
            best_sc = sc_new
            best_f = f_new.copy()

        f = f_new

    return best_f, best_sc


def _grad_refine(f_init, seed, stop_time):
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = np.clip(f, 0.0, 1000.0)
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return f, math.inf
    f = f / s * n

    best_f = f.copy()
    best_sc = _sc(f)
    fs = next_fast_len(2 * n - 1)
    rng = np.random.default_rng(seed)
    lr = 0.12
    stuck = 0
    tau_scale = 0.05
    max_updates = max(1, int((stop_time - time.monotonic()) * 20))

    for it in range(max_updates):
        if time.monotonic() > stop_time - 2.0:
            break

        z = np.log(np.maximum(f, 1e-12))
        z -= z.mean()
        e = np.exp(z - z.max())
        a = e / e.sum() * n

        F = rfft(a, fs)
        b = irfft(F * F, fs)[:2 * n - 1]
        np.maximum(b, 0.0, out=b)
        bm = float(b.max())
        sc = 2.0 * n * bm / (a.sum() ** 2)

        if math.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best_f = a.copy()
            f = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Gradient via soft log-sum-exp
        tau = max(1e-12, tau_scale * max(bm, 1e-30))
        bx = (b - bm) / tau
        ex = np.exp(bx)
        p = ex / ex.sum()

        # Get active indices
        thresh = max(1e-12, p.max() * 1e-8)
        idx = np.where(p >= thresh)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(p))])
        if idx.size > 300:
            idx = np.argpartition(p, -300)[-300:]

        hg = np.zeros(n)
        for k in idx:
            wk = float(p[k])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            hg[lo:hi + 1] += 2.0 * wk * a[k - hi:k - lo + 1][::-1]

        # Center (project to sum-0 for the normalization)
        h = a * hg
        h -= h.mean()
        hmax = float(np.max(np.abs(h)))

        if hmax > 1e-15:
            f -= lr * h / hmax
            f = np.maximum(f, 0.0)
            s = f.sum()
            if s < 0.01:
                f = best_f.copy()
            else:
                f = f / s * n
        else:
            f = f + rng.normal(0.0, 0.3 * f.mean(), n)
            f = np.maximum(f, 0.0)
            s = f.sum()
            if s > 0.01:
                f = f / s * n
            else:
                f = best_f.copy()

        if stuck > 60:
            tau_scale = max(0.01, tau_scale * 0.7)
            lr = 0.12
            f = f * (1.0 + rng.normal(0, 0.1, n))
            f = np.maximum(f, 0.0)
            s = f.sum()
            if s > 0.01:
                f = f / s * n
            stuck = 0

    return best_f, best_sc
