        if res.x is not None:
            rx = np.clip(res.x, 0.0, None)
            rs = rx.sum()
            if rs > 0:
                rx = rx / rs
                rf = _score(rx)
                if rf < best_f:
                    best_x = rx
                    best_f = rf
    except Exception:
        pass
    
    return best_x
