        add_seed(0.25 + 0.75 * (0.5 + 0.5 * np.cos(np.pi * xx)))
        add_seed(np.maximum(base * (0.25 + 0.75 * (0.5 + 0.5 * np.cos(np.pi * xx))), 0.0))

    scored = []
    for a in raw_seeds:
        p = normalize(a)
        sc = fast_score(p)
        if np.isfinite(sc):
            scored.append((sc, p))

    if not scored:
        scored = [(2.0, np.full(n, 1.0 / n, dtype=np.float64))]

    scored.sort(key=lambda z: z[0])

    unique = []
    for sc, p in scored:
        dup = False
        for _, q in unique:
            try:
                if float(np.linalg.norm(p - q)) <= 1.0e-7:
                    dup = True
                    break
            except Exception:
                pass
        if not dup:
            unique.append((sc, p))
        if len(unique) >= 4:
            break

    seeds = unique[:3]

    main_phases = [
        (2.0, 60),
        (5.0, 80),
        (10.0, 120),
        (20.0, 150),
        (50.0, 200),
        (150.0, 260),
        (500.0, 320),
        (2000.0, 380),
        (7500.0, 320),
    ]
    second_phases = [
        (10.0, 80),
        (50.0, 130),
        (200.0, 180),
        (1000.0, 250),
        (5000.0, 180),
    ]
    third_phases = [
        (50.0, 80),
        (500.0, 130),
        (2500.0, 170),
    ]
    final_phases = [
        (20.0, 60),
        (200.0, 110),
        (1000.0, 160),
        (5000.0, 120),
    ]

    if seeds:
        remaining = max(0.0, opt_deadline - time.monotonic())
        main_secs = min(780.0, 0.60 * remaining)
        if main_secs > 20.0:
            polish(seeds[0][1], main_phases, main_secs)

    if len(seeds) > 1:
        remaining = max(0.0, opt_deadline - time.monotonic())
        sec_secs = min(300.0, 0.23 * remaining)
        if sec_secs > 20.0:
            polish(seeds[1][1], second_phases, sec_secs)

    if len(seeds) > 2:
        remaining = max(0.0, opt_deadline - time.monotonic())
        third_secs = min(150.0, 0.12 * remaining)
        if third_secs > 15.0:
            polish(seeds[2][1], third_phases, third_secs)

    remaining = max(0.0, opt_deadline - time.monotonic())
    final_secs = min(180.0, 0.45 * remaining)
    if final_secs > 15.0:
        polish(best_x, final_phases, final_secs)

    # Final validation with the authoritative evaluator when feasible.
    candidates = []

    def add_candidate(a):
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
            if a.size != n:
                return
            a = normalize(a)
            if a.size and np.all(np.isfinite(a)):
                candidates.append(a)
        except Exception:
            pass

    add_candidate(best_x)
    add_candidate(base)
    if seeds:
        add_candidate(seeds[0][1])
    if len(seeds) > 1:
        add_candidate(seeds[1][1])

    best_list = None
    best_val = np.inf

    # Evaluate a small number of candidates with the authoritative metric.
    eval_deadline = t0 + B - 3.0
    for a in candidates[:3]:
        if time.monotonic() > eval_deadline - 5.0:
            break
        try:
            lst = [float(v) for v in a]
            val = evaluate_sequence(lst)
        except Exception:
            val = np.inf
        if np.isfinite(val) and val < best_val:
            best_val = val
            best_list = lst

    if best_list is None:
        a = normalize(best_x)
        best_list = [float(v) for v in a]

    if (not best_list) or any((not np.isfinite(v)) or (v < 0.0) for v in best_list):
        best_list = [float(v) for v in normalize(base)]

    return best_list
