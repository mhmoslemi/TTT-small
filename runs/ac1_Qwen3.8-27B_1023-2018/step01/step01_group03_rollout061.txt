    except Exception:
        pass

def _subgrad_iter(a, step, temp=0.05):
    """One subgradient step. a is normalized (sum=1)."""
    b = np.convolve(a, a)
    M = float(b.max())
    if M < 1e-15:
        return a
    n = len(a)
    
    # Soft weights
    w = np.exp(b / (M * temp + 1e-15))
    w_sum = float(w.sum())
    if w_sum < 1e-15:
        return a
    w /= w_sum
    
    # h[j] = 2 * sum_k w[k] * a[k-j] (valid)
    C = _conv_mat(a)
    h = 2.0 * (C.T @ w)
    
    # Center h (remove component along uniform to preserve sum)
    h -= h.mean()
    
    # Projected step
    a_new = a - step * h
    a_new = np.clip(a_new, 0.0, None)
    s = float(a_new.sum())
    if s < 1e-12:
        return a
    a_new /= s
    return a_new

    n = len(q)

    for it in range(n_lp):
        if time.monotonic() > deadline - 60:

        if g is None:

        best_alpha = 0.0
        best_alpha_sc = _score_fast(q)
        for alpha in (0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
            cand = _norm1((1 - alpha) * q + alpha * g)
            sc = _score_fast(cand)
            if sc < best_alpha_sc:

                best_alpha = alpha

            best_sc = sc
