def _compute_score(a):

    n = len(a)
    s = a.sum()

    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def _smooth_max_grad(c, beta):
    """
    Compute smooth approximation of max(c) and its gradient w.r.t. c.
    Returns (smooth_max, p) where p_k = exp(beta*c_k) / sum(exp(beta*c_j)).
