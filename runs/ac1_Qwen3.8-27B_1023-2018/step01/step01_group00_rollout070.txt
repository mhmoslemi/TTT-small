def _exact_score(seq):
    try:
        a=np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return math.inf
    if a.size == 0:
        return math.inf
    if not np.isfinite(a).all():
        a=np.where(np.isfinite(a), a, 0.0)
    a=np.clip(a,0.0,1000.0)
    s=float(np.sum(a))
    if not math.isfinite(s) or s < 0.01:
        return math.inf
    try:
        b=np.convolve(a,a)
        mb=float(np.max(b)) if b.size else 0.0
    except Exception:
        return math.inf
    if not math.isfinite(mb): return math.inf
    val = float(2.0*a.size*mb/(s*s))
    return val if math.isfinite(val) else math.inf

def _conv_matrix(f):
    f = np.asarray(f, dtype=np.float64).ravel()
    n = int(f.size)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp_direction(f, M):
    if _linprog is None:
        return None

    f = np.asarray(f, dtype=np.float64).ravel()
    n = int(f.size)
    if n <= 0:
        return None

    M = float(M)
    if not math.isfinite(M) or M <= 0.0:
        return None

    maxf = float(np.max(f)) if n else 0.0
    if not math.isfinite(maxf) or maxf <= 0.0:
        return None

    A = _conv_matrix(f) / maxf
    tol = 1e-10 * max(1.0, M) + 1e-12
    b_ub = np.full(2 * n - 1, (M + tol) / maxf, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)

    target = math.sqrt(2.0 * n)
    ub = 100.0 * max(1.0, target)
    bounds = [(0.0, ub)] * n

    methods = ["highs"]
    if hasattr(_linprog, "interior_point"):
        methods.append("interior_point")
    methods.append("revised_simplex")

    for meth in methods:
        try:
            res = _linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method=meth)
            if getattr(res, "success", False) and getattr(res, "x", None) is not None:
                x = np.asarray(res.x, dtype=np.float64).ravel()
                x = np.where(np.isfinite(x), x, 0.0)
                x = np.clip(x, 0.0, ub)
                s = float(np.sum(x))
                if math.isfinite(s) and s > 1e-12:
                    return x
        except Exception:
            continue
    return None


def _line_search(A, B, C, M0):
    A = np.asarray(A, dtype=np.float64).ravel()
    B = np.asarray(B, dtype=np.float64).ravel()
    C = np.asarray(C, dtype=np.float64).ravel()
    if A.size == 0 or B.size != A.size or C.size != A.size:
        return None, float(M0)

    M0 = float(M0)
    if not math.isfinite(M0) or M0 <= 0.0:
        return None, float(M0)

    const = A
    lin = 2.0 * (B - A)
    quad = A - 2.0 * B + C

    ts_main = np.unique(np.concatenate((
        np.geomspace(1e-5, 1.0, 120),
        np.linspace(0.0, 1.0, 64),
        np.array([1.0], dtype=np.float64),
    )))
    ts_main = ts_main[(ts_main >= 0.0) & (ts_main <= 1.0)]

    if ts_main.size == 0:
        return None, M0

    T = ts_main[:, None]
    scores = const + lin * T + quad * (T * T)
    rowmax = np.max(scores, axis=1)
    rowmax = np.where(np.isfinite(rowmax), rowmax, M0)

    tol = 1e-11 * max(1.0, M0)
    best_val = float(np.min(rowmax))
    best_idx = int(np.argmin(rowmax))

    if not math.isfinite(best_val) or best_val >= M0 - tol:
        ts_fine = np.geomspace(1e-8, 1e-3, 48)
        T = ts_fine[:, None]
        scores = const + lin * T + quad * (T * T)
        rowmax = np.max(scores, axis=1)
        rowmax = np.where(np.isfinite(rowmax), rowmax, M0)
        fine_val = float(np.min(rowmax))
        if math.isfinite(fine_val) and fine_val < M0 - tol:
            idx = int(np.argmin(rowmax))
            return float(ts_fine[idx]), fine_val
        return None, M0

    t = float(ts_main[best_idx])
    if t <= 0.0:
        t = 1e-5
    return t, float(rowmax[best_idx])


def _gradient_step(f, M0, target):
    f = np.asarray(f, dtype=np.float64).ravel()
    n = int(f.size)
    if n <= 0:
        return None, float(M0)

    try:
        b = np.convolve(f, f)
        m = float(np.max(b)) if b.size else 0.0
    except Exception:
        return None, float(M0)

    if not math.isfinite(m) or m <= 0.0:
        return None, float(M0)

    thresh = m - 0.25 * m
    idx = np.where(b >= thresh)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(b))], dtype=int)
    if idx.size > 128:
        idx = np.argpartition(b, -128)[-128:]

    w = b[idx].astype(np.float64) - (thresh - 1e-12)
    w = np.maximum(w, 1e-9)
    ws = float(np.sum(w))
    if not math.isfinite(ws) or ws <= 0.0:
        w = np.ones(idx.size, dtype=np.float64) / max(1, idx.size)
    else:
        w = w / ws

    grad = np.zeros(n, dtype=np.float64)
    for k, wi in zip(idx, w):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi < lo:
            continue
        seg = f[k - lo : k - hi - 1 : -1]
        grad[lo:hi + 1] += 2.0 * float(wi) * seg

    gm = float(np.max(np.abs(grad))) if grad.size else 0.0
    if not math.isfinite(gm) or gm <= 1e-300:
        return None, m

    grad = grad / gm
    target = float(target)
    best_nf = None
    best_m = m

    for lr in (0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4):
        nf = np.clip(f - lr * grad, 0.0, None)
        nf = _normalize(nf, target)
        nm = _max_conv(nf)
        if math.isfinite(nm) and nm < best_m - max(1e-12, 1e-11 * best_m):
            best_nf = nf
            best_m = nm
            break

    if best_nf is None:
        return None, m
    return best_nf, best_m

        if M0 > best_M + 0.15 and stagnant > 20:
            f = best.copy()
            M0 = best_M
            stagnant = 0

        if stagnant > 25:
            f = _perturb(best, rng, sigma)
            M0 = _max_conv(f)
            if not math.isfinite(M0):
                f = best.copy()
                M0 = best_M
            stagnant = 0
            sigma = min(0.8, sigma * 1.25)
            continue

        g = _solve_lp_direction(f, M0)
        accepted = False

        if g is not None:
            gs = float(np.sum(g))
            if math.isfinite(gs) and gs > 1e-12:
                g = g * (target / gs)
                B = np.convolve(f, g)
                C = np.convolve(g, g)
                t, Mest = _line_search(A, B, C, M0)
                if t is not None and math.isfinite(Mest):
                    nf = (1.0 - t) * f + t * g
                    nf = np.clip(nf, 0.0, None)
                    nf = _normalize(nf, target)
                    Mnew = _max_conv(nf)
                    if math.isfinite(Mnew) and Mnew < M0 - max(1e-12, 1e-11 * M0):
                        f = nf
                        M0 = Mnew
                        lp_fail = 0
                        grad_fail = 0
                        stagnant = 0
                        if M0 < best_M - 1e-12:
                            best_M = M0
                            best = f.copy()
                            sigma = max(0.02, sigma * 0.92)
                        accepted = True

        if not accepted:
            lp_fail += 1
            stagnant += 1

            if lp_fail >= 3 or (stagnant > 10 and grad_fail == 0):
                gf, gm = _gradient_step(f, M0, target)
                if gf is not None:
                    f = gf
                    M0 = gm
                    grad_fail = 1
                    lp_fail = 0
                    stagnant = 0
                    if M0 < best_M - 1e-12:
                        best_M = M0
                        best = f.copy()
                        sigma = max(0.02, sigma * 0.92)
                else:
                    f = _perturb(best, rng, sigma)
                    M0 = _max_conv(f)
                    if not math.isfinite(M0):
                        f = best.copy()
                        M0 = best_M
                    lp_fail = 0
                    grad_fail += 1
                    sigma = min(0.8, sigma * 1.2)
            else:
                if stagnant > 8:
                    f = _perturb(best, rng, sigma)
                    M0 = _max_conv(f)
                    if not math.isfinite(M0):
                        f = best.copy()
                        M0 = best_M
                    stagnant = 0
                    sigma = min(0.8, sigma * 1.15)

    return best, best_M


def _polish(seq, stop_time, seed=123456):
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return np.ones(32, dtype=np.float64) / 32.0, math.inf

    if arr.size < 2:
        arr = np.ones(32, dtype=np.float64) / 32.0

    n = int(arr.size)
    target = math.sqrt(2.0 * n)
    best = _normalize(arr, target)
    best_score = _fast_score(best)
    if not math.isfinite(best_score) or best_score <= 0.0:
        best = np.full(n, target / n, dtype=np.float64)
        best_score = _fast_score(best)

    stop_time = float(stop_time)
    rng = np.random.default_rng(int(seed) + 777)
    sigma = 0.05
    noimp = 0

    while time.monotonic() < stop_time - 0.3:
        variants = []
        mean = float(np.mean(best)) if best.size else 1.0
        if not math.isfinite(mean) or mean <= 0.0:
            mean = target / max(1, n)

        logf = np.log(np.clip(best, 1e-9, None))
        variants.append(np.exp(logf + rng.normal(0.0, sigma, n)))
        variants.append(best + rng.normal(0.0, sigma * mean, n))

        p = float(rng.uniform(0.55, 1.85))
        variants.append(np.power(np.clip(best, 0.0, None), p))

        k = int(rng.integers(2, min(5, n) + 1)) if n > 2 else 1
        kernel = np.ones(k, dtype=np.float64) / max(1, k)
        variants.append(np.convolve(best, kernel, mode="same"))

        xx = np.linspace(0.0, 1.0, n)
        shift = float(rng.uniform(-0.2, 0.2))
        w = 0.5 + 0.5 * np.cos(np.pi * (xx + shift))
        variants.append(best * (0.3 + 0.7 * w))

        improved = False
        for cand in variants:
            cand = _normalize(cand, target)
            cs = _fast_score(cand)
            if math.isfinite(cs) and cs < best_score - max(1e-13, 1e-12 * best_score):
                best_score = cs
                best = cand.copy()
                noimp = 0
                improved = True
                sigma = max(0.015, sigma * 0.85)
                break

        if not improved:
            noimp += 1
            if noimp >= 10:
                sigma = min(0.7, sigma * 1.35)
                noimp = 0

        if best_score < 1.0000005:
            break

    return best, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not math.isfinite(budget) or budget <= 10.0:
        budget = 10.0

    deadline = t0 + max(5.0, budget - 8.0)

    parent = _get_parent()

    best_seq = np.clip(parent, 0.0, 1000.0).astype(np.float64).copy()
    if best_seq.size == 0:
        best_seq = np.ones(64, dtype=np.float64) / 64.0

    best_score = _safe_evaluate_exact(best_seq)
    if not math.isfinite(best_score):
        fallback = np.ones(64, dtype=np.float64) / 64.0
        best_seq = fallback
        best_score = _safe_evaluate_exact(fallback)

    fall_seq = best_seq.copy()
    fall_score = best_score

    P = int(parent.size) if parent.size > 0 else 192
    P = int(min(max(24, P), 768))

    seen_n = set()
    runs = []

    rem = max(0.0, deadline - time.monotonic())
    if rem > 30.0:
        main_budgets = (
            ("P", max(20.0, rem * 0.20)),
            ("P", max(20.0, rem * 0.13)),
            ("P", max(15.0, rem * 0.09)),
            ("P", max(10.0, rem * 0.06)),
            ("O1", max(10.0, rem * 0.08)),
            ("O2", max(10.0, rem * 0.07)),
            ("O3", max(10.0, rem * 0.07)),
        )
    else:
        main_budgets = (("P", rem * 0.5, ), ("P", rem * 0.3, ))

    other_factors = [1.25, 0.80, 1.50]
    oi = 0
    variant_idx = 0
    for tag, budget in main_budgets:
        if tag == "P":
            n = P
            init = _make_initial(n, parent, variant_idx, int(seed) + variant_idx * 1000)
            variant_idx += 1
        else:
            if oi >= len(other_factors):
                break
            n = max(24, int(round(P * other_factors[oi])))
            n = min(n, 768)
            init = _make_initial(n, parent, variant_idx, int(seed) + 5000 + variant_idx * 997)
            variant_idx += 1
            oi += 1
        if n in seen_n and len(runs) > 3:
            continue
        seen_n.add(n)
        runs.append((n, float(budget), init))

    total = 0.0
    for i, (n, b, init) in enumerate(runs):
        if total > max(30.0, (deadline - t0) * 0.78) or time.monotonic() > deadline - 60.0:
            break
        start = time.monotonic()
        run_deadline = min(deadline - 25.0, start + max(10.0, b))
        if run_deadline <= start + 5.0:
            continue
        try:
            cand, internal_score = _run_one(n, init, run_deadline, int(seed) + 100000 + 13 * i, parent)
        except Exception:
            cand = None

        if cand is not None:
            try:
                cand = np.asarray(cand, dtype=np.float64).ravel()
            except Exception:
                cand = None
            if cand is not None and cand.size > 0:
                cand = np.clip(cand, 0.0, 1000.0)
                sc = _safe_evaluate_exact(cand)
                if math.isfinite(sc) and sc < best_score - 1e-12:
                    best_score = sc
                    best_seq = cand.copy()
        total += max(0.0, time.monotonic() - start)

    if time.monotonic() < deadline - 18.0:
        polish_deadline = min(deadline - 2.0, time.monotonic() + max(20.0, (deadline - time.monotonic()) * 0.35))
        try:
            cand, _ = _polish(best_seq, polish_deadline, int(seed) + 555)
            cand = np.asarray(cand, dtype=np.float64).ravel()
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _safe_evaluate_exact(cand)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_seq = cand.copy()
        except Exception:
            pass

    final = np.asarray(best_seq, dtype=np.float64).ravel()
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)

    if final.size < 1:
        final = np.ones(32, dtype=np.float64) / 32.0

    nfin = int(final.size)
    target = math.sqrt(2.0 * max(1, nfin))
    if float(np.sum(final)) < 0.01:
        final = np.full(nfin, target / max(1, nfin), dtype=np.float64)
    else:
        final = _normalize(final, target)

    final_score = _safe_evaluate_exact(final)
    if not math.isfinite(final_score):
        final = fall_seq
        final_score = fall_score

    if not math.isfinite(final_score):
        final = np.ones(32, dtype=np.float64) / 32.0
        final_score = _safe_evaluate_exact(final)

    if not math.isfinite(final_score):
        return [1.0]

    return [float(x) for x in np.asarray(final, dtype=np.float64).ravel()]
