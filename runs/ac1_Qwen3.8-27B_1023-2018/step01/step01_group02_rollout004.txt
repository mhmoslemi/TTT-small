import numpy as np
import time
import math

try:
    from scipy.optimize import minimize
    _HAVE_MINIMIZE = True
except Exception:
    _HAVE_MINIMIZE = False

try:
    from scipy.optimize import linprog
    _HAVE_LINPROG = True
except Exception:
    _HAVE_LINPROG = False

_INF = float("inf")


def _clean_nonnegative(x):
    try:
        a = np.asarray(x, dtype=np.float64).ravel()
    except Exception:
        a = np.ones(1, dtype=np.float64)
    if a.size == 0:
        a = np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    return a


def _normalize(a):
    a = _clean_nonnegative(a)
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _next_power2(m):
    m = int(max(1, m))
    p = 1
    while p < m:
        p <<= 1
    return p


def _autoconv(a):
    n = a.size
    if n == 1:
        return np.array([float(a[0] * a[0])], dtype=np.float64)
    if n <= 300:
        c = np.convolve(a, a)
    else:
        size = 2 * n - 1
        N = _next_power2(size)
        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A * A, N)[:size]
        c = np.where(c > 0.0, c, 0.0)
    return c


def _score_array(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return _INF
    c = _autoconv(a)
    if c.size == 0:
        return _INF
    mx = float(np.max(c))
    if not math.isfinite(mx) or mx < 0.0:
        return _INF
    return float(2.0 * n * mx)


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.ones(1, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    m = float(np.max(z))
    if not math.isfinite(m):
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    e = np.exp(z - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s


def _logit(a):
    a = _normalize(a)
    z = np.log(np.maximum(a, 1e-8))
    z = np.where(np.isfinite(z), z, 0.0)
    z -= float(np.mean(z))
    return z


def _resample(a, m):
    a = _normalize(a)
    m = int(max(1, m))
    if a.size == m:
        return a.copy()
    if a.size == 1:
        return np.ones(m, dtype=np.float64)
    x0 = np.linspace(0.0, 1.0, a.size)
    x1 = np.linspace(0.0, 1.0, m)
    y = np.interp(x1, x0, a)
    return _normalize(np.maximum(y, 0.0))


def _cross(a, p, pre_fft=None, N=None):
    n = a.size
    if n == 1:
        return np.array([float(a[0] * p[0])], dtype=np.float64)

    if pre_fft is not None and N is not None:
        B = np.fft.rfft(p, N)
        conv = np.fft.irfft(pre_fft * B, N)
        r = conv[n - 1:2 * n - 1]
        return np.where(r > 0.0, r, 0.0)

    if n <= 300:
        conv = np.convolve(a[::-1], p)
        r = conv[n - 1:2 * n - 1]
        return np.where(r > 0.0, r, 0.0)

    size = n + len(p) - 1
    N = _next_power2(size)
    A = np.fft.rfft(a[::-1], N)
    B = np.fft.rfft(p, N)
    conv = np.fft.irfft(A * B, N)
    r = conv[n - 1:2 * n - 1]
    return np.where(r > 0.0, r, 0.0)


def _compute_val_grad(z, beta, lam, n, local_best, global_best):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return 1e300, np.zeros(n, dtype=np.float64)

    zc = z - float(np.mean(z))
    a = _softmax(zc)
    if a.size != n:
        return 1e300, np.zeros(n, dtype=np.float64)

    c = _autoconv(a)
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return 1e300, np.zeros(n, dtype=np.float64)

    mx = float(np.max(c))
    sc = float(2.0 * n * mx / (s * s))
    if math.isfinite(sc):
        if sc < local_best["score"]:
            local_best["score"] = sc
            local_best["arr"] = a.copy()
            local_best["z"] = zc.copy()
            local_best["time"] = time.monotonic()
            if sc < global_best["score"]:
                global_best["score"] = sc
                global_best["arr"] = a.copy()

    L = c.size
    if beta is None or beta <= 1e-3:
        p = np.full(L, 1.0 / float(L), dtype=np.float64)
        fsoft = float(np.mean(c))
    else:
        m = float(np.max(c))
        zc_b = float(beta) * (c - m)
        zc_b = np.where(zc_b > -745.0, zc_b, -745.0)
        ps = np.exp(zc_b)
        ss = float(ps.sum())
        if ss <= 0.0:
            p = np.zeros(L, dtype=np.float64)
            p[int(np.argmax(c))] = 1.0
            fsoft = m
        else:
            p = ps / ss
            fsoft = m + math.log(ss) / float(beta)

    if n > 300:
        N_cr = _next_power2(3 * n - 2)
        A_rev = np.fft.rfft(a[::-1], N_cr)
        rp = _cross(a, p, pre_fft=A_rev, N=N_cr)
        if lam:
            rc = _cross(a, c, pre_fft=A_rev, N=N_cr)
            G = rp + float(lam) * rc
        else:
            G = rp
    else:
        rp = _cross(a, p)
        if lam:
            rc = _cross(a, c)
            G = rp + float(lam) * rc
        else:
            G = rp

    dot = float(np.dot(a, G))
    grad = 2.0 * a * (G - dot)

    val = fsoft
    if lam:
        q = float(np.dot(c, c))
        val += 0.5 * float(lam) * q

    if not math.isfinite(val):
        val = 1e300
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)
    return float(val), grad


def _adam_phase(z, phase_end, beta, lam, n, local_best, global_best, rng):
    z = np.asarray(z, dtype=np.float64).copy()
    if z.size != n or not np.all(np.isfinite(z)):
        z = np.zeros(n, dtype=np.float64)
    z -= float(np.mean(z))

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1 = 0.9
    b2 = 0.999
    cb1 = 0.1
    cb2 = 0.999
    eps = 1e-8
    lr = 0.35
    t = 0

    while time.monotonic() < phase_end:
        _, grad = _compute_val_grad(z, beta, lam, n, local_best, global_best)
        if not np.all(np.isfinite(grad)):
            break
        t += 1
        cb1 += 1.0 - b1
        cb2 += 1.0 - b2
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        step = lr * (m / cb1) / (np.sqrt(v / cb2) + eps)
        step = np.where(np.isfinite(step), step, 0.0)
        step = np.clip(step, -1.0, 1.0)
        z = z - step
        z = np.where(np.isfinite(z), z, 0.0)
        z -= float(np.mean(z))

        if t % 32 == 0:
            stuck = time.monotonic() - local_best["time"]
            if stuck > max(10.0, 0.25 * max(1.0, phase_end - time.monotonic())):
                z = _perturb_z(local_best["z"] if local_best["z"] is not None else z,
                               rng, 0.12 + 0.25 * float(rng.random()))
    return z


def _phase_specs(n, available):
    if available <= 0.0:
        return []
    if available < 4.0:
        return [(float(max(16, n)), 0.2, 1.0)]

    factors = [1, 4, 15, 50, 150, 400, 800]
    lams = [0.45, 0.30, 0.18, 0.10, 0.05, 0.02, 0.0]
    weights = [0.05, 0.10, 0.15, 0.20, 0.22, 0.18, 0.10]

    if available < 20.0:
        k = 3
    elif available < 60.0:
        k = 4
    elif available < 150.0:
        k = 6
    else:
        k = 7
    k = min(k, len(factors))

    scale = max(16, int(n))
    total_w = 0.0
    specs = []
    for i in range(k):
        b = float(factors[i] * scale)
        specs.append((b, lams[i], weights[i]))
        total_w += weights[i]
    if total_w <= 0.0:
        return [(1.0, 0.0, 1.0)]

    return [(b, l, w / total_w) for b, l, w in specs]


def _lbfgs_phase(z, phase_end, beta, lam, n, local_best, global_best, rng):
    z = np.asarray(z, dtype=np.float64).copy()
    if z.size != n or not np.all(np.isfinite(z)):
        z = np.zeros(n, dtype=np.float64)
    z -= float(np.mean(z))

    if phase_end <= time.monotonic() + 0.3:
        return z

    last_x = {"x": z.copy()}

    def fun(x):
        return _compute_val_grad(x, beta, lam, n, local_best, global_best)

    def callback(x):
        if time.monotonic() >= phase_end:
            try:
                last_x["x"] = np.asarray(x, dtype=np.float64).copy()
            except Exception:
                pass
            return True
        return False

    if _HAVE_MINIMIZE:
        try:
            options = {
                "maxiter": 100000,
                "maxfun": 100000,
                "maxls": 30,
                "ftol": 1e-12,
                "gtol": 1e-9,
            }
            res = minimize(fun, z, jac=True, method="L-BFGS-B",
                           callback=callback, options=options)
            x = getattr(res, "x", None)
            if x is not None and np.isfinite(x).all():
                x = np.asarray(x, dtype=np.float64).copy()
                x -= float(np.mean(x))
                return x
        except Exception:
            pass

    return _adam_phase(last_x["x"], phase_end, beta, lam, n, local_best, global_best, rng)


def _smooth_noise(n, rng, low_fraction=0.35):
    noise = rng.normal(size=n)
    if n > 8:
        f = np.fft.rfft(noise)
        k = np.arange(f.size, dtype=np.float64)
        cutoff = max(1.0, float(n) * low_fraction)
        f *= np.exp(-(k / cutoff) ** 2)
        noise = np.fft.irfft(f, n)
    noise -= float(np.mean(noise))
    return np.where(np.isfinite(noise), noise, 0.0)


def _perturb_z(z, rng, scale):
    z = np.asarray(z, dtype=np.float64).copy()
    if z.size == 0:
        return z
    if not np.all(np.isfinite(z)):
        z = np.zeros(z.size, dtype=np.float64)
    z -= float(np.mean(z))
    noise = _smooth_noise(z.size, rng)
    z += float(scale) * noise
    z = np.where(np.isfinite(z), z, 0.0)
    z -= float(np.mean(z))
    return z


def _optimize_array(a0, stop_time, rng, global_best):
    a0 = _normalize(a0)
    n = a0.size
    start_time = time.monotonic()
    available = max(0.0, float(stop_time) - start_time)
    if available < 1.0 or n == 0:
        return a0

    init_score = _score_array(a0)
    local_best = {
        "score": init_score if math.isfinite(init_score) else _INF,
        "arr": a0.copy(),
        "z": _logit(a0),
        "time": start_time,
    }

    current_z = local_best["z"].copy()
    specs = _phase_specs(n, available)
    t_acc = 0.0

    for beta, lam, wt in specs:
        if time.monotonic() >= stop_time:
            break
        phase_start = start_time + t_acc
        phase_end = min(float(stop_time), phase_start + wt * available)
        t_acc += wt * available

        if phase_end <= time.monotonic() + 0.5:
            continue

        before = local_best["score"]
        current_z = _lbfgs_phase(current_z, phase_end, beta, lam, n,
                                 local_best, global_best, rng)

        if local_best["z"] is not None and np.all(np.isfinite(local_best["z"])):
            current_z = local_best["z"].copy()

        if time.monotonic() < stop_time - 1.0:
            stuck = time.monotonic() - local_best["time"]
            threshold = max(12.0, 0.18 * available)
            if stuck > threshold:
                scale = 0.12 + 0.30 * float(rng.random())
                if stuck > 2.0 * threshold:
                    scale += 0.25
                current_z = _perturb_z(local_best["z"], rng, scale)

    return _normalize(local_best["arr"])


def _analytic_inits(n, rng):
    n = int(max(1, n))
    if n <= 1:
        return [np.ones(n, dtype=np.float64)]

    x = np.linspace(0.0, 1.0, n)
    out = []

    def add(v):
        v = np.asarray(v, dtype=np.float64).ravel()
        v = np.where(np.isfinite(v), v, 0.0)
        v = np.maximum(v, 0.0)
        if v.size != n:
            return
        s = float(v.sum())
        if math.isfinite(s) and s > 1e-300:
            out.append(v / s)

    add(np.ones(n))

    for pw in (0.5, 1.0, 1.5, 2.0, 3.0):
        add(np.cos(0.5 * np.pi * x) ** pw)

    for al in (-3.0, -2.0, -1.0, -0.5, 0.0, 0.5, 1.0, 2.0, 3.0):
        add(np.exp(al * x))

    for f in (1, 2, 3, 5):
        for amp in (0.15, 0.35, 0.60, 0.85):
            for ph in (0.0, 0.5 * np.pi, np.pi):
                add(1.0 + amp * np.sin(2.0 * np.pi * f * x + ph))
                add(1.0 + amp * np.cos(2.0 * np.pi * f * x + ph))

    for pw in (-1.0, -0.5, 0.5, 1.0):
        add((1.0 + 0.25 * x) ** pw)
        add((1.0 + 0.25 * (1.0 - x)) ** pw)

    add(1.0 + 0.6 * np.exp(-3.0 * (x - 0.5) ** 2))
    add(1.0 + 0.6 * np.exp(-6.0 * (x - 0.5) ** 2))
    add(1.0 + 0.6 * np.exp(-12.0 * (x - 0.5) ** 2))

    if len(out) > 14:
        out = out[:14]
    if not out:
        out = [np.ones(n, dtype=np.float64)]
    return out


def _lp_polish(a0, stop_time, rng, global_best):
    a0 = _normalize(a0)
    n = a0.size
    if not _HAVE_LINPROG or n <= 1 or n > 160:
        return a0

    best = a0.copy()
    best_score = _score_array(best)

    for _ in range(12):
        if time.monotonic() >= stop_time:
            break

        c = _autoconv(best)
        M = float(np.max(c))
        if not math.isfinite(M) or M <= 1e-300:
            break

        L = 2 * n - 1
        A = np.zeros((L, n), dtype=np.float64)
        for k in range(L):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if hi >= lo:
                js = np.arange(lo, hi + 1, dtype=np.int64)
                A[k, js] = best[k - js]

        b_ub = np.full(L, M * 1.000000001, dtype=np.float64)
        bounds = [(0.0, 1000.0)] * n

        res = None
        try:
            res = linprog(-np.ones(n), A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
        except Exception:
            try:
                res = linprog(-np.ones(n), A_ub=A, b_ub=b_ub, bounds=bounds)
            except Exception:
                break

        if res is None or getattr(res, "status", None) != 0:
            break

        S = -float(res.fun)
        if not math.isfinite(S) or S <= 1.0 + 1e-5:
            break

        b = np.asarray(res.x, dtype=np.float64).ravel()
        b = np.where(np.isfinite(b), b, 0.0)
        b = np.maximum(b, 0.0)
        sb = float(b.sum())
        if not math.isfinite(sb) or sb <= 1e-12:
            break
        b /= sb

        improved = False
        for tt in (0.1, 0.3, 0.6, 1.0):
            cand = (1.0 - tt) * best + tt * b
            cand = _normalize(cand)
            sc = _score_array(cand)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                best = cand
                best_score = sc
                improved = True
                if sc < global_best["score"]:
                    global_best["score"] = sc
                    global_best["arr"] = cand.copy()
        if not improved:
            break

    return best


def _valid_list(arr):
    arr = _normalize(arr)
    return [float(x) for x in arr.tolist()]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    start = time.monotonic()
    try:
        budget = max(1.0, float(budget_s))
    except Exception:
        budget = 1300.0

    safety = min(45.0, max(5.0, 0.04 * budget))
    hard_end = start + max(1.0, budget - safety)
    work_end = max(start + 1.0, hard_end - 5.0)
    if work_end > hard_end:
        work_end = hard_end

    rng = np.random.default_rng(seed)

    parent = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        raw = np.where(np.isfinite(raw), raw, 0.0)
        raw = np.clip(raw, 0.0, 1000.0)
        if raw.size > 0:
            s = float(raw.sum())
            if math.isfinite(s) and s > 1e-12:
                parent = raw / s
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(127, dtype=np.float64) / 127.0
    parent = _normalize(parent)

    if parent.size > 5000:
        parent = _resample(parent, 1023)

    main_n = int(parent.size)
    global_best = {
        "score": _score_array(parent),
        "arr": parent.copy(),
    }
    if not math.isfinite(global_best["score"]):
        global_best = {
            "score": _score_array(np.ones(main_n, dtype=np.float64) / float(main_n)),
            "arr": np.ones(main_n, dtype=np.float64) / float(main_n),
        }

    def add_start(starts, arr, max_count=16):
        if len(starts) >= max_count:
            return
        a = _normalize(arr)
        if a.size == 0:
            return
        sc = _score_array(a)
        if math.isfinite(sc):
            starts.append((sc, a.copy()))

    starts = []
    add_start(starts, parent)

    for sd in (0.02, 0.05, 0.10, 0.18):
        z = _logit(parent)
        zp = _perturb_z(z, rng, sd)
        add_start(starts, _softmax(zp))

    for a in _analytic_inits(main_n, rng):
        add_start(starts, a)

    if not starts:
        add_start(starts, np.ones(main_n, dtype=np.float64) / float(main_n))

    T0 = max(1.0, work_end - start)

    if T0 > 500.0 and main_n > 12:
        low_budget_total = min(25.0, 0.02 * T0)
        if low_budget_total > 8.0:
            low_lengths = [L for L in (15, 31, 63) if L < main_n]
            if low_lengths:
                per = low_budget_total / float(len(low_lengths))
                for L in low_lengths:
                    stop = min(work_end, time.monotonic() + per)
                    if stop - time.monotonic() < 2.0:
                        continue
                    best_small = None
                    best_small_score = _INF
                    for init in [_resample(parent, L)] + _analytic_inits(L, rng)[:4]:
                        arr = _normalize(init)
                        sc = _score_array(arr)
                        if math.isfinite(sc) and sc < best_small_score:
                            best_small = arr
                            best_small_score = sc
                    if best_small is None:
                        best_small = np.ones(L, dtype=np.float64) / float(L)
                    small = _optimize_array(best_small, stop, rng, global_best)
                    add_start(starts, _resample(small, main_n))
                starts.sort(key=lambda item: item[0])

    T0 = max(1.0, work_end - start)
    main_budget = min(800.0, 0.50 * T0)
    second_budget = min(220.0, 0.14 * T0)
    third_budget = min(150.0, 0.09 * T0)
    lp_budget = min(80.0, 0.05 * T0)
    large_budget_each = min(120.0, 0.06 * T0)
    final_budget = min(100.0, 0.07 * T0)

    if main_budget > 10.0:
        stop = min(work_end, time.monotonic() + main_budget)
        if len(starts) > 0:
            starts[0][1] = _optimize_array(starts[0][1], stop, rng, global_best)

    if second_budget > 20.0 and len(starts) > 1 and time.monotonic() < work_end - 20.0:
        stop = min(work_end, time.monotonic() + second_budget)
        starts[1][1] = _optimize_array(starts[1][1], stop, rng, global_best)

    if third_budget > 20.0 and len(starts) > 2 and time.monotonic() < work_end - 20.0:
        stop = min(work_end, time.monotonic() + third_budget)
        starts[2][1] = _optimize_array(starts[2][1], stop, rng, global_best)

    if lp_budget > 15.0 and global_best["arr"].size <= 160 and time.monotonic() < work_end - 15.0:
        stop = min(work_end, time.monotonic() + lp_budget)
        _lp_polish(global_best["arr"], stop, rng, global_best)

    if time.monotonic() < work_end - 50.0 and large_budget_each > 15.0:
        best_n = int(global_best["arr"].size)
        large_lengths = []
        for m in (255, 511, 1023, 2047):
            if m > best_n:
                large_lengths.append(m)
            if len(large_lengths) >= 2:
                break
        if not large_lengths:
            m = int(min(2047, max(255, 2 * best_n)))
            if m > best_n:
                large_lengths.append(m)

        for m in large_lengths:
            if time.monotonic() >= work_end - 20.0:
                break
            stop = min(work_end, time.monotonic() + large_budget_each)
            if stop - time.monotonic() < 15.0:
                continue
            init = _resample(global_best["arr"], m)
            _optimize_array(init, stop, rng, global_best)

    if time.monotonic() < work_end - 25.0 and final_budget > 15.0:
        stop = min(work_end, time.monotonic() + final_budget)
        _optimize_array(global_best["arr"], stop, rng, global_best)

    candidates = []
    if global_best.get("arr") is not None:
        candidates.append(global_best["arr"])
    if parent is not None:
        candidates.append(parent)
    candidates.append(np.ones(max(32, main_n), dtype=np.float64) / float(max(32, main_n)))

    final_list = None
    final_score = _INF
    for cand in candidates:
        try:
            lst = _valid_list(cand)
            if not lst:
                continue
            if any(not math.isfinite(v) for v in lst):
                continue
            sc = float(evaluate_sequence(lst))
            if math.isfinite(sc) and sc < final_score:
                final_score = sc
                final_list = lst
        except Exception:
            continue

    if final_list is None:
        return [1.0]

    return final_list
