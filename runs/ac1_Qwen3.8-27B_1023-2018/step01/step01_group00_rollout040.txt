```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _fft_autocorr(a, fsize):
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)
    n = len(a)
    c = c[:2*n-1]  # full linear autocorrelation
    # Due to FFT, small negative values can appear
    c = np.maximum(c, 0)
    return c


def _lse_gradient(a, b, tau, n, fsize):
    """
    Compute gradient of 2n*LSE_tau(b) w.r.t. z (log-softmax parameters).
    Returns grad_z = 4n * a * (c - d)
    where c_m = sum_k p_k * a_{k-m}, d = sum_k p_k * b_k
    """
    # Compute p = softmax(b/tau)
    b_shifted = (b - b.max()) / tau
    p = np.exp(b_shifted)
    p_sum = p.sum()
    if p_sum <= 0 or not np.isfinite(p_sum):
        return None
    p = p / p_sum
    
    # Compute c_m = sum_k p_k * a_{k-m}
    # Direct computation using top values of p
    # For efficiency, use FFT
    # c is cross-correlation: c_m = sum_k p_k * a_{k-m}
    # This can be computed as: if we define b_idx as 0..2n-2, then
    # c_m = sum_k p_k * a[k-m] where a[j]=0 for j<0 or j>=n
    
    # Using FFT: pad p to fsize, pad reversed a to fsize
    # c_m = (p * a_rev_shifted)[m + offset]
    
    # Alternative: use the direct formula with top-p indices
    top_k = min(64, len(p))
    idx = np.argpartition(p, -top_k)[-top_k:] if top_k < len(p) else np.arange(len(p))
    
    c = np.zeros(n, dtype=np.float64)
    d = 0.0
    for k in idx:
        wk = p[k]
        if wk < 1e-12:
            continue
        d += wk * b[k]
        # c_m += wk * a[k-m] for m where 0 <= k-m < n
        # m ranges from max(0, k-n+1) to min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            # a[k-m] for m in [lo, hi] = a[k-hi : k-lo+1][::-1]
            c[lo:hi+1] += wk * a[k-hi:k-lo+1][::-1]
    
    grad = 4.0 * n * a * (c - d)
    return grad


def _optimize_single(n, z_init, stop_time, seed=0):
    """Optimize softmax(z) to minimize 2n*max(autocorr)."""
    rng = np.random.default_rng(seed)
    fsize = next_fast_len(2*n - 1)
    
    z = z_init.copy().astype(np.float64)
    z -= z.mean()
    z = np.clip(z, -15, 15)
    
    best_a = _softmax(z)
    best_score = 2.0 * n  # upper bound (uniform gives score=2)
    
    # Adam
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    lr = 0.4
    b1, b2 = 0.9, 0.999
    
    # Temperature
    tau_frac = 0.3  # fraction of (bmax - bmean)
    
    stuck = 0
    no_improve_count = 0
    last_improve = time.monotonic()
    
    while time.monotonic() < stop_time - 0.5:
        a = _softmax(z)
        b = _fft_autocorr(a, fsize)
        bmax = float(b.max())
        score = 2.0 * n * bmax
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            last_improve = time.monotonic()
            no_improve_count = 0
        
        # Compute gradient
        bmean = float(b.mean())
        b_range = max(bmax - bmean, 1e-10)
        tau = tau_frac * b_range
        
        grad = _lse_gradient(a, b, tau, n, fsize)
        if grad is None:
            stuck += 1
            if stuck > 10:
                break
            continue
        
        # Adam update
        t_adam += 1
        m_adam = b1 * m_adam + (1-b1) * grad
        v_adam = b2 * v_adam + (1-b2) * grad * grad
        mhat = m_adam / (1 - b1**t_adam)
        vhat = v_adam / (1 - b2**t_adam)
        
        z_new = z - lr * mhat / (np.sqrt(vhat) + 1e-8)
        
        # Remove mean (gauge freedom)
        z_new -= z_new.mean()
        z_new = np.clip(z_new, -20, 20)
        z_new[~np.isfinite(z_new)] = 0
        
        # Evaluate new point
        a_new = _softmax(z_new)
        b_new = _fft_autocorr(a_new, fsize)
        score_new = 2.0 * n * float(b_new.max())

            z = z_new
            stuck = 0

                lr = 0.4
                tau_frac = 0.3
                stuck = 0

        if no_improve_count > 500 and time.monotonic() - last_improve > 10:
            # Larger perturbation
            z += rng.normal(0, 1.5, n)
            z -= z.mean()
            z = np.clip(z, -15, 15)
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            t_adam = 0
            lr = 0.4
            tau_frac = max(0.15, tau_frac)
            stuck = 0
            no_improve_count = 0
    
    return best_a, best_score


def _polish_direct(a_init, score_init, stop_time, seed=12345):
    """
    Direct polish: optimize a (not z) using subgradient of the max autocorrelation.
    Use the fact that near optimum, only a few lags are near the max.
    """
    rng = np.random.default_rng(seed)
    n = len(a_init)
    fsize = next_fast_len(2*n - 1)
    
    a = np.maximum(a_init, 1e-10).copy()
    a = a / a.sum()  # normalize to sum=1
    best_a = a.copy()
    best_score = score_init
    
    lr = 0.01
    stuck = 0
    
    while time.monotonic() < stop_time - 1.0:
        b = _fft_autocorr(a, fsize)
        bmax = float(b.max())
        score = 2.0 * n * bmax
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stuck = 0
        
        # Find top lags (near the max)
        threshold = bmax * 0.99
        top_lags = np.where(b >= threshold)[0]
        
        if len(top_lags) == 0:
            top_lags = np.array([int(np.argmax(b))])
        
        # Compute subgradient: sum of gradients of top lags
        grad = np.zeros(n)
        for k in top_lags:
            # grad_m = 2 * a[k-m] (for valid m)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            stuck += 1
            if stuck > 100:
                break
            continue
        
        grad = grad / gnorm
        
        # Update: a_new = a - lr * grad, then project
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            a_new = a.copy()
        else:
            a_new /= a_new_sum
        
        # Evaluate
        b_new = _fft_autocorr(a_new, fsize)
        score_new = 2.0 * n * float(b_new.max())
        
        if score_new < score * (1 + 1e-12):
            a = a_new
            if stuck == 0:
                lr = max(0.001, lr * 1.02)  # slowly increase lr
        else:
            stuck += 1
            lr *= 0.9
            if stuck > 80:
                # Perturb
                a = a + rng.normal(0, 0.01, n)
                a = np.maximum(a, 0)
                a /= a.sum()
                stuck = 0
                lr = 0.01
    
    return best_a, best_score
```

```python
def _lse_grad(a, b, tau, n):
    b_max = b.max()
    p = np.exp((b - b_max) / tau)

    if ps <= 0 or not np.isfinite(ps):
        return None
    p = p / ps

    topk = min(48, b.size)
    if topk < b.size:

    else:
        idx = np.arange(b.size)
```
