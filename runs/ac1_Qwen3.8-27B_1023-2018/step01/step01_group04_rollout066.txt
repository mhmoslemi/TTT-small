            idx_l.append(k - j)
    return (np.array(rows_l, dtype=np.int32),
            np.array(cols_l, dtype=np.int32),
            np.array(idx_l, dtype=np.int32))

def _lp_refine(a0, best, time_limit, rng, deadline):
    a = _normalize(a0.copy())
    n = a.size
    if n < 2:
        return a
    local = _score_fast(a)
    _update_best(a, best)
    rows, cols, idx = _conv_pattern(n)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    stop = time.monotonic() + time_limit
    stagnant = 0
    for it in range(20000):
        if time.monotonic() >= min(stop, deadline - 5):
            break
        b = np.convolve(a, a)
        M = float(np.max(b))
        if not math.isfinite(M) or M <= 0:
            break
        data = a[idx]
        try:
            A = sps.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
            res = linprog(c_obj, A_ub=A,
                          b_ub=np.full(2 * n - 1, M * 1.000001 + 1e-15),
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 8.0})
        except Exception:
            break
        if res is None or not getattr(res, 'success', False) or res.x is None:
            stagnant += 1
            if stagnant > 10:
                break
            continue
        g = np.clip(np.asarray(res.x, dtype=np.float64).ravel(), 0.0, None)
        sg = float(g.sum())
        if sg < 1e-15:
            break
        g /= sg
        best_sc, best_a = local, a.copy()
        for t in (0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0):
            cand = np.clip((1.0 - t) * a + t * g, 0.0, None)
            sc = _score_fast(cand)
            if sc < best_sc:
                best_sc, best_a = sc, cand.copy()
        if best_sc < local * (1.0 - 1e-9):
            a = _normalize(best_a)
            local = best_sc
            stagnant = 0
            _update_best(a, best)
        else:
            stagnant += 1
            if stagnant % 5 == 1:
                pa = _normalize(a * np.exp(0.06 * rng.standard_normal(n)))
                if _score_fast(pa) < local:
                    a, local = pa, _score_fast(pa)
                    stagnant = 0
                    _update_best(a, best)
            if stagnant > 35:
                break
    _update_best(a, best)
    return a

def _gradient_refine(a0, best, time_limit, rng, deadline):
    a = _normalize(a0.copy())
    n = a.size
    if n < 2:
        return a
    local = _score_fast(a)
    _update_best(a, best)
    stop = time.monotonic() + time_limit
    eta = 0.5
    stale = 0
    for it in range(50000):
        if time.monotonic() >= min(stop, deadline - 5):
            break
        b = np.convolve(a, a)
        M = float(np.max(b))
        if M <= 0:
            break
        w = np.exp((b - M) / max(M * 0.02, 1e-15))
        w /= w.sum()
        g = np.zeros(n)
        for j in range(n):
            k0 = j
            k1 = min(j + n, 2 * n - 1)
            g[j] = 2.0 * np.dot(w[k0:k1], a[:k1 - j])
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        g /= gn
        best_sc, best_a = local, a.copy()
        for e in (eta * 0.01, eta * 0.1, eta * 0.5, eta, eta * 2.0, eta * 10.0):
            cand = a - e * g
            cand = np.clip(cand, 0.0, None)
            cs = float(cand.sum())
            if cs > 1e-15:
                cand /= cs
                sc = _score_fast(cand)
                if sc < best_sc:
                    best_sc, best_a = sc, cand.copy()
        if best_sc < local * (1.0 - 1e-9):
            a = _normalize(best_a)
            local = best_sc
            stale = 0
            eta = min(eta * 1.5, 50.0)
            _update_best(a, best)
        else:
            stale += 1
            eta *= 0.5
            if stale % 8 == 0:
                pa = _normalize(a * np.exp(0.04 * rng.standard_normal(n)))
                if _score_fast(pa) < local:
                    a, local = pa, _score_fast(pa)
                    stale = 0
                    _update_best(a, best)
            if stale > 80:
                break
    _update_best(a, best)
    return a

def _nm_polish(a0, best, time_limit, rng, deadline):
    a = _normalize(a0.copy())
    n = a.size
    if n < 3 or n > 300:
        return
    stop = time.monotonic() + time_limit

    def obj(x):
        v = np.exp(np.clip(x, -40.0, 40.0))
        sv = float(v.sum())
        if sv < 1e-30:
            return 1e10
        v /= sv
        return _score_fast(v)

    x0 = np.log(np.clip(a, 1e-30, None))
    try:
        res = minimize(obj, x0, method='Nelder-Mead',
                       options={'maxiter': 8000, 'xatol': 1e-9, 'fatol': 1e-11,
                                'adaptive': True})
        v = _normalize(np.exp(np.clip(res.x, -40.0, 40.0)))
        sc = _score_fast(v)
        if sc < best[0]:
            _update_best(v, best)
    except Exception:
        pass

def _make_inits(n, rng, parent):
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0.0, None)
        if a.size != n:
            a = np.interp(x, np.linspace(0, 1, a.size), a)
        s = float(a.sum())
        if s > 1e-15:
            inits.append(a / s)

    add(np.ones(n))
    for c in (0.3, 0.6, 1.0, 1.5, 2.0):
        add(np.exp(c * (x - 0.5)))
        add(np.exp(-c * (x - 0.5)))
    for k in (1, 2, 3, 5, 8):
        for amp in (0.3, 0.5, 0.7, 0.85, 0.95):
            add(1.0 + amp * np.cos(2.0 * np.pi * k * x))
            add(1.0 + amp * np.sin(2.0 * np.pi * k * x))
    for alpha in (0.5, 1.0, 3.0):
        for _ in range(6):
            try:
                add(rng.dirichlet(np.full(n, alpha)))
            except Exception:
                pass
    if parent is not None and parent.size >= 2:
        p = np.clip(np.asarray(parent, dtype=np.float64).ravel(), 0, None)
        if p.sum() > 0:
            add(np.interp(x, np.linspace(0, 1, p.size), p))
            gp = np.interp(x, np.linspace(0, 1, p.size), p)
            add(np.clip(gp * np.exp(0.03 * rng.standard_normal(n)), 0.01, None))
    half = max(1, n // 2)
    for _ in range(6):
        v = np.clip(1.0 + 0.7 * rng.standard_normal(half), 0.01, None)
        if n % 2 == 0:
            f = np.concatenate([v, v[::-1] * 0.4])
        else:
            f = np.concatenate([v, np.array([1.0]), v[::-1] * 0.4])[:n]
        add(f)
    for pf in (0.15, 0.35, 0.5, 0.65, 0.85):
        pos = int(pf * (n - 1))
        for dec in (0.8, 1.5, 2.5):
            add(np.exp(-dec * np.abs(np.arange(n) - pos) / max(1.0, n * 0.15)))
    if not inits:
        inits.append(np.ones(n) / n)
    scored = []
    for a in inits:
        sc = _score_fast(a)
        if math.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda z: z[0])
    return [a for _, a in scored]

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    hard_start = time.monotonic()
    try:
        remaining = max(0.0, float(budget_s))
    except Exception:
        remaining = 1300.0
    deadline = hard_start + max(0.0, remaining) - 18.0

    best = [float('inf'), np.array([1.0, 1.0], dtype=np.float64)]
    _update_best([1.0, 1.0], best)

    parent = None
    try:
        h = globals().get('height_sequence_1', None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 0:
                _update_best(parent, best)
    except Exception:
        parent = None

    if parent is not None:
        a_p = _normalize(parent.copy())
        alloc = min(420.0, max(0.0, deadline - time.monotonic()) * 0.42)
        if alloc > 30:
            _lp_refine(a_p, best, alloc, rng, deadline)

    lengths = [16, 24, 32, 48, 64, 80, 96, 112, 128, 192, 256]
    if parent is not None:
        pn = parent.size
        for extra in (pn - 8, pn + 8, pn // 2, pn * 2, int(pn * 0.75), int(pn * 1.33)):
            if 8 <= extra <= 512 and extra not in lengths:
                lengths.append(extra)
    lengths = sorted(set(l for l in lengths if 8 <= l <= 512))

    for n_try in lengths:
        rem = max(0.0, deadline - time.monotonic())
        if rem < 80:
            break
        if n_try <= 32:
            alloc_each = min(25.0, rem * 0.12)
            max_starts = 14
        elif n_try <= 96:
            alloc_each = min(50.0, rem * 0.12)
            max_starts = 10
        elif n_try <= 256:
            alloc_each = min(60.0, rem * 0.10)
            max_starts = 6
        else:
            alloc_each = min(50.0, rem * 0.08)
            max_starts = 4
        inits = _make_inits(n_try, rng, parent)
        for a0 in inits[:max_starts]:
            if time.monotonic() >= deadline - 60:
                break
            _lp_refine(a0, best, max(8.0, alloc_each), rng, deadline)
            rem = max(0.0, deadline - time.monotonic())
            if rem > 50 and n_try <= 160:
                _nm_polish(a0, best, min(25.0, rem * 0.2), rng, deadline)

    rem = max(0.0, deadline - time.monotonic())
    if rem > 60 and best[1] is not None and best[1].size <= 300:
        _gradient_refine(best[1], best, min(200.0, rem * 0.55), rng, deadline)

    rem = max(0.0, deadline - time.monotonic())
    if rem > 40 and best[1] is not None and best[1].size <= 300:
        _nm_polish(best[1], best, min(80.0, rem * 0.7), rng, deadline)

    rem = max(0.0, deadline - time.monotonic())
    if rem > 30 and best[1] is not None and best[1].size <= 200:
        _gradient_refine(best[1], best, min(50.0, rem * 0.7), rng, deadline)
        _nm_polish(best[1], best, min(30.0, max(0.0, deadline - time.monotonic()) * 0.5), rng, deadline)

    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.array([1.0], dtype=np.float64)
    s = float(np.sum(final))
    if s < 0.01:
        final = np.ones_like(final, dtype=np.float64)
    else:
        final = final / s
    final = np.clip(final, 0.0, 1000.0)
    _update_best(final, best)
    final = np.asarray(best[1], dtype=np.float64).ravel()
    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc):
        final = np.array([1.0, 1.0], dtype=np.float64)
    return [float(x) for x in final]
