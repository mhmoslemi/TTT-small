def _normalize_array(x):
    try:
        a = np.asarray(x, dtype=np.float64).ravel()
    except Exception:
        return None
    if a.size == 0:
        return None
    a = a[np.isfinite(a)]
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    a = a / s
    a = np.where(np.isfinite(a) & (a >= 0.0), a, 0.0)

def _uniform(n):
    return np.full(max(1, int(n)), 1.0 / max(1, int(n)), dtype=np.float64)


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return None
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(z.size, 1.0 / z.size, dtype=np.float64)
    return e / s


def _upscale(p, n):
    n = max(1, int(n))
    p = _normalize_array(p)
    if p is None:
        return _uniform(n)
    if p.size == n:
        return p
    if p.size < 2:
        return _uniform(n)
    x = np.linspace(0.0, 1.0, p.size, dtype=np.float64)
    y = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(y, x, p)
    out = _normalize_array(out)
    return out if out is not None else _uniform(n)


def _convolution(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 1024:
        return np.convolve(p, p)
    need = 2 * n - 1
    L = 1 << (need - 1).bit_length()
    fp = np.fft.rfft(p, L)
    c = np.fft.irfft(fp * fp, L)[:need]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def _score_array(p):
    p = np.asarray(p, dtype=np.float64).ravel()

        return float("inf")

    if s <= 0.0 or not np.isfinite(s):

    if c.size == 0:
        return float("inf")
    mx = float(c.max())
    if mx < 0.0 or not np.isfinite(mx):
        return float("inf")
    val = 2.0 * float(p.size) * mx / (s * s)
    return float(val) if np.isfinite(val) else float("inf")


def _hankel(w, n):

    if w.size < n:
        w = np.pad(w, (0, n - w.size), mode="constant")

def _finish_weights(w):
    w = np.asarray(w, dtype=np.float64).ravel()
    w = np.where(np.isfinite(w) & (w > 0.0), w, 0.0)
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(w.size, 1.0 / max(1, w.size), dtype=np.float64)
    w = w / s
    eps = 1e-7
    w = (1.0 - eps) * w + eps * np.full(w.size, 1.0 / w.size, dtype=np.float64)
    return w / float(w.sum())


def _weights(c, frac=-1.0, mix=0.0, top_k=0):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0:
        return None
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    cs = float(c.sum())
    if cs <= 0.0 or not np.isfinite(cs):
        return np.full(L, 1.0 / L, dtype=np.float64)
    base = c / cs

    if top_k > 0:
        k = int(min(max(1, top_k), L))
        if k >= L:
            return _finish_weights(base)
        idx = np.argsort(c)[-k:]
        w = np.zeros(L, dtype=np.float64)
        vals = c[idx]
        vsum = float(vals.sum())
        if vsum > 0.0 and np.isfinite(vsum):
            w[idx] = vals / vsum
        else:
            w[idx] = 1.0 / k
        top_mix = 0.2
        w = (1.0 - top_mix) * w + top_mix * base
        return _finish_weights(w)

    if frac < 0.0:
        return _finish_weights(base)

    cmax = float(c.max())
    temp = max(float(frac) * cmax, 1e-18)
    z = (c - cmax) / temp
    z = np.where(z > 0.0, 0.0, z)
    z = np.clip(z, -60.0, 0.0)
    ws = np.exp(z)
    ss = float(ws.sum())
    if ss <= 0.0 or not np.isfinite(ss):
        ws = np.full(L, 1.0 / L, dtype=np.float64)
    else:
        ws = ws / ss
    mix = min(1.0, max(0.0, float(mix)))
    w = (1.0 - mix) * base + mix * ws
    return _finish_weights(w)


def _newton_solve(H, r):
    n = H.shape[0]
    if n <= 0:
        return None
    try:
        hn = float(np.max(np.abs(H)))
    except Exception:
        hn = 1.0
    if not np.isfinite(hn) or hn <= 0.0:
        hn = 1.0
    eye = np.eye(n, dtype=np.float64)
    A = np.zeros((n + 1, n + 1), dtype=np.float64)
    b = np.zeros(n + 1, dtype=np.float64)
    b[:n] = -r
    ridges = (1e-9 * hn, 1e-7 * hn, 1e-5 * hn, 1e-3 * hn, 1e-1 * hn)
    for ridge in ridges:
        A[:n, :n] = H
        if ridge > 0.0:
            A[:n, :n] += ridge * eye
        A[:n, n] = 1.0
        A[n, :n] = 1.0
        A[n, n] = 0.0
        try:
            d = np.linalg.solve(A, b)
            d = np.asarray(d[:n], dtype=np.float64)
            if not np.all(np.isfinite(d)):
                continue
            d = d - float(np.mean(d))
            if float(np.linalg.norm(d)) > 1e-14:
                return d
        except Exception:
            continue
    return None


def _direction(p, w, use_newton):
    n = p.size
    H = _hankel(w, n)
    gp = 2.0 * (H @ p)
    gp = np.where(np.isfinite(gp), gp, 0.0)
    r = gp - float(np.mean(gp))
    r = r - float(np.mean(r))
    if float(np.linalg.norm(r)) < 1e-16:
        return None, gp

    if use_newton:
        d = _newton_solve(H, r)
        if d is not None:
            if float(np.linalg.norm(d)) > 1e-14 and float(np.dot(gp, d)) < 0.0:
                return d, gp
    return -r, gp


def _line_search(cur, delta, cur_sc):
    cur = np.asarray(cur, dtype=np.float64).ravel()
    delta = np.asarray(delta, dtype=np.float64).ravel()
    if delta.size != cur.size:
        return cur, cur_sc

    neg = delta < -1e-12
    tpos = 1.0
    if neg.any():
        vat = np.maximum(cur[neg], 1e-300)
        den = np.maximum(-delta[neg], 1e-300)
        rt = vat / den
        rt = rt[np.isfinite(rt) & (rt > 0.0)]
        if rt.size:
            tpos = min(tpos, 0.98 * float(np.min(rt)))
    if tpos <= 1e-12 or not np.isfinite(tpos):
        return cur, cur_sc

    pn = float(np.linalg.norm(cur))
    dn = float(np.linalg.norm(delta))
    trust = min(1.0, (0.5 * pn / dn) if dn > 1e-15 else 1.0)
    trust = max(1e-5, trust)

    bases = (
        1.0,
        0.5,
        0.25,
        0.1,
        0.05,
        0.01,
        trust,
        0.5 * trust,
        0.2 * trust,
    )
    seen = set()
    cands = []
    for b in bases:
        try:
            t = float(min(tpos, b))
        except Exception:
            continue
        if not np.isfinite(t) or t <= 1e-12:
            continue
        if t > 1.0:
            t = 1.0
        key = round(t, 12)
        if key not in seen:
            seen.add(key)
            cands.append(t)
    cands = sorted(cands, reverse=True)

    best_p = cur
    best_sc = cur_sc
    for t in cands:
        cand = cur + t * delta
        if float(np.min(cand)) < 0.0:
            cand = np.maximum(cand, 0.0)
        s = float(cand.sum())
        if s <= 0.0 or not np.isfinite(s):
            continue
        cand = cand / s
        sc = _score_array(cand)
        if np.isfinite(sc) and sc < best_sc - 1e-14:
            best_sc = sc
            best_p = cand
    return best_p, best_sc


def _lp_direction(p, time_limit=8.0):
    if linprog is None:
        return None
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n <= 0:
        return None
    L = 2 * n - 1
    c = _convolution(p)
    C = float(c.max()) if c.size else 0.0
    if C <= 0.0 or not np.isfinite(C):
        return None

    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        pj = float(p[j])
        if pj > 0.0:
            A[j:j + n, j] = pj

    Afull = np.empty((L, n + 1), dtype=np.float64)
    Afull[:, 0] = -1.0
    Afull[:, 1:] = A

    cobj = np.zeros(n + 1, dtype=np.float64)
    cobj[0] = 1.0
    b_ub = np.zeros(L, dtype=np.float64)

    Aeq = np.zeros((1, n + 1), dtype=np.float64)
    Aeq[0, 1:] = 1.0
    beq = np.array([1.0], dtype=np.float64)

    m_ub = max(10.0 * C + 1e-8, 1.0)
    bounds = [(0.0, m_ub)] + [(0.0, 1.0 + 1e-8)] * n

    def _solve(opts):
        return linprog(
            cobj,
            A_ub=Afull,
            b_ub=b_ub,
            A_eq=Aeq,
            b_eq=beq,
            bounds=bounds,
            method="highs",
            options=opts,
        )

    res = None
    try:
        tl = max(1.0, min(8.0, float(time_limit)))
        res = _solve({"presolve": True, "time_limit": tl})
    except Exception:
        try:
            res = _solve({"presolve": True})
        except Exception:
            res = None

    if res is None or not getattr(res, "success", False):
        return None
    q = np.asarray(res.x[1:], dtype=np.float64).ravel()
    q = np.where(np.isfinite(q) & (q > 0.0), q, 0.0)
    s = float(q.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    q = q / s
    if float(np.max(np.abs(q - p))) < 1e-8:
        return None
    return q


def _perturb(p, rng, sigma):
    p = np.asarray(p, dtype=np.float64).ravel()
    z = np.log(np.maximum(p, 1e-15))
    z = z - float(np.mean(z))
    z = z + float(sigma) * rng.standard_normal(p.size)
    q = _softmax(z)
    if q is None:
        return p.copy()
    return q


def _random_dirichlet(n, rng, alpha):
    n = max(1, int(n))
    return _normalize_array(rng.dirichlet(np.full(n, max(1e-3, float(alpha)))))


def _gaussian_start(n, centers, widths):
    n = max(1, int(n))
    if n == 1:
        return _uniform(1)
    t = np.arange(n, dtype=np.float64) / float(n - 1)
    v = np.zeros(n, dtype=np.float64)
    for c0, w0 in zip(centers, widths):
        if w0 <= 0.0:
            continue
        v += np.exp(-((t - c0) / w0) ** 2)
    out = _normalize_array(v)
    return out if out is not None else _uniform(n)


def _fourier_start(n, rng, order=5):
    n = max(1, int(n))
    if n <= 1:
        return _uniform(n)
    t = np.arange(n, dtype=np.float64) / float(n - 1)
    z = np.zeros(n, dtype=np.float64)
    for m in range(1, order + 1):
        z += 0.55 * float(rng.normal()) * np.cos(2.0 * np.pi * m * t)
        z += 0.55 * float(rng.normal()) * np.sin(2.0 * np.pi * m * t)
    p = _softmax(z)
    return p if p is not None else _uniform(n)

    except Exception:
        best_list = [1.0]

    return [float(x) for x in best_list]
