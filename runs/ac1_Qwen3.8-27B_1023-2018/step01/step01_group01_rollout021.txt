        v_buf = np.zeros(n)
        step = 0
        b1, b2, eps = 0.9, 0.999, 1e-8
        lr = lr_init
        temp = temp_init
        rng = np.random.default_rng(seed + hash(tag) % 10000)

        start = time.monotonic()
        last_eval = start
        best_local = a.copy()
        best_local_sc = safe_eval(a)
        stall = 0

        while time.monotonic() - start < seconds and time.monotonic() < deadline:
            # Autocorrelation
            C = autocorr(a)
            Cmax = float(C.max())

            # Smooth max
            z = temp * (C - Cmax)
            w = np.exp(z)
            wsum = float(w.sum())
            if wsum < 1e-300 or not np.isfinite(wsum):
                w = np.zeros(2 * n - 1)
                idx = np.argmax(C)
                w[idx] = 1.0
                wsum = 1.0
            w /= wsum
            M_sm = Cmax + np.log(wsum) / temp

            # Gradient
            dM = grad_smooth_max(a, w)
            S = float(a.sum())
            grad = (2.0 * n / (S * S)) * dM - (4.0 * n * M_sm) / (S ** 3)
            grad -= grad.mean()
            # Zero gradient where a is at boundary
            grad[a <= 1e-12] = 0.0

            # Adam update
            step += 1
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            m_hat = m_buf / (1.0 - b1 ** step)
            v_hat = v_buf / (1.0 - b2 ** step)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1000.0)
            a = normalize(a)

            # Schedules
            if step % 1500 == 0:
                temp *= 1.25
            if step % 8000 == 0:
                lr *= 0.93

            # Periodic evaluation
            now = time.monotonic()
            if step % 400 == 0 and now - last_eval > 0.8:
                sc = safe_eval(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local = a.copy()
                    stall = 0
                else:
                    stall += 400
                last_eval = now

            # Anti-stall perturbation
            if stall > 30000:
                a = a * (1.0 + 0.005 * rng.standard_normal(n))
                a = np.clip(a, 0.0, 1000.0)
                a = normalize(a)
                m_buf[:] = 0
                v_buf[:] = 0
                lr = lr_init * 0.5
                temp = temp * 0.5
                stall = 0

        # Final eval
        sc = safe_eval(a)
        if sc < best_local_sc:
            best_local_sc = sc
            best_local = a.copy()
        return best_local, best_local_sc

    # Initialize
    a0 = normalize(base.copy())
    best_a = a0.copy()
    best_score = safe_eval(best_a)

    rng = np.random.default_rng(seed)
    avail = max(30.0, deadline - time.monotonic())

    # Phase 1: Main optimization from parent (60% of time)
    t1 = 0.55 * avail
    a_res, sc_res = run_adam(a0, t1, 0.012, 40.0, "main1")
    if sc_res < best_score:
        best_score = sc_res
        best_a = a_res.copy()

    # Phase 2: Continue from best with lower lr (20% of time)
    if time.monotonic() < deadline:
        t2 = 0.20 * avail
        a_res, sc_res = run_adam(best_a, t2, 0.005, 80.0, "main2")
        if sc_res < best_score:
            best_score = sc_res
            best_a = a_res.copy()

    # Phase 3: Try a couple of perturbed restarts (10% of time each)
    for ri in range(2):
        if time.monotonic() >= deadline:
            break
        pert = 0.03 * rng.standard_normal(n)
        a_try = best_a * (1.0 + pert)
        a_try = np.clip(a_try, 0.0, 1000.0)
        a_try = normalize(a_try)
        t_remain = deadline - time.monotonic()
        t3 = 0.10 * avail if ri == 0 else max(10.0, t_remain * 0.7)
        a_res, sc_res = run_adam(a_try, t3, 0.008, 50.0, f"r{ri}")
        if sc_res < best_score:
            best_score = sc_res
            best_a = a_res.copy()

    # Phase 4: Very fine polish (remaining time)
    if time.monotonic() < deadline:
        t4 = max(10.0, deadline - time.monotonic() - 5.0)
        a_res, sc_res = run_adam(best_a, t4, 0.002, 150.0, "polish")
        if sc_res < best_score:
            best_score = sc_res
            best_a = a_res.copy()

    # Final validation
    final_check = safe_eval(best_a)
    if not np.isfinite(final_check) or final_check > best_score + 1e-9:
        sc_alt = safe_eval(a0)
        if np.isfinite(sc_alt):
            best_a = a0.copy()
            best_score = sc_alt

    out = [float(x) for x in np.clip(best_a, 0.0, 1000.0)]
    if not out or any((not np.isfinite(x)) for x in out):
        out = [1.0 / n] * n
    return out
