import numpy as np
import time

try:
    from scipy.optimize import minimize as _scipy_minimize
    from scipy.optimize import linprog as _scipy_linprog
    _HAVE_MINIMIZE = True
    _HAVE_LINPROG = True
except Exception:
    _scipy_minimize = None
    _scipy_linprog = None
    _HAVE_MINIMIZE = False
    _HAVE_LINPROG = False


class _Timeout(Exception):
    pass


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0 / float(z.size), dtype=np.float64)
    return e / s


def _logit(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    floor = 1e-12
    z = np.log(np.maximum(a, floor))
    z = np.where(np.isfinite(z), z, 0.0)
    if z.size:
        z = z - float(np.mean(z))
    return z


def _maxconv(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    c = np.convolve(a, a)
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    return m if np.isfinite(m) else np.inf


def _score_internal(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    m = _maxconv(a)
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * a.size * m / (s * s))


class _Corr:
    """
    For current a (length n) and weight p over convolution lags (length 2n-1),
    return grad_base[j] = sum_k a[k] * p[k + j], so the derivative of
    sum_l p_l c_l is 2 * grad_base.
    """
    def __init__(self, n):
        self.n = int(n)
        self.L = 2 * self.n - 1
        need = self.n + self.L - 1
        size = 1
        while size < need:
            size <<= 1
        self.fft_len = size
        self.idx = np.arange(self.L - 1, self.L - self.n - 1, -1, dtype=np.int64)

    def apply(self, a, p):
        n = self.n
        L = self.L
        a = np.asarray(a, dtype=np.float64)
        p = np.asarray(p, dtype=np.float64)
        if p.size != L:
            tmp = np.zeros(L, dtype=np.float64)
            r = min(L, p.size)
            if r > 0:
                tmp[:r] = p[:r]
            p = tmp
        if a.size != n:
            a = a.copy()
            if a.size < n:
                newa = np.zeros(n, dtype=np.float64)
                newa[:a.size] = a
                a = newa
            else:
                a = a[:n]

        try:
            q = np.zeros(self.fft_len, dtype=np.float64)
            q[:L] = p[::-1]
            A = np.fft.rfft(a, self.fft_len)
            Q = np.fft.rfft(q, self.fft_len)
            out = np.fft.irfft(A * Q, self.fft_len)
            return out[self.idx]
        except Exception:
            out = np.empty(n, dtype=np.float64)
            for j in range(n):
                out[j] = float(np.dot(a, p[j:j + n]))
            return out


def _adam_state(dim):
    return {
        "m": np.zeros(dim, dtype=np.float64),
        "v": np.zeros(dim, dtype=np.float64),
        "b1": 0.1,
        "b2": 0.001,
        "t": 0.0,
    }


def _adam_update(state, grad, lr, cap):
    g = np.asarray(grad, dtype=np.float64)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(g)
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    state["t"] += 1.0
    state["b1"] += 1.0 - b1
    state["b2"] += 1.0 - b2
    state["m"] = b1 * state["m"] + (1.0 - b1) * g
    state["v"] = b2 * state["v"] + (1.0 - b2) * (g * g)
    mhat = state["m"] / state["b1"]
    vhat = state["v"] / state["b2"]
    step = float(lr) * mhat / (np.sqrt(vhat) + eps)
    if not np.all(np.isfinite(step)):
        step = np.zeros_like(g)
    if cap is not None:
        step = np.clip(step, -float(cap), float(cap))
    return step

def _lp_move(a, M):
    """LP: max 1^T b s.t. a*b <= M, b >= 0.  Returns normalized b if possible."""
    if not _HAVE_LINPROG:
        return None, 0.0

    n = int(a.size)
    if n <= 0:
        return None, 0.0
    if n > 420:
        return None, 0.0

    L = 2 * n - 1
    max_a = float(a.max())
    if not np.isfinite(max_a) or max_a <= 0.0:
        return None, 0.0
    if not np.isfinite(M) or M <= 0.0:
        return None, 0.0

    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a

    ub = M / max_a
    ub = max(float(ub), float(np.max(a)) * (1.0 + 1e-10), 1e-12)
    ub *= (1.0 + 1e-10)
    bounds = [(0.0, ub)] * n
    b_ub = np.full(L, M * (1.0 + 1e-10), dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)

    try:
        try:
            res = _scipy_linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
        except TypeError:
            res = _scipy_linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds)

        if res is not None and res.status in (0, 2) and getattr(res, "x", None) is not None:
            b = np.asarray(res.x, dtype=np.float64)
            b = np.where(np.isfinite(b), b, 0.0)
            b = np.maximum(b, 0.0)
            mass = float(b.sum())
            if mass > 1e-12:
                return b / mass, mass
    except Exception:
        pass
    return None, 0.0

            def fun_poly(z, beta=beta, corr=corr):
                z = np.asarray(z, dtype=np.float64)
                aa = _softmax(z)
                cc = np.convolve(aa, aa)
                cmax = float(np.max(cc))
                y = beta * (cc - cmax)
                y = np.where(y > -745.0, y, -745.0)
                e = np.exp(y)
                s = float(e.sum())
                if not np.isfinite(s) or s <= 0.0:
                    return cmax
                return cmax + float(np.log(s)) / float(beta)

            def grad_poly(z, beta=beta, corr=corr):
                z = np.asarray(z, dtype=np.float64)
                aa = _softmax(z)
                cc = np.convolve(aa, aa)
                cmax = float(np.max(cc))
                y = beta * (cc - cmax)
                y = np.where(y > -745.0, y, -745.0)
                e = np.exp(y)
                s = float(e.sum())
                if not np.isfinite(s) or s <= 0.0:
                    p = np.zeros(cc.size, dtype=np.float64)
                    p[int(np.argmax(cc))] = 1.0
                else:
                    p = e / s
                ga = 2.0 * corr.apply(aa, p)
                gz = aa * (ga - float(np.dot(ga, aa)))
                if not np.all(np.isfinite(gz)):
                    gz = np.zeros_like(gz)
                return gz

            calls = [0]

            def callback(xk):
                calls[0] += 1
                if calls[0] % 10 == 0:
                    update_from_z(xk)
                if time.monotonic() >= end_time - 6.0:
                    raise _Timeout()

            try:
                maxit = 220 if beta <= base_beta else 120
                res = _scipy_minimize(
                    fun_poly,
                    z0,
                    jac=grad_poly,
                    method="L-BFGS-B",
                    callback=callback,
                    options={
                        "maxiter": maxit,
                        "maxfun": 4 * maxit,
                        "ftol": 1e-15,
                        "gtol": 1e-7,
                        "maxcor": 50,
                        "maxls": 20,
                    },
                )
                if res is not None and getattr(res, "x", None) is not None:
                    update_from_z(res.x)
            except Exception:
                pass

        # Occasional exact random polish between beta levels.
        if polish:
            pol, polM = _random_polish(holder["a"], rng, trials=3)
            if polM < holder["M"] - 1e-12:
                holder["M"] = polM
                holder["a"] = pol

    return holder["a"], holder["M"]


def _local_search(a0, end_time, rng, use_lp=True, max_lp_calls=250):
    a = _normalize(a0)
    n = int(a.size)
    best = a.copy()
    best_M = _maxconv(best)

    if n < 2:
        return best, best_M

    corr = _Corr(n)
    z = _logit(best)
    curr = best.copy()
    curr_M = best_M

    state = _adam_state(n)
    lr = 0.38
    cap = 1.35
    no_imp = 0
    restart_thr = 130

    it = 0
    lp_calls = 0
    next_lp = 0
    lp_no_improve = 0

    use_lp = bool(use_lp) and _HAVE_LINPROG and n <= 420

    while time.monotonic() < end_time - 1.2:
        # If current is much worse, reset to best occasionally.
        if curr_M > best_M + 1e-7 and no_imp > 20:
            z = _logit(best)
            curr = best.copy()
            curr_M = best_M
            state = _adam_state(n)
            no_imp = 0

        # LP polar move: strong global-ish direction from the parent state idea.
        if (
            use_lp
            and it >= next_lp
            and lp_calls < int(max_lp_calls)
            and time.monotonic() < end_time - 50.0
        ):
            base = best if curr_M >= best_M - 1e-12 else curr
            baseM = _maxconv(base)
            g, mass = _lp_move(base, baseM)
            lp_calls += 1

            if g is not None:
                improved = False
                for t in (0.03, 0.08, 0.15, 0.25, 0.4, 0.6, 0.85, 1.0):
                    cand = (1.0 - float(t)) * base + float(t) * g
                    cand = _normalize(cand)
                    M = _maxconv(cand)
                    if not np.isfinite(M):
                        continue
                    if M < best_M - 1e-12:
                        best = cand.copy()
                        best_M = M
                        improved = True
                    if M < curr_M - 1e-12:
                        curr = cand.copy()
                        curr_M = M
                        z = _logit(curr)
                        state = _adam_state(n)
                if improved:
                    lp_no_improve = 0
                    next_lp = it + 4
                    if best_M < curr_M - 1e-12:
                        curr = best.copy()
                        curr_M = best_M
                        z = _logit(curr)
                        state = _adam_state(n)
                    continue
                else:
                    lp_no_improve += 1
                    gap = 6 + min(80, 2 * lp_no_improve)
                    next_lp = it + gap

        # Exact top-conv gradient descent in logits, with line search.
        c = np.convolve(curr, curr)
        Mcur = float(np.max(c))
        if not np.isfinite(Mcur):
            z = _logit(best)
            curr = best.copy()
            curr_M = best_M
            state = _adam_state(n)
            continue

        if Mcur < best_M - 1e-12:
            best_M = Mcur
            best = curr.copy()
            if Mcur < curr_M - 1e-12:
                curr_M = Mcur

        L = int(c.size)

        # Top-k active set with mild annealing and randomization.
        if it == 0 or it % 35 == 0 or rng.random() < 0.08:
            frac = max(0.012, 0.10 * (0.97 ** (it // 35)))
            frac *= float(rng.uniform(0.45, 1.6))
            k = int(min(L, max(1, frac * L)))
        else:
            k = int(min(L, max(1, 0.04 * L + rng.integers(1, 8))))

        cap_k = min(L, max(1, int(0.38 * L)))
        k = int(min(k, cap_k))
        if k < 1:
            k = 1

        if k >= L:
            p = np.full(L, 1.0 / float(L), dtype=np.float64)
        elif k == 1:
            p = np.zeros(L, dtype=np.float64)
            p[int(np.argmax(c))] = 1.0
        else:
            idx = np.argpartition(c, -k)[-k:]
            p = np.zeros(L, dtype=np.float64)
            p[idx] = 1.0 / float(k)

        ga = 2.0 * corr.apply(curr, p)
        if not np.all(np.isfinite(ga)):
            ga = np.zeros_like(ga)

        gz = curr * (ga - float(np.dot(ga, curr)))
        if not np.all(np.isfinite(gz)):
            gz = np.zeros_like(gz)

        step = _adam_update(state, gz, lr, cap)
        if not np.all(np.isfinite(step)) or np.all(step == 0.0):
            no_imp += 1
            if no_imp > 10:
                lr = max(0.04, lr * 0.85)
            if no_imp > restart_thr:
                zb = _logit(best) if rng.random() > 0.25 else _logit(best[::-1])
                sig = 0.025 + 0.05 * min(4, no_imp // restart_thr)
                z = zb + sig * rng.standard_normal(n)
                curr = _softmax(z)
                curr_M = _maxconv(curr)
                state = _adam_state(n)
                lr = 0.35
                no_imp = 0
                restart_thr = int(restart_thr * 1.2)
                if np.isfinite(curr_M) and curr_M < best_M - 1e-12:
                    best_M = curr_M
                    best = curr.copy()
            it += 1
            continue

        base_z = z.copy()
        accepted = False
        for fac in (1.0, 0.6, 0.35, 0.18, 0.08):
            z_try = base_z + float(fac) * step
            a_try = _softmax(z_try)
            M_try = _maxconv(a_try)
            if not np.isfinite(M_try):
                continue
            if M_try < curr_M - 1e-12:
                z = z_try
                curr = a_try
                curr_M = M_try
                no_imp = 0
                accepted = True
                if M_try < best_M - 1e-12:
                    best_M = M_try
                    best = curr.copy()
                    lr = min(0.85, lr * 1.025)
                break
            if M_try < best_M - 1e-12 and M_try <= curr_M + 0.08 * (curr_M - best_M + 1e-12):
                z = z_try
                curr = a_try
                curr_M = M_try
                no_imp = 0
                accepted = True
                if M_try < best_M - 1e-12:
                    best_M = M_try
                    best = curr.copy()
                break

        if not accepted:
            no_imp += 1
            if no_imp == 12:
                lr = max(0.04, lr * 0.82)
            if no_imp > restart_thr:
                zb = _logit(best) if rng.random() > 0.3 else _logit(best[::-1])
                sig = 0.025 + 0.05 * min(4, no_imp // restart_thr)
                z = zb + sig * rng.standard_normal(n)
                curr = _softmax(z)
                curr_M = _maxconv(curr)
                state = _adam_state(n)
                lr = 0.35
                no_imp = 0
                restart_thr = int(restart_thr * 1.2)
                if np.isfinite(curr_M) and curr_M < best_M - 1e-12:
                    best_M = curr_M
                    best = curr.copy()

        if it % 25 == 0:
            pol, polM = _random_polish(best, rng, trials=2)
            if polM < best_M - 1e-12:
                best_M = polM
                best = pol
                if curr_M > best_M + 1e-8:
                    curr = best.copy()
                    curr_M = best_M
                    z = _logit(curr)
                    state = _adam_state(n)

        it += 1

    # Final local polish on the best point.
    try:
        if time.monotonic() < end_time - 0.5:
            pol, polM = _random_polish(best, rng, trials=4)
            if polM < best_M - 1e-12:
                best = pol
                best_M = polM
    except Exception:
        pass

    return best, best_M


def _initial_pool(a0, rng):
    a0 = _normalize(a0)
    n = int(a0.size)
    cands = [a0, a0[::-1]]

    if n > 1:
        shifts = [1, 2, 3, 5, 8, 13, 21, n // 3, n // 2]
        for s in shifts:
            s = int(s) % n
            if 0 < s < n:
                cands.append(np.roll(a0, s))

    cands.append((a0 + a0[::-1]) * 0.5)

    if n > 5:
        x = np.linspace(0.0, 1.0, n)
        for amp in (0.05, 0.12, 0.22, 0.35):
            for fr in (1, 2, 3, 5, 7):
                w = 1.0 + float(amp) * np.cos(2.0 * np.pi * float(fr) * x)
                cands.append(np.maximum(w, 1e-6))

    best = None
    bestM = np.inf
    for c in cands:
        try:
            c = _normalize(c)
            m = _maxconv(c)
            if np.isfinite(m) and m < bestM - 1e-13:
                bestM = m
                best = c.copy()
        except Exception:
            pass

    if best is None:
        best = a0.copy()
        bestM = _maxconv(best)
    return best, bestM


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    start = time.monotonic()
    try:
        budget = float(max(10.0, float(budget_s)))
    except Exception:
        budget = 1300.0

    safety = 18.0
    hard_deadline = start + budget - safety
    final_deadline = hard_deadline - 6.0

    rng = np.random.default_rng(seed)

    # Get the supplied parent directly; do not copy its literal values into code.
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.size > 0 and float(p.sum()) > 1e-300:
            parent = p / float(p.sum())
    except Exception:
        parent = None

    if parent is None:
        n_default = 127
        x = np.linspace(0.0, 1.0, n_default)
        parent = np.maximum(1.0 + 0.18 * np.cos(2.0 * np.pi * x), 1e-4)
        parent = parent / parent.sum()

    n = int(parent.size)
    fallback = [float(v) for v in parent]

    best_list = None
    best_score = np.inf

    def consider(arr):
        nonlocal best_list, best_score
        try:
            lst = [float(x) for x in np.asarray(arr, dtype=np.float64)]
            if not lst:
                return
            if any(not np.isfinite(v) for v in lst):
                return
            try:
                sc = float(evaluate_sequence(lst))
            except Exception:
                sc = _score_internal(lst)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_list = lst
        except Exception:
            pass

    consider(parent)

    if time.monotonic() >= final_deadline - 2.0:
        return best_list if best_list is not None else fallback

    try:
        # Initial multi-start within the parent's length.
        a_init, _ = _initial_pool(parent, rng)
        consider(a_init)

        remaining = max(1.0, final_deadline - time.monotonic())

        lbf_end = start + 0.0
        if remaining > 180.0 and _HAVE_MINIMIZE and n <= 2048:
            lbf_end = start + min(180.0, 0.18 * remaining)
            a_lbf, _ = _lbfgs_polish(a_init, lbf_end, rng, polish=True)
            consider(a_lbf)
        else:
            a_lbf = a_init

        local_end = final_deadline - 3.0
        a_loc, _ = _local_search(a_lbf, local_end, rng, use_lp=True, max_lp_calls=260)
        consider(a_loc)

        # If time remains, do a shorter polish from best returned by exact evaluator.
        if time.monotonic() < final_deadline - 60.0 and best_list is not None:
            a_best = _normalize(np.asarray(best_list, dtype=np.float64))
            a_fin, _ = _local_search(a_best, final_deadline - 4.0, rng, use_lp=False, max_lp_calls=0)
            consider(a_fin)
    except Exception:
        pass

    if best_list is None:
        best_list = fallback

    # Final exact validation and return a safe finite list.
    try:
        out = [float(x) for x in best_list]
        if not out or any(not np.isfinite(x) for x in out):
            raise ValueError
    except Exception:
        out = [1.0]

    return out
