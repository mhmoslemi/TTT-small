        # Using the same formula: dM = 2 * conv2[n-1:2n-1][::-1]
        wfl = w[::-1]
        fa = np.fft.rfft(a, sizeG)
        fw = np.fft.rfft(wfl, sizeG)
        conv2 = np.fft.irfft(fa * fw, sizeG)[:2 * n - 1]
        dM = 2.0 * conv2[n - 1:2 * n - 1][::-1]
        
        invS2 = 1.0 / (S * S)
        grad = 2.0 * n * invS2 * dM - 4.0 * n * Cmax * invS2 / S
        val = 2.0 * n * Cmax * invS2
        return val, grad, C, idx
    
    def adam_loop(a, seconds, temp_fn, lr0, use_exact=False, K=3):
        """Adam optimization loop."""
        nonlocal best, best_score
        a = normalize(a.copy())
        m_adam = np.zeros(n)
        v_adam = np.zeros(n)
        step = 0
        b1, b2 = 0.9, 0.999
        
        sc = safe_eval(a)
        if sc < best_score:
            best, best_score = a.copy(), sc
        
        start = time.monotonic()
        last_eval = start
        
        while time.monotonic() < deadline and (time.monotonic() - start) < seconds:
            if use_exact:
                _, grad, C, idx = exact_subgrad(a, K)
            else:
                _, grad = smax_grad(a, temp_fn(step))
            
            step += 1
            m_adam = b1 * m_adam + (1.0 - b1) * grad
            v_adam = b2 * v_adam + (1.0 - b2) * grad * grad
            mb = m_adam / (1.0 - b1 ** step)
            vb = v_adam / (1.0 - b2 ** step)
            
            # Learning rate schedule
            lr = lr0 * (0.97 ** (step / 3000.0))
            
            a = a - lr * mb / (np.sqrt(vb) + 1e-8)
            a = np.clip(a, 0.0, 1000.0)
            a = normalize(a)
            
            # Periodic exact evaluation
            now = time.monotonic()
            if now - last_eval > 0.4:
                last_eval = now
                sc = safe_eval(a)
                if sc < best_score:
                    best, best_score = a.copy(), sc
        
        sc = safe_eval(a)
        if sc < best_score:
            best, best_score = a.copy(), sc
        return a
    
    # ---- Optimization phases ----
    
    # Phase 1: Main search from parent, moderate temperature
    elapsed = time.monotonic() - t0
    avail = max(30.0, deadline - time.monotonic())
    a = adam_loop(base, 0.45 * avail, lambda s: 8.0 + s * 0.003, 0.04)
    
    # Phase 2: Refinement from multiple starting points
    if time.monotonic() < deadline:
        elapsed = time.monotonic() - t0
        avail = max(20.0, deadline - time.monotonic())
        per = avail / 4.0
        
        # From current best
        a = adam_loop(a, per, lambda s: 20.0 + s * 0.01, 0.015)
        
        # From best
        if time.monotonic() < deadline:
            a2 = adam_loop(best.copy(), per, lambda s: 25.0 + s * 0.01, 0.01)
        
        # From a perturbed version

            pert = best + 0.05 * rng.standard_normal(n)
            a3 = adam_loop(pert, per, lambda s: 30.0 + s * 0.01, 0.008)

            smooth = np.convolve(best, np.ones(21) / 21.0, mode='same')
            a4 = adam_loop(smooth, per, lambda s: 35.0 + s * 0.01, 0.006)
    
    # Phase 3: Exact subgradient polish
    if time.monotonic() < deadline:
        avail = max(10.0, deadline - time.monotonic())
        a = adam_loop(best.copy(), 0.5 * avail, lambda s: 1000.0, 0.004, use_exact=True, K=3)
    
    # Phase 4: Final micro-polish

        avail = max(5.0, deadline - time.monotonic())
        a = adam_loop(best.copy(), avail, lambda s: 1000.0, 0.002, use_exact=True, K=1)
    
    # Final validation
    final = normalize(np.clip(np.asarray(best), 0.0, 1000.0))
    fs = safe_eval(final)
    if not np.isfinite(fs) or fs > best_score + 1e-15:
        final = best.copy()
    
    # Ensure valid output
    out_list = [float(x) for x in final]
    if not out_list or any(not np.isfinite(x) or x < 0 for x in out_list):
        out_list = [1.0] * n
    
    return out_list
