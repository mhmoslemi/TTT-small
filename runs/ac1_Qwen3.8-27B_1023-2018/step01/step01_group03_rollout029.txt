    return 1 << (n - 1).bit_length() if n > 1 else 1

def _fconv(a, b):
    n = len(a) + len(b) - 1
    f = _npow2(n)
    return np.real(ifft(fft(a, f) * fft(b, f)))[:n]

    q = np.clip(np.asarray(q, float), 0, 1000)
    s = q.sum()
    if s < 1e-15:
        return np.ones(len(q)) / len(q)
    return q / s

    n = len(a)
    b = _fconv(a, a)
    s = a.sum()
    return 2.0 * n * b.max() / (s * s) if s > 0.01 else float('inf')

def _grad(a, b, tol=0.03):
    M = b.max()
    if M <= 0:
        return np.zeros(len(a))
    thr = M * (1 - tol)
    act = np.where(b >= thr)[0]
    w = np.exp(8.0 * (b[act] - M))

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft

def _np2(n):
    if n <= 1: return 1
    return 1 << (n - 1).bit_length()

def _fc(a, b):
    n = len(a) + len(b) - 1
    f = _np2(n)
    return np.real(ifft(fft(a, f) * fft(b, f)))[:n]

def _nz(q):
    q = np.clip(np.asarray(q, float), 0.0, 1000.0)
    s = q.sum()
    if s < 1e-15:
        return np.ones(len(q)) / len(q)
    return q / s

def _sc(a):
    b = _fc(a, a)
    s = a.sum()
    if s < 0.01: return float('inf')
    return 2.0 * len(a) * b.max() / (s * s)

def _grad_max(a, b):
    n = len(a)
    M = b.max()
    if M <= 0: return np.zeros(n)
    w = np.exp(12.0 * (b - M))
    ws = w.sum()
    if ws < 1e-15: return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    act = np.where(w > w.max() * 1e-3)[0]
    for k in act:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += 2.0 * w[k] * a[k-hi:k-lo+1]
    return g

def _lp_linearized(a, b, deadline):
    """Solve min t s.t. b + a*δ <= t, sum(δ)=0, a+δ >= 0"""
    n = len(a)
    # Build A: A[k, j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    # LP: min t
    # s.t. sum_j a[k-j]*δ[j] - t <= -b[k]  for each k
    #      sum(δ) = 0
    #      δ >= -a (i.e., a + δ >= 0)
    #      t free
    
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = -b
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    c = np.zeros(n+1)
    c[-1] = 1.0
    bounds = [(-a[i], None) for i in range(n)] + [(None, None)]
    
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': max(1, deadline - time.monotonic() - 5)})
        if r.success:
            delta = r.x[:n]
            if np.isfinite(delta).all():
                return delta
    except Exception:
        pass
    return None

def _lp_maxsum(a, b, deadline):
    """Solve max sum(g) s.t. a*g <= M, g >= 0"""
    n = len(a)
    M = b.max()
    if M <= 0: return None
    
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                    bounds=[(0, 1e6)]*n, method='highs',
                    options={'presolve': True, 'time_limit': max(1, deadline - time.monotonic() - 5)})
        if r.success and r.x.sum() > 1e-15:
            return _nz(r.x)
    except Exception:
        pass
    return None

def _optimize(a, rng, deadline, use_lp=True, max_lp_iter=200, max_sg_iter=100):
    a = _nz(a)
    best_a, best_s = a.copy(), _sc(a)
    n = len(a)
    stag = 0
    
    # Phase 1: Linearized LP
    if use_lp:
        for it in range(max_lp_iter):
            if time.monotonic() > deadline - 5:
                break
            b = _fc(a, a)
            delta = _lp_linearized(a, b, deadline)
            if delta is None:
                break
            # Line search on step size
            improved = False
            for step in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
                cand = _nz(a + step * delta)
                cs = _sc(cand)
                if cs < best_s - 1e-12:
                    best_a, best_s = cand, cs
                    a = cand
                    improved = True
                    break
            s_now = _sc(a)
            if s_now < best_s + 1e-10:
                stag = 0
            else:
                stag += 1
            if stag > 25:
                # Random perturbation
                eps = 0.05 * max(1.0/n, 0.01)
                a = _nz(a + eps * rng.standard_normal(n))
                stag = 0
                if stag > 40:
                    break
    
    # Phase 2: Subgradient polish
    b = _fc(a, a)
    cur_s = _sc(a)
    eta = 0.15
    for it in range(max_sg_iter):
        if time.monotonic() > deadline - 3:
            break
        b = _fc(a, a)
        g = _grad_max(a, b)
        g -= g.mean()
        gr = g.max() - g.min()
        if gr < 1e-15:
            break
        step = eta / gr
        for _ in range(5):
            cand = _nz(a - step * g)
            cs = _sc(cand)
            if cs < cur_s * (1 - 1e-6):
                a = cand
                cur_s = cs
                break
            step *= 0.5
        if _sc(a) < best_s - 1e-12:
            best_a, best_s = a.copy(), _sc(a)
    
    return best_a, best_s

def _ups(a, n):

        return _nz(a)
    return _nz(np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(a)), a))
