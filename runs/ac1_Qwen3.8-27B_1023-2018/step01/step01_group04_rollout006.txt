def _project_simplex(v):
    v=_clean(v)
    n=v.size
    if n==0: return v
    s=v.sum()
    if s>0 and np.min(v)>=0:
       u=v/s
       if np.min(u)>=0: return u
    b=np.sort(v)[::-1]
    css=np.cumsum(b)
    rho=np.arange(1,n+1,dtype=float)
    cond=b - (css-1.0)/rho > 0

    u=np.maximum(v-theta,0.0)
    su=u.sum()
    if su<=1e-15: return np.ones(n)/n
    return u/su

def _clean(a, upper=None):
    a = np.asarray(a, dtype=np.float64).ravel()

def _sum1(a):
    a = _clean(a, 1000.0)
    n = a.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _fast_score(a):
    a = _clean(a, 1000.0)
    n = a.size
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-15:
        return float("inf")

    m = float(np.max(p))
    if not math.isfinite(m):
        return float("inf")
    return float(2.0 * n * m / (s * s))

            return

def _project_simplex(v):
    v = _clean(v)
    n = v.size
    if n == 0:
        return v
    if np.min(v) >= 0.0:
        s = float(v.sum())
        if math.isfinite(s) and s > 1e-15:
            return v / s
    b = np.sort(v)[::-1]
    css = np.cumsum(b)
    rho = np.arange(1, n + 1, dtype=np.float64)
    idx = np.nonzero(b - (css - 1.0) / rho > 0.0)[0]
    if idx.size == 0:
        theta = (float(v.sum()) - 1.0) / float(n)
    else:
        z = int(idx[-1])
        theta = (float(css[z]) - 1.0) / float(z + 1)
    u = np.maximum(v - theta, 0.0)
    su = float(u.sum())
    if not math.isfinite(su) or su <= 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return u / su


def _softmax(x):
    x = np.asarray(x, dtype=np.float64).ravel()

    if n == 0:
        return np.ones(1, dtype=np.float64)
    x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
    xc = x - float(np.mean(x))
    m = float(np.max(xc))
    y = np.exp(xc - m)
    s = float(y.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return y / s


def _h_flat(n):
    n = int(max(2, n))
    a = np.zeros(n, dtype=np.float64)

        a[i] = s
    a = np.maximum(a, 0.0)
    ss = float(a.sum())
    if ss <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        ss = float(n)
    return a / ss


def _interp_to_length(seq, n):

def _build_A(a):
    if sps is None:
        return None
    n = int(a.size)
    if n <= 0:
        return None
    rows = np.repeat(np.arange(n, dtype=np.int64), n) + np.tile(np.arange(n, dtype=np.int64), n)
    cols = np.tile(np.arange(n, dtype=np.int64), n)
    data = np.tile(a.astype(np.float64, copy=False), n)
    try:
        A = sps.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        A.sum_duplicates()
        return A
    except Exception:
        return None

def _optimize_n(parent, n, time_limit, rng, best, hard_deadline=None):

        return None

        outer_stop = now + max(0.0, time_limit)
    else:
        outer_stop = min(now + max(0.0, time_limit), hard_deadline - 10.0)
    eff = outer_stop - now
    if eff < 12:
        return None

    starts = _make_starts(parent, n, rng)
    if not starts:
        return None

    weights = (1.6, 1.2, 1.0, 0.85, 0.7, 0.55, 0.45, 0.40, 0.35, 0.30)
    uses = min(len(starts), len(weights))
    wsum = float(sum(weights[:uses]))

    n_best = starts[0].copy()
    n_best_sc = _fast_score(n_best)
    _update_best(n_best, best)

    t_start = time.monotonic()
    for i, s in enumerate(starts[:uses]):
        elapsed = time.monotonic() - t_start
        rem = eff - elapsed
        if rem < 8.0:
            break
        alloc = min(rem * 0.92, eff * weights[i] / max(1e-12, wsum))
        if alloc < 6.0:
            break

        a_opt = _smooth_optimize(s, alloc, best)
        if a_opt is not None and a_opt.size == n:
            sc = _fast_score(a_opt)
            if math.isfinite(sc) and sc < n_best_sc - 1e-15:
                n_best_sc = float(sc)
                n_best = a_opt.copy()
                _update_best(n_best, best)

    rem = max(0.0, eff - (time.monotonic() - t_start))
    if n_best.size == n and rem >= 8.0 and time.monotonic() < outer_stop - 5.0:
        tp = min(30.0, rem - 2.0, outer_stop - time.monotonic() - 4.0)
        if tp > 5.0:
            n_best = _subgrad_polish(n_best, tp, best, rng)

    if n_best.size == n and n <= 400 and time.monotonic() < outer_stop - 12.0:
        rem = max(0.0, outer_stop - time.monotonic() - 6.0)
        tp = min(50.0, rem)
        if tp > 10.0:
            n_best = _lp_polish(n_best, tp, best, rng)

    if (
        n_best.size == n and n <= 768
        and time.monotonic() < outer_stop - 30.0
    ):
        rem = max(0.0, outer_stop - time.monotonic() - 12.0)
        ts = min(40.0, rem)
        if ts > 18.0:
            n_best = _smooth_optimize(n_best, ts, best)
            rem = max(0.0, outer_stop - time.monotonic() - 5.0)
            if rem > 8.0:
                n_best = _subgrad_polish(n_best, min(15.0, rem), best, rng)

    _update_best(n_best, best)
    return n_best


def _finalize(best, parent):
    cands = []

    def add(arr):
        if arr is None:
            return
        arr = _sum1(arr)
        if arr.size:
            cands.append(arr.copy())

    add(best[1])
    if parent is not None:
        add(parent)

    la = _sum1(best[1])
    if la.size > 1:
        for r in (0.1, 0.3, 0.5):
            add((1.0 - r) * la + r * la[::-1])

    add(_h_flat(max(4, parent.size if parent is not None else 96)))

    ev = globals().get("evaluate_sequence", None)
    best_list = None
    best_sc = float("inf")
    for arr in cands:
        lst = [float(x) for x in arr]
        sc = float("inf")
        if ev is not None:
            try:
                sc = float(ev(lst))
            except Exception:
                sc = float("inf")
        if not math.isfinite(sc):
            sc = _fast_score(arr)
        if math.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_list = lst

    if best_list is None:
        fb = _sum1(_h_flat(96))
        best_list = [float(x) for x in fb]
    return best_list


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    try:
        remaining_budget = max(0.0, float(budget_s))
    except Exception:
        remaining_budget = 1300.0
    hard_deadline = t0 + max(0.0, remaining_budget - 28.0)

    best = [float("inf"), np.array([1.0, 1.0], dtype=np.float64)]

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = _sum1(h)
            if p.size:
                sc = _fast_score(p)
                if math.isfinite(sc):
                    best = [float(sc), p.copy()]
                parent = p.copy()
    except Exception:
        parent = None

    if parent is None:
        parent = _h_flat(96)
    _update_best(parent, best)
    parent_len = int(max(4, parent.size)) if parent.size else 96

    if time.monotonic() >= hard_deadline - 35.0:
        return _finalize(best, parent)

    raw_lengths = [
        parent_len,
        int(round(parent_len * 1.25)),
        int(round(parent_len * 1.5)),
        int(round(parent_len * 2.0)),
        int(round(parent_len * 4.0)),
        int(round(parent_len * 0.75)),
        int(round(parent_len * 0.5)),
        512,
    ]
    lengths = []
    for n in raw_lengths:
        n = int(n)
        if 4 <= n <= MAX_OPT_N and n not in lengths:
            lengths.append(n)
    lengths = lengths[:8]

    fracs = (0.46, 0.18, 0.13, 0.10, 0.08, 0.06, 0.05, 0.05)
    caps = (450.0, 180.0, 130.0, 100.0, 85.0, 65.0, 55.0, 55.0)

    for idx, n in enumerate(lengths):
        now = time.monotonic()
        rem = hard_deadline - now
        if rem < 120.0:
            break
        f = fracs[idx] if idx < len(fracs) else 0.03
        alloc = min(f * rem, caps[idx] if idx < len(caps) else 50.0)
        alloc = max(0.0, alloc - 70.0)
        if alloc < 18.0:
            break
        _optimize_n(parent, n, alloc, rng, best, hard_deadline)

    # Final polishing pass on the globally best candidate.
    if time.monotonic() < hard_deadline - 45.0:
        a = _sum1(best[1])
        n = int(a.size)
        if 2 <= n <= 768:
            tp = min(35.0, max(0.0, hard_deadline - time.monotonic() - 30.0))
            if tp > 8.0:
                a = _subgrad_polish(a, tp, best, rng)
            if 2 <= a.size <= 400:
                tp2 = min(45.0, max(0.0, hard_deadline - time.monotonic() - 35.0))
                if tp2 > 10.0:
                    a = _lp_polish(a, tp2, best, rng)
            if time.monotonic() < hard_deadline - 55.0:
                tp3 = min(70.0, max(0.0, hard_deadline - time.monotonic() - 55.0))
                if tp3 > 20.0:
                    a = _smooth_optimize(a, tp3, best)
                    tp4 = min(20.0, max(0.0, hard_deadline - time.monotonic() - 22.0))
                    if tp4 > 6.0:
                        a = _subgrad_polish(a, tp4, best, rng)

    return _finalize(best, parent)
