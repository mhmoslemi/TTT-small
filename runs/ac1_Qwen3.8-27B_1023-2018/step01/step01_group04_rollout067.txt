    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
        arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size == 0:
            return
        s = float(np.sum(arr))
        if s <= 1e-12:
            arr = np.ones_like(arr, dtype=np.float64) * 0.01 / max(1, arr.size)
            s = float(arr.size)
        arr = arr / s
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(np.sum(arr))
        if s < 0.01:
            arr = np.clip(arr + 0.001, 0.0, 1000.0)
            s = float(np.sum(arr))
            if s > 0.0:
                arr = arr / s
        sc = evaluate_sequence([float(x) for x in arr])
        if math.isfinite(sc) and sc < float(best[0]):
            best[0] = float(sc)
            best[1] = arr.copy()
    except Exception:
        pass


def _build_A(a):
    if sps is None:
        return None
    n = len(a)
    if n <= 0:
        return None
    j = np.repeat(np.arange(n, dtype=np.int64), n)
    k = np.tile(np.arange(n, dtype=np.int64), n)
    rows = j + k
    cols = j
    data = np.tile(np.asarray(a, dtype=np.float64), n)
    try:
        return sps.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    except Exception:
        return None


def _make_starts(n, rng, parent):
    starts = []
    x = np.linspace(0.0, 1.0, n)

    def add(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0.0, None)
        if a.size != n:
            return
        s = float(a.sum())
        if s > 1e-15:
            starts.append(a / s)

    add(np.ones(n, dtype=np.float64))

    # Monotonically-decreasing ramps: skews the self-convolution peak downward.
    for g in (0.5, 0.7, 0.9, 1.2):
        add(np.linspace(1.0 + g, 1.0 - g, n))
    for g in (0.3, 0.6, 1.0):
        add(np.power(np.linspace(1.0, 0.0, n), g) + 1e-3)

    for c in (0.5, 1.0, 1.5, 2.5, 3.5):
        add(np.exp(c * (x - 0.5)))

    for k in range(1, min(6, max(1, n // 2)) + 1):
        for amp in (0.3, 0.5, 0.7, 0.9):
            add(1.0 + amp * np.sin(2.0 * math.pi * k * (x - 0.5)))
            add(1.0 - amp * np.sin(2.0 * math.pi * k * (x - 0.5)))

    for alpha in (0.25, 0.5, 1.0):
        try:
            add(rng.dirichlet(np.full(n, alpha)))
        except Exception:
            pass

    half = max(1, n // 2)
    for _ in range(6):
        v = np.tanh(0.8 * rng.standard_normal(half))
        v = np.clip(v, -1.0, 1.0)
        f = np.ones(n, dtype=np.float64)
        f[:half] = 1.0 + v
        if n % 2 == 0:
            f[half:] = 1.0 - v[::-1]
        else:
            f[half:] = 1.0 - v[::-1][:n - half]
            f[half] = 1.0
        add(f)

    if parent is not None:
        try:
            p = np.asarray(parent, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
            p = np.clip(p, 0.0, None)
            if p.size >= 2 and p.sum() > 0.0:
                tp = np.linspace(0.0, 1.0, p.size)
                xp = np.linspace(0.0, 1.0, n)
                gp = np.interp(xp, tp, p)
                add(gp)
                for sgn in (0.02, 0.05, 0.10):
                    gp2 = gp * np.exp(sgn * rng.standard_normal(n))
                    add(gp2)
        except Exception:
            pass

    if not starts:
        starts.append(np.ones(n, dtype=np.float64) / n)

    scored = []
    for a in starts:
        sc = _score_fast(a)
        if math.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda z: z[0])
    return [a for _, a in scored]


def _search_lp(n, time_limit, rng, best, parent, stop):
    n = int(n)
    n < 2 and return
    if linprog is None or sps is None:
        return
    if time_limit <= 0:
        return

    loc_stop = min(time.monotonic() + time_limit, stop)
    starts = _make_starts(n, rng, parent)
    if not starts:
        return

    if n <= 64:
        max_starts = 8
        max_iters = 220
    elif n <= 160:
        max_starts = 8
        max_iters = 150
    else:
        max_starts = 6
        max_iters = 90

    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, 1.0)] * n

    for a0 in starts[:max_starts]:
        a = np.asarray(a0, dtype=np.float64).copy()
        local = _score_fast(a)
        if math.isfinite(local):
            _update_best(a, best)

        stagnant = 0
        for it in range(max_iters):
            if time.monotonic() >= loc_stop - 1:
                return

            b = np.convolve(a, a)
            M = float(np.max(b))
            if not math.isfinite(M) or M <= 0.0:
                break

            A = _build_A(a)
            if A is None:
                break

            try:
                res = linprog(
                    c,
                    A_ub=A,
                    b_ub=np.full(2 * n - 1, M * (1.0 + 1e-11) + 1e-14, dtype=np.float64),
                    bounds=bounds,
                    method="highs",
                    options={"presolve": True, "time_limit": max(1.0, loc_stop - time.monotonic() - 1.0)},
                )
            except Exception:
                break

            if res is None or getattr(res, "x", None) is None:
                stagnant += 1
                if stagnant > 4:
                    break
                continue

            g = np.asarray(res.x, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g = np.clip(g, 0.0, 1.0)
            sg = float(g.sum())
            if not math.isfinite(sg) or sg <= 1e-15:
                g = a
                sg = 1.0
            g = g / sg

            best_sc = local
            best_cand = None
            for t in (0.15, 0.3, 0.5, 0.7, 0.85, 1.0):
                cand = (1.0 - t) * a + t * g
                cand = np.clip(cand, 0.0, None)
                sc = _score_fast(cand)
                if math.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_cand = cand

            if best_cand is not None and best_sc < local * (1.0 - 1e-10):
                a = _normalize(best_cand)
                local = best_sc
                stagnant = 0
                if local < float(best[0]) + 1e-12:
                    _update_best(a, best)
                if float(best[0]) < TARGET and local < TARGET and it > 6:
                    break
            else:
                stagnant += 1
                if stagnant % 8 == 0:
                    pa = a * np.exp(0.12 * rng.standard_normal(n))
                    pa = _normalize(pa)
                    psa = _score_fast(pa)
                    if math.isfinite(psa) and psa < local * (1.0 - 1e-10):
                        a = pa
                        local = psa
                        stagnant = 0
                        if local < float(best[0]) + 1e-12:
                            _update_best(a, best)
                if stagnant > 24:
                    break


def _v_from_f(f, n):
    f = np.asarray(f, dtype=np.float64).ravel()
    f = np.nan_to_num(f, nan=0.0, posinf=0.0, neginf=0.0)
    f = np.clip(f, 0.0, None)
    s = float(f.sum())
    if f.size >= 2 and s > 1e-15:
        old = np.linspace(0.0, 1.0, f.size)
        new = np.linspace(0.0, 1.0, n)
        g = np.interp(new, old, f)
        gs = float(g.sum())
        if gs > 0.0:
            g = g * (float(n) / gs)
        else:
            g = np.ones(n, dtype=np.float64)
    else:
        g = np.ones(n, dtype=np.float64)

    half = n // 2
    if half <= 0:
        return np.zeros(1, dtype=np.float64)
    rev = g[n - 1: n - half - 1:-1] if half > 0 else g[:half]
    v = 0.5 * (g[:half] - rev)
    if v.size != half:
        v = np.zeros(half, dtype=np.float64)
    return np.clip(v, -1.0, 1.0)


def _f_from_v(v, n):
    v = np.asarray(v, dtype=np.float64).ravel()
    half = n // 2
    v = np.clip(v, -1.0, 1.0)
    f = np.ones(n, dtype=np.float64)
    f[:half] = 1.0 + v[:half]
    if n % 2 == 0:
        f[half:] = 1.0 - v[:half][::-1]
    else:
        f[half:] = 1.0 - v[:half][::-1][:n - half]
        f[half] = 1.0
    return _normalize(f)


def _gradient_search(n, time_limit, rng, best, init_f, stop):
    n = int(n)
    if n < 8:
        return
    if n % 2:
        n -= 1
    half = n // 2
    loc_stop = min(time.monotonic() + time_limit, stop)

    def score_f(f):
        return _score_fast(f)

    inits = []
    if init_f is not None and len(init_f) > 1:
        inits.append(_v_from_f(init_f, n))

    x = np.linspace(0.0, 1.0, n)
    for amp in (0.5, 0.7, 0.9):
        for k in (1, 2, 3, 5):
            v = amp * np.sin(2.0 * math.pi * k * (x[:half] - 0.5))
            inits.append(np.clip(v, -1.0, 1.0))
    for _ in range(4):
        base = np.zeros(half, dtype=np.float64)
        for k in range(1, min(8, half) + 1):
            base += (rng.normal(0.0, 0.22 / max(1, k)) *
                     np.sin(2.0 * math.pi * k * (x[:half] - 0.5)))
        v = np.tanh(base)
        rms = float(np.sqrt(np.mean(np.square(v)))) if half else 1.0
        if rms > 1e-12:
            v = 0.75 * v / rms
        inits.append(np.clip(v, -1.0, 1.0))

    if not inits:
        inits.append(np.zeros(half, dtype=np.float64))

    for v0 in inits[:6]:
        if time.monotonic() >= loc_stop - 1:
            return
        v = np.asarray(v0, dtype=np.float64).copy()
        if v.size != half:
            continue
        v = np.clip(v, -1.0, 1.0)
        f = _f_from_v(v, n)
        local = score_f(f)
        if math.isfinite(local):
            _update_best(f, best)

        stale = 0
        for it in range(4000):
            if time.monotonic() >= loc_stop - 1:
                return

            b = np.convolve(f, f)
            S = float(f.sum())
            if not math.isfinite(S) or S <= 1e-15:
                break
            R = b * float(n) / (S * S)
            rmax = float(np.max(R))
            if not math.isfinite(rmax):
                break

            for beta in (4.0, 7.0, 10.0):
                w = np.exp(beta * np.clip(R - rmax, -10.0, 10.0))
                w = np.clip(w, 1e-30, 1e6)
                try:
                    ccorr = np.convolve(f, w[::-1])
                    idx = (2 * n - 2) - np.arange(n, dtype=np.int64)
                    h0 = ccorr[idx]
                except Exception:
                    break
                mirror = (n - 1) - np.arange(half, dtype=np.int64)
                gv = h0[:half] - h0[mirror]
                gnorm = float(np.linalg.norm(gv))
                if not math.isfinite(gnorm) or gnorm <= 1e-14:
                    continue
                scale = float(np.std(v)) + 0.08
                d = gv / gnorm * scale
                best_sc = local
                best_vc = None
                for alpha in (0.1, 0.3, 0.7, 1.5, 3.0, 7.0):
                    vc = np.clip(v - alpha * d, -1.0, 1.0)
                    fc = _f_from_v(vc, n)
                    sc = score_f(fc)
                    if math.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_vc = vc
                if best_vc is not None and best_sc < local * (1.0 - 1e-10):
                    v = best_vc
                    f = _f_from_v(v, n)
                    local = best_sc
                    stale = 0
                    if local < float(best[0]) + 1e-12:
                        _update_best(f, best)
                    break
            else:
                stale += 1
                if stale % 5 == 0:
                    vp = np.clip(v + 0.05 * rng.standard_normal(half), -1.0, 1.0)
                    fp = _f_from_v(vp, n)
                    scp = score_f(fp)
                    if math.isfinite(scp) and scp < local * (1.0 - 1e-10):
                        v = vp
                        f = fp
                        local = scp
                        stale = 0
                        if local < float(best[0]) + 1e-12:
                            _update_best(f, best)
                if stale > 60:
                    break
            if float(best[0]) < TARGET and local < TARGET and it > 10:
                break


def _fullf_grad(n, time_limit, rng, best, init_f, stop):
    n = int(n)
    if n < 4 or n > 700 or time_limit <= 0:
        return
    loc_stop = min(time.monotonic() + time_limit, stop)

    def score_f(f):
        return _score_fast(f)

    inits = []
    if init_f is not None and len(init_f) > 1:
        g = np.interp(np.linspace(0.0, 1.0, n),
                      np.linspace(0.0, 1.0, init_f.size),
                      np.asarray(init_f, dtype=np.float64))
        inits.append(_normalize(g))
    inits.extend(_make_starts(n, rng, init_f)[:3])
    if not inits:
        inits.append(np.ones(n, dtype=np.float64) / n)

    L = 2 * n - 1
    idx = np.arange(n)
    ii = idx[:, None]

    for f0 in inits[:4]:
        if time.monotonic() >= loc_stop - 1:
            return
        f = np.asarray(f0, dtype=np.float64).ravel().copy()
        f = _normalize(f)
        local = score_f(f)
        if math.isfinite(local):
            _update_best(f, best)

        stale = 0
        for it in range(6000):
            if time.monotonic() >= loc_stop - 1:
                return
            S = float(f.sum())
            if S <= 1e-15:
                break
            b = np.convolve(f, f)
            R = b * float(n) / (S * S)
            rmax = float(np.max(R))
            if not math.isfinite(rmax):
                break

            used = False
            for beta in (6.0, 10.0, 16.0):
                sh = beta * (R - rmax)
                sh = np.clip(sh, -30.0, 30.0)
                p = np.exp(sh)
                pz = float(p.sum())
                if pz <= 0 or not math.isfinite(pz):
                    continue
                p = p / pz
                m1 = ii + idx[None, :]
                m2 = ii - idx[None, :]
                w = p
                w1 = np.where((m1 >= 0) & (m1 < L), w[np.clip(m1, 0, L - 1)], 0.0)
                w2 = np.where((m2 >= 0) & (m2 < L), w[np.clip(m2, 0, L - 1)], 0.0)
                K = w1 + w2
                gv = (float(n) / (S * S)) * (K @ f)
                gnorm = float(np.linalg.norm(gv))
                if not math.isfinite(gnorm) or gnorm <= 1e-14:
                    continue
                d = gv / gnorm * float(np.std(f)) * 2.0
                best_sc = local
                best_fc = None
                for alpha in (0.05, 0.15, 0.4, 0.9, 1.8, 3.5):
                    fc = np.clip(f - alpha * d, 0.0, None)
                    fcn = _normalize(fc)
                    sc = score_f(fcn)
                    if math.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_fc = fcn
                if best_fc is not None and best_sc < local * (1.0 - 1e-10):
                    f = best_fc
                    local = best_sc
                    stale = 0
                    used = True
                    if local < float(best[0]) + 1e-12:
                        _update_best(f, best)
                    break
            if not used:
                stale += 1
                if stale % 5 == 0:
                    fp = _normalize(f * np.exp(0.1 * rng.standard_normal(n)))
                    scp = score_f(fp)
                    if math.isfinite(scp) and scp < local * (1.0 - 1e-10):
                        f = fp
                        local = scp
                        stale = 0
                        if local < float(best[0]) + 1e-12:
                            _update_best(f, best)
                if stale > 70:
                    break
            if float(best[0]) < TARGET and local < TARGET and it > 12:
                break


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    hard_start = time.monotonic()
    try:
        remaining = max(0.0, float(budget_s))
    except Exception:
        remaining = 1300.0
    deadline = hard_start + max(0.0, remaining - 12.0)

    best = [2.0, np.array([1.0, 1.0], dtype=np.float64) * 0.5]
    _update_best(best[1], best)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size:
                sc = evaluate_sequence([float(x) for x in parent])
                if math.isfinite(sc) and sc < float(best[0]):
                    best[0] = float(sc)
                    best[1] = parent.copy()
    except Exception:
        parent = None

    n0 = int(parent.size) if parent is not None and parent.size >= 8 else 96
    if n0 % 2:
        n0_even = n0 - 1

    # Concentrated refinement loop on the parent length (and its even twin).
    ref_lens = []
    for cand in (n0, n0_even, n0 - 8, n0 + 8, n0 + 16):
        if cand >= 12 and cand <= 400 and cand not in ref_lens:
            ref_lens.insert(0, cand)
    ref_lens = ref_lens[:3]

    rounds = 8
    for r in range(rounds):
        now = time.monotonic()
        if now >= deadline - 90.0:
            break
        budget = max(0.0, deadline - now - 70.0)
        per = max(8.0, budget / (3 * max(1, len(ref_lens)) * 2))
        before = best[0]
        improved = False
        for n in ref_lens:
            if time.monotonic() >= deadline - 30.0:
                break
            _search_lp(n, per, rng, best, parent, deadline)
            _gradient_search(n, per, rng, best, best[1], deadline)
            _fullf_grad(n, per, rng, best, best[1], deadline)
        if best[0] < before - 1e-9:
            improved = True
        else:
            # one more perturbed round, then stop
            for n in ref_lens[:1]:
                if time.monotonic() >= deadline - 30.0:
                    break
                _fullf_grad(n, min(60.0, max(8.0, deadline - time.monotonic() - 30.0)),
                            rng, best, best[1], deadline)
            if best[0] < before - 1e-9:
                improved = True
            if not improved:
                break

    # Broaden over a few adjacent lengths if time remains.
    adj = [64, 80, 112, 128, 160]
    for n in adj:
        now = time.monotonic()
        if now >= deadline - 45.0:
            break
        if abs(n - n0) <= 24:
            continue
        alloc = min(70.0, max(8.0, deadline - now - 45.0))
        _search_lp(n, alloc * 0.5, rng, best, parent, deadline)
        _gradient_search(n, alloc * 0.5, rng, best, best[1], deadline)
        _fullf_grad(n, alloc * 0.6, rng, best, best[1], deadline)

    # Final focused polish on the exact length of the current best.
    now = time.monotonic()
    if now < deadline - 25.0:
        bn = int(np.asarray(best[1]).size)
        if bn >= 12 and bn <= 400:
            _fullf_grad(bn, min(45.0, max(6.0, deadline - now - 15.0)),
                        rng, best, best[1], deadline)

    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.array([1.0], dtype=np.float64)
    s = float(np.sum(final))
    if s <= 1e-12:
        final = np.ones_like(final, dtype=np.float64)
        s = float(final.size)
    final = final / s
    final = np.clip(final, 0.0, 1000.0)
    _update_best(final, best)

    final = np.asarray(best[1], dtype=np.float64).ravel()
    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc) or fin_sc >= best[0]:
        final = np.asarray(best[1], dtype=np.float64).ravel()
    if final.size == 0:
        final = np.array([1.0], dtype=np.float64)
    return [float(x) for x in final]
