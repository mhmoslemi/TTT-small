        ws = float(np.sum(w))
        if not np.isfinite(ws) or ws <= 1e-300:
            w = np.full(L, 1.0 / L, dtype=np.float64)
        else:
            w = w / ws

        wfl = w[::-1]
        fw = rfft(wfl, size)
        convg = irfft(fa * fw, size)[: 3 * n - 2]
        gM = 2.0 * convg[n - 1: 2 * n - 1][::-1]
        gM = np.nan_to_num(gM, nan=0.0, posinf=0.0, neginf=0.0)

        if not np.isfinite(M):
            M = Cmax
        return float(M), gM, S, Cmax

    def optimize_log(a_init, seconds, P0, P1, lr0, lr1, use_lse=False, beta_abs=10.0):
        nonlocal best_surr_a, best_surr_score

        seconds = float(seconds)
        if seconds < 2.0 or time.monotonic() >= deadline:
            return project(a_init)

        a = project(a_init)
        x = np.log(a)
        m = np.zeros(n, dtype=np.float64)
        v = np.zeros(n, dtype=np.float64)
        u1 = 0.0
        u2 = 0.0
        b1 = 0.9
        b2 = 0.999
        eps = 1e-8
        it = 0
        no_move = 0

        start = time.monotonic()
        end = min(start + seconds, deadline)

        try:
            update_surr(a, surrogate_score(a))
        except Exception:
            pass
        update_exact(a)
        last_eval = time.monotonic()

        while True:
            now = time.monotonic()
            if now >= end or now >= deadline:
                break

            it += 1
            elapsed = now - start
            frac = 0.0 if seconds <= 0 else min(1.0, max(0.0, elapsed / seconds))
            P = P0 + (P1 - P0) * frac
            lr = lr0 + (lr1 - lr0) * frac

            a_old = a
            if use_lse:
                M, gM, S, Cmax = smooth_direction(a_old, 1.0, beta_abs=beta_abs)
            else:
                M, gM, S, Cmax = smooth_direction(a_old, P)

            if S <= 1e-300 or not np.isfinite(S):
                a = project(a_init)
                x = np.log(a)
                continue

            grad_a = (2.0 * n) * (gM * S - 2.0 * M) / (S * S * S)
            grad_a = np.nan_to_num(grad_a, nan=0.0, posinf=0.0, neginf=0.0)
            git = grad_a * a_old
            git = np.nan_to_num(git, nan=0.0, posinf=0.0, neginf=0.0)

            maxg = float(np.max(np.abs(git))) if git.size else 0.0
            if not np.isfinite(maxg) or maxg <= 1e-14:
                no_move += 1
                if no_move > 30:
                    no_move = 0
                    x_tmp = x + 0.04 * rng.standard_normal(n)
                    x_tmp = np.nan_to_num(x_tmp, nan=0.0, posinf=0.0, neginf=0.0)
                    x_tmp = np.clip(x_tmp, XMIN_RAW, XMAX_RAW)
                    a_new = np.exp(x_tmp)
                    a = project(a_new)
                    x = np.log(a)
                continue
            no_move = 0

            if maxg > 1e3:
                git *= 1e3 / maxg

            m = b1 * m + (1.0 - b1) * git
            v = b2 * v + (1.0 - b2) * (git * git)
            u1 = b1 * u1
            u2 = b2 * u2
            mb = m / (1.0 - u1)
            vb = v / (1.0 - u2)

            x_tmp = x - lr * mb / (np.sqrt(vb) + eps)
            if not np.all(np.isfinite(x_tmp)):
                x_tmp = x
            x_tmp = np.clip(x_tmp, XMIN_RAW, XMAX_RAW)
            a_new = np.exp(x_tmp)
            a = project(a_new)
            x = np.log(a)

            if S > 0.0 and Cmax >= 0.0:
                sc_surr = 2.0 * n * Cmax / (S * S)
                if np.isfinite(sc_surr) and sc_surr < best_surr_score:
                    best_surr_score = float(sc_surr)
                    best_surr_a = a_old.copy()

            now2 = time.monotonic()
            if now2 - last_eval > 8.0:
                update_exact(a)
                update_exact(best_surr_a)
                last_eval = now2

        update_exact(a)
        update_exact(best_surr_a)
        return a

    def lbf_stage(a_init, seconds):
        seconds = float(seconds)
        if _scipy_minimize is None or seconds < 25.0 or time.monotonic() >= deadline:
            return project(a_init)

        a = project(a_init)
        x0 = np.log(a)
        mean0 = float(np.mean(x0))
        if np.isfinite(mean0):
            x0 = x0 - mean0

        bounds = [(XMIN, XMAX)] * n

        def objgrad(x):
            x = np.asarray(x, dtype=np.float64)
            a = np.exp(np.clip(x, -40.0, 40.0))
            S = float(a.sum())
            if S <= 1e-300 or not np.isfinite(S):
                return 1e100, np.zeros_like(x)

            M, gM, S2, _ = smooth_direction(a, 1.0, beta_abs=10.0)
            if S2 <= 1e-300 or not np.isfinite(S2):
                return 1e100, np.zeros_like(x)

            f = (2.0 * n * M) / (S2 * S2)
            mn = float(np.mean(x)) if np.all(np.isfinite(x)) else 0.0
            if not np.isfinite(mn):
                mn = 0.0
            f += 1e-8 * mn * mn
            if not np.isfinite(f):
                f = 1e100

            grad_a = (2.0 * n) * (gM * S2 - 2.0 * M) / (S2 * S2 * S2)
            grad_a = np.nan_to_num(grad_a, nan=0.0, posinf=0.0, neginf=0.0)
            grad_x = grad_a * a + 2.0e-8 * mn
            grad_x = np.nan_to_num(grad_x, nan=0.0, posinf=0.0, neginf=0.0)
            mg = float(np.max(np.abs(grad_x))) if grad_x.size else 0.0
            if np.isfinite(mg) and mg > 1e4:
                grad_x *= 1e4 / mg
            return float(f), grad_x

        try:
            maxiter = int(min(120, max(25, seconds / 4.0)))
            maxfun = int(min(4000, max(700, seconds * 10.0)))
            res = _scipy_minimize(
                objgrad,
                x0,
                method="L-BFGS-B",
                bounds=bounds,
                jac=True,
                options={
                    "maxiter": maxiter,
                    "maxfun": maxfun,
                    "ftol": 1e-12,
                    "gtol": 1e-8,
                    "maxls": 12,
                },
            )
            x1 = np.asarray(res.x, dtype=np.float64)
            x1 = np.nan_to_num(x1, nan=0.0, posinf=0.0, neginf=0.0)
            a1 = project(np.exp(np.clip(x1, XMIN_RAW, XMAX_RAW)))
            update_exact(a1)
            try:
                update_surr(a1, surrogate_score(a1))
            except Exception:
                pass
            return a1
        except Exception:
            return a

    # ------------------------------------------------------------------
    # Main schedule.
    # ------------------------------------------------------------------
    T0 = max(0.0, deadline - time.monotonic())

    if T0 > 80.0:
        # Broad smooth averaging: escapes first-order stationary platooes.
        optimize_log(best_exact_a, 0.30 * T0, 1.0, 5.0, 0.30, 0.08)
        update_exact(best_surr_a)

    if T0 > 80.0:
        # Medium sharpness: reduce major peaks.
        optimize_log(best_exact_a, 0.17 * T0, 5.0, 25.0, 0.15, 0.03)
        update_exact(best_surr_a)

    if T0 > 80.0:
        # Sharpen toward exact max.
        optimize_log(best_exact_a, 0.20 * T0, 25.0, 100.0, 0.07, 0.006)
        update_exact(best_surr_a)

    if T0 > 120.0:
        # Local quasi-Newton polish on a bounded smooth objective.
        lbf_stage(best_exact_a, 0.12 * T0)
        update_exact(best_surr_a)

    if T0 > 100.0:
        # Very sharp exact-max polishing.
        optimize_log(best_exact_a, 0.09 * T0, 100.0, 500.0, 0.03, 0.001)
        update_exact(best_surr_a)

    # A few smoothed restarts from the best exact candidate.
    restart_done = 0
    while restart_done < 3:
        rem = max(0.0, deadline - time.monotonic())
        if rem < 50.0:
            break
        sec = min(70.0, max(15.0, 0.20 * rem))
        if sec < 12.0:
            break

        a0 = project(best_exact_a)
        x0 = np.log(a0)
        noise = rng.standard_normal(n)
        if n > 1:
            ff = np.fft.rfft(noise, n)
            freqs = np.arange(n // 2 + 1, dtype=np.float64)
            filt = np.exp(-((freqs / max(1, n // 40)) ** 2))
            noise = np.fft.irfft(ff * filt, n)
        noise -= float(np.mean(noise))
        std = 0.18 * (restart_done + 1)
        x_noisy = x0 + std * noise
        x_noisy = np.nan_to_num(x_noisy, nan=0.0, posinf=0.0, neginf=0.0)
        a_noisy = project(np.exp(np.clip(x_noisy, XMIN_RAW, XMAX_RAW)))

        Pstart = 40.0 + 20.0 * restart_done
        Bend = min(600.0, Pstart + 160.0)
        lrist = max(0.002, 0.05 * (0.75 ** restart_done))
        lend = max(0.0015, 0.6 * lrist)

        if time.monotonic() + sec < deadline:
            optimize_log(a_noisy, sec, Pstart, Bend, lrist, lend)
            update_exact(best_surr_a)
        restart_done += 1

    # Final validation with the authoritative evaluator.
    update_exact(best_exact_a)
    update_exact(best_surr_a)

    final_candidates = []
    for a in (best_exact_a, best_surr_a, base_clean):
        if a is not None:
            try:
                final_candidates.append(prepare_final(a))
            except Exception:
                pass
    final_candidates.append(prepare_final(np.ones(n, dtype=np.float64)))

    final_best_list = [1.0] * n
    final_best_score = float("inf")

    for a in final_candidates:
        try:
            lst = [float(x) for x in a]
            if not lst or any((not np.isfinite(x)) for x in lst):
                continue
            sc = evaluate_sequence(lst)
            if np.isfinite(sc) and sc < final_best_score:
                final_best_score = float(sc)
                final_best_list = lst
        except Exception:
            pass

    if not np.isfinite(final_best_score):
        final_best_list = [1.0] * n
        try:
            final_best_score = evaluate_sequence(final_best_list)
        except Exception:
            final_best_score = float("inf")

    if not final_best_list:
        final_best_list = [1.0] * n

    return final_best_list
