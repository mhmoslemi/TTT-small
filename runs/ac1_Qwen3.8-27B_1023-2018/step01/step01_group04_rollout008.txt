                break
        else:
            stagnant += 1
            if stagnant >= 15:
                scale = 0.08 + 0.04 * min(stagnant / 20.0, 2.0)
                fp = f * np.exp(scale * rng.standard_normal(n))
                fp = np.clip(fp, 0, None)
                sp_ = float(fp.sum())
                if sp_ > 1e-15:
                    fp = fp / sp_
                    scp = _score(fp)
                    if math.isfinite(scp) and scp < local:
                        f = fp
                        local = scp
                        stagnant = 0
                        if local < best[0]:
                            _update(f, best)
                if stagnant > 40:
                    break

    _update(f, best)
    return f


def _gradient_polish(f, time_limit, rng, best):
    """Subgradient / smooth-approximation gradient descent on max autocorrelation."""
    n = len(f)
    f = np.asarray(f, dtype=np.float64).ravel().copy()
    f = np.clip(f, 0.0, None)
    s = float(f.sum())
    if s <= 1e-15:
        f = np.ones(n) / n
        s = float(n)
    f = f / s

    stop = time.monotonic() + time_limit
    local = _score(f)
    if math.isfinite(local):
        _update(f, best)

    lr = 0.08
    stale = 0
    beta = 8.0  # sharpness of soft-max weighting

    for it in range(3000):
        if time.monotonic() >= stop - 1.5:
            break

        b = np.convolve(f, f)
        mb = float(b.max())
        if mb <= 0 or not math.isfinite(mb):
            break

        # Soft-max weights over autocorrelation values
        R = b / mb
        w = np.exp(beta * (R - 1.0))
        sw = float(w.sum())
        if sw <= 0:
            w = np.zeros(2 * n - 1)
            w[np.argmax(b)] = 1.0
        else:
            w = w / sw

        # Weighted gradient: grad_j = sum_k w_k * f[k-j] (for valid indices)
        # This is cross-correlation of f with w (reversed w)
        w_rev = w[::-1]
        cc = np.convolve(f, w_rev)
        # The relevant part: for index j in 0..n-1, grad_j = cc[2(n-1)-j]... 
        # Actually, (f*w)_k = sum_i f_i w_{k-i}. We want for each j:
        # grad_j = sum_k w_k f_{k-j} = (w * f_rev)_{...}
        # Let's just compute directly via the convolution result
        # (f * w_rev)_m = sum_i f_i w_rev[m-i] = sum_i f_i w[2n-2-m+i]
        # For the gradient of (a*a)_k w.r.t. a_j: it's a_{k-j}
        # Weighted: grad_j = sum_k w_k a_{k-j} = (a * w_reversed... )
        # Let me just do: for each j, sum over valid k of w_k * f[k-j]
        # This equals np.convolve(f, w)[:n] reversed... 
        # Actually: defined[h] = sum_i f[i] w[h-i], h=0..2n-2
        # grad_j = sum_k w[k] f[k-j] for k-j in [0,n-1]
        #        = sum_i w[i+j] f[i] for i in [0, n-1-j]
        # This is the cross-correlation = np.correlate(w, f, 'full') shifted
        # Simpler: use the convolution f * w_rev
        # (f * w_rev)_m = sum_i f[i] w_rev[m-i] = sum_i f[i] w[2n-2-m+i]
        # Let k = 2n-2-m+i, i = k-2n+2+m
        # This is getting complex. Let me just compute directly.
        grad = np.zeros(n)
        for j in range(n):
            # grad_j = sum_k w_k * f[k-j] where 0<=k-j<n i.e. j<=k<n+j
            kmin = max(0, j)
            kmax = min(2 * n - 1, n - 1 + j)
            if kmin <= kmax:
                grad[j] = np.dot(w[kmin:kmax + 1], f[0:kmax - kmin + 1])
        # Wait, that's not right either. Let me be more careful.
        # (a*a)_k = sum_{i} a_i * a_{k-i} where 0<=i<n, 0<=k-i<n
        # d/d a_j (a*a)_k = a_{k-j} (when 0<=k-j<n) + a_j (when 0<=k-j<n, the second factor)
        # Actually by symmetry: d/d a_j sum_i a_i a_{k-i} = a_{k-j} + a_{k-j} (if j in range)
        # No wait: (a*a)_k = sum_i a_i a_{k-i}. d/d a_j = a_{k-j} (from first factor when i=j)
        # + sum_i a_i * d/d a_j a_{k-i} (from second factor when k-i=j, i=k-j)
        # = a_{k-j} + a_{k-j} (if both valid) = 2*a_{k-j}
        # But we need 0 <= j < n AND 0 <= k-j < n for both terms.
        # So grad_j = sum_k w_k * (a_{k-j} if 0<=k-j<n else 0)
        #            + sum_k w_k * (a_{k-j} if 0<=k-j<n else 0)  [by symmetry of a*a]
        # Wait no, a*a is not exactly symmetric in this sense for the gradient.
        # (a*a)_k = sum_i a_i a_{k-i}
        # d/d a_j = a_{k-j} [term where i=j] + [term where k-i=j, i.e., i=k-j] a_i = a_{k-j}
        # So d/d a_j (a*a)_k = 2 * a_{k-j} if 0 <= k-j < n, plus a_j if k=2j... 
        # No: when i=j AND k-i=j simultaneously, i.e., i=j and i=k-j, so k=2j.
        # In that case the two terms are the same and we get a_j (not 2*a_j).
        # For k != 2j: d/d a_j = 2*a_{k-j} (if valid)
        # For k = 2j: d/d a_j = a_j (the term is counted once in the product)
        # 
        # Actually, let me just use: for the objective sum_k w_k (a*a)_k,
        # the gradient w.r.t. a_j is:
        #   sum_k w_k [a_{k-j} * 1_{0<=k-j<n} + a_{k-j} * 1_{0<=k-j<n} - a_j * 1_{k=2j}]
        # Hmm, this is getting complicated. Let me just use the simpler version:
        # grad_j ≈ sum_k w_k * a[max(0, min(n-1, k-j))] (approximate)
        # Or better: use the cross-correlation approach.
        # 
        # The cleanest: grad of (a*a) at index k w.r.t. a is the vector with components
        # [a_{k-j}]_{j=0}^{n-1} (zero-padded where k-j out of range) + same reversed
        # Actually for autocorrelation: (a*a)_k = <a, shift_k(a)>
        # d/d a_j = a_{k-j} + a_{k-j} (both factors contribute)
        # = 2*a_{k-j} for k != 2j boundary cases... 
        # For simplicity, let me just use: grad_j = sum_k w_k * a[(k-j) mod 2n]... no.
        
        # Let me just recompute properly:
        grad = np.zeros(n)
        for j in range(n):
            # Contribution to (a*a)_k from a_j: a_j appears as a_i (i=j) 
            # giving a_j * a_{k-j}, and as a_{k-i} (k-i=j, i=k-j) giving a_{k-j}*a_j
            # Total contribution to (a*a)_k: 2*a_j*a_{k-j} (if 0<=k-j<n, k-j!=j)
            # or a_j^2 (if k-j=j, i.e., k=2j)
            # Gradient w.r.t. a_j of (a*a)_k:
            #   if 0<=k-j<n and k-j != j: 2*a_{k-j}
            #   if k-j==j (k=2j): 2*a_j (derivative of a_j^2)
            #   if k-j==j: actually it's just 2*a_j since d(a_j^2)/da_j = 2*a_j
            #   Wait: (a*a)_k contains the term a_j * a_{k-j} (when i=j) 
            #   AND the term a_{k-j} * a_j (when i=k-j, if k-j != j)
            #   If k-j == j: only one term a_j * a_j = a_j^2, derivative = 2*a_j
            #   If k-j != j: two terms a_j*a_{k-j} and a_{k-j}*a_j, derivative = 2*a_{k-j}
            # So in all cases where 0<=k-j<n: derivative = 2*a_{k-j}
            # (including the k=2j case where a_{k-j} = a_j)
            
            # So grad_j = sum_k w_k * 2*a_{k-j} * 1_{0<=k-j<n}
            # = 2 * sum_{i=0}^{n-1} w_{i+j} * a_i  (where i = k-j, so k = i+j)
            # valid when 0 <= i+j <= 2n-2, i.e., 0<=i<n and j<=i+j<=2n-2
            # Since i in [0,n-1] and j in [0,n-1], i+j in [j, n-1+j] subset [0, 2n-2]
            # So: grad_j = 2 * sum_{i=0}^{n-1} w[i+j] * a[i]
            
            valid_i = slice(0, n)
            valid_w = slice(j, j + n)
            grad[j] = 2.0 * np.dot(w[valid_w], f[valid_i])
        
        # Normalize gradient
        gnorm = float(np.linalg.norm(grad))
        if gnorm <= 1e-14:
            break
        
        # Gradient direction (we want to decrease max autocorrelation)
        # f_new = f - lr * grad / gnorm (projected to simplex)
        best_sc = local
        best_fc = None
        for alpha in (0.02, 0.05, 0.1, 0.2, 0.5):
            fc = f - alpha * grad / gnorm
            fc = np.clip(fc, 0.0, None)
            sc = _score(fc)
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_fc = fc

        if best_fc is not None and best_sc < local * (1.0 - 1e-10):
            f = best_fc
            s = float(f.sum())
            if s > 1e-15:
                f = f / s
            local = _score(f)
            stale = 0
            if math.isfinite(local) and local < best[0]:
                _update(f, best)
        else:
            stale += 1
            if stale > 30:
                # Perturb
                fp = f * np.exp(0.05 * rng.standard_normal(n))
                fp = np.clip(fp, 0, None)
                sp_ = float(fp.sum())
                if sp_ > 1e-15:
                    fp = fp / sp_
                    scp = _score(fp)
                    if math.isfinite(scp) and scp < local:
                        f = fp
                        local = scp
                        stale = 0
                        if local < best[0]:
                            _update(f, best)
                if stale > 60:
                    break
            # Decrease learning rate
            lr = max(lr * 0.9, 0.005)

    _update(f, best)
    return f


def _interpolate_to_length(f, n_target):
    f = np.asarray(f, dtype=np.float64).ravel()
    f = np.clip(f, 0, None)
    if f.size == n_target:
        return f.copy()
    if f.size < 2:
        return np.ones(n_target) / n_target
    old_x = np.linspace(0.0, 1.0, f.size)
    new_x = np.linspace(0.0, 1.0, n_target)
    g = np.interp(new_x, old_x, f)
    g = np.clip(g, 0, None)
    s = float(g.sum())
    if s > 1e-15:
        g = g / s
    else:
        g = np.ones(n_target) / n_target
    return g


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    hard_start = time.monotonic()
    try:
        remaining = max(0.0, float(budget_s))
    except Exception:
        remaining = 1300.0
    deadline = hard_start + remaining - 12.0

    best = [float('inf'), np.array([1.0, 1.0], dtype=np.float64)]

    # Get parent
    parent = None
    try:
        h = globals().get('height_sequence_1', None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 0 and parent.sum() > 1e-15:
                _update(parent, best)
    except Exception:
        pass

    if parent is None or not math.isfinite(best[0]):
        parent = np.ones(96, dtype=np.float64)
        _update(parent, best)

    n0 = len(parent)
    f = parent.copy()
    s = float(f.sum())
    if s > 0:
        f = f / s

    # Phase 1: LP search at parent length (primary workhorse)
    if time.monotonic() < deadline - 60.0:
        t1 = min(deadline - time.monotonic() - 60.0, 450.0)
        f = _lp_search(f, t1, rng, best)

    # Phase 2: Gradient polish at parent length
    if time.monotonic() < deadline - 40.0:
        t2 = min(deadline - time.monotonic() - 40.0, 80.0)
        f = _gradient_polish(f, t2, rng, best)

    # Phase 3: More LP at parent length (second round with perturbation)
    if time.monotonic() < deadline - 50.0:
        fp = f * np.exp(0.1 * rng.standard_normal(n0))
        fp = np.clip(fp, 0, None)
        sp_ = float(fp.sum())
        if sp_ > 0:
            fp = fp / sp_
        t3 = min(deadline - time.monotonic() - 50.0, 150.0)
        _lp_search(fp, t3, rng, best)

    # Phase 4: Try longer lengths
    for n_try in [128, 192, 256]:
        if time.monotonic() >= deadline - 50.0:
            break
        init = _interpolate_to_length(best[1], n_try)
        if time.monotonic() < deadline - 40.0:
            t4 = min(deadline - time.monotonic() - 40.0, 100.0)
            f2 = _lp_search(init, t4, rng, best)
        if time.monotonic() < deadline - 35.0 and best[0] > TARGET:
            t5 = min(deadline - time.monotonic() - 35.0, 30.0)
            _gradient_polish(f2, t5, rng, best)

    # Phase 5: Final sprint LP at the best length
    if time.monotonic() < deadline - 15.0:
        best_arr = np.asarray(best[1], dtype=np.float64)
        init = best_arr.copy()
        s = float(init.sum())
        if s > 0:
            init = init / s
        t6 = min(deadline - time.monotonic() - 15.0, 60.0)
        if t6 > 10.0:
            _lp_search(init, t6, rng, best)

    # Final validation and return
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.ones(96, dtype=np.float64)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(96, dtype=np.float64)

    sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(sc) or sc >= best[0]:
        # Fallback to parent
        if parent is not None and parent.size > 0:
            final = parent
        else:
            final = np.ones(96, dtype=np.float64)

    return [float(x) for x in final]
