def _solve_lp(a, M):
    """Solve: max 1'b s.t. conv(a,b) <= M, b >= 0"""
    n = len(a)
    # Constraint matrix: (a*b)_k <= M for k = 0..2n-2
    # (a*b)_k = sum_{j} a_{k-j} * b_j (where 0 <= k-j < n and 0 <= j < n)
    
    # Build A matrix (2n-1 x n), where A[k,j] = a[k-j] if 0 <= k-j < n else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n-1, k)
        for j in range(j_lo, j_hi + 1):
            A[k, j] = a[k - j]
    
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_vec = np.full(2*n-1, M)
    
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_vec, bounds=(0, None), method='highs')

= `sum_j sum_m p_ext[j] a[m] [1/N sum_k e^{2πi(l-j+m)k/N}]`
= `sum_j sum_m p_ext[j] a[m] δ(l, j-m mod N)`
= `sum_m p_ext[l+m] a[m]` (with appropriate wrapping)

So `corr(p_ext, a)[l] = sum_m p_ext[l+m] · a[m]` (for the non-wrapped part).

This is exactly what we want! `(p ⊛ a)[i] = sum_m p[i+m] · a[m]`.

So the gradient is `2 · corr(p_ext, a)`.

Using numpy:
```python
def _gradient_conv(a, p, n):
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, L)
    R = A.conj()  # conj(rfft(a))
    P = np.fft.rfft(p, L)
    corr = np.fft.irfft(P * R, L)[:n]
    return 2.0 * corr
```

Wait, I need to be careful. `np.fft.rfft(p, L)` where `p` has length `2n-1` and `L >= 2n-1`. Then:

`irfft(rfft(p) * conj(rfft(a)))[l] = sum_m p[m] * a[(l-m) mod L]`... 

Hmm, the correlation via FFT is a bit tricky with zero-padding. Let me just use the direct O(n^2) computation for n=127. It's 16K operations, which is nothing.

```python
def _grad_smoothed_max(a, p, n):
    """Compute d(sum_k p_k * (a*a)_k)/da_i for i = 0..n-1"""
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for m in range(n):
            k = m + i
            if k < 2*n - 1:
                s += p[k] * a[m]
        grad[i] = 2.0 * s
    return grad
```

This is O(n^2) = O(127^2) ≈ 16K operations. Fine.

Now, the overall objective (normalized to sum 1) is:
`f(a) = 2n · max_k (a*a)_k` (since sum(a) = 1)

The gradient of the smoothed version:
`∇f ≈ 2n · grad_smoothed_max(a, p, n)`

But we also need to account for the normalization constraint `sum(a) = 1`. In the softmax parameterization, `a = softmax(z)`, so the gradient w.r.t. `z` is:
`∂f/∂z_i = J · ∂f/∂a` where `J` is the Jacobian of softmax.

The softmax Jacobian: `∂a_i/∂z_j = a_i(δ_ij - a_j)`.

So `∇_z f = diag(a) · (∇_a f - (∇_a f · a) · 1)`

In the previous code, this was handled by:
```python
dot = float(np.dot(v, a))
grad = a * (v - dot)
```

where `v` is the gradient in `a`-space. This is correct.

        b1, b2, eps = 0.9, 0.999, 1e-8
        am = b1 * am + (1 - b1) * gz
        av = b2 * av + (1 - b2) * gz * gz
        mh = am / (1.0 - b1 ** at)
        vh = av / (1.0 - b2 ** at)
        step = lr * mh / (np.sqrt(vh) + eps)
        step = np.clip(step, -0.6, 0.6)
        z = z - step
        z = np.where(np.isfinite(z), z, 0.0)

        if no_imp > max_ni:
            z = np.log(np.maximum(best_a, 1e-10))
            z -= float(np.mean(z))
            if rng.random() < 0.35:
                z = z[::-1]
            z += 0.25 * rng.standard_normal(n)
            am = np.zeros(n)
            av = np.zeros(n)
            at = 0
            lr = max(0.08, lr * 0.65)
            no_imp = 0
            max_ni = min(3000, int(max_ni * 1.3))

    for _ in range(200):
        if time.monotonic() > deadline - 10.0:
            break
        cand = best_a.copy()
        nb = int(rng.integers(1, 4))
        for _i in range(nb):
            w = int(rng.integers(2, max(3, n // 8)))
            st = int(rng.integers(0, n - w + 1))
            cand[st:st + w] *= np.exp(rng.normal(0, 0.12))
        cand = np.clip(cand, 0.0, 1000.0)
        cs = float(cand.sum())
        if cs > 0.01:
            cand /= cs
            sc_c = _sc(cand)
            if sc_c < best_sc - 1e-10:
                best_sc = sc_c
                best_a = cand.copy()

                final_sc = sv
