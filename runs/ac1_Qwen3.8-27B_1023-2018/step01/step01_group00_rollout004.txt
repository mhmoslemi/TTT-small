from scipy.optimize import minimize

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")


def _get_parent():
    try:
        h = globals().get("height_sequence_1", None)
        p = np.asarray(h, dtype=np.float64).ravel() if h is not None else np.array([])
    except Exception:
        p = np.array([])
    try:
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0.0, 1000.0)
    except Exception:
        p = np.array([])
    if p.size == 0 or float(p.sum()) <= 0.0:
        n = 192
        x = np.linspace(0.0, 1.0, n)
        p = np.exp(-0.7 * x)
        p = np.maximum(p, 1e-12)
    return p.astype(np.float64, copy=False)

def _softmax_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    n = z.size
    if n == 0:
        return np.array([1.0], dtype=np.float64)
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(z.mean())
    m = float(np.max(z))
    e = np.exp(z - m)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    q = e / s
    q = np.where(np.isfinite(q), q, 0.0)
    s2 = float(q.sum())
    if s2 <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return q / s2


def _score_q(q, fsize=None):
    q = np.asarray(q, dtype=np.float64).ravel()
    n = q.size
    if n <= 0:
        return math.inf
    q = np.clip(q, 0.0, 1000.0)
    s = float(q.sum())
    if not np.isfinite(s) or s < 0.01:
        return math.inf
    if fsize is None:
        fsize = next_fast_len(2 * n - 1)
    F = rfft(q, fsize)
    c = irfft(F * F, fsize)[: 2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    H = float(c.max()) if c.size else 0.0
    if not np.isfinite(H) or H < 0.0:
        return math.inf
    return float(2.0 * n * H / (s * s))


def _score_z(z, fsize=None):
    q = _softmax_z(z)
    return _score_q(q, fsize)


class _SoftOptimizer:
    def __init__(self, n):
        self.n = int(n)
        self.k = 2 * self.n - 1
        self.fsize_conv = next_fast_len(self.k)
        self.fsize_grad = next_fast_len(3 * self.n - 2)
        self.cache_z = None
        self.cache_tau = -1.0
        self.cache_grad = None

    def _compute(self, z, tau):
        n = self.n
        z = np.asarray(z, dtype=np.float64).ravel()
        if z.size != n:
            return math.inf, np.zeros(n, dtype=np.float64)
        if not np.all(np.isfinite(z)):
            z = np.where(np.isfinite(z), z, 0.0)
        zz = z - float(z.mean())

        m = float(np.max(zz))
        e = np.exp(zz - m)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 0.0:
            q = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            q = e / s
            q = np.where(np.isfinite(q), q, 0.0)
            qs = float(q.sum())
            if qs <= 0.0:
                q = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                q = q / qs

        F = rfft(q, self.fsize_conv)
        c = irfft(F * F, self.fsize_conv)[: self.k]
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        H = float(c.max()) if c.size else 0.0
        if not np.isfinite(H) or H < 0.0:
            return math.inf, np.zeros(n, dtype=np.float64)

        tau = float(tau)
        if not np.isfinite(tau) or tau <= 0.0:
            tau = 1e-12
        x = (c - H) / tau
        ex = np.exp(x)
        ss = float(ex.sum())
        if not np.isfinite(ss) or ss <= 0.0:
            p = np.zeros(self.k, dtype=np.float64)
            idx = np.where(c == H)[0]
            if idx.size == 0:
                idx = np.array([int(np.argmax(c))], dtype=int)
            p[idx] = 1.0 / idx.size
            U = H
        else:
            p = ex / ss
            U = H + tau * math.log(ss)
        val = float(2.0 * n * U)

        if val == 2.0 * n * H:
            maxp = float(p.max())
            thresh = max(1e-12, maxp * 1e-7)
            idx = np.where(p > thresh)[0]
            if n <= 512:
                max_loop = 160
            elif n <= 2048:
                max_loop = 80
            else:
                max_loop = 40

            if idx.size > 0 and idx.size <= max_loop:
                p_use = p[idx].astype(np.float64, copy=False)
                p_sum = float(p_use.sum())
                if p_sum <= 0.0:
                    p_use = None
                else:
                    p_use /= p_sum
            else:
                p_use = None

            if p_use is not None:
                gq = np.zeros(n, dtype=np.float64)
                for kk, ww in zip(idx, p_use):
                    ww = float(ww)
                    if ww <= 0.0:
                        continue
                    kk = int(kk)
                    lo = max(0, kk - n + 1)
                    hi = min(n - 1, kk)
                    if hi < lo:
                        continue
                    gq[lo:hi + 1] += (2.0 * ww) * q[kk - hi:kk - lo + 1][::-1]
            else:
                qr = q[::-1]
                G = rfft(p, self.fsize_grad)
                G *= rfft(qr, self.fsize_grad)
                h = irfft(G, self.fsize_grad)
                gq = 2.0 * h[n - 1:n - 1 + n]

            gq = np.where(np.isfinite(gq), gq, 0.0)
            gq = np.maximum(gq, 0.0)
            A = float(np.dot(q, gq))
            grad = q * (gq - A)
            grad -= float(grad.mean())
            if not np.all(np.isfinite(grad)):
                grad = np.where(np.isfinite(grad), grad, 0.0)

            self.cache_z = zz.copy()
            self.cache_tau = tau
            self.cache_grad = grad.copy()
            return val, grad

    def fun(self, z, tau):
        val, _ = self._compute(z, tau)
        return val

    def grad(self, z, tau):
        if self.cache_grad is not None and self.cache_tau == tau:
            zz = np.asarray(z, dtype=np.float64).ravel()
            if zz.size == self.n:
                zz = zz - float(zz.mean())
                if np.array_equal(zz, self.cache_z):
                    return self.cache_grad
        _, g = self._compute(z, tau)
        return g


def _table_iters(n):
    if n <= 128:
        return (120, 170, 220, 280)
    if n <= 256:
        return (95, 135, 180, 230)
    if n <= 512:
        return (75, 105, 140, 180)
    if n <= 1024:
        return (55, 80, 110, 145)
    if n <= 2048:
        return (35, 55, 75, 105)
    return (20, 30, 45, 60)


def _iter_time_est(n):
    if n <= 128:
        return 0.003
    if n <= 256:
        return 0.006
    if n <= 512:
        return 0.012
    if n <= 1024:
        return 0.025
    if n <= 2048:
        return 0.06
    return 0.12


def _optimize_z(n, z0, stop_time, seed):
    if time.monotonic() > stop_time - 1.5:
        z0 = np.asarray(z0, dtype=np.float64).ravel()
        return z0, _score_z(z0)

    z = np.asarray(z0, dtype=np.float64).ravel().copy()
    if z.size != n or not np.all(np.isfinite(z)):
        z = np.zeros(n, dtype=np.float64)

    z = z - float(z.mean())
    z = np.clip(z, -12.0, 12.0)

    fsize = next_fast_len(2 * n - 1)
    sc_work = _score_z(z, fsize)
    if not np.isfinite(sc_work):
        z = np.zeros(n, dtype=np.float64)
        sc_work = _score_z(z, fsize)

    sc_best = sc_work
    z_best = z.copy()

    opt = _SoftOptimizer(n)
    iters = _table_iters(n)
    rels = (0.06, 0.012, 0.0018, 3e-5)
    time_fracs = (0.18, 0.22, 0.26, 0.34)

    for ideal_tau_frac, mx_tab, rel in zip(time_fracs, iters, rels):
        now = time.monotonic()
        if now > stop_time - 1.8:
            break

        remaining = max(0.5, stop_time - now - 1.0)
        call_budget = max(1.0, remaining * ideal_tau_frac)
        call_stop = min(stop_time - 0.8, now + call_budget)
        est = _iter_time_est(n)
        mx = int(max(8, min(mx_tab, call_budget / max(1e-6, est))))

        H = max(1e-12 / float(n), min(2.0, float(sc_work)) / (2.0 * n))
        tau = max(1e-15, rel * H)

        def fun(x, tau=tau):
            return opt.fun(x, tau)

        def jac(x, tau=tau):
            return opt.grad(x, tau)

        opts = {
            "maxiter": mx,
            "ftol": 1e-12,
            "gtol": 1e-12,
            "maxcor": max(8, min(48, n // 32)),
            "maxls": 30,
            "iprint": -1,
        }

        init_val = opt.fun(z, tau)
        if not np.isfinite(init_val):
            break

        try:
            res = minimize(fun, z.copy(), jac=jac, method="L-BFGS-B", options=opts)
        except Exception:
            break

        if hasattr(res, "x") and res.x is not None:
            zc = np.asarray(res.x, dtype=np.float64).ravel()
            if zc.size == n and np.all(np.isfinite(zc)):
                zc = zc - float(zc.mean())
                zc = np.clip(zc, -14.0, 14.0)
                scz = _score_z(zc, fsize)
                if np.isfinite(scz):
                    if scz < sc_best - 1e-12:
                        sc_best = scz
                        z_best = zc.copy()
                    if scz < max(2.0, sc_work) + 0.35 and np.isfinite(getattr(res, "fun", math.nan)):
                        z_work = zc
                        sc_work = scz

    return z_best, sc_best


def _prepare_z(z, rng, scale):
    z = np.asarray(z, dtype=np.float64).ravel()
    if not np.all(np.isfinite(z)):
        return None
    z = z - float(z.mean())
    if scale:
        s = float(z.std())
        if s > 1e-8:
            target = float(rng.uniform(0.45, 1.75))
            z = z * (target / s)
        else:
            z = z + rng.normal(0.0, 1.0, z.size)
            s = float(z.std())
            if s > 1e-8:
                target = float(rng.uniform(0.45, 1.75))
                z = z * (target / s)
    return np.clip(z, -9.0, 9.0)


def _initial_z_list(n, rng, parent_log):
    raw_parent = []
    raw_random = []

    x = np.linspace(0.0, 1.0, n)

    if parent_log is not None and len(parent_log) > 0:
        p = np.asarray(parent_log, dtype=np.float64).ravel()
        if p.size == n:
            raw_parent.append(p.copy())
        else:
            idx = np.linspace(0.0, p.size - 1, n)
            raw_parent.append(np.interp(idx, np.arange(p.size, dtype=np.float64), p))
            if p.size > 1:
                raw_parent.append(np.interp(idx, np.arange(p.size, dtype=np.float64), p[::-1]))

    for a in (-1.3, -0.8, -0.4, 0.4, 0.8, 1.3):
        raw_random.append(a * (2.0 * x - 1.0) + rng.normal(0.0, 0.15, n))

    for deg in (1, 2, 3, 4):
        c = rng.normal(0.0, 1.0, deg + 1)
        raw_random.append(np.polyval(c, x))

    for _ in range(4):
        z = np.zeros(n, dtype=np.float64)
        K = int(rng.integers(2, 6))
        fmax = max(2, n // 8)
        for _ in range(K):
            f = int(rng.integers(1, fmax + 1))
            amp = float(rng.uniform(0.45, 1.35))
            ph = float(rng.uniform(0.0, 2.0 * math.pi))
            z += amp * np.sin(2.0 * math.pi * f * x + ph)
            if rng.random() < 0.5:
                z += float(rng.uniform(0.2, 0.8)) * np.cos(2.0 * math.pi * f * x + ph)
        raw_random.append(z)

    for _ in range(4):
        z = np.zeros(n, dtype=np.float64)
        K = int(max(4, min(n // 3, rng.integers(5, max(6, n // 5)))))
        fmax = max(2, n // 2)
        freqs = rng.integers(1, fmax + 1, size=K)
        amps = rng.uniform(0.20, 0.85, size=K)
        phases = rng.uniform(0.0, 2.0 * math.pi, size=K)
        for f, amp, ph in zip(freqs, amps, phases):
            z += amp * np.sin(2.0 * math.pi * int(f) * x + ph)
        raw_random.append(z)

    raw_random.append(rng.normal(0.0, 1.0, n))

    nb = int(max(2, min(n // 5, rng.integers(2, max(3, n // 8 + 1)))))
    edges = np.sort(rng.integers(0, n, nb + 1))
    vals = rng.normal(0.0, 1.0, nb)
    z = np.empty(n, dtype=np.float64)
    for j in range(nb):
        a = int(edges[j])
        b = int(edges[j + 1])
        if b > a:
            z[a:b] = float(vals[j])
        else:
            z[a] = float(vals[j])
    z[:n * nb // 10] += rng.normal(0.0, 0.1, max(1, n * nb // 10))
    raw_random.append(z)

    for _ in range(3):
        pos = float(rng.uniform(0.05, 0.95))
        width = float(rng.uniform(0.05, 0.35))
        A = float(rng.uniform(0.8, 2.5))
        z = A * np.exp(-(((x - pos) / width) ** 2))
        raw_random.append(z)

    out = []
    for z in raw_parent:
        zz = _prepare_z(z, rng, False)
        if zz is not None:
            out.append(zz)
    for z in raw_random:
        zz = _prepare_z(z, rng, True)
        if zz is not None:
            out.append(zz)

    if not out:
        out.append(np.zeros(n, dtype=np.float64))
    return out


def _polish(seq, score, stop_time, seed):
    seq = np.asarray(seq, dtype=np.float64).ravel()
    n = seq.size
    if n < 2 or not np.all(np.isfinite(seq)):
        return seq, score

    seq = np.clip(seq, 0.0, 1000.0)
    s = float(seq.sum())
    if s <= 0.0:
        seq = np.ones(n, dtype=np.float64) / n
        score = 2.0
    else:
        seq = seq / s

    z = np.log(np.maximum(seq, 1e-300))
    z = z - float(z.mean())
    best_z = z.copy()
    best_seq = seq.copy()
    best_score = float(score)
    fsize = next_fast_len(2 * n - 1)
    exact_calls = 0
    max_exact_calls = 40 if n <= 2048 else 0

    rng = np.random.default_rng(seed)
    sigma = 0.12
    fails = 0

    while time.monotonic() < stop_time - 0.7:
        r = float(rng.random())
        if r < 0.45:
            zc = best_z + rng.normal(0.0, sigma, n)
        elif r < 0.70:
            zc = best_z.copy()
            start = int(rng.integers(0, n))
            width = max(1, int(rng.integers(1, max(2, n // 30))))
            end = min(n, start + width)
            zc[start:end] += rng.normal(0.0, sigma * 1.8, end - start)
        elif r < 0.85:
            zc = 0.82 * best_z + 0.18 * rng.normal(0.0, sigma * 0.8, n)
        else:
            zc = best_z + rng.normal(0.0, sigma * 0.35, n)

        zc = zc - float(zc.mean())
        zc = np.clip(zc, -14.0, 14.0)
        q = _softmax_z(zc)
        sc = _score_q(q, fsize)
        if not np.isfinite(sc):
            fails += 1
            continue

        if sc < best_score - 1e-10:
            use = sc
            if max_exact_calls > 0 and n <= 2048 and exact_calls < max_exact_calls:
                exact = _eval_exact(q)
                exact_calls += 1
                if np.isfinite(exact) and exact < best_score - 1e-10:
                    use = exact
            if use < best_score - 1e-10:
                best_score = use
                best_seq = q.copy()
                best_z = zc.copy()
                sigma = max(0.035, sigma * 0.82)
 fails = 0
        else:
            fails += 1
            if fails % 18 == 0:
                sigma = min(0.95, sigma * 1.35)
            if fails > 1400:
                break
            if time.monotonic() > stop_time - 1.6:
                break

    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 30.0
    deadline = t0 + max(2.0, budget - 18.0)

    parent = _get_parent()
    ps = float(parent.sum())
    if ps > 0.0:
        parent = parent / ps
    else:
        parent = np.ones(192, dtype=np.float64) / 192.0

    try:
        base_score = _eval_exact(parent)
    except Exception:
        base_score = math.inf
    if not np.isfinite(base_score):
        base_score = 2.0

    best_seq = np.array(parent, dtype=np.float64).copy()
    best_score = float(base_score)
    best_n = int(best_seq.size)
    best_z = np.log(np.maximum(best_seq, 1e-300))
    best_z = best_z - float(best_z.mean())

    parent_log = np.log(np.maximum(parent, 1e-300))
    parent_log = parent_log - float(parent_log.mean())

    fallback_seq = np.array(parent, dtype=np.float64).copy()
    fallback_score = float(base_score)

    def consider_candidate(seq, surrogate_score, z=None):
        nonlocal best_seq, best_score, best_n, best_z
        try:
            a = np.asarray(seq, dtype=np.float64).ravel()
        except Exception:
            return
        if a.size < 2 or not np.all(np.isfinite(a)):
            return
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s <= 0.0:
            return
        a = a / s
        n = a.size
        if n <= 1200:
            sc = _eval_exact(a)
            if not np.isfinite(sc):
                sc = surrogate_score
        else:
            sc = surrogate_score
        if np.isfinite(sc) and sc < best_score - 1e-12:
            if n > 1200 and sc > best_score - 1e-7 and best_score < 2.0:
                return
            best_seq = a.copy()
            best_score = float(sc)
            best_n = n
            if z is not None:
                zz = np.asarray(z, dtype=np.float64).ravel()
                if zz.size == n and np.all(np.isfinite(zz)):
                    best_z = zz - float(zz.mean())

    parent_best = parent.size
    base_len = int(parent_best) if 2 <= parent_best <= 4096 else 192

    standard = [
        96, 128, 160, 192, 224, 256, 320, 384, 512, 640,
        768, 896, 1024, 1280, 1536, 2048, 3072, 4096
    ]
    lengths = []
    for L in standard:
        lengths.append(L)
    lengths.append(base_len)
    for mult in (2, 3, 4):
        L = int(np.clip(base_len * mult, 2, 4096))
        lengths.append(L)

    lengths = sorted({int(x) for x in lengths if 48 <= int(x) <= 4096})
    if not lengths:
        lengths = [base_len]

    order = [base_len]
    for L in sorted(x for x in lengths if x != base_len):
        order.append(L)

    weights = []
    for L in order:
        w = 0.8 + 0.45 * math.log2(max(64.0, float(L)) / 64.0)
        if L == base_len:
            w += 1.4
        weights.append(w)
    wsum = float(sum(weights))
    avail = max(1.0, min(deadline, time.monotonic() + (deadline - t0)) - t0 - 30.0)
    allocs = {L: max(3.0, avail * float(w) / wsum) for L, w in zip(order, weights)}

    # If allocations overflow due to minimums, trim non-base lengths lightly.
    if sum(allocs.values()) > avail:
        scale = max(0.35, avail / sum(allocs.values()))
        for L in list(allocs.keys()):
            if L != base_len:
                allocs[L] = max(2.0, allocs[L] * scale)

    start_est_base = {
        "small": 0.7,
        "medium": 1.7,
        "large": 4.5
    }

    for L in order:
        now = time.monotonic()
        if now > deadline - 20.0:
            break
        if best_score < 1.48 and L > 1024:
            # Still allow a bit more, but stop spending on huge lengths early.
            pass

        alloc = float(allocs.get(L, 4.0))
        length_stop = min(deadline - 22.0, now + alloc)
        if length_stop - time.monotonic() < 2.5:
            continue

        rng = np.random.default_rng(int(seed) + 7919 * int(L) + 12345)
        base_inits = _initial_z_list(L, rng, parent_log)
        if not base_inits:
            base_inits = [np.zeros(L, dtype=np.float64)]

        if L <= 256:
            max_starts = int(max(3, min(90, alloc / 0.7)))
        elif L <= 1024:
            max_starts = int(max(3, min(45, alloc / 1.7)))
        else:
            max_starts = int(max(2, min(18, alloc / 4.5)))

        good_zs = []
        sigma_outer = 0.55

        for st in range(max_starts):
            now = time.monotonic()
            if now > length_stop - 4.0:
                break

            if good_zs and rng.random() < 0.72:
                gz = good_zs[int(rng.integers(0, len(good_zs)))]
                z0 = gz + rng.normal(0.0, sigma_outer, L)
            else:
                z0 = base_inits[st % len(base_inits)] + rng.normal(0.0, 0.18, L)

            z0 = np.asarray(z0, dtype=np.float64).ravel()
            if z0.size != L or not np.all(np.isfinite(z0)):
                continue
            z0 = z0 - float(z0.mean())
            z0 = np.clip(z0, -10.0, 10.0)

            quick_sc = _score_z(z0, next_fast_len(2 * L - 1))
            if not np.isfinite(quick_sc):
                quick_sc = 2.0

            run_it = (st < 4) or (quick_sc < best_score + 0.28) or (len(good_zs) == 0 and st < 10)
            if not run_it:
                continue

            if L <= 256:
                start_alloc = 2.2
            elif L <= 1024:
                start_alloc = 4.8
            else:
                start_alloc = 9.5

            start_stop = min(length_stop - 3.0, now + start_alloc)
            if start_stop - time.monotonic() < 1.2:
                break

            try:
                z_out, sc_out = _optimize_z(L, z0, start_stop, int(seed) + 1000 + 31 * L + st)
            except Exception:
                z_out, sc_out = z0, math.inf

            if np.isfinite(sc_out):
                q_out = _softmax_z(z_out)
                consider_candidate(q_out, sc_out, z_out)
                if sc_out < max(2.0, best_score) + 0.08:
                    good_zs.append(np.array(z_out, dtype=np.float64).copy())
                    if len(good_zs) > 12:
                        good_zs = good_zs[-12:]
                if best_score < quick_sc:
                    sigma_outer = max(0.12, sigma_outer * 0.80)
                else:
                    sigma_outer = min(0.95, sigma_outer * 1.06)

    # Final polish.
    final_stop = min(deadline - 8.0, time.monotonic() + max(0.0, deadline - time.monotonic() - 12.0))
    if time.monotonic() < final_stop - 2.0:
        try:
            best_seq, best_score = _polish(
                best_seq,
                best_score,
                final_stop,
                int(seed) + 987654
            )
            if best_seq.size != int(best_n):
                # restore z if length changed
                try:
                    best_z = np.log(np.maximum(best_seq, 1e-300))
                    best_z = best_z - float(best_z.mean())
                except Exception:
                    best_z = None
        except Exception:
            pass

    # Exact final validation and safe fallback.
    final_score = _eval_exact(best_seq)
    if not np.isfinite(final_score):
        final_score = fallback_score
        best_seq = fallback_seq.copy()

    fb_score = _eval_exact(fallback_seq)
    if np.isfinite(fb_score) and fb_score < final_score - 1e-12:
        best_seq = fallback_seq.copy()
        final_score = fb_score

    if not np.isfinite(best_score):
        best_score = final_score

    if best_seq.size < 1 or not np.all(np.isfinite(best_seq)):
        best_seq = np.ones(256, dtype=np.float64) / 256.0
        final_score = _eval_exact(best_seq)
        if not np.isfinite(final_score):
            return [1.0]

    return [float(x) for x in np.clip(best_seq, 0.0, 1000.0)]
