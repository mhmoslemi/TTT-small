            r = np.abs(rng.standard_normal(n)) + 0.1
            k_s = max(5, n // 80)
            r = np.convolve(r, np.ones(k_s) / k_s, mode='same')
            blend = rng.uniform(0.4, 0.7)
            a = (1 - blend) * a + blend * r / r.sum()
            a = np.maximum(a, 0)
            a /= a.sum()
            grad_ma[:] = 0
            patience = 0
            eta = 0.35

        # Anneal
        if phase_iter % 5000 == 0:
            eta = max(0.03, eta * 0.93)
            momentum_coeff = min(0.92, momentum_coeff * 1.01)

        # Hard time check
        if time.monotonic() > deadline - 10.0:
            break

    # --- Final polish: a few more careful steps ---
    if time.monotonic() < deadline - 5.0:
        eta_p = 0.02
        for _ in range(500):
            if time.monotonic() > deadline - 5.0:
                break
            b = fast_conv(a)
            max_b = b.max()
            k_top = max(3, conv_len // 50)
            thresh = np.partition(b, conv_len - k_top)[conv_len - k_top]
            spread = max_b - thresh
            if spread > 1e-14:
                w = 1.0 + 4.0 * np.clip((b - thresh) / spread, 0.0, 1.0)
            else:
                w = np.where(b >= thresh, 5.0, 1.0)
            load = fast_load(w, a)
            lm = load.max()
            if lm > 1e-15:
                load /= lm
            a = a * np.exp(-eta_p * load)
            a = np.maximum(a, 1e-20)
            a /= a.sum()
            s = score_fast(a)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_a = a.copy()

    # --- Final validation ---
    result = [float(x) for x in best_a]
    final_score = evaluate_sequence(result)
    if not np.isfinite(final_score) or final_score > 2.5:
        result = [1.0] * n
        final_score = evaluate_sequence(result)

    return result
