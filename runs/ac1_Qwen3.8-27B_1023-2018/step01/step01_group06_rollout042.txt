from scipy import optimize
from scipy.sparse import csr_matrix
import time


def propose_candidate(seed=42, budget_s=1300, **kwargs):

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)

    n = len(height_sequence_1)
    conv_len = 2 * n - 1

    while fft_n < 3 * n - 2:
        fft_n <<= 1

    def get_autocorr(a):
        """Return (autocorrelation, FFT of a padded to fft_n)."""
        F = np.fft.rfft(a, fft_n)
        ac = np.fft.irfft(F * np.conj(F), fft_n)
        return ac[:conv_len], F

    def get_gradient(p, F):
        """Cross-correlation of p (len 2n-1) and a (len n) at lags 0..n-1 via FFT."""
        P = np.fft.rfft(p, fft_n)
        R = np.fft.irfft(P * np.conj(F), fft_n)
        return R[:n]

        return 2.0 * n * ac.max()

    a = np.clip(a, 0.0, 1000.0)

    if a_sum < 1e-6:
        a = np.ones(n, dtype=np.float64)
        a_sum = float(n)
    a = a / a_sum

    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))

    alts.append(cw / cw.sum())
