from scipy.optimize import linprog

def lp_refine(f, M, n):
    # f: array of length n (the current sequence)
    # M: the current max of f*f
    # Find g >= 0 maximizing sum(g) subject to (f*g)[k] <= M for all k
    
    # (f*g)[k] = sum_i f[i] * g[k-i] for valid i, k-i
    # This is a linear function of g for each k
    # f*g has 2n-1 entries
    # (f*g)[k] for k=0,...,2n-2
    # (f*g)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[i] * g[k-i]
    
    # For each k, the constraint is: sum_i A[k,i] * g[i] <= M
    # where A[k,i] = f[k-i] if 0 <= k-i <= n-1, else 0
    
    # Build A matrix (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = f[j]
    
    c = -np.ones(n)  # minimize -sum(g)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    return res.x if res.success else None

    n = len(f)

    z = z - z.mean()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best = f.copy()
    best_score = _score(f)
    lr = 0.5
    stall = 0

        if time.monotonic() > deadline:
            break
        
        f = _softmax(z)
        c = np.convolve(f, f)
        s = float(f.sum())
        sc = float(2.0 * n * np.max(c) / (s*s))
        
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = f.copy()
            stall = 0
            lr = min(lr * 1.005, 1.2)
        else:
            stall += 1
        
        # Compute smooth gradient
        frac = min(1.0, it / max(1, max_iters))
        beta = 5000 + 50000 * frac
        cmax = float(np.max(c))
        zc = beta * (c - cmax)
        p = np.exp(np.clip(zc, -700, 0))
        ps = float(p.sum())
        if ps < 1e-300:
            p = np.zeros(2*n-1)
            p[int(np.argmax(c))] = 1.0
        else:
            p = p / ps
        
        # dL/df[j] = 2 * sum_k p[k] * f[k-j] (valid range)
        # This is a cross-correlation
        # (p_padded * f_padded) at specific positions
        # For efficiency with small n, just compute directly
        grad_f = np.zeros(n)
        for j in range(n):
            val = 0.0
            for k in range(max(0, j), min(2*n-1, j+n)):
                val += p[k] * f[k-j]
            grad_f[j] = 2.0 * val
        
        # Convert to softmax gradient: dL/dz_i = grad_f[i] - mean(grad_f * f) * ... 
        # Actually: dL/dz_i = (1/S) * (grad_f[i] - sum_j f_j * grad_f[j])
        # where S = sum(f) = 1 (normalized)
        dot = float(np.dot(grad_f, f))
        grad_z = grad_f - f * dot
        
        if not np.all(np.isfinite(grad_z)):
            grad_z = np.zeros(n)
        
        # Adam update
        t += 1
        m = b1 * m + (1-b1) * grad_z
        v = b2 * v + (1-b2) * grad_z**2
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -1.0, 1.0)
        z = z - step
        z = np.where(np.isfinite(z), z, 0.0)
        
        if stall > 500:
            z = np.log(np.maximum(best, 1e-30)) - np.log(np.maximum(best, 1e-30)).mean()
            z += 0.2 * rng.standard_normal(n)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = max(0.1, lr * 0.5)
            stall = 0
    
    return best, best_score

import time

            parent = np.maximum(p.ravel(), 0.0).copy()
            s = float(parent.sum())
            if s > 1e-12:
                parent /= s
                n_p = parent.size
            else:
                parent = None
    except Exception:
        pass
    if parent is None:
        parent = np.ones(n_p, dtype=np.float64) / n_p

    def _norm(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s < 1e-15:
            return np.ones(a.size, dtype=np.float64) / a.size
        return a / s

    def _score_arr(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            return np.inf
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s < 1e-12:
            return np.inf
        c = np.convolve(a, a)
        mx = float(np.max(c))
        return float(2.0 * a.size * mx / (s * s))

    def _softmax(z):
        z = np.asarray(z, dtype=np.float64).ravel()
        z = z - float(z.max())
        e = np.exp(np.clip(z, -700, 700))
        s = float(e.sum())
        if s < 1e-15:
            return np.ones(z.size, dtype=np.float64) / z.size
        return e / s

        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == m:
            return _norm(a)
        if a.size == 1:
            return np.ones(m) / m
        x1 = np.linspace(0, 1, m)
        x0 = np.linspace(0, 1, a.size)
        out = np.interp(x1, x0, a)
        return _norm(out)

            v_ad = b2 * v_ad + (1 - b2) * grad_z ** 2

                m_ad = np.zeros(n)
                v_ad = np.zeros(n)
                t_ad = 0
                lr = max(0.1, lr * 0.6)
                stall = 0

    # --- Phase 3: Random polish ---
    def _random_polish(f0, n_trials, rng_p):
        f = _norm(f0)
        n = len(f)
        best_sc = _score_arr(f)

                best_sc = sc
                best_f = cand

        return best_f, best_sc
