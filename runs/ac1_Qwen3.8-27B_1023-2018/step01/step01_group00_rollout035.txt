```python
def lp_iteration(f, n):
    """
    Given f (length n), solve:
    max sum(g) s.t. (f*g)(k) <= M for all k=0,...,2n-2, g >= 0
    where M = max(f*f).
    Returns g.
    """
    M = np.max(np.convolve(f, f))
    
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # (f*g)(k) = sum_j A[k,j] * g[j]
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    # max sum(g) s.t. A @ g <= M, g >= 0
    # Equivalent to: min -sum(g) s.t. A @ g <= M, g >= 0
    
    from scipy.optimize import linprog
    c = -np.ones(n)

    if res.success:
        g = res.x
    else:
        g = np.zeros(n)
    
    return g, M
```

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _fft_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a):
    """Fast score (scale-invariant, same as evaluator for valid inputs)."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = _fft_conv(a)
    m = float(b.max())
    if not np.isfinite(m) or m < 0:
        return float('inf')
    return 2.0 * n * m / (s * s)

def _lp_solve(f, n):
    """
    Solve: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0.
    f should be normalized (sum = 1).
    Returns g (unnormalized) and M.
    """
    b = _fft_conv(f)
    M = float(b.max())
    if M < 1e-15:
        return np.ones(n) / n, M
    
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            A[k, lo:hi+1] = f[k - hi : k - lo + 1][::-1]
    
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'dual_feasibility_tolerance': 1e-10,
                           'primal_feasibility_tolerance': 1e-10,
                           'time_limit': 5.0})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        if g.sum() > 0:
            return g, M
    return np.ones(n) / n, M

def _lp_optimize(f0, n, stop_time, max_iters=10000):
    """Run LP-based iteration to minimize score."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_f = f.copy()
    best_score = _score(f)
    
    t = 0.5  # step size
    stagnant = 0
    last_improve_iter = 0
    iter_count = 0
    
    while time.monotonic() < stop_time - 3.0:
        iter_count += 1
        
        b = _fft_conv(f)
        M = float(b.max())
        
        if M < 1e-15:
            break
        
        g, M = _lp_solve(f, n)
        g_sum = g.sum()
        if g_sum < 1e-10:
            stagnant += 1
            if stagnant > 50:
                break
            continue
        g = g / g_sum
        
        # Check if g is essentially the same as f
        if np.allclose(g, f, atol=1e-8):
            stagnant += 1
            if stagnant > 20:
                break
            continue
        
        # Try different step sizes
        improved = False
        for trial_t in [t, 0.7*t, 0.4*t, 0.2*t, 0.1*t]:
            f_new = (1 - trial_t) * f + trial_t * g
            f_new = np.maximum(f_new, 1e-15)
            f_new = f_new / f_new.sum()
            s_new = _score(f_new)
            if s_new < best_score - 1e-12:
                f = f_new
                best_f = f_new.copy()
                best_score = s_new
                improved = True
                t = trial_t  # use this step size
                break
            elif s_new < _score(f) + 1e-10:
                # Accept as progress even if not best
                f = f_new
                improved = True
                t = trial_t
                break
        
        if improved:
            stagnant = 0
            last_improve_iter = iter_count
        else:
            stagnant += 1
            t = max(0.01, t * 0.7)
            if stagnant > 100:
                break
        
        # Periodically increase t to escape slow convergence
        if iter_count - last_improve_iter > 30:
            t = min(0.8, t * 1.3)
    
    return best_f, best_score

def _smooth_gradient_optimize(f0, n, stop_time):
    """Use smooth (log-sum-exp) approximation for gradient-based refinement."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    
    best_f = f.copy()
    best_score = _score(f)
    
    lr = 0.1
    tau = 0.05  # smoothing parameter
    stuck = 0
    
    m_acc = np.zeros(n)
    v_acc = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    t_count = 0
    
    while time.monotonic() < stop_time - 3.0:
        b = _fft_conv(f)
        bmax = float(b.max())
        score = 2.0 * n * bmax  # with sum(f)=1, score = 2n * bmax
        
        # Soft log-sum-exp for smooth max approximation
        tau_cur = max(1e-10, 0.03 * max(bmax, 1e-10) * (1.0 + 0.5*math.cos(stuck*0.1)))
        b_shifted = (b - bmax) / tau_cur
        exp_b = np.exp(b_shifted)
        logsumexp = bmax + tau_cur * math.log(exp_b.sum())
        p = exp_b / exp_b.sum()  # "attention" weights on lags
        
        # Compute gradient of smooth max w.r.t. f
        # d/d(f[i]) of logsumexp(b/tau) = (1/tau) * sum_k p[k] * (f*f)(k)'[i]
        # (f*f)(k)'[i] = 2*f[k-i] (with bounds)
        grad = np.zeros(n)
        # Use top lags for efficiency
        top_k = min(200, max(20, n // 2))
        idx = np.argpartition(p, -top_k)[-top_k:]
        idx = idx[p[idx] > 1e-10 * p.max()]
        
        for k in idx:
            k = int(k)
            pk = p[k]
            if pk < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                contrib = 2.0 * pk * f[k - hi : k - lo + 1][::-1]
                grad[lo:hi+1] += contrib
        
        # Adagrad-style update
        t_count += 1
        m_acc = beta1 * m_acc + (1 - beta1) * grad
        v_acc = beta2 * v_acc + (1 - beta2) * grad * grad
        m_hat = m_acc / (1 - beta1**t_count)
        v_hat = v_acc / (1 - beta2**t_count)
        
        direction = m_hat / (np.sqrt(v_hat) + 1e-10)
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-15:
            stuck += 1
            if stuck > 100:
                break
            continue
        
        direction /= dnorm
        
        # Multiplicative (mirror) gradient step
        f_new = f * np.exp(-lr * direction)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        s_new = _score(f_new)
        if s_new < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = s_new
            f = f_new
            stuck = 0
            lr = min(0.5, lr * 1.05)
        else:
            stuck += 1
            lr = max(0.01, lr * 0.9)
            if stuck > 150:
                break
    
    return best_f, best_score

def _local_search(f0, n, stop_time, seed=123):
    """Random local search: perturb and accept if improvement."""
    rng = np.random.default_rng(seed)
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_f = f.copy()
    best_score = _score(f)
    
    stuck = 0
    sigma = 0.1
    
    while time.monotonic() < stop_time - 3.0:
        r = rng.random()
        if r < 0.4:
            # Log-space perturbation
            log_f = np.log(np.maximum(f, 1e-15))
            log_f += rng.normal(0, sigma, n)
            f_new = np.exp(log_f - log_f.max())
            f_new = f_new / f_new.sum()
        elif r < 0.6:
            # Block perturbation
            width = rng.integers(1, max(2, n // 10))
            start = rng.integers(0, n - width)
            f_new = f.copy()
            f_new[start:start+width] *= np.exp(rng.normal(0, 0.3))
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
        elif r < 0.75:
            # Symmetrize
            f_new = 0.5 * (f + f[::-1])
            f_new = f_new / f_new.sum()
        elif r < 0.85:
            # Blend with uniform
            alpha = rng.uniform(0.1, 0.5)
            f_new = (1 - alpha) * f + alpha / n
            f_new = f_new / f_new.sum()
        else:
            # Laplace perturbation
            f_new = f + rng.laplace(0, sigma * 0.5, n)
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
        
        s_new = _score(f_new)
        if s_new < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = s_new
            f = f_new
            stuck = 0
            sigma = max(0.05, sigma * 0.95)
        else:
            stuck += 1
            if stuck % 30 == 0:
                sigma = min(0.5, sigma * 1.2)
            if stuck > 500:
                break
    
    return best_f, best_score
