    a = _normalize(a_init.copy())
    local_score = _score(a)
    best_local = a.copy()
    best_local_score = local_score

    if max_lp_seconds is None:
        max_lp_seconds = 200.0

    stop = time.monotonic() + max_lp_seconds
    stop = min(stop, deadline - 30)

    # Precompute pattern
    rows, cols = _make_conv_pattern(n)
    c_vec = -np.ones(n, dtype=np.float64)

    it = 0
    stagnant = 0
    while time.monotonic() < stop:
        it += 1
        b_conv = np.convolve(a, a)
        M = float(b_conv.max())
        if M <= 1e-15:
            break

        # Build LP matrix
        A = _build_conv_matrix(a, rows, cols)
        b_ub = np.full(2 * n - 1, M * (1.0 + 1e-10))

        # Solve LP: max sum(b) s.t. A@b <= b_ub, b >= 0
        try:
            tl = max(1.0, min(10.0, stop - time.monotonic() - 5))
            res = linprog(
                c_vec, A_ub=A, b_ub=b_ub,
                bounds=[(0.0, None)] * n,
                method='highs',
                options={'time_limit': tl, 'presolve': True}
            )
        except Exception:
            break

        if res is None or not res.success or res.x is None:
            stagnant += 1
            if stagnant > 10:
                break
            continue

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if sg < 1e-12:
            stagnant += 1
            if stagnant > 10:
                break
            continue
        g = g / sg

        # Line search for best mixing parameter
        best_t = 0.0
        best_sc = local_score
        # Try a range of t values, focused on small values
        t_values = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for t in t_values:
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0.0, None)
            sc = _score_normalized(cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = _normalize(a)
            local_score = _score_normalized(a)
            stagnant = 0
            if local_score < best_local_score - 1e-12:
                best_local_score = local_score
                best_local = a.copy()
        else:
            stagnant += 1
            if stagnant > 3:
                # Try perturbation to escape
                if stagnant > 15:
                    break
                if stagnant % 5 == 0:
                    pert = a * np.exp(0.1 * rng.standard_normal(n))
                    pert = _normalize(pert)
                    ps = _score_normalized(pert)
                    if ps < local_score:
                        a = pert
                        local_score = ps
                        stagnant = 0

        # Early exit if at target
        if best_local_score < TARGET and it > 20:
            break

    return best_local


def _gradient_polish(a_init, deadline, rng, max_seconds=120):
    """Gradient-based polish using soft-max smoothing."""
    n = a_init.size
    a = _normalize(a_init.copy())
    local_score = _score(a)
    best_a = a.copy()
    best_score = local_score

    stop = min(time.monotonic() + max_seconds, deadline - 20)
    # Use soft temperature for the max
    tau = 5.0  # softness parameter

    def smooth_objective(a):
        b = np.convolve(a, a)
        # Soft max approximation
        b_shifted = b / max(b.max(), 1e-15)
        soft_max = float(np.log(np.sum(np.exp(tau * b_shifted - tau))) / tau)
        return 2.0 * n * soft_max

    stale = 0
    for it in range(5000):
        if time.monotonic() >= stop:
            break

        b = np.convolve(a, a)
        bmax = float(b.max())
        if bmax < 1e-15:
            break

        # Compute gradient of soft-max of convolution
        # d/d a_i [soft_max(conv(a))] ≈ sum_k w_k * a[k-i] / a[k-i] structure
        # More precisely: d soft_max / d conv_k = exp(tau*(conv_k - soft_max)) / sum(exp(...))
        # And d conv_k / d a_i = a[k-i] (for valid indices)

        b_shifted = b / bmax
        exp_vals = np.exp(tau * b_shifted - tau * b_shifted.max())
        weights = exp_vals / exp_vals.sum()

        # Gradient: grad_i = 2n * sum_k weights[k] * a[k-i] (for valid k-i)
        # conv(a, a)[k] = sum_j a[j]*a[k-j], so d/d a_i conv(k) = a[k-i] + a[k-i] (symmetric?)
        # Actually d/d a_i conv(a,k) = a[k-i] (treating the two a's as the same)
        # For the convolution of a with itself: d/d a_i sum_j a[j]*a[k-j] = a[k-i] + a[k-i] = 2*a[k-i]
        # Wait no. conv(a,a)[k] = sum_j a[j] * a[k-j]. If we differentiate w.r.t. a[i]:
        # d/d a[i] sum_j a[j] * a[k-j] = a[k-i] (from the a[j] term when j=i)
        #   + a[i] * ... no wait.
        # d/d a[i] of a[j]*a[k-j] is delta(j,i)*a[k-j] + a[j]*delta(k-j,i)
        # = delta(j,i)*a[k-i] + a[i]*delta(j,k-i) [when j=k-i]
        # So summing over j: a[k-i] + a[k-i] = 2*a[k-i] (if both i and k-i are in [0,n-1])
        # But if k-i < 0 or k-i >= n, then only one term survives.

        # Simpler: use the matrix A. grad = 2n * A^T @ weights
        # where A is the conv matrix (A @ a = conv(a,a))
        # Actually A @ b = conv(a, b), so for conv(a,a) we need A_a @ a where A_a depends on a.
        # d/d a_i conv(a,a) = sum_k (d conv(k)/d a_i) * ... this is getting complex.
        # Let me just use: d/d a_i [max_k conv(a,k)] ≈ 2 * sum_k w_k * a[k-i] (for valid k-i)
        # This is 2 * (A^T @ w)_i where A is the convolution matrix with a's as data.

        # Build A and compute gradient
        rows_p, cols_p = _make_conv_pattern(n)
        data = a[rows_p - cols_p]
        A_mat = sps.csp_matrix((data, (rows_p, cols_p)), shape=(2 * n - 1, n))
        grad = 2.0 * n * (A_mat.T @ weights)
        grad = np.asarray(grad).ravel()

        # Project gradient (for sum=1 constraint): remove mean
        grad = grad - grad.mean()

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        # Line search
        best_alpha = 0.0
        best_sc = local_score
        for alpha in [0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0, 5.0]:
            step = alpha * grad / gnorm
            cand = a - step
            cand = np.clip(cand, 0.0, None)
            cs = float(cand.sum())
            if cs < 1e-12:
                continue
            cand = cand / cs
            sc = _score_normalized(cand)
            if sc < best_sc:
                best_sc = sc
                best_alpha = alpha

        if best_alpha > 0 and best_score > TARGET:
            step = best_alpha * grad / gnorm
            a = np.clip(a - step, 0.0, None)
            a = _normalize(a)
            local_score = _score_normalized(a)
            stale = 0
            if local_score < best_score - 1e-12:
                best_score = local_score
                best_a = a.copy()
        else:
            stale += 1
            if stale % 10 == 0:
                pert = a * np.exp(0.05 * rng.standard_normal(n))
                pert = _normalize(pert)
                ps = _score_normalized(pert)
                if ps < local_score:
                    a = pert
                    local_score = ps
                    stale = 0
                    if local_score < best_score:
                        best_score = local_score
                        best_a = a.copy()
            if stale > 40:
                break

            # Adapt tau
            if stale % 20 == 0:
                tau *= 1.5

    return best_a


def _direct_optimize(a_init, deadline, rng, max_seconds=150):
    """Combined LP + gradient optimization with adaptive switching."""
    n = a_init.size
    a = _normalize(a_init.copy())
    current_score = _score(a)
    best = a.copy()
    best_score = current_score

    stop = min(time.monotonic() + max_seconds, deadline - 30)
    phase = 0  # 0=LP, 1=gradient, 2=LP again

    while time.monotonic() < stop:
        if phase == 0:
            # LP phase
            result = _lp_refine(a, stop - (stop - (time.monotonic())) - 5, rng,
                               max_lp_seconds=min(60.0, stop - time.monotonic() - 5))
            sc = _score(result)
            if sc < best_score:
                best_score = sc
                best = result.copy()
            a = result
            phase = 1
        elif phase == 1:
            # Gradient phase
            result = _gradient_polish(a, stop - 5, rng,
                                     max_seconds=min(40.0, stop - time.monotonic() - 5))
            sc = _score(result)
            if sc < best_score:
                best_score = sc
                best = result.copy()
            a = result
            phase = 2
        else:
            # Back to LP with perturbation
            pert = a * np.exp(0.03 * rng.standard_normal(n))
            pert = _normalize(pert)
            sc = _score(pert)
            if sc < best_score:
                best_score = sc
                best = pert.copy()
            result = _lp_refine(pert, stop - 5, rng,
                               max_lp_seconds=min(40.0, stop - time.monotonic() - 5))
            sc = _score(result)
            if sc < best_score:
                best_score = sc
                best = result.copy()
            a = result
            phase = 0

    return best


def _try_lengths(a_parent, deadline, rng, best_state):
    """Try different sequence lengths, resampling from parent if needed."""
    if a_parent is None:
        return

    parent_n = a_parent.size
    best_score = best_state[0]
    best_seq = best_state[1]

    for target_n in [parent_n, parent_n + 16, parent_n - 16, parent_n + 32]:
        if target_n < 8 or target_n > 256:
            continue
        if target_n == parent_n:
            continue  # Already handled
        if time.monotonic() > deadline - 80:
            break

        # Resample parent to target_n
        old_x = np.linspace(0, 1, parent_n)
        new_x = np.linspace(0, 1, target_n)
        a0 = np.interp(new_x, old_x, a_parent)
        a0 = _normalize(a0)

        result = _lp_refine(a0, deadline - 60, rng, max_lp_seconds=min(80.0, deadline - time.monotonic() - 60))
        sc = _score(result)
        if sc < best_score:
            best_score = sc
            best_seq = result.copy()

    best_state[0] = best_score
    best_state[1] = best_seq


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(10.0, float(budget_s)) - 20.0

    rng = np.random.default_rng(seed)

    # Get parent sequence
    parent = None
    try:
        h = globals().get('height_sequence_1', None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            s = float(parent.sum())
            if s > 0.01:
                parent = parent / s
            else:
                parent = None
    except Exception:
        parent = None

    # Best tracking
    best_score = float('inf')
    best_seq = None

    def update_best(a):
        nonlocal best_score, best_seq
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s < 0.01:
            return
        sc = _score(a)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = _normalize(a).copy()

    # Initialize with parent
    if parent is not None:
        update_best(parent)

    # Fallback: uniform
    if best_score >= 2.0:
        update_best(np.ones(96) / 96)

    # Ensure we have something
    if best_seq is None:
        best_seq = np.ones(96) / 96
        best_score = _score(best_seq)

    # Phase 1: Direct combined optimization on parent
    if parent is not None and time.monotonic() < deadline - 200:
        result = _direct_optimize(parent, deadline - 150, rng, max_seconds=min(200.0, deadline - time.monotonic() - 150))
        update_best(result)

    # Phase 2: Try different initializations at parent length
    n = best_seq.size
    if time.monotonic() < deadline - 200:
        # Perturbed parent
        for attempt in range(4):
            if time.monotonic() > deadline - 120:
                break
            if attempt == 0:
                a0 = parent.copy() if parent is not None else best_seq.copy()
            elif attempt == 1:
                a0 = best_seq * np.exp(0.2 * rng.standard_normal(n))
                a0 = _normalize(a0)
            elif attempt == 2:
                # Binomial mixture
                a0 = 0.7 * best_seq + 0.3 * (np.ones(n) / n)
                a0 = _normalize(a0)
            else:
                # Smoothed random
                base = rng.dirichlet(np.full(n, 0.8))
                a0 = 0.5 * best_seq + 0.5 * base
                a0 = _normalize(a0)

            alloc = min(80.0, deadline - time.monotonic() - 100)
            if alloc < 10:
                break
            result = _direct_optimize(a0, deadline - 80, rng, max_seconds=alloc)
            update_best(result)

    # Phase 3: Try other lengths
    if parent is not None and time.monotonic() < deadline - 80:
        _try_lengths(parent, deadline - 40, rng, [best_score, best_seq])
        # Re-read
        if [best_score, best_seq][0] < best_score:
            pass  # Already updated via reference... let me fix this

    # Actually let me redo the _try_lengths to properly update
    if parent is not None and time.monotonic() < deadline - 80:
        for target_n in [128, 120, 112]:
            if time.monotonic() > deadline - 60:
                break
            old_x = np.linspace(0, 1, parent.size)
            new_x = np.linspace(0, 1, target_n)
            a0 = np.interp(new_x, old_x, parent)
            a0 = _normalize(a0)
            alloc = min(60.0, deadline - time.monotonic() - 50)
            if alloc < 15:
                break
            result = _lp_refine(a0, deadline - 40, rng, max_lp_seconds=alloc)
            update_best(result)

    # Phase 4: Final gradient polish on the best
    if time.monotonic() < deadline - 30:
        polish_time = min(60.0, deadline - time.monotonic() - 25)
        if polish_time > 15:
            result = _gradient_polish(best_seq, deadline - 20, rng, max_seconds=polish_time)
            update_best(result)

    # Phase 5: One more LP pass on best
    if time.monotonic() < deadline - 25:
        alloc = min(40.0, deadline - time.monotonic() - 20)
        if alloc > 10:
            result = _lp_refine(best_seq, deadline - 15, rng, max_lp_seconds=alloc)
            update_best(result)

    # Final validation
    final = best_seq
    if final is None:
        final = np.ones(96) / 96

    final = np.asarray(final, dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.array([1.0])
    else:
        final = final / s
        final = np.clip(final, 0.0, 1000.0)

    # Validate with exact evaluator
    try:
        sc = evaluate_sequence([float(x) for x in final])
        if not math.isfinite(sc):
            final = np.ones(96) / 96
            sc = evaluate_sequence([float(x) for x in final])
    except Exception:
        final = np.ones(96) / 96

    return [float(x) for x in final]
