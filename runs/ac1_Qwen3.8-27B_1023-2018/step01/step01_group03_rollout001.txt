        _power_search(512, deadline - 220, consider)

    starts = _make_starts(n_main, parent, rng, limit=8)
    for a in starts[:5]:
        consider(a)

    # Initial local refinement on several strong starts.
    for i, a0 in enumerate(starts[:6]):
        now = time.monotonic()
        if now > deadline - 350:
            break
        remaining = deadline - now - 280
        if remaining < 40:
            break
        alloc = min(180.0, max(30.0, remaining / max(1, 6 - i)))
        end = min(deadline - 25, now + alloc)
        if end - now < 20:
            continue
        q1, _ = _soft_polish(a0, n_main, end)
        if q1 is not None:
            consider(q1)

    # Deep polish on the current best at the primary resolution.
    if best_arr is not None and best_arr.size == n_main:
        seed_deep = best_arr.copy()
    elif starts:
        seed_deep = starts[0]
    else:
        seed_deep = _central(n_main)

    if time.monotonic() < deadline - 200:
        qd, _ = _soft_polish(seed_deep, n_main, deadline - 110)
        if qd is not None:
            consider(qd)
            if n_main <= 850 and time.monotonic() < deadline - 110:
                ql, _ = _lp_polish(qd, deadline - 90, max_iters=25)
                if ql is not None:
                    consider(ql)
                    if time.monotonic() < deadline - 80:
                        qf, _ = _soft_polish(ql, n_main, deadline - 35)
                        if qf is not None:
                            consider(qf)
            else:
                if time.monotonic() < deadline - 80:
                    qf, _ = _soft_polish(qd, n_main, deadline - 35)
                    if qf is not None:
                        consider(qf)

    # Optional higher-resolution refinement.
    if n_main != 1024 and time.monotonic() < deadline - 320:
        base10 = best_arr if best_arr is not None else _central(n_main)
        p10 = _upscale(base10, 1024)
        consider(p10)
        starts10 = _make_starts(1024, p10, rng, limit=4)
        for a in starts10[:3]:
            consider(a)

        for i, a0 in enumerate(starts10[:3]):
            now = time.monotonic()
            if now > deadline - 180:
                break
            remaining = deadline - now - 130
            if remaining < 35:
                break
            alloc = min(130.0, max(30.0, remaining / max(1, 3 - i)))
            end = min(deadline - 25, now + alloc)
            if end - now < 20:
                continue
            q10, _ = _soft_polish(a0, 1024, end)
            if q10 is not None:
                consider(q10)

        if time.monotonic() < deadline - 220:
            if best_arr is not None and best_arr.size == 1024:
                seed10 = best_arr.copy()
            else:
                seed10 = starts10[0]
            q10d, _ = _soft_polish(seed10, 1024, deadline - 25)
            if q10d is not None:
                consider(q10d)

    if best_arr is None:
        best_arr = _central(n_main)
    if best_arr is None or best_arr.size == 0:
        best_arr = np.array([1.0], dtype=np.float64)

    lst = []
    for x in best_arr:
        try:
            xf = float(x)
        except Exception:
            continue
        if math.isfinite(xf):
            lst.append(min(1000.0, max(0.0, xf)))
    if not lst:
        lst = [1.0]

    ok = False
    if eval_fn is not None:
        try:
            v = float(eval_fn(lst))
            ok = math.isfinite(v)
        except Exception:
            ok = False
    if not ok:
        lst = [1.0]

    return lst
