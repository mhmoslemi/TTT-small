def _score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    n = a.size
    if n == 0:
        return float('inf')
    s = float(a.sum())
    if s < 1e-30:
        return float('inf')
    b = np.convolve(a, a)
    mb = float(b.max())
    if not math.isfinite(mb):
        return float('inf')
    return 2.0 * n * mb / (s * s)

def _lp_iteration(a, n, max_iter=100, time_limit=60, rng=None, best=None):
    """LP-based alternating optimization."""
    if linprog is None:
        return a
    stop = time.monotonic() + time_limit
    c = -np.ones(n)
    
    for it in range(max_iter):
        if time.monotonic() >= stop - 2:
            break
        
        b = np.convolve(a, a)
        M = float(b.max())
        if M <= 0 or not math.isfinite(M):
            break
        
        # Build constraint matrix A: (a*g)[k] <= M
        # A[k, j] = a[j] if 0 <= k-j <= n-1, else 0
        # Actually: (a*g)[k] = sum_j a[j] * g[k-j]
        # For fixed k, the valid j are max(0,k-n+1) <= j <= min(n-1, k)
        # So A[k,j] = a[j] for those j, 0 otherwise
        # This is a banded Toeplitz-like matrix
        
        # For efficiency with small-to-medium n, use dense
        if n <= 200:
            A = np.zeros((2*n-1, n))
            for k in range(2*n-1):
                j_min = max(0, k - n + 1)
                j_max = min(n - 1, k)
                A[k, j_min:j_max+1] = a[j_min:j_max+1]
        else:
            # Use sparse
            rows = []
            cols = []
            data = []
            for k in range(2*n-1):
                j_min = max(0, k - n + 1)
                j_max = min(n - 1, k)
                for j in range(j_min, j_max+1):
                    rows.append(k)
                    cols.append(j)
                    data.append(a[j])
            A = sps.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        
        b_ub = np.full(2*n-1, M * (1.0 + 1e-10) + 1e-12)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)]*n,
                         method='highs',
                         options={'time_limit': min(5, max(1, stop - time.monotonic() - 2))})
        except:
            break
        
        if res is None or res.x is None or not res.success:
            break
        
        g = np.clip(res.x, 0, None)
        sg = float(g.sum())
        if sg < 1e-15:
            break
        g = g / sg
        
        # Blend
        best_sc = _score_fast(a)
        best_cand = a.copy()
        for t in (0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
            cand = (1-t)*a + t*g
            cand = np.clip(cand, 0, None)
            if cand.sum() < 1e-15:
                continue
            cand = cand / cand.sum()
            sc = _score_fast(cand)
            if sc < best_sc:
                best_sc = sc
                best_cand = cand
        
        if best_cand is not a:  # Had improvement
            a = best_cand
            if best is not None:
                _update_best(a, best)
        else:
            # No improvement from blending, try perturbation
            if rng is not None:
                ap = a * np.exp(0.05 * rng.standard_normal(n))
                ap = np.clip(ap, 0, None)
                if ap.sum() > 1e-15:
                    ap = ap / ap.sum()
                    if _score_fast(ap) < _score_fast(a):
                        a = ap
                        if best is not None:
                            _update_best(a, best)
    
    return a

            cols.extend(range(jlo, jhi + 1))
            data.extend(a[jlo:jhi + 1].tolist())

        A = sps.csr_matrix((data, (np.array(rows), np.array(cols))), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M * (1.0 + 1e-12) + 1e-15)

        try:
            tl = max(0.5, min(8.0, stop - time.monotonic() - 2.0))
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0.0, None)] * n,
                          method='highs',
                          options={'time_limit': tl, 'mip_rel_gap': 1e-6})
        except Exception:
            break

        if res is None or getattr(res, 'x', None) is None:
            stagnant += 1
            if stagnant > 6:
                break
            continue

        g = np.clip(np.asarray(res.x, dtype=np.float64), 0, None)
        sg = float(g.sum())
        if sg < 1e-15:
            stagnant += 1
            if stagnant > 6:
                break
            continue
        g = g / sg

        # Blend with line search
        cur_sc = _score_fast(a)
        best_sc = cur_sc
        best_cand = a.copy()
        for t in (0.15, 0.3, 0.5, 0.7, 1.0):
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0, None)
            cs = float(cand.sum())
            if cs < 1e-15:
                continue
            cand = cand / cs
            sc = _score_fast(cand)
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_cand = cand

        if best_cand is not a and best_sc < cur_sc * (1.0 - 1e-8):
            a = best_cand
            stagnant = 0
            if best_sc < best[0] + 1e-12:
                _update_best(a, best)
            if best[0] < TARGET and it > 10:
                break
        else:
            stagnant += 1
            if stagnant % 8 == 0 and rng is not None:
                ap = a * np.exp(0.08 * rng.standard_normal(n))
                ap = _normalize(ap)
                if _score_fast(ap) < cur_sc * (1.0 - 1e-8):
                    a = ap
                    stagnant = 0
                    if a.size > 0:
                        _update_best(a, best)
            if stagnant > 25:
                break

    return a


def _gradient_refine(a_in, n, time_limit, rng, best):
    """Smoothed-max gradient descent on {a>=0, sum(a)=1}."""
    a = np.asarray(a_in, dtype=np.float64).copy()
    stop = time.monotonic() + time_limit
    best_local = _score_fast(a)

    for it in range(800):
        if time.monotonic() >= stop - 1.0:
            break

        b = np.convolve(a, a)
        S = float(a.sum())
        if S < 1e-30:
            break
        R = b * n / (S * S)
        rmax = float(R.max())

        # Softmax weights (peaked at max)
        beta = 40.0 + 20.0 * min(1.0, it / 50.0)
        w = np.exp(np.clip((R - rmax) * beta, -60.0, 0.0))
        ws = float(w.sum())
        if ws < 1e-30:
            w = np.zeros_like(R)
            w[np.argmax(R)] = 1.0
        else:
            w /= ws

        # Gradient: d/da_j sum_k w_k (a*a)_k = 2 * (w correlated with a) at j
        try:
            grad = 2.0 * np.correlate(w, a, mode='full')[:n]
        except Exception:
            break

        gn = float(np.linalg.norm(grad))
        if gn < 1e-14:
            break
        grad = grad / gn

        # Line search (projected: clip + normalize)
        cur_sc = _score_fast(a)
        best_sc = cur_sc
        best_cand = a.copy()
        for step in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4):
            cand = a - step * grad
            cand = np.clip(cand, 0, None)
            cs = float(cand.sum())
            if cs < 1e-15:
                continue
            cand = cand / cs
            sc = _score_fast(cand)
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_cand = cand

        if best_cand is not a and best_sc < cur_sc * (1.0 - 1e-9):
            a = best_cand
            if best_sc < best_local:
                best_local = best_sc
                if best_sc < best[0] + 1e-12:
                    _update_best(a, best)
            if best[0] < TARGET and it > 20:
                break
        else:
            if rng is not None:
                scale = 0.01 + 0.02 * (it % 10) / 10.0
                ap = a + scale * rng.standard_normal(n)
                ap = np.clip(ap, 0, None)
                aps = float(ap.sum())
                if aps > 1e-15:
                    ap = ap / aps
                    if _score_fast(ap) < cur_sc:
                        a = ap
                        if a.size > 0:
                            _update_best(a, best)

    return a


def _make_starts(n, rng, parent):
    """Generate diverse starting points for length n."""
    starts = []
    x = np.linspace(0, 1, n)

    def add(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0, None)
        if a.size != n or a.size == 0:
            return
        s = float(a.sum())
        if s > 1e-15:
            starts.append(a / s)

    add(np.ones(n))
    for c in (0.5, 1.0, 2.0, -0.5, -1.0, -2.0):
        add(np.exp(c * (x - 0.5)))
    for k in range(1, min(5, n // 2 + 1)):
        for amp in (0.4, 0.7, 0.95):
            add(1.0 + amp * np.sin(2 * math.pi * k * x))
            add(1.0 - amp * np.sin(2 * math.pi * k * x))
    for _ in range(3):
        add(rng.dirichlet(np.full(n, 0.5)))
    for _ in range(3):
        add(rng.dirichlet(np.full(n, 1.5)))

    # Edge-heavy
    e = x ** 3
    add(e / e.sum() if e.sum() > 0 else np.ones(n) / n)
    e2 = (1 - x) ** 3
    add(e2 / e2.sum() if e2.sum() > 0 else np.ones(n) / n)
    # U-shape
    u = 2 * (x - 0.5) ** 2 + 0.1
    add(u / u.sum())

    # From parent
    if parent is not None and len(parent) >= 2:
        p = np.asarray(parent, dtype=np.float64).ravel()
        p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
        p = np.clip(p, 0, None)
        if p.sum() > 1e-15:
            gp = np.interp(x, np.linspace(0, 1, len(p)), p)
            add(gp)
            gp2 = gp * (1.0 + 0.05 * rng.standard_normal(n))
            add(np.clip(gp2, 0, None))

    if not starts:
        starts.append(np.ones(n) / n)

    # Score and sort
    scored = sorted((_score_fast(s), s) for s in starts if _score_fast(s) < float('inf'))
    return [s for _, s in scored]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    try:
        remaining = max(0.0, float(budget_s))
    except Exception:
        remaining = 1300.0
    deadline = t0 + max(0.0, remaining - 15.0)

    best = [float('inf'), np.array([1.0, 1.0], dtype=np.float64)]

    # Initialize from parent
    parent = None
    try:
        h = globals().get('height_sequence_1', None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 0:
                sc = evaluate_sequence([float(x) for x in parent])
                if math.isfinite(sc):
                    best[0] = float(sc)
                    best[1] = parent.copy()
    except Exception:
        parent = None

    if best[0] == float('inf'):
        best[0] = _score_fast(best[1])

    # Phase 1: LP search at parent length (n=96) with multiple starts
    if parent is not None and parent.size >= 2:
        n_parent = parent.size
        if time.monotonic() < deadline - 300.0:
            alloc = min(250.0, max(30.0, deadline - time.monotonic() - 320.0))
            starts = _make_starts(n_parent, rng, parent)
            for si, s in enumerate(starts[:min(8, len(starts))]):
                rem = deadline - time.monotonic() - 280.0
                if rem < 15.0:
                    break
                per = rem / max(1, len(starts) - si)
                per = min(60.0, per)
                _lp_iter(s, n_parent, max_iters=80, time_limit=per, rng=rng, best=best)

    # Phase 2: LP at other lengths
    other_lengths = [64, 128, 160, 192, 256]
    for n in other_lengths:
        rem = deadline - time.monotonic() - 100.0
        if rem < 20.0:
            break
        alloc = min(100.0, max(15.0, rem / (len(other_lengths) + 1)))
        starts = _make_starts(n, rng, best[1] if best[1].size != n else parent)
        for s in starts[:min(5, len(starts))]:
            sub_rem = deadline - time.monotonic() - 100.0
            if sub_rem < 8.0:
                break
            _lp_iter(s, n, max_iters=50, time_limit=min(alloc / max(1, len(starts[:5])),
                                                        sub_rem - 80.0),
                     rng=rng, best=best)
        if deadline - time.monotonic() < 80.0:
            break

    # Phase 3: Gradient refinement on best
    if time.monotonic() < deadline - 60.0:
        n_grad = best[1].size
        _gradient_refine(best[1], n_grad, min(80.0, deadline - time.monotonic() - 50.0),
                         rng, best)

    # Phase 4: More LP with gradient output as new start
    if time.monotonic() < deadline - 50.0:
        n_lp2 = best[1].size
        _lp_iter(best[1], n_lp2, max_iters=60,
                 time_limit=min(60.0, deadline - time.monotonic() - 40.0),
                 rng=rng, best=best)

    # Phase 5: Final gradient polish
    if time.monotonic() < deadline - 25.0:
        _gradient_refine(best[1], best[1].size,
                         min(30.0, deadline - time.monotonic() - 20.0),
                         rng, best)

    # Final validation
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.array([1.0])
    fs = float(final.sum())
    if fs < 0.01:
        final = np.clip(final + 0.001, 0.0, 1000.0)
        fs = float(final.sum())
        if fs > 0:
            final = final / fs
    _update_best(final, best)

    final = np.asarray(best[1], dtype=np.float64).ravel()
    if final.size == 0:
        final = np.array([1.0])
    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc) or fin_sc > best[0] + 1e-6:
        # Fallback to a simple valid sequence
        final = np.ones(2, dtype=np.float64)
        fin_sc = evaluate_sequence([1.0, 1.0])

    return [float(x) for x in np.asarray(final, dtype=np.float64)]
