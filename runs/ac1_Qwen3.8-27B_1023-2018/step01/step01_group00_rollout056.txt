def _solve_lp(f, n, stop_time):
    M = float(np.max(f * f))  # Wait, this is element-wise, not convolution
    # Actually M = max of the full convolution
    fsize = next_fast_len(2*n-1)
    F = rfft(f, fsize)
    conv = irfft(F*F, fsize)[:2*n-1]
    M = float(np.max(conv))
    
    if M <= 0:
        return None, 0
    
    # Build constraint matrix
    # A @ b <= M * ones(2n-1)
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1 else 0
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    c = -np.ones(n)  # minimize -sum b
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if res.success:
        b = res.x
        return b, M
    return None, M
