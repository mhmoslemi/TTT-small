Standard Moreau algorithm:
```
def project_simplex(v, s):
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.where(u > (css - s) / np.arange(1, n+1))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - s) / (rho + 1)
    x = np.maximum(v - theta, 0.0)

        for amp in (0.05, 0.15, 0.30):
            if time.monotonic() >= init_stop:
                break
            add_init(base + lowfreq_noise(amp, 5.0))
            add_init(base + lowfreq_noise(amp, 20.0))

        for frac in (0.35, 0.45, 0.50, 0.55, 0.65):
            if time.monotonic() >= init_stop:
                break
            a = np.ones(n, dtype=np.float64)
            cut = int(n * frac)
            a[:cut] = 1.4
            a[cut:] = 0.6
            add_init(a)

    if init_tuples:
        init_tuples.sort(key=lambda t: t[0])
        top_init = [t[1] for t in init_tuples[:3]]
    else:
        top_init = []
        best_x = np.full(n, 1.0, dtype=np.float64)
        best_M = float(conv_self(best_x).max())

    if best_x is None:
        best_x = np.full(n, 1.0, dtype=np.float64)
        best_M = float(conv_self(best_x).max())

    def refine(x, end_time, max_frac=0.02, min_frac=0.002, eta_e0=0.5, eta_m0=0.3, dual_period=150):
        x = feasible(x)
        B = conv_self(x)
        cur_M = float(B.max())
        best = x.copy()
        best_M_local = cur_M

        now = time.monotonic()
        if end_time <= now:
            return feasible(best), best_M_local

        eta_e = float(eta_e0)
        eta_m = float(eta_m0)
        start_t = now
        dur = max(1e-6, end_time - start_t)
        it = 0
        stuck = 0
        last_best_it = 0
        dual_w = None
        last_dual_fit = -10**9

        while time.monotonic() < end_time:
            it += 1

            if not np.all(np.isfinite(x)) or not np.isfinite(cur_M):
                x = feasible(best)
                B = conv_self(x)
                cur_M = float(B.max())

            rem = max(0.0, (end_time - time.monotonic()) / dur)
            rem = 1.0 if rem > 1.0 else rem
            sf = max_frac + (min_frac - max_frac) * (1.0 - rem)
            sigma = max(0.0, sf * cur_M)

            w_exp = lag_weights(B, sigma)

            if _HAVE_SCIPY_OPT and (it - last_dual_fit >= int(dual_period)) and (end_time - time.monotonic() > 90.0):
                dw = fit_dual_weights(x, B)
                last_dual_fit = it
                if dw is not None:
                    dual_w = dw

            if dual_w is not None:
                w = 0.5 * w_exp + 0.5 * dual_w
                ws = float(w.sum())
                if ws > 0.0 and np.isfinite(ws):
                    w = w / ws
                else:
                    w = w_exp
            else:
                w = w_exp

            c = correlate(w, x)
            pos = c > 1e-12
            cscale = float(np.median(c[pos])) if pos.any() else 1.0
            if not np.isfinite(cscale) or cscale < 1e-12:
                cscale = 1.0
            cu = c / cscale
            cu = np.where(np.isfinite(cu), cu, 0.0)

            old_M = cur_M
            wd0 = float(np.dot(w, B))

            cands = []
            y1 = proj_simplex(x - eta_e * cu, n)
            B1 = conv_self(y1)
            M1 = float(B1.max())
            wd1 = float(np.dot(w, B1))
            cands.append((M1, B1, y1, wd1))

            if (it % 3 == 0) or (stuck > 20) or (eta_e < 0.08):
                delta = cu - cu.mean()
                expo = np.clip(-eta_m * delta, -6.0, 6.0)
                y2 = x * np.exp(expo)
                y2 = np.clip(y2, 0.0, 1000.0)
                s2 = float(y2.sum())
                if s2 > 0.01:
                    y2 *= n / s2
                    B2 = conv_self(y2)
                    M2 = float(B2.max())
                    wd2 = float(np.dot(w, B2))
                    cands.append((M2, B2, y2, wd2))

            if not cands:
                continue

            best_cand = None
            tol = 1e-9 * max(1.0, cur_M)
            for Mc, Bc, xc, wdc in cands:
                if not np.isfinite(Mc):
                    continue
                if best_cand is None:
                    best_cand = (Mc, Bc, xc, wdc)
                else:
                    Mb, _, _, wdb = best_cand
                    if Mc < Mb - tol or (abs(Mc - Mb) <= tol and wdc < wdb - tol):
                        best_cand = (Mc, Bc, xc, wdc)

            accepted = False
            if best_cand is not None:
                Mc, Bc, xc, wdc = best_cand
                if Mc < cur_M - tol or (abs(Mc - cur_M) <= tol and wdc < wd0 - tol):
                    accepted = True
                elif stuck > 25 and Mc <= cur_M * 1.003 + 10.0 * tol:
                    accepted = True

            if accepted:
                x = feasible(best_cand[2])
                B = best_cand[1]
                cur_M = float(B.max())
                if cur_M < best_M_local - tol:
                    best_M_local = cur_M
                    best = x.copy()
                    last_best_it = it

                if old_M > cur_M + tol:
                    eta_e = min(1.25, eta_e * 1.05)
                    eta_m = min(0.75, eta_m * 1.05)
                else:
                    eta_e = min(1.05, eta_e * 1.02)
                    eta_m = min(0.55, eta_m * 1.02)
                stuck = 0

                # Do not drift too far above the current best basin.
                if cur_M > best_M_local * 1.05 + 20.0 * tol:
                    x = feasible(best)
                    B = conv_self(x)
                    cur_M = float(B.max())
                    eta_e = max(0.30, 0.7 * eta_e0)
                    eta_m = max(0.15, 0.7 * eta_m0)
            else:
                stuck += 1
                if stuck < 20:
                    eta_e = max(0.03, eta_e * 0.98)
                    eta_m = max(0.01, eta_m * 0.98)
                elif stuck < 120:
                    eta_e = max(0.03, eta_e * 0.92)
                    eta_m = max(0.01, eta_m * 0.92)
                else:
                    base = best if best_M_local <= cur_M else x
                    perturb_ok = False
                    for amp, sigw in ((0.04, 4.0), (0.08, 6.0), (0.15, 10.0), (0.25, 15.0), (0.40, 25.0)):
                        if time.monotonic() >= end_time - 5.0:
                            break
                        z = base + lowfreq_noise(amp, sigw)
                        x = feasible(z)
                        B = conv_self(x)
                        Mp = float(B.max())
                        if np.isfinite(Mp) and Mp < max(old_M, cur_M) * 1.25:
                            cur_M = Mp
                            perturb_ok = True
                            break
                    if not perturb_ok:
                        x = feasible(best)
                        B = conv_self(x)
                        cur_M = float(B.max())
                    eta_e = max(0.30, 0.6 * eta_e0)
                    eta_m = max(0.15, 0.6 * eta_m0)
                    stuck = 0

            # If the optimizer can keep improving sub-max terms but not the
            # global max for a long time, force a controlled basin move.
            if it - last_best_it > 2200 and it > 300 and (end_time - time.monotonic() > 60.0):
                base = best
                z = base + lowfreq_noise(0.10 + 0.05 * (it // 5000), 8.0)
                x = feasible(z)
                B = conv_self(x)
                cur_M = float(B.max())
                eta_e = max(0.30, 0.7 * eta_e0)
                eta_m = max(0.15, 0.7 * eta_m0)
                stuck = 0

        return feasible(best), float(conv_self(best).max())

    # Stage 1: broad annealing from the best initialization.
    now = time.monotonic()
    rem = deadline - now
    if rem > 80.0 and best_x is not None:
        limit1 = now + 0.55 * rem
        bx, bm = refine(
            best_x,
            limit1,
            max_frac=0.028,
            min_frac=0.004,
            eta_e0=0.55,
            eta_m0=0.35,
            dual_period=150,
        )
        if np.isfinite(bm) and bm < best_M:
            best_M = bm
            best_x = bx

    # Stage 2: sharper sharpening / active-set polishing.
    now = time.monotonic()
    rem = deadline - now
    if rem > 60.0 and best_x is not None:
        limit2 = now + 0.55 * rem
        bx, bm = refine(
            best_x,
            limit2,
            max_frac=0.006,
            min_frac=0.0,
            eta_e0=0.45,
            eta_m0=0.25,
            dual_period=90,
        )
        if np.isfinite(bm) and bm < best_M:
            best_M = bm
            best_x = bx

    # Try one strong alternate initialization if time clearly remains.
    if len(top_init) > 1:
        now = time.monotonic()
        rem = deadline - now
        if rem > 90.0:
            alt = top_init[1]
            if alt is not best_x:
                limit_alt = now + 0.40 * rem
                bx, bm = refine(
                    alt,
                    limit_alt,
                    max_frac=0.015,
                    min_frac=0.001,
                    eta_e0=0.50,
                    eta_m0=0.30,
                    dual_period=120,
                )
                if np.isfinite(bm) and bm < best_M:
                    best_M = bm
                    best_x = bx

    # Low-dimensional mode polish: try small structured perturbations of the
    # current best.  This is cheap enough to run near the end and can clean up
    # residual low-frequency asymmetries.
    def mode_polish(x, end_time, max_evals=260):
        x = feasible(x)
        B = conv_self(x)
        best = x.copy()
        best_Mp = float(B.max())
        now = time.monotonic()
        if end_time <= now + 10.0:
            return best, best_Mp

        idx = np.arange(n, dtype=np.float64)
        modes = []
        for m in (1, 2, 3, 5, 8, 13, 21, 34):
            for ph in (0.0, 0.5 * np.pi):
                v = np.sin(2.0 * np.pi * m * idx / n + ph)
                v -= v.mean()
                modes.append(v)
        for m in (1, 2, 3, 5):
            v = np.cos(2.0 * np.pi * m * idx / n)
            v -= v.mean()
            modes.append(v)

        v = idx.copy()
        v -= v.mean()
        modes.append(v)

        center = -np.exp(-((idx - 0.5 * (n - 1)) ** 2) / (2.0 * (n / 8.0) ** 2))
        center -= center.mean()
        modes.append(center)

        blk = np.zeros(n, dtype=np.float64)
        h = n // 2
        blk[:h] = 1.0
        blk[h:] = -1.0
        blk -= blk.mean()
        modes.append(blk)

        for sigw in (5.0, 10.0, 20.0):
            modes.append(lowfreq_noise(1.0, sigw))

        evals = 0
        tol = 1e-9 * max(1.0, best_Mp)

        def score_t(t):
            nonlocal evals
            cand = x + float(t) * v
            cand = np.clip(cand, 0.0, 1000.0)
            s = float(cand.sum())
            if s < 0.01:
                return np.inf
            cand *= n / s
            Bc = conv_self(cand)
            M = float(Bc.max())
            evals += 1
            return M

        for v in modes:
            if evals >= max_evals:
                break
            if time.monotonic() >= end_time - 10.0:
                break
            v = np.asarray(v, dtype=np.float64) - np.mean(np.asarray(v, dtype=np.float64))
            v = np.where(np.isfinite(v), v, 0.0)
            vmax = float(np.max(np.abs(v)))
            if vmax <= 1e-12:
                continue
            v /= vmax

            tmin, tmax = -0.5, 0.5
            pos = v > 0.0
            neg = v < 0.0
            if pos.any():
                tmax = min(tmax, float(np.min((1000.0 - x[pos]) / v[pos])))
            if neg.any():
                tmin = max(tmin, float(np.max(-x[neg] / v[neg])))
            tmin = max(tmin, -1.0)
            tmax = min(tmax, 1.0)
            if tmin >= tmax:
                continue

            mbest = np.inf
            tbest = None
            for t in np.linspace(tmin, tmax, 7):
                if evals >= max_evals or time.monotonic() >= end_time - 10.0:
                    break
                Mt = score_t(float(t))
                if Mt < mbest:
                    mbest = Mt
                    tbest = float(t)

            if tbest is None or not np.isfinite(mbest):
                continue

            if mbest < best_Mp - tol:
                best_Mp = mbest
                best = x + mbest?  # placeholder (never used)

        return best, best_Mp

    # The actual mode_polish is easier to implement without the placeholder;
    # redefine a clean version:
    def mode_polish(x, end_time, max_evals=260):
        x = feasible(x)
        B = conv_self(x)
        best = x.copy()
        best_Mp = float(B.max())
        now = time.monotonic()
        if end_time <= now + 10.0:
            return best, best_Mp

        idx = np.arange(n, dtype=np.float64)
        modes = []
        for m in (1, 2, 3, 5, 8, 13, 21, 34):
            for ph in (0.0, 0.5 * np.pi):
                v = np.sin(2.0 * np.pi * m * idx / n + ph)
                v -= v.mean()
                modes.append(v)
        for m in (1, 2, 3, 5):
            v = np.cos(2.0 * np.pi * m * idx / n)
            v -= v.mean()
            modes.append(v)

        v = idx.copy()
        v -= v.mean()
        modes.append(v)

        center = -np.exp(-((idx - 0.5 * (n - 1)) ** 2) / (2.0 * (n / 8.0) ** 2))
        center -= center.mean()
        modes.append(center)

        blk = np.zeros(n, dtype=np.float64)
        h = n // 2
        blk[:h] = 1.0
        blk[h:] = -1.0
        blk -= blk.mean()
        modes.append(blk)

        for sigw in (5.0, 10.0, 20.0):
            modes.append(lowfreq_noise(1.0, sigw))

        evals = 0
        tol = 1e-9 * max(1.0, best_Mp)

        for v in modes:
            if evals >= max_evals:
                break
            if time.monotonic() >= end_time - 10.0:
                break
            v = np.asarray(v, dtype=np.float64)
            v = v - v.mean()
            v = np.where(np.isfinite(v), v, 0.0)
            vmax = float(np.max(np.abs(v)))
            if vmax <= 1e-12:
                continue
            v /= vmax

            tmin, tmax = -0.5, 0.5
            pos = v > 0.0
            neg = v < 0.0
            if pos.any():
                tmax = min(tmax, float(np.min((1000.0 - x[pos]) / v[pos])))
            if neg.any():
                tmin = max(tmin, float(np.max(-x[neg] / v[neg])))
            tmin = max(tmin, -1.0)
            tmax = min(tmax, 1.0)
            if tmin >= tmax:
                continue

            mbest = np.inf
            tbest = None
            cbest = None
            for t in np.linspace(tmin, tmax, 7):
                if evals >= max_evals or time.monotonic() >= end_time - 10.0:
                    break
                t = float(t)
                cand = x + t * v
                cand = np.clip(cand, 0.0, 1000.0)
                s = float(cand.sum())
                if s < 0.01:
                    continue
                cand *= n / s
                Bc = conv_self(cand)
                M = float(Bc.max())
                evals += 1
                if M < mbest:
                    mbest = M
                    tbest = t
                    cbest = cand.copy()

            if tbest is None or cbest is None or not np.isfinite(mbest):
                continue

            if mbest < best_Mp - tol:
                best_Mp = mbest
                best = cbest.copy()

            # Short golden-section refinement around the best sampled point.
            span = max(1e-5, 0.25 * (tmax - tmin))
            a = max(tmin, tbest - span)
            b = min(tmax, tbest + span)
            if b - a <= 1e-6:
                continue

            def eval_mode(t):
                nonlocal evals
                cand = x + float(t) * v
                cand = np.clip(cand, 0.0, 1000.0)
                s = float(cand.sum())
                if s < 0.01:
                    return np.inf
                cand *= n / s
                Bc = conv_self(cand)
                evals += 1
                return float(Bc.max()), cand.copy()

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            fc, cb = eval_mode(c)
            fd, db = eval_mode(d)

            for _ in range(10):
                if evals >= max_evals or time.monotonic() >= end_time - 10.0:
                    break
                if fc < fd:
                    b, d, fd, db = d, c, fc, cb
                    c = b - gr * (b - a)
                    fc, cb = eval_mode(c)
                else:
                    a, c, fc, cb = c, d, fd, db
                    d = a + gr * (b - a)
                    fd, db = eval_mode(d)
                if b - a <= 1e-6:
                    break

            tm = 0.5 * (a + b)
            fm, _ = eval_mode(tm)
            if np.isfinite(fm) and fm < best_Mp - tol:
                cand = x + tm * v
                cand = np.clip(cand, 0.0, 1000.0)
                cand *= n / max(1e-30, float(cand.sum()))
                best_Mp = fm
                best = feasible(cand)

        return feasible(best), float(conv_self(best).max())

    now = time.monotonic()
    rem = deadline - now
    if rem > 55.0 and best_x is not None:
        limit_polish = now + max(20.0, min(90.0, 0.25 * rem))
        bx, bm = mode_polish(best_x, limit_polish, max_evals=220)
        if np.isfinite(bm) and bm < best_M:
            best_M = bm
            best_x = bx

    # A second short polish can occasionally recover from a poor low-dimensional
    # final step.
    now = time.monotonic()
    rem = deadline - now
    if rem > 35.0 and best_x is not None:
        limit_polish2 = now + max(10.0, min(45.0, 0.35 * rem))
        bx, bm = mode_polish(best_x, limit_polish2, max_evals=80)
        if np.isfinite(bm) and bm < best_M:
            best_M = bm
            best_x = bx

    # Final exact validation with the supplied evaluator, preserving the best
    # exact-scoring valid candidate.
    final_candidates = []
    if best_x is not None:
        final_candidates.append(best_x)
    if parent_feasible is not None and parent_feasible is not best_x:
        final_candidates.append(parent_feasible)
    if len(top_init) > 1:
        final_candidates.append(top_init[1])
    if len(top_init) > 2:
        final_candidates.append(top_init[2])

    best_final = final_candidates[0]
    best_exact = np.inf
    dedup = []
    for cand in final_candidates:
        cand = feasible(cand)
        sig = (cand.shape, float(cand.sum()), float(cand[0]) if cand.size else 0.0, float(cand[-1]) if cand.size else 0.0)
        if sig in dedup:
            continue
        dedup.append(sig)
        if time.monotonic() > hard_deadline - 12.0:
            break
        try:
            score = float(evaluate_sequence([float(max(0.0, min(1000.0, v))) for v in np.asarray(cand, dtype=np.float64)]))
        except Exception:
            score = np.inf
        if np.isfinite(score) and score < best_exact:
            best_exact = score
            best_final = cand

    if not np.isfinite(best_exact) or best_final is None:
        best_final = feasible(best_x if best_x is not None else np.full(n, 1.0))

    out_arr = np.asarray(best_final, dtype=np.float64).ravel()
    out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
    np.clip(out_arr, 0.0, 1000.0, out=out_arr)
    if float(out_arr.sum()) < 0.01:
        out_arr = np.full(n, 1.0, dtype=np.float64)

    # Hard safety: finish with headroom.
    if time.monotonic() > hard_deadline - 2.0:
        return [float(v) for v in out_arr]

    return [float(v) for v in out_arr]
