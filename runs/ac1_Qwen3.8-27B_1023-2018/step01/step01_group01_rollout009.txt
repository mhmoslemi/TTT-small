    # ---- Final validation ----
    best_a = norm_a(np.clip(np.asarray(holder[0], dtype=np.float64), 0.0, 1000.0))
    fs = safe_eval(best_a)
    if not np.isfinite(fs) or fs > holder[1] + 1e-9:
        best_a = holder[0]
    fs = safe_eval(best_a)
    if not np.isfinite(fs):
        best_a = np.clip(np.asarray(base, dtype=np.float64), 0.0, 1000.0)

    out = [float(x) for x in best_a]
    if not out or any(not np.isfinite(x) for x in out):
        out = [1.0] * n
    return out
