  If i=j and k-j != j, the term is a[j]*a[k-j], derivative w.r.t. a[j] is a[k-j].
- If k-i = j, i.e., i = k-j: contribution is a[k-j] (valid if k-j in S and k-j != j)
  Again, if k-j = j, it's the same case.

So:
Case 1: j in S and k-j in S and j != k-j (i.e., k != 2j):
  grad[j] = a[k-j] (from i=j) + a[k-j] (from i=k-j) = 2*a[k-j]
  
  Wait, from i=k-j: the term is a[k-j]*a[j], derivative w.r.t. a[j] is a[k-j]. Yes.
  
Case 2: j in S and k-j in S and j == k-j (i.e., k == 2j):
  Term is a[j]*a[j] = a[j]^2, derivative = 2*a[j]
  
Case 3: j not in S or k-j not in S:
  grad[j] = 0

So: if 0 <= j < n and 0 <= k-j < n:
  grad[j] = 2 * a[k-j]  (this covers both cases, since when k=2j, k-j=j, so 2*a[k-j] = 2*a[j])
else:
  grad[j] = 0

So the gradient is simply: grad[j] = 2*a[k*-j] if 0<=k*-j < n, else 0.

This is very simple! The gradient of the k*-th convolution element w.r.t. a is just 2 times a shifted/reversed version of a.

But we want the gradient of max(conv), which is the gradient of the active (max) elements. If there are multiple near-max elements, we use a weighted combination.

OK so the gradient descent step would be:
1. Compute conv(a,a), find max position k*
2. grad[j] = 2*q[k*-j] if 0 <= k*-j < n
3. Normalize grad to zero-mean (to preserve sum=1)
4. Step: a_new = a - step * grad_normalized
5. Project to a >= 0, sum = 1

This is very cheap - O(n) per step! Much faster than the LP.

import numpy as np
import time
from scipy.optimize import linprog


def _norm1(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01 or not np.isfinite(s):
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s


def _quick_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float('inf')
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    kv = np.convolve(a, a)
    return 2.0 * n * float(kv.max()) / (s * s)


def _grad_at(q, k_pos):
    """Gradient of conv(q,q)[k_pos] w.r.t. q. O(n)."""
    n = len(q)
    grad = np.zeros(n, dtype=np.float64)
    lo = max(0, int(k_pos) - n + 1)
    hi = min(int(k_pos) + 1, n)
    if lo >= hi:
        return grad
    j = np.arange(lo, hi)
    grad[lo:hi] = 2.0 * q[int(k_pos) - j]
    return grad


def _active_grad(q, alpha=12.0):
    """Weighted subgradient using near-max convolution positions."""
    n = len(q)
    kv = np.convolve(q, q)
    m = float(kv.max())
    if m <= 0 or not np.isfinite(m):
        return np.zeros(n), kv, m
    thresh = (1.0 - 0.05) * m
    active = np.where(kv >= thresh)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(kv))])
    w = np.exp(alpha * (kv[active] / m - 1.0))
    wsum = float(w.sum())
    if wsum < 1e-30:
        w = np.ones(active.size) / active.size
    else:
        w = w / wsum
    grad = np.zeros(n, dtype=np.float64)
    for kk, ww in zip(active, w):
        gk = _grad_at(q, int(kk))
        grad += float(ww) * gk
    grad -= grad.mean()
    return grad, kv, m


def _grad_polish(q, n_iter, step0, deadline, rng, decay=0.995, min_step=0.002):
    """Subgradient polishing with adaptive step."""
    q = _norm1(q)
    best = q.copy()
    best_sc = _quick_score(q)
    step = step0
    for it in range(n_iter):
        if time.monotonic() > deadline - 8.0:
            break
        grad, kv, m = _active_grad(q)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-14:
            break
        grad_dir = -grad / gn
        cand = q + step * grad_dir
        cand = _norm1(cand)
        sc = _quick_score(cand)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best = cand.copy()
        if sc < _quick_score(q) - 1e-14:
            q = cand
            step = min(0.3, step * 1.03)
        else:
            step *= decay
            if step < min_step:
                break
    # Add small random perturbations to escape plateaus
    for _ in range(5):
        if time.monotonic() > deadline - 8.0:
            break
        pert = rng.normal(size=len(best)) * 0.05
        cand = _norm1(best + pert)
        sc = _quick_score(cand)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best = cand
    return best, best_sc


def _lp_step(q, deadline):
    """One LP-based direction: max sum(g) s.t. conv(q,g) <= M, g >= 0."""
    n = len(q)
    kv = np.convolve(q, q)
    M = float(kv.max())
    if M <= 0 or not np.isfinite(M):
        return None
    n_cons = 2 * n - 1
    # Build A: A[k,j] = q[k-j] if 0<=k-j<n else 0
    A = np.zeros((n_cons, n), dtype=np.float64)
    for k in range(n_cons):
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        if lo < hi:
            A[k, lo:hi] = q[k - lo : k - hi + 1][::-1]
    b_ub = np.full(n_cons, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, 1000.0)] * n
    try:
        tl = max(1.0, deadline - time.monotonic() - 15.0)
        if tl < 2.0:
            return None
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': min(tl, 30.0)})
        if res is not None and res.success and res.x is not None:
            g = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, 1000.0)
            gs = float(g.sum())
            if gs > 0.01 and np.isfinite(gs):
                g = g / gs
                return g
    except Exception:
        pass
    return None


def _lp_iterate(q, n_iters, deadline):
    """Multiple LP steps with step-size search."""
    q = _norm1(q)
    best = q.copy()
    best_sc = _quick_score(q)
    stagnant = 0
    for it in range(n_iters):
        if time.monotonic() > deadline - 10.0:
            break
        g = _lp_step(q, deadline)
        if g is None:
            stagnant += 2
            if stagnant >= 6:
                break
            continue
        improved = False
        for t in (0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.7, 1.0):
            cand = (1.0 - t) * q + t * g
            cand = _norm1(cand)
            sc = _quick_score(cand)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best = cand.copy()
                improved = True
        if improved:
            q = best.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant >= 6:
                break
    return best, best_sc


def _make_candidates(k, prev, rng, max_starts):
    """Generate diverse initializations of length k."""
    starts = []
    if prev is not None:
        starts.append(_upscale(prev, k))
    # 4-element seed tiled
    base4 = np.array([0.2, 0.4, 0.2, 0.2], dtype=np.float64)
    if k == 4:
        starts.append(base4 / base4.sum())
    else:
        L = max(1, k // 4)
        tiled = np.repeat(base4, L)
        if tiled.size < k:
            tiled = np.concatenate([tiled, base4[:k - tiled.size]])
        starts.append(_norm1(tiled[:k]))
    # Dirichlet samples
    for alpha in (0.5, 1.0, 3.0, 10.0):
        if len(starts) >= max_starts:
            break
        starts.append(_norm1(rng.dirichlet(np.full(k, alpha))))
    # Sine-wave combinations
    while len(starts) < max_starts:
        t = np.arange(k, dtype=np.float64)
        raw = np.ones(k, dtype=np.float64)
        for m in range(1, min(8, max(2, k // 16)) + 1):
            raw += 0.4 * float(rng.normal()) * np.cos(2.0 * np.pi * m * t / k + float(rng.uniform(0, 2*np.pi)))
            raw += 0.4 * float(rng.normal()) * np.sin(2.0 * np.pi * m * t / k + float(rng.uniform(0, 2*np.pi)))
        raw = np.clip(raw, 0.0, None)
        if raw.sum() > 0.01:
            starts.append(_norm1(raw))
        else:
            starts.append(_norm1(tiled[:k]))
    return starts[:max(1, max_starts)]


def _upscale(q, k):
    q = np.asarray(q, dtype=np.float64)
    if q.size == k:
        return _norm1(q)
    if q.size == 0:
        return _norm1(np.ones(k))
    x = np.linspace(0.0, 1.0 - 1e-10, q.size)
    y = np.linspace(0.0, 1.0 - 1e-10, k)
    out = np.interp(y, x, q)
    return _norm1(out)


def _optimize_size(k, prev, rng, deadline, max_starts=4, n_lp=40, n_grad=300):
    """Optimize at a given size. Returns (best_q, best_score)."""
    starts = _make_candidates(k, prev, rng, max_starts)
    best_q = None
    best_sc = float('inf')
    n_starts = len(starts)
    for i, q0 in enumerate(starts):
        if time.monotonic() > deadline - 20.0:
            break
        remaining = max(1, n_starts - i)
        alloc = (deadline - time.monotonic() - 20.0) / remaining
        if alloc < 8.0:
            break
        end_i = min(deadline - 10.0, time.monotonic() + alloc)
        if end_i - time.monotonic() < 5.0:
            break
        # LP phase
        q_lp, sc_lp = _lp_iterate(q0, n_lp, end_i)
        # Gradient phase
        n_grad_this = max(20, min(n_grad, int(alloc * 2)))
        q_final, sc_final = _grad_polish(q_lp, n_grad_this, 0.15, end_i, rng)
        # Additional LP + grad cycle if time
        if time.monotonic() < end_i - 30.0 and k <= 1024:
            q_lp2, sc_lp2 = _lp_iterate(q_final, 15, end_i)
            q_final, sc_final = _grad_polish(q_lp2, max(20, int((end_i - time.monotonic()) * 1.5)), 0.1, end_i, rng)
        if sc_final < best_sc:
            best_sc = sc_final
            best_q = q_final.copy()
    if best_q is None:
        best_q = starts[0]
        best_sc = _quick_score(best_q)
    return best_q, best_sc
