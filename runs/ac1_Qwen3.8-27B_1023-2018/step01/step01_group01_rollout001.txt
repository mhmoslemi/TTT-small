            and no_improve < 4
        ):
            if hard - time.monotonic() < 18.0:
                break

            K, mode = settings[idx % len(settings)]
            idx += 1
            K = max(8, min(K, Lb // 10))

            try:
                b = conv_a(a)
                Mcur = float(np.max(b))
                if (not np.isfinite(Mcur)) or Mcur <= 0.0:
                    break

                topK = max(1, K // 2)
                top = np.argpartition(b, -topK)[-topK:]
                extra_count = max(2, K // 4)
                extra = np.linspace(0, Lb - 1, extra_count).astype(np.int64)

                if mode == 2:
                    randK = max(1, K // 2)
                    ridx = rng.integers(0, Lb, size=randK)
                    sel = np.unique(np.concatenate((top, extra, ridx)))
                else:
                    sel = np.unique(np.concatenate((top, extra)))

                if len(sel) < K:
                    order = np.argsort(b)[::-1]
                    sel = np.unique(np.concatenate((sel, order[: max(0, K - len(sel))])))
                sel = sel[:K].astype(np.int64)

                gcap = float(min(1000.0, max(4.0, 4.0 * float(np.max(a)) + 1.0)))
                A = build_active_matrix(sel, a)
                c = -np.ones(n, dtype=np.float64)
                b_ub = np.full(len(sel), Mcur, dtype=np.float64)

                rem = max(3.0, min(15.0, hard - time.monotonic() - 5.0))
                res = linprog(
                    c,
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=(0.0, gcap),
                    method="highs",
                    options={"time_limit": rem, "presolve": True},
                )
            except Exception:
                if first_call:
                    no_improve = 10
                else:
                    no_improve += 1
                first_call = False
                continue

            first_call = False
            improved = False

            if res.status == 0 and getattr(res, "x", None) is not None:
                g = np.asarray(res.x, dtype=np.float64)
                if np.all(np.isfinite(g)):
                    g = np.clip(g, 0.0, 1000.0)
                    gs = float(np.sum(g, dtype=np.float64))
                    if gs > 1e-6:
                        g = g * (n_float / gs)
                        g = np.clip(g, 0.0, 1000.0)

                        for tt in (0.02, 0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.85):
                            at = normalize((1.0 - tt) * a + tt * g)
                            st = surrogate(at)
                            if np.isfinite(st) and st < best_sc - 1e-9:
                                best_a = at
                                best_sc = st
                                improved = True

                        if improved:
                            a = best_a
                            no_improve = 0
                            if best_sc < best_surr[0]:
                                best_surr[0] = best_sc
                                best_surr_a[0] = best_a.copy()
                        else:
                            no_improve += 1
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if np.isfinite(best_sc) and best_sc < best_surr[0]:
            best_surr[0] = best_sc
            best_surr_a[0] = best_a.copy()
        return best_a

    # ------------------------------------------------------------------
    # Time allocation.
    # ------------------------------------------------------------------
    T = max(1.0, hard - t0)
    t1 = t0 + 0.50 * T
    t2 = t0 + 0.72 * T
    t3 = t0 + 0.86 * T

    current = best_surr_a[0].copy()
    current = local_search(current, t1, 5.0, 800.0, 0.020, perturb=True, tag="A")

    if time.monotonic() < hard:
        current = best_surr_a[0].copy()
        current = local_search(current, t2, 300.0, 6000.0, 0.008, perturb=True, tag="B")

    if time.monotonic() < hard:
        current = best_surr_a[0].copy()
        current = lp_polish(current, t3, max_calls=4)

    final_a = best_surr_a[0].copy()
    if time.monotonic() < hard - 5.0:
        final_a = local_search(final_a, hard - 5.0, 3000.0, 30000.0, 0.003, perturb=False, tag="F")
    else:
        final_a = best_surr_a[0].copy()

    # ------------------------------------------------------------------
    # Exact final validation against the authoritative evaluator.
    # Keep the best exact-score candidate found among the important
    # finalists; never return an invalid fallback unless everything fails.
    # ------------------------------------------------------------------
    final_candidates = [base_norm, best_surr_a[0], best_exact_a[0], final_a]
    for cand in final_candidates:
        try:
            candn = normalize(cand)
            ex = safe_exact(candn)
            if np.isfinite(ex) and ex < best_exact[0]:
                best_exact[0] = ex
                best_exact_a[0] = candn.copy()
        except Exception:
            pass

    out = [float(x) for x in best_exact_a[0]]
    if (not out) or any((not np.isfinite(x)) for x in out):
        out = [1.0] * n

    out = [min(1000.0, max(0.0, x)) for x in out]
    return out
