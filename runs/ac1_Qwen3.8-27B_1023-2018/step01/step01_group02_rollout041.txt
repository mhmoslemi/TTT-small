`a = softmax(z)`, so `da_j/dz_k = a_j * (delta_jk - a_k)`

The gradient w.r.t. z is: `grad_z = J^T * grad_a` where J is the Jacobian.

`= a_k * grad_a[k] - a_k * sum_j a_j * grad_a[j]`
`= a_k * (grad_a[k] - a @ grad_a)`

def _score(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * len(a) * float(c.max()) / (s * s))


def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a / s


def _softmax_log(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_step(z, grad, m, v, t, lr):
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = b1 * m + (1.0 - b1) * grad
    v = b2 * v + (1.0 - b2) * grad * grad
    t += 1
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    step = np.clip(step, -1.5, 1.5)
    z_new = z - step
    return z_new, m, v, t


def _compute_grad_a(a, c, beta):
    """Compute gradient of soft-max objective w.r.t. a (treating p as fixed)."""
    n = len(a)
    cmax = float(c.max())
    zc = beta * (c - cmax)
    p = np.exp(np.minimum(zc, 0.0))
    ps = float(p.sum())
    if ps < 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p /= ps
    # grad_a[j] = 2 * sum_{l} p[l+j] * a[l] - p[2j]*a[j]
    flip_a = a[::-1]
    cp = np.convolve(flip_a, p, mode='full')
    grad = 2.0 * cp[n - 1:2 * n - 1]
    idx2 = 2 * np.arange(n)
    valid = idx2 < len(p)

    return grad
