def _score_fast(a):
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(a.shape) / a.size
    return a / s

def _build_C(a):
    """Build convolution constraint matrix C where (C @ b) = conv(a, b).
    C shape: (2n-1, n). C[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    k_idx = np.arange(2*n-1)[:, None]  # (2n-1, 1)
    j_idx = np.arange(n)[None, :]  # (1, n)
    diff = k_idx - j_idx  # (2n-1, n)
    valid = (diff >= 0) & (diff < n)
    C = np.zeros((2*n-1, n))
    C[valid] = a[diff[valid]]
    return C

def _lp_step(a, M):
    """LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns normalized b (sum=1)."""
    n = len(a)
    C = _build_C(a)
    c_obj = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        b = np.maximum(res.x, 0)
        s = b.sum()
        if s > 1e-15:
            return b / s
    return a.copy()

    it = 0

        M = float(c.max())
        s = a.sum()
        score = 2.0 * a.size * M / (s * s)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        b = _lp_step(a, M)
        # Adaptive step size
        a_new = (1 - t) * a + t * b
        a_new = _normalize(a_new)

            t = max(0.05, t * 0.8)
            a = _normalize((1 - t/2) * a + (t/2) * b)
        
        it += 1
    
    return best_a, best_score

    """Smooth max gradient for conv(a,a)."""
    n = len(a)
    cmax = c.max()
    z = beta * (c - cmax)
    w = np.exp(z)
    ws = w.sum()
    if ws < 1e-300:

        return g

def _grad_single(a, k):
    """Gradient of conv(a,a)[k]."""
    n = len(a)
    m = np.arange(n)
    idx = k - m
    valid = (idx >= 0) & (idx < n)
    g = np.zeros(n)
    g[valid] = 2.0 * a[idx[valid]]

def _adam_state(n):
    return {'m': np.zeros(n), 'v': np.zeros(n), 't': 0}

def _adam_update(params, grad, state, lr):
    b1, b2, eps = 0.9, 0.999, 1e-8
    state['t'] += 1
    t = state['t']
    state['m'] = b1*state['m'] + (1-b1)*grad
    state['v'] = b2*state['v'] + (1-b2)*grad*grad
    m_hat = state['m'] / (1 - b1**t)
    v_hat = state['v'] / (1 - b2**t)
    return params - lr * m_hat / (np.sqrt(v_hat) + eps)

    state = _adam_state(n)

        frac = (time.monotonic() - start) / max(1e-3, total_time)

        score = 2.0 * n * float(c.max()) / (s*s)

            no_improve = 0

    return best_a, best_score
