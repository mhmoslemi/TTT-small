import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(192, dtype=np.float64)
    try:
        p = np.clip(p, 0.0, 1000.0)
    except Exception:
        p = np.ones(192, dtype=np.float64)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(192, dtype=np.float64)
    return p


def _fft_conv(a, fsize):
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: 2 * len(a) - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a, fsize):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return float("inf")
    b = _fft_conv(a, fsize)
    return 2.0 * n * float(b.max()) / (S * S)


def _soft_max(b, tau):
    m = float(b.max())
    x = np.clip((b - m) / tau, -500.0, 500.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros_like(b)
        idx = np.where(b >= m - 1e-15)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(b))])
        p[idx] = 1.0 / idx.size
        return m, p
    p = ex / s
    val = m + tau * math.log(s)
    return val, p


def _grad_soft_score(a, b, p, tau, n, fsize2):
    S = float(a.sum())
    if S <= 0:
        return np.zeros(n)
    # q[j] = sum_m a[m]*p[m+j]  (cross-correlation)
    Fa = rfft(a, fsize2)
    Fp = rfft(p, fsize2)
    corr = irfft(Fa * np.conj(Fp), fsize2)[:n]
    q = np.maximum(corr, 0.0)
    # w[j] = sum_{k=j}^{j+n-1} p[k]  (sliding window)
    prefix = np.zeros(2 * n)
    prefix[1 : 2 * n - 1 + 1] = np.cumsum(np.concatenate([[0.0], p]))
    w = prefix[n : 2 * n] - prefix[:n]
    # M_t
    m = float(b.max())
    x = np.clip((b - m) / tau, -500.0, 500.0)
    M_t = m + tau * math.log(float(np.exp(x).sum()))
    dM_da = a * w + q
    grad = (2.0 * n / S**3) * (S * dM_da - 2.0 * M_t)
    return grad


def _softmax_from_z(z):
    z = z - np.max(z)
    a = np.exp(z)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(z.size, 1.0 / z.size)
    return a / s


def _init_z(n, rng, variant, parent=None):
    i = np.arange(n)
    eps = 1e-9
    v = int(variant)
    if v % 7 == 0:
        z = rng.normal(0.0, rng.uniform(0.3, 0.8), n)
    elif v % 7 == 1:
        z = rng.normal(0.0, 0.4, n)
        for _ in range(3):
            f = int(rng.integers(1, max(2, n // 2)))
            ph = float(rng.uniform(0, 2 * math.pi))
            amp = float(rng.uniform(0.3, 0.8))
            z += amp * np.sin(2 * math.pi * f * i / n + ph)
    elif v % 7 == 2:
        z = np.where(i < n // 2, 0.3, -0.3) + rng.normal(0, 0.25, n)
    elif v % 7 == 3:
        z = np.linspace(-0.5, 0.5, n) + rng.normal(0, 0.3, n)
    elif v % 7 == 4:
        a = np.max(np.abs(rng.normal(0, 1, n)) + 0.3, eps)
        z = np.log(a) - np.log(a).mean()
    elif v % 7 == 5 and parent is not None and parent.size >= 2:
        P = parent.size
        idx = (np.arange(n) * P // n).astype(int)
        a = np.maximum(parent[idx], eps)
        a = a + rng.normal(0, 0.1 * max(1, a.mean()), n)
        a = np.maximum(a, eps)
        z = np.log(a) - np.log(a).mean()
    else:
        z = rng.uniform(-0.8, 0.8, n)
    z -= z.mean()
    return np.clip(z, -10.0, 10.0)


def _optimize_length(n, seed, time_limit, parent, global_best, global_best_score):
    deadline = time.monotonic() + time_limit
    rng = np.random.default_rng(seed)
    fsize = next_fast_len(2 * n - 1)
    fsize2 = next_fast_len(2 * n)

    local_best = None
    local_best_score = global_best_score if (global_best is not None and global_best.size == n) else float("inf")

    n_restarts = max(2, int(time_limit / 25))

    for restart in range(n_restarts):
        if time.monotonic() > deadline - 3:
            break
        rb = min(deadline - time.monotonic(), max(5.0, time_limit / n_restarts * 1.5))
        if rb < 3:
            break

        if restart == 0 and local_best is not None and local_best.size == n:
            z = np.log(np.clip(local_best, 1e-15, None))
        elif local_best is not None and local_best.size == n and rng.random() < 0.25:
            z = np.log(np.clip(local_best * (0.6 + 0.8 * rng.random()), 1e-15, None))
        else:
            z = _init_z(n, rng, restart, parent)

        a = _softmax_from_z(z)
        z = z - z.mean()

        m_a = np.zeros(n)
        v_a = np.zeros(n)
        tc = 0
        b1, b2 = 0.85, 0.999
        lr = 0.4
        stuck = 0
        t_start = time.monotonic()

        while time.monotonic() < t_start + rb and time.monotonic() < deadline - 2:
            b = _fft_conv(a, fsize)
            bmax = float(b.max())
            S = float(a.sum())
            score = 2.0 * n * bmax / (S * S)

            if np.isfinite(score) and score < local_best_score * (1 - 1e-12):
                local_best_score = score
                local_best = a.copy()
                stuck = 0

            progress = min(1.0, (time.monotonic() - t_start) / max(rb, 1e-9))
            tau = max(1e-6 * max(bmax, 1e-15), (0.08 * (1 - progress) ** 2 + 0.002 * progress ** 2) * max(bmax, 1e-15))

            M_t, p = _soft_max(b, tau)
            grad = _grad_soft_score(a, b, p, tau, n, fsize2)

            # Softmax Jacobian

            tc += 1
            m_a = b1 * m_a + (1 - b1) * grad_z
            v_a = b2 * v_a + (1 - b2) * grad_z ** 2
            m_hat = m_a / (1 - b1**tc)
            v_hat = v_a / (1 - b2**tc)
            step = m_hat / (np.sqrt(v_hat) + 1e-8)
            step = np.clip(step, -50.0, 50.0)

            improved = False
            for fct in (1.0, 0.5, 0.25, 0.125, 0.0625):
                nz = z - lr * fct * step
                nz = np.clip(nz, -10.0, 10.0)
                nz = nz - nz.mean()
                na = _softmax_from_z(nz)
                nb = _fft_conv(na, fsize)
                ns = 2.0 * n * float(nb.max()) / (float(na.sum()) ** 2)
                if np.isfinite(ns) and ns < score * (1 - 1e-10):
                    z, a, b = nz, na, nb
                    score = ns
                    improved = True
                    break

            if improved:
                if tc % 40 == 0:
                    lr = min(1.0, lr * 1.03)
            else:
                stuck += 1
                lr = max(0.02, lr * 0.88)
                if stuck > 35:
                    z = z + rng.normal(0, 0.6, n)
                    z = np.clip(z - z.mean(), -10, 10)
                    a = _softmax_from_z(z)
                    m_a = np.zeros(n)
                    v_a = np.zeros(n)
                    tc = 0
                    lr = 0.4
                    stuck = 0
                    break

    # Validate
    if local_best is not None and local_best.size == n:
        local_best = np.clip(local_best, 0.0, 1000.0)
        try:
            es = float(evaluate_sequence([float(x) for x in local_best]))
        except Exception:
            es = float("inf")
        if np.isfinite(es) and es < global_best_score:
            global_best = local_best.copy()
            global_best_score = es
    return global_best, global_best_score


def _polish(best_seq, best_score, time_limit):
    deadline = time.monotonic() + time_limit
    rng = np.random.default_rng(987654321)
    n = best_seq.size
    fsize = next_fast_len(2 * n - 1)
    a_best = np.clip(best_seq, 0.0, 1000.0).astype(np.float64).copy()
    s_best = best_score
    sigma = 0.12
    stuck = 0

    while time.monotonic() < deadline - 1:
        a = a_best.copy()
        r = rng.random()
        if r < 0.35:
            a += rng.normal(0, sigma, n)
        elif r < 0.55:
            w = max(1, int(rng.integers(1, n // 8 + 1)))
            st = int(rng.integers(0, max(1, n - w)))
            a[st : st + w] *= np.exp(rng.normal(0, 0.25))
        elif r < 0.70:
            a = 0.65 * a + 0.35 * a[::-1]
        elif r < 0.82:
            d = rng.uniform(0.93, 1.07)
            a *= np.linspace(d, 1.0 / d, n) ** 0.3
        elif r < 0.92:
            idx = int(rng.integers(0, n))
            w = max(1, int(rng.integers(1, max(2, n // 15))))
            st = min(idx, n - w)
            a[st : st + w] *= rng.uniform(0.5, 2.0)
        else:
            a = 0.8 * a + 0.2 * np.ones(n) / n
            a += rng.normal(0, sigma * 0.5, n)

        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            continue
        b = _fft_conv(a, fsize)
        fs = 2.0 * n * float(b.max()) / (float(a.sum()) ** 2)
        if not np.isfinite(fs) or fs >= s_best * (1 - 1e-9):
            stuck += 1
            if stuck % 25 == 0:
                sigma = min(1.0, sigma * 1.4)
            continue
        try:
            es = float(evaluate_sequence([float(x) for x in a]))
        except Exception:
            es = float("inf")
        if np.isfinite(es) and es < s_best - 1e-12:
            a_best = a.copy()
            s_best = es
            stuck = 0
            sigma = max(0.04, sigma * 0.92)
        else:
            stuck += 1
            if stuck > 150:
                stuck = 0
                sigma = 0.3

    return a_best, s_best
