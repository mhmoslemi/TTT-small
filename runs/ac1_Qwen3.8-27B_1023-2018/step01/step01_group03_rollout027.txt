                return

                return
            lst = [float(x) for x in arr]

            if np.isfinite(val) and val < best_score:
                best_score = val

        except Exception:
            pass

        q = np.asarray(q, dtype=np.float64).ravel()
        if q.size != n:
            q = np.resize(q, n)
        q = np.clip(q, 0.0, 1000.0)
        s = float(q.sum())
        if s <= 0.0 or not np.isfinite(s):
            return np.full(n, 1.0 / n)
        return q / s

        """Return 2n*max(conv(q,q)) assuming sum(q)=1."""
        b = np.convolve(q, q)
        return 2.0 * n * float(b.max())

    # --- Get parent ---
    parent_q = None

        consider(parent_q)

    # --- Build sparse LP matrix for given q ---
    def build_A(q):
        rows = []
        cols = []
        vals = []
        for j in range(n):
            lo = j
            hi = min(j + n, 2 * n - 1)

    def lp_solve(q, M_fact, tl=15.0):

        B = float(b.max())
        if B <= 0 or not np.isfinite(B):

        except Exception:
            return None
        if not res.success:
            return None
        g = np.asarray(res.x, dtype=np.float64)

        s = float(g.sum())
