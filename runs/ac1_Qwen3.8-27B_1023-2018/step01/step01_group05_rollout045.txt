    inits = []

    # Parent (normalized)
    Sp = parent.sum()
    if Sp > 0.01:
        inits.append(parent * n / Sp)

    # All ones
    inits.append(np.ones(n, dtype=np.float64))

    # Cosine/sine perturbations
    idx = np.arange(n, dtype=np.float64)
    for freq in [1, 2, 3, 5]:
        for amp in [0.15, 0.3, 0.5]:
            init = 1.0 + amp * np.sin(2.0 * np.pi * freq * idx / n)
            init = np.maximum(init, 0.0)
            s = init.sum()
            if s > 0.01:
                inits.append(init * n / s)
            init = 1.0 + amp * np.cos(2.0 * np.pi * freq * idx / n)
            init = np.maximum(init, 0.0)
            s = init.sum()
            if s > 0.01:
                inits.append(init * n / s)

    # Two-level blocks
    for ratio in [0.4, 0.5, 0.6]:
        for d in [0.2, 0.4, 0.6]:
            split = int(n * ratio)
            init = np.ones(n)
            init[:split] = 1.0 + d
            init[split:] = 1.0 - d
            init = np.maximum(init, 0.0)
            s = init.sum()
            if s > 0.01:
                inits.append(init * n / s)

    # Random
    for _ in range(4):
        init = 0.5 + 0.5 * rng.random(n)
        init = init * n / init.sum()
        inits.append(init)

    # Evaluate all initializations, pick best
    best = np.ones(n)
    best_score = np.inf
    for init in inits:
        init = np.clip(init, 0, 1000)
        sc = float(evaluate_fast(init))
        if sc < best_score:
            best_score = sc
            best = init.copy()

    # Verify with exact evaluator
    exact_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(exact_sc):
        best = np.ones(n)
        best_score = float(evaluate_sequence(best.tolist()))
    else:
        best_score = min(best_score, exact_sc)

    # --- Phase 1: Coarse optimization (40% time) ---
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a1, sc1 = optimize_gradient(best, time.monotonic() + remaining * 0.35, lr0=1.0, smooth_eps=0.02)
        if sc1 < best_score:
            best = a1
            best_score = sc1

    # --- Phase 2: Medium optimization (30% time) ---
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a2, sc2 = optimize_gradient(best, time.monotonic() + remaining * 0.35, lr0=0.3, smooth_eps=0.01)
        if sc2 < best_score:
            best = a2
            best_score = sc2

    # --- Phase 3: Fine optimization (25% time) ---
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a3, sc3 = optimize_gradient(best, time.monotonic() + remaining * 0.55, lr0=0.08, smooth_eps=0.005)
        if sc3 < best_score:
            best = a3
            best_score = sc3

    # --- Random restarts from best (remaining time) ---
    remaining = deadline - time.monotonic()
    n_restarts = max(1, int(remaining / 30))
    for r in range(n_restarts):
        if time.monotonic() > deadline - 15:
            break
        pert = best + rng.normal(0, 0.03, n)
        pert = np.maximum(pert, 0.0)
        pert = pert * n / pert.sum()
        a_r, sc_r = optimize_gradient(pert, time.monotonic() + remaining / (n_restarts + 1), lr0=0.1, smooth_eps=0.005)
        if sc_r < best_score:
            best = a_r
            best_score = sc_r

    # --- Final validation with exact evaluator ---
    best = np.clip(best, 0.0, 1000.0)
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = np.ones(n, dtype=np.float64)
        final_sc = float(evaluate_sequence(best.tolist()))

    return [float(max(0.0, min(1000.0, x))) for x in best]
