```python
import math

    except Exception:

    """Full convolution a*a, length 2n-1."""
    n = len(a)
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    c = irfft(F*F, fsize)[:2*n-1]
    c = np.maximum(c, 0)

    n = len(a)
    c = _conv(a)
    return 2*n*c.max()

    Gradient of soft_logsumexp(conv(a,a), tau) w.r.t. a.
    prob[k] = softmax weight on lag k.
    grad[j] = sum_i prob[j+i] * a[i] for valid indices.
```

```python
def _lp_blend(a, M, n):
    """
    Solve LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0.
    Returns optimal b.
    """
    # C[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Build C explicitly (banded Toeplitz-like)
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            C[k, j] = a[k-j]
    
    c_obj = -np.ones(n)  # minimize -sum(b)
    A_ub = C
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
```

```python
def _build_conv_matrix(a):

    for k in range(2*n-1):
        lo = max(0, k-n+1)
        hi = min(n-1, k)

        # That's a[k-hi:k-lo+1][::-1]
```

Also, I realize I should think more carefully about the gradient. The soft-logsumexp gradient gives us the direction in the ambient space, but we need to move within the simplex. The projection is: subtract the mean (to stay on sum=1 hyperplane) and clip to non-negative.

Actually, since the score is scale-invariant, I don't even need to maintain sum=1. I can just minimize max(conv(a,a)) subject to a >= 0, sum(a) = 1 (or any positive constant). The gradient direction projected onto the simplex tangent is: grad - mean(grad).

But actually, there's a subtlety. If I'm minimizing max_k(a*a)[k] over the simplex {a >= 0, sum a = 1}, the feasible directions are those with sum(dir) = 0 and a + t*dir >= 0 for small t. The gradient projected onto the tangent space of the affine constraint (sum a = 1) is grad - mean(grad). The non-negativity constraint is handled by the barrier/clip.

Let me also think: should I work in log-space (like the previous program) or directly in the simplex? 

Log-space (softmax parameterization) ensures a > 0 automatically. The mapping from z to a = softmax(z) has the property that:
da_i/dz_j = a_i * (delta_ij - a_j)

So the gradient w.r.t. z is: dF/dz_j = sum_i (dF/da_i) * (delta_ij - a_j) * ... 
= (dF/da_j) - a_j * sum_i (dF/da_i)

This is just grad_a_j - a_j * mean(grad_a / a) ... hmm, it's a bit different.

Actually: dF/dz_j = sum_i (dF/da_i)(da_i/dz_j) = sum_i grad_a[i] * a[i] * (δ_ij - a[j])
= grad_a[j] * a[j] - a[j] * sum_i grad_a[i] * a[i]
= a[j] * (grad_a[j] - sum_i grad_a[i]*a[i])

So the gradient in z-space is: a * (grad_a - <grad_a, a>)  where <.,.> is the dot product.

This is nice because it automatically gives a direction that, when applied to z, keeps a on the probability simplex.

Let me use this parameterization. It's cleaner.

OK, let me also reconsider the LP. The LP I described finds b that maximizes sum(b) subject to (a*b) <= M. But there's another useful LP:

Given the current max is M at lags k1, k2, ..., km (active set), find the direction d that:
- Minimizes the increase in max_k(a+d * a * d * ... ) ... this is getting complicated.

Actually, let me stick with the simpler LP:
max sum(b)
s.t. (a*b)[k] <= M for all k
     b >= 0

This finds the "best" shape to blend toward. The idea is: b is where we can put mass (given the current shape a) without exceeding the bound M. If b is very different from a, it suggests a is far from optimal.

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(192, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(192, dtype=np.float64)
    return p


def _conv(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    if not np.isfinite(c).all():
        c = np.where(np.isfinite(c), c, 0.0)
    return c


def _score_norm(a):
    n = len(a)
    c = _conv(a)
    return 2.0 * n * float(c.max())


def _grad_softmax(a, prob):
    n = len(a)
    prob_rev = prob[::-1]
    fsize = next_fast_len(3 * n)
    A = rfft(a, fsize)
    P = rfft(prob_rev, fsize)
    cr = irfft(A * P, fsize)
    idx = 2 * n - 2 - np.arange(n)
    return cr[idx]


def _build_conv_mat(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        C[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return C


def _solve_lp(a, M, n, C):
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def _refine_gradient(a, n, stop_time, rng):
    best_a = a.copy()
    best_m = float(_conv(a).max())
    tau = 0.015 * best_m
    lr = 0.15
    m_ad = np.zeros(n)
    v_ad = np.zeros(n)
    t = 0
    b1, b2 = 0.9, 0.999
    stuck = 0
    anneal = 1.0

    while time.monotonic() < stop_time - 2.0:
        c = _conv(a)
        cm = float(c.max())
        if cm < best_m - 1e-13:
            best_m = cm
            best_a = a.copy()
            stuck = 0

        tau = max(5e-5 * best_m, tau * 0.992)
        x = (c - cm) / tau
        e = np.exp(x - x.max())
        s_e = e.sum()
        if s_e < 1e-300:
            break
        prob = e / s_e

        grad_a = _grad_softmax(a, prob)
        # Gradient in log-space (z-space)
        inner = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - inner)

        t += 1
        m_ad = b1 * m_ad + (1 - b1) * grad_z
        v_ad = b2 * v_ad + (1 - b2) * grad_z ** 2
        m_hat = m_ad / (1 - b1 ** t)
        v_hat = v_ad / (1 - b2 ** t)
        update = m_hat / (np.sqrt(v_hat) + 1e-10)

        # Line search in z-space
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            z_step = -lr * factor * update
            # Apply step in implicit z-space: a_new = softmax(log(a) + z_step)
            log_a = np.log(np.maximum(a, 1e-30))
            z = log_a + z_step
            z = z - z.max()
            a_new = np.exp(z)
            a_new_sum = a_new.sum()
            if a_new_sum < 1e-300:
                continue
            a_new /= a_new_sum
            cn = _conv(a_new)
            mn = float(cn.max())
            if mn < cm * (1 - 1e-10):
                a = a_new
                improved = True
                break

        if improved:
            lr = min(lr * 1.02, 0.5)
            if t % 40 == 0:
                anneal = max(0.1, anneal * 0.85)
        else:
            stuck += 1
            lr = max(0.005, lr * 0.85)
            if stuck > 30:
                # Perturb and reset
                log_a = np.log(np.maximum(a, 1e-30))
                log_a += rng.normal(0, 0.15, n)
                log_a -= log_a.max()
                a = np.exp(log_a)
                a /= a.sum()
                m_ad = np.zeros(n)
                v_ad = np.zeros(n)
                t = 0
                lr = 0.15
                stuck = 0
                anneal = 1.0

    return best_a, best_m


def _lp_phase(a, n, stop_time, rng):
    best_a = a.copy()
    best_m = float(_conv(a).max())
    M = best_m
    C = _build_conv_mat(a)
    lp_fail = 0

    while time.monotonic() < stop_time - 3.0:
        b_opt = _solve_lp(a, M, n, C)
        if b_opt is None or b_opt.sum() < 1e-15:
            lp_fail += 1
            if lp_fail > 10:
                break
            a *= np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
            a /= a.sum()
            C = _build_conv_mat(a)
            M = float(_conv(a).max())
            continue

        lp_fail = 0
        b_norm = b_opt / b_opt.sum()
        improved = False
        for tb in (0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7):
            a_new = (1 - tb) * a + tb * b_norm
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-300:
                continue
            a_new /= sn
            cn = _conv(a_new)
            mn = float(cn.max())
            if mn < M - 1e-14:
                a = a_new
                M = mn
                C = _build_conv_mat(a)
                if mn < best_m:
                    best_m = mn
                    best_a = a.copy()
                improved = True
                break

        if not improved:
            # Small random kick
            a_p = a * np.exp(rng.normal(0, 0.03, n))
            a_p = np.maximum(a_p, 0)
            sp = a_p.sum()
            if sp < 1e-300:
                break
            a_p /= sp
            cp = _conv(a_p)
            mp = float(cp.max())
            if mp < M:
                a = a_p
                M = mp
                C = _build_conv_mat(a)
            else:
                lp_fail += 1
                if lp_fail > 15:
                    break

    return best_a, best_m


def _polish(a, n, stop_time, rng):
    best_a = a.copy()
    best_m = float(_conv(a).max())
    c = _conv(a)
    current_m = best_m
    no_improve = 0

    while time.monotonic() < stop_time - 2.0:
        c = _conv(a)
        current_m = float(c.max())
        if current_m < best_m:
            best_m = current_m
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        peak_idx = int(np.argmax(c))
        # Gradient direction that reduces the peak
        # d(a*a)[k]/da[j] = a[k-j] for valid j
        k = peak_idx
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[lo:hi + 1]
        orig_len = hi - lo + 1
        # The contribution to a[j] is a[k-j] = a[lo:hi+1] reversed for j in [lo,hi]
        contrib = np.zeros(n)
        contrib[lo:hi + 1] = a[k - hi:k - lo + 1]
        # Direction to reduce (a*a)[k]: move mass from j where a[k-j] is large to j where it's small
        # Simplified: try shifting mass from high-contrib positions to low-contrib
        if orig_len >= 2:
            top = lo + int(np.argmax(seg))
            bot = lo + int(np.argmin(seg))
            for frac in (0.001, 0.003, 0.01, 0.03):
                delta = frac * a[top]
                a_try = a.copy()
                a_try[top] -= delta
                a_try[bot] += delta
                if a_try[top] < 0:
                    continue
                a_try /= a_try.sum()
                cn = _conv(a_try)
                mn = float(cn.max())
                if mn < current_m - 1e-15:
                    a = a_try
                    current_m = mn
                    no_improve = 0
                    break
            else:
                # Random coordinate move
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                delta = 0.002 * max(a[i], 1e-6)
                if rng.random() < 0.5:
                    a[i] -= delta
                    a[j] += delta
                else:
                    a[j] -= delta
                    a[i] += delta
                a = np.maximum(a, 0)
                a /= a.sum()

        if no_improve > 200:
            # Larger perturbation
            a = a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 0)
            a /= a.sum()
            no_improve = 0

    return best_a, best_m
```
