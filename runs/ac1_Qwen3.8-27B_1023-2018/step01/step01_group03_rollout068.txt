def _grad_polish(q, sc, steps, frac, deadline):
    n=q.size
    if n==0: return q,sc,False,frac
    steps=max(0,int(steps)); frac=max(1e-7,float(frac))
    improved=False
    for _ in range(steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        q=_clean_prob(q)
        b=np.convolve(q,q)
        if b.size==0: break
        m=float(b.max())
        if not np.isfinite(m): break
        cur=2*n*m
        if cur < sc - 1e-12:
            sc=cur; improved=True
        g=_active_subgrad(q,b)
        gn=np.linalg.norm(g)
        if gn <= 1e-15:
            break
        d=-g/gn
        qn=np.linalg.norm(q)
        if qn <= 1e-15:
            q=np.full(n,1.0/n); continue
        best_cand=None; best_cs=cur
        for mult in (1.0, 0.5, 0.2, 2.0, 4.0):
            if deadline is not None and time.monotonic() > deadline: break
            cand = q + (frac*qn*mult)*d
            cand = _clean_prob(cand)
            cb=np.convolve(cand,cand)
            if cb.size==0: continue
            vc=2*n*float(cb.max())
            if np.isfinite(vc) and vc < best_cs - 1e-12:
                best_cs=vc; best_cand=cand
        if best_cand is not None:
            q=best_cand; sc=best_cs; improved=True
            frac=min(0.6, frac*1.15)
        else:
            frac=max(1e-7, frac*0.5)
            if frac <= 1e-6:
                break
    return _clean_prob(q), sc, improved, frac

def _clean_prob(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.atleast_1d(v)
    if v.size == 0:
        return np.array([1.0], dtype=np.float64)
    v = np.where(np.isfinite(v) & (v > 0.0), v, 0.0)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(v.size, 1.0 / v.size, dtype=np.float64)
    return v / s


def _norm1(q):
    return _clean_prob(q)


def _score1(q):
    q = np.asarray(q, dtype=np.float64)
    n = q.size
    if n == 0:
        return float("inf")
    b = np.convolve(q, q)
    if b.size == 0:
        return float("inf")
    val = 2.0 * float(n) * float(np.max(b))
    return val if np.isfinite(val) else float("inf")


def _lower_toeplitz_A(q):
    n = q.size
    if n == 0:
        return np.zeros((0, 0), dtype=np.float64)

    if toeplitz is not None:
        c = np.zeros(2 * n - 1, dtype=np.float64)
        c[:n] = q
        r = np.zeros(n, dtype=np.float64)
        r[0] = q[0]
        A = toeplitz(c, r)
        return np.asarray(A, dtype=np.float64)

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[k, lo:hi + 1] = q[k - hi:k - lo + 1][::-1]
    return A

def _active_subgrad(q, b):
    n = q.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if b.size == 0:
        return np.zeros(n, dtype=np.float64)

    bmax = float(np.max(b))
    if not np.isfinite(bmax) or bmax <= 0.0:
        return np.zeros(n, dtype=np.float64)

    L = b.size
    count = int(round(0.035 * L))
    count = max(8, min(128, count))
    count = min(count, L)

    thr = bmax - max(1e-12, 0.02 * bmax)
    idx = np.flatnonzero(b >= thr)

    if idx.size < max(1, count // 2):
        idx = np.argsort(b)[-count:]
    elif idx.size > count:
        idx = idx[np.argsort(b[idx])[-count:]]

    if idx.size == 0:
        idx = np.array([int(np.argmax(b))], dtype=int)

    w = np.exp(15.0 * (b[idx] - bmax))
    sw = float(np.sum(w))
    if not np.isfinite(sw) or sw <= 0.0:
        w = np.full(idx.size, 1.0 / max(1, idx.size), dtype=np.float64)
    else:
        w = w / sw

    grad = np.zeros(n, dtype=np.float64)
    for ix, ww in zip(idx, w):
        k = int(ix)
        if ww <= 1e-16:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * ww * q[k - hi:k - lo + 1][::-1]

    grad -= float(np.mean(grad))
    return grad

    if n <= 0:
        return q, float("inf"), False, float(frac)
    if not np.isfinite(sc):

    steps = max(0, int(steps))

def _fourier_start(k, rng):

            raw += amp * float(rng.normal()) * np.cos(2.0 * np.pi * m * t / float(k))
            raw += amp * float(rng.normal()) * np.sin(2.0 * np.pi * m * t / float(k))

    raw = np.where(np.isfinite(raw) & (raw > 0.0), raw, 0.0)
    s = float(raw.sum())
    if not np.isfinite(s) or s <= 1e-12:
        raw = _tile_q4(k)
        s = float(raw.sum())
    return raw / s


def _make_starts(k, prev, hs1, rng, extras):
    k = max(1, int(k))
    starts = []

    def add(q):
        if q is None:
            return None
        q = _clean_prob(q)
        if q.size != k:
            q = _upscale(q, k)
        for p in starts:
            if p.size == q.size and np.allclose(p, q, rtol=1e-6, atol=1e-9):
                return p
        starts.append(q)
        return q

    if hs1 is not None:
        try:
            add(_upscale(hs1, k))
        except Exception:
            pass

    if prev is not None:
        add(_upscale(prev, k))

    add(_tile_q4(k))
    add(_fourier_start(k, rng))
    for _ in range(max(0, int(extras))):
        add(_fourier_start(k, rng))

    if extras >= 2:
        try:
            dirich = rng.dirichlet(np.full(k, 2.0))
            add(dirich)
        except Exception:
            pass

    if not starts:
        starts.append(np.full(k, 1.0 / k, dtype=np.float64))

    return starts[:6]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = max(20.0, float(budget_s))
    rng = np.random.default_rng(seed)

    t0 = time.monotonic()
    hard_deadline = t0 + budget_s - 2.0
    deadline = t0 + max(5.0, budget_s - 45.0)
    if deadline > hard_deadline:
        deadline = hard_deadline

    best_list = None
    best_score = float("inf")

    def consider(seq):
        nonlocal best_list, best_score
        try:
            arr = np.asarray(seq, dtype=np.float64)
            arr = np.atleast_1d(arr)
            if arr.size == 0:
                return
            arr = np.where(np.isfinite(arr) & (arr > 0.0), arr, 0.0)
            s = float(arr.sum())
            if not np.isfinite(s) or s <= 1e-12:
                return
            arr = arr / s
            arr = np.clip(arr, 0.0, 1000.0)
            lst = [float(x) for x in arr]
            val = float(evaluate_sequence(lst))
            if np.isfinite(val) and val < best_score:
                best_score = val
                best_list = lst
        except Exception:
            pass

    hs1 = globals().get("height_sequence_1", None)
    if hs1 is not None:
        consider(hs1)

    q4_fallback = _tile_q4(4)
    consider(q4_fallback)

    if best_list is None or time.monotonic() > deadline - 10.0:
        return [float(x) for x in (best_list if best_list is not None else q4_fallback)]

    prev = None
    prev_score = float("inf")
    prev_k = 512

    def run_size(k, starts, budget):
        nonlocal prev, prev_score, prev_k
        size_deadline = min(deadline - 30.0, time.monotonic() + max(10.0, float(budget)))
        size_best = None
        size_score = float("inf")

        if not starts:
            starts = [q.copy() for q in [prev if prev is not None else _tile_q4(k)]]

        nstarts = len(starts)
        for i, q0 in enumerate(starts):
            if time.monotonic() > size_deadline - 20.0:
                break

            remaining = max(1, nstarts - i)
            alloc = (size_deadline - time.monotonic()) / float(remaining)
            alloc = min(350.0, max(20.0, alloc))

            if i == 0:
                alloc = min(
                    size_deadline - time.monotonic(),
                    max(alloc, min(420.0, budget * 0.45)),
                )

            end_start = min(size_deadline, time.monotonic() + alloc)
            if end_start - time.monotonic() < 12.0:
                break

            try:
                qo, sco = _optimize_start(q0, end_start - time.monotonic(), deadline, rng)
            except Exception:
                continue

            if qo is None:
                continue
            if not np.isfinite(sco):
                sco = _score1(qo)
                if not np.isfinite(sco):
                    continue

            consider(qo)
            if sco < size_score:
                size_score = sco
                size_best = qo

            if np.isfinite(sco) and sco < prev_score - 1e-9:
                prev = np.asarray(qo, dtype=np.float64)
                prev_score = float(sco)
                prev_k = int(k)

        if size_best is not None:
            consider(size_best)
        return size_best, size_score

    # Light warm-up at short lengths to create good upscaled seeds.
    if deadline - time.monotonic() > 220.0:
        small_deadline = min(deadline - 160.0, t0 + 30.0)
        for k in (16, 32, 64, 128):
            if time.monotonic() > small_deadline - 20.0:
                break
            starts = _make_starts(k, prev, hs1, rng, 1)
            for q0 in starts:
                if time.monotonic() > small_deadline - 20.0:
                    break
                alloc = min(12.0, small_deadline - time.monotonic() - 10.0)
                if alloc < 6.0:
                    continue
                try:
                    qo, sco = _optimize_start(q0, alloc, deadline, rng)
                except Exception:
                    continue
                if qo is not None:
                    consider(qo)
                    if np.isfinite(sco) and sco < prev_score - 1e-9:
                        prev = np.asarray(qo, dtype=np.float64)
                        prev_score = float(sco)
                        prev_k = int(k)

    # Main effort where the supplied family already lives.
    if time.monotonic() < deadline - 180.0:
        remaining = deadline - time.monotonic()
        b512 = min(480.0, max(120.0, remaining - 560.0))
        starts512 = _make_starts(512, prev, hs1, rng, 3)
        try:
            best512, sc512 = run_size(512, starts512, b512)
        except Exception:
            best512, sc512 = None, float("inf")

    # Larger resolution if there is meaningful time left.
    if time.monotonic() < deadline - 170.0:
        remaining = deadline - time.monotonic()
        b1024 = min(620.0, max(100.0, remaining - 90.0))
        starts1024 = _make_starts(1024, prev, hs1, rng, 3)
        try:
            best1024, sc1024 = run_size(1024, starts1024, b1024)
        except Exception:
            best1024, sc1024 = None, float("inf")

    # Final extra polish on the currently stronger family.
    if time.monotonic() < deadline - 130.0:
        k_extra = prev_k if prev_k in (512, 1024) else 512
        b_extra = max(60.0, deadline - time.monotonic() - 70.0)
        b_extra = min(420.0, b_extra)
        starts_extra = _make_starts(k_extra, prev, hs1, rng, 4)
        try:
            best_extra, sc_extra = run_size(k_extra, starts_extra, b_extra)
        except Exception:
            best_extra, sc_extra = None, float("inf")

    # Dedicated last polish on the exact best candidate if time remains.
    if best_list is not None and deadline - time.monotonic() > 130.0:
        try:
            arr = _norm1(np.asarray(best_list, dtype=np.float64))
            alloc = min(80.0, deadline - time.monotonic() - 80.0)
            if alloc > 30.0:
                qo, sco = _optimize_start(arr, alloc, deadline, rng)
                consider(qo)
        except Exception:
            pass

    if best_list is None:
        consider(q4_fallback)
        if best_list is None:
            best_list = [float(x) for x in q4_fallback]

    # Final validation against the authoritative evaluator.
    try:
        if best_list is None or len(best_list) == 0:
            raise ValueError("empty candidate")
        val = float(evaluate_sequence(best_list))
        if not np.isfinite(val):
            best_list = [float(x) for x in q4_fallback]
    except Exception:
        best_list = [float(x) for x in q4_fallback]

    try:
        val = float(evaluate_sequence(best_list))
        if not np.isfinite(val):
            best_list = [1.0]
    except Exception:
        best_list = [1.0]

    return [float(x) for x in best_list]
