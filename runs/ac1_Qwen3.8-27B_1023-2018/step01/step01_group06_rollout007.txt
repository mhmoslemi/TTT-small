    sm = np.maximum(sm, 1e-10)
    sm /= sm.sum()
    inits.append(sm)

    for init in inits:
        sc = float(evaluate_sequence(init.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = init.copy()

    a = best_a.copy()

    # --- Main optimization loop ---
    eta = 0.5
    lr_add = 0.005
    iter_count = 0
    no_improve = 0
    best_ever = best_score

    while time.monotonic() < deadline:
        iter_count += 1
        t_now = time.monotonic()

        # Compute convolution via FFT
        conv = fast_conv(a)
        max_conv = conv.max()

        # Softmax weights with adaptive temperature
        beta = 15.0 + 5.0 * min(no_improve / 30.0, 4.0)
        shifted = beta * (conv - max_conv)
        np.maximum(shifted, -45.0, out=shifted)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            idx_max = int(conv.argmax())
            w = np.zeros(conv_len, dtype=np.float64)
            w[idx_max] = 1.0
        else:
            w /= w_sum

        # Load: load[m] = sum_k w[k] * a[k-m] for valid indices
        load = np.correlate(w, a, mode='valid')

        # Adaptive step size
        load_max = load.max()
        if load_max < 1e-15:
            load_max = 1e-15

        # Multiplicative (mirror descent) update
        scale = eta * load / load_max
        np.clip(scale, 0.0, 3.0, out=scale)
        a = a * np.exp(-scale)

        # Small additive gradient correction
        a = a - lr_add * load / load_max
        a = np.maximum(a, 1e-15)

        # Renormalize
        s = a.sum()
        if s < 1e-6:
            a = np.ones(n, dtype=np.float64) / n
        else:
            a /= s

        # Fast score estimate
        conv_new = fast_conv(a)
        est_score = 2.0 * n * conv_new.max()

        # Periodic exact validation
        do_exact = (iter_count % 30 == 0) or (est_score < best_score * 0.9995)
        if do_exact:
            exact = float(evaluate_sequence(a.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            if est_score >= best_score * 1.0005:
                no_improve += 1

        # Perturbation strategies
        if no_improve >= 80:
            n_pert = max(5, n // 60)
            idx = rng.choice(n, size=n_pert, replace=False)
            a[idx] *= np.exp(rng.normal(0.0, 0.15, size=n_pert))
            a = np.clip(a, 1e-12, 1000.0)
            a /= a.sum()
            no_improve = 0

        if no_improve >= 250:
            # Try LP-based direction
            if t_now < deadline - 40.0:
                try:
                    M_val = conv_new.max()
                    top_k = 150
                    k_indices = np.argsort(conv_new)[-top_k:]

                    rows, cols, data = [], [], []
                    for ii, ki in enumerate(k_indices):
                        k = int(ki)
                        j_lo = max(0, k - n + 1)
                        j_hi = min(n, k + 1)
                        for jj in range(j_lo, j_hi):
                            rows.append(ii)
                            cols.append(jj)
                            data.append(a[k - jj])

                    A_sub = csr_matrix((data, (rows, cols)), shape=(top_k, n))
                    res = linprog(
                        c=-np.ones(n),
                        A_ub=A_sub,
                        b_ub=np.full(top_k, M_val),
                        bounds=[(0.0, None)] * n,
                        method='highs',
                        options={'presolve': True, 'time_limit': 15}
                    )
                    if res.success and res.x is not None:
                        g = np.clip(res.x, 0.0, 1000.0)
                        if g.sum() > 1e-6:
                            g = g / g.sum()
                            blend = 0.25
                            a = (1.0 - blend) * a + blend * g
                            a = np.maximum(a, 1e-12)
                            a = a / a.sum()
                except Exception:
                    pass

            # Bigger random kick
            n_pert = max(10, n // 25)
            idx = rng.choice(n, size=n_pert, replace=False)
            a[idx] *= np.exp(rng.normal(0.0, 0.4, size=n_pert))
            a = np.clip(a, 1e-12, 1000.0)
            a = a / a.sum()
            no_improve = 0
            eta = min(1.2, eta * 1.15)

        if no_improve >= 500:
            # Symmetry blend + re-heat
            if rng.random() < 0.5:
                blend = rng.uniform(0.1, 0.4)
                a = (1.0 - blend) * a + blend * a[::-1]
                a = a / a.sum()
            n_pert = max(20, n // 15)
            idx = rng.choice(n, size=n_pert, replace=False)
            a[idx] *= np.exp(rng.normal(0.0, 0.6, size=n_pert))
            a = np.clip(a, 1e-12, 1000.0)
            a = a / a.sum()
            no_improve = 0
            eta = 0.8

        # Adaptive annealing
        if iter_count % 3000 == 0:
            eta = max(0.15, eta * 0.93)
            lr_add = max(0.001, lr_add * 0.93)

        # Time check with headroom
        if time.monotonic() > deadline - 5.0:
            break

    # Final: try a few quick local refinements on best_a
    for _ in range(50):
        if time.monotonic() > deadline - 3.0:
            break
        conv = fast_conv(best_a)
        max_c = conv.max()
        beta_f = 50.0
        shifted = beta_f * (conv - max_c)
        np.maximum(shifted, -40.0, out=shifted)
        w_f = np.exp(shifted)
        w_f /= w_f.sum()
        load_f = np.correlate(w_f, best_a, mode='valid')
        lm = max(load_f.max(), 1e-15)
        best_a = best_a * np.exp(-0.3 * load_f / lm)
        best_a = np.maximum(best_a, 1e-15)
        best_a /= best_a.sum()

    # Final validation
    result = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(result))
    if not np.isfinite(final_score) or final_score > 2.5:
        # Fallback
        fb = np.asarray(height_sequence_1, dtype=np.float64)
        fb = np.clip(fb, 0, 1000)
        if fb.sum() < 0.01:
            fb = np.ones(n)
        result = [float(x) for x in fb]
        final_score = float(evaluate_sequence(result))

    return result
