def build_A(F, n):
    # A is (2n-1) x n
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(k, n-1)
        if j_start <= j_end:
            js = range(j_start, j_end+1)
            ks = [k]*len(js)
            vals = F[k - np.array(js)]
            rows.extend(ks)
            cols.extend(js)
            data.extend(vals)
    return sp.csc_matrix((data, (rows, cols)), shape=(2*n-1, n))

            b_new = compute_b(a_new)
            mb_new = b_new.max()

            if mb_new <= bmax_loc + 1e-14:
                a_loc = a_new
                b_loc = b_new
                bmax_loc = mb_new
                lr = min(lr * 1.012, 5.0)
                stag = 0
            else:
                if mb_new < bmax_loc:
                    a_loc = a_new
                    b_loc = b_new
                    bmax_loc = mb_new
                    lr = max(lr * 0.7, 1e-12)
                    stag = max(0, stag - 2)
                else:
                    lr = max(lr * 0.45, 1e-12)
                    stag += 1

            # Anneal tau when making progress
            if stag == 0 and it % 400 == 0:
                tau_frac = max(tau_frac * 0.96, tau_min)

            # Stagnation: structured restart
            if stag > 50:
                r2 = np.random.default_rng(it * 7 + 13)
                ptype = it % 5
                if ptype == 0:
                    a_loc += r2.normal(0, 0.04 / np.sqrt(n), n)
                elif ptype == 1:
                    a_loc *= np.exp(r2.normal(0, 0.03, n))
                elif ptype == 2:
                    f = r2.integers(1, 12)
                    a_loc += 0.02 * np.sin(2 * np.pi * f * np.arange(n) / n)
                elif ptype == 3:
                    f = r2.integers(1, 12)
                    ph = r2.uniform(0, 2 * np.pi)
                    a_loc += 0.02 * np.cos(2 * np.pi * f * np.arange(n) / n + ph)
                else:
                    c = r2.integers(n // 4, 3 * n // 4)
                    sig = r2.uniform(n // 20, n // 4)
                    a_loc += 0.02 * np.exp(-0.5 * ((np.arange(n) - c) / sig) ** 2)
                a_loc = np.maximum(a_loc, 0.0)
                ss = a_loc.sum()
                if ss > 1e-15:
                    a_loc /= ss
                else:
                    a_loc = best_a.copy()
                b_loc = compute_b(a_loc)
                bmax_loc = b_loc.max()
                lr = lr_init
                tau_frac = min(tau_frac * 2.0, 0.4)
                momentum = np.zeros(n)
                stag = 0

            # Periodic score check
            if it % 80 == 0:
                sc = float(evaluate_sequence(a_loc.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_loc.copy()

            if time.monotonic() > t_end - 3.0:
                break

        # Final check of this phase
        if time.monotonic() < t_end - 1.0:
            sc = float(evaluate_sequence(a_loc.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_loc.copy()

        return a_loc, best_score, best_a

    # Phase 1: Coarse optimization (broad tau)
    t_phase1 = t0 + 0.40 * total
    b0 = compute_b(a)
    a, best_score, best_a = optimize_phase(a, b0, 0.18, 0.02, 0.5, t_phase1, mu=0.9)

    # Phase 2: Medium optimization
    t_phase2 = t0 + 0.70 * total
    b1 = compute_b(a)
    a, best_score, best_a = optimize_phase(a, b1, 0.08, 0.005, 0.3, t_phase2, mu=0.88)

    # Phase 3: Restart from best with perturbation
    if time.monotonic() < t_phase2 + 0.15 * total:
        t_phase3 = t0 + 0.85 * total
        a3 = best_a + rng.normal(0, 0.03 / np.sqrt(n), n)
        a3 = np.maximum(a3, 0.0)
        a3 /= a3.sum()
        b3 = compute_b(a3)
        a, best_score, best_a = optimize_phase(a3, b3, 0.06, 0.003, 0.3, t_phase3, mu=0.88)

    # Phase 4: Fine polishing
    if time.monotonic() < deadline - 2.0:
        t_phase4 = deadline
        a4 = best_a.copy()
        b4 = compute_b(a4)
        a, best_score, best_a = optimize_phase(a4, b4, 0.015, 0.001, 0.15, t_phase4, mu=0.82)

    # Final: try a few targeted improvements
    if time.monotonic() < deadline - 5.0:
        for attempt in range(3):
            if time.monotonic() > deadline - 4.0:
                break
            af = best_a + rng.normal(0, 0.01 / np.sqrt(n), n)
            af = np.maximum(af, 0.0)
            af /= af.sum()
            bf = compute_b(af)
            # Quick 200-iter polish
            lr_f = 0.08
            tau_f = 0.005
            mom_f = np.zeros(n)
            for _ in range(200):
                if time.monotonic() > deadline - 3.0:
                    break
                tau = tau_f * max(bf.max(), 1e-15)
                grad = compute_grad(af, bf, tau)
                gc = grad - grad.mean()
                mom_f = 0.8 * mom_f + gc
                anew = af - lr_f * mom_f
                anew = np.maximum(anew, 0.0)
                anew /= anew.sum()
                bn = compute_b(anew)
                if bn.max() <= bf.max() + 1e-14:
                    af, bf = anew, bn
                    lr_f = min(lr_f * 1.005, 0.3)
                else:
                    lr_f = max(lr_f * 0.6, 1e-12)
            sc = float(evaluate_sequence(af.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = af.copy()

    # Final validation
    try:
        sc = float(evaluate_sequence(best_a.tolist()))
        if not np.isfinite(sc):
            best_a = np.ones(n) / n
    except Exception:
        best_a = np.ones(n) / n

    return [float(x) for x in best_a]
