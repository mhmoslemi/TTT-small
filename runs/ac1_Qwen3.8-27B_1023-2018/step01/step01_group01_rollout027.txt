            return np.inf
    
    def conv_aa(a):
        fa = np.fft.rfft(a, sizeC)
        return np.fft.irfft(fa * fa, sizeC)[:2*n-1]
    
    def normalize(a):
        s = float(a.sum())
        if s < 1e-15:
            return np.full(n, target_S/n)
        a = a * (target_S / s)
        return np.clip(a, 0.0, 1000.0)
    
    # Initial best
    best_a = normalize(base.copy())
    best_score = safe_eval(best_a)
    
    # Adam state
    a = best_a.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    
    # Phase 1: Main optimization (65% of time)
    p1_end = t0 + 0.65 * (deadline - t0)
    last_eval = t0
    
    while time.monotonic() < p1_end:
        step += 1
        S = float(a.sum())
        
        # Temperature schedule
        frac = min(1.0, step / 4000.0)
        temp = 3.0 * (1.0 + 37.0 * frac)  # 3 to 40
        lr = 0.018 * (0.96 ** (step / 2000.0))
        
        # Convolution
        C = conv_aa(a)
        Cmax = float(C.max())
        
        # Soft max weights
        z = temp * (C - Cmax)
        z = np.minimum(z, 0.0)
        w = np.exp(z)
        wsum = float(w.sum())
        M = Cmax + np.log(wsum) / temp
        w = w / wsum
        
        # Gradient of smooth M w.r.t. a
        # dM/da_i = 2 * sum_k w_k * a(k-i) = 2 * (w_flipped * a)(i)
        wfl = w[::-1]
        fa = np.fft.rfft(a, sizeC)
        fwfl = np.fft.rfft(wfl, sizeC)
        corr = np.fft.irfft(fa * fwfl, sizeC)[:n]
        
        # Full gradient of F = 2n*M/S^2
        grad = 4.0 * n * corr / (S*S) - 4.0 * n * M / (S**3)
        
        # Adam
        m = 0.9*m + 0.1*grad
        v = 0.999*v + 0.001*(grad*grad)
        mb = m / (1.0 - 0.9**step)
        vb = v / (1.0 - 0.999**step)
        
        a = a - lr * mb / (np.sqrt(vb) + 1e-8)
        a = np.clip(a, 0.0, 1000.0)
        a = normalize(a)
        
        # Periodic eval
        now = time.monotonic()
        if step % 250 == 0 and now - last_eval > 0.3:
            last_eval = now
            sc = safe_eval(a)
            if sc < best_score:
                best_a, best_score = a.copy(), sc
    
    # End of phase 1
    sc = safe_eval(a)
    if sc < best_score:
        best_a, best_score = a.copy(), sc
    
    # Phase 2: Low-temperature polish (20% of time)
    p2_end = t0 + 0.85 * (deadline - t0)
    a = best_a.copy()
    m *= 0.5
    v *= 0.5
    step2 = 0
    last_eval = time.mon
