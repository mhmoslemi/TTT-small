**Better approach: Active set / projected subgradient**

At each iteration:
1. Find `k*` = argmax b_k

**LP-based update (dual method):**

Given `a` with `M = max(a*a)`:
- Solve LP: maximize `sum g` subject to `g >= 0`, `(a*g)_k <= M * sum(g)` for all k
  (I normalize by `sum(g)` to make it scale-invariant)
  
  Actually, let me re-read the parent: "maximizing sum_j b_j subject to b_j >= 0 and ||f * g_0||_inf <= M"
  
  So the LP is:
  - Variables: `g_0, ..., g_{n-1}`
  - Maximize: `sum g`
  - Subject to: `g_j >= 0`
  - `(a * g)_k <= M` for all k = 0, ..., 2n-2

  Since `(a*g)_k = sum_i a_i g_{k-i}`, this is linear in `g`.

The solution `g*` satisfies `(a*g*)_k <= M` for all k. Then `a_new = (1-t)*a + t*g*/sum(g*)` is the update.

Why does this help? The key observation is:
- `(a*a)_k <= M` for all k (by definition of M)
- `(a*g*)_k <= M` for all k (by LP constraint)
- By convexity of the max: `((1-t)a + tg*)*((1-t)a + tg*)_k`
  `= (1-t)^2 (a*a)_k + t(1-t)(a*g*)_k + t(1-t)(g*_{norm}*a)_k + t^2 (g*_{norm}*g*_{norm})_k`
  `<= (1-t)^2 M + t(1-t) M + t(1-t) M + t^2 ||g*_{norm}*g*_{norm}||_∞`
  `= M - t^2 M + t^2 ||g*_{norm}*g*_{norm}||_∞`
  `= M - t^2 (M - ||g*_{norm}*g*_{norm}||_∞)`

So if `||g*_{norm}*g*_{norm}||_∞ < M` (which is likely since `g*` is "spread out" by the LP), then the new max is less than M. The improvement is `t^2 * (M - ||g*_{norm}*g*_{norm}||_∞)`.

This is a quadratic improvement in t, which is a fixed-point iteration. It should converge (slowly) but is guaranteed to decrease the max.

For n=192, the LP has 192 variables and 383 inequality constraints (plus nonnegativity). The constraint matrix is:
`A[i, j]` = coefficient of `g_j` in constraint i.
Constraint i: `(a*g)_i <= M`, which is `sum_j a_{i-j} g_j <= M` (summed over valid j).

So `A[i, j] = a[i-j]` if `0 <= i-j < n`, else 0.

This is a (2n-1) × n matrix. For n=192, that's 383 × 192 = 73,536 entries. Very manageable.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(192, dtype=np.float64)
    p = np.clip(p, 0, 1000.0)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(192, dtype=np.float64)
    return p


def _convolve_fft(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    A = rfft(a, fsize)
    c = irfft(A * A, fsize)[:2 * n - 1]
    c[c < 0] = 0.0
    return c


def _eval_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return math.inf
    n = len(a)
    b = _convolve_fft(a)
    return float(2.0 * n * b.max() / (s * s))


def _grad_bk(a, k):
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if hi >= lo:
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] = 2.0 * seg
    return grad


def _grad_smooth_max(a, b, tau, top_k=None):
    n = len(a)
    m = b.max()
    x = (b - m) / tau
    ex = np.exp(x)
    p = ex / ex.sum()
    if top_k is not None and top_k < len(p):
        order = np.argsort(p)[::-1][:top_k]
        p2 = np.zeros_like(p)
        p2[order] = p[order]
        p2 /= p2.sum()
        p = p2
    grad = np.zeros(n)
    for k in range(len(p)):
        if p[k] < 1e-14:
            continue
        grad += p[k] * _grad_bk(a, k)
    return grad


def _smooth_max_val(b, tau):
    m = b.max()
    return m + tau * np.log(np.exp((b - m) / tau).sum())


def _smooth_grad_descent(a, stop_time, lr_init=0.05, tau_seq=None, iters_per_tau=200, top_k=None, momentum=0.9):
    a = np.asarray(a, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a, math.inf
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = _convolve_fft(a).max()

    if tau_seq is None:
        tau_seq = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]

    vel = np.zeros(n)
    fsize = next_fast_len(2 * n - 1)

    for tau in tau_seq:
        if time.monotonic() > stop_time - 2:
            break
        lr = lr_init
        for it in range(iters_per_tau):
            if time.monotonic() > stop_time - 2:
                break
            b = _convolve_fft(a)
            cur_max = b.max()
            if cur_max < best_max - 1e-15:
                best_max = cur_max
                best_a = a.copy()

            grad = _grad_smooth_max(a, b, tau, top_k=top_k)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-16:
                break

            # Momentum step in z-space (log-space implicitly via normalization)
            vel = momentum * vel - lr * grad / gnorm
            a_new = a + vel * 0.01
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn < 1e-10:
                break
            a_new = a_new / sn

            b_new = _convolve_fft(a_new)
            max_new = b_new.max()
            if max_new <= cur_max * (1.0 + 1e-9):
                a = a_new
            else:
                lr *= 0.7
                vel *= 0.5
                if lr < 1e-8:
                    break
    return best_a, 2.0 * n * best_max


def _lp_dual_update(a, stop_time, t=0.1):
    n = len(a)
    b = _convolve_fft(a)
    M = b.max()
    if M < 1e-15:
        return a

    # Build constraint matrix: (a*g)_k <= M for k = 0, ..., 2n-2
    # (a*g)_k = sum_j a[k-j] * g_j for valid j
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            A[k, lo:hi + 1] = a[hi:lo - 1:-1] if lo > 0 else a[hi::-1]

    # Maximize sum(g) subject to A@g <= M, g >= 0
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=[(0, None)] * n,
                  method='highs', options={'time_limit': max(0.5, stop_time - time.monotonic() - 3)})
    if not res.success:
        return a
    g = res.x
    gs = g.sum()
    if gs < 1e-10:
        return a
    g = g / gs
    a_new = (1 - t) * a + t * g
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-10:
        return a
    a_new /= sn
    return a_new


def _polish_local(a, stop_time, rng):
    n = len(a)
    best = a.copy()
    best_score = _eval_fast(best)
    sigma = 0.05
    stuck = 0

    while time.monotonic() < stop_time - 2:
        cand = best.copy()
        r = rng.random()
        if r < 0.4:
            idx = rng.integers(0, n)
            width = max(1, int(rng.integers(1, max(2, n // 30))))
            lo = max(0, idx)
            hi = min(n, idx + width)
            cand[lo:hi] *= rng.uniform(0.5, 2.0)
        elif r < 0.7:
            z = np.log(np.maximum(best, 1e-12))
            z += rng.normal(0, sigma, n)
            e = np.exp(z - z.max())
            cand = e / e.sum()
        else:
            i = rng.integers(0, n - 1)
            j = rng.integers(0, n - 1)
            tmp = cand[i]
            cand[i] = cand[j]
            cand[j] = tmp
            if rng.random() < 0.5:
                cand = (1 - 0.3) * cand + 0.3 * cand[::-1]

        cand = np.clip(cand, 0, 1000.0)
        cs = cand.sum()
        if cs < 1e-10:
            continue
        sc = _eval_fast(cand)
        if sc < best_score - 1e-12:
            best = cand
            best_score = sc
            stuck = 0
            sigma = max(0.02, sigma * 0.9)
        else:
            stuck += 1
            if stuck > 50:
                sigma = min(0.5, sigma * 1.5)
                stuck = 0
    return best, best_score


def _try_length(a_template, L, stop_time, seed, parent):
    """Resize template to length L and optimize."""
    n_orig = len(a_template)
    if L == n_orig:
        a = a_template.copy()
    else:
        # Interpolate
        x_old = np.linspace(0, 1, n_orig)
        x_new = np.linspace(0, 1, L)
        a = np.interp(x_new, x_old, a_template)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(L) / L
        else:
            a = a / s

    rng = np.random.default_rng(seed)

    # Method 1: Smooth gradient descent
    t1 = min(stop_time - 5, time.monotonic() + 120)
    a1, s1 = _smooth_grad_descent(a, t1, lr_init=0.03,
                                  tau_seq=[0.01, 0.005, 0.002, 0.001, 0.0005],
                                  iters_per_tau=150, top_k=8, momentum=0.85)

    # Method 2: LP dual
    t2 = min(stop_time - 8, time.monotonic() + 30)
    a2 = a1.copy()
    ss = a2.sum()
    a2 = a2 / ss
    for _ in range(5):
        if time.monotonic() > t2:
            break
        a2 = _lp_dual_update(a2, stop_time - 5, t=0.15)
        a2 = np.maximum(a2, 0)
        a2 /= a2.sum()

    # Method 3: Polish
    t3 = min(stop_time - 3, time.monotonic() + 60)
    best_a = a1.copy()
    best_s = _eval_fast(a1)
    if _eval_fast(a2) < best_s:
        best_a = a2
        best_s = _eval_fast(a2)
    best_a, best_s = _polish_local(best_a, t3, rng)

    return best_a, best_s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 10.0  # 10s safety margin

    parent = _get_parent()
    parent = np.clip(parent, 0, 1000.0)
    ps = parent.sum()
    if ps < 1e-10:
        parent = np.ones(192) / 192
        ps = 1.0
    parent = parent / ps  # normalize to sum=1

    best_seq = parent.copy()
    best_score = _eval_fast(parent)
    if not np.isfinite(best_score):
        best_seq = np.ones(192) / 192
        best_score = 2.0

    rng = np.random.default_rng(seed)

    # Try the parent length and nearby lengths
    n_parent = len(parent)
    lengths_to_try = sorted(set([n_parent]))
    # Add nearby lengths
    for d in [16, 32, 48, 64, 96, 128, 160, 256, 320, 384]:
        for L in [n_parent - d, n_parent + d]:
            if L >= 8:

    lengths_to_try = [L for L in lengths_to_try if 8 <= L <= 512]
    lengths_to_try = sorted(set(lengths_to_try))

    # Sort by priority: parent length first, then nearby
    def length_priority(L):
        return abs(L - n_parent)
    lengths_to_try.sort(key=length_priority)

    tried = set()
    for L in lengths_to_try:
        if L in tried:

        tried.add(L)
        if time.monotonic() > deadline - 60:

        remaining = deadline - time.monotonic()

    # Convert to list of floats, ensuring validity

    if not all(np.isfinite(x) for x in result) or sum(result) < 0.01:
        result = [1.0]

    return result
