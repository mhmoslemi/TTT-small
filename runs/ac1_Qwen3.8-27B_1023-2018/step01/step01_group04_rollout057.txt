    a = np.clip(np.nan_to_num(a), 0, 1000)
    n = a.size

    mb = float(b.max())

def _build_A(a):
    """Sparse matrix A where A @ g = conv(g, a[reversed]).
    Wait, I need to be careful about the indexing.
    
    conv(g, a)[k] = sum_j g[j] * a[k-j]
    So A[k, j] = a[k-j] if 0 <= k-j < len(a), else 0.
    """

    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    A = _build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success and res.x is not None:
        return np.clip(res.x, 0, None)
    return None

def _gradient_step(a, eta):
    """One gradient step to reduce the score."""
    n = a.size
    s = float(a.sum())
    if s <= 0:
        return a.copy()
    b = np.convolve(a, a)
    i_star = int(np.argmax(b))
    b_max = float(b[i_star])
    
    grad = np.zeros(n)
    for j in range(n):
        idx = i_star - j
        if 0 <= idx < n:
            grad[j] = 4.0 * n * (a[idx] * s - b_max) / (s ** 3)
        else:

    a_new = np.clip(a_new, 0, None)
    s_new = a_new.sum()
    if s_new > 0:
