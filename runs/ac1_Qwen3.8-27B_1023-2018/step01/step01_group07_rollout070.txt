            time_limit = time.monotonic() + min(remaining - 60, 120)
            L2 = _next_pow2(3 * n_try - 2)
            # Random nonnegative initialization
            a2 = rng.random(n_try)
            a2 += 0.1
            a2 = a2 / a2.sum()

            m2 = np.zeros(n_try)
            v2 = np.zeros(n_try)
            t2 = 0
            lr2 = 0.3
            tau2 = 0.1
            stag2 = 0

            for _ in range(20000):
                if time.monotonic() > time_limit:
                    break
                t2 += 1
                a2_pad = np.zeros(L2)
                a2_pad[:n_try] = a2
                F2 = np.fft.rfft(a2_pad)
                c2 = np.fft.irfft(F2 * F2, L2)[:2*n_try-1]
                cm2 = c2.max()
                if cm2 < 1e-30:
                    break
                t_val = tau2 * cm2
                sh2 = np.clip((c2 - cm2) / t_val, -500.0, 0.0)
                w2 = np.exp(sh2)
                ws2 = w2.sum()
                if ws2 < 1e-30:
                    w2 = np.zeros(2*n_try-1)
                    w2[int(np.argmax(c2))] = 1.0
                    ws2 = 1.0
                w2 /= ws2
                w2_pad = np.zeros(L2)
                w2_pad[:2*n_try-1] = w2
                Fw2 = np.fft.rfft(w2_pad)
                corr2 = np.fft.irfft(np.conj(Fw2) * F2, L2)[:n_try]
                g2 = 2.0 * corr2 / t_val
                g2 -= g2.mean()

                m2 = 0.85 * m2 + 0.15 * g2
                v2 = 0.999 * v2 + 0.001 * g2 * g2
                m2h = m2 / (1.0 - 0.85 ** t2)
                v2h = v2 / (1.0 - 0.999 ** t2)

                a2_new = a2 - lr2 * m2h / (np.sqrt(v2h) + 1e-10)
                a2_new = np.maximum(a2_new, 0.0)
                s2 = a2_new.sum()
                if s2 < 1e-15:
                    a2_new = a2.copy()
                else:
                    a2_new = a2_new / s2

                a2_pad_t = np.zeros(L2)
                a2_pad_t[:n_try] = a2_new
                Ft2 = npfft.rfft(a2_pad_t)
                c2_new = np.fft.irfft(Ft2 * Ft2, L2)[:2*n_try-1]
                if c2_new.max() <= cm2 + 1e-12:
                    a2 = a2_new
                    stag2 = 0
                else:
                    lr2 = max(lr2 * 0.5, 0.01)
                    stag2 += 1

                if t2 % 100 == 0:
                    sc2 = float(evaluate_sequence(a2.tolist()))
                    if np.isfinite(sc2) and sc2 < best_score:
                        best_score = sc2
                        best_a = a2.copy()
                        n = n_try
                        L = L2

                if stag2 > 200:
                    a2 = a2 * np.exp(rng.normal(0, 0.01, n_try))
                    a2 = np.maximum(a2, 0)
                    a2 /= a2.sum()
                    m2[:] = 0
                    v2[:] = 0
                    t2 = 0
                    lr2 = 0.3
                    tau2 = 0.1
                    stag2 = 0

            remaining = deadline - time.monotonic()

    # === Final validation ===
    sc_final = float(evaluate_sequence(a.tolist()))
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    # Validate best_a
    bs = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(bs):
        best_a = np.ones(n, dtype=np.float64) / n
        bs = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(bs) or bs > 2.0:
        best_a = np.ones(1000, dtype=np.float64) / 1000.0

    return [float(x) for x in best_a]
