try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _norm9(q):
    q = np.asarray(q, dtype=np.float64).ravel()
    k = q.size
    if k == 0:
        return np.array([1.0])
    q = np.where(np.isfinite(q) & (q > 0.0), q, 0.0)
    s = float(q.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(k, 1.0 / k)
    q /= s
    return np.clip(q, 0.0, 1000.0)


def _sc(q):
    q = np.asarray(q, dtype=np.float64)
    k = q.size
    if k == 0:
        return float('inf')
    b = np.convolve(q, q)
    v = 2.0 * k * float(b.max())
    return v if np.isfinite(v) else float('inf')


def _buildA(q, k):
    A = np.zeros((2 * k - 1, k), dtype=np.float64)
    for m in range(2 * k - 1):
        lo = max(0, m - k + 1)
        hi = min(k - 1, m)
        A[m, lo:hi + 1] = q[lo:hi + 1]
    return A


def _lp(q, k, tlim=5.0):
    if linprog is None:
        return None
    b = np.convolve(q, q)
    M = float(b.max())
    if not np.isfinite(M) or M <= 0:
        return None
    qs = q + 1e-12 * max(1e-12, float(q.max()))
    A = _buildA(qs, k)
    bu = np.full(2 * k - 1, M)
    c = -np.ones(k) - 1e-10 * q
    bd = [(0.0, 1e4)] * k
    try:
        r = linprog(c, A_ub=A, b_ub=bu, bounds=bd, method='highs',
                    options={'time_limit': tlim, 'presolve': True})
    except Exception:
        try:
            r = linprog(c, A_ub=A, b_ub=bu, bounds=bd, method='simplex')
        except Exception:
            return None
    if r is None or not getattr(r, 'success', False):
        return None
    g = np.asarray(r.x, np.float64)
    g = np.where(np.isfinite(g) & (g > 0), g, 0.0)
    s = g.sum()
    if s <= 0 or not np.isfinite(s):
        return None
    return g / s


def _mix(q, g, scq):
    best_q, best_s = q.copy(), scq
    for t in (0.05, 0.1, 0.15, 0.25, 0.35, 0.5, 0.65, 0.8, 1.0):
        c = _norm9((1 - t) * q + t * g)
        s = _sc(c)
        if s < best_s - 1e-12:
            best_s, best_q = s, c
    return best_q, best_s


def _grad_sub(q, k):
    b = np.convolve(q, q)
    pk = float(b.max())
    if pk <= 0:
        return None, pk
    thr = 0.75 * pk
    act = np.where(b >= thr)[0]
    tn = min(24, act.size)
    top = act[np.argsort(b[act])[::-1]][:tn]
    w = np.exp((b[top] - pk) * 12.0 / max(pk, 1e-15))
    ws = w.sum()
    if ws <= 0 or not np.isfinite(ws):
        return None, pk
    w /= ws
    g = np.zeros(k)
    for kk, ww in zip(top, w):
        m = int(kk)
        lo = max(0, m - k + 1)
        hi = min(k - 1, m)
        seg = q[m - hi:m - lo + 1][::-1]
        g[lo:hi + 1] += 2.0 * float(ww) * seg
    g -= g.mean()
    return g, pk


def _polish(q, k, rng, iters, dl):
    q = _norm9(q)
    best, bs = q.copy(), _sc(q)
    st = 0.15
    for _ in range(max(1, int(iters))):
        if time.monotonic() > dl - 3:
            break
        gi, pk = _grad_sub(q, k)
        if gi is not None:
            gn = float(np.linalg.norm(gi))
            if gn > 1e-14:
                c = _norm9(q - st * gi / gn)
                s = _sc(c)
                if s < bs - 1e-12:
                    bs, best = s, c.copy()
        r = rng.standard_normal(k)
        r -= r.mean()
        rn = float(np.linalg.norm(r))
        if rn > 1e-14:
            c = _norm9(q + 0.5 * st * r / rn)
            s = _sc(c)
            if s < bs - 1e-12:
                bs, best = s, c.copy()
        rs = _sc(best)
        if rs < _sc(q) - 1e-12:
            q = best
            st = min(0.5, st * 1.04)
        else:
            st = max(0.015, st * 0.88)
            if st < 0.016:
                break
    return best, _sc(best)


def _upscale(p, k):
    p = np.asarray(p, dtype=np.float64)
    if p.size == k:
        return _norm9(p)
    if p.size == 0:
        return _norm9(np.ones(k))
    if k % p.size == 0:
        return _norm9(np.repeat(p, k // p.size))
    x = np.linspace(0, 1, p.size)
    y = np.linspace(0, 1, k)
    return _norm9(np.interp(y, x, p))


def _runs_lp(q, k, rng, dl, max_it):
    q = _norm9(q)
    sc = _sc(q)
    stag = 0
    for _ in range(max(1, int(max_it))):
        if time.monotonic() > dl - 6:
            break
        g = _lp(q, k, tlim=3.0)
        if g is None:
            break
        nq, nsc = _mix(q, g, sc)
        if nsc < sc - 1e-10:
            q, sc, stag = nq, nsc, 0
        else:
            stag += 1
            if stag >= 5:
                break
    return q, sc
