import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    return 1 << max(0, (n - 1).bit_length())

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    
    # Get base
    base = None
    try:
        b = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if b.size and np.all(np.isfinite(b)):
            base = np.clip(b, 0.0, 1000.0)
    except Exception:
        pass
    if base is None:
        base = np.ones(6962, dtype=np.float64)
    
    n = len(base)
    sizeC = _next_pow2(2 * n - 1)
    sizeG = _next_pow2(3 * n - 2)
    
    def normalize(a):
        s = a.sum()
        if s < 1e-15:
            return np.ones(n)
        return a * (n / s)
    
    a_init = normalize(base.copy())
    
    def compute_SM(a, temp):
        """Compute soft-max of conv(a,a) and its gradient."""
        fa = np.fft.rfft(a, sizeC)
        C = np.fft.irfft(fa * fa, sizeC)[:2*n-1]
        Cmax = C.max()
        z = np.clip((C - Cmax) * temp, -700.0, 700.0)
        ez = np.exp(z)
        ez_sum = ez.sum()
        w = ez / ez_sum
        M = Cmax + np.log(ez_sum) / temp
        fa2 = np.fft.rfft(a, sizeG)
        fw = np.fft.rfft(w, sizeG)
        gM = np.fft.irfft(np.conj(fa2) * fw, sizeG)[:n]
        return M, gM
    
    def obj_grad(x, temp):
        S = x.sum()
        if S < 1e-15:
            return 0.0, np.zeros(n)
        M, gM = compute_SM(x, temp)
        f = 2.0 * n * M / (S * S)
        g = (2.0 * n / (S * S)) * (gM - 2.0 * M / S)
        return f, g
    
    # Track best
    best_a = a_init.copy()
    best_score = evaluate_sequence([float(x) for x in a_init])
    
    def try_update(cand):
        nonlocal best_a, best_score
        cand = np.clip(cand, 0.0, 1000.0)
        if not np.all(np.isfinite(cand)) or cand.sum() < 0.01:
            return
        cand = normalize(cand)
        s = evaluate_sequence([float(x) for x in cand])
        if np.isfinite(s) and s < best_score:
            best_a = cand.copy()
            best_score = s
    
    def lbfgsb_round(a0, temp, tlimit):
        start = time.monotonic()
        def fun(x):
            if time.monotonic() - start > tlimit:
                return 1e10, np.zeros(n)
            return obj_grad(x, temp)
        try:
            res = minimize(fun, a0, method='L-BFGS-B',
                          bounds=[(0.0, None)] * n,
                          options={'maxiter': 1000, 'maxfun': 5000,
                                   'maxls': 50, 'ftol': 1e-14,
                                   'gtol': 1e-10})
            return np.clip(res.x, 0.0, 1000.0)
        except:
            return a0
    
    def adam_round(a0, temp, tlimit, lr=0.2):
        a = a0.copy()
        m = np.zeros(n); v = np.zeros(n); step = 0
        start = time.monotonic()
        while time.monotonic() - start < tlimit and time.monotonic() < deadline:
            f, g = obj_grad(a, temp)
            step += 1
            m = 0.9*m + 0.1*g
            v = 0.999*v + 0.001*g*g
            mh = m / (1 - 0.9**step)
            vh = v / (1 - 0.999**step)
            a = a - lr * (0.995**step) * mh / (np.sqrt(vh) + 1e-8)
            a = np.clip(a, 0.0, 1000.0)
            a = normalize(a)
            if step % 300 == 0:
                try_update(a)
        try_update(a)
        return a
    
    # Temperature schedule
    schedule = [
        # (temp, lbfgs_time, adam_time)
        (3.0,   40, 30),
        (6.0,   50, 30),
        (12.0,  60, 40),
        (25.0,  70, 50),
        (50.0,  70, 50),
        (100.0, 80, 60),
        (200.0, 90, 60),
        (400.0, 90, 60),
        (800.0, 80, 50),
        (2000.0, 60, 40),
        (5000.0, 40, 30),
        (10000.0, 20, 20),
    ]
    
    for temp, lt, at in schedule:
        if time.monotonic() > deadline - 30:
            break
        remain = deadline - time.monotonic()
        lt_actual = min(lt, remain * 0.4)
        at_actual = min(at, remain * 0.2)
        if lt_actual + at_actual < 10:
            break
        
        # L-BFGS-B
        a_new = lbfgsb_round(best_a.copy(), temp, lt_actual)
        try_update(a_new)
        
        # Adam
        if time.monotonic() < deadline - 30:
            a_new = adam_round(best_a.copy(), temp, at_actual, lr=0.15)
    
    # Final polish
    if time.monotonic() < deadline - 10:
        t_rem = deadline - time.monotonic() - 10
        a = adam_round(best_a.copy(), 10000.0, t_rem, lr=0.03)
    
    # Return
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = normalize(best_a)
    sc = evaluate_sequence([float(x) for x in best_a])
    if not np.isfinite(sc):
        return [1.0] * n
    return [float(x) for x in best_a]
