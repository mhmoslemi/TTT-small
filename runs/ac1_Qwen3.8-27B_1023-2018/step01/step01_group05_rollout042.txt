    for d in [0.1, 0.2, 0.3, 0.4]:
        t = np.ones(n); t[:h] = 1.0 + d; t[h:] = 1.0 - d
        t = np.maximum(t, 0.0)
        if t.sum() > 0.01:
            inits.append(t * n / t.sum())
    for _ in range(5):
        t = 0.2 + 0.8 * rng.random(n)
        inits.append(t * n / t.sum())
    # Multi-frequency
    for f1, a1, f2, a2 in [(1, 0.3, 3, 0.1), (2, 0.2, 5, 0.05), (1, 0.4, 2, 0.2)]:
        t = 1.0 + a1 * np.cos(2 * np.pi * f1 * idx / n) + a2 * np.cos(2 * np.pi * f2 * idx / n)
        t = np.maximum(t, 0.0)
        if t.sum() > 0.01:
            inits.append(t * n / t.sum())

    for t in inits:
        t = np.clip(t, 0, 1000)
        sc = float(evaluate_sequence(t.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = t.copy()
            best_score = sc
            a = best.copy()

    # Main gradient optimization with momentum and adaptive parameters
    beta = 50.0
    K_top = 5
    momentum = np.zeros(n)
    mu = 0.8
    lr = 0.1
    stuck = 0
    it = 0

    phase1_end = start + (deadline - start) * 0.65

    while time.monotonic() < deadline:
        it += 1
        if it % 200 == 0 and time.monotonic() > deadline:
            break

        B = self_conv(a)
        M = B.max()
        S_a = a.sum()
        score = 2.0 * n * M / (S_a * S_a)

        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            stuck = 0
        else:
            stuck += 1

        # Adaptive parameters
        if time.monotonic() < phase1_end:
            frac = min(1.0, it / 6000.0)
            beta = 30.0 + 150.0 * frac
            K_top = min(30, 5 + int(25 * frac))
        else:
            beta = min(10000.0, 180.0 + (it - 6000) * 0.15)
            K_top = min(80, 30 + (it - 6000) // 100)

        # Compute top-K soft-max weights
        if M <= 0:
            break
        Kb = max(1, K_top)
        if Kb < len(B):
            top_idx = np.argpartition(B, -Kb)[-Kb:]
            top_B = B[top_idx]
            w = np.zeros(len(B), dtype=np.float64)
            top_shift = top_B - top_B.max()
            top_shift = np.maximum(top_shift, -50.0 / beta)
            exp_top = np.exp(beta * top_shift)
            w[top_idx] = exp_top / exp_top.sum()
        else:
            shift = B - M
            shift = np.maximum(shift, -50.0 / beta)
            exp_B = np.exp(beta * shift)
            w = exp_B / exp_B.sum()

        # Gradient: grad_j = 2 * sum_k w_k * a_{k-j} - mean
        c = wgrad(w, a)
        grad = 2.0 * c
        grad -= grad.mean()

        # Momentum
        momentum = mu * momentum + grad
        direction = -momentum
        dn = np.linalg.norm(direction)

        if dn < 1e-14:
            km = int(np.argmax(B))
            eg = np.zeros(n)
            lo = max(0, km - (n - 1))
            hi = min(n - 1, km)
            for j in range(lo, hi + 1):
                eg[j] = 2.0 * a[km - j]
            eg -= eg.mean()
            direction = -eg
            dn = np.linalg.norm(direction)
            if dn < 1e-14:
                break
            momentum = np.zeros(n)

        direction /= max(dn, 1e-15)

        # Backtracking line search
        alpha = lr
        accepted = False
        for ls in range(6):
            a_try = a + alpha * direction
            np.maximum(a_try, 0.0, out=a_try)
            st = a_try.sum()
            if st < 0.01:
                alpha *= 0.5
                continue
            a_try *= (n / st)
            B_try = self_conv(a_try)
            M_try = B_try.max()
            if M_try <= M + 1e-10:
                accepted = True
                break
            alpha *= 0.5

        if accepted:
            a = a_try
            lr = min(1.0, lr * 1.03)
            mu = min(0.92, mu + 0.003)
        else:
            # Step rejected
            if stuck > 1500:
                scale = 0.015 + 0.01 * min(3.0, (stuck - 1500) / 1500.0)
                a = a + scale * rng.normal(0, 1, n)
                np.maximum(a, 0.0, out=a)
                a *= (n / max(a.sum(), 0.01))
                momentum *= 0.2
                mu = 0.5
                lr = 0.05
                stuck = 0
            else:
                small = rng.normal(0, 1, n)
                small -= small.mean()
                sn = np.linalg.norm(small)
                if sn > 1e-15:
                    a = a + 0.005 * small / sn
                    np.maximum(a, 0.0, out=a)
                    a *= (n / max(a.sum(), 0.01))
                momentum *= 0.8

            lr = max(0.01, lr * 0.85)

        # Periodic exact evaluation
        if it % 2000 == 0 and time.monotonic() < deadline - 25:
            esc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(esc) and esc < best_score:
                best = a.copy()
                best_score = esc
                stuck = 0

    # Final validation
    fs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fs):
        best = np.ones(n)
        fs = float(evaluate_sequence(best.tolist()))

    return [float(max(0.0, min(1000.0, x))) for x in best]
