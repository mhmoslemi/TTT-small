        return None
    return np.maximum(res.x, 0.0)


def _lp_optimize(a0, stop_time, max_iters=50000):
    a = np.asarray(a0, dtype=np.float64).copy()
    s = float(a.sum())
    if s > 0:
        a = a / s
    else:
        a = np.ones_like(a)
    n = len(a)
    fsize = next_fast_len(2 * n - 1)

    best_a = a.copy()
    best_score = _score_fast(a, fsize)
    no_improve = 0
    rng = np.random.default_rng(31337)

    for it in range(max_iters):
        if time.monotonic() > stop_time - 0.3:
            break

        F = rfft(a, fsize)
        b = np.real(irfft(F * F, fsize)[:2 * n - 1])
        M = float(b.max())
        if M < 1e-18:
            break

        g = _lp_direction(a, M, n)
        if g is None:
            no_improve += 1
            if no_improve > 100:
                break
            a = best_a + rng.normal(0, 0.008, n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = max(0, no_improve - 20)
            continue

        S = float(g.sum())
        if S < 1e-14:
            no_improve += 1
            if no_improve > 100:
                break
            a = best_a + rng.normal(0, 0.008, n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            continue

        g = g / S

        # Line search: try several step sizes
        found_better = False
        for t in [1.0, 0.8, 0.65, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
            a_try = (1.0 - t) * a + t * g
            sc = _score_fast(a_try, fsize)
            if sc < best_score - 1e-11:
                a = a_try
                best_a = a.copy()
                best_score = sc
                no_improve = 0
                found_better = True
                break

        if not found_better:
            no_improve += 1
            if no_improve > 80:
                a = best_a + rng.normal(0, 0.012, n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 10

    return best_a, best_score


def _smooth_init(n, seed, variant=0):
    rng = np.random.default_rng(seed)
    i = np.arange(n, dtype=np.float64)
    if variant == 0:
        a = np.ones(n)
        for _ in range(3):
            freq = float(rng.integers(1, max(2, n // 3)))
            phase = float(rng.uniform(0, 2 * math.pi))
            amp = float(rng.uniform(0.2, 0.6))
            a += amp * np.cos(2 * math.pi * freq * i / n + phase)
        a = np.maximum(a, 0.01)
    elif variant == 1:
        a = np.exp(-0.5 * ((i - n / 2) / (n * 0.3)) ** 2)
        a += 0.1 * np.ones(n)
    elif variant == 2:
        a = 0.5 + 0.4 * np.cos(2 * math.pi * 5 * i / n) + 0.1 * np.cos(2 * math.pi * 13 * i / n)
        a = np.maximum(a, 0.01)
    else:
        a = np.abs(rng.normal(0.5, 0.3, n))
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def _resample_parent(parent, n):
    p = parent
    P = len(p)
    if P == n:
        return p.copy()
    idx = (np.arange(n) * P // n).astype(int)
    # Bilinear interpolation for smoother result
    result = np.zeros(n)
    for k in range(n):
        t = k * (P - 1) / max(1, n - 1) if n > 1 else 0
        k0 = int(np.floor(t))
        k1 = min(k0 + 1, P - 1)
        frac = t - k0
        result[k] = (1 - frac) * p[k0] + frac * p[k1]
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 0:
        result = result / s
    else:
        result = np.ones(n) / n
    return result


def _polish(a0, score, stop_time):
    a = np.asarray(a0, dtype=np.float64).copy()
    s = float(a.sum())
    if s > 0:
        a = a / s
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    best_a = a.copy()
    best_score = float(score)
    rng = np.random.default_rng(424242)
    sigma = 0.08

    while time.monotonic() < stop_time - 0.5:
        r = rng.random()
        if r < 0.35:
            z = np.log(np.maximum(a, 1e-8))
            z += rng.normal(0, sigma, n)
            cand = np.exp(z - z.max())
        elif r < 0.6:
            cand = a.copy()
            width = max(1, min(n, int(rng.integers(1, max(2, n // 20)))))
            start = int(rng.integers(0, max(1, n - width + 1)))
            end = min(n, start + width)
            cand[start:end] *= float(np.exp(rng.normal(0, 0.3)))
        elif r < 0.75:
            cand = 0.5 * (a + a[::-1])
            cand += rng.normal(0, 0.02, n)
        elif r < 0.9:
            cand = (1 - 0.1) * a + 0.1 * (1.0 / n)
        else:
            cand = a.copy()
            idx = int(rng.integers(0, n))
            w = max(1, n // 40)
            s2 = max(0, idx - w // 2)
            e2 = min(n, s2 + w)
            cand[s2:e2] *= float(rng.uniform(0.4, 2.5))

        cand = np.maximum(cand, 0.0)
        cs = float(cand.sum())
        if cs < 0.01:
            continue
        sc = _score_fast(cand, fsize)
        if sc < best_score - 1e-10:
            best_a = cand.copy()
            best_score = sc
            sigma = max(0.03, sigma * 0.9)
        else:
            sigma = min(0.3, sigma * 1.05)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 12.0

    parent = _get_parent()
    parent_score = float(_score(parent))

    best_a = parent.copy()
    best_score = parent_score

    n_parent = len(parent)

    # Phase 1: LP optimization at parent length with parent as start
    alloc1 = min(350.0, budget * 0.30)
    stop1 = t0 + alloc1
    if time.monotonic() < stop1 - 1:
        try:
            a, sc = _lp_optimize(parent, stop1)
            if sc < best_score:
                best_a = a
                best_score = sc
        except Exception:
            pass

    # Phase 2: LP at parent length with different initializations
    stops_remaining = deadline - 60.0
    alloc2_each = min(80.0, (stops_remaining - time.monotonic()) / 4.0)
    for variant in range(4):
        if time.monotonic() > deadline - 50:
            break
        a0 = _smooth_init(n_parent, seed + 100 * variant, variant)
        stop2 = time.monotonic() + max(15.0, alloc2_each)
        try:
            a, sc = _lp_optimize(a0, stop2)
            if sc < best_score:
                best_a = a.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 3: Try other lengths
    other_lengths = [160, 224, 128, 256, 96, 384, 64, 512]
    other_lengths = [L for L in other_lengths if L != n_parent and L >= 32]
    other_lengths.sort(key=lambda L: abs(L - n_parent))

    for n in other_lengths[:4]:
        if time.monotonic() > deadline - 40:
            break
        # Initialize from resampled parent
        a0 = _resample_parent(parent, n)
        alloc3 = min(100.0, max(20.0, (deadline - time.monotonic() - 30.0) * 0.4))
        stop3 = time.monotonic() + alloc3
        try:
            a, sc = _lp_optimize(a0, stop3)
            if sc < best_score:
                best_a = a.copy()
                best_score = sc
        except Exception:
            pass

    # Phase 4: Polish
    if time.monotonic() < deadline - 3:
        try:
            a, sc = _polish(best_a, best_score, deadline - 1.0)
            if sc < best_score:
                best_a = a
                best_score = sc
        except Exception:
            pass

    # Final exact validation
    try:
        fs = float(evaluate_sequence([float(x) for x in best_a]))
        if math.isfinite(fs):
            best_score = fs
        else:
            raise ValueError("invalid")
    except Exception:
        try:
            fs = float(evaluate_sequence([float(x) for x in parent]))
            if math.isfinite(fs):
                best_a = parent.copy()
                best_score = fs
            else:
                best_a = np.ones(32) / 32.0
                best_score = 2.0
        except Exception:
            best_a = np.ones(32) / 32.0
            best_score = 2.0

    # Ensure valid output
    out = np.clip(np.asarray(best_a, dtype=np.float64), 0.0, 1000.0)
    out = out[np.isfinite(out)]
    if out.size == 0:
        out = np.ones(32) / 32.0
    return [float(x) for x in out]
