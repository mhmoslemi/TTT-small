import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = None
    if p is None or p.size == 0 or not np.isfinite(p).all():
        p = np.ones(32, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    return p

def _fft_convolve(a):
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score(arr):
    arr = np.clip(arr.astype(np.float64), 0.0, 1000.0)
    n = len(arr)
    s = arr.sum()
    if s < 0.01:
        return math.inf
    b = _fft_convolve(arr)
    mb = b.max()
    return float(2.0 * n * mb / (s * s))

def _build_A(a):
    """Build the convolution constraint matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            # A[k, lo:hi] = a[k-lo : k-hi+1, -1] i.e. a[k-lo], a[k-lo-1], ..., a[k-hi+1]
            A[k, lo:hi] = a[k - hi + 1 : k - lo + 1][::-1]
    return A

def _lp_iteration(a, t, method='highs'):
    """One LP-based iteration."""
    n = len(a)
    b = _fft_convolve(a)
    M = float(b.max())
    if not np.isfinite(M) or M <= 0:
        return a.copy()
    
    A = _build_A(a)
    
    # max sum(g) s.t. A g <= M, g >= 0
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                  bounds=[(0, None)]*n, method=method,
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
    
    if not res.success:
        # Fallback: use a itself (sum(a) is feasible)
        g = a.copy()
    else:
        g = res.x.copy()
    
    gs = g.sum()
    if gs < 1e-15:
        return a.copy()
    
    # Rescale g to same sum as a
    g = g * (a.sum() / gs)
    
    # Blend
    a_new = (1 - t) * a + t * g
    a_new = np.maximum(a_new, 0.0)
    
    return a_new

    n = len(a)
    a = np.clip(a, 0.0, 1000.0).copy()
    best_a = a.copy()
    best_score = _score(a)
    
    t = 0.15

    while time.monotonic() < total_time:

        s_new = _score(a_new)
        if np.isfinite(s_new) and s_new < best_score - 1e-12:
            best_a = a_new.copy()
            best_score = s_new
            no_improve = 0
            a = a_new.copy()
            # Slightly increase step size

            a = a_new.copy()
            no_improve += 1
        else:

Hmm, this is getting complicated. Let me simplify. For coordinate descent, I'll compute the optimal `a[j]` analytically.

Given all other coordinates fixed, `a[j] = x`, the convolution entries are:
`(a*a)_k = c_k + x * a_{k-j} + a_{k-j} * x = c_k + 2x * a_{k-j}` (for `0 <= k-j < n, k != 2j`)
`(a*a)_k = c_k + x * a_{k-j} + a_{k-j} * x` → same thing.
For `k = 2j`: `(a*a)_k = c_k + x^2` (since both terms give `x * x`).

Wait, actually:
`(a*a)_k = sum_i a_i * a_{k-i}`

The terms involving `x = a_j`:
- `i = j`: `a_j * a_{k-j} = x * a_{k-j}` (if `0 <= k-j < n`)
- `k-i = j`, i.e., `i = k-j`: `a_{k-j} * x` (if `0 <= k-j < n`)

If `j = k-j` (i.e., `k = 2j`), these are the SAME term, giving `x^2`.
Otherwise, they're two different terms, giving `2x * a_{k-j}`.

So:
- For `k ≠ 2j` with `0 <= k-j < n`: `(a*a)_k = c_k + 2x * a_{k-j}`
- For `k = 2j`: `(a*a)_k = c_k + x^2`
- For `k` with `k-j < 0` or `k-j >= n`: `(a*a)_k = c_k`

where `c_k` is the sum of all terms NOT involving `x`.

Now, `max_k (a*a)_k` as a function of `x` is:
`max( max_{k ∈ S1} (c_k + 2x * a_{k-j}), c_{2j} + x^2, max_{k ∈ S2} c_k )`

where S1 = {k : 0 <= k-j < n, k ≠ 2j} and S2 = {k : k-j < 0 or k-j >= n}.

This is the max of (|S1|) affine functions in x, one quadratic in x, and (|S2|) constants.

The minimum of this max over x >= 0 can be found by:
1. If the quadratic term dominates: `c_{2j} + x^2 <= max_k (c_k + 2x*a_{k-j})` for all k. The minimum is where these are equal.
2. Otherwise, the minimum is where two affine terms (or an affine and the quadratic) are equal.

This can be found by checking all pairs of "active" constraints. For the quadratic case: `c_{2j} + x^2 = c_k + 2x*a_{k-j}` → `x^2 - 2a_{k-j}*x + (c_{2j} - c_k) = 0`. For affine-afine: `c_k + 2x*a_{k-j} = c_l + 2x*a_{l-j}` → `x = (c_l - c_k) / (2(a_{k-j} - a_{l-j}))`.

Wait, I need to double-check: is `max(a_new * a_new) <= M` guaranteed? The LP constraint is `(a * g)_k <= M`, not `(g * g)_k <= M`.     try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size < 2 or not np.isfinite(p).all():
            raise ValueError
        return p

        return np.ones(192, dtype=np.float64)


def _convolve(a):
    n = len(a)
    if n < 2:
        return np.array([float(a[0] * a[0])])
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    b = _convolve(a)
    mb = float(b.max())
    return 2.0 * n * mb / (s * s)


def _build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            A[k, lo:hi] = a[k - hi + 1: k - lo + 1][::-1]
    return A


def _lp_step(a, t, lp_options=None):
    n = len(a)
    b = _convolve(a)
    M = float(b.max())
    if M <= 0 or not np.isfinite(M):
        return a.copy()
    
    A = _build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    opts = {'presolve': True, 'time_limit': 5}
    if lp_options:
        opts.update(lp_options)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options=opts)
    
    if not res.success or res.x is None:
        return a.copy()
    
    g = res.x
    gs = float(g.sum())
    if gs < 1e-15:
        return a.copy()
    
    g *= float(a.sum()) / gs
    a_new = (1.0 - t) * a + t * g
    np.maximum(a_new, 0.0, out=a_new)
    return a_new

def _convolve(a):
    n = a.size
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    a = a.copy()
    np.clip(a, 0.0, 1000.0, out=a)
    s = a.sum()
    if s < 0.01:
        return math.inf
    b = _convolve(a)
    return float(2 * a.size * b.max() / (s * s))


def _build_A(a):
    n = a.size
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            A[k, lo:hi] = a[k - hi + 1: k - lo + 1][::-1]
    return A


def _lp_solve(A, M, n):
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub,

                  method='highs',

        return res.x
    return None

    Fg = rfft(g, fsize)

    L = 2 * n - 1

        B = 2.0 * (float(b_ag[k]) - float(b_a[k]))
        C = float(b_a[k]) - M
        
        if A > 1e-12:
            disc = B * B - 4.0 * A * C
            if disc <= 0:
