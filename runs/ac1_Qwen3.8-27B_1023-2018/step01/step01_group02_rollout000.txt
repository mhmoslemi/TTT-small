import numpy as np
import time


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def _score_internal(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n <= 0:
        return float("inf")
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return float("inf")
    p = np.convolve(a, a)
    if p.size == 0:
        return float("inf")
    m = float(np.max(p))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m / (s * s))


def _softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    if z.size == 0:
        return np.ones(1, dtype=np.float64)
    z = z - float(np.max(z))
    e = np.exp(np.clip(z, -745.0, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0 / float(z.size), dtype=np.float64)
    return e / s


def _logit_from_a(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.zeros(1, dtype=np.float64)
    floor = max(1e-12, float(np.max(a)) * 1e-9)
    z = np.log(np.maximum(a, floor))
    z = np.where(np.isfinite(z), z, 0.0)
    return z - float(np.mean(z))


def _upsample(a, m):
    a = np.asarray(a, dtype=np.float64)
    m = int(m)
    if m <= 1:
        return np.ones(m, dtype=np.float64)
    if a.size == m:
        return _normalize(a)
    if a.size == 0:
        return np.ones(m, dtype=np.float64)
    x1 = np.linspace(0.0, 1.0, m, dtype=np.float64)
    x0 = np.linspace(0.0, 1.0, a.size, dtype=np.float64)
    y = np.interp(x1, x0, a)
    return _normalize(y)


_pair_cache = {}


def _pair_idx(n):
    n = int(n)
    if n not in _pair_cache:
        idx = (np.arange(n, dtype=np.intp)[:, None] + np.arange(n, dtype=np.intp)[None, :])
        _pair_cache[n] = idx
    return _pair_cache[n]


def _smooth_logmax_gradient(a, p, beta):
    n = int(a.size)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    pm = float(np.max(p))
    if not np.isfinite(pm):
        idx = int(np.argmax(p))
        w = np.zeros_like(p, dtype=np.float64)
        w[idx] = 1.0
    else:
        z = beta * (p - pm)
        e = np.exp(np.clip(z, -745.0, 0.0))
        s = float(e.sum())
        if not np.isfinite(s) or s <= 0.0:
            w = np.zeros_like(p, dtype=np.float64)
            w[int(np.argmax(p))] = 1.0
        else:
            w = e / s
        w = w + 1e-9 / float(w.size)
        w = w / float(w.sum())

    idx = _pair_idx(n)
    grad_a = 2.0 * (w[idx] @ a)
    grad_a = np.where(np.isfinite(grad_a), grad_a, 0.0)

    dot = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot)
    grad_z = np.where(np.isfinite(grad_z), grad_z, 0.0)
    return grad_z


def _adam_state(n):
    return {
        "m": np.zeros(n, dtype=np.float64),
        "v": np.zeros(n, dtype=np.float64),
        "b1": 1e-8,
        "b2": 1e-16,
        "t": 0,
    }


def _adam_step(z, grad, state, lr, cap=0.8):
    g = np.asarray(grad, dtype=np.float64)
    if g.size == z.size and np.all(np.isfinite(g)):
        pass
    else:
        g = np.zeros_like(z, dtype=np.float64)

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    state["t"] += 1
    state["b1"] += 1.0 - b1
    state["b2"] += 1.0 - b2

    state["m"] = b1 * state["m"] + (1.0 - b1) * g
    state["v"] = b2 * state["v"] + (1.0 - b2) * (g * g)

    mhat = state["m"] / state["b1"]
    vhat = state["v"] / state["b2"]
    step = float(lr) * mhat / (np.sqrt(vhat) + eps)
    step = np.where(np.isfinite(step), step, 0.0)
    if cap is not None:
        step = np.clip(step, -float(cap), float(cap))

    z = z - step
    z = np.where(np.isfinite(z), z, 0.0)
    return z

            if hi > lo:
                local = np.linspace(lo, hi, 10)
                scs2 = _evaluate_xs(local, base, lin, sq_idx, S0, n)
                if scs2.size:
                    jx = int(np.argmin(scs2))
                    if np.isfinite(scs2[jx]) and scs2[jx] < best_scand:
                        best_x = float(local[jx])
                        best_scand = float(scs2[jx])

            if best_scand + 1e-15 < sc:
                a[j] = best_x
                S = S0 + best_x
                p = base + best_x * lin
                p[sq_idx] += best_x * best_x
                sc = best_scand
                improved = True

            if 1000 * int(j) % (1000 * n) == 0:
                if S > 10.0 or S < 0.1:
                    a = a / S
                    p = p / (S * S)
                    S = 1.0
                    sc = 2.0 * n * float(np.max(p))
                if time.monotonic() >= time_end:
                    break

        passes += 1
        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if time.monotonic() >= time_end:
            break

    a = _normalize(a)
    sc = _score_internal(a)
    return a, sc


def _random_inits(n, parent, rng, limit=14):
    inits = []

    def add(a):
        try:
            a = _normalize(a)
            if a.size == n and np.all(np.isfinite(a)):
                inits.append(a)
        except Exception:
            pass

    if parent is not None and parent.size == n:
        add(parent)
        add(parent[::-1])
        for sigma in (0.03, 0.08, 0.16):
            add(parent * (1.0 + sigma * rng.standard_normal(n)))
    else:
        parent = None

    add(np.ones(n, dtype=np.float64) / float(n))

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for pw in (0.3, 0.6, 1.0, 2.0, 3.0, 4.0, -0.5, -1.0):
        if pw >= 0.0:
            v = 1.0 + 0.5 * (x ** pw)
        else:
            v = 1.0 + 0.5 / ((x + 0.05) ** (-pw))
        add(v)

    for d in (0.5, 1.0, 2.0, 4.0, 10.0):
        add(np.exp(-d * x) + 1e-3)
        add(np.exp(-d * (1.0 - x)) + 1e-3)
        add(np.exp(-4.0 * d * (x - 0.5) ** 2) + 1e-3)

    for p in (1, 2, 4):
        add(1.0 + 0.8 * (np.cos(np.pi * x) ** p) + 1e-3)

    for _ in range(6):
        z = np.zeros(n, dtype=np.float64)
        for f in (1, 2, 3):
            z += rng.uniform(0.0, 1.5) * np.cos(2.0 * np.pi * f * x + rng.uniform(0.0, 2.0 * np.pi))
        z += 0.08 * rng.standard_normal(n)
        add(np.exp(z))

    for _ in range(3):
        add(np.exp(2.0 * rng.standard_normal(n)))

    if parent is not None:
        for _ in range(2):
            add(parent + 0.05 * np.exp(z))

    if len(inits) > limit:
        inits = inits[:limit]
    if not inits:
        inits.append(np.ones(n, dtype=np.float64) / float(n))
    return inits


def _adam_run(a0, best, archive, deadline, rng, max_time):
    if max_time <= 2.0:
        return
    n = int(a0.size)
    z = _logit_from_a(a0)
    state = _adam_state(n)
    lr = 0.32
    cap = 0.8
    patience = 420
    no_improve = 0
    start_t = time.monotonic()
    end_t = min(deadline - 5.0, start_t + max_time)
    span = max(1.0, end_t - start_t)
    last_polish = start_t

    while time.monotonic() < end_t:
        a = _softmax_from_z(z)
        p = np.convolve(a, a)
        S = float(a.sum())
        pm = float(np.max(p))
        sc = float(2.0 * n * pm / (S * S))

        if np.isfinite(sc) and sc < best["s"] - 1e-12:
            best["s"] = sc
            best["a"] = a.copy()
            no_improve = 0
            lr = min(1.0, lr * 1.02)
            if len(archive) < 24:
                archive.append(a.copy())

        else:
            no_improve += 1

        frac = min(1.0, max(0.0, (time.monotonic() - start_t) / span))
        beta = float(n) * (8.0 + 300.0 * frac)
        g = _smooth_logmax_gradient(a, p, beta)
        z = _adam_step(z, g, state, lr, cap)

        if not np.all(np.isfinite(z)):
            z = _logit_from_a(best["a"] + 1e-12) + 0.10 * rng.standard_normal(n)
            state = _adam_state(n)
            no_improve = 0

        now = time.monotonic()
        if now - last_polish > 8.0 and deadline - now > 25.0:
            pol_end = min(deadline - 10.0, now + min(5.0, 0.10 * max(0.0, end_t - now) + 1.0))
            try:
                pa, ps = _coordinate_polish(best["a"], pol_end, rng)
                if np.isfinite(ps) and ps < best["s"] - 1e-12:
                    best["s"] = ps
                    best["a"] = pa
                    if len(archive) < 24:
                        archive.append(pa.copy())
                    z = _logit_from_a(pa) + 0.04 * rng.standard_normal(n)
                    state = _adam_state(n)
                    no_improve = 0
            except Exception:
                pass
            last_polish = time.monotonic()

        if no_improve > patience:
            z = _logit_from_a(best["a"] + 1e-12) + (0.08 + 0.35 * float(rng.random())) * rng.standard_normal(n)
            state = _adam_state(n)
            lr = max(0.08, lr * 0.78)
            no_improve = 0
            patience = max(220, int(patience * 1.25))

    return


def _lp_restart(best, deadline, rng):
    if time.monotonic() > deadline - 90.0:
        return
    a = np.asarray(best["a"], dtype=np.float64)
    n = int(a.size)
    if n < 2 or n > 260:
        return
    try:
        from scipy.optimize import linprog
    except Exception:
        return

    try:
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = a

        pmax = float(np.max(np.convolve(a, a)))
        if not np.isfinite(pmax) or pmax <= 0.0:
            return

        for fac in (1.0, 0.95, 0.85, 0.70, 0.50):
            tau = pmax * fac
            if tau <= 0.0:
                continue
            if time.monotonic() > deadline - 40.0:
                break
            try:
                res = linprog(
                    -np.ones(n, dtype=np.float64),
                    A_ub=A,
                    b_ub=np.full(2 * n - 1, tau, dtype=np.float64),
                    bounds=[(0.0, None)] * n,
                    method="highs",
                    options={"time_limit": 5.0},
                )
            except Exception:
                break
            if not getattr(res, "success", False):
                continue
            b = _normalize(res.x)
            ns = _score_internal(b)
            if np.isfinite(ns) and ns < best["s"] - 1e-12:
                best["s"] = ns
                best["a"] = b.copy()

            for t in (0.10, 0.30, 0.60):
                c = (1.0 - t) * best["a"] + t * b
                ns = _score_internal(c)
                if np.isfinite(ns) and ns < best["s"] - 1e-12:
                    best["s"] = ns
                    best["a"] = _normalize(c)
    except Exception:
        return


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    start_t = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1300.0
    deadline = start_t + max(1.0, budget - 25.0)

    parent = None
    n = 127
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            q = np.maximum(p, 0.0)
            s = float(q.sum())
            if s > 0.0:
                parent = q / s
                n = int(parent.size)
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(n, dtype=np.float64) / float(n)

    best = {
        "a": _normalize(parent).copy(),
        "s": _score_internal(parent),
    }
    archive = [best["a"].copy()]

    def try_update(a):
        try:
            a = _normalize(a)
            sc = _score_internal(a)
            if np.isfinite(sc) and (not np.isfinite(best["s"]) or sc < best["s"] - 1e-12):
                best["s"] = sc
                best["a"] = a.copy()
                if len(archive) < 32:
                    archive.append(a.copy())
        except Exception:
            pass

    # Quick coordinate polish of diverse small initializations around parent length.
    inits = _random_inits(n, parent, rng, limit=12)
    quick_budget = min(90.0, max(0.0, deadline - start_t - 220.0))
    if quick_budget > 5.0:
        slot = max(1.0, quick_budget / max(1.0, len(inits)))
        for a0 in inits:
            if time.monotonic() >= deadline - 200.0:
                break
            try:
                a1, _ = _coordinate_polish(a0, time.monotonic() + slot, rng)
                try_update(a1)
            except Exception:
                pass

    # Main smoothed log-sum-exp optimization from a few strong starts.
    starts = [best["a"].copy(), parent.copy()]
    if len(inits) > 1:
        starts.append(inits[1].copy())
    adds = [
        best["a"] * (1.0 + 0.10 * rng.standard_normal(best["a"].size)),
        _softmax_from_z(2.0 * rng.standard_normal(best["a"].size)),
    ]
    starts.extend(adds)

    adam_deadline = deadline - 160.0
    while time.monotonic() < adam_deadline:
        rem = max(0.0, adam_deadline - time.monotonic())
        if rem < 35.0:
            break
        a0 = starts.pop(0) if starts else best["a"]
        try:
            _adam_run(_normalize(a0), best, archive, deadline, rng, min(220.0, rem))
        except Exception:
            pass
        if time.monotonic() >= deadline - 160.0:
            break

    # Optional LP/polar restart if enough time remains.
    try:
        _lp_restart(best, deadline, rng)
    except Exception:
        pass

    # Explore a few neighboring lengths by interpolation and coordinate polishing.
    if time.monotonic() < deadline - 180.0:
        length_set = {63, 95, 127, 159, 191, 255, 383}
        if 2 * n <= 511:
            length_set.add(2 * n)
        lengths = [L for L in length_set if 2 <= L <= 511 and L != n]
        lengths.sort(key=lambda L: (abs(L - n), -L))
        attempts = 0
        total_len_budget = min(240.0, max(0.0, deadline - time.monotonic() - 150.0))
        for L in lengths:
            attempts += 1
            if attempts > 4:
                break
            if total_len_budget <= 0.0:
                break
            slot = min(60.0, total_len_budget / max(1.0, 5 - attempts))
            if slot < 15.0:
                break
            if time.monotonic() >= deadline - 130.0:
                break
            try:
                initL = _upsample(best["a"], L)
                aL, _ = _coordinate_polish(initL, time.monotonic() + slot, rng)
                try_update(aL)
                total_len_budget -= (time.monotonic() - (deadline - total_len_budget - slot))
            except Exception:
                pass

    # Final exact single-coordinate polish on the current best candidate.
    final_end = max(time.monotonic() + 1.0, deadline - 12.0)
    try:
        fa, fs = _coordinate_polish(best["a"], final_end, rng)
        try_update(fa)
    except Exception:
        pass

    # Validate and return the best candidate under the authoritative evaluator.
    candidates = []
    seen = set()

    def add_final(a):
        try:
            a = np.asarray(a, dtype=np.float64)
            if a.size == 0:
                return
            key = (int(a.size), float(np.sum(a)))
            if key in seen:
                return
            seen.add(key)
            candidates.append(_normalize(a))
        except Exception:
            pass

    add_final(best["a"])
    add_final(parent)
    for a in archive:
        add_final(a)
        if len(candidates) >= 24:
            break
    add_final(np.ones(best["a"].size, dtype=np.float64) / float(best["a"].size))
    add_final([1.0, 1.0])

    final_list = None
    final_score = float("inf")
    try:
        exact_eval = evaluate_sequence
    except Exception:
        exact_eval = None

    for a in candidates:
        try:
            lst = [float(min(1000.0, max(0.0, float(x)))) for x in a if np.isfinite(x)]
            if not lst:
                continue
            if exact_eval is not None:
                sc = float(exact_eval(lst))
            else:
                sc = _score_internal(np.asarray(lst, dtype=np.float64))
            if np.isfinite(sc) and sc < final_score:
                final_score = sc
                final_list = lst
        except Exception:
            continue

    if final_list is None:
        lst = [float(min(1000.0, max(0.0, float(x)))) for x in best["a"] if np.isfinite(x)]
        if not lst:
            lst = [1.0]
        final_list = lst

    return final_list
