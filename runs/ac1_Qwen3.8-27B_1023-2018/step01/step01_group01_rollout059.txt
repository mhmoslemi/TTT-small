import numpy as np
import time
from scipy.optimize import minimize


def _p2(l):
    return 1 << max(0, (int(l) - 1).bit_length())


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(1.0, float(budget_s))
    deadline = t0 + budget - 15.0
    rng = np.random.default_rng(seed)

    # ---- base sequence ------------------------------------------------
    try:
        a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64).ravel(), 0.0, 1000.0)
    except Exception:
        a0 = np.ones(6962, dtype=np.float64)
    n = int(a0.size)
    sizeC = _p2(2 * n - 1)
    sizeG = _p2(3 * n - 2)

    def safe_eval(a):
        try:
            s = float(evaluate_sequence([float(x) for x in a]))
            return s if np.isfinite(s) else np.inf
        except Exception:
            return np.inf

    def normalize(a):
        s = float(a.sum())
        if s < 1e-12:
            return np.ones(n, dtype=np.float64)
        return np.clip(a * (float(n) / s), 0.0, 1000.0)

    def conv_full(a):
        fa = np.fft.rfft(a, sizeC)
        return np.fft.irfft(fa * fa, sizeC)[:2 * n - 1]

    def smooth_max_grad(a, T):
        C = conv_full(a)
        TC = T * C
        m = TC.max()
        e = np.exp(TC - m)
        ss = e.sum()
        w = e / ss
        M = (m + np.log(ss)) / T
        wr = w[::-1].copy()
        fa = np.fft.rfft(a, sizeG)
        fw = np.fft.rfft(wr, sizeG)
        c2 = np.fft.irfft(fa * fw, sizeG)[:3 * n - 2]
        dM = 2.0 * c2[n - 1:2 * n - 1][::-1]
        return M, dM

    def obj_grad(a, T):
        S = float(a.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        M, dM = smooth_max_grad(a, T)
        val = 2.0 * n * M / (S * S)
        g = (2.0 * n / (S * S)) * dM - (4.0 * n * M) / (S * S * S)
        return val, g

    best_a = a0.copy()
    best_score = safe_eval(best_a)

    # ---- Phase 1: Adam with multiple seeds ----------------------------
    def run_adam(a_init, seconds, lr0, T0, T1):
        nonlocal best_a, best_score
        a = normalize(a_init.copy())
        mu = np.zeros(n)
        nu = np.zeros(n)
        t = 0
        sc = safe_eval(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc
        start = time.monotonic()
        last_ev = start
        while time.monotonic() < deadline - 25 and (time.monotonic() - start) < seconds:
            frac = min(1.0, (time.monotonic() - start) / max(seconds, 0.1))
            T = T0 + (T1 - T0) * frac
            _, g = obj_grad(a, T)
            t += 1
            mu = 0.9 * mu + 0.1 * g
            nu = 0.999 * nu + 0.001 * (g * g)
            mu_h = mu / (1.0 - 0.9 ** t)
            nu_h = nu / (1.0 - 0.999 ** t)
            lr = lr0 * (0.92 ** (t / 2000.0))
            a = np.clip(a - lr * mu_h / (np.sqrt(nu_h) + 1e-8), 0.0, 1000.0)
            a = normalize(a)
            if t % 500 == 0:
                now = time.monotonic()
                if now - last_ev > 0.25:
                    last_ev = now
                    sc = safe_eval(a)
                    if sc < best_score:
                        best_a, best_score = a.copy(), sc
        sc = safe_eval(a)
        if sc < best_score:

        return a

    scored.sort(key=lambda t: t[0])
    scored = scored[:5]

    avail1 = max(30.0, deadline - time.monotonic() - 180.0)
    per1 = avail1 / max(1, len(scored))
    for i, (_, s) in enumerate(scored):
        if time.monotonic() > deadline - 60:

        run_adam(s, per1, 0.014, 5.0, 30.0)

    # ---- Phase 2: L-BFGS-B polishing ----------------------------------

        T_schedule = [10.0, 18.0, 30.0, 50.0]

        final = a0
    out = [float(v) for v in final]
    if not out or any((not np.isfinite(v)) for v in out):
        out = [1.0] * n
    return out
