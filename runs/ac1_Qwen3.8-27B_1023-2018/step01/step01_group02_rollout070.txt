def _clean_a(a):
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-15:

    arr = arr / s
    if not np.all(np.isfinite(arr)):
        return None
    return arr


def _score_array(a):
    arr = _clean_a(a)
    if arr is None:
        return float("inf")
    n = arr.size
    c = np.convolve(arr, arr)
    if c.size == 0:
        return float("inf")
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx <= 0.0:
        return float("inf")
    return float(2.0 * n * mx)


def _z_from_a(a):
    arr = _clean_a(a)
    if arr is None:
        return np.zeros(1, dtype=np.float64)
    n = arr.size
    z = np.log(np.maximum(arr, 1e-12))
    z -= float(np.mean(z))
    return np.where(np.isfinite(z), z, 0.0)


def _resample(a, m):
    arr = _clean_a(a)
    m = int(m)
    if m <= 0:
        return None
    if arr is None:
        arr = np.ones(1, dtype=np.float64)
    if arr.size == m:
        return arr.copy()
    if arr.size == 1:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    x0 = np.linspace(0.0, 1.0, arr.size)
    x1 = np.linspace(0.0, 1.0, m)
    out = np.interp(x1, x0, arr)
    return _clean_a(out)

    if a is None:
        return None, float("inf")
    n = a.size
    best = a.copy()
    best_sc = _score_array(a)
    eval_budget = max(2, int(eval_budget))
    half = max(1, eval_budget // 2)

    for _ in range(half):
        if t_stop is not None and time.monotonic() > t_stop:
            break
        x = best.copy()
        if n > 6 and rng.random() < 0.35:
            off = int(rng.integers(-max(1, n // 25), max(1, n // 25) + 1))
            x = 0.5 * (x + np.roll(x, off))
        nb = 1 + (1 if rng.random() < 0.5 else 0)
        for _i in range(nb):
            if n > 2:
                maxw = max(2, min(n, n // 6))
                w = int(rng.integers(2, maxw + 1))
                w = max(1, min(w, n))
                st = int(rng.integers(0, n - w + 1))
                x[st:st + w] *= float(np.exp(rng.normal(0.0, 0.25)))
        x = _clean_a(x)
        if x is None:
            continue
        sc = _score_array(x)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best = x
            best_sc = sc

    for _ in range(eval_budget - half):
        if t_stop is not None and time.monotonic() > t_stop:
            break
        x = best.copy()
        j = int(rng.integers(0, n))
        x[j] *= float(np.exp(rng.normal(0.0, 0.20)))
        x = _clean_a(x)
        if x is None:
            continue
        sc = _score_array(x)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best = x
            best_sc = sc

    return best, float(best_sc)


def _lp_update(a):
    a = _clean_a(a)
    if a is None or _so is None:
        return a
    n = a.size
    if n <= 1:
        return a
    span = 2 * n - 1
    c = np.convolve(a, a)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return a

    C = np.zeros((span, n), dtype=np.float64)
    for t in range(span):
        lo = max(0, t - n + 1)
        hi = min(n - 1, t)
        if hi >= lo:
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            C[t, idx] = a[t - idx]

    try:
        res = _so.linprog(
            -np.ones(n),
            A_ub=C,
            b_ub=np.full(span, M, dtype=np.float64),
            bounds=[(0.0, None)] * n,
            method="highs",
        )
    except Exception:
        return a

    if not getattr(res, "success", False):
        return a

    g = np.asarray(res.x, dtype=np.float64).ravel()
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    sg = float(g.sum())
    if not np.isfinite(sg) or sg <= 1e-12:
        return a
    g /= sg

    best = best0 = a.copy()
    best_sc = _score_array(a)
    for mix in (0.02, 0.05, 0.10, 0.20, 0.35, 0.50, 0.70):
        x = _clean_a((1.0 - mix) * a + mix * g)
        if x is None:
            continue
        sc = _score_array(x)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best = x
            best_sc = sc
    best0 = best
    return best0


def _make_inits(n, parent, rng):
    n = int(n)
    if n <= 0:
        return [np.ones(1, dtype=np.float64)]
    parent = _clean_a(parent)
    if parent is None or parent.size != n:
        parent = _resample(parent if parent is not None else np.ones(1), n)
        if parent is None:
            parent = np.ones(n, dtype=np.float64) / float(n)

    out = []

    def add(x):
        y = _resample(x, n)
        if y is not None:
            out.append(y)

    add(parent)
    add(parent[::-1])
    add(parent * (1.0 + 0.05 * rng.standard_normal(n)))
    add(parent * (1.0 + 0.15 * rng.standard_normal(n)))
    add(np.ones(n, dtype=np.float64))
    add(rng.exponential(1.0, size=n) + 1e-3)
    add(np.exp(0.6 * rng.standard_normal(n)))

    if n > 3:
        x = np.arange(n, dtype=np.float64) / float(n)
        for g in (0.5, 1.0, 2.0, 4.0, 8.0):
            add(np.exp(-g * x))
            add(np.exp(-g * (1.0 - x)))
        for mu in (0.15, 0.35, 0.50, 0.65, 0.85):
            for sig in (0.08, 0.16, 0.30):
                add(np.exp(-((x - mu) ** 2) / (2.0 * sig * sig)))
        for p, q in ((0.3, 1.0), (1.0, 0.3), (0.5, 2.0), (2.0, 0.5)):
            add((x ** p) + ((1.0 - x) ** q) + 0.05)
        add(np.sin(np.pi * x) ** 2 + 0.10)

    scored = []
    for a in out:
        sc = _score_array(a)
        if np.isfinite(sc):
            scored.append((sc, a))
    if not scored:
        return [parent.copy()]
    scored.sort(key=lambda z: z[0])
    uniq = []
    for sc, a in scored:
        dup = False
        for _, b in uniq:
            if b.size == a.size and float(np.linalg.norm(a - b)) < 1e-3:
                dup = True
                break
        if not dup:
            uniq.append((sc, a))
        if len(uniq) >= 12:
            break
    return [a for _, a in uniq]


def _adam_core(z0, a0, start_t, end_t, rng):
    a0 = _clean_a(a0)
    if a0 is None:
        a0 = np.ones(1, dtype=np.float64)
    n = a0.size
    if n <= 1:
        return a0.copy(), _score_array(a0)

    span = 2 * n - 1
    L = 1 << max(1, (3 * n - 3).bit_length())

    if z0 is not None:
        ztmp = np.asarray(z0, dtype=np.float64).ravel()
        if ztmp.size == n and np.any(np.isfinite(ztmp)):
            z = np.where(np.isfinite(ztmp), ztmp, 0.0)
            z -= float(np.mean(z))
        else:
            z = _z_from_a(a0)
    else:
        z = _z_from_a(a0)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    bcum1 = 1.0
    bcum2 = 1.0
    b1 = 0.90
    b2 = 0.999
    eps = 1e-9

    best_a = a0.copy()
    best_sc = _score_array(a0)
    best_z = z.copy()
    lr = 0.38
    no_imp = 0
    it = 0
    total = max(1e-3, end_t - start_t)
    next_polish = start_t + 20.0
    tau = _TARGET_BOUND / (2.0 * float(n))

    while True:
        now = time.monotonic()
        if now >= end_t - 0.2:
            break
        frac = min(1.0, max(0.0, (now - start_t) / total))

        zc = z - float(np.mean(z))
        zmax = float(np.max(zc)) if n else 0.0
        if not np.isfinite(zmax):
            zc = np.zeros(n, dtype=np.float64)
            zmax = 0.0
        e = np.exp(zc - zmax)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 0.0:
            a = np.ones(n, dtype=np.float64) / float(n)
        else:
            a = e / s

        A = np.fft.rfft(a, L)
        c = np.fft.irfft(A * A, L)[:span]
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        if c.size == 0:
            break
        cmax = float(np.max(c))
        if not np.isfinite(cmax) or cmax <= 0.0:
            break

        hard = float(2.0 * n * cmax)
        if np.isfinite(hard) and hard < best_sc - 1e-11:
            best_sc = hard
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
            lr = min(0.55, lr * 1.02)
        else:
            no_imp += 1

        beta = 25.0 + 65000.0 * (frac ** 2)
        arg = -beta * (c - cmax)
        arg = np.clip(arg, -745.0, 0.0)
        p = np.exp(arg)
        ps = float(p.sum())
        imax = int(np.argmax(c))
        if not np.isfinite(ps) or ps <= 1e-300:
            p = np.zeros(span, dtype=np.float64)
            p[imax] = 1.0
        else:
            p /= ps

        q = p.copy()
        reg = 0.55 * (1.0 - frac) * (1.0 - frac)
        if reg > 0.0:
            q += reg * c / (float(span) * cmax)

        thr = 0.90 * (1.0 - 0.35 * frac)
        if tau > 0.0 and thr > 0.0:
            over = np.maximum(c - tau, 0.0)
            if float(np.max(over)) > 1e-15:
                q += thr * over / (float(span) * tau)

        q = np.where(np.isfinite(q), q, 0.0)
        if float(q.sum()) <= 0.0:
            q = p

        P = np.fft.rfft(q[::-1], L)
        corr = np.fft.irfft(P * A, L)
        h = corr[n - 1:2 * n - 1][::-1]
        h = np.where(np.isfinite(h), h, 0.0)
        h = np.maximum(h, 0.0)

        grad_a = 4.0 * float(n) * h
        if not np.all(np.isfinite(grad_a)):
            grad_a = np.zeros(n, dtype=np.float64)
        dot = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - dot)
        grad_z = np.where(np.isfinite(grad_z), grad_z, 0.0)
        grad_z -= float(np.mean(grad_z))

        gmax = float(np.max(np.abs(grad_z))) if n else 0.0
        if no_imp > 80 and gmax < 1e-14:
            z = best_z + 0.08 * rng.standard_normal(n)
            z = np.where(np.isfinite(z), best_z, z)
            z -= float(np.mean(z))
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            bcum1 = 1.0
            bcum2 = 1.0
            lr = 0.28
            no_imp = 0
            it += 1
            continue

        t += 1
        bcum1 += 1.0 - b1
        bcum2 += 1.0 - b2
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / bcum1
        vh = v / bcum2
        step = lr * mh / (np.sqrt(vh) + eps)
        step = np.where(np.isfinite(step), step, 0.0)
        cap = 0.70 * (1.0 - 0.45 * frac) + 0.07
        step = np.clip(step, -cap, cap)
        z = z - step

        if not np.all(np.isfinite(z)):
            z = best_z + 0.02 * rng.standard_normal(n)
            z = np.where(np.isfinite(z), best_z, z)
            z -= float(np.mean(z))
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            bcum1 = 1.0
            bcum2 = 1.0
            it += 1
            continue

        z -= float(np.mean(z))

        if no_imp > 900 + 3000 * frac:
            z = best_z + (0.03 + 0.25 * frac) * rng.standard_normal(n)
            z = np.where(np.isfinite(z), best_z, z)
            z -= float(np.mean(z))
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            bcum1 = 1.0
            bcum2 = 1.0
            lr = max(0.07, lr * 0.68)
            no_imp = 0

        it += 1
        if now >= next_polish:
            pol_a, pol_sc = _local_polish(best_a, rng, 55, None)
            if pol_a is not None and np.isfinite(pol_sc) and pol_sc < best_sc - 1e-11:
                best_a = pol_a
                best_sc = pol_sc
                z = _z_from_a(best_a)
                m[:] = 0.0
                v[:] = 0.0
                t = 0
                bcum1 = 1.0
                bcum2 = 1.0
                lr = 0.34
                no_imp = 0
            next_polish = now + max(18.0, 0.15 * total)

    return _clean_a(best_a), float(best_sc)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    T = max(1.0, float(budget_s))
    safety = min(20.0, 0.20 * T)
    deadline = start + max(1.0, T - safety)

    parent = None
    try:
        parent = _clean_a(height_sequence_1)
    except Exception:
        parent = None
    if parent is None:
        parent = np.ones(127, dtype=np.float64) / 127.0

    n = int(parent.size)
    best_a = parent.copy()
    best_score = _score_array(parent)
    pool = [parent.copy()]

    def consider(a):
        nonlocal best_a, best_score
        a = _clean_a(a)
        if a is None:
            return
        sc = _score_array(a)
        if not np.isfinite(sc):
            return
        if sc < best_score - 1e-12:
            best_a = a.copy()
            best_score = sc
        if sc < best_score + 0.05 and len(pool) < 90:
            dup = False
            for b in pool[-40:]:
                if b.size == a.size and float(np.linalg.norm(a - b)) < 1e-3:
                    dup = True
                    break
            if not dup:
                pool.append(a.copy())

    try:
        inits = _make_inits(n, parent, rng)
    except Exception:
        inits = [parent.copy(), parent[::-1].copy(),
                 parent * (1.0 + 0.12 * rng.standard_normal(n)),
                 np.ones(n, dtype=np.float64)]
    for a in inits:
        consider(a)

    try:
        scored = []
        for a in inits:
            a = _clean_a(a)
            if a is None:
                continue
            sc = _score_array(a)
            if np.isfinite(sc):
                scored.append((sc, a))
        if scored:
            scored.sort(key=lambda z: z[0])
            remaining = max(1, int((deadline - time.monotonic()) -
                                   max(60.0, 0.15 * max(1.0, deadline - time.monotonic()))))
            if remaining <= 0:
                remaining = 0
            if remaining > 900:
                max_starts = 6
            elif remaining > 600:
                max_starts = 5
            elif remaining > 300:
                max_starts = 4
            elif remaining > 100:
                max_starts = 3
            else:
                max_starts = 1

            chosen = []
            take_best = max(2, (max_starts + 1) // 2)
            for _, a in scored[:take_best]:
                chosen.append(a)
            rest = [a for _, a in scored[take_best:]]
            rng.shuffle(rest)
            chosen.extend(rest)
            chosen = chosen[:max_starts]
            if not chosen:
                chosen = [parent.copy()]

            now = time.monotonic()
            reserve = max(70.0, 0.16 * max(1.0, deadline - now))
            available = max(0.0, deadline - now - reserve)
            time_per = available / float(max(1, len(chosen)))

            for idx, a0 in enumerate(chosen):
                if time.monotonic() >= deadline - reserve:
                    break
                st = time.monotonic()
                end = min(st + time_per, deadline - reserve)
                if end <= st + 5.0:
                    break
                try:
                    a1, _ = _adam_core(_z_from_a(a0), a0, st, end, rng)
                except Exception:
                    a1 = a0
                consider(a1)

                a2, _ = _local_polish(a1, rng, 70, None)
                consider(a2)

                if time.monotonic() < deadline - max(25.0, 0.5 * reserve):
                    a3 = _lp_update(a2)
                    consider(a3)
                    if time.monotonic() < deadline - max(25.0, 0.5 * reserve):
                        st2 = time.monotonic()
                        end2 = min(st2 + max(10.0, 0.22 * time_per), deadline - reserve)
                        if end2 > st2 + 5.0:
                            try:
                                a4, _ = _adam_core(_z_from_a(a3), a3, st2, end2, rng)
                            except Exception:
                                a4 = a3
                            consider(a4)
    except Exception:
        pass

    try:
        pool_snapshot = list(pool)
        pool_snapshot.sort(key=lambda x: _score_array(x))
        lp_calls = 0
        for a0 in pool_snapshot[:12]:
            if lp_calls >= 10 or time.monotonic() >= deadline - 35.0:
                break
            a1 = _lp_update(a0)
            consider(a1)
            lp_calls += 1
            a2, _ = _local_polish(a1, rng, 45, None)
            consider(a2)
    except Exception:
        pass

    try:
        rem = max(0.0, deadline - time.monotonic())
        if rem > 45.0:
            st = time.monotonic()
            end = min(deadline - 12.0, st + max(20.0, rem - 22.0))
            if end > st + 6.0:
                a1, _ = _adam_core(_z_from_a(best_a), best_a, st, end, rng)
                consider(a1)
                a2, _ = _local_polish(a1, rng, 120, None)
                consider(a2)
                if time.monotonic() < deadline - 12.0:
                    a3 = _lp_update(a2)
                    consider(a3)
                a4, _ = _local_polish(a3, rng, 90, None)
                consider(a4)
    except Exception:
        pass

    candidates = list(pool)
    candidates.append(best_a)
    candidates.append(parent.copy())
    candidates.append(np.ones(1, dtype=np.float64))

    final_list = None
    final_score = float("inf")
    for cand in candidates:
        try:
            arr = _clean_a(cand)
            if arr is None:
                continue
            lst = [float(x) for x in arr]
            if not lst:
                continue
            if any(not np.isfinite(x) for x in lst):
                continue
            try:
                sc = float(evaluate_sequence(lst))
            except Exception:
                sc = _score_array(arr)
            if np.isfinite(sc) and sc < final_score:
                final_score = sc
                final_list = lst
        except Exception:
            pass

    if final_list is None:
        final_list = [float(x) for x in np.ones(1, dtype=np.float64)]

    return final_list
