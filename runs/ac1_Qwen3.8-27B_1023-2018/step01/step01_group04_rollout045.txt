    """Compute gradient of 2n*sum_k w_k * b_k w.r.t. a, where b = a*a.
    w should sum to 1 (softmax weights)."""
    n = len(a)
    # u = time-reverse of w
    u = w[::-1]
    # u_conv_a = convolve(u, a), length 2n-1
    u_conv_a = np.convolve(u, a)  # length 2n-1
    # First term: (u*a)_{2n-2-i} for i=0..n-1
    first = u_conv_a[2*n-2-n+1:2*n-2+1]  # indices n-1, n, ..., 2n-2
    first = first[::-1]  # now first[i] = (u*a)_{2n-2-i}
    
    # Second term: a[i] * sum_{k=i}^{i+n-1} w[k]
    # Sliding window sum of w
    cw = np.concatenate([[0], np.cumsum(w)])  # length 2n
    window_sum = cw[n:n+n] - cw[:n]  # window_sum[i] = sum_{k=i}^{i+n-1} w[k]
    # Wait: cw[j] = sum_{k=0}^{j-1} w[k]
    # sum_{k=i}^{i+n-1} w[k] = cw[i+n] - cw[i]

    # cw[j] for j=0..2n-1: cw[0]=0, cw[j]=sum_{k=0}^{j-1} w[k] for j>=1
    # sum_{k=i}^{i+n-1} w[k] = cw[i+n] - cw[i]

    window_sum = cw[n:n+n] - cw[0:n]

    # cw[0:n] = cw[0], cw[1], ..., cw[n-1] (length n)

    grad = 2.0 * n * (first + second)

    cols = np.tile(np.arange(n, dtype=np.int64), n)
    offsets = np.repeat(np.arange(n, dtype=np.int64), n)
    rows = cols + offsets
    data = np.tile(np.asarray(a, dtype=np.float64).ravel(), n)
    return sps.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _lp_iteration(a, deadline):
    """One LP-based Frank-Wolfe step."""
    n = len(a)
    a = _normalize(a)
    b = np.convolve(a, a)
    M = float(b.max())
    if M <= 0:
        return a
    A = _build_conv_matrix(a)
    if A is None:
        return a
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M * (1.0 + 1e-12) + 1e-15)
    try:
        tl = max(1.0, deadline - time.monotonic() - 3.0)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)]*n,
                      method='highs', options={'time_limit': tl, 'presolve': True})
    except Exception:
        return a
    if res is None or not res.success or res.x is None:
        return a
    g = np.clip(res.x, 0.0, None)
    sg = g.sum()
    if sg <= 1e-15:
        return a
    g = g / sg
    return g

def _gradient_refine(a, deadline, rng, max_iters=5000):
    """Gradient descent on simplex for score minimization."""
    n = len(a)
    a = _normalize(a)
    local = _score_fast(a)
    
    # Adaptive smoothing
    base_beta = 5.0
    
    def get_grad(x):
        b = np.convolve(x, x)
        max_b = b.max()
        if max_b < 1e-30:
            return np.zeros(n)
        beta = base_beta * n / max_b
        mb = beta * b
        mb -= mb.max()
        e = np.exp(mb)
        w = e / e.sum()
        # grad_i = 4n * (convolve(w[::-1], x))[2n-2-i]
        u = w[::-1]
        uc = np.convolve(u, x)
        grad = 4.0 * n * uc[n-1:2*n-1][::-1]
        return grad
    
    stagnant = 0
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        
        grad = get_grad(a)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        best_val = local
        best_a = a.copy()
        lr = 0.1 / (gnorm + 1e-10)
        
        for mult in (0.5, 1.0, 2.0, 5.0, 10.0):
            alpha = mult * lr
            a_new = np.clip(a - alpha * grad, 0.0, None)
            a_new = _normalize(a_new)
            val = _score_fast(a_new)
            if math.isfinite(val) and val < best_val:
                best_val = val
                best_a = a_new
        
        if best_val < local * (1.0 - 1e-10):
            a = best_a
            local = best_val
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 30:
                # Add noise
                a_noisy = np.clip(a + 0.05 * rng.standard_normal(n), 0.0, None)
                a = _normalize(a_noisy)
                local = _score_fast(a)
                stagnant = 0
                if stagnant > 5:
                    break
                continue
            if stagnant > 100:
                break
    
    return a, local
