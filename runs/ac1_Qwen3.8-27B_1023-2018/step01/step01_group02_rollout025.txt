import numpy as np
import time
from scipy.optimize import linprog

def _score_arr(a):
    """Compute the score for array a."""
    a = np.asarray(a, dtype=np.float64)
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * len(a) * np.max(c) / (s * s))

def _convolve_matrix(a):
    """Build (2n-1) x n matrix representing multiplication by a under convolution."""
    n = len(a)
    C = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        C[k, lo:hi+1] = a[lo:hi+1]
    return C

def _lp_refine(a, M):
    """Solve LP: max 1'b s.t. C(a)@b <= M, b >= 0. Return b* and S*."""
    n = len(a)
    C = _convolve_matrix(a)
    # min -1'b s.t. C@b <= M, b >= 0
    c_obj = -np.ones(n)
    A_ub = C
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, float(-res.fun)
    return None, None

def _adam_step(params, grad, state, lr):
    b1, b2, eps = 0.9, 0.999, 1e-8
    state['t'] += 1
    t = state['t']
    state['m'] = b1 * state['m'] + (1-b1) * grad
    state['v'] = b2 * state['v'] + (1-b2) * grad**2
    m_hat = state['m'] / (1 - b1**t)
    v_hat = state['v'] / (1 - b2**t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    return params - step
