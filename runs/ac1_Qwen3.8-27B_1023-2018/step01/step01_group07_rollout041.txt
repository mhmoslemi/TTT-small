            # Adam update
            t_adam += 1
            m = b1 * m + (1 - b1) * gc
            v = b2 * v + (1 - b2) * gc * gc
            m_hat = m / (1 - b1 ** t_adam)
            v_hat = v / (1 - b2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = np.maximum(a - 0.1 * lr * m_hat / (np.sqrt(v_hat) + eps), 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    a_new = a.copy()
                else:
                    a_new = a_new / s_new
            else:
                a_new = a_new / s_new

            # Evaluate new candidate
            b_new = autocorr(a_new)
            max_b_new = b_new.max()

            if max_b_new < max_b - 1e-15:
                a = a_new
                stagnant = 0
                lr = min(lr * 1.005, 3.0)
                tau_frac = max(tau_frac * 0.998, min_tau_frac)
            else:
                stagnant += 1
                lr = max(lr * 0.93, min_lr)

                # Try halved step when moderately stuck
                if 20 < stagnant < 60 and stagnant % 10 == 0:
                    a_try = a - 0.3 * lr * m_hat / (np.sqrt(v_hat) + eps)
                    a_try = np.maximum(a_try, 0.0)
                    st = a_try.sum()
                    if st > 1e-15:
                        a_try /= st
                        bt = autocorr(a_try)
                        if bt.max() < max_b:
                            a = a_try
                            stagnant = max(stagnant - 10, 0)

            # Periodic exact evaluation
            if it % 200 == 0 or max_b_new * 2.0 * n < best_score * 0.999:
                sc = float(evaluate_sequence(a.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    consecutive_no_improve = 0
                    # Soft reset Adam momentum
                    m *= 0.4
                    v *= 0.4
                else:
                    consecutive_no_improve += 1

            # Stagnation: perturb and reset
            if stagnant > 300:
                # Large multiplicative perturbation
                sigma = 0.08 + 0.02 * (stagnant / 1000.0)
                noise = rng.normal(0.0, sigma, n)
                a = a * np.exp(noise)
                a = np.maximum(a, 0.0)
                ss = a.sum()
                if ss > 1e-15:
                    a /= ss
                # Reset optimizer
                m = np.zeros(n)
                v = np.zeros(n)
                t_adam = 0
                lr = 1.0
                tau_frac = min(tau_frac * 1.3, 0.30)
                stagnant = 0
            elif stagnant > 150:
                # Small perturbation
                noise = rng.normal(0.0, 0.03, n)
                a_pert = a * np.exp(noise)
                a_pert = np.maximum(a_pert, 0.0)
                sp = a_pert.sum()
                if sp > 1e-15:
                    a_pert /= sp
                    bp = autocorr(a_pert)
                    if bp.max() < max_b:
                        a = a_pert
                        stagnant = max(stagnant - 50, 0)
                    else:
                        stagnant = max(stagnant - 20, 0)

        except Exception:
            if time.monotonic() > hard_deadline - 3.0:
                break

    # Final evaluation
    try:
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
    except Exception:
        pass

    # Validate final answer
    try:
        bs = float(evaluate_sequence(best_a.tolist()))
        if not np.isfinite(bs) or bs > 10.0:
            best_a = np.ones(n, dtype=np.float64) / n
    except Exception:
        best_a = np.ones(n, dtype=np.float64) / n

    return [float(x) for x in best_a]
