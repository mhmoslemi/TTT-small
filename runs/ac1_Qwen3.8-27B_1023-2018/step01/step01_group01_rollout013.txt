import numpy as np
import time


def _p2(l):
    return 1 << max(1, (int(l) - 1).bit_length())


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s)) - 18.0

    rng = np.random.default_rng(seed)

    # Get base sequence
    try:
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64).ravel(), 0.0, 1000.0)
        if base.size == 0 or not np.all(np.isfinite(base)):
            raise ValueError
    except Exception:
        base = np.ones(6962, dtype=np.float64)
    n = len(base)

    # FFT sizes
    C_size = _p2(2 * n - 1)
    G_size = _p2(3 * n - 2)

    def safe_eval(a):
        try:
            s = float(evaluate_sequence([float(x) for x in a]))
            return s if np.isfinite(s) else np.inf
        except Exception:
            return np.inf

    def normalize(a):
        s = a.sum()
        if s < 1e-10:
            return a.copy()
        target = np.sqrt(2.0 * n)
        return np.clip(a * (target / s), 0.0, 1000.0)

    best = base.copy()
    best_score = safe_eval(best)

    def run_optimizer(a0, seconds, lr0, temp0, temp1, ad
                     am_b1, adam_b2, eval_every):
        nonlocal best, best_score
        a = normalize(np.asarray(a0, dtype=np.float64).copy())
        m_adam = np.zeros(n)
        v_adam = np.zeros(n)
        step_count = 0
        start_time = time.monotonic()
        last_eval = start_time
        local_best = a.copy()
        local_best_score = safe_eval(a)

        while time.monotonic() < deadline:
            elapsed = time.monotonic() - start_time
            if elapsed > seconds:
                break

            # Time-fraction for temperature and LR schedule
            frac = min(1.0, elapsed / max(seconds, 1e-6))
            temperature = temp0 + (temp1 - temp0) * frac
            lr = lr0 * (0.93 ** (step_count / 800.0))

            S = float(a.sum())
            if S < 1e-10:
                break

            # Compute autocorrelation via FFT
            fa_c = np.fft.rfft(a, C_size)
            C = np.fft.irfft(fa_c * fa_c, C_size)[:2 * n - 1]
            Cmax = float(C.max())

            # Smooth-max weights
            z = temperature * (C - Cmax)
            z = np.clip(z, -200.0, 200.0)
            w = np.exp(z)
            wsum = float(w.sum())
            if wsum < 1e-300 or not np.isfinite(wsum):
                w = np.ones(2 * n - 1, dtype=np.float64) / (2 * n - 1)
                wsum = 1.0
            w /= wsum
            M_smooth = Cmax + np.log(wsum) / temperature

            # Gradient of smooth-max via FFT cross-correlation
            wfl = w[::-1].copy()
            fa_g = np.fft.rfft(a, G_size)
            fw = np.fft.rfft(wfl, G_size)
            conv2 = np.fft.irfft(fa_g * fw, G_size)[:3 * n - 2]
            dM = 2.0 * conv2[n - 1:2 * n - 1][::-1]

            # Full gradient of f = 2n*M/S^2
            invS2 = 1.0 / (S * S)
            grad = 2.0 * n * invS2 * dM - 4.0 * n * M_smooth * invS2 / S

            # Adam update
            step_count += 1
            m_adam = ad
                     am_b1 * m_adam + (1.0 - ad
                                 am_b1) * grad
            v_adam = ad
                     am_b2 * v_adam + (1.0 - ad
                                 am_b2) * (grad * grad)
            m_hat = m_adam / (1.0 - ad
                              am_b1 ** step_count)
            v_hat = v_adam / (1.0 - ad
                              am_b2 ** step_count)

            a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            a = np.clip(a, 0.0, 1000.0)
            a = normalize(a)

            # Periodic exact evaluation
            now = time.monotonic()
            if step_count % eval_every == 0 and now - last_eval > 0.4:
                last_eval = now
                sc = safe_eval(a)
                if sc < local_best_score:
                    local_best = a.copy()
                    local_best_score = sc
                if sc < best_score:
                    best = a.copy()
                    best_score = sc

        # Final eval
        sc = safe_eval(a)
        if sc < local_best_score:
            local_best = a.copy()
            local_best_score = sc
        if local_best_score < best_score:
            best = local_best.copy()
            best_score = local_best_score
        return local_best

    # ---- Phase 1: Evaluate a few seeds ---------------------------------
    seeds = [normalize(base.copy())]
    # Add a couple structured perturbations of base
    alpha = 0.1
    pert1 = base.copy() + alpha * rng.standard_normal(n)
    seeds.append(normalize(np.clip(pert1, 0, None)))
    pert2 = base.copy()
    pert2[:n // 4] *= 0.7
    pert2[-n // 4:] *= 0.7
    seeds.append(normalize(pert2))

    scored = sorted([(safe_eval(s), s) for s in seeds], key=lambda t: t[0])
    start_seed = scored[0][1]

    # ---- Phase 2: Main optimization from best seed ---------------------
    avail = max(20.0, deadline - time.monotonic())
    main_time = 0.55 * avail
    run_optimizer(start_seed, main_time, 0.012, 6.0, 30.0, 0.85, 0.999, 800)

    # ---- Phase 3: Polish with higher temp, lower LR --------------------
    if time.monotonic() < deadline:
        remain = deadline - time.monotonic()
        run_optimizer(best.copy(), 0.30 * remain, 0.005, 30.0, 80.0, 0.90, 0.9999, 600)

    # ---- Phase 4: Ultra-fine polish ------------------------------------
    if time.monotonic() < deadline:
        remain = deadline - time.monotonic()
        run_optimizer(best.copy(), remain, 0.0015, 60.0, 150.0, 0.95, 0.9999, 500)

    # ---- Phase 5: Small random restarts if time permits -----------------
    if time.monotonic() < deadline:
        remain = deadline - time.monotonic()
        n_restarts = min(3, int(remain / 30))
        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            pert = best + 0.05 * rng.standard_normal(n)
            pert = normalize(np.clip(pert, 0, None))
            run_optimizer(pert, 0.25 * remain / max(n_restarts, 1),
                          0.006, 20.0, 100.0, 0.90, 0.9999, 500)

    # ---- Final validation ----------------------------------------------
    final = normalize(np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0))
    fs = safe_eval(final)
    if not np.isfinite(fs) or fs > best_score:
        final = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if not np.isfinite(safe_eval(final)):
        final = np.clip(base, 0.0, 1000.0)

    out = [float(x) for x in final]
    if not out or any((not np.isfinite(x)) for x in out):
        out = [1.0] * n
    return out
