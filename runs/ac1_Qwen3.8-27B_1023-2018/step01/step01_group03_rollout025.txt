        b = np.convolve(cur, cur)
        mx = float(b.max())
        w = np.exp((b - mx) * alpha)
        sw = float(w.sum())

            return None

        grad -= grad.mean()

        return grad / gn

        """LP: max sum(g) s.t. conv(cur,g) <= M*frac, g>=0."""

                -np.ones(n, dtype=np.float64),
                A_ub=A, b_ub=np.full(2 * n - 1, M, dtype=np.float64),
                bounds=[(0, None)] * n, method='highs',
                options={'time_limit': max(0.5, tl)})
            if res.success:

                s = float(g.sum())
