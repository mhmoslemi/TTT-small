    c = irfft(F * F, fsize)
    return c[: 2 * a.size - 1]


def _score_normalized(a, fsize):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))"""
    n = a.size
    b = _fft_conv(a, fsize)
    return float(2 * n * b.max())


def _score_general(a, fsize):
    n = a.size
    s = float(a.sum())
    if s < 0.01 or not np.isfinite(s):
        return math.inf
    b = _fft_conv(a, fsize)
    return float(2 * n * b.max() / (s * s))


def _grad_softmax(z, tau, fsize):
    """
    Returns (gradient_of_soft_max_conv w.r.t. z, a, b, U).
    a = softmax(z), conv = fft_conv(a,a)
    grad_j = 2*a_j * (sum_k p_k * a[k-j] - sum_k p_k * conv[k])
    """
    n = z.size
    a = _softmax_fn(z)
    b = _fft_conv(a, fsize)
    bmax = float(b.max())
    if not np.isfinite(bmax) or bmax <= 0:
        return np.zeros(n), a, b, bmax

    tau = max(tau, 1e-12)
    x = (b - bmax) / tau
    ex = np.exp(x)
    s = ex.sum()
    if s <= 0 or not np.isfinite(s):
        p = np.zeros_like(b)
        p[int(np.argmax(b))] = 1.0
    else:
        p = ex / s
    U = bmax + tau * math.log(max(s, 1e-300))

    # S[j] = sum_k p[k] * a[k-j]  (a out-of-range = 0)
    thresh = p.max() * 1e-7
    idx = np.where(p > thresh)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(p))])

    S = np.zeros(n, dtype=np.float64)
    for ki in idx:
        ki = int(ki)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if hi >= lo:
            S[lo: hi + 1] += p[ki] * a[ki - hi: ki - lo + 1][::-1]

    S -= float(np.dot(p, b))
    grad = 2.0 * a * S
    grad -= grad.mean()  # project out shift-null-space
    return grad, a, b, float(U)


def _optimize_round(n, init_a, max_secs, seed, fsize, deadline):
    """Soft-max gradient descent in log-space. Returns (best_a, best_score)."""
    r = np.random.default_rng(seed)
    a = np.clip(init_a.astype(np.float64).copy(), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    z = np.log(a + 1e-30)
    z -= z.mean()
    z = np.clip(z, -20.0, 20.0)

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    tcount = 0
    lr = 0.5
    stuck = 0
    no_improve = 0.0

    b = _fft_conv(a, fsize)
    cur_max = float(b.max())
    local_best = a.copy()
    local_best_score = 2.0 * n * cur_max  # sum=1 normalization

    start = time.monotonic()
    last_improve = start

    # Warm-start tau
    tau = 0.12 * max(cur_max, 1e-10)
    tau_min = 0.003 * max(cur_max, 1e-10)

    while time.monotonic() < start + max_secs and time.monotonic() < deadline - 5.0:
        b = _fft_conv(a, fsize)
        cur_max = float(b.max())
        if not np.isfinite(cur_max):
            break
        cur_score = 2.0 * n * cur_max

        if cur_score < local_best_score - 1e-10:
            local_best_score = cur_score
            local_best = a.copy()
            last_improve = time.monotonic()

        # Anneal tau
        tau = max(tau * 0.998, tau_min)

        grad, a, b, U = _grad_softmax(z, tau, fsize)
        if not np.isfinite(grad).all():
            break
        gnorm = float(np.sqrt(np.sum(grad * grad)))
        if gnorm < 1e-14:
            stuck += 1
            if stuck > 30:
                break
            z += r.normal(0, 0.2, n)
            z -= z.mean()
            z = np.clip(z, -20, 20)
            a = _softmax_fn(z)
            m_adam[:] = 0
            v_adam[:] = 0
            tcount = 0
            continue

        tcount += 1
        m_adam = 0.9 * m_adam + 0.1 * grad
        v_adam = 0.999 * v_adam + 0.001 * grad * grad
        m_hat = m_adam / (1.0 - 0.9 ** tcount)
        v_hat = v_adam / (1.0 - 0.999 ** tcount)
        direction = -m_hat / (np.sqrt(v_hat) + 1e-8)

        improved = False
        for fac in (1.0, 0.5, 0.25, 0.1):
            step = lr * fac * direction / max(gnorm, 1e-10)
            trial_z = z + step
            trial_z -= trial_z.mean()
            trial_z = np.clip(trial_z, -20.0, 20.0)
            trial_a = _softmax_fn(trial_z)
            trial_b = _fft_conv(trial_a, fsize)
            trial_max = float(trial_b.max())
            if np.isfinite(trial_max) and trial_max < cur_max * (1.0 - 1e-9):
                z = trial_z
                a = trial_a
                b = trial_b
                cur_max = trial_max
                improved = True
                break

        if improved:
            stuck = 0
        else:
            stuck += 1
            lr = max(0.02, lr * 0.88)
            if stuck > 50:
                z += r.normal(0, 0.4, n)
                z -= z.mean()
                z = np.clip(z, -20, 20)
                a = _softmax_fn(z)
                m_adam[:] = 0
                v_adam[:] = 0
                tcount = 0
                lr = 0.4
                tau = 0.12 * max(float(_fft_conv(a, fsize).max()), 1e-10)
                stuck = 0

        if time.monotonic() - last_improve > max(15.0, 0.3 * (time.monotonic() - start)):
            break

    # Evaluate with general score (not assuming sum=1)
    gs = _score_general(local_best, fsize)
    if not np.isfinite(gs):
        gs = local_best_score
    return local_best, gs


def _coordinate_polish(a, fsize, n, deadline, max_iters=30):
    """Local coordinate descent: adjust one element at a time."""
    a = np.clip(a.astype(np.float64).copy(), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return a
    a = a / s
    cur_score = _score_normalized(a, fsize)
    r = np.random.default_rng(777)

    for it in range(max_iters):
        if time.monotonic() > deadline - 8.0:
            break
        improved = False
        order = r.permutation(n)
        for j in order:
            if time.monotonic() > deadline - 8.0:
                break
            aj = a[j]
            if aj < 1e-15:
                continue
            for mult in (0.6, 0.8, 0.95, 1.05, 1.2, 1.4):
                trial = a.copy()
                trial[j] = aj * mult
                if trial[j] < 0:
                    continue
                ts = float(trial.sum())
                if ts < 0.01:
                    continue
                tb = _fft_conv(trial, fsize)
                ts_score = 2.0 * n * tb.max() / (ts * ts)
                if ts_score < cur_score - 1e-10:
                    a = trial
                    cur_score = ts_score
                    improved = True
                    break
        if not improved:
            break

    return a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 12.0  # safety margin

    parent = _get_parent()
    n = int(parent.size)

    # Evaluate parent
    try:
        ps = float(evaluate_sequence([float(x) for x in parent]))
    except Exception:
        ps = math.inf

    if not np.isfinite(ps):
        parent = np.ones(n, dtype=np.float64) / n
        try:
            ps = float(evaluate_sequence([float(x) for x in parent]))
        except Exception:
            ps = 2.0

    best_a = parent.astype(np.float64).copy()
    best_score = ps

    fsize = next_fast_len(2 * n - 1)
    rng = np.random.default_rng(seed)

    # --- Strategy: multiple gradient descent rounds ---
    inits = []

    # 1. Parent
    inits.append((parent.copy(), seed + 0, 180.0))

    # 2. Perturbed parent
    p2 = parent.copy()
    s2 = p2.sum()
    if s2 > 0:
        p2 = p2 / s2
    z2 = np.log(p2 + 1e-15)
    z2 += rng.normal(0, 0.4, n)
    z2 -= z2.mean()
    inits.append((_softmax_fn(z2), seed + 100, 150.0))

    # 3. Uniform + noise
    a3 = np.ones(n) + rng.normal(0, 0.3, n)
    a3 = np.maximum(a3, 0.05)
    a3 /= a3.sum()
    inits.append((a3, seed + 200, 130.0))

    # 4. Cosine structure
    i = np.arange(n, dtype=np.float64)
    a4 = 0.6 + 0.35 * np.cos(2 * math.pi * i / n) + 0.25 * np.cos(4 * math.pi * i / n + 0.5)
    a4 = np.maximum(a4, 0.02)
    a4 /= a4.sum()
    inits.append((a4, seed + 300, 110.0))

    # 5. Symmetric random
    half = n // 2
    ah = rng.uniform(0.4, 1.6, half)
    a5 = np.concatenate([ah, ah[::-1]]) if n % 2 == 0 else np.concatenate([ah, [ah[-1]], ah[::-1]])
    a5 = a5[:n]
    a5 = np.maximum(a5, 0.01)
    a5 /= a5.sum()
    inits.append((a5, seed + 400, 100.0))

    # 6. Two-bump
    c1, c2 = n // 3, 2 * n // 3
    a6 = np.zeros(n)
    a6[:c1] = 1.0
    a6[c2:] = 0.8
    a6 += rng.normal(0, 0.15, n)
    a6 = np.maximum(a6, 0.01)
    a6 /= a6.sum()
    inits.append((a6, seed + 500, 100.0))

    for idx, (init_a, s, secs) in enumerate(inits):
        if time.monotonic() > deadline - 25.0:
            break
        remaining = deadline - time.monotonic() - 20.0
        secs = min(secs, remaining)
        if secs < 5.0:
            break
        try:
            cand, sc = _optimize_round(n, init_a, secs, s, fsize, deadline)
            if cand is not None and cand.size == n:
                if np.isfinite(sc) and sc < best_score - 1e-9:
                    best_score = sc
                    best_a = cand.copy()
        except Exception:
            pass

    # --- Additional rounds perturbing the best ---
    for k in range(20):
        if time.monotonic() > deadline - 40.0:
            break
        z = np.log(best_a + 1e-15)
        scale = 0.3 * (1.0 / (1.0 + 0.15 * k))
        z += rng.normal(0, scale, n)
        z -= z.mean()
        a_new = _softmax_fn(z)
        secs = min(60.0, deadline - time.monotonic() - 20.0)
        if secs < 5.0:
            break
        try:
            cand, sc = _optimize_round(n, a_new, secs, seed + 5000 + 100 * k, fsize, deadline)
            if cand is not None and cand.size == n:
                if np.isfinite(sc) and sc < best_score - 1e-9:
                    best_score = sc
                    best_a = cand.copy()
        except Exception:
            pass

    # --- Coordinate polish ---
    if time.monotonic() < deadline - 15.0:
        try:
            polished = _coordinate_polish(best_a, fsize, n, deadline, max_iters=25)
            ps2 = _score_general(polished, fsize)
            if np.isfinite(ps2) and ps2 < best_score - 1e-9:
                best_score = ps2
                best_a = polished.copy()
        except Exception:
            pass

    # --- Final exact validation ---
    best_a = np.clip(best_a, 0.0, 1000.0)
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_a]))
    except Exception:
        final_score = math.inf

    if not np.isfinite(final_score) or final_score > ps:
        try:
            return [float(x) for x in parent]
        except Exception:
            return [1.0]

    return [float(x) for x in best_a]
