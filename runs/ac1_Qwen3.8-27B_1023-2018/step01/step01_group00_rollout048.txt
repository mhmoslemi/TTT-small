    v = beta2*v + (1-beta2)*g*g

    return np.maximum(c, 0.0)


def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = _fftc(a)
    return float(2.0 * n * b.max() / (s * s))


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones(len(z)) / len(z)
    return e / s


def _optimize_run(n, z0, t_budget, seed):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    z = np.asarray(z0, dtype=np.float64).copy()
    z = z - z.mean()

    a = _softmax(z)
    b = _fftc(a)
    best_a = a.copy()
    best_bmax = float(b.max())

    beta1, beta2, eps_adam = 0.85, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    step = 0
    base_lr = 0.6

    tau_frac_start = 0.25
    tau_frac_end = 0.0005
    stuck = 0
    max_stuck = 60

    idx = np.arange(n)
    hankel_idx = idx + idx[:, None]

    while time.monotonic() - t0 < t_budget:
        a = _softmax(z)
        b = _fftc(a)
        bmax = float(b.max())

        if bmax < best_bmax:
            best_bmax = bmax
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > max_stuck:
            elapsed = time.monotonic() - t0
            if elapsed > t_budget * 0.75:
                break
            scale = max(0.1, 0.6 * (1.0 - elapsed / t_budget))
            z += rng.normal(0, scale, n)
            z -= z.mean()
            z = np.clip(z, -25, 25)
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            step = 0
            stuck = 0
            continue

        progress = min(1.0, (time.monotonic() - t0) / max(t_budget, 1e-9))
        tau = bmax * tau_frac_start * (tau_frac_end / tau_frac_start) ** progress
        tau = max(tau, 1e-12)

        bm = b.max()
        x = (b - bm) / tau
        e = np.exp(x)
        s_e = float(e.sum())
        if s_e < 1e-300 or not np.isfinite(s_e):
            p = np.zeros_like(b)
            mask = b >= bm - 1e-14
            p[mask] = 1.0 / mask.sum()
        else:
            p = e / s_e

        P = p[hankel_idx]
        g_a = P @ a
        S_ga = float(g_a @ a)
        g_z = g_a - a * S_ga

        if not np.isfinite(g_z).all():
            g_z = np.where(np.isfinite(g_z), g_z, 0.0)
        g_norm = np.linalg.norm(g_z)
        if g_norm < 1e-14:
            stuck += 5
            continue

        lr = base_lr / (1.0 + 3.0 * progress)
        step += 1
        m_adam = beta1 * m_adam + (1 - beta1) * g_z
        v_adam = beta2 * v_adam + (1 - beta2) * (g_z * g_z)
        m_hat = m_adam / (1.0 - beta1 ** step)
        v_hat = v_adam / (1.0 - beta2 ** step)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z -= z.mean()
        z = np.clip(z, -30, 30)

    return best_a, best_bmax


def _polish_cd(a, n, t_budget):
    t0 = time.monotonic()
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_bmax = float(_fftc(a).max())
    rng = np.random.default_rng(777)
    factors = np.array([0.4, 0.6, 0.8, 0.9, 1.1, 1.2, 1.4, 1.7, 2.5])

    passes = 0
    while time.monotonic() - t0 < t_budget:
        passes += 1
        improved_any = False
        for i in range(n):
            if time.monotonic() - t0 > t_budget:
                break
            old = a[i]
            if old < 1e-15:
                continue
            for f in factors:
                a[i] = old * f
                s = a.sum()
                if s < 0.01:
                    a[i] = old
                    continue
                a_n = a / s
                bm = float(_fftc(a_n).max())
                if bm < best_bmax - 1e-12:
                    best_bmax = bm
                    a = a_n
                    improved_any = True
                    break
            else:
                a[i] = old
                a = a / a.sum()

        if not improved_any:
            for _ in range(max(5, n // 10)):
                if time.monotonic() - t0 > t_budget:
                    break
                i = int(rng.integers(n))
                a[i] *= float(rng.uniform(0.4, 2.0))
                a = np.clip(a, 0.0, None)
                s = a.sum()
                if s < 0.01:
                    continue
                a = a / s
                bm = float(_fftc(a).max())
                if bm < best_bmax:
                    best_bmax = bm

    return a, best_bmax


def _try_length(n, parent, seed, t_budget):
    if time.monotonic() > t_budget:
        return None, float('inf')

    i = np.arange(n, dtype=np.float64)
    inits = []

    if parent is not None and parent.size == n:
        pn = np.clip(parent, 1e-10, 1000.0)
        pn = pn / pn.sum()
        inits.append(pn)

    a_rc = 1.0 + 0.5 * np.cos(2 * np.pi * i / n)
    a_rc = np.maximum(a_rc, 1e-8)
    inits.append(a_rc / a_rc.sum())

    a_sin = 1.0 + 0.6 * np.cos(4 * np.pi * i / n)
    a_sin = np.maximum(a_sin, 1e-8)
    inits.append(a_sin / a_sin.sum())

    a_sin2 = 1.0 + 0.5 * np.sin(2 * np.pi * i / n + 0.5) + 0.3 * np.cos(4 * np.pi * i / n)
    a_sin2 = np.maximum(a_sin2, 1e-8)
    inits.append(a_sin2 / a_sin2.sum())

    sigma = max(2.0, n / 12.0)
    a_gauss = np.exp(-0.5 * ((i - n / 2) / sigma) ** 2) + 0.1
    inits.append(a_gauss / a_gauss.sum())

    rng0 = np.random.default_rng(seed + 99999)
    a_rnd = rng0.random(n) + 0.01
    inits.append(a_rnd / a_rnd.sum())

    inits.append(np.ones(n) / n)

    n_inits = len(inits)
    best_a = None
    best_score = float('inf')

    for idx_a, a0 in enumerate(inits):
        remaining = t_budget - time.monotonic()
        if remaining < 3.0:
            break
        alloc = remaining / max(1, n_inits - idx_a)
        alloc = max(3.0, alloc)

        s0 = _score_fast(a0)
        if s0 < best_score:
            best_score = s0
            best_a = a0.copy()

        z0 = np.log(a0 + 1e-30)
        z0 = z0 - z0.mean()

        try:
            a_opt, bmax_opt = _optimize_run(n, z0, alloc, seed + 50000 * idx_a)
            s_opt = float(2.0 * n * bmax_opt)
            if s_opt < best_score:
                best_score = s_opt
                best_a = a_opt.copy()
        except Exception:
            pass

    if best_a is not None:
        cd_budget = max(0.0, t_budget - time.monotonic() - 2.0)
        if cd_budget > 2.0:
            try:
                a_p, bm_p = _polish_cd(best_a, n, min(cd_budget, 80.0))
                s_p = float(2.0 * n * bm_p)
                if s_p < best_score:
                    best_score = s_p
                    best_a = a_p.copy()
            except Exception:
                pass

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s)) - 8.0

    parent = _get_parent()

    if parent is not None:
        fallback = parent.copy()
        try:
            fb_score = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            fb_score = float('inf')
    else:
        fallback = np.ones(64)
        fb_score = 2.0

    best_seq = fallback.copy()
    best_score = fb_score

    n_primary = int(parent.size) if parent is not None else 192

    # Primary optimization at parent length
    t_budget_primary = min(deadline - time.monotonic() - 30.0, 900.0)
    if t_budget_primary > 20.0:
        a_p, s_p = _try_length(n_primary, parent, int(seed), deadline - 30.0)
        if a_p is not None and s_p < best_score:
            best_score = s_p
            best_seq = a_p.copy()

    # Try a couple other lengths if time permits
    for n_alt in [256, 384]:
        if time.monotonic() > deadline - 25.0:
            break
        if best_score < 1.51:
            break
        t_alloc = min(120.0, deadline - time.monotonic() - 15.0)
        if t_alloc < 15.0:
            break
        a_alt, s_alt = _try_length(n_alt, parent, int(seed) + 31337, time.monotonic() + t_alloc)
        if a_alt is not None and s_alt < best_score:
            best_score = s_alt
            best_seq = a_alt.copy()

    # Final CD polish on the best result
    if time.monotonic() < deadline - 8.0:
        n_best = len(best_seq)
        cd_t = min(40.0, deadline - time.monotonic() - 4.0)
        if cd_t > 3.0:
            try:
                a_final, bm_final = _polish_cd(best_seq, n_best, cd_t)
                s_final = float(2.0 * n_best * bm_final)
                if s_final < best_score:
                    best_score = s_final
                    best_seq = a_final.copy()
            except Exception:
                pass

    # Validate with exact evaluator
    try:
        vs = float(evaluate_sequence([float(x) for x in best_seq]))
        if np.isfinite(vs):
            best_score = vs
        else:
            best_seq = fallback
            best_score = fb_score
    except Exception:
        best_seq = fallback
        try:
            best_score = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            best_seq = [1.0]

    return [float(x) for x in best_seq]
