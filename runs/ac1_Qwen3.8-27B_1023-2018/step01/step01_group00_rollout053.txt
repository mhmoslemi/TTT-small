def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Use the bisection method for the threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def convolve(a):
    """Linear convolution of a with itself."""
    n = len(a)
    N = next_fast_len(2*n - 1)
    A = rfft(a, N)
    b = irfft(A * A, N)[:2*n-1]
    b[~np.isfinite(b)] = 0.0
    b[b < 0] = 0.0  # clip tiny negatives from FFT
    return b

    """Compute 2n * max(conv(a,a)) assuming sum(a) = 1."""
    n = len(a)
    b = convolve(a)
    return 2.0 * n * b.max()

def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}"""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

    rho = np.max(np.where(cond)[0])

    N = next_fast_len(2 * n - 1)
    A = rfft(a, N)
    b = np.real(irfft(A * A, N)[:2 * n - 1])
    np.maximum(b, 0, out=b)  # clip tiny negatives
    return b
