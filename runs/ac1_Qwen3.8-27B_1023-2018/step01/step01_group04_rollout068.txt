    uc = np.convolve(w[::-1], a)  # length 2n-1

    correction[valid] = w[idx[valid]]
    
    second = a * (window_sum - correction)
    
    return first + second

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}. O(n log n)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)

def _score_fast(a):

    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-15:
        a = np.ones(a.size) / a.size
    else:
        a = a / s
    return a

def _build_conv_matrix(a):
    n = len(a)
    if n <= 0:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    cols = np.tile(np.arange(n, dtype=np.int64), n)
    offsets = np.repeat(np.arange(n, dtype=np.int64), n)
    rows = cols + offsets
    data = np.tile(a, n)
    return sps.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def _project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.flatnonzero(u * np.arange(1, n+1) > (cssv - 1.0))
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad_smooth(a, beta):
    """Compute gradient of 2n*sum_k w_k*b_k where w=softmax(beta*b), b=a*a.
    Returns (smooth_objective, gradient)."""
    n = len(a)
    b = np.convolve(a, a)
    mb = beta * b
    mb -= mb.max()
    e = np.exp(mb)
    w = e / e.sum()
    
    # cross[i] = sum_j w[i+j]*a[j] = convolve(w[::-1], a)[2n-2-i]
    uc = np.convolve(w[::-1], a)
    cross = uc[n-1:2*n-1][::-1]
    
    # wsum[i] = sum_{j=0}^{n-1} w[i+j]
    cw = np.zeros(2*n)
    cw[1:] = np.cumsum(w)
    wsum = cw[n:2*n] - cw[:n]
    
    grad = 2.0 * n * (cross + a * wsum)
    obj = 2.0 * n * float(w @ b)
    return obj, grad

def _lp_iteration(a, deadline):
    """One LP Frank-Wolfe step. Returns new normalized sequence."""
    n = len(a)
    a = _normalize(a)
    b = np.convolve(a, a)
    M = float(b.max())
    if M <= 0 or not math.isfinite(M):
        return a
    A = _build_conv_matrix(a)
    if A is None:
        return a
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M * (1.0 + 1e-12) + 1e-15)
    tl = max(0.5, deadline - time.monotonic() - 3.0)
    if tl < 0.5:
        return a
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)]*n,
                      method='highs', options={'time_limit': tl, 'presolve': True})
    except Exception:
        return a
    if res is None or not res.success or res.x is None:
        return a
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    sg = g.sum()
    if sg <= 1e-15:
        return a
    g = g / sg
    return g

def _gradient_refine(a, deadline, rng, max_iters=3000):

    n = len(a)
    a = _normalize(a).copy()
    local = _score_fast(a)
    
    # Adaptive beta
    b0 = np.convolve(a, a)
    M0 = max(float(b0.max()), 1e-30)
    beta = 10.0 * n / M0  # Initial beta
    
    stagnant = 0
    it = 0
    while it < max_iters:
        if time.monotonic() >= deadline:
            break
        it += 1
        
        b = np.convolve(a, a)
        M = float(b.max())
        if M < 1e-30:
            break
        beta = max(10.0 * n / M, 1.0)
        
        try:
            _, grad = _compute_grad_smooth(a, beta)
        except Exception:
            break
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Line search
        best_val = local
        best_a = a.copy()
        base_lr = 0.1 / (gnorm + 1e-10)
        for mult in (0.5, 1.0, 2.0, 5.0, 10.0, 20.0):

            a_new = a - alpha * grad
            a_new = np.clip(a_new, 0.0, None)
            # Project onto simplex

            val = _score_fast(a_new)
            if math.isfinite(val) and val < best_val:
                best_val = val
                best_a = a_new
            if time.monotonic() >= deadline:
                break
        
        if best_val < local * (1.0 - 1e-10):
            a = best_a
            local = best_val
            stagnant = 0

def _make_starts(n, rng, parent):
    starts = []
    x = np.linspace(0.0, 1.0, n)
    
    def add(a):

            return
        s = float(a.sum())
        if s > 1e-15:

    add(x)
    add(1.0 - x)
    
    for c in (0.5, 1.0, 2.0, 5.0, -0.5, -1.0, -2.0):
        add(np.exp(c * (x - 0.5)))

        for amp in (0.3, 0.5, 0.7):
            add(1.0 + amp * np.sin(2.0*math.pi*k*x))
            add(1.0 + amp * np.cos(2.0*math.pi*k*x))

        try:
            add(rng.dirichlet(np.full(n, alpha)))

            pass

    if parent is not None and len(parent) >= 2:
        p = np.asarray(parent, dtype=np.float64).ravel()

        p = np.clip(p, 0.0, None)
        if p.sum() > 1e-15:
            tp = np.linspace(0.0, 1.0, p.size)
            xp = np.linspace(0.0, 1.0, n)
            add(np.interp(xp, tp, p))
            # Scaled parent
            for s in (0.5, 2.0):
                add(p * s)
    
    if not starts:
        starts.append(np.ones(n) / n)

    scored = []

        sc = _score_fast(a)
        if math.isfinite(sc):
