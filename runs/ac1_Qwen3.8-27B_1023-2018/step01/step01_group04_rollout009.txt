```python
def _simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
def _score(a):

    a = np.clip(np.nan_to_num(a), 0, None)
    n = a.size
    if n == 0: return float('inf')

def _project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > css - 1)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (css[rho] - 1) / (rho + 1)

def _lp_step(a):
    n = a.size
    b = np.convolve(a, a)
    M = b.max()
    if M <= 0: return a

    rows, cols, data = [], [], []
    for j in range(n):
```
