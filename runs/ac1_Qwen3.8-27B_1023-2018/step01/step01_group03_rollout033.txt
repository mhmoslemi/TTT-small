def _smoothed_grad(x, beta):

    a /= a.sum()
    n = a.size
    b = np.convolve(a, a)
    scaled = beta * b
    mse = scaled.max()

    w = exp_scaled / exp_scaled.sum()
    
    # Gradient of sum_k w[k]*b[k] w.r.t. a
    # Using 2D approach
    I = np.arange(n)[:, None]
    J = np.arange(n)[None, :]
    K = I + J
    valid1 = (K >= 0) & (K < 2*n-1)
    W1 = np.where(valid1, w[K], 0.0)

    valid2 = (K2 >= 0) & (K2 < 2*n-1)

def _smoothed_obj_grad(x, beta):

def _subgradient_polish(a, n, steps, deadline, step_size=0.01):
    """Direct subgradient method with simplex projection."""
    a = _norm(a.copy())
    best = a.copy()
    best_sc = _exact_score(a)
    
    for step in range(steps):
        if time.monotonic() > deadline:
            break
        
        b = np.convolve(a, a)
        bmax = b.max()
        # Active set: lags where b is within 5% of max
        active = np.where(b >= 0.95 * bmax)[0]
        
        # Subgradient: sum of gradients at active lags
        I = np.arange(n)[:, None]
        J = np.arange(n)[None, :]
        K = I + J
        grad_a = np.zeros(n)
        
        for k in active:
            # db[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad_a[i_lo:i_hi+1] += a[k - i_hi:k - i_lo + 1][::-1]
            if k < n:
                i_lo2 = k
                i_hi2 = min(n - 1, k + n - 1)
                if i_lo2 <= i_hi2:
                    grad_a[i_lo2:i_hi2+1] += a[0:i_hi2 - k + 1]
        
        # Normalize subgradient
        gn = np.linalg.norm(grad_a)
        if gn < 1e-15:
            break
        
        # Projected gradient step
        a_new = a - step_size * grad_a / gn
        a_new = _simplex_project(a_new)
        
        sc = _exact_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best = a_new.copy()
            a = a_new
            # Decrease step size slowly
            step_size *= 0.99
        else:
            a = best  # backtrack
            step_size *= 0.9
    
    return best, best_sc

def _simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    cv = np.cumsum(sv)
    rho = np.nonzero(sv * np.arange(1, n + 1) > (cv - 1))[0]
    if len(rho) == 0:
        return np.full(n, 1.0/n)
    rho = rho[-1]
    mu = (sv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - mu, 0.0)

        A = np.zeros((2*n-1, n))

                          options={'presolve': True, 'time_limit': max(1, (deadline - time.monotonic()) * 0.8)})
        except Exception:
            break
        
        if not res.success:
            break
        
        g = _norm(res.x)
        
        # Line search for best blend
        best_t = 0
        best_sc_t = best_sc
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1-t) * q + t * g
            cand = _norm(cand)
            sc = _exact_score(cand)
            if sc < best_sc_t:

                best_t = t

        else:

def _norm(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000.0)
    s = a.sum()
    if s < 1e-10:
        return np.full(a.shape, 1.0/a.size)
    return a / s

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    return 2.0 * n * float(b.max()) / (s*s)

def _soft_obj_grad(x, beta):
    """Smoothed objective and gradient in softmax parameterization."""
    a = np.exp(x - x.max())
    a /= a.sum()
    n = a.size
    b = np.convolve(a, a)
    
    # Log-sum-exp
    scaled = beta * b
    mse = scaled.max()
    exp_s = np.exp(scaled - mse)
    lse = mse + np.log(exp_s.sum())
    obj = lse / beta
    
    # Weights
    w = exp_s / exp_s.sum()
    
    # Gradient of sum_k w[k] * b[k] w.r.t. a
    idx1 = np.arange(n)[:, None] + np.arange(n)[None, :]
    W1 = np.where((idx1 >= 0) & (idx1 < 2*n-1), w[idx1], 0.0)
    idx2 = np.arange(n)[:, None] - np.arange(n)[None, :]
    W2 = np.where((idx2 >= 0) & (idx2 < 2*n-1), w[idx2], 0.0)
    
    grad_a = W1 @ a + W2 @ a
    
    # Chain rule for softmax
    mean_g = float(grad_a @ a)
    grad_x = a * (grad_a - mean_g)
    
    return obj, grad_x

            break

            return _soft_obj_grad(x, beta)[0]
        
        def jac(x):
            return _soft_obj_grad(x, beta)[1]
        
        try:
            res = minimize(func, x, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iters_per_beta, 'gtol': 1e-8,
                                   'ftol': 1e-12})
            x = res.x

        a_new /= a_new.sum()
        a_new = np.maximum(a_new, 0)
        
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            x = np.log(np.maximum(a, 1e-15))
        else:
            # Try smaller step: revert and increase beta
            pass
    
    return best_a, best_sc

import numpy as np
import time
from scipy.optimize import linprog, minimize


def _norm(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-10:
        return np.full(a.shape, 1.0 / a.size, dtype=np.float64)
    return a / s


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    return 2.0 * n * float(b.max()) / (s * s)


def _soft_obj_grad(x, beta):
    a = np.exp(x - x.max())
    a /= a.sum()
    n = a.size
    b = np.convolve(a, a)
    scaled = beta * b
    mse = scaled.max()
    exp_s = np.exp(scaled - mse)
    lse = mse + np.log(exp_s.sum())
    obj = lse / beta
    w = exp_s / exp_s.sum()
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]
    W = np.where((idx >= 0) & (idx < 2 * n - 1), w[idx], 0.0)
    grad_a = 2.0 * (W @ a)
    mean_g = float(grad_a @ a)
    grad_x = a * (grad_a - mean_g)
    return obj, grad_x


def _optimize_smoothed(a0, n, deadline, max_iters_per_beta=50, max_beta=400):
    a = _norm(a0.copy())
    a_safe = np.maximum(a, 1e-15)
    x = np.log(a_safe)
    best_a = a.copy()
    best_sc = _score(a)
    betas = [5, 15, 50, 100, 200, max_beta]
    for beta in betas:
        if time.monotonic() > deadline - 2:
            break
        best_x = x.copy()
        try:
            res = minimize(
                lambda v: _soft_obj_grad(v, beta)[0],
                x,
                jac=lambda v: _soft_obj_grad(v, beta)[1],
                method='L-BFGS-B',
                options={'maxiter': max_iters_per_beta, 'gtol': 1e-10, 'ftol': 1e-14}
            )
            best_x = res.x
        except Exception:
            pass
        a_new = np.exp(best_x - best_x.max())

            x = np.log(np.maximum(a_new, 1e-15))

def _lp_one_step(q, n, deadline):
    b = np.convolve(q, q)
    M = float(b.max())
    if M <= 0 or not np.isfinite(M):
        return None
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi + 1] = q[j_hi:j_lo - 1:-1]
    c = -np.ones(n)
    remaining = deadline - time.monotonic() - 3.0
    if remaining < 2.0:
        return None
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M, dtype=np.float64),
                      bounds=(0, None), method='highs',
                      options={'presolve': True, 'time_limit': max(1.0, remaining * 0.6)})
    except Exception:
        return None
    if not res.success:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s

def _subgradient_polish(a, n, deadline, max_steps=2000):

    best = a.copy()
    best_sc = _score(a)
    step = 0.05
    stale = 0

            break
        b = np.convolve(a, a)

        active = np.where(b >= 0.9 * bmax)[0]
        if active.size == 0:
            break
        w_active = b[active]
        w_active = (w_active - w_active.min() + 0.1)
        w_active /= w_active.sum()
        grad_a = np.zeros(n)
        for k, wk in zip(active, w_active):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:

        gn = float(np.linalg.norm(grad_a))

            break

def _upscale(a, n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == n:
        return _norm(a)
    if a.size == 0:
        return np.full(n, 1.0 / n)
    x_old = np.linspace(0, 1, a.size)
    x_new = np.linspace(0, 1, n)
    out = np.interp(x_new, x_old, a)
    return _norm(out)
