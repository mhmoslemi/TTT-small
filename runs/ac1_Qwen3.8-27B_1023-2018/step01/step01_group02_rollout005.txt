def _optimize_stage(a0, n, start, end, rng, inits, base_lr=0.6):
    a0 = _floor_norm(a0, 1e-12)
    best = a0.copy()
    best_s = _score_fft(best)
    z = np.log(best)
    z -= z.mean()
    m = np.zeros(n); v = np.zeros(n); adam_t=0; b1c=0.0; b2c=0.0
    lr=base_lr
    last_imp = time.monotonic()

    last_lp = start - 100
    last_random = start - 100

    else: N = 1 << (size-1).bit_length()
    idx = L - 1 - np.arange(n, dtype=np.int64)
    stage_dur = max(1e-3, end-start)
    while True:
        now = time.monotonic()
        if now >= end: break

        a = _floor_norm(a, 1e-12)
        fa = np.fft.rfft(a, N)
        q = np.fft.irfft(fa*fa, N)
        q = q[:L]
        np.maximum(q, 0.0, out=q)
        qs = float(q.sum())
        if qs > 0 and np.isfinite(qs):

            m = np.zeros(n); v = np.zeros(n); adam_t=0; b1c=0; b2c=0
            lr = max(0.1, lr*0.7)
            continue

        beta = 2000.0 + 120000.0*frac

        gamma = 0.02 + 0.4*(1.0-frac)

        b1c += 1 - 0.9

def _normalize_floor(a, eps=1e-12):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    a = a / s
    a = np.maximum(a, eps)
    a /= a.sum()
    return a


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))

    s = float(e.sum())

    return e / s


def _next_pow2_int(size):
    size = int(max(1, size))
    if size <= 1:

    return 1 << (size - 1).bit_length()


def _score_fft(a):

def _upsample(parent, n):

            v = np.where(np.isfinite(v), v, 0.0)
            d = v - float(np.mean(v))
            scale = float(np.max(np.abs(d)))
            if not np.isfinite(scale) or scale <= 1e-13:
                d = rng.standard_normal(n)
                scale = float(np.max(np.abs(d)))
                if scale <= 1e-13:
                    continue
            d /= scale
            if stall > 20:
                d = d + 0.15 * float(np.std(d)) * rng.standard_normal(n)
                scale2 = float(np.max(np.abs(d)))
                if scale2 > 1e-13 and np.isfinite(scale2):
                    d /= scale2

            improved = False
            for alpha in alphas:
                an = best - alpha * d
                an = np.maximum(an, 0.0)
                s = float(an.sum())
                if s <= 1e-12:
                    continue
                an /= s
                sc = _score_direct(an)
                if np.isfinite(sc) and sc < bs - 1e-12:
                    best = an
                    bs = sc
                    improved = True
                    break
            stall = 0 if improved else stall + 1
            if not improved and stall > 55:
                cand = best.copy()
                cand *= np.exp(0.07 * rng.standard_normal(n))
                cand = _normalize_floor(cand)
                sc = _score_direct(cand)
                if np.isfinite(sc) and sc < bs - 1e-12:
                    best = cand
                    bs = sc
                    stall = 0
        except Exception:
            break
    return best, bs


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    start = time.monotonic()
    try:
        budget_s = max(5.0, float(budget_s))
    except Exception:
        budget_s = 1300.0
    safety = 32.0 if budget_s > 120 else max(4.0, 0.10 * budget_s)
    deadline = start + max(1.0, budget_s - safety)
    rng = np.random.default_rng(int(seed) if np.isfinite(float(seed)) else 42)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim != 1:
            p = p.ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if p.size > 0 and np.isfinite(s) and s > 1e-12:
            parent = _normalize_floor(p)
        else:
            parent = None
    except Exception:
        parent = None

    fallback = [0.5, 0.1, 0.4]
    fallback_arr = _normalize_floor(np.asarray(fallback, dtype=np.float64))

    try:
        fb_score = float(evaluate_sequence([float(x) for x in fallback_arr]))
    except Exception:
        fb_score = np.inf

    if parent is not None:
        try:
            parent_score = float(evaluate_sequence([float(x) for x in parent]))
            if np.isfinite(parent_score) and parent_score < fb_score:
                fallback_arr = parent.copy()
                fb_score = parent_score
        except Exception:
            pass

    best_arr = fallback_arr.copy()
    best_score = fb_score
    candidate_arrays = [fallback_arr.copy()]

    parent_n = int(parent.size) if parent is not None else 127
    if parent_n < 3:
        parent_n = 127
    if parent is None:
        parent_n = 255

    usable = max(1.0, deadline - start)
    lengths = [parent_n]
    if usable > 140.0:
        for L in (255, 383, 511, 767, 1023):
            if L > parent_n and L <= 1200:
                lengths.append(L)
    # Deduplicate while preserving order.
    seen = set()
    ordered_lengths = []
    for L in lengths:
        L = int(max(3, L))
        if L not in seen:
            seen.add(L)
            ordered_lengths.append(L)

    weights = []
    for i, L in enumerate(ordered_lengths):
        if i == 0:
            w = 0.46 if len(ordered_lengths) > 1 else 1.0
        else:
            w = max(0.04, 0.26 * (0.78 ** (i - 1)))
        weights.append(w)
    total_w = float(sum(weights))
    if total_w <= 0:
        weights = [1.0]
        total_w = 1.0

    current_time = time.monotonic()
    for k, L in enumerate(ordered_lengths):
        if current_time >= deadline - 25.0:
            break
        frac_time = (weights[k] / total_w) * 0.92 * max(1.0, deadline - start)
        stage_end = min(deadline - 12.0, current_time + frac_time)
        if stage_end <= current_time + 0.8:
            continue

        # Quick initial candidate for this length.
        init_arr = _upsample(parent if parent is not None else best_arr, L)
        try:
            sc0 = _score_fft(init_arr)
        except Exception:
            sc0 = np.inf
        candidate_arrays.append(_normalize_floor(init_arr))

        try:
            inits = _make_inits(L, rng, parent if parent is not None else best_arr)
            # Put the best available initialization first.
            scs = []
            for ix, aa in enumerate(inits):
                try:
                    ss = _score_fft(aa)
                except Exception:
                    ss = np.inf
                scs.append(ss)
            if scs:
                best_init_idx = int(np.argmin(scs))
                inits.insert(0, inits.pop(best_init_idx))

            stage_start = time.monotonic()
            if stage_end < stage_start + 1.0:
                stage_end = stage_start + max(0.5, min(10.0, deadline - stage_start - 8.0))
            base_lr = 0.72 if L <= 256 else (0.62 if L <= 600 else 0.50)
            out, out_score = _optimize_stage(init_arr, L, stage_start, stage_end, rng, inits, base_lr)
            out = _normalize_floor(out)
            candidate_arrays.append(out.copy())

            # Exact re-score of this stage's best and update global.
            try:
                lst = [float(x) for x in out]
                sc = float(evaluate_sequence(lst))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_arr = out.copy()
            except Exception:
                pass
        except Exception:
            pass

        current_time = time.monotonic()
        if best_score <= 1.50274349 and current_time >= deadline - 80.0:
            break

    # Final exact validation over all collected candidates.
    final_list = None
    final_score = np.inf
    unique = []
    sigs = set()
    for arr in candidate_arrays:
        try:
            a = _normalize_floor(arr)
            if a.size == 0:
                continue
            key = (int(a.size), tuple(round(float(x), 12) for x in a[:12] + a[-12:]))
            if key in sigs:
                continue
            sigs.add(key)
            unique.append(a)
        except Exception:
            continue

    # Include fallback and parent again for safety.
    unique.append(fallback_arr.copy())
    if parent is not None:
        unique.append(parent.copy())

    for a in unique:
        try:
            lst = [float(x) for x in a]
            if not lst:
                continue
            if any(not np.isfinite(x) for x in lst):
                continue
            sc = float(evaluate_sequence(lst))
            if np.isfinite(sc) and sc < final_score:
                final_score = sc
                final_list = lst
        except Exception:
            continue

    if final_list is None:
        final_list = [float(x) for x in fallback_arr]

    # If there is meaningful time left, try an exact direct-gradient polish on the
    # globally best dispatch/list.
    try:
        arr = np.asarray(final_list, dtype=np.float64)
        if int(arr.size) <= 750 and time.monotonic() < deadline - 40.0:
            pol, pol_score = _final_local_search(arr, deadline - 6.0, rng)
            pol_list = [float(x) for x in pol]
            sc = float(evaluate_sequence(pol_list))
            if np.isfinite(sc) and sc < final_score:
                final_score = sc
                final_list = pol_list
    except Exception:
        pass

    # Last-resort validity check.
    try:
        out = []
        for x in np.asarray(final_list, dtype=np.float64):
            if not np.isfinite(float(x)):
                out = [0.5, 0.1, 0.4]
                break
            out.append(float(x))
        if not out:
            out = [0.5, 0.1, 0.4]
        if time.monotonic() < deadline - 2.0:
            sc = float(evaluate_sequence(out))
            if not np.isfinite(sc):
                out = [float(x) for x in fallback_arr]
        return out
    except Exception:
        return [0.5, 0.1, 0.4]
