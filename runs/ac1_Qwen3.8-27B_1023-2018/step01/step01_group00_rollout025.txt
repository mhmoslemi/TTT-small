import time
import math
from scipy.fft import rfft, irfft, next_fast_len

    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(z.size, 1.0 / z.size)
    return e / s

    n = len(a)
    nf = next_fast_len(2*n - 1)
    F = rfft(a, nf)
    c = irfft(F*F, nf)[:2*n-1]
    return c

    if s < 0.01:
        return math.inf
    b = _conv(a)
    return 2.0 * len(a) * b.max() / (s*s)

def _grad_softmax(a, tau):
    """
    Gradient of soft_max(b, tau) w.r.t. z where a = softmax(z).
    Returns (grad_z, b, softmax_value)
    """
    n = len(a)
    b = _conv(a)  # length 2n-1
    
    # Soft-max weights
    m = b.max()
    x = (b - m) / tau
    ex = np.exp(x)
    p = ex / ex.sum()
    U = m + tau * math.log(ex.sum())
    
    # G[j] = sum_k p[k] * a[k-j] (cross-correlation)
    # Using FFT
    N = next_fast_len(3*n)
    P = rfft(p, N)
    A = rfft(a, N)
    G = irfft(P * np.conj(A), N)[:n]
    
    grad_a = 2.0 * G
    S = a @ grad_a
    grad_z = a * (grad_a - S)
    grad_z[~np.isfinite(grad_z)] = 0.0
    
    return grad_z, b, U

def _init_z(n, variant, rng, parent):
    """Generate initial z for log-softmax parameterization."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-8
    
    if variant == 0 and parent is not None and parent.size > 0:
        # Resample parent
        idx = (np.arange(n) * parent.size // n).astype(int)
        a = np.maximum(parent[idx], 0.0)
        a = a + eps
        z = np.log(a)
    elif variant == 1 and parent is not None and parent.size > 0:
        # Interpolate parent
        x = np.linspace(0, parent.size - 1, n)
        a = np.interp(x, np.arange(parent.size), parent)
        a = np.maximum(a, 0.0) + eps
        z = np.log(a)
    elif variant == 2:

        z = np.zeros(n)
        n_freq = rng.integers(2, 6)
        for _ in range(n_freq):

    elif variant == 3:

        a = w * (1.0 + 0.7*np.sin(2*math.pi*f*i/n + ph)) + 0.1
        a = np.maximum(a, eps)
        z = np.log(a)
    elif variant == 5:

        # Smooth

        z = np.convolve(z, kernel, 'same')
    else:

        z = rng.normal(0, 1, n)

def _adam_optimize(n, seed, deadline, parent, init_variant=0):
    """Optimize z using Adam to minimize soft-max of convolution."""
    rng = np.random.default_rng(seed)
    n_fft = next_fast_len(3*n)
    
    z = _init_z(n, init_variant, rng, parent)
    a = _softmax_z(z)
    
    # Initial score
    b = _conv(a)
    score = 2.0 * n * b.max() / (a.sum()**2)
    best_score = score
    best_z = z.copy()

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    
    # Learning rate schedule
    lr_start = 0.5
    lr_end = 0.005
    lr_decay = 0.995
    lr = lr_start
    
    # Temperature schedule
    bmax = b.max()
    tau = 0.1 * bmax
    tau_min = 1e-6
    
    t0 = time.monotonic()
    stuck = 0
    last_improve = t0
    
    while time.monotonic() < deadline - 1.0:
        # Compute gradient
        grad_z, b, U = _grad_softmax(a, tau)

        t += 1
        m_adam = beta1*m_adam + (1-beta1)*grad_z
        v_adam = beta2*v_adam + (1-beta2)*(grad_z**2)
        m_hat = m_adam / (1 - beta1**t)
        v_hat = v_adam / (1 - beta2**t)
        step = m_hat / (np.sqrt(v_hat) + eps_adam)
        step[~np.isfinite(step)] = 0

            z_new = z_new - z_new.mean()  # Remove DC (sum normalization)
            z_new = np.clip(z_new, -12, 12)

            b_new = _conv(a_new)

            else:

                best_val = mid
                hi = mid

                lo = mid

        s = a.sum()
        score = 2.0 * n * _conv(a).max() / (s*s) if s >= 0.01 else math.inf
        
        if score < best_score:
            best = a.copy()
            best_score = score
    
    return best, best_score
