def lp_improve(a, M, max_time=10):
    """Solve LP: max sum(g) s.t. g>=0, (a*g)[k] <= M"""
    import cvxpy as cp
    g = cp.Variable(n, nonneg=True)
    # Constraint: (a*g)[k] <= M for all k
    # (a*g)[k] = sum_j a[j]*g[k-j]
    # We can express this as: A @ g <= M for all k
    # A is a (2n-1) x n matrix where A[k,j] = a[k-j] if valid
    
    # Build A sparsely
    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    # For row k, nonzeros are at columns j where max(0,k-n+1) <= j <= min(n-1,k)
    
    # This is a banded matrix. Let's use scipy sparse.
    from scipy.sparse import coo_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [A @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    
    start = time.monotonic()
    prob.solve(solver=cp.SCS, max_iters=10000)  # or ECOS, PDLP
    elapsed = time.monotonic() - start
    
    if g.value is not None and elapsed < max_time:
        g_val = np.maximum(g.value, 0)
        g_val = g_val / g_val.sum() if g_val.sum() > 0 else None
        if g_val is not None:
            return g_val
    return None
