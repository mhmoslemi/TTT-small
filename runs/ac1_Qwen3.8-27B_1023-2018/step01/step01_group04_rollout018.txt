Then `d(soft_max)/da_j = (1/tau) * sum_k exp(tau*b_k) * db_k/da_j / sum_k exp(tau*b_k)`
`= sum_k w_k * db_k/da_j`

where `w_k = exp(tau*b_k) / sum exp(tau*b_k)` (softmax weights)

And `db_k/da_j = a_{j+k} + a_{j-k}` (with bounds, as derived above)

For the full score gradient (with S in denominator):
`d(score)/da_j = 2n/S^2 * d(soft_max)/da_j - 2n*soft_max*2/S^3 * a_j`
`= (2n/S^3) * (S * sum_k w_k * (a_{j+k} + a_{j-k}) - 2*soft_max*a_j)`

This is O(n^2) in the worst case (if many lags are active), but with softmax the weights decay quickly so few lags are truly active.

For efficiency, I can compute the convolution of `a` with the weighted lags:
`sum_k w_k * (a_{j+k} + a_{j-k})` is a weighted sum of shifted a's.

I can compute this as: for each active lag k, add `w_k * shift(a, k) + w_k * shift(a, -k)` to the gradient.

For n up to ~512, this is O(n * n_active) which is fine.

import numpy as np
import time
import math

try:
    from scipy.optimize import linprog
    import scipy.sparse as sps
    from scipy.fft import next_fast_len, fft, ifft
except Exception:
    linprog = None
    sps = None
    next_fast_len = None
    fft = None
    ifft = None

TARGET = 1.50274349


def _autocorr(a):
    n = a.size
    if n > 48 and next_fast_len is not None:
        size = next_fast_len(2 * n - 1)
        A = fft(a, size)
        b = ifft(A * np.conj(A)).real[:2 * n - 1]
        return b
    return np.convolve(a, a)


def _score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    n = a.size
    if n == 0:
        return float("inf")
    s = float(np.sum(a))
    if s <= 1e-15:
        return float("inf")
    b = _autocorr(a)
    mb = float(np.max(b))
    if not math.isfinite(mb):
        return float("inf")
    return float(2.0 * n * mb / (s * s))


def _update_best(seq, best):
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
        arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size == 0:
            return
        s = float(np.sum(arr))
        if s <= 1e-12:
            arr = np.ones_like(arr, dtype=np.float64)
        sc = evaluate_sequence([float(x) for x in arr])
        if math.isfinite(sc) and sc < float(best[0]):
            best[0] = float(sc)
            best[1] = arr.copy()
    except Exception:
        pass


def _computegrad(a, tau=15.0):
    """Compute gradient of soft-max score w.r.t. a."""
    n = a.size
    S = float(np.sum(a))
    if S <= 1e-15:
        return None
    b = _autocorr(a)
    M = float(np.max(b))
    if not math.isfinite(M) or M <= 0:
        return None

    # Softmax weights over lags
    shifted = b - M
    shifted = np.clip(shifted, -50.0, 0.0)
    w = np.exp(tau * shifted)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= wsum

    # Compute grad_conv[j] = sum_k w_k * (a[j+k] + a[j-k])
    grad_conv = np.zeros(n, dtype=np.float64)
    active = np.where(w > 1e-14)[0]
    for k in active:
        k = int(k)
        wk = float(w[k])
        if wk < 1e-14:
            continue
        if k >= n:
            k2 = 2 * (n - 1) - k
        else:
            k2 = k
        if k2 == 0:
            grad_conv += wk * 2.0 * a
        else:
            grad_conv[:n - k2] += wk * a[k2:]
            grad_conv[k2:] += wk * a[:n - k2]

    # Score gradient: (2n/S^3) * (S * grad_conv - 2*M*a)
    grad = (2.0 * n / (S ** 3)) * (S * grad_conv - 2.0 * M * a)
    return grad, M, S


def _gradient_optimize(a0, time_limit, rng, best, tau=15.0):
    """Gradient-based optimization with line search."""
    n = a0.size
    if n < 2 or time_limit <= 0:
        return
    stop = time.monotonic() + time_limit
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s > 0:
        a /= s
    local = _score_fast(a)
    if math.isfinite(local):
        _update_best(a, best)

    base_lr = 1.0 / math.sqrt(n)
    stale = 0
    it = 0
    while time.monotonic() < stop - 2:
        it += 1
        res = _computegrad(a, tau)
        if res is None:
            break
        grad, M, S = res

        # Normalize gradient for step size
        gnorm = float(np.linalg.norm(grad))
        if not math.isfinite(gnorm) or gnorm < 1e-30:
            break

        # Adaptive learning rate
        lr = base_lr / (gnorm + 1e-10)
        lr = min(lr, 2.0 / max(n, 1))

        best_sc = local
        best_a = a.copy()
        found = False

        for mult in (1.0, 0.5, 0.25, 0.1, 0.05):
            step = lr * mult
            cand = a - step * grad
            cand = np.clip(cand, 0.0, None)
            cs = float(np.sum(cand))
            if cs > 1e-15:
                cand /= cs
                sc = _score_fast(cand)
                if math.isfinite(sc) and sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = cand.copy()
                    found = True
                    break

        if found and best_sc < local * (1.0 - 1e-10):
            a = best_a
            local = best_sc
            stale = 0
            if local < float(best[0]) + 1e-12:
                _update_best(a, best)
            if float(best[0]) < TARGET and local < TARGET * 1.001:
                break
        else:
            stale += 1
            if stale > 40:
                # Perturb to escape
                pert = 0.05 * base_lr * rng.standard_normal(n)
                ap = np.clip(a + pert, 0.0, None)
                aps = float(np.sum(ap))
                if aps > 1e-15:
                    ap /= aps
                    scp = _score_fast(ap)
                    if math.isfinite(scp) and scp < local * 1.05:
                        a = ap
                        local = scp
                        stale = 0
                        _update_best(a, best)
                if stale > 80:
                    break
    return a


def _lp_update(a, tau=10.0):
    """LP-based dual update: find g maximizing sum(g) s.t. (a*g)_k <= M, g>=0."""
    if linprog is None or sps is None:
        return None
    n = a.size
    b = _autocorr(a)
    M = float(np.max(b))
    if not math.isfinite(M) or M <= 0:
        return None

    # Build constraint matrix: (a * g)_k = sum_j a_{k-j} * g_j <= M for k = 0..2n-2
    # (a*g)[k] = sum_j a[k-j] * g_j, valid when 0<=k-j<=n-1, i.e., j in [max(0,k-n+1), min(k,n-1)]
    # Build sparse matrix A of shape (2n-1, n)
    rows = []
    cols = []
    data = []
    for j in range(n):
        for k in range(j, j + n):  # k-j in [0, n-1]
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    if len(rows) == 0:
        return None
    A = sps.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M + 1e-15, dtype=np.float64)

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0.0, 1000.0)] * n,
            method="highs",
            options={"presolve": True, "dual_feasibility_tolerance": 1e-8},
        )
    except Exception:
        return None

    if res is None or getattr(res, "x", None) is None or not res.success:
        return None

    g = np.asarray(res.x, dtype=np.float64).ravel()
    g = np.clip(g, 0.0, None)
    sg = float(g.sum())
    if sg <= 1e-15:
        return None
    return g / sg


def _lp_iteration(a0, time_limit, rng, best):
    """Iterative LP-based optimization."""
    n = a0.size
    if n < 4 or time_limit <= 0:
        return
    stop = time.monotonic() + time_limit
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s > 0:
        a /= s
    local = _score_fast(a)
    if math.isfinite(local):
        _update_best(a, best)

    stagnant = 0
    max_iters = 200
    for it in range(max_iters):
        if time.monotonic() >= stop - 3:
            break
        if n > 128:
            # Too large for dense LP; use gradient instead
            a = _gradient_optimize(a, min(5.0, stop - time.monotonic() - 3), rng, best, tau=12.0)
            local = _score_fast(a)
            if local < float(best[0]) + 1e-12:
                _update_best(a, best)
            if float(best[0]) < TARGET * 1.001:
                break
            continue

        g = _lp_update(a)
        if g is None:
            break

        best_sc = local
        best_cand = None
        for t in (0.15, 0.3, 0.5, 0.7, 1.0):
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0.0, None)
            cs = float(np.sum(cand))
            if cs > 1e-15:
                cand /= cs
                sc = _score_fast(cand)
                if math.isfinite(sc) and sc < best_sc:

                    best_cand = cand

        if best_cand is not None and best_sc < local * (1.0 - 1e-9):
            a = best_cand
            local = best_sc
            stagnant = 0
            if local < float(best[0]) + 1e-12:
                _update_best(a, best)
            if float(best[0]) < TARGET * 1.001 and it > 5:

        else:
            stagnant += 1
            if stagnant % 15 == 0 and stagnant < 60:
                pa = a * np.exp(0.08 * rng.standard_normal(n))
                pa = np.clip(pa, 0.0, None)

                        local = psa
                        stagnant = 0

def _interpolate(a, n_new):
    """Interpolate sequence to new length."""
    if a.size < 2:
        return np.ones(n_new, dtype=np.float64) / n_new
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, n_new)
    g = np.interp(new_x, old_x, a)
    gs = float(g.sum())
    if gs > 0:
        g /= gs
    else:
        g = np.ones(n_new, dtype=np.float64) / n_new
    return g


def _make_starts(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    starts = []

    def add(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0.0, None)
        if a.size != n:
            return
        s = float(a.sum())
        if s > 1e-15:
            starts.append(a / s)

    # Uniform
    add(np.ones(n))
    # Raised cosine
    x = np.linspace(0.0, 1.0, n)
    add(1.0 + 0.5 * np.cos(2.0 * math.pi * x))
    add(1.0 + 0.8 * np.cos(2.0 * math.pi * x))
    # Linear ramps
    add(x)
    add(1.0 - x)
    # Symmetric shape
    add(1.0 - 2.0 * np.abs(x - 0.5))
    # Quadratic
    add(1.0 - (2.0 * x - 1.0) ** 2)
    # Multi-frequency
    for k in (1, 2, 3, 5):
        for amp in (0.3, 0.5, 0.7, 0.9):
            add(1.0 + amp * np.sin(2.0 * math.pi * k * (x - 0.5)))
            add(1.0 + amp * np.cos(2.0 * math.pi * k * (x - 0.5)))
    # Dirichlet
    for alpha in (0.3, 0.5, 1.0, 2.0):
        try:
            add(rng.dirichlet(np.full(n, alpha)))
        except Exception:
            pass
    # Parent interpolation
    if parent is not None:
        try:
            p = np.asarray(parent, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
            p = np.clip(p, 0.0, None)
            if p.size >= 2 and p.sum() > 0:
                tp = np.linspace(0.0, 1.0, p.size)
                gp = np.interp(x, tp, p)
                add(gp)
                gp2 = gp * np.exp(0.03 * rng.standard_normal(n))
                add(gp2)
        except Exception:
            pass

    if not starts:
        starts.append(np.ones(n) / n)

    scored = []
    for a in starts:
        sc = _score_fast(a)
        if math.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda z: z[0])
    return [a for _, a in scored[:20]]
