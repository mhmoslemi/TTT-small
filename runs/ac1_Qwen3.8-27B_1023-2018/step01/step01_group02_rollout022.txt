    z = np.where(np.isfinite(z), z, 0.0)
    return z, m, v, t


def _local_search(a, rng, n_trials=50, max_s=None, t_start=None, t_end=None):
    """Random local perturbations to polish a candidate."""
    a = np.asarray(a, dtype=np.float64).copy()
    best = a.copy()
    best_s = _score_arr(best)
    if not np.isfinite(best_s):
        return best, best_s

    for _ in range(n_trials):
        if t_end is not None and time.monotonic() > t_end:
            break
        cand = best.copy()
        for _ in range(rng.integers(1, 4)):
            lo = max(1, best.size // 200)
            hi = max(lo + 1, min(best.size, max(3, best.size // 10)))
            w = int(rng.integers(lo, hi + 1))
            w = min(w, best.size)
            st = int(rng.integers(0, max(1, best.size - w + 1)))
            f = float(np.exp(rng.normal(0, 0.25)))
            cand[st:st + w] *= f
        if best.size > 4 and rng.random() < 0.3:
            # Swap two random elements
            i, j = rng.integers(0, best.size, 2)
            if i != j:
                cand[i], cand[j] = cand[j], cand[i]
        if best.size > 8 and rng.random() < 0.2:
            i = int(rng.integers(0, best.size - 1))
            j = int(rng.integers(0, best.size - 1))
            # Robust swap: mix values
            old_i, old_j = cand[i], cand[j]
            cand[i] = old_i * 0.5 + old_j * 0.5
            cand[j] = old_j * 0.5 + old_i * 0.5
        cand = np.maximum(cand, 0.0)
        s = float(cand.sum())
        if s > 1e-10:
            cand = cand / s
        sc = _score_arr(cand)
        if np.isfinite(sc) and sc < best_s - 1e-12:
            best_s = sc
            best = cand
    return best, best_s


def _optimize_from_init(a0, n, t_start, t_end, rng, init_lr=0.4, init_beta=500.0):
    """Adam optimization in logit space with adaptive beta."""
    a0 = np.asarray(a0, dtype=np.float64).copy()
    a0 = np.maximum(a0, 0.0)
    s0 = float(a0.sum())
    if s0 < 1e-10:
        a0 = np.ones(n) / n
        s0 = 1.0
    a0 = a0 / s0

    best_arr = a0.copy()
    best_s = _score_arr(best_arr)
    if not np.isfinite(best_s):
        best_s = 100.0
        best_arr = np.ones(n) / n
        best_s = _score_arr(best_arr)

    z = _logit(best_arr)
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    lr = init_lr
    beta = init_beta
    patience = 0
    max_patience = 800

    while time.monotonic() < t_end:
        a = _softmax(z)
        s_val = _score_arr(a)

        if np.isfinite(s_val) and s_val < best_s - 1e-10:
            best_s = s_val
            best_arr = a.copy()
            patience = 0
            lr = min(0.8, lr * 1.005)
        else:
            patience += 1

        # Adaptive beta: increase over time
        elapsed_frac = min(1.0, (time.monotonic() - t_start) / max(0.1, t_end - t_start))
        beta = 500.0 + 40000.0 * elapsed_frac * elapsed_frac

        g = _grad_smooth(a, beta)

        z, m_adam, v_adam, t_adam = _adam_update(z, g, m_adam, v_adam, t_adam, lr)

        if patience > max_patience:
            # Restart with perturbation
            zr = _logit(best_arr)
            if rng.random() < 0.3:
                zr = zr[::-1].copy()
            z = zr + float(np.exp(rng.normal(0, 0.4))) * rng.standard_normal(n) * 0.3
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            t_adam = 0
            patience = 0
            lr = max(0.12, lr * 0.65)
            max_patience = max(300, int(max_patience * 1.2))

        # Periodic local search
        if patience > 200 and rng.random() < 0.05:
            polished, ps = _local_search(best_arr, rng, n_trials=30,
                                         t_end=time.monotonic() + 2.0)
            if ps < best_s - 1e-10:
                best_s = ps
                best_arr = polished
                z = _logit(best_arr) + 0.05 * rng.standard_normal(n)
                m_adam = np.zeros(n)
                v_adam = np.zeros(n)
                t_adam = 0
                patience = 0

    return best_arr, best_s


def _try_length(n, t_alloc, rng, inits):
    """Optimize at a specific length, allocating time across initializations."""
    if n <= 0 or t_alloc < 2.0:
        return None, np.inf

    best_arr = None
    best_s = np.inf

    n_inits = len(inits)
    time_per_init = max(1.0, t_alloc * 0.7 / max(1, n_inits))

    for a0 in inits:
        if time.monotonic() > time.time() + t_alloc:  # safety
            break
        a0 = np.asarray(a0, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = float(a0.sum())
        if s < 1e-10:
            a0 = np.ones(n) / n
            s = 1.0
        a0 = a0 / s
        if a0.size != n:
            continue

        t_start = time.monotonic()
        t_end = t_start + time_per_init
        arr, sc = _optimize_from_init(a0, n, t_start, t_end, rng)
        if np.isfinite(sc) and sc < best_s:
            best_s = sc
            best_arr = arr

    # Final polish
    if best_arr is not None and time.monotonic() < time.time() + t_alloc - 1:
        best_arr, best_s = _local_search(best_arr, rng, n_trials=80,
                                         t_end=time.time() + t_alloc - 0.5)

    if best_arr is not None:
        best_arr = np.maximum(best_arr, 0.0)
        s = float(best_arr.sum())
        if s > 1e-10:
            best_arr = best_arr / s
        best_s = _score_arr(best_arr)

    return best_arr, best_s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(5.0, float(budget_s) - 8.0)

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            ps = float(parent.sum())
            if ps > 1e-10:
                parent = parent / ps
            else:
                parent = None
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(127, dtype=np.float64) / 127.0

    n_parent = parent.size
    best_arr = parent.copy()
    best_s = _score_arr(best_arr)

    # Fallback
    fallback = [0.5, 0.1, 0.4]
    fallback_s = _score_arr(fallback)

    try:
        # Phase 1: Optimize at parent length (bulk of time)
        n1 = n_parent
        t1_alloc = 0.55 * (deadline - t0)
        if t1_alloc > 3.0:
            inits1 = [parent.copy()]
            # Add perturbations
            for amp in (0.05, 0.15, 0.3, 0.5):
                pert = parent * (1.0 + amp * rng.standard_normal(n1))
                pert = np.maximum(pert, 0.0)
                ps = float(pert.sum())
                if ps > 1e-10:
                    inits1.append(pert / ps)
            # Uniform
            inits1.append(np.ones(n1) / n1)
            # Softmax random
            z_r = rng.normal(0, 0.5, n1)
            inits1.append(_softmax(z_r))
            # Power shape
            x = np.linspace(0, 1, n1)
            for pw in (0.3, 0.7, 1.5, -0.5):
                vals = 1.0 + 0.5 * np.power(x + 0.01, pw) if pw >= 0 else 1.0 + 0.5 * np.power(1.0 - x + 0.01, -pw)
                vals = np.maximum(vals, 0.01)
                inits1.append(vals / vals.sum())

            if len(inits1) > 10:
                inits1 = inits1[:10]

            arr1, s1 = _try_length(n1, t1_alloc, rng, inits1)
            if arr1 is not None and np.isfinite(s1) and s1 < best_s:
                best_s = s1
                best_arr = arr1.copy()

        # Phase 2: Try a different length
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            # Try a smaller or larger length
            for n_try in (64, 100, 192, 253):
                if n_try == n1 or n_try < 3:
                    continue
                t2_alloc = min(remaining * 0.3, max(5.0, 0.25 * (deadline - t0)))
                if t2_alloc < 5.0:
                    break

                # Upsample parent to new length
                if parent.size > 0:
                    x1 = np.linspace(0, 1, n_try)
                    x0 = np.linspace(0, 1, parent.size)
                    p_up = np.interp(x1, x0, parent)
                else:
                    p_up = np.ones(n_try) / n_try
                p_up = np.maximum(p_up, 0.0)
                p_up = p_up / p_up.sum()

                inits2 = [p_up]
                inits2.append(np.ones(n_try) / n_try)
                z_r2 = rng.normal(0, 0.8, n_try)
                inits2.append(_softmax(z_r2))
                # Bump shape
                x_b = np.linspace(0, 1, n_try)
                bump = 1.0 + 0.4 * np.cos(np.pi * x_b)
                bump = np.maximum(bump, 0.05)
                inits2.append(bump / bump.sum())

                arr2, s2 = _try_length(n_try, t2_alloc, rng, inits2)
                if arr2 is not None and np.isfinite(s2) and s2 < best_s:
                    best_s = s2
                    best_arr = arr2.copy()

                remaining = deadline - time.monotonic()
                if remaining < 15.0:
                    break

        # Phase 3: Final intensive local search on best
        remaining = deadline - time.monotonic()
        if remaining > 5.0:
            best_arr, best_s = _local_search(best_arr, rng, n_trials=300,
                                             t_end=time.monotonic() + remaining - 2.0)
            best_arr = np.maximum(best_arr, 0.0)
            bs = float(best_arr.sum())
            if bs > 1e-10:
                best_arr = best_arr / bs
            best_s = _score_arr(best_arr)

        # Phase 4: One more round of Adam if time permits
        remaining = deadline - time.monotonic()
        if remaining > 15.0:
            n_best = best_arr.size
            inits_final = [best_arr.copy()]
            z_pert = _logit(best_arr) + 0.1 * rng.standard_normal(n_best)
            inits_final.append(_softmax(z_pert))
            arr_f, s_f = _try_length(n_best, remaining - 3.0, rng, inits_final)
            if arr_f is not None and np.isfinite(s_f) and s_f < best_s:
                best_s = s_f
                best_arr = arr_f.copy()

    except Exception:
        pass

    # Validate and return
    final_list = [float(x) for x in np.asarray(best_arr, dtype=np.float64)]
    if not final_list or any(not np.isfinite(x) for x in final_list):
        final_list = fallback

    # Final score check
    final_s = _score_arr(final_list)
    if not np.isfinite(final_s) or final_s > fallback_s:
        final_list = fallback
        final_s = fallback_s

    # One more validation with exact evaluator
    try:
        exact_s = float(evaluate_sequence(final_list))
        if not np.isfinite(exact_s):
            final_list = fallback
    except Exception:
        final_list = fallback

    return final_list
