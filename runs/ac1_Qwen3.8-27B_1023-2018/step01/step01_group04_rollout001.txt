def _safe_array(x):
    if x is None:
        return np.array([], dtype=np.float64)
    try:
        a = np.asarray(x, dtype=np.float64).ravel()
    except Exception:

def _normalize(a):
    a = _safe_array(a)
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    s = float(np.sum(a))
    if not math.isfinite(s) or s <= 1e-15:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def _interpolate_sequence(a, n):
    a = _safe_array(a)
    n = int(n)
    if n <= 0:
        return np.array([1.0], dtype=np.float64)
    if a.size == 0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)

    old = np.linspace(0.0, 1.0, a.size, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    g = np.interp(new, old, a)

    s = float(np.sum(g))

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    if n == 1:
        return np.ones(1, dtype=np.float64)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) - (cssv - 1.0)
    nz = np.nonzero(cond > 0.0)[0]
    if nz.size == 0:
        y = np.clip(v, 0.0, None)
        s = float(np.sum(y))
        if not math.isfinite(s) or s <= 1e-15:
            return np.full(n, 1.0 / float(n), dtype=np.float64)
        return y / s

    rho = int(nz[-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    y = np.maximum(v - theta, 0.0)
    s = float(np.sum(y))
    if not math.isfinite(s) or s <= 1e-15:
        y = np.clip(v, 0.0, None)
        s = float(np.sum(y))
        if not math.isfinite(s) or s <= 1e-15:
            return np.full(n, 1.0 / float(n), dtype=np.float64)
    return y / s


def _correlate_weights(a, w):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([w[0] * a[0]], dtype=np.float64)
    try:
        c = np.correlate(w, a, mode="valid")
        c = np.asarray(c, dtype=np.float64)
        if c.size != n:
            c = c[:n]
        return c
    except Exception:
        out = np.zeros(n, dtype=np.float64)
        for i in range(n):
            out[i] = float(np.dot(a, w[i:i + n]))
        return out


def _logG_grad(a, b, p):
    """
    Return (log G_p, ambient gradient wrt a) for
    G_p(a) = (sum_k b_k^p)^(1/p), b = a*a, assuming sum(a)=1.
    The gradient is the exact gradient of log(G_p).
    """
    p = int(max(1, int(p)))
    if b is None:
        b = np.convolve(a, a)
    b = np.asarray(b, dtype=np.float64).ravel()
    if b.size == 0:
        return float("inf"), np.zeros(a.size, dtype=np.float64)

    np.maximum(b, 0.0, out=b)
    bmax = float(np.max(b))
    if not math.isfinite(bmax) or bmax <= 0.0:
        return float("inf"), np.zeros(a.size, dtype=np.float64)

    ratio = b / bmax
    if p == 1:
        s1 = np.ones_like(ratio, dtype=np.float64)
        s2 = float(np.sum(ratio))
    else:
        s1 = ratio ** (p - 1)
        s2 = float(np.sum(ratio ** p))

    if not math.isfinite(s2) or s2 <= 0.0:
        s1 = np.ones_like(ratio, dtype=np.float64)
        s2 = float(ratio.size)

    val = math.log(bmax) + (1.0 / float(p)) * math.log(s2)
    scale = 1.0 / (bmax * s2)
    if not math.isfinite(scale):
        return val, np.zeros(a.size, dtype=np.float64)

    g = 2.0 * scale * _correlate_weights(a, s1)
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    return val, g


def _smooth_stage(a, p, maxiter):

    a = _normalize(a)
    n = a.size

    p = int(max(2, int(p)))

    ap = np.maximum(a, 1e-300)
    ap = ap / max(1e-300, float(np.sum(ap)))
    pivot = int(np.argmax(ap))
    pivval = float(ap[pivot])
    if pivval <= 1e-300:

        ap = ap / max(1e-300, float(np.sum(ap)))
        pivot = int(np.argmax(ap))
        pivval = float(ap[pivot])

    idx = np.array([i for i in range(n) if i != pivot], dtype=int)
    x0 = np.log(ap[idx] / max(1e-300, pivval))
    x0 = np.nan_to_num(x0, nan=-20.0, posinf=20.0, neginf=-20.0)
    x0 = np.clip(x0, -80.0, 80.0)

        u[pivot] = 0.0
        u[idx] = x
        umax = float(np.max(u))

        aa = ez / max(1e-300, float(np.sum(ez)))
        b = np.convolve(aa, aa)
        val, grad_a = _logG_grad(aa, b, p)
        if not math.isfinite(val):
            return 1e18, np.zeros_like(x, dtype=np.float64)

        return float(val), gx

            fun,
            x0,
            method="L-BFGS-B",
            jac=True,

    u[idx] = x0

    aa = ez / max(1e-300, float(np.sum(ez)))
    return _normalize(aa)

                            if sc < float(best[0]) + 0.05:
                                a = cand
                                cur_score = sc
            if stale > 40:
                break
        except Exception:
            break

    _maybe_update_best(a, _score_norm(a), best)
    return a


def _smooth_refine(a, best, rng, stop):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return a
    if n > MAX_OPT_N:
        a = _interpolate_sequence(a, MAX_OPT_N)
        n = MAX_OPT_N
    if n < 1:
        return a

    _maybe_update_best(a, _score_norm(a), best)

    if minimize is not None and n <= MAX_OPT_N:
        stages = [(4, 90), (8, 90), (16, 70), (32, 55), (64, 40), (128, 30)]
        for p, iters in stages:
            if time.monotonic() >= stop - 8:
                break
            if n > 384:
                iters = int(max(12, iters * 0.6))
            try:
                a = _smooth_stage(a, p, iters)
                sc = _score_norm(a)
                _maybe_update_best(a, sc, best)
            except Exception:
                break

    for p in (8, 32, 128):
        if time.monotonic() >= stop - 4:
            break
        try:
            a = _exact_polish(a, best, p, stop, max_steps=180, rng=rng)
        except Exception:
            break

    _maybe_update_best(a, _score_norm(a), best)
    return a


def _lp_direction(a, rng, secondary=0.0, bias=None, bias_weight=1e-4):
    if linprog is None:
        return None, None
    a = _normalize(a)
    n = a.size
    if n < 2 or n > 900:
        return None, None
    m = 2 * n - 1

    try:
        A = np.zeros((m, n + 1), dtype=np.float64)
        A[:, -1] = -1.0
        for i, ai in enumerate(a):
            if ai > 0.0:
                A[i:i + n, i] = ai

        c = np.zeros(n + 1, dtype=np.float64)
        c[-1] = 1.0
        if secondary and rng is not None:
            c[:n] += float(secondary) * (rng.random(n) - 0.5)
        if bias is not None and bias_weight:
            bmat = np.asarray(bias, dtype=np.float64).ravel()
            if bmat.size == n:
                bmat = bmat - float(np.mean(bmat))
                bn = float(np.linalg.norm(bmat))
                if math.isfinite(bn) and bn > 1e-12:
                    c[:n] += float(bias_weight) * (bmat / bn)

        Aeq = np.zeros((1, n + 1), dtype=np.float64)
        Aeq[0, :n] = 1.0
        b_eq = np.array([1.0], dtype=np.float64)
        b_ub = np.zeros(m, dtype=np.float64)

        bounds = [(0.0, None)] * n
        bounds.append((0.0, None))

        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                A_eq=Aeq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
        except Exception:
            try:
                res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=Aeq, b_eq=b_eq, bounds=bounds, method="highs")
            except Exception:
                res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=Aeq, b_eq=b_eq, bounds=bounds)
    except Exception:
        return None, None

    if res is None:
        return None, None
    x = getattr(res, "x", None)
    if x is None:
        return None, None
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size <= n:
        return None, None
    g = x[:n]
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    g = np.clip(g, 0.0, None)
    s = float(np.sum(g))
    if not math.isfinite(s) or s <= 1e-15:
        return None, None
    g = g / s

    try:
        t = float(np.max(A[:n, :n].dot(g)))
    except Exception:
        t = None

    return g, t


def _mix_search(a, g):
    a = _normalize(a)
    g = _normalize(g)
    best_a = a
    best_sc = _score_norm(a)
    for alpha in (0.05, 0.10, 0.20, 0.35, 0.50, 0.65, 0.80, 0.90, 1.00):
        try:
            c = a + alpha * (g - a)
            c = np.clip(c, 0.0, None)
            c = c / max(1e-300, float(np.sum(c)))
            sc = _score_norm(c)
            if math.isfinite(sc) and sc < best_sc:
                best_a = c
                best_sc = sc
        except Exception:
            pass
    return best_a, best_sc


def _lp_loop(a_init, best, rng, stop):
    cur = _normalize(a_init)
    n = cur.size
    if n == 0 or n > 900:
        return cur

    cur_score = _score_norm(cur)
    local_best = cur.copy()
    local_best_score = cur_score
    _maybe_update_best(cur, cur_score, best)
    stale = 0

    for it in range(30000):
        if time.monotonic() >= stop:
            break
        if n > MAX_OPT_N and it == 0:
            return cur

        try:
            M = cur_score / (2.0 * float(n)) if cur_score > 0.0 else 0.0

            bias = None
            if it % 5 == 2:
                try:
                    b = np.convolve(cur, cur)
                    np.maximum(b, 0.0, out=b)
                    _, gvec = _logG_grad(cur, b, 8)
                    bias = gvec - float(np.mean(gvec))
                    if not np.all(np.isfinite(bias)):
                        bias = None
                except Exception:
                    bias = None

            secondary = 1.0e-4 * (0.5 + float(rng.random())) if it % 7 == 0 else 0.0
            g, t = _lp_direction(cur, rng, secondary=secondary, bias=bias, bias_weight=1e-4)
            if g is None:
                break

            cand, sc = _mix_search(cur, g)
            if math.isfinite(sc) and sc < local_best_score * (1.0 - 1e-12):
                cur = cand
                cur_score = sc
                local_best = cand.copy()
                local_best_score = sc
                _maybe_update_best(cur, cur_score, best)
                stale = 0
            else:
                staled = True
                if t is not None and math.isfinite(t) and M > 0.0 and t < M * (1.0 - 1e-5):
                    if stale == 2 or stale == 5:
                        small = 0.05 if stale == 2 else 0.10
                        cand = (1.0 - small) * cur + small * g
                        cand = _normalize(cand)
                        sc = _score_norm(cand)
                        if math.isfinite(sc):
                            if sc < local_best_score:
                                local_best = cand.copy()
                                local_best_score = sc
                            if sc < float(best[0]) + 0.05:
                                cur = cand
                                cur_score = sc
                            staled = False
                if staled or True:
                    stale += 1

                if stale % 8 == 0:
                    noise = 0.05 + 0.01 * min(10.0, float(stale))
                    cand = cur * np.exp(noise * rng.standard_normal(n))
                    cand = _normalize(cand)
                    sc = _score_norm(cand)
                    if math.isfinite(sc):
                        if sc < local_best_score:
                            local_best = cand.copy()
                            local_best_score = sc
                        if sc < float(best[0]) + 0.07:
                            cur = cand
                            cur_score = sc
                if stale > 70:
                    break

            if it % 6 == 0 and best[1].size == n:
                for lam in (0.25, 0.5, 0.75):
                    cand = (1.0 - lam) * cur + lam * best[1]
                    cand = _normalize(cand)
                    sc = _score_norm(cand)
                    if math.isfinite(sc):
                        _maybe_update_best(cand, sc, best)
                        if sc < local_best_score * (1.0 - 1e-12):
                            local_best = cand.copy()
                            local_best_score = sc
                            cur = cand
                            cur_score = sc
                            stale = 0
        except Exception:
            break

    _maybe_update_best(local_best, local_best_score, best)
    if time.monotonic() < stop - 8 and n <= MAX_OPT_N:
        try:
            cur = _exact_polish(cur, best, 32, stop, max_steps=80, rng=rng)
            cur = _exact_polish(cur, best, 128, stop, max_steps=60, rng=rng)
        except Exception:
            pass
    _maybe_update_best(cur, _score_norm(cur), best)
    return cur


def _make_starts(n, rng, parent, top=7):
    n = int(n)
    if n < 2:
        n = 2
    starts = []

    def add(a):
        try:
            a = _safe_array(a)
            if a.size != n:
                a = _interpolate_sequence(a, n)
            if a.size != n:
                return
            a = np.clip(a, 0.0, None)
            s = float(np.sum(a))
            if not math.isfinite(s) or s <= 1e-15:
                return
            a = a / s
            starts.append(a)
        except Exception:
            return

    if parent is not None and parent.size:
        add(_interpolate_sequence(parent, n))

    add(np.ones(n, dtype=np.float64))

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for c in (0.5, 1.0, 1.8, -0.5, -1.0, -1.8):
        add(np.exp(c * (x - 0.5)))

    half = max(1, n // 2)
    t = np.linspace(0.0, 1.0, half, dtype=np.float64)
    for amp in (0.35, 0.65, 0.90):
        for sign in (1.0, -1.0):
            v = sign * amp * (t - 0.5)
            a = np.ones(n, dtype=np.float64)
            if n % 2 == 0:
                a[:half] = 1.0 + v
                a[half:] = 1.0 - v[::-1]
            else:
                a[:half] = 1.0 + v
                a[half] = 1.0
                a[half + 1:] = 1.0 - v[::-1][:max(0, n - half - 1)]
            add(0.5 * a)

            v2 = sign * amp * np.sin(math.pi * t)
            a = np.ones(n, dtype=np.float64)
            if n % 2 == 0:
                a[:half] = 1.0 + v2
                a[half:] = 1.0 - v2[::-1]
            else:
                a[:half] = 1.0 + v2
                a[half] = 1.0
                a[half + 1:] = 1.0 - v2[::-1][:max(0, n - half - 1)]
            add(0.5 * a)

    for k in range(1, min(7, n // 2) + 1):
        for amp in (0.35, 0.65):
            add(1.0 + amp * np.sin(2.0 * math.pi * k * x))
            add(1.0 + amp * np.cos(2.0 * math.pi * k * x))

    for alpha in (0.75, 1.25):
        for _ in range(2):
            try:
                add(rng.dirichlet(np.full(n, alpha, dtype=np.float64)))
            except Exception:
                pass

    if parent is not None and parent.size:
        base = _interpolate_sequence(parent, n)
        try:
            add(base * np.exp(0.06 * rng.standard_normal(n)))
            add(base * np.exp(0.12 * rng.standard_normal(n)))
        except Exception:
            pass

    scored = []
    for a in starts:
        sc = _score_norm(a)
        if math.isfinite(sc):
            scored.append((sc, a))
    if not scored:
        scored.append((2.0, np.full(n, 1.0 / float(n), dtype=np.float64)))
    scored.sort(key=lambda z: z[0])
    return [a for _, a in scored[:max(1, int(top))]]


def _fallback_candidate():
    return [2.0 / 3.0, 1.0 / 3.0]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not math.isfinite(budget) or budget <= 0.0:
        budget = 1.0
    budget = max(0.5, min(budget, 5000.0))
    deadline = t0 + max(0.25, budget - 18.0)

    best = [float("inf"), np.array([2.0 / 3.0, 1.0 / 3.0], dtype=np.float64)]
    _update_best(best[1], best)
    _update_best(np.ones(96, dtype=np.float64), best)

    parent = _safe_array(globals().get("height_sequence_1", None))
    parent_exists = parent.size > 0
    if parent_exists:
        _update_best(parent, best)

    if best[1].size == 0:
        best[1] = np.array(_fallback_candidate(), dtype=np.float64)
        _update_best(best[1], best)

    if parent_exists:
        n_parent = int(parent.size)
        n_main = max(8, min(MAX_OPT_N, n_parent))
    else:
        n_main = 96 if best[1].size > MAX_OPT_N else max(8, min(MAX_OPT_N, int(best[1].size)))
    if n_main < 8:
        n_main = 8

    # Main length: repeated LP/block-descent from a few good starts.
    now = time.monotonic()
    avail = max(0.0, deadline - now)
    lp_time = min(450.0, max(0.0, avail - 340.0))
    if lp_time > 25.0 and n_main <= MAX_OPT_N:
        try:
            starts = _make_starts(n_main, rng, parent, top=8)
            cur_best_norm = _interpolate_sequence(best[1], n_main)
            starts.insert(0, cur_best_norm)
            if parent_exists:
                starts.insert(0, _interpolate_sequence(parent, n_main))

            # de-duplicate approximately
            ded = []
            for s in starts:
                dup = False
                for d in ded:
                    if d.size == s.size and np.linalg.norm(d - s) < 1e-6:
                        dup = True
                        break
                if not dup:
                    ded.append(s)
            starts = ded[:6]

            count = 3 if lp_time > 160.0 else 2 if lp_time > 60.0 else 1
            per = lp_time / float(max(1, count))
            for a0 in starts[:count]:
                stop = min(deadline - 220.0, time.monotonic() + max(20.0, per * 1.05))
                if time.monotonic() >= stop:
                    break
                _lp_loop(a0, best, rng, stop)
        except Exception:
            pass

    # Polish the current global best on its (or an optimized) length.
    now = time.monotonic()
    sm_time = min(210.0, max(0.0, deadline - now - 260.0))
    if sm_time > 35.0:
        try:
            a_start = _interpolate_sequence(best[1], min(MAX_OPT_N, max(int(best[1].size), 8)))
            a_start = _smooth_refine(a_start, best, rng, time.monotonic() + sm_time)
            _maybe_update_best(a_start, _score_norm(a_start), best)
        except Exception:
            pass

    # Explore nearby lengths, especially around the parent length.
    other_ns = []
    base = n_main
    cand_ns = [
        int(base * 2),
        128,
        int(base * 1.5),
        max(8, base // 2),
        64,
        256,
        max(8, base // 3),
        32,
    ]
    for v in cand_ns:
        v = int(v)
        if 8 <= v <= MAX_OPT_N and v != base and v not in other_ns:
            other_ns.append(v)
    other_ns = other_ns[:5]

    now = time.monotonic()
    ot_budget = min(290.0, max(0.0, deadline - now - 170.0))
    if len(other_ns) > 0 and ot_budget > 40.0:
        for idx, nn in enumerate(other_ns):
            time_left = deadline - time.monotonic()
            if time_left < 110.0:
                break
            per_ot = ot_budget / float(len(other_ns) - idx)
            stop = min(deadline - 120.0, time.monotonic() + max(25.0, min(90.0, per_ot * 1.1)))
            if time.monotonic() >= stop:
                break
            try:
                a0 = _interpolate_sequence(best[1], nn)
                if parent_exists and parent.size:
                    p0 = _interpolate_sequence(parent, nn)
                    cr, sc = _mix_search(a0, p0)
                    if sc < _score_norm(a0):
                        a0 = cr
                _lp_loop(a0, best, rng, stop)
            except Exception:
                pass

    # Restart/polish from current best with small multiplicative perturbations.
    now = time.monotonic()
    rs_budget = min(180.0, max(0.0, deadline - now - 70.0))
    if rs_budget > 45.0:
        try:
            n_cur = int(best[1].size)
            if n_cur < 8 or n_cur > MAX_OPT_N:
                n_cur = n_main
            a = _interpolate_sequence(best[1], min(MAX_OPT_N, max(8, n_cur)))

            restarts = max(1, int(rs_budget // 50) if rs_budget > 100 else 1)
            restarts = min(4, restarts)
            per_rs = rs_budget / float(max(1, restarts))
            for _ in range(restarts):
                stop = min(deadline - 45.0, time.monotonic() + max(18.0, per_rs * 0.85))
                if time.monotonic() >= stop:
                    break
                noise = 0.12 + 0.2 * float(rng.random())
                cand = a * np.exp(noise * rng.standard_normal(a.size))
                a = _normalize(cand)
                _lp_loop(a, best, rng, stop)

            final_time = max(0.0, deadline - time.monotonic() - 25.0)
            if final_time > 30.0:
                a = _interpolate_sequence(best[1], min(MAX_OPT_N, max(8, int(best[1].size))))
                a = _smooth_refine(a, best, rng, time.monotonic() + min(120.0, final_time))
        except Exception:
            pass

    # Final safe validation and return.
    final = _safe_array(best[1])
    if final.size == 0:
        final = np.array(_fallback_candidate(), dtype=np.float64)

    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    s = float(np.sum(final))
    if not math.isfinite(s) or s <= 0.01:
        final = np.ones(final.size if final.size > 0 else 2, dtype=np.float64)
        s = float(np.sum(final))
    final = final / s
    final = np.clip(final, 0.0, 1000.0)

    try:
        sc_final = evaluate_sequence([float(x) for x in final])
    except Exception:
        sc_final = _score_norm(final)

    if not math.isfinite(sc_final):
        final = np.array(_fallback_candidate(), dtype=np.float64)
        s = float(np.sum(final))
        if s > 0:
            final = final / s

    return [float(x) for x in final]
