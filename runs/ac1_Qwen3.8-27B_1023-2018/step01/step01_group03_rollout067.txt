import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _clean_norm(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a) & (a > 0.0), a, 0.0)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(a.shape, 1.0 / a.size, dtype=np.float64)
    a = a / s
    return np.clip(a, 0.0, 1000.0)


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if s < 0.01 or not np.isfinite(s):
        return float("inf")
    p = np.convolve(a, a)
    if p.size == 0:
        return float("inf")
    m = float(np.max(p))
    if not np.isfinite(m) or m < 0.0:
        m = 0.0
    return float(2.0 * n * m / (s * s))


def _resize(q, k):
    k = max(1, int(k))
    q = _clean_norm(q)
    if q.size == k:
        return q.copy()
    if q.size % k == 0 and q.size > k:
        out = q.reshape(k, -1).mean(axis=1)
    else:
        x = np.linspace(0.0, 1.0, q.size)
        y = np.linspace(0.0, 1.0, k)
        out = np.interp(y, x, q)
    return _clean_norm(out)


def _tile_pattern(k, pat=None):
    if pat is None:
        pat = np.array([0.2, 0.4, 0.2, 0.2], dtype=np.float64)
    k = max(1, int(k))
    pat = _clean_norm(pat)
    L = max(1, k // pat.size)
    out = np.repeat(pat, L)
    if out.size < k:
        out = np.concatenate([out, pat[: max(0, k - out.size)]])
    return _clean_norm(out[:k])


def _initial_start(k, rng):
    k = max(4, int(k))
    starts = [_tile_pattern(k)]
    starts.append(np.full(k, 1.0 / k, dtype=np.float64))

    t = np.linspace(0.0, 1.0, k)
    for c, w in ((0.5, 0.15), (0.35, 0.12), (0.65, 0.12)):
        v = np.exp(-((t - c) / max(1e-6, w)) ** 2)
        starts.append(_clean_norm(v))

    for alpha in (1.0, 0.5):
        try:
            starts.append(rng.dirichlet(np.full(k, alpha)))
        except Exception:
            pass

    best = starts[0]
    best_sc = _score(best)
    for v in starts[1:]:
        sc = _score(v)
        if sc < best_sc - 1e-14:
            best = v
            best_sc = sc
    return best


def _build_A_full(q, k):
    Af = np.zeros((2 * k - 1, k + 1), dtype=np.float64)
    if k == 1:
        Af[0, 0] = q[0]
        Af[0, 1] = -1.0
        return Af

    qr = q[::-1]
    Af[0, 0] = q[0]

    for r in range(1, k):
        Af[r, :r + 1] = qr[k - 1 - r:k]

    for r in range(k, 2 * k - 1):

        Af[r, s:k] = qr[0:k - s]

    rows = np.arange(2 * k - 1, dtype=np.int64)
    Af[rows, k] = -1.0
    return Af


def _lp_dir(q, k, mscale=1.0, max_seconds=30, deadline=None, bias=True):
    if linprog is None or k <= 0:
        return None

    p = np.convolve(q, q)
    if p.size == 0:
        return None
    M = float(np.max(p))
    if not np.isfinite(M) or M <= 0.0:
        return None

    t_ub = float(mscale) * M
    if not np.isfinite(t_ub) or t_ub <= 0.0:

def _descent_dir(q, k, T):
    p = np.convolve(q, q)
    if p.size == 0:
        return None
    m = float(np.max(p))
    if not np.isfinite(m) or m <= 0.0:
        return None

    T = float(max(1e-6, T))
    if T >= 1e5:
        thr = max(1e-30, 0.97 * m)
        w = (p >= thr).astype(np.float64)
        sw = float(w.sum())
        if sw <= 0.0:
            w = np.zeros_like(w)
            w[int(np.argmax(p))] = 1.0
        else:
            w /= sw
    else:
        z = (p - m) * T
        w = np.exp(z)
        sw = float(w.sum())
        if sw <= 0.0 or not np.isfinite(sw):
            w = np.zeros_like(w)
            w[int(np.argmax(p))] = 1.0
        else:
            w /= sw

    grad = np.zeros(k, dtype=np.float64)

    if T >= 1e5:
        idx = np.nonzero(w > 0.0)[0]
    else:
        idx = np.nonzero(w > 1e-9)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(p))], dtype=np.int64)
            w = np.zeros_like(w)
            w[idx[0]] = 1.0

    if idx.size > 1200:
        order = np.argsort(w[idx])[::-1]
        idx = idx[order[:1200]]

    for rr in idx:
        r = int(rr)
        wt = float(w[r])
        if wt <= 0.0:
            continue
        lo = max(0, r - k + 1)
        hi = min(r, k - 1)
        if lo > hi:
            continue
        grad[lo:hi + 1] += 2.0 * wt * q[r - hi:r - lo + 1][::-1]

    grad -= grad.mean()
    gn = float(np.linalg.norm(grad))
    if gn < 1e-14 or not np.isfinite(gn):
        return None
    return -grad / gn

                    bs = cs
                    improved = True
                    consider(best.copy())

        if improved:
            stag = 0
            ms = 1.0
        else:
            stag += 1
            ms = max(0.84, ms * 0.98)

            if stag in (6, 12, 20, 30, 45):
                r = rng.normal(size=k)
                r -= r.mean()
                rn = float(np.linalg.norm(r))
                if rn > 0.0:
                    cand = best * np.exp(np.clip(0.08 * (r / rn), -1.5, 1.5))
                    cand = _clean_norm(cand)
                    cs = _score(cand)
                    if cs < bs - 1e-13:
                        best = cand
                        bs = cs
                        consider(best.copy())

            if ms <= 0.85 and stag > 30:
                break

    now = time.monotonic()
    local_budget = min((1.0 - lp_frac) * budget, deadline - now - 6.0)
    if local_budget > 8.0:
        best, bs = _local_search(best, k, rng, time.monotonic() + local_budget, consider)

    return best, bs


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = max(5.0, float(budget_s))
    rng = np.random.default_rng(seed)

    t0 = time.monotonic()
    headroom = min(30.0, max(2.0, 0.05 * budget_s))
    deadline = t0 + max(1.0, budget_s - headroom)

    exact_best_list = None
    exact_best_val = float("inf")

    def consider_arr(arr):
        nonlocal exact_best_list, exact_best_val
        try:
            a = np.asarray(arr, dtype=np.float64)
            if a.ndim != 1 or a.size == 0:
                return
            a = np.where(np.isfinite(a) & (a >= 0.0), a, 0.0)
            a = np.clip(a, 0.0, 1000.0)
            if float(a.sum()) < 0.01:
                return
            lst = [float(x) for x in a]
            val = float(evaluate_sequence(lst))
            if np.isfinite(val) and val < exact_best_val:
                exact_best_val = val
                exact_best_list = lst
        except Exception:
            pass

    q4 = np.array([0.2, 0.4, 0.2, 0.2], dtype=np.float64)
    consider_arr(q4)

    parent = None
    try:
        p = globals().get("height_sequence_1", None)
        if p is not None:
            parent = np.asarray(p, dtype=np.float64).ravel()
            if parent.size > 0:
                consider_arr(parent)
    except Exception:
        parent = None

    if exact_best_list is None:
        exact_best_list = [1.0]
        exact_best_val = 2.0

    if time.monotonic() > deadline - 5.0:
        return [float(x) for x in exact_best_list]

    small_best = None
    small_best_score = float("inf")

    if deadline - time.monotonic() > 120.0:
        small_specs = [(64, 18.0), (128, 30.0), (256, 48.0)]
        for k, alloc in small_specs:
            if time.monotonic() > deadline - 120.0:
                break
            alloc = min(alloc, max(0.0, deadline - time.monotonic() - 90.0))
            if alloc < 12.0:
                continue
            if parent is not None and parent.size > 0:
                qk = _resize(parent, k)
            else:
                qk = _initial_start(k, rng)
            qopt, sc = _refine(qk, k, alloc, rng, deadline, consider_arr)
            sc = float(_score(qopt))
            if sc < small_best_score:
                small_best = qopt
                small_best_score = sc

    if parent is not None and parent.size >= 8:
        k_main = max(16, min(1024, int(parent.size)))
    else:
        k_main = 512

    main_candidates = []
    if parent is not None and parent.size > 0:
        main_candidates.append(_resize(parent, k_main))
    if small_best is not None:
        main_candidates.append(_resize(small_best, k_main))
    if not main_candidates:
        main_candidates.append(_initial_start(k_main, rng))

    q_main = min(main_candidates, key=_score)

    remaining = deadline - time.monotonic()
    if remaining < 25.0:
        rc = _score(q_main)
        consider_arr(q_main)
        return [float(x) for x in exact_best_list]

    reserve = 0.0
    if k_main < 1024 and remaining > 380.0:
        reserve = min(320.0, max(70.0, 0.22 * remaining))
    elif k_main < 2048 and remaining > 240.0:
        reserve = min(220.0, max(50.0, 0.25 * remaining))

    main_budget = max(15.0, remaining - reserve - 12.0)
    q_main, sc_main = _refine(q_main, k_main, main_budget, rng, deadline, consider_arr)
    consider_arr(q_main)

    last_q = q_main
    last_k = k_main

    now = time.monotonic()
    k_large = min(2048, max(64, 2 * last_k))
    if k_large != last_k and deadline - now > 90.0:
        large_alloc = min(360.0, max(30.0, 0.40 * (deadline - now)))
        large_alloc = min(large_alloc, max(0.0, deadline - now - 30.0))
        if large_alloc > 45.0:
            starts = [q_main]
            if parent is not None and parent.size > 0:
                starts.append(_resize(parent, k_large))
            if small_best is not None:
                starts.append(_resize(small_best, k_large))
            large_candidates = [_resize(s, k_large) for s in starts]
            q_large_start = min(large_candidates, key=_score)
            q_large, sc_large = _refine(q_large_start, k_large, large_alloc, rng, deadline, consider_arr)
            consider_arr(q_large)
            if _score(q_large) < _score(last_q) - 1e-12:
                last_q = q_large
                last_k = k_large

    now = time.monotonic()
    if last_k < 2048 and deadline - now > 150.0:
        k_extra = 2048
        extra_alloc = min(260.0, max(0.0, deadline - now - 35.0))
        if extra_alloc > 70.0:
            starts = [last_q]
            if parent is not None and parent.size > 0:
                starts.append(_resize(parent, k_extra))
            extra_candidates = [_resize(s, k_extra) for s in starts]
            q_extra_start = min(extra_candidates, key=_score)
            q_extra, sc_extra = _refine(q_extra_start, k_extra, extra_alloc, rng, deadline, consider_arr)
            consider_arr(q_extra)
            if _score(q_extra) < _score(last_q) - 1e-12:
                last_q = q_extra
                last_k = k_extra

    consider_arr(last_q)

    if exact_best_list is None:
        exact_best_list = [float(x) for x in _clean_norm(q4)]

    try:
        val = float(evaluate_sequence([float(x) for x in exact_best_list]))
        if not np.isfinite(val):
            exact_best_list = [1.0]
    except Exception:
        exact_best_list = [1.0]

    return [float(x) for x in exact_best_list]
