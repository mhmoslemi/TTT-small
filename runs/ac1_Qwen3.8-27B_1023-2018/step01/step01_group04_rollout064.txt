def _score(a):

def _build_A_dense(a):

def _lp_solve(a, M, timeout_s=5.0):
    n = len(a)
    A = _build_A_dense(a)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None)] * n
    try:

                      options={'time_limit': timeout_s, 'presolve': True})

def _soft_grad(a, tau=15.0):
    n = a.size
    b = np.convolve(a, a)
    bm = float(b.max())
    if bm < 1e-30:
        return np.zeros(n)
    logits = tau * b / bm
    logits -= logits.max()
    w = np.exp(logits)
    ws = float(w.sum())
    if ws < 1e-30:
        return np.zeros(n)
    w = w / ws
    # grad[p] = 2 * sum_j a[j] * w[p+j]
    W = np.lib.stride_tricks.sliding_window_view(w, n)
    grad = 2.0 * (W @ a)
    return grad


def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    vn = v / max(target, 1e-30)
    u = np.sort(vn)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = n - 1
    idx = np.where(rho_candidates)[0]
    if len(idx) > 0:
        rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(vn - theta, 0.0)
    return w * target

def _resample(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, None)
    n_old = a.size
    if n_old == n_new:
        s = float(a.sum())
        return a / s if s > 0 else np.ones(n_new) / n_new
    xp = np.linspace(0.0, 1.0, n_old)
    xn = np.linspace(0.0, 1.0, n_new)
    g = np.interp(xn, xp, a)
    s = float(g.sum())
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return g / s
