    # Final validation
    out = [float(x) for x in best_arr]
    final = safe_eval(out)
    if not np.isfinite(final):
        out = [float(x) for x in base]
    if not out or any(not np.isfinite(v) for v in out):
        out = [1.0] * n
    return out
