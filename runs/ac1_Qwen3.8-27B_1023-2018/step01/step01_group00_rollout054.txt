import time
import math
from scipy.fft import rfft, irfft, next_fast_len

        z = z - z.max()
        e = np.exp(z)
        s = e.sum()
        if s <= 0 or not np.isfinite(s):
            return np.full(z.size, 1.0 / z.size)
        return e / s

    def _fft_conv(a, fsize):
        F = rfft(a, fsize)
        c = irfft(F * F, fsize)[:2 * len(a) - 1]
        return np.maximum(c, 0.0)

    def _fast_score(a, fsize):
        b = _fft_conv(a, fsize)
        return 2.0 * len(a) * float(b.max())

    def _grad_log_space(a, b, tau):
        """Gradient of soft-max(log-sum-exp of b) w.r.t. log(a)."""
        n = len(a)
        L = len(b)  # 2n-1
        m = float(b.max())
        x = (b - m) / tau
        x -= x.max()
        e = np.exp(x)
        p = e / e.sum()
        # dS/da_j = 2 * sum_{l=0}^{n-1} p[j+l] * a_l  (with p out-of-range = 0)
        grad_a = np.zeros(n)
        for j in range(n):
            # l from 0 to n-1, index j+l in p
            hi = min(n, L - j)
            if hi <= 0:
                continue
            grad_a[j] = 2.0 * float(np.dot(p[j:j+hi], a[:hi]))
        grad_z = grad_a * a
        return grad_z

    def _grad_hard(a, b):
        """Exact subgradient at the argmax."""
        n = len(a)
        k_star = int(np.argmax(b))
        grad_a = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad_a[j] = 2.0 * a[idx]
        return grad_a * a
