            s_trial = a_trial.sum()
            if s_trial > 0.01:
                a = a_trial * n / s_trial
            else:
                a = a * n / S

        # Periodic exact evaluation
        if iter_count % 500 == 0 and time.monotonic() < deadline - 10:
            sc = exact_score(a)
            if np.isfinite(sc) and sc < best_score:
                best = a.copy()
                best_score = sc
                no_imp = 0

        # Stuck detection and restart
        if no_imp > 12000:
            pert = 0.03 + 0.02 * min(5.0, no_imp / 5000.0)
            a = a + rng.normal(0.0, pert, n)
            a = np.maximum(a, 0.0)
            a = a * n / max(a.sum(), 0.01)
            no_imp = 0
        elif no_imp > 6000:
            # Local perturbation
            a = a + rng.normal(0.0, 0.02, n)
            a = np.maximum(a, 0.0)
            a = a * n / max(a.sum(), 0.01)

    # Final polishing: use exact max-point gradient for a few steps
    if time.monotonic() < deadline - 10:
        a = best.copy()
        a = a * n / a.sum()
        for polish_iter in range(200):
            if time.monotonic() > deadline - 8:
                break
            B = convolve_self(a)
            S = a.sum()
            M = B.max()
            cur_score = 2.0 * n * M / (S * S)
            if cur_score < best_score - 1e-12:
                best = a.copy()
                best_score = cur_score

            # Exact subgradient at max set
            tol = 1e-10
            max_mask = B >= M - tol
            n_max = max_mask.sum()
            if n_max == 0:
                break
            # grad_i = (2n/S^3) * sum_{k in max} (2*S*a_{k-i} - 2*M) ... simplified
            # dF/da_i = 4n/S^3 * (S * (1/n_max)*sum_k a_{k-i} - M)
            # sum_k a_{k-i} for k in max set
            w_exact = np.zeros(2 * n - 1)
            w_exact[max_mask] = 1.0 / n_max
            a_rev = a[::-1]
            flip_w = w_exact[::-1]
            Ar = np.fft.rfft(a_rev, corr_n)
            Fw = np.fft.rfft(flip_w, corr_n)
            conv_res = np.fft.irfft(Ar * Fw, corr_n)
            c = conv_res[n - 1:2 * n - 1][::-1]
            grad = (4.0 * n * (S * c - M)) / (S ** 3)
            grad -= grad.mean()
            gnorm = np.sqrt(np.dot(grad, grad))
            if gnorm < 1e-14:
                break
            # Small step with line search
            step_size = 0.1 / gnorm
            improved = False
            for _ in range(5):
                a_t = np.maximum(a - step_size * grad, 0.0)
                s_t = a_t.sum()
                if s_t < 0.01:
                    step_size *= 0.5
                    continue
                a_t = a_t * n / s_t
                B_t = convolve_self(a_t)
                sc_t = 2.0 * n * B_t.max() / (a_t.sum() ** 2)
                if sc_t < cur_score:
                    a = a_t
                    improved = True
                    break
                step_size *= 0.5
            if not improved:
                break

    # Final validation
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc) or final_sc > best_score:
        # Try parent
        p_sc = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(p_sc) and p_sc < final_sc:
            best = parent
            final_sc = p_sc
        if not np.isfinite(final_sc):
            best = np.ones(n, dtype=np.float64)
            final_sc = float(evaluate_sequence(best.tolist()))

    result = [float(max(0.0, min(1000.0, x))) for x in best]
    return result
