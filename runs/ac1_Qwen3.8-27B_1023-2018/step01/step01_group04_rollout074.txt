   freqs=np.arange(1,K+1,dtype=float)
   phases=rng.random(K)*2*np.pi
   amps=rng.normal(0.0, amp/np.sqrt(K))
   x=np.linspace(0,1,n)
   noise=np.zeros(n)
   for k,ph,am in zip(freqs,phases,amps):
      noise += am*np.sin(2*np.pi*k*x + ph)
   b=a*np.exp(noise)
   return _normalize_nonneg(b)

def smooth_perturb(a, r, amp=0.1):

   n = a.size

                if time.monotonic() >= deadline - 45.0:
                    break
                alloc = min(20.0, max(8.0, deadline - time.monotonic() - 35.0))
                init = resize(best[1] if best[1] is not None else parent, nn)
                init = smooth_perturb(init, rng, 0.12)
                subgrad_refine(init, alloc, rng)

    if time.monotonic() < deadline - 20.0 and best[1] is not None:
        arr = best[1]
        if arr.size > 0:
            if arr.size <= 1200:
                limit = min(18.0, max(2.0, deadline - time.monotonic() - 12.0))
                subgrad_refine(arr, limit, rng)
            else:
                arr2 = resize(arr, 768)
                limit = min(12.0, max(2.0, deadline - time.monotonic() - 12.0))
                subgrad_refine(arr2, limit, rng)

    if best[1] is not None:
        consider(best[1])

    fallback = best[1]
    if fallback is None or fallback.size == 0:
        fallback = np.array([0.5, 0.5], dtype=np.float64)
    fallback = _normalize_nonneg(fallback)
    if fallback.size == 0:
        fallback = np.array([0.5, 0.5], dtype=np.float64)
    try:
        sc = evaluate_sequence([float(x) for x in fallback])
    except Exception:
        sc = float("nan")
    if not math.isfinite(sc):
        fallback = np.array([0.5, 0.5], dtype=np.float64)
        fallback = _normalize_nonneg(fallback)
    return [float(x) for x in fallback]
