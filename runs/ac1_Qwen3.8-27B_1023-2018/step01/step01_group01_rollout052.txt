        z = np.clip(beta * (c - cmax), -700.0, 0.0)
        w = np.exp(z)
        wsum = float(w.sum())
        if wsum < 1e-300 or not np.isfinite(wsum):
            return 1e10, np.zeros_like(x)
        w = w / wsum
        M = cmax + np.log(wsum) / beta
        # Cross-correlation for gradient
        a_rev = x[::-1]
        fa_rev = np.fft.rfft(a_rev, sizeG)
        fw = np.fft.rfft(w, sizeG)
        cross = np.fft.irfft(fa_rev * fw, sizeG)
        g_raw = cross[n - 1:2 * n - 1]
        grad_M = 2.0 * g_raw
        grad = (2.0 * n / (S * S)) * (grad_M - 2.0 * M / S)
        F = 2.0 * n * M / (S * S)
        F = float(F) if np.isfinite(F) else 1e10
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
        return F, grad

    def exact_obj(x):
        S = float(x.sum())
        if S < 0.01:
            return np.inf
        fa = np.fft.rfft(x, sizeC)
        c = np.fft.irfft(fa * fa, sizeC)[:2 * n - 1]
        return float(2.0 * n * c.max() / (S * S))

    bounds = [(0.0, 1000.0)] * n

    def lbfgs_stage(a_start, beta, maxfun, maxiter):
        nonlocal best_a, best_score
        a0 = np.clip(a_start, 0.0, 1000.0)
        s = exact_obj(a0)
        if s < best_score:
            best_a, best_score = a0.copy(), s

        def fg(x):
            return obj_grad(x, beta)

        try:
            res = minimize(fg, a0, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxiter, 'maxfun': maxfun,
                                    'ftol': 1e-15, 'gtol': 1e-13,
                                    'maxls': 30, 'm': 8, 'iprint': -1})
            a_new = np.clip(res.x, 0.0, 1000.0)
        except Exception:
            a_new = a0

        s = exact_obj(a_new)
        if s < best_score:
            best_a, best_score = a_new.copy(), s
        return a_new

    def multi_stage_opt(a_start, betas, funs_per_stage, maxiter_base):
        nonlocal best_a, best_score
        a = a_start.copy()
        for i, (beta, mf) in enumerate(zip(betas, funs_per_stage)):
            if time_left() < 15.0:
                break
            mi = max(20, maxiter_base // (i + 1))
            a = lbfgs_stage(a, beta, mf, mi)
        return a

    # ===== Phase 1: Main optimization from parent =====
    betas_main = [4.0, 10.0, 25.0, 60.0, 150.0, 400.0, 1000.0, 3000.0]
    funs_main = [6000, 5000, 4000, 3000, 2000, 1500, 800, 400]
    a = multi_stage_opt(base, betas_main, funs_main, 400)

    # ===== Phase 2: Refinement with perturbation restarts =====
    n_trials = 80
    for trial in range(n_trials):
        if time_left() < 40.0:
            break
        if trial == 0:
            a_start = best_a.copy()
        elif trial < 10:
            # Small perturbations around best
            scale = 0.005 * (1 + trial * 0.3)
            a_start = best_a * (1.0 + rng.standard_normal(n) * scale)
        else:
            # Larger exploratory perturbations
            scale = 0.02 + 0.01 * trial
            mask = rng.random(n) < 0.3
            a_start = best_a.copy()
            a_start[mask] = best_a[mask] * (1.0 + rng.standard_normal(int(mask.sum())) * scale)
            a_start = np.clip(a_start, 0.0, 1000.0)

        betas_ref = [30.0, 100.0, 300.0, 1000.0]
        funs_ref = [2000, 1200, 600, 200]
        a = multi_stage_opt(a_start, betas_ref, funs_ref, 200)

    # ===== Phase 3: Very high-beta polish =====
    if time_left() > 25.0:
        betas_polish = [500.0, 1500.0, 4000.0, 10000.0]
        funs_polish = [500, 300, 150, 80]
        a = multi_stage_opt(best_a, betas_polish, funs_polish, 100)

    # ===== Phase 4: Direct subgradient polish =====
    if time_left() > 15.0:
        a = best_a.copy()
        S = a.sum()
        for it in range(2000):
            if time_left() < 12.0:
                break
            fa = np.fft.rfft(a, sizeC)
            c = np.fft.irfft(fa * fa, sizeC)[:2 * n - 1]
            cmax = float(c.max())
            # Active set: all positions within 1e-10 of max
            active = np.where(c >= cmax - cmax * 1e-10)[0]
            # Average gradient over active set
            grad_zero = np.zeros(n)
            for k in active:
                # d c_k / d a_i = 2 * a_{k-i} for valid indices
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                if lo <= hi:
                    grad_zero[lo:hi + 1] += 2.0 * a[(k - lo):(k - hi + 1)][::-1]
            grad_zero /= len(active)
            # Subtract mean for normalization invariance
            grad_zero -= grad_zero.mean()
            # Scale step
            step = cmax / (S * S) * 0.01
            a = a - step * grad_zero
            a = np.clip(a, 0.0, 1000.0)
            S = a.sum()
            if S < 0.01:
                break
            a = a / S  # renormalize
            s = exact_obj(a)
            if s < best_score:
                best_a, best_score = a.copy(), s

    # ===== Final validation =====
    final_score = safe_eval(best_a)
    if not np.isfinite(final_score) or final_score > best_score + 1e-6:
        # Try the original base
        base_score = safe_eval(base)
        if np.isfinite(base_score) and base_score < final_score:
            best_a = base.copy()

    out = [float(x) for x in np.clip(best_a, 0.0, 1000.0)]
    if not out or any(not np.isfinite(x) for x in out):
        out = [1.0] * n
    if sum(out) < 0.01:
        out = [max(0.01 / n, 1e-6)] * n
    return out
