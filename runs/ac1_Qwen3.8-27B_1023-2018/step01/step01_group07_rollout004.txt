        y = z - float(z.max())
        p = np.exp(y)
        s = float(p.sum())
        if not np.isfinite(s) or s <= 0.0:
            p = np.ones(n, dtype=np.float64) / float(n)
            z = np.zeros(n, dtype=np.float64)
        else:
            p = p / s
        return p, z

    def logp(p):
        q = np.maximum(p, 1e-300)
        z = np.log(q)
        z = z - float(np.mean(z))
        return z

    def exact_score(p):
        try:
            vals = np.asarray(p, dtype=np.float64).tolist()
            sc = float(evaluate_sequence(vals))
            return sc if np.isfinite(sc) else float("inf")
        except Exception:
            return float("inf")

    def make_shape():
        x = np.linspace(-1.0, 1.0, n)
        r = int(rng.integers(0, 4))
        if r == 0:
            sig = float(rng.uniform(0.3, 1.1))
            q = np.exp(-((x / sig) ** 2))
        elif r == 1:
            q = 1.0 / np.sqrt(1.0 - x * x + 1e-3)
        elif r == 2:
            a = float(rng.uniform(0.5, 2.5))
            b = float(rng.uniform(1.0, 3.5))
            q = 1.0 / (1.0 + a * np.abs(x)) ** b
        else:
            q = np.exp(rng.normal(0.0, 0.5, n))
            q = np.convolve(q, np.ones(7, dtype=np.float64) / 7.0, mode="same")
        q = np.maximum(q, 0.0)
        s = float(q.sum())
        if s <= 1e-12:
            q = np.ones(n, dtype=np.float64) / float(n)
        else:
            q = q / s
        return q

    def run_eu(p0, limit_s, tau_start=0.25):
        p = np.asarray(p0, dtype=np.float64).copy()
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if s <= 1e-12:
            p = np.ones(n, dtype=np.float64) / float(n)
        else:
            p = p / s

        best_p_sess = p.copy()
        c = conv_full(p)
        best_M = float(c.max())
        M = best_M
        tau_frac = float(tau_start)
        lr = 0.4
        stall = 0
        it = 0

        last_exact = time.monotonic()
        exact_best_p = best_p_sess.copy()
        exact_best_sc = exact_score(best_p_sess)

        end = time.monotonic() + max(1.0, limit_s)

        while time.monotonic() < end and time.monotonic() < search_deadline:
            if M <= 0.0:
                break

            tau = tau_frac * max(M, best_M, 1e-300)
            if tau < 1e-300:
                tau = 1e-300

            M, G, w = weights_soft(c, tau)
            M_prev = M
            Ap = weighted_Ap(p, w)
            grad = 2.0 * Ap
            gm = float(np.mean(grad))
            direction = grad - gm

            chosen_obj = None
            chosen_p = None
            chosen_c = None
            chosen_step = None
            chosen_G = None

            # Projected-gradient-like steps.
            for mult in (0.5, 1.0, 2.0):
                step = lr * mult
                if step <= 0.0:
                    continue
                pn = p - step * direction
                if (pn < 0.0).any():
                    np.maximum(pn, 0.0, out=pn)
                spn = float(pn.sum())
                if spn < 1e-12:
                    continue
                pn = pn / spn
                cn = conv_full(pn)
                Mn, Gn, _ = weights_soft(cn, tau)

                if Mn < M_prev - 1e-12:
                    obj = (0, Mn, Gn)
                elif Gn < G - 1e-10 and Mn < M_prev * 1.05:
                    obj = (1, Mn, Gn)
                else:
                    continue

                if chosen_obj is None or obj < chosen_obj:
                    chosen_obj = obj
                    chosen_p = pn
                    chosen_c = cn
                    chosen_step = step
                    chosen_G = Gn

            # Occasional LP-inspired inverse-incidence move.
            if stall >= 5 or it % 300 == 0:
                amax = float(Ap.max())
                if amax > 0.0:
                    eps = max(1e-12, 1e-8 * amax)
                    inv = 1.0 / (Ap + eps)
                    s_inv = float(inv.sum())
                    if s_inv > 0.0:
                        g = inv / s_inv
                        for mix in (0.2, 0.4, 0.6, 0.8):
                            pn = (1.0 - mix) * p + mix * g
                            spn = float(pn.sum())
                            if spn < 1e-12:
                                continue
                            pn = pn / spn
                            cn = conv_full(pn)
                            Mn, Gn, _ = weights_soft(cn, tau)

                            if Mn < M_prev - 1e-12:
                                obj = (0, Mn, Gn)
                            elif Gn < G - 1e-10 and Mn < M_prev * 1.05:
                                obj = (1, Mn, Gn)
                            else:
                                continue

                            if chosen_obj is None or obj < chosen_obj:
                                chosen_obj = obj
                                chosen_p = pn
                                chosen_c = cn
                                chosen_step = None
                                chosen_G = Gn

            if chosen_p is not None:
                p = chosen_p
                c = chosen_c
                M = float(c.max())
                G = chosen_G

                progress = (M < M_prev - 1e-12)
                if progress:
                    stall = 0
                    if chosen_step is not None:
                        lr = min(5.0, max(1e-6, chosen_step * 1.05))
                    else:
                        lr = min(5.0, lr * 1.05)
                    tau_frac = max(min_tau, tau_frac * 0.995)
                else:
                    stall += 1
                    if chosen_step is not None:
                        lr = float(chosen_step)

                if M < best_M:
                    best_M = M
                    best_p_sess = p.copy()
                    now = time.monotonic()
                    if now - last_exact > min(10.0, exact_interval):
                        sc = exact_score(best_p_sess)
                        if np.isfinite(sc) and sc < exact_best_sc:
                            exact_best_sc = sc
                            exact_best_p = best_p_sess.copy()
                        last_exact = now
            else:
                stall += 1
                lr = max(1e-6, lr * 0.7)

            if stall >= 60:
                base = best_p_sess if M > 1.05 * best_M else p
                p2 = base.copy()
                if stall > 150:
                    p2 = 0.7 * base + 0.3 * make_shape()
                z = logp(p2) + rng.normal(0.0, 0.1 if stall < 100 else 0.2, n)
                p, _ = softmax_z(z)
                c = conv_full(p)
                M = float(c.max())
                if M > 2.0 * best_M:
                    p = best_p_sess.copy()
                    c = conv_full(p)
                    M = best_M
                tau_frac = min(max_tau, tau_frac * 2.0)
                lr = 0.4
                stall = 0
            elif stall > 20:
                tau_frac = min(max_tau, tau_frac * 1.15)

            if M > 1.5 * best_M and stall > 30:
                p = best_p_sess.copy()
                c = conv_full(p)
                M = best_M
                tau_frac = min(max_tau, tau_frac * 1.5)

            now = time.monotonic()
            if now - last_exact > exact_interval:
                sc = exact_score(best_p_sess)
                if np.isfinite(sc) and sc < exact_best_sc:
                    exact_best_sc = sc
                    exact_best_p = best_p_sess.copy()
                last_exact = now

            it += 1

        sc = exact_score(best_p_sess)
        if np.isfinite(sc) and sc < exact_best_sc:
            exact_best_sc = sc
            exact_best_p = best_p_sess.copy()

        return best_p_sess.copy(), best_M, exact_best_p.copy(), exact_best_sc

    def run_nat(p0, limit_s, tau_start=0.25):
        p = np.asarray(p0, dtype=np.float64).copy()
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if s <= 1e-12:
            p = np.ones(n, dtype=np.float64) / float(n)
        else:
            p = p / s

        z = logp(p)
        m = np.zeros(n, dtype=np.float64)
        v = np.zeros(n, dtype=np.float64)
        adam_t = 0
        b1 = 0.9
        b2 = 0.999

        best_p_sess = p.copy()
        c = conv_full(p)
        best_M = float(c.max())
        M = best_M
        tau_frac = float(tau_start)
        lr = 0.1
        stall = 0
        it = 0

        last_exact = time.monotonic()
        exact_best_p = best_p_sess.copy()
        exact_best_sc = exact_score(best_p_sess)

        end = time.monotonic() + max(1.0, limit_s)

        while time.monotonic() < end and time.monotonic() < search_deadline:
            if M <= 0.0:
                break

            tau = tau_frac * max(M, best_M, 1e-300)
            if tau < 1e-300:
                tau = 1e-300

            M, G, w = weights_soft(c, tau)
            M_prev = M
            Ap = weighted_Ap(p, w)
            p_ap = float(np.dot(p, Ap))
            gradz = Ap - p_ap

            adam_t += 1
            m = b1 * m + (1.0 - b1) * gradz
            v = b2 * v + (1.0 - b2) * gradz * gradz
            mh = m / (1.0 - b1 ** adam_t)
            vh = v / (1.0 - b2 ** adam_t)
            hat = mh / (np.sqrt(vh) + 1e-8)
            hat = hat - float(np.mean(hat))

            chosen_obj = None
            chosen_p = None
            chosen_c = None
            chosen_step = None
            chosen_G = None

            if float(np.dot(hat, hat)) > 1e-24:
                for mult in (0.5, 1.0, 2.0):
                    step = lr * mult
                    if step <= 0.0:
                        continue
                    zt = z - step * hat
                    pn, _ = softmax_z(zt)
                    cn = conv_full(pn)
                    Mn, Gn, _ = weights_soft(cn, tau)

                    if Mn < M_prev - 1e-12:
                        obj = (0, Mn, Gn)
                    elif Gn < G - 1e-10 and Mn < M_prev * 1.05:
                        obj = (1, Mn, Gn)
                    else:
                        continue

                    if chosen_obj is None or obj < chosen_obj:
                        chosen_obj = obj
                        chosen_p = pn
                        chosen_c = cn
                        chosen_step = step
                        chosen_G = Gn

            if stall >= 5 or it % 300 == 0:
                amax = float(Ap.max())
                if amax > 0.0:
                    eps = max(1e-12, 1e-8 * amax)
                    inv = 1.0 / (Ap + eps)
                    s_inv = float(inv.sum())
                    if s_inv > 0.0:
                        g = inv / s_inv
                        for mix in (0.2, 0.4, 0.6, 0.8):
                            pn = (1.0 - mix) * p + mix * g
                            spn = float(pn.sum())
                            if spn < 1e-12:
                                continue
                            pn = pn / spn
                            cn = conv_full(pn)
                            Mn, Gn, _ = weights_soft(cn, tau)

                            if Mn < M_prev - 1e-12:
                                obj = (0, Mn, Gn)
                            elif Gn < G - 1e-10 and Mn < M_prev * 1.05:
                                obj = (1, Mn, Gn)
                            else:
                                continue

                            if chosen_obj is None or obj < chosen_obj:
                                chosen_obj = obj
                                chosen_p = pn
                                chosen_c = cn
                                chosen_step = None
                                chosen_G = Gn

            if chosen_p is not None:
                p = chosen_p
                c = chosen_c
                M = float(c.max())
                G = chosen_G
                z = logp(p)

                progress = (M < M_prev - 1e-12)
                if progress:
                    stall = 0
                    if chosen_step is not None:
                        lr = min(3.0, max(1e-6, chosen_step * 1.05))
                    else:
                        lr = min(3.0, lr * 1.05)
                    tau_frac = max(min_tau, tau_frac * 0.995)
                else:
                    stall += 1
                    if chosen_step is not None:
                        lr = float(chosen_step)

                if M < best_M:
                    best_M = M
                    best_p_sess = p.copy()
                    now = time.monotonic()
                    if now - last_exact > min(10.0, exact_interval):
                        sc = exact_score(best_p_sess)
                        if np.isfinite(sc) and sc < exact_best_sc:
                            exact_best_sc = sc
                            exact_best_p = best_p_sess.copy()
                        last_exact = now
            else:
                stall += 1
                lr = max(1e-5, lr * 0.7)

            if stall >= 60:
                base = best_p_sess if M > 1.05 * best_M else p
                p2 = base.copy()
                if stall > 150:
                    p2 = 0.7 * base + 0.3 * make_shape()
                z = logp(p2) + rng.normal(0.0, 0.1 if stall < 100 else 0.2, n)
                p, z = softmax_z(z)
                c = conv_full(p)
                M = float(c.max())
                if M > 2.0 * best_M:
                    p = best_p_sess.copy()
                    z = logp(p)
                    c = conv_full(p)
                    M = best_M
                tau_frac = min(max_tau, tau_frac * 2.0)
                lr = 0.1
                m = np.zeros(n, dtype=np.float64)
                v = np.zeros(n, dtype=np.float64)
                adam_t = 0
                stall = 0
            elif stall > 20:
                tau_frac = min(max_tau, tau_frac * 1.15)

            if M > 1.5 * best_M and stall > 30:
                p = best_p_sess.copy()
                z = logp(p)
                c = conv_full(p)
                M = best_M
                tau_frac = min(max_tau, tau_frac * 1.5)

            now = time.monotonic()
            if now - last_exact > exact_interval:
                sc = exact_score(best_p_sess)
                if np.isfinite(sc) and sc < exact_best_sc:
                    exact_best_sc = sc
                    exact_best_p = best_p_sess.copy()
                last_exact = now

            it += 1

        sc = exact_score(best_p_sess)
        if np.isfinite(sc) and sc < exact_best_sc:
            exact_best_sc = sc
            exact_best_p = best_p_sess.copy()

        return best_p_sess.copy(), best_M, exact_best_p.copy(), exact_best_sc

    best_p = p_init.copy()
    best_M = conv_max(best_p)
    best_exact_p = best_p.copy()
    best_exact_score = exact_score(best_p)

    if not np.isfinite(best_exact_score):
        uniform_p = np.ones(n, dtype=np.float64) / float(n)
        us = exact_score(uniform_p)
        if np.isfinite(us):
            best_p = uniform_p.copy()
            best_M = conv_max(best_p)
            best_exact_p = uniform_p.copy()
            best_exact_score = us

    def merge_result(res):
        nonlocal best_p, best_M, best_exact_p, best_exact_score
        if res is None:
            return
        try:
            p_int, M_int, p_exact, sc_exact = res
        except Exception:
            return
        if p_int is not None and np.all(np.isfinite(p_int)):
            if M_int < best_M:
                best_M = float(M_int)
                best_p = np.asarray(p_int, dtype=np.float64).copy()
        if p_exact is not None and np.isfinite(sc_exact):
            if sc_exact < best_exact_score:
                best_exact_score = float(sc_exact)
                best_exact_p = np.asarray(p_exact, dtype=np.float64).copy()

    # Stage 1: Euclidean projected-style optimization, close to the parent program.
    rem = search_deadline - time.monotonic()
    if rem > 120.0:
        t1 = min(rem * (0.45 if rem > 900.0 else 0.60), 650.0)
        if t1 > 30.0:
            merge_result(run_eu(best_p, t1, 0.25))

    # Stage 2: Natural-gradient / simplex parameterization for a different geometry.
    rem = search_deadline - time.monotonic()
    if rem > 100.0:
        t2 = min(rem * (0.38 if rem > 800.0 else 0.45), 520.0)
        if t2 > 30.0:
            merge_result(run_nat(best_p, t2, 0.25))

    # Stage 3: Low-temperature Euclidean polish.
    rem = search_deadline - time.monotonic()
    if rem > 60.0:
        t3 = max(0.0, rem - 20.0)
        if t3 > 20.0:
            merge_result(run_eu(best_p, t3, 0.10))

    # Final exact validation and safe return.
    rel = [
        best_p,
        best_exact_p,
        p_init,
    ]

    final_p = best_exact_p if np.isfinite(best_exact_score) else best_p
    final_sc = best_exact_score if np.isfinite(best_exact_score) else float("inf")

    for cand in rel:
        try:
            if cand is None:
                continue
            sc = exact_score(cand)
            if np.isfinite(sc) and sc < final_sc:
                final_sc = sc
                final_p = np.asarray(cand, dtype=np.float64).copy()
        except Exception:
            pass

    if not np.isfinite(final_sc) or final_p is None:
        final_p = np.ones(n, dtype=np.float64) / float(n)
        final_sc = exact_score(final_p)

    a = np.asarray(final_p, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    if a.size == 0 or not np.isfinite(a).all():
        a = np.ones(max(1, n), dtype=np.float64)

    ss = float(a.sum())
    if ss > 0.0:
        a = a / ss
        ss = float(a.sum())

    if ss < 0.01:
        if ss > 0.0:
            a = a / ss
        else:
            a = np.ones(max(1, n), dtype=np.float64)

    if not np.isfinite(a).all():
        a = np.ones(max(1, n), dtype=np.float64) / float(max(1, n))

    return [float(x) for x in a]
