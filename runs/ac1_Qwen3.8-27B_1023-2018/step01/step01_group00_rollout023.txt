p >= 0

In practice, |K| is much smaller than |S| = n (we expect only a few active lags). So the system is underdetermined, and we find the minimum-norm solution.

```python
def _kkt_solve(a, K, S):
    # a: current sequence (normalized, sum=1)
    # K: active lags (list of indices)
    # S: support indices (where a[i] > 0)
    # Solve: sum_k p[k] * g_k[i] = lambda for i in S
    #        sum_k p[k] = 1
    #        p[k] >= 0
    n = len(a)
    |S| = len(S)
    |K| = len(K)
    
    # Build the system: [G | 1] * [p; lambda] = [0; 1]
    # where G[i,k] = g_k[S[i]] = a[K[k]-S[i]]*I(valid) + a[S[i]]*I(valid)
    # and the last row is sum_k p[k] = 1
    
    A = np.zeros((n+1, |K|+1))
    rhs = np.zeros(n+1)
    rhs[-1] = 1
    
    for ii, i in enumerate(S):
        for kk, k in enumerate(K):
            val = 0
            if 0 <= k - i < n:
                val += a[k-i]
            # a[i] term: valid when 0 <= k-i < n
            if 0 <= k - i < n:
                val += a[i]
            A[ii, kk] = val
        A[ii, -1] = 1  # coefficient of lambda
    
    for kk, k in enumerate(K):
        A[-1, kk] = 1  # sum_k p[k] = 1
    
    # Solve the linear system
    try:
        sol = np.linalg.lstsq(A, rhs, rcond=None)[0]
    except:
        return None
    
    p = sol[:|K|]
    lam = sol[-1]
    
    # Check non-negativity
    if np.any(p < -1e-10):
        # Project onto simplex
        p = np.maximum(p, 0)
        p /= p.sum()
    
    return p, lam
```
