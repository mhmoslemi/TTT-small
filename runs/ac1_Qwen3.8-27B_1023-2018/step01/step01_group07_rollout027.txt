        m_hat = m_adam / (1.0 - beta1 ** t_adam)
        v_hat = v_adam / (1.0 - beta2 ** t_adam)

        # Adaptive learning rate
        lr = base_lr / (1.0 + 0.005 * t_adam)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new /= s_new
        a = a_new

        # Periodic exact evaluation
        if eval_counter % 3000 == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Temperature annealing
        if iter_count > 0 and iter_count % 8000 == 0:
            tau_frac = max(tau_frac * 0.8, min_tau_frac)
            # Gradually increase top-K focus
            if tau_frac < 0.05 and K_top == 0:
                K_top = 10
            elif K_top == 10 and tau_frac < 0.01:
                K_top = 5

        # Stall recovery - mild
        if stagnation == 150:
            a *= np.exp(rng.normal(0, 0.003, n))
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a /= s
            m_adam[:] = 0.0
            v_adam[:] = 0.0
            t_adam = 0
            stagnation = 0
            tau_frac = min(tau_frac * 1.3, 0.3)
            K_top = 0

        # Stall recovery - strong
        if stagnation == 400:
            k_perturb = max(n // 8, 10)
            idx = rng.choice(n, size=k_perturb, replace=False)
            a[idx] *= np.exp(rng.normal(0, 0.2, k_perturb))
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a /= s
            m_adam[:] = 0.0
            v_adam[:] = 0.0
            t_adam = 0
            stagnation = 0
            tau_frac = min(tau_frac * 2.0, 0.4)
            K_top = 0
            base_lr = 0.3

        # Stall recovery - very strong (restart from best with perturbation)
        if stagnation == 800:
            a = best_a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a /= s
            m_adam[:] = 0.0
            v_adam[:] = 0.0
            t_adam = 0
            stagnation = 0
            tau_frac = 0.15
            K_top = 0

        # Check deadline with buffer
        if time.monotonic() > deadline - 5.0:
            break

    # Final evaluation of current and best
    try:
        sc_final = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_a = a.copy()
    except Exception:
        pass

    # Validate best candidate
    try:
        final_check = float(evaluate_sequence(best_a.tolist()))
        if not np.isfinite(final_check):
            best_a = np.ones(n, dtype=np.float64) / n
    except Exception:
        best_a = np.ones(n, dtype=np.float64) / n

    # Ensure valid return
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if not result or any(not np.isfinite(x) for x in result):
        result = [1.0 / n] * n

    return result
