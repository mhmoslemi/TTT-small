        Fw = np.fft.rfft(w[::-1], N)
        v = np.fft.irfft(Fa * Fw, N)
        out = v[n - 1:2 * n - 1][::-1]
        return np.maximum(out, 0.0)

    def row_sums(Fa):
        r = np.fft.irfft(Fa * F_ones, N)[:L]
        return np.maximum(r, 0.0)

    def normalize_arr(vec):
        v = np.asarray(vec, dtype=np.float64).copy()
        v = np.maximum(v, 0.0)
        v += 1e-300
        sv = float(v.sum())
        if not np.isfinite(sv) or sv <= 0.0:
            v = np.ones(n, dtype=np.float64)
            return v / n
        return v / sv

    def exact_score(arr):
        try:
            return float(evaluate_sequence(np.asarray(arr, dtype=float).tolist()))
        except Exception:
            return float("inf")

    def make_weights(b, M, q):
        if M <= 0.0 or not np.isfinite(M):
            return np.full(L, 1.0 / max(1, L), dtype=np.float64)
        ratio = b / M
        if q < 1000.0:
            w = np.power(ratio, float(q))
        else:
            cut = max(1, min(L, int(0.03 * L) + 1))
            idx = np.argpartition(b, -cut)[-cut:]
            w = np.zeros(L, dtype=np.float64)
            w[idx] = 1.0
        w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
        w = np.maximum(w, 0.0)
        w += 1.0 / max(1, L)
        ws = float(w.sum())
        if ws <= 0.0 or not np.isfinite(ws):
            w = np.full(L, 1.0 / max(1, L), dtype=np.float64)
        else:
            w /= ws
        return w

    def mirror_update(arr, y, eta):
        y = np.asarray(y, dtype=np.float64)
        y = np.nan_to_num(y, nan=0.0, posinf=0.0, neginf=0.0)
        y = np.maximum(y, 0.0)
        ym = float(y.max())
        if ym <= 0.0 or not np.isfinite(ym):
            return normalize_arr(arr)
        z = y / ym
        arr2 = arr * np.exp(-float(eta) * z)
        arr2 = np.nan_to_num(arr2, nan=0.0, posinf=0.0, neginf=0.0)
        arr2 = np.maximum(arr2, 0.0)
        arr2 += 1e-300
        s2 = float(arr2.sum())
        if not np.isfinite(s2) or s2 <= 0.0:
            return normalize_arr(arr)
        return arr2 / s2

    def perturb_arr(arr, sigma=0.15, reverse_blend=0.0):
        arr = normalize_arr(arr).copy()
        num = max(1, min(n, n // 60 + 2))
        idx = rng.choice(n, size=num, replace=False)
        arr[idx] *= np.exp(rng.normal(0.0, float(sigma), size=num))
        if reverse_blend > 0.0:
            rb = float(min(0.45, max(0.0, reverse_blend)))
            arr = (1.0 - rb) * arr + rb * arr[::-1]
        return normalize_arr(arr)

    def spectral_kick(arr, iters=6, mix=0.55):
        try:
            cur = normalize_arr(arr).copy()
            inv_sqrt_N = 1.0 / math.sqrt(float(N))
            for _ in range(max(1, int(iters))):
                A = np.fft.rfft(cur, N)
                mag = np.abs(A)
                phase = np.zeros_like(A)
                mask = mag > 1e-15
                phase[mask] = A[mask] / mag[mask]
                cand = np.fft.irfft(phase * inv_sqrt_N, N)[:n]
                cand = np.nan_to_num(cand, nan=0.0, posinf=0.0, neginf=0.0)
                cand = np.abs(cand)
                candc = normalize_arr(cand)
                cur = normalize_arr((1.0 - mix) * cur + mix * candc)
            return normalize_arr(cur)
        except Exception:
            return normalize_arr(arr)

    # ------------------------------------------------------------------
    # LP direction from the Frank-Wolfe-style construction:
    #   max sum(g) s.t. (a*g)_k <= M for selected lags, 0 <= g <= U.
    # Also recovers dual weights if available.
    # ------------------------------------------------------------------
    def lp_direction(a_arr, b_arr, M_val, F_a, remain_s, K):
        if linprog is None:
            return None, None
        t_lp = time.monotonic()
        tl = max(2.0, min(18.0, float(remain_s) - 60.0))
        if tl <= 2.0:
            return None, None

        K = int(max(80, min(520, K)))
        row = row_sums(F_a)
        rmax = float(row.max()) if row.size else 0.0
        if rmax <= 0.0:
            rmax = 1.0

        top_b = max(1, int(0.50 * K))
        top_r = max(1, int(0.35 * K))
        even = max(5, int(0.15 * K))

        if top_b >= L:
            idx_b = np.arange(L)
        else:
            idx_b = np.argpartition(b_arr, -top_b)[-top_b:]
        if top_r >= L:
            idx_r = np.arange(L)
        else:
            idx_r = np.argpartition(row, -top_r)[-top_r:]

        idx_even = np.linspace(0, L - 1, even, dtype=int)
        fixed = np.array([0, L // 3, L // 2, 2 * L // 3, L - 1, max(0, n - 1)], dtype=int)
        fixed = fixed[(fixed >= 0) & (fixed < L)]

        idx = np.unique(np.concatenate((idx_b, idx_r, idx_even, fixed)))
        if idx.size > K:
            vals = b_arr[idx] / max(M_val, 1e-300) + 0.65 * row[idx] / rmax
            keep = idx[np.argpartition(vals, -K)[-K:]]
            idx = np.unique(keep)

        # Build dense selected-constraint matrix (small enough for this n).
        A = np.zeros((idx.size, n), dtype=np.float64)
        for ii, k in enumerate(idx):
            lo = int(max(0, k - (n - 1)))
            hi = int(min(n - 1, k))
            if hi >= lo:
                start = int(k - lo)
                stop = int(k - hi - 1)
                if stop < 0:
                    A[ii, lo:hi + 1] = a_arr[start::-1]
                else:
                    A[ii, lo:hi + 1] = a_arr[start:stop:-1]

        b_ub = (M_val * (1.0 + 1e-9)) * np.ones(idx.size, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)
        var_bound = 2.0
        bounds = [(0.0, var_bound)] * n

        res = None
        methods = []
        if linprog is not None:
            methods.append("highs")
            methods.append("interior-point")

        for meth in methods:
            try:
                res = linprog(
                    c,
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=bounds,
                    method=meth,
                    options={"time_limit": tl},
                )
                break
            except Exception:
                try:
                    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method=meth)
                    break
                except Exception:
                    continue

        h = None
        dual = None

        if res is not None:
            try:
                x = getattr(res, "x", None)
                if x is not None:
                    g = np.asarray(x, dtype=np.float64).clip(0.0, 1000.0)
                    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
                    T = float(g.sum())
                    if np.isfinite(T) and T > 1.0 + 1e-8:
                        h = g / T
                        h = np.maximum(h, 0.0)
                        hs = float(h.sum())
                        if hs > 0.0 and np.isfinite(hs):
                            h /= hs
                        else:
                            h = None

                yd = getattr(res, "y", None)
                if yd is not None and idx.size:
                    dy = np.asarray(yd[:idx.size], dtype=np.float64)
                    dy = np.nan_to_num(dy, nan=0.0, posinf=0.0, neginf=0.0)
                    dy = np.clip(dy, 0.0, 1e12)
                    ds = float(dy.sum())
                    if ds > 1e-12 and np.isfinite(ds):
                        full = np.zeros(L, dtype=np.float64)
                        full[idx] = dy / ds
                        dual = full
            except Exception:
                pass

        # Protect against an LP solver consuming too much wall time.
        dt = time.monotonic() - t_lp
        if dt > 60.0 and h is None and dual is None:
            return None, None
        return h, dual

    def apply_lp_blend(a_arr, b_arr, M_val, F_a, h):
        if h is None:
            return None
        best = None
        best_M = float("inf")
        for t in (1.0, 0.5, 0.25, 0.1):
            cand = (1.0 - t) * a_arr + t * h
            cand = normalize_arr(cand)
            if time.monotonic() > deadline - 8.0:
                break
            Fc, bc, Mc = conv_self(cand)
            if np.isfinite(Mc) and Mc < best_M:
                best = (cand, bc, Mc, Fc)
                best_M = Mc
        return best

    # ------------------------------------------------------------------
    # Final L-BFGS polish on a smooth maximum surrogate, using softmax(u).
    # ------------------------------------------------------------------
    def lbfgs_polish(start, maxfun=45, temp=35.0):
        if minimize is None:
            return None
        try:
            u0 = np.log(np.maximum(start, 1e-300))
            u0 = np.clip(u0, -40.0, 40.0)

            def fun(u):
                u = np.clip(u, -50.0, 50.0)
                e = np.exp(u)
                p = e / e.sum()
                p = np.maximum(p, 0.0)
                ps = float(p.sum())
                p = p / ps if ps > 0 else np.ones(n) / n

                Fp = np.fft.rfft(p, N)
                b = np.fft.irfft(Fp * Fp, N)[:L]
                b = np.maximum(b, 0.0)
                M = float(b.max())
                if M <= 0.0 or not np.isfinite(M):
                    return 0.0, np.zeros_like(u)

                s = float(temp) / max(M, 1e-300)
                z = s * (b - M)
                z = np.clip(z, -100.0, 0.0)
                w = np.exp(z)
                ws = float(w.sum())
                if ws <= 0.0 or not np.isfinite(ws):
                    w = np.ones(L) / L
                    ws = float(L)
                else:
                    w = w / ws

                val = M + math.log(ws) / s
                y = load_weights(w, Fp)
                gp = 2.0 * y
                c = float(np.dot(p, gp))
                grad = p * (gp - c)
                grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
                return float(val), grad

            res = minimize(
                fun,
                u0,
                jac=True,
                method="L-BFGS-B",
                options={"maxiter": 40, "maxfun": maxfun, "ftol": 1e-12, "gtol": 1e-8},
            )
            if res is not None and res.x is not None:
                e = np.exp(np.clip(np.asarray(res.x, dtype=np.float64), -50.0, 50.0))
                p = e / e.sum()
                return normalize_arr(p)
        except Exception:
            return None
        return None

    # ------------------------------------------------------------------
    # Initial score and a few cheap transformations of the parent.
    # ------------------------------------------------------------------
    Fa, b, M = conv_self(a)
    best_a = normalize_arr(a).copy()
    best_exact = exact_score(best_a)
    if not np.isfinite(best_exact):
        best_exact = float("inf")
    best_M_exact = (best_exact / (2.0 * n)) if np.isfinite(best_exact) else float(M)
    best_est = float(M)
    last_exact = t0

    def consider_start(arr):
        nonlocal a, Fa, b, M, best_a, best_exact, best_est, best_M_exact
        arr = normalize_arr(arr)
        if arr.size != n:
            return
        Fc, bc, Mc = conv_self(arr)
        if not np.isfinite(Mc):
            return
        if Mc < float(M) - 1e-10:
            a = arr
            Fa, b, M = Fc, bc, Mc
            sc = exact_score(arr)
            if np.isfinite(sc) and sc < best_exact - 1e-12:
                best_exact = sc
                best_a = arr.copy()
                best_M_exact = sc / (2.0 * n)
                best_est = float(M)

    start_variants = [
        a.copy(),
        normalize_arr(0.5 * (a + a[::-1])),
        np.power(np.maximum(a, 1e-300), 0.82),
        np.power(np.maximum(a, 1e-300), 1.22),
    ]

    # Low-discrepancy re-assignments of the same masses.
    try:
        r = (np.arange(n) * 0.6180339887498948482) % 1.0
        order = np.argsort(r)
        vals_desc = np.sort(a)[::-1]
        vals_asc = np.sort(a)
        gd1 = np.empty(n, dtype=np.float64)
        gd1[order] = vals_desc
        gd2 = np.empty(n, dtype=np.float64)
        gd2[order] = vals_asc
        start_variants.append(gd1)
        start_variants.append(gd2)

        sp = spectral_kick(a, iters=5, mix=0.5)
        start_variants.append(sp)
    except Exception:
        pass

    for vv in start_variants:
        if time.monotonic() > deadline - 30.0:
            break
        consider_start(vv)

    # If a transformation is better internally but exact not yet better,
    # still keep it as the working point.

    # ------------------------------------------------------------------
    # Main search loop.
    # ------------------------------------------------------------------
    q = 10.0
    max_q = 130.0
    eta = 0.45
    eta_min = 0.08
    eta_max = 0.85
    patience = 0
    it = 0
    lambda_dual = None
    last_lp = t0 - 30.0
    lp_K = int(min(520, max(220, n // 9)))

    while time.monotonic() < deadline - 8.0:
        it += 1
        now = time.monotonic()
        if now > deadline - 8.0:
            break

        # Track internal best.
        if M < best_est - 1e-12:
            best_est = float(M)
            patience = 0
        else:
            patience += 1

        # Periodic exact scoring. Exact is authoritative, but direct O(n^2)
        # convolution is slower, so it is sampled.
        if (
            M < best_M_exact - 1e-9
            or now - last_exact > 3.0
            or (it % 120 == 0)
        ):
            sc = exact_score(a)
            last_exact = now
            if np.isfinite(sc):
                if sc < best_exact - 1e-12:
                    best_exact = sc
                    best_a = normalize_arr(a).copy()
                    best_M_exact = sc / (2.0 * n)
                    best_est = min(best_est, M)
                    patience = 0

        # Construct active-lag weights.
        w = make_weights(b, M, q)
        if lambda_dual is not None:
            mix = 0.45
            w = (1.0 - mix) * w + mix * lambda_dual
            w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
            w = np.maximum(w, 0.0)
            ws = float(w.sum())
            if ws <= 0.0 or not np.isfinite(ws):
                w = make_weights(b, M, q)
            else:
                w /= ws

        y = load_weights(w, Fa)
        if not np.any(np.isfinite(y)) or float(y.max()) <= 0.0:
            y = np.ones(n, dtype=np.float64) / n

        a2 = mirror_update(a, y, eta)
        F2, b2, M2 = conv_self(a2)

        accepted = False
        if np.isfinite(M2) and M2 <= M * 1.25:
            accepted = True
        else:
            # If the step is too aggressive, try a smaller one.
            a2s = mirror_update(a, y, eta * 0.35)
            F2s, b2s, M2s = conv_self(a2s)
            if np.isfinite(M2s) and M2s <= max(M * 1.10, M2 * 0.98):
                a2, F2, b2, M2 = a2s, F2s, b2s, M2s
                accepted = True

        if accepted:
            a = a2
            b = b2
            Fa = F2
            M = M2
            if M < M_old := a[:0].sum():  # placeholder no-op to avoid lint
                pass

        # Note: M_old needed for numerical adaptation; compute before update.
        # The placeholder above is harmless, but we maintain proper adaptivity below
        # using current M versus previous M from before this iteration.

        # Simple eta adaptation is intentionally conservative.
        if np.isfinite(M) and np.isfinite(best_est):
            if M < best_est * (1.0 - 1e-7):
                eta = min(eta_max, eta * 1.012)
            elif M > best_est * (1.0 + 1e-3):
                eta = max(eta_min, eta * 0.90)

        # Slowly increase focus.
        q = min(max_q, q * 1.0025)

        # Escaping stagnation.
        if patience >= 450:
            a = perturb_arr(a, sigma=0.18, reverse_blend=0.0)
            Fa, b, M = conv_self(a)
            q = max(7.0, q * 0.65)
            eta = max(0.30, eta * 1.15)
            patience = 0

        if patience >= 1200:
            sp = spectral_kick(a, iters=5, mix=0.65)
            Fa_s, b_s, M_s = conv_self(sp)
            if np.isfinite(M_s) and M_s < M * 1.02:
                a = sp
                Fa, b, M = Fa_s, b_s, M_s
            else:
                a = perturb_arr(a, sigma=0.32, reverse_blend=0.20)
                Fa, b, M = conv_self(a)
            q = max(6.0, q * 0.5)
            eta = 0.5
            patience = 0

        # Occasionally exploit the reversal symmetry by trying a symmetric blend.
        if it % 1500 == 0 and time.monotonic() < deadline - 30.0:
            try:
                sym = normalize_arr(0.5 * (a + a[::-1]))
                Fs, bs, Ms = conv_self(sym)
                if np.isfinite(Ms) and Ms < M * (1.0 - 1e-7):
                    a = sym
                    Fa, b, M = Fs, bs, Ms
            except Exception:
                pass

        # Frank-Wolfe/LP direction, using the more global SOTA idea.
        if (
            it >= 50
            and now - last_lp > 70.0
            and time.monotonic() < deadline - 100.0
        ):
            remain = deadline - time.monotonic()
            if remain > 90.0:
                try:
                    h, dual = lp_direction(a, b, M, Fa, remain, lp_K)
                    if dual is not None:
                        lambda_dual = dual
                    if h is not None:
                        blend = apply_lp_blend(a, b, M, Fa, h)
                        if blend is not None:
                            a, b, M, Fa = blend
                            patience = 0
                            sc = exact_score(a)
                            if np.isfinite(sc) and sc < best_exact - 1e-12:
                                best_exact = sc
                                best_a = normalize_arr(a).copy()
                                best_M_exact = sc / (2.0 * n)
                                best_est = min(best_est, M)
                except Exception:
                    pass
                last_lp = time.monotonic()

    # ------------------------------------------------------------------
    # Final polish attempts.
    # ------------------------------------------------------------------
    if minimize is not None and time.monotonic() < deadline - 25.0:
        starts = [best_a]
        if best_a.size == n and np.array_equal(best_a, normalize_arr(a)) is False:
            starts.append(normalize_arr(a))

        for st in starts:
            if time.monotonic() > deadline - 12.0:
                break
            try:
                p = lbfgs_polish(st, maxfun=45, temp=35.0)
                if p is not None:
                    sc = exact_score(p)
                    if np.isfinite(sc) and sc < best_exact - 1e-12:
                        best_exact = sc
                        best_a = normalize_arr(p).copy()
                        best_M_exact = sc / (2.0 * n)
            except Exception:
                pass

    # Final validation and safe return.
    try:
        current = normalize_arr(a)
        sc_cur = exact_score(current)
        if np.isfinite(sc_cur) and sc_cur < best_exact - 1e-12:
            best_exact = sc_cur
            best_a = current.copy()
    except Exception:
        pass

    out = normalize_arr(best_a)
    if out.size == 0:
        out = np.ones(1000)

    sc_out = exact_score(out)
    if (not np.isfinite(sc_out)) or sc_out > 2.5:
        out = np.ones(n, dtype=np.float64) / float(max(1, n))
        sc_out = exact_score(out)
        if not np.isfinite(sc_out):
            out = np.ones(1000, dtype=np.float64)

    # Force a small safety margin before the hard deadline.
    if time.monotonic() > deadline - 0.5:
        pass

    return [float(x) for x in out]
