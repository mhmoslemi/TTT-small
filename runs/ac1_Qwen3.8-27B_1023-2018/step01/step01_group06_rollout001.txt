import numpy as np
import time

_INF = float("inf")


def _clean_array(a):
    """Return a finite 1-D float array clipped to the evaluator's domain."""
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    arr = np.minimum(arr, 1000.0)
    return arr


def _safe_score(a):
    """Evaluate a candidate with the supplied exact evaluator, never raising."""
    arr = _clean_array(a)
    if arr is None:
        return _INF
    try:
        sc = evaluate_sequence(arr.tolist())
        sc = float(sc)
    except Exception:
        return _INF
    if not np.isfinite(sc):
        return _INF
    return sc


def _renorm(a):
    """Nonnegative finite renormalization to sum 1, with small floor."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-18)
    a = np.minimum(a, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.full(a.shape, 1.0 / max(1, a.size), dtype=np.float64)
    return a / s


def _improve(a, best_score, n, seconds, seed):
    """
    Bounded nonnegative local search.

    This is mainly a hedge for environments where the supplied evaluator is
    not the simple nonnegative autocorrelation bound (in which case the exact
    evaluator is used as the oracle).  Under the printed evaluator, the optimum
    is uniform and best_score should be 2, so this path normally is skipped.

    if seconds <= 0 or n <= 1:
        return _renorm(a), best_score

    t0 = time.monotonic()
    end = t0 + max(0.0, float(seconds))

    a = _renorm(a)

    eta = 0.55
    step = 0
    stagnant = 0
    eval_every = 20

    while time.monotonic() < end:

        conv = np.where(conv > 0.0, conv, 0.0)

        else:
            weights = np.ones(conv_len, dtype=np.float64)

        w_rev = weights[::-1]
        FW = np.fft.rfft(w_rev, fft_n)
        c = np.fft.irfft(F * FW, fft_n)
        load = c[idx_load]
        load = np.where(load > 0.0, load, 0.0)

        lm = float(load.max())
        if lm > 1e-15:
            a *= np.exp(-eta * load / lm)

        a = _renorm(a)
        step += 1

        if step % eval_every == 0:
            sc = _safe_score(a)
            if sc < best_s:
                best_s = sc
                best = a.copy()
                stagnant = 0
            else:
                stagnant += 1

        if step % 500 == 0:
            mask = rng.random(n) < max(0.05, 0.02 + 0.01 * min(stagnant, 10))
            if not mask.any():
                mask[int(rng.integers(n))] = True
            sigma = 0.15 + 0.03 * min(stagnant, 20)
            a[mask] *= np.exp(rng.normal(0.0, sigma, int(mask.sum())))
            if rng.random() < 0.25:
                beta = rng.uniform(0.1, 0.4)
                a = (1.0 - beta) * a + beta * a[::-1]
            a = _renorm(a)
            stagnant = 0

        if step % 2000 == 0:
            eta = max(0.25, eta * 0.94)
            top_frac = max(0.06, top_frac * 0.96)
            if rng.random() < 0.15:
                eta = rng.uniform(0.4, 0.9)
                top_frac = rng.uniform(0.10, 0.35)

    sc = _safe_score(a)
    if sc < best_s:
        best_s = sc
        best = a.copy()

    return best, best_s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    """
    Return the best valid coefficient list found before the deadline.

    For the exact evaluator shown in the prompt, every real sequence satisfies
    score = 2*n*max(a*a)/sum(a)^2 >= 2*n*sum(a^2)/sum(a)^2 >= 2,
    with equality for a constant positive sequence.  The program still probes
    a few candidate families and, only if the runtime evaluator reports a
    score significantly below 2, runs a bounded local search.
    """
    try:
        seed_int = int(seed)
    except Exception:
        seed_int = 42

    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    deadline = t0 + max(1.0, budget - 5.0)

    # Operational length.  The supplied construction length is 4498; fall back
    # to that if the parent is unavailable or pathologically large.
    n = 4498
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size > 0 and p.size <= 20000:
            n = int(p.size)
            parent = p
    except Exception:
        parent = None

    if n < 1:
        n = 1

    rng = np.random.default_rng(seed_int)

    uniform = np.ones(n, dtype=np.float64)
    best = uniform.copy()
    best_sc = _safe_score(uniform)

    candidates = [uniform]

    if parent is not None:
        pc = _clean_array(parent)
        if pc is not None and pc.size == n:
            candidates.append(pc)
            s = float(pc.sum())
            if s > 1e-12:
                candidates.append(pc / s)
            candidates.append(pc[::-1])
            if n > 1:
                candidates.append(0.5 * (pc + pc[::-1]))

    if n > 1:
        j = np.arange(n, dtype=np.float64)
        candidates.append(np.sin(np.pi * (j + 0.5) / n))
        candidates.append(np.exp(-((j - 0.5 * n) / (0.20 * n)) ** 2))
        candidates.append(
            np.exp(-((j - 0.30 * n) / (0.10 * n)) ** 2)
            + 0.7 * np.exp(-((j - 0.70 * n) / (0.10 * n)) ** 2)
        )
        if n >= 3:
            candidates.append(np.maximum(0.0, 1.0 - np.abs((j - 0.5 * n) / (0.5 * n))))
        candidates.append(1.0 + 0.2 * rng.random(n))
    else:
        candidates.append(np.ones(1, dtype=np.float64))

    for cand in candidates:
        if time.monotonic() > deadline - 2.0:
            break
        ac = _clean_array(cand)
        if ac is None or ac.size != n:
            continue
        s = float(ac.sum())
        if s <= 1e-12:
            continue
        vv = ac / s
        sc = _safe_score(vv)
        if sc < best_sc:
            best_sc = sc
            best = vv.copy()

    # If the runtime evaluator permits scores materially below the nonnegative
    # autocorrelation lower bound, use remaining time to try to improve it.
    if best_sc < 2.0 - 1e-3 and 1 < n <= 10000:
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 5.0:
            seconds = min(max(10.0, remaining), 1200.0)
            best, best_sc = _improve(best, best_sc, n, seconds, seed_int + 1)

    # Final validation: compare the retained best with a clean uniform vector.
    final_arr = None
    final_sc = _INF

    for arr in (best, uniform):
        ac = _clean_array(arr)
        if ac is None or ac.size != n:
            continue
        if time.monotonic() > deadline - 1.0:
            # If extremely tight, keep the previous validated best if possible.
            if final_arr is None:
                final_arr = ac
                final_sc = _INF
            continue
        sc = _safe_score(ac)
        if sc < final_sc:
            final_sc = sc
            final_arr = ac

    if final_arr is None:
        final_arr = np.ones(1, dtype=np.float64)
        n = 1

    final_arr = _clean_array(final_arr)
    if final_arr is None:
        final_arr = np.ones(n, dtype=np.float64)

    if float(final_arr.sum()) < 0.01:
        final_arr = np.ones(n, dtype=np.float64)

    return [float(x) for x in final_arr.tolist()]
