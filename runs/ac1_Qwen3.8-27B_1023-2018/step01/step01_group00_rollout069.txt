import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _softmax(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return np.array([1.0])
    x = x - np.max(x)
    e = np.exp(np.clip(x, -80.0, 80.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)
    return e / s


def _conv(a, n, fs):
    F = rfft(a, fs)
    return irfft(F * F, fs)[: (2 * n - 1)]


def _cc(q, a, n, fs2):
    ar = a[::-1]
    c = irfft(rfft(ar, fs2) * rfft(q, fs2), fs2)
    return c[n - 1 : 2 * n - 1]


def _hard_score(a, n, fs):
    b = _conv(a, n, fs)
    m = float(b.max())
    if not np.isfinite(m) or m < 0.0:
        return math.inf
    return 2.0 * n * m


def _init_a(n, variant, rng, parent):
    t = (np.arange(n) + 0.5) / n
    x = 2.0 * t - 1.0
    v = int(variant) % 17
    if v == 0:
        a = np.ones(n)
    elif v == 1:
        a = 1.0 - np.abs(x)
    elif v == 2:
        a = 1.0 - x * x
    elif v == 3:
        a = 0.5 * (1.0 + np.cos(np.pi * x))
    elif v == 4:
        a = (1.0 - np.abs(x)) ** 3
    elif v == 5:
        a = np.exp(-2.5 * np.abs(x))
    elif v == 6:
        a = 0.5 + 0.5 * np.sin(np.pi * t)
    elif v == 7:
        a = np.exp(-3.0 * t)
    elif v == 8:
        a = np.exp(-3.0 * (1.0 - t))
    elif v == 9:
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
    elif v == 10:
        a = 0.5 + 0.5 * np.cos(np.pi * x)
    elif v == 11:
        a = np.ones(n)
        for _ in range(3):
            f = int(rng.integers(1, max(2, n // 2)))
            ph = float(rng.uniform(0.0, 2.0 * math.pi))
            amp = float(rng.uniform(0.3, 0.9))
            a += amp * np.sin(2.0 * math.pi * f * t + ph)
    elif v == 12:
        a = np.abs(np.sin(np.pi * t))
    elif v == 13:
        a = 1.0 / (1.0 + np.exp(15.0 * x))
    elif v == 14:
        a = 0.5 * (1.0 - np.abs(x)) + 0.5 / (1.0 + np.exp(20.0 * x))
    elif v == 15 and parent is not None and parent.size > 0:
        P = parent.size
        idx = (np.arange(n) * P // n).astype(int)
        a = parent[idx].copy()
    else:

    a = a + 1e-3 * np.abs(rng.normal(0.0, 1.0, n))
    a = np.maximum(a, 1e-6)
    a = a / max(1e-300, float(a.sum()))
    return a


def _make_job(n, a0, w, seed):
    fs_c = next_fast_len(2 * n - 1)
    fs2 = next_fast_len(3 * n - 2)
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.clip(a0, 1e-6, 1000.0)
    a0 = a0 / max(1e-300, float(a0.sum()))
    z = np.log(a0)
    z = z - z.mean()
    rng = np.random.default_rng(seed)
    return dict(
        n=n, fs_c=fs_c, fs2=fs2, w=float(w), rng=rng,
        z=z, m=np.zeros(n), v=np.zeros(n), t=0,
        tau_frac=0.45, lr=0.45, st=0, it=0,

def _run(job, end_time, hard_deadline):

    fs_c = job['fs_c']
    fs2 = job['fs2']
    z = job['z'].copy()

        g = 2.0 * _cc(p, a, n, fs2)
        ugrad = a * (g - float(np.dot(a, g)))
        grad = ugrad
        if job['w'] > 0.0:
            q = b - 1.0 / (2.0 * n - 1.0)
            vg = 4.0 * _cc(q, a, n, fs2)
            grad = grad + job['w'] * (a * (vg - float(np.dot(a, vg))))

        gn = float(np.abs(grad).max()) if grad.size else 0.0
        if (not np.isfinite(gn)) or gn < 1e-15:
            job['st'] += 1
            if job['st'] > 40:
                z = best_z + job['rng'].normal(0.0, 0.6, n)
                z = z - z.mean()
                job['m'] = np.zeros(n)
                job['v'] = np.zeros(n)
                job['t'] = 0
                job['lr'] = 0.45
                job['tau_frac'] = 0.35
                job['st'] = 0
            continue

        grad = grad / (gn + 1e-300)
        job['t'] += 1
        job['m'] = 0.9 * job['m'] + 0.1 * grad
        job['v'] = 0.999 * job['v'] + 0.001 * (grad * grad)
        mh = job['m'] / (1.0 - 0.9 ** job['t'])
        vh = job['v'] / (1.0 - 0.999 ** job['t'])
        upd = mh / (np.sqrt(vh) + 1e-8)
        um = float(np.abs(upd).max()) if upd.size else 0.0
        if (not np.isfinite(um)) or um <= 0.0:
            continue
        upd = upd / (um + 1e-300)

        z = z - job['lr'] * upd
        z = z - z.mean()
        z = np.clip(z, -30.0, 30.0)
        z[~np.isfinite(z)] = 0.0

        a2 = _softmax(z)
        b2 = _conv(a2, n, fs_c)
        score = 2.0 * n * max(float(b2.max()), 0.0)

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_z = z.copy()
            job['st'] = 0
        else:
            job['st'] += 1

        job['it'] += 1
        if job['it'] % 12 == 0:
            job['tau_frac'] = max(0.03, job['tau_frac'] * 0.96)
        if job['st'] > 25:
            job['lr'] = max(0.05, job['lr'] * 0.85)
        if job['st'] > 140:
            z = best_z + job['rng'].normal(0.0, 0.6, n)
            z = z - z.mean()
            job['m'] = np.zeros(n)
            job['v'] = np.zeros(n)
            job['t'] = 0
            job['lr'] = 0.45
            job['tau_frac'] = 0.35
            job['st'] = 0

    job['z'] = best_z
    a = _softmax(best_z)
    b = _conv(a, n, job['fs_c'])
    s = 2.0 * n * max(float(b.max()), 0.0)
    return a, s


def _polish(a, polish_end, hard_deadline):
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        a = np.ones(32) / 32.0
    if a.size < 2 or not np.isfinite(a).all():
        a = np.ones(32) / 32.0
    a = np.clip(a, 1e-9, 1000.0)
    a = a / max(1e-300, float(a.sum()))
    n = a.size
    fs_c = next_fast_len(2 * n - 1)
    best = a.copy()
    bs = _hard_score(best, n, fs_c)
    if not np.isfinite(bs):
        bs = 2.0

    rng = np.random.default_rng(987651)
    sigma = 0.25
    stuck = 0
    base_mean = float(a.mean())

    while time.monotonic() < min(polish_end, hard_deadline) - 0.2:
        z = np.log(a)
        r = float(rng.random())
        znd = (time.monotonic() > min(polish_end, hard_deadline) - 0.6)
        if r < 0.28 or znd:
            zc = z + rng.normal(0.0, sigma, n)
        elif r < 0.40:
            i = int(rng.integers(0, n))
            w = max(1, min(n, int(rng.integers(1, max(2, n // 25 + 1)))))
            i2 = min(n, i + w)
            zc = z.copy()
            zc[i:i2] += rng.normal(0.0, sigma * 0.6, i2 - i)
        elif r < 0.52:
            zc = z.copy()
            zv = z - z.mean()
            zc = zv * float(rng.uniform(0.4, 2.2))
        elif r < 0.64:
            a2 = a.copy()
            i = int(rng.integers(0, n))
            w = max(1, min(n, int(rng.integers(1, max(2, n // 30 + 1)))))
            i2 = min(n, i + w)
            a2[i:i2] *= float(np.exp(rng.normal(0.0, 0.35)))
            zc = np.log(np.maximum(a2, 1e-12))
        elif r < 0.78:
            k = float(rng.uniform(0.5, 5.0))
            zc = z + float(rng.uniform(-sigma, sigma)) * np.sin(np.pi * np.arange(n) / max(1, n) + float(rng.uniform(0, 2 * math.pi)))
        else:
            zc = 0.8 * z + 0.2 * np.log(np.maximum(base_mean + a, 1e-12) / base_mean)
            zc = zc - zc.mean()

        zc = zc - zc.mean()
        a2 = _softmax(zc)
        s2 = _hard_score(a2, n, fs_c)
        if np.isfinite(s2) and s2 < bs - 1e-11:
            best = a2.copy()
            bs = s2
            a = a2.copy()
            stuck = 0
            sigma = max(0.05, min(sigma, 0.4))
        else:
            stuck += 1
            sigma = min(0.9, sigma * 1.05)
            if stuck > 60:
                stuck = 0
                sigma = 0.5

    a = np.asarray(best, dtype=np.float64).ravel()
    a = np.clip(a, 1e-9, 1000.0)
    a = a / max(1e-300, float(a.sum()))
    s = _hard_score(a, n, fs_c)
    if not np.isfinite(s):
        return np.ones(32) / 32.0, 2.0
    return a, s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    total = max(0.0, float(budget_s))
    hard_deadline = t0 + max(0.0, total) - 5.0
    explore_end = t0 + max(30.0, 0.70 * total)
    polish_end = t0 + max(explore_end, min(0.92 * total, max(0.0, total) - 12.0))
    polish_end = min(polish_end, t0 + max(0.0, total) - 12.0)
    if polish_end < explore_end:
        polish_end = explore_end + max(5.0, (t0 + max(0.0, total) - 12.0) - explore_end)

    seed = int(seed)

    # parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0 and np.isfinite(p).all():
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        parent = None

    # fallback + initial best
    best_a = np.ones(32) / 32.0
    best_score = 2.0
    try:
        if parent is not None and parent.size > 0:
            fs = float(evaluate_sequence([float(v) for v in parent]))
            if np.isfinite(fs):
                best_a = np.clip(parent, 1e-9, 1000.0).astype(np.float64).copy()
                best_a = best_a / max(1e-300, float(best_a.sum()))
                best_score = fs
    except Exception:
        try:
            fs = float(evaluate_sequence([1.0]))
            if np.isfinite(fs):
                best_a = np.array([1.0])
                best_score = fs
        except Exception:
            best_a = np.ones(32) / 32.0
            best_score = 2.0
    if not np.isfinite(best_score):
        best_a = np.ones(32) / 32.0
        best_score = 2.0

    # build jobs
    plan = {192: 4, 256: 3, 384: 3, 512: 5, 768: 4, 1024: 6, 1536: 5, 2048: 4, 3072: 4, 4096: 3}
    wlist = [0.0, 0.0, 0.0, 0.05, 0.1, 0.15, 0.25]
    jobs = []
    i = 0
    parent_len = parent.size if parent is not None else 0
    seeded_parent = False
    for size in [192, 256, 384, 512, 768, 1024, 1536, 2048, 3072, 4096]:
        for k in range(plan.get(size, 3)):
            variant = (i * 5 + 3) % 17
            w = wlist[i % len(wlist)]
            if (not seeded_parent) and parent is not None and parent.size > 1:
                jn = int(round(parent.size / size) * size) if size <= parent.size else size
                if jn == parent.size:
                    jn = size
                    # use parent directly for the matching-size job (first size==parent_len)
                    pass
            a0 = _init_a(size, variant, np.random.default_rng(seed * 1000 + i * 7), parent)
            # for the first job at the parent's length, prefer the parent itself
            if (not seeded_parent) and size == parent_len and parent is not None:
                a0 = parent.copy()
                seeded_parent = True
            elif variant == 15 and parent is not None:
                pass  # already handled inside _init_a
            job = _make_job(size, a0, w, seed * 13 + i * 101 + 5)
            jobs.append(job)
            i += 1

    order = list(range(len(jobs)))

    # round-robin exploration
    def _try_update(a, s):
        nonlocal best_a, best_score
        if a is None or s is None:
            return
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
            if a.size == 0 or not np.isfinite(a).all():
                return
            a = np.clip(a, 1e-9, 1000.0)
            if float(a.sum()) < 1e-6:
                return
            a = a / float(a.sum())
        except Exception:
            return
        if np.isfinite(s) and s < best_score - 1e-12:
            best_a = a.copy()
            best_score = float(s)

    try:
        while time.monotonic() < explore_end:
            for oi in order:
                if time.monotonic() >= explore_end:
                    break
                rem = explore_end - time.monotonic()
                if rem < 0.2:
                    break
                slic = min(1.0, rem)
                end_time = time.monotonic() + slic
                job = jobs[oi]
                try:
                    a, s = _run(job, end_time, hard_deadline)
                    _try_update(a, s)
                except Exception:
                    pass
            if best_score < 1.14:
                break
    except Exception:
        pass

    # polish phase: Adam rerun from current best, then local search
    try:
        if time.monotonic() < polish_end - 0.5:
            zbest = np.log(np.clip(best_a, 1e-9, 1000.0))
            zbest = zbest - zbest.mean()
            pbest = np.exp(np.clip(zbest, -30, 30))
            pbest = pbest / max(1e-300, float(pbest.sum()))
            jobb = _make_job(pbest.size, pbest, 0.0, seed * 777 + 3)
            adam_end = max(explore_end, polish_end - 0.40 * (polish_end - max(explore_end, polish_end - 40.0)))
            adam_end = min(adam_end, polish_end - 0.15 * max(1.0, polish_end - t0))
            adam_end = max(explore_end - 5.0, min(adam_end, polish_end - 0.1 * max(1.0, polish_end - t0)))
            try:
                a, s = _run(jobb, adam_end, hard_deadline)
                _try_update(a, s)
            except Exception:
                pass
            best_a, best_score = _polish(best_a, polish_end, hard_deadline)
    except Exception:
        pass

    # final exact validation
    try:
        cand = [float(v) for v in best_a]
    except Exception:
        cand = [1.0]
    try:
        fs = float(evaluate_sequence(cand))
    except Exception:
        fs = math.inf
    if not np.isfinite(fs) or fs > best_score + 0.5:
        cand = [1.0]
        try:
            fs = float(evaluate_sequence(cand))
        except Exception:
            fs = math.inf
        if not np.isfinite(fs):
            return [1.0]
    return cand
