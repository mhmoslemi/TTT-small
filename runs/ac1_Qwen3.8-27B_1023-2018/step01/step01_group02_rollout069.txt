def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _match_length(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == n:
        return _normalize(a)
    if a.size == 0 or n <= 0:
        return np.ones(max(1, n), dtype=np.float64) / float(max(1, n))
    if n == 1:
        x = float(np.sum(a))
        if not np.isfinite(x) or x <= 0.0:
            return np.ones(1, dtype=np.float64)
        return np.array([1.0], dtype=np.float64)
    x0 = np.linspace(0.0, 1.0, a.size)
    x1 = np.linspace(0.0, 1.0, n)
    y = np.interp(x1, x0, a)
    return _normalize(y)


def _softmax_fast(z):
    z = np.asarray(z, dtype=np.float64)
    n = z.size
    if n == 0:
        return np.array([], dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    zm = float(np.max(z))
    if not np.isfinite(zm):
        zm = 0.0
    w = np.exp(z - zm)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return w / s


def _score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return float("inf")
    c = np.convolve(a, a)
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * float(a.size) * m / (s * s))


def _cross_corr(p, a):
    # For linear convolution c = conv(a, a) and weights p over the 2n-1 lags,
    # d(sum_k p[k] c[k])/da = 2 * cross_corr(p, a) where
    # cross_corr[j] = sum_k p[k] * a[k-j] (a is zero outside [0, n-1]).
    n = a.size
    full = np.convolve(p, a[::-1], mode="full")
    return full[n - 1:2 * n - 1]


def _gradient_adv(a, c, beta):
    n = a.size
    if c.size == 0 or n == 0:
        return np.zeros(n, dtype=np.float64)
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.zeros(n, dtype=np.float64)

    z = beta * (c - m)
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    es = float(e.sum())
    if not np.isfinite(es) or es <= 0.0:
        p = np.zeros_like(c, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
    else:
        p = e / es

    r = _cross_corr(p, a)
    if not np.all(np.isfinite(r)):
        r = np.zeros(n, dtype=np.float64)

    # Objective on the simplex: 2n * soft_max(c).
    # d(soft_max)/da = 2 * r, hence unconstrained a-gradient is 4n*r.
    g = 4.0 * float(n) * r
    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=np.float64)

    adv = g - float(np.dot(a, g))
    adv = np.where(np.isfinite(adv), adv, 0.0)
    return adv

        if c.size == 0:
            break
        s = float(a.sum())
        m = float(np.max(c))
        if not np.isfinite(s) or s <= 1e-300 or not np.isfinite(m):
            z = np.log(np.maximum(best_a, 1e-300))
            z -= float(np.mean(z))
            z = np.clip(z, -50.0, 50.0)
            prev_score = None
            best_no = 0
            stuck_no = 0
            continue

        score = 2.0 * float(n) * m / (s * s)

        if prev_score is not None and score < prev_score - 1e-13:
            stuck_no = 0
        else:
            stuck_no += 1
        prev_score = score

        if score < best_s - 1e-12:
            best_s = score
            best_a = a.copy()
            best_no = 0
            stuck_no = 0
            last_imp = now
            if best_s < state["best_score"] - 1e-12:
                state["best_score"] = best_s
                state["best_a"] = best_a.copy()
                _add_hall(state, best_a, best_s)
        else:
            best_no += 1

        bt = (now - reset_time) / beta_time
        bt = min(1.0, max(0.0, bt))
        beta = beta0 * (beta1 / beta0) ** bt

        if beta > 2000.0 and c.size > 1:
            second = float(np.partition(c, -2)[-2])
            gap = m - second
            if not np.isfinite(gap) or gap < 1e-8:
                gap = 1e-8
            beta = min(beta, 1.0 / gap)

        adv = _gradient_adv(a, c, beta)
        norm = float(np.linalg.norm(adv)) / math.sqrt(float(n))
        if not np.isfinite(norm) or norm < 1e-14:
            dirv = rng.standard_normal(n)
        else:
            dirv = adv / norm

        if best_no > max(200, int(P["patience"]) // 10):
            dirv = dirv + float(P["noise_dir"]) * rng.standard_normal(n)
            nd = float(np.linalg.norm(dirv))
            if np.isfinite(nd) and nd > 1e-14:
                dirv = dirv / nd
            else:
                dirv = rng.standard_normal(n)

        if stuck_no > 350:
            eta = max(float(P["eta_min"]), eta * 0.92)

        tol = (1e-10 if best_no > 200 else 1e-12) * max(1.0, score)
        accepted = False
        trial_score = None

        for mult in (1.0, 0.5, 0.25):
            step = mult * eta
            z_try = z - step * dirv
            z_try = z_try - float(np.mean(z_try))
            z_try = np.clip(z_try, -50.0, 50.0)

            a_try = _softmax_fast(z_try)
            c_try = np.convolve(a_try, a_try)
            s_try = float(a_try.sum())
            if c_try.size == 0 or not np.isfinite(s_try) or s_try <= 1e-300:
                continue
            m_try = float(np.max(c_try))
            sc_try = 2.0 * float(n) * m_try / (s_try * s_try)
            if not np.isfinite(sc_try):
                continue

            if sc_try <= score + tol:
                z = z_try
                trial_score = sc_try
                accepted = True
                break

        if accepted and trial_score is not None:
            if trial_score < best_s - 1e-12:
                best_s = trial_score
                best_a = _softmax_fast(z)
                best_no = 0
                stuck_no = 0
                last_imp = now
                if best_s < state["best_score"] - 1e-12:
                    state["best_score"] = best_s
                    state["best_a"] = best_a.copy()
                    _add_hall(state, best_a, best_s)
        else:
            if rng.random() < 0.35:
                z = z + float(P["noise_z"]) * rng.standard_normal(n)
                z = z - float(np.mean(z))
                z = np.clip(z, -50.0, 50.0)
            else:
                z_try = z - 0.1 * eta * dirv
                z_try = z_try - float(np.mean(z_try))
                z_try = np.clip(z_try, -50.0, 50.0)
                a_try = _softmax_fast(z_try)
                c_try = np.convolve(a_try, a_try)
                s_try = float(a_try.sum())
                if c_try.size and np.isfinite(s_try) and s_try > 1e-300:
                    sc_try = 2.0 * float(n) * float(np.max(c_try)) / (s_try * s_try)
                    if np.isfinite(sc_try) and sc_try < score * 1.05 + 1e-10:
                        z = z_try
            eta = max(float(P["eta_min"]), eta * 0.75)

        if stuck_no > int(P["patience"]) or (now - last_imp) > float(P["imp_window"]):
            hall = state.get("hall")
            if hall and rng.random() < 0.35:
                idx = int(rng.integers(0, len(hall)))
                base = hall[idx][1]
            else:
                base = best_a
            base = _match_length(base, n)
            noise = float(P["noise"]) * (1.0 + 0.5 * float(min(1.0, best_no / 20000.0)))
            z = np.log(np.maximum(base, 1e-300)) + noise * rng.standard_normal(n)
            z = z - float(np.mean(z))
            z = np.clip(z, -50.0, 50.0)
            eta = float(P["eta0"])
            best_no = 0
            stuck_no = 0
            prev_score = None
            reset_time = now
            last_imp = now

    _add_hall(state, best_a, best_s)
    return best_a, best_s


def _polish(a, rng, n, max_trials=100, end=None):
    best = _match_length(a, n)
    best_s = _score_fast(best)
    if end is None:
        end = time.monotonic() + 1.0

    for _ in range(int(max(0, max_trials))):
        if time.monotonic() >= end:
            break
        cand = best.copy()
        mode = int(rng.integers(0, 4))
        if mode == 0:
            w = int(rng.integers(max(3, n // 30), max(4, n // 5)))
            w = min(max(1, w), n)
            st = int(rng.integers(0, n - w + 1))
            cand[st:st + w] *= math.exp(float(rng.normal(0.0, 0.22)))
        elif mode == 1 and n > 6:
            w1 = int(rng.integers(max(3, n // 40), max(4, n // 8)))
            w2 = int(rng.integers(max(3, n // 40), max(4, n // 8)))
            w1 = min(max(1, w1), n)
            w2 = min(max(1, w2), n)
            st1 = int(rng.integers(0, n - w1 + 1))
            st2 = int(rng.integers(0, n - w2 + 1))
            cand[st1:st1 + w1] *= math.exp(float(rng.normal(0.0, 0.25)))
            cand[st2:st2 + w2] /= math.exp(float(rng.normal(0.0, 0.25)))
        elif mode == 2:
            off = int(rng.integers(-max(1, n // 20), max(1, n // 20) + 1))
            cand = 0.5 * (cand + np.roll(best, off))
        else:
            cand *= math.exp(float(rng.normal(0.0, 0.08)))

        cand = _normalize(cand)
        sc = _score_fast(cand)
        if np.isfinite(sc) and sc < best_s - 1e-12:
            best_s = sc
            best = cand

    return best, best_s


def _softmax_floor(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        z = np.interpolate(z, None, n)  # unreachable in normal use
    z = np.where(np.isfinite(z), z, 0.0)
    zm = float(np.max(z))
    if not np.isfinite(zm):
        zm = 0.0
    w = np.exp(z - zm)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    a = w / s
    a = np.maximum(a, 1e-15)
    return a / float(a.sum())


def _lbfgs_refine(a, beta, n):
    if _sp_minimize is None or n <= 1:
        return a
    a = _match_length(a, n)
    z0 = np.log(np.maximum(a, 1e-300)).astype(np.float64)
    z0 -= float(np.mean(z0))

    def fg(zv):
        zv = np.asarray(zv, dtype=np.float64).ravel()
        if zv.size != n or not np.all(np.isfinite(zv)):
            return 1e100, np.zeros(n, dtype=np.float64)

        cur_a = _softmax_floor(zv, n)
        c = np.convolve(cur_a, cur_a)
        m = float(np.max(c))
        if not np.isfinite(m):
            return 1e100, np.zeros(n, dtype=np.float64)

        zz = beta * (c - m)
        zz = np.clip(zz, -745.0, 0.0)
        e = np.exp(zz)
        es = float(e.sum())
        if not np.isfinite(es) or es <= 0.0:
            p = np.zeros_like(c, dtype=np.float64)
            p[int(np.argmax(c))] = 1.0
            m_soft = m
        else:
            p = e / es
            m_soft = m + math.log(es) / beta

        r = _cross_corr(p, cur_a)
        if not np.all(np.isfinite(r)):
            r = np.zeros(n, dtype=np.float64)
        g = 4.0 * float(n) * r
        adv = g - float(np.dot(cur_a, g))
        grad = cur_a * adv
        grad = np.where(np.isfinite(grad), grad, 0.0)

        val = 2.0 * float(n) * m_soft
        if not np.isfinite(val):
            return 1e100, np.zeros(n, dtype=np.float64)
        return float(val), grad

    try:
        res = _sp_minimize(
            fg,
            z0,
            jac=True,
            method="L-BFGS-B",
            options={
                "maxiter": 60,
                "maxcor": 10,
                "ftol": 1e-10,
                "maxls": 20,
                "disp": 0,
            },
        )
        if getattr(res, "x", None) is not None:
            cand = _softmax_floor(res.x, n)
            if cand.size == n and np.all(np.isfinite(cand)):
                return cand
    except Exception:
        pass
    return a


def _clean_list(a):
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s):
        return None
    if s <= 1e-300:
        return None
    if s < 0.01:
        arr = arr / s
    return [float(x) for x in arr]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    budget = max(1.0, float(budget_s))
    safety = 20.0 if budget > 120.0 else max(5.0, 0.1 * budget)
    deadline = start + max(1.0, budget - safety)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if p.size > 0 and np.isfinite(s) and s > 1e-12:
            parent = p / s
    except Exception:
        parent = None

    if parent is None:
        n = 127
        parent = np.ones(n, dtype=np.float64) / float(n)
    else:
        n = int(parent.size)

    state = {
        "best_a": parent.copy(),
        "best_score": _score_fast(parent),
        "hall": [],
    }
    _add_hall(state, parent, state["best_score"])

    inits = _make_inits(n, parent, rng, max_count=18)
    order = list(range(len(inits)))
    rng.shuffle(order)

    phase1_time = 0.0
    if deadline - start > 60.0:
        phase1_time = min(500.0, 0.35 * budget)
        phase1_time = min(phase1_time, max(0.0, deadline - start - 120.0))

    P_A = {
        "beta0": 30.0,
        "beta1": 8000.0,
        "eta0": 1.35,
        "eta_min": 0.22,
        "patience": 700,
        "imp_window": 25.0,
        "noise": 0.55,
        "noise_dir": 0.12,
        "noise_z": 0.08,
        "beta_time": 60.0,
    }
    P_B = {
        "beta0": 900.0,
        "beta1": 80000.0,
        "eta0": 0.55,
        "eta_min": 0.045,
        "patience": 2600,
        "imp_window": 45.0,
        "noise": 0.08,
        "noise_dir": 0.04,
        "noise_z": 0.02,
        "beta_time": 250.0,
    }

    if phase1_time > 15.0:
        phase1_end = start + phase1_time
        count = int(min(len(inits), max(2, phase1_time / 80.0)))
        slice_time = phase1_time / max(1, count)
        for idx in order[:count]:
            if time.monotonic() >= phase1_end:
                break
            a_start = inits[idx]
            end_i = min(phase1_end, max(time.monotonic() + 8.0, time.monotonic() + slice_time))
            if end_i <= time.monotonic() + 2.0:
                continue
            _stage(a_start, end_i, rng, n, state, P_A)

    final_reserve = min(90.0, max(20.0, 0.06 * budget))
    phase2_end = max(time.monotonic(), deadline - final_reserve)

    if phase2_end > time.monotonic() + 30.0:
        _stage(state["best_a"], phase2_end, rng, n, state, P_B)

    if time.monotonic() < deadline - 3.0:
        a_pol, s_pol = _polish(state["best_a"], rng, n, max_trials=200, end=deadline - 5.0)
        if np.isfinite(s_pol) and s_pol < state["best_score"] - 1e-12:
            state["best_score"] = s_pol
            state["best_a"] = a_pol
            _add_hall(state, a_pol, s_pol)

    if _sp_minimize is not None and time.monotonic() < deadline - 12.0:
        for beta in (1200.0, 5000.0, 18000.0):
            if time.monotonic() > deadline - 8.0:
                break
            try:
                cand = _lbfgs_refine(state["best_a"], beta, n)
                sc = _score_fast(cand)
                if np.isfinite(sc) and sc < state["best_score"] - 1e-12:
                    state["best_score"] = sc
                    state["best_a"] = cand
                    _add_hall(state, cand, sc)
            except Exception:
                pass

    candidates = []
    candidates.append(state["best_a"])
    for _, a_h in state.get("hall", [])[:4]:
        candidates.append(a_h)
    candidates.append(parent)
    candidates.append(np.array([0.5, 0.1, 0.4], dtype=np.float64))

    best_list = None
    best_score = float("inf")
    for cand in candidates:
        lst = _clean_list(cand)
        if lst is None:
            continue
        try:
            sc = float(evaluate_sequence(lst))
        except Exception:
            sc = float("inf")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_list = lst

    if best_list is None:
        best_list = [0.5, 0.1, 0.4]

    return best_list
