```python
import math
from scipy.fft import rfft, irfft, next_fast_len

def _eval(a):
    """Fast evaluation using FFT."""
    n = len(a)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return np.inf
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    b = irfft(F*F, fsize)[:2*n-1]
    b = np.maximum(b, 0)
    return 2.0 * n * b.max() / (s*s)

    n = len(a)
    fsize = next_fast_len(2*n-1)

    b = irfft(F*F, fsize)[:2*n-1]
    b = np.maximum(b, 0)
    max_b = b.max()
    
    # Find lags achieving (near) max
    thresh = max_b * (1 - 1e-9)
    peaks = np.where(b >= thresh)[0]
    
    # Subgradient of max_b
    g = np.zeros(n)

    # where g_max is the subgradient of max_b
    grad = 2.0 * n * (g - max_b)
    return grad, max_b, b
```

```python
def _grad_score_smoothed(a, K=5):
    n = len(a)
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    b = irfft(F*F, fsize)[:2*n-1]
    b = np.maximum(b, 0)
    max_b = b.max()
    
    # Use top K peaks with exponential weighting
    sorted_idx = np.argsort(b)[::-1][:K]
    b_sorted = b[sorted_idx]
    weights = np.exp(-(b_sorted - max_b) * 10)  # decay away from max
    weights /= weights.sum()
    
    g = np.zeros(n)
    for k, w in zip(sorted_idx, weights):
        # ∂b_k/∂a_j = 2*a_{k-j} for 0 <= k-j < n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            # j ranges from lo to hi, and k-j ranges from k-hi to k-lo (both in [0, n-1])
            # g[j] += 2*w*a[k-j]
            js = np.arange(lo, hi+1)
            kjs = k - js
            g[lo:hi+1] += 2.0 * w * a[kjs]
    
    # Gradient of score: 2n * (g - max_b)
    grad = 2.0 * n * (g - max_b)
    return grad, max_b, b
```

This is O(K*n) per gradient evaluation, which is very fast for small K.

```python
    n = len(a)
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    b = irfft(F*F, fsize)[:2*n-1]
    b = np.maximum(b, 0.0)
    mb = float(b.max())
    if not np.isfinite(mb):
        return np.inf
    return 2.0 * n * mb / (s*s)

def _fft_convolve(a):
    n = len(a)
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    b = irfft(F*F, fsize)[:2*n-1]
    return np.maximum(b, 0.0)

def _subgradient(a, K=3):
    """Compute subgradient of score w.r.t. a, assuming a >= 0.
    Returns (gradient, score, b)."""
    n = len(a)
    b = _fft_convolve(a)
    s = float(a.sum())
    max_b = float(b.max())
    score = 2.0 * n * max_b / (s*s)
    
    # Find top K peaks
    if n <= 2*(n-1):  # always true
        pass
    sorted_idx = np.argsort(b)[::-1]
    K = min(K, len(b))
    top_idx = sorted_idx[:K]
    top_vals = b[top_idx]
    
    # Weights: give more weight to the actual max
    spread = max_b - top_vals[-1]
    if spread > 1e-15:
        weights = np.exp(-(top_vals[-1] - top_vals) / (spread * 0.3 + 1e-15))
    else:
        weights = np.ones(K)
    weights = weights / weights.sum()
    
    # Compute subgradient of max_b
    g_maxb = np.zeros(n)
    for k, w in zip(top_idx, weights):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            js = np.arange(lo, hi+1)
            kjs = k - js  # indices of a
            g_maxb[lo:hi+1] += 2.0 * w * a[kjs]
    
    # Gradient of score = 2n * (g_maxb - max_b) / s^2  * s^2 ... 
    # score = 2n * max_b / s^2
    # d(score)/da_j = 2n/s^2 * d(max_b)/da_j - 2n*max_b/s^3 * 1
    # = (2n/s^2) * (d(max_b)/da_j - max_b/s)
    grad = (2.0 * n / (s*s)) * (g_maxb - max_b / s)
    return grad, score, b

def _optimize(a_init, budget_s, stop_time, seed=42):
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = np.clip(a_init.astype(np.float64).copy(), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s  # normalize to sum=1
    
    best_a = a.copy()
    best_score = _eval_fast(a)
    
    lr = 0.1
    patience = 0
    max_patience = 200
    
    while time.monotonic() < stop_time:
        t0 = time.monotonic()
        
        grad, score, b = _subgradient(a, K=3)
        
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            # Restart with perturbation
            a = best_a.copy()
            noise = rng.normal(0, 0.05 * n / a.sum(), n)
            a = np.maximum(0, a + noise)
            s = a.sum()
            if s > 0:
                a = a / s
            lr = 0.1
            patience = 0
            continue
        
        # Gradient step
        a_new = a - lr * grad
        a_new = np.maximum(0, a_new)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = np.ones(n) / n
            s_new = 1.0
        a_new = a_new / s_new
        
        new_score = _eval_fast(a_new)
        
        if np.isfinite(new_score) and new_score < score:
            a = a_new
            # Decay patience
            patience = max(0, patience - 1)
        else:
            # Reduce learning rate
            lr *= 0.9
            if lr < 1e-6:
                # Restart
                a = best_a.copy()
                noise = rng.normal(0, 0.03, n)
                a = np.maximum(0, a + noise)
                s = a.sum()
                if s > 0:
                    a = a / s
                lr = 0.1
                patience = 0
    
    return best_a, best_score
```
