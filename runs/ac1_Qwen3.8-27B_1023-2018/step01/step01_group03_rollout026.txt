import numpy as np
import time
from scipy.optimize import linprog

def _conv(a):
    return np.convolve(a, a)

def _score(a):
    n = len(a)
    b = _conv(a)
    s = a.sum()
    if s < 1e-10: return float('inf')
    return 2.0 * n * b.max() / (s * s)

def _grad_maxconv(a, beta=20.0):
    """Smooth gradient of max(a*a) using softmax over active lags."""
    n = len(a)
    b = _conv(a)
    m = b.max()
    w = np.exp((b - m) * beta)
    w /= w.sum()
    
    # grad[j] = sum_l w_l * (a[l-j] + a[j-l]) for valid indices
    grad = np.zeros(n)
    for l in range(len(b)):
        if w[l] < 1e-10:
            continue
        # Contribution to grad from lag l:
        # If 0 <= l-j <= n-1: add a[l-j]
        # If 0 <= j-l <= n-1: add a[j-l]
        lo = max(0, l - (n-1))
        hi = min(n-1, l)
        if lo <= hi:
            # j in [lo, hi], l-j in [l-hi, l-lo] = [max(0,l-n+1), min(n-1,l)]
            idx = np.arange(lo, hi+1)
            grad[lo:hi+1] += w[l] * a[l - idx]  # a[l-j] for j in [lo,hi]
        
        lo2 = max(0, l)  # j-l >= 0 means j >= l
        hi2 = min(n-1, l + (n-1))  # j-l <= n-1 means j <= l+n-1, always true since j <= n-1
        lo2 = max(l, 0)
        hi2 = min(n-1, l + n - 1)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2+1)
            grad[lo2:hi2+1] += w[l] * a[idx - l]  # a[j-l] for j in [lo2,hi2]
    
    return grad

def _lp_step(a, M_scale=1.0):
    n = len(a)
    b = _conv(a)
    M = M_scale * b.max()
    if M <= 0: return a.copy()
    
    # Build constraint matrix: A[l, j] = a[j] if l-j is valid index in [0,n-1], else 0
    # Constraint: sum_j A[l,j] * g[j] <= M for each l
    # A is the "convolution matrix" of a
    
    # A[l, j] = a[j] if 0 <= l-j <= n-1, i.e., l-(n-1) <= j <= l
    A = np.zeros((2*n-1, n))
    for j in range(n):
        lo = j  # l >= j (l-j >= 0)
        hi = j + n - 1  # l <= j+n-1 (l-j <= n-1)
        A[lo:hi+1, j] = a[j]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, 1000), method='highs')
    if not res.success:
        return a.copy()
    g = res.x
    s = g.sum()
    if s <= 0: return a.copy()
    return g / s

So: ∇L = 2 * convolve(a[::-1], w)[n-1 : 2n-1]

Let me verify the length: convolve(a[::-1], w) has length n + (2n-1) - 1 = 3n-1.
We take indices [n-1 : 2n-1], which is n elements. ✓

This is beautiful! The gradient is just a single convolution operation!

So the gradient computation is:
1. Compute b = convolve(a, a) (length 2n-1)
2. Compute w = softmax(β * (b - max(b))) (length 2n-1)
3. Compute grad = 2 * convolve(a[::-1], w)[n-1 : 2n-1] (length n)

Step 3 is a convolution of length n with length 2n-1, giving length 3n-1, and we take n elements. This is O(n * 2n) = O(2n²) for direct convolution, or O(n log n) with FFT.

For n=512: 2*512² = 524K operations. Very fast (~0.1ms).

Now, for the optimization:
- Objective: minimize L(a) = (1/β) log(sum_ℓ exp(β*(a*a)_ℓ)) ≈ max_ℓ (a*a)_ℓ
- Constraint: sum a = 1, a ≥ 0
- Gradient: as derived above

Algorithm:
```
a ← initial (sum=1, a≥0)
for iter:
    b = convolve(a, a)
    w = softmax(β*b - β*max(b))
    grad = 2 * convolve(a[::-1], w)[n-1:2n-1]
    # Project gradient onto tangent space of simplex
    grad -= grad.mean()  # remove mean (tangent to sum=const)
    # Or: use grad directly and project after step
    # Step
    a ← a - step * grad
    a ← max(a, 0)
    a ← a / sum(a)
    # Adaptive step: line search or backtracking
```

This is very efficient! Each iteration is O(n²) for the convolution. For n=512, that's ~0.2ms per iteration. In 100s, we can do 500K iterations. More than enough.

The key question is: does this converge to a good local minimum? The softmax approximation gets closer to the true max as β increases. We can use adaptive β (start small, increase over time).

Let me also add the LP step as a "jump" method between gradient phases:
- Every K gradient steps, do one LP step to potentially jump to a better region
- The LP finds a "far" direction, while gradient does "local" improvement

Here's my final algorithm:

```
Phase 1: LP iterations (100-200 steps)
Phase 2: Gradient polish (10000+ steps, adaptive β and step)
Phase 3: Multi-restart (10× perturb + Phase 2)
Phase 4: Try different n
Phase 5: Final gradient polish on best
```

        s = a.sum()

            return float('inf')

        return 2.0 * n * b.max() / (s * s)

        a = np.asarray(a, dtype=np.float64)

            return

        s = a.sum()

            return

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return np.full(a.shape, 1.0 / a.size)
        return a / s

    def _gradient(a, beta):
        """Compute smoothed gradient of max auto-correlation."""
        n = a.size
        b = np.convolve(a, a)
        bm = b.max()
        w = np.exp(beta * (b - bm))
        wsum = w.sum()
        if wsum < 1e-30:
            return np.zeros(n)
        w /= wsum
        # grad[k] = 2 * sum_m a[m] * w[k+m], for k=0..n-1
        # = 2 * convolve(a[::-1], w)[n-1 : 2n-1]
        conv = np.convolve(a[::-1], w)
        grad = 2.0 * conv[n-1 : 2*n-1]
        return grad
    
    def _grad_descent(a, n, max_iters, deadline, beta_start=5.0, beta_end=50.0,
                      step0=0.05, verbose=False):
        """Smooth gradient descent on max auto-correlation (sum normalized)."""
        a = _norm(a.copy())
        best_local = a.copy()
        best_f = 2.0 * n * np.convolve(a, a).max()
        step = step0
        stall = 0
        
        for it in range(max_iters):
            if time.monotonic() > deadline - 5.0:
                break
            
            # Adaptive beta
            frac = min(1.0, it / max(1, max_iters * 0.6))
            beta = beta_start + (beta_end - beta_start) * frac
            
            b = np.convolve(a, a)
            bm = b.max()
            
            # Gradient
            w = np.exp(beta * (b - bm))
            wsum = w.sum()
            if wsum < 1e-30:
                break
            w /= wsum
            conv = np.convolve(a[::-1], w)
            grad = 2.0 * conv[n-1 : 2*n-1]
            
            # Remove mean for simplex tangency
            grad -= grad.mean()
            
            # Normalize gradient
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            grad /= gn
            
            # Try step
            cand = a - step * grad
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-15:
                step *= 0.5
                continue
            cand /= cs
            
            cf = 2.0 * n * np.convolve(cand, cand).max()
            
            if cf < best_f * (1.0 - 1e-10):
                best_f = cf
                best_local = cand.copy()
                stall = 0
                if step < 0.15:
                    step = min(0.15, step * 1.03)
            else:
                step *= 0.75
                if step < 0.002:
                    break

                    cand2 = a + pert

                            best_local = cand2.copy()
                            stall = 0

                break
        
        return best_local, best_f
    
    def _lp_direction(a, n):
        """LP: find g maximizing sum(g) s.t. (a*g)_l <= M, g >= 0."""
        b = np.convolve(a, a)
        M = b.max()
        if M < 1e-15:
            return a.copy()

        A = np.zeros((2*n-1, n), dtype=np.float64)
        for j in range(n):
            A[j:j+n, j] = a[j]
        
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        
        try:

            g = res.x
            s = g.sum()
            if s < 1e-15:
                return a.copy()
            return g / s
        except Exception:
            return a.copy()
    
    def _lp_phase(a, n, n_iters, deadline):

        a = _norm(a.copy())
        best_local = a.copy()
        best_f = 2.0 * n * np.convolve(a, a).max()
        
        for it in range(n_iters):

                break
            
            g = _lp_direction(a, n)
            if g is None or np.allclose(g, a, atol=1e-8):
                break
            
            best_cand = a
            best_cs = best_f
            for t in (0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
                cand = (1 - t) * a + t * g
                cand = _norm(cand)
                cf = 2.0 * n * np.convolve(cand, cand).max()
                if cf < best_cs:
                    best_cs = cf
                    best_cand = cand
            
            if best_cs < best_f * (1 - 1e-8):
                a = best_cand
                best_f = best_cs
                best_local = a.copy()
            else:
                break
        
        return best_local, best_f
    
    def _multi_restart(a, n, n_restarts, deadline, per_restart_iters):

        best_local = a.copy()
        best_f = 2.0 * n * np.convolve(a, a).max()

            a_p = a + pert
            a_p = _norm(a_p)
            
            # Short gradient polish
            a_r, f_r = _grad_descent(a_p, n, per_restart_iters, deadline,
                                     beta_start=5.0, beta_end=40.0, step0=0.04)

                best_local = a_r.copy()

                    best_f = f_r2
                    best_local = a_r2.copy()
                    a = best_local.copy()
        
        return best_local, best_f

        x_new = np.linspace(0, 1, n_new)
        a_new = np.interp(x_new, x_old, a)
        return _norm(a_new)
    
    # === MAIN OPTIMIZATION ===
    
    # Phase 1: Optimize parent sequence
    if best_a is not None and best_a.size > 4:

    # Phase 2: Multi-restart on parent size
    if best_a is not None and best_a.size > 4:
