def _lp_direction(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0"""
    n = a.size

    # A[k, j] = a_{k-j} if 0 <= k-j < n, else 0
    # Number of constraints: 2n-1
    num_constraints = 2*n - 1

        h = globals()["height_sequence_1"]

def _convolve_fft(a):
    n = a.size
    fsize = next_fast_len(2*n-1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2*n-1]
    return np.maximum(c, 0.0)

def _score_exact(a):
    n = a.size
    b = _convolve_fft(a)
    m = float(b.max()) if b.size > 0 else 0.0
    s = float(a.sum())
    if not np.isfinite(m) or s < 0.01:
        return math.inf
    return 2.0 * n * m / (s * s)

def _lp_step(a, M, dt=0.1):
    """LP-based update step."""
    n = a.size
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[k, j] = a[k - j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs',
                     options={'time_limit': 5, 'presolve': True})
    
    if not result.success:
        return None
    g = np.maximum(result.x, 0.0)
    if g.size == 0 or not np.isfinite(g).all():
        return None
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    
    # Rescale g to match a's sum
    a_sum = float(a.sum())
    g = g * (a_sum / gs)
    
    # Blend
    a_new = (1.0 - dt) * a + dt * g
    a_new = np.maximum(a_new, 0.0)

        M = float(b.max())
        
        a_new = _lp_step(a, M, t)
        if a_new is None:
            stagnant += 1
            if stagnant > 10:

            # Skip this iteration
            t = max(0.01, t * 0.9)
            continue
        
        a_new_s = float(a_new.sum())
        if a_new_s < 1e-15:
            stagnant += 1
            break
        a_new = a_new / a_new_s  # renormalize
        
        new_score = _score_exact(a_new)
        if np.isfinite(new_score) and new_score < best_score - 1e-12:

            best_score = new_score

import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(192, dtype=np.float64)
    try:
        p = np.clip(p, 0.0, 1000.0)
    except Exception:
        p = np.ones(192, dtype=np.float64)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(192, dtype=np.float64)
    return p


def _convolve_fft(a):
    n = a.size
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _fast_score(a):
    n = a.size
    b = _convolve_fft(a)
    m = float(b.max()) if b.size > 0 else 0.0
    s = float(a.sum())
    if not np.isfinite(m) or s < 0.01:
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(a):
    n = a.size
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n), dtype=np.float64)
    for k in range(num_rows):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return A


def _lp_direction(a, M, time_limit=4.0):
    n = a.size
    if not np.isfinite(M) or M <= 0:
        return None
    A = _build_A(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    if not result.success or result.x is None:
        return None
    g = np.maximum(result.x, 0.0)
    if not np.isfinite(g).all() or g.sum() < 1e-15:
        return None
    # Verify constraints
    residual = A @ g - M
    if residual.max() > 1e-5 * np.max(np.abs(M)):
        # Relax: scale down
        scale = M / max(residual.max(), 1e-30)
        g = g * min(1.0, scale * 0.99)
        if g.sum() < 1e-15:
            return None
    return g


def _optimize_lp(a_init, stop_time, max_iters=400, seed=0):
    n = a_init.size
    if n < 2:
        return a_init.copy(), _fast_score(a_init)
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init, 0.0).copy()
    s = float(a.sum())
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s

    best_a = a.copy()
    best_score = _fast_score(a)

    t = 0.06
    stagnant = 0
    lp_fail = 0

    for it in range(max_iters):
        if time.monotonic() > stop_time:
            break

        b = _convolve_fft(a)
        M = float(b.max())
        if not np.isfinite(M) or M <= 0:
            break

        a_new = _lp_direction(a, M, time_limit=3.0)
        if a_new is None:
            lp_fail += 1
            if lp_fail > 8:
                break
            continue
        lp_fail = 0

        g_s = float(a_new.sum())
        a_new = a_new * (s / g_s)

        # Try the blend
        improved = False
        for try_t in (t, t * 0.5, t * 1.5, t * 0.25):
            a_try = (1.0 - try_t) * a + try_t * a_new
            a_try = np.maximum(a_try, 0.0)
            ts = float(a_try.sum())
            if ts < 1e-15:
                continue
            a_try = a_try / ts
            new_score = _fast_score(a_try)
            if np.isfinite(new_score) and new_score < best_score - 1e-12:
                best_a = a_try.copy()
                best_score = new_score
                a = a_try
                t = try_t
                improved = True
                stagnant = 0
                break

        if not improved:
            stagnant += 1
            t = max(0.015, t * 0.88)
            if stagnant > 50:
                t = float(rng.uniform(0.02, 0.12))
                stagnant = 20

    return best_a, best_score


def _grad_optimize(a_init, stop_time, seed=123):
    n = a_init.size
    if n < 2:
        return a_init.copy(), _fast_score(a_init)
    rng = np.random.default_rng(seed)
    fsize = next_fast_len(2 * n - 1)

    a = np.maximum(a_init, 1e-12).copy()
    s = float(a.sum())
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    z = np.log(a)
    z -= z.max()
    best_a = a.copy()
    best_score = _fast_score(a)

    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t_count = 0
    lr = 0.22
    anneal = 1.0
    stagnant = 0

    while time.monotonic() < stop_time:
        a = np.exp(z)
        a_s = float(a.sum())
        if a_s < 1e-300:
            break
        a = a / a_s

        b = _convolve_fft(a)
        bmax = float(b.max())
        score = 2.0 * n * bmax

        if np.isfinite(score) and score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            stagnant = 0

        # Soft logsumexp for subgradient
        tau = max(1e-12, 0.035 * max(bmax, 1e-15) * anneal)
        x = (b - bmax) / tau
        x = np.clip(x, -60, 0)
        ex = np.exp(x)
        ex_sum = float(ex.sum())
        if ex_sum < 1e-300:
            p = np.zeros_like(b)
            p[int(np.argmax(b))] = 1.0
        else:
            p = ex / ex_sum

        # Active set for gradient
        thresh = max(p.max() * 1e-6, 1e-20)
        active = np.where(p > thresh)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(p))], dtype=int)
        if active.size > 300:
            top_idx = np.argpartition(p, -300)[-300:]
            active = np.sort(top_idx)

        w = p[active].astype(np.float64)
        w_s = float(w.sum())
        if w_s < 1e-300:
            stagnant += 1
            if stagnant > 40:
                break
            z += rng.normal(0, 0.35, n)
            z -= z.mean()
            continue
        w = w / w_s

        grad = np.zeros(n, dtype=np.float64)
        for idx in range(active.size):
            ki = int(active[idx])
            wi = float(w[idx])
            if wi <= 0:
                continue
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            if hi < lo:
                continue
            # grad[lo:hi+1] += 2*n*2*wi * a[ki-hi : ki-lo+1][::-1]
            seg = a[ki - hi:ki - lo + 1][::-1]
            factor = 4.0 * n * wi
            grad[lo:hi + 1] += factor * seg

        # Project to tangent space of simplex in log-param
        S = float(np.dot(a, grad))
        h = a * (grad - S)

        if not np.isfinite(h).all():
            h = np.where(np.isfinite(h), h, 0.0)

        hmax = float(np.max(np.abs(h))) if h.size > 0 else 0.0
        if hmax < 1e-14:
            stagnant += 1
            if stagnant > 40:
                break
            z += rng.normal(0, 0.4, n)
            z -= z.mean()
            m_v = np.zeros(n)
            v_v = np.zeros(n)
            t_count = 0
            continue

        t_count += 1
        m_v = 0.9 * m_v + 0.1 * h
        v_v = 0.999 * v_v + 0.001 * h * h
        beta1_t = 1.0 - 0.9 ** t_count
        beta2_t = 1.0 - 0.999 ** t_count
        mh = m_v / beta1_t
        vh = v_v / beta2_t
        direction = mh / (np.sqrt(vh) + 1e-8)
        direction[~np.isfinite(direction)] = 0.0

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            z_try = z - lr * factor * direction
            z_try -= z_try.mean()
            z_try = np.clip(z_try, -40, 40)
            a_try = np.exp(z_try)
            a_ts = float(a_try.sum())
            if a_ts < 1e-300:
                continue
            a_try = a_try / a_ts
            s_try = 2.0 * n * float(_convolve_fft(a_try).max())
            if np.isfinite(s_try) and s_try < score * (1.0 - 5e-10):
                z = z_try
                stagnant = 0
                improved = True
                break

        if not improved:
            stagnant += 1
            lr = max(0.005, lr * 0.93)
            if stagnant > 70:
                z += rng.normal(0, 0.5, n)
                z -= z.mean()
                m_v = np.zeros(n)
                v_v = np.zeros(n)
                t_count = 0
                lr = 0.22
                anneal = 1.0
                stagnant = 0

        if t_count % 300 == 0:
            anneal = max(0.05, anneal * 0.90)

    return best_a, best_score


def _polish(seq, score, stop_time, seed=777):
    a = np.asarray(seq, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0.0, 1000.0)
    best = a.copy()
    best_score = float(score)
    if not np.isfinite(best_score):
        best_score = _fast_score(best)
    rng = np.random.default_rng(seed)
    n = best.size
    sigma = 0.12
    stuck = 0

    while time.monotonic() < stop_time:
        current = best.copy()
        r = rng.random()

        if r < 0.30:
            z = np.log(np.maximum(current, 1e-12))
            z += rng.normal(0, sigma, n)
            cand = np.exp(z)
        elif r < 0.50:
            cand = current.copy()
            seg_len = max(1, min(n // 15, int(rng.integers(1, max(2, n // 6)))))
            start = int(rng.integers(0, max(1, n - seg_len + 1)))
            cand[start:start + seg_len] *= float(np.exp(rng.normal(0, 0.35)))
        elif r < 0.65:
            cand = 0.55 * current + 0.45 * current[::-1]
        elif r < 0.80:
            cand = current.copy()
            idx = int(rng.integers(0, n))
            half = max(1, min(n // 80, int(rng.integers(1, max(2, n // 30)))))
            lo_b = max(0, idx - half)
            hi_b = min(n, idx + half + 1)
            cand[lo_b:hi_b] *= float(rng.uniform(0.5, 2.0))
        elif r < 0.90:
            cand = current.copy()
            # Swap two segments
            s1 = int(rng.integers(0, n // 2))
            s2 = int(rng.integers(n // 2, n))
            length = max(1, min(n // 20, int(rng.integers(1, max(2, n // 10)))))
            e1 = min(n, s1 + length)
            e2 = min(n, s2 + length)
            tmp = cand[s1:e1].copy()
            cand[s1:e1] = cand[s2:e2]
            cand[s2:e2] = tmp
        else:
            cand = 0.8 * current + 0.2 * np.ones(n) / n

        cand = np.maximum(cand, 0.0)
        cs = float(cand.sum())
        if cs < 0.01:
            continue

        fs = _fast_score(cand)
        if np.isfinite(fs) and fs < best_score - 1e-9:
            es = float(evaluate_sequence([float(x) for x in cand]))
            if np.isfinite(es) and es < best_score - 1e-12:
                best = cand.copy()
                best_score = es
                stuck = 0
                sigma = max(0.04, sigma * 0.96)
            else:
                stuck += 1
        else:
            stuck += 1
            if stuck % 60 == 0:
                sigma = min(0.8, sigma * 1.25)
            if stuck > 300:
                stuck = 0
                sigma = 0.25

    return best, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):

            pass

    # Phase 3: LP again from gradient result
    if best_seq.size >= 4 and time.monotonic() < deadline - 100:
        stop3 = min(deadline - 100, t0 + 750)
        try:
            a_out, sc = _optimize_lp(best_seq, stop3, max_iters=300, seed=si + 222)
            if np.isfinite(sc) and sc < best_score - 1e-10:
                es = float(evaluate_sequence([float(x) for x in a_out]))

                    best_score = es
        except Exception:
            pass

    # Phase 4: Gradient again

        stop4 = min(deadline - 80, t0 + 900)
        try:
            a_out, sc = _grad_optimize(best_seq, stop4, seed=si + 333)
            if np.isfinite(sc) and sc < best_score - 1e-10:
                es = float(evaluate_sequence([float(x) for x in a_out]))
                if np.isfinite(es) and es < best_score - 1e-10:
                    best_seq = a_out.copy()
                    best_score = es
        except Exception:
            pass

    # Phase 5: Try other lengths
    other_lengths = [64, 96, 128, 160, 256, 320, 384, 512]
    for L in other_lengths:
        if L == n_p:
            continue
        if time.monotonic() > deadline - 50:
            break
        alloc = min(100, max(15, deadline - time.monotonic() - 35))
        if alloc < 20:
            break

        # Resample parent
        if parent.size >= 2:
            idx = np.minimum((np.arange(L) * parent.size // L).astype(int), parent.size - 1)
            a_init = np.maximum(parent[idx], 1e-6).copy()
        else:
            a_init = np.ones(L, dtype=np.float64)
        a_init = a_init / a_init.sum()

        # LP first
        stop_lp = time.monotonic() + alloc * 0.4
        try:
            a_l, sc_l = _optimize_lp(a_init, stop_lp, max_iters=100, seed=si + 5000 * L)
        except Exception:
            a_l, sc_l = a_init, math.inf

        # Gradient second
        if time.monotonic() < deadline - 35:
            stop_g = min(deadline - 35, time.monotonic() + alloc * 0.4)
            try:
                a_g, sc_g = _grad_optimize(a_l, stop_g, seed=si + 5000 * L + 7)
                if np.isfinite(sc_g) and sc_g < sc_l:
                    a_l, sc_l = a_g, sc_g
            except Exception:
                pass

        if np.isfinite(sc_l) and sc_l < best_score - 1e-8:
            es = float(evaluate_sequence([float(x) for x in a_l]))

            pass

    # Final validation

    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_score = math.inf

    if not np.isfinite(final_score):
        # Last resort fallback
        fallback = np.ones(32, dtype=np.float64) / 32.0
        try:

        except Exception:
            fs = math.inf
        if not np.isfinite(fs):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
