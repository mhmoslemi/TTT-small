    return _norm(g)


def _z_to_p(z, free, anchor, n):
    z = np.asarray(z, dtype=float)
    z = np.where(np.isfinite(z), z, 0.0)
    maxz = 0.0
    if z.size:
        mx = float(np.max(z))
        if mx > 0.0:
            maxz = mx
    uf = np.exp(z - maxz)
    ua = np.exp(-maxz)
    U = float(uf.sum()) + ua
    if not np.isfinite(U) or U <= 0.0:
        p = np.full(n, 1.0 / n, dtype=float)
        return p
    p = np.empty(n, dtype=float)
    p[free] = uf / U
    p[anchor] = ua / U
    return _norm(p)


def _soft_minimize(a0, betas, deadline, maxfun_per_beta=200, check_every=60):
    if minimize is None:
        return _norm(a0)

    a0 = _norm(a0)
    n = a0.size
    if n <= 1:
        return a0

    best_a = a0.copy()
    best_s = _score(a0)
    N = _npow2(2 * n - 1)

    p = a0.copy()
    anchor = int(np.argmax(p))
    if p[anchor] <= 1e-300:
        p = _norm(p + 1e-12)
        anchor = int(np.argmax(p))
    free_list = [i for i in range(n) if i != anchor]
    free = np.asarray(free_list, dtype=int)
    z = np.log(np.clip(p[free], 1e-30, None) / max(float(p[anchor]), 1e-30))
    z = np.clip(z, -40.0, 40.0)
    bounds = [(-40.0, 40.0)] * free.shape[0]

    for beta in betas:
        if time.monotonic() > float(deadline) - 10.0:
            break

        mfbeta = int(max(16, maxfun_per_beta))
        calls = [0]
        last_check = [0]

        def fun(zv):
            nonlocal best_a, best_s
            calls[0] += 1
            if time.monotonic() > float(deadline) - 10.0 or calls[0] > mfbeta + 25:
                raise _SoftStop

            maxz = 0.0
            if zv.size:
                mx = float(np.max(zv))
                if mx > 0.0:
                    maxz = mx
            uf = np.exp(zv - maxz)
            ua = np.exp(-maxz)
            U = float(uf.sum()) + ua
            if not np.isfinite(U) or U <= 0.0:
                return 1.0e100, np.zeros_like(zv)

            p = np.empty(n, dtype=float)
            p[free] = uf / U
            p[anchor] = ua / U

            P = np.fft.rfft(p, N)
            c = np.fft.irfft(P * P, N)[:2 * n - 1]
            m = float(c.max())
            if not np.isfinite(m) or m <= 0.0:
                return 0.0, np.zeros_like(zv)

            X = beta * (c - m)
            E = np.exp(X)
            s = float(E.sum())
            if not np.isfinite(s) or s <= 0.0:
                q = np.zeros_like(c)
                q[int(np.argmax(c))] = 1.0
                logadd = 0.0
            else:
                q = E / s
                logadd = float(np.log(s)) / float(beta)

            Q = np.fft.rfft(q, N)
            corr = np.fft.irfft(Q * np.conj(P), N)[:n]
            grad_p = 2.0 * corr
            gp = float(np.dot(p, grad_p))
            grad_z = p[free] * (grad_p[free] - gp)
            if not np.all(np.isfinite(grad_z)):
                grad_z = np.zeros_like(zv)

            f = m + logadd
            if not np.isfinite(f):
                f = 1.0e100

            if calls[0] - last_check[0] >= max(1, int(check_every)):
                last_check[0] = calls[0]
                sc = _score(p)
                if np.isfinite(sc) and sc < best_s:
                    best_s = sc
                    best_a = p.copy()

            return f, grad_z

        try:
            res = minimize(
                fun,
                np.asarray(z, dtype=float),
                jac=True,
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": max(12, mfbeta // 2),
                    "maxfun": mfbeta,
                    "ftol": 1.0e-16,
                    "gtol": 1.0e-8,
                    "maxls": 12,
                    "disp": False,
                },
            )
            if hasattr(res, "x") and res.x is not None:
                z = np.asarray(res.x, dtype=float)
                z = np.where(np.isfinite(z), z, 0.0)
                z = np.clip(z, -40.0, 40.0)
        except _SoftStop:
            pass
        except Exception:
            pass

        p = _z_to_p(z, free, anchor, n)
        sc = _score(p)
        if np.isfinite(sc) and sc < best_s:
            best_s = sc
            best_a = p.copy()

    return best_a


def _refine_exact(a0, deadline, rng, max_lp=35, max_iter=20000):
    a0 = _norm(a0)
    n = a0.size
    if n <= 1:
        return a0
    if rng is None:
        rng = np.random.default_rng(12345)

    best_a = a0.copy()
    best_s = _score(a0)
    cur_a = a0.copy()
    cur_s = best_s

    N = _npow2(2 * n - 1)
    eta = max(2.0, 0.10 * n)
    lp_every = 16 if n <= 512 else (30 if n <= 1024 else 60)
    fracs = (0.01, 0.03, 0.005, 0.07, 0.15, 0.02)
    betas = (200.0, 1000.0, 3000.0, 8000.0)

    global_stag = 0
    noise_count = 0
    it = 0

    while it < max_iter and time.monotonic() < float(deadline) - 12.0:
        c, P, _ = _conv_p(cur_a, N)

        if it % 6 == 0:
            w = _weights_active(c, fracs[it % 6])
        elif it % 6 == 1:
            w = _weights_soft(c, betas[it % 4])
        else:
            w = _weights_active(c, fracs[(it + 3) % 6])

        corr = _corr_q(w, P, N)[:n]
        grad = 2.0 * corr

        accepted = False
        if float(grad.max() - grad.min()) > 1.0e-15:
            for mult in (1.0, 0.5, 0.25, 0.125, 0.0625):
                cand = _natural_update(cur_a, grad, eta * mult)
                sc = _score(cand)
                if sc < best_s - 1.0e-12:
                    cur_a = cand
                    cur_s = sc
                    best_a = cand.copy()
                    best_s = sc
                    global_stag = 0
                    accepted = True
                    break
                elif sc < cur_s - 1.0e-12:
                    cur_a = cand
                    cur_s = sc
                    accepted = True
                    break

        if accepted:
            eta = min(10.0 * n, eta * 1.05)
        else:
            eta = max(1.0, eta * 0.7)
            global_stag += 1

        if (
            n <= 1536
            and (it % lp_every == 0 or global_stag > 50)
            and time.monotonic() < float(deadline) - 45.0
        ):
            dirn = _lp_maxsum(cur_a, deadline)
            if dirn is not None:
                for t in (0.05, 0.10, 0.20, 0.35, 0.50, 0.75, 1.00):
                    cand = _norm((1.0 - t) * cur_a + t * dirn)
                    sc = _score(cand)
                    if sc < best_s - 1.0e-12:
                        cur_a = cand
                        cur_s = sc
                        best_a = cand.copy()
                        best_s = sc
                        global_stag = 0
                        accepted = True
                        break
                    elif sc < cur_s - 1.0e-12:
                        cur_a = cand
                        cur_s = sc
                        accepted = True
                        break

        if global_stag > 80:
            if noise_count >= 10:
                break
            mix = 0.78 if n <= 512 else 0.85
            pdir = rng.dirichlet(np.full(n, 5.0, dtype=float))
            cand = _norm(mix * cur_a + (1.0 - mix) * pdir)
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_s * 1.005:
                cur_a = cand
                cur_s = sc
            global_stag = 0
            noise_count += 1

        it += 1

    return best_a


def _resize(a, n):
    a = _norm(a)
    n = max(1, int(n))
    if a.size == n:
        return a
    if a.size == 0:
        return np.full(n, 1.0 / n, dtype=float)
    x = np.linspace(0.0, 1.0, a.size)
    y = np.linspace(0.0, 1.0, n)
    return _norm(np.interp(y, x, a))


def _upscale(a, n):
    a = _norm(a)
    n = max(1, int(n))
    if a.size == n:
        return a
    if a.size == 0:
        return np.full(n, 1.0 / n, dtype=float)
    if n % a.size == 0:
        out = np.tile(a, n // a.size)[:n]
    else:
        out = _resize(a, n)
    return _norm(out)


def _default_start(n, rng):
    t = np.linspace(-1.0, 1.0, n)
    a = np.exp(2.0 * t * t)
    b = np.exp(-1.5 * t * t)
    c = rng.dirichlet(np.full(n, 3.0, dtype=float))
    return _norm(a + 0.35 * b + 0.25 * c)


def _make_starts(base, rng, max_starts=14):
    base = _norm(base)
    n = base.size
    starts = [base]
    if n > 2:
        t = np.linspace(-1.0, 1.0, n)
        for lam in (-2.0, -1.0, 1.0, 2.0):
            starts.append(_norm(base * np.exp(lam * t)))
        for alpha in (1.0, 2.5, 5.0):
            starts.append(_norm(np.exp(alpha * t * t)))
            starts.append(_norm(np.exp(-alpha * t * t) + 0.1))
        k = max(3, n // 32)
        if k < n:
            starts.append(_norm(np.convolve(base, np.ones(k) / float(k), mode="same")))
    starts.append(rng.dirichlet(np.full(n, 3.0, dtype=float)))
    starts.append(_norm(base * rng.dirichlet(np.full(n, 2.0, dtype=float))))
    starts = starts[: max(1, int(max_starts))]
    return starts


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    try:
        budget_s = max(60.0, float(budget_s))
    except Exception:
        budget_s = 1300.0

    t0 = time.monotonic()
    deadline = t0 + budget_s - 60.0
    if deadline <= t0 + 30.0:
        deadline = t0 + max(20.0, budget_s - 5.0)

    rng = np.random.default_rng(seed)

    best_list = None
    best_auth = float("inf")
    best_internal = float("inf")
    best_arr = None

    def consider(seq):
        nonlocal best_list, best_auth, best_internal, best_arr
        if seq is None:
            return
        try:
            arr = np.asarray(seq, dtype=float).ravel()
            if arr.size == 0:
                return
            arr = _norm(arr)
            sc_int = _score(arr)
            if not np.isfinite(sc_int):
                return

            lst = [float(x) for x in arr]
            try:
                auth = float(evaluate_sequence(lst))
            except Exception:
                auth = sc_int

            if not np.isfinite(auth):
                return

            if (
                best_list is None
                or auth < best_auth - 1.0e-12
                or (auth <= best_auth + 1.0e-12 and sc_int < best_internal - 1.0e-12)
            ):
                best_list = lst
                best_auth = auth
                best_internal = sc_int
                best_arr = arr.copy()
        except Exception:
            return

    # Start from supplied parent construction when available.
    parent = None
    try:
        h = height_sequence_1
        if h is not None:
            parent = _norm(h)
            if parent.size == 0:
                parent = None
    except Exception:
        parent = None

    if parent is None or parent.size == 0:
        parent = _default_start(512, rng)

    consider(parent)
    base = parent.copy()

    # Keep the workspace reasonable even for unexpected large parent lengths.
    if base.size > 2048:
        base = _resize(base, 1024)
        consider(base)

    consider(np.ones(base.size, dtype=float))

    starts = _make_starts(base, rng, max_starts=16 if base.size <= 1024 else 10)
    for st in starts:
        consider(st)

    scored = []
    for st in starts:
        sc = _score(st)
        if np.isfinite(sc):
            scored.append((sc, st))
    if not scored:
        scored = [(2.0, base)]
    scored.sort(key=lambda x: x[0])
    top = [s for _, s in scored[:3]]
    if not top:
        top = [base]

    # Phase A: short smooth optimization from a few promising starts.
    A_end = min(deadline - 600.0, t0 + 150.0)
    if A_end > time.monotonic() + 30.0:
        per = max(8.0, (A_end - time.monotonic()) / float(max(1, len(top))))
        for a0 in top:
            now = time.monotonic()
            if now > A_end - 12.0:
                break
            end = min(A_end, now + per)
            if end <= now + 5.0:
                break
            a = _soft_minimize(
                a0,
                [30.0, 120.0],
                end,
                maxfun_per_beta=80,
                check_every=80,
            )
            if a is not None:
                consider(a)

    # Phase B: stronger smooth optimization from the best known sequence.
    B_end = min(deadline - 420.0, t0 + 380.0)
    if best_arr is not None and B_end > time.monotonic() + 60.0:
        a = _soft_minimize(
            best_arr,
            [50.0, 200.0, 800.0],
            B_end,
            maxfun_per_beta=260,
            check_every=60,
        )
        consider(a)

    # Phase C: exact local refinement with subgradient / multiplicative steps and LP.
    C_end = min(deadline - 300.0, t0 + 680.0)
    if best_arr is not None and C_end > time.monotonic() + 90.0:
        a = _refine_exact(best_arr, C_end, rng, max_lp=40, max_iter=25000)
        consider(a)

    # Phase D: high-smoothness pass and final primary-length polish.
    D_end = min(deadline - 180.0, t0 + 900.0)
    if best_arr is not None and D_end > time.monotonic() + 60.0:
        d_soft = min(D_end - 60.0, time.monotonic() + 180.0)
        if d_soft > time.monotonic() + 35.0:
            a = _soft_minimize(
                best_arr,
                [500.0, 2000.0, 8000.0],
                d_soft,
                maxfun_per_beta=100,
                check_every=80,
            )
            consider(a)

        now = time.monotonic()
        if best_arr is not None and now < D_end - 50.0:
            a = _refine_exact(best_arr, D_end, rng, max_lp=18, max_iter=15000)
            consider(a)

    # Phase E: larger resolution, if the primary result is small and time remains.
    now = time.monotonic()
    if best_arr is not None and deadline - now > 260.0 and best_arr.size <= 2048:
        n_big = 1024 if best_arr.size < 1024 else min(2048, 2 * best_arr.size)
        if best_arr.size < n_big and n_big <= 2048:
            a0 = _upscale(best_arr, n_big)
            if n_big >= 1024:
                a0 = _norm(np.convolve(a0, np.ones(5) / 5.0, mode="same"))
            consider(a0)

            e1 = min(deadline - 120.0, now + 200.0)
            if e1 > time.monotonic() + 40.0:
                a = _soft_minimize(
                    a0,
                    [20.0, 80.0, 300.0],
                    e1,
                    maxfun_per_beta=160,
                    check_every=100,
                )
                consider(a)

            now = time.monotonic()
            if best_arr is not None and deadline - now > 170.0:
                e2 = min(deadline - 80.0, now + 330.0)
                if e2 > now + 70.0:
                    a = _refine_exact(
                        a0,
                        e2,
                        rng,
                        max_lp=22 if n_big <= 1024 else 8,
                        max_iter=12000,
                    )
                    consider(a)

    # Final short polish on the best valid candidate.
    now = time.monotonic()
    if best_arr is not None and deadline - now > 45.0:
        a = _refine_exact(best_arr, deadline - 8.0, rng, max_lp=4, max_iter=8000)
        consider(a)

    if best_list is None and best_arr is not None:
        consider(best_arr)

    if best_list is None:
        try:
            fallback = [1.0, 1.0]
            consider(fallback)
        except Exception:
            best_list = [1.0, 1.0]

    if best_list is None:
        best_list = [1.0, 1.0]

    # Authoritative final validation.
    try:
        val = float(evaluate_sequence(best_list))
        if not np.isfinite(val):
            raise ValueError
    except Exception:
        if best_arr is not None:
            try:
                fb = [float(x) for x in best_arr]
                if evaluate_sequence(fb) is not None and np.isfinite(float(evaluate_sequence(fb))):
                    best_list = fb
                else:
                    best_list = [1.0, 1.0]
            except Exception:
                best_list = [1.0, 1.0]
        else:
            best_list = [1.0, 1.0]

    out = []
    for x in best_list:
        try:
            y = float(x)
            if np.isfinite(y):
                out.append(float(np.clip(y, 0.0, 1000.0)))
        except Exception:
            pass
    if not out:
        out = [1.0]
    return out
