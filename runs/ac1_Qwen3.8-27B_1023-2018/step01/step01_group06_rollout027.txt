```python
def project(a):
    """Project onto {x >= 0, sum(x) = 1}"""
    # Use bisection on the threshold
    # We want: max(a_i - t, 0) for some t, such that sum = 1
    lo, hi = a.min() - 1, a.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(a - mid, 0))
        if s > 1:
            lo = mid
        else:
            hi = mid
    return np.maximum(a - (lo+hi)/2, 0)
```

```python
def solve_lp_update(a, M, top_indices, mode='HG'):
    """Solve the LP: max sum(b) s.t. (a*b)(k) <= M for k in top_indices, b >= 0"""
    m = len(top_indices)
    n = len(a)
    
    # Build constraint matrix (sparse)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_indices):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(idx)
            cols.append(i)
            vals.append(a[k - i])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(m, n))
    
    # Solve LP
    res = linprog(
        c=-np.ones(n),
        A_ub=A,
        b_ub=np.full(m, M),
        bounds=[(0, None)] * n,
        method='highs'
    )
    
    if res.success:
        return res.x
    return None
```

```python
    # Pick best initial
    best_a = alternatives[0].copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    for alt in alternatives[1:]:
        sc = float(evaluate_sequence(alt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = alt.copy()
    a = best_a.copy()

    # --- FFT setup ---
    conv_len = 2 * n - 1
    # Need fft_n >= 3n-2 for gradient computation, >= 2n-1 for conv
    fft_n = 1
    while fft_n < 3 * n - 2:
        fft_n <<= 1

    def fast_conv(x):
        """FFT-based linear convolution x*x."""
        F = np.fft.rfft(x, fft_n)
        G = np.fft.irfft(F * F, fft_n)
        return G[:conv_len]

    def compute_gradient(a, W):
        """Compute grad[i] = 2 * sum_k W[k] * a[k-i] for valid indices.
        Uses FFT-based cross-correlation."""
        # S[i] = sum_j a[j] * W[i+j] for j=0..n-1
        # C[n-1+i] = S[i] where C = conv(U_flip, V)
        U_flip = np.zeros(fft_n)
        U_flip[:n] = a[::-1]
        V = np.zeros(fft_n)
        V[:conv_len] = W
        C = np.fft.irfft(np.fft.rfft(V, fft_n) * np.fft.rfft(U_flip, fft_n), fft_n)
        S = C[n - 1:n - 1 + n]
        return 2.0 * S

    def project_simplex(x):
        """Project x onto {v >= 0, sum(v) = 1} via bisection."""
        if x.sum() < 1e-15:
            return np.ones(n) / n
        lo = x.min() - 1.0
        hi = x.max() + 1.0
        for _ in range(80):
            mid = (lo + hi) / 2
            s_arr = np.sum(np.maximum(x - mid, 0.0))
            if s_arr > 1.0:
                lo = mid
            else:
                hi = mid
        return np.maximum(x - (lo + hi) / 2.0, 0.0)

    def lp_update(a, M, top_indices):
        """LP-based update: max sum(b) s.t. (a*b)(k)<=M for k in top_indices, b>=0."""
        m = len(top_indices)
        if m == 0 or m > 200:
            return None
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(top_indices):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                rows.append(idx)
                cols.append(i)
                vals.append(a[k - i])
        if len(rows) == 0:
            return None
        A = sparse.csr_matrix((vals, (rows, cols)), shape=(m, n))
        try:
            res = linprog(
                c=-np.ones(n),
                A_ub=A,
                b_ub=np.full(m, M),
                bounds=[(0, None)] * n,
                method='highs',
                options={'presolve': True, 'time_limit': 20}
            )
            if res.success and res.x is not None and res.x.sum() > 1e-10:
                b = np.maximum(res.x, 0.0)
                b = b / b.sum()
                return b
        except Exception:
            pass
        return None

    # --- Main optimization loop ---
    lr = 0.02
    top_m = 40
    iter_count = 0
    patience = 0
    n_since_best = 0
    eta = 0.5  # multiplicative update rate

    while time.monotonic() < deadline - 5.0:
        iter_count += 1
        t_now = time.monotonic()

        # Compute autocorrelation
        c = fast_conv(a)
        M = c.max()
        est_score = 2.0 * n * M

        # Exact evaluation periodically
        do_eval = (iter_count % 8 == 0) or (est_score < best_score * 0.998)
        if do_eval:
            exact = float(evaluate_sequence(a.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_a = a.copy()
                patience = 0
                n_since_best = 0
            else:
                n_since_best += 8 if iter_count % 8 == 0 else 1
        else:
            n_since_best += 1

        if n_since_best > 400:
            patience += 1

        # --- Update strategy selection ---
        strategy = iter_count % 30

        if strategy < 18:
            # Gradient descent step
            # Find top-m lags
            if top_m >= conv_len:
                active = np.arange(conv_len)
            else:
                active = np.argpartition(c, -top_m)[-top_m:]
            # Soft weights
            c_active = c[active]
            thresh = c_active.min()
            spread = M - thresh
            if spread > 1e-14:
                w = 1.0 + 3.0 * (c_active - thresh) / spread
            else:
                w = np.ones(len(active))

            W_vec = np.zeros(conv_len)
            W_vec[active] = w

            # Compute gradient
            grad = compute_gradient(a, W_vec)

            # Step
            a = a - lr * grad
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-10:
                a = a / a_sum
            else:
                a = project_simplex(a)

        elif strategy < 24:
            # LP-based update
            top_m_lp = min(30, conv_len)
            active_lp = np.argpartition(c, -top_m_lp)[-top_m_lp:]
            b_star = lp_update(a, M, active_lp)
            if b_star is not None:
                t_blend = 0.3
                a = (1 - t_blend) * a + t_blend * b_star
                a = np.maximum(a, 0.0)
                a_sum = a.sum()
                if a_sum > 1e-10:
                    a = a / a_sum

        elif strategy < 28:
            # Multiplicative update (EM-like)
            # load[j] = sum_k c_active_weighted[k] * a[k-j] ...
            # Simpler: multiplicative update driven by c(k)
            active = np.argpartition(c, -top_m)[-top_m:]
            c_act = c[active]
            spread = M - c_act.min()
            if spread > 1e-14:
                w = np.ones(len(active)) + 2.0 * (c_act - c_act.min()) / spread
            else:
                w = np.ones(len(active))

            W_vec = np.zeros(conv_len)
            W_vec[active] = w
            load = np.zeros(n)
            # load[i] = sum_k W_vec[k] * a[k-i] (same as compute_gradient / 2)
            load = compute_gradient(a, W_vec) / 2.0

            lm = load.max()
            if lm > 1e-15:
                a = a * np.exp(-eta * load / lm)
            a = np.maximum(a, 1e-18)
            a = a / a.sum()

        else:
            # Perturbation / exploration
            if patience >= 5:
                n_pert = max(10, n // 50)
                idx = rng.choice(n, size=n_pert, replace=False)
                sigma = 0.15 + 0.1 * min(patience, 10)
                a[idx] *= np.exp(rng.normal(0, sigma, size=n_pert))
            else:
                # Small symmetric blend
                blend = rng.uniform(0.05, 0.2)
                a = (1 - blend) * a + blend * a[::-1]
            a = np.maximum(a, 0)
            a_sum = a.sum()
            if a_sum > 1e-10:
                a = a / a_sum
            patience = 0

        # Adaptive learning rate
        if iter_count % 300 == 0:
            lr = max(0.003, lr * 0.93)
            eta = max(0.15, eta * 0.95)

        # Reset patience on improvement
        if est_score <= M * 0.99:
            if best_score - est_score < 0.01:
                patience = max(0, patience - 1)

        # Time check
        if time.monotonic() > deadline - 5.0:
            break

    # --- Try one more round of LP polish on best ---
    if time.monotonic() < deadline - 15.0:
        a = best_a.copy()
        c = fast_conv(a)
        M = c.max()
        for polish_iter in range(5):
            if time.monotonic() > deadline - 10.0:
                break
            top_m_p = min(25, conv_len)
            active_p = np.argpartition(c, -top_m_p)[-top_m_p:]
            b_star = lp_update(a, M, active_p)
            if b_star is not None:
                t_blend = 0.2
                a_new = (1 - t_blend) * a + t_blend * b_star
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                c_check = fast_conv(a_new)
                if c_check.max() <= M * 1.01:
                    a = a_new
                    c = c_check
                    M = c.max()
            else:
                break
        if time.monotonic() < deadline - 5.0:
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

    # --- Final validation ---
    result = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(result))
    if not np.isfinite(final_score) or final_score > 2.5:
        # Fallback: all-ones gives exactly 2
        result = [1.0] * n
        final_score = float(evaluate_sequence(result))

    return result
```
