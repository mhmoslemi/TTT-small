    g = 0.5 * (g + g[::-1])
    s = float(g.sum())
    if not math.isfinite(s) or s <= 1e-12:
        return None
    return g / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        total = float(budget_s)
    except Exception:
        total = 1300.0
    if total <= 5.0:
        timeout = max(0.1, total * 0.8)
    else:
        timeout = max(1.0, total - 5.0)
    deadline = t0 + timeout

    rng = np.random.default_rng(seed)

    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent_arr = np.ones(4498, dtype=np.float64)
    if parent_arr.ndim != 1:
        try:
            parent_arr = parent_arr.reshape(-1)
        except Exception:
            parent_arr = np.ones(4498, dtype=np.float64)
    if parent_arr.size == 0:
        parent_arr = np.ones(4498, dtype=np.float64)
    n = int(parent_arr.size)
    if n <= 1:
        return [1.0]

    parent_arr = np.nan_to_num(parent_arr, nan=0.0, posinf=0.0, neginf=0.0)
    parent_arr = np.clip(parent_arr, 0.0, 1000.0)
    if float(parent_arr.sum()) <= 1e-300:
        parent_arr = np.ones(n, dtype=np.float64)

    uniform = np.ones(n, dtype=np.float64)

    best_seq = None
    best_score = float("inf")
    start_seq = None
    start_score = float("inf")

    def _consider(x):
        nonlocal best_seq, best_score, start_seq, start_score
        if time.monotonic() > deadline - 0.5:
            return
        arr = _norm_nonnegative(x)
        if arr is None or arr.size != n:
            return
        if not np.all(np.isfinite(arr)):
            return
        sc = _safe_eval(arr)
        if sc < best_score:
            best_score = sc
            best_seq = arr.copy()
        sup = _support_count(arr)
        if sc < start_score and sup >= max(4, n // 10):
            start_score = sc
            start_seq = arr.copy()

    _consider(uniform)
    _consider(_spike(n, 0, 0.0))
    _consider(_spike(n, n // 2, 1e-8))
    pnorm = _norm_nonnegative(parent_arr)
    if pnorm is not None:
        _consider(pnorm)
    psym = _norm_nonnegative(_sym_arr(parent_arr))
    if psym is not None:
        _consider(psym)

    if best_seq is None or not math.isfinite(best_score):
        best_seq = uniform
        best_score = 2.0

    if best_score >= 1.99 or best_score <= 0.2 or best_score < _TARGET:
        final_list = _as_list(best_seq)
        if time.monotonic() < deadline - 0.5:
            sc = _safe_eval(final_list)
            if not math.isfinite(sc):
                final_list = [1.0] * n
        return final_list

    if time.monotonic() > deadline - 120.0:
        return _as_list(best_seq)

    j = np.arange(n, dtype=np.float64)

    for w in (3, 5, 11, 31):
        if n > 2 * w:
            kernel = np.ones(w, dtype=np.float64) / float(w)
            _consider(np.convolve(parent_arr, kernel, mode="same"))
        if time.monotonic() > deadline - 120.0:
            break

    _consider(np.sin(np.pi * (j + 1.0) / float(n + 1)) ** 2)

    for c in (0.0, 0.25, 0.5, 0.75, 1.0):
        cc = c * float(n - 1)
        for sig in (0.05, 0.10, 0.20, 0.35, 0.50):
            _consider(np.exp(-0.5 * ((j - cc) / (max(1.0, sig * float(n)))) ** 2))
        if time.monotonic() > deadline - 120.0:
            break

    for _ in range(6):
        x = np.zeros(n, dtype=np.float64)
        x += np.exp(-0.5 * (j - rng.uniform(0.0, float(n))) ** 2 / max(1.0, rng.uniform(0.1, 0.3) * float(n)) ** 2)
        for _ in range(3):
            x += rng.uniform(0.0, 1.0) * np.exp(
                -0.5 * (j - rng.uniform(0.0, float(n))) ** 2 / max(1.0, rng.uniform(0.03, 0.15) * float(n)) ** 2
            )
        _consider(x + 1e-9)
        if time.monotonic() > deadline - 120.0:
            break

    for alpha in (0.5, 1.0, 2.0):
        x = np.linspace(1.0, 0.0, n, dtype=np.float64) ** alpha
        _consider(x + 1e-9)
        _consider(x[::-1] + 1e-9)
        if time.monotonic() > deadline - 120.0:
            break

    for eps in (1e-2, 1e-4):
        for center in (0, n // 2, n - 1):
            _consider(_spike(n, center, eps))
        if time.monotonic() > deadline - 120.0:
            break

    if best_seq is None or not math.isfinite(best_score):
        best_seq = uniform
        best_score = 2.0

    if best_score >= 1.99 or best_score <= 0.2 or best_score < _TARGET:
        final_list = _as_list(best_seq)
        if time.monotonic() < deadline - 0.5:
            sc = _safe_eval(final_list)
            if not math.isfinite(sc):
                final_list = [1.0] * n
        return final_list

    if start_seq is None:
        start_seq = pnorm if pnorm is not None else uniform
    current = _norm_nonnegative(start_seq)
    if current is None:
        current = uniform

    cur_score = _safe_eval(current)
    if cur_score < best_score:
        best_score = cur_score
        best_seq = current.copy()
    sym_cur = _norm_nonnegative(_sym_arr(current))
    if sym_cur is not None:
        sym_score = _safe_eval(sym_cur)
        if sym_score < best_score:
            best_score = sym_score
            best_seq = sym_cur.copy()
        if sym_score < cur_score:
            current = sym_cur
            cur_score = sym_score

    def _lp_update(cur):
        nonlocal best_seq, best_score
        cur = _norm_nonnegative(cur)
        if cur is None:
            return None
        cur_score = _safe_eval(cur)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()

        for Tlow in (32, 64, 128):
            if time.monotonic() > deadline - 120.0:
                break
            if n <= 12000:
                conv = np.convolve(cur, cur)
            else:
                conv = _fft_auto(cur)
            off = np.asarray(conv[1:n], dtype=np.float64)
            if off.size == 0:
                break
            off = np.maximum(off, 0.0)
            M = float(off.max())
            if not math.isfinite(M) or M <= 0.0:
                break

            lows = _select_lows(off, Tlow)
            g = _lp_solve(cur, lows, M)
            if g is None:
                continue

            best_mix = cur.copy()
            best_mix_score = cur_score

            for gg in (g, _norm_nonnegative(_sym_arr(g))):
                if gg is None:
                    continue
                for t in (0.25, 0.50, 0.75, 1.00):
                    cand = (1.0 - t) * cur + t * gg
                    cand = _norm_nonnegative(cand)
                    if cand is None:
                        continue
                    sc = _safe_eval(cand)
                    if sc < best_score:
                        best_score = sc
                        best_seq = cand.copy()
                    if sc < best_mix_score:
                        best_mix_score = sc
                        best_mix = cand.copy()
                    if time.monotonic() > deadline - 120.0:
                        break
                if time.monotonic() > deadline - 120.0:
                    break

            current_lp = best_mix
            if best_mix is not None:
                cur = _norm_nonnegative(best_mix)
                if cur is not None:
                    cur_score = best_mix_score
            if best_score < _TARGET:
                return current_lp
        return cur

    if _HAVE_SCIPY and time.monotonic() < deadline - 180.0:
        updated = _lp_update(current)
        if updated is not None:
            current = updated

    def _mu_search(cur):
        nonlocal best_seq, best_score
        cur = _norm_nonnegative(cur)
        if cur is None:
            return

        eta = 0.7
        top_frac = 0.12
        floor = 1e-12
        it = 0
        stagn = 0
        base_score = _safe_eval(cur)
        if base_score < best_score:
            best_score = base_score
            best_seq = cur.copy()

        while time.monotonic() < deadline - 30.0 and it < 1000:
            conv = _fft_auto(cur)
            off = np.asarray(conv[1:n], dtype=np.float64)
            if off.size == 0:
                break
            off = np.maximum(off, 0.0)
            M = float(off.max())
            if not math.isfinite(M) or M <= 0.0:
                break

            T = min(off.size, max(32, min(256, int(top_frac * off.size))))
            idx = np.argpartition(off, -T)[-T:]
            lags = idx + 1
            thr = float(off[idx].min())
            span = max(M - thr, 1e-300)
            ws = 1.0 + (off[idx] - thr) / span

            load = np.zeros(n, dtype=np.float64)
            for lag, w in zip(lags, ws):
                k = int(lag)
                if k <= 0 or k >= n:
                    continue
                load[:n - k] += w * cur[k:]
                load[k:] += w * cur[:n - k]

            lm = float(load.max())
            if math.isfinite(lm) and lm > 1e-300:
                cur = cur * np.exp(-eta * load / lm)
            cur = np.maximum(cur, floor)
            cur = _norm_nonnegative(cur)
            if cur is None:
                break

            it += 1
            if it % 4 == 0:
                sc = _safe_eval(cur)
                if sc < best_score:
                    best_score = sc
                    best_seq = cur.copy()
                    stagn = 0
                    base_score = min(base_score, sc)
                else:
                    stagn += 1
                if best_score < _TARGET:
                    return

            if it % 25 == 0:
                center = int(rng.integers(n))
                spike = _spike(n, center, 0.0)
                for mix in (0.7, 0.9):
                    cand = (1.0 - mix) * cur + mix * spike
                    cand = _norm_nonnegative(cand)
                    if cand is not None:
                        sc = _safe_eval(cand)
                        if sc < best_score:
                            best_score = sc
                            best_seq = cand.copy()
                        if best_score < _TARGET:
                            return

            if it % 50 == 0:
                top_frac = max(0.02, top_frac * 0.90)
                eta = min(1.5, eta * 1.05)
                if stagn > 6:
                    if best_seq is not None and _support_count(best_seq) >= max(4, n // 20):
                        cur2 = _norm_nonnegative(best_seq)
                        if cur2 is not None:
                            nidx = max(1, n // 50)
                            idx2 = rng.integers(0, n, size=nidx)
                            cur2[idx2] *= np.exp(rng.normal(0.0, 0.2, size=nidx))
                            cur2 = _norm_nonnegative(cur2)
                            if cur2 is not None:
                                cur = cur2
                    stagn = 0

    if best_score > _TARGET and time.monotonic() < deadline - 60.0:
        _mu_search(current)

    if best_seq is None:
        best_seq = uniform

    final_list = _as_list(best_seq)
    final_list = [0.0 if v < 0.0 else (1000.0 if v > 1000.0 else v) for v in final_list]

    if time.monotonic() < deadline - 0.5:
        sc = _safe_eval(final_list)
        if not math.isfinite(sc):
            final_list = [1.0] * n
        elif sc > 2.0 and time.monotonic() < deadline - 1.0:
            u = [1.0] * n
            usc = _safe_eval(u)
            if math.isfinite(usc) and usc < sc:
                final_list = u

    if not final_list:
        final_list = [1.0] * n
    for v in final_list:
        if not math.isfinite(v):
            final_list = [1.0] * n
            break
    return final_list
