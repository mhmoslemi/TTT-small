def _lp_iteration(a, deadline):
    """Single LP iteration: find g that maximizes sum(g) subject to a*g <= M, g >= 0."""
    n = a.size
    b = _fft_convolve(a, a)
    M = float(np.max(b))
    if M <= 0 or not math.isfinite(M):
        return None
    
    # Build constraint matrix: (a*g)[k] = sum_j a[|k-j|... ] 
    # Actually (a*g)[k] = sum_{j: 0<=k-j<n} a[k-j] * g[j]
    # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a Toeplitz-like matrix (lower triangular part from a, upper from reversed a)
    
    # For n=96, the matrix is 191 x 96. Let's build it.
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_min = max(0, k-n+1)
        j_max = min(k, n-1)
        for j in range(j_min, j_max+1):
            A[k, j] = a[k-j]
    
    # Solve LP: max sum(g) s.t. A @ g <= M, g >= 0
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M * (1 + 1e-10)),
                     bounds=[(0, None)]*n, method='highs')
    except Exception:
        return None
    
    if res.x is None:
        return None
    
    g = np.clip(res.x, 0, None)
    sg = float(np.sum(g))
    if sg < 1e-15:
        return None
    g = g / sg
    return g
