import numpy as np
import time
import math


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 5.0)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).flatten()
        if parent.size < 2 or not np.all(np.isfinite(parent)):
            raise ValueError("bad parent")
    except (NameError, AttributeError, ValueError):
        parent = np.ones(6962, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.size
    conv_len = 2 * n - 1
    conv_fft = 1 << conv_len.bit_length()

    def do_conv(a):
        Af = np.fft.rfft(a, conv_fft)
        return np.fft.irfft(Af * Af, conv_fft)[:conv_len]

    def topk_crosscorr(a, B, k):
        k = min(k, conv_len)
        top = np.argpartition(B, -k)[-k:]
        c = np.zeros(n)
        for r in top:
            ri = int(r)
            lo = max(0, ri - n + 1)
            hi = min(n - 1, ri)
            if lo <= hi:
                seg = a[ri - hi : ri - lo + 1][::-1]
                c[lo:hi + 1] += seg
        c /= float(k)
        return c

    def synth_score(a):
        B = do_conv(a)
        S = a.sum()
        if S < 0.01:
            return np.inf, B, S
        return 2.0 * n * B.max() / (S * S), B, S

    # --- Initialization ---
    best = parent.copy()
    best_sc, _, _ = synth_score(best)

    idx = np.arange(n, dtype=np.float64)
    alts = [best.copy(), np.ones(n, dtype=np.float64)]
    for freq in [1, 2, 3, 5, 8]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            for ph in [0.0, 0.5, 1.0, 1.5]:
                c = 1.0 + amp * np.cos(2.0 * np.pi * freq * idx / n + ph)
                c = np.maximum(c, 0.05)
                c *= n / c.sum()
                alts.append(c.copy())
    half = n // 2
    for d in [0.2, 0.4, 0.6, 0.8]:
        c = np.ones(n, dtype=np.float64)
        c[:half] = 1.0 + d
        c[half:] = max(0.05, 1.0 - d)
        c *= n / c.sum()
        alts.append(c.copy())
    for _ in range(4):
        c = rng.random(n) + 0.3
        c *= n / c.sum()
        alts.append(c.copy())

    for c in alts:
        c = np.clip(c, 0.0, 1000.0)
        sc, _, _ = synth_score(c)
        if np.isfinite(sc) and sc < best_sc:
            best = c.copy()
            best_sc = sc

    # --- Main optimization ---
    a = best.copy()
    total_budget = max(1.0, float(budget_s) - 15.0)
    idx_arr = np.arange(n, dtype=np.float64)

    round_iters = 600
    max_rounds = 5000

    for rnd in range(max_rounds):
        if time.monotonic() > deadline - 8.0:
            break

        frac = min(1.0, (time.monotonic() - t0) / total_budget)
        k_top = max(1, int(40.0 * (1.0 - 0.92 * frac) + 1))
        lr = 0.012 * max(0.15, 1.0 - 0.85 * frac)

        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        b1, b2, eps_a = 0.9, 0.999, 1e-8
        patience = 180
        last_imp = 0
        round_stopped = False

        for it in range(round_iters):
            if time.monotonic() > deadline - 8.0:
                break

            B = do_conv(a)
            S = a.sum()
            M = B.max()
            sc = 2.0 * n * M / (S * S)

            if sc < best_sc - 1e-13:
                best = a.copy()
                best_sc = sc
                last_imp = it

            if it - last_imp > patience:
                round_stopped = True
                break

            c = topk_crosscorr(a, B, k_top)
            grad = 4.0 * n * (S * c - M) / (S * S * S)
            grad -= grad.mean()

            t_adam += 1
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** t_adam)
            v_hat = v / (1.0 - b2 ** t_adam)
            step = lr * m_hat / (np.sqrt(v_hat) + eps_a)

            a = a - step
            np.maximum(a, 0.01, out=a)
            s = a.sum()
            if s > 0.01:
                a *= n / s

        # --- Perturbation between rounds ---
        if time.monotonic() > deadline - 8.0:
            break

        r = rng.random()
        amp_base = 0.04 + 0.12 * rng.random()
        if r < 0.35:
            f1 = int(rng.integers(1, 40))
            ph = float(rng.uniform(0, 2.0 * np.pi))
            a = best + amp_base * np.cos(2.0 * np.pi * f1 * idx_arr / n + ph)
        elif r < 0.55:
            f1 = int(rng.integers(1, 40))
            ph = float(rng.uniform(0, 2.0 * np.pi))
            a = best + amp_base * np.sin(2.0 * np.pi * f1 * idx_arr / n + ph)
        elif r < 0.70:
            a = best + amp_base * 0.5 * rng.standard_normal(n)
        elif r < 0.85:
            scale = 1.0 + amp_base * 0.3 * rng.standard_normal(n)
            a = best * scale
        else:
            # Asymmetric two-tone
            f1 = int(rng.integers(1, 20))
            f2 = int(rng.integers(20, 50))
            a = (best
                 + amp_base * 0.5 * np.cos(2.0 * np.pi * f1 * idx_arr / n)
                 + amp_base * 0.3 * np.cos(2.0 * np.pi * f2 * idx_arr / n))

        np.maximum(a, 0.01, out=a)
        s = a.sum()
        if s > 0.01:
            a *= n / s

        # Quick score check after perturbation
        sc_p, _, _ = synth_score(a)
        if np.isfinite(sc_p) and sc_p < best_sc:
            best = a.copy()
            best_sc = sc_p

    # --- Final verification ---
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01 or not np.all(np.isfinite(best)):
        best = np.ones(n)

    final_sc = float(evaluate_sequence(best.tolist()))

    par_sc = float(evaluate_sequence(parent.tolist()))
    if par_sc < final_sc:
        best = parent.copy()
        final_sc = par_sc

    if not np.isfinite(final_sc):
        best = np.ones(n)

    return [float(max(0.0, min(1000.0, x))) for x in best]
