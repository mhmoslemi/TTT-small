```python
def _simplex_project(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u > (cssv - s) / np.arange(1, n+1)
    if not cond.any():
        return np.full(n, s/n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0)
```
