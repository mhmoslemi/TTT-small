        if sc_try < best_score:
            best = a_try.copy()
            best_score = sc_try

    # Try a few structured initializations
    idx = np.arange(n, dtype=np.float64)
    inits = []
    for freq, amp in [(1, 0.3), (1, 0.5), (2, 0.2), (1, 0.7), (3, 0.15)]:
        sig = 1.0 + amp * np.sin(2.0 * np.pi * freq * idx / n)
        sig = np.maximum(sig, 0.0)
        if sig.sum() > 0.01:
            sig = sig * n / sig.sum()
            inits.append(sig.copy())
    for d in [0.1, 0.2, 0.3]:
        sig = np.ones(n)
        sig[n // 2:] = 1.0 - d
        sig = np.maximum(sig, 0.0)
        if sig.sum() > 0.01:
            sig = sig * n / sig.sum()
            inits.append(sig.copy())

    for init in inits:
        sc_i = quick_score(init)
        if sc_i < best_score:
            best = init.copy()
            best_score = sc_i

    # Main optimization: subgradient descent on max_k conv(a,a)[k]
    a = best.copy()
    patience = 0
    max_patience = 5000
    global_iter = 0

    while time.monotonic() < deadline:
        global_iter += 1
        t_elapsed = time.monotonic() - start_time

        B = fft_convolve(a)
        S = a.sum()
        B_max = B.max()
        cur_score = 2.0 * n * B_max / (S * S)

        if cur_score < best_score - 1e-9:
            best = a.copy()
            best_score = cur_score
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            # Perturb and restart from best
            a = best + rng.normal(0.0, 0.05, n)
            a = np.maximum(a, 0.0)
            a = a * n / max(a.sum(), 0.01)
            patience = 0
            max_patience = max(2000, max_patience // 2)
            continue

        # Find top-K peaks (adaptive K)
        if global_iter < 2000:
            K = min(20, 2 * n - 1)
        else:
            K = min(5, 2 * n - 1)
        peak_idx = np.argpartition(B, -K)[-K:]
        # Only keep those within 2% of max
        thresh = B_max * 0.98
        peak_idx = peak_idx[B[peak_idx] >= thresh]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(B)])
        peak_idx = peak_idx[:K]

        # Compute subgradient: grad[j] = 2 * sum_{k in peaks} a[k-j]
        grad = np.zeros(n)
        for k in peak_idx:
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_max < j_min:
                continue
            idx_arr = k - np.arange(j_min, j_max + 1)
            grad[j_min:j_max + 1] += 2.0 * a[idx_arr]

        # Project to preserve sum (subtract mean)
        grad -= grad.mean()

        grad_norm = np.sqrt(np.dot(grad, grad))
        if grad_norm < 1e-12:
            break

        # Adaptive learning rate
        lr = 2.0 / (n * grad_norm)

        # Cap to stay non-negative
        pos_mask = grad > 1e-12
        if pos_mask.any():
            max_allowed = np.min(a[pos_mask] / grad[pos_mask])
            lr = min(lr, 0.85 * max_allowed)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0, dtype=np.float64)
        a_new_sum = a_new.sum()
        if a_new_sum > 0.01:
            a = a_new * n / a_new_sum
        else:
            patience += 10
            continue

        # Periodic LP refinement (every ~60s or every 50K iters)
        if (global_iter % 50000 == 0 and time.monotonic() < deadline - 60):
            try:
                import cvxpy as cp
                B_now = fft_convolve(a)
                M_now = B_now.max()
                top_lags = np.argpartition(B_now, -80)[-80:]
                top_lags = top_lags[B_now[top_lags] >= M_now * 0.95]
                if len(top_lags) == 0:
                    top_lags = np.array([np.argmax(B_now)])

                A_mat = np.zeros((len(top_lags), n))
                for ii, k in enumerate(top_lags):
                    j_min = max(0, k - n + 1)
                    j_max = min(k, n - 1)
                    idx_arr = k - np.arange(j_min, j_max + 1)
                    A_mat[ii, j_min:j_max + 1] = a[idx_arr]

                b_var = cp.Variable(n, nonneg=True)
                prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [A_mat @ b_var <= M_now])
                prob.solve(solver=cp.CBC, time_limit=30, threads=1)

                if b_var.value is not None and np.all(np.isfinite(b_var.value)):
                    b_val = np.maximum(b_var.value, 0.0)
                    b_sum = b_val.sum()
                    if b_sum > 0.01:
                        for t in [0.05, 0.1, 0.2, 0.3]:
                            blend = (1 - t) * a + t * b_val * a.sum() / b_sum
                            blend = np.maximum(blend, 0.0)
                            bs = blend.sum()
                            if bs > 0.01:
                                blend = blend * n / bs
                                sc_blend = quick_score(blend)
                                if sc_blend < best_score:
                                    best = blend.copy()
                                    best_score = sc_blend
                                    a = blend.copy()
            except Exception:
                pass

        # Periodic exact evaluation
        if global_iter % 20000 == 0 and time.monotonic() < deadline - 10:
            sc_exact = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc_exact) and sc_exact < best_score:
                best = a.copy()
                best_score = sc_exact

    # Final polish: a few more LP steps if time allows
    if time.monotonic() < deadline - 40:
        try:
            import cvxpy as cp
            for _ in range(3):
                if time.monotonic() > deadline - 10:
                    break
                B_final = fft_convolve(best)
                M_final = B_final.max()
                top_lags = np.argpartition(B_final, -100)[-100:]
                top_lags = top_lags[B_final[top_lags] >= M_final * 0.90]
                if len(top_lags) == 0:
                    top_lags = np.array([np.argmax(B_final)])

                A_mat = np.zeros((len(top_lags), n))
                for ii, k in enumerate(top_lags):
                    j_min = max(0, k - n + 1)
                    j_max = min(k, n - 1)
                    idx_arr = k - np.arange(j_min, j_max + 1)
                    A_mat[ii, j_min:j_max + 1] = best[idx_arr]

                b_var = cp.Variable(n, nonneg=True)
                prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [A_mat @ b_var <= M_final])
                prob.solve(solver=cp.CBC, time_limit=20, threads=1)

                if b_var.value is not None and np.all(np.isfinite(b_var.value)):
                    b_val = np.maximum(b_var.value, 0.0)
                    b_sum = b_val.sum()
                    if b_sum > 0.01:
                        S_b = best.sum()
                        for t in [0.05, 0.1, 0.15, 0.2, 0.3]:
                            blend = (1 - t) * best + t * b_val * S_b / b_sum
                            blend = np.maximum(blend, 0.0)
                            bs = blend.sum()
                            if bs > 0.01:
                                blend = blend * n / bs
                                sc_b = quick_score(blend)
                                if sc_b < best_score:
                                    best = blend.copy()
                                    best_score = sc_b
        except Exception:
            pass

    # Final validation
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        best = np.ones(n)
        final_score = float(evaluate_sequence(best.tolist()))

    result = [float(max(0.0, min(1000.0, x))) for x in best]
    return result
