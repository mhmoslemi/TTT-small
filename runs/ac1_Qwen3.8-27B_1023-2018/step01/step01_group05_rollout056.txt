        rho = np.nonzero(v_sorted * np.arange(1, len(v)+1) > (cumsum - target_sum))[0][-1]

    fft_corr = 1 << (3 * n - 2).bit_length()

        return np.fft.irfft(A * A, fft_conv)[:2 * n - 1]

        conv = np.fft.irfft(Af * Wf, fft_corr)
        return conv[n - 1:2 * n - 1]
