            a = _project_onto_simplex(a, n)
            no_improve = 0
            max_no_improve = int(max(20, max_no_improve * 1.2))
            continue
        
        # LP direction
        g = _solve_lp(a, M, n)
        
        if g is None:
            # Fallback: subgradient
            grad = _subgradient_direction(a, n)
            gn = float(np.linalg.norm(grad))
            if gn < 1e-15:
                break
            g = a - 0.1 * grad / gn
            g = _project_onto_simplex(g, n)
        
        # Check if g is too similar to a
        cos_sim = float(np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-30))
        if cos_sim > 0.999:
            no_improve += 3
            a = a + rng.normal(0, 0.05, n)
            a = _project_onto_simplex(a, n)
            continue
        
        # Line search
        t, new_mb = _line_search(a, g, n)
        
        if new_mb < M * (1 - 1e-10):
            a_new = (1 - t) * a + t * g
            a = _project_onto_simplex(a_new, n)
        else:
            # Try subgradient step
            grad = _subgradient_direction(a, n)
            gn = float(np.linalg.norm(grad))
            if gn > 1e-15:
                step = 0.5 / gn
                a_new = a - step * grad
                a_new = _project_onto_simplex(a_new, n)
                mb_new = float(_fft_conv_self(a_new).max())
                if mb_new < M:
                    a = a_new
                else:
                    no_improve += 1
            else:
                no_improve += 5
    
    return best_a, best_mb

def _local_search(a, deadline, rng):
    n = len(a)
    a = _project_onto_simplex(a.copy(), n)
    best_a = a.copy()
    best_mb = float(_fft_conv_self(a).max())
    
    sigma = 0.04
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() < deadline:
        iter_count += 1
        r = rng.random()
        
        if r < 0.25:
            cand = a + rng.normal(0, sigma, n)
        elif r < 0.45:
            cand = a.copy()
            idx = int(rng.integers(0, n))
            width = int(rng.integers(1, max(2, n // 30)))
            lo = max(0, idx - width // 2)
            hi = min(n, lo + width)
            cand[lo:hi] *= np.exp(rng.normal(0, 0.25, hi - lo))
        elif r < 0.60:
            cand = a.copy()
            lo = int(rng.integers(0, max(1, n // 5)))
            width = int(rng.integers(max(1, n // 20), max(2, n // 4)))
            hi = min(n, lo + width)
            cand[lo:hi] *= rng.uniform(0.4, 2.0)
        elif r < 0.75:
            cand = a.copy()
            for _ in range(3):
                idx = int(rng.integers(0, n))
                cand[idx] *= rng.uniform(0.5, 1.8)
        elif r < 0.85:
            # Palindrome projection
            cand = 0.5 * (a + a[::-1])
            cand += rng.normal(0, sigma * 0.5, n)
        else:
            cand = 0.92 * a + 0.08 * np.ones(n) / n
            cand += rng.normal(0, sigma * 0.3, n)
        
        cand = _project_onto_simplex(cand, n)
        mb = float(_fft_conv_self(cand).max())
        
        if mb < best_mb * (1 - 1e-10):
            best_mb = mb
            best_a = cand.copy()
            a = cand.copy()
            no_improve = 0
            sigma = max(0.005, sigma * 0.97)
        else:
            no_improve += 1
            if no_improve > 80:
                sigma = min(0.4, sigma * 1.6)
                no_improve = 0
                if no_improve > 500:
                    break
    
    return best_a, best_mb

def _init_variants(n, rng, variant):
    i = np.arange(n, dtype=np.float64)
    v = int(variant) % 7
    
    if v == 0:
        f1 = int(rng.integers(1, n // 2 + 1))
        f2 = int(rng.integers(1, n // 2 + 1))
        a = 1.0 + 0.4*np.sin(2*np.pi*f1*i/n + rng.uniform(0, 6.28)) + 0.3*np.cos(2*np.pi*f2*i/n + rng.uniform(0, 6.28))
    elif v == 1:
        a = np.exp(rng.normal(0, 0.8, n))
    elif v == 2:
        a = 0.5 + 0.5*np.abs(np.sin(np.pi * i / n * rng.uniform(1, 4)))
    elif v == 3:
        a = np.abs(rng.normal(0, 1, n)) + 0.1
    elif v == 4:
        a = np.ones(n)
        for _ in range(4):
            c = rng.uniform(0, n)
            w = rng.uniform(n//8, n//3)
            a += rng.uniform(0.2, 0.8) * np.exp(-(i-c)**2 / (2*w**2))
    elif v == 5:
        a = 1.0 + 0.6*np.sin(2*np.pi*i/n * rng.integers(2, max(3, n//4)))
        a += 0.2*np.sin(2*np.pi*i/n * rng.integers(5, max(6, n//2)))
    else:
        a = rng.uniform(0.3, 1.5, n)
    
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s))
    deadline = t0 + budget - 12.0  # 12s headroom
    
    parent = _get_parent()
    
    # Evaluate parent
    try:
        parent_score = float(evaluate_sequence([float(x) for x in np.clip(parent, 0, 1000)]))
    except Exception:
        parent_score = math.inf
    
    best_seq = np.clip(parent, 0, 1000).astype(np.float64).copy()
    best_score = parent_score
    
    if not np.isfinite(best_score):
        best_seq = np.ones(32) / 32.0
        best_score = 2.0
    
    rng = np.random.default_rng(seed)
    n_parent = len(best_seq)
    
    # Phase 1: LP-optimized at parent length (parent init)
    phase1_deadline = min(deadline - 30, t0 + budget * 0.45)
    if time.monotonic() < phase1_deadline:
        a0 = _project_onto_simplex(best_seq.copy(), n_parent)
        try:
            res_a, res_mb = _lp_optimize(n_parent, a0, phase1_deadline, rng)
            fast_sc = 2.0 * n_parent * res_mb
            if fast_sc < best_score - 1e-9:
                exact = float(evaluate_sequence([float(x) for x in res_a]))
                if np.isfinite(exact) and exact < best_score:
                    best_seq = res_a.copy()
                    best_score = exact
        except Exception:
            pass
    
    # Phase 2: Multiple restarts at parent length
    phase2_deadline = min(deadline - 40, t0 + budget * 0.72)
    if time.monotonic() < phase2_deadline:
        num_restarts = 15
        for rid in range(num_restarts):
            if time.monotonic() > phase2_deadline - 5:
                break
            if best_score < 1.50:
                break
            a_init = _init_variants(n_parent, rng, variant=rid + 20)
            remaining = max(5.0, (phase2_deadline - time.monotonic()) - 10.0)
            sub_dl = time.monotonic() + min(45.0, remaining / max(1, num_restarts - rid))
            try:
                res_a, res_mb = _lp_optimize(n_parent, a_init, sub_dl, rng)
                fast_sc = 2.0 * n_parent * res_mb
                if fast_sc < best_score - 1e-9:
                    exact = float(evaluate_sequence([float(x) for x in res_a]))
                    if np.isfinite(exact) and exact < best_score:
                        best_seq = res_a.copy()
                        best_score = exact
            except Exception:
                pass
    
    # Phase 3: Local search polishing on best
    phase3_deadline = deadline - 10
    if time.monotonic() < phase3_deadline:
        a_norm = _project_onto_simplex(best_seq.copy(), len(best_seq))
        try:
            res_a, res_mb = _local_search(a_norm, phase3_deadline, np.random.default_rng(seed + 7777))
            fast_sc = 2.0 * len(res_a) * res_mb
            if fast_sc < best_score - 1e-12:
                exact = float(evaluate_sequence([float(x) for x in res_a]))
                if np.isfinite(exact) and exact < best_score:
                    best_seq = res_a.copy()
                    best_score = exact
        except Exception:
            pass
    
    # Phase 4: Try a couple of other lengths
    if time.monotonic() < deadline - 10:
        for L in [128, 256]:
            if time.monotonic() > deadline - 8:
                break
            a_init = _init_variants(L, rng, variant=L % 7)
            sub_dl = time.monotonic() + max(5.0, min(20.0, deadline - time.monotonic() - 6.0))
            try:
                res_a, res_mb = _lp_optimize(L, a_init, sub_dl, rng)
                fast_sc = 2.0 * L * res_mb
                if fast_sc < best_score - 1e-6:
                    exact = float(evaluate_sequence([float(x) for x in res_a]))
                    if np.isfinite(exact) and exact < best_score:
                        best_seq = res_a.copy()
                        best_score = exact
                # Also try local search on this
                if time.monotonic() < deadline - 5:
                    res_a2, res_mb2 = _local_search(res_a, deadline - 4, np.random.default_rng(seed + L))
                    fast_sc2 = 2.0 * L * res_mb2
                    if fast_sc2 < best_score - 1e-9:
                        exact2 = float(evaluate_sequence([float(x) for x in res_a2]))
                        if np.isfinite(exact2) and exact2 < best_score:
                            best_seq = res_a2.copy()
                            best_score = exact2
            except Exception:
                pass
    
    # Final validation
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_score = math.inf
    
    if not np.isfinite(final_score):
        fb = np.ones(32) / 32.0
        try:
            fs = float(evaluate_sequence([float(x) for x in fb]))
            if np.isfinite(fs):
                return [float(x) for x in fb]
        except Exception:
            pass
        return [1.0]
    
    return [float(x) for x in best_seq]
