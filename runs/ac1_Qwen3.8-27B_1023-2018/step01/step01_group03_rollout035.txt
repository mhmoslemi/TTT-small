import numpy as np
import time
from scipy.optimize import linprog

def _normalize(q):
    q = np.asarray(q, dtype=np.float64)
    q = np.where(np.isfinite(q) & (q > 0), q, 0.0)
    s = q.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(q.size) / max(1, q.size)
    return q / s

def _score(a):
    n = len(a)
    if n == 0:
        return float('inf')
    b = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * float(b.max()) / (s * s)

def _build_A(q):
    n = len(q)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = q
    return A

def _lp_step(q, M, deadline):
    n = len(q)
    if time.monotonic() > deadline - 30:
        return None
    A = _build_A(q)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 5, 'dual_feasibility_tolerance': 1e-9})
        if res is not None and getattr(res, 'success', False):
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
            s = g.sum()
            if s > 0 and np.isfinite(s):
                return g / s
    except Exception:
        pass
    return None

def _grad_direction(q, active_frac=0.15, sharpness=6.0):

    b = np.convolve(q, q)
    M = float(b.max())
    K = max(2, int(len(b) * active_frac))
    top_k = np.argsort(b)[::-1][:K]
    w = np.exp((b[top_k] - M) * sharpness)
    wsum = w.sum()
    if wsum <= 0:
        return np.zeros(n)
    w = w / wsum
    grad = np.zeros(n, dtype=np.float64)
    for k, wi in zip(top_k, w):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            m = np.arange(lo, hi + 1)
            partners = k - m
            grad[m] += 2.0 * wi * q[partners]
    grad -= grad.mean()
    return grad

    best_cand = None
    best_sc = best_score

def _polish(q, rng, deadline, max_iters=2000):

    best = q.copy()
    best_sc = _score(q)
    step = 0.5 / np.sqrt(n)
    stagnation = 0
