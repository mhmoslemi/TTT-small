                return float("inf")
            val = float(evaluate_sequence(seq))
            return val if np.isfinite(val) else float("inf")
        except Exception:
            return float("inf")

    def _update_approx(score: float, arr):
        if np.isfinite(score) and score < state["best_approx_score"]:
            state["best_approx_score"] = float(score)
            state["best_approx_arr"] = np.asarray(arr, dtype=np.float64).copy()

    def _try_exact(arr, min_interval=10.0):
        now = time.monotonic()
        if now - last_exact["t"] < min_interval:
            return
        if state["best_exact_arr"] is not None and now >= opt_end - 15:
            return
        sc = _exact_score(arr)
        last_exact["t"] = now
        if np.isfinite(sc) and sc < state["best_exact_score"]:
            state["best_exact_score"] = float(sc)
            state["best_exact_arr"] = np.asarray(arr, dtype=np.float64).copy()

    def _current_best_arr():
        if state["best_exact_arr"] is not None:
            return state["best_exact_arr"]
        if state["best_approx_arr"] is not None:
            return state["best_approx_arr"]
        return anchor

    # ------------------------------------------------------------------
    # Initial candidate generation
    # ------------------------------------------------------------------
    if parent is not None:
        anchor = _normalize(parent)
    else:
        anchor = _normalize(np.ones(n, dtype=np.float64))

    cands = []
    init_deadline = t0 + min(60.0, max(5.0, 0.08 * budget))

    def _consider(a):
        if time.monotonic() > init_deadline:
            return
        try:
            arr = _normalize(a)
            M = _fast_M(arr)
            sc = _score_from_M(M)
            if np.isfinite(sc):
                cands.append((sc, arr))
                _update_approx(sc, arr)
        except Exception:
            pass

    _consider(anchor)
    if parent is not None:
        _consider(0.5 * (anchor + anchor[::-1]))

    _consider(np.ones(n, dtype=np.float64))

    x = np.arange(n, dtype=np.float64)
    for p in (0.45, 0.50, 0.55):
        left = (x + 1.0) ** (-p)
        right = (n - x) ** (-p)
        _consider(left)
        _consider(right)
        _consider(left + 0.25 * right)
        _consider(right + 0.25 * left)

    # Smooth random starts
    for _ in range(2):
        d = rng.standard_normal(n)
        D = np.fft.rfft(d)
        cutoff = max(2, n // 40)
        if D.size > cutoff:
            D[cutoff:] *= 0.0
        d = np.fft.irfft(D, n)
        d -= d.mean()
        _consider(np.exp(0.25 * d))

    # Mix top candidates multiplicatively / additive with the anchor
    if cands:
        cands.sort(key=lambda z: z[0])
        top = cands[:6]
        mix_deadline = t0 + min(120.0, max(15.0, 0.12 * budget))
        for _, barr in top:
            if time.monotonic() > mix_deadline:
                break
            if barr is anchor:
                continue
            for t in (0.1, 0.25, 0.5):
                if time.monotonic() > mix_deadline:
                    break
                mixed_add = (1.0 - t) * anchor + t * barr
                _consider(mixed_add)
                mixed_log = np.exp(
                    (1.0 - t) * np.log(anchor + FLOOR)
                    + t * np.log(barr + FLOOR)
                )
                _consider(mixed_log)

    # Exact-validate a few promising starts
    exact_deadline = t0 + min(180.0, max(20.0, 0.16 * budget))
    if cands:
        cands.sort(key=lambda z: z[0])
        for sc, arr in cands[: max(2, min(5, int(budget // 30) + 1))]:
            if time.monotonic() > exact_deadline:
                break
            es = _exact_score(arr)
            last_exact["t"] = time.monotonic()
            if np.isfinite(es) and es < state["best_exact_score"]:
                state["best_exact_score"] = float(es)
                state["best_exact_arr"] = np.asarray(arr, dtype=np.float64).copy()
        if state["best_exact_arr"] is not None:
            _update_approx(state["best_exact_score"], state["best_exact_arr"])

    if state["best_approx_arr"] is None:
        _update_approx(_score_from_M(_fast_M(anchor)), anchor)

    # ------------------------------------------------------------------
    # Optimization primitives
    # ------------------------------------------------------------------
    def _instantiate_update(a, z, eta):
        logf = -eta * z
        if not np.all(np.isfinite(logf)):
            logf = np.nan_to_num(logf, nan=0.0, posinf=50.0, neginf=-50.0)
        logf = np.clip(logf, -50.0, 50.0)
        new = a * np.exp(logf)
        new = np.nan_to_num(new, nan=0.0, posinf=1e300, neginf=0.0)
        s = float(new.sum())
        if not np.isfinite(s) or s <= 1e-12:
            return None
        new = new * (S_TARGET / s)
        new = np.maximum(new, FLOOR)
        s = float(new.sum())
        if not np.isfinite(s) or s <= 1e-12:
            return None
        new = new * (S_TARGET / s)
        if new.max() > 1000.0:
            new = new * (900.0 / float(new.max()))
            s = float(new.sum())
            if s > 1e-12:
                new = new * (S_TARGET / s)
        new = np.clip(new, 0.0, 1000.0)
        if not np.all(np.isfinite(new)):
            return None
        return new

    def _c_direct(a, idx, w):
        c = np.zeros(n, dtype=np.float64)
        for kk, ww in zip(idx, w):
            ww = float(ww)
            if ww <= 0.0 or not np.isfinite(ww):
                continue
            k = int(kk)
            if k < 0 or k >= L:
                continue
            s = 0 if k < n else k - n + 1
            e = n - 1 if k > n - 1 else k
            if s > e:
                continue
            # indices j=s..e receive a[k-j].  Reverse order is a[k-s] ... a[k-e]
            start = k - s
            last = k - e
            if last > 0:
                seg = a[start : last - 1 : -1]
            else:
                seg = a[start :: -1]
            if seg.size == e - s + 1:
                c[s : e + 1] += ww * seg
        return c

    def _c_fft(a, A, p):
        W = np.fft.rfft(p[::-1], N)
        C = np.fft.irfft(W * A, N)
        c = C[n - 1 : 2 * n - 1][::-1]
        c = np.maximum(c, 0.0)
        return c

    def _topk_weights(B, K, alpha, scale):
        K = int(min(max(1, K), L))
        if K >= L:
            return np.arange(L), np.full(L, 1.0 / L, dtype=np.float64)
        idx = np.argpartition(B, L - K)[-K:]
        b = B[idx]
        bmax = float(np.max(b))
        shift = (b - bmax) / scale
        if alpha > 1e-6:
            shift = np.maximum(shift, -40.0 / max(alpha, 1e-6))
            w = np.exp(alpha * shift)
            sw = float(w.sum())
            if not np.isfinite(sw) or sw <= 0.0:
                w = np.full(K, 1.0 / K, dtype=np.float64)
            else:
                w /= sw
        else:
            w = np.full(K, 1.0 / K, dtype=np.float64)
        return idx, w

    def _perturb(a, scale):
        d = rng.standard_normal(n)
        D = np.fft.rfft(d)
        cutoff = max(2, n // 40)
        if D.size > cutoff:
            D[cutoff:] *= 0.0
        d = np.fft.irfft(D, n)
        d -= d.mean()
        sd = float(d.std())
        if sd > 1e-12:
            d /= sd
        d2 = rng.standard_normal(n)
        d2 -= d2.mean()
        sd2 = float(d2.std())
        if sd2 > 1e-12:
            d = d + 0.2 * (d2 / sd2)
        new = a * np.exp(scale * d)
        return _normalize(new)

    def _run_phase(start_arr, end_time, K_start, K_end,
                   alpha_start, alpha_end,
                   target_start, target_end, max_target_log):
        now0 = time.monotonic()
        phase_dur = max(1e-6, end_time - now0)
        a = _normalize(start_arr)
        target = float(target_start)
        no_improve = 0
        step = 0

        while True:
            now = time.monotonic()
            if now >= end_time - 0.5:
                break
            if now0 + phase_dur <= now:
                phase_dur = max(1e-6, end_time - now0)

            frac = min(1.0, max(0.0, (now - now0) / phase_dur))
            K = int(round(K_start + (K_end - K_start) * frac))
            K = int(min(max(1, K), L))
            alpha = float(alpha_start + (alpha_end - alpha_start) * frac)
            target = float(target_start + (target_end - target_start) * frac)
            target = min(max_target_log, target)

            try:
                A = np.fft.rfft(a, N)
                B = np.fft.irfft(A * A, N)[:L]
                B = np.maximum(B, 0.0)
                M = float(np.max(B))
            except Exception:
                break

            if not np.isfinite(M):
                a = _normalize(_current_best_arr())
                no_improve += 50
                continue

            sc = _score_from_M(M)
            if np.isfinite(sc) and sc < state["best_approx_score"]:
                state["best_approx_score"] = float(sc)
                state["best_approx_arr"] = a.copy()
                if time.monotonic() < end_time - 30:
                    _try_exact(a, min_interval=15.0)

            if time.monotonic() - last_exact["t"] > 60.0 and time.monotonic() < end_time - 30:
                if state["best_approx_arr"] is not None:
                    _try_exact(state["best_approx_arr"], min_interval=0.0)

            scale = max(1.0, S_TARGET)
            try:
                idx, w = _topk_weights(B, K, alpha, scale)
                if K <= 64:
                    c = _c_direct(a, idx, w)
                else:
                    p = np.zeros(L, dtype=np.float64)
                    p[idx] = w
                    c = _c_fft(a, A, p)
            except Exception:
                c = None

            if c is None or c.size != n or not np.all(np.isfinite(c)):
                no_improve += 100
            else:
                c = np.maximum(c, 0.0)
                sa = float(a.sum())
                if not np.isfinite(sa) or sa <= 1e-12:
                    no_improve += 100
                else:
                    mu = float(np.dot(a, c) / sa)
                    z = c - mu
                    zmax = float(z.max())
                    zmin = float(z.min())
                    zrange = zmax - zmin

                    if (not np.isfinite(zrange)) or zrange < 1e-12:
                        no_improve += 100
                    else:
                        eta = min(1.5, target / zrange)
                        best_M = M
                        best_a = a
                        improved = False

                        for fac in (1.0, 0.5, 0.25):
                            if time.monotonic() >= end_time - 2.0:
                                break
                            try:
                                new = _instantiate_update(a, z, eta * fac)
                            except Exception:
                                new = None
                            if new is None:
                                continue
                            try:
                                M2 = _fast_M(new)
                            except Exception:
                                continue
                            if np.isfinite(M2) and M2 < best_M - 1e-8:
                                best_M = M2
                                best_a = new
                                improved = True
                                break

                        a = best_a
                        sc_new = _score_from_M(best_M)
                        if np.isfinite(sc_new) and sc_new < state["best_approx_score"]:
                            state["best_approx_score"] = float(sc_new)
                            state["best_approx_arr"] = a.copy()
                            if time.monotonic() < end_time - 30:
                                _try_exact(a, min_interval=15.0)

                        if improved:
                            no_improve = 0
                            target = min(max_target_log, target * 1.03)
                        else:
                            no_improve += 1
                            target = max(0.5 * target_start, target * 0.95)

            if no_improve > 220:
                scale_p = 0.05 + 0.05 * min(3.0, float(no_improve) / 600.0)
                try:
                    a = _perturb(a, scale_p)
                except Exception:
                    a = _normalize(_current_best_arr())
                no_improve = 0
                target = max(0.5 * target_start, target * 0.7)

            step += 1
            if step % 350 == 0 and time.monotonic() < end_time - 10:
                try:
                    sym = _normalize(0.5 * (a + a[::-1]))
                    Ms = _fast_M(sym)
                    scs = _score_from_M(Ms)
                    if np.isfinite(scs) and scs < state["best_approx_score"]:
                        state["best_approx_score"] = float(scs)
                        state["best_approx_arr"] = sym.copy()
                    if np.isfinite(Ms) and Ms < M - 1e-7:
                        a = sym
                except Exception:
                    pass

        if time.monotonic() < opt_end and time.monotonic() < hard_end - 20:
            _try_exact(a, min_interval=0.0)

    # ------------------------------------------------------------------
    # Scheduled optimization
    # ------------------------------------------------------------------
    if time.monotonic() < opt_end:
        start_arr = _normalize(_current_best_arr())
        rem = max(0.0, opt_end - time.monotonic())

        K_base = min(L, max(128, n // 4))

        if rem > 120:
            endA = time.monotonic() + 0.32 * rem
            _run_phase(
                start_arr,
                endA,
                K_start=K_base,
                K_end=min(L, max(96, K_base // 4)),
                alpha_start=6.0,
                alpha_end=80.0,
                target_start=0.06,
                target_end=0.20,
                max_target_log=0.35,
            )

        start_arr = _normalize(_current_best_arr())
        rem = max(0.0, opt_end - time.monotonic())
        if rem > 60:
            endB = time.monotonic() + 0.48 * rem
            _run_phase(
                start_arr,
                endB,
                K_start=min(L, max(128, K_base // 4)),
                K_end=min(L, 32),
                alpha_start=40.0,
                alpha_end=250.0,
                target_start=0.10,
                target_end=0.32,
                max_target_log=0.55,
            )

        start_arr = _normalize(_current_best_arr())
        if time.monotonic() < opt_end:
            _run_phase(
                start_arr,
                opt_end,
                K_start=min(L, 32),
                K_end=max(1, min(L, 4)),
                alpha_start=120.0,
                alpha_end=700.0,
                target_start=0.15,
                target_end=0.50,
                max_target_log=0.75,
            )

    # ------------------------------------------------------------------
    # Final validation and fallback
    # ------------------------------------------------------------------
    final_bases = []

    def _add_final_base(arr):
        try:
            if arr is None:
                return
            arr = _normalize(arr)
            if arr is not None and np.all(np.isfinite(arr)) and arr.sum() > 0.01:
                final_bases.append(arr)
        except Exception:
            pass

    _add_final_base(state["best_exact_arr"])
    _add_final_base(state["best_approx_arr"])
    _add_final_base(anchor)
    _add_final_base(np.ones(n, dtype=np.float64))

    if final_bases:
        used = []
        uniq = []
        for arr in final_bases:
            key = (float(np.max(arr)), float(np.sum(arr * np.arange(n, dtype=np.float64))))
            if key in used:
                continue
            used.append(key)
            uniq.append(arr)
        final_bases = uniq

        for base in list(final_bases)[:6]:
            if time.monotonic() >= hard_end - 20:
                break
            _try_exact(base, min_interval=0.0)

            try:
                sym = _normalize(0.5 * (base + base[::-1]))
                if time.monotonic() < hard_end - 15:
                    _try_exact(sym, min_interval=0.0)
            except Exception:
                pass

            try:
                if base.size == n:
                    arr2 = base.copy()
                    thr = max(FLOOR, 1e-6 * float(arr2.mean()))
                    mask = arr2 < thr
                    if mask.any() and not mask.all():
                        arr2[mask] = 0.0
                        arr2 = _normalize(arr2)
                        if time.monotonic() < hard_end - 15:
                            _try_exact(arr2, min_interval=0.0)
            except Exception:
                pass

    out_arr = None
    if state["best_exact_arr"] is not None:
        out_arr = state["best_exact_arr"]
    elif state["best_approx_arr"] is not None:
        out_arr = state["best_approx_arr"]
    else:
        out_arr = np.ones(n, dtype=np.float64)

    out_arr = _normalize(out_arr)

    if time.monotonic() < hard_end - 10:
        sc = _exact_score(out_arr)
        if not np.isfinite(sc):
            out_arr = _normalize(np.ones(n, dtype=np.float64))
            sc = _exact_score(out_arr)
        if not np.isfinite(sc) and state["best_exact_arr"] is not None:
            out_arr = _normalize(state["best_exact_arr"])

    res = []
    for x in np.asarray(out_arr, dtype=np.float64).ravel():
        try:
            v = float(x)
        except Exception:
            v = 0.0
        if not np.isfinite(v):
            v = 0.0
        if v < 0.0:
            v = 0.0
        elif v > 1000.0:
            v = 1000.0
        res.append(v)

    if not res or sum(res) < 0.01:
        res = [float(S_TARGET / n) for _ in range(n)]

    # Return with a small safety margin if possible.
    if time.monotonic() > hard_end - 2:
        pass
    return res
