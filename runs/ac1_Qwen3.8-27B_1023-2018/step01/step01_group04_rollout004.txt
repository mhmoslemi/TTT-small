def _clean_sequence(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.nan_to_num(a, nan=0.0, posinf=1.0, neginf=0.0)

    return a


def _normalize(a):
    a = _clean_sequence(a)
    n = a.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    s = float(np.sum(a))
    if not math.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return a / s


def _score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float("inf")
    s = float(np.sum(a))
    if not math.isfinite(s) or s <= 1e-300:
        return float("inf")
    try:
        b = np.convolve(a, a)
        mb = float(np.max(b))
    except Exception:
        return float("inf")
    if not math.isfinite(mb):
        return float("inf")
    val = (2.0 * float(n) * mb) / (s * s)
    return float(val) if math.isfinite(val) else float("inf")

def _coordinate_optimize(a, best, rng, time_limit, max_passes=5000):

    n = int(a.size)
    if n < 2 or time_limit <= 0:
        return a

    t_start = time.monotonic()
    stop = t_start + float(time_limit)
    m = 2 * n - 1
    two_n = 2.0 * float(n)

    try:
        c = np.convolve(a, a)
        local = float(two_n * np.max(c))
    except Exception:
        return a

    if math.isfinite(local):
        _update_best(a, best)
    else:
        return a

    no_improve = 0
    perturbs = 0

    while time.monotonic() < stop - 1.0:
        try:
            c = np.convolve(a, a)
            local = float(two_n * np.max(c))
        except Exception:
            break
        if not math.isfinite(local):
            break

        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() >= stop - 1.0:
                break

            xi = float(a[i])
            s0 = 1.0 - xi
            if s0 <= 1e-15:
                continue

            C = c.copy()
            C[i:i + n] -= 2.0 * xi * a
            jc = 2 * i
            if jc < m:
                C[jc] = c[jc] - xi * xi

            A = np.zeros(m, dtype=np.float64)
            A[i:i + n] = 2.0 * a
            if jc < m:
                A[jc] = 0.0

            xmax = 5.0 + 10.0 * xi

            def obj(x, C=C, A=A, jc=jc, s0=s0, two_n=two_n):
                den = s0 + x
                if den <= 1e-300:
                    return math.inf
                term = C + A * x
                if jc >= 0:
                    term[jc] += x * x
                mt = float(np.max(term))
                if mt <= 0.0:
                    return 0.0
                return two_n * mt / (den * den)

            best_v = obj(xi)
            best_x = xi

            if n <= 384:
                pts = (0.0, 0.25 * xmax, 0.5 * xmax, 0.75 * xmax, xmax)
            else:
                pts = (0.0, 0.5 * xmax, xmax)

            for p in pts:
                try:

                    if v < best_v:

                        best_x = p
                except Exception:
                    pass

                        obj,
                        bounds=(0.0, xmax),
                        method="bounded",

                        v = obj(xx)

                    pass

                    a_new = a.copy()

                    if sn > 1e-15:
                        a_new /= sn
                        c_new /= (sn * sn)

                            c = c_new
                            local = new_local

                    pass

        no_improve += 1

        elif no_improve > 2:

def _lp_precompute(n):
    n = int(n)
    if n <= 0:
        return None
    if n in _lp_cache:
        return _lp_cache[n]

    try:
        m = 2 * n - 1
        total = n * n
        if total > 2000000000:
            return None

        indptr = np.zeros(m + 1, dtype=np.int32)
        indices = np.empty(total, dtype=np.int32)
        src = np.empty(total, dtype=np.int32)

        pos = 0
        for k in range(m):
            start = max(0, k - n + 1)
            end = min(n, k + 1)
            count = end - start
            if count > 0:
                js = np.arange(start, end, dtype=np.int32)
                indices[pos:pos + count] = js
                src[pos:pos + count] = (k - js).astype(np.int32)
                pos += count
            indptr[k + 1] = pos

        if pos != total:
            return None

        _lp_cache[n] = (indices, indptr, src)
        return _lp_cache[n]
    except Exception:
        return None


def _lp_search(a, best, rng, time_limit):
    a = _normalize(a)
    n = int(a.size)
    if n < 4 or n > MAX_LP_N or time_limit < 5:
        return a
    if linprog is None or sps is None:
        return a

    t_start = time.monotonic()
    stop = t_start + float(time_limit)

    local = _score_fast(a)
    if not math.isfinite(local):
        return a
    _update_best(a, best)

    stagnant = 0
    perturb_count = 0

    while time.monotonic() < stop - 3.0:
        try:
            info = _lp_precompute(n)
        except Exception:
            break
        if info is None:
            break
        indices, indptr, src = info

        try:
            a_solve = a.copy()
            eps = 1e-10 * max(1.0 / float(n), float(np.max(a_solve)))
            a_solve = a_solve + eps
            ss = float(np.sum(a_solve))
            if ss <= 1e-15:
                break
            a_solve /= ss

            cfull = np.convolve(a_solve, a_solve)
            M = float(np.max(cfull))
            if not math.isfinite(M) or M <= 1e-300:
                break
            M_scaled = float(n * n * M)

            data = a_solve[src] * float(n)
            A = sps.csr_matrix((data, indices, indptr), shape=(2 * n - 1, n))
            A = A.tocsc()
        except Exception:
            break

        cvec = -np.ones(n, dtype=np.float64)
        ymax = 100.0 * float(n)
        bounds = [(0.0, ymax)] * n

        factors = [1.0]
        if stagnant >= 2:
            factors.append(0.97)
        if stagnant >= 4:
            factors.append(0.93)
        if stagnant >= 6:
            factors.append(0.90)

        got_direction = False

        for factor in factors:
            if time.monotonic() >= stop - 3.0:
                break
            b_ub = M_scaled * factor + 1e-10 * max(1.0, M_scaled)
            rem = stop - time.monotonic()
            if rem <= 1.0:
                break

            opts = {"presolve": True}
            tl = min(45.0 if n > 512 else 25.0, max(1.0, rem * 0.8))
            opts["time_limit"] = tl
            res = None
            try:
                res = linprog(
                    cvec,
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=bounds,
                    method="highs",
                    options=opts,
                )
            except Exception:
                opts.pop("time_limit", None)
                try:
                    res = linprog(
                        cvec,
                        A_ub=A,
                        b_ub=b_ub,
                        bounds=bounds,
                        method="highs",
                        options=opts,
                    )
                except Exception:
                    res = None

            if res is None or not getattr(res, "success", False) or getattr(res, "x", None) is None:
                continue

            y = np.asarray(res.x, dtype=np.float64).ravel()
            y = np.nan_to_num(y, nan=0.0, posinf=0.0, neginf=0.0)
            y = np.clip(y, 0.0, None)
            sy = float(np.sum(y))
            if sy <= 1e-12:
                continue
            g = y / sy

            best_sc = local
            best_cand = None
            for t in (0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 1.0):
                cand = g if t >= 1.0 else (1.0 - t) * a + t * g
                cand = np.maximum(cand, 0.0)
                sc_sum = float(np.sum(cand))
                if sc_sum <= 1e-15:
                    continue
                cand = cand / sc_sum
                sc = _score_fast(cand)
                if math.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_cand = cand

            if best_cand is not None and best_sc < local * (1.0 - 1e-10):
                a = best_cand
                local = best_sc
                stagnant = 0
                got_direction = True
                if local < float(best[0]) - 1e-9:
                    _update_best(a, best)
                break

        if not got_direction:
            stagnant += 1
            if stagnant >= 3 and time.monotonic() < stop - 5.0:
                amp = 0.10 / max(1.0, 0.25 * float(stagnant))
                ap = a * np.exp(amp * rng.standard_normal(n))
                ap = _normalize(ap)
                sp = _score_fast(ap)
                if math.isfinite(sp) and sp < local * 1.20:
                    a = ap
                    local = sp
                    if local < float(best[0]) - 1e-9:
                        _update_best(a, best)
                    stagnant = 0
                    perturb_count += 1
                    if perturb_count > 6:
                        break
                    continue
                else:
                    break
            if stagnant > 18:
                break

    _update_best(a, best)
    return a


def _process_length(n, alloc, rng, best, parent_arr):
    n = int(n)
    if n < 2 or n > MAX_SEARCH_N or alloc <= 0:
        return

    t_start = time.monotonic()
    stop = t_start + float(alloc)

    try:
        starts = _make_starts(n, rng, parent_arr, best[1])
    except Exception:
        return
    if not starts:
        return

    if n <= 128:
        limit = 3
    elif n <= 256:
        limit = 2
    else:
        limit = 1

    for a0 in starts[:limit]:
        if time.monotonic() >= stop - 8.0:
            break
        a = a0.copy()

        rem = stop - time.monotonic()
        if n <= 256:
            t1 = min(rem * 0.35, 70.0)
        elif n <= 768:
            t1 = min(rem * 0.35, 40.0)
        else:
            t1 = min(rem * 0.35, 25.0)
        if t1 > 3.0:
            a = _coordinate_optimize(a, best, rng, t1, max_passes=1000)

        rem = stop - time.monotonic()
        if n <= MAX_LP_N and rem > 25.0:
            if n <= 96:
                t_lp = min(rem * 0.45, 250.0)
            elif n <= 256:
                t_lp = min(rem * 0.45, 90.0)
            elif n <= 768:
                t_lp = min(rem * 0.45, 50.0)
            else:
                t_lp = min(rem * 0.45, 30.0)
            if t_lp > 5.0:
                a = _lp_search(a, best, rng, t_lp)

        rem = stop - time.monotonic()
        if rem > 8.0:
            if n <= 256:
                t2 = min(rem * 0.5, 50.0)
            elif n <= 768:
                t2 = min(rem * 0.5, 25.0)
            else:
                t2 = min(rem * 0.5, 15.0)
            if t2 > 3.0:
                a = _coordinate_optimize(a, best, rng, t2, max_passes=500)

    if n <= 512:
        while time.monotonic() < stop - 12.0:
            rem = stop - time.monotonic()
            t_quick = min(18.0, rem * 0.40)
            if t_quick < 6.0:
                break

            r = float(rng.random())
            base = best[1] if best[1] is not None else parent_arr
            if base is not None and r < 0.70:
                amp = 0.05 + 0.20 * float(rng.random())
                a = _resize(base, n) * np.exp(amp * rng.standard_normal(n))
            else:
                alpha = float(rng.choice(np.array([0.3, 0.6, 1.0, 2.0], dtype=np.float64)))
                try:
                    a = rng.dirichlet(np.full(n, alpha, dtype=np.float64))
                except Exception:
                    a = np.ones(n, dtype=np.float64) / float(n)
            a = _coordinate_optimize(a, best, rng, t_quick, max_passes=80)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    global MAX_SEARCH_N

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    try:
        rem = float(budget_s)
    except Exception:
        rem = 1300.0
    if not math.isfinite(rem) or rem <= 0.0:
        rem = 1e-3

    reserve = 25.0 if rem > 80.0 else max(2.0, 0.10 * rem)
    hard_deadline = t0 + max(0.0, rem - reserve)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = _normalize(h)
            if p.size > 0:
                parent = p
    except Exception:
        parent = None

    best_score = None
    if parent is not None:
        try:
            sc = evaluate_sequence([float(x) for x in parent])
            if math.isfinite(sc):
                best_score = float(sc)
        except Exception:
            best_score = None

    fallback = np.ones(96, dtype=np.float64) / 96.0
    if best_score is None:
        sc = _score_fast(fallback)
        best_score = float(sc) if math.isfinite(sc) else 2.0
        best = [float(best_score), fallback.copy()]
    else:
        best = [float(best_score), parent.copy()]

    if not math.isfinite(best[0]):
        best = [2.0, fallback.copy()]

    parent_n = int(parent.size) if parent is not None else 96
    MAX_SEARCH_N = int(min(max(1536, parent_n), 2048))

    allocs = [
        (96, 250.0),
        (128, 80.0),
        (160, 60.0),
        (192, 80.0),
        (256, 110.0),
        (384, 90.0),
        (512, 150.0),
        (768, 110.0),
        (1024, 180.0),
        (1280, 110.0),
        (1536, 90.0),
    ]

    if parent_n != 96 and 1 <= parent_n <= MAX_SEARCH_N:
        allocs = [(parent_n, 180.0), (n, a) for (n, a) in allocs if n != parent_n]

    for n, preset in allocs:
        now = time.monotonic()
        if now >= hard_deadline - 15.0:
            break
        n = int(n)
        if n < 2 or n > MAX_SEARCH_N:
            continue
        alloc = min(float(preset), max(0.0, hard_deadline - now - 10.0))
        if alloc < 10.0:
            break
        try:
            _process_length(n, alloc, rng, best, parent)
        except Exception:
            pass

    now = time.monotonic()
    if now < hard_deadline - 5.0:
        try:
            a = best[1].copy()
            tfin = min(25.0, max(0.0, hard_deadline - now - 2.0))
            a = _coordinate_optimize(a, best, rng, tfin, max_passes=200)
        except Exception:
            pass

    final = _normalize(best[1])
    try:
        fin_sc = evaluate_sequence([float(x) for x in final])
        if not math.isfinite(fin_sc) or fin_sc >= float(best[0]) + 1e-9:
            try:
                pf = _normalize(parent) if parent is not None else fallback
                ps = evaluate_sequence([float(x) for x in pf])
                if math.isfinite(ps) and (not math.isfinite(fin_sc) or ps < fin_sc):
                    final = pf
            except Exception:
                pass
    except Exception:
        pass

    final = _clean_sequence(final)
    final = _normalize(final)
    if final.size == 0:
        final = np.array([1.0], dtype=np.float64)

    try:
        # Final exact check; if it somehow fails, the normalized fallback is still valid.
        final_sc = evaluate_sequence([float(x) for x in final])
        if not math.isfinite(final_sc):
            fallback2 = _normalize(parent if parent is not None else fallback)
            return [float(x) for x in fallback2]
    except Exception:
        pass

    return [float(x) for x in final]
