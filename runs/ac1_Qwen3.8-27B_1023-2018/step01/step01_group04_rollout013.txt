        try:
            add(rng.dirichlet(np.full(n, alpha)))
        except:
            pass

    if parent is not None and parent.size >= 2:
        p = np.asarray(parent, dtype=np.float64).ravel()
        p = np.nan_to_num(p, nan=0.0, posinf=0.0)
        p = np.clip(p, 0.0, None)
        if p.sum() > 0:
            tp = np.linspace(0, 1, p.size)
            add(np.interp(x, tp, p))
            # Perturbed

        if math.isfinite(sc):

        add(g2)
        g3 = np.interp(x, tp, par) * np.exp(0.3 * rng.standard_normal(n))
        add(g3)
    if not S:
        S.append(np.ones(n) / n)
    S.sort(key=_sc)
    return S


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    try:
        rem = max(0.0, float(budget_s))
    except Exception:
        rem = 1300.0
    dl = t0 + rem - 25.0

    best = [float('inf'), np.array([1.0, 1.0], dtype=np.float64)]
    _upd(best[1], best)

    par = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            par = np.asarray(h, dtype=np.float64).ravel()
            par = np.nan_to_num(par, nan=0.0, posinf=0.0, neginf=0.0)
            par = np.clip(par, 0.0, 1000.0)
            if par.size >= 2:
                _upd(par, best)
    except Exception:
        par = None

    n_par = par.size if par is not None else 96

    # Phase 1: LP at parent length – primary workhorse
    if time.monotonic() < dl - 120.0:
        starts = _mk(n_par, rng, best[1])
        for idx, s in enumerate(starts[:8]):
            if time.monotonic() >= dl - 120.0:
                break
            alloc = min(130.0, (dl - 120.0 - time.monotonic()) * 0.35)
            if alloc < 10.0:
                break
            _lpirate(s, time.monotonic() + alloc, best, rng, max_it=2000)

    # Phase 2: LP at n=128
    if time.monotonic() < dl - 100.0:
        starts128 = _mk(128, rng, best[1])
        for s in starts128[:6]:
            if time.monotonic() >= dl - 100.0:
                break
            alloc = min(90.0, (dl - 100.0 - time.monotonic()) * 0.4)
            if alloc < 10.0:
                break
            _lpirate(s, time.monotonic() + alloc, best, rng, max_it=1500)

    # Phase 3: More LP at parent length with fresh starts
    if time.monotonic() < dl - 80.0:
        starts3 = _mk(n_par, rng, best[1])
        for s in starts3[3:10]:
            if time.monotonic() >= dl - 80.0:
                break
            alloc = min(60.0, (dl - 80.0 - time.monotonic()))
            if alloc < 8.0:
                break
            _lpirate(s, time.monotonic() + alloc, best, rng, max_it=1000)

    # Phase 4: LP at n=192 (if time remains)
    if time.monotonic() < dl - 70.0:
        starts192 = _mk(192, rng, best[1])
        for s in starts192[:3]:
            if time.monotonic() >= dl - 70.0:
                break
            alloc = min(60.0, (dl - 70.0 - time.monotonic()))
            if alloc < 10.0:
                break
            _lpirate(s, time.monotonic() + alloc, best, rng, max_it=800)

    # Phase 5: Gradient refinement at best length and n=128
    if time.monotonic() < dl - 40.0:
        for ng in (n_par, 128):
            if time.monotonic() >= dl - 40.0:
                break
            alloc = min(60.0, (dl - 40.0 - time.monotonic()) * 0.5)
            if alloc < 8.0:
                break
            _grad(ng, best[1], alloc, best, rng)

    # Phase 6: Final LP polish at the best candidate's length
    if time.monotonic() < dl - 12.0:
        nb = best[1].size
        alloc = min(40.0, dl - 12.0 - time.monotonic())
        if alloc > 5.0:
            _lpirate(best[1], time.monotonic() + alloc, best, rng, max_it=500)

    # Final: validate and return
    fin = np.asarray(best[1], dtype=np.float64).ravel().copy()
    fin = np.nan_to_num(fin, nan=0.0, posinf=0.0, neginf=0.0)
    fin = np.clip(fin, 0.0, 1000.0)
    if fin.size < 1:
        fin = np.array([1.0], dtype=np.float64)
    if float(fin.sum()) < 0.01:
        fin = np.ones(fin.size, dtype=np.float64)
    _upd(fin, best)
    fin = np.asarray(best[1], dtype=np.float64).ravel()
    v = evaluate_sequence([float(x) for x in fin])
    if not math.isfinite(v):
        fin = np.array([1.0], dtype=np.float64)
    return [float(x) for x in fin]
