a >= 0

- At current point a, compute the gradients of (a*a)[k] at a
- These give valid under-estimators: (a*a)[k'] >= (a*a)[k] + grad_k(a) · (a' - a)
- Add these as linear constraints
- Solve the resulting LP
- Iterate

This is similar to the Kelley cutting plane method. But it might be slow to converge.

- This LP is linear! The constraints (f*b)_k <= M are linear in b.
- The solution b is a "dual" direction that tries to spread mass to where f*b is small.
- Rescale b to sum 1, blend with f.

- The value at row k, column j is f[k-j] (since (f*b)_k = sum_j f[k-j] b[j])

    w: weights on [0, 2n-2], should sum to 1
    Returns grad[i] = sum_k w[k] (x[k-i] + x[i-k])

    # Column j: rows j to min(j+n-1, 2n-2), values f[0] to f[min(n-1, 2n-2-j)]
    n_cols = n

def propose_candidate(seed=42, budget_s=1300, **kwargs):

    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    # ---- Load and normalize parent ----
    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

        a = np.ones(256, dtype=np.float64)
    n = int(a.size)
    a = np.maximum(a, 0.0)
    s0 = a.sum()
    if s0 < 0.01:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s0

    # ---- FFT convolution ----
    fft_len = 1
    while fft_len < 2 * n - 1:
        fft_len <<= 1

    def conv_self(x):
        F = np.fft.rfft(x, fft_len)
        return np.fft.irfft(F * F, fft_len)[:2 * n - 1]

    # ---- Exact subgradient of (x*x)[k] ----
    def grad_peak(x, k):
        k = int(k)
        g = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] = 2.0 * x[k - lo:k - hi - 1:-1]
        return g

    def compute_grad(x, b, K, tau_frac):
        K = min(K, 2 * n - 1)
        idx = np.argpartition(b, -K)[-K:]
        vals = b[idx]
        vmax = vals.max()
        tau = tau_frac * max(vmax, 1e-30)
        w = np.exp((vals - vmax) / tau)
        ws = w.sum()
        if ws < 1e-30:
            return grad_peak(x, int(idx[np.argmax(vals)]))
        w /= ws
        g = np.zeros(n)
        for ii, ww in zip(idx, w):
            g += ww * grad_peak(x, int(ii))
        return g

    # ---- LP direction (occasional rescue) ----
    lp_A_cache = None

    def lp_direction(f, M, tl=45):
        nonlocal lp_A_cache
        if time.monotonic() > deadline - tl - 10:
            return None
        try:
            if lp_A_cache is None:
                ki = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()

            ki, ji = lp_A_cache

            res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),

            if res.success:
                b = np.maximum(res.x, 0)
                sb = b.sum()
                if sb > 1e-15:

            return None
        except Exception:

        b = conv_self(a)
