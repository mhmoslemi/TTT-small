import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if not np.isfinite(s) or s < 1e-300:
        return np.full(z.shape, 1.0 / z.size, dtype=np.float64)
    return e / s


def _fft_autocorr(a, fsize):
    F = rfft(a, fsize)
    b = irfft(F * F, fsize)[:2 * len(a) - 1]
    np.maximum(b, 0.0, out=b)
    return b


def _grad_soft(a, b, tau, n, topk):
    L = len(b)
    bm = float(b.max())
    x = np.clip((b - bm) / tau, -100.0, 0.0)
    e = np.exp(x)
    es = float(e.sum())
    if es < 1e-300:
        idx = np.array([int(np.argmax(b))])
        p = np.zeros(L)
        p[idx] = 1.0
        U = bm
    else:
        p = e / es
        U = bm + tau * math.log(es)

    if L > topk:
        idx = np.argpartition(-p, topk - 1)[:topk]
    else:
        idx = np.arange(L)

    pt = p[idx]
    pts = float(pt.sum())
    if pts < 1e-300:
        return np.zeros(n, dtype=np.float64), U
    pt = pt / pts
    bt = b[idx]
    b_bar = float(np.dot(pt, bt))

    s = np.zeros(n, dtype=np.float64)
    for ki in range(len(idx)):
        k = int(idx[ki])
        w = float(pt[ki])
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            s[lo:hi + 1] += w * a[k - hi:k - lo + 1][::-1]

    grad = 2.0 * a * (s - b_bar)
    grad[~np.isfinite(grad)] = 0.0
    return grad, U


def _init_z(n, method, rng, parent):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    eps = 1e-10
    if method == 0:
        a = np.ones(n)
    elif method == 1:
        a = np.sin(np.pi * (i + 0.5) / n) ** 2
    elif method == 2:
        a = np.exp(-0.8 * ((i - c) / (n * 0.4)) ** 2)
    elif method == 3:
        a = 0.5 * (1 - np.cos(2 * np.pi * (i + 0.5) / n))
    elif method == 4:
        a = np.exp(-((i - c) / (n * 0.3)) ** 4)
    elif method == 5:
        a = np.ones(n)
        for _ in range(3):
            f = int(rng.integers(1, max(2, n // 3)))
            ph = float(rng.uniform(0, 2 * np.pi))
            am = float(rng.uniform(0.3, 0.9))
            a += am * np.cos(2 * np.pi * f * i / n + ph)
        a = np.maximum(a, eps)
    elif method == 6 and parent is not None:
        P = len(parent)
        idx = np.round(np.linspace(0, P - 1, n)).astype(int)
        a = np.maximum(parent[idx], eps)
    elif method == 7 and parent is not None:
        P = len(parent)
        a = np.maximum(parent, eps)
        if P == n:
            a = a.copy()
        elif P > n:
            blocks = np.array_split(a, n)
            a = np.array([b.mean() for b in blocks])
        else:
            xp = np.linspace(0, 1, P)
            xn = np.linspace(0, 1, n)
            a = np.interp(xn, xp, a)
        a = np.maximum(a, eps)
    elif method == 8:
        a = 1.0 + 0.5 * np.cos(2 * np.pi * i / max(1, n - 1))
    elif method == 9:
        a = np.exp(-1.2 * np.abs(i - c) / n)
    elif method == 10:
        a = 1.0 / (1.0 + 0.3 * ((i - c) / (n * 0.5)) ** 6)
    else:
        a = np.abs(rng.normal(0, 1, n)) + 0.3

    a = np.maximum(a, eps)
    a = a / a.sum()
    z = np.log(a + eps)
    z = z - z.mean()
    z = np.clip(z, -12.0, 12.0)
    return z


def _optimize_n(n, parent, seed, time_limit, n_restarts=2, z_methods=None):
    rng = np.random.default_rng(seed)
    fsize = next_fast_len(2 * n)
    L = 2 * n - 1
    topk = min(L, max(80, min(300, n // 2)))

    best_a = None
    best_score = 2.0

    if z_methods is None:
        z_methods = [0, 1, 2, 5, 6, 7]

    for ri in range(n_restarts):
        if time.monotonic() > time_limit - 1.0:
            break

        z = _init_z(n, z_methods[ri % len(z_methods)], rng, parent)
        a = _softmax(z)

        m_adam = np.zeros(n, dtype=np.float64)
        v_adam = np.zeros(n, dtype=np.float64)
        t_adam = 0
        lr = 0.8
        tau_base = 0.06
        beta1, beta2 = 0.9, 0.999
        stuck = 0
        best_local = math.inf
        best_local_a = a.copy()
        no_improve = 0
        max_iter = 8000 if n <= 256 else (4000 if n <= 1024 else 2000)

        for it in range(max_iter):
            if time.monotonic() > time_limit - 1.0:
                break

            b = _fft_autocorr(a, fsize)
            bmax = float(b.max())
            score = 2.0 * n * bmax

            if score < best_local:
                best_local = score
                best_local_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 400:
                if no_improve > 1500:
                    break
                z = z + rng.normal(0, 0.6, n)
                z -= z.mean()
                z = np.clip(z, -12, 12)
                a = _softmax(z)
                m_adam[:] = 0
                v_adam[:] = 0
                t_adam = 0
                no_improve = 0
                continue

            tau = tau_base * max(bmax, 1e-10) * (1.0 + 0.5 * (1.0 - min(1.0, it / 800.0)))
            tau = max(tau, 1e-6 * max(bmax, 1e-10))

            grad, U = _grad_soft(a, b, tau, n, topk)

            if not np.isfinite(grad).all() or np.max(np.abs(grad)) < 1e-18:
                stuck += 1
                if stuck > 10:
                    break
                z = z + rng.normal(0, 0.8, n)
                z -= z.mean()
                a = _softmax(z)
                continue
            stuck = 0

            t_adam += 1
            m_adam = beta1 * m_adam + (1 - beta1) * grad
            v_adam = beta2 * v_adam + (1 - beta2) * (grad * grad)
            m_hat = m_adam / (1 - beta1 ** t_adam)
            v_hat = v_adam / (1 - beta2 ** t_adam)

            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            step[~np.isfinite(step)] = 0.0

            #/project out mean direction
            step -= step.mean()

            # Line search
            improved = False
            for f in (1.0, 0.5, 0.25, 0.125, 0.0625):
                z_new = z - f * step
                z_new -= z_new.mean()
                z_new = np.clip(z_new, -12, 12)
                a_new = _softmax(z_new)
                b_new = _fft_autocorr(a_new, fsize)
                sc_new = 2.0 * n * float(b_new.max())

                if sc_new < score * (1 - 1e-12):
                    z = z_new
                    a = a_new
                    b = b_new
                    bmax = float(b.max())
                    score = sc_new
                    improved = True
                    break

            if not improved:
                lr = max(0.01, lr * 0.85)
                if it % 200 == 0:
                    tau_base = max(0.01, tau_base * 0.8)

        if np.isfinite(best_local) and best_local < best_score:
            best_score = best_local
            best_a = best_local_a.copy()

    return best_a, best_score

    fsize = next_fast_len(2 * n)
    best = a.copy()
    bs = best_score
    rng = np.random.default_rng(123456789)
    sigma = 0.15
    stuck = 0

    while time.monotonic() < time_limit - 2.0:
        cand = best.copy()
        r = rng.random()

        if r < 0.30:

            z += rng.normal(0, sigma, n)
            cand = _softmax(z)
        elif r < 0.50:
            idx = rng.integers(0, n)
            w = max(1, min(n, int(rng.integers(1, max(2, n // 20)))))
            lo = max(0, idx - w // 2)
            hi = min(n, lo + w)
            if hi > lo:
                cand[lo:hi] *= np.exp(rng.normal(0, 0.3))
        elif r < 0.65:
            cand = 0.5 * (cand + cand[::-1])
        elif r < 0.75:
            alpha = rng.uniform(0.1, 0.5)
            cand = (1 - alpha) * cand + alpha * (1.0 / n)
        elif r < 0.85:
            p = rng.uniform(1.05, 3.0)
            mn, mx = cand.min(), cand.max()
            if mx > mn:
                cn = (cand - mn) / (mx - mn)
                cn = cn ** p
                mn2, mx2 = cn.min(), cn.max()
                if mx2 > mn2:
                    cn = (cn - mn2) / (mx2 - mn2)
                else:
                    cn = np.full(n, 1.0 / n)
                cand = cn
        else:
            k = int(rng.integers(1, max(2, n // 4)))
            kernel = np.ones(k) / k
            padded = np.pad(cand, k // 2, mode='edge')
            cand = np.convolve(padded, kernel, mode='valid')[:n]

        cand = np.clip(cand, 0, 1000)
        cs = cand.sum()
        if cs < 1e-10:
            continue
        cand = cand / cs

        b = _fft_autocorr(cand, fsize)
        fs = 2.0 * n * float(b.max())

        if not np.isfinite(fs) or fs >= bs - 1e-10:
            stuck += 1
            if stuck % 30 == 0:

            sigma = max(0.05, sigma * 0.9)
        else:
            stuck += 1
            if stuck > 200:
                sigma = max(0.05, sigma * 0.7)
