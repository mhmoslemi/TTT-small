    def normalize(a, target=1.0):
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        s = a.sum()
        if s < 1e-30:
            return np.full(n, target / n)
        return a * (target / s)

    # Score tracking
    best_a = normalize(base.copy(), 1.0)
    best_score = safe_eval(best_a)

    def local_search(a0, seconds, tau_start, tau_end, lr_start, lr_end):
        nonlocal best_a, best_score
        a = normalize(a0.copy(), 1.0)
        m = np.zeros(n)
        v = np.zeros(n)
        step = 0
        local_best = a.copy()
        local_best_score = safe_eval(a)
        start = time.monotonic()
        last_eval = start
        stall = 0

        while time.monotonic() - start < seconds and time.monotonic() < hard_deadline:
            # Compute convolution
            C = full_conv(a)
            Cmax = float(C.max())

            # Temperature schedule
            progress = min(1.0, (time.monotonic() - start) / max(seconds, 1.0))
            tau = tau_start + (tau_end - tau_start) * progress
            spread = Cmax - float(C.min())
            if spread > 1e-15 and spread < 0.01 * (Cmax + 1e-30):
                tau *= 3.0
            tau = max(1.0, min(tau, 2e5))

            # Soft-max weights
            z = tau * (C - Cmax)
            z = np.clip(z, -700.0, 0.0)
            w_un = np.exp(z)
            wsum = float(w_un.sum())
            if wsum < 1e-30 or not np.isfinite(wsum):
                w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
                M = Cmax
            else:
                w = w_un / wsum
                M = Cmax + np.log(wsum) / tau

            # Gradient of score = 2n*M/S^2 with S=1: grad = 2n*dM - 4n*M
            dM = compute_dM(a, w)
            g = 2.0 * n * dM - 4.0 * n * M

            # Adam
            b1, b2 = 0.9, 0.999
            step += 1
            m = b1 * m + (1 - b1) * g
            v = b2 * v + (1 - b2) * g * g
            mh = m / (1 - b1 ** step)
            vh = v / (1 - b2 ** step)
            lr = lr_start + (lr_end - lr_start) * progress
            a = a - lr * mh / (np.sqrt(vh) + 1e-8)

            # Project: clip & normalize
            a = np.maximum(a, 0.0)
            a = np.minimum(a, 1000.0)
            s = a.sum()
            if s < 1e-30:
                a = np.full(n, 1.0 / n)
            else:
                a = a / s

            # Periodic exact eval
            now = time.monotonic()
            if step % 400 == 0 and (now - last_eval) > 0.4:
                last_eval = now
                sc = safe_eval(a)
                if np.isfinite(sc) and sc < local_best_score:
                    local_best_score = sc
                    local_best = a.copy()
                    stall = 0
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                else:
                    stall += 1

            # Restart if badly stalled
            if stall > 20:
                epsilon = 0.05 * (1.0 + 0.1 * stall)
                a = a + epsilon * rng.standard_normal(n)
                a = normalize(a, 1.0)
                m[:] = 0.0
                v[:] = 0.0
                step = 0
                stall = 0

        # Final eval for this phase
        sc = safe_eval(a)
        if np.isfinite(sc) and sc < local_best_score:
            local_best = a.copy()
            local_best_score = sc
        if local_best_score < best_score:
            best_a = local_best.copy()
            best_score = local_best_score

        return a, safe_eval(a)

    # --- Build seed pool ---
    seeds = []
    seeds.append(normalize(base.copy(), 1.0))

    x = np.linspace(0, 1, n)
    seeds.append(normalize(np.exp(-((x - 0.5) ** 2) / 0.03), 1.0))
    seeds.append(normalize((0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 3 + 0.01, 1.0))
    s_mod = 0.5 + 0.3 * np.sin(4 * np.pi * x) + 0.2 * np.cos(8 * np.pi * x)
    seeds.append(normalize(np.maximum(s_mod, 0.05), 1.0))
    tl = np.exp(-((x - 0.25) ** 2) / 0.01) + np.exp(-((x - 0.75) ** 2) / 0.01) + 0.1
    seeds.append(normalize(tl, 1.0))
    seeds.append(normalize(np.minimum(x, 1 - x) ** 0.7 + 0.2, 1.0))

    noise = rng.standard_normal(n)
    fn = np.fft.rfft(noise, sizeC)
    mask = np.ones_like(fn)
    mask[n // 6:] = 0.0
    sn = np.abs(np.fft.irfft(fn * mask, sizeC)[:n])
    seeds.append(normalize(sn + 0.1, 1.0))
    seeds.append(normalize(0.3 + 0.7 * rng.random(n), 1.0))

    half_n = n // 2
    half = 0.4 + 0.4 * np.sin(np.pi * np.linspace(0, 1, half_n))
    if n % 2 == 0:
        sym = np.concatenate([half, half[::-1]])
    else:
        sym = np.concatenate([half, half[::-1][:-1]])
    seeds.append(normalize(np.maximum(sym, 0.05), 1.0))

    # Rank seeds
    scored = []
    for s in seeds:
        sc = safe_eval(s)
        if np.isfinite(sc):
            scored.append((sc, s))
    scored.sort(key=lambda t: t[0])
    if scored and scored[0][0] < best_score:
        best_score = scored[0][0]
        best_a = scored[0][1].copy()

    # --- Phase 1: Broad search on top seeds ---
    avail = max(5.0, hard_deadline - time.monotonic())
    top_k = min(6, len(scored))
    phase1_total = 0.42 * avail
    per_seed = phase1_total / max(1, top_k)

    for i in range(top_k):
        if time.monotonic() > hard_deadline - 80:
            break
        local_search(scored[i][1].copy(), per_seed,
                     tau_start=5.0, tau_end=40.0,
                     lr_start=0.012, lr_end=0.003)

    # --- Phase 2: Focused optimization ---
    if time.monotonic() < hard_deadline - 50:
        remain = hard_deadline - time.monotonic() - 50
        local_search(best_a.copy(), min(0.32 * avail, remain),
                     tau_start=30.0, tau_end=150.0,
                     lr_start=0.006, lr_end=0.001)

    # --- Phase 3: High-temperature polish (near-exact max) ---
    if time.monotonic() < hard_deadline - 20:
        remain = hard_deadline - time.monotonic() - 20
        local_search(best_a.copy(), min(0.15 * avail, remain),
                     tau_start=150.0, tau_end=800.0,
                     lr_start=0.003, lr_end=0.0005)

    # --- Phase 4: Final micro-polish ---
    if time.monotonic() < hard_deadline - 10:
        remain = hard_deadline - time.monotonic() - 10
        local_search(best_a.copy(), min(60.0, remain),
                     tau_start=500.0, tau_end=2000.0,
                     lr_start=0.001, lr_end=0.0002)

    # --- Final validation ---
    fs = safe_eval(best_a)
    if not np.isfinite(fs):
        best_a = normalize(base.copy(), 1.0)
        fs = safe_eval(best_a)

    out = [float(x) for x in best_a]
    if not out or any(not np.isfinite(x) for x in out):
        out = [1.0 / n] * n

    return out
