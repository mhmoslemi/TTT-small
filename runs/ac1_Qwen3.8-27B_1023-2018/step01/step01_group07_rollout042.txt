def _next_pow2(x):
    p = 1
    while p < x:
        p <<= 1
    return p

def _ideal_sequence(n):
    """Generate the ideal sequence: a_i proportional to binom(2i,i) / 4^i."""
    a = np.zeros(n)
    a[0] = 1.0
    for i in range(n - 1):
        a[i+1] = a[i] * (2*i + 1) / (2*(i+1))
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    else:
        a = a / s
    return a

def _convolve_fft(a, fft_len):
    Fx = np.fft.rfft(a, fft_len)
    return np.fft.irfft(Fx * Fx, fft_len)[:2*len(a)-1]

def _compute_b_and_w(a, fft_len, tau):
    b = _convolve_fft(a, fft_len)
    max_b = b.max()
    shifted = (b - max_b) / tau
    exp_b = np.exp(shifted)
    w = exp_b / exp_b.sum()
    return b, max_b, w

def _compute_grad(a, w, fft_len):
    n = len(a)
    w_R = np.zeros(fft_len)
    w_R[:2*n-1] = w[::-1]
    Fx = np.fft.rfft(a, fft_len)
    FW = np.fft.rfft(w_R, fft_len)
    conv = np.fft.irfft(Fx * FW, fft_len)
