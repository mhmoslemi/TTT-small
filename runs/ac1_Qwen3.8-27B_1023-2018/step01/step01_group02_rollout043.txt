def lp_polish(f, rng, max_iters=20, t_initial=0.3):
    n = f.size
    f = f.copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-12:
        return f
    f = f / s  # normalize to sum = 1
    
    M = max(np.convolve(f, f))
    
    for it in range(max_iters):
        # Build constraint matrix A for f*g <= M
        # (f*g)[k] = sum_i f[i] * g[k-i] for valid i, k
        # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
        A = np.zeros((2*n - 1, n))
        for k in range(2*n - 1):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    A[k, j] = f[idx]
        
        # LP: max sum(g) s.t. A@g <= M, g >= 0
        # linprog minimizes, so: min -sum(g) s.t. A@g <= M, g >= 0
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n)
        
        if res.status != 0:
            break
        
        g = res.x
        S = g.sum()
        if S <= 1.0 + 1e-6:
            break  # No improvement possible
        
        g = g / S  # normalize
        f = (1 - t) * f + t * g
        f = np.maximum(f, 0)
        s = f.sum()
        if s > 1e-12:
            f = f / s
        
        M_new = max(np.convolve(f, f))
        if M_new >= M - 1e-10:
            t *= 0.5
            if t < 0.01:
                break
        else:
            M = M_new
            t = min(t * 1.1, 0.5)
    
    return f

    mx = c.max()
    return 2.0 * n * mx / (s * s)

def _gradient_fft(f, p, n):
    """Compute gradient of sum_k p[k]*conv(f,f)[k] with respect to f.
    Result: 2 * cross_correlate(p, f) (truncated to length n)"""
    L = 1 << (2*n - 1 - 1).bit_length()  # next power of 2 >= 2n-1
    F = np.fft.rfft(f, L)
    P = np.fft.rfft(p, L)
    # conv(f, f_reversed) gives cross-correlation... 
    # Actually: np.convolve(f, f)[k] = sum_i f[i]*f[k-i]
    # d/d(f[j]) = 2*f[k-j] (approximately)
    # So grad[j] = 2*sum_k p[k]*f[k-j]
    # = 2*(p cross-correlate f)[j] where (p xc f)[j] = sum_k p[k]*f[k-j]
    # 
    # Using FFT: ifft(fft(p) * conj(fft(f))) gives corr[k] = sum_j p[j+k]*conj(f[j])
    # For real: corr[k] = sum_j p[j+k]*f[j]
    # That's not quite what we want.

    # Let m = k-j: = sum_m p[m+j]*f[m]
    # = sum_m f[m]*p[m+j]
    # This is the cross-correlation of f and p at lag j.
    #
    # In FFT terms: cross_correlation(f, p)[j] = ifft(conj(FFT(f)) * FFT(p))[j]
    # (with appropriate zero-padding to avoid circular effects)

    P = np.fft.rfft(p, L)
    cross = np.fft.irfft(np.conj(F) * P, L)
    # cross[j] = sum_m f[m]*p[m+j] (for j >= 0)
    # This is what we want: h[j] = cross[j]
    
    h = cross[:n]

                    polished = lp_polish(self.best, max_iters=10, time_limit=5.0)
                    ps = sc(polished)
                    if ps < self.best_score - 1e-10:
                        self.best_score = ps
                        self.best = polished
                        z = np.log(np.maximum(self.best, 1e-10))
                        m_adam = np.zeros(n)
                        v_adam = np.zeros(n)
                        t_adam = 0
                        no_improve = 0

                # Restart logic
                if no_improve > restart_every:
                    if rng.random() < 0.25:
                        z = np.log(np.maximum(self.best, 1e-10))[::-1]
                    else:
                        z = np.log(np.maximum(self.best, 1e-10)) + \
                            0.3 * rng.standard_normal(n)
                    m_adam = np.zeros(n)
                    v_adam = np.zeros(n)
                    t_adam = 0
                    lr = max(0.08, lr * 0.5)
                    no_improve = 0
                    restart_every = max(250, int(restart_every * 1.2))

    opt = Optimizer()

    # --- Phase 1: Initial optimization ---
    remaining = deadline - time.monotonic()
    phase1 = min(remaining * 0.35, 400.0)
    if phase1 > 5:
        z_init = np.log(np.maximum(parent, 1e-10))
        opt.run(z_init, phase1, beta_start=30.0, beta_end=100000.0, lr0=0.5,
                restart_every=400)

    # --- Phase 2: LP polish ---
    if time.monotonic() < deadline - 40:
        polished = lp_polish(opt.best, max_iters=40, time_limit=min(80.0, (deadline - time.monotonic()) * 0.15))
        if sc(polished) < opt.best_score - 1e-12:
            opt.best = polished
            opt.best_score = sc(opt.best)

    # --- Phase 3: Second optimization round ---
    remaining = deadline - time.monotonic() - 30
    if remaining > 20:
        z_init = np.log(np.maximum(opt.best, 1e-10))
        opt.run(z_init, remaining * 0.55, beta_start=200.0, beta_end=800000.0,
                lr0=0.3, restart_every=350)

    # --- Phase 4: LP polish again ---
    if time.monotonic() < deadline - 25:
        polished = lp_polish(opt.best, max_iters=30, time_limit=min(50.0, (deadline - time.monotonic()) * 0.2))
        if sc(polished) < opt.best_score - 1e-12:
            opt.best = polished
            opt.best_score = sc(opt.best)

    # --- Phase 5: Fine-tuning optimization ---
    remaining = deadline - time.monotonic() - 15
    if remaining > 10:
        z_init = np.log(np.maximum(opt.best, 1e-10))
        opt.run(z_init, remaining * 0.7, beta_start=2000.0, beta_end=2000000.0,
                lr0=0.15, restart_every=600)

    # --- Phase 6: Final LP polish ---
    if time.monotonic() < deadline - 10:
        polished = lp_polish(opt.best, max_iters=20, time_limit=min(20.0, (deadline - time.monotonic()) * 0.3))
        if sc(polished) < opt.best_score - 1e-12:
            opt.best = polished
            opt.best_score = sc(opt.best)

    # --- Short-length exploration (backup) ---
    # Try a few short sequences as fallback
    fallbacks = []
    for trial_len in (3, 7):
        for _ in range(5):
            trial = np.abs(rng.standard_normal(trial_len)) + 0.1
            trial = norm(trial[:n]) if trial_len == n else None
            if trial is not None:
                s = sc(trial)
                if s < opt.best_score:
                    fallbacks.append((trial, s))
    # Only useful if somehow better (unlikely but safe)

    # --- Final validation ---
    candidates = [opt.best, parent]
    if fallbacks:
        candidates.append(fallbacks[0][0])

    final_list = None
    final_score = np.inf
    for cand in candidates:
        try:
            lst = [float(x) for x in np.asarray(cand, dtype=np.float64)]
            if not lst or any(not np.isfinite(x) for x in lst):
                continue
            s = float(evaluate_sequence(lst))
            if np.isfinite(s) and s < final_score:
                final_score = s
                final_list = lst
        except Exception:
            continue

    if final_list is None:
        final_list = [float(x) for x in parent]
        evaluate_sequence(final_list)

    return final_list
