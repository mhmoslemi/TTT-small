Let me also think about the Frank-Wolfe approach more carefully:
- At each step, the current max is at lag k*.
- The "linear minimization oracle" for the simplex {a >= 0, sum(a) = 1}:
  We want to find the vertex e_i that minimizes <grad, e_i> = grad_i.
  So we pick i* = argmin_i grad_i.
  Then move: a_new = (1-t)*a + t*e_{i*}.
- This is simple and requires no Feasibility issues.

- Compute p = softmax(b/tau)
- grad_a = 2 * correlate(a, p) (the soft gradient)
- i* = argmin_i grad_a_i
- a_new = (1-t)*a + t*e_{i*}

import time
import math
from scipy.fft import rfft, irfft, next_fast_len

        h = globals()["height_sequence_1"]

        pass

def _convolve(a, fsize=None):
    """Full convolution of a with itself using FFT."""
    n = len(a)
    if fsize is None:
        fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    c[np.abs(c) < 1e-15] = 0.0
    return c

def _exact_score(a):
    """Compute score matching the evaluator (scale-invariant)."""
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    b = _convolve(a, fsize)
    mb = float(b.max())
    if not math.isfinite(mb):
        return math.inf
    return 2.0 * n * mb / (s * s)

def _soft_obj_grad(a, b, tau):
    """
    Compute the soft objective U and its gradient w.r.t. a.
    U = tau * log(sum exp(b_k / tau)) ≈ max(b) when tau is small.
    grad_a[i] = 2 * sum_j a[j] * p[i+j]
    """
    n = len(a)
    bmax = float(b.max())
    x = (b - bmax) / tau
    ex = np.exp(x)

            j_hi = k - lo

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def _interpolate_parent(parent, n):
    """Interpolate parent to length n."""
    m = len(parent)
    if m == n:
        return parent.copy()
    x_old = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, parent)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a

def _struct_init(n, kind, rng):
    """Generate structured initializations."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-10
    if kind == 0:  # Cosine
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * i / n)
    elif kind == 1:  # Gaussian center
        sig = rng.uniform(0.15, 0.4) * n
        a = np.exp(-((i - n//2)**2) / (2*sig*sig))
    elif kind == 2:  # Exponential decay
        a = np.exp(-rng.uniform(0.5, 3.0) * i / n)
    elif kind == 3:  # Two lobes
        c = n // 2
        sig = rng.uniform(0.1, 0.25) * n
        a = np.exp(-((i - c//2)**2) / (2*sig*sig)) + np.exp(-((i - (c+c//2))**2) / (2*sig*sig))
    elif kind == 4:  # Random smooth
        a = rng.uniform(0.5, 1.0, n)
        # Smooth it
        kernel = np.exp(-0.5 * (np.arange(-5, 6) / 3.0)**2)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
    elif kind == 5:  # Pure random positive
        a = rng.uniform(0.1, 1.0, n)
    else:
        a = np.ones(n)
    a = np.maximum(a, eps)
    a /= a.sum()
    return a

def _gradient_descent(a_init, n, fsize, stop_time, seed, max_tau_ratio=0.02, min_tau_ratio=1e-6):
    """
    Soft gradient descent in softmax space.
    a = softmax(z) so sum(a) = 1, a >= 0.
    Minimize U(a) = tau * log(sum exp(c_k / tau)) where c = convolve(a,a).
    """
    rng = np.random.default_rng(seed)
    
    # Start from a_init, convert to z-space
    a = np.array(a_init, dtype=np.float64)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    z = np.log(a)  # This gives a specific point in z-space
    # Center z (doesn't change a)
    z = z - z.mean()
    
    b = _convolve(a, fsize)
    bmax_init = float(b.max())
    
    best_a = a.copy()
    best_score = _exact_score(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    beta1, beta2 = 0.85, 0.999
    
    # Learning rate
    base_lr = 0.3 / math.sqrt(n)
    lr = base_lr
    stuck = 0
    last_improve = time.monotonic()
    
    # Temperature schedule: start high, anneal
    tau0 = 0.1 * max(bmax_init, 1e-10)
    tau = tau0
    
    while time.monotonic() < stop_time - 0.1:
        # Compute current b and score
        b = _convolve(a, fsize)
        bmax = float(b.max())
        if not math.isfinite(bmax) or bmax <= 0:

            best_a = a.copy()

        gnorm = float(np.linalg.norm(grad_a))

            if stuck > 20:

            z -= z.mean()

            m = np.zeros(n)
            v = np.zeros(n)
            t_count = 0

        S = float(a.dot(grad_a))

            a = _softmax(z)

            v = np.zeros(n)

            continue

        v = beta2 * v + (1 - beta2) * (grad_z * grad_z)
        m_hat = m / (1 - beta1**t_count)
        v_hat = v / (1 - beta2**t_count)
        direction = m_hat / (np.sqrt(v_hat) + 1e-8)

            cand_a = _softmax(cand_z)
            cand_b = _convolve(cand_a, fsize)
            cand_bmax = float(cand_b.max())
            cand_score = 2.0 * n * cand_bmax

                z = cand_z
                a = cand_a
                b = cand_b
                score = cand_score
                improved = True
                break

        else:

            lr = max(1e-6, lr * 0.8)

                v = np.zeros(n)

def _time_alloc(n, total_remaining):
    """Allocate time for optimizing a sequence of length n."""
    # FFT cost is O(n log n) per evaluation, gradient is O(n * active)
    # More time for larger n since they're harder but potentially better
    if n <= 32:
        return min(8.0, total_remaining)
    elif n <= 64:
        return min(15.0, total_remaining)
    elif n <= 128:
        return min(25.0, total_remaining)
    elif n <= 256:
        return min(40.0, total_remaining)
    elif n <= 512:
        return min(60.0, total_remaining)
    else:
        return min(90.0, total_remaining)

def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:

def _convolve(a, fsize):
    n = len(a)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[:2 * n - 1]
    c[np.abs(c) < 1e-15] = 0.0
    return c


def _exact_score(a):
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    n = len(a)
    fsize = next_fast_len(2 * n - 1)
    b = _convolve(a, fsize)
    mb = float(b.max())
    if not math.isfinite(mb):
        return math.inf
    return 2.0 * n * mb / (s * s)


def _soft_obj_grad(a, b, tau):
    """Compute U = tau*log(sum exp(b/tau)) and grad_a = 2*correlate(a, p)."""
    n = len(a)
    bmax = float(b.max())

    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-300 or not math.isfinite(s):
        k = int(np.argmax(b))
        U = bmax
        grad_a = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            j_lo = k - hi
            j_hi = k - lo
            grad_a[lo:hi + 1] = 2.0 * a[j_lo:j_hi + 1][::-1]
        return grad_a, U

    p = ex / s
    U = bmax + tau * math.log(s)

    pmx = float(p.max())
    thresh = max(1e-14, pmx * 1e-8)
    active = np.where(p > thresh)[0]

    if len(active) > max_active:

        active = active[top]

    grad_a = np.zeros(n)

        pk = float(p[int(k)])
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))

        j_lo = int(k) - hi
        j_hi = int(k) - lo

def _softmax(z):

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def _fw_step(a, n, fsize):
    """One Frank-Wolfe step: find argmin grad, move toward that vertex."""
    b = _convolve(a, fsize)
    k_star = int(np.argmax(b))
    # grad_a[i] = 2*a[k_star - i] for valid i
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if hi < lo:
        return a
    j_lo = k_star - hi
    j_hi = k_star - lo
    grad_a = np.zeros(n)
    grad_a[lo:hi + 1] = 2.0 * a[j_lo:j_hi + 1][::-1]

    # Frank-Wolfe: move toward e_{argmin grad_a}
    i_star = int(np.argmin(grad_a))
    return a, i_star, k_star


def _frank_wolfe(a_init, n, fsize, stop_time, seed):
    rng = np.random.default_rng(seed)
    a = np.array(a_init, dtype=np.float64)
    a = np.maximum(a, 1e-12)
    a /= a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * float(_convolve(a, fsize).max())

    t = 0
    gamma = 0.15

    while time.monotonic() < stop_time - 0.05:
        a, i_star, k_star = _fw_step(a, n, fsize)
        t += 1
        step = 2.0 * gamma / (t + gamma)
        e = np.zeros(n)
        e[i_star] = 1.0
        a = (1 - step) * a + step * e
        a = np.maximum(a, 1e-14)
        a /= a.sum()

        b = _convolve(a, fsize)
        score = 2.0 * n * float(b.max())
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

    return best_a, best_score

            cand = 0.5 * (a_cur + a_cur[::-1])
        elif r < 0.80:
            w = rng.uniform(0.03, 0.2)
            cand = (1 - w) * a_cur + w / n
        elif r < 0.90:
            idx = rng.choice(n, size=min(2, n), replace=False)
            cand = a_cur.copy()
            for i in idx:
                cand[i] *= float(rng.uniform(0.5, 2.0))
        else:
            mask = rng.random(n) < 0.1
            cand = np.where(mask, rng.uniform(0.05, 1.0, n), a_cur)

        cand = np.clip(cand, 0.0, 1000.0)
        cs = cand.sum()
        if cs < 0.01:
            continue
        cand = cand / cs

        b = _convolve(cand, fsize)
        fast = 2.0 * n * float(b.max())
        if fast >= best_score * 1.02:
            stuck += 1
            if stuck % 40 == 0:
                sigma = min(1.5, sigma * 1.6)
            continue

        es = float(evaluate_sequence([float(x) for x in cand]))
        if math.isfinite(es) and es < best_score - 1e-11:
            best_a = cand.copy()
            best_score = es
            stuck = 0
            sigma = max(0.04, sigma * 0.88)
        else:
            stuck += 1
            if stuck > 100:
                sigma = 0.3
                stuck = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 6.0)

    parent = _get_parent()
    p_parent = np.clip(parent, 0.0, 1000.0).astype(np.float64)

    # Evaluate parent
    try:
        p_score = float(evaluate_sequence([float(x) for x in p_parent]))
    except Exception:
        p_score = math.inf
    if not math.isfinite(p_score):
        p_score = _exact_score(p_parent)

    best_seq = p_parent.copy()
    best_score = p_score

    # Determine lengths to try
    try:
        pl = len(p_parent)
    except Exception:
        pl = 192

    lengths = sorted(set([
        pl,
        max(16, pl // 2), max(16, pl * 3 // 2),
        128, 160, 192, 224, 256, 320, 384, 512, 640, 768, 1024
    ]))
    lengths = [L for L in lengths if 16 <= L <= 2048]

    # Sort by proximity to parent length first, then by size
    lengths.sort(key=lambda L: (abs(L - pl), L))

    n_attempts_per_len = 3

    for n in lengths:
        if time.monotonic() > deadline - 35.0:
            break
        if best_score < 1.49:
            break

        rem = deadline - time.monotonic() - 20.0
        if rem < 10.0:
            break

        # Allocate time per attempt
        if n <= 64:
            t_per_attempt = min(12.0, rem / max(1, n_attempts_per_len))
        elif n <= 256:
            t_per_attempt = min(25.0, rem / max(1, n_attempts_per_len))
        elif n <= 512:
            t_per_attempt = min(40.0, rem / max(1, n_attempts_per_len))
        else:
            t_per_attempt = min(60.0, rem / max(1, n_attempts_per_len))

        if t_per_attempt < 3.0:
            n_attempts_per_len = max(1, int(rem // 5))
            t_per_attempt = max(3.0, rem / max(1, n_attempts_per_len))

        fsize = next_fast_len(2 * n - 1)

        for att in range(n_attempts_per_len):
            if time.monotonic() > deadline - 25.0:
                break

            seed_i = int(seed) + 100000 + n * 7 + att * 137

            # Generate initialization
            if att == 0:
                a_init = _interpolate_parent(p_parent, n)
            elif att == 1:
                rng = np.random.default_rng(seed_i)
                a_init = _struct_init(n, att, rng)
                if abs(n - pl) < n // 3:
                    a_init = 0.5 * a_init + 0.5 * _interpolate_parent(p_parent, n)
                    a_init /= a_init.sum()
            else:
                rng = np.random.default_rng(seed_i + 7)
                a_init = _struct_init(n, att + 2, rng)

            a_init = np.clip(a_init, 0.0, 1000.0)
            s_init = a_init.sum()
            if s_init > 0.01:
                a_init = a_init / s_init

            # Run gradient descent
            stop_i = time.monotonic() + t_per_attempt * 0.65
            try:
                a_opt, sc_opt = _gradient_descent(a_init, n, fsize, stop_i, seed_i)
            except Exception:
                a_opt = a_init
                sc_opt = _exact_score(a_init)

            # Validate
            if a_opt is not None and len(a_opt) == n:
                a_opt = np.clip(a_opt, 0.0, 1000.0)
                try:
                    es = float(evaluate_sequence([float(x) for x in a_opt]))
                except Exception:
                    es = math.inf
                if math.isfinite(es) and es < best_score - 1e-12:
                    best_seq = a_opt.copy()
                    best_score = es

            # Additional Frank-Wolfe refinement if time permits
            if time.monotonic() < stop_i + t_per_attempt * 0.3 - 0.2:
                fw_stop = time.monotonic() + t_per_attempt * 0.3
                try:
                    a_fw, sc_fw = _frank_wolfe(a_opt, n, fsize, fw_stop, seed_i + 9999)
                except Exception:
                    a_fw = None
                if a_fw is not None and len(a_fw) == n:
                    a_fw = np.clip(a_fw, 0.0, 1000.0)
                    try:
                        es_fw = float(evaluate_sequence([float(x) for x in a_fw]))
                    except Exception:
                        es_fw = math.inf
                    if math.isfinite(es_fw) and es_fw < best_score - 1e-12:
                        best_seq = a_fw.copy()
                        best_score = es_fw

    # Polish phase
    if time.monotonic() < deadline - 8.0:
        polish_time_end = min(deadline - 3.0, time.monotonic() +
                             max(15.0, (deadline - time.monotonic()) * 0.5))
        try:
            best_seq, best_score = _local_search(best_seq, polish_time_end, int(seed) + 987654321)
        except Exception:
            pass

    # Final validation
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.size == 0 or not np.isfinite(best_seq).all():
        best_seq = np.ones(64) / 64.0

    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        final_score = math.inf

    if not math.isfinite(final_score):
        fb = np.ones(64) / 64.0
        try:
            fs = float(evaluate_sequence([float(x) for x in fb]))
        except Exception:
            return [1.0 / 64.0] * 64
        if math.isfinite(fs):
            return [float(x) for x in fb]
        return [1.0 / 64.0] * 64

    return [float(x) for x in best_seq]
