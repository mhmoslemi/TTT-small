class _ConvOp:
    def __init__(self, n: int):
        self.n = int(n)
        self.K = 2 * self.n - 1
        rows = []
        cols = []
        offs = []
        n = self.n
        K = self.K
        for k in range(K):
            lo = max(0, k - n + 1)

def _solve_lp_cvxpy(c, A, b, Aeq, b_eq, bounds):

        import cvxpy as cp
    except Exception:
        return None
    try:
        x = cp.Variable(int(c.size), nonneg=True)
        cons = []

        cons.append(A_arr @ x <= np.asarray(b, dtype=np.float64))

        if mode == "maxsum":

def _random_direction(n, rng):
    n = int(n)
    z = rng.normal(0.0, 1.0, n)
    w = max(1, n // 32)
    if w > 1:
        kernel = np.ones(w, dtype=np.float64) / float(w)
        z = np.convolve(z, kernel, mode="same")
    y = np.exp(np.clip(z, -4.0, 4.0))
    y = y + 1e-9
    s = float(y.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / n
    y = y / s
    return y


def _line_search(a, c_a, y, M):

    if not (np.isfinite(c_y).all() and np.isfinite(c_xy).all() and np.isfinite(c_a).all()):
        return None

    ts = np.array([1.0, 0.92, 0.85, 0.78, 0.70, 0.62, 0.55, 0.48, 0.42,
                   0.36, 0.30, 0.25, 0.20, 0.15, 0.10, 0.07, 0.04, 0.02, 0.01], dtype=np.float64)
    alpha = (1.0 - ts) * (1.0 - ts)
    beta = 2.0 * ts * (1.0 - ts)
    gamma = ts * ts

    vals = (alpha[:, None] * c_a[None, :] +
            beta[:, None] * c_xy[None, :] +
            gamma[:, None] * c_y[None, :])
    mts = vals.max(axis=1)
    if not np.all(np.isfinite(mts)):
        return None
    i0 = int(np.argmin(mts))
    best_t = float(ts[i0])
    best_m = float(mts[i0])

    if 0 < i0 < len(ts) - 1:
        t0 = float(ts[i0 - 1])
        t1 = float(ts[i0 + 1])
        rts = np.linspace(t0, t1, 9)
        ralpha = (1.0 - rts) * (1.0 - rts)
        rbeta = 2.0 * rts * (1.0 - rts)
        rgamma = rts * rts
        rvals = (ralpha[:, None] * c_a[None, :] +
                 rbeta[:, None] * c_xy[None, :] +
                 rgamma[:, None] * c_y[None, :])
        rmts = rvals.max(axis=1)
        if np.all(np.isfinite(rmts)):
            j = int(np.argmin(rmts))
            if float(rmts[j]) < best_m:
                best_m = float(rmts[j])
                best_t = float(rts[j])

    if best_m >= M - 1e-13:
        return None

    new_a = (1.0 - best_t) * a + best_t * y
    new_a = np.nan_to_num(new_a, nan=0.0, posinf=0.0, neginf=0.0)
    new_a = np.clip(new_a, 0.0, None)
    s = float(new_a.sum())
    if not np.isfinite(s) or s <= 1e-14:
        return None
    new_a = new_a / s

    try:
        c_new = np.convolve(new_a, new_a)
    except Exception:
        c_new = None
    if c_new is not None and np.isfinite(c_new).all():
        M_new = float(c_new.max())
        if np.isfinite(M_new) and M_new < M - 1e-13:
            return new_a, c_new, M_new

    # If direct recomputation is numerically inconsistent, fall back to the bilinear estimate.
    if best_m < M - 5e-13:
        c_est = ((1.0 - best_t) * (1.0 - best_t)) * c_a
        c_est += (2.0 * best_t * (1.0 - best_t)) * c_xy
        c_est += (best_t * best_t) * c_y
        c_est = np.maximum(c_est, 0.0)
        if np.isfinite(c_est).all():
            return new_a, c_est, float(best_m)

    return None


def _lp_descent(a0, op: _ConvOp, stop_time: float, seed: int):
    n = op.n
    a = _normalise(a0, floor=1e-8)
    try:
        c_a = np.convolve(a, a)
        M = float(c_a.max())
    except Exception:
        return a, math.inf
    if not np.isfinite(M) or M <= 0.0:
        M = 1e-3
        c_a = np.full(2 * n - 1, M, dtype=np.float64)

    best_a = a.copy()
    best_M = float(M)
    rng = np.random.default_rng(int(seed))
    last_improve = time.monotonic()
    stall = 0
    perturbations = 0
    spec_idx = 0
    specs = ["varcap", "maxsum", "fix"]
    max_stall_s = 18.0 if n <= 256 else (30.0 if n <= 512 else 45.0)

    while time.monotonic() < stop_time:
        if time.monotonic() - last_improve > max_stall_s:
            break
        improved = False

        for off in range(len(specs)):
            if time.monotonic() > stop_time:
                break
            kind = specs[(spec_idx + off) % len(specs)]
            try:
                y = _direction(op, a, best_M if M is None else M, kind)
            except Exception:
                y = None
            if y is None:
                continue
            res = _line_search(a, c_a, y, M)
            if res is None:
                continue
            a, c_a, M = res
            if M < best_M - 1e-13:
                best_a = a.copy()
                best_M = float(M)
            last_improve = time.monotonic()
            stall = 0
            spec_idx = 0
            improved = True
            break

        if not improved:
            spec_idx = (spec_idx + 1) % len(specs)
            for _ in range(2):
                if time.monotonic() > stop_time:
                    break
                try:
                    y = _random_direction(n, rng)
                    res = _line_search(a, c_a, y, M)
                    if res is None:
                        continue
                    a, c_a, M = res
                    if M < best_M - 1e-13:
                        best_a = a.copy()
                        best_M = float(M)
                    last_improve = time.monotonic()
                    stall = 0
                    spec_idx = 0
                    improved = True
                    break
                except Exception:
                    pass

        if not improved:
            stall += 1
            if stall >= 6:
                perturbations += 1
                if perturbations > 14:
                    break
                base = best_a if (M > best_M or best_M <= 0.0) else a
                try:
                    noise = np.exp(rng.normal(0.0, 0.35, n))
                    a = _normalise(base * noise, floor=1e-8)
                    c_a = np.convolve(a, a)
                    M = float(c_a.max())
                except Exception:
                    a = base.copy()
                    c_a = np.convolve(a, a)
                    M = float(c_a.max())
                if not np.isfinite(M):
                    a = best_a.copy()
                    c_a = np.convolve(a, a)
                    M = float(c_a.max())
                stall = 0
                spec_idx = 0

    return best_a, best_M


def _resample_parent(parent, n):
    n = int(n)
    p = np.asarray(parent, dtype=np.float64).ravel()
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    p = np.clip(p, 0.0, 1000.0)
    P = p.size
    if P == 0:
        return np.ones(n, dtype=np.float64) / n
    if P == 1:
        return np.full(n, float(p[0]), dtype=np.float64)
    if P == n:
        a = p.copy()
    elif P < n:
        x_old = np.linspace(0.0, 1.0, P)
        x_new = np.linspace(0.0, 1.0, n)
        a = np.interp(x_new, x_old, p)
    else:
        starts = np.floor(np.linspace(0, P, n, endpoint=False)).astype(int)
        stops = np.floor(np.linspace(0, P, n, endpoint=True)).astype(int)
        a = np.empty(n, dtype=np.float64)
        for i, (s, e) in enumerate(zip(starts, stops)):
            s = int(min(max(0, s), P - 1))
            e = int(min(max(s + 1, e), P))
            if e <= s:
                a[i] = float(p[s])
            else:
                a[i] = float(np.mean(p[s:e]))
    return _normalise(a, floor=0.0)


def _make_inits(n, parent, rng, max_inits=6):
    n = int(n)
    base = _resample_parent(parent, n)
    inits = [base]

    sym = 0.5 * (base + base[::-1])
    if float(np.linalg.norm(base - sym)) > 1e-8:
        inits.append(_normalise(sym, floor=0.0))

    inits.append(_random_direction(n, rng))

    try:
        i = np.arange(n, dtype=np.float64)
        vals = np.ones(n, dtype=np.float64)
        for _ in range(3):
            f = int(rng.integers(1, max(2, n // 4) + 1))
            ph = float(rng.uniform(0.0, 2.0 * math.pi))
            amp = float(rng.uniform(0.25, 0.85))
            vals += amp * np.sin(2.0 * math.pi * f * i / float(n) + ph)
        vals = np.maximum(vals, 1e-3)
        vals = vals + rng.normal(0.0, 0.05, n)
        inits.append(_normalise(vals, floor=0.0))
    except Exception:
        pass

    try:
        z = rng.normal(0.0, 1.0, n)
        w = max(1, n // 8)
        z = np.convolve(z + 0.1, np.ones(w, dtype=np.float64) / float(w), mode="same")
        y = np.exp(0.8 * np.clip(z, -3.0, 3.0))
        inits.append(_normalise(y, floor=0.0))
    except Exception:
        pass

    try:
        nb = max(2, n // 12)
        g = np.exp(rng.normal(0.0, 0.6, nb))
        x_old = np.linspace(0.0, 1.0, nb)
        x_new = np.linspace(0.0, 1.0, n)
        b = np.interp(x_new, x_old, g)
        inits.append(_normalise(b, floor=0.0))
    except Exception:
        pass

    # Optional asymmetric pair, only if not too many inits already.
    if len(inits) < max_inits:
        try:
            a = base.copy()
            a = 0.72 * a + 0.28 * rng.uniform(0.2, 1.8, n)
            inits.append(_normalise(a, floor=0.0))
        except Exception:
            pass

    return inits[:max_inits]


def _search_length(n, stop_time, parent, seed):
    n = int(n)
    if n < 4 or time.monotonic() > stop_time:
        return None, math.inf
    try:
        op = _ConvOp(n)
    except Exception:
        return None, math.inf

    rng = np.random.default_rng(int(seed) + 131071 * n)
    inits = _make_inits(n, parent, rng)
    best_a = None
    best_M = math.inf

    for idx, init in enumerate(inits):
        if time.monotonic() > stop_time - 1.0:
            break
        try:
            a0 = _normalise(init, floor=1e-8)
            rem = max(0.0, stop_time - time.monotonic())
            remaining = max(1, len(inits) - idx)
            q = rem / float(remaining)
            cap_q = 70.0 if n <= 256 else (100.0 if n <= 512 else 150.0)
            q = min(q, cap_q)
            if idx == 0:
                q = min(q + 10.0, rem)
            if q < 7.0:
                continue
            init_stop = min(stop_time - 1.0, time.monotonic() + q)
            a, M = _lp_descent(a0, op, init_stop, int(seed) + 1000 * idx + n)
            if a is not None and np.isfinite(M) and M < best_M:
                best_a = np.asarray(a, dtype=np.float64).copy()
                best_M = float(M)
        except Exception:
            continue

    return best_a, best_M


def _softmax_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    n = z.size
    if n == 0:
        return np.array([1.0])
    z = np.nan_to_num(z, nan=0.0, posinf=0.0, neginf=0.0)
    z = z - float(np.max(z))
    e = np.exp(np.clip(z, -50.0, 50.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / n
    return e / s


def _soft_lse(b, tau):
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return 0.0, None
    m = float(np.max(b))
    if not np.isfinite(m):
        m = 0.0
    tau = float(max(1e-15, tau))
    x = (b - m) / tau
    ex = np.exp(np.clip(x, -50.0, 50.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros_like(b, dtype=np.float64)
        idx = np.where(np.isclose(b, m, rtol=1e-12, atol=1e-15))[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(b))])
        p[idx] = 1.0 / float(idx.size)
        return float(m), p
    p = ex / s
    U = m + tau * math.log(s)
    if not np.isfinite(U):
        U = m
        p = np.zeros_like(b, dtype=np.float64)
        idx = np.where(np.isclose(b, m, rtol=1e-12, atol=1e-15))[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(b))])
        p[idx] = 1.0 / float(idx.size)
    return float(U), p


def _grad_z(a, b, tau, n):
    U, p = _soft_lse(b, tau)
    if p is None or p.size == 0:
        return None, U
    L = p.size
    if L <= 512:
        idx = np.where(p > 0.0)[0]
    else:
        maxp = float(p.max())
        thresh = max(1e-13, maxp * 1e-7)
        idx = np.where(p >= thresh)[0]
        max_top = min(350, max(80, L // 8))
        if idx.size > max_top:
            idx = np.argpartition(p, -max_top)[-max_top:]
    if idx.size == 0:
        idx = np.array([int(np.argmax(p))])

    w = np.asarray(p[idx], dtype=np.float64)
    ws = float(w.sum())
    if not np.isfinite(ws) or ws <= 0.0:
        return None, U
    w = w / ws

    hg = np.zeros(n, dtype=np.float64)
    S = 0.0
    for k, wk in zip(idx, w):
        wk = float(wk)
        if wk <= 0.0:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi < lo:
            continue
        seg = a[lo:hi + 1]
        orig = a[k - hi:k - lo + 1][::-1]
        factor = 2.0 * wk
        hg[lo:hi + 1] += factor * orig
        S += factor * float(np.dot(seg, orig))

    if not np.isfinite(S):
        S = 0.0
    h = a * (hg - S)
    h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)
    return h, float(U)


def _fast_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return math.inf
    n = a.size
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return math.inf
    c = np.convolve(a, a)
    m = float(c.max()) if c.size else 0.0
    if not np.isfinite(m):
        return math.inf
    return float(2.0 * n * m / (s * s))


def _soft_polish(a, stop_time, seed):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size < 2 or time.monotonic() >= stop_time:
        return a, _fast_score(a)
    a = _normalise(a, floor=1e-8)
    n = a.size

    try:
        b = np.convolve(a, a)
        cur = 2.0 * n * float(b.max())
    except Exception:
        cur = _fast_score(a)
    best_a = a.copy()
    best_cur = cur

    z = np.log(a + 1e-15)
    z = z - float(z.mean())
    z = np.clip(z, -12.0, 12.0)

    rng = np.random.default_rng(int(seed) + 987654321)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    tcount = 0
    lr = 0.25 if n <= 256 else (0.18 if n <= 512 else 0.12)
    stuck = 0
    last_improve = time.monotonic()

    while time.monotonic() < stop_time - 0.5:
        if time.monotonic() - last_improve > 50.0:
            break

        try:
            b = np.convolve(a, a)
            bmax = float(b.max())
            cur = 2.0 * n * bmax
        except Exception:
            break
        if not np.isfinite(cur) or cur <= 0.0:
            break

        if cur < best_cur - 1e-12:
            best_a = a.copy()
            best_cur = float(cur)
            last_improve = time.monotonic()

        tau = max(1e-12, 0.04 * max(bmax, 1e-30))
        h, U = _grad_z(a, b, tau, n)
        if h is None:
            break

        hmax = float(np.max(np.abs(h))) if h.size else 0.0
        if not np.isfinite(hmax) or hmax < 1e-15:
            stuck += 1
            if stuck > 60:
                break
            z = z + rng.normal(0.0, 0.35, n)
            z = z - z.mean()
            z = np.clip(z, -12.0, 12.0)
            a = _softmax_z(z)
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            tcount = 0
            continue

        tcount += 1
        b1, b2 = 0.90, 0.999
        m = b1 * m + (1.0 - b1) * h
        v = b2 * v + (1.0 - b2) * (h * h)
        mhat = m / (1.0 - b1 ** tcount)
        vhat = v / (1.0 - b2 ** tcount)
        direction = mhat / (np.sqrt(vhat) + 1e-8)
        direction = np.nan_to_num(direction, nan=0.0, posinf=0.0, neginf=0.0)
        dn = float(np.max(np.abs(direction))) if direction.size else 0.0
        if not np.isfinite(dn) or dn <= 1e-15:
            stuck += 1
            continue
        direction = direction / dn

        improved = False
        for fac in (1.0, 0.5, 0.25, 0.125, 0.065):
            cand_z = z - lr * fac * direction
            cand_z = cand_z - float(np.mean(cand_z))
            cand_z = np.clip(cand_z, -12.0, 12.0)
            cand_z = np.nan_to_num(cand_z, nan=0.0, posinf=0.0, neginf=0.0)
            cand = _softmax_z(cand_z)
            try:
                cb = np.convolve(cand, cand)
                cs = 2.0 * n * float(cb.max())
            except Exception:
                continue
            if np.isfinite(cs) and cs < cur * (1.0 - 1e-10):
                z = cand_z
                a = cand
                cur = float(cs)
                improved = True
                last_improve = time.monotonic()
                break

        if improved:
            stuck = 0
            lr = min(0.55, lr * 1.02)
        else:
            stuck += 1
            lr = max(1e-4, lr * 0.86)
            if stuck % 20 == 0:
                z = z + rng.normal(0.0, 0.4, n)
                z = z - z.mean()
                z = np.clip(z, -12.0, 12.0)
                a = _softmax_z(z)
                m = np.zeros(n, dtype=np.float64)
                v = np.zeros(n, dtype=np.float64)
                tcount = 0
                lr = 0.25 if n <= 256 else (0.18 if n <= 512 else 0.12)
            if stuck > 140:
                break

    return best_a, float(best_cur)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 20.0:
        parent = _get_parent()
        fb = _normalise(parent, floor=0.0)
        if _safe_evaluate(fb) == math.inf:
            return [1.0]
        return [float(x) for x in fb]

    hard_deadline = t0 + max(0.0, budget)
    work_deadline = hard_deadline - 12.0

    parent = _get_parent()
    fallback = _normalise(parent, floor=0.0)
    fallback_score = _safe_evaluate(fallback)
    if not np.isfinite(fallback_score):
        fallback = np.ones(192, dtype=np.float64) / 192.0
        fallback_score = _safe_evaluate(fallback)
        if not np.isfinite(fallback_score):
            return [1.0]

    best_seq = np.asarray(fallback, dtype=np.float64).copy()
    best_score = float(fallback_score)

    max_lp_n = 1024 if csr_matrix is not None else 640

    def _try_update(a, est_score):
        nonlocal best_seq, best_score
        try:
            a = np.asarray(a, dtype=np.float64).ravel()
        except Exception:
            return
        if a.size == 0 or not np.isfinite(a).all():
            return
        if not np.isfinite(est_score) or est_score > (best_score + 1e-7):
            return
        ex = _safe_evaluate(a)
        if np.isfinite(ex) and ex < best_score - 1e-12:
            best_seq = a.copy()
            best_score = float(ex)

    plen = int(parent.size) if parent.size > 0 else 192
    base_len = int(np.clip(plen, 16, max_lp_n))
    if plen <= 4:
        base_len = 192

    length_order = [base_len]
    for L in (256, 192, 128, 512, 384, 64, 640, 768, 1024, 96):
        if int(np.clip(L, 16, max_lp_n)) not in length_order:
            length_order.append(int(np.clip(L, 16, max_lp_n)))

    weights = {}
    for L in length_order:
        if L == base_len:
            w = 1.25
        elif L in (192, 256):
            w = 0.90
        elif L in (128, 384, 512):
            w = 0.55
        elif L in (640, 768):
            w = 0.30
        else:
            w = 0.22
        weights[L] = weights.get(L, 0.0) + w

    sumw = float(sum(weights.values()))
    for L in length_order:
        if time.monotonic() > work_deadline - 35.0:
            break
        if best_score < 1.49 and L != base_len:
            # Once comfortably below the current target, leave most remaining time for polishing the best sequence.
            if length_order.index(L) > 0:
                continue
        rem = max(0.0, work_deadline - time.monotonic() - 25.0)
        alloc = rem * (weights[L] / sumw)
        if L > 768:
            alloc = min(alloc, 95.0)
        elif L > 512:
            alloc = min(alloc, 150.0)
        else:
            alloc = min(alloc, 420.0)
        if alloc < 8.0:
            continue
        stop = min(work_deadline - 25.0, time.monotonic() + alloc)
        try:
            a, M = _search_length(L, stop, parent, int(seed) + 700000 + 17 * L)
            if a is not None and np.isfinite(M):
                _try_update(a, 2.0 * L * float(M))
        except Exception:
            continue

    # Extra focused LP descent on the current best sequence.
    try:
        bn = int(best_seq.size)
        if bn <= max_lp_n and time.monotonic() < work_deadline - 55.0:
            op = _ConvOp(bn)
            rem = max(0.0, work_deadline - time.monotonic() - 40.0)
            extra = min(220.0, max(25.0, 0.22 * rem))
            stop = min(work_deadline - 40.0, time.monotonic() + extra)
            a, M = _lp_descent(best_seq, op, stop, int(seed) + 88111)
            if a is not None and np.isfinite(M):
                _try_update(a, 2.0 * bn * float(M))
    except Exception:
        pass

    # Final smooth-gradient polish on the best candidate.
    try:
        if best_seq.size <= 900 and time.monotonic() < work_deadline - 14.0:
            rem = max(0.0, work_deadline - time.monotonic() - 8.0)
            allow = min(180.0, max(12.0, 0.35 * rem))
            stop = min(work_deadline - 8.0, time.monotonic() + allow)
            a, fs = _soft_polish(best_seq, stop, int(seed) + 550123)
            if a is not None and np.isfinite(fs):
                _try_update(a, float(fs))
    except Exception:
        pass

    # Final exact validation and return with headroom.
    try:
        fin_score = _safe_evaluate(best_seq)
    except Exception:
        fin_score = math.inf

    if not np.isfinite(fin_score):
        try:
            fin_score = _safe_evaluate(fallback)
        except Exception:
            fin_score = math.inf
        if not np.isfinite(fin_score):
            return [1.0]
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
