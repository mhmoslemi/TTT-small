            final_list = [1.0 / n] * n
        for x in final_list:
            if not np.isfinite(x):
                final_list = [1.0 / n] * n
                break
        v = evaluate_sequence(final_list)
        if not np.isfinite(v):
            final_list = [1.0 / n] * n
            v = evaluate_sequence(final_list)
    except Exception:
        final_list = [1.0 / n] * n

    # Extra safety: if we're too close to deadline, return a known-good fallback
    if time.monotonic() > deadline - 10.0:
        fallback = [1.0 / n] * n
        try:
            vf = evaluate_sequence(fallback)
            if np.isfinite(vf) and vf <= 2.0:
                # Compare with our best
                try:
                    vb = evaluate_sequence(final_list)
                    if np.isfinite(vb) and vb < vf:
                        return final_list
                    return fallback
                except Exception:
                    return fallback
        except Exception:
            return final_list

    return final_list
