            rng.shuffle(coords)
            for i in coords:
                if time.monotonic() > deadline - 10.0:
                    break
                for delta in (d, -d, 2*d, -2*d, d*0.5, -d*0.5):
                    cand = best_a.copy()
                    cand[i] = max(0.0, cand[i] + delta * cand[i])
                    if cand.sum() < 1e-12:
                        continue
                    cand /= cand.sum()
                    sc = _eval_local(cand)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = cand
                        improve = True
                        break
        # Gradually reduce step
        d *= 0.7

    # --- Phase 3: Fine perturbation polish ---
    for _ in range(200):
        if time.monotonic() > deadline - 8.0:
            break
        scale = 0.005 + 0.02 * rng.random()
        cand = best_a * (1.0 + scale * rng.standard_normal(n))
        cand = np.maximum(cand, 0.0)
        cs = cand.sum()
        if cs < 1e-12:
            continue
        cand /= cs
        sc = _eval_local(cand)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = cand

    # --- Final validation with exact evaluator ---
    final_list = [float(x) for x in best_a]
    final_sc = evaluate_sequence(final_list)
    if not np.isfinite(final_sc):
        final_list = [1.0 / n] * n
        final_sc = evaluate_sequence(final_list)

    # Also check a few alternatives
    alternatives = [
        [float(x) for x in parent],
        [1.0 / n] * n,
    ]
    for alt in alternatives:
        try:
            sc = evaluate_sequence(alt)
            if np.isfinite(sc) and sc < final_sc:
                final_sc = sc
                final_list = alt
        except Exception:
            pass

    return final_list
