import numpy as np
import time
from scipy.optimize import linprog

def _score_local(q):
    q = np.asarray(q, dtype=np.float64)
    n = len(q)
    if n == 0:
        return float('inf')
    s = float(q.sum())
    if s < 0.01 or not np.isfinite(s):
        return float('inf')
    b = np.convolve(q, q)
    mb = float(b.max())
    if not np.isfinite(mb):
        return float('inf')
    return 2.0 * n * mb / (s * s)

def _norm(q):
    q = np.asarray(q, dtype=np.float64)
    q = np.clip(q, 0.0, 1000.0)
    q = np.where(np.isfinite(q), q, 0.0)
    s = float(q.sum())
    if s <= 0 or not np.isfinite(s):
        return np.full(q.shape, 1.0 / max(1, q.size), dtype=np.float64)
    return q / s

def _build_A(q, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    qf = q.compute if hasattr(q, 'compute') else q
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        seg = qf[k - je + 1: k - js + 1][::-1]
        A[k, js:je] = seg
    return A

def _lp_step(q, n):
    """Solve LP: max sum(g) s.t. A(q)@g <= M, g >= 0. Return normalized g or None."""
    b = np.convolve(q, q)
    M = float(b.max())
    if M <= 0 or not np.isfinite(M):
        return None
    A = _build_A(q, n)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, 1000.0)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
    except Exception:
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            return None
    if not getattr(res, 'success', False) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, 1000.0)
    g = np.where(np.isfinite(g), g, 0.0)
    s = float(g.sum())
    if s <= 0 or not np.isfinite(s):
        return None
    return g / s

def _grad_step(q, n, alpha=0.05, top_k=20):
    """Subgradient step to reduce max autocorrelation."""
    c = np.convolve(q, q)
    mx = float(c.max())
    if mx <= 0 or not np.isfinite(mx):
        return None
    # Active set: top autocorrelation values
    thresh = mx * 0.85
    act = np.where(c >= thresh)[0]
    if act.size == 0:
        act = np.array([np.argmax(c)])
    # Weight by proximity to max
    w = np.exp((c[act] - mx) * 10.0)
    wsum = float(w.sum())
    if wsum <= 0:
        return None
    w = w / wsum
    # Compute subgradient
    grad = np.zeros(n, dtype=np.float64)
    for ki, wi in zip(act, w):
        ki = int(ki)
        js = max(0, ki - n + 1)
        je = min(n, ki + 1)
        segment = q[ki - je + 1: ki - js + 1][::-1]
        grad[js:je] += 2.0 * wi * segment
    # Project off the mean (simplex tangent)
    grad -= grad.mean()
    gn = float(np.linalg.norm(grad))
    if gn < 1e-14:
        return None
    grad /= gn
    q_new = q - alpha * grad
    q_new = _norm(q_new)
    return q_new

def _optimize(q0, n, rng, deadline, max_lp_iters=200):

def _upscale(q_small, n_target):
    """Upscale a sequence to target length via interpolation."""
    q_small = np.asarray(q_small, dtype=np.float64)
    m = len(q_small)
    if m == n_target:
        return _norm(q_small)
    if m < 2:
        return _norm(np.ones(n_target))
    x = np.linspace(0, 1, m)
    y = np.linspace(0, 1, n_target)
    out = np.interp(y, x, q_small)
    return _norm(out)
