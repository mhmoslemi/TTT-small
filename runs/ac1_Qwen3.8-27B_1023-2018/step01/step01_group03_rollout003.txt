def _normalize(q, target=1.0):
    arr = np.asarray(q, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.full(max(1, int(target)), 1.0, dtype=np.float64)
    arr = np.where(np.isfinite(arr) & (arr > 0.0), arr, 0.0)
    arr = np.minimum(arr, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(arr.size, target / max(1, arr.size), dtype=np.float64)
    arr = arr * (float(target) / s)
    arr = np.clip(arr, 0.0, 1000.0)
    s2 = float(arr.sum())
    if s2 > 0.0 and np.isfinite(s2):
        arr = arr * (float(target) / s2)
    return np.clip(arr, 0.0, 1000.0)


def _score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.where(np.isfinite(a) & (a > 0.0), a, 0.0)
    a = np.minimum(a, 1000.0)

def _top_indices(b, k):

    return idx


def _active_weights(vals, k, temperature_factor=5.0):
    k = max(1, int(k))
    vals = np.asarray(vals, dtype=np.float64).ravel()[:k]

                b_ub=rhs,

def _active_lbfgs_one(q, active, weights, end_time, maxiter=60):
    q = _normalize(q)
    n = q.size
    if n < 2 or active.size == 0 or minimize is None:

    anchor = int(np.argmax(q))
    var_idx = np.asarray([i for i in range(n) if i != anchor], dtype=int)
    pre = []
    for k, wk in zip(active, weights):

        wk = float(wk)
        if wk <= 1e-15:
            continue
        start = max(0, k - n + 1)
        end = min(n - 1, k)
        if start <= end:
            pre.append((wk, k, start, end))
    if not pre:
        return q

    yclip = 30.0
    tiny = 1.0e-300

    def from_y(y):
        y = np.asarray(y, dtype=np.float64).ravel()
        y = np.where(np.isfinite(y), y, 0.0)
        y = np.clip(y, -yclip, yclip)
        if y.size:
            m = float(y.max())
        else:
            m = 0.0
        if m < 0.0:
            m = 0.0
        e = np.exp(y - m)
        Z = 1.0 + float(e.sum())
        out = np.zeros(n, dtype=np.float64)
        out[anchor] = 1.0 / Z
        out[var_idx] = e / Z
        return out

    def fun_grad(y):
        qf = from_y(y)
        gq = np.zeros(n, dtype=np.float64)
        val = 0.0
        for wk, k, start, end in pre:
            seg = qf[k - end:k - start + 1][::-1]
            val += wk * float(np.dot(qf[start:end + 1], seg))
            gq[start:end + 1] += (2.0 * wk) * seg

        if not np.isfinite(val):
            return 1.0e3, np.zeros(n - 1, dtype=np.float64)

        dot = float(np.dot(qf, gq))
        grad = qf[var_idx] * (gq[var_idx] - dot)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(val), grad

    qa = max(float(q[anchor]), tiny)
    y0 = np.log(np.maximum(q[var_idx], tiny) / qa)

def _smooth(q, width):
    q = np.asarray(q, dtype=np.float64).ravel()
    width = max(1, int(width))
    if width <= 1 or q.size == 0:
        return _normalize(q)
    pad_left = width // 2
    pad_right = width - 1 - pad_left
    padded = np.pad(q, (pad_left, pad_right), mode="edge")
    kernel = np.ones(width, dtype=np.float64) / float(width)
    out = np.convolve(padded, kernel, mode="valid")[:q.size]
    return _normalize(out)

def _default_start(n):
    n = max(1, int(n))
    base = np.array([0.2, 0.4, 0.2, 0.2], dtype=np.float64)
    if n == 4:
        return base / base.sum()
    L = max(1, n // 4)
    out = np.repeat(base, L)
    if out.size < n:
        out = np.concatenate([out, base[:n - out.size]])
    out = out[:n]
    return _normalize(out)


def _upscale_interpolate(q, m):
    q = _normalize(q)
    m = max(1, int(m))
    n = q.size
    if n == 0:
        return np.full(m, 1.0 / max(1, m))
    if n == m:
        return q
    x = np.linspace(0.0, 1.0, n)
    y = np.linspace(0.0, 1.0, m)
    out = np.interp(y, x, q)
    return _normalize(out)

            out.append(float(blk.mean()))
    if not out:
        return _normalize(q)
    return _normalize(np.asarray(out, dtype=np.float64))


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not np.isfinite(budget) or budget < 1.0:
        budget = 1300.0

    head = min(60.0, max(3.0, 0.03 * budget))
    hard_deadline = t0 + max(1.0, budget)
    deadline = max(t0 + 1.0, t0 + budget - head)

    try:
        seed_int = int(seed)
    except Exception:
        seed_int = 42
    rng = np.random.default_rng(seed_int)

    best_list = None
    best_score = float("inf")

    def consider(seq):
        nonlocal best_list, best_score
        try:
            arr = _normalize(seq)
            if arr.size == 0:
                return
            lst = [float(x) for x in arr]
            val = float(evaluate_sequence(lst))
            if np.isfinite(val) and val < best_score:
                best_score = val
                best_list = lst
        except Exception:
            pass

    fallback = _normalize(np.array([0.2, 0.4, 0.2, 0.2], dtype=np.float64))
    consider(fallback)

    parent_arr = None
    try:
        hseq = globals().get("height_sequence_1", None)
        if hseq is not None:
            arr = np.asarray(hseq, dtype=np.float64).ravel()
            if arr.size > 0:
                parent_arr = arr.copy()
                consider(parent_arr)
    except Exception:
        parent_arr = None

    if parent_arr is None or parent_arr.size < 2:
        q0 = _default_start(512)
        consider(q0)
    else:
        q0 = _normalize(parent_arr)

    q0 = _normalize(q0)
    best_arr = q0.copy()
    best_arr_sc = _score_fast(best_arr)
    consider(best_arr)

    def update_arr(a):
        nonlocal best_arr, best_arr_sc
        try:
            an = _normalize(a)
            if an.size == 0:
                return
            sc = _score_fast(an)
            if np.isfinite(sc) and sc < best_arr_sc - 1e-15:
                best_arr = an.copy()
                best_arr_sc = sc
            consider(an)
        except Exception:
            pass

    total = max(10.0, deadline - t0)
    large_reserve = min(300.0, max(0.0, 0.22 * total)) if best_arr.size >= 384 else 0.0

    if total > 30.0:
        lp_end = t0 + min(0.55 * total, 720.0)
        lp_end = min(lp_end, deadline - 12.0)
        if lp_end > t0 + 12.0:
            q_lp, sc_lp, last_h = _lp_iterate(q0, lp_end, max_iters=2500, timeout=2.5)
            update_arr(q_lp)
        else:
            q_lp = best_arr.copy()
            last_h = None
    else:
        q_lp = best_arr.copy()
        last_h = None

    if total > 45.0:
        act_end = t0 + min(0.78 * total, 1080.0)
        act_end = min(act_end, deadline - 12.0 - small_reserve)
        if act_end > time.monotonic() + 10.0:
            q_act, sc_act = _active_polish(
                best_arr,
                act_end,
                rng,
                Ks=(1, 2, 4, 8, 16, 32, 64),
                maxiter=65,
            )
            update_arr(q_act)

    if time.monotonic() < deadline - 25.0:
        mul_end = min(deadline - 15.0, time.monotonic() + 45.0)
        q_mul, sc_mul = _multiplicative_polish(best_arr, mul_end, max_iters=160)
        update_arr(q_mul)

    extra_end = deadline - large_reserve - 10.0 if total > 300.0 else deadline
    variants = []
    if best_arr.size > 0:
        variants.append(_smooth(best_arr, 3))
        variants.append(_smooth(best_arr, 5))
        variants.append(_smooth(best_arr, 9))
        variants.append(_power_q(best_arr, 0.6))
        variants.append(_power_q(best_arr, 2.0))
        for sg in (0.05, 0.15, 0.30):
            variants.append(_random_perturb(best_arr, sg, rng))
        variants.append(_normalize(0.7 * best_arr + 0.3 * (np.ones(best_arr.size) / best_arr.size)))
        if last_h is not None and np.asarray(last_h).ravel().size == best_arr.size:
            variants.append(_normalize(0.5 * best_arr + 0.5 * np.asarray(last_h).ravel()))

    for v in variants[:6]:
        now = time.monotonic()
        if now > extra_end - 20.0:
            break
        per = min(35.0, max(5.0, (extra_end - 20.0 - now) / 2.0))
        end = min(extra_end, now + per)
        qv, scv = _active_polish(v, end, rng, Ks=(1, 2, 4, 8, 16), maxiter=40)
        update_arr(qv)

    if best_arr.size >= 128 and best_arr.size < 512 and time.monotonic() < extra_end - 30.0:
        small = _downsample(best_arr, 2)
        update_arr(small)
        end_small = min(extra_end, time.monotonic() + 45.0)
        qs, scs = _active_polish(small, end_small, rng, Ks=(1, 2, 4, 8), maxiter=35)
        update_arr(qs)

    if total > 700.0 and 384 <= best_arr.size <= 1200:
        now = time.monotonic()
        if now < deadline - 160.0:
            n2 = min(2048, 2 * int(best_arr.size))
            if n2 > best_arr.size:
                qbig = _upscale_interpolate(best_arr, n2)
                update_arr(qbig)

                rem = deadline - now
                if qbig.size >= 2 and rem > 180.0:
                    big_lp_end = min(deadline - 120.0, now + min(120.0, 0.35 * rem))
                    if big_lp_end - now > 15.0:
                        qbig, scb, _ = _lp_iterate(qbig, big_lp_end, max_iters=160, timeout=5.0)
                        update_arr(qbig)

                if time.monotonic() < deadline - 60.0:
                    big_act_end = min(deadline - 25.0, time.monotonic() + min(160.0, 0.45 * rem))
                    qbig, scb = _active_polish(
                        qbig,
                        big_act_end,
                        rng,
                        Ks=(1, 2, 4, 8, 16),
                        maxiter=45,
                    )
                    update_arr(qbig)

    if best_list is None:
        consider(fallback)

    try:
        val = float(evaluate_sequence(best_list))
        if not np.isfinite(val):
            best_list = [float(x) for x in fallback]
    except Exception:
        best_list = [float(x) for x in fallback]

    return [float(x) for x in best_list]
