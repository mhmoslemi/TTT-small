def _lp_step(a, M):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0."""
    n = len(a)
    # Constraint matrix: row k has entries a[k-j] for valid j
    # C[k, j] = a[k-j] if 0 <= k-j < n else 0
    # Using scipy sparse or dense (for n=192, 383x192 is small)

    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                C[k, j] = a[idx]
    
    # Maximize sum(b) = c^T b where c = ones
    # Subject to C @ b <= M (all constraints)
    # b >= 0
    c = -np.ones(n)  # minimize -sum(b)
    A_ub = C
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'maxiter': 1000})
    
    if res.success:
        return res.x
    return None
