import time
from scipy.optimize import linprog


def _score_arr(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.inf
    c = np.convolve(a, a)
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * n * mx / (s * s))


def _norm(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s


def _conv_mat(a):
    n = a.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _lp_direction(a, M, rng, relax=1.0):
    n = a.size
    A = _conv_mat(a)
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = M * relax * np.ones(2 * n - 1, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3})
        if res.success and res.x is not None and np.all(np.isfinite(res.x)):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return a.copy()


def _gradient_peak(a, p):
    n = a.size
    k = np.arange(2 * n - 1)[:, None].astype(np.float64)
    m = np.arange(n)[None, :].astype(np.float64)
    diff = k - m
    valid1 = (diff >= 0) & (diff < n)
    valid2 = (diff <= 0) & (diff > -n)
    idx1 = np.clip(diff, 0, n - 1).astype(int)
    idx2 = np.clip(-diff, 0, n - 1).astype(int)
    part1 = np.where(valid1, p[:, None] * a[idx1], 0.0)
    part2 = np.where(valid2, p[:, None] * a[idx2], 0.0)
    grad = part1.sum(axis=0)
    grad += part2.sum(axis=0)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad


def _soft_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    cmax = float(np.max(c))
    z = beta * (c - cmax)
    z = np.where(np.isfinite(z), z, -1000.0)
    p = np.exp(z)
    ps = float(p.sum())
    if ps < 1e-300 or not np.isfinite(ps):
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p /= ps
    return p


def _adam_step(params, grad, state, lr, cap=0.6):
    b1, b2, eps = 0.9, 0.999, 1e-8
    state['t'] += 1
    t = state['t']
    state['b1'] += 1 - b1
    state['b2'] += 1 - b2
    state['m'] = b1 * state['m'] + (1 - b1) * grad
    state['v'] = b2 * state['v'] + (1 - b2) * grad * grad
    mhat = state['m'] / state['b1']
    vhat = state['v'] / state['b2']
    step = lr * mhat / (np.sqrt(vhat) + eps)
    step = np.clip(step, -cap, cap)
    step = np.where(np.isfinite(step), step, 0.0)
    return params - step


def _make_adam(dim):
    return {'m': np.zeros(dim), 'v': np.zeros(dim), 'b1': 0.1, 'b2': 0.001, 't': 0}
