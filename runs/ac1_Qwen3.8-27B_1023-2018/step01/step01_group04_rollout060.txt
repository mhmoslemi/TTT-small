        return float('inf')
    b = np.convolve(a, a)
    mb = float(np.max(b))
    if not math.isfinite(mb):
        return float('inf')
    return float(2.0 * n * mb / (s * s))


def _normalize(a):
    a = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0.0, None)
    s = float(np.sum(a))
    if s <= 1e-15:
        return np.ones_like(a) / max(1, a.size)
    return a / s


def _update_best(seq, best):
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
        arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size == 0:
            return
        s = float(np.sum(arr))
        if s < 0.01:
            return
        sc = evaluate_sequence([float(x) for x in arr])
        if math.isfinite(sc) and sc < float(best[0]):
            best[0] = float(sc)
            best[1] = arr.copy()
    except Exception:
        pass


def _build_A(a):
    n = len(a)
    if n <= 0 or sps is None:
        return None
    j = np.repeat(np.arange(n, dtype=np.int64), n)
    k = np.tile(np.arange(n, dtype=np.int64), n)
    rows = j + k
    cols = j
    data = np.tile(a, n)
    try:
        return sps.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    except Exception:
        return None


def _lp_one(a, M, tlim):
    if linprog is None:
        return None
    n = len(a)
    A = _build_A(a)
    if A is None:
        return None
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-12)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': tlim, 'presolve': True})
    except Exception:
        return None
    if res is None or getattr(res, 'x', None) is None:
        return None
    g = np.asarray(res.x, dtype=np.float64).ravel()
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    g = np.clip(g, 0.0, None)
    sg = float(g.sum())
    if sg <= 1e-15:
        return None
    return g / sg


def _lp_optimize(a, n, time_limit, rng, best):
    stop = time.monotonic() + time_limit
    a = _normalize(a.copy())
    local = _score_fast(a)
    if math.isfinite(local):
        _update_best(a, best)

    stagnant = 0
    it = 0
    max_iters = 300
    m_scale = 1.0

    while it < max_iters:
        now = time.monotonic()
        if now >= stop - 3:
            break
        it += 1

        b = np.convolve(a, a)
        M = float(np.max(b)) * m_scale
        if not math.isfinite(M) or M <= 0:
            break

        tlim = max(0.3, min(8.0, (stop - now - 4) / 3.0))
        if tlim <= 0:
            break

        g = _lp_one(a, M, tlim)
        if g is None:
            stagnant += 1
            if stagnant > 6:
                m_scale *= 1.02
                if m_scale > 1.05:
                    break
            continue

        best_sc = local
        best_cand = None
        for t in (0.15, 0.3, 0.45, 0.6, 0.75, 0.9, 1.0):
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0.0, None)
            sc = _score_fast(cand)
            if math.isfinite(sc) and sc < best_sc - 1e-13:
                best_sc = sc
                best_cand = cand

        if best_cand is not None and best_sc < local * (1.0 - 1e-10):
            a = _normalize(best_cand)
            local = best_sc
            stagnant = 0
            m_scale = max(0.95, m_scale * 0.99)
            _update_best(a, best)
            if local < TARGET * 0.98:
                break
        else:
            stagnant += 1
            if stagnant % 6 == 0:
                pa = a * np.exp(0.06 * rng.standard_normal(n))
                pa = _normalize(pa)
                psa = _score_fast(pa)
                if math.isfinite(psa) and psa < local * (1.0 - 1e-10):
                    a = pa
                    local = psa
                    stagnant = 0
                    _update_best(a, best)
            if stagnant > 30:
                break

    return a, local


def _grad_score(a, beta=6.0):
    """Compute smoothed gradient of peak autocorrelation."""
    n = len(a)
    b = np.convolve(a, a)
    S = float(a.sum())
    R = b / (S * S)
    rmax = float(np.max(R))
    # Softmax-like weight emphasizes peaks
    w = 1.0 + np.exp(beta * (R - rmax + 1.0 / (n + 1)))
    w = np.clip(w, 1.0, 1e6)
    # gradient: grad[i] = 2 * sum_j a[j] * w[i+j] for valid indices
    # = 2 * np.correlate(w, a, 'full')[i] for i=0..n-1
    corr = np.correlate(w, a, 'full')
    grad = 2.0 * corr[:n]
    return grad, rmax


def _gradient_refine(a, time_limit, rng, best):
    n = len(a)
    stop = time.monotonic() + time_limit
    a = _normalize(a.copy())
    local = _score_fast(a)
    if math.isfinite(local):
        _update_best(a, best)

    stale = 0
    step_scale = 0.3
    beta = 6.0

    for it in range(8000):
        now = time.monotonic()
        if now >= stop - 2:
            break
        if (it + 1) % 200 == 0:
            beta *= 1.5
            step_scale *= 0.9

        grad, rmax = _grad_score(a, beta)
        gn = float(np.linalg.norm(grad))
        if not math.isfinite(gn) or gn < 1e-14:
            break

        best_sc = local
        best_cand = None
        for alpha in (0.05, 0.1, 0.2, 0.5, 1.0):
            step = alpha * step_scale * grad / gn
            cand = a - step
            cand = np.clip(cand, 0.0, None)
            ds = float(np.sum(cand))
            if ds < 0.01:
                continue
            cand = cand / ds
            sc = _score_fast(cand)
            if math.isfinite(sc) and sc < best_sc - 1e-13:
                best_sc = sc
                best_cand = cand

        if best_cand is not None:
            a = _normalize(best_cand)
            local = best_sc
            stale = 0
            _update_best(a, best)
            if local < TARGET * 0.98:
                break
        else:
            stale += 1
            if stale % 8 == 0:
                pa = np.clip(a + 0.03 * rng.standard_normal(n), 0.0, None)
                pa = _normalize(pa)
                psa = _score_fast(pa)
                if math.isfinite(psa) and psa < local:
                    a = pa
                    local = psa
                    stale = 0
                    _update_best(a, best)
            if stale > 80:
                break

    return a, local


def _resample(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    old_n = len(a)
    if old_n < 2 or new_n < 2:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, np.clip(a, 0, None))
    new = np.clip(new, 0.0, None)
    s = float(np.sum(new))
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return new / s


def _truncate(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if new_n >= n:
        return _normalize(a)
    # Remove the smallest elements
    idx = np.argsort(a)
    keep = np.sort(idx[new_n:])
    result = a[keep].copy()
    s = float(np.sum(result))
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return result / s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    hard_start = time.monotonic()
    try:
        remaining = max(1.0, float(budget_s))
    except Exception:
        remaining = 1300.0
    deadline = hard_start + remaining - 18.0

    best = [float('inf'), np.array([1.0, 1.0, 1.0, 1.0], dtype=np.float64)]

    # Load parent
    parent = None
    try:
        h = globals().get('height_sequence_1', None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size >= 2 and float(parent.sum()) >= 0.01:
                _update_best(parent, best)
    except Exception:
        parent = None

    if not math.isfinite(best[0]):
        best[0] = 2.0
        best[1] = np.ones(8) / 8.0

    if parent is None:
        parent = best[1].copy()

    p_len = len(parent)
    p_norm = _normalize(parent)

    # Phases: LP at various lengths, then gradient refinement
    # Prioritize shorter lengths (better 2n factor)
    lengths = []
    for n in range(max(4, p_len - 10), p_len + 6):
        if n not in lengths:
            lengths.append(n)

    # Also try a few much shorter lengths
    for n in (24, 32, 48, 64):
        if n not in lengths:
            lengths.append(n)

    # Sort: shorter first (better 2n factor), but parent length gets priority
    lengths.sort(key=lambda n: (abs(n - p_len), -n))
    # Move parent length to front
    if p_len in lengths:
        lengths.remove(p_len)
        lengths.insert(0, p_len)
    # Then by decreasing n (shorter ones next for the 2n advantage)
    lengths.sort(key=lambda n: -n if n != p_len else -9999)

    total_lp_time = remaining * 0.55
    total_grad_time = remaining * 0.25
    per_lp = total_lp_time / max(1, min(len(lengths), 6))
    per_grad = total_grad_time / max(1, 3)

    done_lengths = 0
    for n in lengths[:8]:
        now = time.monotonic()
        if now >= deadline - 20:
            break

        remaining_here = deadline - now - 15
        if remaining_here < 15:
            break

        alloc = min(per_lp, remaining_here * 0.6)
        if n == p_len:
            alloc *= 1.5
        alloc = max(alloc, 20.0) if remaining_here > 60 else remaining_here * 0.4
        alloc = min(alloc, remaining_here - 10)
        if alloc < 12:
            break

        # Get initial sequence for this length
        if n == p_len:
            init = p_norm.copy()
        elif n < p_len:
            init = _truncate(p_norm, n)
        else:
            init = _resample(p_norm, n)

        # Run LP
        a_opt, local_sc = _lp_optimize(init, n, alloc, rng, best)
        done_lengths += 1

        # Quick gradient refinement
        gr_t = min(15.0, (deadline - time.monotonic() - 10))
        if gr_t > 8:
            a_opt, local_sc = _gradient_refine(a_opt, gr_t, rng, best)

    # Extended gradient refinement on the best sequence found
    if time.monotonic() < deadline - 25:
        best_arr = _normalize(best[1].copy())
        n_best = len(best_arr)
        gr_alloc = min(120.0, deadline - time.monotonic() - 12)
        if gr_alloc > 15:
            _gradient_refine(best_arr, gr_alloc, rng, best)

    # One more LP pass on the best if we have time
    if time.monotonic() < deadline - 30:
        best_arr = _normalize(best[1].copy())
        n_best = len(best_arr)
        lp_alloc = min(80.0, deadline - time.monotonic() - 12)
        if lp_alloc > 15:
            a_opt, _ = _lp_optimize(best_arr, n_best, lp_alloc, rng, best)
            if time.monotonic() < deadline - 12:
                _gradient_refine(a_opt, min(30.0, deadline - time.monotonic() - 8), rng, best)

    # Final validation
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.array([1.0, 1.0, 1.0, 1.0])
    s = float(np.sum(final))
    if s < 0.01:
        final = np.ones(4)

    _update_best(final, best)
    final = np.asarray(best[1], dtype=np.float64).ravel()
    fs = float(np.sum(final))
    if fs < 0.01:
        final = np.array([1.0, 1.0, 1.0, 1.0])

    # Exact evaluation check
    test_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(test_sc):
        final = np.array([1.0, 1.0, 1.0, 1.0])

    return [float(x) for x in final]
