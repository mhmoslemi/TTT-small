    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)

    n = len(height_sequence_1)
    conv_len = 2 * n - 1

    # FFT size for fast convolution
    fft_n = 1
    while fft_n < conv_len:
        fft_n <<= 1

    def fast_conv(a):
        F = np.fft.rfft(a, fft_n)
        return np.fft.irfft(F * F, fft_n)[:conv_len]

    # --- Initialization ---
    candidates = []

    a_p = np.asarray(height_sequence_1, dtype=np.float64).copy()
    a_p = np.clip(a_p, 0.0, 1000.0)
    if a_p.sum() > 1e-6:
        a_p /= a_p.sum()
        candidates.append(a_p)

    j = np.arange(n, dtype=np.float64)

    for p in [0.5, 1.0, 2.0, 3.0, 4.0]:
        c = np.sin(np.pi * (j + 0.5) / n) ** p
        c = np.maximum(c, 0)
        if c.sum() > 1e-6:
            candidates.append(c / c.sum())

    for sig in [0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.5]:
        g = np.exp(-0.5 * ((j - n * 0.5) / (n * sig)) ** 2)
        candidates.append(g / g.sum())

    for frac in [0.25, 0.3, 0.35, 0.4]:
        for w in [0.08, 0.12, 0.15, 0.2]:
            b = (np.exp(-0.5 * ((j - n * frac) / (n * w)) ** 2) +
                 np.exp(-0.5 * ((j - n * (1.0 - frac)) / (n * w)) ** 2))
            candidates.append(b / b.sum())

    for _ in range(10):
        r = rng.random(n)
        k_sm = max(3, n // 40)
        kern = np.ones(k_sm) / k_sm
        r = np.convolve(r, kern, 'same')
        r = np.maximum(r, 0)
        if r.sum() > 1e-6:
            candidates.append(r / r.sum())

    # Blends of parent with diverse candidates
    for i in range(1, min(8, len(candidates))):
        for t in [0.2, 0.35, 0.5]:
            candidates.append((1.0 - t) * candidates[0] + t * candidates[i])

    best_a = candidates[0].copy()
    best_score = float('inf')
    for c in candidates:
        sc = float(evaluate_sequence(c.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = c.copy()

    a = best_a.copy()

    # --- Phase 1: Soft multiplicative update ---
    eta = 0.4
    top_frac = 0.15
    patience = 0
    max_patience = 180
    it = 0
    phase2_budget = max(100.0, float(budget_s) * 0.18)
    phase2_start = deadline - phase2_budget

    while time.monotonic() < phase2_start:
        it += 1

        conv = fast_conv(a)
        max_conv = conv.max()

        k_top = max(5, int(top_frac * conv_len))
        thresh = np.partition(conv, conv_len - k_top)[conv_len - k_top]
        spread = max_conv - thresh
        if spread > 1e-14:
            weights = 1.0 + 2.5 * np.exp(np.clip((conv - thresh) / spread, 0.0, 5.0))
        else:
            weights = np.ones(conv_len)

        load = np.correlate(weights, a, 'valid')
        lm = load.max()
        if lm > 1e-15:
            a = a * np.exp(-eta * load / lm)
        a = np.maximum(a, 1e-20)
        a /= a.sum()

        if it % 10 == 1:
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1

        if patience >= max_patience:
            r = rng.random()
            if r < 0.25:
                np_ = max(5, n // 25)
                idx = rng.choice(n, size=np_, replace=False)
                a[idx] *= np.exp(rng.normal(0, 0.4, size=np_))
            elif r < 0.45:
                t = rng.uniform(0.1, 0.4)
                alt = candidates[rng.integers(len(candidates))].copy()
                a = (1.0 - t) * a + t * alt
            elif r < 0.6:
                a = 0.7 * a + 0.3 * a[::-1]
            elif r < 0.75:
                k = max(3, n // 50)
                kern = np.ones(k) / k
                a = 0.6 * a + 0.4 * np.convolve(a, kern, 'same')
            else:
                a = best_a.copy()
                np_ = max(10, n // 12)
                idx = rng.choice(n, size=np_, replace=False)
                a[idx] *= np.exp(rng.normal(0, 0.5, size=np_))

            a = np.clip(a, 0.0, 1000.0)
            a /= a.sum()
            patience = 0
            eta = min(1.0, eta * 1.1)

        if it % 4000 == 0:
            eta = max(0.12, eta * 0.93)
            top_frac = max(0.05, top_frac * 0.94)

    # --- Phase 2: Coordinate descent polishing ---
    a = best_a.copy()
    current_score = best_score
    conv = fast_conv(a)
    sum_a = a.sum()
    cd_patience = 0
    max_cd_patience = 4

    factors = [0.1, 0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0]

    for cd_pass in range(500):
        if time.monotonic() > deadline - 12:
            break

        pass_improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > deadline - 12:
                break

            ai = a[i]
            if ai < 1e-15:
                continue

            k_lo = i
            k_hi = min(conv_len - 1, i + n - 1)
            sl = k_hi - k_lo + 1  # should be n

            # Unchanged max
            unch_max = 0.0
            if k_lo > 0:
                unch_max = max(unch_max, conv[:k_lo].max())
            if k_hi < conv_len - 1:
                unch_max = max(unch_max, conv[k_hi + 1:].max())

            best_x = ai
            best_x_sc = current_score

            for f in factors:
                x = ai * f
                if x < 0.0 or x > 1000.0:
                    continue
                if abs(x - ai) < 1e-14:
                    continue

                delta = x - ai
                # new_conv in affected range [k_lo, k_hi]:
                # new_vals[j] = conv[k_lo+j] + 2*delta*a[j]  for j=0..sl-1
                # diagonal correction at j = 2i - k_lo = i:
                #   new_vals[i] += delta^2
                new_vals = conv[k_lo:k_hi + 1] + 2.0 * delta * a[:sl]
                diag_j = 2 * i - k_lo  # = i
                if 0 <= diag_j < sl:
                    new_vals[diag_j] += delta * delta

                loc_max = new_vals.max()
                total_max = max(unch_max, loc_max)

                new_sum = sum_a - ai + x
                if new_sum < 1e-10:
                    continue
                x_sc = 2.0 * n * total_max / (new_sum * new_sum)

                if x_sc < best_x_sc:
                    best_x_sc = x_sc
                    best_x = x

            if best_x != ai and best_x_sc < current_score:
                a[i] = best_x
                sum_a = a.sum()
                pass_improved = True
                current_score = best_x_sc

        # End of coordinate pass: validate with exact evaluator
        if time.monotonic() > deadline - 12:
            break

        a_norm = a / a.sum()
        sc = float(evaluate_sequence(a_norm.tolist()))

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_norm.copy()
            a = a_norm.copy()
            sum_a = a.sum()
            conv = fast_conv(a)
            current_score = sc
            cd_patience = 0
        elif not np.isfinite(sc) or sc > current_score * 1.01:
            # Revert bad pass
            a = best_a.copy()
            sum_a = a.sum()
            conv = fast_conv(a)
            current_score = best_score
            cd_patience += 1
        else:
            cd_patience += 1
            a = a_norm.copy()
            sum_a = a.sum()
            conv = fast_conv(a)

        if not pass_improved:
            cd_patience += 2
            if cd_patience >= max_cd_patience:
                # Try a perturbation to escape
                a = best_a.copy()
                sum_a = a.sum()
                conv = fast_conv(a)
                current_score = best_score
                np_ = max(10, n // 20)
                idx = rng.choice(n, size=np_, replace=False)
                a[idx] *= np.exp(rng.normal(0, 0.3, size=np_))
                a = np.clip(a, 0.0, 1000.0)
                a /= a.sum()
                sum_a = a.sum()
                conv = fast_conv(a)
                cd_patience = 0

                # Quick check
                sc2 = float(evaluate_sequence(a.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_a = a.copy()
                    current_score = sc2

    # --- Final validation and return ---
    result = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(result))
    if not np.isfinite(final_score) or final_score > 2.0:
        result = [1.0] * n

    return result
