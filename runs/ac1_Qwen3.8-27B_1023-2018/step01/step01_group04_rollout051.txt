import numpy as np
import time
import math
from scipy.optimize import linprog
import scipy.sparse as sps

def _convolve(a):
    """Linear convolution of a with itself."""
    return np.convolve(a, a)

def _score(a):
    """Compute 2n*max(c)/(sum a)^2"""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, None)
    n = a.size
    if n == 0:
        return float('inf')
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = _convolve(a)
    return 2.0 * n * c.max() / (s * s)

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-15:
        a[:] = 1.0 / a.size
        return a
    return a / s

def _subgradient(a):
    """Compute subgradient of max(c_k) at a (assuming sum a = 1)."""
    # a should be normalized (sum=1)
    n = a.size
    c = _convolve(a)
    M = c.max()
    # Active set: lags where c_k is close to M
    thresh = M * (1 - 1e-8) - 1e-12
    K = np.where(c >= thresh)[0]
    
    # dM/da_i = (2/|K|) * sum_{k in K} a_{k-i} (if 0<=k-i<n)
    # (B_k a)_i = a_{k-i} if 0<=k-i<n
    # So dM/da_i = (2/|K|) * sum_{k in K} a_{k-i} * 1_{0<=k-i<n}
    
    g = np.zeros(n)
    for k in K:
        # (B_k a)_i = a_{k-i} for valid i
        # i range: max(0, k-(n-1)) to min(k, n-1)
        i_lo = max(0, k - (n - 1))

        # a_{k-i} for i in [i_lo, i_hi] is a[k - i_lo : k - i_hi + 1][::-1]
        # Actually: for i = i_lo, k-i = k-i_lo (max); for i=i_hi, k-i = k-i_hi (min)
        g[i_lo:i_hi+1] += a[k - i_hi : k - i_lo + 1][::-1]
    
    g *= 2.0 / len(K)

def _lp_step(a, M):
    """Given a (normalized) and M = max(c), solve LP to find better g."""
    n = a.size
    # Build constraint matrix: (A g)_k = sum_j a_{k-j} * g_j (cross-correlation)
    # This is the same as convolve(a, g)[k]
    
    # For efficient construction:
    # (a * g)_k = sum_{j: 0<=j<n, 0<=k-j<n} a_{k-j} * g_j
    # For column j of A: A[k, j] = a_{k-j} if 0<=k-j<n, else 0
    
    if sps is None:
        return None
    
    # Build sparse matrix
    rows = []
    cols = []
    data = []
    for j in range(n):
        # A[k, j] = a[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    
    A = sps.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    
    # Solve: max sum(g) s.t. A g <= M, g >= 0
    # linprog minimizes, so use c = -ones
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                     bounds=[(0, None)] * n, method='highs',
                     options={'presolve': True, 'time_limit': 30})
    except:
        return None
    
    if res is None or not res.success or res.x is None:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    s = g.sum()
    if s < 1e-15:
        return None
    g = g / s
    return g

def _fast_score(a):

def _norm1(a):

    A = _build_A(a)

def _smoothed_gradient(a, beta=50.0):
    n = a.size
    c = np.convolve(a, a)
    cb = beta * (c - c.max())
    cb = np.clip(cb, -500.0, 0.0)
    p = np.exp(cb)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros_like(c)
        p[c.argmax()] = 1.0
        ps = 1.0
    else:
        p /= ps
    # Gradient of soft_max_beta(c(a)) w.r.t. a_i
    # d softmax/da_i = sum_k p_k * 2 * (B_k a)_i
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        i_lo = max(0, k - (n - 1))
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            g[i_lo:i_hi + 1] += 2.0 * p[k] * a[k - i_hi:k - i_lo + 1][::-1]
    return g


def _simplex_proj(v, s_target=1.0):
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v / s_target)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if rho.size == 0:
        return np.full(n, s_target / n)
    rho = rho[-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta * s_target, 0.0)

            break

        g_tan = g - g.mean()

    scored.sort(key=lambda z: z[0])
    return [a for _, a in scored]


def _upsample(parent, n):
    if parent is None or parent.size < 2:
        return np.ones(n) / n
    tp = np.linspace(0.0, 1.0, parent.size)
    xp = np.linspace(0.0, 1.0, n)
    return np.interp(xp, tp, parent)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s)) - 15.0

    best = [float('inf'), np.array([1.0], dtype=np.float64)]

    def update_best(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        try:
            sc = evaluate_sequence([float(x) for x in a])
        except Exception:
            return
        if math.isfinite(sc) and sc < best[0]:
            best[0] = float(sc)
            best[1] = a.copy()

    update_best(best[1])

    parent = None
    try:
        h = globals().get('height_sequence_1', None)
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            parent = np.nan_to_num(parent, nan=0.0, posinf=0.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.size > 0 and parent.sum() >= 0.01:
                update_best(parent)
    except Exception:
        parent = None

    def run_lp_on(n, time_alloc, parent_seq):
        if time.monotonic() >= deadline - 30:
            return
        starts = _make_init(n, parent_seq, rng)
        max_starts = min(len(starts), max(3, int(time_alloc / 40)))
        for a0 in starts[:max_starts]:
            if time.monotonic() >= deadline - 30:
                return
            rem = deadline - time.monotonic() - 30
            if rem < 5:
                return
            a = _lp_iteration(a0, max_iters=200, stop_time=deadline - 30,
                              rng=rng, best_list=best)
        # Gradient polish on best of this length
        if time.monotonic() < deadline - 20:
            a_pol = _norm1(best[1])
            _gradient_polish(a_pol, max_iters=500, stop_time=deadline - 20,
                             rng=rng, best_list=best)

    # Phase 1: Optimize at parent length
    if parent is not None and parent.size > 1:
        pn = parent.size
        run_lp_on(pn, 350.0, parent)

    # Phase 2: Try nearby lengths
    for n_try in [64, 80, 128, 160, 192]:
        if time.monotonic() >= deadline - 100:
            break
        if n_try == (parent.size if parent is not None else 0):
            continue
        alloc = min(80.0, max(0.0, deadline - time.monotonic() - 100))
        if alloc < 15:
            break
        run_lp_on(n_try, alloc, best[1])

    # Phase 3: Gradient polish on the global best
    if time.monotonic() < deadline - 15:
        a_pol = _norm1(best[1])
        _gradient_polish(a_pol, max_iters=1000, stop_time=deadline - 15,
                         rng=rng, best_list=best)

    # Phase 4: Final LP refinement with more iterations on best
    if time.monotonic() < deadline - 15:
        a = _norm1(best[1])
        _lp_iteration(a, max_iters=100, stop_time=deadline - 15,
                      rng=rng, best_list=best)

    # Final: validate and return
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.array([1.0])
    s = float(final.sum())
    if s >= 0.01:
        final = final / s
        final = np.clip(final, 0.0, 1000.0)
    try:
        sc = evaluate_sequence([float(x) for x in final])
        if math.isfinite(sc) and sc < best[0]:
            best[0] = sc
            best[1] = final.copy()
    except Exception:
        pass

    out = np.asarray(best[1], dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.clip(out, 0.0, 1000.0)
    if out.size == 0:
        out = np.array([1.0])
    if out.sum() < 0.01:
        out = out + 0.01
    return [float(x) for x in out]
