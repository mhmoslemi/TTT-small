import numpy as np

from scipy.optimize import linprog

def _fs(a):
    """Fast internal score."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 1e18
    b = np.convolve(a, a)
    s = float(a.sum())
    if s < 0.01:
        return 1e18
    v = 2.0 * n * float(b.max()) / (s * s)
    return v if np.isfinite(v) else 1e18

def _nm(a):
    """Normalize to nonnegative, unit-sum."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, None)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s

def _build_A(q):
    """Build the Toeplitz-like convolution matrix A[k,i] = q[k-i] if valid."""
    n = len(q)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = q
    return A

def _lp_response(q, tl=10.0):
    """Parent LP: max sum(g) s.t. (q*g)_k <= M, g>=0. Same M = max(q*q)."""
    n = len(q)
    A = _build_A(q)
    M = float(np.convolve(q, q).max())
    if M <= 0 or not np.isfinite(M):

def _sgrad(q, alpha):
    """One subgradient step projected onto the probability simplex."""
    n = len(q)
    b = np.convolve(q, q)
    M = float(b.max())
    if M <= 0:
        return q
    act = np.where(b >= 0.85 * M)[0]
    w = b[act].astype(np.float64)
    ws = float(w.sum())
    if ws <= 0:
        return q
    w /= ws
    s = np.zeros(n, dtype=np.float64)
    for i in range(len(act)):
        k, wi = int(act[i]), float(w[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            s[lo:hi + 1] += 2.0 * wi * q[k - hi:k - lo + 1][::-1]
    s -= s.mean()
