def _project(x):
    """Project x onto {a >= 0, sum(a) = 1}"""
    # Water-filling: find t such that sum(max(0, x_i - t)) = 1
    x_sorted = np.sort(x)[::-1]

    cumsum = np.cumsum(x_sorted)

        if x_sorted[k-1] >= t or k == 1:
            break
    result = np.maximum(0, x - t)
    s = result.sum()
    if s > 0:
        result = result / s  # renormalize (should already be 1 if we found t correctly)
    return result

Wait, the water-filling projection onto `{a >= 0, sum(a) = 1}`:
We want to find `P(x)` that minimizes `||x - a||_2` subject to `a >= 0` and `sum(a) = 1`.

The KKT conditions give: `a_i = max(0, x_i - t)` for some `t`, where `t` is chosen so that `sum(a_i) = 1`.

The standard algorithm:
1. Sort `x` in decreasing order: `x_(1) >= x_(2) >= ... >= x_(n)`
2. For `k = 1, 2, ..., n`: define `t_k = (sum_{i=1}^k x_(i) - 1) / k`
3. Find the largest `k` such that `x_(k) > t_k` (and `t_k > 0` or `k = n`)
4. Then `a_i = max(0, x_i - t_k)`

```python
def _project(x):
    n = len(x)
    x_sorted = np.sort(x)[::-1]
    cumsum = np.cumsum(x_sorted)
    # Find k such that x_sorted[k-1] >= (cumsum[k-1] - 1) / k
    # i.e., k * x_sorted[k-1] >= cumsum[k-1] - 1
    # i.e., cumsum[k-1] - k * x_sorted[k-1] <= 1
    # Note: cumsum[k-1] - k * x_sorted[k-1] = sum_{i=1}^{k-1} (x_sorted[i-1] - x_sorted[k-1])
    # This is non-decreasing in k (since x is sorted decreasing)
    rho = np.arange(1, n+1)
    vals = (cumsum - rho * x_sorted)  # this is cumsum[k-1] - k * x_sorted[k-1]
    # We want the largest k such that vals[k-1] <= 1
    # Since vals is non-decreasing, we can binary search
    k = np.searchsorted(vals, 1.0, side='right')
    if k == 0:
        # All vals > 1, so k=1
        t = x_sorted[0] - 1.0
        if t < 0:
            t = 0.0  # shouldn't happen for nonnegative x with sum >= 1... actually it can
        result = np.maximum(0, x - t)
    else:
        # k is the answer (1-indexed)
        t = (cumsum[k-1] - 1.0) / k
        result = np.maximum(0, x - t)
    s = result.sum()
    if s > 1e-15:
        result = result / s  # safety renormalization
    else:
        result = np.ones(n) / n
    return result
```

    # Phase 2: LP on other lengths
    for n in [96, 64, 128]:
        if time.monotonic() >= deadline - 200.0:
            break
        if n == (parent.size if parent is not None else 0):
            continue
        remaining = deadline - time.monotonic()
        alloc = min(120.0, max(20.0, remaining * 0.25))
        a0 = _resample(best[1], n) if best[1] is not None else np.ones(n) / n
        _lp_refine(a0, n, alloc, best, rng)

    # Phase 3: Gradient polish on best sequence
    if time.monotonic() < deadline - 100.0:
        n_best = best[1].size
        remaining = deadline - time.monotonic() - 80.0
        alloc = min(250.0, max(30.0, remaining * 0.6))
        _gradient_polish(best[1], n_best, alloc, best, rng)

    # Phase 4: Subgradient polish
    if time.monotonic() < deadline - 60.0:
        n_best = best[1].size
        remaining = deadline - time.monotonic() - 50.0
        alloc = min(100.0, max(15.0, remaining * 0.5))
        _subgradient_polish(best[1], n_best, alloc, best)

    # Phase 5: Final gradient polish with higher beta
    if time.monotonic() < deadline - 30.0:
        n_best = best[1].size
        remaining = deadline - time.monotonic() - 20.0
        alloc = min(60.0, max(5.0, remaining * 0.7))
        _gradient_polish(best[1], n_best, alloc, best, rng)

    # Final validation
    final = np.asarray(best[1], dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.array([1.0], dtype=np.float64)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(final.size if final.size > 0 else 1) / max(1, final.size)
    else:
        final = final / s
    final = np.clip(final, 0.0, 1000.0)

    _try_update(final, best)
    final = np.asarray(best[1], dtype=np.float64).ravel()
    sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(sc) or final.size == 0:
        final = np.array([1.0 / 3.0, 2.0 / 3.0])

    return [float(x) for x in final]
