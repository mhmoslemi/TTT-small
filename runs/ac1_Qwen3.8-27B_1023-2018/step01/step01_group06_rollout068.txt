        # grad[j] = sum_m a[m] * weights[j+m]

        wf = weights[::-1]
        Fa = np.fft.rfft(a, grad_fft_size)
        Fw = np.fft.rfft(wf, grad_fft_size)
        conv_result = np.fft.irfft(Fa * Fw, grad_fft_size)
        return conv_result[n - 1:2 * n - 1][::-1]
