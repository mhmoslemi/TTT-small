import numpy as np
import time


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 8.0)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).flatten()
    except (NameError, AttributeError, ValueError):
        parent = np.ones(6962, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.size
    if n < 2 or not np.all(np.isfinite(parent)):
        return [1.0, 1.0]

    fft_n = 1 << (2 * n - 1).bit_length()

    def fft_conv(x):
        X = np.fft.rfft(x, fft_n)
        return np.fft.irfft(X * X, fft_n)[:2 * n - 1]

    def exact_eval(a):
        return float(evaluate_sequence(a.tolist()))

    S = parent.sum()
    if S < 0.01:
        a_init = np.ones(n)
    else:
        a_init = parent * n / S

    best = a_init.copy()
    C_init = fft_conv(best)
    best_Cmax = C_init.max()
    best_score = 2.0 * best_Cmax / n

    es = exact_eval(best)
    if np.isfinite(es):
        best_score = min(best_score, es)

    def compute_gradient(a, C, top_k):
        top_idx = np.argpartition(C, -top_k)[-top_k:]
        top_vals = C[top_idx]
        w = np.maximum(top_vals - top_vals.min(), 0.0) + 0.01
        w = w ** 3
        w /= w.sum()
        g = np.zeros(n)
        for k_val, wi in zip(top_idx, w):
            k_lo = max(0, k_val - n + 1)
            k_hi = min(n - 1, k_val)
            seg = a[k_val - k_hi: k_val - k_lo + 1][::-1]
            g[k_lo: k_hi + 1] += wi * 2.0 * seg
        g -= g.mean()
        return g

    # Phase 1: Subgradient with restarts (35% budget)
    p1_end = t0 + (deadline - t0) * 0.35

    n_restarts = 4
    per_restart_time = (p1_end - time.monotonic()) / n_restarts

    for restart in range(n_restarts):
        if time.monotonic() > p1_end:
            break

        if restart == 0:
            a = best.copy()
        else:
            scale = 0.05 * (restart + 1)
            a = best + rng.normal(0, scale, n)
            a = np.clip(a, 0.0, 1000.0)
            S_a = a.sum()
            if S_a < 0.01:
                continue
            a *= n / S_a

        C = fft_conv(a)
        v = np.zeros(n)
        momentum = 0.85
        lr = 30.0 / (restart + 1)

        for it in range(30000):
            if time.monotonic() > p1_end:
                break

            g = compute_gradient(a, C, 5)
            v = momentum * v - lr * g
            a_new = a + v
            a_new = np.clip(a_new, 0.0, 1000.0)
            S_new = a_new.sum()
            if S_new < 0.01:
                v *= 0.5
                continue
            a_new *= n / S_new

            C_new = fft_conv(a_new)
            Cmax_new = C_new.max()
            Cmax_old = C.max()

            if Cmax_new <= Cmax_old:
                a = a_new
                C = C_new
                if Cmax_new < best_Cmax:
                    best_Cmax = Cmax_new
                    best = a.copy()
            else:

                a_half = a + 0.5 * v
                a_half = np.clip(a_half, 0.0, 1000.0)
                S_half = a_half.sum()
                if S_half > 0.01:
                    a_half *= n / S_half
                    C_half = fft_conv(a_half)
                    if C_half.max() < Cmax_old:
                        a = a_half
                        C = C_half
                        if C.max() < best_Cmax:
                            best_Cmax = C.max()
                            best = a.copy()

            lr *= 0.999
            momentum = min(0.93, momentum + 0.00005)

            if it % 15000 == 0 and it > 0 and time.monotonic() < p1_end - 5:
                es = exact_eval(a)
                if np.isfinite(es) and es < best_score:
                    best_score = es
                    best = a.copy()

        if S_new < 0.01:
            v *= 0.5
            lr *= 0.9
            continue

            if Cmax_cur < best_Cmax:
