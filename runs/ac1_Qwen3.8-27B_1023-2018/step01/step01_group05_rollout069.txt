def _rfft_convolve_self(a, fft_n):
    A = np.fft.rfft(a, fft_n)
    return np.fft.irfft(A * A, fft_n)[:2 * len(a) - 1]


def _rfft_crosscorr(a, w, fft_n):
    """Compute g[m] = sum_d w[m+d]*a[d] for m=0..n-1 via FFT."""
    n = len(a)
    a_flip = a[::-1]
    W = np.fft.rfft(w, fft_n)
    A = np.fft.rfft(a_flip, fft_n)
    out = np.fft.irfft(W * A, fft_n)
    return out[n - 1:2 * n - 1]


def _project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n_v = len(v)
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    k = np.arange(1, n_v + 1, dtype=np.float64)
    valid = u > (cs - target_sum) / k

    rho = int(np.max(np.where(valid)[0])) + 1
    theta = (cs[rho - 1] - target_sum) / rho
    return np.maximum(v - theta, 0.0)
