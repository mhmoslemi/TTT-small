import time
import math
import numpy as np

INF = float('inf')

def _p2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()

def _is_finite_number(x):
    try:
        y = float(x)
    except Exception:
        return False
    return math.isfinite(y)

def _score_sequence(seq):
    try:
        if not isinstance(seq, list):
            seq = [float(x) for x in seq]
        sc = evaluate_sequence(seq)
    except Exception:
        return INF
    if _is_finite_number(sc):
        return float(sc)
    return INF

def _local_search(a0, seconds, lr0, temp_start, temp_end, deadline):
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0.0, 1000.0)
    n = int(a.size)
    if n <= 0:
        a = np.ones(1, dtype=np.float64)
        n = 1
    def exact(arr):
        try:
            seq = [float(x) for x in arr]
            val = evaluate_sequence(seq)
            if _is_finite_number(val):
                return float(val)
        except Exception:
            pass
        return INF
    holder_arr = a.copy()
    holder_score = exact(holder_arr)
    if not _is_finite_number(holder_score):
        # keep a placeholder but valid
        pass
    target_sum = float(a.sum())
    if not _is_finite_number(target_sum) or target_sum <= 1e-12:
        a = np.ones(n, dtype=np.float64)
        target_sum = float(n)
    target_sum = max(target_sum, 1.0)
    def normalize(x):
        s = float(x.sum())
        if not _is_finite_number(s) or s <= 1e-12:
            return np.ones(n, dtype=np.float64)
        x = x * (target_sum / s)
        return np.clip(x, 0.0, 1000.0)
    a = normalize(a)
    sc = exact(a)
    if sc < holder_score: 
       holder_score = sc; holder_arr = a.copy()
    sizeC = _p2(2*n - 1)
    sizeG = _p2(max(1, 3*n - 2))
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    step = 0
    start = time.monotonic()
    last_eval = start
    eval_every = 1200
    while time.monotonic() < deadline and (time.monotonic() - start) < seconds:
        temp = temp_start + (temp_end - temp_start) * min(1.0, step / 5000.0)
        fa = np.fft.rfft(a, sizeC)
        C = np.fft.irfft(fa * fa, sizeC)[:2*n - 1]
        if not np.all(np.isfinite(C)):
            break
        Cmax = float(np.max(C))
        z = temp * (C - Cmax)
        w = np.exp(z)
        wsum = float(w.sum())
        if not _is_finite_number(wsum) or wsum <= 0.0:
            w = np.full(C.shape, 1.0 / float(C.size), dtype=np.float64)
            wsum = 1.0
        else:
            w = w / wsum
        M = Cmax + math.log(wsum) / temp
        wfl = w[::-1]
        fa2 = np.fft.rfft(a, sizeG)
        fw = np.fft.rfft(wfl, sizeG)
        conv2 = np.fft.irfft(fa2 * fw, sizeG)[:max(1, 3*n - 2)]
        dM = 2.0 * conv2[n - 1:2*n - 1][::-1]

        grad = (2.0 * n) * invS2 * dM - (4.0 * n * M) * invS2 / S

        beta1 = 1.0 - 0.9 ** step
        beta2 = 1.0 - 0.999 ** step

        if beta2 <= 0: beta2=1.0
        mb = m / beta1

        y = float(x)

import time
import math
import numpy as np

INF = float("inf")


def _p2(m: int) -> int:
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _is_finite_number(x) -> bool:
    try:
        y = float(x)
    except Exception:
        return False
    return math.isfinite(y)


def _score_sequence(seq) -> float:
    """Exact score via the supplied evaluator, converted to a finite float or inf."""
    try:
        if not isinstance(seq, list):
            seq = [float(x) for x in seq]
        sc = evaluate_sequence(seq)
    except Exception:
        return INF
    if _is_finite_number(sc):
        return float(sc)
    return INF


def _local_search(a0, seconds, lr0, temp_start, temp_end, deadline):
    """
    Bounded smoothed-max Adam local search for the AC1 surrogate.
    It is only invoked when the exact evaluator reports a candidate clearly
    better than the analytic constant optimum of 2 for the stated evaluator.
    """
    a = np.asarray(a0, dtype=np.float64).ravel().copy()

    n = int(a.size)
    if n <= 0:
        a = np.ones(1, dtype=np.float64)
        n = 1

        try:
            seq = [float(x) for x in arr]
            val = evaluate_sequence(seq)
            if _is_finite_number(val):
                return float(val)
        except Exception:
            pass
        return INF

    holder_arr = a.copy()
    holder_score = exact(holder_arr)

    target_sum = float(a.sum())
    if not _is_finite_number(target_sum) or target_sum <= 1e-12:
        a = np.ones(n, dtype=np.float64)
        target_sum = float(n)
    target_sum = max(target_sum, 1.0)

    def normalize(x):
        s = float(x.sum())
        if not _is_finite_number(s) or s <= 1e-12:
            return np.ones(n, dtype=np.float64)
        x = x * (target_sum / s)
        return np.clip(x, 0.0, 1000.0)

    a = normalize(a)
    sc = exact(a)
    if sc < holder_score:
        holder_score = sc
        holder_arr = a.copy()

    sizeC = _p2(2 * n - 1)

    step = 0
    start = time.monotonic()
    last_eval = start
    eval_every = 1200

    while time.monotonic() < deadline and (time.monotonic() - start) < seconds:
        temp = temp_start + (temp_end - temp_start) * min(1.0, step / 5000.0)

        fa = np.fft.rfft(a, sizeC)
        C = np.fft.irfft(fa * fa, sizeC)[: 2 * n - 1]
        if not np.all(np.isfinite(C)):
            break

        Cmax = float(np.max(C))
        z = temp * (C - Cmax)
        w = np.exp(z)
        wsum = float(w.sum())
        if not _is_finite_number(wsum) or wsum <= 0.0:
            w = np.full(C.shape, 1.0 / float(C.size), dtype=np.float64)
            wsum = 1.0
        else:
            w = w / wsum

        M = Cmax + math.log(wsum) / temp

        wfl = w[::-1]
        fa2 = np.fft.rfft(a, sizeG)
        fw = np.fft.rfft(wfl, sizeG)
        conv2 = np.fft.irfft(fa2 * fw, sizeG)[: max(1, 3 * n - 2)]

        if not _is_finite_number(S) or S <= 1e-12:
            break

        invS2 = 1.0 / (S * S)

        gmax = float(np.max(np.abs(grad))) if n else 0.0

        v = 0.999 * v + 0.001 * (grad * grad)

            beta2 = 1.0

        a = np.clip(a, 0.0, 1000.0)
        a = normalize(a)
