    p = np.clip(p, 0.0, 1000.0)
    if p.sum() > 1e-6:
        inits.append(p / p.sum())
    
    j = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n) / n)  # uniform
    
    cw = np.sin(np.pi * (j + 0.5) / n) ** 2
    inits.append(cw / cw.sum())  # cosine
    
    for gap_frac in [0.03, 0.06, 0.10, 0.15]:
        gap = max(10, int(gap_frac * n))
        tb = np.ones(n)
        lo = max(0, n // 2 - gap // 2)
        hi = min(n, n // 2 + gap // 2 + 1)
        tb[lo:hi] = 0.0
        if tb.sum() > 1e-6:
            inits.append(tb / tb.sum())  # two-bump
    
    for frac in [7, 10, 15]:
        sig = n / frac
        g = np.exp(-0.5 * ((j - (n - 1) / 2.0) / sig) ** 2)
        inits.append(g / g.sum())  # gaussian
    
    # Smooth random
    for _ in range(2):
        r = rng.random(n)
        # IIR smoothing (simple)
        sm = np.empty(n)
        sm[0] = r[0]
        alpha = 0.1
        for i in range(1, n):
            sm[i] = alpha * r[i] + (1 - alpha) * sm[i - 1]
        sm = np.maximum(sm, 1e-6)
        inits.append(sm / sm.sum())
    
    # Pick best initial
    best_a = inits[0].copy()
    best_score = fast_score(inits[0])
    for init in inits[1:]:
        s = fast_score(init)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = init.copy()
    
    # Validate with exact evaluator
    best_exact_a = best_a.copy()
    best_exact = float(evaluate_sequence(best_exact_a.tolist()))
    if not np.isfinite(best_exact):
        best_exact = 2.0
        best_exact_a = (np.ones(n) / n).copy()
    
    # --- Main optimization loop ---
    a = best_a.copy()
    eta = 0.5
    no_improve = 0
    iter_count = 0
    top_frac = 0.15
    lp_counter = 0
    
    while time.monotonic() < deadline:
        iter_count += 1
        lp_counter += 1
        
        # Compute convolution
        conv = fft_conv(a)
        
        # Soft hot-set weighting
        k_top = max(20, int(top_frac * conv_len))
        part = np.partition(conv, conv_len - k_top)
        thresh = part[conv_len - k_top]
        spread = conv.max() - thresh
        if spread > 1e-14:
            weights = 1.0 + np.clip(conv - thresh, 0.0, spread) / spread
        else:
            weights = np.ones(conv_len)
        
        # Load computation (correlation)
        load = np.correlate(weights, a, 'valid')
        lm = load.max()
        if lm > 1e-15:
            a = a * np.exp(-eta * load / lm)
        
        a = normalize(a)
        
        # Score tracking
        if iter_count % 5 == 0:
            s = fast_score(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Small perturbation every 80 iters
        if iter_count % 80 == 0:
            n_p = max(3, n // 80)
            idx = rng.choice(n, size=n_p, replace=False)
            sigma = 0.03 + 0.04 * min(no_improve / 80.0, 2.0)
            a[idx] *= np.exp(rng.normal(0.0, sigma, size=n_p))
            a = normalize(a)
        
        # LP refinement every 400 iterations
        if lp_counter >= 400 and iter_count > 50:
            lp_counter = 0
            M = conv.max()
            L = 40  # number of hot constraints
            top_idx = np.argpartition(conv, -L)[-L:]
            # Build constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n
            A_lp = np.zeros((L, n))
            for ii, k in enumerate(top_idx):
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                A_lp[ii, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
            c_obj = -np.ones(n)
            try:
                res = linprog(c_obj, A_ub=A_lp, b_ub=M * np.ones(L),
                              bounds=[(0, None)] * n, method='highs',
                              options={'time_limit': 10})
                if res.success and res.x is not None:
                    b_lp = np.maximum(res.x, 0)
                    if b_lp.sum() > 1e-10:
                        b_lp /= b_lp.sum()
                        blend = 0.15
                        a = (1 - blend) * a + blend * b_lp
                        a = normalize(a)
            except Exception:
                pass
        
        # Large perturbation / restart
        if no_improve >= 300:
            r = rng.random()
            if r < 0.20 and p.sum() > 1e-6:
                a = np.clip(p, 0, 1000).copy()
                a = a / a.sum()
            elif r < 0.40:
                blend = rng.uniform(0.2, 0.6)
                a = (1 - blend) * a + blend * a[::-1]
                a = normalize(a)
            elif r < 0.60:
                gap2 = rng.integers(n // 25, n // 6)
                tb = np.ones(n)
                lo2 = max(0, n // 2 - gap2 // 2)
                hi2 = min(n, n // 2 + gap2 // 2 + 1)
                tb[lo2:hi2] = 0.0
                a = 0.4 * a + 0.6 * tb / tb.sum()
                a = normalize(a)
            else:
                n_p2 = max(10, n // 8)
                idx2 = rng.choice(n, size=n_p2, replace=False)
                a[idx2] *= np.exp(rng.normal(0.0, 0.35, size=n_p2))
                a = normalize(a)
            no_improve = 0
            eta = min(1.5, eta * 1.08)
        
        # Anneal step size
        if iter_count % 2000 == 0:
            eta = max(0.08, eta * 0.90)
            top_frac = max(0.04, top_frac * 0.90)
        
        # Exact validation every 120 iters
        if iter_count % 120 == 0:
            es = float(evaluate_sequence(a.tolist()))
            if np.isfinite(es) and es < best_exact:
                best_exact = es
                best_exact_a = a.copy()
        
        # Hard time check
        if time.monotonic() > deadline - 4.0:
            break
    
    # --- Ensure we have a valid best ---
    fs = float(evaluate_sequence(best_exact_a.tolist()))
    if not np.isfinite(fs) or fs > 2.0:
        best_exact_a = np.ones(n) / n
        best_exact = float(evaluate_sequence(best_exact_a.tolist()))
    
    # One final check: also validate the current a
    fs2 = float(evaluate_sequence(a.tolist()))
    if np.isfinite(fs2) and fs2 < best_exact:
        best_exact = fs2
        best_exact_a = a.copy()
    
    result = [float(x) for x in best_exact_a]
    return result
