def _safe_eval(seq):

def _norm(a, floor=1e-18):
    if a is None:
        return None
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, floor)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    arr = arr / s
    arr = np.minimum(arr, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return arr / s


def _fft_size(n):
    m = max(1, 3 * int(n) - 2)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _conv_f(a, fftn):
    F = np.fft.rfft(a, fftn)
    c = np.fft.irfft(F * F, fftn)
    c = c[:2 * len(a) - 1]
    if c.size and c.min() < 0.0:
        c = c.copy()
        c[c < 0.0] = 0.0
    return c, F


def _load_from_F(F, a, w, fftn):
    W = np.fft.rfft(w[::-1].copy(), fftn)
    full = np.fft.irfft(F * W, fftn)
    n = a.size
    if n == 1:
        load = full[:1]
    else:
        load = full[2 * n - 2:n - 2:-1].copy()
    if load.size and load.min() < 0.0:
        load[load < 0.0] = 0.0
    return load


def _score_array(a):
    a = _norm(a)

    n = a.size
    if n == 1:

def _downsample(a, n_out):
    n_out = int(n_out)
    if n_out <= 0:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    if a.size == n_out:
        return a
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n_out)
    return np.interp(x_new, x_old, a)


def _upsample(a, n_out):
    return _downsample(a, n_out)


def _random_param(n, rng):
    n = int(n)
    if n <= 1:
        return np.ones(n, dtype=np.float64)

    x = np.linspace(0.0, 1.0, n)
    xc = np.clip(x, 1e-10, 1.0 - 1e-10)
    kind = int(rng.integers(0, 5))

    if kind == 0:
        p = float(np.clip(abs(rng.normal(0.0, 1.3)), 0.0, 6.0))
        q = float(np.clip(abs(rng.normal(0.0, 1.3)), 0.0, 6.0))
        f = np.power(xc, p) * np.power(1.0 - xc, q)

    elif kind == 1:
        y = 2.0 * x - 1.0
        K = int(rng.integers(3, 11))
        th = np.sqrt(1.0 / K) * rng.normal(size=K)

        T1 = y.copy()
        terms = th[0] * T1
        for k in range(2, K):
            T2 = 2.0 * y * T1 - T0
            terms = terms + th[k - 1] * T2
            T0, T1 = T1, T2
        f = np.exp(terms)

    elif kind == 2:
        f = 0.25 * np.ones_like(x)
        m = int(rng.integers(2, 6))

            width = 0.04 + 0.28 * float(rng.random())

    else:
        y = 2.0 * x - 1.0
        K = int(rng.integers(3, 9))
        coeffs = rng.normal(size=K) * 0.45
        T0 = np.ones_like(x)

    s = float(f.sum())

def _deterministic_shapes(n):
    out = []
    n = int(n)
    if n <= 1:
