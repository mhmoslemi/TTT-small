        S = x.sum()
        M = B.max()
        cur_sc = 2.0 * n * M / (S * S)

        if np.isfinite(cur_sc) and cur_sc < best_sc - 1e-12:
            best = x.copy()
            best_sc = cur_sc
            last_improve = it
            stagnant = 0
        else:
            stagnant += 1

        # Stagnation: inject structured perturbation
        if stagnant > PATIENCE:
            pert_type = stagnant % 3
            if pert_type == 0:
                x = x + 0.04 * rng.standard_normal(n)
            elif pert_type == 1:
                x = 0.95 * x + 0.05 * rng.random(n)
            else:
                # Flatten perturbation (reduce variance)
                x_dev = x - x.mean()
                x = x - 0.1 * x_dev + 0.02 * rng.standard_normal(n)
            x = np.maximum(x, 0.0)
            x = x * n / max(x.sum(), 0.01)
            v *= 0.1
            stagnant = 0
            continue

        # Adaptively increase beta
        progress = min(1.0, it / 50000.0)
        beta = 30.0 + 5000.0 * progress
        beta = min(beta, 8000.0)

        # Compute smooth max weights
        B_max = B.max()
        B_shift = B - B_max
        B_shift = np.maximum(B_shift, -30.0 / max(beta, 1.0))
        eB = np.exp(beta * B_shift)
        eB_sum = eB.sum()

        if not np.isfinite(eB_sum) or eB_sum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(B))] = 1.0
            M_smooth = B_max
        else:
            w = eB / eB_sum
            M_smooth = B_max + np.log(eB_sum) / beta

        # Gradient of smooth max: c_j = sum_m a_m * w[j+m]
        c = get_cross_corr(x, w)

        # Direction in score (projected to sum-preserving)
        g = S * c - 2.0 * M_smooth
        g = g - g.mean()

        gnorm = np.sqrt(np.dot(g, g))
        if gnorm < 1e-15:
            break

        # Nesterov momentum
        v = mu_mom * v + g
        direction = -v
        dnorm = np.sqrt(np.dot(direction, direction))
        if dnorm < 1e-15:
            break

        # Adaptive step size
        step = 0.15 / dnorm

        # Backtracking line search
        accepted = False
        for ls in range(12):
            x_try = x + step * direction
            x_try = np.maximum(x_try, 0.0)
            x_try = np.minimum(x_try, 1000.0)
            S_try = x_try.sum()
            if S_try < 0.01:
                step *= 0.5
                continue
            x_try = x_try * (n / S_try)
            B_try = get_conv(x_try)
            sc_try = 2.0 * B_try.max() / n
            if np.isfinite(sc_try) and sc_try < cur_sc + 1e-12:
                x = x_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            # Very small step
            x = x + 1e-7 * direction
            x = np.maximum(x, 0.0)
            x = x * n / max(x.sum(), 0.01)

        # Periodic exact evaluation
        if it % 300 == 0 and time.monotonic() < deadline - 30:
            exact = float(evaluate_sequence(x.tolist()))
            if np.isfinite(exact) and exact < best_sc:
                best = x.copy()
                best_sc = exact

    # --- Final validation and return ---
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = np.ones(n)
        final_sc = float(evaluate_sequence(best.tolist()))

    # Ensure all values are valid
    result = [float(max(0.0, min(1000.0, v))) for v in best]
    # Final sanity check
    sanity_sc = float(evaluate_sequence(result))
    if not np.isfinite(sanity_sc):
        return [1.0] * n
    return result
