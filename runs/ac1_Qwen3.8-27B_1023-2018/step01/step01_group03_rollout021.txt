Now let me think about one more thing: after taking a sub-gradient step and clipping to non-negativity, the sum might not be 1. I re-normalize by dividing by the sum. This is a projection onto the positive simplex (approximately; exact projection onto the simplex is more involved, but clipping + normalizing is a common approximation).

The exact projection onto the positive simplex {a ≥ 0, sum(a) = 1} is:
1. a_i^' = a_i - τ, where τ is chosen so that sum(a') = 1 and a' ≥ 0
2. This is equivalent to: find τ such that sum(max(a_i - τ, 0)) = 1

My approach (clip to 0, then divide by sum) is:
1. a_i^' = max(a_i, 0)
2. a_i^' = a_i^' / sum(a_i^')

This is NOT the exact simplex projection, but it's a valid projection onto the positive orthant followed by scaling to the sum=1 hyperplane. It's a common practical approximation.

For better accuracy, I could implement the exact projection:

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) - (cv - 1) > 0)[0][-1]
    theta = (cv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

import numpy as np
import time
from scipy.optimize import linprog


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    start_time = time.monotonic()
    budget_s = max(30.0, float(budget_s))
    deadline = start_time + budget_s - 25.0
    rng = np.random.default_rng(seed)
    best_list = None
    best_score = float('inf')

    def consider(seq):
        nonlocal best_list, best_score
        try:
            arr = np.asarray(seq, dtype=np.float64)
            if arr.ndim != 1 or arr.size < 1:
                return
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            s = float(arr.sum())
            if s < 0.01:
                return
            lst = [float(x) for x in arr]
            v = float(evaluate_sequence(lst))
            if np.isfinite(v) and v < best_score:
                best_score = v
                best_list = lst
        except Exception:
            pass

    # Load parent
    parent_q = None
    n = 512
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 1:
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.clip(p, 0.0, 1000.0)
            ps = float(p.sum())
            if ps >= 0.01:
                parent_q = p / ps
                n = p.size
                consider(parent_q)
    except Exception:
        pass
    if parent_q is None:
        parent_q = np.ones(n) / n
        consider(parent_q)

    def build_conv_mat(q):
        nq = q.size
        A = np.zeros((2 * nq - 1, nq))
        for j in range(nq):
            A[j:j + nq, j] = q
        return A

    def solve_lp(q, M, timeout=25):
        nq = q.size
        A = build_conv_mat(q)
        try:
            res = linprog(-np.ones(nq), A_ub=A, b_ub=np.full(A.shape[0], M),
                          bounds=[(0.0, 1e6)] * nq, method='highs',
                          options={'time_limit': timeout})
            if res.success:
                g = np.maximum(res.x, 0.0)
                gs = float(g.sum())
                if gs > 1e-10:
                    return g / gs
        except Exception:
            pass
        return None

    def subgrad_dir(q):
        c = np.convolve(q, q)
        Mv = float(c.max())
        if Mv < 1e-18:
            return np.zeros(q.size), Mv
        idx = np.where(c >= 0.9 * Mv)[0]
        w = np.exp((c[idx] - Mv) * 25.0)
        ws = float(w.sum())
        if ws < 1e-10:
            w = np.ones(idx.size) / idx.size
        else:
            w = w / ws
        sg = np.zeros(q.size)
        for ki, wi in zip(idx, w):
            ki = int(ki)
            lo = max(0, ki - q.size + 1)
            hi = min(q.size - 1, ki)
            if lo <= hi:
                sg[lo:hi + 1] += wi * 2.0 * q[ki - hi:ki - lo + 1][::-1]
        sg -= sg.mean()
        return sg, Mv

    def optimize(q0, dl, max_it):
        q = np.clip(q0.astype(np.float64).copy(), 0.0, None)
        qs = float(q.sum())
        if qs < 1e-10:
            return q, float('inf')
        q = q / qs
        conv = np.convolve(q, q)
        M = float(conv.max())
        sc = 2.0 * q.size * M
        bq = q.copy()
        bsc = sc
        stg = 0
        for _ in range(max_it):
            if time.monotonic() > dl - 12:
                break
            imp = False
            # LP step
            g = solve_lp(q, M)
            if g is not None:
                for t in (0.5, 0.3, 0.7, 1.0, 0.2, 0.4, 0.8, 0.6, 0.9, 0.15):
                    cd = (1.0 - t) * q + t * g

                    cs2 = float(cd.sum())
                    if cs2 < 1e-10:
                        continue
                    cd = cd / cs2
                    cc = np.convolve(cd, cd)
                    cm = float(cc.max())
                    ns = 2.0 * q.size * cm
                    if ns < sc - 1e-15:
                        q, sc, conv, M = cd, ns, cc, cm
                        imp = True
                        break
            # Sub-gradient step
            sg, Mv = subgrad_dir(q)
            sn = float(np.linalg.norm(sg))
            if sn > 1e-13:
                for st in (0.05, 0.02, 0.1, 0.01, 0.08, 0.15, 0.03):
                    cd = q - (st / sn) * sg
                    cd = np.clip(cd, 0.0, None)
                    cs2 = float(cd.sum())
                    if cs2 < 1e-10:
                        continue
                    cd = cd / cs2
                    cc = np.convolve(cd, cd)
                    cm = float(cc.max())
                    ns = 2.0 * q.size * cm
                    if ns < bsc - 1e-14:
                        bsc, bq = ns, cd.copy()
                        q, sc, conv, M = cd, ns, cc, cm
                        imp = True
                        break
            if imp:
                stg = 0
                if sc < bsc:
                    bsc, bq = sc, q.copy()
            else:
                stg += 1
                if stg >= 50:
                    break
            M = float(conv.max())
        return bq, bsc

    # Phase 1: Optimize parent
    q1, s1 = optimize(parent_q, deadline, 3000)
    consider(q1)

    # Phase 2: Noisy restarts from best
    for r in range(12):

                     time.monotonic() + max(25.0, (deadline - 25 - time.monotonic()) / max(1, rem_starts)))
        sig = 0.01 + 0.008 * r
        noise = rng.normal(0, sig, q1.size)
        qp = np.clip(q1 + noise, 0.0, None)
        qps = float(qp.sum())
        if qps < 1e-10:

        qp /= qps
        qr, sr = optimize(qp, per_dl, 1000)
        consider(qr)

    # Phase 3: Structural initializations
    if time.monotonic() < deadline - 50:
        t = np.arange(n, dtype=np.float64)
        inits = []
        tri = 1.0 - np.abs(t - (n - 1) / 2.0) / ((n - 1) / 2.0 + 1.0)
        inits.append(np.clip(tri, 0.01, None))
        cosb = 0.5 + 0.5 * np.cos(2.0 * np.pi * t / n)
        inits.append(np.clip(cosb, 0.01, None))
        cos2 = 0.7 + 0.3 * np.cos(4.0 * np.pi * t / n)
        inits.append(np.clip(cos2, 0.01, None))
        for ini in inits:
            if time.monotonic() > deadline - 30:
                break
            qi = ini / float(ini.sum())
            per_dl = min(deadline - 15, time.monotonic() + 100)
            qi_opt, si_opt = optimize(qi, per_dl, 800)
            consider(qi_opt)

    # Phase 4: Interpolation between parent and optimized
    if best_list is not None and time.monotonic() < deadline - 20:
        qb = np.asarray(best_list, dtype=np.float64)
        qb = np.clip(qb, 0.0, None)
        qbs = float(qb.sum())
        if qbs > 0:
            qb /= qbs
            if qb.size == n:
                for alpha in (0.3, 0.5):
                    if time.monotonic() > deadline - 10:
                        break
                    mix = (1 - alpha) * parent_q + alpha * qb
                    mix = np.clip(mix, 0.0, None)
                    ms = float(mix.sum())
                    if ms > 1e-10:
                        mix /= ms
                        qm, sm = optimize(mix, deadline, 500)
                        consider(qm)

    # Phase 5: Final polish
    if best_list is not None and time.monotonic() < deadline - 15:
        qb = np.asarray(best_list, dtype=np.float64)
        qb = np.clip(qb, 0.0, None)
        qbs = float(qb.sum())
        if qbs > 0:
            qb /= qbs
            qf, sf = optimize(qb, deadline, 600)
            consider(qf)

    if best_list is None:
        consider(parent_q)
    if best_list is None:
        best_list = [1.0 / n] * n

    try:
        v = float(evaluate_sequence(best_list))
        if not np.isfinite(v):
            raise ValueError()
    except Exception:
        best_list = [1.0 / n] * n

    res = [float(x) for x in best_list]
    return res
