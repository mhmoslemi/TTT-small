    # Convolution via FFT
    def convolve(a):
        A = np.fft.rfft(a, fft_n)
        C = np.fft.irfft(A * A, fft_n)
        return C[:2*n-1]
    
    def compute_grad_topk(a, conv, top_k):
        """Compute gradient using top-k lags."""
        n = len(a)
        # Find top k lags
        if top_k >= 2*n - 1:
            lags = np.arange(2*n-1)
            weights = np.ones(2*n-1)
        else:
            # Partition to get top k
            part = np.argpartition(conv, -top_k)[-top_k:]
            lags = part
            # Soft weight based on proximity to max
            max_c = conv.max()
            min_c = conv[lags].min()
            spread = max_c - min_c
            if spread > 1e-14:
                weights = (conv[lags] - min_c) / spread
            else:
                weights = np.ones(top_k)
            weights /= weights.sum()
        
        grad = np.zeros(n)
        for k, w in zip(lags, weights):
            # grad[j] += w * a[k-j] if 0 <= k-j < n, and w * a[k-j] if 0 <= j-k < n (symmetric term)
            j = np.arange(n)
            t1 = k - j
            mask1 = (t1 >= 0) & (t1 < n)
            grad[mask1] += w * a[t1[mask1]]
            # The symmetric term: for the quadratic form a^T T_k a, the gradient is 2*T_k*a
            # T_k[i,j] = 1 if i+j=k (linear conv), so (T_k a)[j] = a[k-j] if 0<=k-j<n
            # And the gradient is 2*(T_k a) since T_k is symmetric
            # But we already have the factor of 2 in the weight... 
            # Actually, let me just use the factor 2 in the update step
        return grad
    
    # Actually, the gradient of (a*a)[k] w.r.t. a is 2*a[max(0,k-j)] for valid indices
    # For the "interior" (where both j and k-j are in [0,n-1]), it's 2*a[k-j]
    # For "boundary" (where only one is valid), it's a[k-j]
    # But since we're using subgradient of the MAX, we just need one valid subgradient
    # The simplest: for the quadratic form, grad = 2 * T_k @ a where T_k is the (0,1) matrix
    # T_k[i,j] = 1 iff i+j = k
    # (T_k @ a)[j] = sum_i T_k[i,j] a[i] = a[k-j] if 0 <= k-j < n else 0
    # So grad[j] = 2 * a[k-j] if 0 <= k-j < n else 0
    
    # For multiple lags with weights:
    # grad[j] = sum_k w_k * 2 * a[k-j] (where valid)
    
    iter_count = 0
    eta = 0.5
    top_k = 50
    patience = 0
    lp_freq = 500  # solve LP every 500 iterations
    last_lp_time = t_start
    
    while time.monotonic() < deadline:
        iter_count += 1
        t_now = time.monotonic()
        
        # Compute convolution
        conv = convolve(a)
        max_conv = conv.max()
        
        # Estimate score
        est_score = 2.0 * n * max_conv  # sum=1
        
        # Subgradient update
        # Find top-k lags
        k_eff = min(top_k, 2*n-1)
        top_lags = np.argpartition(conv, -k_eff)[-k_eff:]
        
        # Compute weighted gradient
        j = np.arange(n)
        grad = np.zeros(n)
        for k in top_lags:
            # Weight proportional to conv[k]
            t = k - j
            mask = (t >= 0) & (t < n)
            grad[mask] += conv[k] * a[t[mask]]
        # Also add the symmetric contribution (since it's a quadratic form)
        # Actually, for the max subgradient, we just need ONE valid subgradient
        # The gradient at the argmax is sufficient
        # But using all top-k with weights gives a smoother direction
        
        grad_max = grad.max()
        if grad_max > 1e-18:
            a = a * np.exp(-eta * grad / grad_max)
        
        a = np.maximum(a, 1e-12)
        a = a / a.sum()
        
        # Check score periodically
        if iter_count % 50 == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
        
        # Periodic LP solve
        if iter_count % lp_freq == 0 and time.monotonic() - last_lp_time > 10:
            # Try LP best-response
            M = max_conv
            # Build constraint matrix (sparse)
            # B[k, i] = a[i] if 0 <= k-i < n
            # Use COO format
            rows = []
            cols = []
            data = []
            for i in range(n):
                # k ranges from i to min(i+n-1, 2n-2)
                k_start = i
                k_end = min(i + n - 1, 2*n - 2)
                if k_start <= k_end:
                    rows.extend(range(k_start, k_end + 1))
                    cols.extend([i] * (k_end - k_start + 1))
                    data.extend([a[i]] * (k_end - k_start + 1))
            
            from scipy.sparse import coo_matrix
            B = coo_matrix((data, (rows, cols)), shape=(2*n-1, n))
            
            # Solve: max sum(g) s.t. B@g <= M, g >= 0
            c_obj = -np.ones(n)  # minimize -sum(g)
            A_ub = B
            b_ub = np.full(2*n-1, M)
            bounds = [(0, None)] * n
            
            from scipy.optimize import linprog
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            
            if res.success:
                g = res.x
                g_sum = g.sum()
                if g_sum > 1e-6:
                    g /= g_sum
                    # Blend
                    t_blend = 0.1  # small blend
                    a_new = (1 - t_blend) * a + t_blend * g
                    a_new = np.maximum(a_new, 1e-12)
                    a_new = a_new / a_new.sum()
                    
                    sc_new = float(evaluate_sequence(a_new.tolist()))
                    if np.isfinite(sc_new) and sc_new < best_score:
                        best_score = sc_new
                        best_a = a_new.copy()
                    a = a_new  # accept regardless (for exploration)
            
            last_lp_time = time.monotonic()
        
        # Perturbations
        if iter_count % 1000 == 0:
            n_pert = max(10, n // 100)
            idx = rng.choice(n, size=n_pert, replace=False)
            a[idx] *= np.exp(rng.normal(0, 0.15, size=n_pert))
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
        
        # Larger perturbation if stuck
        if patience >= 100:
            n_pert = max(10, n // 20)
            idx = rng.choice(n, size=n_pert, replace=False)
            a[idx] *= np.exp(rng.normal(0, 0.4, size=n_pert))
            a = np.maximum(a, 1e-12)
