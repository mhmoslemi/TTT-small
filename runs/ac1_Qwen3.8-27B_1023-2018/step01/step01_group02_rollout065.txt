import numpy as np

def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(z.shape, 1.0 / float(z.size), dtype=np.float64)
    return e / s


def _fft_conv(a, b):
    size = len(a) + len(b) - 1
    if size <= 1:
        return np.array([float(a[0]) * float(b[0])])
    L = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    out = np.fft.irfft(A * B, L)
    return out[:size]


def _exact_score(c, n, S):
    if not np.isfinite(S) or S <= 1e-300:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return 2.0 * float(n) * mx / (S * S)


def _grad_z(z, beta, n):
    """Return (a, c, S, exact_score, grad_z) for a = softmax(z)."""
    a = _softmax(z)
    c = _fft_conv(a, a)
    S = float(a.sum())

    bc = float(beta) * c
    bc_max = float(np.max(bc))
    e = np.exp(bc - bc_max)
    es = float(e.sum())
    if not np.isfinite(es) or es <= 1e-300:
        es = 1e-300
        e = np.full_like(e, 1.0 / e.size)
    p = e / es
    LSE_x = (bc_max + float(np.log(es))) / float(beta)

    score = _exact_score(c, n, S)

    # r_i = sum_k p_k * a[k-i]   (zero extension)
    r = np.correlate(p, a, mode="full")[n - 1: 2 * n - 1]

    grad_z = a * (grad_f - float(np.dot(a, grad_f)))
    grad_z = np.where(np.isfinite(grad_z), grad_z, 0.0)
    return a, c, S, score, grad_z


def _adam(params, grad, state, lr, cap=2.0):
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = state["t"] + 1
    state["t"] = t
    state["m"] = b1 * state["m"] + (1.0 - b1) * grad
    state["v"] = b2 * state["v"] + (1.0 - b2) * (grad * grad)
    mhat = state["m"] / (1.0 - b1 ** t)
    vhat = state["v"] / (1.0 - b2 ** t)
    step = float(lr) * mhat / (np.sqrt(vhat) + eps)
    step = np.where(np.isfinite(step), step, 0.0)
    if cap is not None:
        step = np.clip(step, -float(cap), float(cap))
    return params - step, state

    z = np.where(np.isfinite(z), z, 0.0)

def _interpolate(src, dst_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == 0:
        return np.ones(dst_n, dtype=np.float64) / dst_n

        return src.copy()
    x1 = np.linspace(0.0, 1.0, dst_n)
