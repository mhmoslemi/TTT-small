        q = np.clip(q, 0.0, 1000.0)
        s = float(q.sum())
        if s < 1e-15:
            return np.full(q.size, 1.0 / q.size)
        return q / s

        if n == 0:
            return float('inf')

        s = float(q.sum())
        if s < 0.01:
            return float('inf')
        v = 2.0 * n * float(b.max()) / (s * s)

            hi = min(k, n - 1)
            A[k, lo:hi + 1] = q[lo:hi + 1][::-1]

            return None

            return None

        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = (np.zeros(n), np.full(n, 1000.0))

                          method='highs', options={'presolve': True})

        a = a0.copy()
