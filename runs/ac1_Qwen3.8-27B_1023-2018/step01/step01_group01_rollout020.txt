        if sc < holder[1]:
            holder[0], holder[1] = a.copy(), sc

        while time.monotonic() < hard_deadline and (time.monotonic() - start) < time_limit:
            S = float(a.sum())
            fa = np.fft.rfft(a, sizeC)
            C = np.fft.irfft(fa * fa, sizeC)[: 2 * n - 1]
            Cmax = float(C.max())
            
            # Use top-k max indices (k=5 for robustness)
            k = min(5, 2 * n - 1)
            top_idx = np.argpartition(C, -k)[-k:]
            top_idx = top_idx[np.argsort(C[top_idx])[::-1]]
            
            # Average subgradient over top-k
            dM = np.zeros(n, dtype=np.float64)
            for j in top_idx:
                lo = max(0, j - n + 1)
                hi = min(n - 1, j)
                idx = np.arange(lo, hi + 1)
                dM[idx] += 2.0 * a[j - idx]
            dM /= len(top_idx)
            
            M = Cmax
            invS2 = 1.0 / (S * S)
            grad = (2.0 * n) * invS2 * dM - (4.0 * n * M) * invS2 / S
            
            lr = lr0 * (0.999 ** (int((time.monotonic() - start) * 100)))
            a = a - lr * grad
            a = np.clip(a, 0.0, 1000.0)
            a = norm_a(a)
            
            if (time.monotonic() - start) % 5.0 < 0.05:
                sc = safe_eval(a)
                if sc < holder[1]:
                    holder[0], holder[1] = a.copy(), sc

        sc = safe_eval(a)
        if sc < holder[1]:
            holder[0], holder[1] = a.copy(), sc
        return a

    # --- seeds ---
    seeds = [base.copy()]
    # Symmetric windows
    x = np.linspace(0, np.pi, n)
    seeds.append(0.4 + 0.6 * np.cos(x))
    seeds.append(0.54 + 0.46 * np.cos(2 * np.pi * np.arange(n) / max(1, n - 1)))
    seeds.append(0.5 - 0.5 * np.cos(2 * np.pi * np.arange(n) / (n + 1)))
    # Flat-top
    xx = 2 * np.pi * np.arange(n) / n
    seeds.append(0.2156 - 0.4166 * np.cos(xx) + 0.2773 * np.cos(2 * xx)
                 - 0.0836 * np.cos(3 * xx) + 0.0069 * np.cos(4 * xx))
    # Slightly asymmetric
    seeds.append(np.linspace(0.25, 1.0, n))
    seeds.append(np.linspace(1.0, 0.25, n))
    # Broad Gaussian-like
    xc = np.arange(n) / n - 0.5
    seeds.append(np.exp(-4.0 * xc * xc))
    # Constant
    seeds.append(np.ones(n, dtype=np.float64))

    clean = []
    for s in seeds:
        s = np.clip(np.array(s, dtype=np.float64), 0.0, 1000.0)
        if s.sum() > 0.01:
            clean.append(norm_a(s))

    # Rank by fast score
    scored = sorted([(fft_score(s), s) for s in clean], key=lambda t: t[0])[:6]

    # --- Phase 1: multi-seed exploration ---
    avail = max(30.0, hard_deadline - time.monotonic())
    per = (0.45 * avail) / max(1, len(scored))
    for i, (_, s0) in enumerate(scored):
        if time.monotonic() > hard_deadline:
            break
        optimize(s0, per, 0.02, 8.0, 40.0, f"seed{i}")

    # --- Phase 2: medium polish ---
    if time.monotonic() < hard_deadline:
        remain = hard_deadline - time.monotonic()
        optimize(holder[0].copy(), 0.30 * remain, 0.01, 40.0, 120.0, "polish1")

    # --- Phase 3: fine polish ---
    if time.monotonic() < hard_deadline:
        remain = hard_deadline - time.monotonic()
        optimize(holder[0].copy(), 0.50 * remain, 0.004, 120.0, 400.0, "polish2")

    # --- Phase 4: subgradient polish ---
    if time.monotonic() < hard_deadline:
        remain = hard_deadline - time.monotonic()
        subgrad_polish(holder[0].copy(), remain * 0.6, 0.008, "sub")

    # --- Phase 5: final micro-polish ---
    if time.monotonic() < hard_deadline:
        remain = hard_deadline - time.monotonic()
        optimize(holder[0].copy(), remain, 0.002, 400.0, 800.0, "final")

    # --- Final validation ---
    best_a = norm_a(np.clip(np.array(holder[0], dtype=np.float64), 0.0, 1000.0))
    fs = safe_eval(best_a)
    if not np.isfinite(fs) or fs > holder[1] + 1e-6:
        best_a = holder[0]
        fs = safe_eval(best_a)
    if not np.isfinite(fs):
        best_a = np.clip(base, 0.0, 1000.0)
        fs = safe_eval(best_a)
    if not np.isfinite(fs):
        best_a = np.ones(n, dtype=np.float64)

    out = [float(x) for x in best_a]
    if not out or any(not np.isfinite(x) for x in out):
        out = [1.0] * n
    return out
