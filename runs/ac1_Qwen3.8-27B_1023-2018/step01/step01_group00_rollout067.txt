    if time.monotonic() < deadline - 80 and best_score > 1.5028:
        try:
            a_cur = best_a.copy()
            b_cur = _fft_conv(a_cur, fsize)
            M_cur = float(b_cur.max())
            g_dual = _lp_dual_step(a_cur, M_cur, n, time.monotonic() + 35.0)
            if g_dual is not None and np.isfinite(g_dual).all() and g_dual.sum() > 1e-10:
                g_dual = g_dual / g_dual.sum()
                for tf in (0.3, 0.5, 0.7, 0.9, 1.0):
                    if time.monotonic() > deadline - 40:
                        break
                    a_mix = (1.0 - tf) * a_cur + tf * g_dual
                    a_mix = np.maximum(a_mix, 0.0)
                    sm = a_mix.sum()
                    if sm < 1e-30:
                        continue
                    a_mix = a_mix / sm
                    a_mix = _coord_descent(a_mix, n, fsize, time.monotonic() + 18.0, max_passes=20)
                    try_update(a_mix)
        except Exception:
            pass

    # Phase 5: final polished coordinate descent
    if time.monotonic() < deadline - 20:
        final_alloc = max(5.0, deadline - time.monotonic() - 10.0)
        try_update(_coord_descent(best_a, n, fsize, time.monotonic() + final_alloc, max_passes=1500))

    # Final exact validation of the returned candidate
    out = np.clip(best_a, 0.0, 1000.0)
    fs = _exact_eval([float(x) for x in out])
    if not np.isfinite(fs):
        out = np.clip(parent, 0.0, 1000.0)
        fs = _exact_eval([float(x) for x in out])
    if not np.isfinite(fs):
        return [1.0]
    return [float(x) for x in out]
