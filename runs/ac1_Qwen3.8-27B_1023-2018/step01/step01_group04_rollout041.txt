```python
    p = p_init.copy()
    p = _normalize(p)
    best_p = p.copy()
    best_score = _exact_score(p)
    
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        
        beta = beta_schedule[it]
        b = np.convolve(p, p)
        bmax = np.max(b)
        
        # Compute weights
        shifted = beta * (b - bmax)
        shifted = np.clip(shifted, -500, 0)  # numerical safety
        exp_vals = np.exp(shifted)
        sum_exp = np.sum(exp_vals)
        w = exp_vals / sum_exp
        
        # Objective (smooth)
        obj = 2.0 * n * (bmax + np.log(sum_exp) / beta)
        
        # Gradient
        conv_pw = np.convolve(p, w[::-1])
        g_conv = conv_pw[n-1:2*n-1][::-1]
        grad = 2.0 * n * (g_conv + p)
        
        # Projected gradient step with backtracking
        alpha = 1.0 / (beta * 2.0 * n * 2.0)  # initial step size estimate
        for _ in range(30):

            p_new = _simplex_project(p_new)
            sc = _exact_score(p_new)

                break
            alpha *= 0.5
        
        p = p_new
        sc = _exact_score(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()
    
    return best_p, best_score
```

**Structural insight**: The problem is equivalent to finding a probability vector p that minimizes the peak of its autocorrelation. **Key insight for improvement**: Make p "anti-symmetric" in some sense, so that `sum p[j]*p[n-1-j]` is small. But p must be nonnegative! So we can't have true anti-symmetry, but we can make p[j] and p[n-1-j] inversely correlated (when one is large, the other is small).

```python
from scipy.optimize import minimize

def _normalize(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.clip(p, 0.0, None)
    s = float(p.sum())
    if s < 1e-300:
        return np.ones_like(p) / len(p)
    return p / s

def _exact_score(p):
    p = _normalize(p)
    n = len(p)
    b = np.convolve(p, p)
    return float(2.0 * n * np.max(b))

def _softmax(x):
    x = x - x.max()
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return ex / s

def _smooth_obj_and_grad(x, beta):
    p = _softmax(x)
    n = len(p)
    b = np.convolve(p, p)
    bmax = float(b.max())
    # log-sum-exp
    shifted = beta * (b - bmax)
    shifted = np.clip(shifted, -700.0, 0.0)
    ev = np.exp(shifted)
    se = float(ev.sum())
    if se < 1e-300:
        return 2.0 * n * bmax, np.zeros_like(x)
    w = ev / se
    obj = 2.0 * n * (bmax + math.log(se) / beta)
    
    # grad w.r.t. p
    conv_pw = np.convolve(p, w[::-1])
    g_conv = conv_pw[n-1:2*n-1][::-1]
    grad_p = 2.0 * n * (g_conv + p)
    
    # grad w.r.t. x
    gp_mean = float(grad_p.dot(p))
    grad_x = p * (grad_p - gp_mean)
    
    return obj, grad_x
```

```python
def _lp_step(p, M, timeout=5.0):
    """Solve LP: max sum(y) s.t. convolve(p,y)[k] <= M, y >= 0."""
    from scipy.optimize import linprog
    import scipy.sparse as sps
    
    n = len(p)
    # Constraint matrix: A[k, j] = p[k-j] if valid (for convolve(p,y)[k])
    # convolve(p,y)[k] = sum_j p[j] * y[k-j] (valid j, k-j)
    # = sum_j A[k,j] * y[j] where A[k,j] = p[k-j] if 0<=k-j<n else 0
    
    # Build sparse matrix
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        for j in range(n):
            ki = k - j
            if 0 <= ki < n:
                rows.append(k)
                cols.append(j)
                data.append(p[ki])
    
    A = sps.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)  # minimize -sum(y) = maximize sum(y)
    
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M * (1+1e-10)),
                     bounds=[(0, None)]*n, method='highs',
                     options={'time_limit': timeout, 'presolve': True})
        if res.x is not None:
            y = np.clip(res.x, 0, None)
            return y
    except:
        pass
    return None
```
