import numpy as np

def _convolve(a, b):
    """Full linear convolution using FFT for n >= 64, direct otherwise."""
    n = len(a)
    if n < 64:
        return np.convolve(a, b)
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize *= 2
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    out = np.fft.irfft(fa * fb, fsize)[:size]
    # Small numerical cleanup
    out[np.abs(out) < 1e-14] = 0.0
    return out

def _ac1_score(q):
    """Compute 2*n*max(conv(q,q)) / (sum(q))^2. Assumes q is clean nonnegative."""
    n = len(q)
    s = float(q.sum())
    if s < 1e-12:
        return float('inf')
    b = _convolve(q, q)
    return 2.0 * n * float(b.max()) / (s * s)

def _build_conv_matrix(q, n):
    """Build dense (2n-1) x n matrix A such that A @ g = conv(q, g)."""
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    qr = q[::-1]
    for k in range(n - 1):

        A[k, :L] = qr[n - L:]
    for k in range(n - 1, m):
        start = k - n + 1
        L = n - start
        A[k, start:start + L] = qr[:L]
    return A

    k_max = int(np.argmax(conv))

    for j in range(n):
        d = abs(k_max - j)
        if d < n:
            grad[j] = 2.0 * q[d]
    return grad

    conv = _convolve(q, q)
    M = float(conv.max())

        return q

    weights = np.exp((conv[active] - M) * 10.0)
    wsum = float(weights.sum())
    if wsum < 1e-14:

    s = float(q_new.sum())
    if s < 1e-12:

    q_new /= s
    return q_new
