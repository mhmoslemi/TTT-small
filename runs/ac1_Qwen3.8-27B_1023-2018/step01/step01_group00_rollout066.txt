    return best_a, best_score


def _polish(seq, score, stop):
    a = np.asarray(seq, dtype=np.float64).ravel()
    if a.size == 0 or not np.isfinite(a).all():
        a = np.ones(32) / 32.0
    a = np.clip(np.nan_to_num(a), 0.0, 1000.0)
    best = a.copy()
    best_score = float(score) if np.isfinite(score) else 2.0
    n = best.size
    fsize = next_fast_len(2 * n - 1)
    s0 = float(best.sum())
    if s0 > 0.01:
        best = best / s0

    rng = np.random.default_rng(987654321)
    sigma = 0.16
    stuck = 0

    while time.monotonic() < stop - 1.0:
        current = best
        r = float(rng.random())
        if r < 0.40:
            z = np.log(np.clip(current, 1e-8, 1e8))
            z += rng.normal(0.0, sigma, n)
            cand = _softmax_z(z)
        elif r < 0.68:
            cand = current.copy()
            width = int(np.clip(rng.integers(1, max(2, n // 30)), 1, max(1, n // 4)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.20)))
        elif r < 0.80:
            cand = 0.5 * (current + current[::-1])
        elif r < 0.92:
            idx = int(rng.integers(0, n))
            width = int(np.clip(rng.integers(1, max(2, n // 60)), 1, max(1, n // 8)))
            cand = current.copy()
            cand[max(0, idx - width):min(n, idx + width)] *= float(rng.uniform(0.5, 1.8))
        else:
            cand = 0.80 * current + 0.20 * (1.0 / n)

        cand = np.clip(np.nan_to_num(cand), 0.0, 1000.0)
        cs0 = float(cand.sum())
        if cs0 < 1e-9:
            continue
        cand = cand / cs0
        cb = _conv_fft(cand, fsize)
        fs = float(2.0 * n * cb.max())
        if not np.isfinite(fs) or fs >= best_score - 1e-9:
            stuck += 1
            if stuck % 30 == 0:
                sigma = min(1.2, sigma * 1.7)
            continue
        best = cand.copy()
        best_score = fs
        stuck = 0
        sigma = max(0.10, min(sigma, 0.35) * 0.92)

    return best, best_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget = max(0.0, float(budget_s))
    deadline = t0 + max(0.0, budget - 8.0)

    parent = _get_parent()
    # Primary: parent length.
    primary_n = int(np.clip(parent.size, 16, 4096)) if parent.size >= 16 else 192

    a0 = parent.astype(np.float64).copy()
    a0 = np.clip(np.nan_to_num(a0), 1e-12, 1000.0)
    s0 = float(a0.sum())
    if s0 < 0.01:
        a0 = np.ones(primary_n) / max(2, primary_n)
    else:
        a0 = a0 / s0

    best_a = a0.copy()
    best_score = 2.0 * primary_n * float(_conv_fft(a0, next_fast_len(2 * primary_n - 1)).max())

    try:
        exact = evaluate_sequence([float(x) for x in best_a])
        if np.isfinite(exact):
            best_score = exact
    except Exception:
        pass

    if primary_n < 64 or primary_n > 1024:
        parent = np.clip(parent, 1e-12, 1000).astype(np.float64)

    # Concentrate the bulk of the budget on the parent length with several seeds.
    seeds = [int(seed) + 17 * 1000 + 3]
    for k in range(12):
        seeds.append(int(seed) + 17 * 1000 + 3 + 131 * k)

    for it, sd in enumerate(seeds):
        if time.monotonic() > deadline - 30.0 or best_score < 1.503:
            break
        remaining = max(0.0, deadline - time.monotonic() - 25.0)
        alloc = remaining / max(1.0, (len(seeds) - it))
        alloc = max(8.0, min(alloc, 120.0))
        stop = time.monotonic() + alloc
        try:
            cand, fs = _optimize(primary_n, a0, stop, sd)
        except Exception:
            continue
        cand = np.clip(np.nan_to_num(np.asarray(cand, dtype=np.float64).ravel()), 0.0, 1000.0)
        if cand.size != primary_n:
            continue
        cf = float(cand.sum())
        if cf < 1e-9:
            continue
        cand = cand / cf
        exact = None
        try:
            exact = evaluate_sequence([float(x) for x in cand])
        except Exception:
            pass
        if exact is None or not np.isfinite(exact):
            cb = _conv_fft(cand, next_fast_len(2 * primary_n - 1))
            exact = 2.0 * primary_n * float(cb.max())
        if np.isfinite(exact) and exact < best_score - 1e-12:
            best_a = cand.copy()
            best_score = exact

    # Final polish on the best found.
    if time.monotonic() < deadline - 6.0:
        stop_pol = min(deadline - 1.0, time.monotonic() + max(0.0, deadline - time.monotonic() - 2.0))
        try:
            best_a, _ = _polish(best_a, best_score, stop_pol)
        except Exception:
            pass

    # Exact final validation.
    best_a = np.clip(np.nan_to_num(np.asarray(best_a, dtype=np.float64).ravel()), 0.0, 1000.0)
    fs_final = float(best_a.sum())
    if fs_final < 0.01:
        best_a = np.ones(primary_n) / primary_n
    final_score = None
    try:
        final_score = evaluate_sequence([float(x) for x in best_a])
    except Exception:
        final_score = math.inf
    if not np.isfinite(final_score):
        fb = np.ones(primary_n) / primary_n
        try:
            final_score = evaluate_sequence([float(x) for x in fb])
        except Exception:
            pass
        if np.isfinite(final_score):
            best_a = fb
        else:
            return [1.0]

    return [float(x) for x in best_a]
