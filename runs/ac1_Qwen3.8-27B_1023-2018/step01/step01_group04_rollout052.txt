Actually, I realize I should also think about the "symmetry" of the problem. If the optimal sequence is symmetric (a[j] = a[n-1-j]), then n must be even (or odd with the middle element being special). The convolution of a symmetric sequence is symmetric around n-1, so the max is at n-1 (the center). In this case, we just need to minimize `b[n-1] = sum_j a[j]^2` subject to `sum a = 1, a >= 0`, which gives the uniform distribution... but that gives score 2, which is worse.

Wait, that's the constraint `b[n-1] <= M` with M being the max. If the sequence is symmetric, `b[n-1]` is the max (by symmetry and the triangle inequality for autocorrelation). So minimizing `b[n-1] = sum a[j]^2` with `sum a = 1` gives uniform.

But the optimal sequence is NOT symmetric! It's asymmetric, which allows the autocorrelation peak at the center to be reduced by having the mass distributed in a way that the "cross terms" cancel in some lags while the "self terms" are concentrated in others.

Actually wait, for a nonnegative sequence, all autocorrelation values are nonnegative, and by the Cauchy-Schwarz inequality, `b[n-1] = sum a[j]^2 >= (sum a)^2 / n = 1/n` (when sum=1). The uniform achieves this bound. But `b[k]` for k ≠ n-1 can be larger or smaller.

For a "block" sequence like [0,...,0, 1/m, ..., 1/m, 0,...,0] (a window of width m), the autocorrelation is a triangle of base 2m-1 with peak 1/m. The score would be `2n*(1/m) = 2n/m`. For m≈n, this is 2. For m=n/2, it's 4. So blocks are bad.

For a "ramp" sequence like [1/n, 2/n, ..., n/n] normalized: the autocorrelation is more complex.

I think the key is that the optimal sequence has a specific "shape" that balances the autocorrelation values. The LP-based approach naturally finds this by "pushing" the sequence toward equalizing the autocorrelation.

def _score_fast(a):

def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    g = np.interp(x_new, x_old, a)
    s = g.sum()
    if s > 0:
        g = g / s
    return g


def _lp_sort(a, n, max_iters, deadline, rng, perturb_every=12):
    """LP-based local search: iteratively solve LP to find direction, take step."""
    if linprog is None or sps is None:
        return a, _score_fast(a)
    c_vec = -np.ones(n, dtype=np.float64)
    local = _score_fast(a)
    best_local_a = a.copy()
    best_local_sc = local

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        b = np.convolve(a, a)
        M = float(np.max(b))
        if not math.isfinite(M) or M <= 0:
            break
        # Build sparse A: A[k,j] = a[k-j] if 0<=k-j<n
        rows = np.repeat(np.arange(2*n-1, dtype=np.int64), 1)
        cols = np.tile(np.arange(n, dtype=np.int64), 1)
        # Actually let me build it properly
        row_list = []
        col_list = []
        data_list = []
        for k in range(2*n-1):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                row_list.append(k)
                col_list.append(j)
                data_list.append(a[k - j])
        if not row_list:
            break
        try:
            A = sps.csr_matrix((data_list, (row_list, col_list)), shape=(2*n-1, n))
        except Exception:
            break

            res = linprog(
                c_vec,
                A_ub=A,

def _subgrad_polish(a, n, max_iters, deadline, rng):
    """Subgradient-based polish on the max autocorrelation."""
    if n < 4:
        return a, _score_fast(a)
    local = _score_fast(a)
    best_a = a.copy()
    best_sc = local
    lr = 0.02
    fails = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        b = np.convolve(a, a)
        M = float(np.max(b))
        # Active set: top few
        thresh = M * 0.995
        active = np.where(b >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        u = np.zeros(2 * n - 1, dtype=np.float64)
        u[active] = 1.0
        a_rev = a[::-1]
        conv_val = np.convolve(a_rev, u)
        # grad[m] = (2/|S|) * conv_val[n-1+m]
        grad = (2.0 / len(active)) * conv_val[n - 1: n + n - 1]
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        improved = False
        for alpha in (lr, lr * 0.5, lr * 0.25, lr * 0.1):
            a_new = a - alpha * grad
            a_new = np.clip(a_new, 0.0, None)
            sn = float(a_new.sum())
            if sn < 1e-15:
                continue
            a_new = a_new / sn
            sc = _score_fast(a_new)
            if math.isfinite(sc) and sc < local * (1.0 - 1e-10):
                a = a_new
                local = sc
                improved = True
                if local < best_sc:
                    best_sc = local
                    best_a = a.copy()
                break
        if not improved:
            lr *= 0.7
            fails += 1
            if fails > 30 or lr < 1e-7:
                break

            fails = 0
            if it % 25 == 24:
                ap = a * (1.0 + 0.03 * rng.standard_normal(n))

                if sp > 1e-12:

                    scp = _score_fast(ap)

def _make_starts(n, rng, parent):
    starts = []
    x = np.linspace(0.0, 1.0, n)

    def add(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0.0, None)
        if a.size != n or a.sum() <= 1e-15:
            return
        starts.append(a / a.sum())

    # Uniform
    add(np.ones(n))
    # Exponentials
    for c in (-2.0, -1.0, -0.5, 0.5, 1.0, 2.0):
        add(np.exp(c * (x - 0.5)))
    # Sinusoids
    for k in range(1, min(5, n // 2) + 1):
        for amp in (0.3, 0.6, 0.9):
            add(1.0 + amp * np.sin(2 * math.pi * k * (x - 0.5)))
    # Dirichlet
    for alpha in (0.5, 1.0, 2.0):
        add(rng.dirichlet(np.full(n, alpha)))
    # Asymmetric ramps
    add(np.linspace(0.1, n, n))
    add(np.linspace(n, 0.1, n))
    # Triangular
    tri = np.abs(1.0 - 2.0 * x)
    add(1.0 - tri)
    # Parent resampled
    if parent is not None and len(parent) > 1:
        add(_resample(parent, n))
        for scale in (0.8, 1.2):
            p = _resample(parent, n)
            add(p * scale)
    if not starts:
        starts.append(np.ones(n) / n)
    # Score and sort
    scored = [(_score_fast(s), s) for s in starts]
    scored = [(sc, s) for sc, s in scored if math.isfinite(sc)]
    scored.sort(key=lambda z: z[0])
    return [s for _, s in scored]
