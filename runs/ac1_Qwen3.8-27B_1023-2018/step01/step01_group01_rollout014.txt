        return np.fft.irfft(fa * fa, sizeC)[:2 * n - 1]

    # Best-so-far tracker
    a_best = normalize(base.copy())
    sc_best = safe_eval(a_best)

    def try_update(a):
        nonlocal a_best, sc_best
        s = safe_eval(a)
        if np.isfinite(s) and s < sc_best:
            sc_best = s
            a_best = a.copy()
            return True
        return False

    # Core optimization loop
    def run_phase(a_init, duration, lr0, lr1, temp0, temp1, do_perturb=False):
        nonlocal a_best, sc_best
        a = normalize(a_init.copy()).astype(np.float64)
        m = np.zeros(n, dtype=np.float64)
        v = np.zeros(n, dtype=np.float64)
        step = 0
        start = time.monotonic()
        last_eval = start
        stall_count = 0
        best_in_phase = safe_eval(a)

        while time.monotonic() < hard_deadline and (time.monotonic() - start) < duration:
            # Schedules
            frac = min(1.0, step / 8000.0)
            temp = max(1.0, temp0 * (temp1 / max(temp0, 1e-6)) ** frac)
            lr = max(1e-5, lr0 * (lr1 / max(lr0, 1e-12)) ** frac)

            # Compute C = a * a (full convolution)
            C = full_conv(a)
            Cmax = float(C.max())
            S = float(np.sum(a))

            # Softmax weights for smooth max
            z = temp * (C - Cmax)
            z = np.maximum(z, -600.0)
            w = np.exp(z)
            wsum = float(np.sum(w))
            if wsum < 1e-300 or not np.isfinite(wsum):
                mask = C >= Cmax - 2.0 / temp
                if not np.any(mask):
                    mask[C.argmax()] = True
                w = np.zeros_like(C)
                w[mask] = 1.0 / float(np.sum(mask))
                wsum = 1.0
            else:
                w = w / wsum

            M_smooth = Cmax + np.log(max(wsum, 1e-300)) / temp

            # Gradient via FFT: G[i] = sum_k w[k] * a[k-i]
            w_rev = w[::-1].copy()
            fa_g = np.fft.rfft(a, sizeG)
            fw_g = np.fft.rfft(w_rev, sizeG)
            conv_g = np.fft.irfft(fa_g * fw_g, sizeG)
            G = conv_g[n - 1:2 * n - 1][::-1]  # cross-correlation

            # df/da_i = (2n/S^2)*2*G_i - (4n*M/S^3)
            invS = 1.0 / S
            invS2 = invS * invS
            grad = (4.0 * n * invS2) * G - (4.0 * n * M_smooth * invS2 * invS)

            # Adam
            step += 1
            b1, b2 = 0.9, 0.999
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * (grad * grad)
            m_hat = m / (1.0 - b1 ** step)
            v_hat = v / (1.0 - b2 ** step)

            a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            a = np.clip(a, 0.0, 1000.0)
            a = normalize(a)

            # Stall detection and perturbation
            if step % 3000 == 0:
                if step > 6000:
                    sc = safe_eval(a)
                    if np.isfinite(sc) and sc < best_in_phase:
                        best_in_phase = sc
                        stall_count = 0
                    else:
                        stall_count += 1
                    if stall_count >= 4 and do_perturb:
                        # Small random perturbation to escape
                        a = a + 0.005 * rng.standard_normal(n)
                        a = np.clip(a, 0.0, 1000.0)
                        a = normalize(a)
                        m[:] = 0.0
                        v[:] = 0.0
                        stall_count = 0

            # Periodic evaluation
            now = time.monotonic()
            if step % 2000 == 0 and now - last_eval > 0.8:
                last_eval = now
                try_update(a)

        # End of phase: final eval
        try_update(a)

    # --- Allocate time across phases ---
    elapsed = time.monotonic() - t0
    remaining = max(10.0, hard_deadline - time.monotonic())

    t1 = 0.45 * remaining  # broad search
    t2 = 0.25 * remaining  # refine
    t3 = 0.20 * remaining  # polish
    # t4 = 0.10 * remaining reserved for validation/return

    # Phase 1: Moderate temperature, larger steps
    run_phase(a_best.copy(), t1, 0.012, 0.004, 5.0, 40.0, do_perturb=True)

    # Phase 2: Higher temperature, smaller steps
    if time.monotonic() < hard_deadline:
        run_phase(a_best.copy(), t2, 0.005, 0.0015, 40.0, 150.0, do_perturb=False)

    # Phase 3: Very high temperature (near-exact max), fine steps
    if time.monotonic() < hard_deadline:
        run_phase(a_best.copy(), t3, 0.002, 0.0003, 150.0, 800.0, do_perturb=False)

    # --- Final micro-polish with exact subgradient (top-5 active set) ---
    if time.monotonic() < hard_deadline - 5.0:
        a = a_best.copy().astype(np.float64)
        polish_steps = min(5000, max(500, int(3.0 / 0.003)))  # ~3 seconds max
        for ps in range(polish_steps):
            if time.monotonic() > hard_deadline - 5.0:
                break
            C = full_conv(a)
            S = float(np.sum(a))
            # Use top-K active set for sharper gradient
            K = min(10, 2 * n - 1)
            topk = np.argpartition(C, -K)[-K:]
            Cmax = float(C[topk].max())
            w_top = np.zeros(2 * n - 1)
            w_top[topk] = 1.0 / K
            G = np.zeros(n)
            for k in topk:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                G[lo:hi + 1] += (1.0 / K) * a[k - hi:k - lo + 1][::-1]
            # Actually: G[i] = sum_k w_k * a[k-i]; for k in topk, k-i in [0,n-1]
            # G[i] += w_k * a[k-i] for each k: i ranges from max(0,k-n+1) to min(n-1,k)
            G = np.zeros(n)
            for k in topk:
                i_lo = max(0, k - (n - 1))
                i_hi = min(n - 1, k)
                G[i_lo:i_hi + 1] += (1.0 / K) * a[k - i_hi:k - i_lo + 1][::-1]

            M_exact = Cmax
            invS = 1.0 / S
            invS2 = invS * invS
            grad = (4.0 * n * invS2) * G - (4.0 * n * M_exact * invS2 * invS)
            lr_p = 0.001 * (0.995 ** ps)
            a = a - lr_p * grad
            a = np.clip(a, 0.0, 1000.0)
            a = normalize(a)
        try_update(a)

    # --- Final validation ---
    final_a = normalize(np.clip(a_best, 0.0, 1000.0))
    sc_final = safe_eval(final_a)
    if np.isfinite(sc_final) and sc_final <= sc_best:
        a_best = final_a
        sc_best = sc_final

    # Ensure valid output
    out = [float(x) for x in a_best]
    if not out or any((not np.isfinite(x)) for x in out) or any(x < 0 or x > 1000 for x in out):
        out = [float(target_sum / n)] * n

    return out
