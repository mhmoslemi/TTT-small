                    return g / s

            pass

            # For 0 <= k-i < n and 0 <= i < n:
            #   If k-i != i: contributes 2*q[k-i]
            #   If k-i == i (i.e., k = 2i): contributes q[i]
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for i in range(lo, hi + 1):

                    if j != i:

                    else:
                        sg[i] += wk * q[j]

                        q_try = norm_q(q - alpha * sg / sg_norm)
                        s_try = conv_max(q_try)
                        if s_try < current_sc - 1e-12:
                            q = q_try
                            current_sc = s_try
                            stagnant = 0
                            break

            stagnant = 8

    # (skip for time)
    
    # Make sure we have a valid result
    consider(q)
    
    if best_list is None:
        # Fallback
        base = np.array([0.4, 0.2, 0.2, 0.2], dtype=np.float64)
        q_fb = np.tile(base, n // 4)[:n]

        val = float(evaluate_sequence(best_list))
        if not np.isfinite(val):
            best_list = [1.0]
    except Exception:
        best_list = [1.0]
    
    return [float(x) for x in best_list]
