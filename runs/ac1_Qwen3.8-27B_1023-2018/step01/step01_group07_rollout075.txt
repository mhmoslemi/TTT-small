import numpy as np

def _clean_positive_vector(x):
    try:
        arr = np.asarray(x, dtype=np.float64).ravel()
    except Exception:
        arr = np.ones(64, dtype=np.float64)
    if arr.size == 0:
        arr = np.ones(64, dtype=np.float64)
    if not np.all(np.isfinite(arr)):
        arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 1e-12:
        arr = np.ones(arr.size if arr.size > 0 else 64, dtype=np.float64)
        s = float(arr.sum())
    return arr / s


def _symmetrize(a):
    a = 0.5 * (np.asarray(a, dtype=np.float64) + np.asarray(a, dtype=np.float64)[::-1])
    s = float(a.sum())
    if s < 1e-12:
        return a.copy()
    return a / s


def _resample(a, m):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = int(max(1, m))
    if a.size == 0:
        return np.ones(m, dtype=np.float64) / m
    if a.size == m:
        s = float(a.sum())
        if s < 1e-12:
            return np.ones(m, dtype=np.float64) / m
        return a / s
    x = np.linspace(0.0, a.size - 1, m)
    y = np.interp(x, np.arange(a.size, dtype=np.float64), a)
    y = np.maximum(y, 0.0)
    s = float(y.sum())
    if s < 1e-12:
        return np.ones(m, dtype=np.float64) / m
    return y / s


def _random_profile(n, rng):
    n = int(max(1, n))
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    a = np.ones(n, dtype=np.float64)
    for _ in range(3):
        a += rng.uniform(0.2, 0.7, 1) * np.sin(
            2.0 * np.pi * rng.uniform(1.0, 6.0, 1) * x + rng.uniform(0.0, 2.0 * np.pi, 1)
        )
    a = a + rng.normal(0.0, 0.02 / np.sqrt(n), n)
    a = np.maximum(a, 0.02)
    s = float(a.sum())
    if s < 1e-12:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _perturb(a, rng, sigma):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n <= 0:
        return _clean_positive_vector(a)
    a2 = a * np.exp(rng.normal(0.0, float(sigma), n))
    a2 = a2 + rng.normal(0.0, 0.01 / np.sqrt(n), n)
    a2 = np.maximum(a2, 0.0)
    s = float(a2.sum())
    if s < 1e-15:
        return a.copy()
    return a2 / s


def _setup_conv(n):
    n = int(n)
    end = 2 * n - 1
    if n <= 64:
        def b_of(a):
            return np.convolve(a, a, mode="full")

        def grad_soft(a, w):
            full = np.convolve(a, w[::-1], mode="full")
            return 2.0 * full[n - 1:2 * n - 1][::-1]
    else:
        L = 1
        while L < end:
            L <<= 1
        wr = np.zeros(L, dtype=np.float64)

        def b_of(a):
            F = np.fft.rfft(a, L)
            return np.fft.irfft(F * F, L)[:end]

        def grad_soft(a, w):
            wr[:end] = w[::-1]
            if end < L:
                wr[end:] = 0.0
            F = np.fft.rfft(a, L)
            W = np.fft.rfft(wr, L)
            c = np.fft.irfft(F * W, L)
            return 2.0 * c[n - 1:end][::-1]
    return b_of, grad_soft


def _softmax_weights(b, M, tau_frac):
    end = int(b.size)
    w = np.zeros(end, dtype=np.float64)
    if M <= 0.0:
        w[int(np.argmax(b))] = 1.0
        return w
    tau = max(float(tau_frac) * float(M), 1e-30)
    sh = (b - M) / tau
    sh = np.clip(sh, -50.0, 0.0)
    e = np.exp(sh)
    s = float(e.sum())
    if not np.isfinite(s) or s < 1e-30:
        w[int(np.argmax(b))] = 1.0
        return w
    return e / s


def _fill_segments(buf, a, idx, n):
    n1 = n - 1
    for r, k in enumerate(idx):
        row = buf[r]
        row.fill(0.0)
        if k < n1:
            row[:k + 1] = a[:k + 1][::-1]
        else:
            s = k - n1
            row[s:] = a[s:][::-1]


def _active_direction(a, b, K, dual_iters, buf, n):
    end = int(b.size)
    K = int(min(max(1, K), end))
    if K >= end:
        idx = np.arange(end)
    else:
        idx = np.argpartition(b, -K)[-K:]
    Kuse = int(idx.size)
    if Kuse <= 0:
        return False, None, None, 0.0
    _fill_segments(buf, a, idx, n)
    w = np.ones(Kuse, dtype=np.float64) / Kuse

    for _ in range(int(max(1, dual_iters))):
        g = w @ buf[:Kuse]
        r = g - float(np.mean(g))
        norm2 = float(r @ r)
        if not np.isfinite(norm2) or norm2 < 1e-30:
            return False, None, None, 0.0
        if Kuse > 1:
            d = buf[:Kuse] @ r
            dc = d - float(np.mean(d))
            dmax = float(np.max(np.abs(dc)))
            if np.isfinite(dmax) and dmax > 1e-30:
                w *= np.exp(-0.7 * dc / dmax)
                ws = float(w.sum())
                if not np.isfinite(ws) or ws < 1e-30:
                    w[:] = 1.0 / Kuse
                else:
                    w /= ws

    g = w @ buf[:Kuse]
    r = g - float(np.mean(g))
    norm = float(np.sqrt(max(0.0, float(r @ r))))
    if not np.isfinite(norm) or norm <= 1e-15:
        return False, None, None, 0.0

    u_add = -r / norm
    v_log = None
    denom = a + 1e-12
    v = -r / denom
    v = v - float(np.mean(v))
    vm = float(np.max(np.abs(v)))
    if np.isfinite(vm) and vm > 1e-30:
        v_log = v / vm
    return True, u_add, v_log, norm


def _line_search(a, kind, dirval, lr, b_of, M, tol, a_norm=None):
    step = float(lr)
    for _ in range(5):
        if step <= 1e-14:
            break
        if kind == 0:
            if a_norm is None:
                a_norm = float(np.linalg.norm(a))
            cand = a + step * a_norm * dirval
        else:
            cand = a * np.exp(np.clip(step * dirval, -50.0, 50.0))
        cand = np.maximum(cand, 0.0)
        s = float(cand.sum())
        if not np.isfinite(s) or s < 1e-15:
            step *= 0.5
            continue
        cand = cand / s
        b_new = b_of(cand)
        M_new = float(b_new.max())
        if np.isfinite(M_new) and M_new <= M - tol:
            return True, cand, b_new, M_new, step
        step *= 0.5
    return False, None, None, None, step


def _run_optimizer(init, t_end, rng, update_best, best_box, hard_deadline, K0=10, dual_iters=6):
    a = _clean_positive_vector(init)
    n = int(a.size)
    if n <= 0:
        return None
    b_of, grad_soft = _setup_conv(n)
    b = b_of(a)
    M = float(b.max())
    if not np.isfinite(M) or M <= 0.0:
        a = np.ones(n, dtype=np.float64) / float(n)
        b = b_of(a)
        M = float(b.max())

    local_best_a = a.copy()
    local_best_M = M
    update_best(a)

    Kmax = int(min(32, max(2, min(n, 32))))
    K = int(min(max(4, K0), Kmax))
    seg_buf = np.empty((Kmax, n), dtype=np.float64)

    add_lr = 0.30
    log_lr = 0.12
    tau_frac = 0.05
    stag = 0
    it = 0
    tol = 1e-16
    last_exact = time.monotonic()

    if n <= 64:
        exact_interval = 0.4
        score_thresh = 1e-6
    elif n <= 1024:
        exact_interval = 1.5
        score_thresh = 2e-5
    else:
        exact_interval = 6.0
        score_thresh = 1e-4

    while time.monotonic() < t_end:
        if time.monotonic() > hard_deadline - 12.0:
            break
        it += 1
        M = float(b.max())
        if not np.isfinite(M) or M <= 1e-300:
            break

        use_soft = (stag > 25) or (it % 11 == 0)
        u_add = None
        v_log = None

        if (not use_soft) and K >= 1:
            use_active, u_add, v_log, _ = _active_direction(
                a, b, min(K, Kmax), dual_iters, seg_buf, n
            )
            if not use_active:
                use_soft = True

        if use_soft or u_add is None:
            w = _softmax_weights(b, M, tau_frac)
            g = grad_soft(a, w)
            gc = g - float(np.mean(g))
            gn = float(np.linalg.norm(gc))
            if np.isfinite(gn) and gn > 1e-30:
                u_add = -gc / gn
                v = -gc / (a + 1e-12)
                v = v - float(np.mean(v))
                vm = float(np.max(np.abs(v)))
                v_log = v / vm if np.isfinite(vm) and vm > 1e-30 else None
            else:
                u_add = None
                v_log = None

        if u_add is None and v_log is None:
            stag += 50
            a = _perturb(local_best_a, rng, 0.15 if n > 500 else 0.30)
            b = b_of(a)
            M = float(b.max())
            continue

        a_norm = float(np.linalg.norm(a))
        improved = False

        if u_add is not None:
            ok, a_n, b_n, M_n, step = _line_search(
                a, 0, u_add, add_lr, b_of, M, tol, a_norm
            )
            if ok:
                a, b, M = a_n, b_n, M_n
                add_lr = float(min(3.0, max(1e-8, step * 1.15)))
                improved = True

        if not improved and v_log is not None:
            ok, a_n, b_n, M_n, step = _line_search(
                a, 1, v_log, log_lr, b_of, M, tol, a_norm
            )
            if ok:
                a, b, M = a_n, b_n, M_n
                log_lr = float(min(3.0, max(1e-8, step * 1.15)))
                improved = True

        if improved:
            stag = 0
            tau_frac = max(tau_frac * 0.995, 0.008)
            if M < local_best_M - 1e-18:
                local_best_M = M
                local_best_a = a.copy()
        else:
            add_lr = max(add_lr * 0.75, 1e-9)
            log_lr = max(log_lr * 0.75, 1e-9)
            stag += 1
            tau_frac = min(max(tau_frac * 1.12, 0.008), 0.5)
            if stag % 12 == 0:
                K = min(Kmax, max(K, 8) + 3)

        if it % 200 == 0:
            asy = 0.5 * (a + a[::-1])
            sasy = float(asy.sum())
            if sasy > 1e-15:
                asy = asy / sasy
                bsy = b_of(asy)
                Masy = float(bsy.max())
                if np.isfinite(Masy) and Masy < M - tol:
                    a, b, M = asy, bsy, Masy
                    stag = 0
                    if M < local_best_M - 1e-18:
                        local_best_M = M
                        local_best_a = a.copy()

        now = time.monotonic()
        if now - last_exact > exact_interval and now < hard_deadline - 15.0:
            ls = float(local_best_a.sum())
            surr = (2.0 * n * local_best_M) / (ls * ls) if ls > 0.0 else np.inf
            if np.isfinite(surr) and surr < best_box[0] - score_thresh:
                update_best(local_best_a)
                last_exact = now

        if stag > 25 and K < Kmax:
            K = min(Kmax, K + 4)
            stag = 10

        if stag > 90:
            base = local_best_a if stag > 180 else a
            sig = 0.07 if n > 1000 else 0.15
            a = _perturb(base, rng, sig)
            b = b_of(a)
            M = float(b.max())
            add_lr = 0.30
            log_lr = 0.12
            tau_frac = min(0.4, max(tau_frac, 0.08))
            K = max(8, min(Kmax, K // 2 + 4))
            stag = 0
        elif stag > 400:
            a = _random_profile(n, rng)
            b = b_of(a)
            M = float(b.max())
            add_lr = 0.30
            log_lr = 0.12
            tau_frac = 0.1
            K = int(min(max(4, K0), Kmax))
            stag = 0

    if time.monotonic() < hard_deadline - 5.0:
        update_best(local_best_a)
    return local_best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    t0 = time.monotonic()
    try:
        budget = max(1.0, float(budget_s))
    except Exception:
        budget = 1300.0
    hard_deadline = t0 + max(1.0, budget - 25.0)

    best_box = [np.inf, None]

    def consider(arr):
        if arr is None:
            return
        if time.monotonic() > hard_deadline - 3.0:
            return
        a = _clean_positive_vector(arr)
        if a.size == 0:
            return
        try:
            sc = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(sc) and sc < best_box[0]:
            best_box[0] = sc
            best_box[1] = a.copy()

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    parent = _clean_positive_vector(parent)
    full_len = int(parent.size)
    if full_len <= 0:
        parent = np.ones(256, dtype=np.float64)
        full_len = int(parent.size)

    if time.monotonic() > hard_deadline - 5.0:
        a = _clean_positive_vector(parent)
        return [float(x) for x in a]

    consider(parent)
    consider(_symmetrize(parent))

    if best_box[1] is not None and best_box[1].size == full_len:
        full_init = best_box[1].copy()
    else:
        full_init = parent.copy()

    if budget < 120.0:
        _run_optimizer(full_init, hard_deadline - 8.0, rng, consider, best_box, hard_deadline)
        if best_box[1] is None:
            best_box[1] = parent.copy()
        return [float(x) for x in _clean_positive_vector(best_box[1])]

    if full_len == 4806:
        all_sizes = [534, 267, 178, 89, 54]
    else:
        all_sizes = [m for m in [64, 128, 256, 512, 1024, 2048] if 32 <= m < full_len // 2]
        if not all_sizes:
            all_sizes = [max(32, full_len // 8)]
    coarse_sizes = []
    seen = set()
    for m in all_sizes:
        mm = int(max(2, m))
        if 8 < mm < full_len and mm not in seen:
            seen.add(mm)
            coarse_sizes.append(mm)
    coarse_sizes.sort(reverse=True)

    cap = max(1.0, hard_deadline - t0)

    if coarse_sizes:
        frac_c, frac_f1, frac_f2 = 0.13, 0.38, 0.16
    else:
        frac_c, frac_f1, frac_f2 = 0.0, 0.50, 0.18
    frac_pol = 1.0 - (frac_c + frac_f1 + frac_f2)
    if frac_pol < 0.12:
        total = frac_c + frac_f1 + frac_f2
        if total > 0.0:
            scale = (1.0 - 0.12) / total
            frac_c *= scale
            frac_f1 *= scale
            frac_f2 *= scale
            frac_pol = 0.12
        else:
            frac_f1 = 0.88
            frac_pol = 0.12

    coarse_total = frac_c * cap
    full1_total = frac_f1 * cap
    full2_total = frac_f2 * cap

    t = t0
    coarse_end = t + coarse_total
    t = coarse_end
    full1_end = t + full1_total
    t = full1_end
    full2_end = t + full2_total
    polish_end = min(t + frac_pol * cap, hard_deadline)

    if coarse_sizes:
        coarse_slice = coarse_total / float(len(coarse_sizes))
        for m in coarse_sizes:
            if time.monotonic() >= coarse_end - 2.0:
                break
            init_c = _resample(parent, m)
            t_end_c = min(time.monotonic() + coarse_slice, coarse_end, hard_deadline - 10.0)
            _run_optimizer(init_c, t_end_c, rng, consider, best_box, hard_deadline)

    if best_box[1] is not None and best_box[1].size == full_len:
        full_init = best_box[1].copy()
    else:
        base = best_box[1] if best_box[1] is not None else parent
        full_init = _resample(base, full_len) if base.size != full_len else base.copy()

    loc1 = None
    if time.monotonic() < full1_end - 5.0:
        loc1 = _run_optimizer(
            full_init, min(full1_end, hard_deadline - 15.0), rng, consider, best_box, hard_deadline
        )

    if time.monotonic() < full2_end - 5.0:
        if loc1 is not None and loc1.size == full_len:
            full2_init = _perturb(loc1, rng, 0.02)
        else:
            full2_init = _symmetrize(full_init)
        _run_optimizer(
            full2_init, min(full2_end, hard_deadline - 15.0), rng, consider, best_box, hard_deadline
        )

    if time.monotonic() < polish_end - 10.0:
        if best_box[1] is not None:
            polish_init = best_box[1].copy()
        else:
            polish_init = full_init.copy()
        _run_optimizer(
            polish_init, min(polish_end, hard_deadline - 8.0), rng, consider, best_box, hard_deadline
        )

    if best_box[1] is None:
        best_box[1] = parent.copy()
    final_arr = _clean_positive_vector(best_box[1])
    if time.monotonic() < hard_deadline - 3.0:
        consider(final_arr)
        if best_box[1] is not None:
            final_arr = _clean_positive_vector(best_box[1])
    return [float(x) for x in final_arr]
