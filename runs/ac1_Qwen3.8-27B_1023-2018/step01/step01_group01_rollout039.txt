    top_peaks = np.argsort(C)[-K:]

    resp = np.zeros(n)
    for k in top_peaks:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                resp[i] += a[j]  # responsibility of a_i for peak at k
    
    # Shrink top contributors
    top_resp_idx = np.argsort(resp)[-K:]
    shrink_amount = a[top_resp_idx].min() * 0.1  # shrink by 10% of minimum
    a[top_resp_idx] -= shrink_amount
    a[top_resp_idx] = np.maximum(a[top_resp_idx], 0)
    
    # Spread to low contributors
    low_resp_idx = np.argsort(resp)[:K]
    total_removed = K * shrink_amount
    a[low_resp_idx] += total_removed / K
    
    return a

import time
from scipy.optimize import linprog

def _np2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()

def propose_candidate(seed=42, budget_s=1300, **kwargs):

    deadline = t0 + max(1.0, float(budget_s)) - 20.0
    rng = np.random.default_rng(seed)

        base = np.clip(base, 0.0, 1000.0)

        base = np.ones(6962, dtype=np.float64)

    sizeC = _np2(2 * n - 1)
    sizeG = _np2(3 * n - 1)

    def safe_eval(a):
        try:
            s = evaluate_sequence([float(x) for x in a])
            return s if np.isfinite(s) else np.inf
        except Exception:
            return np.inf

    def norm_to(a, target):
        s = a.sum()
        if s < 1e-15:
            return np.full_like(a, target / n)
        return a * (target / s)

    def fft_convolve(a, b):
        sz = _np2(len(a) + len(b) - 1)
        return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz), sz)[:len(a) + len(b) - 1]

    def compute_grad(a, temp):
        """Smooth-max gradient via FFT. Returns (grad, M_smooth, C, Cmax)."""
        fa = np.fft.rfft(a, sizeC)
        C = np.fft.irfft(fa * fa, sizeC)[:2 * n - 1]
        Cmax = float(C.max())
        z = temp * (C - Cmax)
        # Clip z to avoid overflow
        z = np.clip(z, -50, 50)
        w = np.exp(z)
        wsum = float(w.sum())
        if wsum < 1e-30 or not np.isfinite(wsum):
            w = np.ones(2 * n - 1) / (2 * n - 1)
            wsum = 1.0
        else:
            w = w / wsum
        M_smooth = Cmax + np.log(wsum) / temp
        # Gradient: dM_i = 2 * sum_k w_k * a[k-i]
        wfl = w[::-1]
        fa2 = np.fft.rfft(a, sizeG)
        fw = np.fft.rfft(wfl, sizeG)
        conv2 = np.fft.irfft(fa2 * fw, sizeG)[:3 * n - 2]
        dM = 2.0 * conv2[n - 1:2 * n - 1][::-1]
        return dM, M_smooth, C, Cmax

    def lp_direction(a, M_target, K=25):
        """Solve reduced LP: max sum(g) s.t. (a*g)_k <= M_target for top-K peaks, g>=0."""
        try:
            fa = np.fft.rfft(a, sizeC)
            C = np.fft.irfft(fa * fa, sizeC)[:2 * n - 1]
            top_idx = np.argsort(C)[-K:]
            A_mat = np.zeros((K, n))
            for idx, k in enumerate(top_idx):
                j_start = max(0, k - n + 1)
                j_end = min(n, k + 1)
                for j in range(j_start, j_end):
                    A_mat[idx, j] = a[k - j]
            c_obj = -np.ones(n)
            b_vec = np.full(K, M_target)
            res = linprog(c_obj, A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'time_limit': 8.0})
            if res.status == 0 and res.x is not None:
                g = np.maximum(res.x, 0.0)
                gs = g.sum()
                if gs > 1e-10:
                    return g / gs
        except Exception:
            pass
        return None

    def shrink_spread(a, C, K=15):
        """Heuristic: shrink mass at peak-contributing positions, spread to valleys."""
        top_peaks = np.argsort(C)[-K:]
        resp = np.zeros(n)
        for k in top_peaks:

            j_end = min(n, k + 1)

                resp[j] += a[k - j]

        a[top_j] = np.maximum(a[top_j] - shr, 0.0)

        a[low_j] += (K * shr) / K

    best_score = scored[0][0]

    # ---- Main optimization loop ----
    def run_optim(a0, seconds, tag="main"):
        nonlocal best_a, best_score
        a = np.asarray(a0, dtype=np.float64).copy()
        S_target = float(a.sum())
        if S_target < 1e-6:

        v_adv = np.zeros(n)
        step = 0

        no_improve = 0
        start = time.monotonic()

        while time.monotonic() < deadline - 5.0 and (time.monotonic() - start) < seconds:

            grad = dM.copy()

            step += 1
            b1, b2, eps = 0.9, 0.999, 1e-8
            m_adv = b1 * m_adv + (1 - b1) * grad
            v_adv = b2 * v_adv + (1 - b2) * (grad * grad)
            m_hat = m_adv / (1.0 - b1 ** step)
            v_hat = v_adv / (1.0 - b2 ** step)

            # Learning rate schedule: larger early, smaller late
            lr = 0.012 * (1.5 - 0.5 * progress)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.maximum(a, 0.0)
            a = norm_to(a, S_target)

            # Periodic exact evaluation

                last_eval = now
                sc = safe_eval(a)
