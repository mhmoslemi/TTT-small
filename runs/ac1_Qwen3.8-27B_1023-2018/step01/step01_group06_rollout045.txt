        FA = np.fft.rfft(a_arr, corr_n)
        corr_full = np.fft.irfft(FW * np.conj(FA), corr_n)
        valid_corr = corr_full[n - 1: n - 1 + n]
        grad = 2.0 * valid_corr
        return grad, b

    while time.monotonic() < phase1_deadline:
        it += 1
        grad, b = compute_smooth_grad(a)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-18)
        a_new /= a_new.sum()

        s_new = fast_score(a_new)
        s_old = fast_score(a)

        if s_new < s_old * 1.0 + 1e-15:
            a = a_new
            patience = 0
        else:
            lr *= 0.7
            patience += 1
            if patience > 50:
                # Perturb
                npert = max(5, n // 100)
                idx = rng.choice(n, size=npert, replace=False)
                a[idx] *= np.exp(rng.normal(0, 0.25, size=npert))
                a = np.maximum(a, 1e-18)
                a /= a.sum()
                lr = 0.03
                patience = 0

        # Fine-tune gamma
        if it % 800 == 0:
            gamma = min(gamma * 1.03, 300.0)
            lr = min(lr * 1.2, 0.1)

        # Validate
        if it % 30 == 0:
            exact = float(evaluate_sequence(a.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_a = a.copy()

        if time.monotonic() > phase1_deadline - 2:
            break

    # ====================================================================
    # Phase 2: Load-based multiplicative updates (refined)
    # ====================================================================
    a = best_a.copy()
    eta = 0.5
    top_frac = 0.15
    it2 = 0
    patience2 = 0
    phase2_deadline = deadline - 150.0

    while time.monotonic() < phase2_deadline:
        it2 += 1

        conv = fft_conv(a)
        max_conv = conv.max()

        k_top = max(10, int(top_frac * conv_len))
        thresh = np.partition(conv, conv_len - k_top)[conv_len - k_top]

        spread = max_conv - thresh
        if spread > 1e-14:
            weights = 1.0 + np.clip(conv - thresh, 0.0, spread) / spread
        else:
            weights = 1.0 + (conv >= thresh).astype(np.float64)

        # Load: load[j] = sum_m a[m] * weights[j + m - (n-1)]
        # Use cross-correlation: load[j] = (a ⋆ weights_shifted)[j]
        # weights is symmetric, length 2n-1
        # load[j] = sum_k a[k] * weights[j+k-(n-1)] for valid k
        # = convolution of a and weights at index j + (n-1)
        FA = np.fft.rfft(a, corr_n)
        FW = np.fft.rfft(weights, corr_n)
        conv_full = np.fft.irfft(FA * FW, corr_n)
        load = conv_full[n - 1: n - 1 + n]

        lm = load.max()
        if lm > 1e-15:
            a = a * np.exp(-eta * load / lm)
        a = np.maximum(a, 1e-18)
        a /= a.sum()

        est = fast_score(a)

        if it2 % 20 == 0:
            exact = float(evaluate_sequence(a.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_a = a.copy()
                patience2 = 0
            else:
                patience2 += 1

        # Perturbation
        if it2 % 200 == 0:
            npert = max(3, n // 200)
            idx = rng.choice(n, size=npert, replace=False)
            sigma = 0.06 + 0.03 * (patience2 / 30.0)
            a[idx] *= np.exp(rng.normal(0, sigma, size=npert))
            a = np.clip(a, 0, 1000)
            s = a.sum()
            if s > 1e-6:
                a /= s
            else:
                a = best_a.copy()

        if patience2 >= 400:
            npert = max(5, n // 20)
            idx = rng.choice(n, size=npert, replace=False)
            a[idx] *= np.exp(rng.normal(0, 0.3, size=npert))
            a = np.clip(a, 0, 1000)
            s = a.sum()
            if s > 1e-6:
                a /= s
            else:
                a = best_a.copy()
            if rng.random() < 0.3:
                blend = rng.uniform(0.05, 0.2)
                a = (1 - blend) * a + blend * a[::-1]
                a /= a.sum()
            patience2 = 0
            eta = min(1.0, eta * 1.05)

        if it2 % 3000 == 0:
            eta = max(0.1, eta * 0.95)
            top_frac = max(0.02, top_frac * 0.93)

        if time.monotonic() > phase2_deadline - 2:
            break

    # ====================================================================
    # Phase 3: Fine local refinement with very small perturbations
    # ====================================================================
    a = best_a.copy()
    it3 = 0
    phase3_deadline = deadline - 5.0

    def try_improve(a_curr, step_size, n_pert):
        a_try = a_curr.copy()
        idx = rng.choice(n, size=n_pert, replace=False)
        a_try[idx] *= np.exp(rng.normal(0, step_size, size=n_pert))
        a_try = np.clip(a_try, 0, 1000)
        s = a_try.sum()
        if s < 1e-6:
            return a_curr, False
        a_try /= s
        exactly = float(evaluate_sequence(a_try.tolist()))
        if np.isfinite(exactly) and exactly < best_score:
            return a_try, True
        return a_curr, False

    while time.monotonic() < phase3_deadline:
        it3 += 1
        # Very small perturbations
        for _ in range(5):
            a, improved = try_improve(a, 0.02, max(2, n // 500))
            if improved:
                best_a = a.copy()

        # Gradient step with tiny lr
        if it3 % 5 == 0:
            grad, _ = compute_smooth_grad(a)
            a_tiny = a - 0.002 * grad
            a_tiny = np.maximum(a_tiny, 1e-18)
            a_tiny /= a_tiny.sum()
            sc = float(evaluate_sequence(a_tiny.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_tiny.copy()
                a = a_tiny.copy()

        if time.monotonic() > phase3_deadline:
            break

    # ====================================================================
    # Final validation
    # ====================================================================
    result = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(result))

    if not np.isfinite(final_score) or final_score > 2.0:
        result = [1.0] * n
        final_score = float(evaluate_sequence(result))

    return result
