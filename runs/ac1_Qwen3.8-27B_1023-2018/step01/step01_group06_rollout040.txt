```python
import numpy as np
from scipy import sparse
import cvxpy as cp

def solve_lp(a, M):
    """Solve max sum(g) s.t. g >= 0, (a*g)[k] <= M for all k."""
    n = len(a)
    conv_len = 2*n - 1
    
    # Build sparse constraint matrix A where (A@g)[k] = (a*g)[k]
    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(conv_len):
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(conv_len, n))
    
    g = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= M]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, max_iter=10000)
    
    if g.value is None:
        return None
    return g.value
```

    """Solve LP: max sum(g) s.t. g >= 0, conv(a, g) <= M."""
    conv_len = 2 * n - 1

    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    rows_list = []
    cols_list = []
    vals_list = []
    
    for k in range(conv_len):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        if length > 0:
            rows_list.append(np.full(length, k, dtype=np.int32))
            cols_list.append(np.arange(j_start, j_end + 1, dtype=np.int32))
            vals_list.append(a[k - j_start : k - j_end + 1][::-1])  # a[k-j] for j=j_start..j_end
    
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(conv_len, n))
    
    g = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= M]
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-6)
    except:
        try:
            prob.solve(solver=cp.ECOS, max_iter=10000)
        except:
            return None
    
    if g.value is None:
        return None
    return np.asarray(g.value, dtype=np.float64).flatten()

def _build_conv_matrix(a, n):
    """Build sparse convolution matrix A where (A@g)[k] = np.convolve(a,g)[k]."""
    conv_len = 2 * n - 1

    cols_parts = []
    vals_parts = []
