import numpy as np
import time
from scipy.optimize import minimize

def _p2(l):
    return 1 << max(0, (int(l) - 1).bit_length())

def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    budget_s = max(1.0, float(budget_s))
    deadline = t0 + budget_s - 15.0

    # Load base
    base = None
    try:
        b = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if b.size and np.all(np.isfinite(b)):
            base = np.clip(b, 0.0, 1000.0)
    except Exception:
        pass
    if base is None:
        base = np.ones(6962, dtype=np.float64)
    n = int(base.size)
    sz_c = _p2(2 * n - 1)
    sz_g = _p2(3 * n - 2)
    bnds = [(0.0, 1000.0)] * n

    def _ev(a):
        try:
            s = float(evaluate_sequence([float(v) for v in a]))
            return s if np.isfinite(s) else np.inf
        except Exception:
            return np.inf

    best_a = base.copy()
    best_sc = _ev(best_a)
    rng = np.random.default_rng(seed)

    def _fg(x, T):
        a = np.clip(x, 0.0, 1000.0)
        S = float(a.sum())
        if S < 0.01:
            return 1e10, np.zeros(n)
        fa = np.fft.rfft(a, sz_c)
        c = np.fft.irfft(fa * fa, sz_c)[:2 * n - 1]
        Tc = T * c
        Tcm = float(Tc.max())
        ex = np.exp(Tc - Tcm)
        exs = float(ex.sum())
        w = ex / exs
        M = (Tcm + np.log(exs)) / T
        wr = w[::-1].copy()
        d = np.fft.irfft(np.fft.rfft(a, sz_g) * np.fft.rfft(wr, sz_g), sz_g)[:3 * n - 2]
        g = 2.0 * d[n - 1:2 * n - 1][::-1]
        iS = 1.0 / S
        val = 2.0 * n * M * iS * iS
        grad = (2.0 * n * iS * iS) * (g - 2.0 * M * iS)
        return val, grad

    class _TE(Exception):
        pass

    def _lb(x0, T, mi, tm):
        ts = time.monotonic()
        def f(x):
            if time.monotonic() - ts > tm:
                raise _TE
            return _fg(x, T)
        try:
            r = minimize(f, x0, method='L-BFGS-B', jac=True, bounds=bnds,
                         options={'maxiter': mi, 'gtol': 1e-9, 'ftol': 1e-12,
                                  'maxfun': 8000, 'maxcor': 12, 'iprint': 0})
            return np.clip(r.x, 0.0, 1000.0)
        except _TE:
            return np.clip(x0, 0.0, 1000.0)
        except Exception:
            return np.clip(x0, 0.0, 1000.0)

    # Check deadline before each phase
    def _ok():
        return time.monotonic() < deadline

    # Phase 1: L-BFGS from parent, T=10
    if _ok():
        x = _lb(best_a.copy(), 10.0, 500, 60.0)
        s = _ev(x)
        if s < best_sc:
            best_a, best_sc = x, s

    # Phase 2: L-BFGS from parent, different T
    for T in [5.0, 20.0, 50.0]:
        if not _ok():
            break
        x = _lb(best_a.copy(), T, 400, 45.0)
        s = _ev(x)
        if s < best_sc:
            best_a, best_sc = x, s

    # Phase 3: Perturbation search
    for k in range(40):
        if not _ok():
            break
        T = 6.0 + 4.0 * (k % 5)
        sc = 0.02 + 0.015 * (k % 4)
        x0 = np.clip(best_a + rng.standard_normal(n) * sc * (best_a.mean() + 1e-8), 0, 1000)
        x = _lb(x0, T, 300, min(25.0, (deadline - time.monotonic()) * 0.15))
        s = _ev(x)
        if s < best_sc:
            best_a, best_sc = x, s

    # Phase 4: Polish
    for T in [60.0, 120.0, 250.0]:
        if not _ok():
            break
        rem = max(5.0, deadline - time.monotonic())
        x = _lb(best_a.copy(), T, 200, 0.25 * rem)
        s = _ev(x)
        if s < best_sc:
            best_a, best_sc = x, s

    # Final validation
    fs = _ev(best_a)
    if not np.isfinite(fs):
        best_a = base.copy()
    out = [float(v) for v in best_a]
    if not out or any(not np.isfinite(v) for v in out):
        out = [1.0] * n
    return out
