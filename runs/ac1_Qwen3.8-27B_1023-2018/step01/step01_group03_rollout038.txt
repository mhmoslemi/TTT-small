```python
def _lp_step(q, M):
    """LP step: find g maximizing sum(g) s.t. 2*q*g - q*q <= M, g >= 0."""
    from scipy.optimize import linprog
    n = len(q)
    
    # Build the constraint matrix
    # Constraint for each k: 2*sum_i q_i * g_{k-i} - (q*q)_k <= M
    # i.e., sum_i q_i * g_{k-i} <= (M + (q*q)_k) / 2

        for k in range(j, min(j+n, 2*n-1)):
```

```python
def _project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    # Algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _subgradient_step(q, weights, alpha):
    n = len(q)
    conv = np.convolve(q, q)
    M = conv.max()
    
    # Active set: indices where conv is close to max
    active = np.where(conv >= 0.9 * M)[0]
    
    # Compute gradient direction
    grad = np.zeros(n)
    for k in active:
        w = weights[k]
        # grad_j += w * q[k-j] for valid j
        for j in range(n):
            if 0 <= k - j < n:
                grad[j] += w * q[k-j]
    
    # Projected gradient step
    q_new = q - alpha * grad
    q_new = _project_simplex(q_new)
    return q_new
```

```python
        s = q.sum()
        if s > 1e-14:
            q /= s

    return q


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgradient_step(q, alpha, thresh_frac=0.9):
    n = len(q)
    conv = np.convolve(q, q)
    M = float(conv.max())
    if M <= 0:
        return q

    active = np.where(conv >= thresh_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])

    # Weighted gradient
    weights = np.exp((conv[active] - M) * 12.0)
    weights /= weights.sum()

    grad = np.zeros(n)
    for k, w in zip(active, weights):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad[j_start:j_end + 1] += w * q[k - np.arange(j_start, j_end + 1)]

    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return q

    q_new = q - alpha * grad / gn
    q_new = _project_simplex(q_new)
    return q_new


def _optimize_sequence(q, rng, deadline, phase_label=""):
    """Main optimization: alternate LP and coordinate descent."""
    n = len(q)
    q = np.asarray(q, dtype=np.float64).copy()
    q = np.maximum(0, q)
    s = q.sum()
    if s < 1e-14:
        q = np.ones(n) / n
        s = 1.0
    q = q / s

    best_q_local = q.copy()
    best_sc_local = _score_fast(q)

    def _consider(qc):
        nonlocal best_q_local, best_sc_local
        sc = _score_fast(qc)
        if sc < best_sc_local - 1e-12:
            best_sc_local = sc
            best_q_local = qc.copy()

    # Phase A: LP iterations
    stag = 0
    for it in range(300):
        if time.monotonic() > deadline - 200:
            break
        g = _lp_step(q)
        if g is None:
            break

        cur_sc = _score_fast(q)
        best_t = q
        best_t_sc = cur_sc
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            cand = (1 - t) * q + t * g
            cand = np.maximum(0, cand)
            cs = cand.sum()
            if cs < 1e-14:
                continue
            cand /= cs
            sc = _score_fast(cand)
            if sc < best_t_sc:
                best_t_sc = sc
                best_t = cand

        if best_t_sc < cur_sc - 1e-11:
            q = best_t
            _consider(q)
            stag = 0
        else:
            stag += 1
            if stag >= 10:
                break

    # Phase B: Coordinate descent
    no_improve_streak = 0
    prev_sc = _score_fast(q)
    pass_count = 0

    while time.monotonic() < deadline - 80:
        q = _cd_pass(q, deadline - 80)
        pass_count += 1
        sc = _score_fast(q)
        if sc < prev_sc - 1e-11:
            prev_sc = sc
            no_improve_streak = 0
            _consider(q)
        else:
            no_improve_streak += 1
            if no_improve_streak >= 20:
                # Perturb and restart
                scale = 1.0 + 0.03 * rng.normal(size=n)
                q = q * scale
                q = np.maximum(0, q)
                qs = q.sum()
                if qs < 1e-14:
                    q = best_q_local.copy()
                else:
                    q /= qs
                prev_sc = _score_fast(q)
                no_improve_streak = 0

    # Phase C: Subgradient refinement
    alpha = 0.1
    for it in range(500):
        if time.monotonic() > deadline - 40:
            break
        q_new = _subgradient_step(q, alpha)
        sc = _score_fast(q_new)
        if sc < _score_fast(q) - 1e-12:
            q = q_new
            _consider(q)
            alpha = min(0.3, alpha * 1.05)
        else:
            alpha = max(0.005, alpha * 0.9)
            if alpha <= 0.006:
                break

    # Phase D: Final CD polish
    while time.monotonic() < deadline - 20:
        q = _cd_pass(q, deadline - 20)
        _consider(q)
        if _score_fast(q) >= best_sc_local:
            break

    return best_q_local, best_sc_local


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    start_time = time.monotonic()
    deadline = start_time + max(20.0, float(budget_s) - 50.0)
    rng = np.random.default_rng(seed)

    # Get initial sequence from parent
    q_init = None
    try:
        h = globals().get('height_sequence_1', None)
        if h is not None:
            q_init = np.array(h, dtype=np.float64).copy()
            q_init = np.maximum(0.0, q_init)
            s = q_init.sum()
            if s > 0.01:
                q_init = q_init / s
    except Exception:
        pass

    # Fallback construction
    if q_init is None:
        n_fb = 64
        q_init = np.ones(n_fb) / n_fb

    best_q = q_init.copy()
    best_score = evaluate_sequence(best_q.tolist())

    def _consider(qc):
        nonlocal best_q, best_score
        try:
            qc = np.asarray(qc, dtype=np.float64)
            qc = np.maximum(0.0, qc)
            s = qc.sum()
            if s < 0.01 or not np.isfinite(s):
                return
            qc = qc / s
            sc = evaluate_sequence(qc.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_q = qc.copy()
        except Exception:
            pass

    _consider(q_init)

    # Main optimization on parent length
    n = len(q_init)
    if time.monotonic() < deadline - 300:
        q_opt, sc_opt = _optimize_sequence(q_init, rng, deadline - 250)
        _consider(q_opt)

    # Try slightly different lengths by upsampling
    if time.monotonic() < deadline - 200:
        for target_n in [int(n * 1.25), int(n * 0.8)]:
            if time.monotonic() > deadline - 100:
                break
            q_up = np.interp(
                np.linspace(0, 1, target_n),
                np.linspace(0, 1, n),
                best_q
            )
            q_up = np.maximum(0, q_up)
            s = q_up.sum()
            if s < 0.01:
                continue
            q_up /= s
            _consider(q_up)

            if time.monotonic() < deadline - 100:
                q_opt2, sc_opt2 = _optimize_sequence(q_up, rng, deadline - 80)
                _consider(q_opt2)

    # Final: ensure we have a valid answer
    _consider(best_q)

    result = [float(x) for x in best_q]

    # Safety: verify with evaluator
    try:
        final_sc = evaluate_sequence(result)
        if not np.isfinite(final_sc):
            result = [1.0 / 4, 1.0 / 4, 1.0 / 4, 1.0 / 4]
    except Exception:
        result = [1.0 / 4, 1.0 / 4, 1.0 / 4, 1.0 / 4]

    return result
```
