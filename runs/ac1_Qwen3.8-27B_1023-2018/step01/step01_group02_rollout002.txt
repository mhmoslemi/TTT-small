        out = a.copy()
    elif a.size == 0:
        out = np.ones(m, dtype=np.float64) / float(m)
    else:
        x0 = np.linspace(0.0, 1.0, a.size)
        x1 = np.linspace(0.0, 1.0, m)
        out = np.interp(x1, x0, a)
    return _normalize(out)


def _conv_full(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n <= 0:
        return np.array([], dtype=np.float64)
    if n <= 512:
        return np.convolve(a, a)
    L = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
    return c[: 2 * n - 1]


def _score_internal(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = _conv_full(a)
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * a.size * m / (s * s))


def _eval_internal(a):
    a = np.asarray(a, dtype=np.float64)
    s = float(a.sum())
    c = _conv_full(a)
    if c.size == 0 or not np.isfinite(s) or s <= 1e-12:
        inf = np.inf
        return c, inf, inf
    m = float(np.max(c))
    if not np.isfinite(m):
        return c, m, np.inf
    score = float(2.0 * a.size * m / (s * s))
    return c, m, score


def _cross(a, p):
    a = np.asarray(a, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = int(a.size)
    m = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n <= 256:
        conv = np.convolve(a, p[::-1])
        return conv[m - 1 - np.arange(n, dtype=np.int64)]
    L = 1 << (n + m - 2).bit_length()
    A = np.fft.rfft(a, L)
    P = np.fft.rfft(p[::-1], L)
    conv = np.fft.irfft(A * P, L)
    return conv[m - 1 - np.arange(n, dtype=np.int64)]


def _weights(c, beta, frac):
    c = np.asarray(c, dtype=np.float64)
    cmax = float(np.max(c))
    if c.size == 0 or not np.isfinite(cmax) or cmax <= 0.0:
        p = np.ones_like(c, dtype=np.float64)
        s = p.sum()
        if s <= 0.0:
            return p
        return p / s

    z = beta * (c - cmax)
    z = np.clip(z, -745.0, 0.0)
    p_soft = np.exp(z)
    s = float(p_soft.sum())
    if not np.isfinite(s) or s <= 0.0:
        p_soft = np.zeros_like(c, dtype=np.float64)
        p_soft[int(np.argmax(c))] = 1.0
    else:
        p_soft = p_soft / s

    order = np.argsort(c)[::-1]
    gap = cmax - float(np.mean(c))
    if not np.isfinite(gap) or gap <= 1e-15:
        gap = max(1e-15, cmax * 1e-6)
    tol = max(1e-15, 0.09 * gap)
    k = 1
    for idx in order[1:]:
        if c[idx] >= cmax - tol:
            k += 1
        else:
            break
    longest = max(5, int(0.08 * c.size))
    k = max(1, min(k, longest))
    if c.size > 3:
        k = max(k, 3)

    p_top = np.zeros_like(c, dtype=np.float64)
    p_top[order[:k]] = 1.0
    p_top /= float(k)

    top_weight = 0.22 + 0.42 * frac
    p = (1.0 - top_weight) * p_soft + top_weight * p_top
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return p_soft
    return p / s


def _adam_state(dim):
    return {
        "m": np.zeros(int(dim), dtype=np.float64),
        "v": np.zeros(int(dim), dtype=np.float64),
        "t": 0.0,
    }


def _adam_update(param, grad, state, lr, cap=0.4):
    g = np.asarray(grad, dtype=np.float64)
    dim = int(param.size)
    if g.size != dim:
        g = np.zeros(dim, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    state["t"] += 1.0
    state["m"] = b1 * state["m"] + (1.0 - b1) * g
    state["v"] = b2 * state["v"] + (1.0 - b2) * (g * g)

    t = float(state["t"])
    mhat = state["m"] / (1.0 - b1 ** t)
    vhat = state["v"] / (1.0 - b2 ** t)
    step = float(lr) * mhat / (np.sqrt(vhat) + eps)
    if not np.all(np.isfinite(step)):
        step = np.zeros(dim, dtype=np.float64)
    step = np.clip(step, -float(cap), float(cap))

    out = param - step
    return out, step


def _choose_lp_method():
    global _LP_METHOD
    if _LP_METHOD is not None:
        return _LP_METHOD
    if _linprog is None:
        _LP_METHOD = "none"
        return _LP_METHOD
    for meth in ("highs", "interior-point", "revised-simplex"):
        try:
            res = _linprog(
                c=np.array([-1.0]),
                A_ub=np.array([[1.0]]),
                b_ub=np.array([1.0]),
                bounds=[(0.0, None)],
                method=meth,
            )
            if res is not None and getattr(res, "success", False):
                _LP_METHOD = meth
                return _LP_METHOD
        except Exception:
            pass
    _LP_METHOD = "none"
    return _LP_METHOD


def _lp_reciprocal(f, M, cap=None):
    method = _choose_lp_method()
    if method == "none" or _linprog is None:
        return None
    f = np.asarray(f, dtype=np.float64)
    n = int(f.size)
    if n <= 0 or not np.isfinite(M) or M <= 0.0:
        return None

    f = np.where(np.isfinite(f), f, 0.0)
    f = np.maximum(f, 1e-12)
    s = float(f.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    f = f / s

    rows = np.arange(2 * n - 1, dtype=np.int64)[:, None]
    cols = np.arange(n, dtype=np.int64)[None, :]
    d = rows - cols
    mask = (d >= 0) & (d < n)
    F = np.zeros((2 * n - 1, n), dtype=np.float64)
    F[mask] = f[d[mask]]

    c = -np.ones(n, dtype=np.float64)
    if cap is None:
        bounds = [(0.0, None)] * n
    else:
        cap = float(cap)
        if not np.isfinite(cap) or cap <= 0.0:
            bounds = [(0.0, None)] * n
        else:
            bounds = [(0.0, cap)] * n

    res = None
    try:
        res = _linprog(c, A_ub=F, b_ub=float(M), bounds=bounds, method=method)
    except Exception:
        res = None

    if (res is None or not getattr(res, "success", False)) and method != "interior-point":
        try:
            res = _linprog(c, A_ub=F, b_ub=float(M), bounds=bounds, method="interior-point")
        except Exception:
            res = None

    if res is None or not getattr(res, "success", False):
        try:
            res = _linprog(c, A_ub=F, b_ub=float(M), bounds=bounds, method="revised-simplex")
        except Exception:
            res = None

    if res is None or not getattr(res, "success", False):
        return None

    x = np.asarray(res.x, dtype=np.float64)
    if x.size != n:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    xs = float(x.sum())
    if not np.isfinite(xs) or xs <= 1e-14:
        return None
    return x / xs


def _random_polish(a, rng, trials=4):
    a = np.asarray(a, dtype=np.float64)
    if a.size <= 0:
        return a, np.inf
    n = int(a.size)
    best = np.maximum(a, 0.0)
    s = float(best.sum())
    if not np.isfinite(s) or s <= 1e-12:
        best = np.ones(n, dtype=np.float64) / float(n)
    else:
        best = best / s
    best_sc = _score_internal(best)
    if not np.isfinite(best_sc):
        best = np.ones(n, dtype=np.float64) / float(n)
        best_sc = _score_internal(best)

    for _ in range(int(max(1, trials))):
        cand = best.copy()

        nb = int(rng.integers(1, 4))
        for _i in range(nb):
            maxw = max(2, n // 20)
            w = int(rng.integers(1, maxw + 1))
            w = min(max(1, w), n)
            if n > 1:
                st = int(rng.integers(0, n - w + 1))
            else:
                st = 0
            factor = float(np.exp(rng.normal(0.0, 0.18)))
            cand[st:st + w] *= factor

        if n > 1 and rng.random() < 0.45:
            m = int(rng.integers(1, max(2, n // 8)))
            idx = rng.integers(0, n, size=m)
            cand[idx] *= float(np.exp(rng.normal(0.0, 0.20)))

        if n > 8 and rng.random() < 0.30:
            off = int(rng.integers(-n // 20, n // 20 + 1))
            cand = 0.5 * (cand + np.roll(cand, off))

        if rng.random() < 0.25:
            cand = cand * np.exp(rng.normal(0.0, 0.08, n))

        cand = np.where(np.isfinite(cand), cand, 0.0)
        cand = np.maximum(cand, 0.0)
        cs = float(cand.sum())
        if not np.isfinite(cs) or cs <= 1e-12:
            continue
        cand = cand / cs
        sc = _score_internal(cand)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best = cand.copy()

    return best, best_sc


def _make_inits(n, parent, rng):
    n = int(max(1, n))
    cands = []

    def add(x):
        try:
            a = _normalize(x)
            if a.size == n and np.all(np.isfinite(a)):
                cands.append(a)
        except Exception:
            pass

    if parent is not None:
        p = _upsample(parent, n)
        add(p)
        add(p[::-1])
        add(p * (1.0 + 0.05 * rng.standard_normal(n)))
        add(p * (1.0 + 0.12 * rng.standard_normal(n)))

    add(np.ones(n, dtype=np.float64) / float(n))

    if n > 2:
        x = np.linspace(0.0, 1.0, n)

        for amp in (0.20, 0.45, 0.70, 0.85):
            for ff in (1, 2, 3, 5, 8):
                ph = float(rng.uniform(0.0, 2.0 * np.pi))
                add(1.0 + amp * np.cos(2.0 * np.pi * ff * x + ph))
                add(1.0 + 0.5 * amp * np.sin(2.0 * np.pi * ff * x + 2.0 * ph))

        for g in (1.0, 2.0, 4.0, 8.0, 14.0):
            add(np.exp(g * (x - 0.5)) + 1e-6)
            add(np.exp(-g * (x - 0.5)) + 1e-6)
            add(np.exp(-g * (x - 0.5) ** 2) + 1e-6)
            add(np.exp(-g * x) + 1e-6)
            add(np.exp(-g * (1.0 - x)) + 1e-6)

        add(1.0 / (np.sqrt(x + 1e-3) + 1e-3))
        add(1.0 / (np.sqrt(1.0 - x + 1e-3) + 1e-3))
        add(1.0 / (np.sqrt(x + 1e-3) + np.sqrt(1.0 - x + 1e-3) + 1e-3))
        add(np.sqrt(np.maximum(x * (1.0 - x), 1e-8)) + 0.25)

        for c1, c2, ss in ((0.25, 0.75, 0.12), (0.30, 0.70, 0.08), (0.20, 0.80, 0.16)):
            add(np.exp(-((x - c1) / ss) ** 2) + np.exp(-((x - c2) / ss) ** 2) + 1e-6)

        for scale in (0.30, 0.80, 1.50, 2.50, 4.00):
            z = rng.standard_normal(n) * scale
            z = z - float(np.mean(z))
            z = np.clip(z, -30.0, 30.0)
            e = np.exp(z)
            e = e / float(e.sum())
            add(e)

    if not cands:
        cands.append(np.ones(n, dtype=np.float64) / float(n))

    scored = []
    for a in cands:
        sc = _score_internal(a)
        if np.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda xx: xx[0])
    if len(scored) <= 14:
        return [a for _, a in scored]
    return [a for _, a in scored[:14]]


def _build_basis(n, rng):
    n = int(max(1, n))
    if n <= 1:
        return np.zeros((n, 0), dtype=np.float64)
    K = min(n - 1, max(30, n // 2))
    k_dct = min(K, max(8, int(0.6 * K)))
    k_rand = K - k_dct
    parts = []

    if k_dct > 0:
        rows = ((np.arange(n, dtype=np.float64) + 0.5) / float(n))[:, None]
        cols = np.arange(1, k_dct + 1, dtype=np.float64)[None, :]
        Vd = np.sqrt(2.0 / float(n)) * np.cos(np.pi * rows * cols)
        parts.append(Vd)

    if k_rand > 0:
        parts.append(rng.standard_normal((n, k_rand)))

    if not parts:
        return np.zeros((n, 0), dtype=np.float64)

    V = np.hstack(parts).astype(np.float64, copy=False)
    V = V - float(V.mean(axis=0, keepdims=True))
    V = np.where(np.isfinite(V), V, 0.0)

    try:
        if K <= n:
            Q, _ = np.linalg.qr(V)
            Q = Q[:, :K]
        else:
            Q = V
    except Exception:
        Q = V

    Q = np.where(np.isfinite(Q), Q, 0.0)
    norms = np.linalg.norm(Q, axis=0)
    norms = np.where(norms > 1e-12, norms, 1.0)
    Q = Q / norms[None, :]
    return Q


def _make_restart_z(n, best_a, pool, V, rng):
    n = int(n)
    r = float(rng.random())
    if pool and r < 0.32:
        base = pool[int(rng.integers(0, len(pool)))]
    elif r < 0.52:
        base = best_a[::-1]
    else:
        base = best_a

    base = _normalize(base)
    z = _logit(base)
    noise = 0.10 + 0.12 * float(rng.random())
    z = z + noise * rng.standard_normal(n)

    if V is not None and V.size > 0 and float(rng.random()) < 0.55:
        K = int(V.shape[1])
        if K > 0:
            z = V @ (V.T @ z)
            z = z + 0.08 * float(K) ** 0.5 * (V @ (rng.standard_normal(K) * 0.25))

    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.mean(z))
    return np.clip(z, -50.0, 50.0)


def _exact_from_array(a):
    try:
        vals = [float(x) for x in np.asarray(a, dtype=np.float64)]
        if not vals:
            return np.inf
        if any((not np.isfinite(x)) for x in vals):
            return np.inf
        val = evaluate_sequence(vals)
        if np.isfinite(val):
            return float(val)
    except Exception:
        pass
    return np.inf


def _optimize_main(n, inits, start, deadline, rng):
    n = int(max(1, n))
    deadline = float(deadline)
    start = float(start)

    scored_inits = []
    seen = set()
    for a in inits:
        try:
            a = np.asarray(a, dtype=np.float64)
            if a.size != n:
                a = _upsample(a, n)
            a = _normalize(a)
            key = tuple(np.round(a, 12).tolist())
            if key in seen:
                continue
            seen.add(key)
            sc = _score_internal(a)
            if np.isfinite(sc):
                scored_inits.append((sc, a))
        except Exception:
            pass

    if not scored_inits:
        arr = np.ones(n, dtype=np.float64) / float(n)
        return arr

    scored_inits.sort(key=lambda xx: xx[0])
    pool = [a.copy() for _, a in scored_inits[:min(12, len(scored_inits))]]

    best_sc, best_a = scored_inits[0][0], scored_inits[0][1].copy()
    best_exact = _exact_from_array(best_a)
    if not np.isfinite(best_exact):
        best_exact = best_sc

    V = _build_basis(n, rng)
    z = _logit(best_a)
    z = z - float(np.mean(z))
    state = _adam_state(n)
    lr = 0.18
    cap = 0.45

    no_improve = 0
    threshold = 80
    it = 0
    lp_calls = 0
    lp_idx = 0
    last_lp = start - 1000.0
    last_polish = start - 1000.0
    last_exact = start

    lp_caps = [None, 5.0 / float(n), 10.0 / float(n), 25.0 / float(n)]

    while True:
        now = time.monotonic()
        if now > deadline - 5.0:
            break

        it += 1
        a = _softmax(z)
        c, M, sc = _eval_internal(a)

        if not np.isfinite(sc):
            z = _logit(best_a)
            z = z - float(np.mean(z))
            state = _ adam_state(n)
            no_improve = 0
            continue

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
            lr = min(0.34, lr * 1.01)
            if it % 25 == 0 or (now - last_exact) > 0.5:
                esc = _exact_from_array(best_a)
                if np.isfinite(esc) and esc < best_exact:
                    best_exact = esc
                last_exact = now
        else:
            no_improve += 1

        if (_linprog is not None and n <= 1024 and
                lp_calls < 120 and
                (now - last_lp) > 12.0 and
                (deadline - now) > 30.0 and
                (no_improve > 25 or it % 250 == 0)):
            M_best = best_sc / (2.0 * float(n))
            cap_opt = lp_caps[lp_idx % len(lp_caps)]
            g = _lp_reciprocal(best_a, M_best, cap_opt)
            if g is not None:
                improved = False
                for t in (0.05, 0.10, 0.18, 0.30, 0.50, 0.75):
                    cand = (1.0 - t) * best_a + t * g
                    cand = np.where(np.isfinite(cand), cand, 0.0)
                    cand = np.maximum(cand, 0.0)
                    ss = float(cand.sum())
                    if not np.isfinite(ss) or ss <= 1e-12:
                        continue
                    cand = cand / ss
                    csc = _score_internal(cand)
                    if np.isfinite(csc) and csc < best_sc - 1e-12:
                        best_sc = csc
                        best_a = cand.copy()
                        no_improve = 0
                        z = _logit(best_a)
                        z = z - float(np.mean(z))
                        state = _adam_state(n)
                        lr = min(0.30, lr * 1.15)
                        esc = _exact_from_array(best_a)
                        if np.isfinite(esc) and esc < best_exact:
                            best_exact = esc
                        improved = True
                        break
                if not improved:
                    lp_idx += 1
            lp_calls += 1
            last_lp = time.monotonic()

        if it % 150 == 0 and (now - last_polish) > 4.0:
            imp_a, imp_sc = _random_polish(best_a, rng, trials=4)
            if np.isfinite(imp_sc) and imp_sc < best_sc - 1e-12:
                best_sc = imp_sc
                best_a = imp_a.copy()
                no_improve = 0
                z = _logit(best_a)
                z = z - float(np.mean(z))
                state = _adam_state(n)
                esc = _exact_from_array(best_a)
                if np.isfinite(esc) and esc < best_exact:
                    best_exact = esc
            last_polish = time.monotonic()

        if no_improve >= threshold or sc > best_sc + 0.02:
            z = _make_restart_z(n, best_a, pool, V, rng)
            state = _adam_state(n)
            lr = max(0.06, lr * 0.82)
            no_improve = 0
            threshold = min(2200, int(threshold * 1.22) + 1)
            continue

        frac = min(1.0, max(0.0, (now - start) / max(1e-6, deadline - start)))
        cmax = float(M) if np.isfinite(M) and M > 0.0 else 1e-12
        gap = cmax - float(np.mean(c)) if np.all(np.isfinite(c)) else cmax * 1e-4
        if not np.isfinite(gap) or gap <= 1e-15:
            gap = max(1e-15, cmax * 1e-5)
        beta = (1.0 / max(1e-12, 0.055 * gap)) * (0.35 + 1.75 * frac)
        beta = min(max(100.0, beta), 1.0e7)

        p = _weights(c, beta, frac)
        d = 2.0 * _cross(a, p)
        d = np.where(np.isfinite(d), d, 0.0)
        gvec = a * (d - float(np.dot(a, d)))
        gvec = np.where(np.isfinite(gvec), gvec, 0.0)

        try:
            z_new, _ = _adam_update(z, gvec, state, lr, cap)
        except Exception:
            z_new = z + 0.02 * rng.standard_normal(n)

        z = np.where(np.isfinite(z_new), z_new, 0.0)
        z = z - float(np.mean(z))
        z = np.clip(z, -50.0, 50.0)

        if no_improve > 12 and float(rng.random()) < 0.12:
            z = z + 0.02 * rng.standard_normal(n)
            z = z - float(np.mean(z))
            z = np.clip(z, -50.0, 50.0)

    for _ in range(4):
        imp_a, imp_sc = _random_polish(best_a, rng, trials=5)
        if np.isfinite(imp_sc) and imp_sc < best_sc - 1e-12:
            best_sc = imp_sc
            best_a = imp_a.copy()

    return _normalize(best_a)


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    try:
        B = float(budget_s)
    except Exception:
        B = 1300.0
    if not np.isfinite(B) or B <= 1.0:
        B = 1.0
    headroom = min(25.0, max(0.5, 0.18 * B))
    deadline = start + max(0.75, B - headroom)

    n = 127
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.maximum(p, 0.0)
            s = float(p.sum())
            if np.isfinite(s) and s > 1e-12:
                parent = p / s
                n = int(parent.size)
    except Exception:
        parent = None

    if parent is None:
        n = 127
        parent = np.ones(n, dtype=np.float64) / float(n)

    if n > 1024:
        parent = _upsample(parent, 1024)
        n = 1024
    if n < 1:
        n = 1
        parent = np.ones(1, dtype=np.float64)

    inits = _make_inits(n, parent, rng)
    if not inits:
        inits = [parent.copy(), np.ones(n, dtype=np.float64) / float(n)]

    if deadline - start < 12.0:
        quick = inits[:6]
    else:
        quick = _make_inits(n, parent, rng)

    final_cands = []

    def add_final(a):
        try:
            a = _normalize(a)
            if a.size > 0 and np.all(np.isfinite(a)):
                final_cands.append(a.copy())
        except Exception:
            pass

    add_final(parent)
    add_final(np.ones(n, dtype=np.float64) / float(n))
    add_final(np.array([0.5, 0.1, 0.4], dtype=np.float64))

    try:
        best = _optimize_main(n, quick, time.monotonic(), deadline, rng)
        add_final(best)
    except Exception:
        best = None

    for a in quick[:4]:
        add_final(a)

    best_list = None
    best_val = np.inf
    for a in final_cands:
        try:
            vals = [float(x) for x in np.asarray(a, dtype=np.float64)]
            if not vals:
                continue
            if any((not np.isfinite(x)) for x in vals):
                continue
            val = evaluate_sequence(vals)
            if np.isfinite(val) and val < best_val:
                best_val = float(val)
                best_list = vals
        except Exception:
            pass

    if best_list is None:
        best_list = [0.5, 0.1, 0.4]

    return best_list
