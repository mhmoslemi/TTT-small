    # We want: for each j, sum_k p[k]*a[k-j]
    # This equals: (flip(p) * a)[j+n-1] ... let me just do it with FFT cross-correlation
    # corr(j) = sum_k p[k] * a[k-j] = (p_reversed * a)[j + len(p) - 1]
    P_rev = np.fft.rfft(p[::-1], L)
    cross = np.fft.irfft(P_rev * A, L)[:L]
    # cross[k] = sum_i p_rev[i] * a[k-i] = sum_i p[len(p)-1-i] * a[k-i]
    # = sum_j p[j] * a[k - len(p) + 1 + j]... this is getting confusing.
    
    # Simpler: just compute the gradient directly
    # grad[j] = 2 * sum_k p[k] * a[k-j] (where 0 <= k-j < n)
    # = 2 * (cross_correlation of p and a at lag j)
    # cross_correlation(lag j) = sum_k p[k] * a[k-j]
    
    # Use FFT: cross_correlation(j) = sum_k p[k] * a[k-j]
    # = (p * a)[2n-2-j]... 
    # (p * a)[k] = sum_i p[i] * a[k-i]
    # We want sum_k p[k] * a[k-j]: let i=k, then sum_i p[i] * a[i-j]
    # This is the cross-correlation. In terms of convolution:
    # sum_i p[i] * a[i-j] = (p * a_flipped)[j + len(a) - 1] ... 
    
    # Let me just use the straightforward approach for moderate n
    if n <= 2048:
        # grad[j] = 2 * sum_k p[k] * a[k-j] for 0 <= k-j < n
        # = 2 * sum_{m: 0<=m<n} p[m+j] * a[m] (where m = k-j, k = m+j)
        # = 2 * sum_m a[m] * p[m+j]
        # This is the cross-correlation of a and p at lag j.
        # = 2 * fft-based cross-correlation
        L2 = 1 << (2 * n).bit_length()
        A2 = np.fft.rfft(a, L2)
        P2 = np.fft.rfft(p, L2)
        # cross[j] = sum_m a[m] * p[m+j] = (a * p_reversed)[j+n-1]
        P2r = np.fft.rfft(p[::-1], L2)
        conv_full = np.fft.irfft(A2 * P2r, L2)[:2*n-1]
        # conv_full[k] = sum_i a[i] * p_rev[k-i] = sum_i a[i] * p[n-1-(k-i)] = sum_i a[i]*p[n-1-k+i]
        # We want sum_m a[m]*p[m+j] = conv_full[...] 
        # p[n-1-k+i] = p[m+j] when m=i, n-1-k+i = i+j → n-1-k = j → k = n-1-j
        # So cross[j] = conv_full[n-1-j]
        grad = 2.0 * np.zeros(n)
        for j in range(n):
            idx = n - 1 - j
            if 0 <= idx < len(conv_full):
                grad[j] = 2.0 * conv_full[idx]
    else:
        # For large n, use the full vector approach
        grad = 2.0 * np.ones(n) * float(np.dot(p, a))  # rough approximation
    
    # Also need to account for the 1/(sum_a)² factor
    s = float(a.sum())
    grad = grad / (s * s)  # d/dc of 1/s² ... actually we need full chain rule
    
    # Full derivative: f = 2n * max_c / s²
    # df/da_j = 2n * [grad_c_j * s^{-2} + max_c * (-2) * s^{-3} * 1]
    # where grad_c_j = d(max_c)/da_j ≈ sum_k p[k] * 2*a[k-j]
    
    cmax = float(np.max(c))
    grad = (2.0 / (s * s)) * (2.0 * grad) - (4.0 * cmax / (s * s * s)) * 1.0
    
    return np.where(np.isfinite(grad), grad, 0.0)


def _optimize(n, a0, start, end, rng, beta0=500, beta1=20000):
    """Optimize using Adam on logit parameters."""
    n = int(n)
    a = np.asarray(a0, dtype=np.float64)
    a = np.maximum(a, 1e-15)
    s = a.sum()
    a = a / s  # normalize to sum=1
    
    best_a = a.copy()
    best_score = _score(a)
    
    z = np.log(a / max(a[-1], 1e-300))  # logit-like parameterization
    z -= np.mean(z)
    
    # Adam state
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 0.1
    cap = 0.5
    
    no_improve = 0
    threshold = 500
    it = 0
    
    while time.monotonic() < end:
        it += 1
        a = _softmax(z)
        c = np.convolve(a, a)
        sc = _score(a)
        
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        frac = min(1.0, it / max(it, 1))
        beta = beta0 + (beta1 - beta0) * min(1.0, it / 5000)
        
        grad = _grad_max_smooth(a, c, beta)
        
        # Update Adam
        t_adam += 1
        m_adam = b1 * m_adam + (1 - b1) * grad
        v_adam = b2 * v_adam + (1 - b2) * grad * grad
        m_hat = m_adam / (1 - b1 ** t_adam)
        v_hat = v_adam / (1 - b2 ** t_adam)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -cap, cap)
        z = z - step
        z = np.where(np.isfinite(z), z, 0.0)
        
        # Periodic random restart / perturbation
        if no_improve > threshold:
            zr = np.log(np.maximum(best_a, 1e-15) / max(best_a.min(), 1e-300))
            zr -= np.mean(zr)
            z = zr + 0.3 * rng.standard_normal(n)
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            t_adam = 0
            lr = max(0.05, lr * 0.5)
            no_improve = 0
            threshold = int(threshold * 1.3)
        
        # Reduce learning rate slowly
        lr = max(0.02, lr * 0.9999)
        
    return best_a, best_score


def _try_polish(a, rng, n_trials=5):
    """Local polish with random perturbations."""
    best = a.copy().astype(np.float64)
    best_s = _score(best)
    for _ in range(n_trials):
        cand = best.copy()
        # Multiply random windows by random factors
        for _ in range(rng.integers(1, 4)):
            w = int(rng.integers(2, max(3, n // 10)))
            start = int(rng.integers(0, len(cand) - w))
            cand[start:start+w] *= np.exp(rng.normal(0, 0.2))
        cand = np.maximum(cand, 0)
        cs = _score(cand)
        if np.isfinite(cs) and cs < best_s:
            best_s = cs
            best = cand.copy()
    return best, best_s


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 10.0

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            ps = parent.sum()
            if ps > 1e-12:
                parent = parent / ps
    except Exception:
        parent = None

    # Initial candidates
    candidates = []

    # Constant sequences (score = 2)
    for n in (3, 5, 11, 127, 2535):
        candidates.append(np.ones(n, dtype=np.float64) / n)

    # Parent
    if parent is not None:
        candidates.append(parent.copy())

    # Evaluate initial candidates
    best_score = np.inf
    best_arr = None
    for cand in candidates:
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_arr = cand.copy()

    # Try to beat the constant sequence (score 2)
    # Use the parent length for optimization
    n_opt = parent.size if parent is not None else 127
    # Also try smaller sizes where we might find better structures
    
    search_configs = []
    if parent is not None:
        search_configs.append((n_opt, parent.copy()))
    # Try medium sizes
    for n_try in (17, 31, 63, 95):
        search_configs.append((n_try, np.ones(n_try) / n_try))
    # Try the parent upsampled to different sizes
    if parent is not None:
        for n_try in (255, 511, 1023):
            x1 = np.linspace(0, 1, n_try)
            x0 = np.linspace(0, 1, parent.size)
            up = np.interp(x1, x0, parent)
            up = np.maximum(up, 0)
            us = up.sum()
            if us > 1e-12:
                up /= us
                search_configs.append((n_try, up))

    # Allocate time to each config
    time_per_config = max(30.0, (deadline - time.monotonic()) / len(search_configs))
    
    for idx, (n_cfg, a_init) in enumerate(search_configs):
        if time.monotonic() > deadline - 15:
            break
        cfg_end = min(deadline - 15, time.monotonic() + time_per_config)
        if cfg_end <= time.monotonic() + 5:
            continue
        
        try:
            a_opt, sc_opt = _optimize(n_cfg, a_init, time.monotonic(), cfg_end, rng)
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_arr = a_opt.copy()
        except Exception:
            pass
        
        # Quick polish
        if time.monotonic() < deadline - 15:
            try:
                polished, sc_pol = _try_polish(best_arr, rng, n_trials=10)
                if np.isfinite(sc_pol) and sc_pol < best_score:
                    best_score = sc_pol
                    best_arr = polished
            except Exception:
                pass

    # Extended optimization on best found
    if time.monotonic() < deadline - 30 and best_arr is not None:
        n_ext = best_arr.size
        ext_end = deadline - 15
        try:
            a_ext, sc_ext = _optimize(n_ext, best_arr, time.monotonic(), ext_end, rng,
                                       beta0=5000, beta1=50000)
            if np.isfinite(sc_ext) and sc_ext < best_score:
                best_score = sc_ext
                best_arr = a_ext.copy()
        except Exception:
            pass

    # Final polish
    if time.monotonic() < deadline - 5 and best_arr is not None:
        try:
            polished, sc_pol = _try_polish(best_arr, rng, n_trials=30)
            if np.isfinite(sc_pol) and sc_pol < best_score:
                best_score = sc_pol
                best_arr = polished
        except Exception:
            pass

    # Convert to list and verify with evaluator
    final_list = None
    final_score = np.inf
    test_candidates = [best_arr]
    if parent is not None:
        test_candidates.append(parent)
    test_candidates.append(np.ones(3) / 3)

    for tc in test_candidates:
        try:
            lst = [float(x) for x in np.asarray(tc, dtype=np.float64)]
            if not lst or any(not np.isfinite(x) for x in lst):
                continue
            sc = float(evaluate_sequence(lst))
            if np.isfinite(sc) and sc < final_score:
                final_score = sc
                final_list = lst
        except Exception:
            pass

    if final_list is None:
        final_list = [1.0/3, 1.0/3, 1.0/3]

    return final_list
