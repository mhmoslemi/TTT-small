import numpy as np
import time


def _next_pow2(x):
    p = 1
    while p < x:
        p <<= 1
    return p

    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s)) - 12.0

    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

        a = np.ones(256, dtype=np.float64)
    n = int(a.size)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    a = a / s

    fft_len = _next_pow2(2 * n - 1)

    def compute_b(x):
        Fx = np.fft.rfft(x, fft_len)
        return np.fft.irfft(Fx * Fx, fft_len)[:2 * n - 1]

    def grad_topk(x, b, top_k, tau_frac):
        """Gradient of weighted max of top peaks of autocorrelation."""
        # Use argpartition for speed when top_k << len(b)
        if top_k < len(b) // 4:
            part = np.argpartition(b, -top_k)[-top_k:]
            top_idx = part[np.argsort(b[part])[::-1]]
        else:
            top_idx = np.argsort(b)[::-1][:top_k]
        top_vals = b[top_idx].copy()
        max_val = top_vals.max()
        tau = max(tau_frac * max_val, 1e-20)
        w = np.exp((top_vals - max_val) / tau)
        w /= w.sum()

        grad = np.zeros(2 * n - 1, dtype=np.float64)
        for k in range(top_k):
            m = int(top_idx[k])
            wk = w[k]
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            if lo <= hi:
                a_slice = x[m - hi:m - lo + 1][::-1]
                grad[lo:hi + 1] += 2.0 * wk * a_slice
        return grad[:n]

    # Small relative perturbation to break exact symmetry

            a_new = a_new / s_new
