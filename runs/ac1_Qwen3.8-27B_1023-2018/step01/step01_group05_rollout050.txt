    n = len(a)

def solve_lp_direction(f, M):
    n = len(f)
    # Build constraint matrix: (f⊛g)_k = sum_j f_{k-j} * g_j ≤ M
    # For each k, the coefficients of g_j are f_{k-j} for j where 0≤k-j≤n-1
    from scipy import sparse
    rows, cols, vals = [], [], []
    for k in range(2*n-1):
        j_lo = max(0, k-(n-1))
        j_hi = min(n-1, k)
        for j in range(j_lo, j_hi+1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize sum(g) = -minimize -sum(g)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
