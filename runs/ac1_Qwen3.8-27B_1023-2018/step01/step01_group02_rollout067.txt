def _softmax(z):
    z = np.asarray(z, dtype=np.float64).copy()
    z = z - np.maximum(np.max(z), 0.0)
    e = np.exp(np.clip(z, -700.0, 700.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.shape, 1.0 / float(len(z)), dtype=np.float64)
    return e / s


def _score_array(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    c = np.convolve(a, a)
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * n * mx / (s * s))


def _cross_corr(a, p, n_out):
    n_a = a.size
    L = 1 << max(1, (n_a + p.size - 1).bit_length())
    A = np.fft.rfft(a, L)
    P = np.fft.rfft(p, L)
    corr = np.fft.irfft(A * np.conj(P), L)
    if corr.size < n_out:
        pad = np.zeros(n_out - corr.size)
        corr = np.concatenate([corr, pad])
    return corr[:n_out]


def _grad_soft_max(a, c, beta):
    n = a.size
    cmax = float(np.max(c))
    zc = beta * (c - cmax)
    p = np.exp(np.clip(zc, -700.0, 700.0))
    ps = float(p.sum())
    if not np.isfinite(ps) or ps <= 0.0:
        p = np.zeros(len(c), dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
    else:
        p = p / ps
    f_val = float(p @ c)
    corr = _cross_corr(a, p, n)
    grad = 2.0 * a * (corr - f_val)
    return grad, f_val, p


def _adam_state(d):
    return {'m': np.zeros(d, dtype=np.float64),
            'v': np.zeros(d, dtype=np.float64),
            't': 0}


def _adam(params, grad, state, lr, cap=0.9):
    b1, b2, eps = 0.9, 0.999, 1e-8
    state['t'] += 1
    t = state['t']
    state['m'] = b1 * state['m'] + (1.0 - b1) * grad
    state['v'] = b2 * state['v'] + (1.0 - b2) * (grad * grad)
    mhat = state['m'] / (1.0 - b1 ** t)
    vhat = state['v'] / (1.0 - b2 ** t)
    step = lr * mhat / (np.sqrt(vhat) + eps)
    if cap is not None:
        step = np.clip(step, -cap, cap)
    out = params - step
    return out


def _lp_direction(f, M, max_time=0.4):
    if linprog is None:
        return None
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[k, j] = f[k - j]
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, float(M), dtype=np.float64)
    try:
        t0 = time.monotonic()
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs')
        if time.monotonic() - t0 > max_time:
            return None
        if res is not None and getattr(res, 'success', False) and res.x is not None:
            b = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
            if float(b.sum()) > 1e-10:
                return b
    except Exception:
        pass
    return None
