import numpy as np
import time
import cvxpy as cp

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    s = a.sum()
    if s < 1e-30:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)

def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

def _conv_matrix(f, n):
    """Build matrix A such that (f*b)_k = A[k,:] @ b for k=0..2n-2"""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        jmin = max(0, k-n+1)
        jmax = min(k+1, n)
        for j in range(jmin, jmax):
            A[k, k-j] = f[j]
    return A

def _lp_direction(f, M, timeout=5.0):
    """Solve LP: max sum(b) s.t. f*b <= M, b >= 0"""
    n = len(f)
    A = _conv_matrix(f, n)
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.GLOP, timeout=timeout)
        if b.value is not None and np.all(np.isfinite(b.value)):
            return np.maximum(b.value, 0)
    except:
        pass
    try:
        prob.solve(solver=cp.CBC, timeout=timeout)
        if b.value is not None and np.all(np.isfinite(b.value)):
            return np.maximum(b.value, 0)
    except:
        pass
    try:
        prob.solve(solver=cp.ECOS, timeout=timeout)
        if b.value is not None and np.all(np.isfinite(b.value)):
            return np.maximum(b.value, 0)
    except:
        pass
    return None

def _soft_max_grad(f, beta):
    """Compute softmax-weighted gradient of max(f*f)"""
    n = len(f)
    conv = np.convolve(f, f)
    cmax = conv.max()
    z = beta * (conv - cmax)
    p = np.exp(np.clip(z, -700, 0))
    p = p / p.sum()
    # Gradient of (f*f)_k w.r.t. f_i is (f*f) contribution... 
    # d/d f_i of (f*f)_k = f_{k-i} (with appropriate bounds)
    # So d/d f_i of sum_k p_k (f*f)_k = sum_k p_k f_{k-i}
    # This is the cross-correlation of p and f
    grad = np.zeros(n)
    for i in range(n):
        for k in range(2*n-1):
            j = k - i
            if 0 <= j < n:
                grad[i] += p[k] * f[j]
    return 2 * grad  # factor of 2 for the quadratic

def _lp_iteration(f, t=0.1):
    """One LP-based iteration"""
    conv = np.convolve(f, f)
    M = conv.max()
    b = _lp_direction(f, M)
    if b is None:
        return f
    bs = b.sum()
    fs = f.sum()
    if bs < 1e-30:
        return f
    b = b / bs * fs
    f_new = (1-t) * f + t * b
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 0:
        f_new /= s
    return f_new

def _adam_update(params, grad, state, lr):
    state['t'] += 1
    state['m'] = 0.9 * state['m'] + 0.1 * grad
    state['v'] = 0.999 * state['v'] + 0.001 * grad**2
    mhat = state['m'] / (1 - 0.9**state['t'])
    vhat = state['v'] / (1 - 0.999**state['t'])
    step = lr * mhat / (np.sqrt(vhat) + 1e-8)
    return params - step

def _logit(a):
    a = np.maximum(a, 1e-15)
    z = np.log(a)
    return z - z.mean()

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def _random_polish(a, rng, n_trials=5):
    best = a.copy()
    best_sc = _score(best)
    n = len(a)
    for _ in range(n_trials):
        cand = best.copy()
        # Random block scaling
        for _ in range(rng.integers(1, 4)):
            w = rng.integers(max(2, n//50), max(3, n//5))
            s = rng.integers(0, n - w)
            cand[s:s+w] *= np.exp(rng.normal(0, 0.15))
        # Optional small shift
        if n > 4 and rng.random() < 0.3:
            off = rng.integers(-n//15, n//15+1)
            cand = 0.5 * (cand + np.roll(cand, off))
        cand = _normalize(cand)
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best = cand.copy()
    return best, best_sc
