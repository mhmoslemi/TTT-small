    best_score = evaluate_sequence(a.tolist())

    def compute_conv(x):
        Fx = np.fft.rfft(x, fft_len)
        return np.fft.irfft(Fx * Fx, fft_len)[:conv_len]

    def compute_grad(x, weights):
        """Gradient of weighted max: grad_j = 2*sum_k w[k]*x[k-j] (range-checked)."""
        w_R = np.zeros(fft_len, dtype=np.float64)
        w_R[:conv_len] = weights[::-1]
        Fx = np.fft.rfft(x, fft_len)
        FW = np.fft.rfft(w_R, fft_len)
        conv_full = np.fft.irfft(Fx * FW, fft_len)
        return 2.0 * conv_full[n - 1:conv_len][::-1]

    # Adam optimizer state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t_adam = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 3.0 / n  # ~0.0006 for n=4806

    tau_frac = 0.10
    stagnation = 0
    iter_count = 0
    lr_min = 0.05 / n
    lr_max = 20.0 / n

    while time.monotonic() < deadline:
        iter_count += 1
        t_adam += 1

        # Time check inside loop
        if time.monotonic() > deadline - 5.0:
            break

        try:
            b = compute_conv(a)
            max_b = b.max()

            # Softmax weights with adaptive temperature
            tau = tau_frac * max(max_b, 1e-30)
            shifted = (b - max_b) / tau
            w = np.exp(shifted)
            w_sum = w.sum()
            w = w / w_sum

            # Gradient computation
            grad = compute_grad(a, w)
            grad -= grad.mean()  # center for scale invariance

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            # Try full step
            a_new = a - step
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()

            accepted = False
            if s_new > 1e-15:
                a_new = a_new / s_new
                max_b_new = compute_conv(a_new).max()
                if max_b_new < max_b * (1.0 - 1e-13):
                    a = a_new
                    accepted = True

            if not accepted:
                # Try half step
                a_try = a - 0.5 * step
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    max_b_try = compute_conv(a_try).max()
                    if max_b_try < max_b * (1.0 - 1e-13):
                        a = a_try
                        accepted = True

            if accepted:
                stagnation = 0
                lr = min(lr * 1.01, lr_max)
            else:
                lr = max(lr * 0.7, lr_min)
                stagnation += 1

            # Periodic exact evaluation
            if iter_count % 200 == 0 or (iter_count < 10 and iter_count % 2 == 0):
                sc = evaluate_sequence(a.tolist())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            # LP refinement when moderately stalled
            if (stagnation > 60 and stagnation <= 300
                    and iter_count % 100 == 0
                    and time.monotonic() < deadline - 60.0):
                try:
                    top_count = min(30, conv_len)
                    peak_idx = np.argsort(b)[-top_count:].astype(int)
                    M_val = float(b.max())

                    # Build constraint matrix: (top_count, n)
                    A_lp = np.zeros((top_count, n), dtype=np.float64)
                    for idx in range(top_count):
                        ki = peak_idx[idx]
                        lo = max(0, ki - (n - 1))
                        hi = min(n - 1, ki)
                        for i in range(lo, hi + 1):
                            j = ki - i
                            if 0 <= j < n:
                                A_lp[idx, i] = a[j]

                    res = linprog(
                        -np.ones(n),
                        A_ub=A_lp,
                        b_ub=np.full(top_count, M_val),
                        bounds=[(0.0, None)] * n,
                        method='highs',
                        options={'time_limit': 25, 'presolve': True}
                    )

                    if res.success and res.x is not None:
                        g = np.maximum(res.x, 0.0)
                        s_g = g.sum()
                        if s_g > 1e-10:
                            g = g / s_g
                            blend = 0.15 + 0.15 * min(float(stagnation) / 200.0, 1.0)
                            a_try = (1.0 - blend) * a + blend * g
                            a_try = np.maximum(a_try, 0.0)
                            s_try = a_try.sum()
                            if s_try > 1e-15:
                                a_try = a_try / s_try
                                max_b_try = compute_conv(a_try).max()
                                if max_b_try < max_b:
                                    a = a_try
                                    lr = 3.0 / n
                                    m = np.zeros(n, dtype=np.float64)
                                    v = np.zeros(n, dtype=np.float64)
                                    t_adam = 0
                                    stagnation = 0
                                    sc = evaluate_sequence(a.tolist())
                                    if np.isfinite(sc) and sc < best_score:
                                        best_score = sc
                                        best_a = a.copy()
                except Exception:
                    pass

            # Stalling recovery strategies
            if stagnation > 400:
                # Aggressive restart
                a = a * np.exp(rng.normal(0.0, 0.08 / n, n))
                a[n // 4:3 * n // 4] *= 1.0 + rng.normal(0.0, 0.05)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                lr = 3.0 / n
                m = np.zeros(n, dtype=np.float64)
                v = np.zeros(n, dtype=np.float64)
                t_adam = 0
                tau_frac = min(tau_frac * 2.0, 0.6)
                stagnation = 0
            elif stagnation > 200:
                # Moderate perturbation
                a = a * np.exp(rng.normal(0.0, 0.03 / n, n))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                tau_frac = min(tau_frac * 1.4, 0.5)
                lr = max(lr * 0.8, lr_min)
                stagnation = 0
            elif stagnation > 80:
                tau_frac = min(tau_frac * 1.15, 0.5)

            # Gradually sharpen softmax when making steady progress
            if stagnation == 0 and iter_count % 3000 == 0:
                tau_frac = max(tau_frac * 0.95, 0.005)

        except Exception:
            pass

    # Final phase: polish with tiny exact-step moves
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        try:
            b = compute_conv(a)
            max_b = b.max()
            # Exact subgradient descent (no smoothing)
            for polish_i in range(200):
                if time.monotonic() > deadline - 5.0:
                    break
                peak_pos = int(np.argmax(b))
                # Exact gradient: grad_j = 2*a[peak_pos - j] if valid
                grad_exact = np.zeros(n, dtype=np.float64)
                lo = max(0, peak_pos - (n - 1))
                hi = min(n - 1, peak_pos)
                for j in range(lo, hi + 1):
                    jj = peak_pos - j
                    if 0 <= jj < n:
                        grad_exact[j] = 2.0 * a[jj]
                grad_exact -= grad_exact.mean()
                grad_norm = np.linalg.norm(grad_exact)
                if grad_norm < 1e-15:
                    break
                step_size = 0.01 / n
                a_try = a - step_size * grad_exact
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    b_try = compute_conv(a_try)
                    if b_try.max() < max_b:
                        a = a_try
                        b = b_try
                        max_b = b.max()
                        sc = evaluate_sequence(a.tolist())
                        if np.isfinite(sc) and sc < best_score:
                            best_score = sc
                            best_a = a.copy()
        except Exception:
            pass

    # Final evaluation
    try:
        sc = evaluate_sequence(a.tolist())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
    except Exception:
        pass

    # Validate best_a
    try:
        final_sc = evaluate_sequence(best_a.tolist())
        if not np.isfinite(final_sc):
            best_a = np.ones(n, dtype=np.float64) / n
    except Exception:
        best_a = np.ones(n, dtype=np.float64) / n

    return [float(x) for x in best_a]
