def _simproj(v, s=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    r = np.arange(1, n + 1, dtype=np.float64)
    cond = mu > (cv - s) / r
    if not np.any(cond):
        return np.full(n, s / n, dtype=np.float64)
    rho = int(np.max(np.where(cond)[0])) + 1
    theta = (cv[rho - 1] - s) / rho
    x = np.clip(v - theta, 0.0, None)
    ss = float(x.sum())
    if ss > 1e-300 and np.isfinite(ss):

def _solve_lp_A(A, rhs, min_sum=0.1, tl=1.0):
    if linprog is None:
        return None

    m = A.shape[0]

        return None

    res = None
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=rhs,

def _top_grad_score(a, K=None, beta=40.0):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64), 0.0
    b = np.convolve(a, a)
    m = b.size
    mb = float(b.max())
    if mb <= 0.0 or not np.isfinite(mb):
        return np.zeros(n, dtype=np.float64), mb
    if K is None:
        K = min(m, max(8, int(np.sqrt(n))))
    K = max(1, int(K))
    if K >= m:
        idx = np.arange(m)
    else:
        idx = np.argpartition(b, -K)[-K:]
        idx = np.unique(np.concatenate((idx, m - 1 - idx))).astype(int)

    x = beta * (b[idx] - mb)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    s = float(e.sum())
    if s > 0.0 and np.isfinite(s):
        w = e / s
    else:
        w = np.zeros(idx.size, dtype=np.float64)
        w[int(np.argmax(b[idx]))] = 1.0

    grad = np.zeros(n, dtype=np.float64)
    for k, ww in zip(idx, w):
        if ww <= 1e-16:
            continue
        k = int(k)
        ist = max(0, k - n + 1)
        ien = min(n - 1, k)
        if ist > ien:
            continue
        js = k - ien
        je = k - ist
        grad[ist:ien + 1] += 2.0 * ww * a[js:je + 1][::-1]
    return (2.0 * float(n)) * grad, mb


def _subgradient_polish(a0, stop_at, max_iters=10000, K=16, beta=50.0):

class _GradientProblem:
    def __init__(self, a0, hard_deadline, K=None, beta=50.0, reg=1e-10):
        self.a0 = _normalize(a0)
        self.n = self.a0.size
        self.deadline = float(hard_deadline)
        self.beta = float(beta)
        self.reg = float(reg)
        self.best_a = self.a0.copy()
        self.best_sc = _score_arr(self.a0)
        self.last_z = None
        self.last_val = None
        self.last_grad = None
        bsize = 2 * self.n - 1
        if K is None:
            K = min(bsize, max(8, int(np.sqrt(self.n))))
        self.K = max(1, min(bsize, int(K)))

        self.seg = [None] * bsize
        n = self.n
        for k in range(bsize):
            ist = max(0, k - n + 1)
            ien = min(n - 1, k)
            js = k - ien
            je = k - ist
            self.seg[k] = (ist, ien, js, je)

    def _softmax(self, z):
        z = np.asarray(z, dtype=np.float64).ravel()
        if not np.all(np.isfinite(z)):
            z = np.where(np.isfinite(z), z, 0.0)
        z0 = z - float(z.mean())
        zwork = z0 - float(z0.max())
        e = np.exp(zwork)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 1e-300:
            a = np.full(self.n, 1.0 / self.n, dtype=np.float64)
        else:
            a = e / s
            sa = float(a.sum())
            if sa > 1e-300 and np.isfinite(sa):
                a = a / sa
            else:
                a = np.full(self.n, 1.0 / self.n, dtype=np.float64)
        return a, z0

    def _compute(self, z):
        a, z0 = self._softmax(z)
        b = np.convolve(a, a)
        mb = float(b.max())
        val = 2.0 * float(self.n) * mb
        if np.isfinite(val) and val < self.best_sc - 1e-14:
            self.best_sc = val
            self.best_a = a.copy()

        m = b.size
        K = self.K
        if K >= m:
            idx = np.arange(m)
        else:
            idx = np.argpartition(b, -K)[-K:]
            idx = np.unique(np.concatenate((idx, m - 1 - idx))).astype(int)

        x = self.beta * (b[idx] - mb)
        x = np.clip(x, -745.0, 0.0)
        e = np.exp(x)
        s = float(e.sum())
        if s > 0.0 and np.isfinite(s):
            w = e / s
        else:
            w = np.zeros(idx.size, dtype=np.float64)
            if idx.size:
                w[int(np.argmax(b[idx]))] = 1.0

        grad_b = np.zeros(self.n, dtype=np.float64)
        seg = self.seg
        for k, ww in zip(idx, w):
            if ww <= 1e-16:
                continue
            k = int(k)
            ist, ien, js, je = seg[k]
            if ist <= ien:
                grad_b[ist:ien + 1] += 2.0 * ww * a[js:je + 1][::-1]

        grad = (2.0 * float(self.n)) * grad_b
        ga = float(np.dot(grad, a))
        gz = grad - a * ga
        zr = np.clip(z0, -1000.0, 1000.0)
        val += self.reg * float(np.dot(zr, zr))
        gz += (2.0 * self.reg) * zr
        return val, gz

    def fun(self, z):
        if time.monotonic() > self.deadline - 2.0:
            raise _TimeUp
        if self.last_z is not None and np.array_equal(z, self.last_z):
            return self.last_val
        val, grad = self._compute(z)
        self.last_z = np.asarray(z, dtype=np.float64)
        self.last_val = float(val)
        self.last_grad = grad
        return self.last_val

    def jac(self, z):
        if time.monotonic() > self.deadline - 2.0:
            raise _TimeUp
        if self.last_z is not None and np.array_equal(z, self.last_z):
            return self.last_grad
        val, grad = self._compute(z)
        self.last_z = np.asarray(z, dtype=np.float64)
        self.last_val = float(val)
        self.last_grad = grad
        return self.last_grad

    def callback(self, z):
        if time.monotonic() > self.deadline - 3.0:
            raise _TimeUp

    def best(self):
        return self.best_a, self.best_sc


def _gradient_lbfgs(a0, stop_at, K=None, beta=50.0, maxiter=250, maxfun=2000):
    if minimize is None:
        return _normalize(a0)
    a0 = _normalize(a0)
    now = time.monotonic()
    if a0.size <= 1 or stop_at - now < 15:
        return a0
    prob = _GradientProblem(a0, max(now + 4.0, stop_at - 4.0), K=K, beta=beta)
    amax = float(a0.max())
    eps = max(1e-300, amax * 1e-12)
    z0 = np.log(np.maximum(a0, eps))
    z0 = z0 - float(z0.mean())

    try:
        minimize(
            prob.fun,
            z0,
            jac=prob.jac,
            method="L-BFGS-B",
            callback=prob.callback,
            options={
                "maxiter": max(20, int(maxiter)),
                "maxfun": max(50, int(maxfun)),
                "gtol": 1e-8,
                "maxls": 20,
            },
        )
    except Exception:
        pass
    return prob.best(self)[0]


def _upscale(a, k):
    k = int(k)
    if k <= 0:
        k = 1
    a = _normalize(a)
    n = a.size
    if n == k:
        return a
    if n == 1:
        return np.full(k, a[0], dtype=np.float64)
    if k > n:
        out = np.interp(
            np.linspace(0.0, 1.0, k, dtype=np.float64),
            np.linspace(0.0, 1.0, n, dtype=np.float64),
            a,
        )
    else:
        if n % k == 0:
            out = a.reshape(k, n // k).mean(axis=1)
        else:
            out = np.interp(
                np.linspace(0.0, 1.0, k, dtype=np.float64),
                np.linspace(0.0, 1.0, n, dtype=np.float64),
                a,
            )
    return _normalize(out)

    else:
        shifts = np.linspace(0, n - 1, min(n, 64), dtype=int)

    base = best.copy()
    for sh in shifts:
        if stop_at is not None and time.monotonic() > stop_at - 1.0:
            break
        sh = int(sh)

        cand = np.roll(base, sh)

    s2 = _score_arr(sym2)
    if s2 < bs - 1e-15:
        best = sym2
        bs = s2

        if np.isfinite(val) and val < best_exact:

            best_list = lst

    q4 = np.array([0.2, 0.4, 0.2, 0.2], dtype=np.float64)
    consider(q4)

    h1 = globals().get("height_sequence_1", None)
    have_parent = False
    champ = None
    if h1 is not None:
        consider(h1)
        champ = _normalize(h1)
        have_parent = True

    if champ is None:
        if budget_s > 350 and time.monotonic() < deadline - 250:
            prev = _upscale(q4, 16)
            for k in (32, 64, 128, 256, 512):
                if time.monotonic() > deadline - 260:
                    break
                a0 = _upscale(prev, k)
                stop_lp = min(deadline - 260, time.monotonic() + max(6.0, min(25.0, 0.02 * budget_s)))
                a = _lp_optimize(a0, stop_lp, max_lp=max(8, 1200 // max(1, k)))[0]
                stop_g = min(deadline - 260, time.monotonic() + max(4.0, min(12.0, 0.01 * budget_s)))
                a = _gradient_lbfgs(a, stop_g, K=8, beta=30.0, maxiter=60, maxfun=300)
                sc = _score_arr(a)
                consider(a)
                if sc < _score_arr(prev):
                    prev = a
            champ = prev
        else:
            champ = _upscale(q4, 256 if budget_s > 120 else 64)

    champ = _normalize(champ)
    champ_sc = _score_arr(champ)

    def accept(a, do_shift=True, shift_time=6.0):
        nonlocal champ, champ_sc
        if a is None:
            return champ
        a = _normalize(a)
        if do_shift:
            stop_shift = min(deadline - 10.0, time.monotonic() + max(1.0, float(shift_time)))
            a = _polish_sym_shift(a, stop_shift)[0]
        sc = _score_arr(a)
        consider(a)
        if np.isfinite(sc) and sc < champ_sc - 1e-14:
            champ = a
            champ_sc = sc
        return champ

    accept(champ, do_shift=True, shift_time=10.0)

    # Moderate exact-subgradient quasi-Newton pass.
    stop = min(deadline - 320.0, time.monotonic() + min(120.0, max(20.0, 0.08 * budget_s)))
    if stop - time.monotonic() > 15:
        champ = accept(
            _gradient_lbfgs(champ, stop, K=16, beta=25.0, maxiter=220, maxfun=1600),
            do_shift=True,
            shift_time=8.0,
        )

    # Primary LP active-set optimization.
    stop = min(deadline - 220.0, time.monotonic() + min(560.0, max(60.0, 0.45 * budget_s)))
    if stop - time.monotonic() > 30:
        champ = accept(_lp_optimize(champ, stop, max_lp=300)[0], do_shift=True, shift_time=10.0)

    # Sharper exact-subgradient polish.
    stop = min(deadline - 150.0, time.monotonic() + min(150.0, max(20.0, 0.12 * budget_s)))
    if stop - time.monotonic() > 15:
        champ = accept(
            _gradient_lbfgs(champ, stop, K=12, beta=80.0, maxiter=280, maxfun=2000),
            do_shift=True,
            shift_time=8.0,
        )

    # Exact projected top-lag subgradient polish.
    stop = min(deadline - 110.0, time.monotonic() + min(80.0, max(10.0, 0.06 * budget_s)))
    if stop - time.monotonic() > 10:
        champ = accept(
            _subgradient_polish(champ, stop, max_iters=20000, K=16, beta=50.0),
            do_shift=True,
            shift_time=8.0,
        )

    # Second LP pass on improved geometry.
    stop = min(deadline - 80.0, time.monotonic() + min(120.0, max(15.0, 0.09 * budget_s)))
    if stop - time.monotonic() > 20:
        champ = accept(_lp_optimize(champ, stop, max_lp=80)[0], do_shift=True, shift_time=6.0)

    # Final sharp polish.
    stop = min(deadline - 45.0, time.monotonic() + min(80.0, max(10.0, 0.06 * budget_s)))
    if stop - time.monotonic() > 15:
        champ = accept(
            _gradient_lbfgs(champ, stop, K=8, beta=120.0, maxiter=280, maxfun=1800),
            do_shift=True,
            shift_time=5.0,
        )

    stop = min(deadline - 30.0, time.monotonic() + min(40.0, max(5.0, 0.03 * budget_s)))
    if stop - time.monotonic() > 8:
        champ = accept(
            _subgradient_polish(champ, stop, max_iters=12000, K=12, beta=70.0),
            do_shift=True,
            shift_time=4.0,
        )

    # Try a longer length if time remains.
    if deadline - time.monotonic() > 130:
        trial_n = int(champ.size) * 2
        if 64 <= trial_n <= 1200:
            allowed = min(150.0, (deadline - 70.0 - time.monotonic()) * 0.45)
            if allowed > 50.0:
                hard = min(deadline - 60.0, time.monotonic() + allowed)
                cand = _quick_optimize(_upscale(champ, trial_n), allowed, hard)
                accept(cand, do_shift=True, shift_time=5.0)

    # Try a shorter length if current is already large.
    if champ.size > 512 and deadline - time.monotonic() > 90:
        trial_n = max(128, int(champ.size) // 2)
        allowed = min(90.0, (deadline - 55.0 - time.monotonic()) * 0.40)
        if allowed > 35.0:
            hard = min(deadline - 50.0, time.monotonic() + allowed)
            cand = _quick_optimize(_upscale(champ, trial_n), allowed, hard)
            accept(cand, do_shift=True, shift_time=4.0)

    # Final safe acceptance and validation.
    accept(champ, do_shift=True, shift_time=5.0)
    consider(champ)

    if best_list is None:
        d = _density(champ)
        if d is not None:
            best_list = [float(x) for x in d]
        else:
            best_list = [float(x) for x in q4]

    d = _density(best_list)
    if d is not None:
        best_list = [float(x) for x in d]

    if not best_list:
        best_list = [1.0]

    try:
        if eval_fn is not None:
            final_val = float(eval_fn(best_list))
            if not np.isfinite(final_val):
                fb = _density(np.array([0.2, 0.4, 0.2, 0.2], dtype=np.float64))
                best_list = [float(x) for x in fb]
    except Exception:
        best_list = [1.0]

    return [float(x) for x in best_list]
