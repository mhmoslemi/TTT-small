    if best_list is not None and time.monotonic() < deadline - 10:
        b_arr = np.asarray(best_list, dtype=np.float64)
        b_norm = _normalize(b_arr)
        fin_sc = _score_norm(b_norm)
        # Try a few gradient steps
        for s in (0.02, 0.01, 0.005):
            cand = _grad_step(b_norm, s)
            cs = _score_norm(cand)
            if cs < fin_sc - 1e-15:
                b_norm = cand
                fin_sc = cs
        consider(b_norm)

    # Safety
    if best_list is None:
        best_list = [float(x) for x in q4]

    # Final validation
    try:
        fv = float(evaluate_sequence(best_list))
        if not np.isfinite(fv):
            best_list = [float(x) for x in q4]
    except Exception:
        best_list = [float(x) for x in q4]

    # Ensure we return before deadline
    if time.monotonic() > deadline - 5.0:
        try:
            fv4 = float(evaluate_sequence([float(x) for x in q4]))
            if np.isfinite(fv4) and fv4 <= best_score:
                pass
        except Exception:
            pass

    return [float(x) for x in best_list]
