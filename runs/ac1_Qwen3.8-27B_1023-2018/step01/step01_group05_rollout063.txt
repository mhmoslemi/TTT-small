import numpy as np

def _fft_conv(a, b, fft_n):
    """Full linear convolution via FFT."""
    A = np.fft.rfft(a, fft_n)
    B = np.fft.rfft(b, fft_n)
    return np.fft.irfft(A * B, fft_n)[:len(a)+len(b)-1]

def _self_conv(a, fft_n):
    """Full linear self-convolution via FFT."""
    A = np.fft.rfft(a, fft_n)
    return np.fft.irfft(A * A, fft_n)[:2*len(a)-1]
