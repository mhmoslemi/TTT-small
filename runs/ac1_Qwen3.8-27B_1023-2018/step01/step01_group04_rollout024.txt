                    break
            
            if improved:
                stale = 0
                try_update(a_new)
            else:
                stale += 1
                if stale % 30 == 0:
                    x += 0.02 * rng.standard_normal(n)
                    stale = 0
                if stale > 100:
                    break
        
        a_final = np.exp(x - np.max(x))
        a_final = a_final / a_final.sum()
        try_update(a_final)
        return a_final
    
    # ===== MAIN SEARCH =====
    
    # Phase 1: LP iteration at parent length (bulk of time)
    n0 = best_arr.size
    stop1 = min(hard_deadline, t0 + budget_s * 0.45)
    a = lp_iteration(best_arr.copy(), stop1, max_iters=50000)
    
    # Phase 2: Gradient polish at same length
    stop2 = min(hard_deadline, t0 + budget_s * 0.60)
    a = gradient_polish(a, stop2, max_iters=50000)
    
    # Phase 3: Softmax optimization for additional local improvement
    if time.monotonic() < hard_deadline - 30:
        stop3 = min(hard_deadline, t0 + budget_s * 0.70)
        a = softmax_optimize(a, stop3, max_iters = 10000)
    
    # Phase 4: Another LP pass after gradient improvements
    if time.monotonic() < hard_deadline - 25:
        stop4 = min(hard_deadline, t0 + budget_s * 0.82)
        a = lp_iteration(a, stop4, max_iters=30000)
    
    # Phase 5: Final gradient polish
    if time.monotonic() < hard_deadline - 15:
        stop5 = min(hard_deadline, t0 + budget_s * 0.90)
        a = gradient_polish(a, stop5, max_iters=30000)
    
    # Phase 6: Try a few other lengths (interpolate from best)
    if time.monotonic() < hard_deadline - 25:
        for n_try in (64, 128, 192, 256):
            if time.monotonic() > hard_deadline - 15:
                break
            # Interpolate best to new length
            x_old = np.linspace(0, 1, best_arr.size)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_arr)
            a_try = normalize(a_try)
            
            sub_stop = min(hard_deadline - 12, time.monotonic() + 40)
            a_try = lp_iteration(a_try, sub_stop, max_iters=5000)
            
            if time.monotonic() < hard_deadline - 12:
                sub_stop2 = min(hard_deadline - 10, time.monotonic() + 20)
                a_try = gradient_polish(a_try, sub_stop2, max_iters=10000)
    
    # Final: one more short LP + gradient at the best length
    if time.monotonic() < hard_deadline - 8:
        final_stop = min(hard_deadline - 5, time.monotonic() + 10)
        a = lp_iteration(best_arr.copy(), final_stop, max_iters=2000)
        if time.monotonic() < hard_deadline - 5:
            a = gradient_polish(a, hard_deadline - 3, max_iters=5000)
    
    # Final validation
    try_update(best_arr)
    
    final = np.asarray(best_arr, dtype=np.float64).ravel()
    final = np.nan_to_num(final, nan=0.0, posinf=0.0, neginf=0.0)
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(96, dtype=np.float64) / 96.0
    else:
        final = final / s
        final = np.clip(final, 0.0, 1000.0)
    
    # Verify
    fin_sc = evaluate_sequence([float(x) for x in final])
    if not math.isfinite(fin_sc):
        final = np.ones(96, dtype=np.float64) / 96.0
        evaluate_sequence([float(x) for x in final])
    
    return [float(x) for x in final]
