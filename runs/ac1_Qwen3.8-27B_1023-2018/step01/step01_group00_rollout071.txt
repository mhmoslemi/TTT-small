def _normalize(a):

def _resize(a, n):
    a = _normalize(a)

        return a

        return np.full(n, float(a[0]) / float(n), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, a.size, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    b = np.interp(x_new, x_old, a)

def _weighted_gradient(a, idx, wts):
    n = int(a.size)
    G = np.zeros(n, dtype=np.float64)
    for k, wt in zip(idx, wts):
        if wt <= 1e-15:
            continue
        kk = int(k)
        lo = kk - (n - 1)
        if lo < 0:
            lo = 0
        hi = kk
        if hi >= n:
            hi = n - 1
        if hi < lo:
            continue
        start = kk - lo
        stop = kk - hi - 1
        if stop < 0:
            seg = a[start:None:-1]
        else:
            seg = a[start:stop:-1]
        if seg.size:
            G[lo:hi + 1] += (2.0 * wt) * seg
    return G


def _soft_score(c, m, tau_frac):
    if c is None or c.size == 0:
        return 0.0
    m = float(m)
    if not math.isfinite(m) or m <= 0.0:
        return 0.0
    tau = max(1e-12, float(tau_frac) * m)
    x = c / tau
    xmax = float(np.max(x))
    if not math.isfinite(xmax):
        return m
    e = np.exp(x - xmax)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return m
    val = m + tau * math.log(s)
    return float(val) if math.isfinite(val) else m

                    best_lp_st = score
                    for theta in (0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.15, 0.3):
                        cand = (1.0 - theta) * a + theta * g
                        cand = _normalize(cand)
                        c_lp = _conv(cand)
                        m_lp = float(np.max(c_lp)) if c_lp.size else math.nan
                        if not math.isfinite(m_lp):
                            continue
                        s_lp = 2.0 * n * m_lp
                        if s_lp < best_lp_st - 1e-12:
                            best_lp_st = s_lp
                            best_lp_cand = cand
                            best_lp_conv = c_lp
                    if best_lp_cand is not None:
                        a = best_lp_cand
                        c = best_lp_conv
                        w = _log_w(a)
                        score = best_lp_st
                        if score < best_score - 1e-15:
                            best_score = score
                            best_a = a.copy()
                            no_improve_best = 0
                        improved = True
                        stuck = 0

            if not improved:
                gamma = max(0.05, gamma * 0.97)
                if stuck % 10 == 0:
                    noise = min(1.5, noise * 1.12)
                    K = int(max(4, min(L, K - 2)))
                    power = min(5.0, power * 1.25)
                if stuck > 50:
                    base = best_a.copy()
                    a = base * np.exp(rng.normal(0.0, 0.22, n))
                    a = _normalize(a)
                    w = _log_w(a)
                    c = _conv(a)
                    m = float(np.max(c)) if c.size else 0.0
                    score = 2.0 * n * m
                    stuck = 0
                    noise = max(0.15, min(1.0, noise * 0.8))
                    K = int(max(6, min(L, int(rng.integers(6, 33)))))
                    power = float(rng.choice([0.5, 1.0, 2.0, 3.0]))
                    gamma = float(rng.uniform(0.2, 0.7))
                elif stuck > 250:
                    break
        else:
            gamma = min(0.9, gamma * 1.03)
            noise = max(0.10, noise * 0.985)
            power = max(0.5, min(5.0, power * 0.99))
            if iter_count % 80 == 0:
                K = int(min(L, max(4, K + 1)))

        if no_improve_best > 650:
            base = best_a.copy()
            a = base * np.exp(rng.normal(0.0, 0.28, n))
            a = _normalize(a)
            w = _log_w(a)
            c = _conv(a)
            m = float(np.max(c)) if c.size else 0.0
            score = 2.0 * n * m
            no_improve_best = 0
            stuck = 0

    return best_a, best_score


def _make_inits(n, parent, rng, limit=30):
    n = int(max(2, n))
    inits = []

    def add(a):
        if len(inits) >= limit:
            return
        try:
            a = _normalize(a)
            if a.size != n:
                a = _resize(a, n)
            if a.size and np.all(np.isfinite(a)) and float(a.sum()) > 0.0:
                inits.append(a)
        except Exception:
            pass

    src = _resize(parent, n)
    add(src)
    add(0.5 * (src + src[::-1]))
    add(1.0 / (src + 1e-6))
    add(np.sqrt(src + 1e-12))
    add(src * src)

    i = np.arange(n, dtype=np.float64)
    for r in (0.15, 0.35, 0.60, 0.85):
        alt = 1.0 + r * np.cos(math.pi * i)
        add(src * alt)

    edge = 1.0 + 0.8 * np.cos(2.0 * math.pi * (i + 0.5) / float(n))
    add(src * edge)

    for _ in range(8):
        z = np.zeros(n, dtype=np.float64)
        for _ in range(3):
            f = int(rng.integers(1, max(2, n // 2)))
            ph = float(rng.uniform(0.0, 2.0 * math.pi))
            amp = float(rng.uniform(0.3, 1.2))
            z += amp * np.cos(2.0 * math.pi * f * i / float(n) + ph)
        z += rng.normal(0.0, 0.3, n)
        add(np.exp(np.clip(z, -3.0, 3.0)))

    for _ in range(6):
        sig = float(rng.uniform(0.15, 1.0))
        add(src * np.exp(rng.normal(0.0, sig, n)))

    for _ in range(4):
        a = np.ones(n, dtype=np.float64)
        for _ in range(3):
            sigma = float(rng.uniform(n * 0.05, n * 0.45))
            center = float(rng.uniform(0.0, n - 1))
            bump = np.exp(-0.5 * ((i - center) / sigma) ** 2)
            a = a + float(rng.uniform(0.2, 1.2)) * bump
        add(a)

    return inits


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    try:
        available = max(0.0, float(budget_s))
    except Exception:
        available = 1300.0

    deadline = t0 + max(0.1, available - 5.0)
    if available > 0.0 and deadline > t0 + available - 0.5:
        deadline = t0 + max(0.0, available - 0.5)

    parent = _parent_array()
    best_a = _normalize(parent)
    best_score = _exact_score(best_a)
    if not math.isfinite(best_score):
        best_a = np.ones(192, dtype=np.float64) / 192.0
        best_score = _exact_score(best_a)
    best_len = int(best_a.size) if best_a.size else 192

    rng = np.random.default_rng(int(seed))
    N = best_len if best_len >= 2 else 192

    inits = _make_inits(N, parent, rng, 26)
    scored = []
    for a in inits:
        sc, _ = _score_unit(a)
        if math.isfinite(sc):
            scored.append((sc, a))
    scored.sort(key=lambda x: x[0])

    for _, a in scored[:12]:
        ex = _exact_score(a)
        if math.isfinite(ex) and ex < best_score:
            best_score = ex
            best_a = np.asarray(a, dtype=np.float64).copy()
            best_len = int(best_a.size)

    start = time.monotonic()
    main_stop = min(
        deadline - 150.0,
        start + max(280.0, 0.72 * max(0.0, deadline - start)),
    )

    if main_stop > start + 25.0 and scored:
        opt_time = max(0.0, main_stop - start - 5.0)
        if opt_time > 15.0:
            ranks = min(6, len(scored))
            if ranks <= 2:
                times = [0.75 * opt_time]
                if ranks == 2:
                    times.append(0.25 * opt_time)
            else:
                fracs = [0.48, 0.20, 0.12, 0.08, 0.06, 0.04]
                times = [fracs[i] * opt_time for i in range(ranks)]
                leftover = opt_time - float(np.sum(times))
                if leftover > 0.05 * opt_time:
                    times[0] += leftover

            for rank in range(ranks):
                if time.monotonic() > main_stop - 6.0:
                    break
                if time.monotonic() < deadline - 50.0 and best_score < 1.495:
                    pass
                sc, a = scored[rank]
                stop = min(main_stop, time.monotonic() + max(15.0, times[rank]))
                if stop <= time.monotonic() + 8.0:
                    continue
                try:
                    local_a, _ = _optimize_local(
                        N, a, int(seed) + 1000 * rank + 17, stop, use_lp=(N <= 512)
                    )
                except Exception:
                    local_a = None
                if local_a is not None:
                    ex = _exact_score(local_a)
                    if math.isfinite(ex) and ex < best_score:
                        best_score = ex
                        best_a = np.asarray(local_a, dtype=np.float64).copy()
                        best_len = int(best_a.size)

    if time.monotonic() < deadline - 130.0:
        base = int(max(best_len, N))
        tried = {best_len, N}
        candidate_ns = []
        for nn in (base // 2, base + 64, base * 2):
            nn = int(nn)
            if 16 <= nn <= 1024 and nn not in tried:
                candidate_ns.append(nn)

        for nn in candidate_ns:
            if time.monotonic() > deadline - 75.0:
                break
            inits2 = _make_inits(nn, parent, rng, 12)
            scored2 = []
            for a in inits2:
                sc, _ = _score_unit(a)
                if math.isfinite(sc):
                    scored2.append((sc, a))
            scored2.sort(key=lambda x: x[0])
            if not scored2:
                continue

            for _, a in scored2[:4]:
                ex = _exact_score(a)
                if math.isfinite(ex) and ex < best_score:
                    best_score = ex
                    best_a = np.asarray(a, dtype=np.float64).copy()
                    best_len = int(best_a.size)

            rem = max(0.0, deadline - time.monotonic() - 40.0)
            if rem < 30.0:
                break

            stop1 = time.monotonic() + min(0.55 * rem, 180.0)
            if stop1 > time.monotonic() + 12.0:
                try:
                    local_a, _ = _optimize_local(
                        nn, scored2[0][1], int(seed) + 50001, stop1, use_lp=(nn <= 512)
                    )
                except Exception:
                    local_a = None
                if local_a is not None:
                    ex = _exact_score(local_a)
                    if math.isfinite(ex) and ex < best_score:
                        best_score = ex
                        best_a = np.asarray(local_a, dtype=np.float64).copy()
                        best_len = int(best_a.size)

            if len(scored2) > 1 and time.monotonic() < deadline - 70.0:
                rem2 = max(0.0, deadline - time.monotonic() - 30.0)
                stop2 = time.monotonic() + min(0.25 * rem2, 80.0)
                if stop2 > time.monotonic() + 10.0:
                    try:
                        local_a, _ = _optimize_local(
                            nn, scored2[1][1], int(seed) + 60002, stop2, use_lp=(nn <= 512)
                        )
                    except Exception:
                        local_a = None
                    if local_a is not None:
                        ex = _exact_score(local_a)
                        if math.isfinite(ex) and ex < best_score:
                            best_score = ex
                            best_a = np.asarray(local_a, dtype=np.float64).copy()
                            best_len = int(best_a.size)

    if time.monotonic() < deadline - 10.0:
        final_n = int(max(2, best_len))
        rem = max(0.0, deadline - time.monotonic() - 4.0)
        final_stop = time.monotonic() + rem
        if final_stop > time.monotonic() + 6.0:
            try:
                local_a, _ = _optimize_local(
                    final_n, best_a, int(seed) + 99999, final_stop, use_lp=(final_n <= 512)
                )
            except Exception:
                local_a = None
            if local_a is not None:
                ex = _exact_score(local_a)
                if math.isfinite(ex) and ex < best_score:
                    best_score = ex
                    best_a = np.asarray(local_a, dtype=np.float64).copy()
                    best_len = int(best_a.size)

    try:
        final_score = _exact_score(best_a)
    except Exception:
        final_score = math.isfinite

    if not math.isfinite(final_score):
        return [float(x) for x in (np.ones(192, dtype=np.float64) / 192.0)]

    out = [float(x) for x in np.asarray(best_a, dtype=np.float64).ravel()]
    if not out:
        out = [1.0]

    cleaned = []
    for v in out:
        if math.isfinite(v):
            v = min(1000.0, max(0.0, v))
            cleaned.append(v)
        else:
            cleaned.append(0.0)

    s = float(sum(cleaned))
    if s <= 0.01:
        return [1.0]

    return [x / s for x in cleaned]
