def _proj_simplex(v):
    v=np.asarray(v,dtype=np.float64).ravel()
    v=np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    n=v.size
    if n==0: return np.array([1.0])
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    rho=np.nonzero(u*(np.arange(1,n+1,dtype=np.float64)) > (cssv-1.0))[0]
    if rho.size==0:
        return np.full(n, 1.0/n)
    k=rho[-1]
    theta=(cssv[k]-1.0)/(k+1.0)
    out=np.maximum(v-theta,0.0)

            add(base * np.exp(amp * rng.standard_normal(n)))
        for p in (0.5, 2.0):
            add(np.power(np.clip(base, 1e-12, None), p))
        for k in (3, 5, 7):
            if k < n:
                ker = np.ones(k, dtype=np.float64) / float(k)
                add(np.convolve(base, ker, "same"))

    if not starts:
        add(np.ones(n, dtype=np.float64))

    scored = []
    for a in starts:
        sc = _score_fast(a)
        if not math.isfinite(sc):
            continue
        dup = False
        for _, other in scored:
            if np.linalg.norm(a - other) < 1e-4:
                dup = True
                break
        if not dup:
            scored.append((sc, a))
    scored.sort(key=lambda z: z[0])
    return [a for _, a in scored[:max(1, int(limit))]]


def _get_lp_indices(n):
    n = int(n)
    if n <= 0:
        return None
    if n in _LP_CACHE:
        return _LP_CACHE[n]
    L = 2 * n - 1
    r = np.arange(L, dtype=np.int64)[:, None]
    j = np.arange(n, dtype=np.int64)[None, :]
    kk = r - j
    mask = (kk >= 0) & (kk < n)
    rows, cols = np.where(mask)
    rr = rows.astype(np.int32, copy=False)
    cc = cols.astype(np.int32, copy=False)
    kk = (rr.astype(np.int64) - cc.astype(np.int64)).astype(np.int32, copy=False)
    idx = (rr, cc, kk)
    _LP_CACHE[n] = idx
    return idx


def _lp_direction(n, a, idx, M, max_solve_time=4.0):
    if linprog is None or sps is None or idx is None:
        return None
    if not math.isfinite(M) or M <= 0.0:
        return None
    rr, cc, kk = idx
    data = _clean(a)[kk]
    if float(np.max(data)) <= 0.0:
        return None
    try:
        A = sps.csr_matrix((data, (rr, cc)), shape=(2 * n - 1, n), dtype=np.float64)
    except Exception:
        return None

    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M * (1.0 + 1e-9) + 1e-15, dtype=np.float64)
    bounds = [(0.0, None)] * n
    tl = float(max(0.1, min(6.0, max_solve_time)))

    try:
        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": tl},
            )
        except Exception:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
    except Exception:
        return None

    x = getattr(res, "x", None)
    if x is None:
        return None
    x = np.asarray(x, dtype=np.float64).ravel()
    x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
    x = np.clip(x, 0.0, None)
    s = float(x.sum())
    if not math.isfinite(s) or s <= 1e-12:
        return None
    return x / s


def _lp_fw_loop(n, a, time_limit, global_stop, rng, best, idx):
    now = time.monotonic()
    stop = min(now + float(max(0.0, time_limit)), max(now, global_stop - 2.0))
    a = _normalize(a)
    local = _score_fast(a)
    local_best = local if math.isfinite(local) else float("inf")
    local_best_a = a.copy()
    _update_best(a, best)
    improved_any = False
    stale = 0

    while time.monotonic() < stop - 0.5:
        remaining = stop - time.monotonic()
        if remaining < 1.0:
            break
        b = np.convolve(a, a)
        if b.size == 0:
            break
        M = float(np.max(b))
        if not math.isfinite(M) or M <= 0.0:
            break
        max_solve = max(0.2, min(4.0, remaining - 1.0))
        g = _lp_direction(n, a, idx, M, max_solve)
        if g is None:
            stale += 1
            if stale > 6:
                break
            continue

        best_sc = local if math.isfinite(local) else float("inf")
        best_cand = None
        for t in (0.2, 0.5, 1.0, 0.3, 0.8, 0.4, 0.1, 1.2, 0.05, 0.7):
            cand = (1.0 - t) * a + t * g
            cand = _clean(cand)
            s = float(cand.sum())
            if not math.isfinite(s) or s <= 1e-15:
                continue
            cand = cand / s
            sc = _score_fast(cand)
            if math.isfinite(sc) and sc < best_sc - 1e-14:
                best_sc = sc
                best_cand = cand

        if best_cand is not None:
            a = best_cand
            local = best_sc
            improved_any = True
            stale = 0
            if local < local_best:
                local_best = local
                local_best_a = a.copy()
            _update_best(a, best)
        else:
            stale += 1
            if stale % 4 == 0:
                pert = _clean(a * np.exp(0.08 * rng.standard_normal(n)))
                pert = _normalize(pert)
                ps = _score_fast(pert)
                if math.isfinite(ps) and ps <= local_best * 1.02 + 1e-12:
                    a = pert
                    local = ps
                    if ps < local_best:
                        local_best = ps
                        local_best_a = a.copy()
                        improved_any = True
                        _update_best(a, best)
            if stale > 20:
                break

    if not math.isfinite(local) or local > local_best:
        a = local_best_a
        local = local_best
    return a, local, improved_any


def _grad_H_from_reverse(ar, w, n):
    c = np.convolve(ar, w, mode="full")
    return c[n - 1:2 * n - 1].copy()


def _make_weights(b, n, it):
    L = int(b.size)
    if L <= 0:
        return [np.ones(1, dtype=np.float64)]
    b = np.asarray(b, dtype=np.float64).ravel()
    bmax = float(np.max(b))
    weights = []

    betas = (10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 100000.0)
    beta = betas[it % len(betas)]
    w = np.exp(beta * (b - bmax))
    s = float(w.sum())
    if math.isfinite(s) and s > 0.0:
        w = w / s
        weights.append(w)

    Kpat = (max(2, L // 8), max(2, L // 16), max(2, L // 32), 4, 2, 1)
    K = int(Kpat[(it // len(betas)) % len(Kpat)])
    K = max(1, min(L, K))
    if K >= L:
        sb = float(b.sum())
        w = b / sb if sb > 1e-15 else np.full(L, 1.0 / L, dtype=np.float64)
    else:
        if K == L:
            idx = np.arange(L)
        else:
            idx = np.argpartition(b, -K)[-K:]
        w = np.zeros(L, dtype=np.float64)
        if idx.size > 0:
            w[idx] = 1.0 / float(idx.size)
    s = float(w.sum())
    if math.isfinite(s) and s > 0.0:
        w = w / s
        weights.append(w)

    if it % 3 == 0 and bmax > 0.0:
        tol = 1e-12 + 1e-7 * bmax
        ties = np.where(b >= bmax - tol)[0]
        if 0 < ties.size < L:
            w = np.zeros(L, dtype=np.float64)
            w[ties] = 1.0 / float(ties.size)
            weights.append(w)

    if len(weights) > 3:
        weights = weights[:3]
    if not weights:
        weights = [np.full(L, 1.0 / L, dtype=np.float64)]
    return weights


def _pg_loop(n, a, time_limit, global_stop, rng, best):
    now = time.monotonic()
    stop = min(now + float(max(0.0, time_limit)), max(now, global_stop - 1.0))
    a = _normalize(a)
    local = _score_fast(a)
    local_best = local if math.isfinite(local) else float("inf")
    local_best_a = a.copy()
    _update_best(a, best)
    improved_any = False
    stale = 0
    it = 0
    muls = (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0)

    while time.monotonic() < stop - 0.1:
        b = np.convolve(a, a)
        if b.size == 0:
            break
        ws = _make_weights(b, n, it)
        step_improved = False
        ar = a[::-1].copy()

        for w in ws:
            if time.monotonic() >= stop - 0.05:
                break
            H = _grad_H_from_reverse(ar, w, n)
            if H.size != n:
                continue
            G = 2.0 * H
            G = G - float(np.mean(G))
            gmax = float(np.max(np.abs(G))) if G.size else 0.0
            if not math.isfinite(gmax) or gmax <= 1e-14:
                continue
            base = (1.0 / float(n)) / gmax
            for mu in muls:
                if time.monotonic() >= stop - 0.02:
                    break
                cand = _proj_simplex(a - mu * base * G)
                sc = _score_fast(cand)
                if math.isfinite(sc) and sc < local - 1e-14:
                    a = cand
                    local = sc
                    step_improved = True
                    break

            if not step_improved:
                i = int(np.argmax(G))
                j = int(np.argmin(G))
                if i != j and a[i] > 1e-15:
                    for frac in (0.02, 0.05, 0.1, 0.2, 0.35, 0.6, 0.9):
                        if time.monotonic() >= stop - 0.02:
                            break
                        cand = a.copy()
                        delta = float(frac) * float(a[i])
                        cand[i] -= delta
                        cand[j] += delta
                        cand = _clean(cand)
                        s = float(cand.sum())
                        if s <= 1e-15:
                            continue
                        cand = cand / s
                        sc = _score_fast(cand)
                        if math.isfinite(sc) and sc < local - 1e-14:
                            a = cand
                            local = sc
                            step_improved = True
                            break
            if step_improved:
                break

        if step_improved:
            stale = 0
            improved_any = True
            if local < local_best:
                local_best = local
                local_best_a = a.copy()
            _update_best(a, best)
        else:
            stale += 1
            if stale % 10 == 0:
                pert = _clean(a * np.exp(0.06 * rng.standard_normal(n)))
                pert = _normalize(pert)
                ps = _score_fast(pert)
                if math.isfinite(ps) and ps <= local_best * 1.001 + 1e-12:
                    a = pert
                    local = ps
                    if ps < local_best:
                        local_best = ps
                        local_best_a = a.copy()
                        improved_any = True
                        _update_best(a, best)
            if stale > 50:
                break
            if stale > 25:
                a = local_best_a.copy()
                local = local_best
                stale = 15
        it += 1

    if not math.isfinite(local) or local > local_best:
        a = local_best_a
        local = local_best
    return a, local, improved_any


def _lbfgs_polish(n, a, time_limit, global_stop, rng, best):
    if minimize is None or n < 2:
        return _normalize(a)
    now = time.monotonic()
    stop = min(now + float(max(0.0, time_limit)), max(now, global_stop - 2.0))
    a = _normalize(a)
    local_best = _score_fast(a)
    best_a = a.copy()
    z = np.log(a + 1e-30)
    z = z - float(np.mean(z))

    betas = (100.0, 1000.0, 10000.0, 50000.0)
    for beta in betas:
        if time.monotonic() >= stop - 2.0:
            break
        aborted = [False]

        def fun_g(x, beta=beta):
            if aborted[0]:
                return float("inf"), np.zeros(n, dtype=np.float64)
            zloc = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
            zloc = np.clip(zloc, -50.0, 50.0)
            zc = zloc - float(np.max(zloc))
            ez = np.exp(zc)
            s = float(ez.sum())
            if not math.isfinite(s) or s <= 1e-300:
                return float("inf"), np.zeros(n, dtype=np.float64)
            aval = ez / s
            b = np.convolve(aval, aval)
            if b.size == 0:
                return float("inf"), np.zeros(n, dtype=np.float64)
            bmax = float(np.max(b))
            eb = np.exp(beta * (b - bmax))
            se = float(eb.sum())
            if not math.isfinite(se) or se <= 0.0:
                return float("inf"), np.zeros(n, dtype=np.float64)
            score = 2.0 * n * (bmax + math.log(se) / beta)
            w = eb / se
            ar = aval[::-1]
            c = np.convolve(ar, w, mode="full")
            H = c[n - 1:2 * n - 1]
            grad_a = 4.0 * n * H
            avg = float(np.dot(aval, grad_a))
            grad_z = aval * (grad_a - avg)
            return float(score), grad_z

        def cb(z):
            if time.monotonic() >= stop - 2.0:
                aborted[0] = True
                return True
            return False

        try:
            res = minimize(
                fun_g,
                z,
                jac=True,
                method="L-BFGS-B",
                options={"maxiter": 40, "ftol": 1e-15, "gtol": 1e-10, "maxls": 10},
                callback=cb,
            )
        except Exception:
            continue

        if hasattr(res, "x") and res.x is not None:
            z2 = np.nan_to_num(res.x, nan=0.0, posinf=0.0, neginf=0.0)
            z2 = np.clip(z2, -50.0, 50.0)
            z2 = z2 - float(np.mean(z2))
            zc = z2 - float(np.max(z2))
            ez = np.exp(zc)
            s = float(ez.sum())
            if math.isfinite(s) and s > 1e-300:
                aval = ez / s
                sc = _score_fast(aval)
                if math.isfinite(sc):
                    if sc < local_best:
                        local_best = sc
                        best_a = aval.copy()
                    z = z2
        if time.monotonic() >= stop - 4.0:
            break

    _update_best(best_a, best)
    return best_a


def _refine_sequence(n, a, budget, global_stop, rng, best, idx):
    now = time.monotonic()
    stop = min(now + float(max(0.0, budget)), max(now, global_stop - 3.0))
    a = _normalize(a)
    local = _score_fast(a)
    local_best = local if math.isfinite(local) else float("inf")
    local_best_a = a.copy()
    _update_best(a, best)

    it = 0
    while time.monotonic() < stop - 1.0:
        remaining = stop - time.monotonic()
        if remaining < 2.0:
            break
        if it % 3 == 0:
            slice_t = min(15.0, max(2.0, remaining * 0.45))
            a, local, imp = _lp_fw_loop(n, a, slice_t, global_stop, rng, best, idx)
        else:
            slice_t = min(18.0, max(2.0, remaining * 0.70))
            a, local, imp = _pg_loop(n, a, slice_t, global_stop, rng, best)

        if math.isfinite(local) and local < local_best:
            local_best = local
            local_best_a = a.copy()
            _update_best(a, best)

        if not imp:
            if time.monotonic() < stop - 3.0:
                pert = _clean(a * np.exp(0.10 * rng.standard_normal(n)))
                pert = _normalize(pert)
                ps = _score_fast(pert)
                if math.isfinite(ps) and ps <= local_best * 1.02 + 1e-12:
                    a = pert
                    local = ps
                    if ps < local_best:
                        local_best = ps
                        local_best_a = a.copy()
                        _update_best(a, best)
            else:
                break
        it += 1

    if time.monotonic() < stop:
        a, local, _ = _pg_loop(n, a, max(1.0, stop - time.monotonic()), global_stop, rng, best)
        if math.isfinite(local) and local < local_best:
            local_best = local
            local_best_a = a.copy()
            _update_best(a, best)

    if not math.isfinite(local_best):
        local_best = _score_fast(local_best_a)
    return local_best_a, local_best


def _search_length(n, budget, global_stop, rng, best, parent):
    n = int(max(2, n))
    if n > 512:
        n = 512
    budget = float(max(0.0, budget))
    if budget < 5.0:
        return

    t0 = time.monotonic()
    idx = _get_lp_indices(n) if (linprog is not None and sps is not None) else None
    starts = _make_starts(n, rng, parent, limit=30)
    if not starts:
        starts = [np.ones(n, dtype=np.float64) / float(n)]

    max_chosen = 6 if n <= 128 else (5 if n <= 192 else 4)
    chosen = starts[:max_chosen]

    if len(starts) > max_chosen:
        add_count = 0
        while add_count < 2 and len(chosen) < max_chosen:
            j = int(rng.integers(max_chosen, len(starts)))
            cand = starts[j]
            if not any(np.linalg.norm(cand - c) < 1e-4 for c in chosen):
                chosen.append(cand)
                add_count += 1

    len_best = (float("inf"), _normalize(starts[0]))
    for s in chosen:
        sc = _score_fast(s)
        if math.isfinite(sc) and sc < len_best[0]:
            len_best = (sc, _normalize(s))

    for i, s in enumerate(chosen):
        now = time.monotonic()
        used = now - t0
        rem = budget - used
        if rem < 5.0:
            break
        if now >= global_stop - 25.0:
            break
        per = rem / float(max(1, len(chosen) - i))
        multi = 1.2 if i == 0 else 1.0
        alloc = min(95.0, max(4.0, per * multi))
        alloc = min(alloc, max(0.0, global_stop - now - 25.0))
        if alloc < 3.0:
            break
        a, local = _refine_sequence(n, s, alloc, global_stop, rng, best, idx)
        if math.isfinite(local) and local < len_best[0]:
            len_best = (local, _normalize(a))

    now = time.monotonic()
    if n <= 128 and now < global_stop - 60.0:
        alloc = min(35.0, max(0.0, global_stop - now - 25.0))
        if alloc > 8.0:
            a = _lbfgs_polish(n, len_best[1], alloc, global_stop, rng, best)
            sc = _score_fast(a)
            if math.isfinite(sc) and sc < len_best[0]:
                len_best = (sc, a)
            _update_best(a, best)


def _final_candidate(best):
    arr = _clean(best[1])
    if arr.size == 0:
        arr = np.array([1.0], dtype=np.float64)
    s = float(arr.sum())
    if not math.isfinite(s) or s <= 1e-9:
        arr = np.ones(arr.size if arr.size > 0 else 1, dtype=np.float64)
        s = float(arr.size)
    arr = arr / s
    arr = np.clip(arr, 0.0, 1000.0)
    try:
        sc = evaluate_sequence([float(x) for x in arr])
    except Exception:
        sc = _score_fast(arr)
    if not math.isfinite(sc):
        arr = np.array([1.0], dtype=np.float64)
    out = [float(x) for x in arr]
    return out


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    try:
        total = float(budget_s)
    except Exception:
        total = 1300.0
    if not math.isfinite(total) or total <= 0.0:
        total = 1.0
    hard_stop = start + max(1.0, total - 30.0)

    best = [2.0, np.array([1.0], dtype=np.float64)]

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            parent = _clean(h)
            if parent.size > 0:
                _update_best(parent, best)
    except Exception:
        parent = None

    if time.monotonic() >= hard_stop - 30.0:
        return _final_candidate(best)

    plans = (
        (96, 0.450),
        (128, 0.220),
        (80, 0.100),
        (160, 0.080),
        (64, 0.070),
        (192, 0.080),
    )
    total_frac = sum(f for _, f in plans)
    base = max(0.0, total - 45.0)

    for n, frac in plans:
        alloc = max(0.0, base * (float(frac) / total_frac))
        alloc = min(alloc, max(0.0, hard_stop - time.monotonic() - 30.0))
        if alloc <= 8.0:
            break
        try:
            _search_length(n, alloc, hard_stop, rng, best, parent)
        except Exception:
            pass

    now = time.monotonic()
    if now < hard_stop - 60.0:
        try:
            final_n = int(np.asarray(best[1], dtype=np.float64).size)
            if 2 <= final_n <= 512:
                idx = _get_lp_indices(final_n) if (linprog is not None and sps is not None) else None
                alloc = min(120.0, max(0.0, hard_stop - now - 25.0))
                if alloc > 15.0:
                    a, _ = _refine_sequence(final_n, best[1], alloc, hard_stop, rng, best, idx)
                    _update_best(a, best)
        except Exception:
            pass

    return _final_candidate(best)
