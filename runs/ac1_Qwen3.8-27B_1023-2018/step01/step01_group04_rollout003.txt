def _z_from_f(f, n):
    f = _clean_array(f)
    n = int(n)
    if n < 2 or n % 2:
        n = max(2, n - 1)
    m = n // 2
    p = 2 * m

    if f.size == 0:
        return np.zeros(p, dtype=np.float64)

    if f.size != n:
        old = np.linspace(0.0, 1.0, f.size)
        new = np.linspace(0.0, 1.0, n)
        f = np.interp(new, old, f, left=f[0], right=f[-1])

    left = f[:m]
    right = f[n - 1:n - m - 1:-1]

    s = left + right
    sv = np.where(s > 1e-300, s, 0.0)
    ss = float(np.sum(sv))

    v = np.zeros(m, dtype=np.float64)
    if ss <= 1e-300:
        st = np.ones(m, dtype=np.float64)
    else:
        st = float(m) * sv / ss
        ok = sv > 1e-15
        v[ok] = (left[ok] - right[ok]) / sv[ok]
        v = np.clip(v, -1.0, 1.0)
        st = np.clip(st, 1e-300, None)

def _basis_vec(m, r):
    if m <= 0:
        return np.array([], dtype=np.float64)
    t = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
    if r % 2 == 0:
        return np.cos((r + 1) * math.pi * t)
    return np.sin((r + 1) * math.pi * t)


def _random_v(m, rng, kmax=None, scale=1.0):
    m = int(m)
    if m <= 0:
        return np.array([], dtype=np.float64)
    K = int(kmax) if kmax is not None else min(8, m)
    K = max(0, min(K, m))
    v = np.zeros(m, dtype=np.float64)
    sd = max(1e-12, scale / math.sqrt(max(1, K)))
    base = []
    for r in range(K):
        base.append(_basis_vec(m, r) * rng.normal(0.0, sd))
    if base:
        v = np.sum(base, axis=0)
    return np.tanh(v)


def _perturb_z(z, rng, uscale=0.25, vscale=None):
    z = np.array(z, dtype=np.float64, copy=True).ravel()
    m = z.size // 2
    if m <= 0:
        return z
    if vscale is None:
        vscale = 0.5 * uscale
    if z.size < 2 * m:
        zz = np.zeros(2 * m, dtype=np.float64)
        k = min(z.size, 2 * m)
        zz[:k] = z[:k]
        z = zz
    z[:m] = z[:m] + rng.normal(0.0, uscale, m)
    z[m:] = z[m:] + rng.normal(0.0, vscale, m)
    z = np.nan_to_num(z, nan=0.0, posinf=0.0, neginf=0.0)
    z[:m] = np.clip(z[:m], -60.0, 60.0)
    z[m:] = np.clip(z[m:], -1.0, 1.0)
    return z


def _add_pool(pool, n, z, cap=80):
    if pool is None:
        return
    try:
        n = int(n)
        if n < 2 or n % 2:
            n = max(2, n - 1)
        f, _, _ = _uv_components(z, n)
        sc = _fast_score(f)
        if not math.isfinite(sc):
            return
        pool.append((float(sc), int(n), np.array(z, dtype=np.float64, copy=True)))
        pool.sort(key=lambda x: x[0])
        if len(pool) > cap:
            del pool[cap:]
    except Exception:
        pass


def _make_inits(n, parent, rng):
    n = int(n)
    if n < 2 or n % 2:
        n = max(2, n - 1)
    m = n // 2
    p = 2 * m
    out = []

    def add(z):
        try:
            z = np.asarray(z, dtype=np.float64).ravel()
            if z.size < p:
                zz = np.zeros(p, dtype=np.float64)
                k = min(z.size, p)
                if k:
                    zz[:k] = z[:k]
                z = zz
            elif z.size > p:
                z = z[:p]
            z = np.nan_to_num(z, nan=0.0, posinf=0.0, neginf=0.0)
            z[:m] = np.clip(z[:m], -60.0, 60.0)
            z[m:] = np.clip(z[m:], -1.0, 1.0)
            out.append(z)
        except Exception:
            pass

    if parent is not None:
        add(_z_from_f(parent, n))
    add(np.zeros(p, dtype=np.float64))

    for amp in (0.35, 0.60, 0.85):
        for r in range(min(5, m)):
            v = np.clip(amp * _basis_vec(m, r), -1.0, 1.0)
            add(np.concatenate((np.zeros(m, dtype=np.float64), v)))

    for _ in range(5):
        v = _random_v(m, rng, kmax=min(8, m), scale=1.2)
        add(np.concatenate((np.zeros(m, dtype=np.float64), v)))

    for _ in range(3):
        f = np.exp(rng.normal(0.0, 0.6, n))
        add(_z_from_f(f, n))

    for _ in range(3):
        u = np.clip(rng.normal(0.0, 0.35, m), -5.0, 5.0)
        v = _random_v(m, rng, kmax=4, scale=0.7)
        add(np.concatenate((u, v)))

    return out[:16]


def _lowdim_basis(n, deadline, best, pool, rng, max_runs=18):
    if minimize is None:
        return
    n = int(n)
    if n < 8 or n % 2:
        n = max(8, n - 1)
    m = n // 2
    K = min(8, m)
    if K <= 0:
        return

    B = np.vstack([_basis_vec(m, r) for r in range(K)])
    u0 = np.zeros(m, dtype=np.float64)
    bounds = [(-6.0, 6.0)] * K

    starts = [np.zeros(K, dtype=np.float64)]
    for r in range(min(4, K)):
        for amp in (0.5, 1.0, 1.7):
            for sign in (1.0, -1.0):
                c = np.zeros(K, dtype=np.float64)
                c[r] = sign * amp
                starts.append(c)
    for _ in range(8):
        starts.append(rng.normal(0.0, 0.8, K))

    betas = [4.0, 30.0, 200.0, 1000.0]

    for c0 in starts[:max_runs]:
        if time.monotonic() >= deadline - 15.0:
            break
        c = np.array(c0, dtype=np.float64, copy=True)
        for beta in betas:
            if time.monotonic() >= deadline - 12.0:
                break

            def fun(c, beta=beta):
                try:
                    V = B.dot(c)
                    v = np.tanh(V)
                    z = np.concatenate((u0, v))
                    L, g = _uv_objgrad(z, n, beta)
                    if not math.isfinite(L):
                        return 1e100, np.zeros(K, dtype=np.float64)
                    gv = g[m:m + m]
                    gc = B.T.dot((1.0 - v * v) * gv)
                    if not np.all(np.isfinite(gc)):
                        return 1e100, np.zeros(K, dtype=np.float64)
                    return float(L), gc
                except Exception:
                    return 1e100, np.zeros(K, dtype=np.float64)

            try:
                res = minimize(
                    fun, c,
                    jac=True,
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={
                        "maxiter": 180,
                        "maxfun": 3600,
                        "ftol": 1e-12,
                        "gtol": 1e-7,
                        "maxls": 50,
                    },
                )
            except Exception:
                res = None

            if res is not None and getattr(res, "x", None) is not None:
                cc = np.asarray(res.x, dtype=np.float64).ravel()
                cc = np.nan_to_num(cc, nan=0.0, posinf=0.0, neginf=0.0)
                if cc.size == K:
                    c = np.clip(cc, -6.0, 6.0)

            V = B.dot(c)
            v = np.tanh(V)
            z = np.concatenate((u0, v))
            f, _, _ = _uv_components(z, n)
            _try_update(f, best)
            _add_pool(pool, n, z)


def _run_uv(n, z, beta_list, maxit, deadline, best, pool=None):
    if minimize is None:
        return None
    n = int(n)
    if n < 2 or n % 2:
        n = max(2, n - 1)
    m = n // 2
    p = 2 * m

    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size < p:
        zz = np.zeros(p, dtype=np.float64)
        k = min(z.size, p)
        if k:
            zz[:k] = z[:k]
        z = zz
    elif z.size > p:
        z = z[:p]
    z = np.nan_to_num(z, nan=0.0, posinf=0.0, neginf=0.0)
    z[:m] = np.clip(z[:m], -60.0, 60.0)
    z[m:] = np.clip(z[m:], -1.0, 1.0)

    f, _, _ = _uv_components(z, n)
    _try_update(f, best)
    _add_pool(pool, n, z)

    bounds = [(-60.0, 60.0)] * m + [(-1.0, 1.0)] * m
    local_best = _fast_score(f)

    if p <= 128:
        miter = max(30, int(maxit))
        maxfun = max(600, int(miter * 14))
    elif p <= 256:
        miter = max(20, int(0.75 * maxit))
        maxfun = max(500, int(miter * 12))
    elif p <= 512:
        miter = max(15, int(0.5 * maxit))
        maxfun = max(400, int(miter * 10))
    else:
        miter = max(10, int(0.35 * maxit))
        maxfun = max(300, int(miter * 8))

    for beta in beta_list:
        if time.monotonic() >= deadline - 8.0:
            break
        beta = float(beta)
        if not math.isfinite(beta) or beta <= 0.0:
            beta = 1.0

        def fun(x, beta=beta):
            try:
                L, g = _uv_objgrad(x, n, beta)
            except Exception:
                return 1e100, np.zeros(x.size, dtype=np.float64)
            if not math.isfinite(L):
                return 1e100, np.zeros(x.size, dtype=np.float64)
            if not np.all(np.isfinite(g)):
                g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            return float(L), g

        try:
            res = minimize(
                fun, z,
                jac=True,
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": miter,
                    "maxfun": maxfun,
                    "ftol": 1e-12,
                    "gtol": 1e-7,
                    "maxls": 50,
                },
            )
        except Exception:
            res = None

        if res is not None and getattr(res, "x", None) is not None:
            zc = np.asarray(res.x, dtype=np.float64).ravel()
            zc = np.nan_to_num(zc, nan=0.0, posinf=0.0, neginf=0.0)
            if zc.size == p:
                zc[:m] = np.clip(zc[:m], -60.0, 60.0)
                zc[m:] = np.clip(zc[m:], -1.0, 1.0)
                fc, _, _ = _uv_components(zc, n)
                sc = _fast_score(fc)
                if math.isfinite(sc):
                    tol = max(0.02, 0.02 * max(1.0, float(local_best)))
                    if sc <= local_best + tol or sc < float(best[0]) + 1e-9:
                        z = zc
                        local_best = min(local_best, sc)

        f, _, _ = _uv_components(z, n)
        sc = _fast_score(f)
        if math.isfinite(sc):
            local_best = min(local_best, sc)
        _try_update(f, best)
        _add_pool(pool, n, z)

        if float(best[0]) < TARGET * 0.999 and beta >= 1000.0 and local_best < TARGET * 0.999:
            break

    return z


def _lp_polish(n, f, deadline, best):
    if linprog is None or sps is None:
        return
    f = _clean_array(f)
    n = int(f.size)
    if n < 2 or n > 256:
        return
    S = float(f.sum())
    if not math.isfinite(S) or S <= 0.0:
        return

    rows = np.repeat(np.arange(n, dtype=np.int64), n) + np.tile(np.arange(n, dtype=np.int64), n)
    cols = np.repeat(np.arange(n, dtype=np.int64), n)
    cvec = -np.ones(n, dtype=np.float64)
    ybound = max(1000.0 * S, 1000.0, 10.0 + 10.0 * float(np.max(f)))

    stagn = 0
    for _ in range(120):
        if time.monotonic() >= deadline - 15.0:
            break
        C = np.convolve(f, f)
        M = float(np.max(C))
        if not math.isfinite(M) or M <= 0.0:
            break
        data = np.tile(f, n)
        try:
            A = sps.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
            res = linprog(
                cvec,
                A_ub=A,
                b_ub=np.full(2 * n - 1, M * (1.0 + 1e-12) + 1e-14),
                bounds=[(0.0, ybound)] * n,
                method="highs",
                options={"presolve": True},
            )
        except Exception:
            break

        if res is None or getattr(res, "x", None) is None:
            stagn += 1
            if stagn > 8:
                break
            continue

        y = np.asarray(res.x, dtype=np.float64).ravel()
        y = np.nan_to_num(y, nan=0.0, posinf=0.0, neginf=0.0)
        y = np.clip(y, 0.0, None)
        Sy = float(np.sum(y))
        if not math.isfinite(Sy) or Sy <= 0.0:
            stagn += 1
            if stagn > 8:
                break
            continue
        y = y * (S / Sy)

        cur = f
        cur_sc = _fast_score(f)
        best_cand = cur
        for t in (0.1, 0.2, 0.35, 0.5, 0.7, 1.0):
            cand = (1.0 - t) * f + t * y
            sc = _fast_score(cand)
            if math.isfinite(sc) and sc < _fast_score(best_cand):
                best_cand = cand
                cur_sc = sc

        if math.isfinite(_fast_score(best_cand)) and _fast_score(best_cand) < _fast_score(f) * (1.0 - 1e-9):
            f = best_cand
            _try_update(f, best)
            stagn = 0
        else:
            stagn += 1
            if stagn > 8:
                break


def _finalize(best):
    try:
        final = _clean_array(best[1])
        if final.size == 0:
            final = np.array([1.0, 1.0], dtype=np.float64)
        final = np.clip(final, 0.0, 1000.0)
        if math.isfinite(_fast_score(final)):
            _try_update(final, best)
    except Exception:
        final = np.array([1.0, 1.0], dtype=np.float64)

    final = _clean_array(best[1]) if best[1] is not None else np.array([1.0, 1.0], dtype=np.float64)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0:
        final = np.array([1.0, 1.0], dtype=np.float64)

    sc = _exact_score_seq(final)
    if not math.isfinite(sc) or sc >= float(best[0]) - 1e-12:
        try:
            final = _clean_array(best[1])
            final = np.clip(final, 0.0, 1000.0)
            if final.size == 0:
                final = np.array([1.0, 1.0], dtype=np.float64)
        except Exception:
            final = np.array([1.0, 1.0], dtype=np.float64)

    if final.size == 0:
        final = np.array([1.0, 1.0], dtype=np.float64)
    return [float(x) for x in final]


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    start = time.monotonic()
    try:
        rem = float(budget_s)
    except Exception:
        rem = 1300.0
    if not math.isfinite(rem) or rem <= 0.0:
        rem = 1300.0

    head = min(25.0, max(1.0, rem * 0.1))
    deadline = start + max(1.0, rem - head)
    budget = max(1.0, deadline - start)

    rng = np.random.default_rng(seed)
    best = [float("inf"), np.array([1.0, 1.0], dtype=np.float64)]

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            parent = _clean_array(h)
            if parent.size:
                _try_update(parent, best)
    except Exception:
        parent = None

    if (not math.isfinite(best[0])) or best[1] is None or best[1].size == 0:
        _try_update(np.ones(96, dtype=np.float64), best)

    if time.monotonic() >= deadline - 30.0:
        return _finalize(best)

    nb = len(best[1]) if best[1] is not None else 96
    if nb % 2:
        nb -= 1
    if nb < 8 or nb > 256:
        base_n = 96
    else:
        base_n = max(8, nb)

    pool = []
    _add_pool(pool, base_n, _z_from_f(best[1], base_n))

    rem_budget = max(1.0, deadline - start)

    lowdim_deadline = min(deadline, start + max(25.0, rem_budget * 0.12))
    if time.monotonic() < lowdim_deadline:
        _lowdim_basis(base_n, lowdim_deadline, best, pool, rng, max_runs=18)

    deep_end = min(deadline, start + max(60.0, rem_budget * 0.62))
    other_end = min(deadline, start + max(120.0, rem_budget * 0.88))

    if deque is not None:
        queue = deque()
    else:
        queue = []

    def enqueue(n, z):
        try:
            if len(queue) < 80:
                if deque is not None:
                    queue.append((int(n), np.array(z, dtype=np.float64, copy=True)))
                else:
                    queue.append((int(n), np.array(z, dtype=np.float64, copy=True)))
        except Exception:
            pass

    for sc, nn, zz in pool[:8]:
        enqueue(nn, zz)
    for z in _make_inits(base_n, parent, rng):
        enqueue(base_n, z)

    deep_betas = [4.0, 20.0, 100.0, 500.0, 2000.0]

    def pop_queue():
        if not queue:
            return None
        if deque is not None:
            return queue.popleft()
        return queue.pop(0)

    runs = 0
    while True:
        item = pop_queue()
        if item is None:
            break
        if time.monotonic() >= deep_end:
            break
        n, z = item
        n = int(n)
        if n < 8 or n > 512:
            continue
        maxit = 220 if n <= 128 else (120 if n <= 256 else 70)
        _run_uv(n, z, deep_betas, maxit, deadline, best, pool)
        runs += 1

        zbest_n = _z_from_f(best[1], n) if len(best[1]) else _z_from_f(parent if parent is not None else np.ones(n), n)
        enqueue(n, _perturb_z(zbest_n, rng, 0.25, 0.15))
        if n == base_n:
            enqueue(n, _perturb_z(z, rng, 0.35, 0.25))

        if runs >= 28:
            break

    for n_try in (64, 96, 128, 192, 256):
        if time.monotonic() >= other_end:
            break
        if n_try == len(best[1]) and math.isfinite(best[0]) and best[0] < TARGET:
            continue
        inits = []
        if parent is not None:
            inits.append(_z_from_f(parent, n_try))
        inits.append(_z_from_f(best[1], n_try))
        extra = _make_inits(n_try, None, rng)
        inits.extend(extra[:3])

        for z in inits[:3]:
            if time.monotonic() >= other_end:
                break
            maxit = 160 if n_try <= 128 else (90 if n_try <= 256 else 60)
            _run_uv(n_try, z, [4.0, 30.0, 200.0, 800.0], maxit, deadline, best, pool)

    if time.monotonic() < deadline - 70.0:
        n_pol = len(best[1]) if best[1] is not None else base_n
        if n_pol % 2:
            n_pol -= 1
        if n_pol < 8:
            n_pol = base_n
        z_pol = _z_from_f(best[1], n_pol)
        maxit = 260 if n_pol <= 128 else (140 if n_pol <= 256 else 80)
        _run_uv(n_pol, z_pol, [50.0, 300.0, 1500.0, 5000.0], maxit, deadline, best, pool)

    if time.monotonic() < deadline - 50.0 and best[1] is not None:
        _lp_polish(len(best[1]), best[1], deadline, best)

    if time.monotonic() < deadline - 35.0 and best[1] is not None:
        n_pol = len(best[1])
        if n_pol % 2:
            n_pol -= 1
        if n_pol >= 8:
            z_pol = _z_from_f(best[1], n_pol)
            maxit = 180 if n_pol <= 128 else (100 if n_pol <= 256 else 60)
            _run_uv(n_pol, z_pol, [300.0, 2000.0, 8000.0], maxit, deadline, best, pool)

    return _finalize(best)
