    def lp_phase(a0, seconds, K=100, blend_t=0.03):

        except ImportError:
            return a0

                        if np.isfinite(sc):
                            a = a_new
                            if sc < best_score:
                                best_a = a.copy()
                                best_score = sc
                            iter_count += 1
            except Exception:
                break
        
        sc = safe_eval(a)
        if sc < best_score:
            best_a = a.copy()
            best_score = sc
        return a

    best_a = normalize(base.copy())
    best_score = safe_eval(best_a)

    total_time = hard_deadline - t0
    
    if total_time < 30:

        out = [float(x) for x in best_a]
        return out if out and all(np.isfinite(x) for x in out) else [1.0]*n

    p1 = 0.35 * total_time
    a = adam_phase(a, p1, temp_start=10.0, temp_end=2.0, lr0=0.03)

        p4 = 0.10 * total_time
        a = lp_phase(a, p4, K=200, blend_t=0.02)

    if time.monotonic() < hard_deadline:
        remain = hard_deadline - time.monotonic()
        if remain > 20:
            a = adam_phase(a, remain - 5, temp_start=0.5, temp_end=0.05, lr0=0.003)

    return out
