```python
def compute_max_grad(a, active_k, K):
    """
    Compute subgradient of max(b) using active set.
    a: shape (n,)
    active_k: array of indices where b is near max
    K: number of active indices
    """
    n = len(a)
    g = np.zeros(n)
    a_rev = a[::-1]
    
    for k in active_k:
        k = int(k)
        # g[m] gets a[k-m] for valid m
        # a[k-m] is valid when 0 <= k-m < n, i.e., k-n+1 <= m <= k
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi < lo:
            continue
        # First term: a[k-m] for m in [lo, hi]
        # This is a_rev[n-1-(k-m)] = a_rev[n-1-k+m]
        # Simpler: just use a directly
        for m in range(lo, hi + 1):
            j = k - m
            g[m] += a[j]
        # Second term (except at center): a[j] = a[k-m] for m in [lo, hi], m != k/2
        if k % 2 == 0:
            center = k // 2
            if lo <= center <= hi:
                for m in range(lo, hi + 1):
                    if m != center:
                        g[m] += a[k - m]
            else:
                for m in range(lo, hi + 1):
                    g[m] += a[k - m]
        else:
            for m in range(lo, hi + 1):
                g[m] += a[k - m]
    
    g /= K
    return g
```

This avoids the softmax distortion. The projection onto {a ≥ 0, sum(a) = 1} is:

```python
    S = a.sum()
    if S < 0.01:
        return math.inf
    b = _fft_conv(a)
    return 2.0 * n * b.max() / (S * S)

def _grad_max_b(a, b, K=20):
    """Subgradient of max(b) w.r.t. a"""
    n = len(a)
    # Find top K indices
    idx = np.argpartition(b, -min(K, b.size))[-min(K, b.size):]
    bmax = b[idx].max()
    # Include all indices close to max
    threshold = bmax * 0.999
    idx = np.where(b >= threshold)[0]
    K = len(idx)

        j = k - m

        diag = (m == j)
        g[m[~diag]] += a[j[~diag]]
    g /= K
    return g
```

```python
import numpy as np
import time
import math
from scipy.fft import rfft, irfft, next_fast_len


def _fft_conv(a):
    n = len(a)
    if n <= 0:
        return np.array([0.0])
    fsize = next_fast_len(2 * n - 1)
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: 2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return math.inf
    b = _fft_conv(a)
    mb = float(b.max())
    if not math.isfinite(mb):
        return math.inf
    return 2.0 * n * mb / (S * S)


def _project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}"""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a[:] = 1.0 / n
    return a


def _grad_max_b(a, b, K=15):
    """Subgradient of max(b) w.r.t. a, using top-K active lags."""
    n = len(a)
    bmax = float(b.max())
    # Adaptive active set: all lags within 0.1% of max, cap at K
    threshold = bmax * (1.0 - 1e-3)
    active = np.where(b >= threshold)[0]
    if len(active) > K:
        # Keep only top K
        top_idx = np.argpartition(-b, K)[:K]
        active = top_idx
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    K = len(active)
    g = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi < lo:
            continue
        m = np.arange(lo, hi + 1)
        j = k - m
        g[m] += a[j]
        diag = (m == j)
        if np.any(~diag):
            g[m[~diag]] += a[j[~diag]]
    g /= K
    return g, K


def _grad_full_score(a, b, K=15):
    """Gradient of score = 2n * max_b / S^2 w.r.t. a"""
    n = len(a)
    S = float(a.sum())
    bmax = float(b.max())
    g_b, Ka = _grad_max_b(a, b, K)
    # d(score)/da = 2n * (g_b * S^2 - bmax * 2S * 1) / S^4
    #            = 2n * (g_b * S - 2*bmax) / S^3
    g = 2.0 * n * (g_b * S - 2.0 * bmax) / (S * S * S)
    # Remove component along (1,...,1) for sum constraint
    g = g - g.mean()
    return g, Ka

        g, Ka = _grad_full_score(a, b, K)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:

            a0 = np.clip(a0, 0, 1000)
            a0 = _project_simplex(a0)
            
            alloc_r = min(8.0, deadline - time.monotonic() - 140)
            if alloc_r < 3:
                break
            try:
                ar, sr = _optimize_gradient(a0, time.monotonic() + alloc_r, seed + 5000 + restart)
                if sr < best_score:
                    best_seq = ar.copy()
                    best_score = sr
            except Exception:
                pass
    
    # === Phase 5: Try other lengths ===
    other_lengths = [128, 160, 224, 256, 320, 192]
    other_lengths = [L for L in other_lengths if L != n_parent]
    
    if time.monotonic() < deadline - 80:
        for L in other_lengths[:5]:
            if time.monotonic() > deadline - 80:
                break
            try:
                aL, sL = _try_length(L, seed + L * 100, deadline, parent)
                if aL is not None and math.isfinite(sL) and sL < best_score:
                    best_seq = aL.copy()
                    best_score = sL
            except Exception:
                pass
    
    # === Phase 6: Final polish ===
    if time.monotonic() < deadline - 10:
        alloc6 = max(5.0, deadline - time.monotonic() - 8.0)
        try:
            a6, s6 = _local_search(best_seq, time.monotonic() + alloc6, seed + 99999)
            if s6 < best_score:
                best_seq = a6.copy()
                best_score = s6
        except Exception:
            pass
    
    # === Final validation ===
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64).ravel(), 0.0, 1000.0)
    if best_seq.size == 0 or not np.isfinite(best_seq).all():
        best_seq = np.ones(64, dtype=np.float64) / 64.0
    
    final_score = math.inf
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    except Exception:
        pass
    
    if not math.isfinite(final_score):
        # Fallback
        fb = np.ones(128, dtype=np.float64) / 128.0
        try:
            fs = float(evaluate_sequence([float(x) for x in fb]))
            if math.isfinite(fs):
                best_seq = fb
                final_score = fs
        except Exception:
            best_seq = np.ones(8, dtype=np.float64) / 8.0
    
    return [float(x) for x in best_seq]
```
