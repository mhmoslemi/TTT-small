    a = init.copy()
    for step in range(steps):
        if deadline and time.monotonic() > deadline:
            break

        grad = 2 * Q @ a
        # Projected gradient step onto {a >= 0, sum(a) = 1}
        # First subtract gradient
        a_new = a - eta * grad
        # Then project: find smallest s such that sum(max(a_new - s, 0)) = 1
        # This is the Euclidean projection onto the simplex

        # Check if it improved
        f_old = a @ Q @ a
        f_new = a_new @ Q @ a
        if f_new < f_old:
            a = a_new
        else:
            eta *= 0.5
            if eta < 1e-10:

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cs = np.arange(1, n+1)
    rho = np.nonzero(u * cs > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def _score(a):

def _build_Q(n, active_lags, weights):
    """Build the Q matrix for the QP: Q[i,j] = sum of weights where i+j = k_l."""
    Q = np.zeros((n, n), dtype=np.float64)

        k = int(k)
        if k < 0 or k >= 2*n - 1:

        i_e = min(k, n - 1)
        if i_s <= i_e:
            ii = np.arange(i_s, i_e + 1)
            jj = k - ii
            Q[ii, jj] += w
    return Q
