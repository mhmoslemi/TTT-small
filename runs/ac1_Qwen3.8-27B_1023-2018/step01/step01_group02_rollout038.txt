```python
def _solve_lp_direction(a, n):
    """Solve LP to find descent direction d that minimizes max_k (a*d)[k]."""
    # Variables: d_0, ..., d_{n-1}, t (index n)
    # Minimize t
    c = np.zeros(n + 1)
    c[n] = 1  # minimize t
    
    # Inequality constraints: (a*d)[k] - t <= 0 for k = 0, ..., 2n-2
    # A_ub @ [d, t] <= b_ub (all zeros)
    n_ineq = 2 * n - 1
    A_ub = np.zeros((n_ineq, n + 1))
    for k in range(n_ineq):
        # (a*d)[k] = sum_i a[i] * d[k-i] for valid i
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            A_ub[k, i] = a[i]
        A_ub[k, n] = -1  # -t
    
    b_ub = np.zeros(n_ineq)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1
    b_eq = np.zeros(1)
    
    # Bounds: -a_i <= d_i <= 1 for i=0,...,n-1; t is free
    bounds = [(-a[i], 1.0) for i in range(n)] + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, 0
    
    d = res.x[:n]
    t = res.x[n]
    return d, t
```
