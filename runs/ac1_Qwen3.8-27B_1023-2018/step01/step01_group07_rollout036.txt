    while p < x:

    return p

        a = np.ones(n, dtype=np.float64)
        s = float(n)
    a = a / s
    
    # FFT setup
    fft_len = _next_pow2(2 * n - 1)
    fft_len2 = _next_pow2(3 * n - 2)  # for gradient computation
    
    def conv(a):
        F = np.fft.rfft(a, fft_len)
        return np.fft.irfft(F * F, fft_len)[:2*n-1]

        # grad[i] = 2 * sum_k w[k] * a[k-i]

        n = len(v)
        if n == 0:

        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
        theta = (cssv[rho] - 1) / (rho + 1.0)

def _next_pow2(x):
    p = 1
    while p < x:

def _project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(v - theta, 0)
