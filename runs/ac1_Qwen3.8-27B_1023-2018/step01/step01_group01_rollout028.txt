import numpy as np
import time


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s)) - 15.0

    # Load base sequence
    base = None
    try:
        b = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if b.size > 0 and np.all(np.isfinite(b)):
            base = np.clip(b, 0.0, 1000.0)
    except Exception:
        pass
    if base is None:
        base = np.ones(6962, dtype=np.float64)
    n = len(base)

    # FFT size: need >= 3n-2 for gradient conv, and >= 2n-1 for self-conv
    sizeG = 1 << (3 * n - 2).bit_length()

    rng = np.random.default_rng(seed)

    def safe_eval(a):
        try:
            s = evaluate_sequence([float(x) for x in a])
            return s if np.isfinite(s) else np.inf
        except Exception:
            return np.inf

    def normalize(a):
        s = a.sum()
        if s < 1e-30:
            return a
        return a / s

    # Initialize best
    best_a = normalize(base.copy().astype(np.float64))
    best_s = safe_eval(best_a)

    # Pre-allocate buffers
    C_buf = np.empty(2 * n - 1, dtype=np.float64)
    w_buf = np.empty(2 * n - 1, dtype=np.float64)
    wr_buf = np.empty(2 * n - 1, dtype=np.float64)
    conv_buf = np.empty(3 * n - 2, dtype=np.float64)
    grad_buf = np.empty(n, dtype=np.float64)

    def compute_grad(a, temp):
        """Compute gradient of smooth max of (a*a), normalized to sum(a)=1."""
        fa = np.fft.rfft(a, sizeG)
        C = np.fft.irfft(fa * fa, sizeG)[:2 * n - 1]
        Cmax = C.max()
        z = temp * (C - Cmax)
        z = np.clip(z, -500.0, 500.0)
        ez = np.exp(z, out=w_buf)
        ez_sum = ez.sum()
        if not np.isfinite(ez_sum) or ez_sum < 1e-30:
            w_buf[:] = 1.0 / (2 * n - 1)
            M = Cmax
        else:
            w_buf /= ez_sum
            M = Cmax + np.log(ez_sum) / temp
        # w_rev
        wr_buf[:] = w_buf[::-1]
        fwr = np.fft.rfft(wr_buf, sizeG)
        conv = np.fft.irfft(fa * fwr, sizeG)[:3 * n - 2]
        # grad[i] = 2 * conv[2n-2-i], i=0..n-1
        # That's conv[2n-2], conv[2n-3], ..., conv[n-1] → reversed slice
        grad_buf[:] = 2.0 * conv[n - 1:2 * n - 1][::-1]
        return grad_buf, M

    def optimize_run(a0, seconds, temp_lo, temp_hi, lr0, lr1):
        """Run one optimization phase."""
        nonlocal best_a, best_s
        a = normalize(a0.copy().astype(np.float64))
        m = np.zeros(n, dtype=np.float64)
        v = np.zeros(n, dtype=np.float64)
        step = 0
        start = time.monotonic()
        b1, b2 = 0.9, 0.999
        last_eval = start

        while True:
            if time.monotonic() > deadline:
                break
            elapsed = time.monotonic() - start
            if elapsed > seconds:
                break

            # Temperature: smooth ramp from temp_lo to temp_hi
            frac = min(1.0, elapsed / max(seconds, 1e-6))
            # Use power-law for smoother transition
            temp = temp_lo * (temp_hi / temp_lo) ** (frac ** 0.7)
            # LR: exponential decay
            lr = lr0 * (lr1 / lr0) ** frac

            grad, M = compute_grad(a, temp)

            step += 1
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * grad * grad
            m_hat = m / (1 - b1 ** step)
            v_hat = v / (1 - b2 ** step)
            a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            np.clip(a, 0.0, 1000.0, out=a)
            a = normalize(a)

            # Periodic exact evaluation
            now = time.monotonic()
            if step % 800 == 0 and now - last_eval > 0.8:
                last_eval = now
                sc = safe_eval(a)
                if np.isfinite(sc) and sc < best_s:
                    best_s = sc
                    best_a = a.copy()

        sc = safe_eval(a)
        if np.isfinite(sc) and sc < best_s:
            best_s = sc
            best_a = a.copy()
        return a

    # Build seed pool
    seeds = [best_a.copy()]
    # Raised cosine
    x = np.linspace(0, 2 * np.pi, n)
    seeds.append(normalize(1.0 + 0.4 * np.cos(x)))
    # Sine-shaped
    seeds.append(normalize(0.5 + 0.5 * np.sin(np.pi * np.arange(n) / n) ** 2))
    # Two-peak
    two = np.ones(n)
    two[n // 4: 3 * n // 4] = 0.3
    seeds.append(normalize(two))
    # Random smooth
    r = rng.random(n)
    kernel = np.exp(-0.5 * (np.arange(41) - 20) ** 2 / 25.0)
    r = np.convolve(r, kernel, 'same')
    seeds.append(normalize(np.clip(r, 0.01, None)))

    # Score seeds and pick top ones
    scored = []
    for s in seeds:
        s = normalize(np.clip(s, 0.0, 1000.0))
        sc = safe_eval(s)
        scored.append((sc, s))
    scored.sort(key=lambda t: t[0])

    # Phase allocation
    avail = max(10.0, hard_time_left())

    def hard_time_left():
        return max(0.0, deadline - time.monotonic())

    avail = hard_time_left()
    # Allocate: 40% top seed, 25% second, 15% third, 10% fourth, 10% reserve for final polish
    n_top = min(4, len(scored))
    fracs = [0.40, 0.25, 0.15, 0.10, 0.10]
    fracs = fracs[:n_top] + [max(0, 1 - sum(fracs[:n_top]))]
    # If only 1 seed, put all in it
    if n_top == 1:
        fracs = [0.85, 0.15]

    for i in range(n_top):
        remaining = hard_time_left()
        if remaining < 15:
            break
        sec_i = fracs[i] * avail * 0.88  # leave room
        sc_i, s_i = scored[i]
        # Temperature schedule: start moderate, end very sharp
        t_lo = 5.0 + 3.0 * i
        t_hi = 300.0 + 200.0 * i
        lr_s = 0.015 - 0.002 * i
        lr_e = 0.002 - 0.001 * i
        lr_s = max(lr_s, 0.002)
        lr_e = max(lr_e, 0.0005)
        optimize_run(s_i, sec_i, t_lo, t_hi, lr_s, lr_e)

    # Final polish on best candidate
    remaining = hard_time_left()
    if remaining > 20:
        # Very high temperature, small LR
        optimize_run(best_a.copy(), remaining * 0.6, 200.0, 2000.0, 0.003, 0.0005)
    remaining = hard_time_left()
    if remaining > 10:
        optimize_run(best_a.copy(), remaining * 0.9, 500.0, 5000.0, 0.001, 0.0002)

    # Final validation and return
    best_a = normalize(np.clip(best_a, 0.0, 1000.0))
    fs = safe_eval(best_a)
    if not np.isfinite(fs):
        # Fallback
        best_a = normalize(base.copy().astype(np.float64))
        fs = safe_eval(best_a)
    if not np.isfinite(fs):
        best_a = np.ones(n, dtype=np.float64) / n
        fs = safe_eval(best_a)

    out = [float(x) for x in best_a]
    if not out or any(not np.isfinite(x) for x in out):
        out = [1.0] * n
    return out
