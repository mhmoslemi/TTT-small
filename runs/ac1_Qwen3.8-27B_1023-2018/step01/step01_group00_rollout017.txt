import time
import math
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog


def _get_parent():
    try:
        h = globals()["height_sequence_1"]
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        p = np.ones(192, dtype=np.float64)
    p = np.clip(p, 0.0, 1000.0)
    if p.size == 0 or not np.isfinite(p).all():
        p = np.ones(192, dtype=np.float64)
    return p


def _fft_convolve(a, n, fsize):
    F = rfft(a, fsize)
    c = irfft(F * F, fsize)[: (2 * n - 1)]
    c[np.abs(c) < 1e-12] = 0.0
    c[c < 0.0] = 0.0
    return c


def _score_fft(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    fsize = next_fast_len(2 * n - 1)
    b = _fft_convolve(a, n, fsize)
    m = float(b.max())
    if not np.isfinite(m):
        return math.inf
    return 2.0 * n * m / (s * s)


def _build_A(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    return A


def _pencil_optimize(f0, n, stop_time, max_iters=8000):

    f = np.array(f0, dtype=np.float64).copy()

    f = f / S

        b = _fft_convolve(f, n, fsize)
