def project_simplex(v):
    x = np.asarray(v, dtype=np.float64).copy()
    if x.size ==0: return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    m = np.arange(1, len(u)+1, dtype=np.float64)
    cond = (u - (cssv - 1.0)/m) > 0
    if not cond.any():
       return np.ones(len(x))/len(x)
    rho = int(np.where(cond)[0][-1])
    theta = (cssv[rho] - 1.0)/(rho + 1.0)
    out = np.maximum(x - theta, 0.0)
    s=out.sum()
    if s>0: out /= s
    else: out = np.ones(len(x))/len(x)

def _proj_simplex(v):
    x=np.array(v,dtype=np.float64).ravel()
    if x.size==0: return x
    if not np.all(np.isfinite(x)): x=np.zeros_like(x)
    u=np.sort(x)[::-1]
    m=np.arange(1,u.size+1,dtype=np.float64)
    cssv=np.cumsum(u)
    cond = (u - (cssv - 1.0)/m) > 0.0
    if not np.any(cond):
       y=np.ones(u.size)/u.size
    else:
       rho = int(np.where(cond)[0][-1])
       theta = (cssv[rho] - 1.0)/(rho+1.0)
       y = np.maximum(x - theta, 0.0)
       s=y.sum()

           y = np.ones(x.size)/x.size
       else:
           y /= s
    return y

```python
import time
import math
import numpy as np


def _next_pow2(x):
    p = 1
    x = int(max(1, x))
    while p < x:
        p <<= 1
    return p


def _norm_nonneg(arr):
    a = np.asarray(arr, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum(dtype=np.float64))
    if not math.isfinite(s) or s < 1e-12:
        a = np.ones(a.size, dtype=np.float64)
    else:
        a = a / s
    s2 = float(a.sum(dtype=np.float64))
    if not math.isfinite(s2) or s2 < 1e-12:
        a = np.ones(a.size, dtype=np.float64) / float(a.size)
    else:
        a = a / s2
    return a


def _proj_simplex(v):
    x = np.asarray(v, dtype=np.float64).ravel()
    m = x.size
    if m == 0:
        return x
    if not np.all(np.isfinite(x)):
        x = np.where(np.isfinite(x), x, 0.0)

    s = float(x.sum(dtype=np.float64))
    mn = float(x.min())
    if math.isfinite(s) and s > 1e-300 and mn >= -1e-12:
        y = np.maximum(x, 0.0)
        ss = float(y.sum(dtype=np.float64))
        if ss > 1e-300:
            return y / ss

    u = np.sort(x)[::-1]
    cssv = np.cumsum(u, dtype=np.float64)
    m_arr = np.arange(1, m + 1, dtype=np.float64)
    cond = (u - (cssv - 1.0) / m_arr) > 0.0
    if not np.any(cond):
        return np.ones(m, dtype=np.float64) / float(m)

    rho = int(np.flatnonzero(cond)[-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    y = np.maximum(x - theta, 0.0)
    ss = float(y.sum(dtype=np.float64))
    if not math.isfinite(ss) or ss <= 1e-300:
        return np.ones(m, dtype=np.float64) / float(m)
    return y / ss


def _same_array(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    if a.size != b.size:
        return False
    if a.size == 0:
        return True
    return bool(np.allclose(a, b, rtol=1e-8, atol=1e-10))


def _full_conv(a, L):
    n = a.size
    Fx = np.fft.rfft(a, L)
    c = np.fft.irfft(Fx * Fx, L)
    Lc = 2 * n - 1
    c = c[:Lc]
    if c.size:
        c = np.where(c > 0.0, c, 0.0)
        Fv = float(c.max())
        if not math.isfinite(Fv):
            Fv = 0.0
    else:
        Fv = 0.0
    return c, Fx, Fv


def _weights(c, F, tau_frac):
    tau = max(float(tau_frac), 1e-12) * max(float(F), 1e-12)
    tau = max(tau, 1e-12 * float(F))
    z = (c - F) / tau
    z = np.where(z > -700.0, z, -700.0)
    z = np.where(z < 0.0, z, 0.0)
    w = np.exp(z)
    s = float(w.sum(dtype=np.float64))
    if s <= 1e-300 or not math.isfinite(s):
        w = np.zeros_like(c)
        if w.size:
            w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    return w


def _gradient(a, w, L, Fx=None):
    n = a.size
    if Fx is None:
        Fx = np.fft.rfft(a, L)
    wR = np.zeros(L, dtype=np.float64)
    wR[:2 * n - 1] = w[::-1]
    FW = np.fft.rfft(wR, L)
    conv = np.fft.irfft(Fx * FW, L)
    C = conv[n - 1:2 * n - 1][::-1]
    return 2.0 * C


def _exact_score(seq):
    try:
        val = float(evaluate_sequence(seq))
    except Exception:
        return math.inf
    return val if math.isfinite(val) else math.inf


def _optimize(a0, deadline, seed_extra=0):
    a = _proj_simplex(np.array(a0, dtype=np.float64).copy())
    n = a.size
    if n <= 1:
        return a

    L = _next_pow2(max(1, 3 * n - 2))
    c, Fx, F = _full_conv(a, L)

    best_a = a.copy()
    best_F = F
    start_F = F

    rng = np.random.default_rng(0x9E3779B97F4A7C15 + int(seed_extra))
    base_lr = 1.0
    tau_frac = 0.08
    stagn = 0
    perturb = 0
    no_improve = 0
    it = 0

    vel = np.zeros(n, dtype=np.float64)
    momentum = 0.5

    while time.monotonic() < deadline - 1.0:
        it += 1

        w = _weights(c, F, tau_frac)
        grad = _gradient(a, w, L, Fx)
        raw_d = grad - float(np.mean(grad))
        raw_d = np.where(np.isfinite(raw_d), raw_d, 0.0)
        raw_gm = math.sqrt(max(float(np.dot(raw_d, raw_d)), 0.0))

        if not math.isfinite(raw_gm) or raw_gm <= 1e-14 * max(1.0, F):
            if perturb > 8 or (it > 200 and F >= start_F - 1e-10):
                break
            scale = 0.25 / math.sqrt(float(n))
            a = np.maximum(a * np.exp(rng.normal(0.0, scale, n)), 0.0)
            a = _proj_simplex(a)
            c, Fx, F = _full_conv(a, L)
            perturb += 1
            tau_frac = min(1.0, tau_frac * 1.25)
            vel = np.zeros(n, dtype=np.float64)
            momentum = 0.5
            if F < best_F - 1e-12:
                best_F = F
                best_a = a.copy()
                no_improve = 0
            continue

        if momentum > 0.0:
            vel = momentum * vel + raw_d
        else:
            vel = raw_d.copy()
            momentum = 0.55

        d = vel - float(np.mean(vel))
        d = np.where(np.isfinite(d), d, 0.0)
        gm2 = float(np.dot(d, d))
        if not math.isfinite(gm2) or gm2 <= 0.0:
            d = raw_d
            gm2 = float(np.dot(raw_d, raw_d))
        gm = math.sqrt(max(gm2, 0.0))

        if not math.isfinite(gm) or gm <= 1e-14 * max(1.0, F):
            if perturb > 8 or (it > 200 and F >= start_F - 1e-10):
                break
            scale = 0.3 / math.sqrt(float(n))
            base_arr = best_a if best_a is not None else a
            a = np.maximum(base_arr * np.exp(rng.normal(0.0, scale, n)), 0.0)
            a = _proj_simplex(a)
            c, Fx, F = _full_conv(a, L)
            perturb += 1
            tau_frac = min(1.0, tau_frac * 1.3)
            vel = np.zeros(n, dtype=np.float64)
            momentum = 0.5
            if F < best_F - 1e-12:
                best_F = F
                best_a = a.copy()
                no_improve = 0
            continue

        a_l2 = math.sqrt(max(float(np.dot(a, a)), 1e-300))
        step = base_lr * min(0.08 * a_l2, 4.0 * F)
        if not math.isfinite(step) or step < 1e-12:
            step = 1e-10
        step = min(step, 0.30 * a_l2)

        unit = d / gm
        accepted = False
        accepted_factor = 0.0

        for factor in (1.0, 0.6, 0.35, 0.2, 0.1):
            trial_step = step * factor
            if trial_step <= 0.0:
                break
            trial = _proj_simplex(a - trial_step * unit)
            if not np.all(np.isfinite(trial)):
                continue
            ct, Fxt, Ft = _full_conv(trial, L)
            if math.isfinite(Ft) and Ft < F - max(1e-13, 1e-10 * F):
                a = trial
                F = Ft
                c = ct
                Fx = Fxt
                accepted = True
                accepted_factor = factor
                break

        if accepted:
            stagn = 0
            if F < best_F - 1e-12:
                best_F = F
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if accepted_factor >= 0.95:
                base_lr = min(base_lr * 1.18, 60.0)
                momentum = min(0.9, momentum + 0.05)
            elif accepted_factor >= 0.55:
                base_lr = min(base_lr * 1.06, 60.0)
                momentum = min(0.9, momentum + 0.02)
            else:
                base_lr = max(base_lr * 0.9, 1e-4)
                momentum = max(0.0, momentum - 0.2)

            if it % 700 == 0:
                tau_frac = max(0.015, tau_frac * 0.92)
        else:
            stagn += 1
            no_improve += 1
            base_lr = max(base_lr * 0.6, 1e-4)
            momentum = 0.0
            if stagn >= 15:
                tau_frac = min(1.0, tau_frac * 1.5)
            if stagn >= 40:
                base_arr = best_a if best_a is not None else a
                scale = 0.45 / math.sqrt(float(n))
                a = np.maximum(base_arr * np.exp(rng.normal(0.0, scale, n)), 0.0)
                a = _proj_simplex(a)
                c, Fx, F = _full_conv(a, L)
                base_lr = 1.2
                tau_frac = min(1.0, tau_frac * 1.5)
                stagn = 0
                perturb += 1
                vel = np.zeros(n, dtype=np.float64)
                momentum = 0.5

            if perturb > 12 or (no_improve > 2500 and F >= start_F - 1e-8):
                break

        if time.monotonic() > deadline - 2.0:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    t0 = time.monotonic()
    total = max(1.0, float(budget_s))
    reserve = max(2.0, 0.05 * total)
    deadline = t0 + max(1.0, total - reserve)
    rng = np.random.default_rng(seed)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size > 0 and np.all(np.isfinite(p)):
            parent = p
    except Exception:
        parent = None

    if parent is None:
        n = 4806
        parent_arr = None
    else:
        n = int(parent.size)
        if n <= 0:
            n = 4806
            parent = None
        if parent is not None and n > 20000:
            newn = 8192
            idx = np.linspace(0, n - 1, newn).astype(np.int64)
            parent = parent[idx]
            n = int(parent.size)
        parent_arr = _norm_nonneg(parent)

    uniform = np.ones(n, dtype=np.float64) / float(n)
    cands = []

    if parent_arr is not None:
        cands.append(parent_arr)
        try:
            nz = np.maximum(parent_arr * np.exp(rng.normal(0.0, 0.18 / math.sqrt(float(n)), n)), 0.0)
            cands.append(_norm_nonneg(nz))
        except Exception:
            cands.append(uniform)

    cands.append(uniform)

    if n >= 32:
        i = np.arange(n, dtype=np.float64)
        try:
            cosv = 0.5 * (1.0 + np.cos(2.0 * math.pi * i / float(n)))
            cands.append(_norm_nonneg(cosv))
        except Exception:
            pass

        try:
            base = parent_arr if parent_arr is not None else uniform
            r = 0.5 * base + 0.5 * np.exp(rng.normal(0.0, 1.0, n))
            r = np.maximum(r, 0.0)
            R = np.fft.rfft(r)
            if R.size > 4:
                R[2:R.size // 2] = 0.0
            r = np.fft.irfft(R, n)
            cands.append(_norm_nonneg(r))
        except Exception:
            pass

    best_score = math.inf
    best_a = None
    unique_arrs = []
    cand_scores = []

    for arr in cands:
        arr = _norm_nonneg(arr)
        if arr.size == 0:
            continue
        if any(_same_array(arr, u) for u in unique_arrs):
            continue
        unique_arrs.append(arr)
        sc = math.inf
        if time.monotonic() < deadline - 1.0:
            sc = _exact_score([float(x) for x in arr])
        cand_scores.append(sc)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = arr.copy()

    starts = []
    if parent_arr is not None:
        for arr in unique_arrs:
            if _same_array(arr, parent_arr):
                starts.append(arr)
                break

    rest = [arr for arr in unique_arrs if parent_arr is None or not _same_array(arr, parent_arr)]
    rest_scores = []
    for arr in rest:
        s = math.inf
        for a2, sc in zip(unique_arrs, cand_scores):
            if _same_array(a2, arr):
                s = sc
                break
        rest_scores.append((arr, s))

    rest_scores.sort(key=lambda t: (0 if math.isfinite(t[1]) else 1, t[1] if math.isfinite(t[1]) else 0.0))
    for arr, _ in rest_scores:
        if not any(_same_array(arr, s) for s in starts):
            starts.append(arr)
        if len(starts) >= 4:
            break

    if not starts:
        starts = [uniform]

    starts = starts[:4]

    def _try_update(arr):
        nonlocal best_score, best_a
        if arr is None:
            return
        if time.monotonic() >= deadline - 2.0:
            return
        try:
            seq = [float(x) for x in np.asarray(arr, dtype=np.float64)]
            sc = _exact_score(seq)
        except Exception:
            return
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = np.asarray(arr, dtype=np.float64).copy()

    for idx, arr in enumerate(starts):
        if time.monotonic() >= deadline - 3.0:
            break
        now = time.monotonic()
        rem = max(1.0, deadline - now)
        if len(starts) == 1 or idx == len(starts) - 1:
            child_dl = deadline
        else:
            frac = 0.85 if idx == 0 else 0.72
            child_dl = now + frac * rem
            child_dl = min(child_dl, deadline)
            child_dl = max(child_dl, now + 1.0)
        try:
            opt_arr = _optimize(arr, child_dl, seed_extra=1000 + idx * 17 + int(seed % 1000))
        except Exception:
            opt_arr = arr
        _try_update(opt_arr)

    _try_update(best_a)

    fallback_len = int(best_a.size) if best_a is not None else int(n)
    if fallback_len <= 0 or fallback_len > 20000:
        fallback_len = 4806

    if time.monotonic() < deadline - 1.0:
        try:
            uni = np.ones(fallback_len, dtype=np.float64)
            sc = _exact_score([1] * fallback_len)
            if math.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = uni
        except Exception:
            pass

    if best_a is None or not math.isfinite(best_score):
        m = fallback_len
        return [1] * m

    if math.isfinite(best_score) and best_score <= 2.0 + 1e-6:
        try:
            uni = np.ones(best_a.size, dtype=np.float64) / float(best_a.size)
            if _same_array(best_a, uni):
                m = int(best_a.size)
                if m <= 0 or m > 20000:
                    m = 4806
                return [1] * m
        except Exception:
            pass

    out = []
    for x in np.asarray(best_a, dtype=np.float64):
        try:
            y = float(x)
        except Exception:
            y = 0.0
        if not math.isfinite(y):
            y = 0.0
        y = max(0.0, min(1000.0, y))
        out.append(y)

    if not out:
        out = [1.0]

    return out
```
