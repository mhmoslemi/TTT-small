    arr=np.asarray(seq, dtype=np.float64).ravel()
    if arr.size == 0: return arr

    s=float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(arr.size, 1.0/arr.size, dtype=np.float64)

def _conv(a):
    n=a.size

def _npow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean(seq):
    arr = np.asarray(seq, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.asarray([], dtype=np.float64)
    arr = np.where(np.isfinite(arr) & (arr > 0.0), arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(arr.size, 1.0 / float(arr.size), dtype=np.float64)
    return arr / s


def _conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.asarray([], dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    out_len = 2 * n - 1
    N = _npow2(out_len)
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)[:out_len]
    b = np.maximum(b, 0.0)
    return b


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return float("inf")
    b = _conv(a)
    if b.size == 0:
        return float("inf")
    val = 2.0 * float(n) * float(b.max()) / (s * s)
    return float(val) if np.isfinite(val) else float("inf")

def _grad_conv(u, a):
    """
    For u over convolution lags 0..2n-1 and a length n,
    return h_i = sum_l u_l a_{l-i} valid for i=0..n-1.
    """
    u = np.asarray(u, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.asarray([], dtype=np.float64)
    out_len = 3 * n - 2
    N = _npow2(out_len)
    ru = a[::-1].astype(np.float64, copy=True)
    fu = np.fft.rfft(u, N)
    fr = np.fft.rfft(ru, N)
    c = np.fft.irfft(fu * fr, N)[:out_len]
    out = c[n - 1:2 * n - 1]
    out = np.where(np.isfinite(out) & (out > 0.0), out, 0.0)
    return out

        ez = np.exp(z)
        s = float(ez.sum())
        if not np.isfinite(s) or s <= 0.0:
            u = np.zeros(p.size, dtype=np.float64)
            u[int(np.argmax(p))] = 1.0
            soft = pm
        else:
            u = ez / s
            soft = pm + tau * float(np.log(s))

    gc = _grad_conv(u, a)
    grad_a = 2.0 * float(n) * gc
    gdot = float(np.dot(a, grad_a))
    gx = a * (grad_a - gdot)
    gx -= float(np.mean(gx))
    gx += 2.0 * pen * x

    obj = soft + pen * float(np.dot(x, x))
    if not np.isfinite(obj):
        obj = float("inf")
    if not np.all(np.isfinite(gx)):
        gx = np.zeros(n, dtype=np.float64)
    return float(obj), gx


def _lbfgs_once(x0, n, tau, stage_iters):
    if _sp_minimize is None:
        return x0.copy()
    n = int(n)
    stage_iters = int(max(1, stage_iters))
    x = np.asarray(x0, dtype=np.float64).copy()
    x -= float(np.mean(x))
    x = np.clip(x, -40.0, 40.0)

    cache = {"x": None, "fg": None}

    def fun(xx):
        val, grad = _soft_objective_gradient(xx, n, tau)
        if not np.isfinite(val):
            val = 1e6
            grad = np.zeros(n, dtype=np.float64)
        cache["x"] = np.asarray(xx, dtype=np.float64).copy()
        cache["fg"] = (float(val), grad)
        return float(val)

    def jac(xx):
        xx = np.asarray(xx, dtype=np.float64)
        if cache["x"] is not None and np.array_equal(xx, cache["x"]):
            return cache["fg"][1]
        val, grad = _soft_objective_gradient(xx, n, tau)
        if not np.isfinite(val):
            grad = np.zeros(n, dtype=np.float64)
        cache["x"] = xx.copy()
        cache["fg"] = (float(val), grad)
        return grad

    try:
        bounds = [(-40.0, 40.0)] * n
        res = _sp_minimize(
            fun,
            x,
            jac=jac,
            method="L-BFGS-B",
            bounds=bounds,
            options={
                "maxiter": stage_iters,
                "ftol": 1e-12,
                "gtol": 1e-7,
                "maxls": 15,
                "disp": False,
            },
        )
        if getattr(res, "x", None) is not None:
            xn = np.asarray(res.x, dtype=np.float64)
            if xn.size == n and np.all(np.isfinite(xn)):
                x = xn
    except Exception:
        pass

    x -= float(np.mean(x))
    return np.clip(x, -40.0, 40.0).astype(np.float64, copy=False)


def _adam_refine(x0, n, limit, lr=0.02, tau_stages=None):
    x = np.asarray(x0, dtype=np.float64).copy()
    n = int(n)
    if x.size != n or n <= 0:
        return x0.copy(), float("inf")

    x -= float(np.mean(x))
    x = np.clip(x, -40.0, 40.0)
    best_x = x.copy()
    bestF = _score_x(x)

    if tau_stages is None:
        tau_stages = [0.20, 0.10, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015]
    per = int(max(20, min(350, 60000 // max(1, n))))
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2 = 0.90, 0.99
    stall = 0

    for tau in tau_stages:
        for _ in range(per):
            if time.monotonic() >= limit - 1.0:
                break

            a = _softmax(x)
            p = float(n) * _conv(a)
            if p.size == 0:
                break
            pm = float(np.max(p))
            if not np.isfinite(pm):
                x = best_x.copy()
                break
            F = 2.0 * pm

            if F < bestF - 1e-13:
                bestF = F
                best_x = x.copy()
                stall = 0
                if lr < 0.05:
                    lr = min(0.05, lr * 1.02)
            else:
                stall += 1
                if stall > 90 and F > bestF + 0.01:
                    x = best_x.copy()
                    m *= 0.5
                    v *= 0.5
                    lr = max(0.002, lr * 0.85)
                    stall = 0
                    continue

            if tau <= 1e-5:
                act = np.where(p >= pm - (1e-4 + 1e-4 * max(1.0, pm)))[0]
                if act.size == 0:
                    act = np.array([int(np.argmax(p))])
                u = np.zeros(p.size, dtype=np.float64)
                u[act] = 1.0 / float(act.size)
            else:
                z = (p - pm) / tau
                z = np.clip(z, -700.0, 0.0)
                ez = np.exp(z)
                s = float(ez.sum())
                if not np.isfinite(s) or s <= 0.0:
                    u = np.zeros(p.size, dtype=np.float64)
                    u[int(np.argmax(p))] = 1.0
                else:
                    u = ez / s

            gc = _grad_conv(u, a)
            grad_a = 2.0 * float(n) * gc
            gdot = float(np.dot(a, grad_a))
            gx = a * (grad_a - gdot)
            gx -= float(np.mean(gx))
            if not np.all(np.isfinite(gx)):
                x = best_x.copy()
                break

            t += 1
            m = beta1 * m + (1.0 - beta1) * gx
            v = beta2 * v + (1.0 - beta2) * (gx * gx)
            mhat = m / (1.0 - beta1 ** t)
            vhat = v / (1.0 - beta2 ** t)
            step = lr * mhat / (np.sqrt(vhat) + 1e-8)
            x = x - step
            x -= float(np.mean(x))
            x = np.clip(x, -40.0, 40.0)

            if not np.all(np.isfinite(x)):
                x = best_x.copy()
                break

        if time.monotonic() >= limit - 1.0:
            break

    F = _score_x(x)
    if np.isfinite(F) and F < bestF:
        bestF = F
        best_x = x.copy()
    return best_x, bestF


def _lp_response(q, n):
    if _sp_linprog is None:
        return None
    n = int(n)
    q = _clean(q)
    if q.size != n or n <= 0 or n > 1000:
        return None

    b = _conv(q)
    M = float(b.max())
    if not np.isfinite(M) or M <= 0.0:
        return None

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = q

    c = -np.ones(n, dtype=np.float64)
    c -= 1e-6 * (q * float(n))
    bub = np.full(2 * n - 1, M, dtype=np.float64)

    res = None
    try:
        res = _sp_linprog(
            c,
            A_ub=A,
            b_ub=bub,
            bounds=(0.0, 1.0e6),
            method="highs",
            options={"presolve": True},
        )
    except Exception:
        res = None

    if res is None or not getattr(res, "success", False):
        try:
            res = _sp_linprog(c, A_ub=A, b_ub=bub, bounds=(0.0, 1.0e6), method="simplex")
        except Exception:
            res = None

    if res is None or not getattr(res, "success", False):
        return None

    g = np.asarray(res.x, dtype=np.float64)
    g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    return g / s


def _lp_polish(a0, n, limit):
    a = _clean(a0)
    n = int(n)
    if a.size != n or n <= 0:
        return a0, float("inf")

    bestF = _score(a)
    if not np.isfinite(bestF):
        return a0, float("inf")

    max_iters = 20 if n <= 256 else (10 if n <= 512 else 6)
    for _ in range(max_iters):
        if time.monotonic() > limit - 8.0:
            break

        g = _lp_response(a, n)
        if g is None:
            break

        best_cand = a.copy()
        best_cand_sc = bestF
        best_t = 0.0
        for t in (1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.05, 0.02):
            cand = (1.0 - t) * a + t * g
            cand = _clean(cand)
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_cand_sc - 1e-12:
                best_cand_sc = sc
                best_cand = cand
                best_t = t

        if best_t > 0.0:
            a = best_cand
            bestF = best_cand_sc
        else:
            break

    return a, _score(a)


def _refine(x0, n, limit, rng=None):
    n = int(n)
    x = np.asarray(x0, dtype=np.float64).copy()
    if x.size != n or n <= 0:
        return x0.copy(), float("inf")

    x -= float(np.mean(x))
    x = np.clip(x, -40.0, 40.0)
    best_x = x.copy()
    bestF = _score_x(x)

    now = time.monotonic()

    if _sp_minimize is not None and now < limit - 12.0 and n <= 2048:
        if n <= 64:
            taus = [0.12, 0.05, 0.02, 0.008]
            iters = 60
        elif n <= 128:
            taus = [0.12, 0.05, 0.02, 0.008]
            iters = 50
        elif n <= 256:
            taus = [0.10, 0.04, 0.016, 0.007]
            iters = 40
        elif n <= 512:
            taus = [0.08, 0.035, 0.015, 0.006, 0.0025]
            iters = 30
        elif n <= 1024:
            taus = [0.07, 0.03, 0.012, 0.005]
            iters = 22
        else:
            taus = [0.06, 0.025, 0.01]
            iters = 16

        max_stages = int(min(len(taus), max(1, (limit - time.monotonic() - 12.0) / 8.0)))
        for tau in taus[:max_stages]:
            if time.monotonic() > limit - 12.0:
                break
            x = _lbfgs_once(x, n, tau, iters)
            F = _score_x(x)
            if np.isfinite(F):
                if F < bestF + 0.05:
                    if F < bestF:
                        bestF = F
                        best_x = x.copy()
                else:
                    x = best_x.copy()

    if time.monotonic() < limit - 3.0:
        x2, F2 = _adam_refine(best_x, n, max(limit - 2.0, time.monotonic() + 1.0))
        if np.isfinite(F2) and F2 < bestF:
            bestF = F2
            best_x = x2.copy()

    if n <= 800 and time.monotonic() < limit - 20.0:
        a_cur = _softmax(best_x)
        a_lp, F_lp = _lp_polish(a_cur, n, limit - 5.0)
        if np.isfinite(F_lp) and F_lp < bestF:
            bestF = F_lp
            best_x = _x_from_a(a_lp, 0.0)

    if time.monotonic() < limit - 2.0:
        x3, F3 = _adam_refine(
            best_x,
            n,
            limit - 1.0,
            lr=0.015,
            tau_stages=[0.02, 0.008, 0.003, 0.0015],
        )
        if np.isfinite(F3) and F3 < bestF:
            bestF = F3
            best_x = x3.copy()

    return best_x, bestF


def _random_lowfreq(n, rng, mmax=8, amp=0.9):
    n = int(n)
    if n <= 0:
        return np.asarray([], dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    ii = np.arange(n, dtype=np.float64) / float(n)
    x = np.zeros(n, dtype=np.float64)
    maxm = min(int(mmax), max(1, n // 2))
    for m in range(1, maxm + 1):
        c = float(rng.normal(0.0, amp))
        s = float(rng.normal(0.0, amp))
        x += c * np.cos(2.0 * np.pi * float(m) * ii)
        x += s * np.sin(2.0 * np.pi * float(m) * ii)
    return _softmax(x)


def _make_start(kind, n, base, rng):
    n = int(n)
    if n <= 0:
        return np.asarray([], dtype=np.float64)
    if base is not None and np.size(base) > 0:
        base_n = _upscale(base, n)
    else:
        base_n = np.full(n, 1.0 / float(n), dtype=np.float64)

    kind = int(kind)
    if kind == 0:
        return base_n.copy()
    if kind == 1:
        return _softmax(_x_from_a(base_n, 0.25, rng))
    if kind == 2:
        return _random_lowfreq(n, rng, mmax=max(2, min(16, n // 12)), amp=0.9)
    if kind == 3:
        slopes = (-3.0, -1.5, 1.5, 3.0, -4.0, 4.0)
        slope = float(slopes[int(rng.integers(len(slopes)))])
        t = np.linspace(0.0, 1.0, n, dtype=np.float64)
        a = np.exp(slope * (t - 0.5))
        s = float(a.sum())
        if not np.isfinite(s) or s <= 0.0:
            return base_n.copy()
        return a / s
    if kind == 4:
        alphas = (0.5, 2.0, 10.0)
        alpha = float(alphas[int(rng.integers(len(alphas)))])
        return rng.dirichlet(np.full(n, alpha, dtype=np.float64)).astype(np.float64)
    if kind == 5:
        return _softmax(_x_from_a(base_n, 0.70, rng))
    if kind == 6:
        return _softmax(_x_from_a(base_n, 1.40, rng))
    if kind == 7:
        return _softmax(_x_from_a(base_n[::-1].copy(), 0.40, rng))
    return base_n.copy()


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget = max(1.0, float(budget_s))
    t0 = time.monotonic()
    headroom = min(50.0, max(1.0, 0.08 * budget))
    deadline = t0 + max(1.0, budget - headroom)

    rng = np.random.default_rng(seed)

    best_list = None
    best_arr = None
    best_val = float("inf")

    def consider(seq):
        nonlocal best_list, best_arr, best_val
        try:
            if time.monotonic() > deadline - 0.5:
                return
            arr = _clean(seq)
            if arr.size == 0:
                return
            lst = [float(x) for x in arr]
            val = float(evaluate_sequence(lst))
        except Exception:
            return
        if np.isfinite(val) and (best_list is None or val < best_val - 1e-15):
            best_val = float(val)
            best_arr = arr.astype(np.float64, copy=True)
            best_list = lst

    # Use the supplied parent construction directly, without copying its values into this program.
    try:
        parent = globals().get("height_sequence_1", None)
        if parent is not None:
            consider(parent)
    except Exception:
        pass

    # Small valid fallbacks in case the parent is unavailable.
    try:
        consider(np.array([1.0 / 3.0, 2.0 / 3.0], dtype=np.float64))
    except Exception:
        pass
    try:
        consider(np.array([0.1, 0.2, 0.2, 0.2, 0.3], dtype=np.float64))
    except Exception:
        pass

    queue = [
        (256, 0, 25),
        (256, 2, 25),
        (512, 0, 110),
        (512, 1, 80),
        (512, 2, 80),
        (512, 3, 70),
        (512, 4, 70),
        (512, 5, 70),
        (512, 2, 70),
        (1024, 0, 120),
        (1024, 1, 90),
        (1024, 2, 90),
        (1024, 3, 80),
        (2048, 0, 120),
        (2048, 1, 80),
        (1024, 6, 70),
        (512, 6, 60),
    ]

    for n, kind, alloc in queue:
        now = time.monotonic()
        if now > deadline - 22.0:
            break
        rem = deadline - now
        if rem < 18.0:
            break
        alloc = min(float(alloc), max(15.0, rem - 12.0))
        if alloc < 12.0:
            continue
        limit = now + alloc
        if limit > deadline - 10.0:
            limit = deadline - 10.0
        try:
            start = _make_start(kind, int(n), best_arr, rng)
            x0 = _x_from_a(start, 0.0, rng)
            a_opt, _ = _refine(x0, int(n), limit, rng)
            consider(a_opt)
        except Exception:
            pass

    # If time remains, run a few focused polish jobs on the current best shape.
    extra = 0
    while time.monotonic() < deadline - 40.0 and extra < 4:
        if best_arr is not None and best_arr.size > 0:
            n = int(best_arr.size)
        else:
            n = 512
        if extra == 1:
            n = 1024
        elif extra == 2:
            n = 2048
        elif extra == 3:
            n = 512
        if n > 2048:
            n = 2048
        kind = [1, 2, 5, 7][extra % 4]
        alloc = min(120.0, max(30.0, deadline - time.monotonic() - 20.0))
        if alloc < 15.0:
            break
        limit = time.monotonic() + alloc
        if limit > deadline - 12.0:
            limit = deadline - 12.0
        try:
            start = _make_start(kind, n, best_arr, rng)
            x0 = _x_from_a(start, 0.0, rng)
            a_opt, _ = _refine(x0, n, limit, rng)
            consider(a_opt)
        except Exception:
            break
        extra += 1

    # Final safe validation and return.
    try:
        if best_list is None:
            raise ValueError("no candidate")
        if len(best_list) == 0:
            raise ValueError("empty candidate")
        val = float(evaluate_sequence(best_list))
        if not np.isfinite(val):
            raise ValueError("invalid score")
    except Exception:
        try:
            fallback = _clean(np.array([1.0 / 3.0, 2.0 / 3.0], dtype=np.float64)).tolist()
            if np.isfinite(float(evaluate_sequence(fallback))):
                best_list = fallback
            else:
                fallback = [1.0]
                best_list = fallback if np.isfinite(float(evaluate_sequence(fallback))) else [1.0]
        except Exception:
            best_list = [1.0]

    return [float(x) for x in best_list]
