import numpy as np
import time
from scipy.optimize import linprog


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _autocorr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    need = 2 * n - 1
    if need <= 2048:
        return np.convolve(a, a)
    L = 1 << (need - 1).bit_length()
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:need]


def _cross(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    need = a.size + b.size - 1
    if need <= 2048:
        return np.convolve(a, b)
    L = 1 << (need - 1).bit_length()
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:need]


def _score_and_c(a, c=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf, None
    if c is None:
        c = _autocorr(a)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf, c
    if c.size == 0:
        return np.inf, c
    mx = float(np.max(c))
    if not np.isfinite(mx):
        mx = 0.0
    if mx < 0.0:
        mx = 0.0
    return float(2.0 * n * mx / (s * s)), c


def _score(a):
    sc, _ = _score_and_c(a)
    return sc


def _upsample(a, m):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = int(m)
    if m <= 0:
        return np.ones(1, dtype=np.float64)
    if a.size == 0:
        return np.ones(m, dtype=np.float64) / float(m)
    if a.size == m:
        out = a.copy()
    elif a.size == 1:
        out = np.full(m, float(a[0]), dtype=np.float64)
    else:
        x0 = np.linspace(0.0, 1.0, a.size)
        x1 = np.linspace(0.0, 1.0, m)
        out = np.interp(x1, x0, a)
    return _normalize(out)


def _smooth(a, width=3):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 2:
        return a.copy()
    w = int(min(max(1, width), n))
    if w <= 1:
        return a.copy()
    k = np.ones(w, dtype=np.float64) / float(w)
    out = np.convolve(a, k, mode="same")
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 0.0 or not np.isfinite(s):
        return a.copy() / max(1e-300, float(a.sum()))
    return out / s


def _make_inits(L, rng, refs):

    def add(x):
        try:
            y = _normalize(x)

        add(up[::-1])

def _build_lp_matrix(a, active):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    m = len(active)
    A = np.zeros((m, n + 1), dtype=np.float64)
    for i, k in enumerate(active):
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            A[i, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
        A[i, n] = -1.0
    return A

def _weighted_direction(a, c, M, top_count=16, beta=40.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return None
    if not np.isfinite(M) or M <= 1e-300:
        return _normalize(a)
    if c.size == 0:
        return _normalize(1.0 / (a + 1e-12))

    order = np.argsort(c)[::-1]
    take = min(max(2, top_count * 2), c.size)
    order = order[:take]

    z = beta * (c[order] - M)
    z = np.clip(z, -700.0, 0.0)
    w = np.exp(z)
    ws = float(w.sum())
    if not np.isfinite(ws) or ws <= 0.0:
        w = np.ones(take, dtype=np.float64) / float(take)
    else:
        w = w / ws

    h = np.zeros(n, dtype=np.float64)
    for lag, wt in zip(order, w):
        if wt <= 1e-12:
            continue
        k = int(lag)
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            h[j_lo:j_hi + 1] += float(wt) * a[k - j_hi:k - j_lo + 1][::-1]

    if float(np.max(h)) <= 1e-15:
        return _normalize(1.0 / (a + 1e-12))

    Aeq = np.zeros((1, n), dtype=np.float64)
    Aeq[0, :] = 1.0
    bounds = [(0.0, None)] * n

    try:
        res = linprog(h, A_eq=Aeq, b_eq=[1.0], bounds=bounds, method="highs")
    except Exception:
        try:
            res = linprog(h, A_eq=Aeq, b_eq=[1.0], bounds=bounds)
        except Exception:
            return None

    if res is not None and getattr(res, "success", False):
        d = np.asarray(res.x, dtype=np.float64).ravel()
        d = np.where(np.isfinite(d), d, 0.0)
        d = np.maximum(d, 0.0)
        s = float(d.sum())
        if np.isfinite(s) and s > 1e-300:
            return d / s
    return None


def _random_polish(a, rng, trials=2):
    a = _normalize(a)
    n = a.size
    best = a.copy()
    bs = _score(best)
    if not np.isfinite(bs):
        best = np.ones(n, dtype=np.float64) / float(n)
        bs = _score(best)

    for _ in range(max(1, int(trials))):
        cand = best.copy()
        nb = int(rng.integers(1, 4))
        lo = max(2, n // 300)
        hi = max(lo + 1, min(n, max(4, n // 20)))
        for _i in range(nb):
            w = int(rng.integers(lo, hi + 1))
            w = min(max(1, w), n)
            start = int(rng.integers(0, max(1, n - w + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start:start + w] *= factor

        if n > 16 and rng.random() < 0.35:
            off = int(rng.integers(-max(1, n // 100), max(1, n // 100) + 1))
            cand = 0.7 * cand + 0.3 * np.roll(cand, off)

        cand = _normalize(cand)
        sc = _score(cand)
        if np.isfinite(sc) and sc < bs - 1e-12:
            best = cand
            bs = sc

    return best, bs


def _optimize_stage(a, end_time, rng):
    a = _normalize(a)
    n = a.size
    cur_score, cur_c = _score_and_c(a)
    if not np.isfinite(cur_score):
        a = np.ones(n, dtype=np.float64) / float(n)
        cur_score, cur_c = _score_and_c(a)

    local_best = a.copy()
    local_score = cur_score
    no_improve = 0
    it = 0

    alphas = [1.0, 0.7, 0.4, 0.2, 0.1] if n <= 1024 else [1.0, 0.6, 0.3, 0.15]
    hard_end = float(end_time) - 2.0

    while time.monotonic() < hard_end - 2.0:
        it += 1

        if not np.isfinite(cur_score):
            a = local_best.copy()
            cur_score, cur_c = _score_and_c(a)

        if (
            np.isfinite(local_score)
            and cur_score > local_score + 1e-6
            and no_improve > 4
        ):
            a = local_best.copy()
            cur_score = local_score
            cur_c = _autocorr(a)
            no_improve = 0

        M = float(np.max(cur_c)) if cur_c.size else np.inf
        d_fw, T_fw, _ = _fw_direction(a, cur_c, M, time_limit=hard_end)

        dirs = []
        if d_fw is not None and np.all(np.isfinite(d_fw)):
            dirs.append(d_fw)
            if n >= 48:
                dirs.append(_smooth(d_fw, 5 if n > 2000 else 3))

        if no_improve > 2 or it % 4 == 0:
            dw = _weighted_direction(a, cur_c, M, top_count=16 if n > 2000 else 24, beta=40.0)
            if dw is not None and np.all(np.isfinite(dw)):
                dirs.append(dw)
                if n >= 48:
                    dirs.append(_smooth(dw, 3))

        if no_improve > 8:
            dirs.append(_normalize(np.exp(rng.normal(0.0, 0.6, size=n))))

        dirs = dirs[:4]
        improved = False

        for e in dirs:
            e = _normalize(e)
            if e.size != n or not np.all(np.isfinite(e)):
                continue
            if n > 0 and float(np.dot(e, a)) > 0.9999999:
                continue

            for alpha in alphas:
                cand = (1.0 - alpha) * a + alpha * e
                cand = np.where(np.isfinite(cand), cand, 0.0)
                cand = np.maximum(cand, 0.0)
                sc, cand_c = _score_and_c(cand)
                if np.isfinite(sc) and sc < cur_score - 1e-12:
                    a = cand
                    cur_score = sc
                    cur_c = cand_c
                    if sc < local_score - 1e-12:
                        local_best = a.copy()
                        local_score = sc
                    improved = True
                    break
            if improved:
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if time.monotonic() < hard_end - 10 and no_improve > 1:
                pol, ps = _random_polish(local_best, rng, trials=1)
                if np.isfinite(ps) and ps < local_score - 1e-12:
                    local_best = pol
                    local_score = ps
                    if cur_score > local_score + 1e-12:
                        a = local_best.copy()
                        cur_score = local_score
                        cur_c = _autocorr(a)
                        no_improve = 0
            if no_improve >= 25:
                break

        if it % 9 == 0 and time.monotonic() < hard_end - 15:
            pol, ps = _random_polish(local_best, rng, trials=1)
            if np.isfinite(ps) and ps < local_score - 1e-12:
                local_best = pol
                local_score = ps
                if cur_score > local_score + 1e-12:
                    a = local_best.copy()
                    cur_score = local_score
                    cur_c = _autocorr(a)

    return a, cur_score, local_best, local_score


def propose_candidate(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1300.0

    final_polish_budget = max(30.0, 0.05 * budget)
    eval_budget = max(15.0, 0.02 * budget)
    hard_deadline = start + max(5.0, budget - 5.0)
    final_stop = start + max(10.0, budget - eval_budget)
    stage_stop = start + max(10.0, budget - final_polish_budget - eval_budget)
    stage_stop = min(stage_stop, final_stop - 10.0)
    if stage_stop <= start + 5.0:
        stage_stop = final_stop - 5.0
    if final_stop <= stage_stop:
        final_stop = hard_deadline
        stage_stop = hard_deadline - 5.0

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if p.size > 0 and s > 1e-12:
            parent = p / s
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(127, dtype=np.float64) / 127.0

    best_seq = parent.copy()
    best_score = _score(best_seq)

    if final_stop - start < 12.0:
        return [0.5, 0.1, 0.4]

    MAXL = 4095
    L = int(parent.size)
    if L > MAXL:
        parent = _upsample(parent, MAXL)
        L = MAXL
        best_seq = parent.copy()
        best_score = _score(parent)

    lengths = []
    curL = L
    while True:
        lengths.append(curL)
        if curL >= MAXL:
            break
        nxt = min(curL * 2 + 1, MAXL)
        if nxt <= curL:
            break
        curL = nxt

    current = parent.copy()
    weights = [float(Lg ** 0.65) for Lg in lengths]
    total_w = float(sum(weights))
    total_stage = max(1.0, stage_stop - start)
    used = 0.0

    for idx, Lg in enumerate(lengths):
        if time.monotonic() >= stage_stop - 10.0:
            break
        stage_time = total_stage * (weights[idx] / total_w)
        used += stage_time
        stage_end = min(stage_stop - 10.0, time.monotonic() + stage_time)
        if stage_end <= time.monotonic() + 2.0:
            continue

        refs = [best_seq, current, parent]
        inits = _make_inits(Lg, rng, refs)
        fallback = parent if parent.size == Lg else _upsample(parent, Lg)
        a0, sc0 = _best_initial(inits, fallback)

        try:
            a_out, sc_out, local_best, local_score = _optimize_stage(a0, stage_end, rng)
        except Exception:
            a_out, sc_out, local_best, local_score = a0, sc0, a0, sc0

        for cand, sc in ((local_best, local_score), (a_out, sc_out)):
            if cand is None:
                continue
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand.copy()

        cur_cands = []
        if local_best is not None and local_best.size == Lg:
            cur_cands.append(local_best)
        if a_out is not None and a_out.size == Lg:
            cur_cands.append(a_out)
        cur_cands.append(_upsample(best_seq, Lg))

        best_cur = None
        best_cur_sc = np.inf
        for cc in cur_cands:
            try:
                sc = _score(cc)
            except Exception:
                sc = np.inf
            if np.isfinite(sc) and sc < best_cur_sc:
                best_cur = cc
                best_cur_sc = sc
        if best_cur is not None:
            current = best_cur

    if final_stop - time.monotonic() > 25.0:
        try:
            a_out, sc_out, local_best, local_score = _optimize_stage(
                best_seq.copy(),
                final_stop - 5.0,
                rng,
            )
        except Exception:
            local_best, local_score = best_seq, best_score
            a_out, sc_out = best_seq, best_score

        for cand, sc in ((local_best, local_score), (a_out, sc_out)):
            if cand is None:
                continue
            try:
                sc = _score(cand)
            except Exception:
                sc = np.inf
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand.copy()

    if hard_deadline - time.monotonic() > 12.0:
        try:
            pol, ps = _random_polish(best_seq, rng, trials=2 if best_seq.size < 2000 else 1)
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = pol
        except Exception:
            pass

    final_candidates = []
    if best_seq is not None and best_seq.size > 0:
        final_candidates.append(best_seq)
    if parent is not None and parent.size > 0:
        final_candidates.append(parent)
    final_candidates.append(np.array([0.5, 0.1, 0.4], dtype=np.float64))

    final_list = None
    final_score = np.inf

    for cand in final_candidates:
        if time.monotonic() > hard_deadline - 1.5:
            break
        try:
            lst = [float(x) for x in np.asarray(cand, dtype=np.float64).ravel()]
            if not lst or any(not np.isfinite(x) for x in lst):
                continue
            sc = float(evaluate_sequence(lst))
        except Exception:
            try:
                sc = float(_score(np.asarray(cand, dtype=np.float64)))
                lst = [float(x) for x in np.asarray(cand, dtype=np.float64).ravel()]
            except Exception:
                sc = np.inf
                lst = None
            if lst is not None and (not lst or any(not np.isfinite(x) for x in lst)):
                sc = np.inf
                lst = None

        if lst is not None and np.isfinite(sc) and sc < final_score:
            final_score = sc
            final_list = lst

    if final_list is None:
        final_list = [0.5, 0.1, 0.4]

    return final_list
