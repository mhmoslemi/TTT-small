    """Solve the LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for top lags"""
    from scipy.optimize import linprog
    
    n = len(a)
    conv = np.convolve(a, a)
    k_top = max(10, min(n_active, 2*n-1))
    top_idx = np.argpartition(-conv, k_top)[:k_top]  # top k_top lags
    
    # Build constraint matrix: A_eq @ b <= M for each active lag
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((k_top, n))
    for ii, k in enumerate(top_idx):
        for j in range(max(0, k-n+1), min(k+1, n)):
            A[ii, j] = a[k-j]
    
    # Maximize sum(b) = -minimize -sum(b)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(k_top, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        b = res.x
        return b / b.sum()  # normalize
    
    return None
