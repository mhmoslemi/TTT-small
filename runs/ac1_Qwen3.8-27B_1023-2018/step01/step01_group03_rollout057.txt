import numpy as np
import time
from scipy.optimize import linprog

def _normalize(q):
    q = np.asarray(q, dtype=np.float64)
    q = np.where(np.isfinite(q) & (q >= 0), q, 0.0)
    s = q.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(q.shape, 1.0/q.size)
    return q / s

def _score(q):
    q = np.asarray(q, dtype=np.float64)
    n = q.size
    if n == 0:
        return float('inf')
    b = np.convolve(q, q)
    s = q.sum()
    if s < 0.01:
        return float('inf')
    val = 2.0 * n * b.max() / (s*s)
    return val if np.isfinite(val) else float('inf')

def _lp_response(q, n):
    """Find g maximizing sum(g) s.t. (q*g)[k] <= M for all k, g >= 0"""
    b = np.convolve(q, q)
    M = b.max()
    if M <= 0:
        return None
    
    # Build constraint matrix: (q*g)[k] <= M
    # (q*g)[k] = sum_i q[i]*g[k-i] (where k-i is valid)
    # This is a Toeplitz-like matrix
    # A[k, i] = q[i] if 0 <= k-i < n, else 0
    # Wait, (q*g)[k] = sum_i q[i] * g[k-i]
    # For this to be valid, we need 0 <= k-i < n, so i is in [k-n+1, k] ∩ [0, n-1]
    
    # A is (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k-n+1)
        hi = min(k, n-1)
        for i in range(lo, hi+1):
            A[k, i] = q[i]
    
    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, 1000)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.clip(res.x, 0, 1000)
            if g.sum() > 0:
                return g / g.sum()
    except:
        pass
    return None

    """Optimize coordinate i of a, given all others fixed."""
    n = a.size
    ai = a[i]
    
    # Compute the "base" b values (without a[i]'s contribution)
    # b[k] = sum_j a[j]*a[k-j]
    # Contribution of a[i] to b[k]:
    #   j=i: a[i]*a[k-i] (if 0<=k-i<n)
    #   j=k-i: a[k-i]*a[i] (if 0<=k-i<n and k-i != i)

    b = np.convolve(a, a)
