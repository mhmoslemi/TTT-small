import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    sum_h = 35.0
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)

        objective,
        bounds,
