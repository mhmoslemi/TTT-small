from scipy.optimize import differential_evolution, Bounds, minimize

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:

        h_initial = (h_initial / np.sum(h_initial) * 35.0)

        h_shifted = np.roll(h_initial, shift)
        shifted_candidates.append(h_shifted)

    # Differential evolution on shifted candidates
    for candidate in shifted_candidates:
