import numpy as np
from scipy.optimize import Bounds
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(70)
        initial_h = (initial_h / np.sum(initial_h) * 35.0)
    
    dx = 2.0 / 70
    n = 70
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    def evaluate(h):
        return objective(h)
    
    pop_size = 50
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_vel = 0.5
    max_iter = int(budget_s / 10)  # ~100 iterations
    
    # Initialize particles
    particles = np.array([initial_h + np.random.normal(0, 0.1, n) for _ in range(pop_size)])
    particles = np.vstack([project_h(p) for p in particles])
    velocities = np.random.uniform(-max_vel, max_vel, (pop_size, n))
    
    pbest = particles.copy()
    pbest_vals = np.array([evaluate(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_vals)]
    gbest_val = np.min(pbest_vals)
    
    for _ in range(max_iter):
        for i in range(pop_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            particles[i] += velocities[i]
            particles[i] = project_h(particles[i])
        
        current_vals = np.array([evaluate(p) for p in particles])
        improve_mask = current_vals < pbest_vals
        pbest[improve_mask] = particles[improve_mask]
        pbest_vals[improve_mask] = current_vals[improve_mask]
        
        if np.min(pbest_vals) < gbest_val:
            gbest = pbest[np.argmin(pbest_vals)]
            gbest_val = np.min(pbest_vals)
    
    return gbest, gbest_val, n
