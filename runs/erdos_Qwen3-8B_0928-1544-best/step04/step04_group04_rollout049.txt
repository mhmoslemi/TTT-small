import numpy as np
from scipy.optimize import differential_evolution, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(70)
        initial_h = (initial_h / np.sum(initial_h) * 35.0)
    
    n = 70
    dx = 2.0 / n
    
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    def objective(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return float('inf')
        scale = 35.0 / sum_h
        h_scaled = h_clip * scale
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Run differential evolution
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=500,
        polish=True,
        workers=-1,
        vectorized=True
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Final refinement with SLSQP
    from scipy.optimize import minimize, Bounds
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds_opt = Bounds([0.0]*n, [1.0]*n)
    
    result_final = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds_opt,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 150, 'ftol': 1e-9}
    )
    
    if result_final.success and result_final.fun < best_c5:
        best_c5 = result_final.fun
        best_h = result_final.x
    
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return best_h, best_c5, n
    
    return best_h, best_c5, n
