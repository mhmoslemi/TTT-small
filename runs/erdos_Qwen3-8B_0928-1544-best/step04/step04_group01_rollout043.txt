    target_sum_fine = n_points_fine / 2.0
    num_particles_fine = 50
    max_iter_fine = 100
    w_fine, c1_fine, c2_fine = 0.9, 2.0, 2.0
    inertia_decay_fine, c1_decay_fine, c2_decay_fine = 0.95, 0.98, 1.02
    
    particles_fine = np.random.uniform(0, 1, (num_particles_fine, n_points_fine))
    velocities_fine = np.random.uniform(-0.05, 0.05, (num_particles_fine, n_points_fine))
    
    for i in range(num_particles_fine):
        particles_fine[i] = generate_initial(n_points_fine)
    
    pbest_fine = particles_fine.copy()
    pbest_vals_fine = np.array([evaluate_c5(p) for p in particles_fine])
    gbest_fine = particles_fine[np.argmin(pbest_vals_fine)]
    gbest_val_fine = np.min(pbest_vals_fine)
    
    start_time_fine = time.time()
    for iter_fine in range(max_iter_fine):
        for i in range(num_particles_fine):
            r1, r2 = np.random.rand(2)
            velocities_fine[i] = w_fine * velocities_fine[i] + c1_fine * r1 * (pbest_fine[i] - particles_fine[i]) + c2_fine * r2 * (gbest_fine - particles_fine[i])
            particles_fine[i] += velocities_fine[i]
            particles_fine[i] = np.clip(particles_fine[i], 0.0, 1.0)
            sum_p_fine = np.sum(particles_fine[i])
            if sum_p_fine == 0.0:
                particles_fine[i] = np.full(n_points_fine, 0.5)
            else:
                particles_fine[i] = particles_fine[i] * (target_sum_fine / sum_p_fine)
            particles_fine[i] = np.clip(particles_fine[i], 0.0, 1.0)
        
        current_vals_fine = np.array([evaluate_c5(p) for p in particles_fine])
        for i in range(num_particles_fine):
            if current_vals_fine[i] < pbest_vals_fine[i]:
                pbest_fine[i] = particles_fine[i].copy()
                pbest_vals_fine[i] = current_vals_fine[i]
        
        if np.min(current_vals_fine) < gbest_val_fine:
            gbest_fine = particles_fine[np.argmin(current_vals_fine)]
            gbest_val_fine = np.min(current_vals_fine)
        
        w_fine *= inertia_decay_fine
        c1_fine *= c1_decay_fine
        c2_fine *= c2_decay_fine
        
        if time.time() - start_time_fine > budget_s * 0.4:
            break
    
    final_h = gbest_fine if gbest_val_fine < gbest_val else gbest
    final_c5 = gbest_val_fine if gbest_val_fine < gbest_val else gbest_val
    return final_h, final_c5, n_points_fine
