import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = 35
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        T = 1e-3
        soft_max = np.log(np.sum(np.exp(corr / T))) / T
        return soft_max * (2.0 / n_points)

    bounds = [(0.0, 1.0)] * n_points

    def constraint_sum(h):
        return np.sum(h) - ones_count

    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    time_used = 0
    max_iter = int(budget_s // 4)

    for _ in range(max_iter):
        h_perturbed = initial_h + np.random.normal(0, 0.1, n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))

        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )

        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

        initial_h = result.x
        time_used += 4

        if time_used >= budget_s:
            break

    return best_h, best_c5, n_points
