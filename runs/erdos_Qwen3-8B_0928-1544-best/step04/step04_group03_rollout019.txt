        if sum_h == 0:

            h_candidate = h_candidate * (35.0 / sum_h)
        # Compute current and new c5
        corr_current = np.correlate(h, 1 - h, mode='full')
        old_c5 = np.max(corr_current) * dx
        corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        new_c5 = np.max(corr_candidate) * dx
        # Acceptance
        if np.random.rand() < np.exp((old_c5 - new_c5)/T):
            h = h_candidate.copy()
        # Update best
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_candidate.copy()
        # Cool down
        T *= alpha
        iter_count += 1
    # Now perform refinement with SLSQP
    # Define objective function
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
