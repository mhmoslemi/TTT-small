import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    required_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        initial_candidates = [initial_h]
        for _ in range(4):

            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)

            initial_candidates.append(h_perturbed)

        initial_h = np.full(n_points, 0.5)
        initial_candidates = [initial_h]

    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_candidates[0].copy()

            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,

        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
