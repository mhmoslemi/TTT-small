import numpy as np
from scipy.optimize import Bounds
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n = 70
    dx = 2.0 / n

    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(n)
        initial_h = project_h(initial_h)

    population_size = 100
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n)
        population.append(project_h(h))

    def tournament_selection(pop, fitness):
        indices = np.random.choice(len(pop), size=5, replace=False)
        candidates = [pop[i] for i in indices]
        best = min(candidates, key=objective)
        return best

    best_h = initial_h
    best_c5 = objective(best_h)

    tournament_size = 5
    crossover_rate = 0.25
    mutation_rate = 0.05
    generations = 300

    for gen in range(generations):
        fitness = [objective(h) for h in population]
        new_population = []
        for _ in range(population_size // 2):
            p1 = tournament_selection(population, fitness)
            p2 = tournament_selection(population, fitness)
            if np.random.rand() < crossover_rate:
                point = np.random.randint(1, n)
                child1 = np.concatenate((p1[:point], p2[point:]))
                child2 = np.concatenate((p2[:point], p1[point:]))
            else:
                child1, child2 = p1.copy(), p2.copy()

            if np.random.rand() < mutation_rate:
                child1 += np.random.normal(0, 0.03, size=n)
                child1 = project_h(child1)
            if np.random.rand() < mutation_rate:
                child2 += np.random.normal(0, 0.03, size=n)
                child2 = project_h(child2)
            new_population.extend([child1, child2])
        
        new_population.append(best_h)
        new_population = sorted(new_population, key=objective)[:int(population_size * 0.1)]
        while len(new_population) < population_size:
            h = np.random.rand(n)
            new_population.append(project_h(h))
        population = new_population

    fitness = [objective(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_h = project_h(best_h)
    best_c5 = fitness[best_idx]

    return best_h, best_c5, n
