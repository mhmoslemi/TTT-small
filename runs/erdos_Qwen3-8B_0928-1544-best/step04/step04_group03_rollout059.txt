import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(70)
        ones = 35
        interval = 70 / ones
        for i in range(ones):
            pos = int(i * interval)
            h_initial[pos] = 1.0
        if np.sum(h_initial) != ones:
            diff = ones - np.sum(h_initial)
            zeros = np.where(h_initial == 0)[0]
            h_initial[zeros[:diff]] = 1.0
    dx = 2.0 / 70
    best_h = h_initial
    best_c5 = float('inf')
    t0 = time.time()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - 35.0

    bounds = Bounds([0.0]*70, [1.0]*70)

    # Simulated Annealing
    T = 1.0
    cooling = 0.999
    max_sa_iter = 1000
    sa_iter = 0
    sa_candidates = []

    while sa_iter < max_sa_iter and time.time() - t0 < budget_s - 500:
        sa_iter += 1
        perturbation = np.random.normal(0, T, 70)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (35.0 / np.sum(h_perturbed))
        current_c5 = objective(h_perturbed)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_perturbed.copy()
        sa_candidates.append((current_c5, h_perturbed))
        if len(sa_candidates) > 10:
            sa_candidates.pop()
        T *= cooling
        h_initial = best_h

    # Refinement with SLSQP
    top_candidates = [h for (c, h) in sa_candidates]
    for h_ref in top_candidates:
        result = minimize(
            fun=objective,
            x0=h_ref,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-8,
            options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, 70
