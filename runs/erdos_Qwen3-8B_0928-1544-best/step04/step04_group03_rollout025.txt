from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:

        for i in range(ones):
            pos = int(i * interval)
            h_initial[pos] = 1.0
        if np.sum(h_initial) != ones:
            diff = ones - np.sum(h_initial)
            zeros = np.where(h_initial == 0)[0]
            h_initial[zeros[:diff]] = 1.0
    best_c5 = float('inf')
    best_h = h_initial.copy()
    start_time = time.time()
    T = 1.0
    alpha = 0.999
    sa_iterations = 1000

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for _ in range(sa_iterations):
        if time.time() - start_time > budget_s - 500:
            break

        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        current_sum = np.sum(h_perturbed)
        if current_sum == 0:
            continue
        h_perturbed *= (target_sum / current_sum)
        new_c5 = compute_c5(h_perturbed)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_perturbed.copy()
