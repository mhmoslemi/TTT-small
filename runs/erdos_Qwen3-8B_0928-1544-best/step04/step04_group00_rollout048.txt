import numpy as np
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    desired_sum = n_points / 2
    
    def generate_sine():
        x = np.linspace(0, 2, n_points)
        return 0.5 + 0.5 * np.sin(2 * np.pi * x)
    
    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return np.full_like(h, 1.0 / n_points)
        return np.clip(h * (desired_sum / s), 0, 1)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    # Initialize population with sine patterns
    population = [normalize(generate_sine() + np.random.normal(0, 0.1, n_points)) for _ in range(30)]
    best_h = population[0]
    best_c5 = evaluate(population[0])
    time_used = 0
    max_gen = int(budget_s / 5)
    
    for _ in range(max_gen):
        fitness = np.array([evaluate(h) for h in population])
        top_idx = np.argsort(fitness)[:15]
        new_pop = []
        for idx in top_idx:
            h = population[idx].copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = normalize(h)
            new_pop.append(h)
        # Replace worst half with new candidates
        for i in range(len(population)):
            if i < len(new_pop):
                population[i] = new_pop[i]
            else:
                population[i] = normalize(generate_sine() + np.random.normal(0, 0.1, n_points))
        # Track best solution
        current = [evaluate(h) for h in population]
        best_idx = np.argmin(current)
        if current[best_idx] < best_c5:
            best_c5 = current[best_idx]
            best_h = population[best_idx]
        time_used += 5
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
