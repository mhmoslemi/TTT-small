import numpy as np
import random
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(70)
        initial_h = (initial_h / np.sum(initial_h) * 35.0)
    dx = 2.0 / 70
    n = 70
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    population = [initial_h.copy()]
    for _ in range(49):
        h = np.random.rand(n)
        h = (h / np.sum(h) * 35.0)
        population.append(h)
    
    best_h = initial_h
    best_c5 = objective(initial_h)
    
    def crossover(parent1, parent2):
        child1 = np.copy(parent1)
        child2 = np.copy(parent2)
        for i in range(n):
            if np.random.rand() < 0.7:
                child1[i] = parent2[i]
                child2[i] = parent1[i]
        return child1, child2
    
    generations = 100
    population_size = 50
    mutation_rate = 0.05
    
    for gen in range(generations):
        fitness = [objective(ind) for ind in population]
        parents = []
        for _ in range(population_size):
            tournament = random.sample(list(enumerate(fitness)), 3)
            winner_idx = min(tournament, key=lambda x: x[1])[0]
            parents.append(population[winner_idx])
        
        offspring = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                p1, p2 = parents[i], parents[i+1]
                c1, c2 = crossover(p1, p2)
                offspring.append(c1)
                offspring.append(c2)
            else:
                offspring.append(parents[i])
        
        for j in range(len(offspring)):
            for k in range(n):
                if np.random.rand() < mutation_rate:
                    offspring[j][k] += np.random.normal(0, 0.05)
            offspring[j] = project_h(offspring[j])
        
        population = offspring
        current_fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < best_c5:
            best_c5 = current_fitness[best_idx]
            best_h = population[best_idx]
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - 35}]
    bounds = Bounds([0.0]*n, [1.0]*n)
    result = minimize(fun=objective, x0=best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
    if result.success:
        best_c5 = result.fun
        best_h = result.x
    return best_h, best_c5, n
