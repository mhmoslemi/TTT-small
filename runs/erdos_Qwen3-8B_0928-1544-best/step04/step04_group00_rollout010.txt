import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    ones_count = n_points // 2
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    initial_h[::2] = 1.0  # create even-spaced initial guess

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h):
        return np.sum(h) - ones_count

    bounds = [(0.0, 1.0)] * n_points
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Perturb initial guess
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturbed = initial_h + perturb
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
        
        # Optimization
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

        initial_h = result.x
        if time.time() - start_time > budget_s:
            break

    return best_h, best_c5, n_points
