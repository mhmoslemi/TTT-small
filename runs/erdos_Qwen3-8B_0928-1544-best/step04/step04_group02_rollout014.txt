import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.full(70, 0.5)
    h_initial = h_initial * (35.0 / np.sum(h_initial))
    dx = 2.0 / 70
    best_h = h_initial.copy()
    best_c5 = float('inf')
    start_time = time.time()
    max_gen = 1000
    population_size = 20
    population = [h_initial]
    for _ in range(population_size - 1):
        h_mut = h_initial + np.random.normal(0, 0.1, 70)
        h_mut = np.clip(h_mut, 0, 1)
        sum_mut = np.sum(h_mut)
        if sum_mut != 0:
            h_mut = h_mut * (35.0 / sum_mut)
        population.append(h_mut)
    
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        current_c5s = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            current_c5s.append(c5)
        sorted_indices = np.argsort(current_c5s)
        new_pop = [population[i] for i in sorted_indices[:5]]
        for _ in range(population_size - 5):
            idx1, idx2 = np.random.choice(len(sorted_indices), 2, replace=False)
            parent1 = population[sorted_indices[idx1]]
            parent2 = population[sorted_indices[idx2]]
            child = (parent1 + parent2) / 2
            child = np.clip(child, 0, 1)
            sum_c = np.sum(child)
            if sum_c == 0:
                child = np.zeros(70)
            else:
                child = child * (35.0 / sum_c)
            sigma = max(0.01, 0.1 - 0.05 * gen / max_gen)
            perturb = np.random.normal(0, sigma, 70)
            child_mut = child + perturb
            child_mut = np.clip(child_mut, 0, 1)
            sum_mut = np.sum(child_mut)
            if sum_mut != 0:
                child_mut = child_mut * (35.0 / sum_mut)
            new_pop.append(child_mut)
        population = new_pop
        top_idx = sorted_indices[:5]
        for i in top_idx:
            h = population[i]
            def obj_func(x):
                h_vec = x.copy()
                corr = np.correlate(h_vec, 1 - h_vec, mode='full')
                return np.max(corr) * dx
            def cons(x):
                return np.sum(x) - 35.0
            result = minimize(obj_func, h, method='SLSQP', bounds=Bounds([0]*70, [1]*70), constraints={'type': 'eq', 'fun': cons}, tol=1e-8, options={'maxiter': 300})
            if result.success:
                population[i] = result.x
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5, 70
