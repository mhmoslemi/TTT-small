import numpy as np
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(n_points)
        initial_h = (initial_h / np.sum(initial_h) * (n_points // 2))
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    def project_h(h):
        clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(clipped)
        if sum_clipped == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / sum_clipped
        return clipped * scale
    pop_size = 50
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_vel = 0.5
    max_iter = 100
    particles = np.zeros((pop_size, n_points))
    velocities = np.zeros((pop_size, n_points))
    for i in range(pop_size):
        if i == 0:
            particles[i] = initial_h.copy()
        else:
            temp = np.random.rand(n_points)
            temp = (temp / np.sum(temp) * (n_points // 2))
            particles[i] = temp
        velocities[i] = np.random.uniform(-max_vel, max_vel, n_points)
    pbest = particles.copy()
    pbest_obj = np.zeros(pop_size)
    for i in range(pop_size):
        pbest_obj[i] = obj(particles[i])
    gbest = particles[0].copy()
    gbest_obj = pbest_obj[0]
    for _ in range(max_iter):
        for p in range(pop_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[p] = inertia * velocities[p] + cognitive * r1 * (pbest[p] - particles[p]) + social * r2 * (gbest - particles[p])
            velocities[p] = np.clip(velocities[p], -max_vel, max_vel)
            particles[p] += velocities[p]
            particles[p] = project_h(particles[p])
        for p in range(pop_size):
            curr_obj = obj(particles[p])
            if curr_obj < pbest_obj[p]:
                pbest[p] = particles[p].copy()
                pbest_obj[p] = curr_obj
            if curr_obj < gbest_obj:
                gbest = particles[p].copy()
                gbest_obj = curr_obj
    return gbest, gbest_obj, n_points
