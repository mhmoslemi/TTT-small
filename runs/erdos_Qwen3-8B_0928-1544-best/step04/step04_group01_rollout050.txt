import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = initial_h_values
    else:
        h = np.random.rand(70)
        h = h * (35.0 / np.sum(h))
        h = np.clip(h, 0, 1)

    particles = np.random.rand(swarm_size, n_points)
    particles *= (n_points / 2.0 / np.sum(particles, axis=1, keepdims=True))
    particles = np.clip(particles, 0, 1)
    velocities = np.random.rand(swarm_size, n_points) * 0.1
    p_best = particles.copy()
    g_best = particles[0].copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    max_iters = 100
    for _ in range(max_iters):
        if time.time() - start_time > budget_s * 0.6:
            break
        corr = np.correlate(particles, 1 - particles, mode='full')
        c5 = np.max(corr, axis=1) * dx
        for i in range(swarm_size):
            if c5[i] < best_c5:
                best_c5 = c5[i]
