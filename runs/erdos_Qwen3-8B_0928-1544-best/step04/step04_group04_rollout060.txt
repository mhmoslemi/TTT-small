import numpy as np
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = (initial_h / np.sum(initial_h) * (n_points / 2))
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros_like(h_clip)
        scale = (n_points / 2) / sum_h
        return h_clip * scale
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def fitness(h):
        return objective(h)
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(n_points):
            if np.random.rand() < 0.7:
                child[i] = parent2[i]
        return child
    
    def mutate(h):
        for i in range(n_points):
            if np.random.rand() < 0.05:
                h[i] += np.random.normal(0, 0.05)
        return h
    
    population = [initial_h.copy()]
    for _ in range(49):
        h = np.random.rand(n_points)
        h = (h / np.sum(h) * (n_points / 2))
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    
    for _ in range(100):
        fit = [fitness(ind) for ind in population]
        best_idx = np.argmin(fit)
        if fit[best_idx] < best_c5:
            best_c5 = fit[best_idx]
            best_h = population[best_idx].copy()
        
        new_pop = []
        for _ in range(n_points):
            t = random.sample(population, 3)
            best_t = min(t, key=fitness)
            new_pop.append(best_t)
        
        for i in range(0, len(new_pop), 2):
            if i + 1 < len(new_pop):
                p1 = new_pop[i]
                p2 = new_pop[i+1]
                child1 = crossover(p1, p2)
                child2 = crossover(p2, p1)
                child1 = mutate(child1)
                child2 = mutate(child2)
                new_pop[i] = project_h(child1)
                new_pop[i+1] = project_h(child2)
        
        population = new_pop
    
    top_idx = np.argsort([fitness(ind) for ind in population])[:5]
    top_h = [population[i].copy() for i in top_idx]
    
    final_h = best_h.copy()
    final_c5 = best_c5
    
    for h in top_h:
        def obj(x):
            return objective(x)
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bnds = [(0.0, 1.0) for _ in range(n_points)]
        
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons, tol=1e-9)
        if res.success:
            current_c5 = res.fun
            if current_c5 < final_c5:
                final_c5 = current_c5
                final_h = res.x
    
    return final_h, final_c5, n_points
