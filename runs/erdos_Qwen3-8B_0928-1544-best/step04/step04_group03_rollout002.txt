import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            h[2 * i] = 1.0
        if np.sum(h) != n_points // 2:
            diff = n_points // 2 - np.sum(h)
            zeros = np.where(h == 0)[0]
            h[zeros[:diff]] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    # Simulated Annealing
    T = 1.0
    alpha = 0.999
    sa_steps = 1000
    sa_best = h.copy()

    for _ in range(sa_steps):
        if time.time() - t0 > budget_s - 500:
            break
        noise = np.random.normal(0, T, n_points)
        h_perturbed = h + noise
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        sum_perturbed = np.sum(h_perturbed)
        if sum_perturbed > 0:
            h_perturbed = h_perturbed * (n_points // 2) / sum_perturbed
        curr_c5 = compute_c5(h_perturbed, dx)
        if curr_c5 < compute_c5(sa_best, dx):
            sa_best = h_perturbed.copy()
        elif np.random.rand() < np.exp((compute_c5(sa_best, dx) - curr_c5) / T):
            sa_best = h_perturbed.copy()
        T *= alpha

    h = sa_best.copy()
    best_c5 = compute_c5(h, dx)

    # Refinement with SLSQP
    def objective(x):
        return compute_c5(x, dx)
    
    def constraint(x):
        return np.sum(x) - n_points // 2
    
    bounds = np.zeros((n_points, 2))
    bounds[:, 0] = 0.0
    bounds[:, 1] = 1.0
    result = minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        tol=1e-10,
        options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    return best_h, best_c5, n_points
