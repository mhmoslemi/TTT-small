import numpy as np
from scipy.optimize import minimize, Bounds
import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n = 70
    dx = 2.0 / n
    target_sum = n / 2
    initial_h = np.zeros(n)
    
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(n)
        initial_h = (initial_h / np.sum(initial_h) * target_sum)
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = target_sum / sum_h
        return h_clip * scale
    
    h = project_h(initial_h)
    
    def objective(h_flat):
        h_reshaped = h_flat.reshape(-1)
        corr = np.correlate(h_reshaped, 1 - h_reshaped, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    bounds = Bounds([0.0]*n, [1.0]*n)
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for _ in range(100):
        perturbation = np.zeros(n)
        indices = np.random.choice(n, size=7, replace=False)
        perturbation[indices] += np.random.normal(0, 0.05, 7)
        h_perturbed = h + perturbation
        h_perturbed = project_h(h_perturbed)
        
        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=1e-8,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    final_result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 150, 'ftol': 1e-9}
    )
    
    if final_result.success and final_result.fun < best_c5:
        best_c5 = final_result.fun
        best_h = final_result.x
    
    return best_h, best_c5, n
