        constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - curr_n / 2}
        bounds = [(0.0, 1.0)] * curr_n
        
        result = minimize(
            fun=objective,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
            time_used += max_time_per_step
        
        if time_used >= budget_s:
            break
    
    return best_h, best_c5, n_points
