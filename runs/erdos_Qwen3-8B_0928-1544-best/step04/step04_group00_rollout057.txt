import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    population_size = 30
    max_iter = int(budget_s // 50)
    
    # Initial sinusoidal pattern
    base = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
    population = [base + np.random.normal(0, 0.1, n_points) for _ in range(population_size)]
    for i in range(population_size):
        h = population[i]
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        population[i] = h * scale
    
    def c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = population[0]
    best_c5 = c5(best_h)
    
    for _ in range(max_iter):
        scores = [c5(h) for h in population]
        idx = np.argsort(scores)
        top = idx[:population_size // 2]
        worst = idx[population_size // 2:]
        
        # Generate new candidates
        new_pop = []
        for i in top:
            h = population[i] + np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            new_pop.append(h * scale)
        
        # Replace worst with new
        population = [population[i] for i in worst] + new_pop
        
        # Check best
        for h in population:
            curr = c5(h)
            if curr < best_c5:
                best_c5 = curr
                best_h = h
    
    return best_h, best_c5, n_points
