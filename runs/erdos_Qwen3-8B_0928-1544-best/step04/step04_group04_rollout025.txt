import numpy as np
from scipy.optimize import minimize, Bounds
import random

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(70)
        h = (h / np.sum(h) * 35.0)
    
    dx = 2.0 / 70
    n = 70
    
    def objective(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_val):
        return np.sum(h_val) - 35.0
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    def project_h(h_val):
        h_clip = np.clip(h_val, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = 35.0 / sum_h
        return h_clip * scale
    
    best_c5 = float('inf')
    best_h = h.copy()
    current_h = project_h(h)
    current_c5 = objective(current_h)
    best_c5 = min(best_c5, current_c5)
    
    for _ in range(30):
        perturb_idx = np.random.choice(n, size=15, replace=False)
        noise = np.random.normal(0, 0.1, size=n)
        perturbed_h = current_h + noise
        perturbed_h = project_h(perturbed_h)
        
        result = minimize(
            fun=objective,
            x0=perturbed_h,
            method='COBYLA',
            bounds=bounds,
            constraints=constraints,
            tol=1e-8,
            options={'maxiter': 100}
        )
        
        if result.success:
            new_c5 = result.fun
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = result.x
                current_h = result.x
    
    final_result = minimize(
        fun=objective,
        x0=best_h,
        method='COBYLA',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 100}
    )
