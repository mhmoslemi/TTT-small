from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    ones_count = 35

    T = 1.0

    sigma = 0.2
    max_iter = int(budget_s // 5)
    
    population = [h_start.copy() for _ in range(pop_size)]
    for _ in range(pop_size):

        new_h = h_start + perturb
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = new_h * (ones_count / np.sum(new_h))
        population.append(new_h)
    
    population = list(set(population))[:pop_size]

    time_used = 0
    
    for iter in range(max_iter):
        population_sorted = sorted(population, key=lambda x: evaluate_erdos_solution(x))
        top_half = population_sorted[:pop_size // 2]
        
        new_population = []
        for candidate in top_half:
            perturb = np.random.normal(0, sigma, n_points)
            new_h = candidate + perturb
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = new_h * (ones_count / np.sum(new_h))
            
            def obj_func(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * (2.0 / n_points)
            
            result = minimize(obj_func, x0=new_h, method='Nelder-Mead', 
                              options={'maxiter': 100, 'ftol': 1e-9})
            new_h_opt = result.x
            new_h_opt = np.clip(new_h_opt, 0.0, 1.0)
            new_h_opt = new_h_opt * (ones_count / np.sum(new_h_opt))
            new_population.append(new_h_opt)
        
        lower_half = population_sorted[pop_size // 2:]
        population = new_population + lower_half[:pop_size - len(new_population)]
        
        T *= alpha
        sigma *= alpha
        
        current_best = population[0]
        current_c5 = evaluate_erdos_solution(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best
        
        time_used += 5
        if time_used >= budget_s:
            break
    
    best_h = best_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    return best_h, best_c5, n_points
