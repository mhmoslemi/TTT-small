import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.full(70, 0.5)
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    best_h = h_initial
    best_c5 = float('inf')
    start_time = time.time()
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 35}]
    
    start_points = []
    start_points.append(h_initial)
    start_points.append(h_initial + 0.1 * np.random.normal(0, 0.1, n_points))
    start_points.append(h_initial + 0.1 * np.sin(2 * np.pi * np.arange(n_points) / n_points))
    start_points.append(h_initial + 0.1 * np.cos(2 * np.pi * np.arange(n_points) / n_points))
    start_points.append(h_initial + 0.1 * np.random.rand(n_points))
    start_points.append(h_initial + 0.15 * np.sin(2 * np.pi * np.arange(n_points) / n_points))
    start_points.append(h_initial + 0.15 * np.cos(2 * np.pi * np.arange(n_points) / n_points))
    
    for hp in start_points:
        hp_clipped = np.clip(hp, 0.0, 1.0)
        hp_normalized = hp_clipped * (35.0 / np.sum(hp_clipped))
        result = minimize(
            fun=objective,
            x0=hp_normalized,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=1e-8,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    if time.time() - start_time > budget_s:
        pass
    
    return best_h, best_c5, n_points
