import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n = 70
    dx = 2.0 / n
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n)
        for i in range(35):
            h[i * 2] = 1.0
        h = h * (35.0 / np.sum(h))
    best_h = h.copy()
    best_c5 = float('inf')
    t0 = time.time()

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    T = 1.0
    cooling_rate = 0.999
    sa_iter = 500
    for _ in range(sa_iter):
        if time.time() - t0 > budget_s - 500:
            break
        noise = np.random.normal(0, T, n)
        h_perturbed = h + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (35.0 / np.sum(h_perturbed))
        new_c5 = compute_c5(h_perturbed)
        if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / T):
            best_c5 = new_c5
            best_h = h_perturbed.copy()
        T *= cooling_rate

    def objective(h):
        return compute_c5(h)

    constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) - 35.0}
    bounds = (0.0, 1.0)
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
    )
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    return best_h, best_c5, n
