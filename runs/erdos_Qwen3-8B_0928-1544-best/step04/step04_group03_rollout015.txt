import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    sum_h = n_points / 2

    def evaluate_erdos(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - sum_h

    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(int(sum_h)):
            pos = round(i * (n_points / sum_h))
            h[pos] = 1.0
        if np.sum(h) != sum_h:
            diff = sum_h - np.sum(h)
            zeros = np.where(h == 0)[0]
            h[zeros[:int(diff)]] = 1.0
        initial_h = h

    swarm = []
    for _ in range(49):
        perturbed = initial_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * sum_h / np.sum(perturbed)
        swarm.append(perturbed)
    swarm = [initial_h] + swarm
    velocities = [np.zeros(n_points) for _ in swarm]
    personal_best = [h.copy() for h in swarm]
    global_best = min(swarm, key=evaluate_erdos)

    best_h = initial_h
    best_c5 = evaluate_erdos(best_h)
    t0 = time.time()

    inertia = 0.9
    c1, c2 = 2.0, 2.0
    velocity_limit = 0.2
    max_iter = 1000

    for iter in range(max_iter):
        if time.time() - t0 > budget_s - 100:
            break
        for i in range(len(swarm)):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = inertia * velocities[i] + c1 * r1 * (personal_best[i] - swarm[i]) + c2 * r2 * (global_best - swarm[i])
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            swarm[i] += velocities[i]
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * sum_h / np.sum(swarm[i])
        fitness = [evaluate_erdos(h) for h in swarm]
        for i in range(len(swarm)):
            if fitness[i] < evaluate_erdos(personal_best[i]):
                personal_best[i] = swarm[i].copy()
        current_best = min(swarm, key=evaluate_erdos)
        if evaluate_erdos(current_best) < evaluate_erdos(global_best):
            global_best = current_best.copy()
        inertia *= 0.999

    top = sorted(swarm, key=evaluate_erdos)[:10]
    for h in top:
        res = minimize(evaluate_erdos, h, method='SLSQP', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': constraint_sum})
        if res.success:
            curr = res.fun
            if curr < best_c5:
                best_c5 = curr
                best_h = res.x

    return best_h, best_c5, n_points
