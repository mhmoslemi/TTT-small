import numpy as np
import scipy.fft as fft
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = initial_h_values.copy().astype(float)
    else:
        h = np.full(n_points, 0.5)
    h = h * 35.0 / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    max_iterations = int((budget_s - 5) / 10)  # Leave 5s for final check

    for _ in range(max_iterations):
        # Frequency domain adjustment
        H = fft.fft(h)
        current_mag = np.abs(H)
        target_mag = np.sqrt(np.sum(current_mag ** 2) / n_points)
        H_new = target_mag * np.exp(1j * np.angle(H))
        h_new = fft.ifft(H_new).real
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * 35.0 / np.sum(h_new) if np.sum(h_new) != 0 else np.full(n_points, 0.5)

        # Local optimization
        def obj_func(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - 35.0}
        bounds = [(0., 1.) for _ in range(n_points)]

        # Use L-BFGS-B with bounds
        try:
            result = optimize.minimize(obj_func, h_new, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100, 'ftol': 1e-7, 'gtol': 1e-7})
            if result.success:
                h = result.x
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        except:
            h = h_new
            current_c5 = obj_func(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

        if time.time() - start_time > budget_s - 5:
            break

    # Final check
    final_corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    best_c5 = np.max(final_corr)
    return best_h, best_c5, n_points
