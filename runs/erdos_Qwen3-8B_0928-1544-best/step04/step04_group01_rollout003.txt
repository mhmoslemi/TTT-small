import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_fine_initial = initial_h_values.copy()
    else:
        h_fine_initial = np.full(70, 0.5)
    n_points = 70
    dx = 2.0 / n_points
    best_h = h_fine_initial
    best_c5 = float('inf')
    start_time = time.time()
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def decorrelate(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def update_pbest(particles, pbest, pbest_obj, velocities, omega, c1, c2, gbest, n_points):
        n_particles = len(particles)
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h > 0:
                particles[i] = particles[i] * (n_points / 2) / sum_h
            current_obj = decorrelate(particles[i])
            if current_obj < pbest_obj[i]:
                pbest[i] = particles[i].copy()
                pbest_obj[i] = current_obj
        return particles, pbest, pbest_obj, velocities
    
    def refine_solution(h_coarse, n_points):
        h_fine = np.repeat(h_coarse, 2)
        sum_h = np.sum(h_fine)
        if sum_h > 0:
            h_fine = h_fine * (n_points / 2) / sum_h
        return h_fine
    
    if 'initial_h_values' in globals():
        h_fine_initial = refine_solution(initial_h_values / 2, 70)
    else:
        h_fine_initial = np.full(70, 0.5)
    
    particles = np.array([h_fine_initial + np.random.normal(0, 0.05, 70) for _ in range(30)])
    particles = particles / np.sum(particles, axis=1, keepdims=True) * 35.0
    pbest = particles.copy()
    pbest_obj = np.array([decorrelate(p) for p in particles])
    gbest = pbest[np.argmin(pbest_obj)]
    velocities = np.random.rand(30, 70) * 0.01
    omega, c1, c2 = 0.9, 2.0, 2.0
    
    while time.time() - start_time < budget_s:
        for _ in range(10):
            particles, pbest, pbest_obj, velocities = update_pbest(particles, pbest, pbest_obj, velocities, omega, c1, c2, gbest, 70)
        current_obj = pbest_obj[np.argmin(pbest_obj)]
        if current_obj < best_c5:
            best_c5 = current_obj
            best_h = pbest[np.argmin(pbest_obj)]
        gbest = pbest[np.argmin(pbest_obj)]
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, n_points
