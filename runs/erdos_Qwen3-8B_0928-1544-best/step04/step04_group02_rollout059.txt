from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 70
    dx = 2.0 / n_points
    POP_SIZE = 15
    MAX_GEN = 200
    MAX_ITER = 10
    init = np.zeros(n_points)
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        init = initial_h_values.copy()
    else:
        init = np.full(n_points, 0.5)
    init = init * (n_points / 2) / np.sum(init)
    population = [init.copy()]
    for _ in range(POP_SIZE-1):
        mut = init + np.random.normal(0, 0.1, n_points)
        mut = np.clip(mut, 0, 1)
        s = np.sum(mut)
        if s != 0:
            mut = mut * (n_points / 2) / s
        population.append(mut)
    best_c5 = float('inf')
    best_h = init.copy()
    start = time.time()
    for gen in range(MAX_GEN):
        if time.time() - start > budget_s:
            break
        scores = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            scores.append(c5)
        sorted_idx = np.argsort(scores)
        elites = [population[i] for i in sorted_idx[:int(POP_SIZE*0.3)]]
        new_pop = elites.copy()
        for _ in range(POP_SIZE - len(elites)):
            a, b = sorted_idx[np.random.choice(len(elites), 2, replace=False)]
            parent1, parent2 = elites[a], elites[b]
            child = 0.5 * (parent1 + parent2)
            child = np.clip(child, 0, 1)
            s = np.sum(child)
            if s != 0:
                child = child * (n_points / 2) / s
            else:
                child = np.zeros(n_points)
            for _ in range(MAX_ITER):
                def obj(x):
                    corr = np.correlate(x, 1 - x, mode='full')
                    return np.max(corr) * dx
                def cons(x):
                    return np.sum(x) - (n_points / 2)
                res = minimize(obj, child, method='SLSQP', bounds=Bounds(0,1), constraints={'type':'eq', 'fun':cons}, tol=1e-8)

            pert = np.random.normal(0, 0.05, n_points)

        population = new_pop
        for i in sorted_idx[:5]:
            h = population[i]
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
