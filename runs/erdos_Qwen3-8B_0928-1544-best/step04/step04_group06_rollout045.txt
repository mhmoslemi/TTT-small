def compute_fourier_coefficients(h, n_points, M):
    x = np.linspace(0, 2, n_points)

        cos_term = np.cos(k * np.pi * x)
        sin_term = np.sin(k * np.pi * x)
        a[k-1] = (dx / 2) * np.sum(h * cos_term)
        b[k-1] = (dx / 2) * np.sum(h * sin_term)
    return a, b
