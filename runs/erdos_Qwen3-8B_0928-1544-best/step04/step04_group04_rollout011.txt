            disp=False
        )
        new_h = project_h(result.x)
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
    
    result_final = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=Bounds([0.0]*n, [1.0]*n),
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 150, 'ftol': 1e-9}
    )
    
    if result_final.success and result_final.fun < best_c5:
        best_c5 = result_final.fun
        best_h = result_final.x
    
    return best_h, best_c5, n
