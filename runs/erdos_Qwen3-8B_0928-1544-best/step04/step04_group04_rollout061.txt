                bounds=bounds,
                constraints=constraints,
                tol=1e-8,
                options={'maxiter': 150, 'ftol': 1e-9}
            )
            
            if result.success:
                new_c5 = result.fun
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = result.x
    
    return best_h, best_c5, n_points
