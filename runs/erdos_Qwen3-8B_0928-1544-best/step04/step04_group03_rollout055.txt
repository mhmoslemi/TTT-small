import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:
        h = np.zeros(70)
        for i in range(35):
            h[int(i * 2)] = 1.0
        initial_h = h.copy()
    dx = 2.0 / 70

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    swarm_size = 50
    particles = [initial_h.copy()]
    for _ in range(swarm_size - 1):
        h_perturbed = initial_h + np.random.normal(0, 0.1, 70)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (35.0 / np.sum(h_perturbed))
        particles.append(h_perturbed)

    velocities = [np.zeros(70) for _ in range(swarm_size)]
    personal_best = particles[:]
    swarm_best = min(personal_best, key=objective)
    best_c5 = objective(swarm_best)
    best_h = swarm_best.copy()

    omega, c1, c2, vel_limit = 0.9, 2.0, 2.0, 0.2
    max_iter = 1000
    t0 = time.time()

    for _ in range(max_iter):
        if time.time() - t0 > budget_s - 500:
            break
        fitness = [objective(p) for p in particles]
        for i in range(swarm_size):
            if fitness[i] < objective(personal_best[i]):
                personal_best[i] = particles[i].copy()
        current_swarm_best = min(personal_best, key=objective)
        if objective(current_swarm_best) < objective(swarm_best):
            swarm_best = current_swarm_best
            best_c5 = objective(swarm_best)
            best_h = swarm_best.copy()
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (swarm_best - particles[i])
            velocities[i] = np.clip(velocities[i], -vel_limit, vel_limit)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            particles[i] = particles[i] * (35.0 / sum_p if sum_p != 0 else 1.0)
        omega *= 0.999

    top_candidates = personal_best[:10]
    for candidate in top_candidates:
        result = minimize(
            fun=objective,
            x0=candidate,
            method='SLSQP',
            bounds=[(0, 1)] * 70,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - 35.0},

                best_h = result.x

    return best_h, best_c5, 70
