import numpy as np
from scipy.optimize import differential_evolution
import random

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:

        tol=1e-8,

        recombination=0.8,
        seed=seed,
        polish=True,
        disp=False,
        maxiter=100,
        x0=initial_h
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n
