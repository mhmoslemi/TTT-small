import numpy as np
from scipy.optimize import differential_evolution, Bounds, minimize

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(70, 0.5)
    
    np.random.seed(seed)
    n = 70
    dx = 2.0 / n

    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale

    def objective(h):
        h_proj = project_h(h)
        corr = np.correlate(h_proj, 1 - h_proj, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n)]

    result_de = differential_evolution(
        objective, bounds,
        strategy='best1bin', popsize=30, maxiter=150,
        tol=1e-8, mutation=(0.5, 1.0), recombination=0.9,
        seed=seed, polish=True, disp=False
    )

    best_h = result_de.x
    best_c5 = result_de.fun

    def constraint_sum(h):
        return np.sum(h) - n / 2

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds_slsqp = Bounds([0.0]*n, [1.0]*n)

    result_slsqp = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 300, 'ftol': 1e-9}
    )

    if result_slsqp.success:
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun

    return best_h, best_c5, n
