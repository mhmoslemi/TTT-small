    current_c5 = objective(current_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h
    
    for _ in range(150):  # Increase number of perturbations
        perturbation = np.random.normal(0, 0.03, size=n)  # Smaller perturbation
        perturbed_h = current_h + perturbation
        perturbed_h = project_h(perturbed_h)
        
        result = minimize(
            fun=objective,
            x0=perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            tol=1e-8,
            options={'maxiter': 50}
        )
        
        if result.success:
            new_c5 = result.fun
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = result.x
                current_h = result.x
    
    result_final = minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 100}
    )
    
    if result_final.success and result_final.fun < best_c5:
        best_c5 = result_final.fun
        best_h = result_final.x
    
    return best_h, best_c5, n
