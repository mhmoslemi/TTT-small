import numpy as np
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    ones_count = n_points // 2
    h_initial = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - ones_count
    
    bounds = [(0.0, 1.0)] * n_points
    
    # Initial optimization
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Perturbation and re-optimization
    for _ in range(3):
        if np.random.rand() < 0.2:
            h_perturbed = np.clip(best_h + np.random.normal(0, 0.05, n_points), 0, 1)
            h_perturbed = h_perturbed * (ones_count / np.sum(h_perturbed))
            result = minimize(
                fun=objective,
                x0=h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
    
    return best_h, best_c5, n_points
