import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = initial_h_values.copy()
    else:
        h = np.full(70, 0.5)
    n_points = 70
    dx = 2.0 / n_points
    h_sum = np.sum(h)
    scale = 35.0 / h_sum
    h = h * scale
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    iter = 0
    max_iter = 50
    while iter < max_iter and time.time() - start_time < budget_s:
        H = fft(h)
        freq = np.fft.fftshift(np.arange(n_points))
        freq_norm = freq / n_points * 2
        # Invert high frequencies to balance energy
        scale = 1.0 / (1 + (freq_norm ** 2) * 0.1)
        H_scaled = H * scale
        h_new = np.real(ifft(H_scaled))
        h_new = np.clip(h_new, 0, 1)
        norm_sum = np.sum(h_new)
        h_new = h_new * (35.0 / norm_sum) if norm_sum > 0 else h_new
        
        def obj_func(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        def cons(x):
            return np.sum(x) - 35.0
        
        res = minimize(obj_func, h_new, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': cons}, tol=1e-8, options={'maxiter': 100})
        
        if res.success:

        current_c5 = np.max(corr) * dx
