from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:

    else:

    best_h = h_initial
    best_c5 = float('inf')
    max_trials = 40
    trials_per_start = 3
    start_time = time.time()
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}]
    
    def generate_start(h_prev):
        perturbation = np.random.normal(0, 0.1, n_points)
        h_perturbed = h_prev + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (n_points/2.0 / np.sum(h_perturbed))
        return h_perturbed
    
    for trial in range(max_trials):
        h_start = generate_start(h_initial)
        result = minimize(
            fun=objective,
            x0=h_start,
            method='L-BFGS-B',

            constraints=constraints,
