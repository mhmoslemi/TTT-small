import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(n_points)
        ones = 35
        interval = n_points / ones
        for i in range(ones):
            pos = int(round(i * interval))
            h_initial[pos] = 1.0
        if np.sum(h_initial) != ones:
            diff = ones - np.sum(h_initial)
            zeros = np.where(h_initial == 0)[0]
            h_initial[zeros[:diff]] = 1.0

    best_h = h_initial
    best_c5 = float('inf')
    t0 = time.time()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - 35.0

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)

    # Simulated Annealing
    temp = 1.0
    alpha = 0.999
    sa_iterations = 2000
    sa_candidates = []

    for _ in range(sa_iterations):
        if time.time() - t0 > budget_s - 500:
            break
        perturbation = np.random.normal(0, temp, n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (35.0 / np.sum(h_perturbed))
        curr_c5 = objective(h_perturbed)
        sa_candidates.append((curr_c5, h_perturbed))
        # Keep top 10 candidates
        sa_candidates.sort()
        sa_candidates = sa_candidates[:10]
        h_initial = sa_candidates[0][1]
        temp *= alpha

    # Refinement with SLSQP on top candidates
    refined_best = best_h
    refined_c5 = best_c5
    for c5, h in sa_candidates:
        if c5 < refined_c5:
            result = minimize(
                fun=objective,
                x0=h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                tol=1e-8,
                options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < refined_c5:
                    refined_c5 = current_c5
                    refined_best = result.x
    return refined_best, refined_c5, n_points
