import numpy as np
from scipy.optimize import differential_evolution, Bounds, minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(70)

    def objective(h):
        h_proj = project_h(h)
        corr = np.correlate(h_proj, 1 - h_proj, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    # Generate diverse initial population from initial_h and random perturbation
    initial_population = np.zeros((20, n))
    for i in range(20):

            vec *= (35.0 / sum_vec)
