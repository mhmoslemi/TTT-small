import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = initial_h_values.copy().astype(float)
    else:
        h = np.full(n_points, 0.5)
    sum_h = np.sum(h)
    h = h * (35.0 / sum_h)
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s:
        iter_count += 1
        time_remaining = budget_s - (time.time() - start_time)
        if time_remaining <= 0:
            break
        
        # Frequency domain optimization
        current_H = np.fft.fft(h)
        mag = np.abs(current_H)
        existing_energy = np.sum(mag**2)
        avg_mag = np.sqrt(existing_energy / n_points)
        target_mag = avg_mag * np.ones_like(mag)
        scale_factor = np.sqrt(existing_energy / (np.sum(target_mag**2)))
        scaled_target_mag = target_mag * scale_factor
        phase = np.angle(current_H)
        new_H = scaled_target_mag * np.exp(1j * phase)
        
        # Back to time domain
        h_new = np.fft.ifft(new_H).real
        h_clipped = np.clip(h_new, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            h_normalized = np.zeros_like(h_clipped)
        else:
            h_normalized = h_clipped * (35.0 / sum_clipped)
        
        # Time domain optimization
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        def cons(x):
            return np.sum(x) - 35.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, h_normalized, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': cons}, tol=1e-8, options={'maxiter': 100})
        if res.success:
            h = res.x
        
        # Evaluate
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
