import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.full(70, 0.5, dtype=np.float64)
    
    n = 70
    dx = 2.0 / n
    
    def objective(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.inf
        h_scaled = h_clipped * (n / 2) / sum_h
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n)]
    start_time = time.time()
    
    # Initial call with seed and constraint-aware bounds
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        popsize=20,
        tol=1e-8,
        mutation=(0.8, 1.0),
        recombination=0.9,
        strategy='best1bin',
        maxiter=300,
        seed=seed,
        disp=False
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Final refinement with local search
    def refine(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.inf
        h_scaled = h_clipped * (n / 2) / sum_h
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    result_refine = differential_evolution(
        fun=refine,
        bounds=bounds,
        popsize=10,
        tol=1e-8,
        mutation=(0.5, 1.0),
        recombination=0.9,
        strategy='best1bin',
        maxiter=100,
        seed=seed + 1,
        disp=False
    )
    
    if result_refine.fun < best_c5:
        best_c5 = result_refine.fun
        best_h = result_refine.x
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Additional refinement if time permits
        result_final = differential_evolution(

            disp=False
        )
