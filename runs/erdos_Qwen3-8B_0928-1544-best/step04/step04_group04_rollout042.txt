import numpy as np
from scipy.optimize import minimize, Bounds
import random

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(70)
        initial_h = (initial_h / np.sum(initial_h) * 35.0)
    
    dx = 2.0 / 70
    n = 70
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds([0.0]*n, [1.0]*n)
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    best_c5 = float('inf')
    best_h = np.copy(initial_h)
    current_h = project_h(initial_h)
    current_c5 = objective(current_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h
    
    for _ in range(75):
        # Systematic perturbation of the current best
        shifts = np.linspace(0, n, 6, endpoint=False).astype(int)
        perturbed_h_list = []
        for shift in shifts:
            perturbed_h = np.roll(current_h, shift)
            perturbed_h = project_h(perturbed_h)
            perturbed_h_list.append(perturbed_h)
        
        # Add random perturbations for diversity
        for _ in range(4):
            perturbation = np.random.normal(0, 0.05, n)
            perturbed_h = current_h + perturbation
            perturbed_h = project_h(perturbed_h)
            perturbed_h_list.append(perturbed_h)
        
        for perturbed_h in perturbed_h_list:
            try:
                result = minimize(

                    x0=perturbed_h,
                    method='SLSQP',

                )
                if result.success:
                    new_c5 = result.fun

    )
    
    if result_final.success and result_final.fun < best_c5:
