import numpy as np
from cvxpy import Problem, Variable, Minimize, Constraint, sum_entries
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h = initial_h_values.copy()
    else:
        h = np.full(70, 0.5)
    h = h * (35.0 / np.sum(h))
    n_points = 70
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    solver_opts = {'max_iter': 100}
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        j_values = [(i + k) % n_points for i in range(n_points)]
        
        h_var = Variable(n_points)
        t_var = Variable()
        
        constraints = [
            sum_entries(h_var) == 35,
            0 <= h_var,
            h_var <= 1,
            sum(h_var[i] * (1 - h_var[j_values[i]]) for i in range(n_points)) * dx <= t_var
        ]
        objective = Minimize(t_var)
        prob = Problem(objective, constraints)
        result = prob.solve(solver='CVXOPT', verbose=False, **solver_opts)

            best_h = h_var.value
