        h = initial_h_values.copy()
    else:
        h = np.full(70, 0.5)
    h = h * (35.0 / np.sum(h))
    dx = 2.0 / 70
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        k_idx = np.argmax(corr)
        current_c5 = max_corr * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.380876:
            break
        
        def objective(x):
            j = (np.arange(70) + k_idx) % 70
            return np.sum(x * (1 - x[j]))
        
        def grad(x):
            grad = np.zeros(70)
            for i in range(70):
                j = (i + k_idx) % 70
                l = (i - k_idx) % 70
                grad[i] = (1 - x[j]) - x[l]
            return grad
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 35}]
        bounds = [(0, 1) for _ in range(70)]
        
        result = minimize(objective, h, method='SLSQP', jac=grad, bounds=bounds, constraints=cons, options={'maxiter': 20})
        if result.success:
            h = result.x
        else:
            h = np.clip(h, 0, 1)
            h = h * (35.0 / np.sum(h))
    
    return best_h, best_c5, 70
