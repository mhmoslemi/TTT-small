import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.full(70, 0.5)
    
    n = 70
    dx = 2.0 / n
    best_c5 = float('inf')
    best_h = np.copy(initial_h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    def project_h(h):
        h_clip = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clip)
        if sum_h == 0:
            return np.zeros(n)
        scale = (n / 2) / sum_h
        return h_clip * scale
    
    np.random.seed(seed)
    current_h = project_h(initial_h)
    best_h = current_h.copy()
    best_c5 = objective(current_h)
    
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Perturb and optimize
        perturbation = np.random.normal(0, 0.03, size=n)
        perturbed_h = current_h + perturbation
        perturbed_h = project_h(perturbed_h)
        
        result = minimize(
            fun=objective,
            x0=perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            tol=1e-8,
            options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        if result.success:
            new_c5 = result.fun
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = result.x
                current_h = result.x
    
    result_final = minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    if result_final.success and result_final.fun < best_c5:
        best_c5 = result_final.fun
        best_h = result_final.x
    
    return best_h, best_c5, n
