import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and len(initial_h_values) == 70:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(70)
        ones = 35
        interval = 70 / ones
        for i in range(ones):
            pos = int(i * interval)
            h_initial[pos] = 1.0
        if np.sum(h_initial) != ones:
            diff = ones - np.sum(h_initial)
            zeros = np.where(h_initial == 0)[0]
            h_initial[zeros[:diff]] = 1.0
    dx = 2.0 / 70
    best_h = h_initial.copy()
    best_c5 = float('inf')
    t0 = time.time()

    # Evaluate initial solution
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Simulated Annealing
    T = 1.0
    alpha = 0.9995
    for _ in range(1000):
        if time.time() - t0 > budget_s - 100:
            break
        perturbation = np.random.normal(0, 0.05, 70)
        h_perturbed = np.clip(h_initial + perturbation, 0, 1)
        h_perturbed = h_perturbed * (35.0 / np.sum(h_perturbed))
        corr = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        new_c5 = np.max(corr) * dx
        if np.random.rand() < np.exp((best_c5 - new_c5) / T):
            h_initial = h_perturbed
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_perturbed.copy()
        T *= alpha

    # Final refinement with SLSQP
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - 35.0

    bounds = Bounds([0.0]*70, [1.0]*70)
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 200, 'ftol': 1e-10}
    )
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    return best_h, best_c5, 70
